    n = len(h)
    h = np.round(h).astype(float)

    best_h = h.copy()
    best_c5 = c5_cur

    zeros = np.where(h < 0.5)[0]
    
    n_evap = max(1, n_iters // 100)
    for it in range(n_iters):
        progress = it / n_iters
        T = T_init * (T_final / T_init) ** progress
        
        if len(ones) == 0 or len(zeros) == 0:

_GT = 0.0

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                s = min(-d, h[i])
                h[i] -= s
                d += s
            if abs(d) < 1e-14:
                break
    return h

def _eq(x):
    return x.sum() - _GT

def _eqj(x):
    return np.ones(len(x))

def _lse(x, T):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    s = T * (c - m)
    return (m + np.log(np.sum(np.exp(s))) / T) * 2.0 / n

def _lseg(x, T):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    s = T * (c - m)
    e = np.exp(s)
    w = e / e.sum()
    ct = n - 1
    dx = 2.0 / n
    g = np.zeros(n)
    for o in range(-(n - 1), n):
        wi = w[ct + o]
        if wi < 1e-14:
            continue
        if o == 0:
            g += wi * (1.0 - 2.0 * x)
        elif o > 0:
            g[o:] += wi * (1.0 - x[:n - o])
            g[:n - o] -= wi * x[o:]
        else:
            p = -o
            g[:n - p] += wi * (1.0 - x[p:])
            g[p:] -= wi * x[:n - p]
    return g * dx

        g[:n - o] -= x[o:]
    else:
        p = -o
        g[:n - p] += 1.0 - x[p:]
        g[p:] -= x[:n - p]
    return g * dx

    global _GT
    _GT = target
    n = len(h)
    b = [(0.0, 1.0)] * n

    try:
        r = minimize(obj, h, method='SLSQP', jac=jac, bounds=b,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(r.x, 0.0, 1.0)
    except Exception:
        return h

def coord_desc(h, target, n_sweeps, steps):

                if c5n < c5c - 1e-16:
                    h = t
                    c5c = c5n
                    imp = True
                    break
        if not imp:
            break
    return h, c5c
