- For each coordinate i (cycling through all n):
  - Compute the overlap as a function of x_i (given all other x fixed)
  - Find the optimal x_i that minimizes the max overlap
  - This is a 1D optimization problem

The 1D problem: given x_i = t, the overlap at shift m is:
c_m(t) = sum_{j≠i} x[j]*(1-x[j+m]) + (contribution from i)

The contribution from i depends on whether i or i+m is in the valid range.

For shift m ≥ 0:
- If i+m < n: x_i appears as x_i*(1-x[i+m]) → (1-x[i+m])*t
- If i-m ≥ 0: x_i appears in x[i-m]*(1-x_i) → -x[i-m]*t

So c_m(t) = c_m(0) + t * [(1-x[i+m])*[i+m<n] - x[i-m]*[i-m>=0]]

This is LINEAR in t! So the optimal t for a single shift is:
- If coefficient > 0: t = 0 (minimize)
- If coefficient < 0: t = 1 (minimize)
- If coefficient = 0: any t

For the MAX over shifts, it's:
f(t) = max_m (c_m(0) + a_m * t) where a_m is the coefficient for shift m.

This is a piecewise-linear function in t (max of linear functions). The minimum is at a breakpoint or at t=0 or t=1.

We can find the minimum by:
1. Evaluating f at t=0 and t=1
2. Finding the breakpoints (where the argmax changes)
3. Evaluating at breakpoints

The breakpoints are at t values where c_m1(t) = c_m2(t) for some pair m1, m2.
c_m1(0) + a_m1*t = c_m2(0) + a_m2*t
t = (c_m2(0) - c_m1(0)) / (a_m1 - a_m2)

There are O(K²) such breakpoints for K active shifts.

This gives an EXACT coordinate descent step! This should be very effective for the final polish.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        m2 = (j + shift) < n
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        m2 = j >= s
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def sm_obj(x, n, T, K, t0, tlim, flag):
    if time.time() - t0 > tlim:
        flag[0] = True
    corr = np.correlate(x, 1.0 - x, mode="full")
    dx = 2.0 / n
    mc = np.max(corr)
    if K < len(corr):
        idx = np.argpartition(corr, -K)[-K:]
        cs = corr[idx]
        mcs = np.max(cs)
    else:
        idx = np.arange(len(corr))
        cs = corr
        mcs = mc
    sh = T * (cs - mcs)
    e = np.exp(sh)
    sw = e / e.sum()
    val = (mcs + np.log(e.sum()) / T) * dx
    center = n - 1
    g = np.zeros(n)
    for i in range(len(sw)):
        if sw[i] < 1e-15:
            continue
        g += sw[i] * shift_grad(x, n, idx[i] - center)
    return val, g * dx

def do_lbfgsb(h, n, T, K, maxiter, t0, tlim):
    flag = [False]
    def obj(x):
        v, _ = sm_obj(x, n, T, K, t0, tlim, flag)
        return v
    def jac(x):
        _, g = sm_obj(x, n, T, K, t0, tlim, flag)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jac, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-13,
                                'maxls': 20, 'maxfun': maxiter * 4})
        if flag[0]:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def do_coord_desc(h, n, K, max_sweeps, t0, tlim):
    """Coordinate descent: for each i, find optimal h[i] given others fixed."""
    center = n - 1
    h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() - t0 > tlim:
            break
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Top K shifts
        if K < len(corr):
            topk = np.argpartition(corr, -K)[-K:]
        else:
            topk = np.arange(len(corr))
        for i in range(n):
            hi = h[i]
            if hi <= 0.0 or hi >= 1.0:
                continue  # already at boundary
            # Compute coefficients for top-K shifts
            a = np.zeros(len(topk))
            for k_idx in range(len(topk)):
                m = topk[k_idx] - center
                coeff = 0.0
                ip = i + m
                im = i - m
                if 0 <= ip < n:
                    coeff += (1.0 - h[ip])
                if 0 <= im < n:
                    coeff -= h[im]
                a[k_idx] = coeff
            c = corr[topk]
            # f(t) = max_k (c[k] + (t - hi) * a[k])
            # Minimize over t in [0,1]
            f0 = np.max(c - hi * a)
            f1 = np.max(c + (1.0 - hi) * a)
            best_t = 0.0 if f0 <= f1 else 1.0
            best_f = min(f0, f1)
            # Check intersection points of top pairs
            # Only check top-K for efficiency
            for p in range(len(a)):
                for q in range(p + 1, len(a)):
                    da = a[p] - a[q]
                    if abs(da) < 1e-15:
                        continue
                    t_bp = (c[q] - c[p]) / da + hi
                    if 0.0 < t_bp < 1.0:
                        f_bp = np.max(c + (t_bp - hi) * a)
                        if f_bp < best_f - 1e-15:
                            best_f = f_bp
                            best_t = t_bp
            h[i] = best_t
    return h

def do_slsqp(h, n, T, K, maxiter, t0, tlim):
    flag = [False]
    def obj(x):
        if time.time() - t0 > tlim:
            flag[0] = True
        v, _ = sm_obj(x, n, T, K, t0, tlim, flag)
        return v
    def jac(x):
        _, g = sm_obj(x, n, T, K, t0, tlim, flag)
        return g
    def cesq(x):
        return x.sum() - n / 2.0
    def cejac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cesq, 'jac': cejac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        if flag[0]:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def perturb(h, n, rng, mode):

    if mode == 0:

        p += rng.normal(0, 0.002, n)
    return np.clip(p, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        m2 = (j + shift) < n
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        m2 = j >= s
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def sm_obj(x, n, T, K, t0, tlim, flag):
    if time.time() - t0 > tlim:
        flag[0] = True
    corr = np.correlate(x, 1.0 - x, mode="full")
    dx = 2.0 / n
    mc = np.max(corr)
    if K < len(corr):
        idx = np.argpartition(corr, -K)[-K:]
        cs = corr[idx]
        mcs = np.max(cs)
    else:
        idx = np.arange(len(corr))
        cs = corr
        mcs = mc
    sh = T * (cs - mcs)
    e = np.exp(sh - sh.max())
    sw = e / e.sum()
    val = (mcs + np.log(e.sum()) / T) * dx
    center = n - 1
    g = np.zeros(n)
    for i in range(len(sw)):
        if sw[i] < 1e-15:
            continue
        g += sw[i] * shift_grad(x, n, idx[i] - center)
    return val, g * dx

def do_lbfgsb(h, n, T, K, maxiter, t0, tlim):
    flag = [False]
    def obj(x):
        v, _ = sm_obj(x, n, T, K, t0, tlim, flag)
        return v
    def jac(x):
        _, g = sm_obj(x, n, T, K, t0, tlim, flag)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jac, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14,
                                'maxls': 20, 'maxfun': maxiter * 5})
        if flag[0]:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def do_slsqp(h, n, T, K, maxiter, t0, tlim):
    flag = [False]
    def obj(x):
        v, _ = sm_obj(x, n, T, K, t0, tlim, flag)
        return v
    def jac(x):
        _, g = sm_obj(x, n, T, K, t0, tlim, flag)
        return g
    def ceq(x):
        return x.sum() - n / 2.0
    def ceqj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': ceq, 'jac': ceqj}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-17})
        if flag[0]:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def do_coord_desc(h, n, K, max_sweeps, t0, tlim):
    h = h.copy()
    center = n - 1
    for sweep in range(max_sweeps):
        if time.time() - t0 > tlim:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        if K < len(corr):
            topk = np.argpartition(corr, -K)[-K:]
        else:
            topk = np.arange(len(corr))
        shifts = topk - center
        c_vals = corr[topk].copy()
        nk = len(topk)
        for i in range(n):
            hi = h[i]
            if hi <= 1e-12 or hi >= 1.0 - 1e-12:
                continue
            ip = i + shifts
            im = i - shifts
            mp = (ip >= 0) & (ip < n)
            mm = (im >= 0) & (im < n)
            a = np.zeros(nk)
            if mp.any():
                a[mp] += (1.0 - h[ip[mp]])
            if mm.any():
                a[mm] -= h[im[mm]]
            f0 = float(np.max(c_vals - hi * a))
            f1 = float(np.max(c_vals + (1.0 - hi) * a))
            best_t = 0.0 if f0 <= f1 else 1.0
            best_f = min(f0, f1)
            # Vectorized breakpoint check
            da = a[:, None] - a[None, :]
            dc = c_vals[None, :] - c_vals[:, None]
            valid = np.abs(da) > 1e-12
            valid = valid & (np.arange(nk)[:, None] < np.arange(nk)[None, :])
            with np.errstate(divide='ignore', invalid='ignore'):
                tb = dc / da + hi
            tb[~valid] = -1.0
            tb[(tb <= 0.0) | (tb >= 1.0)] = -1.0
            tb_valid = tb[tb > 0.0]
            if len(tb_valid) > 0:
                mat = (tb_valid[:, None] - hi) * a[None, :] + c_vals[None, :]
                fb = np.max(mat, axis=1)
                best_idx = int(np.argmin(fb))
                if fb[best_idx] < best_f - 1e-14:
                    best_f = fb[best_idx]
                    best_t = tb_valid[best_idx]
            h[i] = best_t
    return h

def perturb(h, n, rng, mode):
    x = np.arange(n, dtype=float)
    p = h.copy()
    if mode == 0:
        for _ in range(4):
            f = rng.uniform(1, 10)
            ph = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.005, 0.03)
            p += amp * np.sin(2 * np.pi * f * x / n + ph)
        p += rng.normal(0, 0.004, n)
    elif mode == 1:
        sig = 0.003 + 0.008 * (mode % 3)
        p += rng.normal(0, sig, n)
    else:
        bs = max(n // 10, 5)
        nb = n // bs
        for _ in range(int(rng.integers(1, 4))):
            bi = int(rng.integers(0, nb))
            p[bi*bs:min((bi+1)*bs, n)] = 1.0 - p[bi*bs:min((bi+1)*bs, n)]
        p += rng.normal(0, 0.002, n)
    return np.clip(p, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    n0 = 280
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(n0)
        h0[::2] = 1.0
    n0 = len(h0)
    h0 = project_feasible(h0, n0)

    try_ns = []
    for nn in [n0, 240, 320, 200]:
        if nn not in try_ns:
            try_ns.append(nn)

    h_best = h0.copy()
    c5_best = compute_c5(h0)
    n_best = n0

    for n_try in try_ns:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        if n_try == n0:
            h = h0.copy()
        else:
            x_old = np.arange(n0) * (2.0 / n0)
            x_new = np.linspace(0, 2.0, n_try, endpoint=False)
            h = np.interp(x_new, x_old, h0)
        h = project_feasible(h, n_try)
        n = n_try
        c5 = compute_c5(h)

        # L-BFGS-B schedule
        for T, K, mi in [(32, 300, 4000), (8, 200, 6000), (2, 100, 10000), (0.5, 30, 15000)]:
            rem = budget_s - (time.time() - t0)
            if rem < 50:
                break
            h = do_lbfgsb(h, n, T, K, mi, t0, budget_s - 15)
            c5 = compute_c5(h)

        # Coordinate descent
        rem = budget_s - (time.time() - t0)
        if rem > 25:
            t_cd_end = t0 + min(rem - 10, 100)
            h = do_coord_desc(h, n, 40, 8, t0, t_cd_end)
            c5 = compute_c5(h)

        # SLSQP
        rem = budget_s - (time.time() - t0)
        if rem > 30:
            h = do_slsqp(h, n, 1, 30, 4000, t0, budget_s - 10)
            c5 = compute_c5(h)

        # Coordinate descent again
        rem = budget_s - (time.time() - t0)
        if rem > 20:
            h = do_coord_desc(h, n, 25, 5, t0, t0 + rem - 8)
            c5 = compute_c5(h)

        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()
            n_best = n

    # Restarts
    n = n_best
    h = h_best.copy()
    for pi in range(50):
        rem = budget_s - (time.time() - t0)
        if rem < 20:
            break
        hp = perturb(h, n, rng, pi % 3)
        hp = project_feasible(hp, n)
        hp = do_lbfgsb(hp, n, 2, 60, 2500, t0, budget_s - 8)
        hp = do_coord_desc(hp, n, 20, 3, t0, budget_s - 5)
        c5_r = compute_c5(hp)
        if c5_r < c5_best:
            c5_best = c5_r
            h_best = hp.copy()
            h = hp.copy()

    # Final polish
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        h_best = do_coord_desc(h_best, n_best, 20, 5, t0, t0 + rem - 3)
        c5_f = compute_c5(h_best)
        if c5_f < c5_best:
            c5_best = c5_f

    h_best = project_feasible(h_best, n_best)
    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, n_best)
