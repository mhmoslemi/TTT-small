import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def coord_min_1d(alpha, beta, iters=20):
    best_v = 0.5
    best_val = float(np.max(alpha + 0.5 * beta))
    lo_v, hi_v = 0.0, 1.0
    for _ in range(iters):
        mid_v = (lo_v + hi_v) / 2.0
        vals = alpha + mid_v * beta
        mx = float(np.max(vals))
        if mx < best_val - 1e-15:
            best_val = mx
            best_v = mid_v
        thr = mx - 1e-10
        active = vals >= thr
        if not active.any():
            active = vals >= mx - 1e-8
        min_b = float(np.min(beta[active]))
        max_b = float(np.max(beta[active]))
        if min_b > 1e-10:
            hi_v = mid_v
        elif max_b < -1e-10:
            lo_v = mid_v
        else:
            break
    for vc in (0.0, 1.0):
        vcv = float(np.max(alpha + vc * beta))
        if vcv < best_val:
            best_val = vcv
            best_v = vc
    return best_v

def coord_descent_full(h, n, max_sweeps, t0, tlim, rng, perturb_every=20):

    corr = np.correlate(h, 1.0 - h, mode="full")

    for sweep in range(max_sweeps):
        if time.time() - t0 > tlim:
            break
        iters_bi = 10 if sweep < 50 else 18
        order = rng.permutation(n)
        for j in order:
            if time.time() - t0 > tlim:
                break
            v_old = h[j]
            js = j + s_vals
            I1 = ((js >= 0) & (js < n)).astype(float)
            jm = j - s_vals
            I2 = ((jm >= 0) & (jm < n)).astype(float)
            h_js = np.where(I1 > 0.5, h[np.clip(js, 0, n - 1)], 0.0)
            h_jm = np.where(I2 > 0.5, h[np.clip(jm, 0, n - 1)], 0.0)
            alpha = corr - v_old * (1.0 - h_js) * I1 + v_old * h_jm * I2
            beta = (1.0 - h_js) * I1 - h_jm * I2
            i0 = n - 1
            alpha[i0] = corr[i0] - v_old * (1.0 - 2.0 * v_old)
            beta[i0] = 1.0 - 2.0 * v_old
            v_opt = coord_min_1d(alpha, beta, iters=iters_bi)

def _sm_val(x, n, T, K):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    topk = np.argpartition(corr, -K)[-K:]
    c_sel = corr[topk]
    mc_sel = np.max(c_sel)
    sh = T * (c_sel - mc_sel)
    e = np.exp(sh)
    return (mc_sel + np.log(e.sum()) / T) * (2.0 / n)

def _sm_grad(x, n, T, K):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    topk = np.argpartition(corr, -K)[-K:]
    c_sel = corr[topk]
    mc_sel = np.max(c_sel)
    sh = T * (c_sel - mc_sel)
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    j = np.arange(n)
    for ii in range(len(sw)):
        w = sw[ii]
        if w < 1e-15:
            continue
        s_top = topk[ii] - center
        if s_top >= 0:
            m1 = j >= s_top
            m2 = (j + s_top) < n
            grad[m1] += w * (1.0 - x[j[m1] - s_top])
            grad[m2] -= w * x[j[m2] + s_top]
        else:
            ss = -s_top
            m1 = (j + ss) < n
            m2 = j >= ss
            grad[m1] += w * (1.0 - x[j[m1] + ss])
            grad[m2] -= w * x[j[m2] - ss]
    return grad * (2.0 / n)
