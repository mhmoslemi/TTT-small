import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def smooth_max_obj(h, n, T, topk):
    """Log-sum-exp over top-k correlations."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = min(topk, len(corr))
    # Get top-k values
    part = np.argpartition(corr, -k)[-k:]
    topk_vals = corr[part]
    mc = topk_vals.max()
    # Log-sum-exp
    e = np.exp((topk_vals - mc) * T)
    obj = (mc + np.log(e.sum()) / T) * dx
    return obj, part, topk_vals

def grad_smooth_max(h, n, T, topk):
    """Gradient of smooth max objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = min(topk, len(corr))
    part = np.argpartition(corr, -k)[-k:]
    topk_vals = corr[part]
    mc = topk_vals.max()
    e = np.exp((topk_vals - mc) * T)
    w = e / e.sum()
    
    center = n - 1
    j = np.arange(n)
    grad = np.zeros(n)
    for oi in range(k):
        wi = w[oi]
        if wi < 1e-12:
            continue
        shift = part[oi] - center
        if shift >= 0:
            m1 = j >= shift
            idx1 = j[m1] - shift
            grad[m1] += wi * (1.0 - h[idx1])
            m2 = (j + shift) < n
            idx2 = j[m2] + shift
            grad[m2] -= wi * h[idx2]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            idx1 = j[m1] + s2
            grad[m1] += wi * (1.0 - h[idx1])
            m2 = j >= s2
            idx2 = j[m2] - s2
            grad[m2] -= wi * h[idx2]
    grad *= dx
    return grad

def fista_opt(h_init, n, T, topk, n_iters, t_end, lr0=0.01):
    """FISTA with smooth max gradient."""
    h_y = h_init.copy()
    h_z = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    t_mom = 1.0
    lr = lr0
    
    for it in range(n_iters):
        if time.time() > t_end:
            break
        grad = grad_smooth_max(h_y, n, T, topk)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        h_z = h_y + beta * (h_y - h_z)
        h_y = h_z - lr * grad
        h_y = project_feasible(h_y, n)
        t_mom = t_new
        if (it + 1) % 5000 == 0:

            c5 = compute_c5(h_y)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_y.copy()
    return h_best, c5_best

def slsqp_opt(h, n, T, topk, maxiter, t_end):

    target = n / 2.0
    
    def obj(x):
        return smooth_max_obj(x, n, T, topk)[0]
    
    def jac(x):
        return grad_smooth_max(x, n, T, topk)
    
    def cf(x):
        return x.sum() - target
    
    bounds = [(0.0, 1.0)] * n

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return res.x

        return h.copy()
