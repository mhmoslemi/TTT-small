def topk_smooth_obj(h, n, T, k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = topk.max()
    sh = T * (topk - mc)
    sh = np.clip(sh, -400, 400)
    e = np.exp(sh)
    sw = e / e.sum()
    obj = (mc + np.log(e.sum()) / T) * dx
    return obj, idx, sw, topk

def topk_grad(h, n, idx, sw, topk):
    dx = 2.0 / n

        return x.sum() - target

        return np.ones(n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def slsqp_opt(h0, n, T, k, maxiter, ftol):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mk = min(k, len(corr))
        idx = np.argpartition(corr, -mk)[-mk:]
        topk = corr[idx]
        mc = topk.max()
        sh = np.clip(T * (topk - mc), -400, 400)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mk = min(k, len(corr))
        idx = np.argpartition(corr, -mk)[-mk:]
        topk = corr[idx]
        mc = topk.max()
        sh = np.clip(T * (topk - mc), -400, 400)
        e = np.exp(sh)
        sw = e / e.sum()
        center = n - 1
        grad = np.zeros(n)
        j = np.arange(n)
        for oi in range(mk):
            w = sw[oi]
            if w < 1e-12:
                continue
            shift = idx[oi] - center
            if shift >= 0:
                m1 = j >= shift
                grad[m1] += w * (1.0 - x[j[m1] - shift])
                m2 = (j + shift) < n
                grad[m2] -= w * x[j[m2] + shift]
            else:
                s2 = -shift
                m1 = (j + s2) < n
                grad[m1] += w * (1.0 - x[j[m1] + s2])
                m2 = j >= s2
                grad[m2] -= w * x[j[m2] - s2]
        return grad * dx
    def cf(x):
        return x.sum() - target
    def cj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x

    hz = h_init.copy()
    t_m = 1.0

    it = 0

```python
import numpy as np
from scipy.optimize import minimize
import time

_SLSQP_STATE = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def _slsqp_obj(x):
    n = _SLSQP_STATE['n']
    T = _SLSQP_STATE['T']
    k = _SLSQP_STATE['k']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = topk.max()
    sh = np.clip(T * (topk - mc), -400, 400)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * dx

def _slsqp_jac(x):
    n = _SLSQP_STATE['n']
    T = _SLSQP_STATE['T']
    k = _SLSQP_STATE['k']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = topk.max()
    sh = np.clip(T * (topk - mc), -400, 400)
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    j = np.arange(n)
    for oi in range(mk):
        w = sw[oi]
        if w < 1e-12:
            continue
        shift = idx[oi] - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += w * (1.0 - x[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= w * x[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += w * (1.0 - x[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * x[j[m2] - s2]
    return grad * dx

def _slsqp_cfun(x):
    return x.sum() - _SLSQP_STATE['n'] / 2.0

def _slsqp_cjac(x):
    return np.ones(_SLSQP_STATE['n'])

def slsqp_run(h0, n, T, k, maxiter, ftol):
    _SLSQP_STATE['n'] = n
    _SLSQP_STATE['T'] = T
    _SLSQP_STATE['k'] = k
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cfun, 'jac': _slsqp_cjac}]
    try:
        res = minimize(_slsqp_obj, h0, method='SLSQP', jac=_slsqp_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0.copy()

def compute_deltas(h, n):
    """Compute delta[i, ci] for all positions and correlation indices."""
    n_corr = 2 * n - 1
    center = n - 1
    d = np.zeros((n, n_corr))
    j = np.arange(n)
    for ci in range(n_corr):
        shift = ci - center
        if shift >= 0:
            mask1 = j <= n - 1 - shift
            mask2 = j >= shift
            d[mask1, ci] = (1.0 - h[j[mask1] + shift])
            d[mask2, ci] -= h[j[mask2] - shift]
        else:
            s = -shift
            mask1 = j >= s
            mask2 = j <= n - 1 - s
            d[mask1, ci] = (1.0 - h[j[mask1] - s])
            d[mask2, ci] -= h[j[mask2] + s]
    return d

def local_search_binary(h_bin, n, max_sweeps=100):
    """Local search: swap 1s and 0s to reduce max correlation."""
    dx = 2.0 / n
    h = h_bin.astype(float).copy()
    f = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = f.max() * dx
    best = h.copy()
    
    for sweep in range(max_sweeps):
        d = compute_deltas(h, n)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        nn = len(ones)
        nz = len(zeros)
        
        best_improve = 1e-15
        best_i, best_j = -1, -1
        
        for ii in range(nn):
            base = f - d[ones[ii]]
            # For all j: (base + d[zeros[jj]]).max()
            vals = (base[None, :] + d[zeros, :]).max(axis=1)
            jj_min = np.argmin(vals)
            c5_val = vals[jj_min] * dx
            if c5_val < best_c5 - best_improve:
                best_improve = best_c5 - c5_val
                best_i = ones[ii]
                best_j = zeros[jj_min]
        
        if best_j < 0:
            break
        
        h[best_i] = 0.0
        h[best_j] = 1.0
        f = f - d[best_i] + d[best_j]
        c5 = f.max() * dx
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    
    return best, best_c5
