import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def smooth_max_og(x, n, T, k):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = topk.max()
    shifted = T * (topk - mc)
    shifted -= shifted.max()
    e = np.exp(shifted)
    sw = e / e.sum()
    obj = (mc + mc + np.log(e.sum()) / T - shifted.max() / T) * dx
    obj = (mc + np.log(e.sum()) / T) * dx
    center = n - 1
    j = np.arange(n)
    grad = np.zeros(n)
    for oi in range(mk):
        w = sw[oi] * dx
        if w < 1e-13:
            continue
        shift = int(idx[oi] - center)
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += w * (1.0 - x[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= w * x[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += w * (1.0 - x[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * x[j[m2] - s2]
    return obj, grad

def slsqp_opt(h0, n, T, k, maxiter, ftol, t_end):
    target = n / 2.0
    def objf(x):
        return smooth_max_og(x, n, T, k)[0]
    def jacf(x):
        return smooth_max_og(x, n, T, k)[1]
    def cf(x):
        return x.sum() - target
    def cj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(objf, h0, method='SLSQP', jac=jacf, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < np.inf:
            return np.clip(res.x, 0, 1)
    except Exception:
        pass
    return h0.copy()

def fista_opt(h0, n, t_end, lr0=0.015, k_active=5):
    dx = 2.0 / n
    target = n / 2.0
    h_y = h0.copy()
    h_z = h0.copy()
    h_best = h0.copy()
    c5_best = compute_c5(h0)
    t_mom = 1.0
    lr = lr0
    j = np.arange(n)
    center = n - 1
    it = 0
    t_check = time.time()
    while time.time() < t_end:
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        ka = min(k_active, len(corr))
        idxa = np.argpartition(corr, -ka)[-ka:]
        grad = np.zeros(n)
        for oi in idxa:
            shift = int(idxa[oi] - center)
            if shift >= 0:
                m1 = j >= shift
                grad[m1] += (1.0 - h_y[j[m1] - shift])
                m2 = (j + shift) < n
                grad[m2] -= h_y[j[m2] + shift]
            else:
                s2 = -shift
                m1 = (j + s2) < n
                grad[m1] += (1.0 - h_y[j[m1] + s2])
                m2 = j >= s2
                grad[m2] -= h_y[j[m2] - s2]
        grad *= dx / ka
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        h_z = h_y + beta * (h_y - h_z)
        h_y = h_z - lr * grad
        h_y = project_feasible(h_y, n)
        t_mom = t_new
        lr *= 0.99995
        it += 1
        if it % 5000 == 0:
            c5 = compute_c5(h_y)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_y.copy()
    return h_best, c5_best

def perturb(h, n, rng, mode):
    h_p = h.copy()
    if mode == 0:
        s = int(rng.integers(1, 5))
        h_p = np.roll(h_p, s)
    elif mode == 1:
        st = int(rng.integers(0, max(1, n - 20)))
        ln = int(rng.integers(5, min(30, n - st)))
        h_p[st:st+ln] = 1.0 - h_p[st:st+ln]
    elif mode == 2:
        freq = int(rng.integers(2, 12))
        amp = rng.uniform(0.01, 0.05)
        x = np.arange(n) * 2.0 / n
        ph = rng.uniform(0, 2 * np.pi)
        h_p += amp * np.sin(np.pi * freq * x + ph)
    elif mode == 3:
        h_p += rng.uniform(0.003, 0.02) * rng.standard_normal(n)
    elif mode == 4:
        b1 = int(rng.integers(5, 40))
        s1 = int(rng.integers(0, max(1, n - b1)))
        s2 = int(rng.integers(0, max(1, n - b1)))
        tmp = h_p[s1:s1+b1].copy()
        h_p[s1:s1+b1] = h_p[s2:s2+b1]
        h_p[s2:s2+b1] = tmp
    elif mode == 5:
        mask = rng.random(n) < 0.2
        h_p[mask] = 1.0 - h_p[mask]
    elif mode == 6:
        w = int(rng.integers(10, 60))
        st = int(rng.integers(0, max(1, n - w)))
        h_p[st:st+w] = 1.0 - h_p[st:st+w]
        amp = rng.uniform(0.1, 0.3)
        h_p[st:st+w] += amp * rng.standard_normal(w)
    else:
        h_p += rng.uniform(0.005, 0.015) * rng.standard_normal(n)
    return h_p
