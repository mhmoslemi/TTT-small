    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, method='SLSQP', jac=_jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def slsqp_power(h, n, P, maxiter=8000, ftol=1e-14):
    dx = 2.0 / n
    center = n - 1
    target = n / 2.0
    def _obj(x):
        c = np.clip(np.correlate(x, 1.0 - x, mode="full"), 0, None)
        return (c.sum() ** P) ** (1.0 / P) * dx / (len(c) ** (1.0 - 1.0/P))
    def _jac(x):
        c = np.clip(np.correlate(x, 1.0 - x, mode="full"), 0, None)
        cpow = c.sum() ** P
        base = cpow ** (1.0/P - 1.0) * P / len(c) ** (1.0 - 1.0/P) * dx
        g = np.zeros(n)
        for oi in range(len(c)):
            if c[oi] > 1e-15:
                g += c[oi] ** (P-1) * shift_grad_raw(x, n, oi - center)
        return g * base
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, method='SLSQP', jac=_jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def grad_topk(h_init, n, time_budget, K=6, lr0=0.06, mom0=0.85):
    dx = 2.0 / n
    center = n - 1
    h_cur = h_init.copy()
    h_best = h_init.copy()
    c5_best = compute_c5(h_init)
    t_s = time.time()
    lr = lr0
    mom = np.zeros(n)
    for it in range(400000):
        if time.time() - t_s > time_budget:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        top_idx = np.argsort(corr)[-K:]
        w = np.exp(4.0 * (corr[top_idx] - corr[top_idx].max()))
        w /= w.sum()
        g = np.zeros(n)
        for wk, ik in zip(w, top_idx):
            if wk > 1e-10:
                g += wk * shift_grad_raw(h_cur, n, ik - center)
        g *= dx
        mom = mom0 * mom + g
        h_new = h_cur - lr * mom
        h_new = project_feasible(h_new, n)
        h_cur = h_new
        lr = lr0 / (1.0 + 0.003 * it)
        if (it + 1) % 300 == 0:
            cv = compute_c5(h_cur)
            if cv < c5_best:
                c5_best = cv
                h_best = h_cur.copy()
    cv = compute_c5(project_feasible(h_cur, n))
    if cv < c5_best:
        c5_best = cv
        h_best = project_feasible(h_cur, n).copy()
    return h_best, c5_best

def rebalance_local(h_init, n, rng, steps=4000):
    dx = 2.0 / n
    center = n - 1
    h_cur = h_init.copy()
    c5_cur = compute_c5(h_cur)
    for _ in range(steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        top_idx = np.argsort(corr)[-4:]
        for ak in top_idx:
            shift = ak - center
            g = shift_grad_raw(h_cur, n, shift) * dx
            pos = np.where(g > 0.04)[0]
            neg = np.where(g < -0.04)[0]
            if len(pos) == 0 or len(neg) == 0:
                continue
            improved = False
            for _t in range(8):
                ip = pos[rng.integers(len(pos))]
                inn = neg[rng.integers(len(neg))]
                d = min(0.02, h_cur[ip], 1.0 - h_cur[inn])
                if d < 0.001:
                    continue
                h_try = h_cur.copy()
                h_try[ip] -= d
                h_try[inn] += d
                h_try = project_feasible(h_try, n)
                cv = compute_c5(h_try)
                if cv < c5_cur - 1e-10:
                    h_cur = h_try
                    c5_cur = cv
                    improved = True
                    break
            if not improved:
                continue
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP LSE coarse
    temps = [48, 24, 12, 6, 3, 1.5, 1.0, 0.6, 0.4]
    for T in temps:
        if time.time() - t0 > budget_s - 450:
            break
        h_new = slsqp_lse(h_best, n, T, 9000, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        else:
            h_new = slsqp_power(h_best, n, 8, 5000, 1e-14)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 2: Gradient top-K with momentum
    if time.time() - t0 < budget_s - 250:
        h_g, c5_g = grad_topk(h_best, n, budget_s - 250, K=6, lr0=0.05)
        if c5_g < c5_best:
            c5_best = c5_g
            h_best = h_g
        else:
            h_best = project_feasible(h_best, n)

    # Phase 3: Fine SLSQP
    for T in [0.4, 0.25, 0.15, 0.1, 0.07]:
        if time.time() - t0 > budget_s - 120:
            break
        h_new = slsqp_lse(h_best, n, T, 15000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        h_new = slsqp_power(h_best, n, 12, 6000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Rebalance local search
    if time.time() - t0 < budget_s - 60:
        h_rb, c5_rb = rebalance_local(h_best, n, rng, steps=5000)
        if c5_rb < c5_best:
            c5_best = c5_rb
            h_best = h_rb
        h_new = slsqp_lse(h_best, n, 0.15, 8000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Perturbation restarts
    for pi in range(25):
        if time.time() - t0 > budget_s - 15:
            break
        sigma = 0.0015 + 0.004 * (pi % 5)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_new = slsqp_lse(h_p, n, 1.5, 4000, 1e-13)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            h_new = slsqp_lse(h_best, n, 0.3, 6000, 1e-14)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Final SLSQP polish
    if time.time() - t0 < budget_s - 5:
        h_new = slsqp_lse(h_best, n, 0.12, 12000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
