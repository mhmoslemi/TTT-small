def flip_delta(h, j, n):
    """Compute Δcorr when flipping h[j]"""
    d = 1.0 - 2.0 * h[j]
    v = 1.0 - h
    delta = np.zeros(2*n - 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def project_binary(h, n):
    target = n // 2
    h_b = (h >= 0.5).astype(float)
    diff = target - int(h_b.sum())
    if diff > 0:
        below = np.where(h_b == 0)[0]
        order = below[np.argsort(-h[below])]
        for i in range(diff):
            h_b[order[i]] = 1.0
    elif diff < 0:
        above = np.where(h_b == 1)[0]
        order = above[np.argsort(h[above])]
        for i in range(-diff):
            h_b[order[i]] = 0.0
    return h_b

def thue_morse(n):
    t = np.zeros(n, dtype=float)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def flip_delta(h, j, n):
    d = 1.0 - 2.0 * h[j]
    v = 1.0 - h
    delta = np.zeros(2 * n - 1)
    k1s, k1e = max(0, j), min(2 * n - 2, j + n - 1)
    for k in range(k1s, k1e + 1):
        idx = j + n - 1 - k
        if 0 <= idx < n:
            delta[k] += d * v[idx]
    k2s, k2e = max(0, n - 1 - j), min(2 * n - 2, 2 * n - 2 - j)
    for k in range(k2s, k2e + 1):
        idx = k - n + 1 + j
        if 0 <= idx < n:
            delta[k] -= d * h[idx]
    delta[n - 1] -= d * d
    return delta

def binary_sa(h_init, n, t_end, rng, max_iter=2000000):
    h = h_init.copy()

    temp = 0.05
    order = np.arange(n)
    stuck = 0
    for it in range(max_iter):
        if time.time() > t_end:

            stuck = 0

            if time.time() > t_end:
                break

            new_max = np.max(corr + delta)
            acceptance = new_max - c5 + 1e-12
            if acceptance < 0 or (temp > 1e-6 and rng.random() < np.exp(-acceptance / (temp * dx))):

                corr += delta

                    best_h = h.copy()
        if imp:
            stuck = 0
        else:
            stuck += 1

                break
    return best_h, best_c5 * dx

def slsqp_topk(h, n, T, k, maxiter, t_end):
    dx = 2.0 / n
    target = n / 2.0
    center = n - 1
    j_ar = np.arange(n)
    def _topk_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mk = min(k, len(corr))
        idx = np.argpartition(corr, -mk)[-mk:]
        topk = corr[idx]
        mc = topk.max()
        sh = T * (topk - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        obj = (mc + np.log(e.sum()) / T) * dx
        grad = np.zeros(n)
        for oi in range(mk):
            w = sw[oi]
            if w < 1e-11:
                continue
            shft = idx[oi] - center
            if shft >= 0:
                m1 = j_ar >= shft
                grad[m1] += w * (1.0 - x[j_ar[m1] - shft])
                m2 = (j_ar + shft) < n
                grad[m2] -= w * x[j_ar[m2] + shft]
            else:
                s2 = -shft
                m1 = (j_ar + s2) < n
                grad[m1] += w * (1.0 - x[j_ar[m1] + s2])
                m2 = j_ar >= s2
                grad[m2] -= w * x[j_ar[m2] - s2]
        grad *= dx
        return obj, grad
    def f(x):
        return _topk_obj(x)[0]
    def g(x):
        return _topk_obj(x)[1]
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h, method='SLSQP', jac=g, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return res.x
    except Exception:
        return h.copy()
