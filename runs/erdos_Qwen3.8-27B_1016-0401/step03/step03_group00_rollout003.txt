    """Compute new correlation after changing h[i] to new_val, given old corr."""

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_c5(h, n, shift_idx):
    shift = shift_idx - (n - 1)
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

_G_N, _G_TGT, _G_T, _G_K = 0, 0.0, 1.0, 10

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_G_T * (c - m))
    return (m + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_G_T * (c - m))
    w = e / e.sum()
    grad = np.zeros(_G_N)
    for wi, ci in zip(w, idx):
        if wi > 1e-12:
            grad += wi * shift_grad_c5(x, _G_N, ci)
    return grad * (2.0 / _G_N)

def _g_cfun(x):
    return x.sum() - _G_TGT

def _g_cjac(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, top_k, maxiter, ftol):
    global _G_N, _G_TGT, _G_T, _G_K
    _G_N = n
    _G_TGT = n / 2.0
    _G_T = float(T)
    _G_K = min(top_k, 2 * n - 1)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _g_cfun, 'jac': _g_cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def nesterov_subgrad(h_init, n, t_end, top_k=8):
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5a = compute_c5(h_init)
    ha = h_init.copy()
    it = 0
    while time.time() < t_end:
        it += 1
        beta = (it - 1.0) / (it + 1.5) if it > 1 else 0.0
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad_c5(h_y, n, ci)
        g *= dx / top_k
        alpha = 0.4 / (1.0 + 0.0015 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / it
        if it % 4000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < c5a:
                c5a, ha = c1, h_cur.copy()
            c2 = compute_c5(project_feasible(h_avg, n))
            if c2 < c5a:
                c5a, ha = c2, project_feasible(h_avg, n).copy()
    return ha, c5a

def local_flip(h, n, rng, max_rounds=100):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for _ in range(max_rounds):
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            old = h_cur[i]
            for nv in (0.0, 1.0):
                if abs(old - nv) < 1e-6:
                    continue
                h_cur[i] = nv
                ht = project_feasible(h_cur, n)
                c = compute_c5(ht)
                if c < c5_cur - 1e-10:
                    h_cur = ht
                    c5_cur = c
                    improved = True
                    break
                else:
                    h_cur[i] = old
        if not improved:
            break
    return h_cur, c5_cur

def local_pair_flip(h, n, rng, max_rounds=50):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for _ in range(max_rounds):
        improved = False
        for _ in range(200):
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            oi, oj = h_cur[i], h_cur[j]
            for ni in (0.0, 1.0):
                for nj in (0.0, 1.0):
                    if abs(oi - ni) < 1e-6 and abs(oj - nj) < 1e-6:
                        continue
                    h_cur[i], h_cur[j] = ni, nj
                    ht = project_feasible(h_cur, n)
                    c = compute_c5(ht)
                    if c < c5_cur - 1e-10:
                        h_cur = ht
                        c5_cur = c
                        improved = True
                        break
                    h_cur[i], h_cur[j] = oi, oj
                if improved:
                    break
        if not improved:
            break
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n0 = 280
        h_best = np.zeros(n0)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse
    t1 = t0 + budget_s * 0.20
    for T, tk, mi in [(64, 30, 1000), (16, 15, 1500), (4, 10, 2000)]:
        if time.time() > t1:
            break
        hn = do_slsqp(h_best, n, T, tk, mi, 1e-12)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 2: Nesterov
    t2 = t0 + budget_s * 0.40
    hs, c5s = nesterov_subgrad(h_best, n, t2, top_k=8)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs

    # Phase 3: Local single-flip search
    t3 = t0 + budget_s * 0.60
    while time.time() < t3:
        hf, c5f = local_flip(h_best, n, rng, max_rounds=80)
        if c5f < c5_best:
            c5_best, h_best = c5f, hf
        else:
            break

    # Phase 4: SLSQP fine
    t4 = t0 + budget_s * 0.72
    for T, tk, mi in [(2, 8, 2000), (1, 6, 3000), (0.5, 5, 4000)]:
        if time.time() > t4:
            break
        hn = do_slsqp(h_best, n, T, tk, mi, 1e-14)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 5: Pair flip + single flip
    t5 = t0 + budget_s * 0.85
    while time.time() < t5:
        hp, c5p = local_pair_flip(h_best, n, rng, max_rounds=40)
        if c5p < c5_best:
            c5_best, h_best = c5p, hp
        hf, c5f = local_flip(h_best, n, rng, max_rounds=60)
        if c5f < c5_best:
            c5_best, h_best = c5f, hf
        if c5p >= c5_best and c5f >= c5_best:
            break

    # Phase 6: Final polish
    if time.time() < budget_s - 10:
        hn = do_slsqp(h_best, n, 0.5, 5, 3000, 1e-15)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn
        hf, c5f = local_flip(h_best, n, rng, max_rounds=40)
        if c5f < c5_best:
            c5_best, h_best = c5f, hf

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
