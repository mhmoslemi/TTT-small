def compute_c5(h):

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():

            grad[m1] += (1.0 - h[j[m1] + s])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def _sm_val_and_grad(x, n, T, K):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    if K < len(corr):
        topk = np.argpartition(corr, -K)[-K:]
        c_sel = corr[topk]
        mc_sel = np.max(c_sel)
        sh = T * (c_sel - mc_sel)
        e = np.exp(sh)
        val = (mc_sel + np.log(e.sum()) / T) * (2.0 / n)
        sw = e / e.sum()
        center = n - 1
        grad = np.zeros(n)
        for ii in range(len(sw)):
            w = sw[ii]
            if w < 1e-14:
                continue
            grad += w * shift_grad(x, n, topk[ii] - center)
        return val, grad * (2.0 / n)
    else:
        sh = T * (corr - mc)
        e = np.exp(sh)
        val = (mc + np.log(e.sum()) / T) * (2.0 / n)
        sw = e / e.sum()
        center = n - 1
        grad = np.zeros(n)
        for oi in range(len(sw)):
            w = sw[oi]
            if w < 1e-14:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return val, grad * (2.0 / n)

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):
    bounds = [(0.0, 1.0)] * n
    flag = [False]
    def f_obj(x):
        if time.time() - t0 > tlim:
            flag[0] = True
        v, _ = _sm_val_and_grad(x, n, T, K)
        return v
    def f_jac(x):
        if time.time() - t0 > tlim:
            flag[0] = True
        _, g = _sm_val_and_grad(x, n, T, K)
        return g
    try:
        res = minimize(f_obj, h, method='L-BFGS-B', jac=f_jac,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol,
                                'gtol': tol * 0.05, 'maxls': 50,
                                'maxfun': maxiter * 4})
        if flag[0]:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def min_1d_convex(a, b):

            a[n - 1 - t_arr[m1n]] += (1.0 - h[i + t_arr[m1n]])
            a[n - 1 - t_arr[m2n]] -= h[i - t_arr[m2n]]
            b = corr - a * old
            t_star = min_1d_convex(a, b)
            if t_star != old:
                corr += a * (t_star - old)
                h[i] = t_star
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5_new = compute_c5(h)
        if c5_new < c5:
            c5 = c5_new
    return h, c5

def structured_perturb(h, n, rng):
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(7):
        f = rng.uniform(0.5, 12.0)
        ph = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.005, 0.035)
        pert += amp * np.sin(2 * np.pi * f * x / n + ph)
    pert += rng.normal(0, 0.005, n)
    return h + pert

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B at parent resolution
    schedules_280 = [(64, 3000, 200), (16, 3000, 100), (4, 4000, 50), (1.0, 5000, 30)]
    for T, mi, K in schedules_280:
        rem = budget_s - (time.time() - t0)
        if rem < 100:
            break
        h_new = do_lbfgsb(h_best, n, T, K, mi, 1e-14, t0, budget_s - 50)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Coordinate descent at parent resolution
    rem = budget_s - (time.time() - t0)
    if rem > 100:
        h_cd, c5_cd = coord_pure(h_best, n, 4, t0, t0 + rem - 80)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    # Phase 3: Interpolate to n=560 and optimize
    rem = budget_s - (time.time() - t0)
    if rem > 300:
        n2 = n * 2
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = project_feasible(h_hi, n2)
        c5_hi = compute_c5(h_hi)
        if c5_hi < c5_best:
            c5_best = c5_hi
            h_best = h_hi
            n = n2
        else:
            h_best = h_hi
            n = n2
        # L-BFGS-B at n=560
        schedules_560 = [(32, 4000, 200), (8, 4000, 80), (2, 5000, 40), (0.5, 6000, 20)]
        for T, mi, K in schedules_560:
            rem = budget_s - (time.time() - t0)
            if rem < 80:
                break
            h_new = do_lbfgsb(h_best, n, T, K, mi, 1e-14, t0, budget_s - 40)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 4: Coordinate descent at higher resolution
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h_cd, c5_cd = coord_pure(h_best, n, 6, t0, t0 + rem - 50)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    # Phase 5: Diverse restarts
    n_rest = 30
    for pi in range(n_rest):
        rem = budget_s - (time.time() - t0)
        if rem < 12:
            break
        if pi % 4 == 0:
            h_p = structured_perturb(h_best, n, rng)
        elif pi % 4 == 1:
            sig = 0.003 + 0.008 * (pi % 4)
            h_p = h_best + rng.normal(0, sig, n)
        elif pi % 4 == 2:
            h_p = h_best.copy()
            bs = max(1, n // 10)
            bi = int(rng.integers(0, n // bs))
            h_p[bi * bs:(bi + 1) * bs] = 1.0 - h_p[bi * bs:(bi + 1) * bs]
            h_p += rng.normal(0, 0.002, n)
        else:
            # Targeted: perturb near the worst-shift region
            corr = np.correlate(h_best, 1.0 - h_best, mode="full")
            worst_k = int(np.argmax(corr))
            shift = worst_k - (n - 1)
            h_p = h_best.copy()
            lo_i = max(0, -shift)
            hi_i = min(n, n - shift) if shift >= 0 else min(n, n + shift)
            lo_i = max(0, min(lo_i, hi_i))
            hi_i = max(lo_i + 1, hi_i)
            seg = h_p[lo_i:hi_i].copy()
            seg = 1.0 - seg
            h_p[lo_i:hi_i] = seg + rng.normal(0, 0.003, hi_i - lo_i)
        h_p = project_feasible(h_p, n)
        T_r = 1.0 if pi < n_rest // 2 else 0.5
        h_new = do_lbfgsb(h_p, n, T_r, 50, 2000, 1e-13, t0, budget_s - 5)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Final L-BFGS-B + CD polish
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h_new = do_lbfgsb(h_best, n, 0.5, 15, 8000, 1e-15, t0, t0 + rem - 30)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_cd, c5_cd = coord_pure(h_best, n, 5, t0, t0 + rem - 10)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
