    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        # Nesterov extrapolation
        beta = it / (it + 1.5) if it > 0 else 0.0
        h_y = h_cur + beta * (h_cur - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        
        # Compute objective and gradient
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        
        # Subgradient: average of top-k gradients
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h_y, n, ci - (n - 1))
        g *= dx / top_k
        
        # Learning rate with decay

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        v1 = j >= shift
        g[v1] += (1.0 - h[j[v1] - shift])
        v2 = j + shift < n
        g[v2] -= h[j[v2] + shift]
    else:
        s = -shift
        v1 = j >= s
        g[v1] += (1.0 - h[j[v1] - s])
        v2 = j + s < n
        g[v2] -= h[j[v2] - s]
    return g

def topk_corrs(h, k):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-k:]
    return corr, idx

def proj_nesterov(h_init, n, t_end, top_k=8, max_iter=300000):
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    for it in range(max_iter):
        if time.time() > t_end:
            break
        beta = it / (it + 1.5) if it > 0 else 0.0
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h_y, n, ci - (n - 1))
        g *= dx / top_k
        lr = 2.0 / (2.0 + 0.05 * it)

        h_cur = h_new
        if (it + 1) % 3000 == 0:
            c5v = compute_c5(h_cur)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_cur.copy()
    c5v = compute_c5(h_cur)
    if c5v < best_c5:
        best_c5, best_h = c5v, h_cur.copy()
    return best_h, best_c5

# Global state for SLSQP callbacks
_GN = 0
_GTARGET = 0.0
_GT = 1.0
_GK = 10

def _gobj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_GK:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_GT * (c - m))
    return (m + np.log(e.sum()) / _GT) * (2.0 / _GN)

def _gjac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_GK:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_GT * (c - m))
    w = e / e.sum()
    grad = np.zeros(_GN)
    for wi, ci in zip(w, idx):
        if wi > 1e-10:
            grad += wi * shift_grad(x, _GN, ci - (_GN - 1))
    return grad * (2.0 / _GN)

def _gfun(x):
    return x.sum() - _GTARGET

def _gjacf(x):
    return np.ones(_GN)

def slsqp_solve(h0, n, T, top_k, maxiter=5000, ftol=1e-15):
    global _GN, _GTARGET, _GT, _GK
    _GN = n
    _GTARGET = n / 2.0
    _GT = float(T)
    _GK = min(top_k, 2 * n - 1)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_gobj, h0, method='SLSQP', jac=_gjac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _gfun, 'jac': _gjacf}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0.copy()

def lag_equalize(h0, n, t_end, n_lags=5, max_rounds=2000):
    """Alternating minimization on individual active lags."""
    dx = 2.0 / n
    h_cur = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    corr_all = np.correlate(h_cur, 1.0 - h_cur, mode="full")
    top_idx = np.argsort(corr_all)[-n_lags:]
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        # Update active set every 50 rounds
        if rnd % 50 == 0:
            corr_all = np.correlate(h_cur, 1.0 - h_cur, mode="full")
            top_idx = np.argsort(corr_all)[-n_lags:]
        for ci in top_idx:
            shift = ci - (n - 1)
            g = shift_grad(h_cur, n, shift)
            # Line search: reduce this correlation without increasing max
            base_corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
            base_max = base_corr.max() * dx
            best_step_h = h_cur
            best_step_val = compute_c5(h_cur)
            for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005]:
                trial = project_feasible(h_cur - alpha * g * dx, n)
                tc5 = compute_c5(trial)
                if tc5 < best_step_val:
                    best_step_val = tc5
                    best_step_h = trial
            h_cur = best_step_h
            c5v = compute_c5(h_cur)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_cur.copy()
    return best_h, best_c5

def binary_project(h, n):
    """Project to nearest 0-1 pattern with exactly n/2 ones, then smooth."""
    target = n // 2
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    # Smooth: blend with original
    h_smooth = 0.7 * h + 0.3 * h_bin
    return project_feasible(h_smooth, n)

def _perturb(h, n, rng, mode):
    if mode == 0:
        return project_feasible(h + rng.normal(0, 0.003, n), n)
    elif mode == 1:
        hp = h.copy()
        lb = rng.integers(0, n - 15)
        ln = rng.integers(4, 16)
        hp[lb:lb+ln] = 1.0 - hp[lb:lb+ln]
        return project_feasible(hp, n)
    elif mode == 2:
        hp = h.copy()
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        hp[[i, j]] = hp[[j, i]]
        return project_feasible(hp, n)
    else:
        sig = 0.004
        hp = h + rng.normal(0, sig, n)
        hp[rng.integers(0, n)] = 1.0
        return project_feasible(hp, n)
