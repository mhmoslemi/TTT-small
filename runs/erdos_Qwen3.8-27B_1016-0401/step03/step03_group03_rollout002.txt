from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    g = np.zeros(n)
    if shift >= 0:
        m = n - shift
        g[:m] += (1.0 - h[shift:n])
        if shift > 0:
            g[shift:n] -= h[0:m]
    else:
        s = -shift
        m = n - s
        g[s:n] += (1.0 - h[0:m])
        g[0:m] -= h[s:n]
    return g

def smooth_obj(x, n, T, K):
    corr = np.correlate(x, 1.0 - x, mode="full")
    ti = np.argsort(corr)[-K:]
    mc = corr[ti].max()
    ev = np.exp(T * (corr[ti] - mc))
    return (mc + np.log(ev.sum()) / T) * (2.0 / n)

def smooth_grad(x, n, T, K):
    corr = np.correlate(x, 1.0 - x, mode="full")
    ti = np.argsort(corr)[-K:]
    mc = corr[ti].max()
    ev = np.exp(T * (corr[ti] - mc))
    w = ev / ev.sum()
    center = n - 1
    dx = 2.0 / n
    g = np.zeros(n)

        if wi < 1e-12:
            continue
        g += wi * shift_grad(x, n, ii - center)
    return g * dx

def slsqp_step(h, n, T, K, maxiter):
    _n, _T, _K = n, float(T), K
    def _obj(x):
        return smooth_obj(x, _n, _T, _K)
    def _jac(x):
        return smooth_grad(x, _n, _T, _K)
    def _cf(x):
        return x.sum() - n / 2.0
    def _cj(x):
        return np.ones(_n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_obj, h, method='SLSQP', jac=_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return res.x
    except Exception:

        h_new = slsqp_step(h_best, n, T, 8, 6000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 2: Accelerated gradient (multi-shift, adaptive)
    t_ag_end = min(t0 + budget_s * 0.45, time.time() + 350)
    h_ag, c5_ag = accel_gradient(h_best, n, t0, t_ag_end, K=7, alpha0=0.06, gamma=0.9998)
    if c5_ag < c5_best:
        c5_best, h_best = c5_ag, h_ag

    # Phase 3: Second SLSQP round at low temperatures
    for T in [0.5, 0.3, 0.2, 0.15, 0.1, 0.08]:
        if time.time() - t0 > budget_s - 350:
            break
        h_new = slsqp_step(h_best, n, T, 10, 10000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 4: Aggressive local coordinate search
    tl_end = time.time() + min(200, max(60, budget_s - (time.time() - t0) - 80))
    h_ls, c5_ls = local_search(h_best, n, rng, 80000, tl_end)
    if c5_ls < c5_best:
        c5_best, h_best = c5_ls, h_ls

    # Phase 5: SLSQP polish after local search
    if time.time() - t0 < budget_s - 40:
        h_new = slsqp_step(h_best, n, 0.4, 8, 8000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 6: More local search on the polished solution
    if time.time() - t0 < budget_s - 30:
        h_ls2, c5_ls2 = local_search(h_best, n, rng, 40000, time.time() + 80)
        if c5_ls2 < c5_best:
            c5_best, h_best = c5_ls2, h_ls2

    # Phase 7: Final SLSQP
    if time.time() - t0 < budget_s - 15:
        h_new = slsqp_step(h_best, n, 0.2, 12, 12000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
