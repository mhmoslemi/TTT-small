        if sw[oi] < 1e-13:
            continue
        g += sw[oi] * corr_grad_pair(x, _GN, oi - center)
    return g * dx

def _eq_fun(x):
    return x.sum() - _GT

def _eq_jac(x):
    return np.ones(_GN)

def slsqp_solve(h0, n, T, maxiter):
    global _GN, _GT, _GTTEMP
    _GN = n
    _GT = n / 2.0
    _GTTEMP = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_lse_obj, h0, method='SLSQP', jac=_lse_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return res.x
    except Exception:
        return h0.copy()

def nesterov_optimize(h0, n, t0, budget, K=8, Tsoft=5.0, mom_decay=0.003):
    h_cur = h0.copy()
    h_prev = h0.copy()
    best = compute_c5(h_cur)
    best_h = h_cur.copy()
    dx = 2.0 / n
    it = 0
    while time.time() - t0 < budget:
        beta = it / (it + 4.0)
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top_idx = np.argsort(corr)[-K:]
        vals = corr[top_idx]
        w = np.exp(Tsoft * (vals - vals.max()))
        w /= w.sum()
        g = np.zeros(n)
        for wk, ik in zip(w, top_idx):
            g += wk * corr_grad_pair(h_y, n, ik - (n - 1))
        g *= dx
        alpha = 0.12 / (1.0 + mom_decay * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        it += 1
        if it % 15000 == 0:
            cv = compute_c5(h_cur)
            if cv < best:
                best = cv
                best_h = h_cur.copy()
    cv = compute_c5(project_feasible(h_cur, n))
    if cv < best:
        best = cv
        best_h = project_feasible(h_cur, n).copy()
    return best_h, best

def local_search(h0, n, rng, steps):
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    dx = 2.0 / n
    center = n - 1
    for _ in range(steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        top2 = np.argsort(corr)[-2:]
        for ak in top2:
            shift = ak - center
            g = corr_grad_pair(h_cur, n, shift) * dx
            idx_pos = np.where(g > 0.02)[0]
            idx_neg = np.where(g < -0.02)[0]
            if len(idx_pos) == 0 or len(idx_neg) == 0:
                continue
            n_try = min(3, len(idx_pos), len(idx_neg))
            for _t in range(n_try):
                ip = idx_pos[rng.integers(len(idx_pos))]
                ii = idx_neg[rng.integers(len(idx_neg))]
                delta = min(0.04, h_cur[ip] * 0.7, (1.0 - h_cur[ii]) * 0.7)
                if delta < 0.001:
                    continue
                h_try = h_cur.copy()
                h_try[ip] -= delta
                h_try[ii] += delta
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < c5_cur - 5e-10:
                    h_cur = h_try
                    c5_cur = c5_try
                    break
        if time.time() > 999999:
            break
    return h_cur, c5_cur

def block_move_search(h0, n, rng, steps, block_sz=4):
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    dx = 2.0 / n
    for _ in range(steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        ak = int(np.argmax(corr))
        shift = ak - (n - 1)
        g = corr_grad_pair(h_cur, n, shift) * dx
        pos = np.where(g > 0.015)[0]
        neg = np.where(g < -0.015)[0]
        if len(pos) < 2 or len(neg) < 2:
            continue
        bs = min(block_sz, len(pos), len(neg))
        ipos = pos[rng.integers(0, len(pos), bs)]
        ineg = neg[rng.integers(0, len(neg), bs)]
        d = 0.02
        h_try = h_cur.copy()
        ok = True
        for ip, ii in zip(ipos, ineg):
            dv = min(d, h_cur[ip] * 0.6, (1.0 - h_cur[ii]) * 0.6)
            if dv < 0.0005:
                ok = False
                break
            h_try[ip] -= dv
            h_try[ii] += dv
        if not ok:
            continue
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < c5_cur - 3e-10:
            h_cur = h_try
            c5_cur = c5_try
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse-to-fine
    temps1 = [96, 48, 24, 12, 6, 3, 2, 1.2, 0.8, 0.5, 0.3, 0.2]
    for T in temps1:
        if time.time() - t0 > budget_s - 600:
            break
        h_new = slsqp_solve(h_best, n, T, 6000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Nesterov
    if time.time() - t0 < budget_s - 350:
        h_sg, c5_sg = nesterov_optimize(h_best, n, t0, budget_s - 340, K=8, Tsoft=5.0)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase 3: Fine SLSQP
    temps3 = [1.5, 0.8, 0.5, 0.3, 0.2, 0.12, 0.08, 0.05]
    for T in temps3:
        if time.time() - t0 > budget_s - 250:
            break
        h_new = slsqp_solve(h_best, n, T, 15000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Local coordinate search
    if time.time() - t0 < budget_s - 120:
        h_ls, c5_ls = local_search(h_best, n, rng, 12000)
        if c5_ls < c5_best:
            c5_best = c5_ls
            h_best = h_ls
        h_new = slsqp_solve(h_best, n, 0.4, 8000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Block moves
    if time.time() - t0 < budget_s - 60:
        h_bl, c5_bl = block_move_search(h_best, n, rng, 5000, block_sz=4)
        if c5_bl < c5_best:
            c5_best = c5_bl
            h_best = h_bl
        h_new = slsqp_solve(h_best, n, 0.25, 6000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Perturbation restarts
    for pi in range(20):
        if time.time() - t0 > budget_s - 20:
            break
        sigma = 0.002 + 0.005 * (pi % 5)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_new = slsqp_solve(h_p, n, 1.5, 4000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Final polish
    if time.time() - t0 < budget_s - 5:
        h_new = slsqp_solve(h_best, n, 0.15, 10000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        h_new = slsqp_solve(h_best, n, 0.1, 8000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
