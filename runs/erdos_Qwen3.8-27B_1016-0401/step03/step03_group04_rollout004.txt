                    c5_t = compute_c5(h_curr)
                    if c5_t < c5_best - 1e-12:
                        c5_best = c5_t
                    else:
                        h_curr[i] = old
                        h_curr = project_feasible(h_curr, n)
                        break
    return h_curr, c5_best

def struct_perturb(h, n, rng):
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(1.0, 6.0)
        p = rng.uniform(0.0, 2.0 * np.pi)
        a = rng.uniform(0.005, 0.03)
        pert += a * np.sin(2.0 * np.pi * f * x / n + p)
    pert += rng.normal(0.0, 0.005, n)
    return h + pert

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Soft-max L-BFGS-B coarse to fine
    schedules = [(100, 400, 8000), (50, 200, 8000), (25, 150, 8000),
                 (10, 100, 8000), (5, 60, 10000), (2, 30, 10000),
                 (1, 20, 12000), (0.5, 15, 12000), (0.3, 10, 15000)]
    for T, K, mi in schedules:
        rem = budget_s - (time.time() - t0)
        if rem < 120:
            break
        h_new = do_lbfgsb(h_best, n, T, K, mi, 1e-15, t0, budget_s - 80)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Top-m exact subgradient
    rem = budget_s - (time.time() - t0)
    if rem > 200:
        h_sg, c5_sg = top_m_subgrad(h_best, n, t0, t0 + rem - 180, m=3,
                                     n_iters=300000)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase 3: L-BFGS-B fine polish
    for T, K, mi in [(0.3, 8, 20000), (0.2, 6, 20000), (0.15, 5, 25000)]:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        h_new = do_lbfgsb(h_best, n, T, K, mi, 1e-16, t0, budget_s - 40)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Coordinate descent
    rem = budget_s - (time.time() - t0)
    if rem > 150:
        h_cd, c5_cd = coord_descent(h_best, n, t0, t0 + rem - 130, n_passes=200)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    # Phase 5: More subgradient after CD
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        h_sg2, c5_sg2 = top_m_subgrad(h_best, n, t0, t0 + rem - 100, m=2,
                                       n_iters=200000)
        if c5_sg2 < c5_best:
            c5_best = c5_sg2
            h_best = h_sg2
        h_new = do_lbfgsb(h_best, n, 0.2, 6, 10000, 1e-16, t0, budget_s - 30)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Diverse restarts
    n_restarts = 12
    for pi in range(n_restarts):
        rem = budget_s - (time.time() - t0)
        if rem < 50:
            break
        if pi % 3 == 0:
            h_p = struct_perturb(h_best, n, rng)
        elif pi % 3 == 1:
            sig = 0.003 + 0.008 * (pi % 4)
            h_p = h_best + rng.normal(0.0, sig, n)
        else:
            h_p = h_best.copy()
            bs = n // 10
            bi = int(rng.integers(0, 10))
            h_p[bi * bs:(bi + 1) * bs] = 1.0 - h_p[bi * bs:(bi + 1) * bs]
            h_p += rng.normal(0.0, 0.002, n)
        h_p = project_feasible(h_p, n)
        h_new = do_lbfgsb(h_p, n, 1.0, 20, 5000, 1e-14, t0, budget_s - 15)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            h_sg3, c5_sg3 = top_m_subgrad(h_best, n, t0, min(budget_s - 15,
                      time.time() + 20), m=2, n_iters=50000)
            if c5_sg3 < c5_best:
                c5_best = c5_sg3
                h_best = h_sg3

    # Phase 7: Resolution doubling
    rem = budget_s - (time.time() - t0)
    if rem > 300:
        n2 = n * 2
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = project_feasible(h_hi, n2)
        c5_hi = compute_c5(h_hi)
        if c5_hi <= c5_best + 0.001:
            h_hi_best = h_hi
            c5_hi_best = min(c5_hi, c5_best)
            for T, K, mi in [(10, 80, 6000), (3, 30, 8000), (1, 15, 10000),
                             (0.4, 8, 12000)]:
                rem = budget_s - (time.time() - t0)
                if rem < 80:
                    break
                h_new = do_lbfgsb(h_hi_best, n2, T, K, mi, 1e-15, t0,
                                  budget_s - 40)
                c5 = compute_c5(h_new)
                if c5 < c5_hi_best:
                    c5_hi_best = c5
                    h_hi_best = h_new
            h_cd2, c5_cd2 = coord_descent(h_hi_best, n2, t0,
                                          budget_s - 20, n_passes=50)
            if c5_cd2 < c5_hi_best:
                c5_hi_best = c5_cd2
                h_hi_best = h_cd2
            if c5_hi_best < c5_best:
                c5_best = c5_hi_best
                h_best = h_hi_best
                n = n2

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
