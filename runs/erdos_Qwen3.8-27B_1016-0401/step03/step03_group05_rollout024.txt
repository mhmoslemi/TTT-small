def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0

        if abs(s - target) < 1e-13:
            break

            lo = mid
        else:
            hi = mid

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def topk_gradient(y, n, K, power):
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    corr = np.correlate(y, 1.0 - y, mode="full")
    if K >= n_corr:
        idx = np.arange(n_corr)
        w = np.ones(K) / K
    else:
        idx = np.argpartition(corr, -K)[-K:]
        w = corr[idx].copy()
        mc = np.max(w)
        w = (w / max(mc, 1e-15)) ** power
        s = w.sum()
        if s > 1e-15:
            w = w / s
        else:
            w = np.ones(K) / K
    g = np.zeros(n)
    for i in range(len(idx)):
        g += w[i] * shift_grad(y, n, idx[i] - center)
    return g * dx, corr
