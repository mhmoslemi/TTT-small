def project_feasible(y, n):
    target = n / 2.0

        else: hi = mid

def lse_smooth(h, n, T, k_active):
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = min(k_active, 2*n-1)
    idx = np.argpartition(corr, -k)[-k:]
    vals = corr[idx]
    mc = vals.max()
    sh = T * (vals - mc)
    e = np.exp(sh)
    es = e.sum()
    obj = (mc + np.log(es)/T) * (2.0/n)
    grad = np.zeros(n)
    center = n-1
    j = np.arange(n)

    except Exception:
        return h.copy()

def subgrad_refine(h, n, t_deadline, max_rounds=500):
    """Direct subgradient on the max correlation with diminishing step."""
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    lr = 0.01
    for _ in range(max_rounds):
        if time.time() > t_deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = corr.max()
        active = np.where(np.abs(corr - max_val) < 1e-8)[0]
        grad = np.zeros(n)
        center = n - 1
        j = np.arange(n)
        for ai in active:
            s = ai - center
            if s >= 0:
                m1 = j >= s
                grad[m1] += (1.0 - h[j[m1] - s])
                m2 = (j + s) < n
                grad[m2] -= h[j[m2] + s]
            else:
                s2 = -s
                m1 = (j + s2) < n
                grad[m1] += (1.0 - h[j[m1] + s2])
                m2 = j >= s2
                grad[m2] -= h[j[m2] - s2]
        grad /= len(active)
        grad *= 2.0 / n
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            lr *= 0.7
            if lr < 1e-12:
                break
    return best_h, best_c5

def perturb_struct(h, n, rng):
    h_p = h.copy()
    r = rng.random()
    if r < 0.25:
        freq = rng.integers(2, 14)
        amp = rng.uniform(0.008, 0.04)
        x = np.arange(n) * 2.0 / n
        h_p += amp * np.sin(2.0 * np.pi * freq * x / 2.0 + rng.uniform(0, 6.28))
    elif r < 0.45:
        start = rng.integers(0, max(1, n - 12))
        ln = rng.integers(4, min(18, n - start))
        h_p[start:start + ln] = 1.0 - h_p[start:start + ln]
    elif r < 0.65:
        h_p += rng.uniform(0.004, 0.015) * rng.normal(0, 1, n)
    elif r < 0.85:
        b = rng.integers(4, 25)
        s1 = rng.integers(0, max(1, n - b))
        s2 = rng.integers(0, max(1, n - b))
        tmp = h_p[s1:s1 + b].copy()
        h_p[s1:s1 + b] = h_p[s2:s2 + b]
        h_p[s2:s2 + b] = tmp
    else:
        mask = rng.random(n) < 0.12
        h_p[mask] = rng.random(int(mask.sum()))
    return project_feasible(np.clip(h_p, 0, 1), n)

def swap_local_search(h, n, t_deadline, max_passes=200):
    """Try random position swaps to reduce max correlation."""
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    rng_ls = np.random.default_rng(12345)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for _ in range(max_passes):
        if time.time() > t_deadline:
            break
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = rng_ls.choice(ones)
        j = rng_ls.choice(zeros)
        h_try = h.copy()
        h_try[i], h_try[j] = h_try[j], h_try[i]
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try
            h = h_try
            ones[i] = j
            zeros[j] = i
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP with decreasing T (continuation)
    schedule = [(20, 20, 2500), (8, 12, 3000), (4, 10, 3000),
                (2, 8, 3500), (1, 6, 3500), (0.5, 5, 4000), (0.2, 4, 4000)]
    for T, k, mi in schedule:
        if time.time() > t_end - 200:
            break
        h_new = slsqp_round(h_best, n, T, k, mi)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 2: FISTA continuation (low T for fine tuning)
    rem = t_end - time.time()
    if rem > 80:
        h_f, c5_f = fista_run(h_best, n, time.time() + rem * 0.25, 10.0, 0.05, 6, int(rem * 20))
        h_f = project_feasible(h_f, n)
        c5_f = compute_c5(h_f)
        if c5_f < c5_best:
            c5_best, h_best = c5_f, h_f

    # Phase 3: Subgradient refinement
    if time.time() < t_end - 40:
        h_s, c5_s = subgrad_refine(h_best, n, time.time() + 40, 300)
        if c5_s < c5_best:
            c5_best, h_best = c5_s, h_s

    # Phase 4: More SLSQP polish
    for T, k in [(0.5, 4), (0.3, 3), (0.1, 2)]:
        if time.time() > t_end - 80:
            break
        h_new = slsqp_round(h_best, n, T, k, 5000)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 5: Perturbation + re-optimization rounds
    rem = t_end - time.time()
    n_rounds = max(1, int(rem / 50))
    for ri in range(n_rounds):
        if time.time() > t_end - 20:
            break
        h_p = perturb_struct(h_best, n, rng)
        for T2, k2 in [(2, 8), (0.5, 4)]:
            if time.time() > t_end - 10:
                break
            h_new = slsqp_round(h_p, n, T2, k2, 2500)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best, h_best = c5, h_new

    # Phase 6: Swap local search on near-binary solution
    if time.time() < t_end - 30:
        h_bin = (h_best > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n)
        h_sw, c5_sw = swap_local_search(h_bin, n, t_end - 5, 500)
        if c5_sw < c5_best:
            h_new = slsqp_round(h_sw, n, 0.5, 4, 3000)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best, h_best = c5, h_new

    # Phase 7: Final subgradient + SLSQP polish
    if time.time() < t_end - 15:
        h_s, c5_s = subgrad_refine(h_best, n, time.time() + 10, 200)
        if c5_s < c5_best:
            c5_best, h_best = c5_s, h_s
        if time.time() < t_end - 5:
            h_new = slsqp_round(h_best, n, 0.1, 2, 2000)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best, h_best = c5, h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
