        h_y = h_cur + beta * (h_cur - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        h_y = project_feasible(h_y, n)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        m = c.max()
        T_eff = T0 / (1.0 + 0.05 * (it // 2000))
        T_eff = max(T_eff, 0.1)
        e = np.exp(T_eff * (c - m))
        w = e / e.sum()
        g = np.zeros(n)
        for wi, ci in zip(w, idx):
            if wi > 1e-12:
                g += wi * shift_grad(h_y, n, ci - (n - 1))
        g *= dx
        lr = base_lr / (1.0 + 0.003 * it)
        h_new = project_feasible(h_y - lr * g, n)
        h_prev = h_cur
        h_cur = h_new
        if (it + 1) % 3000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < c5a:
                c5a = c1
                ha = h_cur.copy()
    return ha, c5a

def _perturb(h, n, rng, pi):
    mode = pi % 5
    if mode == 0:
        sig = 0.002 + 0.002 * (pi // 5) % 4
        hp = project_feasible(h + rng.normal(0, sig, n), n)
    elif mode == 1:
        hp = h.copy()
        lb = rng.integers(0, n - 30)
        ln = rng.integers(8, 35)
        hp[lb:lb+ln] = 1.0 - hp[lb:lb+ln]
        hp = project_feasible(hp, n)
    elif mode == 2:
        hp = h.copy()
        i = rng.integers(0, n - 8)
        ln = rng.integers(5, 20)
        hp[i:i+ln] = hp[i:i+ln][::-1]
        hp = project_feasible(hp, n)
    elif mode == 3:
        hp = h.copy()
        lb = rng.integers(0, n - 20)
        ln = rng.integers(10, 25)
        hp[lb:lb+ln] = hp[lb:lb+ln] * 0.6
        hp = project_feasible(hp, n)
    else:
        hp = h.copy()
        for _ in range(2):
            hp[rng.integers(0, n)] = 1.0 - hp[rng.integers(0, n)]
        hp = project_feasible(hp, n)
    return hp

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n0 = 280
        h_best = np.zeros(n0)
        h_best[::2] = 1.0

    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Progressive SLSQP (40% time)
    t1 = t0 + budget_s * 0.40
    sched = [
        (48, 40, 1200, 1e-13),
        (24, 35, 1500, 1e-14),
        (12, 30, 2000, 1e-14),
        (6, 25, 2500, 1e-14),
        (3, 20, 3000, 1e-15),
        (2, 16, 3500, 1e-15),
        (1.5, 14, 4000, 1e-15),
        (1.0, 12, 5000, 1e-15),
        (0.7, 10, 6000, 1e-16),
        (0.5, 8, 7000, 1e-16),
    ]
    for T, tk, mi, ft in sched:
        if time.time() > t1:
            break
        hn = _slsqp(h_best, n, T, tk, mi, ft)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 2: Momentum optimization with periodic SLSQP rescue (25% time)
    t2 = t0 + budget_s * 0.65
    for T_m in [6.0, 3.0, 1.5]:
        if time.time() > t2:
            break
        t_chunk = min(t2, time.time() + budget_s * 0.06)
        hs, c5s = _momentum_opt(h_best, n, t_chunk, rng, top_k=12, T0=T_m)
        if c5s < c5_best:
            c5_best, h_best = c5s, hs
        hn = _slsqp(h_best, n, 1.0, 10, 3000, 1e-15)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 3: Targeted multi-restart (25% time)
    t3 = t0 + budget_s * 0.90
    for pi in range(100):
        if time.time() > t3:
            break
        hp = _perturb(h_best, n, rng, pi)
        hn = _slsqp(hp, n, 3.0, 14, 2000, 1e-14)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn
            hn2 = _slsqp(h_best, n, 0.8, 8, 4000, 1e-16)
            hn2 = project_feasible(hn2, n)
            c5_2 = compute_c5(hn2)
            if c5_2 < c5_best:
                c5_best, h_best = c5_2, hn2

    # Phase 4: Final ultra-fine polish (10% time)
    t4 = t0 + budget_s - 5
    if time.time() < t4:
        hn = _slsqp(h_best, n, 0.4, 6, 6000, 1e-16)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn
        if time.time() < t4:
            hn = _slsqp(h_best, n, 0.2, 5, 5000, 1e-16)
            hn = project_feasible(hn, n)
            c5 = compute_c5(hn)
            if c5 < c5_best:
                c5_best, h_best = c5, hn

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
