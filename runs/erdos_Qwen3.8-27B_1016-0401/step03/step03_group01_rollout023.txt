        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:

    return np.clip(y - mid, 0.0, 1.0)

def corr_grad_single(h, n, lag_idx):
    """Gradient of correlate(h, 1-h, 'full')[lag_idx] w.r.t. h"""
    grad = np.zeros(n)
    center = n - 1
    shift = lag_idx - center
    j = np.arange(n)
    if shift >= 0:
        m1 = j >= shift
        grad[m1] += (1.0 - h[j[m1] - shift])
        m2 = (j + shift) < n
        grad[m2] -= h[j[m2] + shift]
    else:
        s2 = -shift
        m1 = (j + s2) < n
        grad[m1] += (1.0 - h[j[m1] + s2])
        m2 = j >= s2
        grad[m2] -= h[j[m2] - s2]
    return grad

def soft_topk_obj_grad(h, n, T, k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = np.max(topk)
    sh = T * (topk - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    obj = (mc + np.log(e.sum()) / T) * dx
    grad = np.zeros(n)
    for oi in range(mk):
        w = sw[oi]
        if w < 1e-12:
            continue
        grad += w * corr_grad_single(h, n, idx[oi])
    grad *= dx
    return obj, grad

def hinge_topk_obj_grad(h, n, threshold, k):
    """Sum of (max(0, corr[i]*dx - threshold))^2 for top-k above threshold"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    active = c > threshold
    n_active = active.sum()
    if n_active == 0:
        return 0.0, np.zeros(n)
    # Focus on top k among active
    if n_active > k:
        top_idx_in_active = np.argpartition(c[active], -k)[-k:]
        active_idx = np.where(active)[0]
        idx = active_idx[top_idx_in_active]
    else:
        idx = np.where(active)[0]
    excess = c[idx] - threshold
    obj = 0.5 * np.sum(excess**2)
    grad = np.zeros(n)
    for ii in range(len(idx)):
        w = excess[ii]
        grad += w * corr_grad_single(h, n, idx[ii])
    grad *= dx
    return obj, grad

def slsqp_opt(h, n, T, k, maxiter, ftol, mode='soft'):
    target = n / 2.0
    if mode == 'soft':
        def obj(x): return soft_topk_obj_grad(x, n, T, k)[0]
        def jac(x): return soft_topk_obj_grad(x, n, T, k)[1]
    else:
        def obj(x): return hinge_topk_obj_grad(x, n, T, k)[0]
        def jac(x): return hinge_topk_obj_grad(x, n, T, k)[1]
    def cf(x): return x.sum() - target
    def cj(x): return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def fista_opt(h_init, n, t_end, lr0=0.01, k_lags=5, T=3.0):

            t_end = t0 + (budget_s - (time.time() - t0)) * 0.10
            h_f, c5_f = fista_opt(h_best, n, t_end, lr0, kl, T)
            h_f = project_feasible(h_f, n)
            c5_f = compute_c5(h_f)
            if c5_f < c5_best:
                c5_best = c5_f
                h_best = h_f

    # Phase 3: Hinge-loss SLSQP targeting current worst lags
    thr = c5_best * 0.999
    for _ in range(3):
        rem = budget_s - (time.time() - t0)
        if rem < 70:
            break
        h_new = slsqp_opt(h_best, n, thr, 5, 5000, 1e-19, 'hinge')
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        thr = c5_best * 0.999

    # Phase 4: Diverse exploration via perturbation + SLSQP
    rem = budget_s - (time.time() - t0)
    n_trials = max(8, int(rem / 22))
    for ti in range(n_trials):
        if time.time() - t0 > budget_s - 35:
            break
        h_p = gen_perturb(h_best, n, rng, ti % 6)
        h_p = project_feasible(h_p, n)
        h_new = slsqp_opt(h_p, n, 4, 8, 2000, 1e-17, 'soft')
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Very fine SLSQP (tiny T, top 2-3)
    for T, k in [(0.5, 3), (0.2, 3), (0.1, 2)]:
        rem = budget_s - (time.time() - t0)
        if rem < 40:
            break
        h_new = slsqp_opt(h_best, n, T, k, 8000, 1e-20, 'soft')
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Alternative resolutions
    rem = budget_s - (time.time() - t0)
    for n_alt in [200, 320]:
        if n_alt == n:
            continue
        if time.time() - t0 > budget_s - 50:
            break
        if rem < 80:
            break
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = project_feasible(h_alt, n_alt)
        for T, k in [(16, 30), (4, 10), (1, 5)]:
            if time.time() - t0 > budget_s - 20:
                break
            h_new = slsqp_opt(h_alt, n_alt, T, k, 3000, 1e-18, 'soft')
            h_new = project_feasible(h_new, n_alt)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
                n = n_alt
                rem = budget_s - (time.time() - t0)

    # Final polish
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        h_new = slsqp_opt(h_best, n, 0.5, 3, 5000, 1e-20, 'soft')
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
