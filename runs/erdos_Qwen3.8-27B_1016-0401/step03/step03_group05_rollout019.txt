import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    """Correct gradient of C(shift) w.r.t. h."""
    j = np.arange(n)
    grad = np.zeros(n)
    if shift == 0:
        grad[:] = 1.0 - 2.0 * h
    elif shift > 0:
        m1 = j < (n - shift)
        m2 = j >= shift
        grad[m1] += (1.0 - h[j[m1] + shift])
        grad[m2] -= h[j[m2] - shift]
    else:
        s = -shift
        m1 = j >= s
        m2 = j < (n - s)
        grad[m1] += (1.0 - h[j[m1] - s])
        grad[m2] -= h[j[m2] + s]
    return grad

def fast_nesterov(h_init, n, t0, time_budget, max_iters=500000):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_best = h_init.copy()
    c5_best = compute_c5(h_init)
    T_smooth = 32.0

    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        frac = min((it + 1) / max_iters, 1.0)
        K = max(1, int(1 + 49 * frac * frac))

        beta = (float(it) / (it + 1.5)) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)

        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)

        if K <= 1:
            idx = int(np.argmax(corr))
            g = shift_grad(h_y, n, idx - center) * dx
        else:
            top_idx = np.argpartition(corr, -K)[-K:]
            top_vals = corr[top_idx]
            w = (top_vals / mc) ** 12.0
            ws = w.sum()
            if ws < 1e-15:
                w = np.ones(K) / K
            else:
                w /= ws
            g = np.zeros(n)
            for i in range(K):
                g += w[i] * shift_grad(h_y, n, top_idx[i] - center)
            g *= dx

        alpha = 0.5 / (1.0 + 0.001 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)

        h_prev = h_sg
        h_sg = h_new

        if (it + 1) % 10000 == 0:
            c5c = compute_c5(h_sg)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h_sg.copy()

    c5f = compute_c5(h_sg)
    if c5f < c5_best:
        h_best = h_sg.copy()
        c5_best = c5f
    return h_best, c5_best

def slsqp_smooth(h, n, T, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(n_corr):
            w = sw[oi]
            if w < 1e-12:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return grad * dx

    def cons_fun(x):
        return x.sum() - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def slsqp_topk(h, n, K, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        top_idx = np.argpartition(corr, -K)[-K:]
        top_vals = corr[top_idx]
        w = (top_vals / max(mc, 1e-14)) ** 10
        ws = w.sum()
        if ws < 1e-14:
            w = np.ones(K) / K
        else:
            w /= ws
        grad = np.zeros(n)
        for i in range(K):
            grad += w[i] * shift_grad(x, n, top_idx[i] - center)
        return grad * dx

    def cons_fun(x):
        return x.sum() - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def perturb_h(h, n, rng, mode):
    x = np.arange(n) * 2.0 / n
    if mode == 0:
        f = 0.5 + 0.5 * (mode % 3)
        return h + 0.02 * np.sin(2 * np.pi * f * x)
    elif mode == 1:
        f = 0.3 + 0.2 * ((mode * 7) % 5)
        return h + 0.015 * np.cos(4 * np.pi * f * x + rng.uniform(0, 6.28))
    elif mode == 2:
        block = n // 9
        return h + rng.normal(0, 0.01, n)
    else:
        sigma = 0.005 + 0.008 * (mode % 4)
        return h + rng.normal(0, sigma, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
