Actually, for coordinate descent, updating a single coordinate h[j]:
The overlap for shift s is: ... + h[j]*(1-h[j+s]) + h[j-s]*(1-h[j]) + ... (terms involving h[j])
= h[j] - h[j]*h[j+s] + h[j-s] - h[j-s]*h[j] + (terms not involving h[j])
= h[j] - h[j](h[j+s] + h[j-s]) + h[j-s] + (const)

This is quadratic in h[j]: -h[j]*(h[j+s]+h[j-s]) + h[j] + const
= -h[j]^2*(h[j+s]+h[j-s])/(h[j+s]+h[j-s]) + ... no, it's linear in h[j] if h[j+s] and h[j-s] are fixed!

This IS linear in h[j]! (Treat h[j+s] and h[j-s] as constants.)

So for a single coordinate update, the overlap for each shift s is:
overlap(s) = a_s * h[j] + b_s
where a_s = (1 - h[j+s] - h[j-s]) (with appropriate boundary handling) and b_s is constant.

We want to minimize max_s (a_s * h[j] + b_s) over h[j] ∈ [0,1].

This is the minimum of the maximum of linear functions, which can be solved in closed form! The minimum of max of linear functions over an interval occurs at either an endpoint or at the intersection of two of the lines.

This is great! We can do exact coordinate descent.

- h[j] = 1

    target = n / 2.0
    lo, hi = -1.5, 2.5

        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def grad_shift(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        s = shift
        left = (j + s) < n
        right = j >= s
        if left.any():
            g[left] += (1.0 - h[j[left] + s])
        if right.any():
            g[right] -= h[j[right] - s]
    else:
        s = -shift
        left = j >= s
        right = (j + s) < n
        if left.any():
            g[left] += (1.0 - h[j[left] - s])
        if right.any():
            g[right] -= h[j[right] + s]
    return g

def sm_val_grad(h, n, T, K):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    m = len(corr)
    if K >= m:
        idx = np.arange(m)
    else:
        idx = np.argpartition(corr, -K)[-K:]
    c_sel = corr[idx]
    mc = float(c_sel.max())
    sh = T * (c_sel - mc)
    e = np.exp(sh)
    es = float(e.sum())
    val = (mc + np.log(es) / T) * dx
    w = e / es
    sw = 0.0
    for ii in range(len(w)):
        if w[ii] > 1e-14:
            sw += w[ii] * grad_shift(h, n, int(idx[ii]) - center)
    return val, sw * dx

_timed_out = [False]

def _mock(x, n, T, K, t0, tlim):
    if time.time() - t0 > tlim:
        _timed_out[0] = True
    v, _ = sm_val_grad(x, n, T, K)
    return v

def _mog(x, n, T, K, t0, tlim):
    if time.time() - t0 > tlim:
        _timed_out[0] = True
    _, g = sm_val_grad(x, n, T, K)
    return g

def lbfgsb_opt(h, n, T, K, maxiter, tol, t0, tlim):
    _timed_out[0] = False
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_mock, h, method='L-BFGS-B', jac=_mog,
                       args=(n, T, K, t0, tlim), bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol,
                                'gtol': tol * 0.05, 'maxls': 40,
                                'maxfun': maxiter * 4})
        if _timed_out[0]:
            return project_feasible(h.copy(), n)
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h.copy(), n)

def slsqp_opt(h, n, T, K, maxiter, tol, t0, tlim):
    _timed_out[0] = False
    bounds = [(0.0, 1.0)] * n
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(_mock, h, method='SLSQP', jac=_mog,
                       args=(n, T, K, t0, tlim), bounds=bounds,

def coord_descent(h, n, sweeps=3, t0=None, tlim=None):

        h_p = perturb_h(h_best, n, rng, mode)
        h_p = project_feasible(h_p, n)
        T_r = [4, 2, 1][pi % 3]
        K_r = [60, 30, 20][pi % 3]
        h_new = lbfgsb_opt(h_p, n, T_r, K_r, 3000, 1e-13, t0, budget_s - 8)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            if c5_best < 0.38086:
                break

    # Phase 7: Final SLSQP + L-BFGS-B polish
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        h_new = slsqp_opt(h_best, n, 1, 20, 5000, 1e-16, t0, t0 + rem - 10)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_new = lbfgsb_opt(h_best, n, 0.5, 10, 8000, 1e-16, t0, t0 + rem - 5)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Final: try upscaling to n*2 if time permits
    rem = budget_s - (time.time() - t0)
    if rem > 200:
        n2 = n * 2
        h_hi = upsample(h_best, n, n2)
        h_hi = project_feasible(h_hi, n2)
        c5_hi = compute_c5(h_hi)
        if c5_hi < c5_best + 0.001:
            for T, mi in [(8, 3000), (2, 5000), (1, 6000)]:
                rem = budget_s - (time.time() - t0)
                if rem < 60:
                    break
                h_hi = lbfgsb_opt(h_hi, n2, T, 100, mi, 1e-14, t0, budget_s - 30)
                c5_hi = compute_c5(h_hi)
            if c5_hi < c5_best:
                c5_best = c5_hi
                h_best = h_hi
                n = n2

    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    if c5_final > c5_best:
        c5_final = c5_best
    return (h_best, c5_final, n)
