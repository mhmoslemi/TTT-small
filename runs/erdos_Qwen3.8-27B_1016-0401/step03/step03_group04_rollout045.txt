    dx = 2.0 / n
    h_curr = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    center = n - 1
    
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        beta = it / (it + 2.5)
        y = h_curr + beta * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        corr = np.correlate(y, 1.0 - y, mode="full")
        if K < len(corr):
            topk = np.argpartition(corr, -K)[-K:]
            c_top = corr[topk]
            mc = np.max(c_top)
            sh = T * (c_top - mc)
            e = np.exp(sh)
            w = e / e.sum()

            topk = np.arange(len(corr))
            w = np.ones(len(corr)) / len(corr)

        if gn < 1e-15:
            break

        gap = max(f_y - 0.37, 1e-6)
        alpha = min(2.0 * gap / (gn * gn + 1e-12), 0.1 / (1.0 + 0.001 * it))
        
        h_new = project_feasible(y - alpha * g, n)
        h_prev = h_curr.copy()

    if c5_now < c5_best:
