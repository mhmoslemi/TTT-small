from scipy.optimize import linprog

def solve_linearized_step(h, corr, top_indices, n, dx):
    """Solve min t s.t. grad_k . (h - h0) + corr_k <= t, 0<=h<=1, sum(h)=n/2"""
    # Variables: [h(0), h(1), ..., h(n-1), t]
    # Constraints: grad_k . h - grad_k . h0 + corr_k - t <= 0 for each k
    #              sum(h) = n/2
    #              0 <= h <= 1
    #              t >= 0 (or t >= -inf, but we want t >= 0)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((len(top_indices), n + 1))
    b_ub = np.zeros(len(top_indices))
    
    for i, idx in enumerate(top_indices):
        shift = idx - (n - 1)
        g = shift_grad(h, n, shift) * dx
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -corr[idx]  # g . h - t <= -corr[idx]  =>  g . h + corr[idx] - t <= 0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    return res.x[:n] if res.success else h

    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    
    def obj(x):

        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return mc + np.log(e.sum()) / T
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)

        grad = np.zeros(n)

            if w < 1e-11:

            grad += w * shift_grad(x, n, oi - center)
        return grad
    
    def cons_fun(x):
        return x.sum() - target
    
    def cons_jac(x):
        return np.ones(n)

                       bounds=bounds, constraints=constraints,

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        grad[m1] += (1.0 - h[j[m1] - shift])
        m2 = (j + shift) < n
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        grad[m1] += (1.0 - h[j[m1] + s])
        m2 = j >= s
        grad[m2] -= h[j[m2] - s]
    return grad

def project_feasible(y, n):

                h_new = project_feasible(h_new, n2)
                c5_alt = compute_c5(h_new)
                if c5_alt < c5_best:
                    c5_best = c5_alt
                    h_best = h_new
                    n = n2

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
