    grad = np.zeros(n)
    if shift == 0:
        grad[:] = 1.0 - 2.0 * h
    elif shift > 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]

        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

    corr = np.correlate(x, 1.0 - x, mode="full") * (2.0/n)
    if K < len(corr):
        topk_idx = np.argpartition(corr, -K)[-K:]
        c = corr[topk_idx]
    else:
        topk_idx = np.arange(len(corr))
        c = corr
    mc = np.max(c)
    sh = T * (c - mc)
    e = np.exp(sh)
    val = mc + np.log(e.sum()) / T
    return val

def soft_max_grad(x, n, T, K, t0, tlim):
    corr = np.correlate(x, 1.0 - x, mode="full")
    if K < len(corr):
        topk_idx = np.argpartition(corr, -K)[-K:]
        c = corr[topk_idx]
    else:

        c = corr

        if time.time() > tlim:
            corr = np.correlate(x, 1.0 - x, 'full')
            s = int(np.argmax(corr)) - (n - 1)
            return shift_grad(x, n, s) * (2.0 / n)
        return sm_grad(x, n, T, K)
    try:
        res = minimize(obj, h.copy(), method='L-BFGS-B', jac=jac, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 50})
        return project_feasible(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return h.copy()

def exact_subgrad(h, n, tlim, max_steps=500):
    dx = 2.0 / n
    h_curr = h.copy()
    best = compute_c5(h_curr)
    best_h = h_curr.copy()
    stall = 0
    for it in range(max_steps):
        if time.time() > tlim:
            break
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        g = shift_grad(h_curr, n, s) * dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        f0 = compute_c5(h_curr)
        alpha = min(2.0 * max(f0 - 0.374, 1e-6) / (gn * gn + 1e-12), 0.4 / (1.0 + 0.002 * it))
        improved = False
        for _ in range(30):
            h_new = project_feasible(h_curr - alpha * g, n)
            f_new = compute_c5(h_new)
            if f_new < f0 - 1e-15:
                h_curr = h_new
                if f_new < best:
                    best = f_new
                    best_h = h_curr.copy()
                    stall = 0
                else:
                    stall += 1
                improved = True
                break
            alpha *= 0.5
        if not improved:
            stall += 2
            if stall > 20:
                break
    return best_h, best

def make_struct(n, rng, pi):
    if pi % 5 == 0:
        period = int(rng.integers(3, 14))
        duty = rng.uniform(0.3, 0.7)
        on = max(1, int(round(period * duty)))
        h = np.zeros(n)
        for i in range(n):
            if (i % period) < on:
                h[i] = 1.0
        h += rng.normal(0, 0.04, n)
    elif pi % 5 == 1:
        sig = 0.002 + 0.008 * (pi % 4)
        h = None  # will use h_best
    elif pi % 5 == 2:
        period1 = int(rng.integers(4, 10))
        period2 = int(rng.integers(11, 20))
        h = np.zeros(n)
        ph1 = rng.uniform(0, 2 * np.pi)
        ph2 = rng.uniform(0, 2 * np.pi)
        h[:] = 0.5 + 0.3 * np.sin(2 * np.pi * np.arange(n) / period1 + ph1)
        h += 0.2 * np.sin(2 * np.pi * np.arange(n) / period2 + ph2)
        h += rng.normal(0, 0.03, n)
    elif pi % 5 == 3:
        h = np.zeros(n)
        nb = int(rng.integers(4, 12))
        bs = n // nb
        for bi in range(nb):
            h[bi * bs:(bi + 1) * bs] = rng.uniform(0, 1)
        h += rng.normal(0, 0.02, n)
    else:
        f1, f2 = rng.uniform(1, 4), rng.uniform(5, 12)
        h = 0.5 + 0.25 * np.sin(2 * np.pi * f1 * np.arange(n) / n)
        h += 0.15 * np.cos(2 * np.pi * f2 * np.arange(n) / n + rng.uniform(0, 6.28))
        h += rng.normal(0, 0.03, n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Coarse-to-fine L-BFGS-B with adaptive K
    schedule = [(32, 80, 2500), (16, 40, 3000), (8, 20, 4000),
                (4, 10, 5000), (2, 5, 6000), (1, 3, 7000), (0.5, 2, 8000), (0.3, 1, 10000)]
    for T, K, mi in schedule:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        h_new = run_lbfgsb(h_best, n, T, K, mi, t0 + rem - 15)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Lower-resolution exploration
    rem = budget_s - (time.time() - t0)
    if rem > 280:
        n_lo = n // 2
        x_hi = np.linspace(0, 2.0, n, endpoint=False)
        x_lo = np.linspace(0, 2.0, n_lo, endpoint=False)
        h_lo = np.interp(x_lo, x_hi, h_best)
        h_lo = project_feasible(h_lo, n_lo)
        for T, K, mi in [(16, 20, 3000), (4, 5, 4000), (1, 2, 5000), (0.5, 1, 5000)]:
            rem = budget_s - (time.time() - t0)
            if rem < 50:
                break
            h_lo = run_lbfgsb(h_lo, n_lo, T, K, mi, t0 + rem - 10)
            c5_lo = compute_c5(h_lo)
            x_hi2 = np.linspace(0, 2.0, n, endpoint=False)
            h_up = np.interp(x_hi2, x_lo, h_lo)
            h_up = project_feasible(h_up, n)
            c5_up = compute_c5(h_up)
            if c5_up < c5_best:
                c5_best = c5_up
                h_best = h_up

    # Phase 3: Diverse restarts
    rem = budget_s - (time.time() - t0)
    n_restarts = min(50, max(10, int(rem // 10)))
    for pi in range(n_restarts):
        rem = budget_s - (time.time() - t0)
        if rem < 12:
            break
        if pi % 5 == 1:
            sig = 0.002 + 0.008 * (pi % 4)
            h_p = project_feasible(h_best + rng.normal(0, sig, n), n)
        else:
            h_p = make_struct(n, rng, pi)
        T_r = 4 if pi < n_restarts // 2 else 2
        K_r = 8 if pi < n_restarts // 2 else 4
        mi_r = 2500 if rem > 40 else 1200
        h_new = run_lbfgsb(h_p, n, T_r, K_r, mi_r, t0 + rem - 5)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Fine polish L-BFGS-B
    for T, K, mi in [(0.5, 1, 12000), (0.3, 1, 15000)]:
        rem = budget_s - (time.time() - t0)
        if rem < 35:
            break
        h_new = run_lbfgsb(h_best, n, T, K, mi, t0 + rem - 8)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Exact subgradient with line search
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        h_sg, c5_sg = exact_subgrad(h_best, n, t0 + rem - 5, 500)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase 6: Final combined polish
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_new = run_lbfgsb(h_best, n, 0.4, 1, 10000, t0 + rem - 15)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        rem = budget_s - (time.time() - t0)
        if rem > 20:
            h_sg2, c5_sg2 = exact_subgrad(h_best, n, t0 + rem - 4, 400)
            if c5_sg2 < c5_best:
                c5_best = c5_sg2
                h_best = h_sg2

    # Phase 7: One more L-BFGS-B + subgradient cycle
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        h_new = run_lbfgsb(h_best, n, 1.0, 2, 8000, t0 + rem - 20)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        rem = budget_s - (time.time() - t0)
        if rem > 25:
            h_sg3, c5_sg3 = exact_subgrad(h_best, n, t0 + rem - 4, 300)
            if c5_sg3 < c5_best:
                c5_best = c5_sg3
                h_best = h_sg3

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
