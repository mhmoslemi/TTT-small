def project_feasible(y, n):
    target = n / 2.0

    except Exception:
        return h.copy()

def line_search_worst(h, n, t_deadline, max_iters=80000):
    dx = 2.0 / n
    center = n - 1
    h_cur = h.copy()
    c5_best = compute_c5(h_cur)
    h_best = h_cur.copy()
    for it in range(max_iters):
        if time.time() > t_deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        ak = int(np.argmax(corr))
        shift = ak - center
        g = shift_grad_raw(h_cur, n, shift) * dx
        alpha = 0.03 / (1.0 + 0.0005 * it)
        h_new = h_cur - alpha * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-12:
            c5_best = c5_new
            h_best = h_new.copy()
        h_cur = h_new
    return h_best, c5_best

def multi_top_pareto(h, n, t_deadline, K=4, max_iters=60000):
    dx = 2.0 / n
    center = n - 1
    h_cur = h.copy()
    h_prev = h.copy()
    c5_best = compute_c5(h_cur)
    h_best = h_cur.copy()
    for it in range(max_iters):
        if time.time() > t_deadline:
            break
        beta = 0.7 * float(it) / (it + 5.0)
        h_y = h_cur + beta * (h_cur - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = corr.max()
        top_idx = np.argsort(corr)[-K:]
        w = np.exp(10.0 * (corr[top_idx] - mc))
        w /= w.sum()
        g = np.zeros(n)
        for wi, ik in zip(w, top_idx):
            g += wi * shift_grad_raw(h_y, n, ik - center)
        g *= dx
        alpha = 0.06 / (1.0 + 0.0008 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_cur
        h_cur = h_new
        if (it + 1) % 10000 == 0:
            cv = compute_c5(h_cur)
            if cv < c5_best:
                c5_best = cv
                h_best = h_cur.copy()
    cv = compute_c5(h_cur)
    if cv < c5_best:
        c5_best = cv
        h_best = h_cur.copy()
    return h_best, c5_best

def coordinate_polish(h, n, rng, steps=4000):
    dx = 2.0 / n
    center = n - 1
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for _ in range(steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        ak = int(np.argmax(corr))
        shift = ak - center
        g = shift_grad_raw(h_cur, n, shift) * dx
        pos_mask = g > 0.01
        neg_mask = g < -0.01
        if not pos_mask.any() or not neg_mask.any():
            break
        ip = int(rng.choice(np.where(pos_mask)[0]))
        inn = int(rng.choice(np.where(neg_mask)[0]))
        delta = min(0.04, h_cur[ip], 1.0 - h_cur[inn])
        h_try = h_cur.copy()
        h_try[ip] -= delta
        h_try[inn] += delta
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < c5_cur - 1e-9:
            h_cur = h_try
            c5_cur = c5_try
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    else:
        n0 = 280
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0

    h_best = project_feasible(h0, n0)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP with moderate temperatures (skip very high T since we start good)
    for T in [4.0, 2.0, 1.0, 0.7, 0.5, 0.35, 0.25, 0.2]:
        if time.time() - t0 > budget_s - 600:
            break
        h_new = slsqp_run(h_best, n0, T, 7000)
        h_new = project_feasible(h_new, n0)
        c5 = compute_c5(h_new)
        if c5 < c5_best - 1e-13:
            c5_best = c5
            h_best = h_new

    # Phase 2: Multi-top Pareto with momentum
    t_dead = t0 + budget_s - 250
    h_p, c5_p = multi_top_pareto(h_best, n0, t_dead, K=4, max_iters=80000)
    if c5_p < c5_best:
        c5_best = c5_p
        h_best = h_p

    # Phase 3: Try alternative n values
    n_alts = [240, 320, 200, 360, 140]
    for n_try in n_alts:
        if time.time() - t0 > budget_s - 120:
            break
        x_old = np.linspace(0, 2, n0, endpoint=False) + 1.0 / n0
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_old, h_best, period=2.0)
        h_interp = project_feasible(h_interp, n_try)
        c5_int = compute_c5(h_interp)
        if c5_int > c5_best + 0.02:
            continue
        h_try = h_interp
        for T in [3.0, 1.0, 0.5, 0.3]:
            if time.time() - t0 > budget_s - 80:
                break
            h_try = slsqp_run(h_try, n_try, T, 5000)
            h_try = project_feasible(h_try, n_try)
        c5_try = compute_c5(h_try)
        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_try
            n0 = n_try

    # Phase 4: Line search on worst shift
    t_dead2 = t0 + budget_s - 60
    h_ls, c5_ls = line_search_worst(h_best, n0, t_dead2, max_iters=50000)
    if c5_ls < c5_best:
        c5_best = c5_ls
        h_best = h_ls

    # Phase 5: Final SLSQP + coordinate polish
    if time.time() - t0 < budget_s - 30:
        h_new = slsqp_run(h_best, n0, 0.3, 10000)
        h_new = project_feasible(h_new, n0)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    if time.time() - t0 < budget_s - 15:
        h_cp, c5_cp = coordinate_polish(h_best, n0, rng, steps=3000)
        if c5_cp < c5_best:
            c5_best = c5_cp
            h_best = h_cp

    # Phase 6: Small perturbations + quick SLSQP
    for pi in range(10):
        if time.time() - t0 > budget_s - 5:
            break
        sigma = 0.004 + 0.006 * (pi % 3)
        h_pert = h_best + rng.normal(0, sigma, n0)
        h_pert = project_feasible(h_pert, n0)
        h_new = slsqp_run(h_pert, n0, 1.5, 3000)
        h_new = project_feasible(h_new, n0)
        c5 = compute_c5(h_new)
        if c5 < c5_best - 1e-10:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n0)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n0)
