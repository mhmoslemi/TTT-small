import numpy as np
from scipy.optimize import minimize
import time

_global_target = 0.0
_global_n = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def _con_fun(x):
    return x.sum() - _global_target

def _con_jac(x):
    return np.ones(len(x))

def slsqp_smooth(h, n, T, maxiter, ftol):
    global _global_target
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    _global_target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(n_corr):
            w = sw[oi]
            if w < 1e-10:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return grad * dx

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def slsqp_topk(h, n, K, maxiter, ftol):
    global _global_target
    dx = 2.0 / n
    center = n - 1
    _global_target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        top_idx = np.argpartition(corr, -K)[-K:]
        top_vals = corr[top_idx]
        w = (top_vals / max(mc, 1e-15)) ** 6
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(K) / K
        else:
            w /= ws
        grad = np.zeros(n)
        for i in range(K):
            grad += w[i] * shift_grad(x, n, top_idx[i] - center)
        return grad * dx

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def nesterov_run(h_init, n, t0, time_limit):
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    stalling = 0

    for it in range(800000):
        if time.time() - t0 > time_limit:
            break
        if (it + 1) % 5000 == 0:
            c5c = compute_c5(h_sg)
            if c5c < c5_best - 1e-7:
                c5_best = c5c
                h_best = h_sg.copy()
                stalling = 0
            else:
                stalling += 1
            if stalling > 80 and it > 10000:
                n2 = len(h_sg)
                h_sg += np.random.default_rng(it).normal(0, 0.003, n2)
                h_sg = project_feasible(h_sg, n2)
                stalling = 0
                h_prev = h_sg.copy()

        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)

        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)

        if it < 3000:
            K = 1
        elif it < 20000:
            K = 3 + (it - 3000) // 5000
        else:
            K = min(8 + (it - 20000) // 50000, 20)

        if K <= 1:
            idx = int(np.argmax(corr))
            g = shift_grad(h_y, n, idx - center) * dx
        else:
            top_idx = np.argpartition(corr, -K)[-K:]
            top_vals = corr[top_idx]
            w = (top_vals / max(mc, 1e-15)) ** 8
            ws = w.sum()
            if ws < 1e-15:
                w = np.ones(K) / K
            else:
                w /= ws
            g = np.zeros(n)
            for i in range(K):
                g += w[i] * shift_grad(h_y, n, top_idx[i] - center)
            g *= dx

        alpha = 0.35 / (1.0 + 0.0025 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new

    c5f = compute_c5(h_sg)
    if c5f < c5_best:
        c5_best = c5f
        h_best = h_sg.copy()
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:

            if c5_alt < c5_best:
                c5_best = c5_alt
                h_best = h_alt
                n = n2

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
