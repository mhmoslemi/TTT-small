```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def topk_gradient(h, n, K, power=8):
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    if K <= 1:
        idx = int(np.argmax(corr))
        return shift_grad(h, n, idx - center) * dx, corr
    top_idx = np.argpartition(corr, -K)[-K:]
    top_vals = corr[top_idx]
    w = (top_vals / max(mc, 1e-15)) ** power
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.ones(K) / K
    else:
        w /= w_sum
    g = np.zeros(n)
    for i in range(K):
        g += w[i] * shift_grad(h, n, top_idx[i] - center)
    return g * dx, corr

    for _ in range(30):
        h_new = project_feasible(h - alpha * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_current or c5_new < c5_target:

        alpha *= 0.7
    return h, c5_current, alpha

    c5_cur = compute_c5(h)
    h_best = h.copy()
    c5_best = c5_cur

    beta = 0.9

            break
        frac = min(it / max_iters, 1.0)
        K = max(1, int(K_start + (K_end - K_start) * frac * frac))

        momentum = beta * momentum - alpha * g
        h_new = h + momentum

        if c5_new < c5_cur:
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(y - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1, m2 = j >= shift, (j + shift) < n
        if m1.any(): g[m1] += 1.0 - h[j[m1] - shift]
        if m2.any(): g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1, m2 = (j + s) < n, j >= s
        if m1.any(): g[m1] += 1.0 - h[j[m1] + s]
        if m2.any(): g[m2] -= h[j[m2] - s]
    return g

_st = {}

def _so(x):
    T, dx = _st['T'], _st['dx']
    c = np.correlate(x, 1.0 - x, mode="full")
    mc = c.max()
    sh = T * (c - mc)
    sh -= sh.max()
    return (mc + np.log(np.exp(sh).sum()) / T) * dx

def _sj(x):
    T, dx, n = _st['T'], _st['dx'], _st['n']
    c = np.correlate(x, 1.0 - x, mode="full")
    mc = c.max()
    sh = T * (c - mc)
    sh -= sh.max()
    e = np.exp(sh)
    w = e / e.sum()
    ctr = n - 1
    g = np.zeros(n)
    for i in range(len(c)):
        if w[i] > 1e-10:
            g += w[i] * shift_grad(x, n, i - ctr)
    return g * dx

    return x.sum() - _st['t']

    return np.ones(_st['n'])

    _st.update(T=T, n=n, dx=2.0/n, t=n/2.0)
    try:
        r = minimize(_so, h, method='SLSQP', jac=_sj,
                     bounds=[(0,1)]*n,
                     constraints=[{'type':'eq','fun':_scf,'jac':_scj}],
                     options={'maxiter':mi,'ftol':ft})
        return project_feasible(r.x, n)
    except:

    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode="full")
    mc = c.max()
    if K <= 1:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(y - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1, m2 = j >= shift, (j + shift) < n
        if m1.any():
            g[m1] += 1.0 - h[j[m1] - shift]
        if m2.any():
            g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1, m2 = (j + s) < n, j >= s
        if m1.any():
            g[m1] += 1.0 - h[j[m1] + s]
        if m2.any():
            g[m2] -= h[j[m2] - s]
    return g

_st = {}

def _so(x):
    T, dx = _st['T'], _st['dx']
    c = np.correlate(x, 1.0 - x, mode="full")
    mc = float(c.max())
    sh = T * (c - mc)
    sh -= sh.max()
    return (mc + np.log(np.exp(sh).sum()) / T) * dx

def _sj(x):
    T, dx, n = _st['T'], _st['dx'], _st['n']
    c = np.correlate(x, 1.0 - x, mode="full")
    mc = float(c.max())
    sh = T * (c - mc)
    sh -= sh.max()
    e = np.exp(sh)
    w = e / e.sum()
    ctr = n - 1
    g = np.zeros(n)
    for i in range(len(c)):
        if w[i] > 1e-10:
            g += w[i] * shift_grad(x, n, i - ctr)
    return g * dx

def _scf(x):
    return x.sum() - _st['t']

def _scj(x):
    return np.ones(_st['n'])

def slsqp(h, n, T, mi, ft):
    _st['T'], _st['n'] = T, n
    _st['dx'], _st['t'] = 2.0 / n, n / 2.0
    try:
        r = minimize(_so, h, method='SLSQP', jac=_sj,
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scf, 'jac': _scj}],
                     options={'maxiter': mi, 'ftol': ft})
        return project_feasible(r.x, n)
    except Exception:
        return project_feasible(h.copy(), n)

def topk_grad(h, n, K, pw):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    mc = float(c.max())
    ctr = n - 1
    if K <= 1:
        return shift_grad(h, n, int(c.argmax()) - ctr) * dx
    ti = np.argpartition(c, -K)[-K:]
    w = (c[ti] / max(mc, 1e-15)) ** pw
    s = float(w.sum())
    w = w / s if s > 1e-15 else np.ones(K) / K
    g = np.zeros(n)
    for i in range(K):
        g += w[i] * shift_grad(h, n, ti[i] - ctr)
    return g * dx

def gd(h, n, tl, mi, K, pw, a0, bm, pe, ps):
    cur = compute_c5(h)
    hb, cb = h.copy(), cur
    a = a0
    m = np.zeros(n)
    bad = 0
    rng = np.random.default_rng(7)
    for it in range(mi):
        if time.time() > tl:
            break
        g = topk_grad(h, n, K, pw)
        m = bm * m - a * g
        hn = project_feasible(h + m, n)
        cn = compute_c5(hn)
        if cn < cur:
            h, cur, bad = hn, cn, 0
            if cur < cb:
                cb, hb = cur, h.copy()
        else:
            bad += 1
            if bad >= 5:
                a *= 0.7
                m *= 0.3
                bad = 0
            h2 = project_feasible(h - a * g, n)
            c2 = compute_c5(h2)
            if c2 < cur:
                h, cur = h2, c2
                m = np.zeros(n)
                if cur < cb:
                    cb, hb = cur, h.copy()
            else:
                a *= 0.8
                m = np.zeros(n)
        if (it + 1) % pe == 0:
            hp = project_feasible(h + rng.normal(0.0, ps, n), n)
            cp = compute_c5(hp)
            if cp < cb * 1.03:
                h = hp
                cur = cp
                m = np.zeros(n)
                a = a0
    return hb, cb

def interp(h, no, nn):
    xo = np.arange(no) * (2.0 / no)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), nn)

def bin_check(h, n):
    c0 = compute_c5(h)
    hb = np.where(h >= 0.5, 1.0, 0.0)
    t = n // 2
    s = int(hb.sum())
    if s > t:
        o = np.argsort(h)[::-1]
        for i in range(s - t):
            hb[o[i]] = 0.0
    elif s < t:
        o = np.argsort(h)
        for i in range(t - s):
            hb[o[i]] = 1.0
    cb = compute_c5(hb)
    if cb < c0:
        return hb, cb
    return h, c0

def symm(h, n):
    hs = (h + 1.0 - h[::-1]) / 2.0
    return project_feasible(hs, n)
```

```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0

    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1, m2 = j >= shift, (j + shift) < n
        if m1.any():

            g[m2] -= h[j[m2] + shift]

        if m2.any():

    return (mc + np.log(np.exp(sh).sum()) / T) * dx

def _sj(x):
    T, dx, n = _st['T'], _st['dx'], _st['n']
    c = np.correlate(x, 1.0 - x, mode="full")
    mc = float(c.max())
    sh = T * (c - mc)
    sh -= sh.max()
    e = np.exp(sh)
    w = e / e.sum()
    ctr = n - 1
    g = np.zeros(n)
    for i in range(len(c)):
        if w[i] > 1e-10:
            g += w[i] * shift_grad(x, n, i - ctr)
    return g * dx

def _scf(x):
    return x.sum() - _st['t']

def _scj(x):
    return np.ones(_st['n'])

def slsqp(h, n, T, mi, ft):
    _st['T'], _st['n'] = T, n
    _st['dx'], _st['t'] = 2.0 / n, n / 2.0
    try:
        r = minimize(_so, h, method='SLSQP', jac=_sj,
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scf, 'jac': _scj}],
                     options={'maxiter': mi, 'ftol': ft})
        return project_feasible(r.x, n)
    except Exception:
        return project_feasible(h.copy(), n)

def topk_grad(h, n, K, pw):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    mc = float(c.max())
    ctr = n - 1
    if K <= 1:
        return shift_grad(h, n, int(c.argmax()) - ctr) * dx
    ti = np.argpartition(c, -K)[-K:]
    w = (c[ti] / max(mc, 1e-15)) ** pw
    s = float(w.sum())
    w = w / s if s > 1e-15 else np.ones(K) / K
    g = np.zeros(n)
    for i in range(K):
        g += w[i] * shift_grad(h, n, ti[i] - ctr)
    return g * dx

def gd(h, n, tl, mi, K, pw, a0, bm, pe, ps):
    cur = compute_c5(h)
    hb, cb = h.copy(), cur
    a = a0
    m = np.zeros(n)
    bad = 0
    rng = np.random.default_rng(7)
    for it in range(mi):
        if time.time() > tl:
            break
        g = topk_grad(h, n, K, pw)
        m = bm * m - a * g
        hn = project_feasible(h + m, n)
        cn = compute_c5(hn)
        if cn < cur:
            h, cur, bad = hn, cn, 0
            if cur < cb:
                cb, hb = cur, h.copy()
        else:
            bad += 1
            if bad >= 5:
                a *= 0.7
                m *= 0.3
                bad = 0
            h2 = project_feasible(h - a * g, n)
            c2 = compute_c5(h2)
            if c2 < cur:
                h, cur = h2, c2
                m = np.zeros(n)
                if cur < cb:
                    cb, hb = cur, h.copy()
            else:
                a *= 0.8
                m = np.zeros(n)
        if (it + 1) % pe == 0:
            hp = project_feasible(h + rng.normal(0.0, ps, n), n)
            cp = compute_c5(hp)
            if cp < cb * 1.03:
                h = hp
                cur = cp
                m = np.zeros(n)
                a = a0
    return hb, cb

def interp(h, no, nn):
    xo = np.arange(no) * (2.0 / no)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), nn)

def bin_check(h, n):
    c0 = compute_c5(h)
    hb = np.where(h >= 0.5, 1.0, 0.0)
    t = n // 2
    s = int(hb.sum())
    if s > t:
        o = np.argsort(h)[::-1]
        for i in range(s - t):
            hb[o[i]] = 0.0
    elif s < t:
        o = np.argsort(h)
        for i in range(t - s):
            hb[o[i]] = 1.0
    cb = compute_c5(hb)
    if cb < c0:
        return hb, cb
    return h, c0

def symm(h, n):
    hs = (h + 1.0 - h[::-1]) / 2.0
    return project_feasible(hs, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    BS = float(budget_s)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        hb = np.array(initial_h_values, dtype=float)
    else:
        hb = np.zeros(280)
        hb[::2] = 1.0
    n = len(hb)
    hb = project_feasible(hb, n)
    cb = compute_c5(hb)

    if time.time() - t0 < BS - 400:
        h1 = slsqp(hb, n, 8, 3000, 1e-14)
        c1 = compute_c5(h1)
        if c1 < cb:
            cb, hb = c1, h1

    if time.time() - t0 < BS - 200:
        h2, c2 = gd(hb, n, t0 + BS - 200, 50000, 10, 6, 0.05, 0.85, 5000, 0.008)
        if c2 < cb:
            cb, hb = c2, h2

    if time.time() - t0 < BS - 150:
        h3 = slsqp(hb, n, 4, 5000, 1e-15)
        c3 = compute_c5(h3)
        if c3 < cb:
            cb, hb = c3, h3

    hs = symm(hb, n)
    cs = compute_c5(hs)
    if cs < cb:
        cb, hb = cs, hs
    if time.time() - t0 < BS - 200:
        hg_s, cg_s = gd(hs, n, t0 + BS - 200, 10000, 8, 6, 0.04, 0.85, 3000, 0.005)
        if cg_s < cb:
            cb, hb = cg_s, hg_s

    n_cands = set([n])
    for f in [6, 7, 8, 9, 10, 11, 12, 14, 16, 18, 20, 24]:
        nt2 = f * 20
        if 80 <= nt2 <= 600:
            n_cands.add(nt2)
    for nt in sorted(n_cands):
        if time.time() - t0 > BS - 120:
            break
        if nt == n:
            continue
        ht = interp(hb, n, nt)
        ct = compute_c5(ht)
        if time.time() - t0 < BS - 80:
            ht = slsqp(ht, nt, 6, 2000, 1e-14)
            ct = compute_c5(ht)
        if time.time() - t0 < BS - 60:
            hg, cg = gd(ht, nt, t0 + BS - 50, 15000, 8, 6, 0.04, 0.85, 4000, 0.006)
            if cg < ct:
                ct, ht = cg, hg
        ht, ct = bin_check(ht, nt)
        hs2 = symm(ht, nt)
        cs2 = compute_c5(hs2)
        if cs2 < ct:
            ht, ct = hs2, cs2
        if ct < cb:
            cb, hb, n = ct, ht.copy(), nt

    if time.time() - t0 < BS - 30:
        h5 = slsqp(hb, n, 3, 4000, 1e-15)
        c5 = compute_c5(h5)
        if c5 < cb:
            cb, hb = c5, h5

    if time.time() - t0 < BS - 10:
        h6, c6 = gd(hb, n, t0 + BS - 3, 8000, 5, 8, 0.02, 0.9, 3000, 0.004)
        if c6 < cb:
            cb, hb = c6, h6

    hs3 = symm(hb, n)
    cs3 = compute_c5(hs3)
    if cs3 < cb:
        hb = hs3
        cb = cs3

    hb = project_feasible(hb, n)
    return (hb, min(compute_c5(hb), cb), n)
```
