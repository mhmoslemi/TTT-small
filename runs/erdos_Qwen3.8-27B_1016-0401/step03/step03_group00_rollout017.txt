import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2.0, 0.0, 1.0)

# Soft-max objective and gradient (top-k shifts)
def softmax_obj(h, T, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    val = (m + np.log(e.sum()) / T) * dx
    return val

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

def softmax_jac(h, T, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    for wi, ci in zip(w, idx):
        if wi > 1e-10:
            grad += wi * shift_grad(h, n, ci - (n - 1))
    return grad * dx

    bounds = [(0.0, 1.0)] * n
    def obj(x): return softmax_obj(x, T, top_k)
    def jac(x): return softmax_jac(x, T, top_k)
    def con_fun(x): return x.sum() - n / 2.0
    def con_jac(x): return np.ones(n)
    try:

                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': ftol})

            #   if j-s < n and j-s >= 0: term (j-s) contributes h[j-s]*(1-x)
            B = np.zeros(2*n-1)

            A = corr_all - B * old
            # Minimize max_k (A[k] + B[k]*x) for x in [0,1]

                f1 = np.max(A + B*m1)

                else:
                    lo = m1
            new_val = (lo+hi)/2
            h[j] = new_val
        # Project to satisfy sum constraint
        h = project_feasible(h, n)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad_vec(h, n, shift):
    """Gradient of corr(shift) w.r.t. h, vectorized."""
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

def slsqp_phase(h, n, T, top_k, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        m = c.max()
        e = np.exp(T * (c - m))
        return (m + np.log(e.sum()) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        m = c.max()
        e = np.exp(T * (c - m))
        w = e / e.sum()
        grad = np.zeros(n)
        for wi, ci in zip(w, idx):
            if wi > 1e-10:
                grad += wi * shift_grad_vec(x, n, ci - (n - 1))
        return grad * dx
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except:
        return h.copy()
