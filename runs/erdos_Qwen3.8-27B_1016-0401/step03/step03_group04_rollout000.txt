import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 5.0
    hi = float(y.max()) + 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def softmax_obj_grad(h, n, T, K):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    if K < len(corr):
        topk = np.argpartition(corr, -K)[-K:]
    else:
        topk = np.arange(len(corr))
    c_sel = corr[topk]
    sh = T * (c_sel - np.max(c_sel))
    e = np.exp(sh)
    sw = e / e.sum()
    val = (np.max(c_sel) + np.log(e.sum()) / T) * (2.0 / n)
    center = n - 1
    grad = np.zeros(n)
    for ii in range(len(sw)):
        if sw[ii] > 1e-14:
            grad += sw[ii] * shift_grad(h, n, topk[ii] - center)
    grad *= (2.0 / n)
    return val, grad

def coord_descent(h, n, n_sweeps=8, t0=None, tlim=None):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    s_arr = np.arange(-(n - 1), n)
    for sweep in range(n_sweeps):
        if t0 is not None and time.time() - t0 > tlim:
            break
        for i in range(n):
            v0 = h[i]
            left_valid = (i + s_arr >= 0) & (i + s_arr < n)
            right_valid = (i - s_arr >= 0) & (i - s_arr < n)
            g = np.zeros(2 * n - 1)
            g[left_valid] += 1.0 - h[i + s_arr[left_valid]]
            g[right_valid] -= h[i - s_arr[right_valid]]
            lo_u = -v0
            hi_u = 1.0 - v0
            if hi_u - lo_u < 1e-14:
                continue
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = lo_u, hi_u
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            fc = np.max(corr + c * g)
            fd = np.max(corr + d * g)
            for _ in range(35):
                if fc < fd:
                    b = d; d = c; fd = fc
                    c = b - gr * (b - a); fc = np.max(corr + c * g)
                else:
                    a = c; c = d; fc = fd
                    d = a + gr * (b - a); fd = np.max(corr + d * g)
            v_new = np.clip((a + b) / 2.0 + v0, 0.0, 1.0)
            dv = v_new - v0
            if abs(dv) > 1e-15:
                h[i] = v_new
                corr += dv * g
    return h

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        if time.time() - t0 > tlim:
            raise TimeoutError()
        v, _ = softmax_obj_grad(x, n, T, K)
        return v
    def jac(x):
        if time.time() - t0 > tlim:
            raise TimeoutError()
        _, g = softmax_obj_grad(x, n, T, K)
        return g
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jac, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol * 0.05, 'maxls': 25})
        return project_feasible(res.x, n)
    except (TimeoutError, Exception):
        return h.copy()

def do_slsqp(h, n, T, K, maxiter, tol, t0, tlim):
    def obj(x):
        if time.time() - t0 > tlim:
            raise TimeoutError()
        v, _ = softmax_obj_grad(x, n, T, K)
        return v
    def jf(x):
        _, g = softmax_obj_grad(x, n, T, K)
        return g
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jf, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': tol})
        return project_feasible(res.x, n)
    except (TimeoutError, Exception):
        return h.copy()

def make_initial(n, rng):
    h = np.zeros(n)
    h[::2] = 1.0
    h = project_feasible(h + rng.normal(0, 0.05, n), n)
    return h

def struct_perturb(h, n, rng):
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(7):
        f = rng.uniform(0.5, 12)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.005, 0.035)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert += rng.normal(0, 0.006, n)
    return h + pert

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:

        h_best = make_initial(280, rng)

    c5_best = compute_c5(h_best)

    tlim_hard = t0 + budget_s - 30

    # Phase 1: Coordinate descent (fast, exact 1D subproblems)
    h_cd = coord_descent(h_best, n, 6, t0, tlim_hard)

        c5_best, h_best = c5, h_cd

    # Phase 2: L-BFGS-B warmup (moderate smoothing)

        if time.time() > deadline - 120:
            break
        h_new = do_lbfgsb(h_best, n, T, K, 3000, 1e-13, t0, tlim_hard)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 3: More coordinate descent sweeps
    if time.time() < deadline - 100:
        h_cd = coord_descent(h_best, n, 10, t0, tlim_hard)
        c5 = compute_c5(h_cd)
        if c5 < c5_best:
            c5_best, h_best = c5, h_cd

    # Phase 4: Fine L-BFGS-B (sharp focus)
    for T, K in [(1.0, 15), (0.5, 10), (0.3, 5), (0.2, 3)]:
        if time.time() > deadline - 80:
            break
        h_new = do_lbfgsb(h_best, n, T, K, 8000, 1e-15, t0, tlim_hard)
        c5 = compute_c5(h_new)
        if c5 < c5_best:

    # Phase 5: SLSQP with exact sum constraint
    for T, K in [(1.0, 10), (0.5, 5)]:
        if time.time() > deadline - 60:

        h_new = do_slsqp(h_best, n, T, K, 3000, 1e-15, t0, tlim_hard)

    # Phase 6: Coordinate descent final
    if time.time() < deadline - 40:

    for pi in range(n_rest):
        if time.time() > deadline - 15:

        c5 = compute_c5(h_new)

    return (h_best, c5_final, n)
