corr[m] = sum_{i} h[i] * (1 - h[i+m]) for i where both i and i+m are in [0, n-1]

- Term where i+m=a (i=a-m) in corr[m]: was h[a-m]*(1-1)=0, becomes h[a-m]*(1-0)=h[a-m]. Change: +h[a-m] if a-m valid

            if 0 <= b + shift_w < n:
                db += (1.0 - h[b + shift_w])
            if 0 <= b - shift_w < n:
                db -= h[b - shift_w]
            if da + db >= -1e-10:
                continue
            h[a], h[b] = 0.0, 1.0
            corr_new = np.correlate(h, 1.0 - h, mode="full")
            c5_new = float(corr_new.max() * dx)
            if c5_new < c5 - 1e-15:
                c5 = c5_new
                corr = corr_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                improved = True
                break
            else:
                h[a], h[b] = 1.0, 0.0
        if improved:
            no_imp = 0
        else:
            no_imp += 1
    return best_h, best_c5

def adam_opt(h0, n, t_end, lr0=0.003, top_m=8):
    h = h0.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    lr = lr0
    c5_best = compute_c5(h)
    h_best = h.copy()
    dx = 2.0 / n
    center = n - 1
    while time.time() < t_end:
        t += 1
        corr = np.correlate(h, 1.0 - h, mode="full")
        T_adapt = 15.0 + 5.0 * t / 10000.0
        exp_c = np.exp(corr * T_adapt - corr.max() * T_adapt)
        w = exp_c / exp_c.sum()
        grad = np.zeros(n)
        for oi in range(2 * n - 1):
            if w[oi] < 1e-10:
                continue
            shift = oi - center
            if shift >= 0:
                m1 = np.arange(shift, n)
                grad[m1] += w[oi] * (1.0 - h[m1 - shift])
                m2 = np.arange(n - shift)
                grad[m2] -= w[oi] * h[m2 + shift]
            else:
                s2 = -shift
                m1 = np.arange(s2, n)
                grad[m1] += w[oi] * (1.0 - h[m1 - s2])
                m2 = np.arange(n - s2)
                grad[m2] -= w[oi] * h[m2 + s2]
        grad *= dx
        m_v = 0.9 * m_v + 0.1 * grad
        v_v = 0.999 * v_v + 0.001 * grad * grad
        m_hat = m_v / (1.0 - 0.9 ** t)
        v_hat = v_v / (1.0 - 0.999 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = project_feasible(h, n)
        if t % 2000 == 0:
            c5 = compute_c5(h)
            if c5 < c5_best:
                c5_best = c5
                h_best = h.copy()
        if t % 20000 == 0:
            lr *= 0.6
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Quick SLSQP polish
    for T, k in [(200, 50), (30, 20), (5, 10), (1, 5)]:
        if time.time() - t0 > budget_s * 0.15:
            break
        h_new = slsqp(h_best, n, T, k, 3000, 1e-17)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Binary local search
    t_end_bl = t0 + budget_s * 0.45
    h_bin, c5_bin = bin_ls(h_best, n, t_end_bl, seed=seed)
    c5_bin = compute_c5(h_bin)
    if c5_bin < c5_best:
        c5_best = c5_bin
        h_best = h_bin.copy()

    # Phase 3: Adam continuation from binary
    rem = budget_s - (time.time() - t0)
    if rem > 100:
        t_end_ad = t0 + budget_s * 0.70
        h_ad, c5_ad = adam_opt(h_best, n, t_end_ad, lr0=0.002)
        h_ad = project_feasible(h_ad, n)
        c5_ad = compute_c5(h_ad)
        if c5_ad < c5_best:
            c5_best = c5_ad
            h_best = h_ad

    # Phase 4: SLSQP fine polish
    for T, k in [(2, 6), (0.5, 4), (0.2, 3)]:
        if time.time() - t0 > budget_s * 0.90:
            break
        h_new = slsqp(h_best, n, T, k, 4000, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Second binary local search from refined solution
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h_bin2, c5_bin2 = bin_ls(h_best, n, t0 + budget_s - 30, seed=seed + 7)
        c5_bin2 = compute_c5(h_bin2)
        if c5_bin2 < c5_best:
            c5_best = c5_bin2
            h_best = h_bin2.copy()
            if time.time() - t0 < budget_s - 15:
                h_new = slsqp(h_best, n, 1, 4, 2000, 1e-18)
                h_new = project_feasible(h_new, n)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new

    # Phase 6: Try higher resolution
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        n2 = n * 2
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = project_feasible(h_hi, n2)
        c5_hi = compute_c5(h_hi)
        t_end_hi = t0 + budget_s - 20
        for T, k in [(50, 30), (5, 10)]:
            if time.time() > t_end_hi:
                break
            h_new = slsqp(h_hi, n2, T, k, 3000, 1e-17)
            h_new = project_feasible(h_new, n2)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
                n = n2

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
