        else: hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any(): g[m1] += (1.0 - h[j[m1] - shift])
        if m2.any(): g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any(): g[m1] += (1.0 - h[j[m1] + s])
        if m2.any(): g[m2] -= h[j[m2] - s]
    return g

def _smooth_max_obj(x, n, T, top_k):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    return (m + np.log(e.sum()) / T) * (2.0 / n)

def _smooth_max_grad(x, n, T, top_k):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    for wi, ci in zip(w, idx):
        if wi > 1e-12:
            grad += wi * shift_grad(x, n, ci - (n - 1))
    return grad * (2.0 / n)

def _slsqp_opt(h, n, T, top_k, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    def fun(x):
        return _smooth_max_obj(x, n, T, top_k)
    def jac(x):
        return _smooth_max_grad(x, n, T, top_k)
    def cf(x):
        return x.sum() - n / 2.0
    def cj(x):
        return np.ones(n)
    try:
        res = minimize(fun, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()
