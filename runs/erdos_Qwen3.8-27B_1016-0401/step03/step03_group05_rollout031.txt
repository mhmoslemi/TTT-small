import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0

        if s > target:

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] = 1.0 - h[j[m1] - shift]
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] = 1.0 - h[j[m1] + s]
        grad[m2] -= h[j[m2] - s]
    return grad

def compute_corr_and_weights(h, n, T, K):
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_corr = 2 * n - 1
    mc = np.max(corr)
    if T is not None:
        sh = T * (corr - mc)
        sh -= sh.max()  # numerical stability
        e = np.exp(sh)
        w = e / e.sum()
    else:
        # Top-K weights
        top_idx = np.argpartition(corr, -K)[-K:]
        top_vals = corr[top_idx]

        wv = (top_vals / max(mc, 1e-15))**8
        wv /= wv.sum()
        w_full[top_idx] = wv
        w = w_full
    return corr, mc, w

    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1

        mc = np.max(corr)
        sh = T * (corr - mc)
        sh -= sh.max()
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")

        sh = T * (corr - mc)

        return grad * dx

        return x.sum() - target
    
    bounds = [(0.0, 1.0)] * n

def perturb(h, n, rng, pi):

To minimize this over a_new ∈ [0,1], I compute for each m:
f_m(a) = corr_m_old + (a - a_old) * slope_m

The minimum of max_m f_m(a) over a ∈ [0,1] is found by:
1. Computing f_m(0) and f_m(1) for all m → get max at 0 and max at 1
2. Finding all breakpoints where f_m(a) = f_k(a) for m≠k
3. Evaluating the max at each breakpoint
4. Taking the minimum

For n=280, there are 559 shifts. The breakpoints are at (corr_k - corr_m - (a_old-a_old)*(slope_k-slope_m)) / (slope_m - slope_k)... this is getting complicated. Let me just use a simpler approach:

Since f_m(a) is linear in a, the max of 559 linear functions is a convex PL function with at most 559 segments. I can find the minimum by:
1. Computing the "upper envelope" of the 559 lines
2. Finding the minimum of the upper envelope on [0,1]

The upper envelope can be found by: for each a in [0,1], the active line is the one with the largest f_m(a). The minimum of the envelope is at a point where two lines intersect (or at 0 or 1).

A simple O(n·k) approach: 
- Sort lines by slope
- Build upper envelope (O(k log k) for k lines)
- Find minimum of envelope on [0,1]

But for n=280 and 559 lines, even a brute-force approach works:
- Evaluate at a=0, a=1
- For each pair of lines, compute intersection a* = (b_m - b_k)/(slope_k - slope_m) where b_m = corr_m_old - a_old*slope_m
- If 0 <= a* <= 1, evaluate max at a*
- Keep track of minimum

But that's O(k²) = O(559²) = O(312481) per coordinate. For 280 coordinates, that's 87M operations per full pass. That might be too slow.

Better: just evaluate at a few candidate points:
- a = 0, a = 1
- For the top 10-20 "active" shifts (those with highest f_m at current a), compute their pairwise intersections with a few others
- Take the best

Actually, for the coordinate descent to work well, I don't need the exact optimum. I can just try a few candidate values:
- a = 0, a = 1
- a = (argmax_t f_t(0) + argmax_t f_t(1)) intersection point
- Clamped to [0,1]

Let me simplify: for each coordinate, try a ∈ {0, 1} and pick the better one. This is the "binary coordinate descent" and it's very fast.

Even simpler: just set each h[i] to either 0 or 1 (whichever gives lower C5), one at a time, sweeping. This reduces the problem to a binary combinatorial optimization.

But this might lose the advantage of fractional values. The optimal solution might have some h[i] values in (0,1).

Let me do a hybrid:
1. Binary coordinate descent: for each i, try h[i]=0 and h[i]=1 (projecting to maintain sum constraint)
2. Then a few rounds of fine-tuning with SLSQP

Actually, maintaining the sum constraint during coordinate descent is tricky. If I change h[i] from 0 to 1, the sum increases by 1, so I need to decrease some other h[j] by 1. This becomes a "swap" operation: change h[i] from 0 to 1 and h[j] from 1 to 0.

This is getting complex. Let me stick with the gradient-based approach but make it better:

3. Adam with T = 2, 1, 0.5 (3 levels, 50000 iters each, decreasing LR)
4. Top-K SLSQP with K = 2, 3, 5 (3 levels, 5000 iters each) - focus on the worst shifts

6. Try n=140 if time permits
7. Final polish

The key difference from before: more temperature levels, more Adam iterations, and a "frozen max" approach where I identify the top few worst shifts and focus optimization on them.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] = 1.0 - h[j[m1] - shift]
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] = 1.0 - h[j[m1] + s]
        grad[m2] -= h[j[m2] - s]
    return grad

def slsqp_lse(h, n, T, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc) - (T * (corr - mc)).max()
        return (mc + np.log(np.exp(sh).sum()) / T) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc) - (T * (corr - mc)).max()
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(n_corr):
            w = sw[oi]
            if w < 1e-13:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return grad * dx

    def cf(x):
        return x.sum() - target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def slsqp_topk(h, n, K, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        ti = np.argpartition(corr, -K)[-K:]
        tv = corr[ti]
        wv = (tv / max(mc, 1e-15))**10
        ws = wv.sum()
        if ws < 1e-15:
            wv = np.ones(K) / K
        else:
            wv /= ws
        grad = np.zeros(n)
        for i in range(K):
            grad += wv[i] * shift_grad(x, n, ti[i] - center)
        return grad * dx

    def cf(x):
        return x.sum() - target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def adam_lse(h_init, n, T, max_itrs, lr0, deadline):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    beta = 0.9
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.85, 0.999
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()

    for it in range(max_itrs):
        if time.time() > deadline:
            break
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh -= sh.max()
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(n_corr):
            w = sw[oi]
            if w < 1e-13:
                continue
            grad += w * shift_grad(h_y, n, oi - center)
        grad *= dx
        it1 = it + 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        lr = lr0 / (1.0 + 0.001 * it)
        h_new = h_y - lr * m / (1 - b1**it1) / (np.sqrt(v / (1 - b2**it1)) + 1e-8)
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 10000 == 0:
            c5c = compute_c5(h_sg)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h_sg.copy()
    c5c = compute_c5(h_sg)
    if c5c < c5_best:
        c5_best = c5c
        h_best = h_sg.copy()
    return h_best, c5_best

def perturb(h, n, rng, pi):
    mode = pi % 5
    x = np.arange(n) * 2.0 / n
    if mode == 0:
        f = 0.4 + 0.3 * (pi % 4)
        return h + 0.025 * np.sin(2 * np.pi * f * x + rng.uniform(0, 6.28))
    elif mode == 1:
        f = 0.15 + 0.2 * (pi % 3)
        return h + 0.02 * np.cos(4 * np.pi * f * x + rng.uniform(0, 6.28))
    elif mode == 2:
        block = max(n // 6, 1)
        h_p = h.copy()
        for b in range(6):
            lo_i = b * block
            hi_i = min(lo_i + block, n)
            h_p[lo_i:hi_i] += rng.uniform(-0.035, 0.035)
        return h_p
    elif mode == 3:
        return h + rng.normal(0, 0.018, n)
    else:
        cp = rng.uniform(0, n)
        w = rng.uniform(n // 8, n // 3)
        bump = 0.025 * np.exp(-0.5 * ((np.arange(n) - cp) / w) ** 2)
        return h + bump * rng.choice([-1.0, 1.0])

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP log-sum-exp geometric cooling
    for T in [32, 16, 8, 4, 2, 1]:
        if elapsed() > budget_s * 0.40:
            break
        h_new = slsqp_lse(h_best, n, T, 400, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Adam with decreasing T
    for T in [3, 1.5, 1, 0.5]:
        if elapsed() > budget_s * 0.55:
            break
        lr = 0.08 if T >= 2 else 0.04
        h_new, c5 = adam_lse(h_best, n, T, 60000, lr, t0 + budget_s * 0.55)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 3: Top-K SLSQP (focused on worst correlations)
    for K in [2, 3, 5, 8]:
        if elapsed() > budget_s * 0.72:
            break
        h_new = slsqp_topk(h_best, n, K, 5000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Perturbations + re-optimize
    for pi in range(15):
        if elapsed() > budget_s * 0.85:
            break
        h_p = perturb(h_best, n, rng, pi + seed)
        h_p = project_feasible(h_p, n)
        h_new = slsqp_lse(h_p, n, 4, 250, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Alternate resolution
    if elapsed() < budget_s * 0.90:
        for n2 in [n * 2, n // 2]:
            if elapsed() > budget_s * 0.93:
                break
            if n2 < 60 or n2 > 1200:
                continue
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0, 2.0, n2, endpoint=False)
            h_alt = np.interp(x_new, x_old, h_best)
            h_alt = project_feasible(h_alt, n2)
            c5_alt = compute_c5(h_alt)
            for T in [6, 2]:
                if elapsed() > budget_s * 0.95:
                    break
                h_new2 = slsqp_lse(h_alt, n2, T, 350, 1e-14)
                h_new2 = project_feasible(h_new2, n2)
                c5_alt = compute_c5(h_new2)
            if elapsed() < budget_s * 0.97:
                h_new2 = slsqp_topk(h_alt, n2, 5, 3000, 1e-15)
                h_new2 = project_feasible(h_new2, n2)
                c5_alt = compute_c5(h_new2)
            if c5_alt < c5_best:
                c5_best = c5_alt
                h_best = h_new2
                n = n2

    # Final polish
    if elapsed() < budget_s - 10:
        h_new = slsqp_lse(h_best, n, 1, 2000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    if c5_final < c5_best:
        c5_best = c5_final
        h_best = project_feasible(h_best, n)
    return (h_best, c5_best, n)
