def project_feasible(y, n):

def _mk_masks(n):
    idx = np.arange(2 * n - 1)
    k = idx - (n - 1)
    masks = []
    for i in range(n):
        m1 = (i - k >= 0) & (i - k < n)
        m2 = (i + k >= 0) & (i + k < n)
        off1 = np.where(m1)[0] - (i + (n - 1))
        off2 = np.where(m2)[0] - (i - (n - 1))
        masks.append((m1, m2, off1, off2))
    return masks

def solve_1d(A, s):
    f0 = A.max()
    f1 = (A + s).max()
    a0 = np.where(A >= f0 - 1e-13)[0]
    if s[a0].min() > 0:
        return 0.0
    a1 = np.where(A + s >= f1 - 1e-13)[0]
    if s[a1].max() < 0:
        return 1.0
    lo, hi = 0.0, 1.0

        if s[act].max() <= 0:
            lo = mid
        elif s[act].min() >= 0:

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def _sm_obj(x, n, T, K, t0, tlim):
    if time.time() - t0 > tlim:
        return 0.37
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    if K < len(corr):
        topk = np.argpartition(corr, -K)[-K:]
        cs = corr[topk]
        mcs = cs.max()
        sh = T * (cs - mcs)
        e = np.exp(sh)
        return (mcs + np.log(e.sum()) / T) * (2.0 / n)
    sh = T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * (2.0 / n)

def _sm_grad(x, n, T, K, t0, tlim):
    if time.time() - t0 > tlim:
        return np.zeros(n)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    center = n - 1
    if K < len(corr):
        topk = np.argpartition(corr, -K)[-K:]
        cs = corr[topk]
        mcs = cs.max()
        sh = T * (cs - mcs)
        e = np.exp(sh)
        sw = e / e.sum()
    else:
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        topk = np.arange(len(corr))
    grad = np.zeros(n)
    for ii in range(len(sw)):
        w = sw[ii]
        if w < 1e-14:
            continue
        grad += w * shift_grad(x, n, topk[ii] - center)
    return grad * (2.0 / n)

                       args=(n, T, K, t0, tlim), bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14,
                                'gtol': 1e-12, 'maxls': 20})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

            h_p = h_best + pert + rng.normal(0, 0.01, n)
        elif pi % 3 == 1:
            h_bin = (h_best > 0.5).astype(float)
            h_bin = project_feasible(h_bin, n)
            flip_n = max(1, n // 20)
            idxs = rng.choice(n, flip_n, replace=False)
            h_bin[idxs] = 1.0 - h_bin[idxs]
            h_p = project_feasible(h_bin, n)
        else:
            h_p = project_feasible(h_best + rng.normal(0, 0.03, n), n)
        h_new, c5_new = coord_descent(h_p, n, t0, t0 + min(rem - 15, 80), 2000, seed + 100 + pi)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 5: Final L-BFGS-B polish
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        for T in [2, 1, 0.5]:
            rem = budget_s - (time.time() - t0)
            if rem < 40:
                break
            h_new = do_lbfgsb(h_best, n, T, 100, 3000, t0, budget_s - 10)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 6: Final CD polish
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_cd, c5_cd = coord_descent(h_best, n, t0, t0 + rem - 5, 3000, seed + 999)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
