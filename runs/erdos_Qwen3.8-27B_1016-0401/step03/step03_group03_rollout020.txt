import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target: lo = mid
        else: hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1, m2 = j >= shift, (j + shift) < n
        g[m1] += 1.0 - h[j[m1] - shift]
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1, m2 = (j + s) < n, j >= s
        g[m1] += 1.0 - h[j[m1] + s]
        g[m2] -= h[j[m2] - s]
    return g

_N, _T = 0, 1.0

def _obj(x):
    c = np.correlate(x, 1.0 - x, mode="full")
    mc = c.max()
    e = np.exp(np.clip(_T * (c - mc), -50, 0))
    return (mc + np.log(e.sum()) / _T) * (2.0 / _N)

def _jac(x):
    c = np.correlate(x, 1.0 - x, mode="full")
    mc = c.max()
    e = np.exp(np.clip(_T * (c - mc), -50, 0))
    w = e / e.sum()
    ctr = _N - 1
    g = np.zeros(_N)
    for i in range(len(w)):
        if w[i] > 1e-14:
            g += w[i] * shift_grad(x, _N, i - ctr)
    return g * (2.0 / _N)

def _cf(x): return x.sum() - _N / 2.0
def _cj(x): return np.ones(_N)

    try:
        r = minimize(_obj, h, method='SLSQP', jac=_jac,
                     bounds=[(0,1)]*_N, constraints=[{'type':'eq','fun':_cf,'jac':_cj}],
                     options={'maxiter':mi,'ftol':1e-16})

def project_feasible(y, n):

    j = np.arange(n)

        g[m1] += 1.0 - h[j[m1] - shift]
        g[m2] -= h[j[m2] + shift]
    else:

        g[m1] += 1.0 - h[j[m1] + s]
        g[m2] -= h[j[m2] - s]
