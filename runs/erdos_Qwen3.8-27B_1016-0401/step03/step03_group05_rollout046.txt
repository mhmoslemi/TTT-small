    j = np.arange(n)
    grad = np.zeros(n)
    center = n - 1

    if shift >= 0:

        m = j < (n - shift)
        grad[m] += (1.0 - h[j[m] + shift])
        m2 = j >= shift
        grad[m2] -= h[j[m2] - shift]
    else:

        m = j >= s
        grad[m] += (1.0 - h[j[m] - s])
        m2 = (j + s) < n
        grad[m2] -= h[j[m2] + s]
    return grad

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        if np.clip(y - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m = j < (n - shift)
        grad[m] += (1.0 - h[j[m] + shift])
        m2 = j >= shift
        grad[m2] -= h[j[m2] - shift]
    else:
        s = -shift
        m = j >= s
        grad[m] += (1.0 - h[j[m] - s])
        m2 = (j + s) < n
        grad[m2] -= h[j[m2] + s]
    return grad

def slsqp_smooth(h, n, T, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        return (mc + np.log(np.exp(sh).sum()) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(n_corr):
            w = sw[oi]
            if w < 1e-12:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return grad * dx
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def slsqp_topk(h, n, K, maxiter, ftol, power=20.0):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        top_idx = np.argpartition(corr, -K)[-K:]
        tv = corr[top_idx]
        w = (tv / max(mc, 1e-15)) ** power
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(K) / K
        else:
            w /= ws
        grad = np.zeros(n)
        for i in range(K):
            grad += w[i] * shift_grad(x, n, top_idx[i] - center)
        return grad * dx
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,

def lp_active_step(h, n):
    """Linearized active-set LP step."""
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    active = np.where(corr >= mc - 0.5 * mc * 0.02)[0]
    if len(active) < 2 or len(active) > 30:
        return h.copy()
    # Minimize t s.t. f_k(h0)+g_k.(h-h0) <= t, sum(h)=n/2, 0<=h<=1
    # Variables: [h(0..n-1), t]
    # obj: [0,...,0, 1]
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = []
    b_ub = []
    for ci in active:
        s = ci - center
        g_k = shift_grad(h, n, s)
        f_k0 = corr[ci]
        # f_k0 + g_k.(h-h0) - t <= 0  =>  g_k.h - t <= -f_k0 + g_k.h0
        row = np.zeros(n + 1)
        row[:n] = g_k
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(-f_k0 + g_k @ h)
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds_lp = [(0, 1)] * n + [(0, None)]
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.status == 0:
        return np.clip(res.x[:n], 0, 1)

        if c < c5_best - 1e-9:
            c5_best = c
            h_best = h_lp
        else:
            break

    # Phase 4: Try n*2
    t_rem = budget_s - (time.time() - t0)
    if t_rem > 120:
        n2 = n * 2
        h2 = interp_h(h_best, n, n2)
        h2 = project_feasible(h2, n2)
        h2, c5_2 = optimize_at_resolution(h2, n2, t0 + budget_s - 60)
        if c5_2 < c5_best:
            c5_best = c5_2
            h_best = h2
            n = n2

    # Phase 5: Try n//2
    t_rem = budget_s - (time.time() - t0)
    if t_rem > 80 and n > 100:
        n3 = n // 2
        h3 = interp_h(h_best, n, n3)
        h3 = project_feasible(h3, n3)
        h3, c5_3 = optimize_at_resolution(h3, n3, t0 + budget_s - 30)
        if c5_3 < c5_best:
            c5_best = c5_3
            h_best = h3
            n = n3

    # Phase 6: Final heavy polish
    if time.time() < t0 + budget_s - 15:
        for K in [3, 5, 10]:
            if time.time() > t0 + budget_s - 5:
                break
            h_new = slsqp_topk(h_best, n, K, 8000, 1e-17, power=32.0)
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new)
            if c < c5_best:
                c5_best = c
                h_best = h_new
        h_new = slsqp_smooth(h_best, n, 0.5, 8000, 1e-17)
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < c5_best:
            c5_best = c
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
