            ino = rng.choice(neg)
            d = min(0.04, h_cur[ip] - 0.005, 1.0 - h_cur[ino] - 0.005)
            if d <= 0.002:
                continue
            h_try = h_cur.copy()
            h_try[ip] -= d
            h_try[ino] += d
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-11:
                h_cur = h_try
                c5_cur = c5_try
                corr_cur = get_corr(h_cur)
                improved = True
                break
        if not improved and it > 5000:
            top_idx = np.argsort(corr_cur)[-4:]
            mc = corr_cur[top_idx].max()
            w = np.exp(1.5 * (corr_cur[top_idx] - mc))
            w /= w.sum()
            g = np.zeros(n)
            for wk, ik in zip(w, top_idx):
                g += wk * shift_grad(h_cur, n, ik - center)
            pos = np.where(g > 0.01)[0]
            neg = np.where(g < -0.01)[0]
            if len(pos) == 0 or len(neg) == 0:
                break
    return h_cur, c5_cur

def multi_temperature_nesterov(h_init, n, t0, budget):
    h = h_init.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    configs = [
        (5, 3.0, 0.08, 25000),
        (4, 2.0, 0.07, 25000),
        (3, 1.5, 0.06, 25000),
        (2, 1.0, 0.05, 25000),
        (2, 0.5, 0.04, 25000),
    ]
    remaining = budget - (time.time() - t0)
    per_phase = remaining / len(configs)
    for K, T, alpha0, iters in configs:
        if time.time() - t0 > budget:
            break
        h_new, _ = nesterov_phase(h, n, time.time(), time.time() + per_phase * 0.95,
                                   K, T, iters)
        h_new = project_feasible(h_new, n)
        cv = compute_c5(h_new)
        if cv < c5_best:
            c5_best = cv
            h_best = h_new
            h = h_new
        else:
            h = h_best.copy()
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase A: Quick SLSQP to enter basin
    T_a = 1.5
    if time.time() - t0 < budget_s - 800:
        h_new = slsqp_phase(h_best, n, T_a, 5000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase B: Multi-temperature Nesterov (main optimization)
    if time.time() - t0 < budget_s - 450:
        t_b = time.time()
        b_budget = t_b + (budget_s - 450 - (t_b - t0))
        h_sg, c5_sg = multi_temperature_nesterov(h_best, n, t0, b_budget)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase C: SLSQP fine polish
    for T_c in [0.8, 0.5, 0.3, 0.2, 0.12]:
        if time.time() - t0 > budget_s - 250:
            break
        h_new = slsqp_phase(h_best, n, T_c, 10000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase D: Pair swap local search
    if time.time() - t0 < budget_s - 120:
        t_d = time.time()
        d_budget = t_d + (budget_s - 120 - (t_d - t0))
        h_ps, c5_ps = pair_swap(h_best, n, t0, d_budget, rng)
        if c5_ps < c5_best:
            c5_best = c5_ps
            h_best = h_ps

    # Phase E: Final SLSQP + Nesterov micro
    if time.time() - t0 < budget_s - 40:
        h_new = slsqp_phase(h_best, n, 0.1, 12000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    if time.time() - t0 < budget_s - 15:
        h_sg, c5_sg = nesterov_phase(h_best, n, time.time(), time.time() + (budget_s - 15 - (time.time() - t0)),
                                      2, 0.5, 20000)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Final: perturbation restart
    if time.time() - t0 < budget_s - 20:
        for si in range(5):
            if time.time() - t0 > budget_s - 10:
                break
            sigma = 0.002 + 0.003 * si
            h_p = h_best + rng.normal(0, sigma, n)
            h_p = project_feasible(h_p, n)
            h_new = slsqp_phase(h_p, n, 0.5, 6000, 1e-15)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, n)
