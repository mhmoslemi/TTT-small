def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

_gl_n = 0
_gl_T = 1.0
_gl_k = 1

    return x.sum() - _gl_n / 2.0

    return np.ones(_gl_n)

def _gl_obj(x):
    n, T, k = _gl_n, _gl_T, _gl_k
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = np.max(topk)
    sh = T * (topk - mc)
    sh -= sh.max()
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * dx

def _gl_jac(x):
    n, T, k = _gl_n, _gl_T, _gl_k
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = np.max(topk)
    sh = T * (topk - mc)
    sh -= sh.max()
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    j_arr = np.arange(n)
    for oi in range(mk):
        w = sw[oi]
        if w < 1e-12:
            continue
        shift = idx[oi] - center
        if shift >= 0:
            m1 = j_arr >= shift
            if m1.any():
                grad[m1] += w * (1.0 - x[j_arr[m1] - shift])
            m2 = (j_arr + shift) < n
            if m2.any():
                grad[m2] -= w * x[j_arr[m2] + shift]
        else:
            s2 = -shift
            m1 = (j_arr + s2) < n
            if m1.any():
                grad[m1] += w * (1.0 - x[j_arr[m1] + s2])
            m2 = j_arr >= s2
            if m2.any():
                grad[m2] -= w * x[j_arr[m2] - s2]
    grad *= dx
    return grad

def slsqp_opt(h, n, T, k, maxiter, ftol):

    _gl_n = n
    _gl_T = T
    _gl_k = k

                       bounds=bounds, constraints=cons,
