    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

_smooth_N = 0
_smooth_T = 1.0
_smooth_target = 0.0

    sh = _smooth_T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _smooth_T) * (2.0 / _smooth_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

_STARG = 0.0

        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

_SN = 0
_ST = 1.0
_STARG = 0.0

def s_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _ST * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _ST) * (2.0 / _SN)

def s_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _ST * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _SN - 1
    dx = 2.0 / _SN
    grad = np.zeros(_SN)
    for oi in range(len(sw)):
        if sw[oi] < 1e-14:
            continue
        k = oi - center
        if k >= 0:
            i1 = np.arange(k, _SN)
            i2 = np.arange(0, _SN - k)
            grad[i1] += sw[oi] * (1.0 - x[i1 - k])
            grad[i2] -= sw[oi] * x[i2 + k]
        else:
            s = -k
            i1 = np.arange(0, _SN - s)
            i2 = np.arange(s, _SN)
            grad[i1] += sw[oi] * (1.0 - x[i1 + s])
            grad[i2] -= sw[oi] * x[i2 - s]
    return grad * dx

def s_cf(x):
    return x.sum() - _STARG

def s_cj(x):
    return np.ones(_SN)

def do_slsqp(h, n, T, maxiter, ftol):
    global _SN, _ST, _STARG
    _SN = n
    _ST = float(T)
    _STARG = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': s_cf, 'jac': s_cj}]
    try:
        res = minimize(s_obj, h, method='SLSQP', jac=s_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def nesterov(h_init, n, deadline, K=12, max_iters=900000):
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        beta = float(it) / (it + 3.0)
        h_y = np.clip(h_sg + beta * (h_sg - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top_idx = np.argsort(corr)[-K:]
        mv = float(corr[top_idx].max())
        w = np.exp(6.0 * (corr[top_idx] - mv))
        w /= w.sum()
        g = np.zeros(n)
        for wk, ik in zip(w, top_idx):
            g += wk * shift_grad_raw(h_y, n, int(ik) - center)
        g *= dx
        alpha = 0.12 / (1.0 + 0.002 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 8000 == 0:
            cv = compute_c5(h_sg)
            if cv < c5_best:
                c5_best = cv
                h_best = h_sg.copy()
    cv = compute_c5(project_feasible(h_sg, n))
    if cv < c5_best:
        c5_best = cv
        h_best = project_feasible(h_sg, n).copy()
    return h_best, c5_best

def local_search(h, n, rng, steps, deadline):
    dx = 2.0 / n
    center = n - 1
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for it in range(steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        ak = int(np.argmax(corr))
        shift = ak - center
        g = shift_grad_raw(h_cur, n, shift) * dx
        pos_idx = np.where(g > 0.008)[0]
        neg_idx = np.where(g < -0.008)[0]
        if len(pos_idx) == 0 or len(neg_idx) == 0:
            i1 = rng.integers(0, n)
            i2 = rng.integers(0, n)
            if i1 == i2:
                continue
            d = 0.01 * (1.0 if h_cur[i1] > h_cur[i2] else -1.0)
            h_try = h_cur.copy()
            h_try[i1] -= d
            h_try[i2] += d
            h_try = np.clip(h_try, 0, 1)
            h_try = project_feasible(h_try, n)
            if compute_c5(h_try) < c5_cur:
                h_cur = h_try
                c5_cur = compute_c5(h_cur)
            continue
        for _ in range(3):
            ip = pos_idx[rng.integers(0, len(pos_idx))]
            inn = neg_idx[rng.integers(0, len(neg_idx))]
            delta = min(0.02, h_cur[ip], 1.0 - h_cur[inn])
            h_try = h_cur.copy()
            h_try[ip] -= delta
            h_try[inn] += delta
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-12:
                h_cur = h_try
                c5_cur = c5_try
                break
    return h_cur, c5_cur

def bimodal_commit(h, n, thresh):
    h2 = h.copy()
    h2[h2 < thresh] = 0.0
    h2[h2 > 1.0 - thresh] = 1.0
    return project_feasible(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Try structured alternatives as starting points
    xg = np.linspace(0.0, 2.0, n, endpoint=False)
    for freq in [1.0, 2.0, 3.0, 5.0]:
        h_alt = np.clip(0.5 + 0.35 * np.sin(np.pi * freq * xg), 0.0, 1.0)
        h_alt = project_feasible(h_alt, n)
        if compute_c5(h_alt) < c5_best:
            c5_best = compute_c5(h_alt)
            h_best = h_alt.copy()

    # Phase 1: SLSQP coarse-to-fine (geometric temp)
    T = 128.0
    for _ in range(22):
        if time.time() > budget_s - 320:
            break
        h_new = do_slsqp(h_best, n, T, 12000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        T *= 0.72

    # Phase 2: Nesterov multi-shift
    if time.time() < budget_s - 220:
        h_sg, c5_sg = nesterov(h_best, n, budget_s - 220, K=12)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase 3: SLSQP fine polish
    for T in [0.5, 0.3, 0.18, 0.1, 0.06, 0.04, 0.025]:
        if time.time() > budget_s - 140:
            break
        h_new = do_slsqp(h_best, n, T, 15000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Local coordinate search
    if time.time() < budget_s - 90:
        h_lp, c5_lp = local_search(h_best, n, rng, 8000, budget_s - 70)
        if c5_lp < c5_best:
            c5_best = c5_lp
            h_best = h_lp
        h_new = do_slsqp(h_best, n, 0.15, 12000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Perturbation restarts
    for pi in range(50):
        if time.time() > budget_s - 45:
            break
        sigma = 0.0015 + 0.004 * (pi % 7)
        h_p = project_feasible(h_best + rng.normal(0, sigma, n), n)
        h_new = do_slsqp(h_p, n, 0.8, 6000, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Bimodal commit + re-optimize
    if time.time() < budget_s - 55:
        for thresh in [0.12, 0.2, 0.3]:
            if time.time() > budget_s - 25:
                break
            h_bm = bimodal_commit(h_best, n, thresh)
            h_new = do_slsqp(h_bm, n, 0.35, 10000, 1e-16)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Final SLSQP polish
    if time.time() < budget_s - 12:
        h_new = do_slsqp(h_best, n, 0.06, 15000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
