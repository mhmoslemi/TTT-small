    g = np.zeros(n)
    for ii in range(len(w)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * shift_grad(x, n, int(topk[ii]) - center)
    return g * (2.0 / n)

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):
    flag = [False]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_obj, h, method='L-BFGS-B', jac=_sm_grad,
                       args=(n, T, K, t0, tlim, flag), bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol * 0.1,
                                'maxls': 20, 'maxfun': maxiter * 3})
        if flag[0]:
            return h
        out = project_feasible(res.x, n)
        return out
    except Exception:
        return h.copy()

def do_slsqp(h, n, T, K, maxiter, tol, t0, tlim):
    flag = [False]
    bounds = [(0.0, 1.0)] * n

    def cfun(x):
        return x.sum() - n / 2.0

    def cjac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(_sm_obj, h, method='SLSQP', jac=_sm_grad,
                       args=(n, T, K, t0, tlim, flag), bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': tol})
        if flag[0]:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def struct_perturb(h, n, rng):
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(6):
        f = rng.uniform(0.5, 12.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.005, 0.05)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert += rng.normal(0, 0.006, n)
    return h + pert

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(280)
        h_init[::2] = 1.0

    n0 = len(h_init)
    h0 = project_feasible(h_init, n0)
    c5_best = compute_c5(h0)
    h_best = h0.copy()
    n_best = n0

    # Multi-resolution coordinate descent
    for n_try in [n0, n0 - 40, n0 + 80, 160, 400]:
        if n_try < 40:
            continue
        rem = budget_s - (time.time() - t0)
        if rem < 60:
            break
        if n_try == n0:
            h_cur = h0.copy()
        else:
            x_old = np.arange(n0) * (2.0 / n0)
            x_new = np.linspace(0, 2.0, n_try, endpoint=False)
            h_cur = np.interp(x_new, x_old, h0)
        h_cur = project_feasible(h_cur, n_try)
        t_sub = min(budget_s - 40, time.time() + 120)
        h_new, c5_new = coord_descent(h_cur, n_try, t0, t_sub, 200)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new
            n_best = n_try

    # Coordinate descent polishing on best
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        t_sub = min(budget_s - 30, time.time() + 150)
        h_new, c5_new = coord_descent(h_best, n_best, t0, t_sub, 400)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # L-BFGS-B with smooth max
    for T, mi in [(8.0, 3000), (4.0, 4000), (2.0, 5000), (1.0, 6000)]:
        rem = budget_s - (time.time() - t0)
        if rem < 90:
            break
        K = min(300, 2 * n_best - 1)
        h_new = do_lbfgsb(h_best, n_best, T, K, mi, 1e-14, t0, budget_s - 30)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        # Follow with coord descent
        rem = budget_s - (time.time() - t0)
        if rem > 40:
            h_new2, c5_2 = coord_descent(h_new, n_best, t0, min(budget_s - 30, time.time() + 80), 150)
            if c5_2 < c5_best:
                c5_best = c5_2
                h_best = h_new2

    # SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h_new = do_slsqp(h_best, n_best, 2.0, 100, 3000, 1e-14, t0, budget_s - 30)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Diverse restarts
    n_restarts = 60
    for pi in range(n_restarts):
        rem = budget_s - (time.time() - t0)
        if rem < 20:
            break
        if pi % 4 == 0:
            h_p = struct_perturb(h_best, n_best, rng)
        elif pi % 4 == 1:
            sig = 0.003 + 0.010 * (pi % 5)
            h_p = h_best + rng.normal(0, sig, n_best)
        elif pi % 4 == 2:
            h_p = h_best.copy()
            bs = max(n_best // 10, 5)
            bi = int(rng.integers(0, max(1, n_best // bs)))
            h_p[bi * bs:min(n_best, (bi + 1) * bs)] = 1.0 - h_p[bi * bs:min(n_best, (bi + 1) * bs)]
            h_p += rng.normal(0, 0.002, n_best)
        else:
            # Random restart at different n
            nr = int(rng.choice([120, 160, 200, 240, 280, 320, 360]))
            h_p = 0.5 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 6) * np.arange(nr) / nr + rng.uniform(0, 6.28))
            h_p += 0.15 * np.sin(2 * np.pi * rng.uniform(3, 15) * np.arange(nr) / nr + rng.uniform(0, 6.28))
            h_p = project_feasible(h_p, nr)
            # Quick coord descent
            t_sub = min(budget_s - 15, time.time() + 25)
            h_p, c5_p = coord_descent(h_p, nr, t0, t_sub, 60)
            if c5_p < c5_best:
                c5_best = c5_p
                h_best = h_p
                n_best = nr
            continue

        h_p = project_feasible(h_p, n_best)
        # Quick L-BFGS-B then coord descent
        rem = budget_s - (time.time() - t0)
        if rem < 15:
            break
        t_sub = min(budget_s - 15, time.time() + 20)
        h_new = do_lbfgsb(h_p, n_best, 2.0, 100, 1500, 1e-13, t0, t_sub)
        h_new2, c5_2 = coord_descent(h_new, n_best, t0, min(budget_s - 15, time.time() + 15), 80)
        c5 = compute_c5(h_new2)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new2

    # Final coordinate descent polish
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_new, c5_new = coord_descent(h_best, n_best, t0, budget_s - 10, 200)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Final SLSQP
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_new = do_slsqp(h_best, n_best, 1.5, 80, 3000, 1e-15, t0, budget_s - 5)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n_best)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n_best)
