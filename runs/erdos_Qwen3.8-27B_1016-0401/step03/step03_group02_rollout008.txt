    n = len(x)

    dx = 2.0 / n
    g = _corr_grad_part(o, x, n) * dx
    g += 2.0 * M * (x.sum() - target)
    return g

def sumtop_obj(x, top_idx, target, M):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    s = np.sum(corr[top_idx]) / len(top_idx)
    return s * (2.0 / n) + M * (x.sum() - target)**2

def sumtop_grad(x, top_idx, target, M):
    n = len(x)
    center = n - 1
    dx = 2.0 / n
    g = np.zeros(n)
    for idx in top_idx:
        o = idx - center
        g += _corr_grad_part(o, x, n)
    g *= dx / len(top_idx)
    g += 2.0 * M * (x.sum() - target)
    return g

def coord_descent(h, n_sweeps):
    n = len(h)
    target = n / 2.0
    dts = [0.01, -0.01, 0.005, -0.005]
    c5_cur = compute_c5(h)
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            best_c5 = c5_cur
            best_h = None
            for dt in dts:
                trial = h.copy()
                trial[j] = np.clip(trial[j] + dt, 0.0, 1.0)
                trial = fix_constraint(trial, target)
                c5_new = compute_c5(trial)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = trial
            if best_h is not None and best_c5 < c5_cur - 1e-15:
                h = best_h
                c5_cur = best_c5
                improved = True
        if not improved:
            break
    return h, c5_cur

def binary_push(h, threshold):
    n = len(h)
    target = n / 2.0
    c5_cur = compute_c5(h)
    for j in range(n):
        if h[j] > threshold:
            trial = h.copy(); trial[j] = 1.0
        elif h[j] < 1.0 - threshold:
            trial = h.copy(); trial[j] = 0.0
        else:
            continue
        trial = fix_constraint(trial, target)
        c5_new = compute_c5(trial)
        if c5_new < c5_cur - 1e-15:
            h = trial; c5_cur = c5_new
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(340)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n
    M = 500.0

    # Phase 1: L-BFGS-B smooth max (50% budget)
    p1_end = t0 + budget_s * 0.45
    for T in [300, 1000, 5000, 20000, 80000]:
        if elapsed() > p1_end:
            break
        try:
            res = minimize(sm_obj, h_best, args=(T, target, M),
                          method='L-BFGS-B', jac=sm_grad,
                          bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-18, 'maxfun': 50000})
            h_new = fix_constraint(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 2: SLSQP on worst lag (12% budget)
    p2_end = t0 + budget_s * 0.58
    for _ in range(20):
        if elapsed() > p2_end:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst = int(np.argmax(corr))
        try:
            res = minimize(lag_obj_real, h_best, args=(worst, target, M),
                          method='SLSQP', jac=lag_grad_real, bounds=bounds,
                          options={'maxiter': 800, 'ftol': 1e-18})
            h_new = fix_constraint(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-16:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 3: Leveling top lags (10% budget)
    p3_end = t0 + budget_s * 0.68
    for n_top in [2, 3, 5]:
        if elapsed() > p3_end:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        top_idx = np.argsort(corr)[-n_top:]
        try:
            res = minimize(sumtop_obj, h_best, args=(top_idx, target, M),
                          method='SLSQP', jac=sumtop_grad, bounds=bounds,
                          options={'maxiter': 1000, 'ftol': 1e-17})
            h_new = fix_constraint(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-16:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 4: Coordinate descent + binary push (8% budget)
    p4_end = t0 + budget_s * 0.76
    for _ in range(6):
        if elapsed() > p4_end:
            break
        h_best, c5_cd = coord_descent(h_best, 2)
        if c5_cd < c5_best:
            c5_best = c5_cd
        h_best, c5_bp = binary_push(h_best, 0.03)
        if c5_bp < c5_best:
            c5_best = c5_bp

    # Phase 5: Multi-n exploration (15% budget)
    p5_end = t0 + budget_s * 0.92
    n_tries = [280, 336, 200]
    for n_alt in n_tries:
        if elapsed() > p5_end:
            break
        target_alt = n_alt / 2.0
        x_old = np.arange(len(h_best)) * (2.0 / len(h_best))
        x_new = np.linspace(0.0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = fix_constraint(h_alt, target_alt)
        c5_alt = compute_c5(h_alt)
        bounds_alt = [(0.0, 1.0)] * n_alt
        # Optimize alternating h_alt
        for T in [2000, 10000]:
            if elapsed() > p5_end:
                break
            try:
                res = minimize(sm_obj, h_alt, args=(T, target_alt, M),
                              method='L-BFGS-B', jac=sm_grad,
                              bounds=bounds_alt, options={'maxiter': 2000, 'ftol': 1e-17})
                h_new = fix_constraint(res.x, target_alt)
                c5_new = compute_c5(h_new)
                if c5_new < c5_alt:
                    c5_alt = c5_new
                    h_alt = h_new
            except Exception:
                pass
        # SLSQP polish
        corr_a = np.correlate(h_alt, 1.0 - h_alt, mode="full")
        worst_a = int(np.argmax(corr_a))
        try:
            res = minimize(lag_obj_real, h_alt, args=(worst_a, target_alt, M),
                          method='SLSQP', jac=lag_grad_real, bounds=bounds_alt,
                          options={'maxiter': 800, 'ftol': 1e-17})
            h_new = fix_constraint(res.x, target_alt)
            c5_new = compute_c5(h_new)
            if c5_new < c5_alt:
                c5_alt = c5_new
                h_alt = h_new
        except Exception:
            pass
        h_alt, c5_cd_a = coord_descent(h_alt, 1)
        if c5_cd_a < c5_alt:
            c5_alt = c5_cd_a
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt
            target = n / 2.0
            bounds = [(0.0, 1.0)] * n

    # Phase 6: Final polish (8% budget)
    p6_end = t0 + budget_s - 3.0
    target = len(h_best) / 2.0
    n = len(h_best)
    bounds = [(0.0, 1.0)] * n
    for _ in range(10):
        if elapsed() > p6_end:
            break
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst_f = int(np.argmax(corr_f))
        try:
            res = minimize(lag_obj_real, h_best, args=(worst_f, target, M),
                          method='SLSQP', jac=lag_grad_real, bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-19})
            h_new = fix_constraint(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass
    h_best, c5_cd_f = coord_descent(h_best, 2)
    if c5_cd_f < c5_best:
        c5_best = c5_cd_f
    h_best, c5_bp_f = binary_push(h_best, 0.02)
    if c5_bp_f < c5_best:
        c5_best = c5_bp_f

    c5_final = min(compute_c5(h_best), c5_best)
    h_best = np.clip(h_best, 0.0, 1.0)
    return (h_best, c5_final, len(h_best))
