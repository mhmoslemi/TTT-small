import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    s = abs(int(shift))
    grad = np.zeros(n)
    if s == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        m1 = np.arange(s, n)
        m2 = np.arange(0, n - s)
        grad[m1] += (1.0 - h[m1 - s])
        grad[m2] -= h[m2 + s]
    return grad

def multi_shift_grad(h, n, shifts, weights):
    grad = np.zeros(n)
    for s, w in zip(shifts, weights):
        if w < 1e-13:
            continue
        as_ = abs(int(s))
        if as_ == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            m1 = np.arange(as_, n)
            m2 = np.arange(0, n - as_)
            grad[m1] += w * (1.0 - h[m1 - as_])
            grad[m2] -= w * h[m2 + as_]
    return grad

def slsqp_logsumexp(h, n, T, maxiter, ftol):
    dx = 2.0 / n
    center = n - 1
    target = n / 2.0
    n_corr = 2 * n - 1
    safe_max = 50.0 / max(T, 1e-6)

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = np.clip(T * (corr - mc), -50, 50)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = np.clip(T * (corr - mc), -50, 50)
        e = np.exp(sh)
        sw = e / e.sum()
        shifts = np.arange(n_corr) - center
        g = multi_shift_grad(x, n, shifts, sw)
        return g * dx

    def cons_fun(x):
        return x.sum() - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def slsqp_topk(h, n, K, maxiter, ftol):
    dx = 2.0 / n
    center = n - 1
    target = n / 2.0
    n_corr = 2 * n - 1

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        top_idx = np.argpartition(corr, -K)[-K:]
        top_vals = corr[top_idx]
        w = (top_vals / max(mc, 1e-15))**6
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.ones(K) / K
        else:
            w /= w_sum
        shifts = top_idx - center
        g = multi_shift_grad(x, n, shifts, w)
        return g * dx

    def cons_fun(x):
        return x.sum() - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def nesterov_opt(h_init, n, t0, max_time, max_iters=500000):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    alpha0 = 0.5
    K_min = 1
    K_max = 40

    for it in range(max_iters):
        if time.time() - t0 > max_time:
            break
        frac = min((it + 1) / max_iters, 1.0)
        K = max(K_min, int(K_min + (K_max - K_min) * frac * frac))

        beta = (float(it) / (it + 1.5)) if it > 0 else 0.0
        h_y = np.clip(h_sg + beta * (h_sg - h_prev), 0.0, 1.0)

        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)

        if K <= 1:
            idx = int(np.argmax(corr))
            g = shift_grad(h_y, n, idx - center)
        else:
            top_idx = np.argpartition(corr, -K)[-K:]
            top_vals = corr[top_idx]
            w = (top_vals / max(mc, 1e-15))**8
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.ones(K) / K
            else:
                w /= w_sum
            shifts = top_idx - center
            g = multi_shift_grad(h_y, n, shifts, w)
        g *= dx

        alpha = alpha0 / (1.0 + 0.003 * it)
        h_new = project_feasible(h_y - alpha * g, n)

        h_prev = h_sg
        h_sg = h_new

        if (it + 1) % 15000 == 0:
            c5c = compute_c5(h_sg)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h_sg.copy()

    c5c = compute_c5(h_sg)
    if c5c < c5_best:
        c5_best = c5c
        h_best = h_sg.copy()
    return h_best, c5_best

def try_resolution(h_best, c5_best, n, n2, t0, budget_s):
