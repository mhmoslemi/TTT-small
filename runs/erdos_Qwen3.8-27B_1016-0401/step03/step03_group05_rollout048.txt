        T = 0.4 * (0.005 / 0.4) ** (frac ** 0.7)

        a_idx = rng.integers(n_ones)
        b_idx = rng.integers(n_ones)
        a = int(ones_pos[a_idx])
        b = int(zeros_pos[b_idx])
        if a == b:
            continue

        h[a] = 0.0
        h[b] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(new_corr)) * dx

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

        delta = new_c5 - c5
        if delta <= 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            c5 = new_c5
            corr = new_corr
            ones_pos[a_idx] = b
            zeros_pos[b_idx] = a
        else:
            h[a] = 1.0
            h[b] = 0.0

    return best_h, best_c5

def coord_descend(h, n, passes, budget):
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    t0 = time.time()
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1

    for p in range(passes):
        if time.time() - t0 > budget:
            break
        for j in range(n):
            if time.time() - t0 > budget:
                break
            old_val = h[j]
            # Compute coefficients for each shift
            # corr_s = a_s * h[j] + b_s for each shift s
            center = n - 1
            for s_val in range(-n + 1, n):
                pass  # Too slow to do per-shift; use the fact it's linear in h[j]
            # Faster: just evaluate at a few points
            candidates = np.array([0.0, 1.0, old_val])
            # Add current + small perturbations
            candidates = np.concatenate([candidates, [old_val - 0.05, old_val + 0.05,
                                                      old_val - 0.01, old_val + 0.01]])
            candidates = candidates[(candidates >= 0) & (candidates <= 1)]
            best_v = old_val
            best_obj = compute_c5(h)
            for v in candidates:
                h[j] = v
                # Adjust another variable to maintain sum
                diff = target - h.sum()
                if abs(diff) < 1e-14:
                    obj = compute_c5(h)
                    if obj < best_obj:
                        best_obj = obj
                        best_v = v
                h[j] = old_val
            h[j] = best_v
            c5_now = compute_c5(h)
            # Re-project to maintain constraint
            h = project_feasible(h, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

def nesterov_fast(h_init, n, t0, remaining, max_iters=500000):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    K = 3

    for it in range(max_iters):
        if time.time() - t0 > t0 + remaining:
            break
        frac = min(it / max_iters, 1.0)
        K_use = max(2, int(2 + 8 * frac * frac))

        beta = (float(it) / (it + 1.5)) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)

        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        top_idx = np.argpartition(corr, -K_use)[-K_use:]
        tv = corr[top_idx]
        w = (tv / max(mc, 1e-15)) ** 8
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(K_use) / K_use
        else:
            w /= ws

        g = np.zeros(n)
        for i in range(K_use):
            g += w[i] * shift_grad(h_y, n, top_idx[i] - center)
        g *= dx

        alpha = 0.5 / (1.0 + 0.003 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)

        h_prev = h_sg
        h_sg = h_new

        if (it + 1) % 50000 == 0:
            c5c = compute_c5(h_sg)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h_sg.copy()

    c5_final = compute_c5(h_sg)
    if c5_final < c5_best:
        c5_best = c5_final
        h_best = h_sg.copy()
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(eval('initial_h_values'), dtype=float)
    else:
        n_init = 280
        h_best = np.zeros(n_init)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP polish with decreasing T
    for T in [8, 4, 2, 1]:
        if time.time() - t0 > budget_s - 700:
            break
        rem = budget_s - (time.time() - t0) - 700
        if rem < 20:
            break
        h_new = slsqp_opt(h_best, n, T, 50, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Nesterov with top-K
    rem = budget_s - (time.time() - t0) - 500
    if rem > 50:
        h_new, c5_new = nesterov_fast(h_best, n, t0, min(rem, 200), 300000)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 3: Binary SA
    rem = budget_s - (time.time() - t0) - 400
    if rem > 60:
        h_bin, c5_bin = binary_sa(h_best, n, min(rem, 180), rng)
        if c5_bin < c5_best:
            c5_best = c5_bin
            h_best = h_bin.copy()

    # Phase 4: SLSQP from binary / best
    for T in [2, 1, 0.5]:
        if time.time() - t0 > budget_s - 250:
            break
        h_new = slsqp_opt(h_best, n, T, 80, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Top-K focused SLSQP
    for K in [2, 3, 5]:
        if time.time() - t0 > budget_s - 150:
            break
        h_new = slsqp_topk(h_best, n, K, 60, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Try nearby n values
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        for n2 in [n + 8, n - 8, n + 16]:
            if time.time() - t0 > budget_s - 60:
                break
            if n2 < 100 or n2 > 1000:
                continue
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0, 2.0, n2, endpoint=False)
            h_alt = np.interp(x_new, x_old, h_best)
            h_alt = project_feasible(h_alt, n2)
            c5_alt = compute_c5(h_alt)
            for T2 in [2, 1]:
                if time.time() - t0 > budget_s - 40:
                    break
                h_alt = slsqp_opt(h_alt, n2, T2, 40, 1e-17)
                h_alt = project_feasible(h_alt, n2)
                c5_alt = compute_c5(h_alt)
                if c5_alt < c5_best:
                    c5_best = c5_alt
                    h_best = h_alt
                    n = n2

    # Phase 7: Final micro-polish
    if time.time() - t0 < budget_s - 30:
        h_new = slsqp_topk(h_best, n, 2, 40, 1e-19)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, n)
