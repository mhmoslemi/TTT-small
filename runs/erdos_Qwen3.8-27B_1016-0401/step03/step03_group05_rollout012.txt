    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def slsqp_smooth(h, n, T, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)

        return grad * dx
    def cons_fun(x):
        return x.sum() - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})

        return h.copy()

def slsqp_balanced(h, n, K, maxiter, ftol):

    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    lam = 10.0

        corr = np.correlate(x, 1.0 - x, mode="full")
        top_vals = np.sort(corr)[-K:]
        return (top_vals.mean() + lam * top_vals.var()) * dx

    dx = 2.0 / n
    center = n - 1
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        tv = np.sort(corr)[-K:]
        return (tv.mean() + lam * tv.var()) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        top_idx = np.argpartition(corr, -K)[-K:]
        tv = corr[top_idx]
        tm = tv.mean()
        w = (1.0 + lam * 2.0 * (tv - tm)) / K
        grad = np.zeros(n)
        for i in range(K):
            if abs(w[i]) > 1e-14:
                grad += w[i] * shift_grad(x, n, top_idx[i] - center)
        return grad * dx
    def cf(x):
        return x.sum() - target
    def cj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def nesterov_adaptive(h_init, n, t0, deadline, max_iters):
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    alpha0 = 0.25
    K_min, K_max = 2, 30
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        K = max(K_min, int(K_min + (K_max - K_min) * frac * frac))
        beta = (float(it) / (it + 1.5)) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        if K <= 1:
            idx = int(np.argmax(corr))
            g = shift_grad(h_y, n, idx - center) * dx
        else:
            top_idx = np.argpartition(corr, -K)[-K:]
            top_vals = corr[top_idx]
            w = (top_vals / max(mc, 1e-15)) ** 10
            ws = w.sum()
            w = np.ones(K) / K if ws < 1e-15 else w / ws
            g = np.zeros(n)
            for i in range(K):
                g += w[i] * shift_grad(h_y, n, top_idx[i] - center)
            g *= dx
        alpha = alpha0 / (1.0 + 0.004 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        h_avg += (h_new - h_avg) / (float(it + 1))
        if (it + 1) % 10000 == 0:
            pa = project_feasible(h_avg, n)
            ca = compute_c5(pa)
            if ca < c5_best:
                c5_best, h_best = ca, pa.copy()
            cc = compute_c5(h_sg)
            if cc < c5_best:
                c5_best, h_best = cc, h_sg.copy()
    return h_best, c5_best

def perturb_polish(h, n, rng, n_trials, deadline):
    hb = h.copy()
    cb = compute_c5(hb)
    for tr in range(n_trials):
        if time.time() > deadline:
            break
        sigma = 0.0015 + 0.002 * (tr % 4)
        hp = hb + rng.normal(0, sigma, n)
        hp = project_feasible(hp, n)
        hnew = slsqp_smooth(hp, n, 5, 2500, 1e-15)
        hnew = project_feasible(hnew, n)
        c5 = compute_c5(hnew)
        if c5 < cb:
            cb, hb = c5, hnew
        if time.time() > deadline:
            break
    return hb, cb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP smooth coarse
    for T in [16, 8, 4]:
        if time.time() - t0 > budget_s - 550:
            break
        hnew = slsqp_smooth(h_best, n, T, 4000, 1e-16)
        hnew = project_feasible(hnew, n)
        c5 = compute_c5(hnew)
        if c5 < c5_best:
            c5_best, h_best = c5, hnew

    # Phase 2: Nesterov long run
    dl2 = t0 + budget_s * 0.50
    hsg, c5sg = nesterov_adaptive(h_best, n, t0, dl2, 600000)
    if c5sg < c5_best:
        c5_best, h_best = c5sg, hsg
    h_best = project_feasible(h_best, n)

    # Phase 3: Balanced SLSQP (equalize top-K)
    for K, lam in [(3, 5.0), (5, 8.0), (8, 12.0), (5, 4.0)]:
        if time.time() - t0 > budget_s - 250:
            break
        hnew = slsqp_balanced(h_best, n, K, lam, 5000, 1e-17)
        hnew = project_feasible(hnew, n)
        c5 = compute_c5(hnew)
        if c5 < c5_best:
            c5_best, h_best = c5, hnew

    # Phase 4: Fine SLSQP polish
    for T in [3, 1]:
        if time.time() - t0 > budget_s - 180:
            break
        hnew = slsqp_smooth(h_best, n, T, 8000, 1e-18)
        hnew = project_feasible(hnew, n)
        c5 = compute_c5(hnew)
        if c5 < c5_best:
            c5_best, h_best = c5, hnew

    # Phase 5: Nesterov short refinement
    dl5 = t0 + budget_s * 0.80
    hsg2, c5s2 = nesterov_adaptive(h_best, n, t0, dl5, 200000)
    if c5s2 < c5_best:
        c5_best, h_best = c5s2, hsg2
    h_best = project_feasible(h_best, n)

    # Phase 6: Small perturb + polish
    if time.time() - t0 < budget_s - 80:
        h_best, c5_best = perturb_polish(h_best, n, rng, 12, t0 + budget_s - 40)

    # Phase 7: Try higher resolution
    el = time.time() - t0
    if el < budget_s - 100 and n < 600:
        n2 = n * 2 if n <= 400 else n + 120
        n2 = min(n2, 1200)
        if n2 > n:
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0, 2.0, n2, endpoint=False)
            hat = np.interp(x_new, x_old, h_best)
            hat = project_feasible(hat, n2)
            cat = compute_c5(hat)
            for T2 in [4, 2]:
                if time.time() - t0 > budget_s - 40:
                    break
                hnew = slsqp_smooth(hat, n2, T2, 3000, 1e-15)
                hnew = project_feasible(hnew, n2)
                cat = compute_c5(hnew)
                if cat < c5_best:
                    c5_best, h_best, n = cat, hnew, n2

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
