            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0: continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0: continue

                diff += sub
            if abs(diff) < 1e-12: break

def slsqp_lag(h, lag_idx, bounds, n_pts, maxiter):
    dx = 2.0 / n_pts
    target = n_pts / 2.0

    def obj(x):
        return np.correlate(x, 1.0 - x, mode="full")[lag_idx] * dx

    def jac(x):
        o = lag_idx - (n_pts - 1)
        g = np.zeros(n_pts)
        if o == 0:
            g[:] = 1.0 - 2.0 * x
        elif o > 0:
            g[o:] += 1.0 - x[:n_pts - o]
            g[:n_pts - o] -= x[o:]
        else:
            p = -o
            g[:n_pts - p] += 1.0 - x[p:]
            g[p:] -= x[:n_pts - p]
        return g * dx

    def eqfun(x):
        return x.sum() - target

    def eqjac(x):
        return np.ones(n_pts)

    cons = [dict(type='eq', fun=eqfun, jac=eqjac)]
    res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-19})
    return np.clip(res.x, 0.0, 1.0)

                    break

    return h, c5_cur

    c5_cur = compute_c5(h)

        c5_new = compute_c5(trial)

            c5_cur = c5_new
    return h, c5_cur
