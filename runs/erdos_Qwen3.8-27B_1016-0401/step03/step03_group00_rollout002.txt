    h_avg = h_init.copy()
    c5a = compute_c5(h_init)
    ha = h_init.copy()
    s_prev = 0
    y_prev = 0
    alpha = 0.1
    for it in range(2000000):
        if time.time() > t_end:
            break
        beta = 2.0 / (it + 5.0)
        h_y = h_cur + beta * (h_cur - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        global _G_N, _G_TGT, _G_T, _G_K
        _G_N = n
        _G_TGT = float(n / 2.0)
        _G_T = float(T)
        _G_K = top_k
        g = _jac(h_y)
        alpha = np.clip(alpha * 0.999, 1e-5, 0.5)
        h_new = project_feasible(h_y - alpha * g, n)
        s_new = h_new - h_cur
        sy = s_new @ (h_new - h_mom)
        if sy > 1e-10:
            alpha = min(0.5, max(1e-5, (s_new @ s_new) / sy))
        h_mom = h_cur
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 3000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < c5a:
                c5a, ha = c1, h_cur.copy()
            c2 = compute_c5(project_feasible(h_avg, n))
            if c2 < c5a:
                c5a, ha = c2, project_feasible(h_avg, n).copy()
    c_final = compute_c5(h_cur)
    if c_final < c5a:
        c5a, ha = c_final, h_cur.copy()
    return ha, c5a

def _perturb(h, n, rng, pi):
    mode = pi % 6
    if mode == 0:
        sig = 0.001 + 0.001 * (pi // 6)
        return project_feasible(h + rng.normal(0, sig, n), n)
    elif mode == 1:
        h_p = h.copy()
        lb = int(rng.integers(0, n - 30))
        ln = int(rng.integers(3, 15))
        h_p[lb:lb+ln] = 1.0 - h_p[lb:lb+ln]
        return project_feasible(h_p, n)
    elif mode == 2:
        h_p = h.copy()
        k = int(rng.integers(1, n // 2))
        h_p = np.concatenate([h_p[k:], h_p[:k]])
        return project_feasible(h_p, n)
    elif mode == 3:
        h_p = h.copy()
        for _ in range(int(rng.integers(2, 6))):
            i = int(rng.integers(0, n - 5))
            h_p[i:i+5] = 1.0 - h_p[i:i+5]
        return project_feasible(h_p, n)
    elif mode == 4:
        sig = 0.003
        h_p = h.copy()
        mask = rng.random(n) < 0.1
        h_p[mask] += rng.normal(0, sig, mask.sum())
        return project_feasible(h_p, n)
    else:
        h_p = h.copy()
        i1 = int(rng.integers(0, n // 3))
        i2 = int(rng.integers(n // 3, 2 * n // 3))
        i3 = int(rng.integers(2 * n // 3, n))
        h_p = np.concatenate([h_p[i3:], h_p[i1:i2], h_p[:i1], h_p[i2:i3]])
        return project_feasible(h_p, n)

def _try_n_interp(h_src, n_src, n_tgt, rng):
    """Interpolate solution from n_src to n_tgt points."""
    x_src = np.linspace(0, 2, n_src)
    x_tgt = np.linspace(0, 2, n_tgt)
    h_interp = np.interp(x_tgt, x_src, h_src)
    h_interp = project_feasible(h_interp, n_tgt)
    return h_interp

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP warm-up with decreasing temperature (25%)
    t1 = t0 + budget_s * 0.25
    for T, tk, mi in [(64, 30, 1500), (16, 20, 2000), (4, 14, 3000),
                      (1, 10, 4000), (0.5, 8, 5000)]:
        if time.time() > t1:
            break
        hn = _slsqp(h_best, n, T, tk, mi, 1e-13)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 2: BB-Nesterov gradient with averaging (20%)
    t2 = t0 + budget_s * 0.45
    hs, c5s = _bb_nesterov(h_best, n, t2, top_k=8, T=2.0)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs

    # Phase 3: SLSQP fine polish (10%)
    t3 = t0 + budget_s * 0.55
    for T, tk, mi in [(1, 8, 5000), (0.5, 6, 7000), (0.3, 5, 8000)]:
        if time.time() > t3:
            break
        hn = _slsqp(h_best, n, T, tk, mi, 1e-14)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 4: Multi-perturbation exploration (25%)
    t4 = t0 + budget_s * 0.80
    attempts = 0
    while attempts < 80 and time.time() < t4:
        hp = _perturb(h_best, n, rng, attempts)
        c5p = compute_c5(hp)
        if c5p < c5_best * 1.01:
            hn = _slsqp(hp, n, 0.5, 6, 4000, 1e-14)
            hn = project_feasible(hn, n)
            c5 = compute_c5(hn)
            if c5 < c5_best:
                c5_best, h_best = c5, hn
        attempts += 1

    # Phase 5: Try n=300 by interpolation (10%)
    t5 = t0 + budget_s * 0.90
    if time.time() < t5:
        n_alt = 300
        h_alt = _try_n_interp(h_best, n, n_alt, rng)
        c5_alt = compute_c5(h_alt)
        if c5_alt < c5_best * 1.001:
            hn = _slsqp(h_alt, n_alt, 1, 8, 3000, 1e-13)
            hn = project_feasible(hn, n_alt)
            c5 = compute_c5(hn)
            if c5 < c5_best:
                c5_best, h_best, n = c5, hn, n_alt

    # Phase 6: Final ultra-fine SLSQP (10%)
    t6 = t0 + budget_s * 0.97
    if time.time() < t6:
        for T, tk, mi in [(0.3, 5, 8000), (0.2, 4, 10000), (0.1, 3, 12000)]:
            if time.time() > t6:
                break
            hn = _slsqp(h_best, n, T, tk, mi, 1e-15)
            hn = project_feasible(hn, n)
            c5 = compute_c5(hn)
            if c5 < c5_best:
                c5_best, h_best = c5, hn

    # Final: verify and return
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    if c5_final > c5_best:
        c5_final = c5_best
    return (h_best, c5_final, n)
