    n = len(h0)
    dx = 2.0 / n

    h = cp.Variable(n)
    C = cp.Variable()
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == target]

    corr0 = np.correlate(h0, 1.0 - h0, mode="full")

    for _ in range(200):
        diff = target - h.sum()

            break
        if diff > 0:

        else:
            idx = np.argsort(h)
        for i in idx:

                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0:

                h[i] -= sub
                diff += sub

    else:

import numpy as np
from scipy.optimize import minimize
import time

try:
    import cvxpy as cp
    HAS_CP = True
except ImportError:
    HAS_CP = False

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-13:
                break
    return h

def corr_grad_vec(h, o):
    n = len(h)
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] += 1.0 - h[:n - o]
        g[:n - o] += -h[o:]
    else:
        p = -o
        g[:n - p] += 1.0 - h[p:]
        g[p:] += -h[:n - p]
    return g

def solve_lp(hb, target, dx):
    n = len(hb)
    if not HAS_CP:
        return None
    corr0 = np.correlate(hb, 1.0 - hb, mode="full")
    ctr = n - 1
    hvar = cp.Variable(n)
    Cvar = cp.Variable()
    cons = [hvar >= 0, hvar <= 1, cp.sum(hvar) == target]
    for k in range(2 * n - 1):
        o = k - ctr
        g = corr_grad_vec(hb, o)
        lin = corr0[k] * dx + dx * cp.sum(g * (hvar - hb))
        cons.append(lin <= Cvar)
    prob = cp.Problem(cp.Minimize(Cvar), cons)
    for slv in [cp.CBC, cp.GLOP, cp.GLPK]:
        try:
            prob.solve(solver=slv, verbose=False)
            if hvar.value is not None and not np.any(np.isnan(hvar.value)):
                return np.array(hvar.value, dtype=float)
        except Exception:
            continue
    return None

def lp_refine(h, t0, tlim, n_iters=40):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    hb = h.copy()
    c5b = compute_c5(hb)
    for _ in range(n_iters):
        if time.time() - t0 > tlim:
            break
        hv = solve_lp(hb, target, dx)
        if hv is None:
            break
        hn = np.clip(hv, 0, 1)
        hn = fix_constraint(hn, target)
        if np.allclose(hn, hb, atol=1e-12):
            break
        best_dm = None
        best_ct = c5b
        for dm in [1.0, 0.7, 0.5, 0.3]:
            ht = (1.0 - dm) * hb + dm * hn
            ht = fix_constraint(ht, target)
            ct = compute_c5(ht)
            if ct < best_ct - 2e-14:
                best_ct = ct
                best_dm = dm
        if best_dm is not None:
            ht = (1.0 - best_dm) * hb + best_dm * hn
            hb = fix_constraint(ht, target)
            c5b = compute_c5(hb)
    return hb, c5b

def slsqp_worst(h, bounds, target, maxiter=800):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    wi = int(np.argmax(corr))
    o = wi - (n - 1)

    def obj(x):
        return np.correlate(x, 1.0 - x, mode="full")[wi] * dx

    def grad(x):
        return corr_grad_vec(x, o) * dx

    def eqf(x):
        return x.sum() - target

    def eqj(x):
        return np.ones(len(x))

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eqf, 'jac': eqj}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h.copy()

def softmax_slsqp(h, bounds, target, T=5000, maxiter=2000):
    n = len(h)
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = float(corr.max())
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = float(corr.max())
        e = np.exp(T * (corr - m))
        sw = e / np.sum(e)
        ctr = n - 1
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if sw[k] < 1e-13:
                continue
            g += sw[k] * corr_grad_vec(x, k - ctr)
        return g * dx

    def eqf(x):
        return x.sum() - target

    def eqj(x):
        return np.ones(len(x))

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eqf, 'jac': eqj}],
                       options={'maxiter': maxiter, 'ftol': 1e-17})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h.copy()

def coord_descend(h, target, n_sweeps=2):
    n = len(h)
    h = h.copy()
    c5c = compute_c5(h)
    dts = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for _ in range(n_sweeps):
        imp = False
        for j in range(n):
            for dt in dts:
                tr = h.copy()
                tr[j] = np.clip(tr[j] + dt, 0, 1)
                tr = fix_constraint(tr, target)
                c = compute_c5(tr)
                if c < c5c - 1e-14:
                    h = tr
                    c5c = c
                    imp = True
                    break
        if not imp:
            break
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(340)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: LP linearization (35% budget)
    h_best, c5_best = lp_refine(h_best, t0, budget_s * 0.35, 50)

    # Phase 2: SLSQP on worst lags (15%)
    for _ in range(15):
        if time.time() - t0 > budget_s * 0.50:
            break
        hn = slsqp_worst(h_best, bounds, target, 1000)
        hn = fix_constraint(hn, target)
        c = compute_c5(hn)
        if c < c5_best - 2e-15:
            c5_best = c
            h_best = hn
        else:
            break

    # Phase 3: Soft-max SLSQP (10%)
    if time.time() - t0 < budget_s * 0.60:
        hn = softmax_slsqp(h_best, bounds, target, T=5000, maxiter=3000)
        hn = fix_constraint(hn, target)
        c = compute_c5(hn)
        if c < c5_best - 2e-15:
            c5_best = c
            h_best = hn
        # Try different T
        if time.time() - t0 < budget_s * 0.58:
            hn = softmax_slsqp(h_best, bounds, target, T=20000, maxiter=2000)
            hn = fix_constraint(hn, target)
            c = compute_c5(hn)
            if c < c5_best - 2e-15:
                c5_best = c
                h_best = hn

    # Phase 4: LP refinement (15%)
    h_best, c5_best = lp_refine(h_best, t0, budget_s * 0.75, 30)

    # Phase 5: Coordinate descent + perturbations (10%)
    for _ in range(5):
        if time.time() - t0 > budget_s * 0.85:
            break
        h_best, c5_cd = coord_descend(h_best, target, 2)
        if c5_cd < c5_best:
            c5_best = c5_cd
        if time.time() - t0 < budget_s * 0.84:
            hpert = h_best + rng.normal(0, 0.008, n)
            hpert = fix_constraint(hpert, target)
            cp2 = compute_c5(hpert)
            if cp2 < c5_best:
                h_best = hpert
                c5_best = cp2

    # Phase 6: Final polish (15%)
    p5_end = budget_s - 5.0
    for _ in range(8):
        if time.time() - t0 > p5_end:
            break
        hn = slsqp_worst(h_best, bounds, target, 1500)
        hn = fix_constraint(hn, target)
        c = compute_c5(hn)
        if c < c5_best - 2e-15:
            c5_best = c
            h_best = hn
    if time.time() - t0 < p5_end - 2.0:
        h_best, c5_best = lp_refine(h_best, t0, p5_end, 15)
    h_best, c5_cd = coord_descend(h_best, target, 3)
    if c5_cd < c5_best:
        c5_best = c5_cd

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
