    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(sw)):
        w = sw[idx]
        if w < 1e-12:
            continue
        o = idx - center
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx

def obj_lag(x, n, lag_idx):
    o = lag_idx - (n - 1)
    dx = 2.0 / n
    if o == 0:
        return np.sum(x * (1.0 - x)) * dx
    elif o > 0:
        return np.sum(x[:n - o] * (1.0 - x[o:])) * dx
    else:
        p = -o
        return np.sum(x[p:] * (1.0 - x[:n - p])) * dx

def grad_lag(x, n, lag_idx):
    o = lag_idx - (n - 1)
    return grad_one_lag(x, o, n)

def optimize_smoothmax(h, n, T, bounds, maxiter=400):
    res = minimize(obj_smoothmax, h, args=(n, T), jac=grad_smoothmax,
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-18, 'maxfun': maxiter * 3})
    return np.clip(res.x, 0.0, 1.0)

def optimize_lag(h, n, lag_idx, bounds, maxiter=400):
    res = minimize(obj_lag, h, args=(n, lag_idx), jac=grad_lag,
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-18, 'maxfun': maxiter * 3})
    return np.clip(res.x, 0.0, 1.0)

def minmax_subgrad(h, n, n_iter, step0):
    target = n / 2.0
    step = step0
    c5_cur = compute_c5(h)
    for _ in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        worst = np.argmax(corr)
        g = grad_one_lag(h, worst - (n - 1), n)
        diff = target - h.sum()
        g += (diff / n) * 0.01
        h = np.clip(h - step * g, 0.0, 1.0)
        h = fix_constraint(h, target)
        c5_new = compute_c5(h)
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
        else:
            step *= 0.6
            h = np.clip(h + step * 0.5 * g, 0.0, 1.0)
            h = fix_constraint(h, target)
    return h, c5_cur

def pair_swap_refine(h, n, max_rounds=60):
    target = n / 2.0
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_improve = -1
        best_pair = None
        for j in ones:
            for k in zeros:
                if j == k:
                    continue
                h2 = (h > 0.5).astype(float).copy()
                h2[j] = 0.0
                h2[k] = 1.0
                h2 = fix_constraint(h2, target)
                c5_new = compute_c5(h2)
                if c5_new < c5_cur + best_improve:
                    improve = c5_new - c5_cur
                    if improve < -1e-14:
                        best_improve = improve
                        best_pair = (j, k, h2)
        if best_pair is not None:
            h = best_pair[2]
            c5_cur = c5_cur + best_improve
        else:
            break
    return h, c5_cur

def diverse_init(n, seed):
    rng = np.random.default_rng(seed)
    inits = []
    # Alternating
    h1 = np.zeros(n)
    h1[::2] = 1.0
    inits.append(h1)
    # Kronecker-like
    for alpha in [np.sqrt(2), 1.618, np.pi / 3]:
        idx = np.arange(n)
        vals = np.sin(2 * np.pi * alpha * idx / n) + 0.3 * np.sin(2 * np.pi * alpha * 2 * idx / n)
        thresh = np.median(vals)
        h = (vals > thresh).astype(float)
        inits.append(h)
    # Periodic patterns
    for period in [5, 7, 9, 13]:
        pattern = (np.arange(period) % 2 == 0).astype(float)
        n_ones_pat = pattern.sum()
        h = np.tile(pattern, n // period + 1)[:n]
        inits.append(h)
    return inits

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    # Initialize from parent if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Diverse initialization and quick BFGS (10% budget)
    p1_end = t0 + budget_s * 0.10
    for h_init in diverse_init(n, seed):
        if elapsed() > p1_end:
            break
        h_init = fix_constraint(h_init, target)
        c5_i = compute_c5(h_init)
        if c5_i < c5_best:
            h_best = h_init.copy()
            c5_best = c5_i

    # Quick warm-up on best
    for T in [300, 1000]:
        if elapsed() > p1_end:
            break
        try:
            h_new = optimize_smoothmax(h_best, n, T, bounds, maxiter=200)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 2: Multi-temperature smooth max optimization (25% budget)
    p2_end = t0 + budget_s * 0.35
    for T in [100, 300, 800, 2000, 5000, 12000, 30000]:
        if elapsed() > p2_end:
            break
        try:
            h_new = optimize_smoothmax(h_best, n, T, bounds, maxiter=500)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass
        # Subgradient step between T values
        h_best, c5_sg = minmax_subgrad(h_best, n, 20, 0.005)
        if c5_sg < c5_best:
            c5_best = c5_sg

    # Phase 3: Targeted worst-lag optimization (20% budget)
    p3_end = t0 + budget_s * 0.55
    for _ in range(50):
        if elapsed() > p3_end:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        top_k = np.argsort(corr)[-3:]
        improved = False
        for lag in top_k:
            if elapsed() > p3_end:
                break
            try:
                h_new = optimize_lag(h_best, n, lag, bounds, maxiter=300)
                h_new = fix_constraint(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-15:
                    c5_best = c5_new
                    h_best = h_new
                    improved = True
            except Exception:
                pass
        # Short subgradient
        h_best, c5_sg = minmax_subgrad(h_best, n, 10, 0.003)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_best  # already updated
        if not improved and elapsed() > p3_end * 0.8:
            break

    # Phase 4: Binarize + pair swap refinement (20% budget)
    p4_end = t0 + budget_s * 0.75
    for _ in range(6):
        if elapsed() > p4_end:
            break
        # Project to binary
        h_bin = (h_best > 0.5).astype(float)
        h_bin = fix_constraint(h_bin, target)
        c5_bin = compute_c5(h_bin)
        if c5_bin < c5_best:
            h_best = h_bin
            c5_best = c5_bin
        # Pair swap local search
        h_best, c5_ps = pair_swap_refine(h_best, n, max_rounds=40)
        if c5_ps < c5_best:
            c5_best = c5_ps
        # BFGS polish on worst lag
        corr_b = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst_b = int(np.argmax(corr_b))
        try:
            h_new = optimize_lag(h_best, n, worst_b, bounds, maxiter=200)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 5: Multi-n exploration (15% budget)
    p5_end = t0 + budget_s * 0.90
    n_candidates = [n, max(280, n - 20), n + 20, max(260, n - 40)]
    for n_alt in n_candidates:
        if elapsed() > p5_end:
            break
        if n_alt == n or n_alt < 100:
            continue
        target_alt = n_alt / 2.0
        x_old = np.arange(len(h_best)) * (2.0 / len(h_best))
        x_new = np.linspace(0.0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = fix_constraint(h_alt, target_alt)
        c5_alt = compute_c5(h_alt)
        bounds_alt = [(0.0, 1.0)] * n_alt
        for T in [500, 3000, 10000]:
            if elapsed() > p5_end:
                break
            try:
                h_new = optimize_smoothmax(h_alt, n_alt, T, bounds_alt, maxiter=250)
                h_new = fix_constraint(h_new, target_alt)
                c5_new = compute_c5(h_new)
                if c5_new < c5_alt:
                    c5_alt = c5_new
                    h_alt = h_new
            except Exception:
                pass
        h_alt, c5_ps_a = pair_swap_refine(h_alt, n_alt, max_rounds=20)
        if c5_ps_a < c5_alt:
            c5_alt = c5_ps_a
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt
            target = n / 2.0
            bounds = [(0.0, 1.0)] * n

    # Phase 6: Final tight polish (5% budget)
    p6_end = t0 + budget_s - 2.0
    target = len(h_best) / 2.0
    n = len(h_best)
    bounds = [(0.0, 1.0)] * n
    for _ in range(25):
        if elapsed() > p6_end:
            break
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst_f = int(np.argmax(corr_f))
        try:
            h_new = optimize_lag(h_best, n, worst_f, bounds, maxiter=600)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass
        # Tiny subgradient
        h_best, c5_sg = minmax_subgrad(h_best, n, 5, 0.001)
        if c5_sg < c5_best:
            c5_best = c5_sg

    c5_final = min(compute_c5(h_best), c5_best)
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_constraint(h_best, len(h_best) / 2.0)
    c5_final = min(compute_c5(h_best), c5_final)
    return (h_best, c5_final, len(h_best))
