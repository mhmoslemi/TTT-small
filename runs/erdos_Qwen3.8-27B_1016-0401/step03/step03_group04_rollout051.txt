import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = (j >= shift)
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        m2 = (j + shift) < n
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        m2 = (j >= s)
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def smooth_obj_grad(x, n, T, K, t0, tlim):
    corr = np.correlate(x, 1.0 - x, mode="full")
    nc = len(corr)
    if K < nc:
        topk = np.argpartition(corr, -K)[-K:]
        c_sel = corr[topk]
        mc = np.max(c_sel)
        sh = T * (c_sel - mc)
        e = np.exp(sh)
        ep = e / e.sum()
        val = (mc + np.log(e.sum()) / T) * (2.0 / n)
        center = n - 1
        grad = np.zeros(n)
        for ii in range(K):

            if w < 1e-14:

            grad += w * shift_grad(x, n, topk[ii] - center)
        return val, grad * (2.0 / n)

        sh = T * (corr - mc)
        e = np.exp(sh)

        val = (mc + np.log(e.sum()) / T) * (2.0 / n)

    best_h = h.copy()

        for j in order:

            # a_s for shift s: (1-h[j-s])*I[j-s valid] - h[j+s]*I[j+s valid]

            a = np.zeros(2 * n - 1)

                coeff = 0.0
                js = j - valid_s
                jsp = j + valid_s
                if 0 <= js < n:
                    coeff += 1.0 - h[js]
                if 0 <= jsp < n:
                    coeff -= h[jsp]
                a[valid_s + n - 1] = coeff

            lo, hi = 0.0, 1.0
            for _ in range(25):

                m2 = hi - (hi - lo) / 3.0
                if np.max(a * m1 + b) <= np.max(a * m2 + b):
                    hi = m2
                else:
                    lo = m1
            v_new = (lo + hi) / 2.0
            h[j] = v_new
            # Update corr incrementally: corr += (v_new - v_old) * a
            corr += (v_new - v_old) * a
        
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

def shift_grad(h, n, shift):
    j = np.arange(n)

    if shift >= 0:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(y - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m = j >= shift
        g[m] += 1.0 - h[j[m] - shift]
        m2 = (j + shift) < n
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m = (j + s) < n
        g[m] += 1.0 - h[j[m] + s]
        m2 = j >= s
        g[m2] -= h[j[m2] - s]
    return g

def sox(x, n, T, K):
    corr = np.correlate(x, 1.0 - x, mode="full")
    nc = len(corr)
    kk = min(K, nc)
    idx = np.argpartition(corr, -kk)[-kk:]
    cs = corr[idx]
    mc = np.max(cs)
    sh = T * (cs - mc)
    e = np.exp(sh)
    ep = e / e.sum()
    val = (mc + np.log(e.sum()) / T) * (2.0 / n)
    ctr = n - 1
    g = np.zeros(n)
    for i in range(len(ep)):
        if ep[i] > 1e-14:
            g += ep[i] * shift_grad(x, n, idx[i] - ctr)
    return val, g * (2.0 / n)

_tf = [False]

def dl(h0, n, T, K, mi, t0, tl):
    _tf[0] = False
    bds = [(0.0, 1.0)] * n
    def f(x):
        if time.time() - t0 > tl:
            _tf[0] = True
        v, _ = sox(x, n, T, K)
        return v
    def jf(x):
        _, g = sox(x, n, T, K)
        return g
    try:
        r = minimize(f, h0, method='L-BFGS-B', jac=jf, bounds=bds,
                     options={'maxiter': mi, 'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 30})
        return h0 if _tf[0] else project_feasible(r.x, n)
    except Exception:
        return h0.copy()
