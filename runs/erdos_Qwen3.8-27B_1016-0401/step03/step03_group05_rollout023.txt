        K = min(8, 3 + it // 80000)
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argpartition(corr, -K)[-K:]
        top_vals = corr[top_idx]
        w = top_vals ** 10
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(K) / K
        else:
            w /= ws
        g = np.zeros(n)
        for i in range(K):
            g += w[i] * shift_grad(h, n, top_idx[i] - center)
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        alpha = 1.5 / (gn * (it + 50))
        alpha = min(alpha, 0.05)
        h_new = h - alpha * g
        h_new = project_feasible(h_new, n)
        h = h_new
        if (it + 1) % 10000 == 0:
            c5 = compute_c5(h)
            if c5 < c5_best:
                c5_best = c5
                h_best = h.copy()
    c5 = compute_c5(h)
    if c5 < c5_best:
        c5_best = c5
        h_best = h.copy()
    return h_best, c5_best

def try_resolution(h_src, n_src, n_new, t0, budget_s):
    if n_new < 60 or n_new > 1200 or time.time() - t0 > budget_s - 60:
        return h_src, n_src, compute_c5(h_src)
    x_old = np.arange(n_src) * (2.0 / n_src)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_alt = np.interp(x_new, x_old, h_src)
    h_alt = project_feasible(h_alt, n_new)
    c5_alt = compute_c5(h_alt)
    for T in [50, 200]:
        if time.time() - t0 > budget_s - 30:
            break
        h_alt = slsqp_smooth(h_alt, n_new, T, 3000, 1e-18)
        h_alt = project_feasible(h_alt, n_new)
        c5_alt = compute_c5(h_alt)
    return h_alt, n_new, c5_alt

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.full(280, 0.5)
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP with increasing T (key fix: high T = closer to true max)
    for T in [30, 100, 300, 800]:
        if time.time() - t0 > budget_s - 400:
            break
        h_new = slsqp_smooth(h_best, n, T, 5000, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Subgradient polishing
    h_sg, c5_sg = subgrad_polish(h_best, n, t0, budget_s - 350, 600000)
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg
    h_best = project_feasible(h_best, n)

    # Phase 3: High-T SLSQP fine polish
    for T in [500, 1500, 3000]:
        if time.time() - t0 > budget_s - 200:
            break
        h_new = slsqp_smooth(h_best, n, T, 8000, 1e-20)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Restart with tiny perturbations
    for attempt in range(12):
        if time.time() - t0 > budget_s - 80:
            break
        sigma = 0.003 * (1 - attempt / 20.0)
        h_pert = h_best + rng.normal(0, sigma, n)
        h_pert = project_feasible(h_pert, n)
        h_new = slsqp_smooth(h_pert, n, 500, 5000, 1e-19)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        h_new2 = slsqp_smooth(h_best, n, 1500, 6000, 1e-20)
        h_new2 = project_feasible(h_new2, n)
        c5 = compute_c5(h_new2)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new2

    # Phase 5: Try adjacent resolutions
    for dn in [-10, 10, -20, 20]:
        if time.time() - t0 > budget_s - 40:
            break
        n2 = n + dn
        h_alt, n2_actual, c5_alt = try_resolution(h_best, n, n2, t0, budget_s - 20)
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n2_actual

    # Phase 6: Final ultra-precise polish
    if time.time() - t0 < budget_s - 30:
        for T in [2000, 5000]:
            if time.time() - t0 > budget_s - 10:
                break
            h_new = slsqp_smooth(h_best, n, T, 10000, 1e-20)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Final SLSQP subgradient step
    if time.time() - t0 < budget_s - 5:
        h_final, c5_final_sg = subgrad_polish(h_best, n, t0, budget_s - 2, 100000)
        if c5_final_sg < c5_best:
            h_best = h_final
            c5_best = c5_final_sg

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
