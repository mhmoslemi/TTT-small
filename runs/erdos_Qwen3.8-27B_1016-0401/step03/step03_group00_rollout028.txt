  - g[j] for j < n-s: -= h[j+s] ✓
  - Combined: g[j] = (1-h[j-s])*[j>=s] - h[j+s]*[j<n-s] ✓

- shift_grad(h, n, -s) for s > 0: g[:n-s] += (1-h[s:]), g[s:] -= h[:n-s]

  - g[j] for j >= s: -= h[j-s] ✓
  - Combined: g[j] = (1-h[j+s])*[j<n-s] - h[j-s]*[j>=s] ✓

Now let me think about potential issues:
1. The Nesterov momentum might cause oscillations. The standard Nesterov uses beta = (it-1)/(it+2) for the extrapolation step. Let me use the standard formula.

Standard Nesterov:
y_k = x_k + (k-1)/(k+2) * (x_k - x_{k-1})
x_{k+1} = proj(y_k - alpha * grad(y_k))

Let me use this exact formulation.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(45):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    g = np.zeros(n)
    if shift == 0:
        g[:] = 1.0 - 2.0 * h
    elif shift > 0:
        g[shift:] += (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def nesterov_opt(h_init, n, T_s, T_e, K_s, K_e, t_end, alpha0):
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    h_best = h_init.copy()
    c5_best = compute_c5(h_init)
    it = 0
    while time.time() < t_end:
        it += 1
        frac = min(1.0, it / 5000.0)
        T = T_s * (T_e / T_s) ** frac
        K = max(3, int(round(K_s + (K_e - K_s) * frac)))
        beta = (it - 1) / (it + 2) if it > 1 else 0.0
        h_y = h_cur + beta * (h_cur - h_prev)
        h_y = project_feasible(h_y, n)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-K:]
        c = corr[idx]
        m = c.max()
        e = np.exp(T * (c - m))
        w = e / e.sum()
        g = np.zeros(n)
        for wi, ci in zip(w, idx):
            if wi > 1e-12:
                g += wi * shift_grad(h_y, n, ci - (n - 1))
        g *= (2.0 / n)
        alpha = alpha0 / (1.0 + 0.0002 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if it % 1200 == 0:
            c5c = compute_c5(h_cur)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h_cur.copy()
            hap = project_feasible(h_avg, n)
            c5a = compute_c5(hap)
            if c5a < c5_best:
                c5_best = c5a
                h_best = hap.copy()
    c5 = compute_c5(h_cur)
    if c5 < c5_best:
        c5_best = c5
        h_best = h_cur.copy()
    return h_best, c5_best

def perturb_fn(h, n, rng, pi):
    mode = pi % 6
    if mode == 0:
        sig = 0.001 + 0.0012 * (pi % 7)
        return project_feasible(h + rng.normal(0, sig, n), n)
    elif mode == 1:
        hp = h.copy()
        lb = int(rng.integers(0, n - 40))
        ln = int(rng.integers(8, 40))
        hp[lb:lb + ln] = 1.0 - hp[lb:lb + ln]
        return project_feasible(hp, n)
    elif mode == 2:
        s = int(rng.integers(1, 6))
        return project_feasible(np.roll(h, s), n)
    elif mode == 3:
        hp = h.copy()
        nf = int(rng.integers(3, 10))
        idx = rng.choice(n, size=nf, replace=False)
        hp[idx] = 1.0 - hp[idx]
        return project_feasible(hp, n)
    elif mode == 4:
        sig = 0.003 + 0.002 * (pi % 3)
        hp = project_feasible(h + rng.normal(0, sig, n), n)
        i = int(rng.integers(0, n))
        hp[i] = 1.0 - hp[i]
        return project_feasible(hp, n)
    else:
        k = np.ones(5) / 5.0
        hp = np.convolve(h, k, mode='same')
        return project_feasible(hp + rng.normal(0, 0.006, n), n)

def slsqp_fine(h_init, n, T, K, maxiter, t_end):
    global _GL_N, _GL_T, _GL_K, _GL_TGT
    _GL_N = n
    _GL_T = float(T)
    _GL_K = K
    _GL_TGT = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        from scipy.optimize import minimize
        res = minimize(_slsqp_obj, h_init, method='SLSQP', jac=_slsqp_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _slsqp_c, 'jac': _slsqp_cj}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if time.time() < t_end:
            return project_feasible(res.x, n)
        return h_init
    except Exception:
        return h_init

def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_GL_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_GL_T * (c - m))
    return (m + np.log(e.sum()) / _GL_T) * (2.0 / _GL_N)

def _slsqp_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_GL_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_GL_T * (c - m))
    w = e / e.sum()
    g = np.zeros(_GL_N)
    for wi, ci in zip(w, idx):
        if wi > 1e-12:
            g += wi * shift_grad(x, _GL_N, ci - (_GL_N - 1))
    return g * (2.0 / _GL_N)

def _slsqp_c(x):
    return x.sum() - _GL_TGT

def _slsqp_cj(x):
    return np.ones(_GL_N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    h0 = None
    if 'initial_h_values' in globals():
        h0 = initial_h_values
    if h0 is not None:
        h = np.array(h0, dtype=float)
    else:
        h = np.zeros(280)
        h[::2] = 1.0
    n = len(h)
    h = project_feasible(h, n)
    c5_best = compute_c5(h)

    # Phase 1: Coarse SLSQP warmup (15%)
    t1 = t0 + budget_s * 0.15
    hn = slsqp_fine(h, n, 32.0, 20, 1500, t1)
    hn = project_feasible(hn, n)
    c5 = compute_c5(hn)
    if c5 < c5_best:
        c5_best, h = c5, hn

    # Phase 2: Nesterov coarse-to-fine (25%)
    t2 = t0 + budget_s * 0.40
    hn, c5 = nesterov_opt(h, n, 32.0, 4.0, 20, 8, t2, 0.35)
    if c5 < c5_best:
        c5_best, h = c5, hn

    # Phase 3: SLSQP fine (10%)
    t3 = t0 + budget_s * 0.50
    hn = slsqp_fine(h, n, 4.0, 8, 3000, t3)
    hn = project_feasible(hn, n)
    c5 = compute_c5(hn)
    if c5 < c5_best:
        c5_best, h = c5, hn

    # Phase 4: Nesterov intermediate (15%)
    t4 = t0 + budget_s * 0.65
    hn, c5 = nesterov_opt(h, n, 8.0, 1.0, 10, 4, t4, 0.18)
    if c5 < c5_best:
        c5_best, h = c5, hn

    # Phase 5: Multi-restart exploration (20%)
    t5 = t0 + budget_s * 0.85
    n_restarts = 25
    time_per = max(5.0, (t5 - time.time()) / n_restarts)
    for pi in range(n_restarts):
        if time.time() > t5:
            break
        hp = perturb_fn(h, n, rng, pi)
        t_loc = min(time.time() + time_per, t5)
        hn, c5n = nesterov_opt(hp, n, 12.0, 1.5, 10, 4, t_loc, 0.22)
        if c5n < c5_best:

    # Phase 6: Ultra-fine SLSQP polish (8%)
    t6 = t0 + budget_s * 0.93
    for T, K in [(2.0, 5), (1.0, 4), (0.5, 3)]:
        if time.time() > t6:
            break
        hn = slsqp_fine(h, n, T, K, 4000, t6)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h = c5, hn

    # Phase 7: Final Nesterov micro-polish (7%)
    t7 = t0 + budget_s * 0.99
    hn, c5 = nesterov_opt(h, n, 1.0, 0.15, 4, 3, t7, 0.05)

    h = project_feasible(h, n)
    c5_final = min(compute_c5(h), c5_best)
    return (h, c5_final, n)
