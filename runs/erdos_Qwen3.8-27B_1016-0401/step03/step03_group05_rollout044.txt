import time

    corr = np.correlate(h, 1.0 - h, mode="full")

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(n_corr):
            w = sw[oi]
            if w < 1e-10:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return grad

    res = minimize(obj, h, method='SLSQP', jac=jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def slsqp_topk(h, n, K, maxiter, ftol):
    dx = 2.0 / n
    center = n - 1
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        top_idx = np.argpartition(corr, -K)[-K:]
        top_vals = corr[top_idx]
        w = (top_vals / max(mc, 1e-15)) ** 10
        ws = w.sum()

            w = np.ones(K) / K
        else:
            w /= ws
        grad = np.zeros(n)
        for i in range(K):
            grad += w[i] * shift_grad(x, n, top_idx[i] - center)
        return grad

    res = minimize(obj, h, method='SLSQP', jac=jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid

def shift_grad(h, n, shift):
    dx = 2.0 / n
    grad = np.zeros(n)
    if shift >= 0:
        m = n - shift
        if m > 0:
            grad[:m] += (1.0 - h[shift:shift + m])
            grad[shift:] -= h[:m]
    else:
        s = -shift
        m = n - s
        if m > 0:
            grad[s:] += (1.0 - h[:m])
            grad[:m] -= h[s:]
    return grad * dx
