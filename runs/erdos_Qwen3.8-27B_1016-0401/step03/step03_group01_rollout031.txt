def bit_flip_search(h, n, t_end, max_rounds=500):
    """Local search by flipping pairs of bits to reduce max correlation."""
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    corr_best = np.correlate(h_best, 1.0 - h_best, mode="full") * (2.0/n)
    ones = np.where(h_best > 0.5)[0]
    zeros = np.where(h_best < 0.5)[0]
    
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        # Focus on the top few lags
        top5 = np.argsort(corr_best)[-5:]
        # Try random pairs of (one, zero) flips
        n_tries = min(len(ones) * len(zeros), 20000)
        rng_local = np.random.default_rng(rnd + 42)
        for t in range(n_tries):
            i = ones[rng_local.integers(len(ones))]
            j = zeros[rng_local.integers(len(zeros))]
            if i == j:
                continue
            # Flip
            h_test = h_best.copy()
            h_test[i] = 1.0
            h_test[j] = 0.0
            corr_test = np.correlate(h_test, 1.0 - h_test, mode="full") * (2.0/n)
            c5_test = corr_test.max()
            if c5_test < c5_best - 1e-12:
                c5_best = c5_test
                h_best = h_test
                corr_best = corr_test
                ones = np.where(h_best > 0.5)[0]
                zeros = np.where(h_best < 0.5)[0]
                improved = True
                break
        if not improved:
            break
    return h_best, c5_best

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def ls_obj_grad(h, n, T, topk=None):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
    else:
        idx = np.arange(len(corr))
    ck = corr[idx]
    mc = np.max(ck)
    sh = (ck - mc) * T
    e = np.exp(sh)
    es = e.sum()
    w = e / es
    obj = (mc + np.log(es) / T) * dx
    center = n - 1
    g = np.zeros(n)
    for oi in range(len(idx)):
        wi = w[oi]
        if wi < 1e-12:
            continue
        shift = idx[oi] - center
        if shift == 0:
            g[:] += wi * (1.0 - 2.0 * h)
        elif shift > 0:
            g[shift:] += wi * (1.0 - h[:n - shift])
            g[:n - shift] -= wi * h[shift:]
        else:
            s = -shift
            g[:n - s] += wi * (1.0 - h[s:])
            g[s:] -= wi * h[:n - s]
    g *= dx
    return obj, g

def slsqp_solve(h0, n, T, maxiter, ftol, topk=None):
    def obj(x):
        return ls_obj_grad(x, n, T, topk)[0]
    def jac(x):
        return ls_obj_grad(x, n, T, topk)[1]
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h0, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0.copy()

def bit_flip_optimize(h_init, n, t_end, max_rounds=200):
    h_best = h_init.copy()
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    corr_best = np.correlate(h_best, 1.0 - h_best, mode="full") * (2.0 / n)
    rng = np.random.default_rng(123)
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        ones = np.where(h_best > 0.45)[0]
        zeros = np.where(h_best < 0.55)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_tries = min(len(ones) * len(zeros), 3000)
        improved = False
        for t in range(n_tries):
            if time.time() > t_end:
                break
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            if i == j:
                continue
            h_best[i] = 1.0
            h_best[j] = 0.0
            c5 = compute_c5(h_best)
            if c5 < c5_best - 1e-12:
                c5_best = c5
                corr_best = np.correlate(h_best, 1.0 - h_best, mode="full") * (2.0 / n)
                improved = True
            else:
                h_best[i] = min(1.0, h_best[i]) if h_init[i] > 0.5 else max(0.0, 1.0 - h_best[i])
                # revert

                h_best[i] = 0.0 if h_init[i] < 0.5 else 1.0
                h_best[j] = 1.0 if h_init[j] > 0.5 else 0.0
        if not improved:
            break
    return h_best, c5_best

def smooth_bit_search(h_init, n, t_end):

    h_best = h_init.copy()
    c5_best = compute_c5(h_best)
    rng = np.random.default_rng(77)

            j = zeros[rng.integers(len(zeros))]
            if i == j:

            h_old_i = h_best[i]
            h_old_j = h_best[j]
            h_best[i] = 1.0
            h_best[j] = 0.0

                improved = True
            else:
                h_best[i] = h_old_i
                h_best[j] = h_old_j
        if not improved:

    return h_best, c5_best

def diverse_start(n, rng, ptype):
    if ptype == 0:
        h = np.zeros(n)
        h[::2] = 1.0
    elif ptype == 1:
        period = rng.integers(n // 4, n // 2 + 1)
        base = np.zeros(period)
        base[:period // 2] = 1.0
        h = np.tile(base, n // period + 1)[:n]
    elif ptype == 2:
        h = (rng.random(n) < 0.5).astype(float)
    elif ptype == 3:
        h = np.zeros(n)
        block = max(3, n // 20)
        for s in range(0, n, 2 * block):
            h[s:s + block] = 1.0
    else:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * rng.integers(2, 8) * np.arange(n) / n + rng.uniform(0, 6.28))
        h = np.clip(h, 0, 1)
    return project_feasible(h, n)

        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new.copy()

    # Phase 5: Final fine polish
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_new = slsqp_solve(h_best, n, 0.1, 600, 1e-19, 1)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new.copy()
        # One more at even lower T
        h_new = slsqp_solve(h_best, n, 0.05, 400, 1e-19, 1)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new.copy()

    # Final verification
    h_best = project_feasible(h_best, n)
    assert np.all(h_best >= -1e-10) and np.all(h_best <= 1 + 1e-10)
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
