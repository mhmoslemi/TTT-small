    dx = 2.0 / n

        return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def corr_and_grad(h, n, T, topk=None):
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    nk = len(corr)
    if topk is not None and topk < nk:
        idx = np.argpartition(corr, -topk)[-topk:]
    else:
        idx = np.arange(nk)
    c = corr[idx]
    mc = np.max(c)
    sh = T * (c - mc) - np.max(T * (c - mc))
    e = np.exp(sh)
    sw = e / e.sum()
    val = (mc + np.log(e.sum()) / T) * dx
    center = n - 1
    grad = np.zeros(n)
    for i in range(len(idx)):
        if sw[i] < 1e-14:
            continue
        grad += sw[i] * shift_grad(h, n, idx[i] - center)
    return val, grad * dx

def do_lbfgsb(h, n, T, topk, maxiter, tol, t0, tlim):
    def obj(x):
        v, _ = corr_and_grad(x, n, T, topk)
        return v
    def jac(x):
        _, g = corr_and_grad(x, n, T, topk)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jac,
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol * 0.1,
                               'maxls': 50, 'maxfun': maxiter * 5})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def nesterov_exact(h_init, n, t0, tlim, max_iters=200000):
    dx = 2.0 / n
    h_curr = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        beta = it / (it + 2.0)
        y = h_curr + beta * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode="full")
        mc = np.max(corr)
        mask = corr > mc * 0.995
        idxs = np.where(mask)[0]
        center = n - 1
        g = np.zeros(n)
        for ix in idxs:
            g += shift_grad(y, n, ix - center)
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        gap = max(mc * dx - 0.375, 1e-7)
        alpha = min(2.0 * gap / (gn * gn + 1e-12), 0.4 / (1.0 + 0.008 * it))
        h_new = project_feasible(y - alpha * g, n)
        h_prev = h_curr.copy()
        h_curr = h_new

            c5_now = compute_c5(h_curr)
            if c5_now < c5_best:
                c5_best = c5_now
                h_best = h_curr.copy()
    return h_best, c5_best

def binary_local_search(h, n, t0, tlim, max_rounds=5):
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        one_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[one_idx])
        for ri in order[:n_ones - target]:
            h_bin[one_idx[ri]] = 0.0
    elif n_ones < target:
        zero_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(h[zero_idx])[::-1]
        for ai in order[:target - n_ones]:
            h_bin[zero_idx[ai]] = 1.0
    c5_cur = compute_c5(h_bin)
    one_pos = np.where(h_bin == 1.0)[0]
    zero_pos = np.where(h_bin == 0.0)[0]
    for rnd in range(max_rounds):
        if time.time() - t0 > tlim:
            break
        improved = False
        np.random.shuffle(one_pos)
