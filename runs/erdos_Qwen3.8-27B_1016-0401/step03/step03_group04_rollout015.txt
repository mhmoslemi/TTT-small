import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 2.0
    hi = float(y.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def sm_val_grad(h, n, T, K):
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    if K < len(corr):
        topk = np.argpartition(corr, -K)[-K:]
        c_sel = corr[topk]
    else:
        topk = np.arange(len(corr))
        c_sel = corr
    mc = np.max(c_sel)
    sh = T * (c_sel - mc)
    e = np.exp(sh)
    E = e.sum()
    val = (mc + np.log(E) / T) * (2.0 / n)
    sw = e / E
    grad = np.zeros(n)
    for ii in range(len(sw)):
        w = sw[ii]
        if w < 1e-14:
            continue
        grad += w * shift_grad(h, n, topk[ii] - center)
    grad *= (2.0 / n)
    return val, grad

def _lb_timeout = [False]

def _lb_obj(x, n, T, K, t0, tlim):
    if time.time() - t0 > tlim:
        _lb_timeout[0] = True
    v, _ = sm_val_grad(x, n, T, K)
    return v

def _lb_grad(x, n, T, K, t0, tlim):
    if time.time() - t0 > tlim:
        _lb_timeout[0] = True
    _, g = sm_val_grad(x, n, T, K)
    return g

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):
    _lb_timeout[0] = False
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lb_obj, h.copy(), method='L-BFGS-B', jac=_lb_grad,
                       args=(n, T, K, t0, tlim), bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol * 0.1,
                                'maxls': 40, 'maxfun': maxiter * 4})
        if _lb_timeout[0]:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def do_slsqp(h, n, T, K, maxiter, tol, t0, tlim):
    def obj(x):
        v, _ = sm_val_grad(x, n, T, K)
        return v
    def jac(x):
        _, g = sm_val_grad(x, n, T, K)
        return g
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    _lb_timeout[0] = False
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h.copy(), method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if _lb_timeout[0]:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_polish(h_init, n, t0, tlim, max_iters=200000):
    dx = 2.0 / n
    h_curr = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    y = h_init.copy()
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        beta = it / (it + 2.0)
        y = h_curr + beta * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = shift_grad(y, n, shift) * dx
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        alpha = 0.15 / (1.0 + 0.01 * it)
        h_new = y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_curr.copy()
        h_curr = h_new
        if (it + 1) % 5000 == 0:
            c5_now = compute_c5(h_curr)
            if c5_now < c5_best:
                c5_best = c5_now
                h_best = h_curr.copy()
    c5_now = compute_c5(h_curr)
    if c5_now < c5_best:
        c5_best = c5_now
        h_best = h_curr.copy()
    return h_best, c5_best

def coordinate_refine(h, n, t0, tlim):
    dx = 2.0 / n
    c5 = compute_c5(h)
    deltas = [0.1, -0.1, 0.02, -0.02, 0.005, -0.005, 0.001, -0.001]

def fib_initial(n):
    tau = (np.sqrt(5.0) - 1.0) / 2.0
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i * tau) % 1.0 < 0.5 else 0.0
    return project_feasible(h, n)

def cos_initial(n, freqs, amps, phases):
    x = np.arange(n, dtype=float)
    h = 0.5 * np.ones(n)
    for f, a, p in zip(freqs, amps, phases):
        h += a * np.cos(2.0 * np.pi * f * x / n + p)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)
