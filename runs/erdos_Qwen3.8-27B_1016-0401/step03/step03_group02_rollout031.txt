import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target, tol=1e-12):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < tol:
            break
        # Distribute uniformly, then clip
        step = d / len(h)
        h += step
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        # Adjust one element
        if diff > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def corr_grad(x, o, n):
    """Gradient of correlation at lag o with respect to x."""
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * x
    elif o > 0:
        g[o:] += 1.0 - x[:n-o]
        g[:n-o] += -x[o:]
    else:
        p = -o
        g[:n-p] += 1.0 - x[p:]
        g[p:] += -x[:n-p]
    return g

def sm_obj(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * (2.0 / n)

def sm_grad(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    # Numerical stability
    shifted -= np.max(shifted)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)

        w = sw[center + o]
        if w < 1e-14:

        g = corr_grad(x, o, n)

    return grad * dx

def slsqp_sm(h, T, bounds, maxiter, ftol, target):
    def obj(x):
        return sm_obj(x, T)
    def jac(x):
        return sm_grad(x, T)
    def eq_fun(x):
        return x.sum() - target
    def eq_jac(x):
        return np.ones(len(x))
    cons = [dict(type='eq', fun=eq_fun, jac=eq_jac)]
    res = minimize(obj, h, jac=jac, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return np.clip(res.x, 0.0, 1.0)

    else:
        h_best = np.zeros(340)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    _TARGET = target
    h_best = fix_sum(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Quick warm-start SLSQP
    p1 = t0 + budget_s * 0.08
    corr = np.correlate(h_best, 1-h_best, mode="full")
    worst = int(np.argmax(corr))
    h_try = slsqp_lag(h_best, worst, bounds, 600, 1e-15)
    h_try = fix_sum(h_try, target)
    c_try = compute_c5(h_try)
    if c_try < c5_best:
        c5_best = c_try
        h_best = h_try

    # Phase 2: Softmax SLSQP with increasing temperature
    p2 = t0 + budget_s * 0.35
    for T in [500, 2000, 10000, 50000, 200000, 500000]:
        if time.time() - t0 > p2:
            break
        try:
            h_try = slsqp_sm(h_best, T, bounds, 4000, 1e-19)
            h_try = fix_sum(h_try, target)
            c_try = compute_c5(h_try)
            if c_try < c5_best - 1e-16:
                c5_best = c_try
                h_best = h_try
        except Exception:
            pass

    # Phase 3: Nesterov subgradient
    p3 = t0 + budget_s * 0.43
    h_try, c_try = nesterov_subgrad(h_best, target, 100, 0.0008)
    if c_try < c5_best:
        c5_best = c_try
        h_best = h_try

    # Phase 4: SLSQP cycling on worst lag
    p4 = t0 + budget_s * 0.63
    stall = 0
    for _ in range(50):
        if time.time() - t0 > p4 or stall > 5:
            break
        corr_b = np.correlate(h_best, 1-h_best, mode="full")
        worst_b = int(np.argmax(corr_b))
        h_try = slsqp_lag(h_best, worst_b, bounds, 800, 1e-18)
        h_try = fix_sum(h_try, target)
        c_try = compute_c5(h_try)
        if c_try < c5_best - 1e-16:
            c5_best = c_try
            h_best = h_try
            stall = 0
        else:
            stall += 1

    # Phase 5: Coordinate descent + binary push
    p5 = t0 + budget_s * 0.78
    for _ in range(8):
        if time.time() - t0 > p5:
            break
        h_best, c_cd = coord_descent(h_best, target, max_sweeps=2)
        if c_cd < c5_best:
            c5_best = c_cd
        for th in [0.05, 0.02, 0.01]:
            h_best, c_bp = binary_push(h_best, target, th)
            if c_bp < c5_best:
                c5_best = c_bp

    # Phase 6: Final high-T softmax + SLSQP polish
    p6 = t0 + budget_s - 10.0
    for T in [100000, 400000, 1000000]:
        if time.time() - t0 > p6:
            break
        try:
            h_try = slsqp_sm(h_best, T, bounds, 6000, 1e-20)
            h_try = fix_sum(h_try, target)
            c_try = compute_c5(h_try)
            if c_try < c5_best:
                c5_best = c_try
                h_best = h_try
        except Exception:
            pass

    # Final coordinate descent polish
    h_best, c_cd = coord_descent(h_best, target, max_sweeps=4)
    if c_cd < c5_best:
        c5_best = c_cd

    h_best, c_bp = binary_push(h_best, target, 0.01)
    if c_bp < c5_best:
        c5_best = c_bp

    c5_final = min(compute_c5(h_best), c5_best)
    h_best = fix_sum(h_best, target)
    return (h_best, c5_final, len(h_best))
