def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            count = mask.sum()
            if count == 0: break
            h[mask] += diff / count
        else:
            mask = h > 0.0
            count = mask.sum()
            if count == 0: break
            h[mask] += diff / count

def to_binary(h, n):

    idx = np.argsort(h)[::-1]

    hb[idx[:target]] = 1.0
    return hb

def sa_binary(h, n, t0_temp, t1_temp, n_iter, rng, c5_thresh=None):
    """Simulated annealing on binary swaps."""
    dx = 2.0 / n
    one_idx = np.where(h == 1.0)[0]
    zero_idx = np.where(h == 0.0)[0]
    cur_c5 = float(np.max(np.correlate(h, 1-h, mode="full")) * dx)
    best_h = h.copy()
    best_c5 = cur_c5
    
    for it in range(n_iter):
        T = t0_temp * (t1_temp / t0_temp) ** (it / n_iter)
        pi = rng.integers(len(one_idx))
        pj = rng.integers(len(zero_idx))
        i = one_idx[pi]
        j = zero_idx[pj]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = float(np.max(np.correlate(h, 1-h, mode="full")) * dx)
        
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-18)):
            cur_c5 = new_c5
            one_idx[pi] = j
            zero_idx[pj] = i
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

def slsqp_sm(h, T, n, maxiter, ftol):
    """SLSQP with soft-max objective."""
    target = n // 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0-x, mode="full")
        mc = np.max(corr)
        shifted = T * (corr - mc)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * (2.0/n)
    
    def grad(x):
        corr = np.correlate(x, 1.0-x, mode="full")
        mc = np.max(corr)
        shifted = T * (corr - mc)
        e = np.exp(shifted)
        sw = e / np.sum(e)
        center = n - 1
        dx = 2.0/n
        g = np.zeros(n)
        for o in range(-(n-1), n):
            w = sw[center + o]
            if w < 1e-14: continue
            if o == 0:
                g += w * (1.0 - 2.0*x)
            elif o > 0:
                g[o:] += w * (1.0 - x[:n-o])
                g[:n-o] += -w * x[o:]
            else:
                p = -o
                g[:n-p] += w * (1.0 - x[p:])
                g[p:] += -w * x[:n-p]
        return g * dx
    
    def cons_fun(x):
        return x.sum() - target
    def cons_jac(x):
        return np.ones(n)
    
    res = minimize(obj, h, jac=grad, method='SLSQP',
                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':cons_fun,'jac':cons_jac}],
                   options={'maxiter':maxiter, 'ftol':ftol})
    return np.clip(res.x, 0.0, 1.0)

The full correlation array has 2n-1 entries, indexed 0 to 2n-2, where entry m corresponds to lag m-(n-1).

Δcorr[ℓ] = contribution from h[i] change + contribution from h[j] change

Contribution from h[i] changing 1→0:
- h[i] appears as "first factor" when i is in range and i+ℓ is valid:
  - Before: 1 * (1-h[i+ℓ]) = 1-h[i+ℓ] (if 0 ≤ i+ℓ ≤ n-1)
  - After: 0
  - Delta: -(1-h[i+ℓ]) if 0 ≤ i+ℓ ≤ n-1, else 0
- h[i] appears as "second factor" (i.e., as h[i+ℓ] where base is i-ℓ):
  - Before: h[i-ℓ] * (1-1) = 0 (if 0 ≤ i-ℓ ≤ n-1)
  - After: h[i-ℓ] * (1-0) = h[i-ℓ]
  - Delta: +h[i-ℓ] if 0 ≤ i-ℓ ≤ n-1, else 0

Contribution from h[j] changing 0→1:
- h[j] as "first factor":
  - Before: 0 * (1-h[j+ℓ]) = 0
  - After: 1 * (1-h[j+ℓ]) = 1-h[j+ℓ]
  - Delta: +(1-h[j+ℓ]) if 0 ≤ j+ℓ ≤ n-1, else 0
- h[j] as "second factor" (base is j-ℓ):
  - Before: h[j-ℓ] * (1-0) = h[j-ℓ]
  - After: h[j-ℓ] * (1-1) = 0
  - Delta: -h[j-ℓ] if 0 ≤ j-ℓ ≤ n-1, else 0

So:
Δcorr[ℓ] = -(1-h[i+ℓ])·[0≤i+ℓ<n] + h[i-ℓ]·[0≤i-ℓ<n] + (1-h[j+ℓ])·[0≤j+ℓ<n] - h[j-ℓ]·[0≤j-ℓ<n]

Now, the indicator [0≤i+ℓ<n] means ℓ ∈ [-i, n-1-i].
The indicator [0≤i-ℓ<n] means ℓ ∈ [i-(n-1), i] = [i-n+1, i].
Similarly for j.

For a given (i,j), the non-zero Δcorr[ℓ] is the sum of up to 4 terms, each being a simple array lookup with an indicator.

```python
def compute_swap_delta(h, i, j, n):
    # ℓ ranges from -(n-1) to (n-1)
    lags = np.arange(-(n-1), n)  # length 2n-1
    
    delta = np.zeros(2*n-1)
    
    # Term 1: -(1-h[i+ℓ]) where 0 ≤ i+ℓ < n
    idx1 = i + lags
    mask1 = (idx1 >= 0) & (idx1 < n)
    delta[mask1] -= (1.0 - h[idx1[mask1]])
    
    # Term 2: +h[i-ℓ] where 0 ≤ i-ℓ < n
    idx2 = i - lags
    mask2 = (idx2 >= 0) & (idx2 < n)
    delta[mask2] += h[idx2[mask2]]
    
    # Term 3: +(1-h[j+ℓ]) where 0 ≤ j+ℓ < n
    idx3 = j + lags
    mask3 = (idx3 >= 0) & (idx3 < n)
    delta[mask3] += (1.0 - h[idx3[mask3]])
    
    # Term 4: -h[j-ℓ] where 0 ≤ j-ℓ < n
    idx4 = j - lags
    mask4 = (idx4 >= 0) & (idx4 < n)
    delta[mask4] -= h[idx4[mask4]]
    
    return delta
```

def compute_corr(h):

        else:

            cnt = mask.sum()
            if cnt == 0:

    return np.clip(h, 0.0, 1.0)

    tgt = n // 2
    idx = np.argsort(h)[::-1]

def swap_delta(h, i, j, n):
    lags = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    idx1 = i + lags
    mk1 = (idx1 >= 0) & (idx1 < n)
    delta[mk1] -= (1.0 - h[idx1[mk1]])
    idx2 = i - lags
    mk2 = (idx2 >= 0) & (idx2 < n)
    delta[mk2] += h[idx2[mk2]]
    idx3 = j + lags
    mk3 = (idx3 >= 0) & (idx3 < n)
    delta[mk3] += (1.0 - h[idx3[mk3]])

def hill_climb_binary(h, n, n_rounds, n_tries, rng, t0, time_limit):

    best_h = h.copy()
    best_c5 = cur_c5

            j = zero_idx[pj]

        h[j] = 1.0

        zero_idx[pj] = i

def sm_obj(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = float(np.max(corr))
    shifted = T * (corr - mc)
    shifted -= shifted.max()
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * (2.0 / n)

def sm_grad(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = float(np.max(corr))
    shifted = T * (corr - mc)
    shifted -= shifted.max()
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx_v = 2.0 / n
    g = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            g += w * (1.0 - 2.0 * x)
        elif o > 0:
            g[o:] += w * (1.0 - x[:n - o])
            g[:n - o] += -w * x[o:]
        else:
            p = -o
            g[:n - p] += w * (1.0 - x[p:])
            g[p:] += -w * x[:n - p]
    return g * dx_v

    return x.sum() - tgt

    return np.ones(n)

            h_alt = fix_sum(h_alt, tgt_a)
            c5_alt = compute_c5(h_alt)
        except Exception:
            pass
        h_alt, c5_cd_a = coord_descent(h_alt, tgt_a, 2)
        c5_alt = min(c5_alt, c5_cd_a)
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt
            target = n / 2.0

    # Phase 6: Final polish (5%)
    p6_end = t0 + budget_s - 3.0
    n = len(h_best)
    target = n / 2.0
    for _ in range(10):
        if time.time() > p6_end:
            break
        h_best, c5_cd = coord_descent(h_best, target, 2)
        if c5_cd < c5_best:
            c5_best = c5_cd
        try:
            h_new = slsqp_sm(h_best, 50000, n, 2000, 1e-19)
            h_new = fix_sum(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    c5_final = min(compute_c5(h_best), c5_best)
    h_best = fix_sum(h_best, len(h_best) / 2.0)
    return (h_best, c5_final, len(h_best))
