def penalty_obj(h, t, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    excess = np.maximum(0, corr - t)
    return t + mu * np.sum(excess**2)

def penalty_grad(h, t, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    excess = np.maximum(0, corr - t)
    # grad_wrt_h = mu * sum_k excess[k] * d(corr[k])/dh
    grad_h = np.zeros(n)
    center = n - 1
    for k in range(2*n-1):
        if excess[k] <= 0:
            continue
        w = mu * excess[k]
        o = k - center  # lag
        if o == 0:
            grad_h += w * (1.0 - 2.0 * h)

            grad_h[:n-o] += -w * h[o:]
        else:

            grad_h[:n-p] += w * (1.0 - h[p:])
            grad_h[p:] += -w * h[:n-p]
    grad_h *= dx
    grad_t = 1.0 - mu * np.sum(excess)
    return np.concatenate([grad_h, [grad_t]])

    # h_vec is the full vector (just h, not augmented)
    n = len(h_vec)
    dx = 2.0 / n
    corr = np.correlate(h_vec, 1.0 - h_vec, mode="full") * dx
    # Soft max
    mc = np.max(corr)
    shifted = T * (corr - mc)
    log_sum = mc + np.log(np.sum(np.exp(shifted))) / T

    s = h_vec.sum() - target

            grad[p:] += -w * h[:n-p]

    return grad

        h_alt = fix_constraint(h_alt, tgt_a)
        c5_alt = compute_c5(h_alt)
        for T in [500, 5000]:
            if elapsed() > p4_end: break
            try:
                h_new = do_lbfgs(sm_obj, sm_grad, h_alt, (T, lam, tgt_a), 1500)
                h_new = fix_constraint(h_new, tgt_a)
                c5_new = compute_c5(h_new)
                if c5_new < c5_alt: c5_alt = c5_new; h_alt = h_new
            except Exception: pass
        h_alt, c5_ca = coord_descent(h_alt, tgt_a, 2)
        if c5_ca < c5_alt: c5_alt = c5_ca
        h_alt, c5_pa = paired_cd(h_alt, tgt_a, 1)
        if c5_pa < c5_alt: c5_alt = c5_pa
        if c5_alt < c5_best:
            c5_best = c5_alt; h_best = h_alt; n = n_alt; target = n / 2.0

    # Phase 5: Final polish (remainder)
    p5_end = t0 + budget_s - 3.0
    target = len(h_best) / 2.0
    n = len(h_best)
    for _ in range(25):
        if elapsed() > p5_end: break
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        wf = int(np.argmax(corr_f))
        try:
            h_new = do_lbfgs(lag_obj, lag_grad, h_best, (wf, lam, target), 2000)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best: c5_best = c5_new; h_best = h_new
        except Exception: pass
        h_best, c5_cf = coord_descent(h_best, target, 1)
        if c5_cf < c5_best: c5_best = c5_cf
    h_best, c5_pf = paired_cd(h_best, target, 1)
    if c5_pf < c5_best: c5_best = c5_pf
    h_best, c5_bf = binary_push(h_best, [0.02, 0.05, 0.1])
    if c5_bf < c5_best: c5_best = c5_bf

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
