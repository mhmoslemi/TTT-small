import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    g = np.zeros(n)
    if shift >= 0:
        v1 = np.arange(0, n - shift)
        g[v1] += (1.0 - h[v1 + shift])
        v2 = np.arange(shift, n)
        g[v2] -= h[v2 - shift]
    else:
        s = -shift
        v1 = np.arange(s, n)
        g[v1] += (1.0 - h[v1 - s])
        v2 = np.arange(0, n - s)
        g[v2] -= h[v2 + s]
    return g

_N = 0
_TGT = 0.0
_T = 1.0
_K = 20

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-(_K):]
    c = corr[idx]
    m = c.max()
    e = np.exp(_T * (c - m))
    return (m + np.log(e.sum()) / _T) * (2.0 / _N)

def _jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-(_K):]
    c = corr[idx]
    m = c.max()
    e = np.exp(_T * (c - m))
    w = e / e.sum()
    grad = np.zeros(_N)
    for wi, ci in zip(w, idx):
        if wi > 1e-12:
            grad += wi * shift_grad(x, _N, ci - (_N - 1))
    return grad * (2.0 / _N)

def _eqf(x):
    return x.sum() - _TGT

def _eqj(x):
    return np.ones(_N)

def slsqp(h, n, T, top_k, maxiter, ftol=1e-15):
    global _N, _TGT, _T, _K
    _N = n
    _TGT = n / 2.0
    _T = float(T)
    _K = min(top_k, 2 * n - 1)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, method='SLSQP', jac=_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eqf, 'jac': _eqj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def nesterov_softmax(h_init, n, t_end, top_k=15, T_soft=4.0):
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5a = compute_c5(h_init)
    ha = h_init.copy()
    for it in range(800000):
        if time.time() > t_end:
            break
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        m = c.max()
        e = np.exp(T_soft * (c - m))
        w = e / e.sum()
        g = np.zeros(n)
        for wi, ci in zip(w, idx):
            if wi > 1e-10:
                g += wi * shift_grad(h_y, n, ci - (n - 1))
        g *= dx
        alpha = 0.8 / (1.0 + 0.004 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 3000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < c5a:
                c5a, ha = c1, h_cur.copy()
            c2 = compute_c5(project_feasible(h_avg, n))
            if c2 < c5a:
                c5a, ha = c2, project_feasible(h_avg, n).copy()
    return ha, c5a

def make_perturb(h_best, n, rng, pi):
    mode = pi % 5
    if mode == 0:
        sig = 0.001 + 0.0015 * (pi // 5) % 5
        return project_feasible(h_best + rng.normal(0, sig, n), n)
    elif mode == 1:
        h_p = h_best.copy()
        lb = rng.integers(0, n - 25)
        ln = rng.integers(4, 25)
        h_p[lb:lb + ln] = 1.0 - h_p[lb:lb + ln]
        return project_feasible(h_p, n)
    elif mode == 2:
        h_p = h_best.copy()
        perm = rng.permutation(n)
        swap_cnt = rng.integers(2, 6)
        for _ in range(swap_cnt):
            i, j = perm[_], perm[_ + 1] if _ + 1 < n else perm[0]
            h_p[[i, j]] = h_p[[j, i]]
        return project_feasible(h_p, n)
    elif mode == 3:
        h_p = h_best.copy()
        b = n // 5
        s = rng.integers(1, b // 2)
        h_p[:b] = np.roll(h_best[:b], s)
        h_p[b:2*b] = np.roll(h_best[b:2*b], -s)
        return project_feasible(h_p, n)
    else:
        h_p = h_best.copy()
        ctr = rng.integers(5, n - 5)
        w = rng.integers(4, 12)
        amp = 0.005 + 0.004 * (pi // 5) % 3
        ks = np.arange(-w, w)
        h_p[ctr - w:ctr + w] += amp * np.cos(np.pi * ks / (2.0 * w))
        return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n0 = 280
        h_best = np.zeros(n0)
        h_best[::2] = 1.0

    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Coarse-to-fine SLSQP
    t1 = t0 + budget_s * 0.22
    schedule1 = [(64, 30, 1500), (32, 25, 2000), (16, 20, 2500),
                 (8, 15, 3000), (4, 12, 4000), (2, 10, 5000)]
    for T, tk, mi in schedule1:
        if time.time() > t1:
            break
        hn = slsqp(h_best, n, T, tk, mi, 1e-14)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 2: Nesterov with softmax
    t2 = t0 + budget_s * 0.42
    hs, c5s = nesterov_softmax(h_best, n, t2, top_k=12, T_soft=5.0)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs

    # Phase 3: Fine SLSQP (multiple passes)
    t3 = t0 + budget_s * 0.65
    schedule3 = [(4, 12, 3500), (2, 10, 4500), (1, 8, 5500),
                 (0.5, 6, 7000), (0.3, 5, 8000)]
    for T, tk, mi in schedule3:
        if time.time() > t3:
            break
        for _ in range(2):
            if time.time() > t3:
                break
            hn = slsqp(h_best, n, T, tk, mi, 1e-15)
            hn = project_feasible(hn, n)
            c5 = compute_c5(hn)
            if c5 < c5_best:
                c5_best, h_best = c5, hn
            else:
                break

    # Phase 4: Perturbation + local search
    t4 = t0 + budget_s * 0.88
    for pi in range(120):
        if time.time() > t4:
            break
        hp = make_perturb(h_best, n, rng, pi)
        hn = slsqp(hp, n, 2, 10, 2000, 1e-14)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn
            # Do a quick follow-up
            if time.time() < t4:
                hn2 = slsqp(hn, n, 1, 6, 3000, 1e-15)
                hn2 = project_feasible(hn2, n)
                c52 = compute_c5(hn2)
                if c52 < c5_best:
                    c5_best, h_best = c52, hn2

    # Phase 5: Ultra-fine final polish
    if time.time() < budget_s - 10:
        for T, tk, mi in [(0.5, 5, 6000), (0.3, 4, 8000), (0.2, 3, 10000)]:
            if time.time() > budget_s - 5:
                break
            hn = slsqp(h_best, n, T, tk, mi, 1e-16)
            hn = project_feasible(hn, n)
            c5 = compute_c5(hn)
            if c5 < c5_best:
                c5_best, h_best = c5, hn

    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    c5_best = min(c5_best, c5_final)
    return (h_best, c5_best, n)
