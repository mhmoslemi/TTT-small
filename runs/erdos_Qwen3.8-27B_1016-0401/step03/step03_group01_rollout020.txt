import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def topk_obj_grad(h, n, T, k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = topk.max()
    sh = T * (topk - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    obj = (mc + np.log(e.sum()) / T) * dx
    grad = np.zeros(n)
    center = n - 1
    j = np.arange(n)
    for oi in range(mk):
        w = sw[oi]
        if w < 1e-11:
            continue
        shift = int(idx[oi]) - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += w * (1.0 - h[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= w * h[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += w * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * h[j[m2] - s2]
    grad *= dx
    return obj, grad

def slsqp_opt(h, n, T, k, maxiter, ftol):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:

                       method='SLSQP', jac=lambda x: topk_obj_grad(x, n, T, k)[1],
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        return res.x
    except Exception:
        return h.copy()

def sharpness_push(h, n, t_end):

    c5_best = compute_c5(h)

    while improved_any and time.time() < t_end:
        improved_any = False
        for i in range(n):
            if time.time() > t_end:
                break
            for new_val in [0.0, 1.0]:
                if abs(h_best[i] - new_val) < 1e-6:
                    continue
                h_try = h_best.copy()
                h_try[i] = new_val
                h_try = project_feasible(h_try, n)
                c5 = compute_c5(h_try)
                if c5 < c5_best - 1e-12:
                    c5_best = c5
                    h_best = h_try
                    improved_any = True
                    break
    return h_best, c5_best
