from scipy.optimize import minimize

def compute_c5(h):

    dx = 2.0 / n

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0

        s = np.clip(y - mid, 0.0, 1.0).sum()

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():

def smooth_obj_grad(h, n, T, K):

    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n

    else:

    center = n - 1

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def smooth_obj_grad(h, n, T, K):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    nc = len(corr)
    if K < nc:
        topk = np.argpartition(corr, -K)[-K:]
        c_sel = corr[topk]
    else:
        topk = np.arange(nc)
        c_sel = corr
    mc = c_sel.max()
    sh = T * (c_sel - mc)
    mx = sh.max()
    e = np.exp(sh - mx)
    val = (mc + (mx + np.log(e.sum())) / T) * dx
    sw = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    for ii in range(len(sw)):
        if sw[ii] > 1e-15:
            grad += sw[ii] * shift_grad(h, n, topk[ii] - center)
    return val, grad * dx

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):
    _done = [False]
    def _obj(x):
        if time.time() - t0 > tlim:
            _done[0] = True
        return smooth_obj_grad(x, n, T, K)[0]
    def _jac(x):
        if time.time() - t0 > tlim:
            _done[0] = True
        return smooth_obj_grad(x, n, T, K)[1]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, method='L-BFGS-B', jac=_jac, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol*0.05,
                                'maxls': 20, 'maxfun': maxiter*2})
        return h if _done[0] else project_feasible(res.x, n)
    except Exception:
        return h.copy()

def do_slsqp(h, n, T, K, maxiter, tol, t0, tlim):
    _done = [False]
    def _obj(x):
        if time.time() - t0 > tlim:
            _done[0] = True
        return smooth_obj_grad(x, n, T, K)[0]
    def _jac(x):
        if time.time() - t0 > tlim:
            _done[0] = True
        return smooth_obj_grad(x, n, T, K)[1]
    def _cfun(x):
        return x.sum() - n / 2.0
    def _cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        res = minimize(_obj, h, method='SLSQP', jac=_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        return h if _done[0] else project_feasible(res.x, n)
    except Exception:
        return h.copy()

def nesterov_subgrad(h_init, n, t0, tlim, max_iters=200000, K=5):
    dx = 2.0 / n
    h_curr = h_init.copy()

    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        beta = it / (it + 2.5)
        y = h_curr + beta * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode="full")
        topk = np.argpartition(corr, -K)[-K:]
        g = np.zeros(n)
        for t in topk:
            g += shift_grad(y, n, t - (n - 1))
        g *= dx / K
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        f_y = compute_c5(y)

    return h_best, c5_best

    dx = 2.0 / n
    c5_best = compute_c5(h)
    order = np.arange(n)

            if time.time() - t0 > tlim:

            old = h[i]
            best_v = old

            for dv in [-0.02, -0.01, -0.005, 0.005, 0.01, 0.02]:
                cv = old + dv
                if cv < 0 or cv > 1:

                h[i] = cv
                # Maintain sum constraint by adjusting a random other element
                diff = old - cv
                j = int((i + n // 2) % n)

                if abs(residual) > 1e-10:

            if h[i] != old:

                if c5_now < c5_best:
                    c5_best = c5_now
