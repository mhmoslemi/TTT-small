import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_TGT = 0.0
_G_T = 1.0
_G_K = 10
_G_DX = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_G_T * (c - m))
    return (m + np.log(e.sum()) / _G_T) * _G_DX

def _slsqp_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_G_T * (c - m))
    w = e / e.sum()
    grad = np.zeros(_G_N)
    for wi, ci in zip(w, idx):
        if wi > 1e-10:
            grad += wi * shift_grad(x, _G_N, ci - (_G_N - 1))
    return grad * _G_DX

def _slsqp_cfun(x):
    return x.sum() - _G_TGT

def _slsqp_cjac(x):
    return np.ones(_G_N)

def slsqp_run(h, n, T, top_k, maxiter, ftol):
    global _G_N, _G_TGT, _G_T, _G_K, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_T = float(T)
    _G_K = min(top_k, 2 * n - 1)
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP', jac=_slsqp_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _slsqp_cfun, 'jac': _slsqp_cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def peak_chase(h_init, n, t_end, top_k=3):
    h_cur = h_init.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    alpha = 0.05
    for it in range(200000):
        if time.time() > t_end:

        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        idx = np.argsort(corr)[-top_k:]

            g += shift_grad(h_cur, n, ci - (n - 1))
        g *= dx / top_k
        alpha = 0.05 / (1.0 + 0.0005 * it)
        h_try = project_feasible(h_cur - alpha * g, n)

    return best_h, best_c5

def binarize(h, n):
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        ones = np.where(h_bin == 1)[0]
