    m = c.max()
    e = np.exp(GT_ * (c - m))
    w = e / e.sum()
    grad = np.zeros(GN)
    for wi, ci in zip(w, idx):
        if wi > 1e-10:
            grad += wi * shift_grad(x, GN, ci - (GN - 1))
    return grad * (2.0 / GN)

def _eqf(x):
    return x.sum() - GTGT

def _eqj(x):
    return np.ones(GN)

def _pobj(x):
    return _sobj(x) + 1e5 * (x.sum() - GTGT) ** 2

def _pjac(x):
    return _sjac(x) + 2e5 * (x.sum() - GTGT)

def _setp(n, T, tk):
    global GN, GTGT, GT_, GK
    GN = n
    GTGT = n / 2.0
    GT_ = T
    GK = min(tk, 2 * n - 1)

def _opt_slb(h, T, tk, mi, gtol):
    _setp(len(h), T, tk)
    bounds = [(0.0, 1.0)] * GN
    try:
        res = minimize(_pobj, h, method='L-BFGS-B', jac=_pjac,
                       bounds=bounds, options={'maxiter': mi, 'gtol': gtol, 'ftol': 1e-18})
        return project_feasible(res.x, GN)
    except Exception:
        return project_feasible(h, GN)

def _opt_slsqp(h, T, tk, mi, ftol):
    _setp(len(h), T, tk)
    bounds = [(0.0, 1.0)] * GN
    try:
        res = minimize(_sobj, h, method='SLSQP', jac=_sjac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eqf, 'jac': _eqj}],
                       options={'maxiter': mi, 'ftol': ftol})
        return project_feasible(res.x, GN)
    except Exception:
        return project_feasible(h, GN)

def _nesterov(h_init, n, t_end, rng, top_k=8):
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5a = compute_c5(h_init)
    ha = h_init.copy()
    it = 0
    while time.time() < t_end:
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h_y, n, ci - (n - 1))
        g *= dx / top_k
        alpha = 0.5 / (1.0 + 0.001 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        it += 1
        if it % 5000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < c5a:
                c5a, ha = c1, h_cur.copy()
            hv = project_feasible(h_avg, n)
            c2 = compute_c5(hv)
            if c2 < c5a:
                c5a, ha = c2, hv
    return ha, c5a

def _rounding(h, n, rng, rounds=300):
    c5_best = compute_c5(h)
    h_best = h.copy()
    for _ in range(rounds):
        mask = (h_best > 0.04) & (h_best < 0.96)
        if not mask.any():
            break
        idx = np.where(mask)[0]
        i = idx[rng.integers(len(idx))]
        h_try = h_best.copy()
        h_try[i] = 0.0 if h_best[i] < 0.5 else 1.0
        h_try = project_feasible(h_try, n)
        c5 = compute_c5(h_try)
        if c5 < c5_best - 1e-14:
            c5_best = c5
            h_best = h_try
    return h_best, c5_best

def _bit_swap(h, n, rng, t_end):
    c5_best = compute_c5(h)
    h_best = h.copy()
    for _ in range(200):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(80):
            if time.time() > t_end:
                break
            zeros = np.where(h_best < 0.15)[0]
            ones = np.where(h_best > 0.85)[0]
            if len(zeros) < 2 or len(ones) < 2:
                break
            i = zeros[rng.integers(len(zeros))]
            j = ones[rng.integers(len(ones))]
            h_try = h_best.copy()
            h_try[i], h_try[j] = h_try[j], h_try[i]
            h_try = project_feasible(h_try, n)
            c5 = compute_c5(h_try)
            if c5 < c5_best - 1e-13:
                c5_best = c5
                h_best = h_try
                improved = True
        if not improved:
            break
    return h_best, c5_best

def _perturb(h, n, rng, pi):
    mode = pi % 5
    if mode == 0:
        sig = 0.001 + 0.002 * (pi // 5) % 4
        return project_feasible(h + rng.normal(0, sig, n), n)
    elif mode == 1:
        hp = h.copy()
        lb = rng.integers(0, n - 30)
        ln = rng.integers(5, 20)
        hp[lb:lb+ln] = 1.0 - hp[lb:lb+ln]
        return project_feasible(hp, n)
    elif mode == 2:
        hp = h.copy()
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        hp[[i, j]] = [1.0 - hp[j], 1.0 - hp[i]]
        return project_feasible(hp, n)
    elif mode == 3:
        sig = 0.003 + 0.001 * (pi // 5) % 3
        return project_feasible(h + rng.normal(0, sig, n), n)
    else:
        hp = h.copy()
        k = rng.integers(1, 8)
        pos = rng.integers(0, n, k)
        hp[pos] = 1.0 - hp[pos]
        return project_feasible(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.default_rng(seed)
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        h_best = np.array(iv, dtype=float)
    else:
        n0 = 280
        h_best = np.zeros(n0)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B coarse (fast, many iterations)
    t1 = t0 + budget_s * 0.22
    for T, tk, mi, gt in [(60, 30, 4000, 1e-7), (30, 20, 5000, 1e-8),
                           (10, 15, 6000, 1e-9), (5, 10, 8000, 1e-10)]:
        if time.time() > t1:
            break
        hn = _opt_slb(h_best, T, tk, mi, gt)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 2: SLSQP mid-refinement
    t2 = t0 + budget_s * 0.42
    for T, tk, mi, ft in [(10, 15, 3000, 1e-14), (5, 12, 4000, 1e-15),
                           (2, 10, 5000, 1e-15), (1, 8, 6000, 1e-16)]:
        if time.time() > t2:
            break
        hn = _opt_slsqp(h_best, T, tk, mi, ft)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 3: Nesterov with momentum
    t3 = t0 + budget_s * 0.55
    hs, c5s = _nesterov(h_best, n, t3, rng, top_k=8)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs

    # Phase 4: Rounding to exploit discrete structure
    hr, c5r = _rounding(h_best, n, rng, rounds=400)
    if c5r < c5_best:
        c5_best, h_best = c5r, hr

    # Phase 5: Bit-swap local search
    hb, c5hb = _bit_swap(h_best, n, rng, t0 + budget_s * 0.62)
    if c5hb < c5_best:
        c5_best, h_best = c5hb, hb

    # Phase 6: Ultra-fine SLSQP with very low T
    t4 = t0 + budget_s * 0.80
    for T, tk, mi, ft in [(1, 6, 8000, 1e-16), (0.5, 5, 10000, 1e-17),
                           (0.3, 4, 12000, 1e-17), (0.15, 3, 15000, 1e-18),
                           (0.08, 3, 15000, 1e-18)]:
        if time.time() > t4:
            break
        hn = _opt_slsqp(h_best, T, tk, mi, ft)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 7: Multi-start with structured perturbations
    t5 = t_end - 35
    for pi in range(50):
        if time.time() > t5:
            break
        hp = _perturb(h_best, n, rng, pi)
        hn = _opt_slsqp(hp, 1, 8, 3000, 1e-15)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn
            # Extra polish after improvement
            if time.time() < t5:
                hn2 = _opt_slsqp(h_best, 0.3, 4, 8000, 1e-17)
                c52 = compute_c5(hn2)
                if c52 < c5_best:
                    c5_best, h_best = c52, hn2

    # Phase 8: Final ultra-high-precision polish
    if time.time() < t_end - 5:
        hn = _opt_slsqp(h_best, 0.05, 3, 20000, 1e-19)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
