The problem is:
minimize max_k f_k(h)
subject to 0 ≤ h[i] ≤ 1, ∑h[i] = n/2

At the optimum, suppose shifts k_1, ..., k_m are active (f_{k_1} = ... = f_{k_m} = C* = max).

The KKT condition (for the max of smooth functions) is that there exist weights w_1, ..., w_m ≥ 0, ∑w_i = 1, such that:
∑_{i=1}^m w_i * ∇f_{k_i}(h*) = λ * 1 + μ
where μ is the vector of bound multiplier contributions (μ[j] = -α_j if h*[j]=0, μ[j] = +β_j if h*[j]=1, μ[j]=0 otherwise, with α_j, β_j ≥ 0).

For free variables (0 < h*[j] < 1):
∑_{i=1}^m w_i * ∂f_{k_i}/∂h[j] = λ

This is a system of n_free equations in n_free + 1 unknowns (h_free and λ), plus the sum constraint gives one more equation. But we also need to find the optimal weights w_i.

This is more complex than I initially thought. For m=1 (single active shift), it simplifies to what I had. For m>1, I need to either:
1. Fix the weights and solve (but how to choose them?)
2. Use a "max" approximation (soft-max) and solve the resulting system

For simplicity, let me use m=1 (just the single worst shift) for the KKT polish. This works well when there's a clear single worst shift, which is often the case after the SLSQP has run.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

def build_gradient_mv(n, shift):
    """Build M, v such that gradient = M@h + v for given shift."""
    M = np.zeros((n, n))
    v = np.zeros(n)
    if shift >= 0:
        v[shift:] = 1.0
        jj = np.arange(shift, n)
        M[jj, jj - shift] += -1.0
        jj = np.arange(0, n - shift)
        M[jj, jj + shift] += -1.0
    else:
        sp = -shift
        v[:n - sp] = 1.0
        jj = np.arange(0, n - sp)
        M[jj, jj + sp] += -1.0
        jj = np.arange(sp, n)
        M[jj, jj - sp] += -1.0
    return M, v

def kkt_polish(h, n, top_k=1, tol=1e-9, max_iter=100):
    """KKT-based polish for the worst shift(s)."""
    h = h.copy()
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argsort(corr)[-top_k:]
        M = np.zeros((n, n))
        v = np.zeros(n)
        for si in idx:
            M_i, v_i = build_gradient_mv(n, si - (n - 1))
            M += M_i
            v += v_i
        M /= top_k
        v /= top_k
        free_mask = (h > tol) & (h < 1.0 - tol)
        n_free = int(free_mask.sum())
        if n_free == 0:
            break
        fi = np.where(free_mask)[0]
        bi = np.where(~free_mask)[0]
        h_B = h[bi]
        const = M[np.ix_(fi, bi)].dot(h_B) + v[fi]
        M_FF = M[np.ix_(fi, fi)]
        C = n / 2.0 - h[~free_mask].sum()
        n_f = len(fi)
        A_sys = np.zeros((n_f + 1, n_f + 1))
        A_sys[:n_f, :n_f] = M_FF
        A_sys[:n_f, n_f] = -1.0
        A_sys[n_f, :n_f] = 1.0
        rhs = np.zeros(n_f + 1)
        rhs[:n_f] = -const
        rhs[n_f] = C
        try:
            sol = np.linalg.solve(A_sys + 1e-12 * np.eye(n_f + 1), rhs)
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(A_sys, rhs, rcond=None)[0]
            except Exception:
                break
        h_F_new = sol[:n_f]
        h_F_clipped = np.clip(h_F_new, 0.0, 1.0)
        if np.array_equal(h_F_clipped, h_F_new):
            h[fi] = h_F_clipped
            break
        else:
            h[fi] = h_F_clipped
            h = project_feasible(h, n)
    h = project_feasible(h, n)
    return h

def _sl_obj(x, n, T, top_k):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    return (m + np.log(e.sum()) / T) * (2.0 / n)

def _sl_jac(x, n, T, top_k):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    for wi, ci in zip(w, idx):
        if wi > 1e-10:
            grad += wi * shift_grad(x, n, ci - (n - 1))
    return grad * (2.0 / n)

def slsqp(h, n, T, top_k, maxiter, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lambda x: _sl_obj(x, n, T, top_k), h,
            method='SLSQP',
            jac=lambda x: _sl_jac(x, n, T, top_k),
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                          'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def perturb(h, n, rng, it):
    mode = it % 5
    if mode == 0:
        sig = 0.003 + 0.002 * (it // 5) % 4
        return project_feasible(h + rng.normal(0, sig, n), n)
    elif mode == 1:
        hp = h.copy()
        lb = int(rng.integers(0, max(1, n - 30)))
        ln = int(rng.integers(5, 30))
        hp[lb:lb+ln] = 1.0 - hp[lb:lb+ln]
        return project_feasible(hp, n)
    elif mode == 2:
        sig = 0.001 + 0.001 * (it // 5) % 3
        return project_feasible(h + rng.normal(0, sig, n), n)
    elif mode == 3:
        hp = h.copy()
        i1 = int(rng.integers(0, n))
        i2 = int(rng.integers(0, n))
        hp[[i1, i2]] = hp[[i2, i1]]
        return project_feasible(hp, n)
    else:
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        g = shift_grad(h, n, s)
        return project_feasible(h - 0.005 * g, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n0 = 280
        h_best = np.zeros(n0)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP annealing (coarse to fine)
    p1_end = t0 + budget_s * 0.28
    for T, tk, mi in [(32, 5, 2000), (16, 5, 2500), (8, 4, 3000), (4, 4, 3500), (2, 3, 4000), (1, 3, 5000)]:
        if time.time() > p1_end:
            break
        h_new = slsqp(h_best, n, T, tk, mi)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 2: KKT polish + perturbation cycles
    p2_end = t0 + budget_s * 0.72
    stall = 0
    for it in range(500):
        if time.time() > p2_end:
            break
        # KKT on current best
        h_kkt = kkt_polish(h_best, n, top_k=1)
        c5 = compute_c5(h_kkt)
        if c5 < c5_best:
            c5_best, h_best = c5, h_kkt
            stall = 0
        else:
            stall += 1
        # Perturb and KKT
        h_pert = perturb(h_best, n, rng, it)
        h_kkt2 = kkt_polish(h_pert, n, top_k=1)
        c5_2 = compute_c5(h_kkt2)
        if c5_2 < c5_best:
            c5_best, h_best = c5_2, h_kkt2
            stall = 0
        elif stall > 40 and it % 5 == 0:
            # Bigger perturbation to escape
            h_big = perturb(h_best, n, rng, it * 3 + 7)
            h_kkt3 = kkt_polish(h_big, n, top_k=1)
            c5_3 = compute_c5(h_kkt3)
            if c5_3 < c5_best:
                c5_best, h_best = c5_3, h_kkt3
                stall = 0

    # Phase 3: Fine SLSQP
    p3_end = t0 + budget_s * 0.90
    for T, tk, mi in [(1, 3, 5000), (0.5, 3, 6000), (0.3, 2, 8000), (0.2, 2, 10000)]:
        if time.time() > p3_end:
            break
        h_new = slsqp(h_best, n, T, tk, mi)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 4: Final KKT + micro-polish
    if time.time() < t0 + budget_s - 10:
        h_final = kkt_polish(h_best, n, top_k=1)
        c5 = compute_c5(h_final)
        if c5 < c5_best:
            c5_best, h_best = c5, h_final
        h_final2 = kkt_polish(h_best, n, top_k=2)
        c5 = compute_c5(h_final2)
        if c5 < c5_best:
            c5_best, h_best = c5, h_final2

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
