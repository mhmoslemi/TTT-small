def projection_onto_box_sum(y, target_sum):
    """Project y onto {x : 0<=x_i<=1, sum(x)=target_sum}"""
    n = len(y)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target_sum) < 1e-15:
            break
        if s > target_sum:

        else:

def smooth_max_obj(h, n, T, topk_frac=1.0):
    """Compute smooth max of correlation and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Select top fraction of correlations
    k = max(1, int(len(corr) * topk_frac))
    idx = np.argpartition(corr, -k)[-k:]
    top = corr[idx]
    # Smooth max
    m = top.max()
    st = T * (top - m)
    et = np.exp(st)
    Z = et.sum()
    obj = (m + np.log(Z) / T) * dx
    # Gradient
    w = et / Z
    center = n - 1
    grad = np.zeros(n)
    j = np.arange(n)
    for oi in range(k):
        wi = w[oi]
        if wi < 1e-12:
            continue
        shift = idx[oi] - center
        if shift >= 0:

            grad[m1] += wi * (1.0 - h[j[m1] - shift])

            grad[m1] += wi * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= wi * h[j[m2] - s2]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(y, target):
    n = len(y)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def corr_gradient(h, n, shifts, weights):
    """Gradient of sum_k w_k * C(k) where C(k)=sum_i h[i]*(1-h[i+k])."""
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    j = np.arange(n)
    for si, wi in zip(shifts, weights):
        shift = si - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += wi * (1.0 - h[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= wi * h[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += wi * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= wi * h[j[m2] - s2]
    grad *= dx
    return grad

def active_set_opt(h0, n, t_end, max_active=12, inner_iters=3000):
    """Active set method: minimize top-k correlations with adaptive k."""
    target = n / 2.0
    h = h0.copy()
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    k = 1
    for phase in range(40):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argsort(corr)[-k:]
        idx = idx[idx >= 0]
        w = np.ones(len(idx)) / len(idx)
        for it in range(inner_iters):
            if time.time() > t_end:
                break
            grad = corr_gradient(h, n, idx, w)
            t = 1.0
            obj0 = corr[idx].sum()
            for _ in range(30):
                h_trial = project_box_sum(h - t * grad, target)
                c_trial = np.correlate(h_trial, 1.0 - h_trial, mode="full")
                if c_trial[idx].sum() < obj0 - 1e-16:
                    h = h_trial
                    corr = c_trial
                    break
                t *= 0.5
            else:
                h = project_box_sum(h - t * grad, target)
                corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
        k = min(k + 1, max_active)
    return best_h, best_c5

    """FISTA with smooth top-k max."""
    target = n / 2.0

    lr = lr0
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    nit = 60000
    for it in range(nit):
        if time.time() > t_end:
            break
        frac = min(1.0, (it + 1) / nit)
        T = T0 * (T1 / T0) ** frac
        corr = np.correlate(hy, 1.0 - hy, mode="full")
        mk = min(ktop, len(corr))
        idx = np.argpartition(corr, -mk)[-mk:]
        top = corr[idx]
        mc = top.max()
        st = T * (top - mc)
        et = np.exp(st)
        Z = et.sum()
        w = et / Z
        wi_list = []
        si_list = []
        for oi in range(mk):
            if w[oi] > 1e-10:
                wi_list.append(w[oi])
                si_list.append(idx[oi])
        grad = corr_gradient(hy, n, si_list, wi_list)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tm * tm)) / 2.0
        beta = (tm - 1.0) / tn
        hz = hy + beta * (hy - hz)
        hy = project_box_sum(hz - lr * grad, target)
        tm = tn
        lr *= decay
        if (it + 1) % 8000 == 0:
            c5 = compute_c5(hy)
            if c5 < best_c5:
                best_c5, best_h = c5, hy.copy()
    c5 = compute_c5(hy)
    if c5 < best_c5:
        best_c5, best_h = c5, hy.copy()
    return best_h, best_c5

def binary_local_search(h0, n, t_end, max_rounds=3000):
    """Directed binary swap search focusing on worst lag."""
    target = int(n / 2)
    h = (h0 > 0.5).astype(float)
    ones = np.where(h == 1)[0]
    if len(ones) > target:
        rm = np.random.default_rng(7).choice(ones, len(ones) - target, replace=False)
        h[rm] = 0
    elif len(ones) < target:
        zeros = np.where(h == 0)[0]
        add = np.random.default_rng(7).choice(zeros, target - len(ones), replace=False)
        h[add] = 1
    best_c5 = compute_c5(h)
    best_h = h.copy()
    rng = np.random.default_rng(99)
    center = n - 1
    stagnant = 0
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        widx = corr.argmax()
        wval = corr[widx]
        shift = widx - center
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        best_m = wval + 1.0
        best_a, best_b = -1, -1
        ntr = 400
        for _ in range(ntr):
            a = ones_idx[rng.integers(len(ones_idx))]
            b = zeros_idx[rng.integers(len(zeros_idx))]
            if a == b:
                continue
            h[a], h[b] = 0.0, 1.0
            cv = np.correlate(h, 1.0 - h, mode="full")
            m = cv.max()
            h[a], h[b] = 1.0, 0.0
            if m < best_m:
                best_m = m
                best_a, best_b = a, b
        if best_a < 0:
            stagnant += 1
            if stagnant > 200:
                break
            if stochastic_move说法 if stochastic_move(rng, h, n):
                pass
            continue
        h[best_a], h[best_b] = 0.0, 1.0
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
            stagnant = 0
        else:
            stagnant += 1
    return best_h, best_c5

def stochastic_move(rng, h, n):
    a = rng.integers(n)
    b = rng.integers(n)
    if a == b or h[a] == h[b]:
        return False
    h[a], h[b] = 1.0 - h[a], 1.0 - h[b]
    return True

def multi_start_try(n, t_end, rng, c5_ref):
    """Try several structured initializations and optimize the best."""
    candidates = []
    # Block pattern
    h1 = np.zeros(n)
    h1[::2] = 1.0
    candidates.append(h1)
    # Sinusoidal
    x = np.linspace(0, 2, n, endpoint=False)
    h2 = ((np.sin(2 * np.pi * 3 * x / 2) + 1) / 2)
    candidates.append(h2)
    # Golden ratio spacing
    gr = (1 + np.sqrt(5)) / 2
    pos = sorted((int((i * gr) % n) for i in range(n // 2)))
    h3 = np.zeros(n)
    h3[pos[:n // 2]] = 1.0
    candidates.append(h3)
    # Random
    h4 = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h4[idx] = 1.0
    candidates.append(h4)
    best_h, best_c5 = None, c5_ref
    for hc in candidates:
        hc = project_box_sum(hc.astype(float), n / 2.0)
        c5 = compute_c5(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc
    if best_h is None:
        return None, c5_ref
    h_opt, c5_opt = active_set_opt(best_h, n, t_end, max_active=8, inner_iters=1500)
    return h_opt, min(c5_opt, best_c5)

        c5_best, h_best = c5b, hb

    # Phase 4: FISTA refinement from binary
    t4 = t0 + budget_s * 0.78
