def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def nesterov_subgrad(h_init, n, t_end, ntop=10, tol=0.05):
    """Nesterov-accelerated subgradient with top-k active shifts."""
    dx = 2.0 / n
    h_y = h_init.copy()
    h_old = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    t_mom = 1.0
    lr = 0.1
    center = n - 1
    j = np.arange(n)
    
    for it in range(500000):
        if time.time() > t_end:
            break
        # Compute all correlations
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        cmax = np.max(corr)
        
        # Find active shifts (within tol of max)
        active = np.where(corr >= cmax - tol * cmax)[0]
        n_active = min(len(active), ntop)
        if n_active == 0:
            active = np.array([np.argmax(corr)])
            n_active = 1
        active = active[:n_active]
        
        # Compute subgradient (sum of gradients of active terms, normalized)
        grad = np.zeros(n)
        for ai in range(n_active):
            shift = active[ai] - center
            if shift >= 0:
                m1 = j >= shift
                grad[m1] += (1.0 - h_y[j[m1] - shift])
                m2 = (j + shift) < n
                grad[m2] -= h_y[j[m2] + shift]
            else:

                m1 = (j + s2) < n
                grad[m1] += (1.0 - h_y[j[m1] + s2])
                m2 = j >= s2
                grad[m2] -= h_y[j[m2] - s2]

        beta = (t_mom - 1.0) / t_new
        h_old = h_y.copy()
        h_y = h_y + beta * (h_y - h_old)
        h_y = h_y - lr * grad
        h_y = project_feasible(h_y, n)
        t_mom = t_new
        
        # Adaptive learning rate
        lr = lr * 0.99995 + 0.0001 * (1.0 / (1.0 + it * 0.001))
        lr = np.clip(lr, 0.001, 0.1)
        
        # Track best

            if c5 < c5_best:
                c5_best = c5
                h_best = h_y.copy()

        if it % 10000 == 0:

    return h_best, c5_best

        topk = corr[idx]

        return (mc + np.sum(np.log(1.0 + np.exp(T * (topk - mc)))) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = min(k, len(corr))
        idx = np.argpartition(corr, -m)[-m:]
        topk = corr[idx]
        mc = np.max(topk)
        e = np.exp(T * (topk - mc))
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(m):
            wt = sw[oi]
            if wt < 1e-10:
                continue
            shift = idx[oi] - center
            if shift >= 0:
                m1 = j_arr >= shift
                grad[m1] += wt * (1.0 - x[j_arr[m1] - shift])
                m2 = (j_arr + shift) < n
                grad[m2] -= wt * x[j_arr[m2] + shift]
            else:
                s2 = -shift
                m1 = (j_arr + s2) < n
                grad[m1] += wt * (1.0 - x[j_arr[m1] + s2])
                m2 = j_arr >= s2
                grad[m2] -= wt * x[j_arr[m2] - s2]
        return grad * dx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return res.x
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Nesterov subgradient top-5
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h_new, c5 = nesterov_opt(h_best, n, t0 + min(rem - 20, 250), ntop=5, lr0=0.04)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 2: Coordinate polish
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h_new = coord_polish(h_best, n, n_sweeps=50, ntop=5)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 3: Nesterov top-3 fine
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h_new, c5 = nesterov_opt(h_best, n, t0 + min(rem - 20, 200), ntop=3, lr0=0.02)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 4: Coordinate polish again
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_new = coord_polish(h_best, n, n_sweeps=60, ntop=4)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 5: SLSQP polish with very low T
    rem = budget_s - (time.time() - t0)
    if rem > 100:
        for T, k in [(2.0, 5), (1.0, 4), (0.5, 3)]:
            if time.time() - t0 > budget_s - 40:
                break
            h_new = fast_slsqp(h_best, n, T, k, 3000)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best, h_best = c5, h_new
        h_new = coord_polish(h_best, n, n_sweeps=40, ntop=5)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 6: Try n=200
    rem = budget_s - (time.time() - t0)
    if rem > 150:
        for n_try in [200, 140]:
            if time.time() - t0 > budget_s - 50:
                break
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_hi = np.interp(x_new, x_old, h_best)
            h_hi = project_feasible(h_hi, n_try)
            c5_hi = compute_c5(h_hi)
            h_tmp = h_hi.copy()
            c5_tmp = c5_hi
            # Nesterov on new resolution
            h_new2, c5_2 = nesterov_opt(h_tmp, n_try, t0 + min(rem - 30, 120), ntop=5, lr0=0.03)
            h_new2 = project_feasible(h_new2, n_try)
            c5_2 = compute_c5(h_new2)
            if c5_2 < c5_tmp:
                h_tmp, c5_tmp = h_new2, c5_2
            h_tmp = coord_polish(h_tmp, n_try, n_sweeps=40, ntop=5)
            c5_tmp = compute_c5(h_tmp)
            # SLSQP
            if time.time() - t0 < budget_s - 30:
                h_new2 = fast_slsqp(h_tmp, n_try, 1.0, 4, 2000)
                h_new2 = project_feasible(h_new2, n_try)
                c5_2 = compute_c5(h_new2)
                if c5_2 < c5_tmp:
                    h_tmp, c5_tmp = h_new2, c5_2
            if c5_tmp < c5_best:
                c5_best, h_best, n = c5_tmp, h_tmp, n_try
                h_best = coord_polish(h_best, n, n_sweeps=30, ntop=5)
                c5_best = compute_c5(h_best)

    # Phase 7: Final coordinate polish + Nesterov
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_new, c5 = nesterov_opt(h_best, n, t0 + min(rem - 5, 60), ntop=4, lr0=0.015)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new
        h_best = coord_polish(h_best, n, n_sweeps=30, ntop=5)
        c5_best = min(c5_best, compute_c5(h_best))

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
