def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def _shift_grad(h, n, shift):
    dx = 2.0 / n
    grad = np.zeros(n)
    j = np.arange(n)
    if shift >= 0:
        m1 = j >= shift
        if np.any(m1):
            grad[m1] += (1.0 - h[j[m1] - shift])
        m2 = (j + shift) < n
        if np.any(m2):
            grad[m2] -= h[j[m2] + shift]
    else:
        s2 = -shift
        m1 = (j + s2) < n
        if np.any(m1):
            grad[m1] += (1.0 - h[j[m1] + s2])
        m2 = j >= s2
        if np.any(m2):
            grad[m2] -= h[j[m2] - s2]
    return grad * dx

        vals = corr[idx]
        mc = vals.max()
        sh = T * (vals - mc)
        e = np.exp(sh)

        h_new = slsqp_opt(h_best, n, T, k, min(mi, int(rem * 0.6)), 1e-19)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Local refinement (subgradient focused on top lags)
    rem = budget_s - (time.time() - t0)
    if rem > 100:
        t_loc = t0 + rem * 0.4
        h_best, c5_best = local_refine(h_best, n, t_loc, rng, topk=7)

    # Phase 6: Try shorter NAG with different topk
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        t_nag2 = t0 + rem * 0.5
        h2, c5_2 = nag_opt(h_best, n, t_nag2, topk=3, T_sm=6.0)
        if c5_2 < c5_best:
            c5_best = c5_2
            h_best = h2

    # Phase 7: Final SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_new = slsqp_opt(h_best, n, 0.5, 3, 3000, 1e-19)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 8: Try n=560 if time allows
    rem = budget_s - (time.time() - t0)
    if rem > 200:
        n_hi = n * 2
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = project_feasible(h_hi, n_hi)
        c5_hi = compute_c5(h_hi)
        if c5_hi < c5_best + 0.002:
            for T, k, mi in [(16, 20, 3000), (4, 8, 3500), (1, 4, 3500)]:
                rem2 = budget_s - (time.time() - t0)
                if rem2 < 60:
                    break
                h_new = slsqp_opt(h_hi, n_hi, T, k, min(mi, int(rem2 * 0.6)), 1e-18)
                h_new = project_feasible(h_new, n_hi)
                c5 = compute_c5(h_new)
                if c5 < c5_hi:
                    c5_hi = c5
                    h_hi = h_new
            if c5_hi < c5_best:
                c5_best = c5_hi
                h_best = h_hi
                n = n_hi

    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    return (h_best, min(c5_final, c5_best), n)
