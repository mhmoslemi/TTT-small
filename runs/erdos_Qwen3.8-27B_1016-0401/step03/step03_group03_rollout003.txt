        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)

            g += sw[oi] * shift_grad(x, n, oi - center)

        return x.sum() - n / 2.0

        return np.ones(n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = -3.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def _smooth_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    return (mc + np.log(np.exp(sh).sum()) / T) * dx

def _smooth_jac(x, n, T):
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    g = np.zeros(n)
    center_idx = n - 1
    for oi in range(len(sw)):
        if sw[oi] < 1e-10:
            continue
        g += sw[oi] * shift_grad(x, n, oi - center)
    return g * dx

        return _smooth_jac(x, n, T)

        return np.ones(n)

                       bounds=bounds, constraints=cons,

        return h.copy()

    h_sg = h.copy()
    h_prev = h.copy()
    c5_best = compute_c5(h)
