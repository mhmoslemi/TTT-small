def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()

            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def grad_at_shift(h, shift, n):
    """Gradient of f_k(h) = sum_i h[i](1-h[i+shift]) w.r.t. h."""
    center = n - 1
    k = shift - center  # actual shift
    grad = np.zeros(n)
    j = np.arange(n)
    if k >= 0:
        # corr[center+k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # d/dh[j] = (1-h[j-k]) for j >= k (from h[j] in first factor)
        #           - h[j+k] for j < n-k (from h[j] in second factor)
        idx1 = np.arange(k, n)
        grad[idx1] += (1.0 - h[idx1 - k])
        idx2 = np.arange(0, n - k)
        grad[idx2] -= h[idx2 + k]
    else:

        grad[idx1] += (1.0 - h[idx1 + k2])
        idx2 = np.arange(k2, n)
        grad[idx2] -= h[idx2 - k2]
    return grad

def grad_at_shift(h, shift, n):

    else:

        grad[:n-k2] += (1.0 - h[k2:])
        grad[k2:] -= h[:n-k2]
    return grad

def nesterov_run(h_init, n, t_end, lr0=0.005, topk=3, restart_every=5000):

    h_y = h_init.copy()
    h_z = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    t_mom = 1.0
    lr = lr0
    it = 0
    while time.time() < t_end:
        if it % restart_every == restart_every - 1 and it > 0:
            # Soft restart: dampen momentum, increase lr slightly
            h_z = h_y.copy()
            t_mom = 1.0
            lr = lr0 * (1.0 + 0.1 * (it / restart_every))
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")

        idx_top = np.argpartition(corr, -k_use)[-k_use:]
        grad = np.zeros(n)
        for idx in idx_top:
            grad += grad_at_shift(h_y, idx, n)
        grad *= dx / k_use
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2)) / 2.0
        beta = (t_mom - 1.0) / t_new
        h_z = h_y + beta * (h_y - h_z)
        h_y = h_z - lr * grad
        h_y = project_feasible(h_y, n)
        t_mom = t_new
        lr *= 0.99998
        it += 1
        if it % 2000 == 0:

    if c5 < c5_best:

def slsqp_run(h, n, T, k, maxiter, ftol):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mk = min(k, len(corr))
        idx = np.argpartition(corr, -mk)[-mk:]
        topk = corr[idx]
        mc = np.max(topk)
        sh = T * (topk - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mk = min(k, len(corr))
        idx = np.argpartition(corr, -mk)[-mk:]
        topk = corr[idx]
        mc = np.max(topk)
        sh = T * (topk - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(len(x))
        for oi in range(mk):
            w = sw[oi]
            if w > 1e-13:
                grad += w * grad_at_shift(x, idx[oi], len(x))
        grad *= dx
        return grad
    def cf(x):
        return x.sum() - target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()
