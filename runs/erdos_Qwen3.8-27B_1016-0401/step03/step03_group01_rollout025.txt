import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def grad_lag(h, lag, n):
    dx = 2.0 / n
    center = n - 1
    shift = lag - center
    g = np.zeros(n)
    j = np.arange(n)
    if shift >= 0:
        ms = j >= shift
        mp = (j + shift) < n
        g[ms] += (1.0 - h[j[ms] + shift])
        g[mp] -= h[j[mp]]
    else:
        s = -shift
        ms = (j + s) < n
        mp = j >= s
        g[ms] += (1.0 - h[j[ms] - s])
        g[mp] -= h[j[mp] + s]
    return g * dx

def fista_sub(h_init, n, t_end, K, lr0, decay=0.99999, check_every=2000):
    h_y = h_init.copy()
    h_z = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    t_m = 1.0
    lr = lr0
    for it in range(500000):
        if time.time() > t_end:
            break
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        kk = min(K, len(corr))
        idx = np.argpartition(corr, -kk)[-kk:]
        g = np.zeros(n)
        for k in idx:
            g += grad_lag(h_y, k, n)
        g /= kk
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_m * t_m)) / 2.0
        beta = (t_m - 1.0) / t_new
        h_z = h_y + beta * (h_y - h_z)
        h_y = project_feasible(h_z - lr * g, n)
        t_m = t_new
        lr *= decay
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h_y)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_y.copy()
    return h_best, c5_best

def slsqp_run(h, n, T, k, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    j = np.arange(n)
    def obj_fn(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mk = min(k, len(corr))
        idx = np.argpartition(corr, -mk)[-mk:]
        topk = corr[idx]
        mc = np.max(topk)
        sh = T * (topk - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    def jac_fn(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mk = min(k, len(corr))
        idx = np.argpartition(corr, -mk)[-mk:]
        topk = corr[idx]
        mc = np.max(topk)
        sh = T * (topk - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        g = np.zeros(n)
        for oi in range(mk):
            w = sw[oi]
            if w < 1e-11:
                continue
            shift = idx[oi] - center
            if shift >= 0:
                ms = j >= shift
                g[ms] += w * (1.0 - x[j[ms] - shift])
                mp = (j + shift) < n
                g[mp] -= w * x[j[mp] + shift]
            else:
                s2 = -shift
                ms = (j + s2) < n
                g[ms] += w * (1.0 - x[j[ms] + s2])
                mp = j >= s2
                g[mp] -= w * x[j[mp] - s2]
        return g * dx
    def cf(x):
        return x.sum() - target
    def cj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h, method='SLSQP', jac=jac_fn, bounds=bounds,
                       constraints=[{'type':'eq','fun':cf,'jac':cj}],
                       options={'maxiter':maxiter,'ftol':ftol})
        return res.x
    except Exception:
        return h.copy()

def perturb_diverse(h, n, rng):

    ptype = rng.integers(0, 7)
    if ptype == 0:

    elif ptype == 1:

            h_p[s2:s2+b] = tmp
    elif ptype == 3:
