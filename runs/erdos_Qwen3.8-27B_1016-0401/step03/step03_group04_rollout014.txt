            if rem > 50:
                h_new = lbfgsb_polish(h_best, n, t0, t0 + rem - 10, 1.0, 20, 6000)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
