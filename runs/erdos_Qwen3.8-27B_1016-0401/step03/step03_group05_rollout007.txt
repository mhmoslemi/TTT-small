import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def slsqp_focus(h, n, K, lam_spread, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1

    def obj_func(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        top_idx = np.argpartition(corr, -K)[-K:]
        top_vals = corr[top_idx]
        m = top_vals.max()
        if lam_spread > 0 and K > 1:
            return (m + lam_spread * (m - top_vals.min())) * dx
        return m * dx

    def jac_func(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        top_idx = np.argpartition(corr, -K)[-K:]
        top_vals = corr[top_idx]
        mc = top_vals.max()
        w = (top_vals / max(mc, 1e-15)) ** 12
        wsum = w.sum()
        if wsum < 1e-15:
            w = np.ones(K) / K
        else:
            w = w / wsum
        grad = np.zeros(n)
        for i in range(K):
            grad += w[i] * shift_grad(x, n, top_idx[i] - center)
        return grad * dx

    def cons_func(x):
        return x.sum() - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_func, 'jac': cons_jac}]
    try:
        res = minimize(obj_func, h, method='SLSQP', jac=jac_func,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def slsqp_lse(h, n, T, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1

    def obj_func(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(np.clip(sh, -500, 500))
        return (mc + np.log(e.sum()) / T) * dx

    def jac_func(x):

        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(np.clip(sh, -500, 500))
        sw = e / e.sum()
        grad = np.zeros(n)
        ncorr = len(corr)
        for oi in range(ncorr):
            if sw[oi] < 1e-10:
                continue
            grad += sw[oi] * shift_grad(x, n, oi - center)
        return grad * dx

    def cons_func(x):
        return x.sum() - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_func, 'jac': cons_jac}]
    try:
        res = minimize(obj_func, h, method='SLSQP', jac=jac_func,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)

        frac = min(it / 150000.0, 1.0)

        mc = np.max(corr)

def coord_descent(h, n, rng, n_steps=5000):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    dx = 2.0 / n
    idx_order = np.random.permutation(n)
    for step in range(n_steps):
        i = idx_order[step % n]

            h_try = h_cur.copy()

            c = compute_c5(h_try)

                c5_cur = c
