            grad[:n - s2] -= w * h[s2:]
    grad *= dx
    return obj, grad

def _f(x):
    return corr_grad(x, _g_T, _g_k)[0]

def _jf(x):
    return corr_grad(x, _g_T, _g_k)[1]

def _cf(x):
    return x.sum() - _g_n / 2.0

def _cj(x):
    return np.ones(_g_n)

def do_slsqp(h0, T, k, maxiter, ftol):
    global _g_n, _g_T, _g_k
    _g_n = len(h0)
    _g_T = T
    _g_k = k
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_f, h0, method='SLSQP', jac=_jf, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0.copy()

def do_fista(h0, n, t_end, lr0=0.008):
    dx = 2.0 / n
    h_y = h0.copy()
    h_z = h0.copy()
    c5_best, _ = compute_c5(h0)
    h_best = h0.copy()
    t = 1.0
    lr = lr0
    for it in range(800000):
        if time.time() > t_end:
            break
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        k3 = 3
        idx3 = np.argpartition(corr, -k3)[-k3:]
        center = n - 1
        grad = np.zeros(n)
        for oi in idx3:
            s = int(oi) - center
            if s == 0:
                grad += (1.0 - 2.0 * h_y)
            elif s > 0:
                grad[:n - s] += (1.0 - h_y[s:])
                grad[s:] -= h_y[:n - s]
            else:
                s2 = -s
                grad[s2:] += (1.0 - h_y[:n - s2])
                grad[:n - s2] -= h_y[s2:]
        grad *= dx / k3
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        h_z = h_y + beta * (h_y - h_z)
        h_y = proj(h_z - lr * grad, n)
        t = t_new
        lr *= 0.99997
        if (it + 1) % 5000 == 0:
            c5, _ = compute_c5(h_y)
            if c5 < c5_best:
                c5_best, h_best = c5, h_y.copy()
    c5, _ = compute_c5(h_y)
    if c5 < c5_best:
        c5_best, h_best = c5, h_y.copy()
    return h_best, c5_best

def pair_descent(h, n, t_end, rng):
    dx = 2.0 / n
    c5_cur, _ = compute_c5(h)
    best = h.copy()
    best_c5 = c5_cur
    amp = 0.08
    for it in range(3000000):
        if time.time() > t_end:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        d = rng.uniform(-amp, amp)
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5_try = float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)
        if c5_try < c5_cur:
            c5_cur = c5_try
            if c5_cur < best_c5:
                best_c5, best = c5_cur, h.copy()
        else:
            h[i], h[j] = h[i] - d, h[j] + d
        if (it + 1) % 50000 == 0:
            amp *= 0.7
    return best, best_c5

def make_periodic(n, period):
    h = np.zeros(n)
    half = period // 2
    for i in range(period):
        h[i] = 1.0 if i < half else 0.0
    tiles = n // period
    return np.tile(h[:n], tiles)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = proj(h_best, n)
    c5_best, _ = compute_c5(h_best)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Coarse-to-fine SLSQP on initial solution
    schedule = [
        (5, 20, 2500), (3, 10, 2500), (3, 5, 2500),
        (5, 3, 2000), (3, 2, 2000), (2, 1, 2000),
        (3, 0.5, 2000), (2, 0.3, 1500), (1, 0.2, 1500),
    ]
    for k_val, T_val, mi in schedule:
        if remaining() < 80:
            break
        h_new = do_slsqp(h_best, T_val, k_val, mi, 1e-18)
        h_new = proj(h_new, n)
        c5, _ = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 2: FISTA with top-3 gradient
    if remaining() > 200:
        t_fista_end = time.time() + min(remaining() * 0.3, 200)
        h_f, c5_f = do_fista(h_best, n, t_fista_end, lr0=0.006)
        h_f = proj(h_f, n)
        c5_f, _ = compute_c5(h_f)
        if c5_f < c5_best:
            c5_best, h_best = c5_f, h_f

    # Phase 3: Fine SLSQP polish
    for T_val, k_val, mi in [(1.0, 3, 3000), (0.5, 2, 3000), (0.25, 2, 2500)]:
        if remaining() < 50:
            break
        h_new = do_slsqp(h_best, T_val, k_val, mi, 1e-19)
        h_new = proj(h_new, n)
        c5, _ = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 4: Pair descent (random coordinate swaps)
    if remaining() > 60:
        t_pd_end = time.time() + min(remaining() * 0.35, 120)
        h_pd, c5_pd = pair_descent(h_best.copy(), n, t_pd_end, rng)
        h_pd = proj(h_pd, n)
        c5_pd, _ = compute_c5(h_pd)
        if c5_pd < c5_best:
            c5_best, h_best = c5_pd, h_pd

    # Phase 5: Try alternative structured starting points
    alts = []
    for p in [14, 20, 28, 35, 40]:
        if p < n and n % p == 0:
            alts.append(make_periodic(n, p))
    # Also try a thick-thin pattern
    h_tt = np.zeros(n)
    blk = n // 7
    for bi in range(7):
        start = bi * blk
        length = min(blk, n - start)
        if bi % 2 == 0:
            h_tt[start:start + length] = 1.0
    alts.append(proj(h_tt, n))

    for h_alt in alts:
        if remaining() < 120:
            break
        h_alt = proj(h_alt, n)
        h_tmp = do_slsqp(h_alt, 5, 5, 1500, 1e-16)
        h_tmp = proj(h_tmp, n)
        h_tmp = do_slsqp(h_tmp, 1, 3, 2000, 1e-18)
        h_tmp = proj(h_tmp, n)
        c5, _ = compute_c5(h_tmp)
        if c5 < c5_best:
            c5_best, h_best = c5, h_tmp

    # Phase 6: Final pair descent + SLSQP on best
    if remaining() > 40:
        t_pd2 = time.time() + min(remaining() * 0.4, 80)
        h_pd2, c5_pd2 = pair_descent(h_best.copy(), n, t_pd2, rng)
        h_pd2 = proj(h_pd2, n)
        c5_pd2, _ = compute_c5(h_pd2)
        if c5_pd2 < c5_best:
            c5_best, h_best = c5_pd2, h_pd2

    if remaining() > 30:
        h_new = do_slsqp(h_best, 0.3, 2, 2000, 1e-19)
        h_new = proj(h_new, n)
        c5, _ = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Final pair descent squeeze
    if remaining() > 20:
        h_final, c5_final = pair_descent(h_best.copy(), n, time.time() + remaining() - 5, rng)
        h_final = proj(h_final, n)
        c5_final, _ = compute_c5(h_final)
        if c5_final < c5_best:
            c5_best, h_best = c5_final, h_final

    h_best = proj(h_best, n)
    c5_final, _ = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, n)
