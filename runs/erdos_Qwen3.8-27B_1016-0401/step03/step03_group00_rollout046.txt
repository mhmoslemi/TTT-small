def project_feasible(y, n):
    target = n / 2.0

    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:

def coord_descent(h, n, t_end, top_k=30, max_cycles=200000):
    h = h.copy()

    best_h = h.copy()
    stagnant = 0
