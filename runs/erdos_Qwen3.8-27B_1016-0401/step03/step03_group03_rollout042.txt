            if len(pos) == 0 or len(neg) == 0:
                continue
            improved = False
            for _t in range(4):
                ip = pos[rng.integers(len(pos))]
                inn = neg[rng.integers(len(neg))]
                for delta in (0.02, 0.01, 0.04):
                    h_try = h_cur.copy()
                    h_try[ip] -= delta
                    h_try[inn] += delta
                    h_try = np.clip(h_try, 0.0, 1.0)
                    h_try = project_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < c5_cur - 1e-9:
                        h_cur = h_try
                        c5_cur = c5_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
    return h_cur, c5_cur

def downsample(h, new_n):
    old_n = len(h)
    old_x = np.arange(old_n)
    new_x = np.linspace(0, old_n - 1, new_n)
    return np.interp(new_x, old_x, h)

def upsample(h, new_n):
    old_n = len(h)
    old_x = np.arange(old_n)
    new_x = np.linspace(0, old_n - 1, new_n)
    return np.interp(new_x, old_x, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(280)
        h_init[::2] = 1.0
    n = len(h_init)
    h_best = project_feasible(h_init, n)
    c5_best = compute_c5(h_best)

    # Phase 0: Multi-resolution basin finding
    for ln in [80, 120, 180]:
        if time.time() - t0 > budget_s - 750:
            break
        h_lo = downsample(h_best, ln)
        h_lo = project_feasible(h_lo, ln)
        h_lo = do_slsqp(h_lo, ln, 5.0, 4000, 1e-12)
        h_lo = project_feasible(h_lo, ln)
        h_up = upsample(h_lo, n)
        h_up = project_feasible(h_up, n)
        c5_up = compute_c5(h_up)
        # Quick SLSQP on upsampled
        h_up2 = do_slsqp(h_up, n, 2.0, 3000, 1e-12)
        h_up2 = project_feasible(h_up2, n)
        c5_up2 = compute_c5(h_up2)
        if c5_up2 < c5_best:
            c5_best = c5_up2
            h_best = h_up2

    # Phase 1: Temperature-scheduled SLSQP
    temps = [8.0, 4.0, 2.5, 1.5, 1.0, 0.7, 0.45, 0.3, 0.2]
    for T in temps:
        if time.time() - t0 > budget_s - 520:
            break
        h_new = do_slsqp(h_best, n, T, 12000, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Nesterov with extended run
    h_sg, c5_sg = nesterov_opt(h_best, n, t0, budget_s - 420, K=10, max_iters=250000)
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg
    else:
        h_best = project_feasible(h_best, n)

    # Phase 3: Fine SLSQP
    for T in [0.5, 0.3, 0.2, 0.1]:
        if time.time() - t0 > budget_s - 220:
            break
        h_new = do_slsqp(h_best, n, T, 15000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Coordinate + SLSQP interleave
    if time.time() - t0 < budget_s - 100:
        h_lp, c5_lp = coord_polish(h_best, n, rng, steps=5000)
        if c5_lp < c5_best:
            c5_best = c5_lp
            h_best = h_lp
        h_new = do_slsqp(h_best, n, 0.25, 10000, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Perturbation restarts
    for pi in range(20):
        if time.time() - t0 > budget_s - 25:
            break
        sigma = 0.001 + 0.003 * (pi % 5)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_new = do_slsqp(h_p, n, 0.8, 6000, 1e-13)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Final polish
    if time.time() - t0 < budget_s - 8:
        h_new = do_slsqp(h_best, n, 0.15, 12000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
