    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2, 0.0, 1.0)

    h = h.copy()
    xgrid = np.linspace(0.0, 1.0, grid_pts)

        for i in range(n):

            has_quad = False
            for j in range(2*n-1):

    T = _G['T']
    K = _G['K']
    n = _G['n']
    idx = np.argsort(corr)[-K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    return (m + np.log(e.sum()) / T) * (2.0 / n)

    T = _G['T']
    K = _G['K']
    n = _G['n']
    idx = np.argsort(corr)[-K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, ci in zip(w, idx):

            g = np.zeros(n)

                m2 = np.arange(n)[np.arange(n)[m1] + s < n]
                g[m1] += (1.0 - x[np.arange(n)[m1] - s])
                g[m2] -= x[np.arange(n)[m2] + s]
            else:

                valid1 = np.arange(n) + ss < n
                valid2 = np.arange(n) >= ss
                g[valid1] += (1.0 - x[np.arange(n)[valid1] + ss])
                g[valid2] -= x[np.arange(n)[valid2] - ss]

    return x.sum() - _G['n'] / 2.0

    return np.ones(_G['n'])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def coord_descent(h, n, sweeps=10, grid_pts=80, t_end=None):
    h = h.copy()
    xgrid = np.linspace(0.0, 1.0, grid_pts)
    quad = xgrid - xgrid * xgrid
    s_all = np.arange(-(n - 1), n)
    for sweep in range(sweeps):
        if t_end is not None and time.time() > t_end:
            break
        for i in range(n):
            idx_a = i + s_all
            idx_b = i - s_all
            valid_s0 = s_all != 0
            valid_a = (idx_a >= 0) & (idx_a < n) & valid_s0
            valid_b = (idx_b >= 0) & (idx_b < n) & valid_s0
            a_c = np.zeros(2 * n - 1)
            b_c = np.zeros(2 * n - 1)
            a_c[valid_a] = 1.0 - h[idx_a[valid_a]]
            a_c[valid_b] -= h[idx_b[valid_b]]
            b_c[valid_b] = h[idx_b[valid_b]]
            active = (a_c != 0.0) | (b_c != 0.0)
            if not active.any():
                h[i] = 0.0
                continue
            idx = np.where(active)[0]
            X = xgrid[:, np.newaxis]
            vals = a_c[idx][np.newaxis, :] * X + b_c[idx][np.newaxis, :]
            vals = np.maximum(vals, quad[:, np.newaxis])
            maxvals = vals.max(axis=1)
            h[i] = xgrid[np.argmin(maxvals)]
    return h

_G = {}

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    T, K, n = _G['T'], _G['K'], _G['n']
    idx = np.argsort(corr)[-K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    return (m + np.log(e.sum()) / T) * (2.0 / n)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    T, K, n = _G['T'], _G['K'], _G['n']
    idx = np.argsort(corr)[-K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    dx = 2.0 / n
    p = np.arange(n)
    grad = np.zeros(n)
    for wi, ci in zip(w, idx):
        if wi > 1e-10:
            s = ci - (n - 1)
            g = np.zeros(n)
            va = (p + s >= 0) & (p + s < n)
            g[va] += (1.0 - x[p[va] + s])
            vb = (p - s >= 0) & (p - s < n)
            g[vb] -= x[p[vb] - s]
            grad += wi * g
    return grad * dx

def _g_cf(x):
    return x.sum() - _G['n'] / 2.0

def _g_cj(x):
    return np.ones(_G['n'])

def slsqp_polish(h, n, T=2, K=8, maxiter=3000, ftol=1e-14):
    _G['T'], _G['K'], _G['n'] = T, K, n
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _g_cf, 'jac': _g_cj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def perturb(h, n, rng, pi):
    mode = pi % 5
    if mode == 0:
        sig = 0.01 + 0.005 * (pi // 5) % 4
        return project_feasible(h + rng.normal(0, sig, n), n)
    elif mode == 1:
        hp = h.copy()
        lb = int(rng.integers(0, max(1, n - 40)))
        ln = int(rng.integers(10, 40))
        hp[lb:lb + ln] = 1.0 - hp[lb:lb + ln]
        return project_feasible(hp, n)
    elif mode == 2:
        hp = h.copy()
        hp[hp > 0.55] = 1.0
        hp[hp < 0.45] = 0.0
        hp += rng.normal(0, 0.04, n)
        return project_feasible(hp, n)
    elif mode == 3:
        hp = h.copy()
        a1 = int(rng.integers(0, max(1, n - 30)))
        a2 = int(rng.integers(0, max(1, n - 30)))
        la = int(rng.integers(10, 25))
        hp[a1:a1 + la] = h[a2:a2 + la]
        return project_feasible(hp, n)
    else:
        hp = h.copy()
        mask = rng.random(n) < 0.08
        hp[mask] = 1.0 - hp[mask]
        return project_feasible(hp, n)

def optimize_track(h_init, n, t_start, t_end, rng, label=""):
    h = project_feasible(h_init.copy(), n)
    c5 = compute_c5(h)
    # Coordinate descent
    h = coord_descent(h, n, sweeps=15, grid_pts=100, t_end=min(t_end, t_start + (t_end - t_start) * 0.3))
    c5 = min(c5, compute_c5(h))
    # SLSQP
    for T, K, mi in [(6, 10, 2000), (2, 8, 3000), (1, 5, 4000)]:
        if time.time() > min(t_end, t_start + (t_end - t_start) * 0.7):
            break
        hn = slsqp_polish(h, n, T, K, mi, 1e-14)
        c5n = compute_c5(hn)
        if c5n < c5:
            c5, h = c5n, hn
        hn2 = coord_descent(hn, n, sweeps=5, grid_pts=80, t_end=t_end)
        c5n = compute_c5(hn2)
        if c5n < c5:
            c5, h = c5n, hn2
    # Perturbations
    for pi in range(25):
        if time.time() > t_end - 5:
            break
        hp = perturb(h, n, rng, pi)
        hp = coord_descent(hp, n, sweeps=6, grid_pts=80, t_end=t_end - 5)
        c5p = compute_c5(hp)
        if c5p < c5:
            c5, h = c5p, hp
            if time.time() < t_end - 10:
                hp2 = slsqp_polish(h, n, 1.5, 5, 2000, 1e-14)
                c5p2 = compute_c5(hp2)

        c5 = min(c5, compute_c5(hf))
        h = hf if compute_c5(hf) < compute_c5(h) else h
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
