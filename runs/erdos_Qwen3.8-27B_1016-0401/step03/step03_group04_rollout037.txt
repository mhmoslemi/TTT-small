def project_feasible(y, n):

    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:

        else:

def shift_grad(h, n, shift):
    # Gradient of sum_i h[i]*(1-h[i+shift]) w.r.t. h
    grad = np.zeros(n)
    j = np.arange(n)
    if shift >= 0:
        m = j <= n - 1 - shift
        if m.any():
            grad[m] += (1.0 - h[j[m] + shift])
        m2 = j >= shift
        if m2.any():
            grad[m2] -= h[j[m2] - shift]

            grad[m] += (1.0 - h[j[m] + s])
        m2 = j >= s

    return grad

    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def smooth_obj_grad(x, n, T, K):
    corr = np.correlate(x, 1.0 - x, mode="full")
    center = n - 1
    L = len(corr)
    if K < L:
        topk = np.argpartition(corr, -K)[-K:]
        csel = corr[topk]
    else:
        topk = np.arange(L)
        csel = corr
    mc = float(csel.max())
    sh = T * (csel - mc)
    e = np.exp(sh)
    es = e.sum()
    val = (mc + np.log(es) / T) * (2.0 / n)
    sw = e / es
    grad = np.zeros(n)
    for ii in range(len(sw)):
        if sw[ii] > 1e-14:
            grad += sw[ii] * shift_grad(x, n, int(topk[ii]) - center)
    return val, grad * (2.0 / n)

def lbfgsb_opt(h, n, T, K, maxiter, tol, t0, tlim):
    bounds = [(0.0, 1.0)] * n
    def f(x):
        return smooth_obj_grad(x, n, T, K)[0]
    def g(x):
        return smooth_obj_grad(x, n, T, K)[1]
    try:
        res = minimize(f, h, method='L-BFGS-B', jac=g, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol * 0.5,
                                'maxls': 60, 'maxfun': maxiter * 8})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_opt(h, n, T, K, maxiter, tol, t0, tlim):
    bounds = [(0.0, 1.0)] * n
    def f(x):
        return smooth_obj_grad(x, n, T, K)[0]
    def g(x):
        return smooth_obj_grad(x, n, T, K)[1]
    def c_eq(x):
        return x.sum() - n / 2.0
    def c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': c_eq, 'jac': c_jac}]
    try:
        res = minimize(f, h, method='SLSQP', jac=g, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': tol, 'iprint': -1})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def nesterov_phase(h_init, n, t0, tlim, max_iters, f_star):
    dx = 2.0 / n
    h_curr = h_init.copy()
    h_prev = h_init.copy()
    h_best = h_init.copy()
    c5_best = compute_c5(h_init)
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        beta = it / (it + 2.5)
        y = np.clip(h_curr + beta * (h_curr - h_prev), 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = shift_grad(y, n, shift) * dx
        gn2 = float(np.dot(g, g))
        f_y = compute_c5(y)
        gap = max(f_y - f_star, 5e-8)
        alpha = min(2.0 * gap / (gn2 + 1e-12), 0.25 / (1.0 + 0.001 * it))
        h_new = project_feasible(y - alpha * g, n)
        h_prev = h_curr.copy()
        h_curr = h_new
        if (it + 1) % 1500 == 0:
            c = compute_c5(h_curr)
            if c < c5_best:
                c5_best = c
                h_best = h_curr.copy()
    return h_best, c5_best

def worst_shift_local(h, n, t0, tlim, nsteps=5000):
    """Local search targeting only the single worst shift."""
    dx = 2.0 / n
    h = h.copy()
    c5_best = compute_c5(h)
    for it in range(nsteps):
        if time.time() - t0 > tlim:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        shift = int(np.argmax(corr)) - (n - 1)
        g = shift_grad(h, n, shift) * dx
        gn2 = float(np.dot(g, g))
        if gn2 < 1e-20:
            break
        # Adaptive step: try decreasing step sizes
        best_step = 0.0
        best_c5 = c5_best
        for frac in [1.0, 0.5, 0.25, 0.125]:
            alpha = frac * 2.0 * (best_c5 - 0.37) / (gn2 + 1e-12)
            alpha = max(alpha, 1e-6)
            h_try = project_feasible(h - alpha * g, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_step = alpha
        if best_step > 0:
            h = project_feasible(h - best_step * g, n)
            c5_best = best_c5
        else:
            if it > 200:
                break
    return h, c5_best

def gen_start(n, rng, style):
    x = np.linspace(0, 2.0, n, endpoint=False)
    if style == 0:
        return np.full(n, 0.5)
    elif style == 1:
        f = rng.uniform(2, 8)
        return project_feasible(0.5 + 0.35 * np.sin(np.pi * f * x + rng.uniform(0, 6.28)), n)
    elif style == 2:
        f1, f2 = rng.uniform(2, 5), rng.uniform(5, 12)
        return project_feasible(0.5 + 0.22*np.sin(np.pi*f1*x) + 0.12*np.sin(np.pi*f2*x + 1.0), n)
    elif style == 3:
        h = (rng.random(n) < 0.5).astype(float)
        k = max(3, n // 50)
        h = np.convolve(h, np.ones(k)/k, mode='same')
        return project_feasible(h, n)
    else:
        return project_feasible(0.5 + rng.normal(0, 0.2, n), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except (NameError, UnboundLocalError):
        h_best = gen_start(280, rng, 1)

    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Coarse-to-fine L-BFGS-B
    for T, K, mi in [(128, 80, 2000), (32, 60, 3000), (16, 40, 4000), (8, 30, 5000), (4, 20, 6000), (2, 10, 8000)]:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        h_new = lbfgsb_opt(h_best, n, T, K, mi, 1e-14, t0, budget_s - 20)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Nesterov subgradient
    rem = budget_s - (time.time() - t0)
    if rem > 180:
        h_sg, c5_sg = nesterov_phase(h_best, n, t0, t0 + rem - 100, 500000, 0.378)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase 3: SLSQP with equality constraint
    for T, K, mi in [(8, 30, 4000), (4, 20, 5000), (2, 10, 6000), (1, 5, 8000)]:
        rem = budget_s - (time.time() - t0)
        if rem < 60:
            break
        h_new = slsqp_opt(h_best, n, T, K, mi, 1e-15, t0, budget_s - 15)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Worst-shift local search
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        h_w, c5_w = worst_shift_local(h_best, n, t0, t0 + rem - 20, 8000)
        if c5_w < c5_best:
            c5_best = c5_w
            h_best = h_w

    # Phase 5: Try different resolutions
    for n_try in [200, 240, 264, 300, 320, 340]:
        rem = budget_s - (time.time() - t0)
        if rem < 90:
            break
        if n_try == n:
            continue
        x_old = np.arange(len(h_best)) * (2.0 / len(h_best))
        x_new = np.linspace(0, 2.0, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, h_best)
        h_try = project_feasible(h_try, n_try)
        for T, K, mi in [(16, 40, 2000), (4, 20, 3000), (1, 8, 4000)]:
            rem = budget_s - (time.time() - t0)
            if rem < 40:
                break
            h_try = lbfgsb_opt(h_try, n_try, T, K, mi, 1e-13, t0, budget_s - 10)
        c5_try = compute_c5(h_try)
        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_try
            n = n_try

    # Phase 6: Diverse restarts
    for pi in range(35):
        rem = budget_s - (time.time() - t0)
        if rem < 18:
            break
        s = pi % 6
        if s == 0:
            h_try = h_best + rng.normal(0, 0.006, n)
        elif s == 1:
            x = np.arange(n) / n * 2.0
            h_try = h_best + 0.015 * np.sin(2*np.pi*rng.uniform(3,9)*x + rng.uniform(0,6.28))
        elif s == 2:
            h_try = h_best.copy()
            bs = max(8, n // 10)
            bi = int(rng.integers(0, n - bs))
            h_try[bi:bi+bs] = 1.0 - h_try[bi:bi+bs]
        elif s == 3:
            h_try = gen_start(n, rng, int(rng.integers(0, 5)))
        elif s == 4:
            # Half-swap
            h_try = h_best.copy()
            mid = n // 2
            h_try[:mid], h_try[mid:] = h_try[mid:].copy(), h_try[:mid].copy()
        else:
            h_try = h_best + rng.normal(0, 0.003, n)
        h_try = project_feasible(h_try, n)
        h_new = lbfgsb_opt(h_try, n, 4, 30, 1500, 1e-12, t0, budget_s - 8)
        c5 = compute_c5(h_new)
        if c5 < c5_best - 5e-6:
            h_new = lbfgsb_opt(h_new, n, 1, 5, 6000, 1e-14, t0, budget_s - 8)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 7: Final high-precision polish
    for T, K, mi in [(1, 5, 12000), (0.5, 3, 10000)]:
        rem = budget_s - (time.time() - t0)
        if rem < 25:
            break
        h_new = slsqp_opt(h_best, n, T, K, mi, 1e-16, t0, t0 + rem - 8)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 8: Ultra-fine L-BFGS-B
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        h_new = lbfgsb_opt(h_best, n, 1.0, 2, 15000, 1e-17, t0, t0 + rem - 5)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 9: Final worst-shift micro-search
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        h_w, c5_w = worst_shift_local(h_best, n, t0, t0 + rem - 3, 3000)
        if c5_w < c5_best:
            c5_best = c5_w
            h_best = h_w

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
