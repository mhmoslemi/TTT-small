def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Moderate SLSQP to settle into basin
    for T, k, mi in [(50, 5, 6000), (30, 3, 6000)]:
        rem = budget_s - (time.time() - t0)
        if rem < 100:
            break
        h_new = slsqp_focus(h_best, n, T, k, min(mi, int(rem * 2.5)))
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Sharp SLSQP focused on the max (k=1)
    for T, k, mi in [(200, 1, 6000), (100, 1, 6000), (50, 1, 8000), (25, 1, 8000)]:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        h_new = slsqp_focus(h_best, n, T, k, min(mi, int(rem * 2.5)))
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 3: Subgradient polish on the single worst shift
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        h_sg, c5_sg = subgrad_descent(h_best, n, t0 + min(rem * 0.25, 200), lr_init=0.003)
        h_sg = project_feasible(h_sg, n)
        c5_sg = compute_c5(h_sg)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase 4: Very sharp SLSQP (near-exact max)
    for T, k, mi in [(300, 1, 8000), (150, 1, 8000), (75, 1, 8000), (40, 2, 6000), (20, 2, 6000)]:
        rem = budget_s - (time.time() - t0)
        if rem < 60:
            break
        h_new = slsqp_focus(h_best, n, T, k, min(mi, int(rem * 2.5)))
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Micro-perturbation search (very small, diverse)
    rem = budget_s - (time.time() - t0)
    n_perts = max(8, int(rem / 18))
    for pi in range(n_perts):
        if time.time() - t0 > budget_s - 20:
            break
        scale = 0.001 + 0.003 * (pi % 4)
        h_p = h_best + rng.uniform(-scale, scale, n)
        h_p = project_feasible(h_p, n)
        h_new = slsqp_focus(h_p, n, 60, 1, 3000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Final SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        for T, k in [(200, 1), (100, 1), (50, 1)]:
            if time.time() - t0 > budget_s - 10:
                break
            h_new = slsqp_focus(h_best, n, T, k, min(5000, int((budget_s - (time.time()-t0))*2)))
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 7: Try alternative resolution (only if time allows)
    rem = budget_s - (time.time() - t0)
    if rem > 150:
        for n_try in [200, 168]:
            if time.time() - t0 > budget_s - 40:
                break
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_hi = np.interp(x_new, x_old, h_best)
            h_hi = project_feasible(h_hi, n_try)
            c5_hi = compute_c5(h_hi)
            for T, k, mi in [(100, 1, 4000), (30, 2, 4000)]:
                if time.time() - t0 > budget_s - 10:
                    break
                h_new = slsqp_focus(h_hi, n_try, T, k, min(mi, int((budget_s-(time.time()-t0))*2)))
                h_new = project_feasible(h_new, n_try)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
                    n = n_try

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
