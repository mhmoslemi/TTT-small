import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def get_worst_shift(h, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = int(np.argmax(corr))
    return k - (n - 1)

def shift_grad(h, n, s):
    j = np.arange(n)
    g = np.zeros(n)
    if s >= 0:
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    else:
        ss = -s
        m1 = (j + ss) < n
        m2 = j >= ss
        g[m1] += (1.0 - h[j[m1] + ss])
        g[m2] -= h[j[m2] - ss]
    return g

def subgradient_descent(h_init, n, t_end, rng, max_iter=200000):
    """Subgradient descent on the single worst shift."""
    dx = 2.0 / n

        e = np.exp(T * (c - m))

    j = np.arange(n)
    g = np.zeros(n)

        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]

        m2 = j >= ss
        g[m1] += (1.0 - h[j[m1] + ss])
        g[m2] -= h[j[m2] - ss]
    return g

        return (m + np.log(np.exp(T*(cv-m)).sum()) / T) * dx

        idx = rng.integers(0, n, 2)
        hp[idx[0]], hp[idx[1]] = hp[idx[1]], hp[idx[0]]
        return project_feasible(hp, n)
    else:
        hp = h.copy()
        idx = rng.integers(0, n, 3)
        hp[idx] = 1.0 - hp[idx]
        return project_feasible(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse to medium
    t1 = t0 + budget_s * 0.16
    for T, tk, mi in [(64, 30, 1200), (16, 15, 2000), (4, 10, 3000)]:
        if time.time() > t1:
            break
        hn = slsqp_opt(h_best, n, T, tk, mi, 1e-14)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 2: Subgradient on worst shift
    t2 = t0 + budget_s * 0.35
    hs, c5s = subgrad_worst(h_best, n, t2)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs

    # Phase 3: SLSQP fine (very low T)
    t3 = t0 + budget_s * 0.50
    for T, tk, mi in [(1.0, 6, 5000), (0.4, 5, 7000), (0.15, 4, 9000)]:
        if time.time() > t3:
            break
        hn = slsqp_opt(h_best, n, T, tk, mi, 1e-15)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 4: Pair swaps (new heuristic)
    t4 = t0 + budget_s * 0.65
    hl, c5l = pair_swaps(h_best, n, t4, 25000, rng)
    if c5l < c5_best:
        c5_best, h_best = c5l, hl

    # Phase 5: Another SLSQP + subgrad cycle
    t5 = t0 + budget_s * 0.76
    hn = slsqp_opt(h_best, n, 0.5, 4, 6000, 1e-15)
    c5 = compute_c5(hn)
    if c5 < c5_best:
        c5_best, h_best = c5, hn
    if time.time() < t5:
        hs2, c5s2 = subgrad_worst(h_best, n, t5, max_iter=100000)
        if c5s2 < c5_best:
            c5_best, h_best = c5s2, hs2

    # Phase 6: Diverse perturbations + SLSQP
    for pi in range(35):
        if time.time() > budget_s - 50:
            break
        hp = perturb(h_best, n, rng, pi % 4)
        hn = slsqp_opt(hp, n, 2, 8, 1800, 1e-13)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 7: Final SLSQP polish with multiple T
    if time.time() < budget_s - 15:
        for T, mi in [(0.2, 5000), (0.1, 6000)]:
            if time.time() > budget_s - 5:
                break
            hn = slsqp_opt(h_best, n, T, 3, mi, 1e-16)
            hn = project_feasible(hn, n)
            c5 = compute_c5(hn)
            if c5 < c5_best:
                c5_best, h_best = c5, hn

    # Phase 8: Final swap pass
    if time.time() < budget_s - 5:
        hl2, c5l2 = pair_swaps(h_best, n, budget_s - 3, 5000, rng)
        if c5l2 < c5_best:
            c5_best, h_best = c5l2, hl2

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
