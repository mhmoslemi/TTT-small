from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def eval_corr_vector(h, n):
    """Compute all correlation values (the full vector, times dx)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def shift_grad_single(h, n, d):
    """Gradient of corr[center+d] w.r.t. h, times dx."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[:m] += (1.0 - h[d:])
            grad[d:] -= h[:m]
    else:
        e = -d
        m = n - e
        if m > 0:
            grad[e:] += (1.0 - h[:m])
            grad[:m] -= h[e:]
    return grad * dx

def active_set_slsqp(h, n, K, maxiter, ftol, t0, budget_s):
    """Active-set SLSQP: minimize sum of top-K correlations."""
    target = n / 2.0
    center = n - 1
    
    for outer in range(10):
        if time.time() - t0 > budget_s - 60:
            break
        # Find top-K shifts
        cv = eval_corr_vector(h, n)
        top_idx = np.argpartition(cv, -K)[-K:]
        top_val = cv[top_idx].max()
        
        # Objective: weighted sum of top-K correlations
        # Weights: proportional to how close to max
        rel = cv[top_idx] / max(top_val, 1e-15)
        w = rel ** 4
        w /= w.sum()
        
        def obj(x):
            c2 = np.correlate(x, 1.0 - x, mode="full") * (2.0/n)
            return float(np.sum(w * c2[top_idx - center]))
        
        def jac(x):
            g = np.zeros(n)
            for i in range(K):
                g += w[i] * shift_grad_single(x, n, top_idx[i] - center)
            return g
        
        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
        
        try:
            res = minimize(obj, h, method='SLSQP', jac=jac,
                          bounds=bounds, constraints=constraints,

        c5_old = compute_c5(h)

        else:

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    mid = (lo + hi) / 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_single(h, n, d):
    dx = 2.0 / n
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[:m] += (1.0 - h[d:])
            grad[d:] -= h[:m]
    else:
        e = -d
        m = n - e
        if m > 0:
            grad[e:] += (1.0 - h[:m])
            grad[:m] -= h[e:]
    return grad * dx

def active_set_slsqp(h, n, K, maxiter, t0, time_limit):
    target = n / 2.0
    center = n - 1
    for _ in range(8):
        if time.time() - t0 > time_limit:
            break
        cv = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        top_idx = np.argpartition(cv, -K)[-K:]
        top_max = cv[top_idx].max()
        rel = cv[top_idx] / max(top_max, 1e-15)
        w = rel ** 6
        w /= w.sum()
        shifts = (top_idx - center).astype(int)

        def obj(x):
            c2 = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
            return float(np.sum(w * c2[top_idx]))

        def jac(x):
            g = np.zeros(n)
            for i in range(K):
                g += w[i] * shift_grad_single(x, n, shifts[i])
            return g

        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                        'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, h, method='SLSQP', jac=jac,
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, n)
        except Exception:
            h_new = h.copy()
        if compute_c5(h_new) < compute_c5(h):
            h = h_new
    return h
