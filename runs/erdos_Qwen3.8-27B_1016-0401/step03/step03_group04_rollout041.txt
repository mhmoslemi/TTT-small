def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = (j >= shift) & (j - shift < n)
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def smooth_max_val_grad(x, n, T, K):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    center = n - 1
    if K < len(corr):
        topk = np.argpartition(corr, -K)[-K:]
        c_sel = corr[topk]
        mc_sel = np.max(c_sel)
    else:
        topk = np.arange(len(corr))
        c_sel = corr
        mc_sel = np.max(c_sel)
    sh = T * (c_sel - mc_sel)
    sh -= sh.max()  # numerical stability
    e = np.exp(sh)
    sw = e / e.sum()
    val = (mc_sel + np.log(e.sum()) / T) * dx

    return val, grad * dx

        return v

        _, g = smooth_max_val_grad(x, n, T, K)
        return g

                       bounds=bounds,

                                'maxls': 20, 'maxfun': maxiter * 20})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def do_slsqp(h, n, T, K, maxiter, tol, t0, tlim):

        return v

        _, g = smooth_max_val_grad(x, n, T, K)
        return g

        return x.sum() - n / 2.0

        return np.ones(n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        m2 = (j + shift) < n
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        m2 = j >= s
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def smooth_max_val_grad(x, n, T, K):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    center = n - 1
    if K < len(corr):
        topk = np.argpartition(corr, -K)[-K:]
        c_sel = corr[topk]
        mc_sel = np.max(c_sel)
    else:
        topk = np.arange(len(corr))
        c_sel = corr
        mc_sel = np.max(c_sel)
    sh = T * (c_sel - mc_sel)
    sh -= sh.max()
    e = np.exp(sh)
    sw = e / e.sum()
    val = (mc_sel + np.log(e.sum()) / T) * dx
    grad = np.zeros(n)
    for ii in range(len(sw)):
        w = sw[ii]
        if w < 1e-14:
            break
        grad += w * shift_grad(x, n, topk[ii] - center)
    return val, grad * dx

_time_out = [False]

def _obj_lb(x, n, T, K, t0, tlim):

        _time_out[0] = True
    v, _ = smooth_max_val_grad(x, n, T, K)
    return v

def _jac_lb(x, n, T, K, t0, tlim):

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):

def _c_fun(x):

def do_slsqp(h, n, T, K, maxiter, tol, t0, tlim):

def nesterov_subgrad(h_init, n, t0, tlim, max_iters=200000):

    h_curr = h_init.copy()
    h_prev = h_init.copy()

    h_best = h_init.copy()
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        beta = it / (it + 2.5)
        y = h_curr + beta * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)

        shift = idx - (n - 1)
        g = shift_grad(y, n, shift) * dx
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        f_y = float(np.max(corr) * dx)

            if c5_now < c5_best:
