from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def smooth_max_obj(h, n, T, K):
    """Softmax-smoothed max of top-K correlations."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    mk = min(K, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = topk.max()
    if T < 1e-6:
        return mc * dx, None
    sh = T * (topk - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    obj = (mc + np.log(e.sum()) / T) * dx
    # Gradient
    grad = np.zeros(n)
    center = n - 1
    j = np.arange(n)
    for oi in range(mk):
        w = sw[oi] * dx
        if w < 1e-12:
            continue
        shift = idx[oi] - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += w * (1.0 - h[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= w * h[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += w * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * h[j[m2] - s2]
    return obj, grad

def slsqp_opt(h, n, T, K, maxiter, t_end):
    def obj(x):
        return smooth_max_obj(x, n, T, K)[0]
    def jac(x):
        return smooth_max_obj(x, n, T, K)[1]
    def feq(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': feq, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return res.x
    except Exception:
        return h.copy()

def subgradient_step(h, n, T, K, lr, t_end):
    """One projected subgradient step with backtracking."""
    obj, grad = smooth_max_obj(h, n, T, K)
    if grad is None:
        return h, obj
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new, obj

def opt_subgradient(h, n, t_end, T_start=50, T_end=1, K=5, lr0=0.05):
    """Adaptive subgradient descent with cooling."""
    h_c = h.copy()
    best_obj = compute_c5(h)
    best_h = h.copy()
    n_iters = 0
    lr = lr0
    for it in range(100000):
        if time.time() > t_end:
            break
        T = max(T_end, T_start * (T_end / T_start) ** (it / 5000.0))
        h_c, _ = subgradient_step(h_c, n, T, K, lr, t_end)
        lr *= 0.9999
        n_iters += 1
        if n_iters % 5000 == 0:
            c5 = compute_c5(h_c)
            if c5 < best_obj:
                best_obj = c5
                best_h = h_c.copy()
    return best_h, best_obj

def modulated_init(n, freq, rng):
    """Create initial pattern with modulation."""
    h = np.zeros(n)

    return h

def block_init(n, block_size, rng):

    h = np.zeros(n)
    offset = rng.integers(0, block_size)
    for i in range(n):
        h[i] = 1.0 if ((i + offset) % (2*block_size)) < block_size else 0.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def _obj_grad(h, n, T, K):
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    mk = min(K, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = topk.max()
    if T < 1e-8:
        obj = mc * dx
        grad = np.zeros(n)
        center = n - 1
        j = np.arange(n)
        shift = idx[np.argmax(topk)] - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += (1.0 - h[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= h[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= h[j[m2] - s2]
        grad *= dx
        return obj, grad
    sh = T * (topk - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    obj = (mc + np.log(e.sum()) / T) * dx
    grad = np.zeros(n)
    center = n - 1
    j = np.arange(n)
    for oi in range(mk):
        w = sw[oi] * dx
        if w < 1e-13:
            continue
        shift = idx[oi] - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += w * (1.0 - h[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= w * h[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += w * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * h[j[m2] - s2]
    return obj, grad

def slsqp_opt(h, n, T, K, maxiter, t_deadline):
    def obj(x):
        return _obj_grad(x, n, T, K)[0]
    def jac(x):
        return _obj_grad(x, n, T, K)[1]
    def feq(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': feq, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.success or res.fun < obj(h):
            return res.x
        return h
    except Exception:
        return h

def small_subgrad(h, n, t_deadline, T=1.0, K=3, lr0=0.01, n_steps=20000):
    h_c = h.copy()
    best = compute_c5(h_c)
    best_h = h_c.copy()
    lr = lr0
    for it in range(n_steps):
        if time.time() > t_deadline:
            break
        _, g = _obj_grad(h_c, n, T, K)
        h_c = h_c - lr * g
        h_c = project_feasible(h_c, n)
        lr *= 0.99995
        if (it + 1) % 5000 == 0:
            c = compute_c5(h_c)
            if c < best:
                best = c
                best_h = h_c.copy()
    c = compute_c5(h_c)
    if c < best:
        best = c
        best_h = h_c
    return best_h, best

def perturb(h, n, rng, scale=0.03):
    h_p = h.copy()
    pt = rng.integers(0, 5)
    if pt == 0:
        h_p += scale * rng.normal(0, 1, n)
    elif pt == 1:
        s = rng.integers(0, max(1, n - 15))
        ln = rng.integers(4, min(15, n - s))
        h_p[s:s+ln] = 1.0 - h_p[s:s+ln]
    elif pt == 2:
        f = rng.integers(1, 6)
        a = rng.uniform(0.02, 0.06)
        x = np.arange(n) * 2.0 / n
        h_p += a * np.sin(2*np.pi*f*x + rng.uniform(0, 6.28))
    elif pt == 3:
        b = rng.integers(4, 20)
        s1 = rng.integers(0, n - b)
        s2 = rng.integers(0, n - b)
        tmp = h_p[s1:s1+b].copy()
        h_p[s1:s1+b] = h_p[s2:s2+b]
        h_p[s2:s2+b] = tmp
    else:
        m = rng.random(n) < 0.08
        h_p[m] = rng.random(int(m.sum()))
    return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    
    # Phase 1: Coarse SLSQP to sharpen
    rem = budget_s - (time.time() - t0)
    if rem > 180:
        for (T, K, mi) in [(100, 20, 2000), (20, 10, 3000), (5, 6, 3000)]:
            rem = budget_s - (time.time() - t0)
            if rem < 100:
                break
            h_new = slsqp_opt(h_best, n, T, K, mi, t0 + rem * 0.75)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
    
    # Phase 2: Fine SLSQP focus on top-2
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h_new = slsqp_opt(h_best, n, 1.0, 2, 8000, t0 + rem * 0.6)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    
    # Phase 3: Subgradient micro-steps
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h_new, c5 = small_subgrad(h_best, n, t0 + rem * 0.5, T=1.0, K=3, lr0=0.005, n_steps=30000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    
    # Phase 4: Very fine SLSQP with K=2, T=0.5
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        h_new = slsqp_opt(h_best, n, 0.5, 2, 10000, t0 + rem * 0.7)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    
    # Phase 5: Targeted random restarts
    rem = budget_s - (time.time() - t0)
    n_perts = max(0, int(rem / 12))
    for pi in range(n_perts):
        if time.time() - t0 > budget_s - 12:
            break
        h_p = perturb(h_best, n, rng, scale=rng.uniform(0.01, 0.04))
        h_new = slsqp_opt(h_p, n, 3.0, 5, 1500, t0 + 9)
        h_new = project_feasible(h_new, n)
        h_new = slsqp_opt(h_new, n, 0.5, 2, 2000, t0 + 10)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    
    # Phase 6: Try n=140 (downsample)
    rem = budget_s - (time.time() - t0)
    if rem > 100 and n > 140:
        n2 = n // 2
        h2 = (h_best[0::2] + h_best[1::2]) / 2.0
        h2 = project_feasible(h2, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < c5_best:
            c5_best = c5_2
            h_best = h2
            n = n2
        else:
            # Still try to optimize n2 version
            h2o = slsqp_opt(h2, n2, 5.0, 5, 3000, t0 + 40)
            h2o = project_feasible(h2o, n2)
            h2o = slsqp_opt(h2o, n2, 1.0, 2, 4000, t0 + 60)
            h2o = project_feasible(h2o, n2)
            c5_2 = compute_c5(h2o)
            if c5_2 < c5_best:
                c5_best = c5_2
                h_best = h2o
                n = n2
    
    # Phase 7: Try n=420 (upsample) if time allows
    rem = budget_s - (time.time() - t0)
    if rem > 120 and n < 500:
        n3 = n * 2
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n3, endpoint=False)
        h3 = np.interp(x_new, x_old, h_best)
        h3 = project_feasible(h3, n3)
        h3 = slsqp_opt(h3, n3, 30, 15, 2000, t0 + 40)
        h3 = project_feasible(h3, n3)
        h3 = slsqp_opt(h3, n3, 2.0, 3, 3000, t0 + 60)
        h3 = project_feasible(h3, n3)

            c5_best = c5_3
            h_best = h3
            n = n3
    
    # Final polish
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_new = slsqp_opt(h_best, n, 0.5, 2, 5000, t0 + rem * 0.8)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    
    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
