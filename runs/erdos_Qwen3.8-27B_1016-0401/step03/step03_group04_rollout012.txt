import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    grad = np.zeros(n)
    j = np.arange(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def smooth_obj_grad(h, n, T, K):
    corr = np.correlate(h, 1.0 - h, mode="full")
    nc = len(corr)
    center = (nc - 1) // 2
    if K < nc:
        topk = np.argpartition(corr, -K)[-K:]
    else:
        topk = np.arange(nc)
    c_sel = corr[topk]
    mc = float(np.max(c_sel))
    sh = T * (c_sel - mc)
    sh -= sh.max()
    e = np.exp(sh)
    sw = e / e.sum()
    val = mc + np.log(e.sum()) / T
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(len(sw)):
        w = sw[i]
        if w < 1e-16:
            continue
        shift = topk[i] - center
        grad += w * shift_grad(h, n, shift)
    imax = int(np.argmax(c_sel))
    grad += shift_grad(h, n, topk[imax] - center)
    return val * dx, grad * dx

def nesterov_solve(h_init, n, t0, tlim, max_iters=300000):
    dx = 2.0 / n
    h_curr = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        beta = it / (it + 2.5)
        y = h_curr + beta * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = shift_grad(y, n, shift) * dx
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        f_y = corr[idx] * dx
        gap = max(f_y - 0.37, 1e-7)
        alpha = min(2.0 * gap / (gn * gn + 1e-12), 0.2 / (1.0 + 0.001 * it))
        h_new = project_feasible(y - alpha * g, n)
        h_prev = h_curr.copy()
        h_curr = h_new
        if (it + 1) % 4000 == 0:
            c5_now = compute_c5(h_curr)
            if c5_now < c5_best:
                c5_best = c5_now
                h_best = h_curr.copy()
    return h_best, c5_best

def lbfgsb_opt(h, n, T, K, maxiter, t0, tlim):
    bounds = [(0.0, 1.0)] * n
    def objf(x):
        if time.time() - t0 > tlim:
            raise RuntimeError("t")
        return smooth_obj_grad(x, n, T, K)[0]
    def jacf(x):
        if time.time() - t0 > tlim:
            raise RuntimeError("t")
        return smooth_obj_grad(x, n, T, K)[1]
    try:
        res = minimize(objf, h.copy(), method='L-BFGS-B', jac=jacf,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                                'maxls': 40, 'maxfun': maxiter * 4})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def sigmoid_optim(h, n, t0, tlim):
    eps = 1e-6
    hc = np.clip(h, eps, 1.0 - eps)
    a = np.log(hc / (1.0 - hc))
    a = np.clip(a, -15.0, 15.0)
    c5_best = compute_c5(h)
    h_best = h.copy()
    for T, K, mi in [(6, 60, 2000), (3, 40, 3000), (1.5, 25, 4000), (1.0, 15, 5000)]:
        if time.time() - t0 > tlim:
            break
        def sobj(a_):
            h_ = project_feasible(1.0 / (1.0 + np.exp(-a_)), n)
            corr = np.correlate(h_, 1.0 - h_, mode="full")
            nc = len(corr)
            ctr = (nc - 1) // 2
            if K < nc:
                topk = np.argpartition(corr, -K)[-K:]
            else:
                topk = np.arange(nc)
            c_s = corr[topk]
            mc = float(np.max(c_s))
            sh = T * (c_s - mc)
            sh -= sh.max()
            e = np.exp(sh)
            sw = e / e.sum()
            val = mc + np.log(e.sum()) / T
            dx = 2.0 / n
            grad_h = np.zeros(n)
            for ii in range(len(sw)):
                w = sw[ii]
                if w < 1e-16:
                    continue
                grad_h += w * shift_grad(h_, n, topk[ii] - ctr)
            grad_h += shift_grad(h_, n, topk[int(np.argmax(c_s))] - ctr)
            grad_a = grad_h * h_ * (1.0 - h_) * dx
            return val * dx, grad_a
        try:
            res = minimize(sobj, a.copy(), method='L-BFGS-B',
                           options={'maxiter': mi, 'ftol': 1e-13, 'gtol': 1e-9})
            h_new = project_feasible(1.0 / (1.0 + np.exp(-res.x)), n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
                hc2 = np.clip(h_best, eps, 1.0 - eps)
                a = np.clip(np.log(hc2 / (1.0 - hc2)), -20.0, 20.0)
        except Exception:
            pass
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    def rem():

    # Phase 1: L-BFGS-B coarse-to-fine

    # Phase 2: Sigmoid parameterization

            h_low = project_feasible(np.interp(x_low, x_old, h_best), n_low)
            best_low = h_low.copy()
            best_c5_low = compute_c5(h_low)
            for ri in range(10):
                if rem() < 25:
                    break
                if ri == 0:
                    h_try = h_low.copy()
                elif ri % 3 == 1:

        h_new = lbfgsb_opt(h_p, n, 1.5, 30, 2000, t0, budget_s - 8)

    # Phase 6: Quantization nudge (push toward binary)

            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, n)
