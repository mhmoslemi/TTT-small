import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_TARGET = 0.0
_G_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j < (n - shift)
        grad[m1] += (1.0 - h[m1 + shift])
        if shift > 0:
            m2 = j >= shift
            grad[m2] -= h[m2 - shift]
    else:
        s = -shift
        m1 = j >= s
        grad[m1] += (1.0 - h[m1 - s])
        m2 = j < (n - s)
        grad[m2] -= h[m2 + s]
    return grad

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-10:
            continue
        grad += w * shift_grad_raw(x, _G_N, oi - center)
    return grad * dx

def _g_cons_fun(x):
    return x.sum() - _G_TARGET

def _g_cons_jac(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter, ftol):
    global _G_N, _G_TARGET, _G_T
    _G_N = n
    _G_TARGET = n / 2.0
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_cons_fun, 'jac': _g_cons_jac}]
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def active_set_gradient(h, n, K=10, max_iters=80000, t0=None, budget=600.0):
    dx = 2.0 / n
    center = n - 1
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    momentum = np.zeros(n)
    mu = 0.9
    for it in range(max_iters):
        if t0 is not None and time.time() - t0 > budget + t0:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        top_idx = np.argsort(corr)[-K:]
        top_vals = corr[top_idx]
        max_v = top_vals.max()
        w = np.exp(8.0 * (top_vals - max_v))
        w /= w.sum()
        g = np.zeros(n)
        for wk, ik in zip(w, top_idx):
            if wk < 1e-10:
                continue
            g += wk * shift_grad_raw(h_cur, n, ik - center)
        g *= dx
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-12:
            break
        g_normed = g / norm_g
        # Backtracking line search
        alpha = 0.5 / (1.0 + 0.001 * it)
        best_h = h_cur.copy()
        best_c5 = c5_cur
        for back in range(10):
            h_try = h_cur - alpha * g
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-10:
                best_h = h_try
                best_c5 = c5_try
                break
            alpha *= 0.7
        if best_c5 < c5_cur:
            momentum = mu * momentum + (1 - mu) * (best_h - h_cur)
            h_cur = project_feasible(best_h + 0.5 * momentum, n)
            if compute_c5(h_cur) > best_c5:
                h_cur = best_h
            c5_cur = min(c5_cur, compute_c5(h_cur))
        else:
            alpha *= 0.5
            if alpha < 1e-8:
                break
    return h_cur, compute_c5(h_cur)

def coord_descent(h, n, rng, steps=8000):
    dx = 2.0 / n
    center = n - 1
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
    for it in range(steps):
        ak = int(np.argmax(corr))
        shift = ak - center
        g = shift_grad_raw(h_cur, n, shift) * dx
        # Find best single coordinate move
        pos_idx = np.where(g > 0.005)[0]
        neg_idx = np.where(g < -0.005)[0]
        if len(pos_idx) == 0 or len(neg_idx) == 0:
            # Try sum-preserving pair move
            break
        ip = rng.choice(pos_idx[:50] if len(pos_idx) > 50 else pos_idx)
        inn = rng.choice(neg_idx[:50] if len(neg_idx) > 50 else neg_idx)
        delta = min(0.03, h_cur[ip], 1.0 - h_cur[inn])
        h_try = h_cur.copy()
        h_try[ip] -= delta
        h_try[inn] += delta
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < c5_cur - 1e-9:
            h_cur = h_try
            c5_cur = c5_try
            corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
    return h_cur, c5_cur

def round_and_reopt(h, n, T=2.0, maxiter=3000):
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    h_new = do_slsqp(h_bin, n, T, maxiter, 1e-14)
    h_new = project_feasible(h_new, n)
    return h_new
