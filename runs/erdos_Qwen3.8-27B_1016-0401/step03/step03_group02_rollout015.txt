        if diff > 0:
            mask = h < 1.0
            if not mask.sum():
                break
            step = diff / mask.sum()
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            if not mask.sum():
                break
            step = diff / mask.sum()  # negative
            h[mask] = np.maximum(h[mask] + step, 0.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] = np.minimum(h[mask] + diff / mask.sum(), 1.0)
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] = np.maximum(h[mask] + diff / mask.sum(), 0.0)
    return h

def sm_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / e.sum()
    c = n - 1
    g = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[c + o]
        if w < 1e-14:
            continue
        if o == 0:
            g += w * (1.0 - 2.0 * h)
        elif o > 0:
            g[o:] += w * (1.0 - h[:n - o])
            g[:n - o] -= w * h[o:]
        else:
            p = -o
            g[:n - p] += w * (1.0 - h[p:])
            g[p:] -= w * h[:n - p]
    return g * dx

def sm_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx

def momentum_gd(h, T, n_steps, lr, beta, target, time_limit):
    h = h.astype(float).copy()
    v = np.zeros_like(h)
    best = h.copy()
    best_c5 = compute_c5(h)
    t_end = time.time() + time_limit
    for s in range(n_steps):
        if time.time() > t_end:
            break
        g = sm_grad(h, T)
        v = beta * v + g
        h_new = h - lr * v
        h_new = project_h(h_new, target)
        if not np.allclose(h_new, h, atol=1e-16):
            h = h_new
        else:
            lr *= 0.85
        if s % 400 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
        if s % 2000 == 0:
            lr *= 0.92
    return best, best_c5

def slsqp_lag_opt(h, lag_idx, target, bounds, maxiter, ftol):
    n = len(h)
    dx_val = 2.0 / n

    def obj(x):
        return np.correlate(x, 1.0 - x, mode="full")[lag_idx] * dx_val

    def jac(x):
        o = lag_idx - (n - 1)
        g = np.zeros(n)
        if o == 0:
            g[:] = 1.0 - 2.0 * x
        elif o > 0:
            g[o:] += 1.0 - x[:n - o]
            g[:n - o] += -x[o:]
        else:
            p = -o
            g[:n - p] += 1.0 - x[p:]
            g[p:] += -x[:n - p]
        return g * dx_val

    def eq_f(x):
        return x.sum() - target

    def eq_j(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq_f, 'jac': eq_j}]
    res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_h(np.clip(res.x, 0, 1), target)

def binary_local_search(h, target, n_sweeps, time_limit):
    h = h.copy()
    t_end = time.time() + time_limit
    c5_cur = compute_c5(h)
    for s in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for i in ones:
            for j in zeros:
                if time.time() > t_end:
                    break
                trial = h.copy()
                trial[i] = 0.0
                trial[j] = 1.0
                trial = project_h(trial, target)
                c5_new = compute_c5(trial)
                if c5_new < c5_cur - 1e-15:
                    h = trial
                    c5_cur = c5_new
                    improved = True
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    break
            if time.time() > t_end:
                break
        if not improved:
            break
    return h, c5_cur

def coord_descent(h, target, n_sweeps, time_limit):
    h = h.copy()
    t_end = time.time() + time_limit
    c5_cur = compute_c5(h)
    dts = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    for s in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for j in range(len(h)):
            for dt in dts:
                trial = h.copy()
                trial[j] = np.clip(trial[j] + dt, 0, 1)
                trial = project_h(trial, target)
                c5_new = compute_c5(trial)
                if c5_new < c5_cur - 1e-15:
                    h = trial
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def interpolate_to_n(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    target = n_new / 2.0
    h_new = project_h(h_new, target)
    return h_new
