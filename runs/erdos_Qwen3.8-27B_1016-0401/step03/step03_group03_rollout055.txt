import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    mid = 0.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

# SLSQP helper with globals
_N = 0
_TARGET = 0.0
_T = 1.0

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _T * (corr - mc)
    e = np.exp(sh)
    se = e.sum()
    return (mc + np.log(se) / _T) * (2.0 / _N)

def _jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _N - 1
    dx = 2.0 / _N
    grad = np.zeros(_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-13:
            continue
        grad += w * shift_grad(x, _N, oi - center)
    return grad * dx

def _cons_fun(x):
    return x.sum() - _TARGET

def _cons_jac(x):
    return np.ones(_N)

def do_slsqp(h, n, T, maxiter, ftol):
    global _N, _TARGET, _T
    _N = n
    _TARGET = n / 2.0
    _T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', jac=_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def topK_descent(h, n, t0, time_budget, K0=10, K1=40):
    dx = 2.0 / n
    center = n - 1
    h_cur = h.copy()
    h_mom = h.copy()
    c5_best = compute_c5(h_cur)
    h_best = h_cur.copy()
    it = 0
    while time.time() - t0 < time_budget:
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        # Adaptive K
        frac = it / 20000.0
        K = int(K0 + (K1 - K0) * min(1.0, frac))
        top_idx = np.argsort(corr)[-K:]
        top_vals = corr[top_idx]
        mv = top_vals.max()
        T_w = 3.0 + 10.0 / (1.0 + 0.001 * it)
        w = np.exp(T_w * (top_vals - mv))
        w /= w.sum()
        g = np.zeros(n)
        for wk, ik in zip(w, top_idx):
            if wk > 1e-10:
                g += wk * shift_grad(h_cur, n, ik - center)
        g *= dx
        # Adaptive step with momentum
        alpha = 0.05 / (1.0 + 0.002 * it)
        beta = 0.85
        h_mom = beta * h_mom + (1.0 - beta) * (h_cur - alpha * g)
        h_new = project_feasible(h_mom, n)
        if (it + 1) % 2000 == 0:
            cv = compute_c5(h_new)
            if cv < c5_best:
                c5_best = cv
                h_best = h_new.copy()
            h_cur = h_new
        else:
            h_cur = h_new
        it += 1
    cv = compute_c5(project_feasible(h_cur, n))
    if cv < c5_best:
        c5_best = cv
        h_best = project_feasible(h_cur, n).copy()
    return h_best, c5_best

def pair_swap_search(h, n, rng, time_budget, t0):
    """Targeted pair swaps to reduce the max correlation."""

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    mid = (lo + hi) / 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

_N = 0
_TGT = 0.0
_TW = 1.0

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _TW * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _TW) * (2.0 / _N)

def _jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _TW * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _N - 1
    dx = 2.0 / _N
    grad = np.zeros(_N)
    for oi in range(len(sw)):
        if sw[oi] < 1e-13:
            continue
        grad += sw[oi] * shift_grad(x, _N, oi - center)
    return grad * dx

    return x.sum() - _TGT

    return np.ones(_N)

    _TGT = n / 2.0

                       bounds=bounds, constraints=cons,
