    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def solve_qp_polish(h0, n, active_shifts, C5_target, max_iter=100):
    """Solve QP: minimize ||h - h0||² s.t. linearized active constraints, bounds, sum=1"""
    from scipy.optimize import minimize as opt_min
    dx = 2.0 / n
    corr = np.correlate(h0, 1.0 - h0, mode="full")
    center = n - 1
    j = np.arange(n)
    
    # Build linear constraint matrices for active shifts
    # For shift k (index in corr array), the linearized constraint is:
    # sum_i [(1-h0[i+shift])*(h[i]-h0[i]) - h0[shift+i]*(h[i+shift]-h0[i+shift])] <= (C5_target/dx) - corr[k]
    # where shift = k - center
    
    A_rows = []
    b_rows = []
    
    for k in active_shifts:
        shift = k - center
        row = np.zeros(n)
        if shift >= 0:
            # corr[k] = sum_{i=0}^{n-1-shift} h[i]*(1-h[i+shift])
            # d/dh[i+shift]: -(h[i]) -> contributes to index i+shift
            # d/dh[i]: (1-h[i+shift]) -> contributes to index i
            for i in range(n - shift):
                row[i] += (1.0 - h0[i + shift])
                row[i + shift] -= h0[i]
        else:
            s2 = -shift
            for i in range(n - s2):
                row[i + s2] += (1.0 - h0[i])
                row[i] -= h0[i + s2]
        
        # c[k] * dx is the current value, we want it <= C5_target
        # Linearized: c[k] + row·(h-h0) <= C5_target/dx
        # So: row·h <= C5_target/dx - c[k] + row·h0
        rhs = C5_target / dx - corr[k] + row @ h0
        A_rows.append(row)
        b_rows.append(rhs)
    
    A = np.array(A_rows)
    b = np.array(b_rows)
    
    # Optimization: minimize (h - h0)ᵀ(h - h0) s.t. A h ≤ b, sum(h) = n/2, 0 ≤ h ≤ 1
    # This is a convex QP
    # Using SLSQP with the QP formulation
    
    def obj_q(h):
        d = h - h0
        return d @ d
    
    def jac_q(h):
        return 2.0 * (h - h0)
    
    cons = []
    for i in range(len(A_rows)):
        cons.append({'type': 'ineq', 'fun': lambda h, i=i: b[i] - A[i] @ h, 
                     'jac': lambda h, i=i: -A[i]})
    cons.append({'type': 'eq', 'fun': lambda h: h.sum() - n/2.0,
                 'jac': lambda h: np.ones(n)})
    
    bounds = [(0.0, 1.0)] * n
    
    h_new = h0.copy()
    for _ in range(max_iter):

                     bounds=bounds, constraints=cons,

        row = np.zeros(n)
        if shift >= 0:
            row[shift:] += (1.0 - h0[0:n - shift])
            row[0:n - shift] -= h0[shift:n]
        else:
            s2 = -shift
            row[0:n - s2] += (1.0 - h0[s2:n])
            row[s2:] -= h0[0:n - s2]
        A_list.append(row)
        b_list.append(c5_target / dx - corr[ki] + row @ h0)
    A = np.array(A_list)
    b = np.array(b_list)
    h_var = cp.Variable(n)
    obj = cp.sum_squares(h_var - h0)
    constraints = [A @ h_var <= b, cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)
    for solver in [cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, max_iter=5000, abstol=1e-12, reltol=1e-12)
            if h_var.value is not None and not np.any(np.isnan(h_var.value)):
                return project_feasible(np.array(h_var.value), n)
        except Exception:
            continue
    return h0.copy()

def coordinate_polish(h, n, n_iters=5000, time_limit=None):
    best_c5 = compute_c5(h)
    h_best = h.copy()
    t_start = time.time()
    rng = np.random.default_rng(12345)
    j_arr = np.arange(n)
    for it in range(n_iters):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        i = rng.integers(n)
        delta = rng.uniform(-0.005, 0.005)
        new_val = h[i] + delta
        if new_val < 0 or new_val > 1:
            continue
        h[i] = new_val
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h.copy()
        else:
            h[i] -= delta
            h = project_feasible(h, n)
    return h_best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Coarse-to-fine SLSQP
    schedule = [(64, 30, 1500), (32, 20, 2000), (16, 12, 2500),
                (8, 8, 3000), (4, 6, 3000), (2, 4, 4000), (1, 3, 4000)]
    for T, k, mi in schedule:
        rem = budget_s - (time.time() - t0)
        if rem < 120:
            break
        h_new = slsqp_phase(h_best, n, T, k, min(mi, int(rem * 0.4)), 1e-18)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: QP polish with active set iteration
    rem = budget_s - (time.time() - t0)
    if rem > 100:
        for k_act in [5, 8, 12, 20]:
            if time.time() - t0 > budget_s - 50:
                break
            for _ in range(8):
                h_new = qp_polish(h_best, n, k_act, c5_best * 1.0001)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
                else:
                    break

    # Phase 3: High-T SLSQP fine polish
    for T, k in [(50, 3), (100, 3), (200, 2), (80, 4), (50, 5)]:
        rem = budget_s - (time.time() - t0)
        if rem < 60:
            break
        h_new = slsqp_phase(h_best, n, T, k, 3000, 1e-19)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        if rem > 30:
            h_new2 = qp_polish(h_best, n, 5, c5_best * 1.0001)
            c52 = compute_c5(h_new2)
            if c52 < c5_best:
                c5_best = c52
                h_best = h_new2

    # Phase 4: Coordinate polish (local random search)
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_new, c5_new = coordinate_polish(h_best, n, n_iters=8000, time_limit=rem * 0.5)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 5: Try nearby resolutions
    rem = budget_s - (time.time() - t0)
    if rem > 150:
        for n_try in [n + 20, n + 40, n - 20, n + 60]:
            if n_try < 100 or n_try > 600:
                continue
            if time.time() - t0 > budget_s - 60:
                break
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_hi = np.interp(x_new, x_old, h_best)
            h_hi = project_feasible(h_hi, n_try)
            c5_hi = compute_c5(h_hi)
            if c5_hi < c5_best * 1.01:
                for T, k in [(16, 10), (4, 5), (1, 3)]:
                    if time.time() - t0 > budget_s - 20:
                        break
                    h_new = slsqp_phase(h_hi, n_try, T, k, 2500, 1e-18)
                    c5 = compute_c5(h_new)
                    if c5 < c5_best:
                        c5_best = c5
                        h_best = h_new
                        n = n_try
                        c5_hi = c5
                    h_new2 = qp_polish(h_best, n_try, 5, c5_best * 1.0001)
                    c52 = compute_c5(h_new2)
                    if c52 < c5_best:
                        c5_best = c52
                        h_best = h_new2

    # Phase 6: Final QP + coordinate polish
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_new = qp_polish(h_best, n, 8, c5_best * 1.00005)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        if rem > 20:
            h_new, c5_new = coordinate_polish(h_best, n, n_iters=3000, time_limit=rem * 0.4)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
