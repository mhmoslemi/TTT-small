def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0

            break

        else:
            hi = mid

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:

        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad
