- c_k as a function of h[j] is linear (since h[j] appears linearly in each c_k)
- So we can find the optimal h[j] by checking where the max of these linear functions is minimized
- This is a 1D minimax problem!

For coordinate j:
- c_k = a_k * h[j] + b_k where a_k and b_k don't depend on h[j]
  (a_k depends on whether j is in the range for lag k, and on the value of the other coordinate)
- We want to minimize max_k (a_k * h[j] + b_k) subject to 0 ≤ h[j] ≤ 1 and the sum constraint

The sum constraint couples all coordinates, but if we fix the sum of all other coordinates, then h[j] is determined. So actually, for a single coordinate change, we need to also adjust another coordinate to maintain the sum.

This is why the 2-flip approach is natural: change h[i] by δ and h[j] by -δ to maintain the sum.

For a 2-coordinate change (i gets +δ, j gets -δ):
- c_k changes by δ * (a_k_i - a_k_j) where a_k_i is the coefficient of h[i] in c_k and a_k_j is the coefficient of h[j] in c_k
- We want to choose δ to minimize the resulting max c_k

This is a 1D problem in δ! We can solve it efficiently.

def two_coord_optimal(h, i, j, target):
    """Find optimal δ for h[i]+=δ, h[j]-=δ (or vice versa)."""
    n = len(h)
    dx = 2.0 / n
    # Current correlations
    corr_base = np.correlate(h, 1.0 - h, mode="full")
    
    # Coefficient of h[i] in each corr:
    # corr[sigma] = sum over valid indices
    # For h[i], it appears in corr[n-1+k] when i + k < n (as h[i]) 
    # and in corr[n-1-k] when i - k >= 0 (as the "other" term)
    
    # Actually let's just compute the derivative directly
    # d(corr)/d(h[i]) for each lag
    # corr[sigma] involves h[a]*(1-h[b]) where sigma encodes the lag
    # For the full correlation: corr[n-1+k] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
    # d/d(h[j]) of corr[n-1+k]:
    #   = (1-h[j+k]) if j <= n-1-k
    #   + (-h[j-k]) if j-k >= 0 and j-k <= n-1-(-k) ... 
    #   Actually for lag -k: corr[n-1-k] = sum_{i=k}^{n-1} h[i]*(1-h[i-k])
    #   d/d(h[j]): = (1-h[j-k]) if j >= k
    #   + (-h[j+k]) if j+k < n ... wait I'm getting confused with indices.
    
    # Let me just use finite differences for the coefficients
    eps = 1e-8
    # coeff_i[sigma] = (corr[h + eps*e_i] - corr[h - eps*e_i]) / (2*eps)
    # But this is O(n) for each sigma, and there are 2n-1 sigmas, so O(n²)
    # For n=340, that's 115K operations. Fine.
    
    h_p = h.copy(); h_p[i] += eps
    h_m = h.copy(); h_m[i] -= eps
    coeff_i = (np.correlate(h_p, 1-h_p, mode="full") - np.correlate(h_m, 1-h_m, mode="full")) / (2*eps)
    
    h_p = h.copy(); h_p[j] += eps
    h_m = h.copy(); h_m[j] -= eps
    coeff_j = (np.correlate(h_p, 1-h_p, mode="full") - np.correlate(h_m, 1-h_m, mode="full")) / (2*eps)
    
    # For δ: h'[i] = h[i]+δ, h'[j] = h[j]-δ
    # c_k(δ) = corr_base[k] + δ*(coeff_i[k] - coeff_j[k])
    # Minimize max_k c_k(δ) subject to 0 <= h[i]+δ <= 1 and 0 <= h[j]-δ <= 1
    
    delta_coeff = coeff_i - coeff_j  # shape (2n-1,)
    # c_k(δ) = corr_base[k] + δ * delta_coeff[k]
    
    # Bounds on δ:
    delta_min = max(-h[i], h[j] - 1.0)
    delta_max = min(1.0 - h[i], -h[j])
    
    # Minimize max of linear functions: this is a 1D convex problem
    # The max of linear functions is convex, so we can use ternary search or
    # find where the max is minimized
    
    # For linear functions f_k(δ) = a_k * δ + b_k, the max is minimized at
    # a point where at least two functions are equal and both are at the max.
    
    # Simple approach: ternary search on [delta_min, delta_max]
    if delta_min >= delta_max:
        return 0
    
    def max_c(delta):
        vals = corr_base + delta * delta_coeff
        return np.max(vals)
    
    # Grid search + refinement
    deltas = np.linspace(delta_min, delta_max, 200)
    max_vals = np.array([max_c(d) for d in deltas])
    best_idx = np.argmin(max_vals)
    
    # Refine
    if best_idx == 0:
        best_delta = delta_min
    elif best_idx == len(deltas) - 1:
        best_delta = delta_max
    else:
        # Ternary search in surrounding interval
        lo = deltas[max(0, best_idx-1)]
        hi = deltas[min(len(deltas)-1, best_idx+1)]
        for _ in range(30):
            m1 = lo + (hi - lo) / 3
            m2 = hi - (hi - lo) / 3
            if max_c(m1) < max_c(m2):
                hi = m2
            else:
                lo = m1
        best_delta = (lo + hi) / 2
    
    return best_delta

    cj /= (2.0 * eps)
    dc = ci - cj
    lo = max(-h[i], h[j] - 1.0)
    hi_ = min(1.0 - h[i], -h[j])
    if lo >= hi_:
        return 0.0
    grids = np.linspace(lo, hi_, 80)
    vals = np.array([np.max(corr + d * dc) for d in grids])
    bi = int(np.argmin(vals))
    if bi == 0:
        return lo
    if bi == len(grids) - 1:
        return hi_
    a = grids[max(0, bi - 1)]
    b = grids[min(len(grids) - 1, bi + 1)]
    for _ in range(35):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if np.max(corr + m1 * dc) < np.max(corr + m2 * dc):
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    rng = np.random.default_rng(seed)
    h0 = None
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    if h0 is None:
        h0 = np.zeros(280)
        h0[::2] = 1.0
    n = len(h0)
    target = n / 2.0
    h = fix_sum(h0, target)
    c5_best = compute_c5(h)

    # Phase 1: Nesterov PG with descending T (35% budget)
    t1 = t0 + budget_s * 0.35
    eta0 = 0.03
    for T in [50, 150, 400, 1000, 3000]:
        if elapsed() > t1:
            break
        h_new, c_new = nesterov_pg(h, target, [T], 60, eta0 / T)
        if c_new < c5_best:
            h, c5_best = h_new, c_new

    # Phase 2: Binary SA (25% budget)
    t2 = t0 + budget_s * 0.60
    while elapsed() < t2:
        h_bin = (h > 0.5).astype(float)
        h_bin = fix_sum(h_bin, target)
        h_bin, c_bin = two_flip_sa(h_bin, target, 800, 0.008, 0.0005, rng)
        if c_bin < c5_best:
            c5_best = c_bin
            h = h_bin.copy()
        # Alternate threshold
        h_b2 = (h > 0.4).astype(float)
        h_b2 = fix_sum(h_b2, target)
        h_b2, c_b2 = two_flip_sa(h_b2, target, 400, 0.005, 0.0005, rng)
        if c_b2 < c5_best:
            c5_best = c_b2
            h = h_b2.copy()

    # Phase 3: Optimal 2-coordinate moves (25% budget)
    t3 = t0 + budget_s * 0.85
    stall = 0
    while elapsed() < t3 and stall < 30:
        stall += 1
        improved_any = False
        for _ in range(80):
            if elapsed() > t3:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            if h[i] < 1e-8 and h[j] < 1e-8:
                continue
            if h[i] > 1.0 - 1e-8 and h[j] > 1.0 - 1e-8:
                continue
            delta = two_coord_delta(h, i, j)
            if abs(delta) < 1e-10:
                continue
            ht = h.copy()
            ht[i] += delta
            ht[j] -= delta
            ht = project(ht, target)
            c_new = compute_c5(ht)
            if c_new < c5_best - 1e-14:
                h = ht
                c5_best = c_new
                improved_any = True
                stall = 0
        if not improved_any:
            stall += 1

    # Phase 4: Final gradient polish + try other n (15% budget)
    t4 = t0 + budget_s - 5.0
    for T in [5000, 15000, 40000]:
        if elapsed() > t4:
            break
        h_new, c_new = nesterov_pg(h, target, [T], 40, 0.01 / T)
        if c_new < c5_best:
            h, c5_best = h_new, c_new
    # Systematic single-coordinate polish
    for _ in range(3):
        if elapsed() > t4:
            break
        for j in range(n):
            if elapsed() > t4:
                break
            for dv in [0.5, -0.5, 0.1, -0.1, 0.02, -0.02]:
                trial = h.copy()
                trial[j] = np.clip(trial[j] + dv, 0.0, 1.0)
                trial = fix_sum(trial, target)
                if compute_c5(trial) < c5_best - 1e-14:
                    h = trial
                    c5_best = compute_c5(h)
                    break
    # Try nearby n values
    if elapsed() < t4:
        for dn in [-20, 20, 40, -40]:
            n_alt = n + dn
            if n_alt < 100 or elapsed() > t4:
                continue
            x_old = np.arange(n) * 2.0 / n
            x_new = np.linspace(0.0, 2.0, n_alt, endpoint=False)
            h_alt = fix_sum(np.interp(x_new, x_old, h), n_alt / 2.0)
            h_alt, c_alt = nesterov_pg(h_alt, n_alt / 2.0, [200, 800, 3000], 40, 0.02)
            if c_alt < c5_best:
                h = h_alt
                c5_best = c_alt
                n = n_alt
                target = n / 2.0
            h_alt2 = (h_alt > 0.5).astype(float)
            h_alt2 = fix_sum(h_alt2, n_alt / 2.0)
            h_alt2, c_alt2 = two_flip_sa(h_alt2, n_alt / 2.0, 300, 0.005, 0.001, rng)
            if c_alt2 < c5_best:
                h = h_alt2
                c5_best = c_alt2
                n = n_alt
                target = n / 2.0

    c5_final = compute_c5(h)
    return (h, min(c5_final, c5_best), len(h))
