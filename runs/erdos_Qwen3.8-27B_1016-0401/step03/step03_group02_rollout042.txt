The minimum of a convex PL function on [0,1] occurs at:
- A vertex (kink point) of the upper envelope
- Or at a boundary (0 or 1)

The kink points are where two of the linear functions cross each other AND both are on the upper envelope. In the worst case there are O(n) such points, but in practice much fewer.

4. If the slope is negative at t=0, the minimum is to the right
5. Find the next kink by finding which pair of active/inactive functions crosses first

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def exact_coord_sweep(h, n_sweeps, tol=1e-10):

def slsqp_lag(h, lag_idx, maxiter=800, ftol=1e-18):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n

        return float(np.correlate(x, 1.0 - x, mode="full")[lag_idx] * dx)

        o = lag_idx - (n - 1)

            g[:] = 1.0 - 2.0 * x

        else:

            g[p:] += -x[:n-p]
        return g * dx

        return np.ones(n)

                   options={'maxiter': maxiter, 'ftol': ftol})
    return np.clip(res.x, 0.0, 1.0), cfun(res.x)

    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * (2.0 / n)

def sm_jac(x, T):
    n = len(x)

            grad[:n-p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n-p]
    return grad * dx
