    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = np.max(topk)
    sh = T * (topk - mc)
    e = np.exp(np.clip(sh, -50, 50))
    se = e.sum()
    obj = (mc + np.log(se) / T) * dx
    center = n - 1
    j = np.arange(n)
    grad = np.zeros(n)
    sw = e / se
    for oi in range(mk):
        w = sw[oi]
        if w < 1e-12:
            continue
        shift = idx[oi] - center
        if shift >= 0:
            m1 = j >= shift
            gi = j[m1] - shift
            grad[m1] += w * (1.0 - x[gi])
            m2 = (j + shift) < n
            gj = j[m2] + shift
            grad[m2] -= w * x[gj]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            gi = j[m1] + s2
            grad[m1] += w * (1.0 - x[gi])
            m2 = j >= s2
            gj = j[m2] - s2
            grad[m2] -= w * x[gj]
    grad *= dx
    return obj, grad

def slsqp_run(h, n, T, k, maxiter, ftol):
    target = n / 2.0
    def obj(x):
        return lse_obj_grad(x, n, T, k)[0]
    def jac(x):
        return lse_obj_grad(x, n, T, k)[1]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def subgrad_opt(h, n, t_end, n_lags=5, n_iter=50000, lr0=0.015, decay=0.9997):
    dx = 2.0 / n
    h_best = h.copy()
    c5_best = compute_c5(h)
    h_cur = h.copy()
    lr = lr0
    j = np.arange(n)
    center = n - 1
    for it in range(n_iter):
        if time.time() > t_end:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        nl = min(n_lags, len(corr))
        idx = np.argpartition(corr, -nl)[-nl:]
        topk = corr[idx]
        mc = np.max(topk)
        sw = np.exp((topk - mc) * 8.0)
        sw /= sw.sum()
        grad = np.zeros(n)
        for oi in range(nl):
            w = sw[oi]
            if w < 1e-12:
                continue
            shift = idx[oi] - center
            if shift >= 0:
                m1 = j >= shift
                gi = j[m1] - shift
                grad[m1] += w * (1.0 - h_cur[gi])
                m2 = (j + shift) < n
                gj = j[m2] + shift
                grad[m2] -= w * h_cur[gj]
            else:
                s2 = -shift
                m1 = (j + s2) < n
                gi = j[m1] + s2
                grad[m1] += w * (1.0 - h_cur[gi])
                m2 = j >= s2
                gj = j[m2] - s2
                grad[m2] -= w * h_cur[gj]
        grad *= dx
        h_cur = project_feasible(h_cur - lr * grad, n)
        lr *= decay
        if (it + 1) % 4000 == 0:
            c5 = compute_c5(h_cur)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_cur.copy()
    return h_best, c5_best

def binary_local_search(h, n, rng, n_trials=500):
    c5_cur = compute_c5(h)
    h_best = h.copy()
    for _ in range(n_trials):
        i = rng.integers(0, n)
        h_try = h_best.copy()
        h_try[i] = 1.0 - h_best[i]
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < c5_cur - 1e-10:
            h_best = h_try
            c5_cur = c5_try
    return h_best, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Coarse-to-fine SLSQP
    for T, k, mi in [(32, 40, 3000), (16, 20, 3000), (8, 12, 4000),
                     (4, 8, 4000), (2, 6, 5000), (1, 5, 5000),
                     (0.5, 4, 5000), (0.25, 3, 4000), (0.1, 3, 4000)]:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        h_new = slsqp_run(h_best, n, T, k, mi, 1e-20)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Subgradient with multiple lags
    rem = budget_s - (time.time() - t0)
    if rem > 200:
        h_sg, c5_sg = subgrad_opt(h_best, n, t0 + min(rem * 0.4, 350), n_lags=5, n_iter=70000, lr0=0.012, decay=0.9998)
        h_sg = project_feasible(h_sg, n)
        c5_sg = compute_c5(h_sg)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase 3: Binary local search
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h_bs, c5_bs = binary_local_search(h_best, n, rng, n_trials=max(200, int(rem * 3)))
        if c5_bs < c5_best:
            c5_best = c5_bs
            h_best = h_bs

    # Phase 4: Fine SLSQP after local search
    for T, k in [(0.5, 4), (0.2, 3), (0.1, 3)]:
        rem = budget_s - (time.time() - t0)
        if rem < 50:
            break
        h_new = slsqp_run(h_best, n, T, k, 3000, 1e-20)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Try n=560
    rem = budget_s - (time.time() - t0)
    if rem > 250:
        n2 = 560
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = project_feasible(h_hi, n2)
        c5_hi = compute_c5(h_hi)
        for T, k, mi in [(16, 30, 2500), (8, 15, 3000), (4, 10, 3000), (2, 6, 3500), (1, 5, 4000), (0.5, 4, 3000)]:
            rem2 = budget_s - (time.time() - t0)
            if rem2 < 50:
                break
            h_new = slsqp_run(h_hi, n2, T, k, mi, 1e-19)
            h_new = project_feasible(h_new, n2)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
                n = n2
                h_hi = h_best
        rem3 = budget_s - (time.time() - t0)
        if rem3 > 60 and n == 560:
            h_sg2, c5_sg2 = subgrad_opt(h_best, 560, t0 + rem3 * 0.7, n_lags=4, n_iter=30000, lr0=0.008, decay=0.9999)
            h_sg2 = project_feasible(h_sg2, 560)
            c5_sg2 = compute_c5(h_sg2)
            if c5_sg2 < c5_best:
                c5_best = c5_sg2
                h_best = h_sg2

    # Phase 6: Final subgradient polish at whatever n we have
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_sg3, c5_sg3 = subgrad_opt(h_best, n, t0 + rem * 0.85, n_lags=3, n_iter=40000, lr0=0.005, decay=0.9999)
        h_sg3 = project_feasible(h_sg3, n)
        c5_sg3 = compute_c5(h_sg3)
        if c5_sg3 < c5_best:
            c5_best = c5_sg3
            h_best = h_sg3

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
