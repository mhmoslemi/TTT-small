import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def _topk_smooth(x, n, T, k):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = np.max(topk)
    sh = T * (topk - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    obj = (mc + np.log(e.sum()) / T) * (2.0 / n)
    center = n - 1
    grad = np.zeros(n)
    j = np.arange(n)
    for oi in range(mk):
        w = sw[oi]
        if w < 1e-11:
            continue
        shift = idx[oi] - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += w * (1.0 - x[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= w * x[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += w * (1.0 - x[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * x[j[m2] - s2]
    grad *= 2.0 / n
    return obj, grad

def slsqp_opt(h, n, T, k, maxiter, ftol):
    target = n / 2.0
    def obj_func(x):
        return _topk_smooth(x, n, T, k)[0]
    def jac_func(x):
        return _topk_smooth(x, n, T, k)[1]
    def cf(x):
        return x.sum() - target
    def cj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj_func, h, method='SLSQP', jac=jac_func, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def gradient_topk(h, n, k_active):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    ka = min(k_active, len(corr))
    idx_a = np.argpartition(corr, -ka)[-ka:]
    grad = np.zeros(n)
    center = n - 1
    j = np.arange(n)
    for oi in idx_a:
        shift = oi - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += (1.0 - h[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= h[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= h[j[m2] - s2]
    grad *= dx / ka
    return grad, idx_a

def gradient_guided_flip(h, n, rng, n_flips=8):
    grad, _ = gradient_topk(h, n, 10)
    h_p = h.copy()
    candidates = np.argsort(np.abs(grad))
    for _ in range(n_flips):
        i = candidates[_ % len(candidates)]
        if h_p[i] > 0.5:
            h_p[i] -= rng.uniform(0.3, 0.7)
        else:
            h_p[i] += rng.uniform(0.3, 0.7)
    return project_feasible(h_p, n)

def block_flip(h, n, rng):
    h_p = h.copy()
    bstart = rng.integers(0, max(1, n - 20))
    blen = rng.integers(3, min(20, n - bstart))

    assert abs(h_best.sum() - n / 2.0) < 1e-8, f"Sum={h_best.sum()}, target={n/2}"
    assert h_best.min() >= -1e-10 and h_best.max() <= 1 + 1e-10
    h_best = np.clip(h_best, 0.0, 1.0)
    # Re-project to ensure exact sum
    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, n)
