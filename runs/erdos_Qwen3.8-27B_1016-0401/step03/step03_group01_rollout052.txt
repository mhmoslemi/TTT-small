def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def lse_objective(h, n, T):
    """Compute log-sum-exp smoothed max of correlations and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Log-sum-exp for numerical stability
    mc = np.max(corr)
    e = np.exp((corr - mc) * T)
    sw = e / e.sum()
    obj = mc + np.log(e.sum()) / T
    
    # Gradient
    center = n - 1
    j = np.arange(n)

        shift = k - center
        if w < 1e-10:

            grad[m1] += w * (1.0 - h[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= w * h[j[m2] + shift]
        else:

            m1 = (j + s2) < n
            grad[m1] += w * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * h[j[m2] - s2]
    return obj, grad

    return project_feasible(p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(280)
        h[::2] = 1.0
    n = len(h)
    h = project_feasible(h, n)
    c5b = compute_c5(h)

    # Phase 1: Nesterov warm-up (25% budget)
    t1 = t0 + budget_s * 0.25
    h, c5b = nesterov_opt(h, n, t1, k=3, lr0=0.02)

    # Phase 2: Progressive SLSQP (30% budget)
    t2 = t0 + budget_s * 0.55
    for T, k, mi in [(20, 5, 2000), (10, 4, 3000), (5, 3, 4000), (2, 3, 4000), (1, 2, 5000)]:
        if time.time() > t2:
            break
        h2 = slsqp_step(h, n, T, k, mi)
        c5 = compute_c5(h2)
        if c5 < c5b:
            c5b, h = c5, h2

    # Phase 3: Restart loop with small perturbations (35% budget)
    t3 = t0 + budget_s * 0.90
    n_r = max(2, int((t3 - time.time()) // 14))
    for ri in range(n_r):
        if time.time() > t3:
            break
        hp = perturb(h, n, rng, ri % 6)
        hp, _ = nesterov_opt(hp, n, t0 + (t3 - time.time()) * 0.35, k=3, lr0=0.012)
        hp = slsqp_step(hp, n, 3, 2, 2000)
        c5 = compute_c5(hp)
        if c5 < c5b:
            c5b, h = c5, hp

    # Phase 4: Final exact-max polish (remaining budget)
    rem = max(5.0, budget_s - (time.time() - t0))
    h2 = slsqp_step(h, n, 1.0, 1, max(2000, int(rem * 2)))
    c5 = compute_c5(h2)
    if c5 < c5b:
        c5b, h = c5, h2
    h, c5b = exact_max_step(h, n, steps=200, lr=0.003)

    # Phase 5: Try n=200 as alternative resolution
    rem = max(5.0, budget_s - (time.time() - t0))
    if rem > 60:
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.arange(200) * 0.01
        h200 = np.interp(x_new, x_old, h)
        h200 = project_feasible(h200, 200)
        h200, c5_200 = nesterov_opt(h200, 200, t0 + min(30, rem * 0.4), k=3, lr0=0.02)
        h200 = slsqp_step(h200, 200, 2, 2, 3000)
        c5_200 = compute_c5(h200)
        if c5_200 < c5b:
            c5b, h, n = c5_200, h200, 200

    h = project_feasible(h, n)
    c5_final = compute_c5(h)
    return (h, min(c5_final, c5b), n)
