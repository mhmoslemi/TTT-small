import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad_exact(h, n, shift):
    """Gradient of the overlap integral at a given shift."""
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = (j >= shift) & (j - shift < n)
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def nesterov_max(h, n, t0, tlim, max_iters=200000):
    """Nesterov accelerated gradient descent with exact max."""
    dx = 2.0 / n
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    f_star = 0.37
    
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        beta = it / (it + 2.5)
        y = h_curr + beta * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        corr = np.correlate(y, 1.0 - y, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = shift_grad_exact(y, n, shift) * dx
        
        f_y = corr[idx] * dx
        gap = max(f_y - f_star, 1e-8)
        gn_sq = float(g @ g)
        alpha = min(2.0 * gap / (gn_sq + 1e-12), 0.5 / (1.0 + 0.001 * it))
        
        h_new = project_feasible(y - alpha * g, n)
        h_prev = h_curr.copy()
        h_curr = h_new
        
        if (it + 1) % 5000 == 0:
            c5_now = compute_c5(h_curr)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h_curr.copy()
    
    c5_final = compute_c5(h_curr)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_curr.copy()
    return best_h, best_c5

def eval_sg(x, n, T, K, t0, tl):
    if time.time() - t0 > tl:

        cs = corr[idx]

        cs = corr
    mcs = cs.max()
    sh = T * (cs - mcs)

def nesterov(h, n, t0, tl, maxit=500000):
    dx = 2.0 / n
    hc = h.copy()
    hp = h.copy()
    bh = h.copy()
    bc = compute_c5(h)
    fs = 0.37
    for it in range(maxit):
        if time.time() - t0 > tl:
            break
        beta = it / (it + 2.5)
        y = np.clip(hc + beta * (hc - hp), 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode="full")
        idx = int(corr.argmax())
        g = shift_grad(y, n, idx - (n - 1)) * dx
        fy = corr[idx] * dx
        gap = max(fy - fs, 1e-7)
        gn = float(g @ g)
        a = min(2.0 * gap / (gn + 1e-12), 0.3 / (1.0 + 0.001 * it))
        hn = project_feasible(y - a * g, n)
        hp = hc.copy()
        hc = hn
        if (it + 1) % 5000 == 0:
            c = compute_c5(hc)
            if c < bc:
                bc = c
                bh = hc.copy()
    c = compute_c5(hc)
    if c < bc:
        bc = c
        bh = hc.copy()
    return bh, bc
