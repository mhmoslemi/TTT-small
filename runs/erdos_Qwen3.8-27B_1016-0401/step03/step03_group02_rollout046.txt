    if c5_s < c5_best:
        c5_best, h_best = c5_s, h_best
    corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
    for li_i in np.argsort(corr_f)[-8:]:
        if time.time() > t0 + budget_s - 5:
            break
        hn = _slsqp_lag(h_best, int(li_i), tgt, bnds, 2000, 1e-19)
        cv = _c5(hn)
        if cv < c5_best:
            c5_best, h_best = cv, hn
    h_best = _proj(h_best, tgt)
    c5_fin = min(_c5(h_best), c5_best)
    return (h_best, c5_fin, len(h_best))
