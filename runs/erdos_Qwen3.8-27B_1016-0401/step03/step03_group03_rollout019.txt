def project_feasible(y, n):
    target = n / 2.0

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:

        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])

    dx = 2.0 / n

    c5_cur = compute_c5(h_cur)
