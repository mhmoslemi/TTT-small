```python
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:

        else:
            hi = mid

def round_to_binary(h, n):
    """Round h to {0,1} with exactly n/2 ones."""
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        # Need to flip some 1s to 0s
        ones_idx = np.where(h_bin == 1)[0]
        # Flip the ones with lowest h values
        order = np.argsort(h[ones_idx])
        for i in range(n_ones - target):

        order = np.argsort(-h[zeros_idx])
        for i in range(target - n_ones):
            h_bin[zeros_idx[order[i]]] = 1
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]

    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        active = sw > 1e-10
        for oi in np.where(active)[0]:
            grad += sw[oi] * shift_grad(x, n, oi - center)
        return grad * dx
    
    bounds = [(0.0, 1.0)] * n

                   'jac': lambda x: np.ones(n)}]

                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})

            h_cur[j] = 0
        
        it += 1
```

The issue is that we're doing a FULL correlate every time. Can we do better?

For a single pair flip (i: 1→0, j: 0→1), the change in corr[k] for each k can be computed analytically. This would make each evaluation O(n) instead of O(n²).

For flip at position i (1→0):
- affects corr[k] for all k where the pair (i, i+k) or (i+k, i) is in the correlation sum
- Specifically, corr contains the sum h[n+s]*(1-h[n]) for lag s
- i contributes to: as h[n+s] when n = i-s (contribution was (1-h[i-s]), now 0): change = -(1-h[i-s]) for valid s
  - And as (1-h[n]) when n = i (contribution was h[i+s]*(-1), now 0): change = h[i+s] for valid s

For flip at position j (0→1):
- j now contributes: as h[n+s] when n = j-s: change = +(1-h[j-s]) for valid s
  - And as (1-h[n]) when n = j: change = h[j+s]*(-1) for valid s

This is getting complex but would give O(n) per flip. However, implementing it correctly is error-prone. Let me just use the full correlate for simplicity, since n=280 is small enough.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def slsqp(h, n, T, maxit, ftol):
    target = n / 2.0
    dx = 2.0 / n
    ct = n - 1
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        mc = c.max()
        return (mc + np.log(np.exp(T * (c - mc)).sum()) / T) * dx
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        mc = c.max()
        w = np.exp(T * (c - mc))
        w /= w.sum()
        g = np.zeros(n)
        act = w > 1e-12
        for oi in np.where(act)[0]:
            g += w[oi] * shift_grad(x, n, oi - ct)
        return g * dx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                     constraints=cons, options={'maxiter': maxit, 'ftol': ftol})
        return r.x
    except:
        return h.copy()

def binarize(h, n):
    hb = (h >= 0.5).astype(float)
    t = n // 2
    no = int(hb.sum())
    if no > t:
        idx = np.where(hb == 1)[0]
        ord_ = np.argsort(h[idx])
        hb[idx[ord_[:no - t]]] = 0
    elif no < t:
        idx = np.where(hb == 0)[0]
        ord_ = np.argsort(-h[idx])
        hb[idx[ord_[:t - no]]] = 1
    return hb

def binary_anneal(h, n, t0, budget_s, rng):
    dx = 2.0 / n
    hc = h.copy()
    c5c = compute_c5(hc)
    hb = hc.copy()
    c5b = c5c
    T0, Tmin = 0.04, 0.0005
    T = T0
    tcool = 0.99995
    ones = np.where(hc == 1)[0]
    zeros = np.where(hc == 0)[0]
    while time.time() - t0 < budget_s:
        T = max(Tmin, T * tcool)
        # Weighted sampling: bias toward worst-neighborhood positions
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        hc[i], hc[j] = 0.0, 1.0
        c5n = compute_c5(hc)
        d = c5n - c5c
        if d < 0 or rng.random() < np.exp(-d / T):
            c5c = c5n
            ones = np.where(hc == 1)[0]
            zeros = np.where(hc == 0)[0]
            if c5n < c5b:
                c5b = c5n
                hb = hc.copy()
        else:
            hc[i], hc[j] = 1.0, 0.0
    return hb, c5b

def focused_slsqp(h, n, K, maxit, ftol):
    # Focus on top-K shifts
    target = n / 2.0
    dx = 2.0 / n
    ct = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_k = np.argpartition(corr, -K)[-K:]
    w_init = corr[top_k] / max(corr[top_k].max(), 1e-15)
    w_init /= w_init.sum()
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        return np.dot(w_init, c[top_k]) * dx
    
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        mc = c.max()
        # Adaptive weights: recompute
        w = np.maximum(c[top_k] - mc + 1, 1e-6)
        w = w ** 4
        w /= w.sum()
        g = np.zeros(n)
        for i in range(K):
            g += w[i] * shift_grad(x, n, top_k[i] - ct)
        return g * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                     constraints=cons, options={'maxiter': maxit, 'ftol': ftol})
        return r.x
    except:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    
    # Phase 1: Quick SLSQP (40s)
    for T in [20, 10]:
        if time.time() - t0 > budget_s - 800:
            break
        h_new = slsqp(h_best, n, T, 3000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    
    # Phase 2: Binary local search (main, ~500s)
    h_bin = binarize(h_best, n)
    c5_bin = compute_c5(h_bin)
    h_al, c5_al = binary_anneal(h_bin, n, t0, budget_s - 350, rng)
    if c5_al < c5_best:
        c5_best = c5_al
        h_best = h_al.copy()
    
    # Phase 3: SLSQP continuous polish from binary (150s)
    if time.time() - t0 < budget_s - 200:
        for T in [30, 15, 8]:
            if time.time() - t0 > budget_s - 150:
                break
            h_new = slsqp(h_best, n, T, 5000, 1e-16)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
    
    # Phase 4: Focused SLSQP on worst shifts (100s)
    for K in [3, 5, 8]:
        if time.time() - t0 > budget_s - 60:
            break
        h_new = focused_slsqp(h_best, n, K, 4000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    
    # Phase 5: Multi-start perturbations (final)
    for pi in range(15):
        if time.time() - t0 > budget_s - 20:
            break
        pert = 0.02 * rng.normal(0, 1, n)
        h_p = project_feasible(h_best + pert, n)
        h_new = slsqp(h_p, n, 10, 3000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    
    # Final
    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.clip(y - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

def shift_grad(h, n, s):
    j = np.arange(n)
    g = np.zeros(n)
    if s >= 0:
        m1, m2 = j >= s, (j + s) < n
        if m1.any(): g[m1] += 1.0 - h[j[m1] - s]
        if m2.any(): g[m2] -= h[j[m2] + s]
    else:
        t = -s
        m1, m2 = (j + t) < n, j >= t
        if m1.any(): g[m1] += 1.0 - h[j[m1] + t]
        if m2.any(): g[m2] -= h[j[m2] - t]
    return g

def slsqp(h0, n, T, mi, ft):
    tgt, dx, ct = n/2, 2.0/n, n-1
    def obj(x):
        c = np.correlate(x, 1-x, 'full')
        mc = c.max()
        return (mc + np.log(np.sum(np.exp(T*(c-mc))))/T)*dx
    def jac(x):
        c = np.correlate(x, 1-x, 'full')
        mc = c.max()
        w = np.exp(T*(c-mc)); w /= w.sum()
        g = np.zeros(n)
        for oi in np.where(w > 1e-12)[0]:
            g += w[oi]*shift_grad(x, n, oi-ct)
        return g*dx
    bnds = [(0,1)]*n
    cons = [{'type':'eq','fun':lambda x: x.sum()-tgt,'jac':lambda x: np.ones(n)}]
    try:
        return minimize(obj, h0, method='SLSQP', jac=jac, bounds=bnds,
                        constraints=cons, options={'maxiter':mi,'ftol':ft}).x
    except: return h0.copy()

    elif no < tgt:
```

```python
            c5c = compute_c5(h)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h.copy()
        eta = 0.5/(1+0.001*count)
    return h_best, c5_best
```

This is similar to what the previous code did with Nesterov. The key difference would be in the step size schedule and the smoothness parameter T.

Actually, let me think about what's really limiting the previous approach. The Nesterov method in the previous code used:
- K up to 50 shifts (top-K weighting)
- Alpha = 0.6/(1+0.002*it)
- Projected gradient with momentum

For this to work well, the step size needs to be in the right range. Too large → oscillation. Too small → slow convergence.

One key improvement: **adaptive step size via backtracking line search**. At each iteration, try the step, and if C₅ increases, halve the step size.

Another key improvement: **better handling of the max**. Instead of a fixed T for the log-sum-exp, use T that adapts:
- Early iterations: large T (smooth, gradient is well-defined)
- Late iterations: small T (close to true max, sharper gradient)
