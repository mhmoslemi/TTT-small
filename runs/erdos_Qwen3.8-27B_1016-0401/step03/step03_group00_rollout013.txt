def binarize(h, n):
    """Convert to binary with exactly n/2 ones."""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        # Flip some 1s to 0s (prefer those closest to 0.5)
        one_idx = np.where(h_bin == 1)[0]
        # Sort by h value (lower h means closer to 0.5 boundary)
        one_idx = one_idx[np.argsort(h[one_idx])]
        for i in range(n_ones - target):
            h_bin[one_idx[i]] = 0
    elif n_ones < target:

        zero_idx = zero_idx[np.argsort(-h[zero_idx])]
        for i in range(target - n_ones):
            h_bin[zero_idx[i]] = 1
    return h_bin

def two_flip_anneal(h, n, rng, t_start, t_end):

    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_corr = get_corr(h).copy()
    
    elapsed = 0
    total_time = t_end - t_start
    T = 0.05  # initial temperature
    
    while time.time() < t_end:
        # Adaptive temperature
        frac = (time.time() - t_start) / total_time
        T = 0.05 * (0.01 ** frac)
        
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        if len(one_idx) < 2 or len(zero_idx) < 2:
            break
        
        a = one_idx[rng.integers(len(one_idx))]
        b = zero_idx[rng.integers(len(zero_idx))]
        
        h[a] = 0
        h[b] = 1
        
        new_corr = get_corr(h)
        new_max = float(np.max(new_corr) * dx)
        delta = new_max - best_c5
        
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            best_corr = new_corr
            if new_max < best_c5:
                best_c5 = new_max
                best_h = h.copy()
        else:
            h[a] = 1
            h[b] = 0
    
    return best_h, best_c5

def slsqp_optimize(h, n, T_smooth, top_k, maxiter):
    """SLSQP with smooth max approximation."""
    dx = 2.0 / n
    
    def obj(x):
        corr = get_corr(x)
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        m = c.max()
        if T_smooth < 1e-10:
            s = m + (c[c > m * 0.99] - m).max() if (c > m * 0.99).any() else m
        else:
            e = np.exp(T_smooth * (c - m))
            s = m + np.log(e.sum()) / T_smooth
        return s * dx
    
    def jac(x):
        corr = get_corr(x)
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        m = c.max()
        if T_smooth < 1e-10:
            # Subgradient: average of active shifts
            active = idx[c >= m - 1e-6]
            w = np.ones(len(active)) / len(active)
        else:
            e = np.exp(T_smooth * (c - m))
            w = e / e.sum()
        
        g = np.zeros(n)
        for wi, ci in zip(w, idx):
            if wi > 1e-10:
                s = ci - (n - 1)  # actual shift
                # grad of corr at shift s
                if s >= 0:
                    valid = np.arange(0, n - s)
                    g[valid] += wi * (1.0 - x[valid + s])
                    g[valid + s] -= wi * x[valid]
                else:
                    ss = -s
                    valid = np.arange(0, n - ss)
                    g[valid + ss] += wi * (1.0 - x[valid])
                    g[valid] -= wi * x[valid + ss]
        return g * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return res.x
    except Exception:
        return h.copy()

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")

def get_corr(h):

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0

        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()

        else:

    n = _G_N
    g = np.zeros(n)
    if shift >= 0:
        valid = np.arange(0, n - shift)
