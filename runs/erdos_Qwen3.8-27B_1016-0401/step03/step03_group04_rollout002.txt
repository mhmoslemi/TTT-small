import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def topk_gradient(h, n, K, T=None):
    """Compute smoothed max value and gradient."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    dx = 2.0 / n
    
    if T is None or T < 1.0:
        # Use top-K with uniform weights
        idx = np.argpartition(corr, -K)[-K:]
        c_sel = corr[idx]
        mc = np.max(c_sel)
        val = mc * dx
        # Gradient: average of gradients of top-K
        g = np.zeros(n)
        for ii in range(K):
            g += shift_grad(h, n, idx[ii] - center)
        g *= dx / K
        return val, g
    else:
        # Log-sum-exp smoothing
        topk_idx = np.argpartition(corr, -min(K, len(corr)))[-min(K, len(corr)):]
        c_sel = corr[topk_idx]
        mc_sel = np.max(c_sel)
        sh = T * (c_sel - mc_sel)
        e = np.exp(sh)
        sw = e / e.sum()
        val = (mc_sel + np.log(sw.sum()) / T) * dx
        g = np.zeros(n)
        for ii in range(len(sw)):
            if sw[ii] > 1e-15:
                g += sw[ii] * shift_grad(h, n, topk_idx[ii] - center)
        g *= dx
        return val, g

def prod_subgrad(h_init, n, t0, tlim, max_iters=300000, K=50, T_init=10.0):
    """Proximal subgradient with momentum and adaptive parameters."""
    h_curr = h_init.copy()
    h_mom = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    dx = 2.0 / n
    f_star = 0.375  # known lower bound
    
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        T = max(1.0, T_init * 0.9995**it)  # anneal temperature
        K_eff = min(K, 2*n-1)
        
        y = h_curr + 0.9 * (h_curr - h_mom) if it > 0 else h_curr
        y = np.clip(y, 0.0, 1.0)
        
        val, g = topk_gradient(y, n, K_eff, T)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        gap = max(val - f_star, 1e-8)
        alpha = 2.0 * gap / (gn * gn + 1e-12)
        alpha = min(alpha, 0.5 / (1.0 + 0.01 * it))
        
        h_new = y - alpha * g
        h_new = project_feasible(h_new, n)
        h_mom = h_curr.copy()
        h_curr = h_new
        
        if (it + 1) % 5000 == 0:
            c5_now = compute_c5(h_curr)
            if c5_now < c5_best:

    if c5_final < c5_best:

    return h_best, c5_best

def block_cd(h, n, t0, tlim, block_size=6, max_sweeps=20):
    """Block coordinate descent."""
    h_curr = h.copy()
    c5_best = compute_c5(h_curr)
    h_best = h_curr.copy()
    n_blocks = (n + block_size - 1) // block_size
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > tlim:
            break
        for bi in range(n_blocks):
            if time.time() - t0 > tlim:
                break
            i0 = bi * block_size
            i1 = min(i0 + block_size, n)
            B = i1 - i0
            h_block = h_curr[i0:i1].copy()
            
            # Local SLSQP on this block
            def local_obj(xb):
                h_temp = h_curr.copy()
                h_temp[i0:i1] = xb
                return compute_c5(h_temp)
            
            bounds = [(0.0, 1.0)] * B
            # Equality: sum of block + sum of rest = n/2
            rest_sum = h_curr.sum() - h_block.sum()
            def req(xb):
                return xb.sum() + rest_sum - n / 2.0
            
            try:
                res = minimize(local_obj, h_block, method='SLSQP',
                             bounds=bounds,
                             constraints=[{'type': 'eq', 'fun': req, 'jac': lambda x: np.ones(B)}],
                             options={'maxiter': 100, 'ftol': 1e-16})
                h_curr[i0:i1] = np.clip(res.x, 0.0, 1.0)
            except:
                pass
            # Re-project for safety
            h_curr = project_feasible(h_curr, n)
        
        c5_now = compute_c5(h_curr)
        if c5_now < c5_best:
            c5_best = c5_now
            h_best = h_curr.copy()
    
    return h_best, c5_best

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):
    def obj(x):
        v, _ = topk_gradient(x, n, K, T)
        return v
    def jac(x):
        _, g = topk_gradient(x, n, K, T)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jac,
                     bounds=bounds,
                     options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol*0.1,
                              'maxls': 30, 'maxfun': maxiter*3})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def do_slsqp(h, n, T, K, maxiter, tol, t0, tlim):
    def obj(x):
        v, _ = topk_gradient(x, n, K, T)
        return v
    def jac(x):
        _, g = topk_gradient(x, n, K, T)
        return g
    def cfun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': lambda x: np.ones(len(x))}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': tol})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def fourier_perturb(h, n, rng):
    """Perturb in Fourier domain to maintain structure."""
    h_dev = h - 0.5  # deviation from 1/2
    spec = np.fft.rfft(h_dev)
    n_freq = len(spec)
    # Randomize phases in high frequencies, damp in low
    for i in range(1, n_freq):
        damp = rng.uniform(0.9, 1.1)
        phase = rng.uniform(-0.1, 0.1)
        spec[i] *= damp * (np.cos(phase) + 1j * np.sin(phase))
    h_new = 0.5 + np.fft.irfft(spec, n=n)
    return np.clip(h_new, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def smooth_val_grad(h, n, T, K):
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    dx = 2.0 / n
    Ke = min(K, len(corr))
    idx = np.argpartition(corr, -Ke)[-Ke:]
    c_sel = corr[idx]
    mc = np.max(c_sel)
    sh = T * (c_sel - mc)
    sh -= sh.max()
    e = np.exp(sh)
    sw = e / e.sum()
    val = (mc + np.log(e.sum()) / T) * dx
    g = np.zeros(n)
    for ii in range(len(sw)):
        if sw[ii] > 1e-14:
            g += sw[ii] * shift_grad(h, n, idx[ii] - center)
    return val * 1.0, g * dx

def prox_subgrad(h_init, n, t0, tlim, max_it, K, T0, T1, rng):

    best_v, best_h = compute_c5(h_init), h_init.copy()
    for it in range(max_it):
        if time.time() - t0 > tlim:

        fr = it / max(1, max_it)
        T = T0 * (T1 / T0) ** fr

        y = h_c + beta * (h_c - h_p) if it > 0 else h_c
        y = np.clip(y, 0.0, 1.0)

        gn = np.linalg.norm(g)

        h_new = project_feasible(y - a * g, n)
        h_p = h_c.copy()
        h_c = h_new
        if (it + 1) % 4000 == 0:
            v = compute_c5(h_c)
            if v < best_v:
                best_v, best_h = v, h_c.copy()
    v_end = compute_c5(h_c)
    if v_end < best_v:

    return best_h, best_v

def lbfgsb_opt(h, n, T, K, miter, tol, t0, tlim):
    def obj(x):
        v, _ = smooth_val_grad(x, n, T, K)
        return v
    def jac(x):
        _, g = smooth_val_grad(x, n, T, K)
        return g
    bds = [(0.0, 1.0)] * n
    try:
        r = minimize(obj, h, method='L-BFGS-B', jac=jac, bounds=bds,
                     options={'maxiter': miter, 'ftol': tol, 'gtol': tol * 0.1,
                              'maxls': 30, 'maxfun': miter * 3})
        return project_feasible(r.x, n)
    except Exception:
        return h.copy()

def slsqp_opt(h, n, T, K, miter, tol, t0, tlim):
    def obj(x):
        v, _ = smooth_val_grad(x, n, T, K)
        return v
    def jac(x):
        _, g = smooth_val_grad(x, n, T, K)
        return g

        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(len(x))

def block_cd(h, n, t0, tlim, bs, sweeps):

    best_v, best_h = compute_c5(h_c), h_c.copy()
