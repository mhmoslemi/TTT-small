    # Phase 1: Coarse SLSQP (high T for global exploration)
    for T in [48, 24, 12]:
        if time.time() - t0 > budget_s - 600:
            break
        h_new = slsqp_pass(h_best, n, T, 2500, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Nesterov with periodic SLSQP restarts
    t_lim = budget_s - 300
    h_best, c5_best = nesterov_with_restarts(h_best, n, t0, t_lim, 600000)

    # Phase 3: Focused SLSQP top-K
    for T, K in [(8, 20), (4, 10), (3, 6), (2, 4), (2, 3)]:
        if time.time() - t0 > budget_s - 180:
            break
        h_new = slsqp_pass(h_best, n, T, 4000, 1e-15, topK=K)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Very fine SLSQP
    for T in [1.5, 1.0]:
        if time.time() - t0 > budget_s - 100:
            break
        h_new = slsqp_pass(h_best, n, T, 6000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Reheating - small perturb + re-optimize
    for pi in range(25):
        if time.time() - t0 > budget_s - 30:
            break
        sig = 0.005 + 0.008 * (pi % 4)
        h_p = h_best + rng.normal(0, sig, n)
        h_p = project_feasible(h_p, n)
        h_new = slsqp_pass(h_p, n, 3, 2500, 1e-15, topK=10)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Try alternate n if time remains
    remaining = budget_s - (time.time() - t0)
    if remaining > 250:
        for n2 in [n + 8, n + 16, n - 8]:
            if n2 < 60 or n2 > 600 or n2 % 2 != 0:
                continue
            if time.time() - t0 > budget_s - 80:
                break
            x_old = np.linspace(0, 2.0, n, endpoint=False)
            x_new = np.linspace(0, 2.0, n2, endpoint=False)
            h_alt = np.interp(x_new, x_old, h_best)
            h_alt = project_feasible(h_alt, n2)
            for T2, K2 in [(8, 15), (3, 6)]:
                if time.time() - t0 > budget_s - 40:
                    break
                h_new = slsqp_pass(h_alt, n2, T2, 3000, 1e-15, topK=K2)
                h_new = project_feasible(h_new, n2)
                c5_alt = compute_c5(h_new)
                if c5_alt < c5_best:
                    c5_best = c5_alt
                    h_best = h_new
                    n = n2

    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, n)
