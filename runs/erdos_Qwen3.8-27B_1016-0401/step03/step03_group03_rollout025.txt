def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0

        if s > target:

def shift_grad(h, n, s):
    g = np.zeros(n)
    lo1 = max(0, s)
    hi1 = min(n, n + s)
    if lo1 < hi1:
        j1 = np.arange(lo1, hi1)
        g[j1] += 1.0 - h[j1 - s]
    lo2 = max(0, -s)
    hi2 = min(n, n - s)
    if lo2 < hi2:
        j2 = np.arange(lo2, hi2)
        g[j2] -= h[j2 + s]

# Global state for SLSQP
_N = 0
_T = 1.0

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    sh = _T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _T) * (2.0 / _N)

def _jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    sh = _T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _N - 1
    dx = 2.0 / _N
    grad = np.zeros(_N)
    for oi in range(len(sw)):
        if sw[oi] < 1e-14:
            continue
        grad += sw[oi] * shift_grad(x, _N, oi - center)
    return grad * dx

def _cons_fun(x):
    return x.sum() - _N / 2.0

    return np.ones(_N)

    _T = T
