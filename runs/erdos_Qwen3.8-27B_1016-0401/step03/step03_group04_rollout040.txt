import numpy as np
from scipy.optimize import minimize
import time

_CTX = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = float(y.min()) - 3.0, float(y.max()) + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m = j >= shift
        g[m] += (1.0 - h[j[m] - shift])
        m = (j + shift) < n
        g[m] -= h[j[m] + shift]
    else:
        s = -shift
        m = (j + s) < n
        g[m] += (1.0 - h[j[m] + s])
        m = j >= s
        g[m] -= h[j[m] - s]
    return g

def _sm_val_grad(x):
    n, T, K, dx = _CTX['n'], _CTX['T'], _CTX['K'], _CTX['dx']
    corr = np.correlate(x, 1.0 - x, mode="full")
    if K < len(corr):
        idx = np.argpartition(corr, -K)[-K:]
        c = corr[idx]
    else:
        idx = np.arange(len(corr))
        c = corr
    mc = c.max()
    sh = T * (c - mc)
    e = np.exp(sh)
    w = e / e.sum()
    val = (mc + np.log(e.sum()) / T) * dx
    center = n - 1
    g = np.zeros(n)
    for ii in range(len(w)):
        if w[ii] > 1e-12:
            g += w[ii] * shift_grad(x, n, idx[ii] - center)
    return val, g * dx

def _sm_obj(x):
    if _CTX.get('expired'):
        return 0.0
    v, _ = _sm_val_grad(x)
    return v

def _sm_jac(x):
    if _CTX.get('expired'):
        return np.zeros(_CTX['n'])
    _, g = _sm_val_grad(x)
    return g

def _c_eq_fun(x):
    return x.sum() - _CTX['n'] / 2.0

def _c_eq_jac(x):
    return np.ones(_CTX['n'])

def do_lbfgsb(h, n, T, K, maxiter, tol, tlim):
    _CTX.update(n=n, T=T, K=K, dx=2.0/n, expired=False)
    old = h.copy()
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_obj, h, method='L-BFGS-B', jac=_sm_jac,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol*0.1,
                                'maxls': 40, 'maxfun': maxiter*4})
        if time.time() > tlim:
            return old
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return old.copy()

def do_slsqp(h, n, T, K, maxiter, tol, tlim):
    _CTX.update(n=n, T=T, K=K, dx=2.0/n, expired=False)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_eq_fun, 'jac': _c_eq_jac}]
    try:
        res = minimize(_sm_obj, h, method='SLSQP', jac=_sm_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol, 'iprint': -1})
        if time.time() > tlim:
            return h.copy()
        return project_feasible(np.clip(res.x, 0, 1), n)
    except:
        return h.copy()

def greedy_refine(h, n, tlim, passes=3):
    h = h.copy()
    best = compute_c5(h)
    dx = 2.0 / n
    for p in range(passes):
        if time.time() > tlim - 5:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top = np.argsort(corr)[-5:]
        for _ in range(3):
            grad = np.zeros(n)
            for k in top:
                grad += shift_grad(h, n, k - (n - 1))
            grad *= dx
            # Try a few step sizes
            for alpha in [0.02, 0.01, 0.005, 0.002]:
                h_try = h - alpha * grad
                h_try = project_feasible(h_try, n)
                c = compute_c5(h_try)
                if c < best:
                    best = c
                    h = h_try
                    break
    return h, best

def border_refine(h, n, tlim):
    """Only optimize coordinates where 0.02 < h[i] < 0.98."""
    mask = (h > 0.02) & (h < 0.98)
    if mask.sum() < 3:
        return h, compute_c5(h)
    idx = np.where(mask)[0]
    h = h.copy()
    best = compute_c5(h)
    for T in [1.0, 0.3, 0.1]:
        if time.time() > tlim - 5:
            break
        _CTX.update(n=n, T=T, K=5, dx=2.0/n, expired=False)
        bounds = [(0.0, 1.0)] * n
        # Fix non-border coords
        fixed = ~mask
        h_try = h.copy()
        try:
            obj = _sm_obj
            jac = _sm_jac
            res = minimize(obj, h_try, method='L-BFGS-B', jac=jac,
                           bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-14, 'gtol': 1e-10,
                                    'maxls': 30, 'maxfun': 8000})
            if time.time() > tlim:
                break
            h_new = project_feasible(np.clip(res.x, 0, 1), n)
            c = compute_c5(h_new)
            if c < best:
                best = c
                h = h_new
                mask = (h > 0.02) & (h < 0.98)
                fixed = ~mask
        except:
            pass
    return h, best

def struct_perturb(h, n, rng):
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(4):
        f = rng.uniform(1, 10)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.005, 0.03)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert += rng.normal(0, 0.006, n)
    return h + pert
