def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:

        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

    n = len(corr)
    if K >= n:
        return np.arange(n)
    idx = np.argpartition(corr, -K)[-K:]
    return idx

        grad += 2.0 * d * shift_grad(h, n, sh) * dx * dx

            h_curr[j_idx] = h_curr[j_idx] + diff  # restore
            h_curr[j_idx] = np.clip(h_curr[j_idx], 0, 1)
    return h_best, c5_best

    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def sm_obj(x, n, T, K):
    dx = 2.0 / n

    center = n - 1

        c_sel = corr
    mc = np.max(c_sel)
    sh = T * (c_sel - mc)

    val = (mc + np.log(e.sum()) / T) * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def sm_val_grad(x, n, T, K):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    center = n - 1
    if K < len(corr):
        top_idx = np.argpartition(corr, -K)[-K:]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    mc = np.max(c_sel)
    sh = T * (c_sel - mc)
    e = np.exp(sh)
    val = (mc + np.log(e.sum()) / T) * dx
    sw = e / e.sum()
    grad = np.zeros(n)
    for ii in range(len(sw)):
        w = sw[ii]
        if w < 1e-15:
            continue
        grad += w * shift_grad(x, n, top_idx[ii] - center)
    return val, grad * dx

def do_lbfgsb(h, n, T, K, maxiter, t0, tlim):
    bounds = [(0.0, 1.0)] * n
    def f(x):
        v, _ = sm_val_grad(x, n, T, K)
        return v
    def g(x):
        _, gr = sm_val_grad(x, n, T, K)
        return gr
    try:
        res = minimize(f, h, method='L-BFGS-B', jac=g, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 40})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def do_slsqp(h, n, T, K, maxiter, t0, tlim):
    bounds = [(0.0, 1.0)] * n
    def f(x):
        v, _ = sm_val_grad(x, n, T, K)
        return v
    def g(x):
        _, gr = sm_val_grad(x, n, T, K)
        return gr
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(f, h, method='SLSQP', jac=g, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coord_descent(h, n, t0, tlim, max_rounds=600):
    dx = 2.0 / n
    h_curr = h.copy()
    c5_best = compute_c5(h_curr)
    h_best = h_curr.copy()
    stall = 0

        for i in order[:40]:
            if time.time() - t0 > tlim:

            old = h_curr[i]
            if old < 0.003 or old > 0.997:

            if g_comb[i] > 0:
                deltas = [0.008, 0.004, 0.002, 0.001, 0.0005]

            for d in deltas:

def subgrad_polish(h, n, t0, tlim, max_iters=20000):
    dx = 2.0 / n
    h_curr = h.copy()
    c5_best = compute_c5(h_curr)
    h_best = h_curr.copy()
    alpha = 0.02
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        top_k = np.argsort(corr)[-3:]
        g = np.zeros(n)
        for kk in top_k:
            g += shift_grad(h_curr, n, int(kk) - (n - 1))
        g /= len(top_k)
        moved = False
        for a in [alpha, alpha * 0.5, alpha * 0.1]:
            h_try = h_curr - a * g * dx
            h_try = project_feasible(h_try, n)
            c5t = compute_c5(h_try)
            if c5t < c5_best - 1e-15:
                h_curr = h_try
                c5_best = c5t
                h_best = h_try.copy()
                moved = True
                break
        if not moved:
            alpha *= 0.7
            if alpha < 1e-7:
                break
        else:
            alpha = min(alpha * 1.1, 0.05)
    return h_best, c5_best
