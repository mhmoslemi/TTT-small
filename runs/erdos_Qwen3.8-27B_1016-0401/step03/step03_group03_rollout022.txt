def do_slsqp_constrained(h_init, n, K, maxiter, ftol):
    # Variables: [h_0, ..., h_{n-1}, t]
    # Objective: t
    # Constraints: corr_k * dx <= t for k in top_K, sum(h) = n/2, 0<=h<=1, t>=0
    dx = 2.0 / n
    center = n - 1
    
    h_curr = h_init.copy()
    corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
    top_k = np.argsort(corr)[-K:]
    
    x0 = np.concatenate([h_curr, [float(corr.max() * dx) + 0.01]])
    
    def obj(x):
        return x[-1]
    
    def jac_obj(x):
        j = np.zeros(len(x))
        j[-1] = 1.0
        return j
    
    def corr_shift(h, shift):
        if shift >= 0:
            return np.sum(h[shift:] * (1.0 - h[:n-shift]))
        else:
            s = -shift
            return np.sum(h[s:] * (1.0 - h[:n-s]))
    
    # Build constraints for top-K shifts
    cons_list = []
    for ik in top_k:
        shift = ik - center
        def mk_fun(x, s=shift):
            h_part = x[:n]
            return corr_shift(h_part, s) * dx - x[-1]
        def mk_jac(x, s=shift):
            j = np.zeros(len(x))
            if s >= 0:
                j[s:] += (1.0 - x[:n-s]) * dx
                j[:n-s] -= x[s:] * dx
            else:
                ss = -s
                j[ss:] += (1.0 - x[:n-ss]) * dx
                j[:n-ss] -= x[ss:] * dx
            j[-1] = -1.0
            return j
        cons_list.append({'type': 'ineq', 'fun': mk_fun, 'jac': mk_jac})
    
    # Equality constraint: sum(h) = n/2
    def eq_fun(x):
        return x[:n].sum() - n/2.0
    def eq_jac(x):
        j = np.zeros(len(x))
        j[:n] = 1.0
        return j
    cons_list.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=jac_obj,
                       bounds=bounds, constraints=cons_list,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x[:n], res.x[-1]
    except Exception:
        return h_curr.copy(), float(corr.max() * dx)
