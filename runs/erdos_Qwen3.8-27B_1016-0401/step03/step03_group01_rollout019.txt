def project_feasible(y, n):

def topk_og(x, n, T, k):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = topk.max()
    sh = T * (topk - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    obj = mc + np.log(e.sum()) / T
    center = n - 1
    j = np.arange(n)

        h_new = slsqp_call(h_best, n, 0.1, 1, 5000, 1e-20)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
