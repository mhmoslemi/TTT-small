def simulated_annealing(h_init, n, t0, tlim, seed=42):
    best = h_init.copy()
    best_c5 = compute_c5(h_init)
    rng = np.random.default_rng(seed)
    h_curr = h_init.copy()
    c5_curr = best_c5
    T_sa = 0.001  # initial temperature
    T_min = 1e-7
    cooldown = 0.999
    
    while time.time() - t0 < tlim:
        # Generate perturbation
        pert_type = rng.integers(0, 4)
        h_try = h_curr.copy()
        if pert_type == 0:
            # Random Gaussian
            sig = 0.01
            h_try += rng.normal(0, sig, n)
        elif pert_type == 1:
            # Block flip
            bs = max(5, n // 14)
            bi = rng.integers(0, n - bs)
            h_try[bi:bi+bs] = 1.0 - h_try[bi:bi+bs]
        elif pert_type == 2:
            # Sine perturbation
            f = rng.uniform(1, 6)
            p = rng.uniform(0, 2*np.pi)
            a = rng.uniform(0.01, 0.05)
            x = np.arange(n, dtype=float)
            h_try += a * np.sin(2*np.pi*f*x/n + p)

            b2 = rng.integers(n//2, n-5)
            l1 = rng.integers(3, n//6)
            l2 = rng.integers(3, n//6)
            h_try[b1:b1+l1], h_try[b2:b2+l2] = h_try[b2:b2+l2].copy(), h_try[b1:b1+l1].copy()
        
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        
        # Metropolis acceptance
        if c5_try < c5_curr or rng.random() < np.exp(-(c5_try - c5_curr) / T_sa):
            h_curr = h_try
            c5_curr = c5_try

def shift_grad(x, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1, m2 = j >= shift, (j + shift) < n
        if m1.any(): g[m1] += (1.0 - x[j[m1] - shift])
