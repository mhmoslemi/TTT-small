def corr_and_grad(h, n, K, p=8):
    """Correlations and gradient of weighted top-K."""
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_corr = 2 * n - 1
    if K >= n_corr:
        top_idx = np.arange(n_corr)
        w = np.ones(n_corr) / n_corr

        w = (top_vals / mc) ** p

def slsqp_smooth(h, n, T, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        _, g = corr_and_grad(x, n, n_corr, 1)
        # For log-sum-exp, gradient is weighted sum of individual grads

                m2 = (j + si) < n
                grad[m1] += w * (1.0 - x[j[m1] - si])
                grad[m2] -= w * x[j[m2] + si]

                m1 = (j + s) < n
                m2 = j >= s
                grad[m1] += w * (1.0 - x[j[m1] + s])
                grad[m2] -= w * x[j[m2] - s]
        return grad * dx

    dx = 2.0 / n
    center = n - 1
    nc = 2 * n - 1
    j = np.arange(n)

        c = np.correlate(x, 1.0 - x, mode="full")
        mc = c.max()
        return (mc + np.log(np.exp(T * (c - mc)).sum()) / T) * dx
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        mc = c.max()
        sw = np.exp(T * (c - mc))
        sw /= sw.sum()

                m1, m2 = (j + s) < n, j >= s
                g[m1] += w * (1.0 - x[j[m1] + s])
                g[m2] -= w * x[j[m2] - s]
        return g * dx
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=[(0.0,1.0)]*n,
                       constraints=[{'type':'eq','fun':lambda x: x.sum()-target,'jac':lambda x: np.ones(n)}],
