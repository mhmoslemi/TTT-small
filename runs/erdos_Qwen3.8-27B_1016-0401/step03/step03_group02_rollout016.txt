    return x.sum() - _TARGET

def eq_jac(x):
    return np.ones(len(x))

def make_cons():
    return [dict(type='eq', fun=eq_fun, jac=eq_jac)]

def lag_obj(x, lag_idx):
    n = len(x)
    return float(corr_vec(x)[lag_idx] * (2.0 / n))

def lag_grad(x, lag_idx):
    n = len(x)
    o = lag_idx - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * x
    elif o > 0:
        g[o:] += 1.0 - x[:n-o]
        g[:n-o] += -x[o:]
    else:
        p = -o
        g[:n-p] += 1.0 - x[p:]
        g[p:] += -x[:n-p]
    return g * dx

def smax_obj(x, T):
    n = len(x)
    c = corr_vec(x)
    m = np.max(c)
    shifted = T * (c - m)
    e = np.exp(shifted)
    return (m + np.log(np.sum(e)) / T) * (2.0 / n)

def smax_grad(x, T):
    n = len(x)
    c = corr_vec(x)
    m = np.max(c)
    shifted = T * (c - m)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n-o])
            grad[:n-o] += -w * x[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n-p]
    return grad * dx

def slsqp_lag(h, lag_idx, bounds, target, maxiter, ftol):
    global _TARGET
    _TARGET = target
    cons = make_cons()
    try:
        res = minimize(lag_obj, h, args=(lag_idx,), method='SLSQP',
                      jac=lag_grad, bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        h_new = np.clip(res.x, 0.0, 1.0)
        if compute_c5(h_new) <= compute_c5(h) + 1e-15:
            return h_new
    except Exception:
        pass
    return h

def slsqp_smax(h, T, bounds, target, maxiter, ftol):
    global _TARGET
    _TARGET = target
    cons = make_cons()
    try:
        res = minimize(smax_obj, h, args=(T,), method='SLSQP',
                      jac=smax_grad, bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        h_new = np.clip(res.x, 0.0, 1.0)
        if compute_c5(h_new) <= compute_c5(h) + 1e-15:
            return h_new
    except Exception:
        pass
    return h

def coord_fine(h, n_sweeps, dts):
    n = len(h)
    target = n / 2.0
    c5_cur = compute_c5(h)
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            for dt in dts:
                trial = h.copy()
                trial[j] = np.clip(trial[j] + dt, 0.0, 1.0)
                trial = fix_constraint(trial, target)
                c5_new = compute_c5(trial)
                if c5_new < c5_cur - 1e-16:
                    h = trial
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def binary_push(h, threshold):
    n = len(h)
    target = n / 2.0
    c5_cur = compute_c5(h)
    for j in range(n):
        if h[j] > threshold or h[j] < 1.0 - threshold:
            trial = h.copy()
            trial[j] = 1.0 if h[j] > 0.5 else 0.0
            trial = fix_constraint(trial, target)
            c5_new = compute_c5(trial)
            if c5_new < c5_cur - 1e-16:
                h = trial
                c5_cur = c5_new
    return h, c5_cur

def thue_morse_init(n):
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    t = (t + 1) / 2.0
    t = fix_constraint(t, n / 2.0)
    return t

def run(seed=42, budget_s=1100, **kwargs):
    global _TARGET
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = thue_morse_init(340)

    n = len(h_best)
    target = n / 2.0
    _TARGET = target
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Intensive SLSQP cycling on top lags (35% budget)
    p1_end = t0 + budget_s * 0.35
    for outer in range(200):
        if elapsed() > p1_end:
            break
        c = corr_vec(h_best)
        top_idx = np.argsort(c)[-5:]
        for lag in sorted(top_idx):
            if elapsed() > p1_end:
                break
            h_new = slsqp_lag(h_best, lag, bounds, target, 1200, 1e-19)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-17:
                c5_best = c5_new
                h_best = h_new

    # Phase 2: Softmax SLSQP (12% budget)
    p2_end = t0 + budget_s * 0.47
    for T in [3000, 10000, 50000, 150000]:
        if elapsed() > p2_end:
            break
        h_new = slsqp_smax(h_best, T, bounds, target, 4000, 1e-17)
        h_new = fix_constraint(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-17:
            c5_best = c5_new
            h_best = h_new

    # Phase 3: Fine coordinate descent + binary push + SLSQP (20% budget)
    p3_end = t0 + budget_s * 0.67
    fine_dts = [0.001, -0.001, 0.0005, -0.0005, 0.002, -0.002,
                0.0002, -0.0002, 0.005, -0.005]
    for _ in range(15):
        if elapsed() > p3_end:
            break
        h_best, c5_cd = coord_fine(h_best, 1, fine_dts)
        if c5_cd < c5_best:
            c5_best = c5_cd
        h_best, c5_bp = binary_push(h_best, 0.08)
        if c5_bp < c5_best:
            c5_best = c5_bp
        c = corr_vec(h_best)
        worst = int(np.argmax(c))
        h_new = slsqp_lag(h_best, worst, bounds, target, 1000, 1e-19)
        h_new = fix_constraint(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 4: Nearby n values (13% budget)
    p4_end = t0 + budget_s * 0.80
    n_tries = [336, 338, 342, 344, 330, 348, 328, 352]
    for n_alt in n_tries:
        if elapsed() > p4_end:
            break
        target_alt = n_alt / 2.0
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = fix_constraint(h_alt, target_alt)
        c5_alt = compute_c5(h_alt)
        bounds_alt = [(0.0, 1.0)] * n_alt
        for _ in range(6):
            if elapsed() > p4_end:
                break
            c_a = corr_vec(h_alt)
            worst_a = int(np.argmax(c_a))
            h_new = slsqp_lag(h_alt, worst_a, bounds_alt, target_alt, 800, 1e-18)
            h_new = fix_constraint(h_new, target_alt)
            c5_new = compute_c5(h_new)
            if c5_new < c5_alt:
                c5_alt = c5_new
                h_alt = h_new
        h_alt, c5_cd_a = coord_fine(h_alt, 1, fine_dts)
        if c5_cd_a < c5_alt:
            c5_alt = c5_cd_a
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt
            target = n / 2.0
            bounds = [(0.0, 1.0)] * n

    # Phase 5: Final SLSQP polish + ultra-fine CD (13% budget)
    p5_end = t0 + budget_s - 3.0
    for _ in range(20):
        if elapsed() > p5_end:
            break
        c = corr_vec(h_best)
        worst = int(np.argmax(c))
        h_new = slsqp_lag(h_best, worst, bounds, target, 2000, 1e-20)
        h_new = fix_constraint(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-18:
            c5_best = c5_new
            h_best = h_new
    ultra_dts = [0.0001, -0.0001, 0.00005, -0.00005, 0.0003, -0.0003]
    h_best, c5_ultra = coord_fine(h_best, 2, ultra_dts)
    if c5_ultra < c5_best:
        c5_best = c5_ultra
    h_best = fix_constraint(h_best, target)

    c5_final = compute_c5(h_best)
    if c5_final < c5_best:
        c5_best = c5_final

    return (h_best, c5_best, len(h_best))
