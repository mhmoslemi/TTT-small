    return grad * dx

def pmean_obj(x, p=16.0):
    corr = np.correlate(x, 1.0 - x, mode="full")
    shift = np.min(corr) - 1.0
    c = corr - shift + 1e-10
    return (np.power(c, p).sum() ** (1.0 / p)) * (2.0 / _N)

def run_slsqp(h, n, T, maxiter, ftol, pmode=False, pval=32.0):
    global _N, _TARGET, _T
    _N = n
    _TARGET = n / 2.0
    _T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - _TARGET, 'jac': lambda x: np.ones(n)}]
    if pmode:
        res = minimize(pmean_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    else:
        res = minimize(lse_obj, h, method='SLSQP', jac=lse_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def nesterov_multi(h_init, n, t0, tm Budget, K_start=3, K_end=20, max_iters=400000):
    dx = 2.0 / n
    center = n - 1
    nm = 2 * n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    step_base = 0.12
    for it in range(max_iters):
        if time.time() - t0 > t_budget:
            break
        frac = it / max_iters
        K = K_start + int((K_end - K_start) * frac)
        beta = float(it) / (it + 3.0)
        h_y = np.clip(h_sg + beta * (h_sg - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top_idx = np.argsort(corr)[-K:]
        top_corr = corr[top_idx]
        mv = top_corr.max()
        w = np.exp(8.0 * (top_corr - mv))
        w /= w.sum()
        g = np.zeros(n)
        for wk, ik in zip(w, top_idx):
            if wk > 1e-14:
                g += wk * shift_grad(h_y, n, ik - center)
        g *= dx
        alpha = step_base / (1.0 + 0.003 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 20000 == 0:
            cv = compute_c5(h_new)
            if cv < c5_best:
                c5_best = cv
                h_best = h_new.copy()
    h_new = project_feasible(h_sg, n)
    cv = compute_c5(h_new)
    if cv < c5_best:
        c5_best = cv
        h_best = h_new.copy()
    return h_best, c5_best

def coord_polish(h, n, rng, steps=8000):
    dx = 2.0 / n
    center = n - 1
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for _ in range(steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        top = np.argsort(corr)[-3:]
        g = np.zeros(n)
        for ik in top:
            g += shift_grad(h_cur, n, ik - center)
        g *= dx
        pos = np.where(g > 0.005)[0]
        neg = np.where(g < -0.005)[0]
        if len(pos) == 0 or len(neg) == 0:
            break
        ip = rng.choice(pos)
        in_ = rng.choice(neg)
        delta = min(0.04, h_cur[ip] * 0.5, (1.0 - h_cur[in_]) * 0.5)
        if delta < 1e-5:
            delta = 0.01
        h_try = h_cur.copy()
        h_try[ip] -= delta
        h_try[in_] += delta
        h_try = project_feasible(np.clip(h_try, 0.0, 1.0), n)
        c5_try = compute_c5(h_try)
        if c5_try < c5_cur - 1e-9:
            h_cur = h_try
            c5_cur = c5_try
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP LSE continuation
    temps = [96, 48, 24, 12, 6, 3, 1.5, 0.8, 0.5, 0.3, 0.2, 0.15]
    for T in temps:
        if time.time() - t0 > budget_s - 600:
            break
        h_new = run_slsqp(h_best, n, T, 6000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Nesterov multi-shift
    h_sg, c5_sg = nesterov_multi(h_best, n, t0, budget_s - 380, K_start=3, K_end=20, max_iters=400000)
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg
    else:
        h_best = project_feasible(h_best, n)

    # Phase 3: SLSQP fine polish with power-mean
    for T, pm in [(0.5, False), (0.3, False), (0.2, True), (0.15, True), (0.1, False)]:
        if time.time() - t0 > budget_s - 280:
            break
        if pm:
            h_new = run_slsqp(h_best, n, 1.0, 8000, 1e-15, pmode=True, pval=48.0)
        else:
            h_new = run_slsqp(h_best, n, T, 8000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Coordinate polish
    if time.time() - t0 < budget_s - 120:
        h_lp, c5_lp = coord_polish(h_best, n, rng, steps=10000)
        if c5_lp < c5_best:
            c5_best = c5_lp
            h_best = h_lp
        h_new = run_slsqp(h_best, n, 0.3, 10000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Restart with structured perturbations
    for pi in range(25):
        if time.time() - t0 > budget_s - 50:
            break
        if pi < 8:
            sigma = 0.004 + 0.005 * (pi % 4)
        elif pi < 16:
            sigma = 0.015 + 0.01 * (pi % 4)
        else:
            sigma = 0.04 + 0.02 * (pi % 3)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_new = run_slsqp(h_p, n, 3.0, 4000, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            h_new = run_slsqp(h_best, n, 0.5, 6000, 1e-15)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 6: Try downsampling to n/2 if time permits
    if time.time() - t0 < budget_s - 60 and n >= 200:
        n2 = n // 2
        h2 = project_feasible(h_best[::2], n2)
        c5_2 = compute_c5(h2)
        if c5_2 < c5_best + 0.005:
            for T in [8, 4, 2, 1, 0.5, 0.3]:
                if time.time() - t0 > budget_s - 20:
                    break
                h_new = run_slsqp(h2, n2, T, 5000, 1e-15)
                h_new = project_feasible(h_new, n2)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
                    n = n2
            h2_final = run_slsqp(h2, n2, 0.3, 8000, 1e-15)
            h2_final = project_feasible(h2_final, n2)
            c5 = compute_c5(h2_final)
            if c5 < c5_best:
                c5_best = c5
                h_best = h2_final
                n = n2

    # Final polish
    if time.time() - t0 < budget_s - 15:
        h_new = run_slsqp(h_best, n, 0.2, 12000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
