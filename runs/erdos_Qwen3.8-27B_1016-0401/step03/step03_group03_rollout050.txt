import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    """Gradient of correlation at given shift."""
    grad = np.zeros(n)
    j = np.arange(n)
    if shift >= 0:
        m1 = j >= shift
        grad[m1] += (1.0 - h[np.minimum(j[m1] - shift, n-1)])
        m2 = (j + shift) < n
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        grad[m1] += (1.0 - h[j[m1] + s])
        m2 = j >= s
        grad[m2] -= h[j[m2] - s]
    return grad

def top_k_grad(h, n, K, temp):
    """Compute weighted gradient from top-K correlations."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    top_idx = np.argsort(corr)[-K:]
    mv = corr[top_idx].max()
    w = np.exp(temp * (corr[top_idx] - mv))
    w /= w.sum()
    g = np.zeros(n)
    for wk, ik in zip(w, top_idx):
        if wk > 1e-10:
            g += wk * shift_grad(h, n, ik - center)
    return g, corr.max()

def adam_optimize(h0, n, time_budget, t0, lr=0.01, K=3, temp=3.0, beta1=0.9, beta2=0.999):
    dx = 2.0 / n
    h = h0.copy()
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    while time.time() - t0 < time_budget:
        t += 1
        # Adaptive learning rate (warmup)
        lr_t = lr * np.sqrt(1 - beta2**t) / (1 - beta1**t)
        
        g, _ = top_k_grad(h, n, K, temp)
        g *= dx
        
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        
        h_try = h - lr_t * m / (np.sqrt(v) + 1e-8)
        h_try = project_feasible(h_try, n)
        
        if t % 100 == 0:
            c5 = compute_c5(h_try)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_try.copy()
        
        h = h_try
    
    c5 = compute_c5(h)
    if c5 < c5_best:
        c5_best = c5
        h_best = h.copy()
    
    return h_best, c5_best

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    grad = np.zeros(n)
    j = np.arange(n)
    if shift >= 0:
        m1 = j >= shift
        grad[m1] += (1.0 - h[j[m1] - shift])
        m2 = (j + shift) < n
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        grad[m1] += (1.0 - h[j[m1] + s])
        m2 = j >= s
        grad[m2] -= h[j[m2] - s]
    return grad

def top_k_grad(h, n, K, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    top_idx = np.argsort(corr)[-K:]
    mv = corr[top_idx].max()
    w = np.exp(temp * (corr[top_idx] - mv))
    w /= w.sum()
    g = np.zeros(n)
    for wk, ik in zip(w, top_idx):
        if wk > 1e-10:
            g += wk * shift_grad(h, n, ik - center)
    return g

def adam_phase(h0, n, t0, duration, lr=0.005, K=3, temp=3.0):
    dx = 2.0 / n
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    c5_best = compute_c5(h)
    h_best = h.copy()
    beta1, beta2 = 0.9, 0.999
    total_iters = max(1, int(duration * 2000))  # estimate ~2000 iters/sec
    
    while time.time() - t0 < t0 + duration:
        t += 1
        # Cosine decay learning rate
        frac = min(1.0, t / total_iters)
        lr_t = lr * 0.5 * (1 + np.cos(np.pi * frac))
        bias1 = 1 - beta1**t
        bias2 = 1 - beta2**t
        
        g = top_k_grad(h, n, K, temp) * dx
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / bias1
        v_hat = v / bias2
        
        h_try = h - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_try = project_feasible(h_try, n)
        
        if t % 50 == 0:
            c5 = compute_c5(h_try)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_try.copy()
        h = h_try
    
    c5 = compute_c5(h)
    if c5 < c5_best:
        c5_best = c5
        h_best = h.copy()
    return h_best, c5_best

# SLSQP entropic helpers
_S_N = 0
_S_T = 1.0

def _s_corr(x):
    return np.correlate(x, 1.0 - x, mode="full")

def _s_obj(x):
    corr = _s_corr(x)
    mc = np.max(corr)
    T = _S_T
    e = np.exp(T * (corr - mc))
    return (mc + np.log(e.sum()) / T) * (2.0 / _S_N)

def _s_jac(x):
    corr = _s_corr(x)
    mc = np.max(corr)
    T = _S_T
    e = np.exp(T * (corr - mc))
    sw = e / e.sum()
    center = _S_N - 1
    dx = 2.0 / _S_N
    grad = np.zeros(_S_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-12:
            continue
        grad += w * shift_grad(x, _S_N, oi - center)
    return grad * dx

def _s_cons_fun(x):
    return x.sum() - _S_N / 2.0

def _s_cons_jac(x):
    return np.ones(_S_N)

def do_slsqp(h, n, T, maxiter, ftol):
    global _S_N, _S_T
    _S_N = n
    _S_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _s_cons_fun, 'jac': _s_cons_jac}]
    try:
        res = minimize(_s_obj, h, method='SLSQP', jac=_s_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    grad = np.zeros(n)
    j = np.arange(n)
    if shift >= 0:
        m1 = j >= shift
        grad[m1] += (1.0 - h[j[m1] - shift])
        m2 = (j + shift) < n
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        grad[m1] += (1.0 - h[j[m1] + s])
        m2 = j >= s
        grad[m2] -= h[j[m2] - s]
    return grad

def top_k_grad(h, n, K, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    top_idx = np.argsort(corr)[-K:]
    mv = corr[top_idx].max()
    w = np.exp(temp * (corr[top_idx] - mv))
    w /= w.sum()
    g = np.zeros(n)
    for wk, ik in zip(w, top_idx):
        if wk > 1e-10:
            g += wk * shift_grad(h, n, ik - center)
    return g

def adam_run(h0, n, t_end, lr, K, temp, beta1=0.9, beta2=0.999):
    dx = 2.0 / n
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    c5_best = compute_c5(h)
    h_best = h.copy()
    best_t = 0
    while time.time() < t_end:
        t += 1
        frac = min(1.0, t / 300000.0)
        lr_t = lr * 0.5 * (1.0 + np.cos(np.pi * frac))
        lr_t = max(lr_t, lr * 0.01)
        b1 = 1.0 - beta1 ** t
        b2 = 1.0 - beta2 ** t
        g = top_k_grad(h, n, K, temp) * dx
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * g * g
        h_try = h - (lr_t / b1) * m / (np.sqrt(v / b2) + 1e-8)
        h_try = project_feasible(h_try, n)
        h = h_try
        if t % 100 == 0:
            c5 = compute_c5(h)
            if c5 < c5_best:
                c5_best = c5
                h_best = h.copy()
                best_t = t
            elif t - best_t > 50000:
                # Reset momentum
                m *= 0.5
                v *= 0.5
                best_t = t
    c5 = compute_c5(h)
    if c5 < c5_best:
        c5_best = c5
        h_best = h.copy()

_SN = 0
_ST = 1.0

def _s_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    e = np.exp(_ST * (corr - mc))
    return (mc + np.log(e.sum()) / _ST) * (2.0 / _SN)

def _s_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    e = np.exp(_ST * (corr - mc))
    sw = e / e.sum()
    center = _SN - 1
    dxv = 2.0 / _SN

    for oi in range(len(sw)):
        if sw[oi] < 1e-12:
            continue
        grad += sw[oi] * shift_grad(x, _SN, oi - center)
    return grad * dxv

def _s_cf(x):
    return x.sum() - _SN / 2.0

def _s_cj(x):
    return np.ones(_SN)

def do_slsqp(h, n, T, maxiter, ftol):
    global _SN, _ST
    _SN = n
    _ST = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _s_cf, 'jac': _s_cj}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h.copy()

        h_new = do_slsqp(h_best, n, T, 6000, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 4: Adam very fine (10% time)
    h_a, c5_a = adam_run(h_best, n, t0 + 0.92 * budget_s, lr=0.0002, K=3, temp=6.0)
    if c5_a < c5_best:

        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    grad = np.zeros(n)
    j = np.arange(n)
    if shift >= 0:
        m1 = j >= shift
        grad[m1] += (1.0 - h[j[m1] - shift])
        m2 = (j + shift) < n
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        grad[m1] += (1.0 - h[j[m1] + s])
        m2 = j >= s
        grad[m2] -= h[j[m2] - s]
    return grad

def top_k_grad(h, n, K, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    top_idx = np.argsort(corr)[-K:]
    mv = corr[top_idx].max()
    w = np.exp(temp * (corr[top_idx] - mv))
    w /= w.sum()
    g = np.zeros(n)
    for wk, ik in zip(w, top_idx):
        if wk > 1e-10:
            g += wk * shift_grad(h, n, ik - center)
    return g

def adam_run(h0, n, t_end, lr, K, temp):
    dx = 2.0 / n
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2 = 0.9, 0.999
    c5_best = compute_c5(h)
    h_best = h.copy()
    stall = 0
    while time.time() < t_end:
        t += 1
        decay = 1.0 / (1.0 + 8e-6 * t)
        lr_t = lr * decay
        g = top_k_grad(h, n, K, temp) * dx
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        bh1 = 1.0 - b1 ** t
        bh2 = 1.0 - b2 ** t
        h_try = h - (lr_t / bh1) * m / (np.sqrt(v / bh2) + 1e-8)
        h_try = project_feasible(h_try, n)
        h = h_try
        if t % 200 == 0:
            c5 = compute_c5(h)
            if c5 < c5_best - 1e-12:
                c5_best = c5
                h_best = h.copy()
                stall = 0
            else:
                stall += 1
                if stall > 80:
                    m *= 0.3
                    v *= 0.3
                    stall = 0
    c5 = compute_c5(h)
    if c5 < c5_best:
        c5_best = c5
        h_best = h.copy()
    return h_best, c5_best

_SN = 0
_ST = 1.0

def _s_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    e = np.exp(_ST * (corr - mc))
    return (mc + np.log(e.sum()) / _ST) * (2.0 / _SN)

def _s_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    e = np.exp(_ST * (corr - mc))
    sw = e / e.sum()
    center = _SN - 1
    dxv = 2.0 / _SN
    grad = np.zeros(_SN)
    for oi in range(len(sw)):
        if sw[oi] < 1e-12:
            continue
        grad += sw[oi] * shift_grad(x, _SN, oi - center)
    return grad * dxv

def _s_cf(x):
    return x.sum() - _SN / 2.0

def _s_cj(x):
    return np.ones(_SN)

def do_slsqp(h, n, T, maxiter, ftol):
    global _SN, _ST
    _SN = n
    _ST = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _s_cf, 'jac': _s_cj}]
    try:
        res = minimize(_s_obj, h, method='SLSQP', jac=_s_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Adam top-3, temp=4 (50% time)
    h_a, c5_a = adam_run(h_best, n, t0 + 0.50 * budget_s, 0.005, 3, 4.0)
    if c5_a < c5_best:
        c5_best, h_best = c5_a, h_a

    # Phase 2: Adam top-5, temp=3, lower lr (20% time)
    h_a, c5_a = adam_run(h_best, n, t0 + 0.70 * budget_s, 0.001, 5, 3.0)
    if c5_a < c5_best:
        c5_best, h_best = c5_a, h_a

    # Phase 3: SLSQP coarse-to-fine (up to 20% time)
    for T in [3.0, 1.5, 0.8, 0.4]:
        if time.time() - t0 > budget_s - 180:
            break
        h_new = do_slsqp(h_best, n, T, 4000, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 4: Adam very fine, high temp (8% time)
    h_a, c5_a = adam_run(h_best, n, t0 + 0.92 * budget_s, 0.0003, 3, 6.0)
    if c5_a < c5_best:
        c5_best, h_best = c5_a, h_a

    # Phase 5: Final SLSQP polish
    if time.time() - t0 < budget_s - 12:
        h_new = do_slsqp(h_best, n, 0.15, 8000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
