def project_feasible(y, n):
    target = n / 2.0

def grad_at_max(h, n, topk=1, T=0.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    kk = min(topk, len(corr))
    idx = np.argpartition(corr, -kk)[-kk:]
    if T > 0:
        tv = corr[idx]
        w = np.exp((tv - tv.max()) / T)
        w /= w.sum()
    else:
        w = np.ones(kk) / kk
    grad = np.zeros(n)
    j = np.arange(n)
    for oi in range(kk):
        wi = w[oi]
        if wi < 1e-12: continue
        sh = idx[oi] - center
        if sh >= 0:
            m1 = j >= sh
            grad[m1] += wi * (1.0 - h[j[m1] - sh])
            m2 = (j + sh) < n
            grad[m2] -= wi * h[j[m2] + sh]
        else:

            m1 = (j + s2) < n
            grad[m1] += wi * (1.0 - h[j[m1] + s2])

            grad[m2] -= wi * h[j[m2] - s2]

    return grad

        if sh >= 0:
            m1 = j >= sh
            grad[m1] += wi * (1.0 - h[j[m1] - sh])
            m2 = (j + sh) < n
            grad[m2] -= wi * h[j[m2] + sh]
        else:
            s2 = -sh
            m1 = (j + s2) < n
            grad[m1] += wi * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= wi * h[j[m2] - s2]
    grad *= dx
    return grad

def projected_grad(h, n, t_end, lr0=0.04, topk=1, T=0.0):
    h_y = h.copy()
    h_prev = h.copy()
    c5b = compute_c5(h)
    hb = h.copy()
    t_m = 1.0
    lr = lr0
    it = 0
    while time.time() < t_end:
        g = grad_at_max(h_y, n, topk, T)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_m * t_m)) * 0.5
        beta = (t_m - 1.0) / t_new
        h_z = h_y + beta * (h_y - h_prev)
        h_prev = h_y.copy()
        h_y = h_z - lr * g
        h_y = project_feasible(h_y, n)
        t_m = t_new
        lr = max(0.002, lr0 * (0.9999 ** it))
        it += 1
        if it % 800 == 0:
            c5 = compute_c5(h_y)
            if c5 < c5b:
                c5b = c5
                hb = h_y.copy()
    return hb, c5b

def slsqp_run(h, n, T=1.0, topk=10, maxiter=1500):
    target = n / 2.0
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        kk = min(topk, len(corr))
        idx = np.argpartition(corr, -kk)[-kk:]
        tv = corr[idx]
        mc = tv.max()
        return (mc + np.log(np.exp((tv - mc) / T).sum()) / T) * dx

    def jac(x):
        return grad_at_max(x, n, topk, T)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def perturb_h(h, n, rng):
    h_p = h.copy()
    pt = rng.integers(0, 6)
    if pt == 0:
        s = rng.integers(1, 5)
        h_p = np.roll(h_p, int(s))
    elif pt == 1:
        a = int(rng.integers(0, max(1, n - 12)))
        b = int(rng.integers(4, 18))
        h_p[a:a+b] = 1.0 - h_p[a:a+b]
    elif pt == 2:
        freq = int(rng.integers(2, 9))
        amp = rng.uniform(0.01, 0.04)
        x = np.arange(n) * 2.0 / n
        ph = rng.uniform(0, 2 * np.pi)
        h_p += amp * np.sin(np.pi * freq * x + ph)
    elif pt == 3:
        i1 = int(rng.integers(0, max(1, n - 16)))
        i2 = int(rng.integers(0, max(1, n - 16)))
        b = int(rng.integers(6, 14))
        h_p[i1:i1+b], h_p[i2:i2+b] = h_p[i2:i2+b].copy(), h_p[i1:i1+b].copy()
    elif pt == 4:
        h_p += rng.uniform(0.005, 0.02) * rng.normal(0, 1, n)
    else:
        mask = rng.random(n) < 0.08
        h_p[mask] = rng.random(int(mask.sum()))
    return project_feasible(h_p, n)

def try_resolution(h_old, n_old, n_new, t_end):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_hi = np.interp(x_new, x_old, h_old)
    h_hi = project_feasible(h_hi, n_new)
    for T, kk in [(4.0, 30), (1.0, 10), (0.4, 5)]:
        if time.time() > t_end:
            break
        h_hi = slsqp_run(h_hi, n_new, T=T, topk=kk, maxiter=1200)
    return h_hi

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Projected gradient with exact max (30% budget)
    t1 = t0 + budget_s * 0.30
    h_new, c5_new = projected_grad(h_best, n, t1, lr0=0.05, topk=1, T=0.0)
    if c5_new < c5_best:
        c5_best, h_best = c5_new, h_new
    # Second pass with topk=2 for smoother descent
    t1b = t0 + budget_s * 0.38
    if time.time() < t1b:
        h_new, c5_new = projected_grad(h_best, n, t1b, lr0=0.03, topk=2, T=0.2)
        if c5_new < c5_best:
            c5_best, h_best = c5_new, h_new

    # Phase 2: SLSQP cascade (25% budget)
    t2 = t0 + budget_s * 0.63
    for T, kk, mi in [(3.0, 30, 1200), (1.0, 12, 1500), (0.4, 6, 1500), (0.2, 4, 1200)]:
        if time.time() > t2:
            break
        h_new = slsqp_run(h_best, n, T=T, topk=kk, maxiter=mi)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best, h_best = c5_new, h_new

    # Phase 3: Fine projected gradient (15% budget)
    t3 = t0 + budget_s * 0.78
    h_new, c5_new = projected_grad(h_best, n, t3, lr0=0.02, topk=1, T=0.0)
    if c5_new < c5_best:
        c5_best, h_best = c5_new, h_new
    # Short SLSQP polish
    if time.time() < t0 + budget_s * 0.83:
        h_new = slsqp_run(h_best, n, T=0.3, topk=3, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best, h_best = c5_new, h_new

    # Phase 4: Perturbation restarts (15% budget)
    t4 = t0 + budget_s * 0.95
    n_perts = 8
    for pi in range(n_perts):
        rem = t4 - time.time()
        if rem < 40:
            break
        h_p = perturb_h(h_best, n, rng)
        t_sub = time.time() + min(rem * 0.35, 35)
        h_new, c5_new = projected_grad(h_p, n, t_sub, lr0=0.04, topk=1, T=0.0)
        if c5_new < c5_best:
            c5_best, h_best = c5_new, h_new
        if time.time() < t4 - 5:
            h_new = slsqp_run(h_best, n, T=0.5, topk=6, maxiter=800)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best, h_best = c5_new, h_new

    # Phase 5: Try alternate resolution if time permits
    t5 = t0 + budget_s * 0.95
    rem = t5 - time.time()
    if rem > 120 and n == 280:
        h_alt = try_resolution(h_best, n, 420, t5)
        c5_alt = compute_c5(h_alt)
        if c5_alt < c5_best:
            c5_best, h_best, n = c5_alt, h_alt, 420

    # Final polish
    rem = t0 + budget_s - 3 - time.time()
    if rem > 15:
        h_new = slsqp_run(h_best, n, T=0.25, topk=3, maxiter=int(min(1500, rem * 2)))
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best, h_best = c5_new, h_new

    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n)
