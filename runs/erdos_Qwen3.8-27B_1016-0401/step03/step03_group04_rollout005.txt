import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j < (n - shift)
        m2 = j >= shift
        if m1.any():
            grad[j[m1]] += (1.0 - h[j[m1] + shift])
        if m2.any():
            grad[j[m2]] -= h[j[m2] - shift]
    else:
        s = -shift
        m1 = j < (n - s)
        m2 = j >= s
        if m1.any():
            grad[j[m1]] -= h[j[m1] + s]
        if m2.any():
            grad[j[m2]] += (1.0 - h[j[m2] - s])
    return grad

def smooth_max(h, n, T, K):
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    L = len(corr)
    if K < L:
        idx = np.argpartition(corr, -K)[-K:]
        c = corr[idx]
    else:
        idx = np.arange(L)
        c = corr
    mc = c.max()
    exp_arg = T * (c - mc)
    e = np.exp(exp_arg)
    sw = e / e.sum()
    val = (mc + np.log(e.sum()) / T) * dx
    return val, sw, idx

def lp_relax(h, n, m_active=20, max_lp=50):
    """Iterate: linearize top-m correlations, solve LP, update h."""
    dx = 2.0 / n
    center = n - 1
    h_curr = h.copy()
    c5_curr = compute_c5(h_curr)
    
    for it in range(max_lp):
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        top_idx = np.argpartition(corr, -m_active)[-m_active:]
        
        # Build LP: min t s.t. f_s(h0) + g_s(h0)·(h-h0) <= t
        # Variables: h[0..n-1], t (total n+1)
        # c vector (objective): minimize t -> c = [0,...,0,1]
        c_lp = np.zeros(n + 1)
        c_lp[-1] = 1.0
        
        # Inequality constraints: A_ub @ x <= b_ub
        # For each active shift: g_s · (h - h0) + f_s(h0) <= t
        # => g_s · h - t <= f_s(h0) - g_s · h0
        A_rows = []
        b_rows = []
        for ii in range(m_active):
            s = top_idx[ii] - center
            g = shift_grad(h_curr, n, s) * dx
            f_s = corr[top_idx[ii]] * dx
            # g · (h - h0) + f_s <= t  =>  g·h - t <= f_s - g·h0
            row = np.zeros(n + 1)
            row[:n] = g
            row[-1] = -1.0
            A_rows.append(row)
            b_rows.append(f_s - g @ h_curr)
        
        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)
        
        # Equality: sum(h) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        # Bounds: 0 <= h_i <= 1, t >= 0
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        
        try:
            res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                h_new = res.x[:n]
                t_new = res.x[-1]
                # Check actual improvement
                c5_new = compute_c5(h_new)
                if c5_new < c5_curr:
                    h_curr = h_new
                    c5_curr = c5_new
                # Also try projected version

                c5_proj = compute_c5(h_proj)
                if c5_proj < c5_curr:
                    h_curr = h_proj
                    c5_curr = c5_proj
        except Exception:
            break
    
    return h_curr, c5_curr

Wait, there's an issue. The LP finds h that minimizes the linearized max, but the actual max at h_new might be worse (since the linearization is only a lower bound on the quadratic). I need to check this carefully.

Actually, for a quadratic f(h) = h^T A h, the linearization at h₀ is:
f(h) ≥ f(h₀) + ∇f(h₀)·(h-h₀)  (if A is positive semidefinite)
f(h) ≤ f(h₀) + ∇f(h₀)·(h-h₀)  (if A is negative semidefinite)

For our case, f_s(h) = (2/n)∑h[i](1-h[i+s]) = 1 - (2/n)∑h[i]h[i+s].
The Hessian of ∑h[i]h[i+s] is positive semi-definite (as argued above).
So the Hessian of f_s is negative semi-definite.
Therefore f_s is concave, and f_s(h) ≤ f_s(h₀) + ∇f_s(h₀)·(h-h₀).

So if the LP ensures f_s(h₀) + ∇f_s(h₀)·(h-h₀) ≤ t for all active s, it does NOT guarantee f_s(h) ≤ t. The LP is a "relaxation" that might be optimistic.

However, this doesn't mean the LP step is useless. The solution h_new from the LP will have direction d = h_new - h₀ that reduces the linearized correlations. If we then do a line search along d to find where the true max correlation is minimized, we might improve.
