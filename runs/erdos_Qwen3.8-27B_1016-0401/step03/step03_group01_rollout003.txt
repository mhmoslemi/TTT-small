        best_i, best_j, best_d = -1, -1, 0
        
        # For each pair i < j, compute the best delta
        for i in range(n):
            for j in range(i + 1, n):
                # d_s for this pair
                d_vals = np.zeros(ka)
                for s in range(ka):
                    m = shifts[s]
                    # h[j+m] - h[i+m] + h[j-m] - h[i-m]
                    jm = j + m
                    im = i + m
                    jmm = j - m
                    imm = i - m
                    v = 0.0
                    if 0 <= jm < n: v += h_cur[jm]
                    if 0 <= im < n: v -= h_cur[im]
                    if 0 <= jmm < n: v += h_cur[jmm]
                    if 0 <= imm < n: v -= h_cur[imm]
                    d_vals[s] = v
                
                # Find optimal delta to minimize max_s (c_vals[s] + delta * d_vals[s])
                # Constraints: -h[i] <= delta <= 1-h[i] and -(1-h[j]) <= delta <= h[j]
                # i.e., max(-h[i], h[j]-1) <= delta <= min(1-h[i], h[j])
                d_lo = max(-h_cur[i], h_cur[j] - 1.0)
                d_hi = min(1.0 - h_cur[i], h_cur[j])
                
                if d_lo >= d_hi:
                    continue
                
                # Candidate deltas: boundaries and crossings
                candidates = [d_lo, d_hi, 0.0]
                for a in range(ka):
                    for b in range(a + 1, ka):
                        if abs(d_vals[a] - d_vals[b]) < 1e-15:
                            continue
                        dc = (c_vals[b] - c_vals[a]) / (d_vals[a] - d_vals[b])
                        if d_lo < dc < d_hi:
                            candidates.append(dc)
                
                best_c_new = np.inf
                best_delta = 0.0
                for dc in candidates:
                    vals = c_vals + dc * d_vals
                    mx = np.max(vals)
                    if mx < best_c_new:
                        best_c_new = mx
                        best_delta = dc
                
                improvement = c_max - best_c_new
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_i, best_j, best_d = i, j, best_delta
        
        if best_i >= 0 and best_improvement > 1e-15:
            h_cur[best_i] += best_d
            h_cur[best_j] -= best_d
            # Project (should be exact within floating point)
            h_cur = np.clip(h_cur, 0, 1)
            # Force exact sum
            diff = h_cur.sum() - target
            if abs(diff) > 1e-10:
                h_cur = project_feasible(h_cur, n)
            c5_cur = compute_c5(h_cur)
    
    return h_cur, c5_cur
