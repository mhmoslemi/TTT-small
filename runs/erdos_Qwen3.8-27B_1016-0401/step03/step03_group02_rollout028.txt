def make_eq(target):
    return [dict(type='eq', fun=lambda x: x.sum() - target, jac=lambda x: np.ones(len(x)))]

def sm_obj(x, T, target):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    smax = np.log(np.sum(np.exp(shifted))) / T + mc
    pen = 0.5 * (x.sum() - target) ** 2 * 100.0
    return smax * (2.0 / n) + pen

def sm_grad(x, T, target):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")

    e = np.exp(shifted)

            h_new = slsqp_lag(h_best, int(lag), target, bounds, 1000, 1e-17)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                h_best = h_new
    
    # Phase 3: Binary swap local search (30%)
    p3_end = t0 + budget_s * 0.70
    h_bin = (h_best > 0.5).astype(float)
    h_bin = fix_constraint(h_bin, target)
    c5_bin = compute_c5(h_bin)
    time_left = max(10, p3_end - time.time())
    h_bin, c5_bin = binary_swap_search(h_bin, max_passes=500, time_limit=time_left)
    if c5_bin < c5_best:
        c5_best = c5_bin
        h_best = h_bin.copy()
    
    # Back to mixed: use binary as starting point for SLSQP
    h_mixed = h_best.copy()
    # Slightly relax: add small perturbation to break symmetry
    for _ in range(5):
        if elapsed() > p3_end + budget_s * 0.05:
            break
        h_try = h_mixed.copy()
        for k in range(len(h_try)):
            if 0.05 < h_try[k] < 0.95:
                h_try[k] = np.clip(h_try[k] + rng.uniform(-0.02, 0.02), 0, 1)
        h_try = fix_constraint(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < c5_best - 1e-14:
            h_mixed = h_try
            c5_best = c5_try
            h_best = h_try
    
    # Phase 4: Fine coordinate descent + SLSQP polish (15%)
    p4_end = t0 + budget_s * 0.85
    for _ in range(10):
        if elapsed() > p4_end:
            break
        h_best, c5_cd = coord_fine(h_best, 2)
        if c5_cd < c5_best:
            c5_best = c5_cd
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst_f = int(np.argmax(corr_f))
        h_new = slsqp_lag(h_best, worst_f, target, bounds, 800, 1e-18)
        h_new = fix_constraint(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new
        # Binary push: push values near 0 or 1
        h_bp = h_best.copy()
        for k in range(len(h_bp)):
            if h_bp[k] > 0.8:
                h_bp[k] = 1.0
            elif h_bp[k] < 0.2:
                h_bp[k] = 0.0
        h_bp = fix_constraint(h_bp, target)
        c5_bp = compute_c5(h_bp)
        if c5_bp < c5_best:
            c5_best = c5_bp
            h_best = h_bp
    
    # Phase 5: Try alternative n values (10%)
    p5_end = t0 + budget_s - 15.0
    for n_alt in [280, 300, 320, 360, 380, 400]:
        if elapsed() > p5_end:
            break
        h_alt = interpolate_h(h_best, n_alt)
        c5_alt = compute_c5(h_alt)
        t_alt = n_alt / 2.0
        bnd_alt = [(0.0, 1.0)] * n_alt
        # Quick SLSQP
        for T in [2000, 1000]:
            if elapsed() > p5_end:
                break
            h_new = slsqp_sm(h_alt, T, t_alt, bnd_alt, 1500, 1e-16)
            h_new = fix_constraint(h_new, t_alt)
            c5_new = compute_c5(h_new)
            if c5_new < c5_alt:
                c5_alt = c5_new
                h_alt = h_new
        # Quick binary search
        h_ab = (h_alt > 0.5).astype(float)
        h_ab = fix_constraint(h_ab, t_alt)
        time_left = max(5, min(30, p5_end - time.time()))
        h_ab, c5_ab = binary_swap_search(h_ab, max_passes=100, time_limit=time_left)
        if c5_ab < c5_alt:
            c5_alt = c5_ab
            h_alt = h_ab
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt
            target = n / 2.0
    
    # Final polish
    p6_end = time.time() + 12.0
    h_best, c5_final_cd = coord_fine(h_best, 2, dt_base=0.005)
    if c5_final_cd < c5_best:
        c5_best = c5_final_cd
        h_best = h_best.copy()
    
    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    h_best = np.clip(h_best, 0.0, 1.0)
    return (h_best, c5_final, len(h_best))
