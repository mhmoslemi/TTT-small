def golden_search(f, a, b, tol=1e-9, max_iter=50):
    gr = (np.sqrt(5) - 1) / 2
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)

            d = c

            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)

Wait, for binary h: h(i)(1-h(i+k)) = h(i) - h(i)h(i+k). Sum over i: sum h(i) - sum h(i)h(i+k) = n/2 - autocorr_k.

So overlap_k = dx * (n/2 - autocorr_k) = 1 - dx * autocorr_k.

And we want to minimize max_k overlap_k = 1 - min_k dx*autocorr_k.

So equivalently, maximize min_k autocorr_k * dx.

For a binary sequence, autocorr_k = number of i such that h[i]=h[i+k]=1 (for valid i).

The total sum of autocorr over all k: sum_k autocorr_k = (n/2)^2 (counting all pairs). There are 2n-1 shifts. Average autocorr = (n/2)^2 / (2n-1) ≈ n/8.

So the minimum autocorr is at most n/8 on average, giving minimum overlap ≥ 1 - 2/8 = 3/4. Wait, that's the average, not the minimum.

Actually, for the specific shifts:
- k=0: autocorr_0 = n/2 (all 1s pair with themselves)
- k=1: autocorr_1 ≤ n/2 (limited by the number of 1s)
- k=n/2: could be very small

The constraint is that autocorr_k ≥ 0 for all k, and sum_k autocorr_k = (n/2)^2.

If all autocorr_k were equal (except k=0), we'd have autocorr_k = (n/2)^2 / (2n-2) ≈ n/8 for k≠0.
Then overlap_k = 1 - (n/8)*(2/n) = 1 - 1/4 = 3/4 for k≠0.
And overlap_0 = 1 - (n/2)*(2/n) = 1 - 1 = 0.

So the theoretical lower bound on C₅ for binary sequences is at most 3/4 (if we could achieve perfectly flat autocorrelation). But in practice, we can't achieve perfectly flat autocorrelation for binary sequences.

import numpy as np
from scipy.optimize import minimize
import time

_G = {'n': 0, 'T': 1.0, 'target': 0.0}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    g = np.zeros(n)
    if shift > 0:
        g[:n - shift] += 1.0 - h[shift:]
        g[shift:] -= h[:n - shift]
    elif shift < 0:
        s = -shift
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    else:
        g[:] = 1.0 - 2.0 * h
    return g

def _sm_obj(x):
    n, T = _G['n'], _G['T']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    sh = T * (corr - mc)
    return (mc + np.log(np.exp(sh).sum()) / T) * dx

def _sm_jac(x):
    n, T = _G['n'], _G['T']
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    w = e / e.sum()
    g = np.zeros(n)
    nz = np.where(w > 1e-13)[0]
    for oi in nz:
        g += w[oi] * shift_grad(x, n, oi - center)
    return g * dx

def _eq_fun(x):
    return x.sum() - _G['target']

def _eq_jac(x):
    return np.ones(_G['n'])

def slsqp_call(h, n, T, maxiter, ftol):

    _G['T'] = float(T)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def nesterov_opt(h_init, n, t0, budget, K=10, max_iters=180000):
    dx = 2.0 / n
    center = n - 1
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    for it in range(max_iters):

            break

        h_y = h_cur + beta * (h_cur - h_prev)

def pair_swap_polish(h, n, rng, steps=3000):
