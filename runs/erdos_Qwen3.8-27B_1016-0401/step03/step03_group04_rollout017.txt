        c_sel = corr[topk]
        mc_sel = np.max(c_sel)
        sh = T * (c_sel - mc_sel)
        e = np.exp(sh)
        val = (mc_sel + np.log(e.sum()) / T) * (2.0 / n)
        sw = e / e.sum()
        center = n - 1
        grad = np.zeros(n)
        for ii in range(len(sw)):
            w = sw[ii]
            if w < 1e-15:
                continue
            grad += w * shift_grad(x, n, topk[ii] - center)
        return val, grad * (2.0 / n)
    else:
        sh = T * (corr - mc)
        e = np.exp(sh)
        val = (mc + np.log(e.sum()) / T) * (2.0 / n)
        sw = e / e.sum()
        center = n - 1
        grad = np.zeros(n)
        for oi in range(len(sw)):
            w = sw[oi]
            if w < 1e-15:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return val, grad * (2.0 / n)

_tflag = [False]

def _obj_fn(x, n, T, K, t0, tlim):
    if time.time() - t0 > tlim:
        _tflag[0] = True
    v, _ = sm_val_and_grad(x, n, T, K)
    return v

def _grad_fn(x, n, T, K, t0, tlim):
    if time.time() - t0 > tlim:
        _tflag[0] = True
    _, g = sm_val_and_grad(x, n, T, K)
    return g

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):
    _tflag[0] = False
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_fn, h, method='L-BFGS-B', jac=_grad_fn,
                       args=(n, T, K, t0, tlim),
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol * 0.05,
                                'maxls': 40, 'maxfun': maxiter * 4})
        if _tflag[0]:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def alternating_subgrad(h_init, n, t0, tlim, max_iters=300000):
    """Subgradient with alternating focus on top shifts and momentum."""
    dx = 2.0 / n
    h_curr = h_init.copy()
    v_mom = np.zeros(n)
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    f_star = c5_best - 0.001
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        y = h_curr
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_K = min(5, len(corr))
        idxs = np.argpartition(corr, -top_K)[-top_K:]
        g = np.zeros(n)
        for idx in idxs:
            g += shift_grad(y, n, idx - (n - 1))
        g *= dx / top_K
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        f_y = float(np.max(corr) * dx)
        gap = max(f_y - f_star, 1e-8)
        alpha = min(2.0 * gap / (gn * gn + 1e-12), 0.4 / (1.0 + 0.005 * it))
        v_mom = 0.85 * v_mom + 0.15 * (-alpha * g)
        h_new = y + v_mom
        h_new = project_feasible(h_new, n)
        h_curr = h_new
        if (it + 1) % 2000 == 0:
            c5_now = compute_c5(h_curr)
            if c5_now < c5_best:
                c5_best = c5_now
                h_best = h_curr.copy()
                f_star = c5_best - 0.0005
    c5_now = compute_c5(h_curr)
    if c5_now < c5_best:
        c5_best = c5_now
        h_best = h_curr.copy()
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Coarse-to-fine L-BFGS-B, wider K for broader averaging
    schedules = [(96, 2500, 250), (48, 3500, 150), (24, 4500, 100), (12, 5500, 60),
                 (6, 7000, 30), (3, 8000, 20), (1.5, 10000, 10), (1.0, 12000, 8)]
    for T, mi, K in schedules:
        rem = budget_s - (time.time() - t0)
        if rem < 90:
            break
        tl = min(budget_s - 40, t0 + rem - 50)
        h_new = do_lbfgsb(h_best, n, T, K, mi, 1e-14, t0, tl)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Alternating subgradient (different from previous Nesterov)
    rem = budget_s - (time.time() - t0)
    if rem > 180:
        tl = t0 + rem - 120
        h_sg, c5_sg = alternating_subgrad(h_best, n, t0, tl, 200000)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase 3: Diverse restarts with different structures
    x = np.arange(n, dtype=float)
    n_restarts = 0
    while True:
        rem = budget_s - (time.time() - t0)
        if rem < 40 or n_restarts >= 40:
            break
        n_restarts += 1
        r = n_restarts
        if r % 5 == 1:
            sig = 0.003 + 0.004 * (r % 4)
            h_p = h_best + rng.normal(0, sig, n)
        elif r % 5 == 2:
            f = rng.uniform(0.5, 6.0)
            p = rng.uniform(0, 2 * np.pi)
            a = rng.uniform(0.005, 0.025)
            h_p = h_best + a * np.sin(2 * np.pi * f * x / 2.0 + p)
        elif r % 5 == 3:
            h_p = h_best.copy()
            bs = n // 6
            bi = int(rng.integers(0, 6))
            h_p[bi*bs:(bi+1)*bs] = 1.0 - h_p[bi*bs:(bi+1)*bs]
            h_p += rng.normal(0, 0.004, n)
        elif r % 5 == 4:
            # Mirror perturbation
            h_p = h_best.copy()
            h_p[:n//4] = 1.0 - h_best[n//4:n//2]
            h_p[n//4:n//2] = 1.0 - h_best[:n//4]
            h_p += rng.normal(0, 0.003, n)
        else:
            # Tail emphasis: perturb edges more
            w = 1.0 + 2.0 * np.abs(x - 1.0) / 1.0
            h_p = h_best + rng.normal(0, 0.008, n) * w / w.mean()
        h_p = project_feasible(h_p, n)
        tl = min(budget_s - 20, t0 + rem - 25)
        h_new = do_lbfgsb(h_p, n, 2.0, 80, 2500, 1e-13, t0, tl)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Ultra-fine polish - very low T, many iterations
    for T in [0.5, 0.8, 1.0]:
        rem = budget_s - (time.time() - t0)
        if rem < 45:
            break
        tl = min(budget_s - 15, t0 + rem - 20)
        h_new = do_lbfgsb(h_best, n, T, 15, 20000, 1e-16, t0, tl)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Resolution experiment - downsample to n/2, optimize, upscale
    rem = budget_s - (time.time() - t0)
    if rem > 250:
        n2 = n // 2
        x_full = np.arange(n) * (2.0 / n)
        x_half = np.arange(n2) * (2.0 / n2)
        h_lo = np.interp(x_half, x_full, h_best)
        h_lo = project_feasible(h_lo, n2)
        c5_lo = compute_c5(h_lo)
        for T, mi in [(16, 4000), (4, 6000), (1, 10000), (0.5, 10000)]:
            rem2 = budget_s - (time.time() - t0)
            if rem2 < 50:
                break
            tl = min(budget_s - 30, t0 + rem2 - 30)
            h_new_lo = do_lbfgsb(h_lo, n2, T, min(80, 2*n2-1), mi, 1e-14, t0, tl)
            c5_new_lo = compute_c5(h_new_lo)
            if c5_new_lo < c5_lo:
                c5_lo = c5_new_lo
                h_lo = h_new_lo
        # Upscale
        x_up = np.linspace(0, 2.0, n, endpoint=False)
        h_up = np.interp(x_up, x_half, h_lo)
        h_up = project_feasible(h_up, n)
        c5_up = compute_c5(h_up)
        if c5_up < c5_best:
            c5_best = c5_up
            h_best = h_up

    # Phase 6: Final micro-polish
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        tl = t0 + rem - 5
        h_new = do_lbfgsb(h_best, n, 0.5, 10, 20000, 1e-16, t0, tl)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 7: One more subgradient pass on the refined solution
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h_sg2, c5_sg2 = alternating_subgrad(h_best, n, t0, t0 + rem - 10, 50000)
        if c5_sg2 < c5_best:
            c5_best = c5_sg2
            h_best = h_sg2
            h_best = project_feasible(h_best, n)

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
