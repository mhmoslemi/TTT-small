    """Compute gradient of log-sum-exp(beta * corr) * dx"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * corr
    # Numerically stable LSE
    mc = np.max(scaled)
    e = np.exp(scaled - mc)
    se = e.sum()
    w = e / se  # softmax weights
    
    # Gradient: (dx/beta) * sum_k w_k * [(1-h_{i+s_k}) - h_{i-s_k}]
    # = (dx/beta) * [conv(w, 1-h) - conv(w, h)] at appropriate indices
    # 
    # w has length 2n-1, h has length n
    # conv(w, 1-h) has length 3n-2
    # We need the values at indices n-1 to 2n-2 (centered)
    
    # Actually, let's think about this more carefully.
    # corr[m] = sum_j h[j+m-(n-1)] * (1-h[j]) for valid j
    # s_m = m - (n-1), m ranges from 0 to 2n-2
    # 
    # ∂corr[m]/∂h[i] = (1-h[i+s_m]) if i+s_m in [0,n-1]
    #                  - h[i-s_m] if i-s_m in [0,n-1]

    #

    # sum_m w_m * u[i+s_m]: this is like a cross-correlation of w with u
    # sum_m w_m * v[i-s_m]: this is like a convolution of w with v (reversed)
    
    # Use FFT for efficiency
    # Actually for n=280, direct computation is fine: O(n * (2n-1)) = O(4n²) ≈ 313k
    grad = np.zeros(n)
    for m in range(2 * n - 1):

            continue
        s = m - (n - 1)
        # Contribution to positions i where i+s is valid (i in [max(0,-s), min(n-1, n-1-s)])
        # and i-s is valid (i in [max(0,s), min(n-1, n-1+s)])
        if s >= 0:

            # i-s in [0, n-1] means i in [s, n-1]

            i_lo2, i_hi2 = s, n - 1  # for v[i-s]
            grad[i_lo1:i_hi1+1] += w[m] * (1.0 - h[i_lo1+s:i_hi1+1+s])
            grad[i_lo2:i_hi2+1] -= w[m] * h[i_lo2-s:i_hi2+1-s]

            grad[s2:n] += w[m] * (1.0 - h[0:n-s2])
            grad[0:n-s2] -= w[m] * h[s2:n]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def lse_obj_grad(h, n, beta):
    """Objective and gradient of LSE(beta * corr) * dx."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * corr
    mc = scaled.max()
    e = np.exp(scaled - mc)
    se = e.sum()
    w = e / se
    obj = (mc + np.log(se)) / beta * dx
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        s = m - (n - 1)
        if s >= 0:
            if n - 1 - s >= 0:
                grad[:n-s] += wm * (1.0 - h[s:n])
            if s < n:
                grad[s:] -= wm * h[:n-s]
        else:
            s2 = -s
            if n - 1 - s2 >= 0:
                grad[s2:] += wm * (1.0 - h[:n-s2])
                grad[:n-s2] -= wm * h[s2:]
    grad *= dx / beta
    return obj, grad

def lse_pg(h, n, beta, lr, n_steps, time_limit, t0):
    """Projected gradient descent on LSE."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        obj, grad = lse_obj_grad(h, n, beta)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        h = h_new
        if step % 500 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def lse_fista(h, n, beta, lr, n_steps, time_limit, t0):
    """FISTA on LSE objective."""
    h_y = h.copy()
    h_z = h.copy()
    t_mom = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        obj, grad = lse_obj_grad(h_y, n, beta)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta_m = (t_mom - 1.0) / t_new
        h_z = h_y + beta_m * (h_y - h_z)
        h_y = h_z - lr * grad
        h_y = project_feasible(h_y, n)
        t_mom = t_new
        if step % 500 == 0:
            c5 = compute_c5(h_y)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_y.copy()
    c5 = compute_c5(h_y)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_y.copy()
    return best_h, best_c5

    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = topk.max()
    e = np.exp(T * (topk - mc))
    se = e.sum()
    return (mc + np.log(se) / T) * (2.0 / n)

def topk_smooth_grad(h, n, T, k):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]

    e = np.exp(T * (topk - mc))

    B = v2 * h[i2]
    grad = (A @ sw - B @ sw) * (2.0 / n)

def slsqp_run(h, n, T, k, maxiter):
    def obj(x):
        return topk_smooth_obj(x, n, T, k)
    def jac(x):
        return topk_smooth_grad(x, n, T, k)[1]
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return res.x
    except:
        return h.copy()

def coord_exact(h, n, max_rounds=5, t0=None, t_end=None):
    """Coordinate descent with exact 1D minimization."""
    dx = 2.0 / n
    h = h.copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    
    # Precompute the "coefficient matrix" M[i,m] = dx * [(1-h[i+s_m]) - h[i-s_m]] (valid)
    # This changes as h changes, so we recompute each round
    
    for rnd in range(max_rounds):
        if t_end and time.time() > t_end:
            break
        # Recompute M and corr
        corr = np.correlate(h, 1.0 - h, mode="full")
        s = np.arange(2 * n - 1) - (n - 1)
        i_idx = np.arange(n)
        idx1 = i_idx[:, None] + s[None, :]
        idx2 = i_idx[:, None] - s[None, :]
        v1 = (idx1 >= 0) & (idx1 < n)
        v2 = (idx2 >= 0) & (idx2 < n)
        i1 = np.clip(idx1, 0, n - 1)
        i2 = np.clip(idx2, 0, n - 1)
        M = (v1 * (1.0 - h[i1]) - v2 * h[i2]) * dx  # shape (n, 2n-1)
        
        # Coordinate descent
        for i in range(n):
            if t_end and time.time() > t_end:
                break
            # f(h_i) = max_m (M[i,m] * h_i + (corr[m] - M[i,m] * h[i]))
            # Let x = h_i, then f(x) = max_m (M[i,m] * x + c_m)
            # where c_m = corr[m] - M[i,m] * h[i]
            c = corr - M[i, :] * h[i]
            a = M[i, :]  # slopes
            
            # Find min_{x in [0,1]} max_m (a[m]*x + c[m])
            # Using the upper envelope approach
            # For simplicity, use ternary search or golden section
            # (the function is convex piecewise-linear)
            lo, hi = 0.0, 1.0
            for _ in range(50):  # 50 bisections for 1/2^50 precision
                mid = (lo + hi) / 2.0
                subgrad_lo = (a * lo + c).max() - (a * mid + c).max()
                # Actually, for convex PWL, the subgradient at mid is:
                # sum of a[m] for active m (those achieving the max)
                vals = a * mid + c
                mx = vals.max()
                active = (vals == mx)
                subgrad = a[active].sum()
                if subgrad < 0:
                    hi = mid
                elif subgrad > 0:
                    lo = mid
                else:
                    break
            x_opt = (lo + hi) / 2.0
            h[i] = x_opt
        
        # Project to fix sum
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    
    return best, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def lse_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    sc = beta * corr
    mc = sc.max()
    e = np.exp(sc - mc)
    se = e.sum()
    w = e / se
    obj = (mc + np.log(se)) / beta * dx
    s = np.arange(2 * n - 1) - (n - 1)
    ii = np.arange(n)
    ix1 = ii[:, None] + s[None, :]
    ix2 = ii[:, None] - s[None, :]
    v1 = (ix1 >= 0) & (ix1 < n)
    v2 = (ix2 >= 0) & (ix2 < n)
    A = v1 * (1.0 - h[np.clip(ix1, 0, n-1)])
    B = v2 * h[np.clip(ix2, 0, n-1)]
    grad = (A @ w - B @ w) * dx
    return obj, grad

        tt = tn
