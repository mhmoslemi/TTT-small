def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_G_T * (c - m))
    return (m + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _shift_grad_one(h, n, s):
    j = np.arange(n)
    g = np.zeros(n)
    if s >= 0:
        m1 = j >= s
        m2 = (j + s) < n
        if np.any(m1):
            g[m1] += (1.0 - h[j[m1] - s])
        if np.any(m2):
            g[m2] -= h[j[m2] + s]
    else:
        ss = -s
        m1 = (j + ss) < n
        m2 = j >= ss
        if np.any(m1):
            g[m1] += (1.0 - h[j[m1] + ss])
        if np.any(m2):
            g[m2] -= h[j[m2] - ss]
    return g

def _jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_G_T * (c - m))
    w = e / e.sum()
    grad = np.zeros(_G_N)
    for wi, ci in zip(w, idx):
        if wi > 1e-12:
            grad += wi * _shift_grad_one(x, _G_N, int(ci) - (_G_N - 1))
    return grad * (2.0 / _G_N)

    return x.sum() - _G_TGT

def _cjam(x):
    return np.ones(_G_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def _shift_grad_one(h, n, s):
    j = np.arange(n)
    g = np.zeros(n)
    if s >= 0:
        m1 = j >= s
        m2 = (j + s) < n
        if np.any(m1):
            g[m1] += (1.0 - h[j[m1] - s])
        if np.any(m2):
            g[m2] -= h[j[m2] + s]
    else:
        ss = -s
        m1 = (j + ss) < n
        m2 = j >= ss
        if np.any(m1):
            g[m1] += (1.0 - h[j[m1] + ss])
        if np.any(m2):
            g[m2] -= h[j[m2] - ss]
    return g

# Globals for SLSQP
_G_N = 0
_G_TGT = 0.0
_G_T = 1.0
_G_K = 10

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_G_T * (c - m))
    return (m + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_G_T * (c - m))
    w = e / e.sum()
    grad = np.zeros(_G_N)
    for wi, ci in zip(w, idx):
        if wi > 1e-12:
            grad += wi * _shift_grad_one(x, _G_N, int(ci) - (_G_N - 1))
    return grad * (2.0 / _G_N)

def _cfun(x):
    return x.sum() - _G_TGT

def _cjam(x):
    return np.ones(_G_N)

def _slsqp(h, n, T, top_k, maxiter, ftol):
    global _G_N, _G_TGT, _G_T, _G_K
    _G_N = n
    _G_TGT = n / 2.0
    _G_T = float(T)
    _G_K = min(top_k, 2 * n - 1)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h.copy(), method='SLSQP', jac=_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjam}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < _obj(h.copy()):
            return res.x
    except Exception:
        pass
    return h.copy()

def _exact_subgrad(h_init, n, t_end):
    """Projected subgradient with exact max, plus momentum."""
    h = h_init.copy()
    v = np.zeros(n)
    beta = 0.85
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(500000):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        g = _shift_grad_one(h, n, s)
        
        # Adaptive momentum
        b = 0.9 if it < 10000 else 0.7
        alpha = 0.03 / (1.0 + 0.002 * it)
        v = b * v - alpha * g
        h = project_feasible(h + v, n)
        
        if (it + 1) % 5000 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
    
    return best_h, best_c5

def _multi_shift_subgrad(h_init, n, t_end, top_k=5):
    """Subgradient with top-k shifts and softmax weighting."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(500000):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        T_w = max(1.0, 20.0 * (1.0 - it / 30000.0))
        w = np.exp(T_w * (c - c.max()))
        w /= w.sum()
        
        g = np.zeros(n)
        for wi, ci in zip(w, idx):
            if wi > 1e-10:
                g += wi * _shift_grad_one(h, n, int(ci) - (n - 1))
        
        alpha = 0.02 / (1.0 + 0.003 * it)
        h = project_feasible(h - alpha * g, n)
        
        if (it + 1) % 5000 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
    
    return best_h, best_c5
