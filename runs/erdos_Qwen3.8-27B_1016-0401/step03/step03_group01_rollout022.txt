def slsqp_opt(h, n, T, k, maxiter, ftol):
    """SLSQP with smooth max."""
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    j = np.arange(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mk = min(k, len(corr))
    
    def _eval(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argpartition(c, -mk)[-mk:]
        topk = c[idx]
        mc = topk.max()
        sh = T * (topk - mc)
        e = np.exp(sh)
        ss = e.sum()
        obj = (mc + np.log(ss) / T) * dx
        # Gradient
        sw = e / ss
        gr = np.zeros(n)
        for oi, w in zip(idx, sw):
            if w < 1e-12:
                continue
            s = int(oi) - center
            if s >= 0:
                m1 = j >= s
                gr[m1] += w * (1.0 - x[j[m1] - s])
                m2 = (j + s) < n
                gr[m2] -= w * x[j[m2] + s]
            else:
                s2 = -s
                m1 = (j + s2) < n
                gr[m1] += w * (1.0 - x[j[m1] + s2])
                m2 = j >= s2
                gr[m2] -= w * x[j[m2] - s2]
        gr *= dx
        return obj, gr
    
    def obj(x):
        return _eval(x)[0]
    def jac(x):
        return _eval(x)[1]
    def cf(x):
        return x.sum() - target
    def cj(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
