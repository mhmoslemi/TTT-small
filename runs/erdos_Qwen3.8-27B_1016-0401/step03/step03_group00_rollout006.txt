import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

_G_N = 280
_G_TGT = 140.0
_G_T = 1.0
_G_K = 10

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_G_T * (c - m))
    return (m + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_G_T * (c - m))
    w = e / e.sum()
    grad = np.zeros(_G_N)
    for wi, ci in zip(w, idx):
        if wi > 1e-12:
            grad += wi * shift_grad(x, _G_N, ci - (_G_N - 1))
    return grad * (2.0 / _G_N)

def _g_cf(x):
    return x.sum() - _G_TGT

def _g_cj(x):
    return np.ones(_G_N)

def _slsqp(h, n, T, top_k, maxiter, ftol):
    global _G_N, _G_TGT, _G_T, _G_K
    _G_N = n
    _G_TGT = n / 2.0
    _G_T = float(T)
    _G_K = min(top_k, 2 * n - 1)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _g_cf, 'jac': _g_cj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def _grad_polish(h, n, n_iter, lr0, n_top, t_end):
    dx = 2.0 / n
    h_cur = h.copy()
    best_c = compute_c5(h_cur)
    best_h = h_cur.copy()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        idx = np.argsort(corr)[-n_top:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h_cur, n, ci - (n - 1))
        g *= dx / n_top
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g_unit = g / g_norm
        lr = lr0 / (1.0 + 0.005 * it)
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            trial = project_feasible(h_cur - lr * factor * g_unit, n)
            c = compute_c5(trial)
            if c < best_c - 1e-12:
                best_c = c
                best_h = trial.copy()
                h_cur = trial
                improved = True
                break
        if not improved:
            break
    return best_h, best_c

def _light_nesterov(h, n, n_iter, top_k, t_end):
    dx = 2.0 / n
    h_cur = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h_cur)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h_y, n, ci - (n - 1))
        g *= dx / top_k
        alpha = 0.3 / (1.0 + 0.001 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        if (it + 1) % 2000 == 0:
            c = compute_c5(h_cur)
            if c < best_c:
                best_c = c
                best_h = h_cur.copy()
    return best_h, best_c
