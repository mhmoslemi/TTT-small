        w = np.exp(temp * (c - m))
        w /= w.sum()
        g = np.zeros(n)
        for wi, ci in zip(w, idx):
            if wi > 1e-8:
                g += wi * shift_grad(h_y, n, ci - (n - 1))
        g *= dx
        gnorm = np.linalg.norm(g)
        alpha = 0.8 / (1.0 + 0.003 * it + gnorm * n * 0.1)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        if it % 4000 == 0:
            c5 = compute_c5(h_cur)
            if c5 < c5_best:
                c5_best, h_best = c5, h_cur.copy()
    c5 = compute_c5(h_cur)
    if c5 < c5_best:
        c5_best, h_best = c5, h_cur.copy()
    return h_best, c5_best

def perturb_h(h_best, n, rng, pi):
    mode = pi % 5
    if mode == 0:
        sig = 0.001 + 0.002 * (pi // 5) % 4
        return project_feasible(h_best + rng.normal(0, sig, n), n)
    elif mode == 1:
        h_p = h_best.copy()
        l = rng.integers(0, n - 30)
        ln = rng.integers(8, 30)
        h_p[l:l+ln] = 1.0 - h_p[l:l+ln]
        return project_feasible(h_p, n)
    elif mode == 2:
        h_p = h_best.copy()
        k = rng.integers(3, 15)
        for _ in range(3):
            i = rng.integers(0, n)
            h_p[i] = 1.0 - h_p[i]
        return project_feasible(h_p, n)
    elif mode == 3:
        h_p = h_best.copy()
        sig = 0.003 + 0.001 * (pi // 5) % 3
        h_p += rng.normal(0, sig, n)
        h_p[rng.integers(0, n)] = 1.0
        h_p[rng.integers(0, n)] = 0.0
        return project_feasible(h_p, n)
    else:
        h_p = h_best.copy()
        a, b = sorted(rng.choice(n, 2, replace=False))
        ln = min(b - a, 20)
        if ln > 2:
            h_p[a:a+ln] = 1.0 - h_p[a:a+ln]
        return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Nesterov with top-30, softmax (30% time)
    t1 = t0 + budget_s * 0.30
    h_best, c5_best = nesterov_opt(h_best, n, t1, top_k=30, temp=4.0)

    # Phase 2: SLSQP annealing (30% time)
    t2 = t0 + budget_s * 0.60
    slsqp_params = [(32, 30, 2500), (16, 25, 3000), (8, 20, 4000),
                    (4, 15, 5000), (2, 12, 6000), (1, 10, 7000),
                    (0.5, 8, 8000), (0.2, 6, 10000), (0.1, 5, 12000)]
    for T, topk, mi in slsqp_params:
        if time.time() > t2:
            break
        hn = slsqp_smooth(h_best, n, T, topk, mi)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 3: Nesterov again with smaller top-k and lower temp (15% time)
    t3 = t0 + budget_s * 0.75
    h_best, c5_best = nesterov_opt(h_best, n, t3, top_k=20, temp=8.0)

    # Phase 4: Perturbation + quick SLSQP (15% time)
    t4 = t0 + budget_s * 0.93
    trial = 0
    while time.time() < t4:
        trial += 1
        hp = perturb_h(h_best, n, rng, trial)
        hn = slsqp_smooth(hp, n, 2, 12, 2500)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 5: Final ultra-fine polish
    if time.time() < budget_s - 5:
        hn = slsqp_smooth(h_best, n, 0.05, 5, 15000)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn
    if time.time() < budget_s - 2:
        hn = slsqp_smooth(h_best, n, 0.01, 3, 8000)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
