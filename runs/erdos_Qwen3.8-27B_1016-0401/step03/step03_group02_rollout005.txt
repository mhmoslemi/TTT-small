def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)

            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

    n = len(x)
    o = lag_idx - center
    g = np.zeros(n)
    if o == 0:

    elif o > 0:

    else:
        p = -o
        g[:n-p] += 1.0 - x[p:]
        g[p:] += -x[:n-p]
    return g * dx

    dx = 2.0 / n

    # Active set: lags within 10% of max

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c5_cur = c5_new
        else:
            h[i], h[j] = h[j], h[i]
        ones = np.where(h > 0.7)[0]
        zeros = np.where(h < 0.3)[0]
    return h, c5_cur

def coord_descent_fine(h, target, n_passes=3):
    n = len(h)
    c5_cur, _ = compute_c5(h)
    dts = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for _ in range(n_passes):
        improved = False
        for j in range(n):
            for dt in dts:
                trial = h.copy()
                trial[j] = np.clip(trial[j] + dt, 0.0, 1.0)
                trial = fix_constraint(trial, target)
                c5_new, _ = compute_c5(trial)
                if c5_new < c5_cur - 1e-16:
                    h = trial
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def bipy_swap(h, target, radius=5, max_rounds=500):
    n = len(h)
    c5_cur, _ = compute_c5(h)
    for _ in range(max_rounds):
        best_delta = 1e-16
        best_move = None
        for i in range(n):
            for j in range(max(0, i-radius), min(n, i+radius)):
                if i == j:
                    continue
                if h[i] > 0.5 and h[j] < 0.5:
                    trial = h.copy()
                    trial[i] = 0.0
                    trial[j] = 1.0
                    trial = fix_constraint(trial, target)
                    c5_new, _ = compute_c5(trial)
                    delta = c5_new - c5_cur
                    if delta < best_delta:
                        best_delta = delta
                        best_move = (i, j, trial, c5_new)
                elif h[i] < 0.5 and h[j] > 0.5:
                    trial = h.copy()
                    trial[i] = 1.0
                    trial[j] = 0.0
                    trial = fix_constraint(trial, target)
                    c5_new, _ = compute_c5(trial)
                    delta = c5_new - c5_cur
                    if delta < best_delta:
                        best_delta = delta
                        best_move = (i, j, trial, c5_new)
        if best_move is None:
            break
        h = best_move[2]
        c5_cur = best_move[3]
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(340)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best, _ = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP on worst lag with high iterations (25% budget)
    p1_end = t0 + budget_s * 0.25
    for cycle in range(20):
        if elapsed() > p1_end:
            break
        h_new = slsqp_worst_lag(h_best, bounds, maxiter=3000, ftol=1e-20)
        h_new = fix_constraint(h_new, target)
        c5_new, _ = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 2: Annealing swaps (25% budget)
    p2_end = t0 + budget_s * 0.50
    T_start = 0.05
    while elapsed() < p2_end:
        n_steps = min(5000, int((p2_end - elapsed()) / 0.005))
        if n_steps < 100:
            break
        h_new, c5_new = annealing_swaps(h_best, target, n_steps,
                                         T_init=T_start, T_end=0.0005, rng=rng)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new
        T_start = max(0.01, T_start * 0.7)

    # Phase 3: SLSQP + bipy swap (20% budget)
    p3_end = t0 + budget_s * 0.70
    for cycle in range(15):
        if elapsed() > p3_end:
            break
        h_new = slsqp_worst_lag(h_best, bounds, maxiter=2000, ftol=1e-19)
        h_new = fix_constraint(h_new, target)
        c5_new, _ = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new
        if elapsed() > p3_end:
            break
        h_new, c5_new = bipy_swap(h_best, target, radius=6, max_rounds=200)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 4: Multi-n exploration (15% budget)
    p4_end = t0 + budget_s * 0.85
    x_old = np.arange(n) * (2.0 / n)
    for n_alt in [336, 344, 342, 338, 360, 320]:
        if elapsed() > p4_end:
            break
        x_new = np.linspace(0.0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        target_alt = n_alt / 2.0
        h_alt = fix_constraint(h_alt, target_alt)
        c5_alt, _ = compute_c5(h_alt)
        bounds_alt = [(0.0, 1.0)] * n_alt
        for _ in range(4):
            if elapsed() > p4_end:
                break
            h_new = slsqp_worst_lag(h_alt, bounds_alt, maxiter=1500, ftol=1e-18)
            h_new = fix_constraint(h_new, target_alt)
            c5_new, _ = compute_c5(h_new)
            if c5_new < c5_alt:
                c5_alt = c5_new
                h_alt = h_new
        h_alt, c5_a = annealing_swaps(h_alt, target_alt, 2000, 0.03, 0.002, rng)
        if c5_a < c5_alt:
            c5_alt = c5_a
            h_alt = h_alt
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt
            target = n / 2.0
            bounds = [(0.0, 1.0)] * n

    # Phase 5: Final polish (15% budget)
    p5_end = t0 + budget_s - 5.0
    n = len(h_best)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    for _ in range(10):
        if elapsed() > p5_end:
            break
        h_new = slsqp_worst_lag(h_best, bounds, maxiter=4000, ftol=1e-20)
        h_new = fix_constraint(h_new, target)
        c5_new, _ = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new
        if elapsed() > p5_end:
            break
        h_best, c5_cd = coord_descent_fine(h_best, target, n_passes=3)
        if c5_cd < c5_best:
            c5_best = c5_cd

    c5_final, _ = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    h_best = fix_constraint(h_best, len(h_best) / 2.0)
    c5_final, _ = compute_c5(h_best)
    return (h_best, c5_final, len(h_best))
