import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] + shift])
        if shift > 0:
            m2 = j >= shift
            g[m2] -= h[j[m2] - shift]
    else:
        s = -shift
        m1 = j >= s
        g[m1] += (1.0 - h[j[m1] - s])
        if s > 0:
            m2 = (j + s) < n
            g[m2] -= h[j[m2] + s]
    return g

_G_N = 280
_G_TGT = 140.0
_G_T = 1.0
_G_K = 10

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_G_T * (c - m))
    return (m + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_G_T * (c - m))
    w = e / e.sum()
    grad = np.zeros(_G_N)
    for wi, ci in zip(w, idx):
        if wi > 1e-12:
            grad += wi * shift_grad(x, _G_N, ci - (_G_N - 1))
    return grad * (2.0 / _G_N)

def _g_cf(x):
    return x.sum() - _G_TGT

def _g_cj(x):
    return np.ones(_G_N)

def slsqp_opt(h, n, T, top_k, maxiter, ftol):
    global _G_N, _G_TGT, _G_T, _G_K
    _G_N = n
    _G_TGT = n / 2.0
    _G_T = float(T)
    _G_K = min(top_k, 2 * n - 1)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _g_cf, 'jac': _g_cj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def peaked_subgradient(h, n, t_max):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = corr.max()

    idx = np.argsort(corr)[-top_k:]

    if not HAS_CP or len(active_shifts) == 0:
