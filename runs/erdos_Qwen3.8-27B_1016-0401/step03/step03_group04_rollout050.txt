        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Diverse restarts
    for pi in range(50):
        rem = budget_s - (time.time() - t0)
        if rem < 20:
            break
        r = rng
        if pi % 5 == 0:
            h_p = h_best + r.normal(0, 0.004, n)
        elif pi % 5 == 1:
            h_p = h_best.copy()
            idx = r.integers(0, n, 8)
            h_p[idx] = 1.0 - h_p[idx]
            h_p += r.normal(0, 0.003, n)
        elif pi % 5 == 2:
            x = np.arange(n, dtype=float)
            h_p = h_best + 0.015 * np.sin(2 * np.pi * r.uniform(2, 8) * x / n + r.uniform(0, 6.28))
            h_p += r.normal(0, 0.003, n)
        elif pi % 5 == 3:
            h_p = h_best.copy()
            bs = n // 6
            bi = int(r.integers(0, 6))
            h_p[bi * bs:(bi + 1) * bs] = 1.0 - h_p[bi * bs:(bi + 1) * bs]
            h_p += r.normal(0, 0.002, n)
        else:
            h_p = h_best + r.normal(0, 0.008, n)
        h_p = project_feasible(h_p, n)
        h_new = do_lbfgsb(h_p, n, 0.5, 8, 3000, t0, budget_s - 15)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            h_new, c5 = grad_local_search(h_best, n, t0, t0 + 25, rng, 10)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 7: Final local search + polish
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_new, c5 = local_search(h_best, n, t0, t0 + min(rem - 10, 120), rng, 30)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_new, c5 = grad_local_search(h_best, n, t0, t0 + min(rem - 10, 80), rng, 25)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        h_new = do_lbfgsb(h_best, n, 0.05, 2, 5000, t0, t0 + rem - 5)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 8: Try at different resolution if time remains
    rem = budget_s - (time.time() - t0)
    if rem > 200:
        n2 = n + 40
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, h_best)
        h2 = project_feasible(h2, n2)
        c5_2 = compute_c5(h2)
        for T, K, mi in [(4, 30, 3000), (0.3, 5, 4000)]:
            rem2 = budget_s - (time.time() - t0)
            if rem2 < 60:
                break
            h2_new = do_lbfgsb(h2, n2, T, K, mi, t0, budget_s - 40)
            c5_2 = compute_c5(h2_new)
            h2 = h2_new
        h2, c5_2 = grad_local_search(h2, n2, t0, budget_s - 10, rng, 20)
        if c5_2 < c5_best:
            c5_best = c5_2
            h_best = h2
            n = n2

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
