import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = float(y.min()) - 3.0, float(y.max()) + 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def shift_grad(h, n, sh):
    j = np.arange(n)
    g = np.zeros(n)
    if sh >= 0:
        m1, m2 = j >= sh, (j + sh) < n
        if m1.any(): g[m1] += 1.0 - h[j[m1] - sh]
        if m2.any(): g[m2] -= h[j[m2] + sh]
    else:
        s = -sh
        m1, m2 = (j + s) < n, j >= s
        if m1.any(): g[m1] += 1.0 - h[j[m1] + s]
        if m2.any(): g[m2] -= h[j[m2] - s]
    return g

def smvj(x, n, T, K):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    K = min(K, len(corr))
    topk = np.argpartition(corr, -K)[-K:]
    c_sel = corr[topk]
    msel = np.max(c_sel)
    sh = T * (c_sel - msel)
    e = np.exp(sh)
    val = (msel + np.log(e.sum()) / T) * (2.0 / n)
    sw = e / e.sum()
    ctr = n - 1
    g = np.zeros(n)
    for ii in range(len(sw)):
        w = sw[ii]
        if w < 1e-15: continue
        g += w * shift_grad(x, n, topk[ii] - ctr)
    return val, g * (2.0 / n)

_TG = [0.0, False]

def _f(x, n, T, K):
    if time.time() > _TG[0]: _TG[1] = True
    return smvj(x, n, T, K)[0]

def _g(x, n, T, K):
    if time.time() > _TG[0]: _TG[1] = True
    return smvj(x, n, T, K)[1]

def lbl(x0, n, T, K, mi, ft, tlim):
    _TG[0], _TG[1] = tlim, False
    bd = [(0.0, 1.0)] * n
    try:
        r = minimize(_f, x0, method='L-BFGS-B', jac=_g, args=(n, T, K),
                     bounds=bd, options={'maxiter': mi, 'ftol': ft,
                                         'gtol': ft * 0.04, 'maxls': 40,
                                         'maxfun': mi * 5})
        return x0 if _TG[1] else project_feasible(r.x, n)
    except Exception:
        return x0.copy()

def slq(x0, n, T, K, mi, ft, tlim):
    _TG[0], _TG[1] = tlim, False
    bd = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(_f, x0, method='SLSQP', jac=_g, args=(n, T, K),
                     bounds=bd, constraints=cons, options={'maxiter': mi, 'ftol': ft})
        return x0 if _TG[1] else project_feasible(r.x, n)
    except Exception:
        return x0.copy()
