def exact_coord_step(h, j):
    """Find optimal h[j] minimizing max corr, all else fixed."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    old = h[j]
    coef = np.zeros(2 * n - 1)
    # Terms where h[j] appears in h[i+k] (i+k=j → i=j-k): contributes t*(1-h[j-k])
    k_lo = max(0, j - n + 1)
    k_hi = min(j, 2 * n - 2)
    if k_lo <= k_hi:
        idx = np.arange(k_lo, k_hi + 1)
        coef[k_lo:k_hi + 1] += 1.0 - h[j - idx]
    # Terms where h[j] appears in (1-h[i]) (i=j): contributes h[j+k]*(1-t)
    k_lo2 = max(0, 0)  # j+k >= 0 always for k>=0
    k_hi2 = min(2 * n - 2, n - 1 - j)
    if k_lo2 <= k_hi2:
        idx2 = np.arange(k_lo2, k_hi2 + 1)
        coef[k_lo2:k_hi2 + 1] -= h[j + idx2]
    
    d_lo = -old
    d_hi = 1.0 - old
    # ternary search on [d_lo, d_hi] for min of max(corr + d*coef)
    lo, hi = d_lo, d_hi
    for _ in range(50):
        m1 = lo + (hi - lo) * 0.3333
        m2 = hi - (hi - lo) * 0.3333
        if np.max(corr + m1 * coef) <= np.max(corr + m2 * coef):
            hi = m2
        else:
            lo = m1
    d_opt = (lo + hi) * 0.5
    return np.clip(old + d_opt, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.argsort(h)
        else:
            idx = np.argsort(-h)
        remaining = diff
        for i in idx:
            if remaining > 0:
                add = min(remaining, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                remaining -= add
            else:
                sub = min(-remaining, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                remaining += sub
            if abs(remaining) < 1e-14:
                break
    return h

def exact_coord_opt(h, j, corr):
    n = len(h)
    old = h[j]
    coef = np.zeros(2 * n - 1)
    kl1 = max(0, j - n + 1)

        coef[kl1:kh1 + 1] += 1.0 - h[j - idx]
    kh2 = min(2 * n - 2, 2 * n - 2 - j)
    kl2 = max(0, n - 1 - j)
    if kl2 <= kh2:
        idx2 = np.arange(kl2, kh2 + 1)
        coef[kl2:kh2 + 1] -= h[j + idx2 - (n - 1)]
    self_m = n - 1
    d_lo, d_hi = -old, 1.0 - old
    base = corr + d_lo * coef

    vals = np.array([np.max(corr + d * coef - (d * (d + 2.0 * old)) if False else 0) for d in t_grid])
    # Proper evaluation including quadratic self-term

        new_corr = corr + d * coef

        v1 = max(v1, (corr[self_m] + m1 * coef[self_m]) - m1 * (m1 + 2.0 * old))
        v2 = np.max(corr + m2 * coef)
        v2 = max(v2, (corr[self_m] + m2 * coef[self_m]) - m2 * (m2 + 2.0 * old))
        if v1 <= v2:
            hi_d = m2
        else:
            lo_d = m1
    d_opt = (lo_d + hi_d) * 0.5
    return float(np.clip(old + d_opt, 0.0, 1.0))

def exact_cd(h, n_sweeps):
    n = len(h)
    target = n / 2.0
    c5_cur = compute_c5(h)
    for _ in range(n_sweeps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        any_change = False
        for j in range(n):
            old = h[j]
            t_new = exact_coord_opt(h, j, corr)
            if abs(t_new - old) > 1e-12:
                delta = t_new - old
                self_m = n - 1
                kl1 = max(0, j - n + 1)
                kh1 = min(j, 2 * n - 2)
                if kl1 <= kh1:
                    idx = np.arange(kl1, kh1 + 1)
                    corr[kl1:kh1 + 1] += delta * (1.0 - h[j - idx])
                kh2 = min(2 * n - 2, 2 * n - 2 - j)
                kl2 = max(0, n - 1 - j)

                    idx2 = np.arange(kl2, kh2 + 1)
                    corr[kl2:kh2 + 1] -= delta * h[j + idx2 - (n - 1)]

                    (1.0 - h[self_m - (n - 1)]) if 0 <= self_m - (n - 1) < n else 0.0
                    - h[self_m + 0 - (n - 1)] if 0 <= self_m + 0 - (n - 1) < n else 0.0)
                # Simpler: just recompute self term

        c5_new = compute_c5(h)
        if c5_new < c5_cur:

        if not any_change:
            break
    return h, c5_cur

def sm_obj(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    return (mc + np.log(np.sum(e)) / T) * (2.0 / n)

def sm_grad(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx

def lag_obj(x, lag_idx, n):
    return np.correlate(x, 1.0 - x, mode="full")[lag_idx] * (2.0 / n)

def lag_grad(x, lag_idx, n):
    o = lag_idx - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * x
    elif o > 0:
        g[o:] += 1.0 - x[:n - o]
        g[:n - o] += -x[o:]
    else:
        p = -o
        g[:n - p] += 1.0 - x[p:]
        g[p:] += -x[:n - p]
    return g * dx

_global_target = 0.0

def _eq_fun(x):
    return x.sum() - _global_target

def _eq_jac(x):
    return np.ones(len(x))

def slsqp_lag(h, lag_idx, bounds, maxiter, n):
    global _global_target
    _global_target = n / 2.0
    cons = [dict(type='eq', fun=_eq_fun, jac=_eq_jac)]
    try:
        res = minimize(lag_obj, h, args=(lag_idx, n), method='SLSQP',
                       jac=lambda x: lag_grad(x, lag_idx, n),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = np.clip(res.x, 0.0, 1.0)
        h_new = fix_constraint(h_new, n / 2.0)
        return h_new
    except Exception:
        return h

def slsqp_sm(h, T, bounds, maxiter, n):
    global _global_target
    _global_target = n / 2.0
    cons = [dict(type='eq', fun=_eq_fun, jac=_eq_jac)]
    try:
        res = minimize(sm_obj, h, args=(T, n), method='SLSQP',
                       jac=lambda x: sm_grad(x, T, n),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-17})
        h_new = np.clip(res.x, 0.0, 1.0)
        h_new = fix_constraint(h_new, n / 2.0)
        return h_new
    except Exception:
        return h
