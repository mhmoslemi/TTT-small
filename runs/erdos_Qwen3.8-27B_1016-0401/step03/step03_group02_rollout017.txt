            for i in order:
                s = min(rem, h[i])
                h[i] -= s
                rem -= s
                if rem < 1e-14:
                    break
    return h

def eq_fun(x):
    return x.sum() - _TARGET

def eq_jac(x):
    return np.ones(len(x))

def corr_grad_at_lag(x, lag_idx):
    n = len(x)
    o = lag_idx - (n - 1)
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * x
    elif o > 0:
        g[o:] += 1.0 - x[:n - o]
        g[:n - o] += -x[o:]
    else:
        p = -o
        g[:n - p] += 1.0 - x[p:]
        g[p:] += -x[:n - p]
    return g

def topK_sm_obj(x, T, K):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    if K < len(corr):
        idx = np.argsort(corr)[-K:]
        vals = corr[idx]
    else:
        idx = np.arange(len(corr))
        vals = corr
    mc = np.max(vals)
    sh = T * (vals - mc)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def topK_sm_grad(x, T, K):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    if K < len(corr):
        idx = np.argsort(corr)[-K:]
        vals = corr[idx]
    else:
        idx = np.arange(len(corr))
        vals = corr
    mc = np.max(vals)
    sh = T * (vals - mc)
    e = np.exp(sh)
    sw = e / np.sum(e)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(len(idx)):
        if sw[i] < 1e-12:
            continue
        grad += sw[i] * corr_grad_at_lag(x, idx[i])
    return grad * dx

def slsqp_topK(h, T, K, bounds, maxiter, ftol):
    global _TARGET
    _TARGET = len(h) / 2.0
    cons = [dict(type='eq', fun=eq_fun, jac=eq_jac)]
    res = minimize(topK_sm_obj, h, args=(T, K), method='SLSQP',
                   jac=topK_sm_grad, bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return np.clip(res.x, 0.0, 1.0)

def lag_obj_wrap(x, lag, dx):
    return np.correlate(x, 1.0 - x, mode="full")[lag] * dx

def lag_grad_wrap(x, lag, dx):
    return corr_grad_at_lag(x, lag) * dx

def slsqp_lag(h, lag_idx, bounds, maxiter, ftol):
    global _TARGET
    _TARGET = len(h) / 2.0
    cons = [dict(type='eq', fun=eq_fun, jac=eq_jac)]
    n = len(h)
    dx = 2.0 / n
    res = minimize(lag_obj_wrap, h, args=(lag_idx, dx), method='SLSQP',
                   jac=lag_grad_wrap, bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return np.clip(res.x, 0.0, 1.0)

def fine_cd(h, n_sweeps):
    n = len(h)
    target = n / 2.0
    c5_cur = compute_c5(h)
    dts = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.02, -0.02, 0.001, -0.001]
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            for dt in dts:
                trial = h.copy()
                trial[j] = np.clip(trial[j] + dt, 0.0, 1.0)
                trial = fix_constraint(trial, target)
                c5_new = compute_c5(trial)
                if c5_new < c5_cur - 1e-16:
                    h = trial
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def sa_cd(h, n_iters, T_start, T_end, seed_val):
    n = len(h)
    target = n / 2.0
    c5_cur = compute_c5(h)
    rng = np.random.default_rng(seed_val)
    for it in range(n_iters):
        frac = it / max(n_iters, 1)
        T_sa = T_start * (T_end / T_start) ** frac
        j = rng.integers(0, n)
        dt = rng.choice([0.02, -0.02, 0.01, -0.01, 0.04, -0.04])
        trial = h.copy()
        trial[j] = np.clip(trial[j] + dt, 0.0, 1.0)
        trial = fix_constraint(trial, target)
        c5_new = compute_c5(trial)
        delta = c5_new - c5_cur
        if delta < 0 or (T_sa > 1e-12 and rng.random() < np.exp(-delta / T_sa)):
            h = trial
            c5_cur = c5_new
    return h, c5_cur

def project_grad(h, T, K, lr, n_steps, bounds):
    global _TARGET
    n = len(h)
    target = n / 2.0
    _TARGET = target
    v = np.zeros(n)
    mom = 0.8
    for _ in range(n_steps):
        g = topK_sm_grad(h, T, K)
        v = mom * v - lr * g
        h_new = fix_constraint(h + v, target)
        if compute_c5(h_new) < compute_c5(h):
            h = h_new
        else:
            lr *= 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global _TARGET
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    _TARGET = target
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Top-K soft-max sweeping (30%)
    p1_end = t0 + budget_s * 0.30
    configs = [(100, 5), (500, 8), (1000, 10), (3000, 12), (8000, 15),
               (20000, 20), (50000, 30), (100000, 40), (300000, 60)]
    for T, K in configs:
        if elapsed() > p1_end:
            break
        try:
            h_new = slsqp_topK(h_best, T, K, bounds, 2500, 1e-18)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass
        if elapsed() < p1_end:
            corr_b = corr_full(h_best)
            worst = int(np.argmax(corr_b))
            try:
                h_new = slsqp_lag(h_best, worst, bounds, 400, 1e-18)
                h_new = fix_constraint(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-15:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass
    # Second refined pass
    for T, K in [(200, 3), (800, 5), (3000, 8), (15000, 15)]:
        if elapsed() > p1_end:
            break
        try:
            h_new = slsqp_topK(h_best, T, K, bounds, 3500, 1e-19)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 2: Projected gradient + SA + lag SLSQP (25%)
    p2_end = t0 + budget_s * 0.55
    for _ in range(8):
        if elapsed() > p2_end:
            break
        h_best, c5_cd = fine_cd(h_best, 2)
        if c5_cd < c5_best:
            c5_best = c5_cd
        h_sa, c5_sa = sa_cd(h_best, 3000, 0.005, 0.0005, seed + 100)
        if c5_sa < c5_best:
            c5_best = c5_sa
            h_best = h_sa
        # Projected gradient step
        try:
            h_pg = project_grad(h_best, 3000, 12, 0.3, 5, bounds)
            h_pg = fix_constraint(h_pg, target)
            c5_pg = compute_c5(h_pg)
            if c5_pg < c5_best:
                c5_best = c5_pg
                h_best = h_pg
        except Exception:
            pass
        # SLSQP on worst lag
        corr_b = corr_full(h_best)
        worst = int(np.argmax(corr_b))
        try:
            h_new = slsqp_lag(h_best, worst, bounds, 600, 1e-18)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 3: Multi-n exploration (15%)
    p3_end = t0 + budget_s * 0.70
    current_n = len(h_best)
    for n_alt in [320, 360, 380, 300, 400, 340]:
        if elapsed() > p3_end:
            break
        if n_alt == current_n:
            continue
        target_alt = n_alt / 2.0
        x_old = np.arange(len(h_best)) * (2.0 / len(h_best))
        x_new = np.linspace(0.0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = fix_constraint(h_alt, target_alt)
        c5_alt = compute_c5(h_alt)
        bounds_alt = [(0.0, 1.0)] * n_alt
        for T, K in [(500, 8), (5000, 15), (30000, 30)]:
            if elapsed() > p3_end:
                break
            try:
                h_new = slsqp_topK(h_alt, T, K, bounds_alt, 2000, 1e-17)
                h_new = fix_constraint(h_new, target_alt)
                c5_new = compute_c5(h_new)
                if c5_new < c5_alt - 1e-15:
                    c5_alt = c5_new
                    h_alt = h_new
            except Exception:
                pass
        h_alt, _ = fine_cd(h_alt, 1)
        c5_alt = compute_c5(h_alt)
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            current_n = n_alt
            target = n_alt / 2.0
            bounds = [(0.0, 1.0)] * n_alt

    # Phase 4: Intensive final polish (20%)
    p4_end = t0 + budget_s - 2.0
    target = len(h_best) / 2.0
    n = len(h_best)
    bounds = [(0.0, 1.0)] * n
    _TARGET = target
    forouter in range(30):
        if elapsed() > p4_end:
            break
        # Cycle through top lags with SLSQP
        corr_f = corr_full(h_best)
        top_lags = np.argsort(corr_f)[-4:]
        for lag in top_lags:
            if elapsed() > p4_end:
                break
            try:
                h_new = slsqp_lag(h_best, int(lag), bounds, 1000, 1e-19)
                h_new = fix_constraint(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass
        # Top-K SLSQP
        if elapsed() < p4_end:
            try:
                h_new = slsqp_topK(h_best, 5000, 10, bounds, 2500, 1e-19)
                h_new = fix_constraint(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass
        # Fine coordinate descent
        if elapsed() < p4_end:
            h_best, c5_cd_f = fine_cd(h_best, 2)
            if c5_cd_f < c5_best:
                c5_best = c5_cd_f

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
