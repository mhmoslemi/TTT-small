def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

_G = {}

def _obj(x):
    n, T, top_k = _G['n'], _G['T'], _G['tk']
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    return (m + np.log(e.sum()) / T) * (2.0 / n)

def _jac(x):
    n, T, top_k = _G['n'], _G['T'], _G['tk']
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    for wi, ci in zip(w, idx):
        if wi > 1e-12:
            grad += wi * shift_grad(x, n, ci - (n - 1))
    return grad * (2.0 / n)

def _eq_fun(x):
    return x.sum() - _G['target']

def _eq_jac(x):
    return np.ones(_G['n'])

def slsqp_opt(h, n, T, top_k, maxiter, ftol=1e-14):
    _G['n'], _G['T'], _G['tk'], _G['target'] = n, T, top_k, n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h.copy(), method='SLSQP', jac=_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def subgradient_polish(h, n, t_end, n_iter=200000):
    dx = 2.0 / n
    h_cur, h_prev = h.copy(), h.copy()
    h_avg = h.copy()
    best_c5, best_h = compute_c5(h_cur), h_cur.copy()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        beta = float(it) / (it + 2.5) if it > 0 else 0.0
        h_y = h_cur + beta * (h_cur - h_prev)
        h_y = project_feasible(h_y, n)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        m = corr.max()
        thresh = m - 0.015 * max(m, 1e-10)
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        g = np.zeros(n)
        for ci in active:
            g += shift_grad(h_y, n, ci - (n - 1))
        g *= dx / len(active)
        alpha = 0.4 / (1.0 + 0.0008 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 3000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < best_c5:
                best_c5, best_h = c1, h_cur.copy()
            c2 = compute_c5(project_feasible(h_avg, n))
            if c2 < best_c5:
                best_c5, best_h = c2, project_feasible(h_avg, n).copy()
    c1 = compute_c5(h_cur)
    if c1 < best_c5:
        best_c5, best_h = c1, h_cur.copy()
    return best_h, best_c5

def coord_search(h, n, t_end, n_passes=25):
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    deltas = [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]
    for p in range(n_passes):
        if time.time() > t_end:
            break
        improved = False
        step = [deltas[0:2] if p < 10 else deltas[0:4] if p < 20 else deltas]
        for i in range(n):
            if time.time() > t_end:
                break
            for d in step:
                h_try = h_cur.copy()
                h_try[i] = np.clip(h_try[i] + d, 0.0, 1.0)
                h_try = project_feasible(h_try, n)
                c5 = compute_c5(h_try)
                if c5 < best_c5 - 1e-9:
                    h_cur, best_c5 = h_try, c5
                    improved = True
                    break
        if not improved and p >= 5:
            break
    return h_cur, best_c5

def perturb(h_best, n, rng, strength=1.0):
    mode = int(rng.integers(0, 5))
    if mode == 0:
        sig = 0.002 + 0.003 * (1.5 - strength)
        return project_feasible(h_best + rng.normal(0, sig, n), n)
    elif mode == 1:
        h_p = h_best.copy()
        lb = int(rng.integers(0, n - 12))
        ln = int(rng.integers(4, 14))
        h_p[lb:lb+ln] = 1.0 - h_p[lb:lb+ln]
        return project_feasible(h_p, n)
    elif mode == 2:
        h_p = h_best.copy()
        idx = rng.choice(n, size=4, replace=False)
        h_p[idx] = 1.0 - h_p[idx]
        return project_feasible(h_p, n)
    elif mode == 3:
        sig = 0.008 * strength
        h_p = h_best + rng.normal(0, sig, n)
        h_p[int(rng.integers(n))] = 1.0
        h_p[int(rng.integers(n))] = 0.0
        return project_feasible(h_p, n)
    else:
        h_p = h_best.copy()
        sig = 4.0
        k_range = np.arange(-12, 13)
        kernel = np.exp(-0.5 * (k_range / sig) ** 2)
        h_p = np.convolve(h_p, kernel, mode='same')
        return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n = 280
        h_best = np.zeros(n)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse to fine (25% budget)
    t1 = t0 + budget_s * 0.25
    schedule = [(32, 20, 1200), (16, 16, 1500), (8, 12, 2000),
                (4, 10, 2500), (2, 8, 3000), (1, 6, 3500), (0.5, 5, 4000)]
    for T, tk, mi in schedule:
        if time.time() > t1:
            break
        hn = slsqp_opt(h_best, n, T, tk, mi, 1e-15)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 2: Subgradient Nesterov (20% budget)
    t2 = t0 + budget_s * 0.45
    hs, c5s = subgradient_polish(h_best, n, t2, n_iter=250000)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs

    # Phase 3: Multi-restart (30% budget)
    t3 = t0 + budget_s * 0.75
    n_restarts = 45
    for pi in range(n_restarts):
        if time.time() > t3:
            break
        strength = max(0.2, 1.0 - pi / (n_restarts * 0.8))
        hp = perturb(h_best, n, rng, strength)
        # Quick SLSQP
        hn = slsqp_opt(hp, n, 4, 10, 1200, 1e-14)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn
        # Brief subgradient if promising
        if c5 < c5_best + 0.002 and time.time() < t3 - 5:
            hs2, c5s2 = subgradient_polish(hn, n, min(time.time() + 15, t3), n_iter=40000)
            if c5s2 < c5_best:
                c5_best, h_best = c5s2, hs2

    # Phase 4: Coordinate search (12% budget)
    t4 = min(time.time() + budget_s * 0.12, t_end - 15)
    hc, c5c = coord_search(h_best, n, t4, n_passes=30)
    if c5c < c5_best:
        c5_best, h_best = c5c, hc

    # Phase 5: Final polish (remaining budget)
    if time.time() < t_end - 10:
        hn = slsqp_opt(h_best, n, 1, 5, 3000, 1e-16)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn
    if time.time() < t_end - 5:
        hf, c5f = subgradient_polish(h_best, n, t_end - 2, n_iter=50000)
        if c5f < c5_best:
            c5_best, h_best = c5f, hf
    if time.time() < t_end - 2:
        hc2, c5c2 = coord_search(h_best, n, t_end - 2, n_passes=10)
        if c5c2 < c5_best:
            c5_best, h_best = c5c2, hc2

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
