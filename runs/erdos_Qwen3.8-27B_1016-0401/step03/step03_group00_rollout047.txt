import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, s):
    j = np.arange(n)
    g = np.zeros(n)
    if s == 0:
        return 1.0 - 2.0 * h
    if s > 0:
        m1 = j < (n - s)
        g[m1] += (1.0 - h[j[m1] + s])
        m2 = j >= s
        g[m2] -= h[j[m2] - s]
    else:
        t = -s
        m1 = j >= t
        g[m1] += (1.0 - h[j[m1] - t])
        m2 = j < (n - t)
        g[m2] -= h[j[m2] + t]
    return g

def _slsqp_obj(x, n, T, topk):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-topk:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    return (m + np.log(np.sum(e)) / T) * (2.0 / n)

def _slsqp_jac(x, n, T, topk):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-topk:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    for wi, ci in zip(w, idx):
        if wi > 1e-12:
            grad += wi * shift_grad(x, n, ci - (n - 1))
    return grad * (2.0 / n)

def _slsqp_run(h, n, T, topk, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    def cf(x): return x.sum() - n / 2.0
    def cj(x): return np.ones(n)
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP', jac=lambda x: _slsqp_jac(x, n, T, topk),
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def _fista(h_init, n, t_end, rng, topk=20):

    h_cur = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5a = compute_c5(h_init)
    ha = h_init.copy()
    L = 4.0 / dx + 1.0  # Lipschitz estimate
    step = 1.0 / np.sqrt(L)
    for it in range(200000):
        if time.time() > t_end:
            break
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-topk:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h_y, n, ci - (n - 1))
        g *= dx / topk
        # Backtracking
        alpha = step
        h_new = project_feasible(h_y - alpha * g, n)
        f_new = compute_c5(h_new)
        for _ in range(10):
            if f_new < c5a + 0.01 / (it + 1) or alpha < 1e-8:

            h_new = project_feasible(h_y - alpha * g, n)

        step = min(2.0 * step, 1.0 / (4.0 * (it + 1)))
        h_prev = h_cur
        h_cur = h_new
