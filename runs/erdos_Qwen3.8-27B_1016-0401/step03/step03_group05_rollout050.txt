        mx = vals.max()
        active = np.where(vals >= mx - 1e-14)[0]

            idx2 = m + shift

            b[m2] -= h[idx2[m2]]

            delta_lo = -h[m]
            delta_hi = 1.0 - h[m]

            vals = corr + 0.0 * b  # just corr

                sg_max = b[active].max()

                    continue  # already optimal

                vals = corr + mid * b
                mx = vals.max()
                active = np.where(vals >= mx - 1e-14)[0]

                sg_max = b[active].max()

                elif sg_max <= 0:

        return res.x
    except Exception:
        return h.copy()

def coord_grad_vec(h, n, m):
    n_corr = 2 * n - 1
    i = np.arange(n_corr)
    shift = i - (n - 1)
    idx1 = m - shift
    idx2 = m + shift
    b = np.zeros(n_corr)
    m1 = (idx1 >= 0) & (idx1 < n)
    m2 = (idx2 >= 0) & (idx2 < n)
    b[m1] = 1.0 - h[idx1[m1]]
    b[m2] -= h[idx2[m2]]
    return b

def minimize_1d_max(corr, b, lo, hi):
    if lo > hi:
        lo, hi = hi, lo
    def subgrad(t):
        vals = corr + t * b
        mx = vals.max()
        active = np.where(vals >= mx - 1e-13)[0]
        return b[active].min(), b[active].max()
    sg0_lo, sg0_hi = subgrad(lo)
    if sg0_lo >= 0:
        return lo
    sgn_lo, sgn_hi = subgrad(hi)
    if sgn_hi <= 0:
        return hi
    for _ in range(50):
        mid = (lo + hi) / 2.0
        sgm_lo, sgm_hi = subgrad(mid)
        if sgm_lo >= 0:
            hi = mid
        elif sgm_hi <= 0:
            lo = mid
        else:
            return mid
    return (lo + hi) / 2.0

def coord_descent_v2(h_init, n, rng, time_limit_s):
    h = h_init.copy()
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_best = float(np.max(corr) * 2.0 / n)
    h_best = h.copy()
    t_start = time.time()
    sweep = 0
    while time.time() - t_start < time_limit_s:
        order = rng.permutation(n)
        for m in order:
            b = coord_grad_vec(h, n, m)
            d_lo = -h[m]
            d_hi = 1.0 - h[m]
            if d_lo >= d_hi:
                d_lo, d_hi = d_hi, d_lo
            if d_lo == d_hi:
                continue
            mx0 = corr.max()
            act0 = np.where(corr >= mx0 - 1e-13)[0]
            if act0.size > 0 and b[act0].min() <= 0.0 and b[act0].max() >= 0.0:
                continue
            delta = minimize_1d_max(corr, b, d_lo, d_hi)
            if abs(delta) < 1e-14:
                continue
            old_val = h[m]
            new_val = old_val + delta
            if new_val < 0.0 or new_val > 1.0:
                new_val = np.clip(new_val, 0.0, 1.0)
            delta_actual = new_val - old_val
            if abs(delta_actual) < 1e-14:
                continue
            h[m] = new_val
            corr += delta_actual * b
        # Project to maintain sum
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5_cur = float(np.max(corr) * 2.0 / n)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()
        sweep += 1
    return h_best, c5_best

def nesterov_adaptive(h_init, n, t0, remain_s, max_iters=500000):
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    alpha0 = 0.5
    for it in range(max_iters):
        if time.time() - t0 > remain_s:
            break
        frac = min(it / max_iters, 1.0)
        K = max(1, int(1 + 49 * frac * frac))
        beta = (float(it) / (it + 1.5)) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        if K <= 1:
            idx = int(np.argmax(corr))
            g = coord_grad_vec(h_y, n, 0)  # placeholder, need shift_grad
            # Use direct computation
            j = np.arange(n)
            shift = idx - center
            g = np.zeros(n)
            if shift >= 0:
                m1 = j >= shift
                m2 = (j + shift) < n
                g[m1] = (1.0 - h_y[j[m1] - shift])
                g[m2] -= h_y[j[m2] + shift]
            else:
                s = -shift
                m1 = (j + s) < n
                m2 = j >= s
                g[m1] = (1.0 - h_y[j[m1] + s])
                g[m2] -= h_y[j[m2] - s]
            g *= dx
        else:
            top_idx = np.argpartition(corr, -K)[-K:]
            top_vals = corr[top_idx]
            w = (top_vals / max(mc, 1e-15)) ** 8
            ws = w.sum()
            if ws < 1e-15:
                w = np.ones(K) / K
            else:
                w /= ws
            j = np.arange(n)
            g = np.zeros(n)
            for ii in range(K):
                shift = top_idx[ii] - center
                if shift >= 0:
                    m1 = j >= shift
                    m2 = (j + shift) < n
                    g[m1] += w[ii] * (1.0 - h_y[j[m1] - shift])
                    g[m2] -= w[ii] * h_y[j[m2] + shift]
                else:
                    s = -shift
                    m1 = (j + s) < n
                    m2 = j >= s
                    g[m1] += w[ii] * (1.0 - h_y[j[m1] + s])
                    g[m2] -= w[ii] * h_y[j[m2] - s]
            g *= dx
        alpha = alpha0 / (1.0 + 0.003 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 50000 == 0:
            c5c = compute_c5(h_sg)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h_sg.copy()
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse
    for T in [12, 6, 3]:
        if time.time() - t0 > budget_s - 500:
            break
        h_new = run_slsqp(h_best, n, T, 4000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Coordinate descent
    remain_cd = budget_s - 550
    if remain_cd > 120:
        h_cd, c5_cd = coord_descent_v2(h_best, n, rng, min(remain_cd, 400))
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    # Phase 3: SLSQP fine
    for T in [2, 1, 0.5, 0.3]:
        if time.time() - t0 > budget_s - 200:
            break
        h_new = run_slsqp(h_best, n, T, 5000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_cd if hasattr(h_best, '_orig') else h_new
            h_best = h_new

    # Phase 4: More coordinate descent
    remain_cd2 = budget_s - 150
    if remain_cd2 > 60:
        h_cd, c5_cd = coord_descent_v2(h_best, n, rng, min(remain_cd2, 300))
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    # Phase 5: Final SLSQP + coord descent
    if time.time() - t0 < budget_s - 80:
        h_new = run_slsqp(h_best, n, 0.2, 3000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    if time.time() - t0 < budget_s - 30:
        h_cd, c5_cd = coord_descent_v2(h_best, n, rng, budget_s - 30)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    # Try alternate resolution
    elapsed = time.time() - t0
    if elapsed < budget_s - 150:
        for n2 in [n + 20, n + 40]:
            if n2 > 600 or time.time() - t0 > budget_s - 80:
                break
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0, 2.0, n2, endpoint=False)
            h_alt = np.interp(x_new, x_old, h_best)
            h_alt = project_feasible(h_alt, n2)
            c5_alt = compute_c5(h_alt)
            if time.time() - t0 < budget_s - 50:
                h_cd2, c5_cd2 = coord_descent_v2(h_alt, n2, rng, budget_s - 50)
                if c5_cd2 < c5_alt:
                    c5_alt = c5_cd2
                    h_alt = h_cd2
            if c5_alt < c5_best:
                c5_best = c5_alt
                h_best = h_alt
                n = n2

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
