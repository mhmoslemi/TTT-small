import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

def corr_grad(h, n, T, top_k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    se = e.sum()
    val = (m + np.log(se) / T) * dx
    w = e / se
    grad = np.zeros(n)
    for wi, ci in zip(w, idx):
        if wi > 1e-12:
            grad += wi * shift_grad(h, n, ci - (n - 1))
    grad *= dx
    return val, grad

def nesterov_loop(h0, n, t_end, iters):
    h_cur = h0.copy()
    h_prev = h0.copy()
    h_avg = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    L = 20.0
    for it in range(iters):
        if time.time() > t_end:
            break
        prog = min(it / max(iters, 1), 1.0)
        T = max(1.5, 50.0 * (1.0 - prog) ** 1.5 + 1.5)
        top_k = max(5, int(14 - 8 * prog))
        beta = float(it) / (it + 2.5) if it > 0 else 0.0
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        val_y, g_y = corr_grad(h_y, n, T, top_k)
        alpha = 1.0 / (L * (it + 1))
        h_new = project_feasible(h_y - alpha * g_y, n)
        val_new, _ = corr_grad(h_new, n, T, top_k)
        if val_new > val_y + 1e-10:
            alpha *= 0.4
            h_new = project_feasible(h_y - alpha * g_y, n)
        gd = np.linalg.norm(g_y)
        sd = np.linalg.norm(h_new - h_y)
        if sd > 1e-10:
            L = max(L * 0.98, gd / sd + 1e-4)
            L = min(L, 200.0)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 4000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < best_c5:
                best_c5, best_h = c1, h_cur.copy()
            c2 = compute_c5(project_feasible(h_avg, n))
            if c2 < best_c5:
                best_c5, best_h = c2, project_feasible(h_avg, n).copy()
    c_av = compute_c5(project_feasible(h_avg, n))
    if c_av < best_c5:
        best_c5, best_h = c_av, project_feasible(h_avg, n)
    return best_h, best_c5

def slsqp_polish(h0, n, T, top_k, maxiter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        m = c.max()
        e = np.exp(T * (c - m))
        return (m + np.log(e.sum()) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        m = c.max()
        e = np.exp(T * (c - m))
        w = e / e.sum()
        grad = np.zeros(n)
        for wi, ci in zip(w, idx):
            if wi > 1e-12:
                grad += wi * shift_grad(x, n, ci - (n - 1))
        return grad * dx
    def cf(x):
        return x.sum() - n / 2.0
    def cj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h0.copy()

def make_perturb(h_best, n, rng, pi):

    if mod == 0:
        sig = 0.002 + 0.001 * (pi // 5) % 4
        return project_feasible(h_best + rng.normal(0, sig, n), n)
    elif mod == 1:

        lb = int(rng.integers(0, n - 8))

        hp[b1:b1+bl], hp[b2:b2+bl] = hp[b2:b2+bl].copy(), hp[b1:b1+bl].copy()
        return project_feasible(hp, n)
    elif mod == 3:
        h_s = h_best.copy()
        h_s[h_s < 0.2] = 0.0
        h_s[h_s > 0.8] = 1.0

    else:

        lb = int(rng.integers(0, n - 20))

        hp[lb:lb+ln] = 0.5 + 0.5 * np.sin(2.0 * np.pi * np.arange(ln) / ln)
        return project_feasible(hp, n)
