def topk_lob(x, n, T, k):

    corr = np.correlate(x, 1.0 - x, mode="full")

    lr = lr0
    T_ad = T0
    j = np.arange(n)
    center = n - 1
    for it in range(400000):
        if time.time() > t_end:
            break
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        ke = min(topk, len(corr))
        idx = np.argpartition(corr, -ke)[-ke:]
        tv = corr[idx]
        mc = np.max(tv)
        T_eff = max(T_ad, 0.05)
        sh = T_eff * (tv - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(ke):
            w = sw[oi]
            if w < 1e-12:
                continue
            shift = idx[oi] - center
            if shift >= 0:
                m1 = j >= shift
                grad[m1] += w * (1.0 - h_y[j[m1] - shift])
                m2 = (j + shift) < n
                grad[m2] -= w * h_y[j[m2] + shift]
            else:
                s2 = -shift
                m1 = (j + s2) < n
                grad[m1] += w * (1.0 - h_y[j[m1] + s2])
                m2 = j >= s2
                grad[m2] -= w * h_y[j[m2] - s2]
        grad *= dx / ke
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        h_z = h_y + beta * (h_y - h_z)
        h_y = h_z - lr * grad
        h_y = project_feasible(h_y, n)
        t_mom = t_new
        lr *= dec
        T_ad = max(0.05, T_ad * 0.9995)
        if (it + 1) % 4000 == 0:
            c5 = compute_c5(h_y)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_y.copy()
    return h_best, c5_best

def perturb(h, n, rng):
    hp = h.copy()
    pt = rng.integers(0, 6)
    if pt == 0:
        hp += rng.uniform(0.003, 0.012) * rng.normal(0, 1, n)
    elif pt == 1:
        ln = rng.integers(4, min(18, n - 2))
        st = rng.integers(0, max(1, n - ln))
        hp[st:st+ln] = 1.0 - hp[st:st+ln]
    elif pt == 2:
        fr = rng.integers(3, 14)
        am = rng.uniform(0.01, 0.04)
        x = np.arange(n) * 2.0 / n
        ph = rng.uniform(0, 2*np.pi)
        hp += am * np.sin(np.pi * fr * x + ph)
    elif pt == 3:
        ln = rng.integers(5, 18)
        s1 = rng.integers(0, max(1, n - ln))
        s2 = (s1 + rng.integers(ln+1, n)) % (n - ln)
        tmp = hp[s1:s1+ln].copy()
        hp[s1:s1+ln] = hp[s2:s2+ln]
        hp[s2:s2+ln] = tmp
    elif pt == 4:
        s = rng.integers(1, 3)
        hp = np.roll(hp, int(s))
    else:
        mask = rng.random(n) < 0.06
        hp[mask] = rng.random(int(mask.sum()))
    return np.clip(hp, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Coarse-to-fine SLSQP
    for T, k, mi in [(32, 40, 3000), (8, 15, 4000), (2, 8, 5000), (1, 5, 5000)]:
        if time.time() > t_end - 350:
            break
        h_new = slsqp_run(h_best, n, T, k, mi, 1e-18, t_end - 300)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: FISTA with top-8 adaptive
    rem = budget_s - (time.time() - t0)
    if rem > 250:
        h_f, c5_f = fista_run(h_best, n, t0 + rem * 0.28, topk=8, T0=3.0, lr0=0.01)
        h_f = project_feasible(h_f, n)
        c5_f = compute_c5(h_f)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f

    # Phase 3: Very fine SLSQP (multiple passes at small T)
    fine_sched = [(0.5, 4, 5000), (0.25, 3, 5000), (0.15, 2, 5000),
                  (0.08, 1, 5000), (0.05, 1, 5000), (0.03, 1, 5000)]
    for T, k, mi in fine_sched:
        if time.time() > t_end - 250:
            break
        for _ in range(2):
            if time.time() > t_end - 200:
                break
            h_new = slsqp_run(h_best, n, T, k, mi, 1e-19, t_end - 150)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 4: Perturb + SLSQP cycles
    rem = budget_s - (time.time() - t0)
    n_perts = max(1, int(rem / 22))
    for pi in range(n_perts):
        if time.time() > t_end - 35:
            break
        h_p = perturb(h_best, n, rng)
        h_p = project_feasible(h_p, n)
        h_new = slsqp_run(h_p, n, 1.0, 6, 2500, 1e-17, t_end - 25)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            if time.time() < t_end - 80:
                h_new2 = slsqp_run(h_new, n, 0.3, 2, 4000, 1e-19, t_end - 25)
                h_new2 = project_feasible(h_new2, n)
                c5_2 = compute_c5(h_new2)
                if c5_2 < c5_best:
                    c5_best = c5_2
                    h_best = h_new2

    # Final polish
    if time.time() < t_end - 25:
        h_new = slsqp_run(h_best, n, 0.1, 1, 4000, 1e-19, t_end - 10)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
