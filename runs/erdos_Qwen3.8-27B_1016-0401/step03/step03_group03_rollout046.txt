from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        elif s < target:
            hi = mid
        else:
            break
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        i = np.arange(n - shift)
        g[i] += (1.0 - h[i + shift])
        g[i + shift] -= h[i]
    else:
        s = -shift
        i = np.arange(s, n)
        g[i] += (1.0 - h[i - s])
        g[i - s] -= h[i]
    return g

def _gn(h_arr, n, K, T):
    """Compute weighted subgradient using top-K shifts."""
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h_arr, 1.0 - h_arr, mode="full")
    top_idx = np.argsort(corr)[-K:]
    mv = corr[top_idx].max()
    w = np.exp(T * (corr[top_idx] - mv))
    w /= w.sum()
    g = np.zeros(n)
    for wk, ik in zip(w, top_idx):
        g += wk * shift_grad(h_arr, ik - center)
    return g * dx, corr

def slsqp_polish(h, n, T, maxiter, ftol=1e-14):

    center = n - 1
    target = n / 2.0
    
    def obj(x):

        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    
    def jac(x):

            w = sw[oi]
            if w < 1e-13:

            grad += w * shift_grad(x, oi - center)

        return np.ones(n)

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 1.5
    hi = y.max() + 1.5
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        elif s < target:
            hi = mid
        else:
            break
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        i = np.arange(n - shift)
        g[i] += (1.0 - h[i + shift])
        g[i + shift] -= h[i]
    else:
        s = -shift
        i = np.arange(s, n)
        g[i] += (1.0 - h[i - s])
        g[i - s] -= h[i]
    return g

def _slsqp_obj(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _G_T) * dx

def _slsqp_jac(x):
    n = _G_N
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    grad = np.zeros(n)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-13:
            continue
        grad += w * shift_grad(x, oi - center)
    return grad * dx

def _slsqp_cons(x):
    return x.sum() - _G_N / 2.0

    global _G_N, _G_T
    _G_N = n
    _G_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons, 'jac': _slsqp_cons_jac}]
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP', jac=_slsqp_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return res.x
    except Exception:
        return h.copy()

                   a0=0.05, max_it=200000):

    h_p = h_init.copy()
    c5b = compute_c5(h_init)
    hb = h_init.copy()

    dx = 2.0 / n
    center = n - 1

        if time.time() > t_end:

        top = np.argsort(corr)[-K:]
        mv = corr[top].max()
        T = T_s + (T_e - T_s) * min(1.0, it / 50000.0)
        w = np.exp(T * (corr[top] - mv))
        w /= w.sum()
