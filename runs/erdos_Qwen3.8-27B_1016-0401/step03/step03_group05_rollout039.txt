import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def _slsqp_obj(x, T, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    return (mc + np.log(np.exp(sh).sum()) / T) * dx

def _slsqp_jac(x, T, dx, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    j_arr = np.arange(n)
    g = np.zeros(n)
    for oi in range(2 * n - 1):
        w = sw[oi]
        if w < 1e-11:
            continue
        s = oi - center
        if s >= 0:
            m1 = j_arr >= s
            m2 = (j_arr + s) < n
            g[m1] += w * (1.0 - x[j_arr[m1] - s])
            g[m2] -= w * x[j_arr[m2] + s]
        else:
            ss = -s
            m1 = (j_arr + ss) < n
            m2 = j_arr >= ss
            g[m1] += w * (1.0 - x[j_arr[m1] + ss])
            g[m2] -= w * x[j_arr[m2] - ss]
    return g * dx

def slsqp_refine(h, n, T, maxiter):
    target = n / 2.0
    dx = 2.0 / n

    def obj(x):
        return _slsqp_obj(x, T, dx)

    def jac(x):
        return _slsqp_jac(x, T, dx, n)

    def cfun(x):
        return x.sum() - target

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return res.x
    except Exception:
        return h.copy()

def fista_run(h_init, n, t_start, duration):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    h = h_init.copy()
    h_prev = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    c = 1.0
    it = 0
    j_arr = np.arange(n)

    while time.time() - t_start < duration:
        frac = min(it / 40000.0, 1.0)
        K = max(1, int(14 - 11 * frac))
        sharp = 4.0 + 10.0 * frac

        beta = (1.0 - 1.0 / (c + 1.0)) / 2.0
        h_y = np.clip(h + beta * (h - h_prev), 0.0, 1.0)

        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)

        if K == 1:
            idx = int(np.argmax(corr))
            s = idx - center
            if s >= 0:
                m1 = j_arr >= s
                m2 = (j_arr + s) < n
                g = np.zeros(n)
                g[m1] += (1.0 - h_y[j_arr[m1] - s])
                g[m2] -= h_y[j_arr[m2] + s]
            else:
                ss = -s
                m1 = (j_arr + ss) < n
                m2 = j_arr >= ss
                g = np.zeros(n)
                g[m1] += (1.0 - h_y[j_arr[m1] + ss])
                g[m2] -= h_y[j_arr[m2] - ss]
        else:
            top_idx = np.argpartition(corr, -K)[-K:]
            top_vals = corr[top_idx]
            w = (top_vals / max(mc, 1e-15)) ** sharp
            ws = w.sum()
            if ws < 1e-15:
                w = np.ones(K) / K
            else:
                w /= w.sum()
            g = np.zeros(n)
            for i in range(K):
                s = top_idx[i] - center
                if s >= 0:
                    m1 = j_arr >= s
                    m2 = (j_arr + s) < n
                    g[m1] += w[i] * (1.0 - h_y[j_arr[m1] - s])
                    g[m2] -= w[i] * h_y[j_arr[m2] + s]
                else:
                    ss = -s
                    m1 = (j_arr + ss) < n
                    m2 = j_arr >= ss
                    g[m1] += w[i] * (1.0 - h_y[j_arr[m1] + ss])
                    g[m2] -= w[i] * h_y[j_arr[m2] - ss]

        g *= dx
        step = 1.2 / (1.0 + 0.002 * it)
        h_new = np.clip(h_y - step * g, 0.0, 1.0)
        h_new = project_feasible(h_new, n)

        h_prev = h.copy()
        h = h_new
        c = (1.0 + np.sqrt(1.0 + 4.0 * c * c)) / 2.0
        it += 1

        if it % 4000 == 0:
            c5 = compute_c5(h)
            if c5 < c5_best:
                c5_best = c5
                h_best = h.copy()

    c5 = compute_c5(h)
    if c5 < c5_best:
        c5_best = c5
        h_best = h.copy()
    return h_best, c5_best

def binary_local_search(h_init, n, t_start, duration, seed_ls=777):
    h = h_init.copy()
    h = np.where(h > 0.5, 1.0, 0.0)
    target = int(n / 2)
    ns = int(h.sum())
    if ns > target:
        ones = np.where(h == 1.0)[0]
        h[ones[:ns - target]] = 0.0
    elif ns < target:
        zeros = np.where(h == 0.0)[0]
        h[zeros[:target - ns]] = 1.0

    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    rng = np.random.default_rng(seed_ls)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    n_stale = 0
    temp = 0.004

    while time.time() - t_start < duration and temp > 1e-9:
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)

        if c5_new < c5_cur:
            c5_cur = c5_new
            n_stale = 0
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            accept = rng.random() < np.exp(-(c5_new - c5_cur) / (temp + 1e-12))
            if accept:
                c5_cur = c5_new
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
                n_stale += 1
                if n_stale > 600:
                    temp *= 0.72
                    n_stale = 0

    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(280)
        h[::2] = 1.0
    n = len(h)
    h = project_feasible(h, n)
    c5_best = compute_c5(h)
    h_best = h.copy()

    # Phase 1: SLSQP medium-T polish
    for T in [40, 20, 10]:
        if time.time() - t0 > budget_s - 450:
            break

        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h = h_new
            h_best = h.copy()

    # Phase 2: FISTA (primary gradient descent)
    dur2 = (t0 + budget_s * 0.42) - time.time()
    if dur2 > 15:
        h_f, c5_f = fista_run(h, n, t0, dur2)
        h_f = project_feasible(h_f, n)
        c5_f = compute_c5(h_f)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f.copy()

    # Phase 3: Binary local search (Simulated Annealing on swaps)
    dur3 = (t0 + budget_s * 0.62) - time.time()
    if dur3 > 15:
        h_bin, c5_bin = binary_local_search(h_best, n, time.time(), dur3, 12345)
        if c5_bin < c5_best:
            c5_best = c5_bin
            h_best = h_bin.copy()

    # Phase 4: SLSQP fine polish
    for T in [6, 3, 2]:
        if time.time() - t0 > budget_s - 200:
            break
        h_new = slsqp_refine(h_best, n, T, 2500)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new.copy()

    # Phase 5: Second binary search
    dur5 = (t0 + budget_s * 0.82) - time.time()
    if dur5 > 10:
        h_bin2, c5_bin2 = binary_local_search(h_best, n, time.time(), dur5, 54321)
        if c5_bin2 < c5_best:
            c5_best = c5_bin2
            h_best = h_bin2.copy()

    # Phase 6: Final SLSQP + perturbation restarts
    if time.time() - t0 < budget_s - 40:
        h_new = slsqp_refine(h_best, n, 2, 2000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new.copy()

    # Phase 7: Small perturbation + SLSQP
    for trial in range(8):
        if time.time() - t0 > budget_s - 20:
            break
        sigma = 0.008 + 0.004 * trial
        h_pert = h_best + rng.normal(0, sigma, n)
        h_pert = project_feasible(h_pert, n)
        h_new = slsqp_refine(h_pert, n, 4, 1000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new.copy()

    h_best = project_feasible(h_best, n)
    assert abs(h_best.sum() * (2.0 / n) - 1.0) < 1e-10
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
