def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:

def shift_grad(h, n, shift):
    """Gradient of corr[shift+n-1] w.r.t. h for shift in [-n+1, n-1]"""
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        # ∂/∂h[j] of sum h[i]*(1-h[i+shift]) = (1-h[j-shift]) - h[j+shift]
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]

        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]

We want: max_h min_k A_k subject to 0≤h≤1, ∑h = n/2

**Alternative: Use the structure**

For a periodic function with period p (in the discrete sense), the autocorrelation has a specific structure. If h has period p where p divides n, then only shifts that are multiples of p have non-trivial autocorrelation (others average out). This could simplify the problem.

But the optimal solution likely doesn't have exact periodicity.

**Another idea: Interior point / barrier method**

Use a barrier for the box constraints and a smooth max for the objective:
min_h (1/T) log(∑_k exp(T * corr_k)) + μ * barrier(h)

Subject to: ∑h = n/2

This can be solved with L-BFGS-B (which handles some constraints) or plain gradient descent.

Actually, L-BFGS-B handles box constraints natively! So:
min_h (1/T) log(∑_{k in active} exp(T * (corr_k * dx))) + λ * (∑h - n/2)²

The (∑h - n/2)² penalty handles the sum constraint. As λ→0, we enforce it.

But L-BFGS-B needs a smooth objective. The log-sum-exp of correlations is smooth. The penalty is smooth. So this works!

    _S_N = n; _S_TGT = n/2.0; _S_DX = 2.0/n; _S_T = float(T); _S_K = K

def slsqp_opt(h, n, T, top_k, maxiter, ftol):

# Global state for L-BFGS-B
_B_N = 280; _B_TGT = 140.0; _B_DX = 2.0/280; _B_T = 1.0; _B_K = 10; _B_LAM = 1e10

def _b_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_B_K:]
    c = corr[idx]; m = c.max()
    e = np.exp(_B_T * (c - m))
    f = (m + np.log(e.sum()) / _B_T) * _B_DX
    sl = x.sum() - _B_TGT
    return f + _B_LAM * sl * sl

def _b_grad(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_B_K:]
    c = corr[idx]; m = c.max()
    e = np.exp(_B_T * (c - m)); w = e / e.sum()
    g = np.zeros(_B_N)
    for wi, ci in zip(w, idx):
        if wi > 1e-12:
            g += wi * shift_grad(x, _B_N, ci - (_B_N - 1))
    g *= _B_DX
    sl = x.sum() - _B_TGT
    g += 2.0 * _B_LAM * sl
    return g

def _b_set(n, T, K, lam):
    global _B_N, _B_TGT, _B_DX, _B_T, _B_K, _B_LAM
    _B_N = n; _B_TGT = n/2.0; _B_DX = 2.0/n; _B_T = float(T); _B_K = K; _B_LAM = lam

def lbfgsb_opt(h, n, T, top_k, maxiter, lam=1e10):
    _b_set(n, T, min(top_k, 2*n-1), lam)
    bounds = [(0.0, 1.0)] * n
    try:
        r = minimize(_b_obj, h, method='L-BFGS-B', jac=_b_grad, bounds=bounds,
                     options={'maxiter':maxiter, 'ftol':1e-20, 'gtol':1e-14})
        return r.x
    except Exception:
        return h.copy()

def local_search(h, n, rng, n_trials, t_end):
    best = h.copy(); c5b = compute_c5(h)
    for _ in range(n_trials):
        if time.time() > t_end: break
        hp = best.copy()
        m = rng.integers(0, 5)
        if m == 0:
            i, j = rng.integers(0, n, 2)
            hp[[i, j]] = hp[[j, i]]
        elif m == 1:
            k = rng.integers(1, n-5); i = rng.integers(0, n-k)
            d = rng.uniform(0.003, 0.04)
            hp[i] = np.clip(hp[i]+d, 0, 1); hp[i+k] = np.clip(hp[i+k]-d, 0, 1)
        elif m == 2:
            lb = rng.integers(0, n-15); ln = rng.integers(3, 12)
            hp[lb:lb+ln] = 1.0 - hp[lb:lb+ln]
        elif m == 3:
            sig = 0.001 + 0.002*rng.integers(0,5)
            hp += rng.normal(0, sig, n)
        else:
            i = rng.integers(0, n)
            hp[i] = rng.uniform(0, 1)
        hp = project_feasible(hp, n)
        c5 = compute_c5(hp)
        if c5 < c5b: c5b, best = c5, hp
    return best, c5b

def nesterov_flow(h, n, t_end, rng, top_k=6):
    dx = 2.0/n
    hc = h.copy(); hp = h.copy(); ha = h.copy()
    c5a = compute_c5(h); hb = h.copy()
    for it in range(500000):
        if time.time() > t_end: break
        beta = min(0.9, float(it)/(it+8.0)) if it > 0 else 0.0
        hy = np.clip(hc + beta*(hc - hp), 0, 1)
        corr = np.correlate(hy, 1.0-hy, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(hy, n, ci-(n-1))
        g *= dx / top_k
        a = 0.5 / (1.0 + 0.0005*it)
        hn = project_feasible(hy - a*g, n)
        hp = hc; hc = hn; ha += (hn - ha)/(it+1)
        if (it+1) % 20000 == 0:
            c1 = compute_c5(hc)
            if c1 < c5a: c5a, hb = c1, hc.copy()
            c2 = compute_c5(project_feasible(ha, n))
            if c2 < c5a: c5a, hb = c2, project_feasible(ha, n).copy()
    return hb, c5a
