import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def solve_1d_minmax(A, C, n_bisect=40):
    n = A.shape[1]
    pos = A > 1e-14
    neg = A < -1e-14
    t_lo = np.zeros(n)
    t_hi = np.max(A + C, axis=0)
    zero_mask = ~pos & ~neg
    if zero_mask.any():
        t_lo = np.maximum(t_lo, C[zero_mask].max(axis=0))
    
    for _ in range(n_bisect):
        t_mid = 0.5 * (t_lo + t_hi)
        tm = t_mid[None, :]
        x_low = np.full(n, 0.0)
        x_high = np.full(n, 1.0)
        if neg.any():
            with np.errstate(divide='ignore', invalid='ignore'):
                x_low = np.maximum(x_low, ((tm - C[neg]) / (-A[neg])).max(axis=0))
        if pos.any():
            with np.errstate(divide='ignore', invalid='ignore'):
                x_high = np.minimum(x_high, ((tm - C[pos]) / A[pos]).min(axis=0))
        feasible = x_low <= x_high + 1e-12
        t_hi[~feasible] = t_mid[~feasible]
        t_lo[feasible] = t_mid[feasible]
    
    tm = t_lo[None, :]
    x_low = np.full(n, 0.0)
    x_high = np.full(n, 1.0)
    if neg.any():
        with np.errstate(divide='ignore', invalid='ignore'):
            x_low = np.maximum(x_low, ((tm - C[neg]) / (-A[neg])).max(axis=0))
    if pos.any():
        with np.errstate(divide='ignore', invalid='ignore'):
            x_high = np.minimum(x_high, ((tm - C[pos]) / A[pos]).min(axis=0))
    return np.clip(0.5 * (x_low + x_high), 0.0, 1.0)

def coord_descent(h, n, top_k, n_sweeps, t_end):
    h = h.copy()
    shift_vec = np.arange(2 * n - 1) - (n - 1)
    for _ in range(n_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argsort(corr)[-top_k:]
        A = np.zeros((top_k, n))
        C = np.zeros((top_k, n))
        for si in range(top_k):
            sh = int(shift_vec[int(idx[si])])
            if sh >= 0:
                A[si, :n - sh] += (1.0 - h[sh:n])
                if sh > 0:
                    A[si, sh:] -= h[:n - sh]
            else:
                sh2 = -sh
                A[si, sh2:] += (1.0 - h[:n - sh2])
                if sh2 > 0:
                    A[si, :n - sh2] -= h[sh2:]
            C[si, :] = corr[int(idx[si])] - A[si, :] * h
        h = project_feasible(solve_1d_minmax(A, C), n)
    return h

# SLSQP helper
_S_N = 0
_S_TGT = 0.0
_S_T = 1.0
_S_K = 20

def _s_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_S_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_S_T * (c - m))
    return (m + np.log(e.sum()) / _S_T) * (2.0 / _S_N)

def _shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        g[:n - shift] += (1.0 - h[shift:n])
        if shift > 0:
            g[shift:] -= h[:n - shift]
    else:
        s = -shift
        g[s:] += (1.0 - h[:n - s])
        if s > 0:
            g[:n - s] -= h[s:]
    return g

def _s_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_S_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_S_T * (c - m))
    w = e / e.sum()
    grad = np.zeros(_S_N)
    for wi, ci in zip(w, idx):
        if wi > 1e-10:
            grad += wi * _shift_grad(x, _S_N, int(ci) - (_S_N - 1))
    return grad * (2.0 / _S_N)

def _s_cf(x):
    return x.sum() - _S_TGT

def _s_cj(x):
    return np.ones(_S_N)

def _slsqp(h, n, T, top_k, maxiter, ftol):
    global _S_N, _S_TGT, _S_T, _S_K
    _S_N = n
    _S_TGT = n / 2.0
    _S_T = float(T)
    _S_K = min(top_k, 2 * n - 1)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_s_obj, h, method='SLSQP', jac=_s_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _s_cf, 'jac': _s_cj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()
