from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    """Project x onto {0<=x<=1, sum(x)=target}"""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Bisection for threshold
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

def correlate_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smax_obj(x, T):
    n = len(x)
    corr = correlate_full(x)
    mc = corr.max()
    s = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(s))) / T) * (2.0 / n)

def smax_grad(x, T):
    n = len(x)
    corr = correlate_full(x)
    mc = corr.max()
    s = T * (corr - mc)
    e = np.exp(s)
    sw = e / e.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n-o])
            grad[:n-o] += -w * x[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n-p]
    return grad * dx

def fazer_fista(h, T, target, n_iter, lr=None):
    """FISTA on soft-max with projection"""
    n = len(h)
    if lr is None:
        # Lipschitz constant estimate
        lr = 1.0 / (4.0 * T * (2.0/n) * 2.0)  # rough
        lr = min(lr, 0.5)
    x = h.copy()
    y = h.copy()
    t = 1.0
    best_x = h.copy()
    best_f = smax_obj(h, T)
    for k in range(n_iter):
        g = smax_grad(y, T)
        y_new = y - lr * g
        x_new = project_feasible(y_new, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + (t - 1.0) / t_new * (x_new - x)
        x = x_new
        t = t_new
        f = smax_obj(x, T)
        if f < best_f:
            best_f = f
            best_x = x.copy()
        # Check constraint
        diff = x.sum() - target
        if abs(diff) > 1e-8:
            x = project_feasible(x, target)
    return best_x, best_f

    n = len(h)
    # Make binary
    h_bin = np.zeros(n)

        h_bin[j] = 1.0
        c5_new = compute_c5(h_bin)
        if c5_new < c5_cur:

            # Update tracking
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
            if c5_new < c5_best:

        else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def smax_grad(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    s = T * (corr - mc)
    e = np.exp(s)
    sw = e / e.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-13:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n-o])
            grad[:n-o] += -w * x[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n-p]
    return grad * dx

def smax_val(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    s = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(s))) / T) * (2.0 / n)

def fista_solve(h0, T, target, n_iter, lr):
    n = len(h0)
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best_x = h0.copy()
    best_f = smax_val(h0, T)
    for k in range(n_iter):
        g = smax_grad(y, T)
        x_new = project_feasible(y - lr * g, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + (t - 1.0) / t_new * (x_new - x)
        x = x_new
        t = t_new
        f = smax_val(x, T)
        if f < best_f:
            best_f = f
            best_x = x.copy()
    return best_x, best_f

def binary_sa(h0, target, n_flips, T_start, T_end, rng):
    n = len(h0)
    h = np.zeros(n)
    order = np.argsort(h0)[::-1]
    h[order[:int(target)]] = 1.0
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for it in range(n_flips):
        T_sa = T_start * (T_end / T_start) ** (it / max(n_flips - 1, 1))
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        else:
            d = c5_new - c5_cur
            if T_sa > 1e-12 and rng.random() < np.exp(-d / T_sa):
                c5_cur = c5_new
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
    return h_best, c5_best

def make_equil(x, target):
    return x.sum() - target

def make_eqjac(x):
    return np.ones(len(x))

def slsqp_lag_opt(h, lag_idx, target, bounds, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        return np.correlate(x, 1.0 - x, mode="full")[lag_idx] * dx
    def objg(x):
        o = lag_idx - (n - 1)
        g = np.zeros(n)
        if o == 0:
            g[:] = 1.0 - 2.0 * x
        elif o > 0:
            g[o:] += 1.0 - x[:n-o]
            g[:n-o] += -x[o:]
        else:
            p = -o
            g[:n-p] += 1.0 - x[p:]
            g[p:] += -x[:n-p]
        return g * dx
    cons = [dict(type='eq', fun=lambda x: x.sum() - target, jac=lambda x: np.ones(len(x)))]
    res = minimize(obj, h, jac=objg, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, target)

def coord_refine(h, target, n_sweeps):
    n = len(h)
    dts = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    c5_cur = compute_c5(h)
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            for dt in dts:
                trial = h.copy()
                trial[j] = np.clip(trial[j] + dt, 0.0, 1.0)
                trial = project_feasible(trial, target)
                c5_new = compute_c5(trial)
                if c5_new < c5_cur - 1e-16:
                    h = trial
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(340)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = project_feasible(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Binary SA (25% budget)
    p1_end = t0 + budget_s * 0.25
    for restart in range(4):
        if elapsed() > p1_end:
            break
        if restart == 0:
            h_sa = h_best.copy()
        elif restart == 1:
            h_sa = np.zeros(n)
            h_sa[rng.choice(n, int(target), replace=False)] = 1.0
        else:
            h_sa = h_best.copy()
            h_sa[rng.integers(n)] = rng.random()
            h_sa = project_feasible(h_sa, target)
        T_s = 0.005 if restart == 0 else 0.01
        T_e = 1e-5
        nf = 80000 if restart == 0 else 50000
        h_sa, c5_sa = binary_sa(h_sa, target, nf, T_s, T_e, rng)
        if c5_sa < c5_best:
            c5_best = c5_sa
            h_best = h_sa.copy()

    # Phase 2: FISTA soft-max annealing (25%)
    p2_end = t0 + budget_s * 0.50
    for T in [2000, 8000, 32000, 128000]:
        if elapsed() > p2_end:
            break
        lr = 0.3 / (T * 2.0 / n * 4.0)
        lr = min(lr, 0.1)
        h_f, _ = fista_solve(h_best, T, target, 3000, lr)
        c5_f = compute_c5(h_f)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f.copy()

    # Phase 3: SA again from FISTA result (20%)
    p3_end = t0 + budget_s * 0.70
    for _ in range(3):
        if elapsed() > p3_end:
            break
        h_sa, c5_sa = binary_sa(h_best, target, 60000, 0.003, 1e-6, rng)
        if c5_sa < c5_best:
            c5_best = c5_sa
            h_best = h_sa.copy()

    # Phase 4: SLSQP on worst + top lags (15%)
    p4_end = t0 + budget_s * 0.85
    for rnd in range(20):
        if elapsed() > p4_end:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        top_k = np.argsort(corr)[-4:]
        for lag in top_k:
            if elapsed() > p4_end:
                break
            try:
                h_new = slsqp_lag_opt(h_best, int(lag), target, bounds, 600, 1e-18)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    # Phase 5: Final polish (10%)
    p5_end = t0 + budget_s - 3.0
    h_best, c5_cd = coord_refine(h_best, target, 2)
    if c5_cd < c5_best:
        c5_best = c5_cd
    for _ in range(5):
        if elapsed() > p5_end:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst = int(np.argmax(corr))
        try:
            h_new = slsqp_lag_opt(h_best, worst, target, bounds, 1000, 1e-19)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
