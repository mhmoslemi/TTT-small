        if m1.any():

        if m2.any():
            grad[m2] -= h[j[m2] - s]

def smooth_max_val_grad(h, n, T, K):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    if K is not None and K < len(corr):
        topk = np.argpartition(corr, -K)[-K:]
        c_sel = corr[topk]
        mc_sel = np.max(c_sel)
        sh = T * (c_sel - mc_sel)
        e = np.exp(sh)
        val = (mc_sel + np.log(e.sum()) / T) * dx
        sw = e / e.sum()
        grad = np.zeros(n)
        for ii in range(len(sw)):
            w = sw[ii]
            if w < 1e-15:
                continue
            grad += w * shift_grad(h, n, topk[ii] - center)
        return val, grad * dx
    else:
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        val = (mc + np.log(e.sum()) / T) * dx
        sw = e / e.sum()
        grad = np.zeros(n)

            w = sw[oi]

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):

    def obj(x):

        v, _ = smooth_max_val_grad(x, n, T, K)
        return v
    def jac(x):

        _, g = smooth_max_val_grad(x, n, T, K)
        return g
    try:

                       options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol*0.1,

def subgradient_descent(h, n, t0, tlim, max_iters=200000):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def smooth_val_grad(h, n, T, K):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    if K is not None and K < len(corr):
        topk = np.argpartition(corr, -K)[-K:]
        c_sel = corr[topk]
        mc_sel = np.max(c_sel)
        sh = T * (c_sel - mc_sel)
        e = np.exp(sh)
        val = (mc_sel + np.log(e.sum()) / T) * dx
        sw = e / e.sum()
        grad = np.zeros(n)
        for ii in range(len(sw)):
            if sw[ii] < 1e-14:
                continue
            grad += sw[ii] * shift_grad(h, n, topk[ii] - center)
        return val, grad * dx
    else:
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        val = (mc + np.log(e.sum()) / T) * dx
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(len(sw)):
            if sw[oi] < 1e-14:
                continue
            grad += sw[oi] * shift_grad(h, n, oi - center)
        return val, grad * dx

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        if time.time() - t0 > tlim:
            raise TimeoutError
        return smooth_val_grad(x, n, T, K)[0]
    def jac(x):
        if time.time() - t0 > tlim:
            raise TimeoutError
        return smooth_val_grad(x, n, T, K)[1]
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jac, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol * 0.5,
                                'maxls': 40, 'maxfun': maxiter * 4})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_descent(h, n, t0, tlim):
    dx = 2.0 / n
    h_curr = h.copy()
    h_mom = h.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    max_iters = 150000
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        beta = 2.0 / (it + 2.0)
        y = np.clip(h_curr + beta * (h_curr - h_mom), 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = shift_grad(y, n, shift) * dx
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        f_y = float(np.max(corr) * dx)
        gap = max(f_y - 0.36, 1e-7)
        alpha = min(2.0 * gap / (gn * gn + 1e-12), 0.15 / (1.0 + 0.001 * it))
        h_new = project_feasible(np.clip(y - alpha * g, 0.0, 1.0), n)
        h_mom = h_curr.copy()
        h_curr = h_new
        if (it + 1) % 4000 == 0:
            c5_now = compute_c5(h_curr)
            if c5_now < c5_best:
                c5_best = c5_now
                h_best = h_curr.copy()
    c5_fin = compute_c5(h_curr)
    if c5_fin < c5_best:
        c5_best = c5_fin
        h_best = h_curr.copy()
    return h_best, c5_best

def local_search(h, n, max_iters, t0, tlim, rng):
    c5 = compute_c5(h)
    h_work = h.copy()
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        corr = np.correlate(h_work, 1.0 - h_work, mode="full")
        top_indices = np.argsort(corr)[-5:]
        improved = False
        for idx in top_indices:
            if time.time() - t0 > tlim:
                break
            shift = int(idx) - (n - 1)
            g = shift_grad(h_work, n, shift)
            pos = np.where(g > 0.005)[0]
            neg = np.where(g < -0.005)[0]
            if len(pos) == 0 or len(neg) == 0:
                continue
            j_dec = pos[np.argmax(g[pos])]
            j_inc = neg[np.argmin(g[neg])]
            max_t = min(1.0 - h_work[j_dec], h_work[j_inc], 0.04)
            if max_t < 1e-7:
                continue
            for frac in [1.0, 0.6, 0.3, 0.15]:
                delta = max_t * frac
                if delta < 1e-9:
                    break
                old_a, old_b = h_work[j_dec], h_work[j_inc]
                h_work[j_dec] -= delta
                h_work[j_inc] += delta
                c5_new = compute_c5(h_work)
                if c5_new < c5 - 1e-15:
                    c5 = c5_new
                    improved = True
                    break
                else:
                    h_work[j_dec] = old_a
                    h_work[j_inc] = old_b
        if not improved:
            # Random perturbation to escape
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                break
            d = rng.uniform(0.002, 0.015)
            if h_work[i1] < d or h_work[i2] > 1.0 - d:
                d = min(h_work[i1], 1.0 - h_work[i2], 0.005)
            if d < 1e-8:
                break
            oa, ob = h_work[i1], h_work[i2]
            h_work[i1] -= d
            h_work[i2] += d
            c5_new = compute_c5(h_work)
            if c5_new < c5:
                c5 = c5_new
            else:
                h_work[i1] = oa
                h_work[i2] = ob
                if it > 200:
                    break
    return h_work, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:

        h_init = np.zeros(280)

    h_best = project_feasible(h_init, n)

    # Phase 1: L-BFGS-B progressive smoothing
    T_schedule = [(64, 3000, 200), (32, 4000, 150), (16, 5000, 120),
                  (8, 5000, 100), (4, 6000, 60), (2, 7000, 40),
                  (1, 8000, 30), (0.5, 8000, 20), (0.25, 9000, 10), (0.1, 10000, 5)]
    for T, mi, K in T_schedule:

            h_best = h_new

    # Phase 2: Subgradient (Nesterov)
    rem = budget_s - (time.time() - t0)
    if rem > 150:
        h_sg, c5_sg = subgradient_descent(h_best, n, t0, t0 + rem - 100)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase 3: Intensive local search (the key improvement)
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        lc_iters = int(min(80000, rem * 1200))
        h_ls, c5_ls = local_search(h_best, n, lc_iters, t0, budget_s - 15, rng)
        if c5_ls < c5_best:
            c5_best = c5_ls

        if c5_ls2 < c5_best:
            c5_best = c5_ls2
            h_best = h_ls2

    # Phase 6: Restart with perturbation and re-optimize
    for ri in range(8):
        rem = budget_s - (time.time() - t0)
        if rem < 25:
            break
        h_p = h_best + rng.normal(0, 0.003 + 0.004 * ri, n)
        if ri % 3 == 0:
            bs = n // 4
            bi = int(rng.integers(0, 4))
            h_p[bi * bs:(bi + 1) * bs] *= 0.97

                h_best = h_ls3

    # Final project and evaluate
    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, n)
