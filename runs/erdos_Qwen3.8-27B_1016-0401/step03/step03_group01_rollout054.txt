    """Project y onto {x: 0≤x≤1, sum(x)=n/2} via bisection on shift."""
    target = n / 2.0

    topk = corr[idx]
    mc = topk.max()
    sh = (topk - mc) * T
    smx = sh.max()
    e = np.exp(sh - smx)
    sw = e / e.sum()
    lse_val = mc + (smx + np.log(e.sum())) / T
    return lse_val, idx, sw

def smooth_obj_grad(h, T, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    lse_val, idx, sw = _lse_topk(corr, T, k)
    obj = lse_val * dx
    center = n - 1
    j = np.arange(n)
    grad = np.zeros(n)
    for oi in range(len(idx)):
        w = T * sw[oi] * dx
        if w < 1e-14:
            continue
        s = idx[oi] - center
        if s >= 0:
            m1 = j >= s
            grad[m1] += w * (1.0 - h[j[m1] - s])
            m2 = (j + s) < n
            grad[m2] -= w * h[j[m2] + s]
        else:
            s2 = -s
            m1 = (j + s2) < n
            grad[m1] += w * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * h[j[m2] - s2]
    return obj, grad

def lbfgsb_opt(h0, n, T, k, lam, maxiter):
    target = n / 2.0
    def _f(x):
        o, _ = smooth_obj_grad(x, T, k)
        return o + 0.5 * lam * (x.sum() - target) ** 2
    def _g(x):
        _, gr = smooth_obj_grad(x, T, k)
        return gr + lam * (x.sum() - target)
    res = minimize(_f, h0, jac=_g, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'maxfun': 30000})
    return res.x

def slsqp_opt(h0, n, T, k, maxiter):
    target = n / 2.0
    def _f(x):
        return smooth_obj_grad(x, T, k)[0]
    def _g(x):
        return smooth_obj_grad(x, T, k)[1]
    def _c(x):
        return x.sum() - target
    res = minimize(_f, h0, jac=_g, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _c, 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return res.x

def coord_descent_step(h, n, T, k, lam, n_grid=25):
    """One coordinate descent sweep with 1D grid search per coordinate."""
    target = n / 2.0
    dx = 2.0 / n
    h_new = h.copy()
    for i in range(n):
        old_val = h_new[i]
        if old_val <= 1e-8 or old_val >= 1.0 - 1e-8:
            continue
        best_val = old_val
        best_obj = smooth_obj_grad(h_new, T, k)[0] + 0.5 * lam * (h_new.sum() - target) ** 2
        grid = np.linspace(max(0.0, old_val - 0.15), min(1.0, old_val + 0.15), n_grid)
        for gv in grid:
            if abs(gv - old_val) < 1e-10:
                continue
            h_new[i] = gv
            o, _ = smooth_obj_grad(h_new, T, k)
            pen = 0.5 * lam * (h_new.sum() - target) ** 2
            if o + pen < best_obj:
                best_obj = o + pen
                best_val = gv
        h_new[i] = best_val
    return h_new

def fista_opt(h0, n, T, k, n_iter, time_limit):
    dx = 2.0 / n
    target = n / 2.0
    h_y = h0.copy()
    h_z = h0.copy()
    t_mom = 1.0
    lr = 0.04
    for it in range(n_iter):
        if time.time() > time_limit:
            break
        _, grad = smooth_obj_grad(h_y, T, k)
        grad = grad + 8.0 * (h_y.sum() - target) / n
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        h_z = h_y + beta * (h_y - h_z)
        h_y = np.clip(h_z - lr * grad, 0.0, 1.0)
        h_y = project_feasible(h_y, n)
        t_mom = t_new
        lr *= 0.9997
    return h_y

def perturb(h, n, rng, strength=0.04):
    h_p = h.copy()
    ptype = rng.integers(0, 7)
    if ptype == 0:
        h_p += strength * rng.normal(0, 1, n)
    elif ptype == 1:
        start = rng.integers(0, max(1, n - 30))
        ln = rng.integers(5, min(35, n - start))
        h_p[start:start + ln] = 1.0 - h_p[start:start + ln]
    elif ptype == 2:
        x = np.arange(n) * 2.0 / n
        for _ in range(4):
            f = rng.integers(1, 20)
            a = rng.uniform(0.01, 0.07)
            ph = rng.uniform(0, 2 * np.pi)
            h_p += a * np.sin(np.pi * f * x + ph)
    elif ptype == 3:
        b = rng.integers(8, 40)
        i1 = rng.integers(0, max(1, n - b))
        i2 = rng.integers(0, max(1, n - b))
        if i1 != i2:
            tmp = h_p[i1:i1+b].copy()
            h_p[i1:i1+b] = h_p[i2:i2+b]
            h_p[i2:i2+b] = tmp
    elif ptype == 4:
        mask = rng.random(n) < 0.08
        h_p[mask] = rng.random(int(mask.sum()))
    elif ptype == 5:
        s = rng.integers(1, 5)
        h_p = np.roll(h_p, int(s))
    else:
        # Butterfly perturbation
        h_p[:n//2] += strength * 0.5 * rng.normal(0, 1, n//2)
        h_p[n//2:] -= strength * 0.5 * rng.normal(0, 1, n - n//2)
    return np.clip(h_p, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    def elapsed():
        return time.time() - t0

    # Phase 1: L-BFGS-B with decreasing T (broad to focused)
    schedule1 = [(40, 25, 600), (20, 20, 800), (10, 15, 1000),
                 (5, 10, 1200), (3, 8, 1500), (1.5, 6, 2000), (0.8, 5, 2500)]
    for T, k, mi in schedule1:
        if elapsed() > budget_s - 350:
            break
        h_new = lbfgsb_opt(h_best, n, T, k, 200.0, mi)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: SLSQP focused polish
    for T, k in [(1.0, 6), (0.5, 4), (0.25, 3)]:
        if elapsed() > budget_s - 250:
            break
        try:
            h_new = slsqp_opt(h_best, n, T, k, 4000)
        except Exception:
            continue
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 3: Coordinate descent sweeps
    if elapsed() < budget_s - 200:
        for T, k, ns in [(3, 8, 15), (1, 5, 20), (0.5, 4, 30)]:
            if elapsed() > budget_s - 150:
                break
            for _s in range(200):
                if elapsed() > budget_s - 120:
                    break
                h_new = coord_descent_step(h_best, n, T, k, 200.0, ns)
                h_new = project_feasible(h_new, n)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
                elif c5 - c5_best > 0.005:
                    break

    # Phase 4: FISTA
    if elapsed() < budget_s - 120:
        h_f = fista_opt(h_best, n, 2, 6, 30000, t0 + budget_s - 130)
        h_f = project_feasible(h_f, n)
        c5 = compute_c5(h_f)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_f

    # Phase 5: Multi-restart with perturbations
    rem = budget_s - elapsed()
    n_restarts = max(2, int(rem / 35))
    for ri in range(n_restarts):
        if elapsed() > budget_s - 20:
            break
        h_p = perturb(h_best, n, rng, strength=0.03 + 0.02 * (ri % 3))
        h_p = project_feasible(h_p, n)
        h_new = lbfgsb_opt(h_p, n, 5, 10, 200.0, 500)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        # Quick polish
        if elapsed() < budget_s - 10:
            h_new2 = lbfgsb_opt(h_best, n, 1, 5, 200.0, 800)
            h_new2 = project_feasible(h_new2, n)
            c5_2 = compute_c5(h_new2)
            if c5_2 < c5_best:
                c5_best = c5_2
                h_best = h_new2

    # Phase 6: Try n=200 for possibly better structure
    if elapsed() < budget_s - 100:
        n2 = 200
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_200 = np.interp(x_new, x_old, h_best)
        h_200 = project_feasible(h_200, n2)
        for T, k in [(20, 15), (5, 8), (1, 5)]:
            if elapsed() > budget_s - 20:
                break
            h_n = lbfgsb_opt(h_200, n2, T, k, 200.0, 1000)
            h_n = project_feasible(h_n, n2)
            c5 = compute_c5(h_n)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_n
                n = n2

    # Final SLSQP polish on the best
    if elapsed() < budget_s - 5:
        try:
            h_fp = slsqp_opt(h_best, n, 0.5, 4, 3000)
            h_fp = project_feasible(h_fp, n)
            c5_fp = compute_c5(h_fp)
            if c5_fp < c5_best:
                c5_best = c5_fp
                h_best = h_fp
        except Exception:
            pass

    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, n)
