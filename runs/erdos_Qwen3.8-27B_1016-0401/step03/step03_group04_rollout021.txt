where f_k(h) = ∫ h(x)(1-h(x+k)) dx

By minimax duality (for non-convex problems this isn't exact, but it's a lower bound):
≤ min_h max_k f_k(h)
≥ max_{p∈Δ} min_h Σ p_k f_k(h)

The inner minimization min_h Σ p_k f_k(h) is a quadratic program (since each f_k is quadratic in h). We could:
1. Start with a distribution p over shifts
2. Solve the QP to get h*
3. Update p to put more weight on the shifts that are "tightest"
4. Repeat

= 1 - h^T (Σ_k p_k R_k) h

        # Linear constraint: f_k + g^T (h_var - h) <= t_var
        constraints.append(f_k + g @ (h_var - h) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.GLOP)  # or CPLEX, CBC
        if h_var.value is not None:
            return np.array(h_var.value).flatten()
    except:
        pass
    return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

_tg = [0, 0.0]

def _sobj(x, n, K, T):
    if time.time() > _tg[1]:
        _tg[0] = 1
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    N = len(corr)
    if K >= N:
        idx = np.arange(N)
    else:
        idx = np.argpartition(corr, -K)[-K:]
    c = corr[idx]
    cm = c.max()
    if T < 0.005:
        bi = int(np.argmax(c))
        return c[bi] * dx
    sh = T * (c - cm)
    e = np.exp(sh)
    return (cm + np.log(e.sum()) / T) * dx

def _sgrad(x, n, K, T):
    if time.time() > _tg[1]:
        _tg[0] = 1
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    N = len(corr)
    center = n - 1
    if K >= N:
        idx = np.arange(N)
    else:
        idx = np.argpartition(corr, -K)[-K:]
    c = corr[idx]
    cm = c.max()
    g = np.zeros(n)
    if T < 0.005:
        bi = int(np.argmax(c))
        return shift_grad(x, n, idx[bi] - center) * dx
    sh = T * (c - cm)
    e = np.exp(sh)
    sw = e / e.sum()
    for ii in range(len(sw)):
        if sw[ii] > 1e-14:
            g += sw[ii] * shift_grad(x, n, idx[ii] - center)
    return g * dx

def do_lbfgs(h, n, K, T, maxiter, tol, tdeadline):
    _tg[0] = 0
    _tg[1] = tdeadline
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sobj, h, method='L-BFGS-B', jac=_sgrad,
                       args=(n, K, T), bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol,
                                'gtol': max(tol * 0.001, 1e-18),
                                'maxls': 50, 'maxfun': maxiter * 5})
        if _tg[0]:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def do_fista(h_init, n, tdeadline, max_iter=200000):
    dx = 2.0 / n
    h = h_init.copy()
    h_prev = h_init.copy()
    best = h_init.copy()
    best_c5 = compute_c5(h_init)
    t_k = 1.0
    for it in range(max_iter):
        if time.time() > tdeadline:
            break
        mom = (t_k - 1.0) / (t_k + 1.0)
        y = h + mom * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode="full")
        topk = np.argpartition(corr, -4)[-4:]
        center = n - 1
        g = np.zeros(n)
        for k in topk:
            g += shift_grad(y, n, k - center)
        g *= dx / len(topk)
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        f_y = corr[topk].max() * dx
        gap = max(f_y - 0.375, 1e-6)
        alpha = min(2.0 * gap / (gn * gn + 1e-12), 0.8 / (1.0 + 0.0005 * it))
        h_new = project_feasible(y - alpha * g, n)

        h = h_new
        t_k += 1.0
        if (it + 1) % 8000 == 0:
            c5 = compute_c5(h)

    c5 = compute_c5(h)

        best = h.copy()
    return best, best_c5

    H = np.fft.rfft(h)
    nf = min(10, len(H) - 1)
    for k in range(1, nf + 1):

        H[k] *= np.exp(1j * rng.normal(0, 0.25))
    return np.fft.irfft(H, n=n)

def surgical_2d(h_base, n, s1, s2, tdeadline):
    dx = 2.0 / n
    g1 = shift_grad(h_base, n, s1)
    g2 = shift_grad(h_base, n, s2)
    n1 = np.linalg.norm(g1)
    n2 = np.linalg.norm(g2)
    if n1 < 1e-15 or n2 < 1e-15:
        return h_base, compute_c5(h_base)
    g1 /= n1
    g2 /= n2
    best_h = h_base.copy()
    best_c5 = compute_c5(h_base)
    def obj2d(ab):
        if time.time() > tdeadline:
            return 1e9
        h_try = project_feasible(h_base + ab[0] * g1 + ab[1] * g2, n)
        return compute_c5(h_try)
    starts = [[0.002, 0.002], [-0.002, 0.002], [0.002, -0.002],
              [-0.002, -0.002], [0.005, 0.0], [0.0, 0.005]]
    for st in starts:
        if time.time() > tdeadline:
            break
        try:
            res = minimize(obj2d, st, method='Nelder-Mead',
                           options={'xatol': 1e-10, 'fatol': 1e-13, 'maxiter': 300})
            v = res.fun
            if v < best_c5:
                best_c5 = v
                best_h = project_feasible(h_base + res.x[0] * g1 + res.x[1] * g2, n)
        except Exception:
            pass
    return best_h, best_c5
