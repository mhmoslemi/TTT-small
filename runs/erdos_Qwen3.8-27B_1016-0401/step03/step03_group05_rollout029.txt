from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 3.0
    hi = y.max() + 3.0
    mid = 0.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

    alpha = 0.5
    K_base = 3
    K_max = 35
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        frac = min(it / max_iters, 1.0)
        K = max(2, int(K_base + (K_max - K_base) * frac**1.5))
        beta = (t_fist - 1.0) / (0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fist**2)))
        t_fist = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fist**2))

        h_y = np.clip(h_y, 0.0, 1.0)

            top_idx = np.argpartition(corr, -K)[-K:]
