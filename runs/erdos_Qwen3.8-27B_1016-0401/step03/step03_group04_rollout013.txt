            grad += sw[ii] * shift_grad(x, n, topk[ii] - center)
        return val, grad * (2.0 / n)
    sh = T * (corr - mc)
    e = np.exp(sh)
    val = (mc + np.log(e.sum()) / T) * (2.0 / n)
    sw = e / e.sum()
    grad = np.zeros(n)
    for oi in range(len(sw)):
        if sw[oi] < 1e-15:
            continue
        grad += sw[oi] * shift_grad(x, n, oi - center)
    return val, grad * (2.0 / n)

_flag = [False]

def _obj(x, n, T, K, t0, tlim):
    if time.time() - t0 > tlim:
        _flag[0] = True
    v, _ = _sm_val_and_grad(x, n, T, K)
    return v

def _jac(x, n, T, K, t0, tlim):
    if time.time() - t0 > tlim:
        _flag[0] = True
    _, g = _sm_val_and_grad(x, n, T, K)
    return g

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):
    _flag[0] = False
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, method='L-BFGS-B', jac=_jac,
                       args=(n, T, K, t0, tlim), bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol * 0.1, 'maxls': 30})
        if _flag[0]:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def do_coord_desc(h_init, n, t0, tlim, max_sweeps=2000, grid_size=30, rng=None):
    h = h_init.copy()
    L = 2 * n - 1
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full").copy()
    best_c5 = float(np.max(corr) * 2.0 / n)
    best_h = h.copy()
    arange_L = np.arange(L)
    if rng is None:
        rng = np.random.default_rng(0)
    for sweep in range(max_sweeps):
        if time.time() - t0 > tlim:
            break
        order = np.arange(n)
        if sweep > 0:
            rng.shuffle(order)
        max_change = 0.0
        for i in order:
            alpha = i + arange_L - center
            beta = i - arange_L + center
            valid_a = (alpha >= 0) & (alpha < n)
            valid_b = (beta >= 0) & (beta < n)
            delta = np.zeros(L)
            if valid_a.any():
                delta[valid_a] += (1.0 - h[alpha[valid_a]])
            if valid_b.any():
                delta[valid_b] -= h[beta[valid_b]]
            old = h[i]
            grid = np.linspace(0.0, 1.0, grid_size)
            new_corrs = corr[None, :] + (grid[:, None] - old) * delta[None, :]
            new_corrs[:, center] = corr[center] + grid * (1.0 - grid) - old * (1.0 - old)
            vals = np.max(new_corrs, axis=1)
            bi = int(np.argmin(vals))
            best_v = grid[bi]
            best_val = vals[bi]
            span = 1.0 / grid_size
            lo_r = max(0.0, best_v - span)
            hi_r = min(1.0, best_v + span)
            if hi_r > lo_r + 1e-15:
                grid2 = np.linspace(lo_r, hi_r, 16)
                nc2 = corr[None, :] + (grid2[:, None] - old) * delta[None, :]
                nc2[:, center] = corr[center] + grid2 * (1.0 - grid2) - old * (1.0 - old)
                vals2 = np.max(nc2, axis=1)
                bi2 = int(np.argmin(vals2))
                if vals2[bi2] < best_val:
                    best_v = grid2[bi2]
                    best_val = vals2[bi2]
            if abs(best_v - old) > 1e-14:
                dc = best_v - old
                corr += dc * delta
                corr[center] += best_v * (1.0 - best_v) - old * (1.0 - old) - dc * delta[center]
                h[i] = best_v
                max_change = max(max_change, abs(dc))
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5_now = float(np.max(corr) * 2.0 / n)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        if max_change < 1e-12:
            break
    return best_h, best_c5

def struct_perturb(h, n, rng):
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(6):
        f = rng.uniform(1, 12)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.004, 0.025)
        pert += a * np.sin(2 * np.pi * f * x / n + p)
    pert += rng.normal(0, 0.004, n)
    return h + pert

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B coarse-to-fine
    for T, mi in [(32, 3000), (8, 4000), (2, 5000), (1, 6000)]:
        rem = budget_s - (time.time() - t0)
        if rem < 180:
            break
        h_new = do_lbfgsb(h_best, n, T, 80, mi, 1e-13, t0, t0 + min(rem - 40, 45))
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Coordinate descent (main workhorse)
    rem = budget_s - (time.time() - t0)
    if rem > 150:
        h_cd, c5_cd = do_coord_desc(h_best, n, t0, t0 + rem - 200, 2000, 30, rng)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    # Phase 3: L-BFGS-B polish
    for T in [1.0, 0.5]:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        h_new = do_lbfgsb(h_best, n, T, 50, 4000, 1e-14, t0, t0 + min(rem - 30, 50))
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: More coordinate descent with finer grid
    rem = budget_s - (time.time() - t0)
    if rem > 100:
        h_cd, c5_cd = do_coord_desc(h_best, n, t0, t0 + rem - 120, 3000, 50, rng)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    # Phase 5: Diversification
    for pi in range(25):
        rem = budget_s - (time.time() - t0)
        if rem < 25:
            break
        if pi % 4 == 0:
            h_p = struct_perturb(h_best, n, rng)
        elif pi % 4 == 1:
            h_p = h_best + rng.normal(0, 0.003 + 0.008 * (pi % 3), n)
        elif pi % 4 == 2:
            h_p = h_best.copy()
            bs = n // 12
            bi = int(rng.integers(0, min(12, n // bs)))
            h_p[bi * bs:(bi + 1) * bs] = 1.0 - h_p[bi * bs:(bi + 1) * bs]
            h_p += rng.normal(0, 0.002, n)
        else:
            h_p = h_best + rng.normal(0, 0.001, n)
        h_p = project_feasible(h_p, n)
        h_new = do_lbfgsb(h_p, n, 2, 80, 1500, 1e-12, t0, t0 + min(rem - 8, 8))
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        rem2 = budget_s - (time.time() - t0)
        if rem2 > 15:
            h_cd, c5_cd = do_coord_desc(h_new, n, t0, t0 + min(rem2 - 5, 7), 80, 25, rng)
            if c5_cd < c5_best:
                c5_best = c5_cd
                h_best = h_cd

    # Phase 6: Final polish
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_new = do_lbfgsb(h_best, n, 1, 30, 3000, 1e-15, t0, t0 + rem - 15)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_cd, c5_cd = do_coord_desc(h_best, n, t0, t0 + rem - 8, 500, 40, rng)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
