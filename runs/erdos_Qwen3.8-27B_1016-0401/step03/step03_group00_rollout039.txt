    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def vsg(h, n, shift):
    """Vectorized shift gradient."""
    g = np.zeros(n)
    if shift >= 0:
        if n - shift > 0:
            g[:n - shift] += 1.0 - h[shift:]
        if shift > 0:
            g[shift:] -= h[:n - shift]
    else:
        s = -shift
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    return g

def adam_opt(h0, n, n_iter, T0, T1, tk, lr, t_end):
    dx = 2.0 / n
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)

        c = corr[idx]
        cm = c.max()
        e = np.exp(T * (c - cm))
        w = e / e.sum()
        g = np.zeros(n)
        for i in range(len(idx)):
            if w[i] > 1e-12:
                g += w[i] * vsg(h, n, int(idx[i]) - (n - 1))
        g *= dx
        m = b1 * m + (1 - b1) * g

_G = {}

def _sobj(x):
    n = _G['n']
    T = _G['T']
    K = _G['K']
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(corr)[-K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    return (m + np.log(e.sum()) / T) * (2.0 / n)

def _sjac(x):
    n = _G['n']
    T = _G['T']
    K = _G['K']
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(corr)[-K:]
    c = corr[idx]
    cm = c.max()
    e = np.exp(T * (c - cm))
    w = e / e.sum()
    g = np.zeros(n)
    for i in range(len(idx)):
        if w[i] > 1e-12:
            g += w[i] * vsg(x, n, int(idx[i]) - (n - 1))
    return g * (2.0 / n)

def _sconstraint(x):
    return x.sum() - _G['n'] / 2.0

def _scon_jac(x):
    return np.ones(_G['n'])

def slsqp_polish(h0, n, T, K, maxiter=3000):
    _G['n'] = n
    _G['T'] = T
    _G['K'] = min(K, 2 * n - 1)
    bounds = [(0.0, 1.0)] * n

                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sconstraint, 'jac': _scon_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new

    if shift >= 0:
        if n - shift > 0:
            g[:n - shift] += 1.0 - h[shift:]
        if shift > 0:
            g[shift:] -= h[:n - shift]
    else:
        s = -shift
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    return g

def adam_run(h0, n, nit, T0, T1, tk, lr, t_end):
    dx = 2.0 / n
    h = h0.astype(float)
    m = np.zeros(n); v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    b1p = b2p = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for t in range(1, nit + 1):
        if time.time() > t_end:
            break
        fr = t / nit
        T = T0 * (T1 / T0) ** fr
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argpartition(corr, -tk)[-tk:]
        c = corr[idx]; cm = c.max()
        e = np.exp(T * (c - cm)); w = e / e.sum()
        g = np.zeros(n)
        for i in range(tk):
            if w[i] > 1e-12:
                g += w[i] * vsg(h, n, int(idx[i]) - (n - 1))
        g *= dx
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        b1p *= b1; b2p *= b2
        h = project_feasible(h - lr * (m / (1 - b1p)) / (np.sqrt(v / (1 - b2p)) + eps), n)
        if t % 2000 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    return best_h, best_c5
