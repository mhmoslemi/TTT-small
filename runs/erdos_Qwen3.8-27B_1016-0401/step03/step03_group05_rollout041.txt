import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def grad_shift_vec(h, n, s):
    grad = np.zeros(n)
    if s >= 0:
        if s == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
    else:
        ss = -s
        grad[:n - ss] += (1.0 - h[ss:])
        grad[ss:] -= h[:n - ss]
    return grad

def slsqp_smooth(h0, n, T, maxiter, ftol):
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh -= np.max(sh)
        return (mc + np.log(np.exp(sh).sum()) / T) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh -= np.max(sh)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(n_corr):
            w = sw[oi]
            if w < 1e-12:
                continue
            grad += w * grad_shift_vec(x, n, oi - center)
        return grad * dx

    def cf(x):
        return x.sum() - n / 2.0

    def cj(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h0, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0.copy()

def slsqp_maxcon(h0, n, K, maxiter, ftol):
    dx = 2.0 / n
    center = n - 1
    N = n + 1
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    top_idx = np.argpartition(corr0, -K)[-K:]
    max_c5 = np.max(corr0)

    def obj(x):
        return x[n]

    def jac_o(x):
        g = np.zeros(N)
        g[n] = 1.0
        return g

    def cons_f(x):
        hx = x[:n]
        corr = np.correlate(hx, 1.0 - hx, mode="full")
        return x[n] - corr[top_idx]

    def cons_j(x):
        hx = x[:n]
        J = np.zeros((K, N))
        for i, oi in enumerate(top_idx):
            J[i, :n] = -grad_shift_vec(hx, n, oi - center) * dx
            J[i, n] = 1.0
        return J

    def eq_f(x):
        return x[:n].sum() - n / 2.0

    def eq_j(x):
        g = np.zeros(N)
        g[:n] = 1.0
        return g

    x0 = np.concatenate([h0, [max_c5 * dx + 0.01]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [{'type': 'ineq', 'fun': cons_f, 'jac': cons_j},
            {'type': 'eq', 'fun': eq_f, 'jac': eq_j}]
    try:
        res = minimize(obj, x0, method='SLSQP', jac=jac_o, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x[:n]
    except Exception:
        return h0.copy()

def nesterov_loop(h0, n, t0, time_limit, max_iters=500000):
    dx = 2.0 / n
    center = n - 1
    h_sg = h0.copy()
    h_prev = h0.copy()
    h_avg = h0.copy()
    c5_best = compute_c5(h0)
    h_best = h0.copy()
    alpha0 = 0.5
    K_schedule = [(0, 3), (0.3, 8), (0.6, 15), (1.0, 30)]

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        frac = min(it / max_iters, 1.0)
        K = K_schedule[0][1]
        for thr, kval in K_schedule:
            if frac >= thr:
                K = kval
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        if K <= 1:
            idx = int(np.argmax(corr))
            g = grad_shift_vec(h_y, n, idx - center) * dx
        else:
            top_idx = np.argpartition(corr, -K)[-K:]
            top_vals = corr[top_idx]
            w = (top_vals / max(mc, 1e-15)) ** 10
            ws = w.sum()
            if ws < 1e-15:
                w = np.ones(K) / K
            else:
                w /= ws
            g = np.zeros(n)
            for i in range(K):
                g += w[i] * grad_shift_vec(h_y, n, top_idx[i] - center)
            g *= dx
        alpha = alpha0 / (1.0 + 0.003 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        h_avg += (h_new - h_avg) / (float(it + 1))
        if (it + 1) % 15000 == 0:
            ha = project_feasible(h_avg, n)
            c5a = compute_c5(ha)
            if c5a < c5_best:
                c5_best = c5a
                h_best = ha.copy()
            c5c = compute_c5(h_sg)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h_sg.copy()
    return h_best, c5_best

def perturb_structured(h, n, rng, mode, scale):
    x = np.arange(n) * 2.0 / n
    if mode == 0:
        f = 0.5 + 0.5 * (mode + rng.integers(0, 4))
        return h + scale * np.sin(2 * np.pi * f * x + rng.uniform(0, 6.28))
    elif mode == 1:
        f = 0.3 + 0.2 * rng.integers(0, 6)
        return h + scale * 0.7 * np.cos(4 * np.pi * f * x + rng.uniform(0, 6.28))
    elif mode == 2:
        nb = int(n / 5)
        hp = h.copy()
        for b in range(5):
            lo = b * nb
            hi = min(lo + nb + n % 5, n)
            hp[lo:hi] += scale * rng.uniform(0.5, 1.5)
        return hp
    else:
        return h + scale * rng.normal(0, 1.0, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.ones(280)
        h_best[1::2] = 0.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP smooth with decreasing T
    for T in [60, 25, 10, 5, 2, 1]:
        if time.time() - t0 > budget_s - 500:
            break
        h_new = slsqp_smooth(h_best, n, T, 8000, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Nesterov
    h_sg, c5_sg = nesterov_loop(h_best, n, t0, budget_s - 350, 600000)
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg
    h_best = project_feasible(h_best, n)

    # Phase 3: SLSQP max-constraint (exact max via linear constraints)
    for K in [4, 8, 12]:
        if time.time() - t0 > budget_s - 150:
            break
        h_new = slsqp_maxcon(h_best, n, K, 6000, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: More SLSQP smooth passes on refined solution
    for T in [1, 0.5]:
        if time.time() - t0 > budget_s - 80:
            break
        h_new = slsqp_smooth(h_best, n, T, 10000, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Structured perturbations + local SLSQP
    for pi in range(20):
        if time.time() - t0 > budget_s - 30:
            break
        mode = pi % 4
        scale = 0.01 + 0.015 * (pi % 3)
        h_p = perturb_structured(h_best, n, rng, mode, scale)
        h_p = project_feasible(h_p, n)
        h_new = slsqp_maxcon(h_p, n, 5, 3000, 1e-17)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Try different resolutions
    elapsed = time.time() - t0
    if elapsed < budget_s - 150:
        for n2 in [n // 2, n * 2, n + 40]:
            if n2 < 60 or n2 > 600:
                continue
            if time.time() - t0 > budget_s - 60:
                break
            x_old = np.linspace(0, 2.0, n, endpoint=False)
            x_new = np.linspace(0, 2.0, n2, endpoint=False)
            h_alt = np.interp(x_new, x_old, h_best)
            h_alt = project_feasible(h_alt, n2)
            c5_alt = compute_c5(h_alt)
            h_new = slsqp_smooth(h_alt, n2, 10, 4000, 1e-17)
            h_new = project_feasible(h_new, n2)
            c5_alt = compute_c5(h_new)
            h_new2 = slsqp_maxcon(h_new, n2, 5, 3000, 1e-17)
            h_new2 = project_feasible(h_new2, n2)
            c5_alt = min(c5_alt, compute_c5(h_new2))
            if c5_alt < c5_best:
                c5_best = c5_alt
                h_best = h_new2 if compute_c5(h_new2) <= c5_alt else h_new
                n = n2

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
