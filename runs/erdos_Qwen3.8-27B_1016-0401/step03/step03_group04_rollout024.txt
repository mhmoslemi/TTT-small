                    improved = True
                else:
                    h_c[i], h_c[i+1] = o1, o2
                    h_c = project_feasible(h_c, n)
        if not improved:
            break
    return h_c, c5_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        h_best = np.array(iv, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B coarse-to-fine
    for T, mi in [(20, 2000), (8, 3000), (4, 4000), (2, 5000), (1, 5000)]:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        K = min(60, 2 * n - 1)
        h_new = do_lbfgsb(h_best, n, T, K, mi, 1e-14, t0, budget_s - 40)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 2: Local + block search
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        h_ls, c5_ls = local_search(h_best, n, t0, t0 + rem - 100, 200, 11)
        if c5_ls < c5_best:
            c5_best, h_best = c5_ls, h_ls
    rem = budget_s - (time.time() - t0)
    if rem > 100:
        h_bs, c5_bs = block_search(h_best, n, t0, t0 + rem - 80, 80, 22)
        if c5_bs < c5_best:
            c5_best, h_best = c5_bs, h_bs

    # Phase 3: L-BFGS-B fine polish
    for T, mi in [(1, 5000), (0.5, 7000), (0.3, 8000), (0.1, 10000)]:
        rem = budget_s - (time.time() - t0)
        if rem < 50:
            break
        h_new = do_lbfgsb(h_best, n, T, min(30, 2*n-1), mi, 1e-15, t0, budget_s - 15)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 4: More local search
    rem = budget_s - (time.time() - t0)
    if rem > 100:
        h_ls, c5_ls = local_search(h_best, n, t0, t0 + rem - 80, 300, 33)
        if c5_ls < c5_best:
            c5_best, h_best = c5_ls, h_ls

    # Phase 5: Very fine L-BFGS-B
    for T in [0.05, 0.02]:
        rem = budget_s - (time.time() - t0)
        if rem < 35:
            break
        h_new = do_lbfgsb(h_best, n, T, min(15, 2*n-1), 8000, 1e-16, t0, budget_s - 10)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 6: Multi-restart with diverse perturbations
    rng = np.random.default_rng(seed + 100)
    for ri in range(30):
        rem = budget_s - (time.time() - t0)
        if rem < 15:
            break
        if ri % 3 == 0:
            x = np.arange(n, dtype=float)
            pert = np.zeros(n)
            for _ in range(6):
                f = rng.uniform(1, 10)
                ph = rng.uniform(0, 2 * np.pi)
                pert += rng.uniform(0.005, 0.03) * np.sin(2 * np.pi * f * x / n + ph)
            h_p = h_best + pert
        elif ri % 3 == 1:
            sig = 0.003 + 0.008 * rng.random()
            h_p = h_best + rng.normal(0, sig, n)
        else:
            h_p = h_best.copy()
            bs = max(1, n // 10)
            bi = int(rng.integers(0, max(1, n // bs)))
            h_p[bi*bs:min(n,(bi+1)*bs)] = 1.0 - h_p[bi*bs:min(n,(bi+1)*bs)]
        h_p = project_feasible(h_p, n)
        h_new = do_lbfgsb(h_p, n, 1, min(40, 2*n-1), 2000, 1e-13, t0, budget_s - 5)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 7: Final polish + upsample
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h_new = do_lbfgsb(h_best, n, 0.1, min(20, 2*n-1), 10000, 1e-16, t0, t0 + rem - 20)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    rem = budget_s - (time.time() - t0)
    if rem > 180 and n < 600:
        n2 = n * 2
        xo = np.arange(n) * (2.0 / n)
        xn = np.arange(n2) * (2.0 / n2)
        h_hi = np.interp(xn, xo, h_best)
        h_hi = project_feasible(h_hi, n2)
        c5_hi = compute_c5(h_hi)
        if c5_hi < c5_best:
            c5_best, h_best, n = c5_hi, h_hi, n2
        else:
            c5_hi = min(c5_hi, c5_best)
            n = n2
            h_best = h_hi
            c5_best = min(c5_best, c5_hi)
        for T, mi in [(8, 2000), (2, 3000), (0.5, 4000)]:
            rem = budget_s - (time.time() - t0)
            if rem < 50:
                break
            h_new = do_lbfgsb(h_best, n, T, min(50, 2*n-1), mi, 1e-14, t0, budget_s - 20)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best, h_best = c5, h_new
        rem = budget_s - (time.time() - t0)
        if rem > 60:
            h_ls, c5_ls = local_search(h_best, n, t0, t0 + rem - 40, 150, 55)
            if c5_ls < c5_best:
                c5_best, h_best = c5_ls, h_ls

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
