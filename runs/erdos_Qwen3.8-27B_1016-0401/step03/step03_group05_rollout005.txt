from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0

        else:

def shift_grad(h, n, shift):
    """Gradient of the correlation value at the given linear shift.
    shift = corr_index - (n-1), so shift=0 is the center."""
    j = np.arange(n)
    grad = np.zeros(n)
    s = shift
    if s >= 0:
        m = min(n - s, n)
        for i in range(m):
            grad[i + s] += (1.0 - h[i])
            grad[i] -= h[i + s]
    else:
        t = -s
        m = min(n - t, n)
        for i in range(m):

            grad[i + t] -= h[i]
    return grad

def lse_obj_and_grad(h, n, T):
    """Log-sum-exp smooth approximation of max(corr)*dx."""
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    # Temperature relative to max
    sh = T * (corr - mc)
    # Clip for numerical stability
    sh = np.clip(sh, -500, 0)
    e = np.exp(sh)
    ex = e.sum()
    val = (mc + np.log(ex) / T) * dx
    
    sw = e / ex
    grad = np.zeros(n)
    for oi in range(len(corr)):
        w = sw[oi]
        if w < 1e-10:
            continue
        grad += w * shift_grad(h, n, oi - center)
    grad *= dx
    return val, grad

    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        val, _ = lse_obj_and_grad(x, n, T)
        return val

        _, g = lse_obj_and_grad(x, n, T)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    grad = np.zeros(n)
    s = shift
    if s >= 0:
        m = min(n - s, n)
        for i in range(m):
            grad[i + s] += (1.0 - h[i])
            grad[i] -= h[i + s]
    else:
        t = -s
        m = min(n - t, n)
        for i in range(m):
            grad[i] += (1.0 - h[i + t])
            grad[i + t] -= h[i]
    return grad

def lse_val_grad(h, n, T):
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = np.clip(sh, -400.0, 0.0)
    e = np.exp(sh)
    ex = e.sum()
    val = (mc + np.log(max(ex, 1e-300)) / T) * dx
    sw = e / max(ex, 1e-300)
    grad = np.zeros(n)
    n_corr = len(corr)
    for oi in range(n_corr):
        w = sw[oi]
        if w < 1e-10:
            continue
        grad += w * shift_grad(h, n, oi - center)
    grad *= dx
    return val, grad

    target = n / 2.0
    def obj(x):
        v, _ = lse_val_grad(x, n, T)
        return v
    def jac(x):
        _, g = lse_val_grad(x, n, T)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, method='SLSQP', jac=jac, bounds=bounds,

        h_best = 0.5 + 0.3 * np.sin(np.linspace(0, 2 * np.pi, 280))

    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP smooth (coarse → fine)
    d1 = t0 + budget_s * 0.20
    for T in [24, 12, 6, 3, 1.5]:
        if time.time() > d1:
            break
        h_new = slsqp_smooth(h_best, n, T, 6000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: FISTA annealing
    d2 = t0 + budget_s * 0.45
    h_f, c5_f = fista_anneal(h_best, n, t0, d2, T0=30, T1=0.8)
    if c5_f < c5_best:
        c5_best = c5_f
        h_best = h_f
    h_best = project_feasible(h_best, n)

    # Phase 3: Top-K SLSQP (focus on worst shifts)
    d3 = t0 + budget_s * 0.65
    for K in [4, 7, 12, 20, 35]:
        if time.time() > d3:
            break
        h_new = slsqp_topk(h_best, n, K, 7000, 1e-17)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Fine SLSQP
    d4 = t0 + budget_s * 0.80
    for T in [1.0, 0.5, 0.25]:
        if time.time() > d4:
            break
        h_new = slsqp_smooth(h_best, n, T, 10000, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Subgradient polish with averaging
    d5 = t0 + budget_s * 0.92
    h_sg, c5_sg = subgradient_polish(h_best, n, t0, d5, 300000)
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg
    h_best = project_feasible(h_best, n)

    # Phase 6: Final top-3 SLSQP
    d6 = t0 + budget_s - 15
    if time.time() < d6:
        h_new = slsqp_topk(h_best, n, 3, 12000, 1e-19)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        h_new = slsqp_topk(h_best, n, 5, 8000, 1e-19)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
