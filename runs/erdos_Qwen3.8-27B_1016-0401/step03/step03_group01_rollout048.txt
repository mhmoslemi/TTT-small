from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def grad_topk(h, n, k=3):
    """Gradient of sum of top-k correlations."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    grad = np.zeros(n)
    center = n - 1
    j = np.arange(n)
    dx = 2.0 / n
    for oi in idx:
        shift = oi - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += (1.0 - h[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= h[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= h[j[m2] - s2]
    return grad * dx

def to_bangbang(h, n):
    target = n // 2
    thresh = np.partition(h, n - target)[n - target]
    bb = (h >= thresh).astype(float)
    n_ones = int(bb.sum())
    if n_ones > target:
        ones_idx = np.where(bb == 1)[0]
        order = np.argsort(h[ones_idx])
        for cnt in range(n_ones - target):
            bb[ones_idx[order[cnt]]] = 0
    elif n_ones < target:
        zeros_idx = np.where(bb == 0)[0]
        order = np.argsort(h[zeros_idx])[::-1]

def slsqp_quick(h, n, T, k, maxiter):
    dx = 2.0 / n
    target = n / 2.0
    def fval(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mk = min(k, len(corr))
        idx = np.argpartition(corr, -mk)[-mk:]
        topk = corr[idx]
        mc = np.max(topk)
        sh = T * (topk - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    def fjac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mk = min(k, len(corr))
        idx = np.argpartition(corr, -mk)[-mk:]
        topk = corr[idx]
        mc = np.max(topk)
        sh = T * (topk - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        g = np.zeros(n)
        center = n - 1
        j = np.arange(n)
        for oi in range(mk):
            w = sw[oi]
            if w < 1e-12:

            shift = idx[oi] - center
            if shift >= 0:
                m1 = j >= shift
                g[m1] += w * (1.0 - x[j[m1] - shift])
                m2 = (j + shift) < n
                g[m2] -= w * x[j[m2] + shift]
            else:
                s2 = -shift
                m1 = (j + s2) < n
                g[m1] += w * (1.0 - x[j[m1] + s2])
                m2 = j >= s2
                g[m2] -= w * x[j[m2] - s2]
        return g * dx

        res = minimize(fval, h.copy(), method='SLSQP', jac=fjac,
                      bounds=[(0,1)]*n,
                      constraints=[{'type':'eq','fun':lambda x: x.sum()-target,'jac':lambda x: np.ones(n)}],
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return res.x

def compute_c5(h):

def project_feasible(y, n):

            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

    return x.sum() - _global_target

        mc = np.max(topk)
        sh = T * (topk - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    def fjac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mk = min(k, len(corr))
        idx = np.argpartition(corr, -mk)[-mk:]
