import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

_G = {}

def _obj(x):
    n = _G['n']
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G['tk']:]
    c = corr[idx]
    m = c.max()
    T = _G['T']
    dx = 2.0 / n
    pen = _G['pen']
    cv = x.sum() - n / 2.0
    if T < 1e-6:
        return m * dx + pen * cv * cv
    e = np.exp(T * (c - m))
    sm = m + np.log(e.sum()) / T
    return sm * dx + pen * cv * cv

def _grad(x):
    n = _G['n']
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G['tk']:]
    c = corr[idx]
    m = c.max()
    T = _G['T']
    dx = 2.0 / n
    pen = _G['pen']
    cv = x.sum() - n / 2.0
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for wi, ci in zip(w, idx):
        if wi > 1e-10:
            g += wi * shift_grad(x, n, ci - (n - 1))
    g *= dx
    g += 2.0 * pen * cv * np.ones(n)
    return g

def lbfgs_step(h, n, T, top_k, pen, maxiter):
    _G['n'] = n
    _G['T'] = T
    _G['tk'] = top_k
    _G['pen'] = pen
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        return res.x
    except:
        return h.copy()

def nesterov_refine(h_init, n, t_end, rng, top_k=8):
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    step = 1.0
    for it in range(200000):
        if time.time() > t_end:
            break
        if it % 100 == 0:
            step = 1.0 / (1.0 + 0.001 * it)
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        # Project to feasible
        h_y = project_feasible(h_y, n)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h_y, n, ci - (n - 1))
        g *= dx / top_k
        g /= (np.linalg.norm(g) + 1e-12)  # normalize
        # Line search
        alpha = step
        for ls in range(10):
            h_new = project_feasible(h_y - alpha * g, n)
            f_new = compute_c5(h_new)
            if f_new <= compute_c5(h_y):
                break
            alpha *= 0.5
        h_prev = h_cur
        h_cur = h_new
        c1 = compute_c5(h_cur)
        if c1 < best_c5:
            best_c5, best_h = c1, h_cur.copy()
    return best_h, best_c5

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

_GN = 0
_GT = 1.0
_GK = 10
_GP = 1.0

def _o(x):

    corr = np.correlate(x, 1.0 - x, mode="full")

    m = c.max()
    T = _GT
    dx = 2.0 / n

        return m * dx + _GP * cv * cv
    e = np.exp(T * (c - m))
    return (m + np.log(e.sum()) / T) * dx + _GP * cv * cv

def _g(x):
    n = _GN
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_GK:]
    c = corr[idx]
    m = c.max()
    T = _GT
    dx = 2.0 / n
    cv = x.sum() - n * 0.5
    e = np.exp(T * (c - m))

    g *= dx
    g += 2.0 * _GP * cv

        r = minimize(_o, h, jac=_g, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': mi, 'ftol': 1e-16, 'gtol': 1e-13, 'iprint': -1})
        return r.x
    except:
        return h.copy()

def nesterov(h0, n, tel, rng, tk=8):

    hc, hp, ha = h0.copy(), h0.copy(), h0.copy()
    bc, bh = compute_c5(h0), h0.copy()
    for it in range(500000):
        if time.time() > tel:
            break
        b = float(it) / (it + 1.5) if it else 0.0
        hy = project_feasible(np.clip(hc + b * (hc - hp), 0, 1), n)
        corr = np.correlate(hy, 1.0 - hy, mode="full")
        idx = np.argsort(corr)[-(tk):]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(hy, n, ci - (n - 1))
        g *= dx / tk
        nm = np.linalg.norm(g)
        if nm < 1e-14:

        if np.clip(y - mid, 0.0, 1.0).sum() > tgt:

            hi = mid

    g = np.zeros(n)
    if shift >= 0:
        m1 = (j + shift) < n
        m2 = j >= shift
        g[m1] += 1.0 - h[j[m1] + shift]
