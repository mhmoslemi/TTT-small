    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0

        mid = (lo + hi) / 2.0

        if abs(s - target) < 1e-14:
            break
        if s > target:

        else:

    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def solve_lp(h0, n, K, maxiter_lp=5000):

    b_rows = []

def slsqp_smooth(h, n, T, maxiter, ftol):
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx

        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)

            w = sw[oi]
            if w < 1e-11:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return grad * dx
    
    def cons_fun(x):
        return x.sum() - n / 2.0

        return np.ones(n)

    except Exception:

def nesterov_step(h, n, it, K):
    """Single Nesterov step."""
    dx = 2.0 / n
    center = n - 1
    # Just compute gradient for top K shifts
    corr = np.correlate(h, 1.0 - h, mode="full")
    if K <= 1:
        idx = int(np.argmax(corr))
        g = shift_grad(h, n, idx - center) * dx
    else:
        top_idx = np.argpartition(corr, -K)[-K:]
        top_vals = corr[top_idx]
        w = (top_vals / max(np.max(corr), 1e-15))**8
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.ones(K) / K
        else:
            w /= w_sum
        g = np.zeros(n)
        for i in range(K):
            g += w[i] * shift_grad(h, n, top_idx[i] - center)
        g *= dx
    return g, np.max(corr) * dx

One more idea: **use the dual problem**. The primal is:
min_h max_k v_k(h) s.t. 0 ≤ h ≤ 1, sum h = n/2

The Lagrangian dual (for the max) would involve weights w_k ≥ 0, sum w_k = 1:

The inner problem: min_h sum_k w_k * v_k(h) is a QP (quadratic in h). If the weighted sum of Q_k matrices is positive semi-definite, this is convex and can be solved exactly!

The weighted sum: sum_k w_k * Q_k. For this to be PSD, we need the weighted sum of shift matrices to be PSD. The eigenvalues of Q_k are exp(2πi*j*k/n), so the eigenvalues of sum_k w_k * Q_k are sum_k w_k * exp(2πi*j*k/n) = ŵ(-j) (the Fourier transform of w at frequency -j).

This is the Fejér-Riesz condition! It's not always satisfied, but for many weight vectors w it will be.

If it IS satisfied, the inner QP can be solved exactly, and the dual becomes:
max_w (QP minimum value) s.t. w ≥ 0, sum w = 1, and ŵ ≥ 0 (positive definite condition)

        row[:n] = g
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(float(g @ h0 - corr[oi]))
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = []
    for i in range(n):
        lo = max(0.0, h0[i] - delta)
        hi = min(1.0, h0[i] + delta)
        bounds.append((lo, hi))
    bounds.append((-1.0, 1.0))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 10000})
        if res.status == 0:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return h0, np.inf

def nesterov_run(h_init, n, t0, time_limit, max_iters=400000):
    dx = 2.0 / n
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_best = h_init.copy()
    c5_best = compute_c5(h_init)
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        frac = min(it / max_iters, 1.0)
        K = max(1, int(2 + 28 * frac * frac))
        alpha = 0.45 / (1.0 + 0.0015 * it)
        beta = (float(it) / (it + 1.5)) if it > 0 else 0.0
        h_y = np.clip(h_sg + beta * (h_sg - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        if K <= 1:
            idx = int(np.argmax(corr))
            g = shift_grad(h_y, n, idx - (n-1)) * dx
        else:
            top_idx = np.argpartition(corr, -K)[-K:]
            top_vals = corr[top_idx]
            w = (top_vals / max(mc, 1e-15))**6
            ws = w.sum()
            w = w / ws if ws > 1e-15 else np.ones(K) / K
            g = np.zeros(n)
            for i in range(K):
                g += w[i] * shift_grad(h_y, n, top_idx[i] - (n-1))
            g *= dx
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_sg.copy()
        h_sg = h_new
        if (it + 1) % 8000 == 0:
            c5c = compute_c5(h_sg)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h_sg.copy()
    h_best = project_feasible(h_best, n)
    c5_best = min(compute_c5(h_best), c5_best)
    return h_best, c5_best

def lp_polish(h0, n, t0, time_limit, n_iters=150):
    c5_best = compute_c5(h0)
    h_best = h0.copy()
    delta = 0.15
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        K = min(8 + it // 4, 60)
        h_lp, _ = solve_lp_trust(h_best, n, K, delta)
        h_lp = project_feasible(h_lp, n)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_best - 1e-8:
            c5_best = c5_lp
            h_best = h_lp
            delta = min(delta * 1.3, 0.4)
        else:
            improved = False
            for alpha2 in [0.5, 0.3, 0.15]:
                h_try = project_feasible((1 - alpha2) * h_best + alpha2 * h_lp, n)
                c5_try = compute_c5(h_try)
                if c5_try < c5_best - 1e-9:
                    c5_best = c5_try
                    h_best = h_try
                    improved = True
                    break
            if not improved:
                delta *= 0.7
                if delta < 0.01:
                    delta = 0.1
    return h_best, c5_best

def coordinate_refine(h0, n, t0, time_limit, n_passes=5):
    dx = 2.0 / n
    h_best = h0.copy()
    c5_best = compute_c5(h0)
    for p in range(n_passes):
        if time.time() - t0 > time_limit:
            break
        n = len(h_best)
        order = np.random.default_rng(100 + p * 7).permutation(n)
        for j in order:
            if time.time() - t0 > time_limit:
                break
            saved = h_best[j]
            for trial in np.linspace(0, 1, 11):
                if abs(trial - saved) < 1e-6:
                    continue
                h_best[j] = trial
                h_best = project_feasible(h_best, n)
                c5_t = compute_c5(h_best)
                if c5_t < c5_best:
                    c5_best = c5_t
                else:
                    h_best[j] = saved
                    break
            h_best[j] = saved
            h_best = project_feasible(h_best, n)
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP moderate polish
    for T in [12, 6]:
        if time.time() - t0 > budget_s - 550:
            break
        h_new = slsqp_smooth(h_best, n, T, 2500, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: LP linearization (new method)
    h_best, c5_best = lp_polish(h_best, n, t0, budget_s - 350, 120)

    # Phase 3: Nesterov from LP result
    if time.time() - t0 < budget_s - 250:
        h_ng, c5_ng = nesterov_run(h_best, n, t0, budget_s - 250, 250000)
        if c5_ng < c5_best:
            c5_best = c5_ng
            h_best = h_ng

    # Phase 4: SLSQP fine
    for T in [3, 1.5, 1]:
        if time.time() - t0 > budget_s - 150:
            break
        h_new = slsqp_smooth(h_best, n, T, 3500, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: More LP
    h_best, c5_best = lp_polish(h_best, n, t0, budget_s - 80, 80)

    # Phase 6: Coordinate refinement
    if time.time() - t0 < budget_s - 30:
        h_cr, c5_cr = coordinate_refine(h_best, n, t0, budget_s - 20, 3)
        if c5_cr < c5_best:
            c5_best = c5_cr
            h_best = h_cr

    # Phase 7: Final SLSQP
    if time.time() - t0 < budget_s - 15:
        h_new = slsqp_smooth(h_best, n, 1, 4000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
