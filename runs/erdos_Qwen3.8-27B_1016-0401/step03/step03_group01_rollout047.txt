def slsqp_smooth(h, n, T, k, maxiter, ftol):
    target = n / 2.0
    j = np.arange(n)
    center = n - 1
    
    def obj_func(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mk = min(k, len(corr))
        idx = np.argpartition(corr, -mk)[-mk:]
        topk = corr[idx]
        mc = np.max(topk)
        sh = T * (topk - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * (2.0 / n)
    
    def jac_func(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mk = min(k, len(corr))
        idx = np.argpartition(corr, -mk)[-mk:]
        topk = corr[idx]
        mc = np.max(topk)
        sh = T * (topk - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(len(idx)):
            w = sw[oi]
            if w < 1e-10:
                continue
            shift = idx[oi] - center
            if shift >= 0:
                m1 = j >= shift
                grad[m1] += w * (1.0 - x[j[m1] - shift])
                m2 = (j + shift) < n
                grad[m2] -= w * x[j[m2] + shift]
            else:
                s2 = -shift
                m1 = (j + s2) < n
                grad[m1] += w * (1.0 - x[j[m1] + s2])
                m2 = j >= s2
                grad[m2] -= w * x[j[m2] - s2]
        grad *= 2.0 / n
        return grad
    
    def cf(x):
        return x.sum() - target
    def cj(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj_func, h, method='SLSQP', jac=jac_func, bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except:
        return h.copy()

    dx = 2.0 / n
    h = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h.copy()
    grad_avg = np.zeros(n)
    ema_decay = 0.9
    
    for it in range(500000):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.argmax(corr)
        shift = m - (n - 1)

        grad = np.zeros(n)
        j = np.arange(n)
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += (1.0 - h[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= h[j[m2] + shift]
        else:

            grad[m1] += (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= h[j[m2] - s2]

        grad_avg = ema_decay * grad_avg + (1 - ema_decay) * grad
