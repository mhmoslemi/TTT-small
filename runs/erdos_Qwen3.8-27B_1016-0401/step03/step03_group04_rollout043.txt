import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

_N = [280]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def sm_val_and_grad(x, n, T, K):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    center = n - 1
    if K < len(corr):
        topk = np.argpartition(corr, -K)[-K:]
        c_sel = corr[topk]
        mc_sel = np.max(c_sel)
        sh = T * (c_sel - mc_sel)
        e = np.exp(sh)
        val = (mc_sel + np.log(e.sum()) / T) * (2.0 / n)
        sw = e / e.sum()
        grad = np.zeros(n)
        for ii in range(len(sw)):
            w = sw[ii]
            if w < 1e-15:
                continue
            grad += w * shift_grad(x, n, topk[ii] - center)
        return val, grad * (2.0 / n)
    else:
        sh = T * (corr - mc)
        e = np.exp(sh)
        val = (mc + np.log(e.sum()) / T) * (2.0 / n)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(len(sw)):
            w = sw[oi]
            if w < 1e-15:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return val, grad * (2.0 / n)

def _obj_lb(x, n, T, K, t0, tl):
    if time.time() - t0 > tl:
        _tf[0] = True
    v, _ = sm_val_and_grad(x, n, T, K)
    return v

def _grad_lb(x, n, T, K, t0, tl):
    if time.time() - t0 > tl:
        _tf[0] = True
    _, g = sm_val_and_grad(x, n, T, K)
    return g

    return x.sum() - _N[0] / 2.0

def _slsqp_eq_jac(x):

def lbfgsb_opt(h, n, T, K, mi, tol, t0, tl):
    _tf[0] = False
    try:
        res = minimize(_obj_lb, h, method='L-BFGS-B', jac=_grad_lb,
                       args=(n, T, K, t0, tl),
