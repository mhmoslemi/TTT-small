```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_fg(h, n, T):
    """Soft-max value and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = float(corr.max())
    sh = T * (corr - mc)
    sh = np.clip(sh, -100.0, 0.0)
    e = np.exp(sh)
    Z = e.sum()
    val = (mc + np.log(Z) / T) * dx
    sw = e / Z
    center = n - 1
    grad = np.zeros(n)
    grad += sw[center] * (1.0 - 2.0 * h)
    for s in range(1, n):
        wp = sw[center + s]
        wm = sw[center - s]
        if wp < 1e-8 and wm < 1e-8:
            continue
        if wp >= 1e-8:
            grad[s:] += wp * (1.0 - h[:n-s])
            grad[:n-s] -= wp * h[s:]
        if wm >= 1e-8:
            grad[:n-s] += wm * (1.0 - h[s:])
            grad[s:] -= wm * h[:n-s]
    return val, grad * dx

def adam_optimize(h0, n, t0, tlim, max_iters=500000):
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    c5_best = compute_c5(h)
    h_best = h.copy()
    T_start = 20.0
    T_end = 0.5
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        T = T_start * (T_end / T_start) ** (it / max_iters)
        T = max(T, T_end)
        lr = 0.8 / (1.0 + 0.001 * it)
        val, g = soft_max_fg(h, n, T)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, n)
        if (it + 1) % 5000 == 0:
            c5_now = compute_c5(h)
            if c5_now < c5_best:
                c5_best = c5_now
                h_best = h.copy()
    return h_best, c5_best

def lbfgsb_polish(h, n, T, K, maxiter, t0, tlim):
    """L-BFGS-B with soft-max."""
    dx = 2.0 / n
    center = n - 1
    
    def obj(x):
        return soft_max_fg(x, n, T)[0]
    
    def jac(x):
        return soft_max_fg(x, n, T)[1]
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jac,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 20})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    
    # Phase 1: Adam optimization with temperature annealing
    rem = budget_s - (time.time() - t0)
    if rem > 300:
        h_adam, c5_adam = adam_optimize(h_best, n, t0, t0 + rem - 250)
        if c5_adam < c5_best:
            c5_best = c5_adam
            h_best = h_adam
    
    # Phase 2: L-BFGS-B polish
    for T in [4.0, 1.0, 0.3]:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        h_new = lbfgsb_polish(h_best, n, T, 559, 5000, t0, budget_s - 30)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 2.0
    hi = float(y.max()) + 0.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_fg(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = float(corr.max())
    sh = T * (corr - mc)
    sh = np.clip(sh, -100.0, 0.0)
    e = np.exp(sh)
    Z = e.sum()
    val = (mc + np.log(Z) / T) * dx
    sw = e / Z
    center = n - 1
    grad = np.zeros(n)
    grad += sw[center] * (1.0 - 2.0 * h)
    for s in range(1, n):
        wp = sw[center + s]
        wm = sw[center - s]
        if wp < 1e-8 and wm < 1e-8:
            continue
        if wp >= 1e-8:
            grad[s:] += wp * (1.0 - h[:n-s])
            grad[:n-s] -= wp * h[s:]
        if wm >= 1e-8:
            grad[:n-s] += wm * (1.0 - h[s:])
            grad[s:] -= wm * h[:n-s]
    return val, grad * dx

def _sm_obj(x, n_val, T_val):
    return soft_max_fg(x, n_val, T_val)[0]

def _sm_jac(x, n_val, T_val):
    return soft_max_fg(x, n_val, T_val)[1]

def _sum_con_fun(x):
    return x.sum() - x.size / 2.0

def _sum_con_jac(x):
    return np.ones(x.size)

def adam_optimize(h0, n, t0, tlim, max_iters=600000):
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    c5_best = compute_c5(h)
    h_best = h.copy()
    T_start, T_end = 25.0, 0.3
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        frac = it / max_iters
        T = T_start * (T_end / T_start) ** frac
        T = max(T, T_end)
        lr = 0.9 / (1.0 + 0.0005 * it)
        val, g = soft_max_fg(h, n, T)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1 ** (it + 1))
        v_hat = v / (1 - b2 ** (it + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, n)
        if (it + 1) % 3000 == 0:
            c5_now = compute_c5(h)
            if c5_now < c5_best:
                c5_best = c5_now
                h_best = h.copy()
    c5_last = compute_c5(h)
    if c5_last < c5_best:
        c5_best = c5_last
        h_best = h.copy()
    return h_best, c5_best

def nesterov_subgrad(h0, n, t0, tlim, max_iters=200000):
    h_curr = h0.copy()
    h_prev = h0.copy()
    c5_best = compute_c5(h0)
    h_best = h0.copy()
    f_star = c5_best
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        beta = it / (it + 2.0)
        y = h_curr + beta * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        dx = 2.0 / n
        corr = np.correlate(y, 1.0 - y, mode="full")
        idx = int(np.argmax(corr))
        center = n - 1
        shift = idx - center
        j = np.arange(n)
        g = np.zeros(n)
        if shift >= 0:
            m1 = j >= shift
            m2 = (j + shift) < n
            if m1.any():
                g[m1] += (1.0 - y[j[m1] - shift])
            if m2.any():
                g[m2] -= y[j[m2] + shift]
        else:
            s = -shift
            m1 = (j + s) < n
            m2 = j >= s
            if m1.any():
                g[m1] += (1.0 - y[j[m1] + s])
            if m2.any():
                g[m2] -= y[j[m2] - s]
        g *= dx
        gn2 = float(g @ g)
        if gn2 < 1e-28:
            break
        f_y = compute_c5(y)
        gap = max(f_y - f_star, 1e-8)
        alpha = min(2.0 * gap / gn2, 0.5 / (1.0 + 0.001 * it))
        h_new = y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_curr.copy()
        h_curr = h_new
        if (it + 1) % 5000 == 0:
            c5_now = compute_c5(h_curr)
            if c5_now < c5_best:
                c5_best = c5_now
                h_best = h_curr.copy()
                f_star = c5_best
    return h_best, c5_best

def clip_sweep(h, n):
    h_new = h.copy()
    best_c5 = compute_c5(h_new)
    for thr in [0.01, 0.03, 0.05, 0.08]:
        h_test = h.copy()
        h_test[h_test < thr] = 0.0
        h_test[h_test > 1.0 - thr] = 1.0
        h_test = project_feasible(h_test, n)
        c5_t = compute_c5(h_test)
        if c5_t < best_c5:
            best_c5 = c5_t
            h_new = h_test
    return h_new, best_c5

    rem = budget_s - (time.time() - t0)
    if rem > 250:
        h_adam, c5_adam = adam_optimize(h_best, n, t0, t0 + rem - 200)
        if c5_adam < c5_best:
            c5_best = c5_adam
            h_best = h_adam

    # Phase 2: Nesterov subgradient on true max
    rem = budget_s - (time.time() - t0)

        if c5_ng < c5_best:
            c5_best = c5_ng
            h_best = h_ng

    # Phase 3: L-BFGS-B polish with decreasing T
    for T in [6.0, 2.0, 0.5, 0.2]:
        rem = budget_s - (time.time() - t0)
        if rem < 70:
            break
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(_sm_obj, h_best, method='L-BFGS-B', jac=_sm_jac,
                           args=(n, T), bounds=bounds,
                           options={'maxiter': 8000, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 20})
            h_new = project_feasible(res.x, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new
        except Exception:
            pass

    # Phase 4: Clip sweep (binary-ify near-extreme values)

    n_restarts = 40

        rem = budget_s - (time.time() - t0)
        if rem < 15:
            break

        elif pi % 3 == 1:

            bs = max(2, n // 12)

        h_p = project_feasible(h_p, n)
```
