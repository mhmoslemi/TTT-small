import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

def adam_optimize(h_init, n, t_end, rng, top_k=8, lr_init=0.01, n_iter_max=200000):
    """Adam-style optimization focused on worst shifts."""
    dx = 2.0 / n
    h = h_init.copy()
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    for it in range(n_iter_max):
        if time.time() > t_end:
            break
        
        # Compute current worst shifts
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argsort(corr)[-top_k:]
        
        # Compute gradient as average of worst shifts
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h, n, ci - (n - 1))
        g *= dx / top_k
        
        # Adam update
        lr = lr_init / (1.0 + 0.01 * it)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        h_new = project_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
        
        h = h_new
    
    return h_best, c5_best

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

def adam_optimize(h_init, n, t_end, rng, top_k=5, lr_init=0.04, max_iter=300000):
    dx = 2.0 / n
    h = h_init.copy()
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    c5_best = compute_c5(h)
    h_best = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    active = set(np.argsort(corr)[-top_k:].tolist())
    for it in range(max_iter):
        if time.time() > t_end:
            break
        if it % 500 == 0:
            corr = np.correlate(h, 1.0 - h, mode="full")
            tk_now = max(3, top_k - it // 30000)
            active = set(np.argsort(corr)[-tk_now:].tolist())
        g = np.zeros(n)
        for ci in active:
            g += shift_grad(h, n, ci - (n - 1))
        g *= dx / len(active)
        lr = lr_init / (1.0 + 0.003 * it)
        m1 = b1 * m1 + (1 - b1) * g
        m2 = b2 * m2 + (1 - b2) * g * g
        m1h = m1 / (1 - b1 ** (it + 1))
        m2h = m2 / (1 - b2 ** (it + 1))
        h = project_feasible(h - lr * m1h / (np.sqrt(m2h) + eps), n)
        if (it + 1) % 2000 == 0:
            c = compute_c5(h)
            if c < c5_best:
                c5_best, h_best = c, h.copy()
    c = compute_c5(h)
    if c < c5_best:
        c5_best, h_best = c, h
    return h_best, c5_best

def slsqp_focused(h_init, n, T, top_k, maxiter, ftol):
    N = n
    tgt = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        mx = c.max()
        e = np.exp(T * (c - mx))
        return (mx + np.log(e.sum()) / T) * (2.0 / N)
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        mx = c.max()
        e = np.exp(T * (c - mx))
        w = e / e.sum()
        grad = np.zeros(N)
        for wi, ci in zip(w, idx):
            if wi > 1e-10:
                grad += wi * shift_grad(x, N, ci - (N - 1))
        return grad * (2.0 / N)
    def cfun(x):
        return x.sum() - tgt
    def cjac(x):
        return np.ones(N)
    bounds = [(0.0, 1.0)] * N
    try:
        res = minimize(obj, h_init, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init.copy()

def coordinate_descent(h_init, n, t_end, rng, n_rounds=500):
    dx = 2.0 / n
    h = h_init.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    active = set(np.argsort(corr)[-8:].tolist())
    for rnd in range(n_rounds):

        if rnd % 50 == 0:
            corr = np.correlate(h, 1.0 - h, mode="full")

        for _ in range(3):
            g = np.zeros(n)
            for ci in active:
                g += shift_grad(h, n, ci - (n - 1))
            g *= dx / len(active)
            for _coord in range(40):
                i = rng.integers(0, n)
                step = 0.5 / (1.0 + 0.01 * rnd)
                h_trial = h.copy()
                h_trial[i] = np.clip(h[i] - step * g[i], 0.0, 1.0)
                h_trial = project_feasible(h_trial, n)

                if c <= c5_best:

                else:

                    c = compute_c5(h_trial)
                    if c < c5_best:

    return h_best, c5_best
