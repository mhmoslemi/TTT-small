def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0

        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break

            lo = mid
        else:
            hi = mid

def shift_grad(h, n, shift):
    j = np.arange(n)

def binary_project(h, n):
    """Threshold to 0-1, adjust to meet sum constraint"""
    dB = (h > 0.5).astype(float)
    # Adjust: if too many ones, flip some; if too few, flip others
    target = n // 2
    n_ones = int(dB.sum())
    if n_ones > target:
        # Flip some 1s to 0s: prefer those with lowest h
        ones_idx = np.where(dB == 1)[0]
        order = np.argsort(h[ones_idx])
        for idx in ones_idx[order[:n_ones - target]]:
            dB[idx] = 0
    elif n_ones < target:
        zeros_idx = np.where(dB == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for idx in zeros_idx[order[:target - n_ones]]:
            dB[idx] = 1
    return dB

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = float(y.min()) - 3.0, float(y.max()) + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def smooth_max_val_grad(x, n, T, K):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    if K < len(corr):
        topk = np.argpartition(corr, -K)[-K:]
        c_sel = corr[topk]
        mc_sel = np.max(c_sel)
        sh = T * (c_sel - mc_sel)
        e = np.exp(sh)
        val = (mc_sel + np.log(e.sum()) / T) * (2.0 / n)
        sw = e / e.sum()
        center = n - 1
        grad = np.zeros(n)
        for ii in range(len(sw)):
            if sw[ii] > 1e-12:
                grad += sw[ii] * shift_grad(x, n, topk[ii] - center)
        return val, grad * (2.0 / n)
    sh = T * (corr - mc)
    e = np.exp(sh)
    val = (mc + np.log(e.sum()) / T) * (2.0 / n)
    sw = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    for oi in range(len(sw)):
        if sw[oi] > 1e-12:
            grad += sw[oi] * shift_grad(x, n, oi - center)
    return val, grad * (2.0 / n)

_fl = [False]

def _lb_obj(x, n, T, K, t0, tlim):
    if time.time() - t0 > tlim:
        _fl[0] = True
    v, _ = smooth_max_val_grad(x, n, T, K)
    return v

def _lb_grad(x, n, T, K, t0, tlim):
    if time.time() - t0 > tlim:
        _fl[0] = True
    _, g = smooth_max_val_grad(x, n, T, K)
    return g

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):
    _fl[0] = False
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lb_obj, h, method='L-BFGS-B', jac=_lb_grad,
                       args=(n, T, K, t0, tlim), bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol * 0.05,
                                'maxls': 40, 'maxfun': maxiter * 4})
        if _fl[0]:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def do_slsqp(h, n, T, K, maxiter, tol, t0, tlim):
    _fl[0] = False
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: smooth_max_val_grad(x, n, T, K)[0], h, method='SLSQP',
                       jac=lambda x: smooth_max_val_grad(x, n, T, K)[1],
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if _fl[0]:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def binary_cd(h, n, n_sweeps, t0, tlim):
    """Binary coordinate descent: try setting each h[i] to 0 or 1"""
    dx = 2.0 / n
    h_c = h.copy()
    c5_cur = compute_c5(h_c)
    for sw in range(n_sweeps):
        if time.time() - t0 > tlim:
            break
        order = np.random.permutation(n)
        for i in order:
            old = h_c[i]
            for val in (0.0, 1.0):
                h_c[i] = val
                c5_try = compute_c5(h_c)
                if c5_try < c5_cur - 1e-14:
                    c5_cur = c5_try
                    break
            else:
                h_c[i] = old
        if (sw + 1) % 10 == 0:
            h_c = project_feasible(h_c, n)
            c5_cur = compute_c5(h_c)
    return h_c, c5_cur

def nesterov_step(h, n, t0, tlim, max_it=200000):
    dx = 2.0 / n
    h_c, h_p = h.copy(), h.copy()
    c5_b, h_b = compute_c5(h), h.copy()
    for it in range(max_it):
        if time.time() - t0 > tlim:
            break
        beta = it / (it + 2.5)
        y = np.clip(h_c + beta * (h_c - h_p), 0, 1)
        corr = np.correlate(y, 1.0 - y, mode="full")
        idx = int(np.argmax(corr))
        g = shift_grad(y, n, idx - (n - 1)) * dx
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        gap = max(float(np.max(corr)) * dx - 0.375, 1e-7)
        alpha = min(2.0 * gap / (gn * gn + 1e-12), 0.2 / (1.0 + 0.001 * it))
        h_new = project_feasible(y - alpha * g, n)
        h_p = h_c.copy()
        h_c = h_new
        if (it + 1) % 2000 == 0:
            c5_now = compute_c5(h_c)
            if c5_now < c5_b:
                c5_b = c5_now
                h_b = h_c.copy()
    return h_b, c5_b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Coarse-to-fine L-BFGS-B
    for T, mi in [(32, 2000), (8, 3000), (2, 4000), (0.5, 5000)]:
        rem = budget_s - (time.time() - t0)
        if rem < 100:
            break
        h_new = do_lbfgsb(h_best, n, T, min(100, 2*n-1), mi, 1e-14, t0, budget_s - 40)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 2: Binary coordinate descent
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        h_b, c5_b = binary_cd(h_best, n, 60, t0, t0 + rem - 60)
        if c5_b < c5_best:
            c5_best, h_best = c5_b, h_b
        # Re-optimize from binary
        h_new = do_lbfgsb(h_best, n, 2, 80, 3000, 1e-14, t0, budget_s - 40)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 3: Nesterov subgradient
    rem = budget_s - (time.time() - t0)
    if rem > 200:
        h_ng, c5_ng = nesterov_step(h_best, n, t0, t0 + rem - 150, 150000)
        if c5_ng < c5_best:
            c5_best, h_best = c5_ng, h_ng

    # Phase 4: SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h_new = do_slsqp(h_best, n, 1, 50, 3000, 1e-15, t0, budget_s - 30)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 5: Multi-restart
    for pi in range(25):
        rem = budget_s - (time.time() - t0)
        if rem < 15:
            break
        if pi % 4 == 0:
            h_p = h_best + rng.normal(0, 0.015 + 0.01 * (pi % 3), n)
        elif pi % 4 == 1:
            hb = (h_best > 0.5).astype(float)
            nf = max(2, n // 30)
            fi = rng.choice(n, size=nf, replace=False)
            hb[fi] = 1.0 - hb[fi]
            h_p = hb + rng.normal(0, 0.01, n)
        elif pi % 4 == 2:
            h_p = h_best.copy()
            bs = max(4, n // 14)
            bi = int(rng.integers(0, n // bs - 1))
            h_p[bi*bs:(bi+1)*bs] = 1.0 - h_p[bi*bs:(bi+1)*bs]
            h_p += rng.normal(0, 0.004, n)
        else:
            x = np.arange(n, dtype=float)
            pert = sum(0.015 * np.sin(2*np.pi*f*x/n + rng.uniform(0, 6.28)) for f in [3, 7, 19])
            h_p = h_best + pert + rng.normal(0, 0.008, n)
        h_p = project_feasible(h_p, n)
        h_new = do_lbfgsb(h_p, n, 2, 100, 2500, 1e-13, t0, budget_s - 10)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 6: Final fine L-BFGS-B
    for T in [0.3, 0.8, 1.5]:
        rem = budget_s - (time.time() - t0)
        if rem < 40:
            break
        h_new = do_lbfgsb(h_best, n, T, 60, 10000, 1e-16, t0, budget_s - 20)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 7: Try alternate resolution
    for n_try in [200, 350]:
        rem = budget_s - (time.time() - t0)
        if rem < 180:
            break
        x_old = np.linspace(0, 2.0, n, endpoint=False)
        x_new = np.linspace(0, 2.0, n_try, endpoint=False)
        h_t = np.interp(x_new, x_old, h_best)
        h_t = project_feasible(h_t, n_try)
        h_t = do_lbfgsb(h_t, n_try, 8, min(120, 2*n_try-1), 3000, 1e-13, t0, budget_s - 80)
        c5_t = compute_c5(h_t)
        if c5_t < c5_best:
            c5_best = c5_t
            h_best = do_lbfgsb(h_t, n_try, 1, 80, 4000, 1e-14, t0, budget_s - 40)
            n = n_try
            c5_best = min(c5_best, compute_c5(h_best))

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
