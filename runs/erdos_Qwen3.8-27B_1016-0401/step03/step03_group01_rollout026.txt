from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 3.0, y.max() + 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def softmax_obj_grad(h, n, T, kt):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mk = min(kt, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = np.max(topk)
    sh = T * (topk - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    obj = (mc + np.log(e.sum()) / T) * dx
    center = n - 1
    grad = np.zeros(n)
    for oi in range(mk):
        w = sw[oi]
        if w < 1e-12:
            continue
        k = idx[oi] - center
        if k >= 0:
            i1 = np.arange(max(0, 0), min(n, n - k))
            if len(i1) > 0:
                grad[i1] += w * (1.0 - h[i1 + k])
            i2 = np.arange(k, n)
            if len(i2) > 0:
                grad[i2] -= w * h[i2 - k]
        else:
            s = -k
            i1 = np.arange(s, n)
            if len(i1) > 0:
                grad[i1] += w * (1.0 - h[i1 - s])
            i2 = np.arange(0, n - s)
            if len(i2) > 0:
                grad[i2] -= w * h[i2 + s]
    grad *= dx
    return obj, grad

        h_best = h_y.copy()
    return h_best, c5_best

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < obj(h):

    except Exception:
        pass
    return h.copy()
