def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

# Global state for SLSQP callbacks
_G_N = 0
_G_TARGET = 0.0
_G_T = 1.0

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-12:
            continue
        grad += w * shift_grad_raw(x, _G_N, oi - center)
    return grad * dx

def _g_cons_fun(x):
    return x.sum() - _G_TARGET

def _g_cons_jac(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter, ftol):
    global _G_N, _G_TARGET, _G_T
    _G_N = n
    _G_TARGET = n / 2.0
    _G_T = float(T)

    except Exception:

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-11:
            continue
        grad += w * shift_grad_raw(x, _G_N, oi - center)
    return grad * dx

def _g_cons_fun(x):
    return x.sum() - _G_N / 2.0

def _g_cons_jac(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter, ftol):
    global _G_N, _G_T
    _G_N = n
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_cons_fun, 'jac': _g_cons_jac}]
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if getattr(res, 'x', None) is not None:
            return res.x
    except Exception:
        pass
    return h.copy()

def nesterov_phase(h_init, n, t0, time_budget, max_iters=400000):
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    base_alpha = 0.12
    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        beta = float(it) / (it + 4.0) if it > 0 else 0.0
        h_y = np.clip(h_sg + beta * (h_sg - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top_idx = np.argsort(corr)[-6:]
        mv = corr[top_idx].max()
        gap = mv - corr[top_idx].min()
        K_eff = max(1, int(np.sum(corr[top_idx] > mv - max(0.02, gap * 0.6))))
        top_idx = top_idx[-K_eff:]
        mv = corr[top_idx].max()
        w = np.exp(6.0 * (corr[top_idx] - mv))
        w /= w.sum()
        g = np.zeros(n)
        for wk, ik in zip(w, top_idx):
            g += wk * shift_grad_raw(h_y, n, ik - center)
        g *= dx
        alpha = base_alpha / (1.0 + 0.003 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_sg.copy()
        h_sg = h_new
        if (it + 1) % 12000 == 0:
            cv = compute_c5(h_sg)
            if cv < c5_best:
                c5_best = cv
                h_best = h_sg.copy()
    cv = compute_c5(h_sg)
    if cv < c5_best:
        c5_best = cv
        h_best = h_sg.copy()
    return h_best, c5_best

def coord_descent(h, n, rng, t0, time_budget):
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    deltas = [0.02, 0.05, 0.1, 0.2]
    order = rng.permutation(n)
    pass_count = 0
    while time.time() - t0 < time_budget:
        pass_count += 1
        improved = 0
        for i in order:
            if time.time() - t0 > time_budget:
                break
            vi = h_cur[i]
            for d in deltas:
                for sign in (-1, 1):
                    nv = vi + sign * d
                    if nv < 0.0 or nv > 1.0:
                        continue
                    h_try = h_cur.copy()
                    h_try[i] = nv
                    h_try = project_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < c5_cur - 1e-9:
                        h_cur = h_try
                        c5_cur = c5_try
                        improved += 1
                        break
                else:
                    continue
                break
        if improved == 0 and pass_count > 1:
            break
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None and len(initial_h_values) > 10:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse-to-fine
    temps = [64, 16, 4, 1.5, 0.7]
    for T in temps:
        if time.time() - t0 > budget_s - 700:
            break
        h_new = do_slsqp(h_best, n, T, 4000, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Adaptive Nesterov subgradient
    remaining = budget_s - (time.time() - t0)
    nest_budget = min(remaining * 0.35, 350)
    h_sg, c5_sg = nesterov_phase(h_best, n, t0, t0 + nest_budget, max_iters=500000)
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg

    # Phase 3: SLSQP fine polish
    temps_fine = [0.5, 0.25, 0.12, 0.06]
    for T in temps_fine:
        if time.time() - t0 > budget_s - 300:
            break
        h_new = do_slsqp(h_best, n, T, 6000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Coordinate descent
    if time.time() - t0 < budget_s - 150:
        cd_budget = min(budget_s - (time.time() - t0) - 80, 200)
        h_cd, c5_cd = coord_descent(h_best, n, rng, t0, t0 + cd_budget)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    # Phase 5: Perturbation restarts
    restarts = 8
    for pi in range(restarts):
        if time.time() - t0 > budget_s - 60:
            break
        sigma = 0.005 + 0.005 * (pi % 4)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_new = do_slsqp(h_p, n, 1.5, 3000, 1e-13)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Final SLSQP polish
    if time.time() - t0 < budget_s - 15:
        h_new = do_slsqp(h_best, n, 0.08, 8000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    h_best = np.clip(h_best, 0.0, 1.0)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
