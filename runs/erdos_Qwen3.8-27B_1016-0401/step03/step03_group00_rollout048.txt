    s_vals = k - (n - 1)  # actual shifts
    
    for idx, s in enumerate(s_vals):
        # h[i] appears in this correlation if:
        # As first factor: i + s is valid, i.e., 0 <= i+s < n
        # As second factor: i - s is valid, i.e., 0 <= i-s < n
        coeff = 0.0
        if 0 <= i + s < n:
            coeff += (1.0 - h[i + s])
        if 0 <= i - s < n:
            coeff -= h[i - s]
        c[idx] = coeff
    
    d = corr0 - c * h[i]
    
    # Now find z in [0,1] that minimizes max_k (c[k]*z + d[k])
    # This is a 1D convex piecewise linear minimization
    # Simple approach: ternary search (doesn't work for PWL) 
    # Better: find the point where 0 enters the subgradient
    
    # Evaluate at endpoints
    f0 = np.max(c * 0 + d)  # max(d)
    f1 = np.max(c * 1 + d)  # max(c+d)
    
    # Grid search first (coarse), then refine
    z_grid = np.linspace(0, 1, 100)
    f_vals = np.max(c[:, None] * z_grid[None, :] + d[:, None], axis=0)
    z_best = z_grid[np.argmin(f_vals)]
    
    # Refine: find the exact breakpoint
    # At z_best, the active set is all k where c[k]*z_best + d[k] ≈ max
    # The minimum is at z where the largest and smallest slopes in the active set straddle 0
    
    # Actually for a convex PWL function, the min is where:
    # min{c[k] : c[k]*z + d[k] = f(z)} <= 0 <= max{c[k] : c[k]*z + d[k] = f(z)}
    
    # Let's just do a fine grid search
    z_fine = np.linspace(0, 1, 1000)
    f_fine = np.max(c[:, None] * z_fine[None, :] + d[:, None], axis=0)
    z_opt = z_fine[np.argmin(f_fine)]
    
    return z_opt

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, "full")) * 2.0 / n)

def project(y, n):
    t = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(y - mid, 0.0, 1.0).sum() > t:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, s):
    g = np.zeros(n)
    j = np.arange(n)
    if s >= 0:
        if s == 0:
            g[:] += (1.0 - h)
            g[:] -= h
        else:
            m1 = j < (n - s)
            g[m1] += (1.0 - h[j[m1] + s])
            m2 = j >= s
            g[m2] -= h[j[m2] - s]
    else:
        ss = -s
        m1 = j < (n - ss)
        g[m1] += (1.0 - h[j[m1] + ss])
        m2 = j >= ss
        g[m2] -= h[j[m2] - ss]
    return g

_S = {}

def _obj(x):
    n = _S['n']
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_S['tk']:]
    c = corr[idx]
    m = c.max()
    T = _S['T']
    e = np.exp(T * (c - m))
    return (m + np.log(e.sum()) / T) * (2.0 / n)

def _jac(x):
    n = _S['n']
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_S['tk']:]
    c = corr[idx]
    m = c.max()
    T = _S['T']
    e = np.exp(T * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, idx):
        if wi > 1e-10:
            grad += wi * shift_grad(x, n, ki - (n - 1))
    return grad * (2.0 / n)

def _cf(x):
    return x.sum() - _S['tgt']

def _cj(x):
    return np.ones(_S['n'])

def slsqp_step(h, n, T, tk, maxiter, ftol):
    _S['n'] = n
    _S['T'] = float(T)
    _S['tk'] = min(tk, 2 * n - 1)
    _S['tgt'] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, method='SLSQP', jac=_jac, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def coord_descent(h, n, n_grid=300, max_rounds=30):
    dx = 2.0 / n
    z_grid = np.linspace(0.0, 1.0, n_grid)
    s_vals = np.arange(2 * n - 1) - (n - 1)
    corr = np.correlate(h, 1.0 - h, "full").copy()
    h_work = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        improved = 0
        for i in range(n):
            c = np.zeros(2 * n - 1)
            v1 = (i + s_vals >= 0) & (i + s_vals < n)
            v2 = (i - s_vals >= 0) & (i - s_vals < n)
            idx1 = np.where(v1)[0]
            idx2 = np.where(v2)[0]
            c[idx1] = 1.0 - h_work[i + s_vals[idx1]]
            c[idx2] -= h_work[i - s_vals[idx2]]
            d = corr - c * h_work[i]
            f_vals = np.max(c[:, None] * z_grid[None, :] + d[:, None], axis=0)
            z_opt = z_grid[np.argmin(f_vals)]
            delta = z_opt - h_work[i]
            if abs(delta) > 1e-8:
                corr += c * delta
                h_work[i] = z_opt
                improved += 1
        if improved == 0:
            break
    h_work = project(h_work, n)
    return h_work

def nesterov_step(h_init, n, t_end, rng, top_k=10, max_iter=200000):
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5a = compute_c5(h_init)
    ha = h_init.copy()
    for it in range(max_iter):
        if time.time() > t_end:
            break
        if it == 0:
            h_y = h_cur.copy()
        else:
            beta = float(it) / (it + 1.5)
            h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for ki in idx:
            g += shift_grad(h_y, n, ki - (n - 1))
        g *= dx / top_k
        alpha = 1.2 / (1.0 + 0.001 * it)
        h_new = project(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 5000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < c5a:
                c5a, ha = c1, h_cur.copy()
    return ha, c5a

def perturb(h, n, rng, mode):

        sig = 0.003 + 0.004 * (mode // 4) % 4
        return project(h + rng.normal(0, sig, n), n)
    elif mode == 1:
        hp = h.copy()
        lb = int(rng.integers(0, max(1, n - 30)))
        ln = int(rng.integers(5, 30))
        hp[lb:lb+ln] = 1.0 - hp[lb:lb+ln]
        return project(hp, n)
    elif mode == 2:
        hp = h.copy()
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        hp[i], hp[j] = hp[j], hp[i]
        return project(hp, n)
    else:
        sig = 0.008 + 0.003 * (mode // 4) % 3
        hp = h + rng.normal(0, sig, n)
        hp[int(rng.integers(0, n))] = 1.0
        return project(hp, n)
