def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def coord_descent(h, n, t_end, max_sweeps=2000, top_k=15):
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    shifts = np.arange(n_corr) - center
    c5_best = compute_c5(h)
    h_best = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full").copy()
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        any_change = False
        for i in range(n):
            hi_i = h[i]
            ip = i + shifts
            im = i - shifts
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            D = np.zeros(n_corr)
            D[vp] = 1.0 - h[ip[vp]]
            D[vm] -= h[im[vm]]
            
            # Find optimal t minimizing max_k (corr[k] + (t-hi_i)*D[k])
            best_t = hi_i
            best_f = np.max(corr)
            
            f0 = np.max(corr - hi_i * D)
            f1 = np.max(corr + (1.0 - hi_i) * D)
            if f0 < best_f:
                best_f, best_t = f0, 0.0
            if f1 < best_f:
                best_f, best_t = f1, 1.0
            
            # Breakpoints
            top_idx = np.argsort(corr)[-top_k:]
            for a in range(len(top_idx)):
                for b in range(a + 1, len(top_idx)):
                    ka, kb = int(top_idx[a]), int(top_idx[b])
                    dk = D[ka] - D[kb]
                    if abs(dk) < 1e-14:
                        continue
                    t_bp = hi_i + (corr[kb] - corr[ka]) / dk
                    if 0.0 < t_bp < 1.0:
                        f_bp = np.max(corr + (t_bp - hi_i) * D)
                        if f_bp < best_f:
                            best_f, best_t = f_bp, t_bp
            
            if abs(best_t - hi_i) > 1e-10:
                corr += (best_t - hi_i) * D
                h[i] = best_t
                any_change = True
        
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = compute_c5(h)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()
        if not any_change:
            break
    return h_best, c5_best

def fista_opt(h, n, t_end, top_k=10, lr0=0.005):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    center = n - 1
    shifts = np.arange(n_corr) - center
    h_y = h.copy()
    h_z = h.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    t_mom = 1.0
    lr = lr0
    for it in range(200000):

            break
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argpartition(corr, -top_k)[-top_k:]

        for oi in idx:
            s = int(oi) - center
            ip = np.arange(n) + s
            im = np.arange(n) - s
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            g = np.zeros(n)
            g[vp] = 1.0 - h_y[ip[vp]]
            g[vm] -= h_y[im[vm]]
            grad += g

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0

            if c5 < c5_best:

        # Also try CD directly on interpolated
        hc3, cc3 = coord_descent(h_hi, n2, t0 + (time.time() - t0) + rem * 0.5, max_sweeps=1500, top_k=15)
        best_hi = min(cc2, cc3)
        best_hi_h = hc2 if cc2 <= cc3 else hc3
        if best_hi < c5_best:
            c5_best, h_best, n = best_hi, best_hi_h, n2
    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
