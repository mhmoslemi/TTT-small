def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def project_feasible(y, n):
    tgt = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > tgt: lo = mid
        else: hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def sgrad(h, n, sh):
    j = np.arange(n)
    g = np.zeros(n)
    if sh >= 0:
        m = j >= sh
        g[m] += 1.0 - h[j[m] - sh]
        m2 = j + sh < n
        g[m2] -= h[j[m2] + sh]
    else:
        s = -sh
        m = j + s < n
        g[m] += 1.0 - h[j[m] + s]
        m2 = j >= s
        g[m2] -= h[j[m2] - s]
    return g

_N, _T, _TG = 0, 1.0, 0.0

def _ob(x):
    c = np.correlate(x, 1.0 - x, mode="full")
    mx = c.max()
    e = np.exp(_T * (c - mx))
    return (mx + np.log(e.sum()) / _T) * (2.0 / _N)

def _jc(x):
    c = np.correlate(x, 1.0 - x, mode="full")
    mx = c.max()
    e = np.exp(_T * (c - mx))
    w = e / e.sum()
    g = np.zeros(_N)
    ctr = _N - 1
    for i in range(len(w)):
        if w[i] > 1e-12:
            g += w[i] * sgrad(x, _N, i - ctr)
    return g * (2.0 / _N)

def _cf(x): return x.sum() - _TG
def _cj(x): return np.ones(_N)

def slq(h, n, T, mi, ft):

                     options={'maxiter': mi, 'ftol': ft})
        return r.x
    except Exception:
        return h.copy()
