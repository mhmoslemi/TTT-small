    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', jac=_obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def active_descent(h, n, steps=25000, K=5, t0=None, budget_s=0):
    dx = 2.0 / n
    center = n - 1
    h_cur = h.copy()
    h_prev = h.copy()
    h_avg = h.copy()
    c5_best = compute_c5(h_cur)
    h_best = h_cur.copy()
    for it in range(steps):
        if t0 is not None and time.time() - t0 > budget_s:
            break
        beta = float(it) / (it + 1.5)
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = float(np.max(corr))
        if K <= 1:
            g = shift_grad(h_y, n, int(np.argmax(corr)) - center) * dx
        else:
            top = np.argpartition(corr, -K)[-K:]
            tv = corr[top]
            w = (tv / max(mc, 1e-15)) ** 10
            ws = w.sum()
            if ws < 1e-15:
                w = np.ones(K) / K
            else:
                w /= ws
            g = np.zeros(n)
            for kk in range(K):
                g += w[kk] * shift_grad(h_y, n, top[kk] - center)
            g *= dx
        alpha = 1.0 / (1.0 + 0.005 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / (it + 1.0)
        if (it + 1) % 5000 == 0:
            for cand in [h_avg.copy(), h_cur.copy()]:
                cand = project_feasible(cand, n)
                c = compute_c5(cand)
                if c < c5_best:
                    c5_best = c
                    h_best = cand
    return project_feasible(h_best, n), c5_best

def manual_line_search(h, n, steps=3000, alphas=None):
    dx = 2.0 / n
    center = n - 1
    if alphas is None:
        alphas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    h_cur = h.copy()
    c5_best = compute_c5(h_cur)
    h_best = h_cur.copy()
    stall = 0
    for it in range(steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        mc = float(np.max(corr))
        top_k = 3
        top = np.argpartition(corr, -top_k)[-top_k:]
        g = np.zeros(n)
        for kk in range(top_k):
            g += shift_grad(h_cur, n, top[kk] - center)
        g *= dx / top_k
        improved = False
        for a in alphas:
            h_try = project_feasible(h_cur - a * g, n)
            c_try = compute_c5(h_try)
            if c_try < c5_best - 1e-15:
                c5_best = c_try
                h_best = h_try
                h_cur = h_try
                improved = True
                break
        if not improved:
            stall += 1
            if stall > 200:
                break
        else:
            stall = 0
    return h_best, c5_best

def perturb_block(h, n, rng, strength=0.03):
    h_p = h.copy()
    nblocks = int(rng.integers(3, 9))
    for b in range(nblocks):
        bs = int(rng.integers(n // 20, max(n // 4, n // 20 + 1)))
        start = int(rng.integers(0, max(1, n - bs)))
        delta = rng.uniform(-strength, strength)
        h_p[start:start + bs] += delta
    return project_feasible(h_p, n)

def perturb_smooth(h, n, rng):
    h_p = h.copy()
    x = np.arange(n) * 2.0 / n
    f = rng.uniform(0.5, 5.0)
    ph = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.01, 0.04)
    h_p += amp * np.sin(2 * np.pi * f * x + ph)
    return project_feasible(h_p, n)

def perturb_noise(h, n, rng):
    return project_feasible(h + rng.normal(0, 0.008, n), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None and len(initial_h_values) > 0:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n_init = 280
        h_best = np.zeros(n_init)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    for T in [16, 32, 64, 128, 256]:
        if time.time() - t0 > budget_s * 0.30:
            break
        h_new = slsqp(h_best, n, T, maxiter=8000, ftol=1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    if time.time() - t0 < budget_s * 0.45:
        h_ad, c5_ad = active_descent(h_best, n, steps=25000, K=5, t0=t0, budget_s=budget_s * 0.45)
        if c5_ad < c5_best:
            c5_best = c5_ad
            h_best = h_ad

    for T in [64, 128, 256, 512]:
        if time.time() - t0 > budget_s * 0.55:
            break
        h_new = slsqp(h_best, n, T, maxiter=10000, ftol=1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    for pi in range(12):
        if time.time() - t0 > budget_s - 250:
            break
        mode = pi % 3
        if mode == 0:
            h_p = perturb_block(h_best, n, rng)
        elif mode == 1:
            h_p = perturb_smooth(h_best, n, rng)
        else:
            h_p = perturb_noise(h_best, n, rng)
        h_new = slsqp(h_p, n, 256, maxiter=6000, ftol=1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    if time.time() - t0 < budget_s - 200:
        for n2 in [n // 2, n * 2]:
            if n2 < 70 or n2 > 900:
                continue
            if time.time() - t0 > budget_s - 100:
                break
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
            h_alt = np.interp(x_new, x_old, h_best)
            h_alt = project_feasible(h_alt, n2)
            for T2 in [64, 256]:
                if time.time() - t0 > budget_s - 50:
                    break
                h_new = slsqp(h_alt, n2, T2, maxiter=6000, ftol=1e-15)
                h_new = project_feasible(h_new, n2)
                c5_alt = compute_c5(h_new)
                if c5_alt < c5_best:
                    c5_best = c5_alt
                    h_best = h_new
                    n = n2

    if time.time() - t0 < budget_s - 60:
        h_best, c5_m = manual_line_search(h_best, n, steps=3000)
        c5_best = min(c5_best, c5_m)

    if time.time() - t0 < budget_s - 20:
        for T in [256, 512]:
            if time.time() - t0 > budget_s - 10:
                break
            h_new = slsqp(h_best, n, T, maxiter=8000, ftol=1e-16)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    if time.time() - t0 < budget_s - 10:
        h_best, c5_m2 = manual_line_search(h_best, n, steps=2000)
        c5_best = min(c5_best, c5_m2)

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
