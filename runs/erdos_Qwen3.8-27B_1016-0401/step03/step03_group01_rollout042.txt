def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    lo = y.min() - 2.0
    hi = y.max() + 2.0

def grad_single_lag(h, shift, n):
    """Gradient of f(h) = sum_i h[i]*(1-h[i+shift]) w.r.t. h"""
    j = np.arange(n)
    grad = np.zeros(n)

        grad[m] += (1.0 - h[j[m] - shift])
        m2 = (j + shift) < n

        m2 = j >= s2
        grad[m2] -= h[j[m2] - s2]
    return grad

    T = T_init
    center = n - 1

            grad += w[oi] * grad_single_lag(h, shift, n)

            h_best = h.copy()

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_values(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def grad_lag(h, shift, n):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m = j >= shift
        grad[m] += (1.0 - h[j[m] - shift])
        m2 = (j + shift) < n
        grad[m2] -= h[j[m2] + shift]
    else:
        s2 = -shift
        m = (j + s2) < n
        grad[m] += (1.0 - h[j[m] + s2])
        m2 = j >= s2
        grad[m2] -= h[j[m2] - s2]
    return grad

def adam_phase(h, n, t_end, lr, T_start, T_end, focus_k, max_iter, restart_iters=3000):
    dx = 2.0 / n
    h_best = h.copy()
    c5_best = compute_c5(h)
    center = n - 1
    m_amt = np.zeros(n)
    v_amt = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t_a = 0
    stall = 0

    for it in range(max_iter):
        if time.time() > t_end:
            break
        corr = corr_values(h)
        k = min(focus_k, len(corr))
        idx = np.argpartition(corr, -k)[-k:]

        frac = it / max(1, max_iter)
        T = T_start * (T_end / T_start) ** frac
        T = max(T, 0.3)

        topk = corr[idx]
        sh = T * (topk - topk.max())
        e = np.exp(sh)
        w = e / e.sum()

        grad = np.zeros(n)
        for oi in range(k):
            if w[oi] < 1e-12:
                continue
            shift = idx[oi] - center
            grad += w[oi] * grad_lag(h, shift, n)
        grad *= dx

        t_a += 1
        m_amt = b1 * m_amt + (1.0 - b1) * grad
        v_amt = b2 * v_amt + (1.0 - b2) * grad * grad
        m_h = m_amt / (1.0 - b1 ** t_a)
        v_h = v_amt / (1.0 - b2 ** t_a)

        h_new = h - lr * m_h / (np.sqrt(v_h) + eps)
        h = project_feasible(h_new, n)

        if it % 100 == 0:
            c5 = compute_c5(h)
            if c5 < c5_best - 1e-10:
                c5_best = c5
                h_best = h.copy()
                stall = 0
            else:
                stall += 100
            if stall > restart_iters:
                h = h_best.copy()
                m_amt = np.zeros(n)
                v_amt = np.zeros(n)
                t_a = 0
                stall = 0

    c5 = compute_c5(h)
    if c5 < c5_best:
        c5_best = c5
        h_best = h.copy()
    return h_best, c5_best

def binary_push(h, n, thresh_lo=0.2, thresh_hi=0.8):
    h_b = h.copy()
    h_b[h_b >= thresh_hi] = 1.0
    h_b[h_b <= thresh_lo] = 0.0
    return project_feasible(h_b, n)

def local_flip_search(h, n, t_end, rng, max_rounds=30, flips_per_round=300):
    h_best = h.copy()
    c5_best = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(flips_per_round):
            if time.time() > t_end:
                break
            i = int(rng.integers(0, n))
            h_try = h_best.copy()
            h_try[i] = 1.0 - h_try[i]
            h_try = project_feasible(h_try, n)
            c5 = compute_c5(h_try)
            if c5 < c5_best - 1e-11:
                c5_best = c5
                h_best = h_try
                improved = True
        if not improved:
            break
    return h_best, c5_best

def nudge_worst_lag(h, n, t_end, rng, n_steps=8000):
    """Directly nudge to reduce the worst lag."""
    h_best = h.copy()
    c5_best = compute_c5(h)
    center = n - 1
    dx = 2.0 / n
    lr = 0.003
    m_amt = np.zeros(n)
    v_amt = np.zeros(n)
    t_a = 0
    b1, b2, eps = 0.8, 0.999, 1e-8

    for it in range(n_steps):
        if time.time() > t_end:
            break
        corr = corr_values(h)
        widx = int(np.argmax(corr))
        shift = widx - center
        grad = grad_lag(h, shift, n) * dx
        # Add small noise to escape flat regions
        if it % 500 == 0:
            grad += 0.01 * rng.standard_normal(n) * dx
        t_a += 1
        m_amt = b1 * m_amt + (1.0 - b1) * grad
        v_amt = b2 * v_amt + (1.0 - b2) * grad * grad
        m_h = m_amt / (1.0 - b1 ** t_a)
        v_h = v_amt / (1.0 - b2 ** t_a)
        h_new = h - lr * m_h / (np.sqrt(v_h) + eps)
        h = project_feasible(h_new, n)
        if it % 200 == 0:
            c5 = compute_c5(h)
            if c5 < c5_best:
                c5_best = c5
                h_best = h.copy()
    c5 = compute_c5(h)
    if c5 < c5_best:
        c5_best = c5
        h_best = h.copy()
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n_init = 280
        h_best = np.zeros(n_init)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Adam broad focus
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h_new, c5_new = adam_phase(h_best, n, t0 + rem * 0.28, 0.012, 25.0, 3.0, 5, 25000)
        if c5_new < c5_best:
            c5_best, h_best = c5_new, h_new

    # Phase 2: Binary push + focused Adam
    h_bp = binary_push(h_best, n, 0.3, 0.7)
    c5_bp = compute_c5(h_bp)
    if c5_bp < c5_best:
        c5_best, h_best = c5_bp, h_bp

    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h_new, c5_new = adam_phase(h_best, n, t0 + rem * 0.22, 0.007, 12.0, 1.0, 3, 20000)
        if c5_new < c5_best:
            c5_best, h_best = c5_new, h_new

    # Phase 3: Nudge worst lag
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_new, c5_new = nudge_worst_lag(h_best, n, t0 + rem * 0.18, rng, 12000)
        if c5_new < c5_best:
            c5_best, h_best = c5_new, h_new

    # Phase 4: Local flip search
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_new, c5_new = local_flip_search(h_best, n, t0 + rem * 0.2, rng)
        if c5_new < c5_best:
            c5_best, h_best = c5_new, h_new

    # Phase 5: Very sharp Adam (single lag focus)
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        h_new, c5_new = adam_phase(h_best, n, t0 + rem * 0.25, 0.004, 60.0, 0.5, 1, 25000)
        if c5_new < c5_best:
            c5_best, h_best = c5_new, h_new

    # Phase 6: Try higher resolution
    rem = budget_s - (time.time() - t0)
    if rem > 130:
        for n_try in [n + 80, 400, 240]:
            if time.time() - t0 > budget_s - 50:
                break
            if n_try == n or n_try < 100:
                continue
            t_sub = min(t0 + (budget_s - (time.time() - t0)) * 0.12, budget_s - 20)
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_hi = np.interp(x_new, x_old, h_best)
            h_hi = project_feasible(h_hi, n_try)
            h_new, c5_new = adam_phase(h_hi, n_try, t_sub, 0.008, 20.0, 1.0, 3, 8000)
            h_new, c5_new2 = adam_phase(h_new, n_try, t_sub, 0.004, 30.0, 0.5, 1, 5000)
            c5_new = min(c5_new, c5_new2)
            if c5_new2 < c5_new:
                h_new = h_new
            if c5_new < c5_best:
                c5_best, h_best, n = c5_new, h_new, n_try

    # Phase 7: Final Adam polish
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        h_new, c5_new = adam_phase(h_best, n, t0 + rem * 0.9, 0.002, 30.0, 0.5, 2, 10000)
        if c5_new < c5_best:
            c5_best, h_best = c5_new, h_new

    # Phase 8: Final flip search
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        h_new, c5_new = local_flip_search(h_best, n, t0 + rem, rng, 10, 150)
        if c5_new < c5_best:
            c5_best, h_best = c5_new, h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
