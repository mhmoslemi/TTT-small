        m2 = j >= s
        grad[m2] -= h[j[m2] - s]

    corr, mc = corr_and_max(x)
    dx = 2.0 / n
    sh = T * (corr - mc)
    # Clip for numerical stability
    sh = np.clip(sh, -50, 50)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * dx

def lse_grad(x, n, T):

    dx = 2.0 / n

    sh = np.clip(sh, -50, 50)
    e = np.exp(sh)

    center = n - 1

    for oi in range(len(corr)):

        if w < 1e-12:

    elif mode == 1:
        f = 0.3 + 0.25 * (pi % 6)
        h_p = h + scale * 0.7 * np.cos(4.0 * np.pi * f * x + rng.uniform(0, 6.28))
    elif mode == 2:
        block = n // 9
        h_p = h.copy()
        for b in range(9):
            lo_i = b * block
            hi_i = min(lo_i + block, n)
            h_p[lo_i:hi_i] += scale * 0.5 * rng.uniform(-1.0, 1.0)
    elif mode == 3:
        h_p = h + rng.normal(0, scale * 0.5, n)
    else:
        idx = rng.choice(n, size=n // 10, replace=False)
        h_p = h.copy()
        h_p[idx] += rng.uniform(-scale, scale, len(idx))
    return h_p

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n = 280
        h_best = np.zeros(n)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse polish
    for T in [24.0, 12.0, 6.0]:
        if time.time() - t0 > budget_s - 550:
            break
        h_new = run_slsqp(h_best, n, T, 2500, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Nesterov with adaptive T and K
    h_new, c5_new = nesterov_adaptive(h_best, n, t0, budget_s - 400)
    if c5_new < c5_best:
        c5_best = c5_new
        h_best = h_new
    h_best = project_feasible(h_best, n)

    # Phase 3: SLSQP fine polish (multi-temperature)
    for T in [4.0, 2.0, 1.0]:
        if time.time() - t0 > budget_s - 180:
            break
        h_new = run_slsqp(h_best, n, T, 5000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Perturbation + quick SLSQP
    for trial in range(20):
        if time.time() - t0 > budget_s - 40:
            break
        scale = 0.015 / (1.0 + trial * 0.6)
        h_p = perturb_h(h_best, n, rng, trial, scale)
        h_p = project_feasible(h_p, n)
        T_p = 2.0 if trial < 10 else 1.0
        h_new = run_slsqp(h_p, n, T_p, 2500, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Alternate resolution
    elapsed = time.time() - t0
    if elapsed < budget_s - 150:
        for n2 in [360, 200]:
            if time.time() - t0 > budget_s - 80:
                break
            x_old = np.linspace(0.0, 2.0, n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
            h_alt = np.interp(x_new, x_old, h_best)
            h_alt = project_feasible(h_alt, n2)
            for T2 in [6.0, 2.0]:
                if time.time() - t0 > budget_s - 20:
                    break
                h_new = run_slsqp(h_alt, n2, T2, 3000, 1e-14)
                h_new = project_feasible(h_new, n2)
                c5_alt = compute_c5(h_new)
                if c5_alt < c5_best:
                    c5_best = c5_alt
                    h_best = h_new
                    n = n2

    # Phase 6: Final tight polish
    if time.time() - t0 < budget_s - 5:
        h_new = run_slsqp(h_best, n, 1.0, 4000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
