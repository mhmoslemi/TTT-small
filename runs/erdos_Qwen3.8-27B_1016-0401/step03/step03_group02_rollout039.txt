def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(3000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                if a <= 0:
                    continue
                h[i] += a
                d -= a
            else:
                s = min(-d, h[i])
                if s <= 0:
                    continue
                h[i] -= s
                d += s
            if abs(d) < 1e-14:
                break
    return h

def _lag_g(g, x, o, n, wi):
    if o == 0:
        g += wi * (1.0 - 2.0 * x)
    elif o > 0:
        g[o:] += wi * (1.0 - x[:n - o])
        g[:n - o] -= wi * x[o:]
    else:
        g[:n + o] += wi * (1.0 - x[-o:])
        g[-o:] -= wi * x[:n + o]

    return float(np.dot(w, np.correlate(x, 1.0 - x, mode="full")) * 2.0 / n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(3000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                if a <= 0: continue
                h[i] += a; d -= a
            else:
                s = min(-d, h[i])
                if s <= 0: continue
                h[i] -= s; d += s
            if abs(d) < 1e-14: break
    return h

def _lag_g(g, x, o, n, wi):
    if o == 0:
        g += wi * (1.0 - 2.0 * x)
    elif o > 0:
        g[o:] += wi * (1.0 - x[:n-o])
        g[:n-o] -= wi * x[o:]
    else:
        g[:n+o] += wi * (1.0 - x[-o:])
        g[-o:] -= wi * x[:n+o]

def rw_obj(x, w, n):
    return float(np.dot(w, np.correlate(x, 1.0-x, mode="full")) * 2.0/n)

def rw_grad(x, w, n):
    g = np.zeros(n); ctr = n-1; dx = 2.0/n
    for i in range(len(w)):
        wi = w[i]
        if wi > 1e-14:
            _lag_g(g, x, i-ctr, n, wi)
    return g * dx

def lp_obj(x, p, n):
    c = np.correlate(x, 1.0-x, mode="full")
    return float(np.sum(c**p)**(1.0/p) * 2.0/n)

def lp_grad(x, p, n):
    c = np.correlate(x, 1.0-x, mode="full")
    s = np.sum(c**p)
    if s < 1e-30: return np.zeros(n)
    sc = s**(1.0/p - 1.0) * 2.0/n
    g = np.zeros(n); ctr = n-1
    for i in range(2*n-1):
        ci = c[i]
        if ci > 1e-14:
            wi = ci**(p-1) * sc
            if wi > 1e-16:
                _lag_g(g, x, i-ctr, n, wi)
    return g

def _eq(x, t): return x.sum() - t
def _eqj(x): return np.ones(len(x))

def slsqp_solve(obj, grad, h0, oargs, tgt, mi=800, ft=1e-18):
    n = len(h0)
    r = minimize(obj, h0, args=oargs, method='SLSQP', jac=grad,
                 bounds=[(0.0,1.0)]*n,
                 constraints=[dict(type='eq', fun=_eq, jac=_eqj, args=(tgt,))],
                 options={'maxiter': mi, 'ftol': ft})
    return np.clip(r.x, 0.0, 1.0)

def mk_w(corr, shp=4.0):
    mx = np.max(corr)
    if mx < 1e-15: return np.ones(len(corr))/len(corr)
    w = (corr/mx)**shp; w /= w.sum()
    return w

def coord_refine(h, tgt, ns=3):
    n = len(h); c5 = compute_c5(h)
    dts = [0.002,-0.002,0.008,-0.008,0.02,-0.02]
    for _ in range(ns):
        imp = False
        for j in range(n):
            if h[j] < 0.005 or h[j] > 0.995: continue
            base = h.copy()
            for dt in dts:
                nv = base[j]+dt
                if nv < 0 or nv > 1: continue
                tr = base.copy(); tr[j] = nv
                tr = fix_constraint(tr, tgt)
                cn = compute_c5(tr)
                if cn < c5 - 1e-15:
                    h = tr; c5 = cn; imp = True; break
        if not imp: break
    return h, c5

def bin_push(h, tgt):
    n = len(h); c5 = compute_c5(h); go = True
    while go:
        go = False
        for j in range(n):
            for v in (0.0, 1.0):
                if abs(h[j]-v) < 1e-9: continue
                tr = h.copy(); tr[j] = v; tr = fix_constraint(tr, tgt)
                cn = compute_c5(tr)
                if cn < c5 - 1e-15:
                    h = tr; c5 = cn; go = True; break
    return h, c5

def sa_bin(h, tgt, tb, rng):
    n = len(h)
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    h = fix_constraint(h, tgt)
    c5c = compute_c5(h); c5b = c5c; hb = h.copy()
    te = time.time() + tb

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(3000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                if a <= 0:
                    continue
                h[i] += a
                d -= a
            else:
                s = min(-d, h[i])
                if s <= 0:
                    continue
                h[i] -= s
                d += s
            if abs(d) < 1e-14:
                break

    if o == 0:
        g += wi * (1.0 - 2.0 * x)
    elif o > 0:
        g[o:] += wi * (1.0 - x[:n - o])
        g[:n - o] -= wi * x[o:]
    else:
        g[:n + o] += wi * (1.0 - x[-o:])

    return float(np.dot(w, np.correlate(x, 1.0 - x, mode="full")) * 2.0 / n)

        wi = w[i]
        if wi > 1e-14:
            _lag_g(g, x, i - ctr, n, wi)
    return g * dx
