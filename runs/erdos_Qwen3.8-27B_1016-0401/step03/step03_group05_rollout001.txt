    h2 = h + strength * (h - mid)
    return project_feasible(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Quick SLSQP polish on initial
    if time.time() - t0 < budget_s - 350:
        h_new = slsqp_smooth(h_best, n, 10, 2000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Nesterov main optimization
    if time.time() - t0 < budget_s - 250:
        h_sg, c5_sg = nesterov_opt(h_best, n, t0, budget_s - 250, 500000)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg
        h_best = project_feasible(h_best, n)

    # Phase 3: Binarize + re-optimize (exploit step structure)
    if time.time() - t0 < budget_s - 200:
        for st in [0.05, 0.1, 0.2]:
            if time.time() - t0 > budget_s - 180:
                break
            h_p = binarize_push(h_best, n, st)
            h_new = slsqp_topk(h_p, n, 5, 12, 3000)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 4: Multi-T SLSQP fine polish
    if time.time() - t0 < budget_s - 120:
        for T in [6, 4, 3, 2]:
            if time.time() - t0 > budget_s - 100:
                break
            h_new = slsqp_smooth(h_best, n, T, 5000)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 5: Top-K SLSQP with various K
    if time.time() - t0 < budget_s - 80:
        for K, pw in [(2, 10), (3, 12), (5, 10), (8, 8)]:
            if time.time() - t0 > budget_s - 60:
                break
            h_new = slsqp_topk(h_best, n, K, pw, 4000)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 6: Small perturbations + SLSQP restart
    if time.time() - t0 < budget_s - 40:
        for pi in range(15):
            if time.time() - t0 > budget_s - 30:
                break
            x = np.arange(n) * 2.0 / n
            freq = rng.uniform(0.5, 3.0)
            h_p = h_best + rng.normal(0, 0.004, n)
            h_p += 0.003 * np.sin(2 * np.pi * freq * x + rng.uniform(0, 6.28))
            h_p = project_feasible(h_p, n)
            h_new = slsqp_topk(h_p, n, 5, 10, 2500)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 7: Try coarser resolution
    if time.time() - t0 < budget_s - 150:
        n2 = n // 2
        if n2 >= 70:
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0, 2.0, n2, endpoint=False)
            h_alt = np.interp(x_new, x_old, h_best)
            h_alt = project_feasible(h_alt, n2)
            c5_alt = compute_c5(h_alt)
            h_alt_best = h_alt.copy()
            for K2 in [3, 5, 8]:
                if time.time() - t0 > budget_s - 80:
                    break
                h_new = slsqp_topk(h_alt_best, n2, K2, 10, 3000)
                h_new = project_feasible(h_new, n2)
                c5_alt = compute_c5(h_new)
                if c5_alt < c5_best:
                    c5_best = c5_alt
                    h_best = h_new
                    n = n2
                    h_alt_best = h_new
            # Try smooth SLSQP too
            if time.time() - t0 < budget_s - 60:
                h_new = slsqp_smooth(h_alt_best, n2, 3, 4000)
                h_new = project_feasible(h_new, n2)
                c5_alt = compute_c5(h_new)
                if c5_alt < c5_best:
                    c5_best = c5_alt
                    h_best = h_new
                    n = n2

    # Final refinement
    if time.time() - t0 < budget_s - 10:
        h_new = slsqp_topk(h_best, n, 3, 10, 3000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
