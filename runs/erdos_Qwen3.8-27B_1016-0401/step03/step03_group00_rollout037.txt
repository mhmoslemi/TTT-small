def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

def all_shift_grads(h, n, shifts):

def slsqp_optimize(h, n, T, top_k, maxiter, ftol):
    """SLSQP with soft-max objective."""
    N = n
    TGT = n / 2.0
    T_val = float(T)
    K = min(top_k, 2 * n - 1)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(corr)[-K:]
        c = corr[idx]
        m = c.max()
        e = np.exp(T_val * (c - m))
        return (m + np.log(e.sum()) / T_val) * (2.0 / N)
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(corr)[-K:]
        c = corr[idx]
        m = c.max()
        e = np.exp(T_val * (c - m))
        w = e / e.sum()
        grad = np.zeros(N)
        for wi, ci in zip(w, idx):
            if wi > 1e-10:
                grad += wi * shift_grad(x, N, ci - (N - 1))
        return grad * (2.0 / N)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - TGT, 
             'jac': lambda x: np.ones(N)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def shift_grad(h, n, shift):

    g = np.zeros(n)

        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]

        m2 = j >= s

    corr = np.correlate(h, 1.0 - h, mode="full")

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

_G_T = 1.0

_G_LAM = 0.0

    idx = np.argsort(corr)[-_G_K:]

    val = m + np.log(e.sum()) / _G_T
