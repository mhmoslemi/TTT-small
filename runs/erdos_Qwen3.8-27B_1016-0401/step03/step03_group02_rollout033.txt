def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def eq_fun(x):
    return x.sum() - _TARGET

def eq_jac(x):
    return np.ones(len(x))

def lag_obj(x, lag_idx):
    n = len(x)
    return float(np.correlate(x, 1.0 - x, mode="full")[lag_idx] * (2.0 / n))

def lag_grad(x, lag_idx):
    n = len(x)
    o = lag_idx - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * x
    elif o > 0:
        g[o:] += 1.0 - x[:n - o]
        g[:n - o] -= x[o:]
    else:
        p = -o
        g[:n - p] += 1.0 - x[p:]
        g[p:] -= x[:n - p]
    return g * dx

def slsqp_one(h, lag_idx, bounds, maxiter=800, ftol=1e-18):
    global _TARGET
    _TARGET = len(h) / 2.0
    cons = [dict(type='eq', fun=eq_fun, jac=eq_jac)]
    try:
        res = minimize(lag_obj, h.copy(), args=(lag_idx,), method='SLSQP',
                       jac=lag_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def binary_push(h, thresh=0.3):
    n = len(h)
    target = n / 2.0
    c5 = compute_c5(h)
    changed = True
    while changed:
        changed = False
        for j in range(n):
            if h[j] > thresh or h[j] < 1.0 - thresh:
                trial = h.copy()
                trial[j] = 1.0 if h[j] > 0.5 else 0.0
                trial = fix_constraint(trial, target)
                c5n = compute_c5(trial)
                if c5n < c5 - 1e-15:
                    h = trial
                    c5 = c5n
                    changed = True
    return h, c5

def pair_flip(h, sweeps=5):
    n = len(h)
    target = n / 2.0
    c5 = compute_c5(h)
    for _ in range(sweeps):
        imp = False
        zi = np.where(h < 0.01)[0]
        oi = np.where(h > 0.99)[0]
        if len(zi) == 0 or len(oi) == 0:
            break
        for i in zi:
            for j in oi:
                t = h.copy()
                t[i] = 1.0
                t[j] = 0.0
                cn = compute_c5(t)
                if cn < c5 - 1e-14:
                    h = t
                    c5 = cn
                    imp = True
                    zi = np.where(h < 0.01)[0]
                    oi = np.where(h > 0.99)[0]
        if not imp:
            break
    return h, c5

def anneal_binary(h, steps=500, tstart=0.04, tstop=0.0005, seed=42):
    n = len(h)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    c5c = compute_c5(h)
    c5b = c5c
    hb = h.copy()
    for s in range(steps):
        T = tstart * (tstop / tstart) ** (s / float(steps))
        zi = np.where(h < 0.5)[0]
        oi = np.where(h > 0.5)[0]
        if len(zi) == 0 or len(oi) == 0:
            break
        i = zi[rng.integers(len(zi))]
        j = oi[rng.integers(len(oi))]
        t = h.copy()
        t[i] = 1.0
        t[j] = 0.0
        t = fix_constraint(t, target)
        cn = compute_c5(t)
        d = cn - c5c
        if d < 0 or rng.random() < np.exp(-max(d, -5.0) / max(T, 1e-10)):
            h = t
            c5c = cn
            if c5c < c5b:
                c5b = c5c
                hb = h.copy()
    return hb, c5b

def coord_fine(h, sweeps=3):
    n = len(h)
    target = n / 2.0
    c5 = compute_c5(h)
    steps = [0.01, -0.01, 0.005, -0.005, 0.02, -0.02, 0.003, -0.003]
    for _ in range(sweeps):
        imp = False
        for j in range(n):
            for dt in steps:
                t = h.copy()
                t[j] = np.clip(t[j] + dt, 0.0, 1.0)
                t = fix_constraint(t, target)
                cn = compute_c5(t)
                if cn < c5 - 1e-15:
                    h = t
                    c5 = cn
                    imp = True
                    break
        if not imp:
            break
    return h, c5

def kick_perturb(h, nkicks, seed):
    n = len(h)
    target = n / 2.0
    rng = np.random.default_rng(seed)
    h = h.copy()
    for _ in range(nkicks):
        zi = np.where(h < 0.5)[0]
        oi = np.where(h > 0.5)[0]
        if len(zi) == 0 or len(oi) == 0:
            break
        i = zi[rng.integers(len(zi))]
        j = oi[rng.integers(len(oi))]
        h[i] = 1.0
        h[j] = 0.0
    h = fix_constraint(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global _TARGET
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(340)
        h[::2] = 1.0
    n = len(h)
    target = n / 2.0
    _TARGET = target
    h = fix_constraint(h, target)
    c5b = compute_c5(h)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Binary push + pair flip local search (15%)
    end1 = t0 + budget_s * 0.15
    h, c5 = binary_push(h, 0.3)
    if c5 < c5b:
        c5b = c5
    h, c5 = pair_flip(h, 8)
    if c5 < c5b:
        c5b = c5

    # Phase 2: SLSQP cycling on top lags + inner binary search (30%)
    end2 = t0 + budget_s * 0.45
    for outer in range(50):
        if time.time() > end2:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top = np.argsort(corr)[-4:]
        for lag in top:
            if time.time() > end2:
                break
            hn = slsqp_one(h, int(lag), bounds, 700, 1e-18)
            hn = fix_constraint(hn, target)
            cn = compute_c5(hn)
            if cn < c5b - 1e-16:
                c5b = cn
                h = hn
        if time.time() < end2 - 3:
            h, c5 = binary_push(h, 0.3)
            if c5 < c5b:
                c5b = c5
            h, c5 = pair_flip(h, 3)
            if c5 < c5b:
                c5b = c5

    # Phase 3: Annealing cycles with kick-perturb + local search (20%)
    end3 = t0 + budget_s * 0.65
    for r in range(6):
        if time.time() > end3:
            break
        hk = kick_perturb(h, 4 + r * 2, seed + r * 7919)
        ha, ca = pair_flip(hk, 3)
        ha2, ca2 = anneal_binary(ha, 400, 0.03, 0.0008, seed + r * 313)
        if ca2 < ca:
            ha, ca = ha2, ca2
        ha, ca = pair_flip(ha, 4)
        if ca < c5b:
            c5b = ca
            h = ha
        if time.time() < end3 - 8:
            corr = np.correlate(h, 1.0 - h, mode="full")
            hn = slsqp_one(h, int(np.argmax(corr)), bounds, 800, 1e-18)
            hn = fix_constraint(hn, target)
            cn = compute_c5(hn)
            if cn < c5b:
                c5b = cn
                h = hn
            h, c5 = binary_push(h, 0.3)
            if c5 < c5b:
                c5b = c5
            h, c5 = pair_flip(h, 3)
            if c5 < c5b:
                c5b = c5

    # Phase 4: Multi-n exploration (20%)
    end4 = t0 + budget_s * 0.85
    for na in [320, 360, 380, 300, 280]:
        if time.time() > end4:
            break
        ta = na / 2.0
        xo = np.arange(n) * (2.0 / n)
        xn = np.linspace(0.0, 2.0, na, endpoint=False)
        ha = np.interp(xn, xo, h)
        ha = fix_constraint(ha, ta)
        ca = compute_c5(ha)
        ba = [(0.0, 1.0)] * na
        for _ in range(6):
            if time.time() > end4:
                break
            corr = np.correlate(ha, 1.0 - ha, mode="full")
            hn = slsqp_one(ha, int(np.argmax(corr)), ba, 500, 1e-17)
            hn = fix_constraint(hn, ta)
            cn = compute_c5(hn)
            if cn < ca - 1e-16:
                ca = cn
                ha = hn
        ha, ca = binary_push(ha, 0.3)
        ha, ca = pair_flip(ha, 3)
        if ca < c5b:
            c5b = ca
            h = ha
            n = na
            target = n / 2.0
            bounds = ba

    # Phase 5: Final polish (15%)
    end5 = t0 + budget_s - 5.0
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    for _ in range(15):
        if time.time() > end5:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        hn = slsqp_one(h, int(np.argmax(corr)), bounds, 1500, 1e-19)
        hn = fix_constraint(hn, target)
        cn = compute_c5(hn)
        if cn < c5b:
            c5b = cn
            h = hn
    h, c5 = binary_push(h, 0.2)
    if c5 < c5b:
        c5b = c5
    h, c5 = pair_flip(h, 5)
    if c5 < c5b:
        c5b = c5
    h, c5 = coord_fine(h, 3)
    if c5 < c5b:
        c5b = c5
    h, c5 = pair_flip(h, 3)
    if c5 < c5b:
        c5b = c5

    c5f = min(compute_c5(h), c5b)
    return (h, c5f, len(h))
