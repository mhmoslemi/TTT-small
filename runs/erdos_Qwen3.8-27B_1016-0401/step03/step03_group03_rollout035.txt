        h_new = do_slsqp(h_best, n, T, mi, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 2: Focused subgradient (top-2 shifts)
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        alloc = min(remaining - 60, 250)
        h_sg, c5_sg = proj_subgrad(h_best, n, t0, time.time() + alloc, top_k=2, scale=0.05)
        if c5_sg < c5_best:
            c5_best, h_best = c5_sg, h_sg

    # Phase 3: Very sharp SLSQP (T going down for tighter convergence)
    sharp_temps = [(64.0, 12000), (32.0, 10000), (16.0, 10000), (8.0, 12000)]
    for T, mi in sharp_temps:
        if time.time() - t0 > budget_s - 200:
            break
        h_new = do_slsqp(h_best, n, T, mi, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 4: Block perturb + SLSQP restarts
    if time.time() - t0 < budget_s - 60:
        for r in range(14):
            if time.time() - t0 > budget_s - 30:
                break
            h_p = block_perturb(h_best, n, rng)
            h_new = do_slsqp(h_p, n, 8.0, 4000, 1e-13)
            h_new = project_feasible(h_new, n)
            h_new = do_slsqp(h_new, n, 48.0, 6000, 1e-14)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best, h_best = c5, h_new

    # Phase 5: Subgradient polish on top-1 shift
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        h_sg, c5_sg = proj_subgrad(h_best, n, t0, time.time() + min(remaining - 20, 150), top_k=1, scale=0.04)
        if c5_sg < c5_best:
            c5_best, h_best = c5_sg, h_sg

    # Phase 6: Final ultra-sharp SLSQP
    if time.time() - t0 < budget_s - 10:
        h_new = do_slsqp(h_best, n, 64.0, 15000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new
        h_new = do_slsqp(h_best, n, 16.0, 8000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
