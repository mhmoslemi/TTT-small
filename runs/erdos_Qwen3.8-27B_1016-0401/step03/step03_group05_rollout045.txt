import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def topk_gradient(h, n, K, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    center = n - 1
    if K >= len(corr):
        top_idx = np.arange(len(corr))
        w = np.ones(len(corr)) / len(corr)
    else:
        top_idx = np.argpartition(corr, -K)[-K:]
        tv = corr[top_idx]
        w = (tv / max(mc, 1e-15)) ** 6
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(K) / K
        else:
            w /= ws
    g = np.zeros(n)
    for i in range(K):
        if w[i] < 1e-10:
            continue
        g += w[i] * shift_grad(h, n, top_idx[i] - center)
    return g * dx, corr

def nesterov_refined(h_init, n, t0, budget_s, alpha0=0.4, K_start=2, K_end=12):
    target = n / 2.0
    dx = 2.0 / n
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    max_iters = 1200000
    it = 0
    while True:
        if time.time() - t0 > budget_s:
            break
        it += 1
        frac = min(it / 300000.0, 1.0)
        K = max(K_start, int(K_start + (K_end - K_start) * frac ** 0.5))

        beta = it / (it + 1.8)
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)

        g, corr = topk_gradient(h_y, n, K, dx)
        alpha = alpha0 / (1.0 + 0.005 * it) ** 0.45
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)

        h_prev = h_sg
        h_sg = h_new
        h_avg += (h_new - h_avg) / it

        if it % 30000 == 0:
            h_ap = project_feasible(h_avg, n)
            c5a = compute_c5(h_ap)
            if c5a < c5_best:
                c5_best = c5a
                h_best = h_ap.copy()
            c5c = compute_c5(h_sg)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h_sg.copy()
    h_ap = project_feasible(h_avg, n)
    c5a = compute_c5(h_ap)
    if c5a < c5_best:
        c5_best = c5a
        h_best = h_ap.copy()
    c5c = compute_c5(h_sg)
    if c5c < c5_best:
        c5_best = c5c
        h_best = h_sg.copy()
    return h_best, c5_best

def slsqp_lse(h, n, T, maxiter, tol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    nc = 2 * n - 1

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(nc):
            w = sw[oi]
            if w < 1e-12:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return grad * dx

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        return res.x if res.success or res.fun < np.inf else h.copy()
    except Exception:
        return h.copy()

def slsqp_topk_small(h, n, K, maxiter, tol):

def coord_descent_1d(h, n, dx, max_rounds=5):
    """1D coordinate minimization of max correlation, with sum constraint."""
    target = n / 2.0
    center = n - 1
    h_c = h.copy()
    for rd in range(max_rounds):
        improved = False
        for j in range(n):
            # Compute current correlation values
            corr = np.correlate(h_c, 1.0 - h_c, mode="full")
            # For each lag, the gradient coefficient of h[j]:
            coeffs = np.zeros(2 * n - 1)
            for oi in range(2 * n - 1):
                s = oi - center
                if s >= 0:
                    if j >= s and (j + s) < n:
                        coeffs[oi] = (1.0 - h_c[j - s])
                    if (j + s) < n:
                        coeffs[oi] -= h_c[j + s]
                else:
                    ss = -s
                    if (j + ss) < n:
                        coeffs[oi] = (1.0 - h_c[j + ss])
                    if j >= ss:
                        coeffs[oi] -= h_c[j - ss]
            # h_c[j] appears linearly: f_oi = corr_oi + coeffs[oi]*delta where delta = new - old
            old = h_c[j]
            # Minimize max_i (corr[i] + coeffs[i]*delta) over delta in [0-old, 1-old]
            lo_d, hi_d = -old, 1.0 - old
            if lo_d > hi_d:
                continue
            # Sample candidate deltas
            best_d = 0.0
            best_val = np.max(corr)
            # Check boundaries and intersections
            candidates = np.array([lo_d, hi_d, 0.0])
            # Intersections of top few pairs
            top5 = np.argpartition(corr + coeffs * 0, -5)[-5:]
            for a in range(len(top5)):
                for b in range(a + 1, len(top5)):
                    ia, ib = top5[a], top5[b]
                    dc = coeffs[ib] - coeffs[ia]
                    if abs(dc) > 1e-12:
                        d_int = (corr[ia] - corr[ib]) / dc
                        if lo_d <= d_int <= hi_d:
                            candidates = np.append(candidates, d_int)
            for d in candidates:
                if d < lo_d - 1e-12 or d > hi_d + 1e-12:
                    continue
                d = np.clip(d, lo_d, hi_d)
                val = np.max(corr + coeffs * d)
                if val < best_val - 1e-12:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-10:
                h_c[j] = old + best_d
                # Restore sum constraint: adjust nearest non-boundary element
                h_c[j] = np.clip(h_c[j], 0.0, 1.0)
                diff = target - h_c.sum()
                if abs(diff) > 1e-10:
                    # Find an element not at boundary to adjust
                    mask = (h_c > 0.05) & (h_c < 0.95)
                    if mask.any():
                        idxs = np.where(mask)[0]
                        adj = np.random.choice(idxs)
                        h_c[adj] = np.clip(h_c[adj] + diff, 0.0, 1.0)
                        diff2 = target - h_c.sum()
                        h_c[adj] = np.clip(h_c[adj] + diff2, 0.0, 1.0)
                    else:
                        h_c = project_feasible(h_c, n)
                improved = True
        if not improved:
            break
    h_c = project_feasible(h_c, n)
    return h_c
