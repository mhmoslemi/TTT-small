    def cf(x):
        return x.sum() - target
    def cj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def nesterov_opt(h, n, T, k_top, n_iter, lr_init=0.015, decay=0.9999):
    h_y = h.copy().astype(float)
    h_z = h.copy().astype(float)
    c5_best = compute_c5(h)
    h_best = h.copy()
    t_mom = 1.0
    lr = lr_init
    for it in range(n_iter):
        _, grad = smooth_max_obj_grad(h_y, n, T, k_top)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        h_z = h_y + beta * (h_y - h_z)
        h_y = h_z - lr * grad
        h_y = project_feasible(h_y, n)
        t_mom = t_new
        lr *= decay
        if (it + 1) % 3000 == 0:
            c5 = compute_c5(h_y)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_y.copy()
    c5 = compute_c5(h_y)
    if c5 < c5_best:
        c5_best = c5
        h_best = h_y
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B coarse-to-fine temperature schedule
    schedule = [(128, 60, 200), (64, 50, 300), (32, 40, 300), (16, 25, 400),
                (8, 20, 500), (4, 15, 600), (2, 12, 700), (1, 10, 800),
                (0.5, 8, 1000), (0.25, 6, 1200), (0.125, 5, 1500)]
    for T, k, mi in schedule:
        rem = budget_s - (time.time() - t0)
        if rem < 150:
            break
        h_new = lbfgsb_opt(h_best, n, T, k, mi)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Nesterov accelerated gradient
    rem = budget_s - (time.time() - t0)
    if rem > 250:
        n_iter = min(150000, int(rem * 8))
        h_f, c5_f = nesterov_opt(h_best, n, T=4, k_top=10, n_iter=n_iter, lr_init=0.01, decay=0.99995)
        h_f = project_feasible(h_f, n)
        c5_f = compute_c5(h_f)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f

    # Phase 3: SLSQP fine polish
    for T, k in [(1.0, 5), (0.5, 5), (0.25, 4), (0.1, 3)]:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        h_new = slsqp_opt(h_best, n, T, k, 4000, 1e-19)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Multi-resolution upsample + optimize
    rem = budget_s - (time.time() - t0)
    if rem > 250:
        for n_try in [n * 2, n * 3]:
            if n_try > 1680 or time.time() - t0 > budget_s - 120:
                break
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_hi = np.interp(x_new, x_old, h_best)
            h_hi = project_feasible(h_hi, n_try)
            for T, k, mi in [(32, 50, 300), (8, 25, 400), (2, 12, 500), (0.5, 6, 400)]:
                if time.time() - t0 > budget_s - 40:
                    break
                h_new = lbfgsb_opt(h_hi, n_try, T, k, mi)
                h_new = project_feasible(h_new, n_try)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
                    n = n_try

    # Phase 5: Perturbation-based local search
    rem = budget_s - (time.time() - t0)
    n_perts = max(0, int(rem / 20))
    for pi in range(n_perts):
        if time.time() - t0 > budget_s - 15:
            break
        h_p = h_best.copy()
        ptype = rng.integers(0, 4)
        if ptype == 0:
            h_p += rng.uniform(0.005, 0.03) * rng.normal(0, 1, n)
        elif ptype == 1:
            freq = rng.integers(2, 10)
            x = np.arange(n) * 2.0 / n
            h_p += rng.uniform(0.01, 0.04) * np.sin(np.pi * freq * x + rng.uniform(0, 6.28))
        elif ptype == 2:
            b = rng.integers(3, 15)
            s1 = rng.integers(0, max(1, n - b))
            s2 = rng.integers(0, max(1, n - b))
            tmp = h_p[s1:s1+b].copy()
            h_p[s1:s1+b] = h_p[s2:s2+b]
            h_p[s2:s2+b] = tmp
        else:
            start = rng.integers(0, max(1, n - 8))
            ln = rng.integers(3, min(10, n - start))
            h_p[start:start+ln] = 1.0 - h_p[start:start+ln]
        h_p = project_feasible(h_p, n)
        h_new = lbfgsb_opt(h_p, n, 8, 12, 300)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Final SLSQP squeeze
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_new = slsqp_opt(h_best, n, 0.25, 4, 5000, 1e-20)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
