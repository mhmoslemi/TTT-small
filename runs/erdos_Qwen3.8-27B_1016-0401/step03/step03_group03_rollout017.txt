def nesterov_adaptive(h_init, n, t0, time_budget, K=10, max_iters=500000):
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    alpha = 0.07
    no_improve = 0
    import math
    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        beta = float(it) / (it + 8.0) if it > 1 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top_idx = np.argsort(corr)[-K:]
        mv = corr[top_idx].max()
        w = np.exp(6.0 * (corr[top_idx] - mv))
        w /= w.sum()
        g = np.zeros(n)
        for wk, ik in zip(w, top_idx):
            g += wk * shift_grad_raw(h_y, n, ik - center)
        g *= dx
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        no_improve += 1
        if no_improve >= 4000:
            cv = compute_c5(h_sg)
            if cv < c5_best - 1e-11:
                c5_best = cv
                h_best = h_sg.copy()
                no_improve = 0
                alpha = min(0.3, alpha * 1.02)
            else:
                alpha = max(0.005, alpha * 0.85)
                no_improve = 0
                if alpha < 0.008:
                    break
    cv = compute_c5(h_sg)
    if cv < c5_best:
        c5_best = cv
        h_best = h_sg.copy()
    return h_best, c5_best

def targeted_swap_ls(h, n, rng, max_steps, t0, budget):
    """Local search: target worst shifts via 1<->0 swaps."""
    h_cur = project_feasible(h, n)
    c5_cur = compute_c5(h_cur)
    ones_mask = h_cur > 0.5
    zeros_mask = h_cur < 0.5
    stall = 0
    for step in range(max_steps):
        if time.time() - t0 > budget:
            break
        if stall > 80000:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        max_k = int(np.argmax(corr))
        max_v = corr[max_k]
        # Try to find good swap: flip a 1 near the worst interaction
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(zeros_mask)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        shift_abs = abs(max_k - (n - 1))
        if shift_abs == 0:
            ip = rng.choice(ones_idx)
            jz = rng.choice(zeros_idx)
        else:
            # Prefer flipping 1s that contribute to the worst shift
            if max_k >= n - 1:
                s = max_k - (n - 1)
                good_ones = [i for i in ones_idx if i >= s and (i + s) < n and h_cur[i + s] < 0.5]
            else:
                s = (n - 1) - max_k
                good_ones = [i for i in ones_idx if i < s and h_cur[i - s] < 0.5]
            if good_ones and len(good_ones) < 200:
                ip = good_ones[rng.integers(len(good_ones))]
            else:
                ip = rng.choice(ones_idx)
            jz = rng.choice(zeros_idx)
            if jz == ip:
                jz = zeros_idx[rng.integers(len(zeros_idx))]
        if ip == jz:
            continue
        h_try = h_cur.copy()
        h_try[ip] = 1.0 - h_try[ip]
        h_try[jz] = 1.0 - h_try[jz]
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < c5_cur - 1e-12:
            h_cur = h_try
            c5_cur = c5_try
            ones_mask = h_cur > 0.5
            zeros_mask = h_cur < 0.5
            stall = 0
        else:
            stall += 1
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse-to-fine (16 temps)
    temps = [256, 128, 64, 32, 16, 8, 4, 2, 1, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
    for Ti, T in enumerate(temps):
        if time.time() - t0 > budget_s - 480:
            break
        if T > 8:
            hrs = 5000
        elif T > 1:
            hrs = 8000
        else:
            hrs = 12000
        h_new = do_slsqp(h_best, n, T, hrs, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        else:
            # Try even lower temp for extra iterations
            if T <= 0.5 and time.time() - t0 < budget_s - 400:
                h_new = do_slsqp(h_best, n, T * 0.5, 8000, 1e-16)
                h_new = project_feasible(h_new, n)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new

    # Phase 2: Nesterov with larger K, adaptive step
    rem = budget_s - (time.time() - t0)
    if rem > 250:
        h_sg, c5_sg = nesterov_adaptive(h_best, n, t0, budget_s - 380, K=12, max_iters=500000)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase 3: SLSQP fine polish at very low temps
    for T in [0.3, 0.1, 0.05, 0.02, 0.01]:
        if time.time() - t0 > budget_s - 180:
            break
        h_new = do_slsqp(h_best, n, T, 14000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Targeted swap local search
    if time.time() - t0 < budget_s - 120:
        rem2 = int((budget_s - (time.time() - t0) - 120) * 2500)
        h_lp, c5_lp = targeted_swap_ls(h_best, n, rng, rem2, t0, budget_s - 120)
        if c5_lp < c5_best:
            c5_best = c5_lp
            h_best = h_lp
        # SLSQP after binary search
        if time.time() - t0 < budget_s - 60:
            h_new = do_slsqp(h_best, n, 0.2, 12000, 1e-16)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 5: Perturbation + reoptimize
    for pi in range(12):
        if time.time() - t0 > budget_s - 25:
            break
        sigma = 0.0015 + 0.004 * (pi % 4)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_new = do_slsqp(h_p, n, 1.0, 6000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            h_new = do_slsqp(h_best, n, 0.1, 10000, 1e-16)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Final polish
    if time.time() - t0 < budget_s - 8:
        h_new = do_slsqp(h_best, n, 0.05, 15000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
