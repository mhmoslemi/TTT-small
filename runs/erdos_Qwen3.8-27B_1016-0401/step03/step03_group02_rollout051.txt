    """Return smooth max of correlation and its gradient"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)

    """Projected gradient with soft-max objective"""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.1

            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                h = h_trial
                best_c5 = c5_trial
                best_h = h_trial.copy()
                break
        else:
            step *= 0.5
            if step < 1e-8:
                break
        if it % 50 == 0:
            step = min(step * 1.2, 0.5)
    return best_h, best_c5

    n = len(h)
    c5_cur = compute_c5(h)
    dts = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]

                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:

    return h, c5_cur

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lag_grad(h, o):
    n = len(h)
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] += 1.0 - h[:n-o]
        g[:n-o] -= h[o:]
    else:
        p = -o
        g[:n-p] += 1.0 - h[p:]
        g[p:] -= h[:n-p]
    return g

def slsqp_lag(h, lag_idx, bounds, target, maxiter):
    n = len(h)
    dx = 2.0 / n
    o = lag_idx - (n - 1)
    def obj(x):
        return np.correlate(x, 1.0 - x, mode="full")[lag_idx] * dx

        return lag_grad(x, o) * dx

                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
    return np.clip(res.x, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

_TARGET = 0.0
_LAG_IDX = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lag_grad(x, o):
    n = len(x)
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * x
    elif o > 0:
        g[o:] += 1.0 - x[:n-o]
        g[:n-o] -= x[o:]
    else:
        p = -o
        g[:n-p] += 1.0 - x[p:]
        g[p:] -= x[:n-p]
    return g

def _obj(x):
    n = len(x)
    return np.correlate(x, 1.0 - x, mode="full")[_LAG_IDX] * (2.0 / n)

def _jac(x):
    n = len(x)
    o = _LAG_IDX - (n - 1)
    return lag_grad(x, o) * (2.0 / n)

def _c_fun(x):
    return x.sum() - _TARGET

def _c_jac(x):
    return np.ones(len(x))

def slsqp_lag(h, lag_idx, bounds, target, maxiter):
    global _TARGET, _LAG_IDX
    _TARGET = target
    _LAG_IDX = lag_idx
    cons = [dict(type='eq', fun=_c_fun, jac=_c_jac)]
    res = minimize(_obj, h, jac=_jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
    return np.clip(res.x, 0.0, 1.0)

def sm_pg(h, target, T, max_iter, t_end):
    n = len(h)
    step = 0.08
    for it in range(max_iter):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        mc = np.max(corr)
        sw = np.exp(T * (corr - mc))
        sw /= sw.sum()
        center = n - 1
        grad = np.zeros(n)
        for ki in range(len(sw)):
            w = sw[ki]
            if w < 1e-14:
                continue
            grad += w * lag_grad(h, ki - center)
        accepted = False
        for s in [step, step*0.5, step*0.2, step*0.05]:
            ht = project_h(h - s * grad, target)
            c5t = compute_c5(ht)
            c5h = compute_c5(h)
            if c5t < c5h - 1e-16:
                h = ht
                step = min(step * 1.05, 0.25)
                accepted = True
                break
        if not accepted:
            step *= 0.5
            if step < 1e-9:
                break
    return h

def fine_coord(h, target, dts, max_sweeps, t_end):
    n = len(h)
    c5_cur = compute_c5(h)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for j in range(n):
            for dt in dts:
                trial = h.copy()
                trial[j] = np.clip(trial[j] + dt, 0.0, 1.0)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < c5_cur - 1e-15:
                    h = trial
                    c5_cur = c
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def slsqp_cycle(h, bounds, target, max_rounds, t_end):
    n = len(h)
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top = np.argsort(corr)[-4:]
        for lag in top:
            if time.time() > t_end:
                break
            try:
                h_new = slsqp_lag(h, lag, bounds, target, 500)
                h_new = project_h(h_new, target)
                c = compute_c5(h_new)
                if c < c5_cur - 1e-18:
                    h, c5_cur = h_new, c
            except Exception:
                pass
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    global _TARGET
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(340)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    _TARGET = target
    h_best = project_h(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: SLSQP cycling (25% budget)
    e1 = t0 + budget_s * 0.25
    h_best, c5_best = slsqp_cycle(h_best, bounds, target, 150, e1)

    # Phase 2: Soft-max projected gradient (20%)
    e2 = t0 + budget_s * 0.45
    for T in [2000, 8000, 25000, 80000]:
        if elapsed() > e2:
            break
        h_sm = sm_pg(h_best.copy(), target, T, 400, e2)
        h_sm = project_h(h_sm, target)
        c = compute_c5(h_sm)
        if c < c5_best:
            h_best, c5_best = h_sm, c

    # Phase 3: Fine coordinate descent (20%)
    e3 = t0 + budget_s * 0.65
    dts_med = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    h_best, c5_best = fine_coord(h_best, target, dts_med, 40, e3)

    # Phase 4: Try n=300 (15%)
    e4 = t0 + budget_s * 0.80
    n_alt = 300
    target_alt = n_alt / 2.0
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.linspace(0.0, 2.0, n_alt, endpoint=False)
    h_alt = np.interp(x_new, x_old, h_best)
    h_alt = project_h(h_alt, target_alt)
    c5_alt = compute_c5(h_alt)
    bounds_alt = [(0.0, 1.0)] * n_alt
    h_alt, c5_alt = slsqp_cycle(h_alt, bounds_alt, target_alt, 80, e4)
    h_alt, c5_alt = fine_coord(h_alt, target_alt, dts_med, 15, e4)
    if c5_alt < c5_best:
        h_best, c5_best, n, target, bounds = h_alt, c5_alt, n_alt, target_alt, bounds_alt

    # Phase 5: Perturbation + re-optimize (10%)
    e5 = t0 + budget_s * 0.90
    for k in range(30):
        if elapsed() > e5:
            break
        scale = 0.004 + 0.006 * (k % 3)
        hp = project_h(h_best + rng.normal(0, scale, len(h_best)), target)
        corr_p = np.correlate(hp, 1.0 - hp, mode="full")
        wl = int(np.argmax(corr_p))
        try:
            h_new = slsqp_lag(hp, wl, bounds, target, 300)
            h_new = project_h(h_new, target)
            c = compute_c5(h_new)
            if c < c5_best:
                h_best, c5_best = h_new, c
        except Exception:
            pass

    # Phase 6: Ultra-fine polish (10%)
    e6 = t0 + budget_s - 3.0
    dts_fine = [0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005, 0.0002, -0.0002]
    h_best, c5_best = fine_coord(h_best, target, dts_fine, 50, e6)

    # Final SLSQP
    corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
    wf = int(np.argmax(corr_f))
    try:
        h_new = slsqp_lag(h_best, wf, bounds, target, 3000)
        h_new = project_h(h_new, target)
        c = compute_c5(h_new)
        if c < c5_best:
            h_best, c5_best = h_new, c
    except Exception:
        pass

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
