    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def _gobj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _GT * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _GT) * (2.0 / _GN)

def _gjac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _GT * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _GN - 1
    dx = 2.0 / _GN
    grad = np.zeros(_GN)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-12:
            continue
        grad += w * shift_grad(x, _GN, oi - center)
    return grad * dx

def _gcons(x):
    return x.sum() - _GTT

def _gcons_jac(x):
    return np.ones(_GN)

def do_slsqp(h, n, T, maxiter, ftol):
    global _GN, _GT, _GTT
    _GN = n
    _GT = float(T)
    _GTT = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _gcons, 'jac': _gcons_jac}]
    try:
        res = minimize(_gobj, h, method='SLSQP', jac=_gjac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def nesterov_adv(h_init, n, t0, time_budget, K=12, max_iters=500000):
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    alpha0 = 0.12
    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        beta = float(it) / (it + 8.0) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top_idx = np.argsort(corr)[-K:]
        mv = corr[top_idx].max()
        w = np.exp(7.0 * (corr[top_idx] - mv))
        w /= w.sum()
        g = np.zeros(n)
        for wk, ik in zip(w, top_idx):
            g += wk * shift_grad(h_y, n, ik - center)
        g *= dx
        alpha = alpha0 / (1.0 + 0.003 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 15000 == 0:
            h_proj = project_feasible(h_sg, n)
            cv = compute_c5(h_proj)
            if cv < c5_best:
                c5_best = cv
                h_best = h_proj.copy()
    h_proj = project_feasible(h_sg, n)
    cv = compute_c5(h_proj)
    if cv < c5_best:
        c5_best = cv
        h_best = h_proj.copy()
    return h_best, c5_best

def pushpull_loop(h, n, t0, time_budget, K=10):
    dx = 2.0 / n
    center = n - 1
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    step = 0.015
    for it in range(300000):
        if time.time() - t0 > time_budget:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        top_idx = np.argsort(corr)[-K:]
        g = np.zeros(n)
        for ik in top_idx:
            g += shift_grad(h_cur, n, ik - center)
        g *= dx
        pos_idx = np.where(g > 0.008)[0]
        neg_idx = np.where(g < -0.008)[0]
        if len(pos_idx) == 0 or len(neg_idx) == 0:
            break
        ip = pos_idx[np.argmax(g[pos_idx])]
        in_ = neg_idx[np.argmin(g[neg_idx])]
        delta = min(step, h_cur[ip], 1.0 - h_cur[in_])
        if delta < 0.0005:
            continue
        h_try = h_cur.copy()
        h_try[ip] -= delta
        h_try[in_] += delta
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try <= c5_cur + 1e-13:
            h_cur = h_try
            c5_cur = c5_try
    return h_cur, c5_cur

def coord_search(h, n, rng, steps=8000):
    dx = 2.0 / n
    center = n - 1
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for s in range(steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        ak = int(np.argmax(corr))
        g = shift_grad(h_cur, n, ak - center) * dx
        pos_idx = np.where(g > 0.01)[0]
        neg_idx = np.where(g < -0.01)[0]
        if len(pos_idx) > 0 and len(neg_idx) > 0:
            ip = rng.choice(pos_idx)
            in_ = rng.choice(neg_idx)
            delta = min(0.04, h_cur[ip], 1.0 - h_cur[in_])
            h_try = h_cur.copy()
            h_try[ip] -= delta
            h_try[in_] += delta
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-10:
                h_cur = h_try
                c5_cur = c5_try
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP annealing (coarse to fine)
    temps_p1 = [64, 32, 16, 8, 4, 2, 1.5, 1.0, 0.7, 0.5, 0.35, 0.2, 0.12]
    for T in temps_p1:
        if time.time() - t0 > budget_s - 520:
            break
        h_new = do_slsqp(h_best, n, T, 10000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Advanced Nesterov with K=12
    h_sg, c5_sg = nesterov_adv(h_best, n, t0, budget_s - 420, K=12, max_iters=450000)
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg

    # Phase 3: Fine SLSQP polish
    temps_p3 = [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05]
    for T in temps_p3:
        if time.time() - t0 > budget_s - 230:
            break
        h_new = do_slsqp(h_best, n, T, 15000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Deterministic push-pull
    if time.time() - t0 < budget_s - 80:
        h_pp, c5_pp = pushpull_loop(h_best, n, t0, budget_s - 80, K=10)
        if c5_pp < c5_best:
            c5_best = c5_pp
            h_best = h_pp
        h_new = do_slsqp(h_best, n, 0.2, 12000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Randomized perturbations + SLSQP
    for pi in range(12):
        if time.time() - t0 > budget_s - 30:
            break
        sigma = 0.002 + 0.003 * (pi % 4)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_new = do_slsqp(h_p, n, 0.8, 8000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Final ultra-fine polish
    if time.time() - t0 < budget_s - 12:
        for T in [0.05, 0.03, 0.02]:
            if time.time() - t0 > budget_s - 12:
                break
            h_new = do_slsqp(h_best, n, T, 20000, 1e-16)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
