def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break

            corr_a = np.correlate(h_alt, 1.0 - h_alt, mode="full")
            worst_a = int(np.argmax(corr_a))
            try:
                h_new = try_slsqp_lag(h_alt, worst_a, bounds_alt, 400, 1e-17)
                h_new = fix_constraint(h_new, target_alt)
                c5_new = compute_c5(h_new)
                if c5_new < c5_alt:
                    c5_alt = c5_new
                    h_alt = h_new
            except Exception:
                pass
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt
            target = n / 2.0
            bounds = [(0.0, 1.0)] * n
            _TARGET = target

    # Phase 7: Final polish (5s buffer)
    p7_end = t0 + budget_s - 3.0
    _TARGET = target
    for it in range(6):
        if elapsed() > p7_end:
            break
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst_f = int(np.argmax(corr_f))
        try:
            h_new = try_slsqp_lag(h_best, worst_f, bounds, 800, 1e-19)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass
    h_new, c5_cd = coord_descent(h_best, 3, target)
    if c5_cd < c5_best:
        c5_best = c5_cd
        h_best = h_new

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
