import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):

    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):

        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def slse_obj_grad(h, n, T, k_top):
    """Log-sum-exp smoothed max with top-k restriction."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    idx = np.argpartition(corr, -k_top)[-k_top:]
    topk = corr[idx]
    mc = np.max(topk)
    sh = T * (topk - mc)
    e = np.exp(sh)
    obj = (mc + np.log(e.sum()) / T) * dx
    
    # Gradient
    sw = e / e.sum()
    grad = np.zeros(n)
    j = np.arange(n)
    for oi in range(len(idx)):
        w = sw[oi]
        if w < 1e-12:
            continue
        shift = idx[oi] - center
        if shift >= 0:
            m1 = j < (n - shift)
            grad[m1] += w * (1.0 - h[j[m1] + shift])
            m2 = j >= shift
            grad[m2] -= w * h[j[m2] - shift]
        else:
            s2 = -shift
            m1 = j < (n - s2)
            grad[m1] += w * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * h[j[m2] - s2]
    grad *= dx
    return obj, grad

def slsqp_opt(h, n, T, k_top, maxiter, ftol, t_end):
    def obj(x):
        return slse_obj_grad(x, n, T, k_top)[0]
    def jac(x):
        return slse_obj_grad(x, n, T, k_top)[1]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        return res.x
    except Exception:
        return h.copy()
