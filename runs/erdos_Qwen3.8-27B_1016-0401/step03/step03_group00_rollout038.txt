```python
def all_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_shift(h, n, s):
    """Gradient of c_s = sum_i h[i]*(1-h[i+s]) w.r.t. h (vectorized)"""
    g = np.zeros(n)
    if s >= 0:
        L = n - s
        g[:L] += (1.0 - h[s:])
        g[s:] -= h[:L]
    else:
        ss = -s
        L = n - ss
        g[:L] += (1.0 - h[ss:])
        g[ss:] -= h[:L]
    return g

def softmax_grad(h, n, T, top_k):
    """Gradient of soft-max objective (top-k approximation)"""
    corr = all_corr(h)
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    cmax = c.max()
    e = np.exp(T * (c - cmax))
    w = e / e.sum()
    # Gradient includes the max term + weighted soft part
    grad = np.zeros(n)
    # Max term: gradient at argmax
    imax = np.argmax(c)
    grad += grad_shift(h, n, idx[imax] - (n-1))
    # Soft part: weighted average
    for wi, ki in zip(w, idx):
        if wi > 1e-12:
            grad += wi * grad_shift(h, n, ki - (n-1))
    dx = 2.0 / n
    return grad * dx

def softmax_val(h, n, T, top_k):
    corr = all_corr(h)
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    cmax = c.max()
    e = np.exp(T * (c - cmax))
    dx = 2.0 / n
    return (cmax + np.log(e.sum()) / T) * dx
```

```python
def nesterov_opt(h0, n, T, top_k, n_iter, t_end, rng):

        if it == 0:
            h_y = h_cur
        else:
            beta = (1 + np.sqrt(1 + 4*it*it)) / (2*it + 2)  # or it/(it+1.5)

        g = softmax_grad(h_y_p, n, T, top_k)

        alpha = 1.0 / (np.sqrt(it + 1) * np.sqrt(n) / 2)
        
        # Update
        h_new = project_feasible(h_y_p - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        
        # Check improvement periodically
        if (it+1) % 500 == 0:

            if c1 < best:

            c2 = compute_c5(project_feasible(h_avg, n))
            if c2 < best:
                best, best_h = c2, project_feasible(h_avg, n).copy()
    
    return best_h, best
```

```python
import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def grad_shift(h, n, s):
    g = np.zeros(n)
    if s >= 0:
        L = n - s
        if L > 0:
            g[:L] += (1.0 - h[s:])
            g[s:] -= h[:L]
    else:
        ss = -s
        L = n - ss
        if L > 0:
            g[:L] += (1.0 - h[ss:])
            g[ss:] -= h[:L]
    return g

def sf_obj(x, n, T, top_k, pen):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    cm = c.max()
    e = np.exp(T * (c - cm))
    v = (cm + np.log(e.sum()) / T) * dx
    v += pen * (x.sum() - n / 2.0) ** 2
    return v

def sf_grad(x, n, T, top_k, pen):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    cm = c.max()
    e = np.exp(T * (c - cm))
    w = e / e.sum()
    g = np.zeros(n)
    for wi, ki in zip(w, idx):
        if wi > 1e-12:
            g += wi * grad_shift(x, n, ki - (n - 1))
    g *= dx
    g += 2.0 * pen * (x.sum() - n / 2.0)
    return g

def lbfgs_run(h, n, T, top_k, pen, maxiter, t_end):
    if time.time() > t_end:
        return h
    bounds = [(0.0, 1.0)] * n
    try:
        res = fmin_l_bfgs_b(sf_obj, h, fprime=sf_grad,
                           args=(n, T, top_k, pen),
                           pgtol=1e-14, maxiter=maxiter,
                           maxfun=maxiter * 5, bounds=bounds, disp=False)
        return res[0]
    except Exception:
        return h

def nesterov_run(h0, n, t_end, T, top_k, n_max=500000):
    dx = 2.0 / n
    h_cur = project_feasible(h0.copy(), n)
    h_prev = h_cur.copy()
    h_avg = h_cur.copy()
    c5_best = compute_c5(h_cur)
    h_best = h_cur.copy()
    for it in range(n_max):
        if time.time() > t_end:
            break
        if it > 0:
            beta = float(it) / (float(it) + 1.5)
            h_y = h_cur + beta * (h_cur - h_prev)
        else:
            h_y = h_cur.copy()
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for ki in idx:
            g += grad_shift(h_y, n, ki - (n - 1))
        g *= dx / top_k
        alpha = 2.0 / (1.0 + 0.001 * float(it))
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / (float(it) + 1.0)
        if (it + 1) % 2000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < c5_best:
                c5_best, h_best = c1, h_cur.copy()
            h_avg_p = project_feasible(h_avg, n)
            c2 = compute_c5(h_avg_p)
            if c2 < c5_best:
                c5_best, h_best = c2, h_avg_p.copy()
    return h_best, c5_best

def make_perturbed(h, n, rng, mode):
    if mode == 0:
        sig = 0.003 + 0.002 * (int(time.time() * 1000) % 5)
        return project_feasible(h + rng.normal(0, sig, n), n)
    elif mode == 1:
        h_p = h.copy()
        lb = int(rng.integers(0, max(1, n - 30)))
        ln = int(rng.integers(5, 31))
        h_p[lb:lb+ln] = 1.0 - h_p[lb:lb+ln]
        return project_feasible(h_p, n)
    elif mode == 2:
        h_p = h.copy()
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        h_p[[i, j]] = h_p[[j, i]]
        return project_feasible(h_p, n)
    else:
        sig = 0.008
        return project_feasible(h + rng.normal(0, sig, n), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n0 = 280
        h_best = np.zeros(n0)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    
    # Phase 1: L-BFGS-B with decreasing T
    t1 = t0 + budget_s * 0.35
    params = [(64, 30, 10, 3000), (32, 25, 10, 4000), (16, 20, 50, 5000),
              (8, 15, 100, 5000), (4, 12, 200, 6000)]
    for T, tk, pen, mi in params:
        if time.time() > t1:
            break
        hn = lbfgs_run(h_best, n, T, tk, pen, mi, t1)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn
    
    # Phase 2: Nesterov exploration
    t2 = t0 + budget_s * 0.55
    hs, c5s = nesterov_run(h_best, n, t2, T=4, top_k=8, n_max=300000)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs
    
    # Phase 3: Fine L-BFGS-B
    t3 = t0 + budget_s * 0.75
    params3 = [(2, 10, 500, 5000), (1, 8, 1000, 6000), (0.5, 6, 2000, 8000)]
    for T, tk, pen, mi in params3:
        if time.time() > t3:
            break
        hn = lbfgs_run(h_best, n, T, tk, pen, mi, t3)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn
    
    # Phase 4: Multi-restart
    t4 = t0 + budget_s - 30
    for pi in range(80):
        if time.time() > t4:
            break
        hp = make_perturbed(h_best, n, rng, pi % 4)
        hn = lbfgs_run(hp, n, 2, 10, 500, 3000, t4)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn
    
    # Phase 5: Final polish
    if time.time() < budget_s - 5:
        hn = lbfgs_run(h_best, n, 0.5, 5, 5000, 5000, budget_s - 5)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn
    
    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
```

```python
import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    if abs(y.sum() - target) < 1e-14:
        return np.clip(y, 0.0, 1.0)
    lo, hi = -3.0, 4.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def grad_shift(h, n, s):
    g = np.zeros(n)
    if s >= 0:
        L = n - s
        if L > 0:
            g[:L] += (1.0 - h[s:])
            g[s:] -= h[:L]
    else:
        ss = -s
        L = n - ss
        if L > 0:
            g[:L] += (1.0 - h[ss:])
            g[ss:] -= h[:L]
    return g

def sf_obj(x, n, T, top_k, pen):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    cm = c.max()
    e = np.exp(T * (c - cm))
    v = (cm + np.log(e.sum()) / T) * dx
    v += pen * (x.sum() - n / 2.0) ** 2
    return v

def sf_grad(x, n, T, top_k, pen):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    cm = c.max()
    e = np.exp(T * (c - cm))
    w = e / e.sum()
    g = np.zeros(n)
    for wi, ki in zip(w, idx):
        if wi > 1e-12:
            g += wi * grad_shift(x, n, ki - (n - 1))
    g *= dx
    g += 2.0 * pen * (x.sum() - n / 2.0)
    return g

def lbfgs_opt(h, n, T, top_k, pen, maxiter, t_end):
    if time.time() > t_end:
        return h.copy()
    bounds = [(0.0, 1.0)] * n
    try:
        res = fmin_l_bfgs_b(sf_obj, h, fprime=sf_grad,
                           args=(n, T, top_k, pen),
                           pgtol=1e-15, maxiter=maxiter,
                           maxfun=maxiter * 6, bounds=bounds, disp=False)
        return res[0]
    except Exception:
        return h.copy()

def nesterov_opt(h0, n, t_end, top_k=8):
    dx = 2.0 / n
    h_cur = project_feasible(h0.copy(), n)
    h_prev = h_cur.copy()
    h_avg = h_cur.copy()
    c5_best = compute_c5(h_cur)
    h_best = h_cur.copy()
    for it in range(600000):
        if time.time() > t_end:
            break
        if it > 0:
            beta = float(it) / (float(it) + 1.5)
            h_y = h_cur + beta * (h_cur - h_prev)
        else:
            h_y = h_cur.copy()
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for ki in idx:
            g += grad_shift(h_y, n, ki - (n - 1))
        g *= dx / top_k
        alpha = 1.5 / (1.0 + 0.003 * float(it))
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / (float(it) + 1.0)
        if (it + 1) % 3000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < c5_best:
                c5_best, h_best = c1, h_cur.copy()
            h_avg_p = project_feasible(h_avg, n)
            c2 = compute_c5(h_avg_p)
            if c2 < c5_best:
                c5_best, h_best = c2, h_avg_p.copy()
    return h_best, c5_best

def do_perturb(h, n, rng, mode):
    if mode == 0:
        sig = 0.003 + 0.002 * (mode * 7 + int(np.random.randint(5)))
        return project_feasible(h + rng.normal(0, 0.004, n), n)
    elif mode == 1:
        h_p = h.copy()
        lb = int(rng.integers(0, max(1, n - 30)))
        ln = int(rng.integers(4, 26))
        h_p[lb:lb + ln] = 1.0 - h_p[lb:lb + ln]
        return project_feasible(h_p, n)
    elif mode == 2:
        h_p = h.copy()
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i != j:
            h_p[[i, j]] = h_p[[j, i]]
        else:
            h_p[i] = 1.0 - h_p[i]
        return project_feasible(h_p, n)
    else:
        return project_feasible(h + rng.normal(0, 0.009, n), n)

def try_upsample(h_small, n_small, n_large, t_end):
    if time.time() > t_end:
        return None, None
    ratio = n_large // n_small
    h_big = np.repeat(h_small, ratio)
    h_big = project_feasible(h_big, n_large)
    c5_big = compute_c5(h_big)
    hn = lbfgs_opt(h_big, n_large, 4, 15, 5000, 4000, t_end)
    hn = project_feasible(hn, n_large)
    c5 = compute_c5(hn)
    if c5 < c5_big:
        c5_big = c5
    hn2 = lbfgs_opt(hn, n_large, 1, 8, 10000, 5000, t_end)
    hn2 = project_feasible(hn2, n_large)
    c5_2 = compute_c5(hn2)
    if c5_2 < c5_big:
        hn, c5_big = hn2, c5_2
    return hn, c5_big

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n_init = 280
        h_best = np.zeros(n_init)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B coarse to fine (35% time)
    t1 = t0 + budget_s * 0.35
    schedule1 = [(64, 40, 500, 2000), (32, 30, 1000, 3000),
                 (16, 25, 2000, 4000), (8, 18, 5000, 5000),
                 (4, 12, 10000, 5000), (2, 10, 10000, 5000)]
    for T, tk, pen, mi in schedule1:
        if time.time() > t1:
            break
        hn = lbfgs_opt(h_best, n, T, tk, pen, mi, t1)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 2: Nesterov multi-shift (20% time)
    t2 = t0 + budget_s * 0.55
    hs, c5s = nesterov_opt(h_best, n, t2, top_k=10)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs

    # Phase 3: Fine L-BFGS-B polish (15% time)
    t3 = t0 + budget_s * 0.70
    schedule3 = [(4, 12, 10000, 6000), (2, 8, 10000, 8000),
                 (1, 6, 20000, 8000), (0.5, 5, 20000, 10000)]
    for T, tk, pen, mi in schedule3:
        if time.time() > t3:
            break
        hn = lbfgs_opt(h_best, n, T, tk, pen, mi, t3)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 4: Multi-restart perturbation + optimize (20% time)
    t4 = t0 + budget_s - 60
    for pi in range(100):
        if time.time() > t4:
            break
        hp = do_perturb(h_best, n, rng, pi % 4)
        hn = lbfgs_opt(hp, n, 2, 10, 10000, 3000, t4)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 5: Try upsampled version if time permits
    if time.time() < t0 + budget_s - 40 and n * 2 < 1000:
        hu, c5u = try_upsample(h_best, n, n * 2, t0 + budget_s - 10)
        if hu is not None and c5u < c5_best:
            c5_best, h_best, n = c5u, hu, n * 2

    # Phase 6: Final ultra-fine polish
    if time.time() < budget_s - 5:
        hn = lbfgs_opt(h_best, n, 1, 5, 30000, 8000, budget_s - 2)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    if c5_final > c5_best:
        c5_final = c5_best
    return (h_best, c5_final, n)
```
