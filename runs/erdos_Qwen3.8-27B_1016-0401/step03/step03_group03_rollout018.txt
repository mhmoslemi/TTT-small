import time

_N = 0
_TARGET = 0.0
_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _T) * (2.0 / _N)

def _jac(x):

    center = _N - 1

import time

_N = 0
_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def sg(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += 1.0 - h[j[m1] - shift]
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += 1.0 - h[j[m1] + s]
        g[m2] -= h[j[m2] - s]
    return g

def _f(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    e = np.exp(_T * (corr - mc))
    return (mc + np.log(e.sum()) / _T) * (2.0 / _N)

def _g(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    e = np.exp(_T * (corr - mc))
    w = e / e.sum()
    c = _N - 1
    d = 2.0 / _N
    g = np.zeros(_N)
    for i in range(len(w)):
        if w[i] > 1e-12:
            g += w[i] * sg(x, _N, i - c)
    return g * d

def solve_slsqp(h, n, T, mi, ftol):
    global _N, _T
    _N, _T = n, float(T)
    try:
        r = minimize(_f, h, method='SLSQP', jac=_g,
                     bounds=[(0.0,1.0)]*n,

                                   'jac':lambda x: np.ones(n)}],
                     options={'maxiter':mi,'ftol':ftol})
        return r.x
    except:
        return h.copy()

def nesterov(h0, n, t0, tb, K=6, mi=500000):

    hs, hp = h0.copy(), h0.copy()

        b = it/(it+3.0) if it else 0.0
        hy = np.clip(hs + b*(hs-hp), 0, 1)
        corr = np.correlate(hy, 1.0-hy, mode="full")
        ti = np.argsort(corr)[-K:]
        w = np.exp(5.0*(corr[ti]-corr[ti].max()))
        w /= w.sum()
        g = sum(wi*sg(hy,n,ii-c) for wi,ii in zip(w,ti))*dx
        a = 0.06/(1.0+0.002*it)

        hs = project_feasible(hy - a*g, n)
        if (it+1)%10000 == 0:
            cv = compute_c5(hs)
            if cv < cb: cb, hb = cv, hs.copy()
    cv = compute_c5(hs)
    if cv < cb: cb, hb = cv, hs.copy()
    return hb, cb

def coord_search(h, n, rng, t0, tb, ms=20000):

                    ht = project_feasible(ht, n)
                    ct = compute_c5(ht)
                    if ct < cc: hc, cc = ht, ct; break
    return hc, cc
