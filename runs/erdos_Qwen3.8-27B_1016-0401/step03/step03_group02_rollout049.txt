from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target, tol=1e-12):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < tol:
            break
        # Distribute evenly among elements that can absorb
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-14
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            room = h
            mask = room > 1e-14
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def project(h, target):
    """PROJECT h onto {0<=h<=1, sum(h)=target} via bisection on nu"""
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        nu = (lo + hi) / 2
        h_new = np.clip(h - nu, 0.0, 1.0)
        s = h_new.sum()
        if s > target:
            lo = nu
        else:
            hi = nu
    h_new = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Fix tiny residual
    diff = target - h_new.sum()
    if abs(diff) > 1e-10:
        h_new += diff / len(h_new)
    return np.clip(h_new, 0.0, 1.0)

def smooth_max_obj(x, T, n, target, lam_eq):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    # Numerically stable log-sum-exp
    shifted = T * (corr - mc)
    lse = mc + np.log(np.sum(np.exp(shifted))) / T
    eq_pen = lam_eq * (x.sum() - target) ** 2
    return lse * dx + eq_pen

def smooth_max_grad(x, T, n, target, lam_eq):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    w = e / np.sum(e)
    center = n - 1
    grad = np.zeros(n)
    # For each lag contribution
    for d in range(-(n-1), n):
        wi = w[center + d]
        if wi < 1e-14:

        if d == 0:
            grad[:] += wi * (1.0 - 2.0 * x)
        elif d > 0:
            grad[d:] += wi * (1.0 - x[:n-d])

        else:
            p = -d
            grad[:n-p] += wi * (1.0 - x[p:])
            grad[p:] += -wi * x[:n-p]
    grad *= dx
    # Add equality constraint gradient
    grad += 2 * lam_eq * (x.sum() - target)

def fix_sum(h, target):
    """Fix sum of h to target, maintaining 0<=h<=1. Water-filling projection."""
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h2 = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix tiny residual
    diff = target - h2.sum()
    if abs(diff) > 1e-12:
        # Distribute among non-bound elements
        mask = (h2 > 1e-8) & (h2 < 1.0 - 1e-8)
        if mask.sum() > 0:
            h2[mask] += diff / mask.sum()
        else:
            h2 += diff / len(h2)

def thue_morse(n):
    """Generate Thue-Morse sequence of length n (0 or 1)."""
    h = np.zeros(n)
    h[0] = 0
    for i in range(1, n):
        h[i] = h[i // 2] ^ 1 if i % 2 == 0 else 1 - h[i - 1]
    # Actually the standard definition: t(i) = popcount(i) % 2
    # Let me use the standard
    h = np.zeros(n, dtype=float)
    for i in range(n):
        h[i] = bin(i).count('1') % 2
    return h

def thue_morse_fast(n):
    """Generate Thue-Morse using recursive construction."""
    h = np.array([0.0])
    while len(h) < n:
        h = np.concatenate([h, 1.0 - h])
    return h[:n]

            h2[mask] += diff / mask.sum()
        else:
            h2 += diff / len(h2)
    return np.clip(h2, 0.0, 1.0)

def thue_morse_gen(n):
    h = np.array([0.0])
    while len(h) < n:
        h = np.concatenate([h, 1.0 - h])
    return h[:n]

def sm_obj(x, T, n, lam):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    lse = mc + np.log(np.sum(np.exp(shifted))) / T
    return lse * dx + lam * (x.sum() - n * 0.5) ** 2

def sm_grad(x, T, n, lam):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    w = e / np.sum(e)
    center = n - 1
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - center
        if d == 0:
            grad[:] += wk * (1.0 - 2.0 * x)
        elif d > 0:
            grad[d:] += wk * (1.0 - x[:n - d])
            grad[:n - d] += -wk * x[d:]
        else:
            p = -d
            grad[:n - p] += wk * (1.0 - x[p:])
            grad[p:] += -wk * x[:n - p]
    grad *= dx
    grad += 2.0 * lam * (x.sum() - n * 0.5)
    return grad

def slsqp_multilag(h, n, n_lags, maxiter, ftol):
    target = n / 2.0
    cons = [dict(type='eq', fun=lambda x: x.sum() - target, jac=lambda x: np.ones(n))]
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full")
        top = np.sort(corr)[::-1][:n_lags]
        mc = np.max(corr)
        shifted = 300.0 * (top - mc)
        return (mc + np.log(np.sum(np.exp(shifted))) / 300.0) * dx
    def grad(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        top_idx = np.argsort(corr)[::-1][:n_lags]
        sub = corr[top_idx]
        shifted = 300.0 * (sub - mc)
        e = np.exp(shifted)
        sw = e / np.sum(e)
        center = n - 1
        g = np.zeros(n)
        for ii, k in enumerate(top_idx):
            wk = sw[ii]
            if wk < 1e-14:
                continue
            d = k - center
            if d == 0:
                g[:] += wk * (1.0 - 2.0 * x)
            elif d > 0:
                g[d:] += wk * (1.0 - x[:n - d])
                g[:n - d] += -wk * x[d:]
            else:
                p = -d
                g[:n - p] += wk * (1.0 - x[p:])
                g[p:] += -wk * x[:n - p]
        return g * dx
    res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return np.clip(res.x, 0.0, 1.0)

def coord_descent(h, n_sweeps, max_rounds=3):
    n = len(h)
    target = n / 2.0
    c5_cur = compute_c5(h)
    dts = [0.02, -0.02, 0.01, -0.01, 0.04, -0.04, 0.005, -0.005]
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            for dt in dts:
                trial = h.copy()
                trial[j] = np.clip(trial[j] + dt, 0.0, 1.0)
                trial = fix_sum(trial, target)
                c5_new = compute_c5(trial)
                if c5_new < c5_cur - 1e-14:
                    h = trial
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def pair_swap_refine(h, max_iter=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    c5_cur = compute_c5(h)
    for _ in range(max_iter):
        improved = False
        ones = np.where(h > 0.7)[0]
        zeros = np.where(h < 0.3)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for i in zeros:
            for j in ones:
                if i == j:
                    continue
                trial = h.copy()
                trial[i] = 1.0
                trial[j] = 0.0
                trial = fix_sum(trial, target)
                c5_new = compute_c5(trial)
                if c5_new < c5_cur - 1e-14:
                    h = trial
                    c5_cur = c5_new
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, c5_cur

def optimize_one(h, n, t_end, label=""):
    target = n / 2.0
    h = fix_sum(h.copy(), target)
    c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    # Smooth max SLSQP with increasing T
    for T in [100, 300, 800, 2000, 5000]:
        if time.time() > t_end:
            break
        lam = 50.0
        try:
            res = minimize(sm_obj, h, args=(T, n, lam), jac=sm_grad,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-10})
            h_new = fix_sum(np.clip(res.x, 0, 1), target)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-15:
                c5 = c5_new
                h = h_new
        except Exception:
            pass
    # Multi-lag SLSQP
    for nl in [3, 5, 8]:
        if time.time() > t_end:
            break
        try:
            h_new = slsqp_multilag(h, n, nl, 600, 1e-18)
            h_new = fix_sum(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-15:
                c5 = c5_new
                h = h_new
        except Exception:
            pass
    # Coordinate descent
    if time.time() < t_end - 30:
        h, c5_cd = coord_descent(h, 3)
        if c5_cd < c5:
            c5 = c5_cd
            h = h
    # Pair swap
    if time.time() < t_end - 10:
        h, c5_ps = pair_swap_refine(h, 100)
        if c5_ps < c5:
            c5 = c5_ps
            h = h
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = 1.0
    best_n = 0
    t_final = t0 + budget_s - 5.0

    # Build candidate list
    candidates = []
    # From initial_h_values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        h0 = fix_sum(h0, n0 / 2.0)
        candidates.append((h0.copy(), n0))
    # Thue-Morse at various n
    for n_tm in [256, 320, 384, 448]:
        htm = thue_morse_gen(n_tm)
        htm = fix_sum(htm, n_tm / 2.0)
        c5tm = compute_c5(htm)
        c5tm = 1.0 if np.isnan(c5tm) else c5tm
        if c5tm < 0.45:
            candidates.append((htm.copy(), n_tm))
    # Alternating with slight perturbation
    for n_alt in [280, 340]:
        ha = np.zeros(n_alt)
        ha[::2] = 1.0
        ha = fix_sum(ha, n_alt / 2.0)
        c5a = compute_c5(ha)
        if c5a < 0.5:
            candidates.append((ha.copy(), n_alt))
    # Random smooth initializations
    for n_rand in [256, 320]:
        hr = 0.5 + 0.3 * np.sin(np.linspace(0, 4 * np.pi, n_rand)) + 0.1 * rng.standard_normal(n_rand)
        hr = np.clip(hr, 0, 1)
        hr = fix_sum(hr, n_rand / 2.0)
        c5r = compute_c5(hr)
        if c5r < 0.45:
            candidates.append((hr.copy(), n_rand))

    # Sort candidates by initial C5
    scored = []
    for hc, nc in candidates:
        c = compute_c5(hc)
        if c < best_c5:
            best_c5 = c
            best_h = hc.copy()
            best_n = nc
        scored.append((c, hc, nc))
    scored.sort(key=lambda x: x[0])
    # Deduplicate by n (keep best per n)
    seen_n = set()
    unique_cands = []
    for c, hc, nc in scored:
        if nc not in seen_n:
            seen_n.add(nc)
            unique_cands.append((hc, nc))

    # Optimize each candidate
    n_cands = len(unique_cands)
    per_cand_time = (t_final - t0) / max(n_cands, 1)
    for idx, (hc, nc) in enumerate(unique_cands):
        t_cand_end = min(t0 + (idx + 1) * per_cand_time, t_final)
        if time.time() > t_final:
            break
        h_opt, c5_opt = optimize_one(hc, nc, t_cand_end)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = nc
        # If we have a very good result, spend remaining time polishing
        if best_c5 < 0.381 and idx < n_cands - 1:
            remaining = t_final - time.time()
            if remaining > 120:
                h_opt, c5_opt = optimize_one(best_h, best_n, t_final - 20)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
                    best_n = nc
    # Final polish on best
    if time.time() < t_final - 30:
        h_final, c5_final = optimize_one(best_h, best_n, t_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
            best_n = best_n
    best_c5 = min(best_c5, compute_c5(best_h))
    best_h = fix_sum(best_h, best_n / 2.0)
    best_c5 = min(best_c5, compute_c5(best_h))
    return (best_h, best_c5, len(best_h))
