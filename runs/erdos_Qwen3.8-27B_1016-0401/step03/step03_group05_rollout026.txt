def project_feasible(y, n):

def shift_grad(h, n, shift):

    grad = np.zeros(n)

        m1 = j >= shift
        m2 = (j + shift) < n

        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx

        return x.sum() - target

        return np.ones(n)

        return h.copy()

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad
