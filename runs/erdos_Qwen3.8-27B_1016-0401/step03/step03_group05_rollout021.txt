import numpy as np
from scipy.optimize import minimize
import time

_OPT = {'n': 280, 'T': 1.0, 'K': 5, 'pwr': 12}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    mid = 0.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j < (n - shift)
        m2 = j >= shift
        g[m1] += (1.0 - h[j[m1] + shift])
        g[m2] -= h[j[m2] - shift]
    else:
        s = -shift
        m1 = j >= s
        m2 = j < (n - s)
        g[m1] += (1.0 - h[j[m1] - s])
        g[m2] -= h[j[m2] + s]
    return g

def _obj_sm(x):
    n, T = _OPT['n'], _OPT['T']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * dx

def _jac_sm(x):
    n, T = _OPT['n'], _OPT['T']
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    g = np.zeros(n)
    for oi in range(2 * n - 1):
        w = sw[oi]
        if w < 1e-10:
            continue
        g += w * shift_grad(x, n, oi - center)
    return g * dx

def _obj_tk(x):
    n, K = _OPT['n'], _OPT['K']
    corr = np.correlate(x, 1.0 - x, mode="full")
    return np.max(corr) * (2.0 / n)

def _jac_tk(x):
    n, K, pw = _OPT['n'], _OPT['K'], _OPT['pwr']
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    top_idx = np.argpartition(corr, -K)[-K:]
    tv = corr[top_idx]
    w = (tv / max(mc, 1e-15)) ** pw
    ws = w.sum()
    if ws < 1e-15:
        w = np.ones(K) / K
    else:
        w /= ws
    g = np.zeros(n)
    for i in range(K):
        g += w[i] * shift_grad(x, n, top_idx[i] - center)
    return g * dx

def _cfun(x):
    return x.sum() - _OPT['n'] / 2.0

def _cjac(x):
    return np.ones(_OPT['n'])

def slsqp_run(h, n, obj_fn, jac_fn, maxiter, ftol):
    _OPT['n'] = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', jac=jac_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def mom_desc(h, n, t0, tlimit, kmax=12):
    dx = 2.0 / n
    center = n - 1
    hb = h.copy()
    c5b = compute_c5(h)
    hm, hp = h.copy(), h.copy()
    lr, mu = 0.25, 0.85
    for it in range(400000):
        if time.time() > tlimit:
            break
        frac = min(it / 20000.0, 1.0)
        K = max(1, int(1 + (kmax - 1) * frac * frac))
        hy = np.clip(hm + mu * (hm - hp), 0.0, 1.0)
        corr = np.correlate(hy, 1.0 - hy, mode="full")
        mc = np.max(corr)
        if K <= 1:
            g = shift_grad(hy, n, int(np.argmax(corr)) - center) * dx
        else:
            ti = np.argpartition(corr, -K)[-K:]
            tv = corr[ti]
            w = (tv / max(mc, 1e-15)) ** 10
            ws = w.sum()
            if ws < 1e-15:
                w = np.ones(K) / K
            else:
                w /= ws
            g = np.zeros(n)
            for i in range(K):
                g += w[i] * shift_grad(hy, n, ti[i] - center)
            g *= dx
        hn = project_feasible(hy - lr * g, n)
        c5n = compute_c5(hn)
        if c5n < c5b:
            c5b = c5n
            hb = hn.copy()
            lr = min(lr * 1.03, 0.8)
        else:
            lr = max(lr * 0.92, 0.01)
        hp, hm = hm, hn
    return hb, c5b

    n = len(h_best)

    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP smooth max, decreasing T
    for T in [16, 8, 4, 2, 1]:

        _OPT['T'] = T
        iters = 3000 if T > 2 else 5000
        hx = slsqp_run(h_best, n, _obj_sm, _jac_sm, iters, 1e-18)
        hx = project_feasible(hx, n)
        c5 = compute_c5(hx)
        if c5 < c5_best:
            c5_best, h_best = c5, hx

    # Phase 2: Top-K SLSQP
    for K, pw in [(2, 20), (3, 15), (5, 10), (8, 8)]:
        if time.time() - t0 > budget_s - 400:

        _OPT['K'], _OPT['pwr'] = K, pw

        hx = project_feasible(hx, n)
        c5 = compute_c5(hx)
        if c5 < c5_best:
            c5_best, h_best = c5, hx
    _OPT['pwr'] = 12

    # Phase 3: Momentum descent
    rem = budget_s - (time.time() - t0) - 250

        hx = slsqp_run(h_best, n, _obj_tk, _jac_tk, 3000, 1e-18)

        hx = project_feasible(hx, n)
        c5 = compute_c5(hx)
        if c5 < c5_best:
            c5_best, h_best = c5, hx
    _OPT['pwr'] = 12
