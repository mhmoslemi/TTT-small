def fix_constraint(h, target):

def alt_lp_solve(h, target):
    """Solve one LP iteration of the alternating linearization."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    
    # Build A_ub (n_lags x (n+1)), b_ub, A_eq, b_eq, c_obj, bounds
    A_ub = np.zeros((n_lags, n + 1))
    for idx, o in enumerate(range(-(n-1), n)):
        j_start = max(0, o)
        j_end = min(n, n + o)
        if j_end > j_start:
            js = np.arange(j_start, j_end)
            A_ub[idx, js] = (1.0 - h[js - o]) * dx
        A_ub[idx, n] = -1.0
    
    b_ub = np.zeros(n_lags)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds_lp = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.success:
        return np.clip(res.x[:n], 0, 1)
    return None

def alt_linearize(h, target, n_iter=200):
    n = len(h)
    h_cur = h.copy()
    c5_best = compute_c5(h_cur)
    for it in range(n_iter):
        h_new = alt_lp_solve(h_cur, target)
        if h_new is None:
            break
        h_new = fix_constraint(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_cur = h_new
        else:

            h_damp = 0.5 * h_new + 0.5 * h_cur
            h_damp = np.clip(h_damp, 0, 1)
            h_damp = fix_constraint(h_damp, target)
            c5_d = compute_c5(h_damp)
            if c5_d < c5_best:
                c5_best = c5_d
                h_cur = h_damp
            else:
                break
    return h_cur, c5_best

    h_sorted = np.argsort(-h)

    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]

        corr_old = compute_corr(h_bin)

        delta = max_new - max_old
        if delta < 0 or rng.random() < np.exp(-delta / T):

        else:
            # Reject
            h_bin[i] = 1; h_bin[j] = 0
        
        T = max(T_min, T * 0.999)
    
    return h_bin, c5_cur

def corr_delta_pair(h, i, j):
    """Compute the change in all correlations when flipping h[i]:1->0, h[j]:0->1."""
    n = len(h)
    delta = np.zeros(2*n-1)
    
    # Effect of h[i]: 1->0
    # In corr[m] = sum_l h[l]*(1-h[l+o]) where o = m-(n-1):
    # Term l=i: h[i]*(1-h[i+o]) was 1*(1-h[i+o]), now 0. Change: -(1-h[i+o])
    #   Valid when 0 <= i+o < n, i.e., -i <= o < n-i
    # Term l+o=i, i.e., l=i-o: h[i-o]*(1-h[i]) was h[i-o]*0=0, now h[i-o]*1. Change: +h[i-o]
    #   Valid when 0 <= i-o < n, i.e., i-n < o <= i
    
    for o in range(-(n-1), n):
        d = 0.0
        # l = i term
        if 0 <= i + o < n:
            d -= (1.0 - h[i + o])
        # l = i-o term
        if 0 <= i - o < n:
            d += h[i - o]

        if 0 <= j + o < n:
            d += (1.0 - h[j + o])
        # l = j-o term
        if 0 <= j - o < n:
            d -= h[j - o]
        delta[o + n - 1] += d
    
    return delta

def compute_delta(h, i, j, n):
    # Returns delta array of length 2n-1
    # Effect of h[i]: 1->0
    # Effect of h[j]: 0->1
    
    # For h[i]: 1->0
    # corr contribution: -(1-h[i+o]) for o in range(-i, n-i) -> indices in full corr
    #                    + h[i-o] for o in range(max(-n+1, i-n+1), min(n-1, i)+1)
    
    delta = np.zeros(2*n-1)
    
    # h[i] 1->0: remove +h[i]*(1-h[i+o]) = +(1-h[i+o])
    # o ranges: -i <= o <= n-1-i (so that 0 <= i+o <= n-1)
    o_min = -i
    o_max = n-1-i
    if o_max >= o_min:
        idx = np.arange(o_min, o_max+1) + (n-1)  # index in full corr
        delta[idx] -= (1.0 - h[i+o_min : i+o_max+1])
    
    # h[i] 1->0: add h[i-o]*(1-h[i]) = h[i-o]*1 (was 0 since h[i]=1)
    # o ranges: i-n+1 <= o <= i (so that 0 <= i-o <= n-1)
    o_min2 = max(-(n-1), i-(n-1))
    o_max2 = min(n-1, i)
    if o_max2 >= o_min2:
        idx2 = np.arange(o_min2, o_max2+1) + (n-1)
        # h[i-o] for o in [o_min2, o_max2] -> h[i-o_min2 : i-o_max2+1][::-1]
        # i-o ranges from i-o_min2 down to i-o_max2
        # h[i-o_min2], h[i-o_min2-1], ..., h[i-o_max2]
        vals = h[i-o_max2 : i-o_min2+1][::-1]
        delta[idx2] += vals
    
    # Similarly for h[j]: 0->1
    # Add h[j]*(1-h[j+o]) = 1*(1-h[j+o])
    o_min = -j
    o_max = n-1-j
    if o_max >= o_min:
        idx = np.arange(o_min, o_max+1) + (n-1)
        delta[idx] += (1.0 - h[j+o_min : j+o_max+1])
    
    # Remove h[j-o]*(1-h[j]) = h[j-o]*1 (was h[j-o], now 0 since h[j]=1)
    o_min2 = max(-(n-1), j-(n-1))
    o_max2 = min(n-1, j)
    if o_max2 >= o_min2:

        vals = h[j-o_max2 : j-o_min2+1][::-1]
        delta[idx2] -= vals
    
    return delta

                h[i] -= sub; diff += sub
                if abs(diff) < 1e-14:
                    break
    return h

def to_binary(h, target):
    n = len(h)
    h_bin = np.zeros(n)
    h_bin[np.argsort(-h)[:int(target)]] = 1.0
    return h_bin

def compute_delta_pair(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    o_min = -i; o_max = n - 1 - i
    if o_max >= o_min:
        idx = np.arange(o_min, o_max + 1) + (n - 1)
        delta[idx] -= (1.0 - h[i + o_min:i + o_max + 1])
    o_min2 = max(-(n - 1), i - (n - 1)); o_max2 = min(n - 1, i)
    if o_max2 >= o_min2:
        idx2 = np.arange(o_min2, o_max2 + 1) + (n - 1)
        delta[idx2] += h[i - o_max2:i - o_min2 + 1][::-1]
    o_min = -j; o_max = n - 1 - j
    if o_max >= o_min:
        idx = np.arange(o_min, o_max + 1) + (n - 1)
        delta[idx] += (1.0 - h[j + o_min:j + o_max + 1])
    o_min2 = max(-(n - 1), j - (n - 1)); o_max2 = min(n - 1, j)
    if o_max2 >= o_min2:
        idx2 = np.arange(o_min2, o_max2 + 1) + (n - 1)
        delta[idx2] -= h[j - o_max2:j - o_min2 + 1][::-1]
    return delta

def binary_sa(h, target, n_iter, rng, t_end):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    max_c = float(corr.max())
    c5_cur = max_c * dx
    ones = np.where(h_cur > 0.99)[0].tolist()
    zeros = np.where(h_cur < 0.01)[0].tolist()
    T = max_c * dx * 0.05
    T_min = 1e-7
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        if not ones or not zeros:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        if i == j:
            continue
        delta = compute_delta_pair(h_cur, i, j, n)
        new_max = float((corr + delta).max())
        if new_max <= max_c + 1e-15 or rng.random() < np.exp(-(new_max - max_c) * dx / max(T, 1e-15)):
            h_cur[i] = 0.0; h_cur[j] = 1.0
            corr = corr + delta
            max_c = new_max
            c5_cur = max_c * dx
            ones.remove(i); zeros.remove(j)
            zeros.append(i); ones.append(j)
        T = max(T_min, T * 0.9997)
    return h_cur, c5_cur

def alt_lp(h, target, n_iter, t_end):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    c5_best = compute_c5(h_cur)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        A_ub = np.zeros((2 * n - 1, n + 1))
        for idx in range(2 * n - 1):
            o = idx - (n - 1)
            js = np.arange(max(0, o), min(n, n + o))
            if len(js) > 0:
                A_ub[idx, js] = (1.0 - h_cur[js - o]) * dx
            A_ub[idx, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        c_obj = np.zeros(n + 1); c_obj[n] = 1.0
        bounds_lp = [(0, 1)] * n + [(None, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs',
                          options={'maxiter': 5000})
            if not res.success:
                break
            h_new = fix_constraint(np.clip(res.x[:n], 0, 1), target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new; h_cur = h_new
            else:
                h_d = 0.5 * h_new + 0.5 * h_cur
                h_d = fix_constraint(np.clip(h_d, 0, 1), target)
                c5_d = compute_c5(h_d)
                if c5_d < c5_best:
                    c5_best = c5_d; h_cur = h_d
                else:
                    break
        except Exception:
            break
    return h_cur, c5_best

def _TARGET:
    pass

_target_val = 0.0

def eq_fun(x):
    return x.sum() - _target_val

def eq_jac(x):
    return np.ones(len(x))

def make_cons():
    return [dict(type='eq', fun=eq_fun, jac=eq_jac)]

def lag_obj(x, lag_idx):
    n = len(x)
    return float(compute_corr(x)[lag_idx] * (2.0 / n))

def lag_grad(x, lag_idx):
    n = len(x)
    o = lag_idx - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * x
    elif o > 0:
        g[o:] += 1.0 - x[:n - o]
        g[:n - o] += -x[o:]
    else:
        p = -o
        g[:n - p] += 1.0 - x[p:]
        g[p:] += -x[:n - p]
    return g * dx

def slsqp_lag(h, lag_idx, maxiter):
    global _target_val
    _target_val = len(h) / 2.0
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lag_obj, h, args=(lag_idx,), method='SLSQP',
                       jac=lag_grad, bounds=bounds,
                       constraints=make_cons(),
                       options={'maxiter': maxiter, 'ftol': 1e-17})
        return fix_constraint(np.clip(res.x, 0, 1), len(h) / 2.0)
    except Exception:
        return h

def slsqp_sm_obj(x, T):
    n = len(x)
    corr = compute_corr(x)
    mc = float(corr.max())
    shifted = T * (corr - mc)
    shifted -= shifted.max()
    return (mc + np.log(np.exp(shifted).sum()) / T) * (2.0 / n)

def slsqp_sm_grad(x, T):
    n = len(x)
    corr = compute_corr(x)
    mc = float(corr.max())
    shifted = T * (corr - mc)
    shifted -= shifted.max()
    sw = np.exp(shifted) / np.exp(shifted).sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx

def slsqp_sm(h, T, maxiter):
    global _target_val
    _target_val = len(h) / 2.0
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(slsqp_sm_obj, h, args=(T,), method='SLSQP',
                       jac=slsqp_sm_grad, bounds=bounds,
                       constraints=make_cons(),
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return fix_constraint(np.clip(res.x, 0, 1), len(h) / 2.0)
    except Exception:
        return h

def coord_descent(h, n_sweeps):
    n = len(h)
    target = n / 2.0
    dts = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.015, -0.015]
    c5_cur = compute_c5(h)
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            for dt in dts:
                trial = h.copy()
                trial[j] = np.clip(trial[j] + dt, 0.0, 1.0)
                trial = fix_constraint(trial, target)
                c5_new = compute_c5(trial)
                if c5_new < c5_cur - 1e-15:
                    h = trial; c5_cur = c5_new; improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def full_pipeline(h_start, t0, t_end, seed=42):
    global _target_val
    rng = np.random.default_rng(seed)
    n = len(h_start)
    target = n / 2.0
    _target_val = target
    h_best = fix_constraint(np.clip(h_start, 0, 1), target)
    c5_best = compute_c5(h_best)
    t_end_save = t_end

    # Binary round + SA
    h_bin = to_binary(h_best, target)
    c5_bin = compute_c5(h_bin)
    if c5_bin <= c5_best:
        h_best = h_bin; c5_best = c5_bin
    h_sa, c5_sa = binary_sa(h_best, target, 80000, rng, t0 + (t_end_save - t0) * 0.25)
    if c5_sa < c5_best:
        h_best = h_sa; c5_best = c5_sa

    # Alternating LP
    h_lp, c5_lp = alt_lp(h_best, target, 80, t0 + (t_end_save - t0) * 0.45)
    if c5_lp < c5_best:
        h_best = h_lp; c5_best = c5_lp

    # More SA after LP
    h_bin2 = to_binary(h_best, target)
    h_sa2, c5_sa2 = binary_sa(h_bin2, target, 60000, rng, t0 + (t_end_save - t0) * 0.60)
    if c5_sa2 < c5_best:
        h_best = h_sa2; c5_best = c5_sa2

    # SLSQP on worst lag
    corr_w = compute_corr(h_best)
    worst = int(corr_w.argmax())
    h_sl = slsqp_lag(h_best, worst, 800)
    c5_sl = compute_c5(h_sl)
    if c5_sl < c5_best:
        h_best = h_sl; c5_best = c5_sl

    # Soft-max SLSQP
    h_sm = slsqp_sm(h_best, 3000, 2000)
    c5_sm = compute_c5(h_sm)
    if c5_sm < c5_best:
        h_best = h_sm; c5_best = c5_sm

    # Coordinate descent
    h_cd, c5_cd = coord_descent(h_best, 3)
    if c5_cd < c5_best:
        h_best = h_cd; c5_best = c5_cd

    # Final SLSQP
    if time.time() < t_end_save - 5:
        corr_f = compute_corr(h_best)
        h_f = slsqp_lag(h_best, int(corr_f.argmax()), 1000)
        c5_f = compute_c5(h_f)
        if c5_f < c5_best:
            h_best = h_f; c5_best = c5_f

    c5_final = min(compute_c5(h_best), c5_best)
    return h_best, c5_final

def run(seed=42, budget_s=1100, **kwargs):
    global _target_val
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(340)
        h_init[::2] = 1.0
    n = len(h_init)
    target = n / 2.0
    _target_val = target
    h_init = fix_constraint(np.clip(h_init, 0, 1), target)

    h_best = h_init.copy()
    c5_best = compute_c5(h_init)

    # Main pipeline on initial n
    h1, c5_1 = full_pipeline(h_init, t0, min(t_end - 30, t0 + budget_s * 0.6), seed)
    if c5_1 < c5_best:
        h_best = h1; c5_best = c5_1

    # Try a few alternative n values
    remaining = t_end - time.time()
    n_alts = [280, 320, 360, 380]
    for n_a in n_alts:
        if time.time() > t_end - 25:
            break
        t_pass = min(t_end - 10, time.time() + max(remaining * 0.3, 40))
        x_old = np.arange(len(h_best)) * (2.0 / len(h_best))
        x_new = np.linspace(0.0, 2.0, n_a, endpoint=False)
        h_a = np.interp(x_new, x_old, h_best)
        h_a = fix_constraint(h_a, n_a / 2.0)
        h_a2, c5_a = full_pipeline(h_a, time.time(), t_pass, seed + n_a)
        if c5_a < c5_best:
            c5_best = c5_a; h_best = h_a2

    # Final polish
    if time.time() < t_end - 5:
        h_best, c5_pol = coord_descent(h_best, 2)
        if c5_pol < c5_best:
            c5_best = c5_pol
        corr_p = compute_corr(h_best)
        h_pol = slsqp_lag(h_best, int(corr_p.argmax()), 600)
        c5_pol2 = compute_c5(h_pol)
        if c5_pol2 < c5_best:
            h_best = h_pol; c5_best = c5_pol2

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
