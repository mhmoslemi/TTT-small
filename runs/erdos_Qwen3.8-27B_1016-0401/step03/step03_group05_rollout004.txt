import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    idx = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = idx >= shift
        m2 = (idx + shift) < n
        grad[m1] += (1.0 - h[idx[m1] - shift])
        grad[m2] -= h[idx[m2] + shift]
    else:
        s = -shift
        m1 = (idx + s) < n
        m2 = idx >= s
        grad[m1] += (1.0 - h[idx[m1] + s])
        grad[m2] -= h[idx[m2] - s]
    return grad

def grad_topk(h, n, K, power):
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    if K <= 1:
        idx = int(np.argmax(corr))
        return shift_grad(h, n, idx - center) * dx, mc * dx
    top_idx = np.argpartition(corr, -K)[-K:]
    top_vals = corr[top_idx]
    w = (top_vals / max(mc, 1e-15)) ** power
    ws = w.sum()
    if ws < 1e-15:
        w = np.ones(K) / K
    else:
        w /= ws
    g = np.zeros(n)
    for i in range(K):
        g += w[i] * shift_grad(h, n, top_idx[i] - center)
    return g * dx, mc * dx

def nesterov_phase(h_init, n, t0, budget_s, max_iters, K_lo, K_hi, alpha0, decay, power):
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    for it in range(max_iters):
        if time.time() - t0 > budget_s:
            break
        frac = min(float(it + 1) / max_iters, 1.0)
        K = max(K_lo, int(K_lo + (K_hi - K_lo) * frac ** 1.2))
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        g, _ = grad_topk(h_y, n, K, power)
        alpha = alpha0 / (1.0 + decay * it)
        h_new = project_feasible(h_y - alpha * g, n)
        if (it + 1) % 20000 == 0:
            h_prev = h_sg.copy()
        h_prev = h_sg
        h_sg = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 4000 == 0:
            c5a = compute_c5(project_feasible(h_avg, n))
            if c5a < c5_best:
                c5_best = c5a
                h_best = project_feasible(h_avg, n)
            c5s = compute_c5(h_sg)
            if c5s < c5_best:
                c5_best = c5s
                h_best = h_sg.copy()
    c5s = compute_c5(h_sg)
    if c5s < c5_best:

        h_best = h_sg.copy()
    return h_best, c5_best

def slsqp_smooth(h, n, T, maxiter):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(n_corr):
            w = sw[oi]
            if w < 1e-12:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return grad * dx
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-18})
        return res.x
    except Exception:
        return h.copy()

def slsqp_topk(h, n, K, maxiter):

        if ws < 1e-15:
            w = np.ones(K) / K
        else:
            w /= ws
        grad = np.zeros(n)

        return grad * dx
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-18})
        return res.x
    except Exception:
        return h.copy()
