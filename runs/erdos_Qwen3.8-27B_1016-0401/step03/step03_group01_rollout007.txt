        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        lo, hi = (mid, hi) if s > target else (lo, mid)
    return np.clip(y - (lo+hi)/2, 0.0, 1.0)

def linearized_coeff(h0, shift, n):
    """Compute coefficient vector for h in the linearized correlation at given shift."""
    coeff = np.zeros(n)
    const = 0.0
    if shift >= 0:
        ns = n - shift  # number of valid pairs
        # Term 1: sum_{j=shift}^{n-1} h[j] * (1-h0[j-shift])
        coeff[shift:] += (1.0 - h0[:ns])
        # Term 2: -sum_{j=0}^{ns-1} h[j] * h0[j+shift]
        coeff[:ns] -= h0[shift:]
        # Constant: sum_{i=0}^{ns-1} h0[i+shift]*h0[i]
        const = np.sum(h0[shift:] * h0[:ns])
    else:
        s = -shift
        ns = n - s
        # By symmetry, shift -s is same as shift s with h and (1-h) swapped? No.
        # For shift < 0, let s = -shift > 0
        # f(-s) = sum_i h[i-s]*(1-h[i]) for i=s to n-1
        # = sum_{j=0}^{n-1-s} h[j]*(1-h[j+s])
        # Linearization at h0:
        # sum h0[j]*(1-h[j+s]) + (h[j]-h0[j])*(1-h0[j+s])
        # = sum h0[j] - sum h0[j]*h[j+s] + sum h[j]*(1-h0[j+s]) - sum h0[j]*(1-h0[j+s])
        # = sum h[j]*(1-h0[j+s]) - sum h[j+s]*h0[j] + sum h0[j]*h0[j+s]
        # Term 1: sum_{j=0}^{ns-1} h[j]*(1-h0[j+s])
        coeff[:ns] += (1.0 - h0[s:])
        # Term 2: -sum_{j=0}^{ns-1} h[j+s]*h0[j]
        coeff[s:] -= h0[:ns]
        const = np.sum(h0[:ns] * h0[s:])
    return coeff, const

Wait, I need to double-check the correlation definition. Let me verify:

np.correlate(a, v, mode="full") where both have length n:
result has length 2n-1
result[k] for k=0,...,2n-2

The definition: result[k] = sum_i a[i+k-(n-1)] * conj(v[i])
where the sum is over all i where both indices are valid (0 ≤ i < n, 0 ≤ i+k-(n-1) < n)

So for k = n-1+m (where m is the "lag" from -(n-1) to n-1):
result[n-1+m] = sum_i a[i+m] * v[i] for valid i

With a=h, v=1-h:
result[n-1+m] = sum_{i: 0≤i≤n-1 AND 0≤i+m≤n-1} h[i+m] * (1-h[i])

For m ≥ 0: i ranges from 0 to n-1-m
For m < 0 (let s=-m): i ranges from s to n-1, equivalently j=i-s ranges from 0 to n-1-s
  = sum_{j=0}^{n-1-s} h[j] * (1-h[j+s])

So:
- For shift m ≥ 0: f(m) = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
- For shift m < 0, s=-m: f(-s) = sum_{j=0}^{n-1-s} h[j]*(1-h[j+s])

Note: f(m) for m≥0 and f(-m) for m>0 are related:
f(m) = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])  [h is 1-shifted forward, 1-h is at base]
f(-m) = sum_{j=0}^{n-1-m} h[j]*(1-h[j+m])  [h is at base, 1-h is m-shifted forward]

These are NOT the same. f(m) counts positions where h is 1 at position i+m and 0 at position i.
f(-m) counts positions where h is 1 at position i and 0 at position i+m.

Both are important.

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def linearized_coeff(h0, shift, n):
    coeff = np.zeros(n)
    if shift >= 0:
        ns = n - shift
        coeff[shift:] += (1.0 - h0[:ns])
        coeff[:ns] -= h0[shift:]
        d = float(np.sum(h0[shift:] * h0[:ns]))
    else:
        s = -shift
        ns = n - s
        coeff[:ns] += (1.0 - h0[s:])
        coeff[s:] -= h0[:ns]
        d = float(np.sum(h0[:ns] * h0[s:]))
    return coeff, d

def solve_lp(h0, n, n_lags=50, solver=None):
    h = cp.Variable(n)
    t = cp.Variable()
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h0, 1.0 - h0, mode="full")
    top_lags = np.argsort(corr)[-n_lags:]
    constraints = [h >= 0, h <= 1, cp.sum(h) * dx == 1.0]
    for li in top_lags:
        shift = int(li) - center
        coeff, d = linearized_coeff(h0, shift, n)
        constraints.append((coeff @ h + d) * dx <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    if solver is None:
        solver = cp.ECOS
    try:
        prob.solve(solver=solver, max_iters=10000, tol_abs=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=10000)
        except Exception:
            return h0.copy()
    if h.value is None:
        return h0.copy()
    return np.clip(h.value, 0.0, 1.0)

def nesterov_polish(h_init, n, t_end, n_iters=50000):
    dx = 2.0 / n
    h_y = h_init.copy()
    h_z = h_init.copy()
    best = h_init.copy()
    best_c5 = compute_c5(h_init)
    t_mom = 1.0
    lr = 0.05
    corr_full = np.zeros(2*n-1)
    center = n - 1
    j_arr = np.arange(n)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        # Softmax weights over all lags
        T = 2.0
        cs = corr - corr.max()
        exps = np.exp(cs / T * (2*n - 1) * dx)  # scale
        # Actually let's use a simpler temperature
        T2 = 3.0
        exps = np.exp((corr - corr.max()) / T2)
        w = exps / exps.sum()
        # Gradient: weighted sum of gradients at each lag
        grad = np.zeros(n)
        for m in range(2*n-1):
            wi = w[m]
            if wi < 1e-8:
                continue
            shift = m - center
            if shift >= 0:
                m1 = j_arr >= shift
                grad[m1] += wi * (1.0 - h_y[j_arr[m1] - shift])
                m2 = (j_arr + shift) < n
                grad[m2] -= wi * h_y[j_arr[m2] + shift]
            else:
                s2 = -shift
                m1 = (j_arr + s2) < n
                grad[m1] += wi * (1.0 - h_y[j_arr[m1] + s2])
                m2 = j_arr >= s2
                grad[m2] -= wi * h_y[j_arr[m2] - s2]
        grad *= dx
        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2)) / 2.0
        beta = (t_mom - 1.0) / t_new
        h_z = h_y + beta * (h_y - h_z)
        h_y = project_feasible(h_z - lr * grad, n)
        t_mom = t_new
        lr *= 0.9998
        if (it+1) % 1000 == 0:
            c5 = compute_c5(h_y)
            if c5 < best_c5:
                best_c5 = c5
                best = h_y.copy()
    return best, best_c5

Let me optimize: instead of computing the full correlation after each flip, compute the delta:

When we flip h[pos] from 0 to 1 (or 1 to 0), the change in correlation at lag m is:
- For m ≥ 0 (shift = m): 
  - h[pos+m] term: if pos+m < n, the term h[pos+m]*(1-h[pos]) changes by h[pos+m]*(h[pos]_new - h[pos]_old) ... wait
  - Actually: old contribution = h[pos+m]*(1-h[pos]) if 0≤pos+m<n and 0≤pos<n
  - new contribution = h[pos+m]*(1-h[pos]_new)
  - delta = h[pos+m]*((1-h[pos]_new) - (1-h[pos])) = h[pos+m]*(h[pos] - h[pos]_new)
  - If flipping 0→1: delta = h[pos+m]*(0-1) = -h[pos+m]
  - If flipping 1→0: delta = h[pos+m]*(1-0) = h[pos+m]
  
  - Also h[pos] term: if pos-m ≥ 0 (i.e., pos is the "h[i+m]" part with i=pos-m)
  - Wait, for lag m≥0: corr[m] = Σ_{i=0}^{n-1-m} h[i+m]*(1-h[i])
  - h[pos] appears as:
    - h[pos] in the "h[i+m]" part when i+m=pos, i=pos-m, valid if 0≤pos-m≤n-1-m, i.e., m≤pos≤n-1
    - h[pos] in the "(1-h[i])" part when i=pos, valid if 0≤pos≤n-1-m, i.e., 0≤pos≤n-1-m
  - So delta for flipping h[pos] by δ (=+1 or -1):
    - From "h[i+m]" part (if m≤pos≤n-1): δ*(1-h[pos-m])
    - From "(1-h[i])" part (if pos≤n-1-m): -δ*h[pos+m]
    - Total: δ*(1-h[pos-m])*[m≤pos≤n-1] - δ*h[pos+m]*[0≤pos≤n-1-m]

This is O(1) per lag per flip! But we have 2n-1 lags, so O(n) per flip. With 100 flips per round, O(n²) per round. For n=280, that's 78400 per round. Much faster.

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def linearized_coeff(h0, shift, n):
    coeff = np.zeros(n)
    if shift >= 0:
        ns = n - shift
        if ns > 0:
            coeff[shift:] += (1.0 - h0[:ns])
            coeff[:ns] -= h0[shift:]
            d = float(np.sum(h0[shift:] * h0[:ns]))
        else:
            d = 0.0
    else:
        s = -shift
        ns = n - s
        if ns > 0:
            coeff[:ns] += (1.0 - h0[s:])
            coeff[s:] -= h0[:ns]
            d = float(np.sum(h0[:ns] * h0[s:]))
        else:
            d = 0.0
    return coeff, d

def solve_lp(h0, n, n_lags=50, trust=0.5):
    h = cp.Variable(n)
    t = cp.Variable()
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h0, 1.0 - h0, mode="full")
    top_lags = np.argsort(corr)[-n_lags:]
    constraints = [h >= 0, h <= 1, cp.sum(h) * dx == 1.0,
                   h >= h0 - trust, h <= h0 + trust]
    for li in top_lags:
        shift = int(li) - center
        coeff, d = linearized_coeff(h0, shift, n)
        constraints.append((coeff @ h + d) * dx <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    for solver in [cp.ECOS, cp.PDLP, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, max_iters=20000, tol_abs=1e-10)
            break
        except Exception:
            continue
    if h.value is None:
        return h0.copy()
    return np.clip(h.value, 0.0, 1.0)

def nesterov_polish(h_init, n, t_end, n_top=8, lr0=0.04):
    dx = 2.0 / n
    h_y = h_init.copy()
    h_z = h_init.copy()
    best = h_init.copy()
    best_c5 = compute_c5(h_init)
    t_mom = 1.0
    lr = lr0
    center = n - 1
    ja = np.arange(n)
    for it in range(100000):
        if time.time() > t_end:
            break
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top_idx = np.argpartition(corr, -n_top)[-n_top:]
        cv = corr[top_idx]
        w = np.exp((cv - cv.max()) / 1.0)
        w /= w.sum()
        grad = np.zeros(n)
        for oi in range(n_top):
            wi = w[oi]
            if wi < 1e-7:
                continue
            shift = int(top_idx[oi]) - center
            if shift >= 0:
                m1 = ja >= shift
                grad[m1] += wi * (1.0 - h_y[ja[m1] - shift])
                m2 = (ja + shift) < n
                grad[m2] -= wi * h_y[ja[m2] + shift]
            else:
                s2 = -shift
                m1 = (ja + s2) < n
                grad[m1] += wi * (1.0 - h_y[ja[m1] + s2])
                m2 = ja >= s2
                grad[m2] -= wi * h_y[ja[m2] - s2]
        grad *= dx
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        h_z = h_y + beta * (h_y - h_z)
        h_y = project_feasible(h_z - lr * grad, n)
        t_mom = t_new
        lr *= 0.99985
        if (it + 1) % 3000 == 0:
            c5 = compute_c5(h_y)
            if c5 < best_c5:
                best_c5 = c5
                best = h_y.copy()
    c5 = compute_c5(h_y)
    if c5 < best_c5:
        best_c5 = c5
        best = h_y.copy()
    return best, best_c5

def slsqp_smooth(h, n, T, k, maxiter=4000):
    def topk_obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mk = min(k, len(corr))
        idx = np.argpartition(corr, -mk)[-mk:]
        topk = corr[idx]
        mc = topk.max()
        sh = T * (topk - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        obj = (mc + np.log(e.sum()) / T) * (2.0 / n)
        grad = np.zeros(n)
        center = n - 1
        j = np.arange(n)
        for oi in range(mk):
            ww = sw[oi]
            if ww < 1e-11:
                continue
            shift = int(idx[oi]) - center
            if shift >= 0:
                m1 = j >= shift
                grad[m1] += ww * (1.0 - x[j[m1] - shift])
                m2 = (j + shift) < n
                grad[m2] -= ww * x[j[m2] + shift]
            else:
                s2 = -shift
                m1 = (j + s2) < n
                grad[m1] += ww * (1.0 - x[j[m1] + s2])
                m2 = j >= s2
                grad[m2] -= ww * x[j[m2] - s2]
        grad *= 2.0 / n
        return obj, grad
    def obj(x):
        return topk_obj(x)[0]
    def jac(x):
        return topk_obj(x)[1]
    res = minimize(obj, h, method='SLSQP', jac=jac,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                                 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return res.x

def binary_flip_search(h_init, n, t_end, max_rounds=800):

    rng = np.random.default_rng(42)
    for rd in range(max_rounds):

        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        rem2 = budget_s - (time.time() - t0)
        if rem2 > 30:
            h_nf, c5_nf = nesterov_polish(h_best, n, t0 + rem2 * 0.55, n_top=12, lr0=0.02)
            h_nf = project_feasible(h_nf, n)
            c5_nf = compute_c5(h_nf)
            if c5_nf < c5_best:
                c5_best = c5_nf
                h_best = h_nf

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
