        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse-to-fine
    temps = [48, 24, 12, 6, 3, 1.5, 0.8, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08]
    for T in temps:
        if time.time() - t0 > budget_s - 480:
            break
        maxit = 5000 if T > 1 else 7000
        hn = slsqp_solv(h_best, n, T, maxit)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best = c5
            h_best = hn

    # Phase 2: Nesterov
    deadline2 = t0 + budget_s - 350
    h_sg, c5_sg = nesterov(h_best, n, deadline2, K=10)
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg
    else:
        h_best = project_feasible(h_best, n)

    # Phase 3: Fine SLSQP
    for T in [0.6, 0.4, 0.28, 0.2, 0.14, 0.1, 0.07]:
        if time.time() - t0 > budget_s - 220:
            break
        hn = slsqp_solv(h_best, n, T, 9000)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best = c5
            h_best = hn

    # Phase 4: Targeted local search + SLSQP interleave
    if time.time() - t0 < budget_s - 140:
        h_ls, c5_ls = targeted_local(h_best, n, rng, steps=6000)
        if c5_ls < c5_best:
            c5_best = c5_ls
            h_best = h_ls
        hn = slsqp_solv(h_best, n, 0.25, 8000)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best = c5
            h_best = hn

    # Phase 5: Perturbation restarts
    for pi in range(18):
        if time.time() - t0 > budget_s - 25:
            break
        sigma = 0.0015 + 0.004 * (pi % 4)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        hn = slsqp_solv(h_p, n, 1.0, 4000)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best = c5
            h_best = hn

    # Phase 6: Final local + polish
    if time.time() - t0 < budget_s - 15:
        h_ls, c5_ls = targeted_local(h_best, n, rng, steps=4000)
        if c5_ls < c5_best:
            c5_best = c5_ls
            h_best = h_ls
        hn = slsqp_solv(h_best, n, 0.15, 12000)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best = c5
            h_best = hn

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
