import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def sg(h, n, shift):
    grad = np.zeros(n)
    if shift >= 0:
        i2 = np.arange(0, n - shift)
        i1 = np.arange(shift, n)
        grad[i2] += 1.0 - h[i2 + shift]
        grad[i1] -= h[i1 - shift]
    else:
        s = -shift
        i1 = np.arange(0, n - s)
        i2 = np.arange(s, n)
        grad[i1] += 1.0 - h[i1 + s]
        grad[i2] -= h[i2 - s]
    return grad

def do_slsq(h, n, T, maxiter, ftol=1e-15):
    dx, ctr = 2.0 / n, n - 1
    def f(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        mc = c.max()
        return (mc + np.log(np.exp(T * (c - mc)).sum()) / T) * dx
    def g(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        mc = c.max()
        e = np.exp(T * (c - mc))
        w = e / e.sum()
        gr = np.zeros(n)
        for i in range(len(w)):
            if w[i] > 1e-10:
                gr += w[i] * sg(x, n, i - ctr)
        return gr * dx
    def cf(x):
        return x.sum() - n / 2.0
    def cj(x):
        return np.ones(n)
    r = minimize(f, h, method='SLSQP', jac=g,
                 bounds=[(0.0, 1.0)] * n,
                 constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                 options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 50})
    return r.x

def nesterov_phase(h, n, t_end, K=6, steps=400000):
    dx, ctr = 2.0 / n, n - 1

    hc, hp = h.copy(), h.copy()

        if time.time() > t_end:
            break
        a = 0.035 / (1.0 + 0.0004 * it)
        b = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) * a * a)) / (2.0 * (it + 1) + 1.0)
        hy = np.clip(hc + b * (hc - hp), 0.0, 1.0)
        c = np.correlate(hy, 1.0 - hy, mode="full")
        tk = np.argsort(c)[-K:]
        mv = c[tk].max()
        Tw = max(0.8, 4.0 - 2.5 * it / max(1, steps))
        w = np.exp(Tw * (c[tk] - mv))
        w /= w.sum()
        g = np.zeros(n)
        for wi, ki in zip(w, tk):
            g += wi * sg(hy, n, ki - ctr)
        g *= dx
        ht = project_feasible(hy - a * g, n)
        ct = compute_c5(ht)
        if ct < best_c:
            best_c, best_h = ct, ht.copy()
        saT = 0.0004 * np.exp(-0.4 * it / max(1, steps))
        if ct <= best_c or rng.random() < np.exp(-(ct - best_c) / max(saT, 1e-9)):
            hp, hc = hc, ht

            hp = hc
            pz = rng.integers(0, n, size=4)
            hc[pz] += 0.003 * rng.standard_normal(4)
            hc = project_feasible(hc, n)
    return best_h, best_c
