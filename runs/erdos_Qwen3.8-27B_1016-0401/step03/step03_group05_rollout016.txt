        K = max(3, min(K, 25))
        if t <= 1:
            h_y = h_sg.copy()
        else:
            beta = (t - 1.0) / (t + 1.0)
            h_y = h_sg + beta * (h_sg - h_prev)
            np.clip(h_y, 0.0, 1.0, out=h_y)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        top_idx = np.argpartition(corr, -K)[-K:]
        tv = corr[top_idx]
        w = (tv / max(mc, 1e-15)) ** pwr
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(K) / K
        else:
            w /= ws
        g = np.zeros(n)
        for i in range(K):
            g += w[i] * shift_grad(h_y, n, top_idx[i] - center)
        g *= dx
        alpha = alpha0 / (1.0 + decay * t)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        if t % 2500 == 0:
            c5c = compute_c5(h_sg)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h_sg.copy()
            if t % 50000 == 0 and time.time() < deadline - 40:
                h_r = slsqp_topk(h_sg, n, 5, 1000, 1e-15, 4.0)
                h_r = project_feasible(h_r, n)
                c5r = compute_c5(h_r)
                if c5r < c5_best:
                    c5_best = c5r
                    h_best = h_r.copy()
                h_sg = h_r
                h_prev = h_r.copy()
    c5f = compute_c5(h_sg)
    if c5f < c5_best:
        c5_best = c5f
        h_best = h_sg.copy()
    return h_best, c5_best

def coord_refine(h, n, deadline, n_outer=800):
    h_best = h.copy()
    c5_best = compute_c5(h)
    center = n - 1
    rng = np.random.default_rng(12345)
    for _ in range(n_outer):
        if time.time() > deadline:
            break
        idx = int(rng.integers(n))
        for delta in [0.002, -0.002, 0.005, -0.005]:
            h_try = h_best.copy()
            h_try[idx] = np.clip(h_try[idx] + delta, 0.0, 1.0)
            h_try = project_feasible(h_try, n)
            c5t = compute_c5(h_try)
            if c5t < c5_best - 1e-9:
                c5_best = c5t
                h_best = h_try
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n = 280
        h_best = np.full(n, 0.5)
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP smooth
    for T in [10, 5, 3]:
        if time.time() - t0 > budget_s - 450:
            break
        h_new = slsqp_smooth(h_best, n, T, 2500, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Alternative starts from structured patterns
    x = np.arange(n) * 2.0 / n
    alts = []
    h_alt = np.zeros(n); h_alt[::2] = 1.0
    alts.append(project_feasible(h_alt, n))
    h_alt2 = np.clip(0.5 + 0.3 * np.sin(np.pi * x), 0, 1)
    alts.append(project_feasible(h_alt2, n))
    h_alt3 = np.clip(0.5 + 0.2 * np.cos(2.0 * np.pi * x + 0.5), 0, 1)
    alts.append(project_feasible(h_alt3, n))
    for ha in alts:
        if time.time() - t0 > budget_s - 350:
            break
        h_new = slsqp_topk(ha, n, 5, 1500, 1e-15, 4.0)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 3: Nesterov with periodic SLSQP
    dl3 = t0 + budget_s - 260
    if time.time() < dl3:
        h_n, c5_n = nesterov_run(h_best, n, dl3, 500000, alpha0=1.0, decay=0.003, pwr=5.0)
        if c5_n < c5_best:
            c5_best = c5_n
            h_best = h_n
        h_best = project_feasible(h_best, n)

    # Phase 4: SLSQP top-K sweep
    for K in [3, 5, 8, 12, 18]:
        if time.time() - t0 > budget_s - 120:
            break
        h_new = slsqp_topk(h_best, n, K, 2500, 1e-16, 4.0)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Second Nesterov pass
    dl5 = t0 + budget_s - 80
    if time.time() < dl5:
        h_n2, c5_n2 = nesterov_run(h_best, n, dl5, 200000, alpha0=0.7, decay=0.005, pwr=6.0)
        if c5_n2 < c5_best:
            c5_best = c5_n2
            h_best = h_n2
        h_best = project_feasible(h_best, n)

    # Phase 6: Final SLSQP + smooth
    if time.time() - t0 < budget_s - 40:
        h_new = slsqp_topk(h_best, n, 5, 2000, 1e-16, 3.0)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    if time.time() - t0 < budget_s - 15:
        h_new = slsqp_smooth(h_best, n, 2, 1500, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
