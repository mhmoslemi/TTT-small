def slsqp_smooth(h, n, T, maxiter, ftol, t0, budget):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(n_corr):
            w = sw[oi]
            if w < 1e-11:
                continue
            grad += w * shift_grad_single(x, n, oi - center)
        return grad * dx
    def cons_fun(x):
        return x.sum() - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def cvxpy_lp(h, n, t0, budget, max_iters=30):
    target = n / 2.0
    dx = 2.0 / n

    n_corr = 2 * n - 1

        # where c_i = 1 - h_cur[i+s] (fixed) for the i term, and -h_cur[i-s] for the other
        # Actually the LP is: min t s.t. for all s: sum_i [grad_s[i]] * h[i] <= t

        # corr[s] = sum_i h[i]*(1-h[i+s]) [valid] - but this isn't linear in h
        # For the LP relaxation: fix the "other" factor
        # corr[s] ≈ sum_i a_i * h[i] where a_i = (1-h_cur[i+s]) - h_cur[i-s] (the gradient)
        # This is the linearization at h_cur
        A_rows = []

            row = np.zeros(n)
            j = np.arange(n)

                row[m1] += (1.0 - h_cur[j[m1] - t])
                m2 = j + t < n
                row[m2] -= h_cur[j[m2] + t]
            A_rows.append(row)
        A = np.array(A_rows) * dx
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]

        g = np.zeros(n)
        for i in range(len(top_idx)):
            g += w[i] * shift_grad_single(h_y, n, top_idx[i] - center)
        g *= dx
        alpha = alpha0 / (1.0 + 0.005 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 5000 == 0:
            c5 = compute_c5(h_sg)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_sg.copy()
    c5 = compute_c5(h_sg)
    if c5 < c5_best:
        c5_best = c5
        h_best = h_sg.copy()
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    elif 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n_init = 280
        h_best = np.zeros(n_init)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP smooth (quick start)
    for T in [12, 6]:
        if time.time() - t0 > budget_s - 600:
            break
        h_new = slsqp_smooth(h_best, n, T, 3000, 1e-15, t0, budget_s - 600)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: FISTA with sum-of-squares
    if time.time() - t0 < budget_s - 500:
        h_f, c5_f = fista_adaptive(h_best, n, t0, budget_s - 500, 300000)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f

    # Phase 3: Binary local search
    if time.time() - t0 < budget_s - 300:
        h_bin, c5_bin = binary_local_search(h_best, n, t0, budget_s - 300, 200)
        if c5_bin < c5_best:
            c5_best = c5_bin
            h_best = h_bin
        # Un-binary: interpolate and re-optimize
        h_soft = h_bin.astype(float)
        # Smooth slightly
        from scipy.ndimage import uniform_filter1d
        h_soft = uniform_filter1d(h_soft, size=3, mode='wrap')
        h_soft = project_feasible(h_soft, n)
        h_new = slsqp_smooth(h_soft, n, 4, 3000, 1e-15, t0, budget_s - 250)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Nesterov top-K
    if time.time() - t0 < budget_s - 200:
        h_ng, c5_ng = nesterov_topk(h_best, n, t0, budget_s - 200, 300000)
        if c5_ng < c5_best:
            c5_best = c5_ng
            h_best = h_ng

    # Phase 5: CVXPY LP relaxation
    if time.time() - t0 < budget_s - 120:
        h_lp, c5_lp = cvxpy_lp(h_best, n, t0, budget_s - 120, 20)
        if c5_lp < c5_best:
            c5_best = c5_lp
            h_best = h_lp
        h_new = slsqp_smooth(h_best, n, 3, 3000, 1e-15, t0, budget_s - 80)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Final SLSQP fine polish
    if time.time() - t0 < budget_s - 40:
        for T in [2, 1]:
            if time.time() - t0 > budget_s - 20:
                break
            h_new = slsqp_smooth(h_best, n, T, 4000, 1e-16, t0, budget_s - 15)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 7: Try different n if time permits
    elapsed = time.time() - t0
    if elapsed < budget_s - 150:
        for n2 in [n - 40, n + 40]:
            if n2 < 80 or n2 > 800 or n2 == n:
                continue
            if time.time() - t0 > budget_s - 60:
                break
            x_old = np.arange(n) * (2.0 / n)
            x_new2 = np.linspace(0, 2.0, n2, endpoint=False)
            h_alt = np.interp(x_new2, x_old, h_best)
            h_alt = project_feasible(h_alt, n2)
            c5_alt = compute_c5(h_alt)
            h_alt = slsqp_smooth(h_alt, n2, 4, 3000, 1e-15, t0, budget_s - 40)
            h_alt = project_feasible(h_alt, n2)
            c5_alt = compute_c5(h_alt)
            h_ng2, c5_ng2 = nesterov_topk(h_alt, n2, t0, budget_s - 30, 150000)
            c5_alt = min(c5_alt, c5_ng2)
            h_alt = h_ng2 if c5_ng2 < c5_alt else h_alt
            h_alt = project_feasible(h_alt, n2)
            c5_alt = min(compute_c5(h_alt), c5_alt)
            if c5_alt < c5_best:
                c5_best = c5_alt
                h_best = h_alt
                n = n2

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
