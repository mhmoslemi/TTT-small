import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_TARGET = 0.0
_G_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-14:
            continue
        grad += w * shift_grad_raw(x, _G_N, oi - center)
    return grad * dx

def _g_cons_fun(x):
    return x.sum() - _G_TARGET

def _g_cons_jac(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter, ftol):
    global _G_N, _G_TARGET, _G_T
    _G_N = n
    _G_TARGET = n / 2.0
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_cons_fun, 'jac': _g_cons_jac}]
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def nesterov_sg(h_init, n, t0, time_budget, K=3, max_iters=500000):
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        beta = it / (it + 3.0) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top_idx = np.argsort(corr)[-K:]
        mv = corr[top_idx].max()
        # Very sharp softmax
        w = np.exp(20.0 * (corr[top_idx] - mv))
        w /= w.sum()
        g = np.zeros(n)
        for wk, ik in zip(w, top_idx):
            shift = ik - center
            g += wk * shift_grad_raw(h_y, n, shift)
        g *= dx
        alpha = 0.08 / (1.0 + 0.003 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 20000 == 0:
            cv = compute_c5(h_sg)
            if cv < c5_best:
                c5_best = cv
                h_best = h_sg.copy()
    cv = compute_c5(project_feasible(h_sg, n))
    if cv < c5_best:
        c5_best = cv
        h_best = project_feasible(h_sg, n).copy()
    return h_best, c5_best

def swap_local_search(h, n, rng, steps=10000):
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    target = n / 2.0
    for _ in range(steps):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # Try to move mass from i to j or j to i

            if c5_try < c5_cur - 1e-10:
                h_cur = h_try
                c5_cur = c5_try
                break
    return h_cur, c5_cur

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_TARGET = 0.0
_G_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-13:
            continue
        grad += w * shift_grad_raw(x, _G_N, oi - center)
    return grad * dx

def _g_cf(x):
    return x.sum() - _G_TARGET

def _g_cj(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter, ftol):
    global _G_N, _G_TARGET, _G_T
    _G_N = n
    _G_TARGET = n / 2.0
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_cf, 'jac': _g_cj}]
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def nesterov_opt(h_init, n, t0, tmax, K=3, sharp=25.0, max_iters=600000):
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    stall = 0
    for it in range(max_iters):
        if time.time() - t0 > tmax:
            break
        beta = float(it) / (it + 4.0) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top_idx = np.argsort(corr)[-K:]
        mv = corr[top_idx].max()
        w = np.exp(sharp * (corr[top_idx] - mv))
        w /= w.sum()
        g = np.zeros(n)
        for wk, ik in zip(w, top_idx):
            shift = ik - center
            g += wk * shift_grad_raw(h_y, n, shift)
        g *= dx
        alpha = 0.06 / (1.0 + 0.002 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 15000 == 0:
            cv = compute_c5(h_sg)
            if cv < c5_best - 1e-10:
                c5_best = cv
                h_best = h_sg.copy()
                stall = 0
            else:
                stall += 1
            if stall > 3:
                h_sg = h_best + np.random.default_rng(it).normal(0, 0.005, n)
                h_sg = project_feasible(h_sg, n)
                h_prev = h_sg.copy()
                stall = 0
    cv = compute_c5(project_feasible(h_sg, n))
    if cv < c5_best:
        c5_best = cv
        h_best = project_feasible(h_sg, n).copy()
    return h_best, c5_best

def swap_search(h, n, rng, steps=10000):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for _ in range(steps):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = rng.uniform(0.02, 0.08)
        improved = False
        for (di, dj) in [(-1, 1), (1, -1)]:
            vi, vj = h_cur[i] + di * delta, h_cur[j] + dj * delta
            if vi < 0 or vi > 1 or vj < 0 or vj > 1:
                continue
            h_try = h_cur.copy()
            h_try[i] = vi
            h_try[j] = vj
            h_try = project_feasible(h_try, n)
            cv = compute_c5(h_try)
            if cv < c5_cur - 1e-11:
                h_cur = h_try
                c5_cur = cv
                improved = True
                break
        if not improved and rng.random() < 0.05:
            delta2 = rng.uniform(0.005, 0.02)
            for (di, dj) in [(-1, 1), (1, -1)]:
                vi, vj = h_cur[i] + di * delta2, h_cur[j] + dj * delta2
                if vi < 0 or vi > 1 or vj < 0 or vj > 1:
                    continue
                h_try = h_cur.copy()
                h_try[i] = vi
                h_try[j] = vj
                h_try = project_feasible(h_try, n)
                cv = compute_c5(h_try)
                if cv < c5_cur - 1e-11:
                    h_cur = h_try
                    c5_cur = cv
                    break
    return h_cur, c5_cur

def interp_up(h, n_old, n_new):
    if n_new == n_old:
        return h.copy()
    scale = n_new / n_old
    if scale == 2:
        return np.repeat(h, 2)
    idx = np.linspace(0, n_old - 1, n_new)
    return np.interp(idx, np.arange(n_old), h)
