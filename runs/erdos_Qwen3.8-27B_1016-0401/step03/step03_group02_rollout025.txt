import numpy as np
from scipy.optimize import minimize
import time

_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def lag_gradient(h, lag_idx):
    n = len(h)
    dx = 2.0 / n
    o = lag_idx - (n - 1)
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] += 1.0 - h[:n - o]
        g[:n - o] += -h[o:]
    else:
        p = -o
        g[:n - p] += 1.0 - h[p:]
        g[p:] += -h[:n - p]
    return g * dx

def softmax_og(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    obj = (mc + np.log(np.sum(e)) / T) * dx
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-12:
            continue
        grad += w * lag_gradient(h, center + o)
    return obj, grad

def eq_fun(x):
    return x.sum() - _TARGET

def eq_jac(x):
    return np.ones(len(x))

def make_cons():
    return [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

def lag_obj(x, lag_idx):
    n = len(x)
    return np.correlate(x, 1.0 - x, mode="full")[lag_idx] * (2.0 / n)

def lag_jac(x, lag_idx):
    return lag_gradient(x, lag_idx)

def sm_obj(x, T):
    return softmax_og(x, T)[0]

def sm_jac(x, T):
    return softmax_og(x, T)[1]

def slsqp_lag(h, lag_idx, bounds, maxiter, ftol):
    global _TARGET
    _TARGET = len(h) / 2.0
    cons = make_cons()
    try:
        res = minimize(lag_obj, h, args=(lag_idx,), method='SLSQP',
                       jac=lag_jac, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = np.clip(res.x, 0.0, 1.0)
        if compute_c5(h_new) < compute_c5(h) + 1e-12:
            return h_new
    except Exception:
        pass
    return h

def slsqp_sm(h, T, bounds, maxiter, ftol):
    global _TARGET
    _TARGET = len(h) / 2.0
    cons = make_cons()
    try:
        res = minimize(sm_obj, h, args=(T,), method='SLSQP',
                       jac=sm_jac, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = np.clip(res.x, 0.0, 1.0)
        if compute_c5(h_new) < compute_c5(h) + 1e-12:
            return h_new
    except Exception:
        pass
    return h

def nesterov_step(h, grad, v, alpha, target):
    v_new = 0.75 * v + grad
    h_new = h - alpha * v_new
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = fix_constraint(h_new, target)
    return h_new, v_new

def nesterov_optimize(h0, T, n_iters, alpha, target, t_end):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    obj0, grad0 = softmax_og(h, T)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        h_new, v = nesterov_step(h, grad0, v, alpha, target)
        obj1, grad1 = softmax_og(h_new, T)
        if obj1 <= obj0 + 1e-14:
            h = h_new
            obj0 = obj1
            grad0 = grad1
            alpha = min(alpha * 1.02, 0.5)
        else:
            h = h0 if it == 0 else h
            alpha *= 0.5
            grad0, _ = softmax_og(h, T)
            if alpha < 1e-6:
                alpha = 0.01
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def binary_push(h, threshold, target):
    n = len(h)
    c5_cur = compute_c5(h)
    changed = True
    while changed:
        changed = False
        for j in range(n):
            if h[j] > threshold:
                trial = h.copy()
                trial[j] = 1.0
            elif h[j] < 1.0 - threshold:
                trial = h.copy()
                trial[j] = 0.0
            else:
                continue
            trial = fix_constraint(trial, target)
            c5_new = compute_c5(trial)
            if c5_new < c5_cur - 1e-15:
                h = trial
                c5_cur = c5_new
                changed = True
    return h, c5_cur

def coord_descent(h, target, n_sweeps):
    n = len(h)
    dts = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    c5_cur = compute_c5(h)
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            for dt in dts:
                tv = h[j] + dt
                if tv < 0 or tv > 1:
                    continue
                trial = h.copy()
                trial[j] = tv
                trial = fix_constraint(trial, target)
                c5_new = compute_c5(trial)
                if c5_new < c5_cur - 1e-15:
                    h = trial
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def anneal_perturb(h, target, n_perturb, rng, t_end):
    n = len(h)
    c5_cur = compute_c5(h)
    h_out = h.copy()
    best_c5 = c5_cur
    best_h = h.copy()
    for trial in range(200):
        if time.time() > t_end:
            break
        h_p = h_out.copy()
        idx = rng.choice(n, size=min(n_perturb, n), replace=False)
        for i in idx:
            h_p[i] = min(1.0, max(0.0, h_p[i] + rng.uniform(-0.15, 0.15)))
        h_p = fix_constraint(h_p, target)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5 - 1e-10:
            best_c5 = c5_p
            best_h = h_p.copy()
            h_out = h_p.copy()
    return best_h, best_c5

        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 2: SLSQP top-12 lag cycling (20% budget)
    p2_end = t0 + budget_s * 0.50

        if elapsed() > p2_end:

            h_new = fix_constraint(h_new, target)
