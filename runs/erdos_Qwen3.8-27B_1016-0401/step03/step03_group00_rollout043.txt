import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

# Active-set SLSQP
_AN = 0
_AT = 1.0
_AK = 10

def _a_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_AK:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_AT * (c - m))
    return (m + np.log(e.sum()) / _AT) * (2.0 / _AN)

def _a_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_AK:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_AT * (c - m))
    w = e / e.sum()
    grad = np.zeros(_AN)
    n = _AN
    for wi, ci in zip(w, idx):
        if wi > 1e-10:
            s = int(ci) - (n - 1)
            j = np.arange(n)
            g = np.zeros(n)
            if s >= 0:
                m1 = j >= s
                m2 = (j + s) < n
                g[m1] += (1.0 - x[j[m1] - s])
                g[m2] -= x[j[m2] + s]
            else:
                ss = -s
                m1 = (j + ss) < n
                m2 = j >= ss
                g[m1] += (1.0 - x[j[m1] + ss])
                g[m2] -= x[j[m2] - ss]
            grad += wi * g
    return grad * (2.0 / _AN)

def _a_cf(x):
    return x.sum() - _AN / 2.0

def _a_cj(x):
    return np.ones(_AN)

def slsqp_opt(h, n, T, top_k, maxiter, ftol):
    global _AN, _AT, _AK
    _AN = n
    _AT = float(T)
    _AK = min(top_k, 2 * n - 1)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_a_obj, h, method='SLSQP', jac=_a_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _a_cf, 'jac': _a_cj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

                h_new = h.copy()
                h_new[i] = a + delta
                h_new[j] = b - delta
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    h = h_new
                    best = h_new.copy()
                    best_c5 = c5_new
                    improved = True
                    break
        
        if not improved and it % 5000 == 0:
            # Periodic full re-evaluation
            c5_check = compute_c5(h)
            if c5_check > best_c5 + 1e-10:
                h = best.copy()
    
    return best, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

_AN = 0
_AT = 1.0
_AK = 10

def _a_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_AK:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_AT * (c - m))
    return (m + np.log(e.sum()) / _AT) * (2.0 / _AN)

def _lag_grad(x, n, lag):
    j = np.arange(n)
    g = np.zeros(n)
    if lag >= 0:
        m1 = j >= lag
        m2 = (j + lag) < n
        g[m1] += (1.0 - x[j[m1] - lag])
        g[m2] -= x[j[m2] + lag]
    else:
        s = -lag
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - x[j[m1] + s])
        g[m2] -= x[j[m2] - s]
    return g

def _a_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_AK:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_AT * (c - m))
    w = e / e.sum()
    grad = np.zeros(_AN)
    for wi, ci in zip(w, idx):
        if wi > 1e-10:
            grad += wi * _lag_grad(x, _AN, int(ci) - (_AN - 1))
    return grad * (2.0 / _AN)

def _a_cf(x):
    return x.sum() - _AN / 2.0

def _a_cj(x):
    return np.ones(_AN)

def slsqp_opt(h, n, T, top_k, maxiter, ftol):
    global _AN, _AT, _AK
    _AN = n
    _AT = float(T)
    _AK = min(top_k, 2 * n - 1)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_a_obj, h, method='SLSQP', jac=_a_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _a_cf, 'jac': _a_cj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def pairwise_search(h, n, t_end, rng, top_k=8):
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(h)
    step_sizes = [0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    stall = 0
    max_stall = 3000
    it = 0
    while time.time() < t_end:
        it += 1
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        a, b = float(h[i]), float(h[j])
        if a < 1e-8 and b < 1e-8:
            continue
        if a > 1 - 1e-8 and b > 1 - 1e-8:
            continue
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_top = np.argsort(corr)[-top_k:]
        gli = 0.0
        glj = 0.0
        for ci in idx_top:
            lag = int(ci) - (n - 1)
            gi = (1.0 - h[i + lag]) if 0 <= i + lag < n else 0.0
            gi -= h[i - lag] if 0 <= i - lag < n else 0.0
            gj = (1.0 - h[j + lag]) if 0 <= j + lag < n else 0.0
            gj -= h[j - lag] if 0 <= j - lag < n else 0.0
            gli += gi
            glj += gj
        diff = gli - glj
        if abs(diff) < 1e-8:
            dirs = [0.005, -0.005]
        else:
            dirs = [-np.sign(diff) * 0.05, np.sign(diff) * 0.02]
        moved = False
        for d0 in dirs:
            for ss in step_sizes:
                delta = d0 * ss
                new_a = a + delta
                new_b = b - delta
                if new_a < -1e-9 or new_a > 1 + 1e-9:
                    continue
                if new_b < -1e-9 or new_b > 1 + 1e-9:
                    continue
                h[i], h[j] = new_a, new_b
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-12:
                    best = h.copy()
                    best_c5 = c5_new
                    stall = 0
                    moved = True
                    break
                else:
                    h[i], h[j] = a, b
            if moved:
                break
        if not moved:
            stall += 1
            if stall >= max_stall:
                h = best.copy()
                stall = 0
                max_stall = min(max_stall * 2, 20000)
                if max_stall > 20000:
                    break
    if compute_c5(h) > best_c5:
        h = best
    return best, best_c5

def nesterov_polish(h_init, n, t_end, top_k=10):
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    best = h_init.copy()
    best_c5 = compute_c5(h_init)
    for it in range(200000):
        if time.time() > t_end:
            break
        beta = float(it) / (it + 1.5) if it > 0 else 0.0

        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)

            g += _lag_grad(h_y, n, int(ci) - (n - 1))
        g *= dx / top_k
        alpha = 0.8 / (1.0 + 0.001 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        if (it + 1) % 2000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < best_c5:
                best_c5 = c1
                best = h_cur.copy()
    return best, best_c5
