```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def optimal_t(corr0, slopes):
    """Find t in [0,1] minimizing max_m (corr0[m] + slopes[m]*t)."""
    n_s = len(corr0)
    pos = slopes > 1e-12
    neg = slopes < -1e-12
    zero = ~pos & ~neg
    
    # Upper bound on the optimal value
    hi_val = max(np.max(corr0), np.max(corr0 + slopes))
    lo_val = 0.0  # safe lower bound (corr values are non-negative for h in [0,1])
    
    # Actually, the minimum value of max of lines at t in [0,1] is at least
    # the value at the best of t=0 and t=1
    v0 = np.max(corr0)
    v1 = np.max(corr0 + slopes)
    lo_val = min(v0, v1)
    
    for _ in range(50):
        mid = (lo_val + hi_val) / 2.0
        # Check feasibility
        if np.any(corr0[zero] > mid):
            lo_val = mid
            continue
        if np.any(neg):
            t_lo = np.max((mid - corr0[neg]) / slopes[neg])
        else:
            t_lo = 0.0
        if np.any(pos):
            t_hi = np.min((mid - corr0[pos]) / slopes[pos])
        else:
            t_hi = 1.0
        if max(0.0, t_lo) <= min(1.0, t_hi):
            hi_val = mid
        else:
            lo_val = mid
    
    # Find optimal t
    if np.any(neg):
        t_lo = np.max((hi_val - corr0[neg]) / slopes[neg])
    else:
        t_lo = 0.0
    if np.any(pos):
        t_hi = np.min((hi_val - corr0[pos]) / slopes[pos])
    else:
        t_hi = 1.0
    
    t_lo = max(0.0, t_lo)
    t_hi = min(1.0, t_hi)
    return (t_lo + t_hi) / 2.0
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_TGT = 0.0
_G_T = 1.0
_G_K = 10

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def optimal_t(corr0, slopes):
    pos = slopes > 1e-12
    neg = slopes < -1e-12
    zero = ~pos & ~neg
    v0 = np.max(corr0)
    v1 = np.max(corr0 + slopes)
    lo_v = min(v0, v1)
    hi_v = max(v0, v1) + 1e-12
    for _ in range(52):
        mid = (lo_v + hi_v) / 2.0
        if np.any(corr0[zero] > mid):
            lo_v = mid
            continue
        if np.any(neg):
            t_lo = np.max((mid - corr0[neg]) / slopes[neg])
        else:
            t_lo = 0.0
        if np.any(pos):
            t_hi = np.min((mid - corr0[pos]) / slopes[pos])
        else:
            t_hi = 1.0
        if max(0.0, t_lo) <= min(1.0, t_hi):
            hi_v = mid
        else:
            lo_v = mid
    if np.any(neg):
        t_lo = np.max((hi_v - corr0[neg]) / slopes[neg])
    else:
        t_lo = 0.0
    if np.any(pos):
        t_hi = np.min((hi_v - corr0[pos]) / slopes[pos])
    else:
        t_hi = 1.0
    t_lo = max(0.0, t_lo)
    t_hi = min(1.0, t_hi)
    return (t_lo + t_hi) / 2.0

def coord_descent(h, n, t_end, rng):
    h = h.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    s_all = np.arange(-(n - 1), n)
    for sweep in range(1000000):
        if time.time() > t_end:
            break
        order = rng.permutation(n)
        for i in order:
            h[i] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode="full")
            valid1 = (i + s_all >= 0) & (i + s_all < n)
            valid2 = (i - s_all >= 0) & (i - s_all < n)
            slopes = np.zeros(2 * n - 1)
            if np.any(valid1):
                slopes[valid1] += (1.0 - h[i + s_all[valid1]])
            if np.any(valid2):
                slopes[valid2] -= h[i - s_all[valid2]]
            h[i] = optimal_t(corr0, slopes)
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()
    return h_best, c5_best

def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_G_T * (c - m))
    return (m + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _slsqp_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_G_T * (c - m))
    w = e / e.sum()
    grad = np.zeros(_G_N)
    n = _G_N
    for wi, ci in zip(w, idx):
        if wi < 1e-10:
            continue
        s = ci - (n - 1)
        val1 = (wi * (1.0 - np.roll(x, -s, axis=0)))[:n]
        val2 = (wi * np.roll(x, -s, axis=0))[:n]
        mask1 = np.ones(n, dtype=bool)
        if s > 0:
            mask1[:s] = False
        elif s < 0:
            mask1[s:] = False
        mask2 = np.ones(n, dtype=bool)
        if s > 0:
            mask2[n - s:] = False
        elif s < 0:
            mask2[:n + s] = False
        grad += np.where(mask1, val1, 0.0) - np.where(mask2, val2, 0.0)
    return grad * (2.0 / _G_N)

def _slsqp_eq(x):
    return x.sum() - _G_TGT

def _slsqp_eq_jac(x):
    return np.ones(_G_N)

def slsqp_refine(h, n, T, top_k, maxiter, ftol):
    global _G_N, _G_TGT, _G_T, _G_K
    _G_N = n
    _G_TGT = n / 2.0
    _G_T = float(T)
    _G_K = min(top_k, 2 * n - 1)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP', jac=_slsqp_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def perturb_solve(h, n, rng, strength, t_end):
    h_p = h + rng.normal(0, strength, n)
    h_p = project_feasible(h_p, n)
    h1, c1 = coord_descent(h_p, n, min(t_end - 30, time.time() + 8), rng)
    if time.time() < t_end - 5:
        h1 = slsqp_refine(h1, n, 4, 12, 3000, 1e-15)
        h1 = project_feasible(h1, n)
        c1 = compute_c5(h1)
    return h1, c1

    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n0 = 280
        h_best = np.zeros(n0)
        h_best[::2] = 1.0

    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Heavy coordinate descent (45% budget)
    t1 = t0 + budget_s * 0.45
    h1, c1 = coord_descent(h_best, n, t1, rng)
    if c1 < c5_best:
        c5_best, h_best = c1, h1

    # Phase 2: SLSQP refinement (15% budget)
    t2 = t0 + budget_s * 0.60
    for T, tk, mi in [(32, 20, 2000), (8, 12, 3000), (2, 8, 4000)]:
        if time.time() > t2:

        hn = slsqp_refine(h_best, n, T, tk, mi, 1e-15)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 3: More coord descent after SLSQP (10% budget)
    t3 = t0 + budget_s * 0.70
    h1, c1 = coord_descent(h_best, n, t3, rng)
    if c1 < c5_best:
        c5_best, h_best = c1, h1

    # Phase 4: Multi-start perturbation (25% budget)
    t4 = t0 + budget_s * 0.95
    strengths = [0.003, 0.005, 0.008, 0.012, 0.02, 0.004, 0.006, 0.01]
    attempt = 0
    for strength in strengths:
        if time.time() > t4 - 10:
            break
        h_p, c_p = perturb_solve(h_best, n, rng, strength, t4 - 5)
        if c_p < c5_best:
            c5_best, h_best = c_p, h_p
        attempt += 1

        if c_p < c5_best + 1e-5 and time.time() < t4 - 20:

            if c_p2 < c5_best:

    if time.time() < t0 + budget_s - 3:

            hn = project_feasible(hn, n)
            c5 = compute_c5(hn)
            if c5 < c5_best:
                c5_best, h_best = c5, hn

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
```
