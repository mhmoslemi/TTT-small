        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def smooth_val_grad(x, n, T, K):
    corr = np.correlate(x, 1.0 - x, mode="full")
    if K < len(corr):
        topk = np.argpartition(corr, -K)[-K:]
    else:
        topk = np.arange(len(corr))
    c_sel = corr[topk]
    mc = np.max(c_sel)
    sh = T * (c_sel - mc)
    e = np.exp(sh)
    es = e.sum()
    val = (mc + np.log(es) / T) * (2.0 / n)
    sw = e / es
    center = n - 1
    grad = np.zeros(n)
    for ii in range(len(sw)):
        if sw[ii] > 1e-13:
            grad += sw[ii] * shift_grad(x, n, topk[ii] - center)
    return val, grad * (2.0 / n)

def lbfgsb_opt(h, n, T, K, maxiter, tol, tlim):
    def f(x):
        if time.time() > tlim:
            raise TimeoutError
        v, _ = smooth_val_grad(x, n, T, K)
        return v
    def g(x):
        if time.time() > tlim:
            raise TimeoutError
        _, gr = smooth_val_grad(x, n, T, K)
        return gr
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h, method='L-BFGS-B', jac=g, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol * 0.03,
                                'maxls': 50, 'maxfun': maxiter * 5})
        return project_feasible(np.clip(res.x, 0, 1), n)
    except (TimeoutError, Exception):
        return h.copy()

def exact_subgrad(h_init, n, tlim, max_iters=200000):
    dx = 2.0 / n
    h_c = h_init.copy()
    c5_b = compute_c5(h_c)
    h_b = h_c.copy()
    alpha = 0.15
    for it in range(max_iters):
        if time.time() > tlim:
            break
        corr = np.correlate(h_c, 1.0 - h_c, mode="full")
        mv = np.max(corr)
        active = np.where(corr >= mv - 1e-4)[0]
        g = np.zeros(n)
        for ai in active:
            g += shift_grad(h_c, n, ai - (n - 1))
        g /= len(active)
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        c5_cur = compute_c5(h_c)
        for _ in range(25):
            h_t = project_feasible(h_c - alpha * g, n)
            if compute_c5(h_t) < c5_cur:
                break
            alpha *= 0.5
        else:
            alpha *= 0.01
            h_t = project_feasible(h_c - alpha * g, n)
        alpha = min(alpha * 1.3, 0.3)
        h_c = h_t
        c5_now = compute_c5(h_c)
        if c5_now < c5_b:
            c5_b, h_b = c5_now, h_c.copy()
    return h_b, c5_b

def try_n(h_ref, n_ref, n_new, tlim):
    if n_new <= 50 or n_new == n_ref:
        return None, 999.0
    x_old = np.linspace(0, 2.0, n_ref, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_ref)
    h_new = project_feasible(h_new, n_new)
    c5 = compute_c5(h_new)
    for T, mi in [(16, 3000), (2, 4000), (0.5, 5000)]:
        if time.time() > tlim - 5:
            break
        h_t = lbfgsb_opt(h_new, n_new, T, min(100, 2*n_new-1), mi, 1e-13, tlim - 3)
        c5_t = compute_c5(h_t)
        if c5_t < c5:
            c5, h_new = c5_t, h_t
    return h_new, c5

def perturb(h, n, rng, mode):
    x = np.arange(n, dtype=float)
    if mode == 0:
        return h + rng.normal(0, 0.004, n)
    elif mode == 1:
        h_p = h.copy()
        bs = max(3, n // 12)
        bi = int(rng.integers(0, max(1, n // bs)))
        h_p[bi*bs:min((bi+1)*bs, n)] = 1.0 - h_p[bi*bs:min((bi+1)*bs, n)]
        return h_p
    elif mode == 2:
        return h + 0.015 * np.sin(2*np.pi*rng.uniform(2,12)*x/n + rng.uniform(0, 6.28))
    elif mode == 3:
        a = int(rng.integers(0, n-5))
        ln = int(rng.integers(3, min(25, n-4)))
        h_p = h.copy()
        h_p[a:a+ln] = 1.0 - h[a:a+ln]
        return h_p
    else:
        h_p = h.copy()
        for _ in range(4):
            a = int(rng.integers(0, n-8))
            b = int(rng.integers(a+4, n))
            h_p[a:b] = h[a:b][::-1]
        return h_p

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B coarse-to-fine
    tlim1 = t0 + 280
    for T, K, mi in [(64, 400, 2000), (16, 250, 3500), (4, 150, 5000), (1, 80, 6000), (0.3, 40, 8000)]:
        if time.time() > tlim1 - 5:
            break
        h_new = lbfgsb_opt(h_best, n, T, K, mi, 1e-14, min(time.time()+80, tlim1 - 3))
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 2: Exact subgradient polish
    tlim2 = t0 + 400
    if time.time() < tlim2:
        h_sg, c5_sg = exact_subgrad(h_best, n, tlim2, 150000)
        if c5_sg < c5_best:
            c5_best, h_best = c5_sg, h_sg

    # Phase 3: Try other resolutions
    tlim3 = t0 + 580
    for dn in [-60, -40, -20, 20, 40, 60, 80]:
        if time.time() > tlim3 - 10:
            break
        n_try = n + dn
        if n_try < 100:
            continue
        h_t, c5_t = try_n(h_best, n, n_try, tlim3)
        if h_t is not None and c5_t < c5_best:
            c5_best, h_best, n = c5_t, h_t, n_try
            # Quick refine at new n
            for T, mi in [(2, 4000), (0.3, 5000)]:
                if time.time() > tlim3 - 5:
                    break
                h_t2 = lbfgsb_opt(h_best, n, T, 60, mi, 1e-14, min(time.time()+60, tlim3 - 3))
                c5_t2 = compute_c5(h_t2)
                if c5_t2 < c5_best:
                    c5_best, h_best = c5_t2, h_t2

    # Phase 4: Fine L-BFGS-B with very low temperature
    tlim4 = t0 + 780
    for T, K, mi in [(0.5, 30, 15000), (0.15, 20, 20000), (0.05, 15, 25000)]:
        if time.time() > tlim4 - 10:
            break
        h_new = lbfgsb_opt(h_best, n, T, K, mi, 1e-15, min(time.time()+100, tlim4 - 5))
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 5: Diverse restarts
    tlim5 = t0 + 1000
    n_rest = 120
    for pi in range(n_rest):
        if time.time() > tlim5 - 5:
            break
        h_p = perturb(h_best, n, rng, pi % 5)
        h_p = project_feasible(np.clip(h_p, 0, 1), n)
        # Quick optimize: 2 stages
        h_new = lbfgsb_opt(h_p, n, 3, 80, 2500, 1e-13, min(time.time()+6, tlim5 - 3))
        c5 = compute_c5(h_new)
        if c5 < c5_best + 0.0001:
            h_new2 = lbfgsb_opt(h_new, n, 0.3, 25, 4000, 1e-14, min(time.time()+8, tlim5 - 3))
            c5_2 = compute_c5(h_new2)
            if c5_2 < c5_best:
                c5_best, h_best = c5_2, h_new2

    # Phase 6: Final subgradient + L-BFGS-B
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        h_sg, c5_sg = exact_subgrad(h_best, n, t0 + rem - 5, 50000)
        if c5_sg < c5_best:
            c5_best, h_best = c5_sg, h_sg
    if rem > 15:
        h_new = lbfgsb_opt(h_best, n, 0.1, 15, 15000, 1e-16, t0 + rem - 3)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
