def project_feasible(y, n):

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])

def slsqp_smooth(h, n, T, maxiter=10000, ftol=1e-15):
    target = n / 2.0
    center = n - 1
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        return (mc + np.log(np.exp(sh).sum()) / T) * dx
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        g = np.zeros(n)
        for oi in range(len(sw)):
            if sw[oi] < 1e-12:
                continue
            g += sw[oi] * shift_grad(x, n, oi - center)
        return g * dx
    
    def cons_fun(x):
        return x.sum() - target
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:

def adaptive_subgrad(h, n, t0, budget, K=8, K_max=30):
    dx = 2.0 / n
    center = n - 1
    h_cur = h.copy()
    h_prev = h.copy()
    c5_best = compute_c5(h_cur)
    h_best = h_cur.copy()
    beta = 0.7
    alpha_base = 0.12
    
    while time.time() - t0 < budget:
        it_elapsed = 1
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        mc = corr.max()
        # Adaptive K: shifts within 10% of max
        thresh = mc - 0.10 * mc
        active = np.where(corr >= thresh)[0]
        if len(active) < 2:
            active = np.argsort(corr)[-K:]
        K_eff = min(len(active), K_max)
        active = active[np.argsort(corr[active])[-K_eff:]]
        
        w = np.exp(6.0 * (corr[active] - mc))
        w /= w.sum()
        g = np.zeros(n)

            if wi > 1e-10:
                g += wi * shift_grad(h_cur, n, ak - center)

        h_mom = h_cur + beta * (h_cur - h_prev)

def cyclic_coord_descent(h, n, t0, budget, n_passes=50):

    c5_best = compute_c5(h_cur)
