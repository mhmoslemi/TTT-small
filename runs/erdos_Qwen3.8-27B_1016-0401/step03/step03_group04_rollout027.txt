        idx = np.argpartition(v, -K)[-K:]
        sel = v[idx]
        mc = sel.max()
        sh = T * (sel - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        val = (mc + np.log(e.sum()) / T)
        weights = np.zeros(n)
        weights[idx] = sw
        return val, weights, idx
    else:
        mc = v.max()
        sh = T * (v - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        val = mc + np.log(e.sum()) / T
        return val, sw, np.arange(n)

def _obj_lb(x, T, K, t0, tlim):

def _grad_lb(x, T, K, t0, tlim):
    if time.time() - t0 > tlim:
        return np.zeros(len(x))

    dx = 2.0 / n

    for i in range(len(sw)):

        h_curr = h_new

            x = np.arange(n, dtype=float)
            pert = np.zeros(n)
            for _ in range(4):
                f = rng.uniform(1, 12)
                p = rng.uniform(0, 2 * np.pi)
                a = rng.uniform(0.005, 0.04)
                pert += a * np.sin(2 * np.pi * f * x / n + p)
            h_p = h_best + pert
        elif pi % 5 == 1:
            h_p = freq_perturb(h_best, n, rng)
        elif pi % 5 == 2:
            h_p = h_best.copy()
            bs = max(4, n // 10)
            bi = int(rng.integers(0, max(1, n // bs)))
            h_p[bi * bs:(bi + 1) * bs] = 1.0 - h_p[bi * bs:(bi + 1) * bs]
            h_p += rng.normal(0, 0.003, n)
        elif pi % 5 == 3:
            h_p = h_best.copy()
            c = int(rng.integers(0, n))
            w = max(2, n // 20)
            s, e = max(0, c - w), min(n, c + w)
            h_p[s:e] = 1.0 - h_p[s:e]
        else:
            sig = 0.002 + 0.008 * (pi % 4)
            h_p = h_best + rng.normal(0, sig, n)
        h_p = project_feasible(h_p, n)
        h_new, c5 = do_lbfgsb(h_p, n, 2, 80, 1500, 1e-13, t0, budget_s - 10)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Final polish
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_new, c5 = do_slsqp(h_best, n, 1, 50, 4000, 1e-16, t0, t0 + rem - 10)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        rem = budget_s - (time.time() - t0)
        if rem > 20:
            h_new, c5 = do_lbfgsb(h_best, n, 1, 50, 6000, 1e-16, t0, t0 + rem - 5)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 7: Try resolution doubling if time allows
    rem = budget_s - (time.time() - t0)
    if rem > 200:
        n2 = n * 2
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = project_feasible(h_hi, n2)
        c5_hi = compute_c5(h_hi)
        if c5_hi < c5_best:
            c5_best = c5_hi
            h_best = h_hi
            n = n2
            for T, mi in [(8, 3000), (2, 5000), (1, 6000)]:
                rem = budget_s - (time.time() - t0)
                if rem < 60:
                    break
                h_new, c5 = do_lbfgsb(h_best, n, T, 150, mi, 1e-14, t0, budget_s - 30)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
