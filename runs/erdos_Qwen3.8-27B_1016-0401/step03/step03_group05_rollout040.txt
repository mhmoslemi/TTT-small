def shift_grad(h, n, shift):
    # Gradient of sum_i h[i]*(1-h[i+shift]) w.r.t. h
    # = sum_i (1-h[i+shift]) * I[i] - sum_i h[i] * I[i-shift]
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def softmax_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * dx

def softmax_grad(x, n, T):

    center = n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")

    bounds = [(0.0, 1.0)] * n

                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})

    center = n - 1
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        top_idx = np.argpartition(corr, -K)[-K:]
        top_vals = corr[top_idx]
        w = (top_vals / max(mc, 1e-15)) ** power
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(K) / K
        else:
            w /= ws
        grad = np.zeros(n)
        for i in range(K):
            grad += w[i] * shift_grad(x, n, top_idx[i] - center)
        return grad * dx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(len(x))}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return res.x
    except Exception:
        return h.copy()

def nesterov_v2(h_init, n, t0, time_limit, max_iters=600000):
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    alpha_base = 0.4
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        frac = min((it + 1) / max_iters, 1.0)
        if frac < 0.25:
            K = max(1, int(1 + 8 * frac / 0.25))
        elif frac < 0.6:
            K = max(9, int(9 + 20 * (frac - 0.25) / 0.35))
        else:
            K = 29 + int(11 * (0.5 + 0.5 * np.sin(frac * 30.0)))
        beta = min(0.85, 0.3 * float(it) / (it + 5.0))
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        if K <= 1:
            idx = int(np.argmax(corr))
            g = shift_grad(h_y, n, idx - center) * dx
        else:
            top_idx = np.argpartition(corr, -K)[-K:]
            top_vals = corr[top_idx]
            w = (top_vals / max(mc, 1e-15)) ** 16
            ws = w.sum()
            if ws < 1e-15:
                w = np.ones(K) / K
            else:
                w /= ws
            g = np.zeros(n)
            for i in range(K):
                g += w[i] * shift_grad(h_y, n, top_idx[i] - center)
            g *= dx
        gnorm2 = np.dot(g, g) + 1e-12
        gap = mc * dx - c5_best
        alpha = min(0.6, max(1e-5, gap / gnorm2))
        alpha = alpha_base * alpha / (1.0 + 0.001 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        h_avg += (h_new - h_avg) / (float(it + 1))
        if (it + 1) % 15000 == 0:
            ca = compute_c5(project_feasible(h_avg, n))
            if ca < c5_best:
                c5_best = ca
                h_best = project_feasible(h_avg, n)
            cc = compute_c5(h_sg)
            if cc < c5_best:
                c5_best = cc
                h_best = h_sg.copy()
    return h_best, c5_best

def perturb_h(h, n, rng, mode):
    x = np.arange(n) * 2.0 / n
    if mode == 0:
        return h + rng.normal(0, 0.015, n)
    elif mode == 1:
        f = 0.3 + 0.15 * (mode + int(rng.uniform(0, 5)))
        return h + 0.025 * np.sin(2 * np.pi * f * x + rng.uniform(0, 6.28))
    elif mode == 2:
        hp = h.copy()
        k = n // 5
        for b in range(5):
            hp[b*k:min((b+1)*k, n)] += rng.uniform(-0.04, 0.04)
        return hp
    else:
        idx = rng.choice(n, size=n // 4, replace=False)
        hp = h.copy()
        hp[idx] += rng.uniform(-0.06, 0.06, len(idx))
        return hp

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n = 280
        h_best = np.zeros(n)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    el = lambda: time.time() - t0
    # Phase 1: L-BFGS-B coarse
    for T in [32, 16, 8]:
        if el() > budget_s - 450:
            break
        h_new = lbfgs_smooth(h_best, n, T, 3000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new
    # Phase 2: SLSQP medium
    for T in [8, 4, 2]:
        if el() > budget_s - 400:
            break
        h_new = slsqp_smooth(h_best, n, T, 4000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new
    # Phase 3: Nesterov with Polyak
    h_sg, c5_sg = nesterov_v2(h_best, n, t0, budget_s - 260, 500000)
    if c5_sg < c5_best:
        c5_best, h_best = c5_sg, h_sg
    h_best = project_feasible(h_best, n)
    # Phase 4: SLSQP fine
    for T in [2, 1, 1]:
        if el() > budget_s - 180:
            break
        h_new = slsqp_smooth(h_best, n, T, 5000)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new
    # Phase 5: Top-K SLSQP
    for K in [3, 5, 8, 12]:
        if el() > budget_s - 100:
            break
        h_new = slsqp_topk(h_best, n, K, 3500, power=20)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new
    # Phase 6: Perturb + re-optimize
    for pi in range(40):
        if el() > budget_s - 30:
            break
        h_p = perturb_h(h_best, n, rng, pi % 4)
        h_p = project_feasible(h_p, n)
        h_new = slsqp_topk(h_p, n, 5, 1500, power=16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new
    # Phase 7: Final micro polish
    if el() < budget_s - 20:
        h_new = slsqp_topk(h_best, n, 3, 3000, power=24)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new
    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
