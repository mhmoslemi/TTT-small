        m2 = (j + shift) < n
        if m1.any():

        if m2.any():

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def sm_val_grad(x, n, T, K):
    corr = np.correlate(x, 1.0 - x, mode="full")
    center = n - 1
    ns = len(corr)
    if K < ns:
        topk = np.argpartition(corr, -K)[-K:]
    else:
        topk = np.arange(ns)
    c_sel = corr[topk]
    mc_sel = float(c_sel.max())
    sh = T * (c_sel - mc_sel)
    e = np.exp(sh)
    e_sum = e.sum()
    sw = e / e_sum
    val = (mc_sel + np.log(e_sum) / T) * (2.0 / n)
    grad = np.zeros(n)
    for ii in range(len(topk)):
        w = sw[ii]
        if w > 1e-14:
            grad += w * shift_grad(x, n, topk[ii] - center)
    return val, grad * (2.0 / n)

    def obj(x):
        v, _ = sm_val_grad(x, n, T, K)
        return v
    def jac(x):
        _, g = sm_val_grad(x, n, T, K)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jac, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol * 0.05,
                                'maxls': 40, 'maxfun': maxiter * 4})
        if time.time() - t0 > tlim:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def do_slsqp(h, n, T, K, maxiter, tol, t0, tlim):
    def obj(x):
        v, _ = sm_val_grad(x, n, T, K)
        return v
    def jac(x):
        _, g = sm_val_grad(x, n, T, K)
        return g
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def sm_val_grad(x, n, T, K):
    corr = np.correlate(x, 1.0 - x, mode="full")
    center = n - 1
    ns = len(corr)
    if K < ns:
        topk = np.argpartition(corr, -K)[-K:]
    else:
        topk = np.arange(ns)
    c_sel = corr[topk]
    mc_sel = float(c_sel.max())
    sh = T * (c_sel - mc_sel)
    e = np.exp(sh)
    e_sum = e.sum()
    sw = e / e_sum
    val = (mc_sel + np.log(e_sum) / T) * (2.0 / n)
    grad = np.zeros(n)
    for ii in range(len(topk)):
        w = sw[ii]
        if w > 1e-14:
            grad += w * shift_grad(x, n, topk[ii] - center)
    return val, grad * (2.0 / n)

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):
    def obj(x):
        v, _ = sm_val_grad(x, n, T, K)
        return v
    def jac(x):
        _, g = sm_val_grad(x, n, T, K)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jac, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol * 0.05,
                                'maxls': 40, 'maxfun': maxiter * 4})
        if time.time() - t0 > tlim:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def do_slsqp(h, n, T, K, maxiter, tol, t0, tlim):
    def obj(x):
        v, _ = sm_val_grad(x, n, T, K)
        return v
    def jac(x):
        _, g = sm_val_grad(x, n, T, K)
        return g
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': tol})
        if time.time() - t0 > tlim:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def nesterov_smooth(h, n, T, K, t0, tlim, max_iter=12000):
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        if time.time() - t0 > tlim:
            break
        beta = it / (it + 2.5)
        y = h_curr + beta * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        v, g = sm_val_grad(y, n, T, K)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        gap = max(v - 0.375, 1e-6)
        alpha = min(2.0 * gap / (gnorm * gnorm + 1e-12), 0.15 / (1.0 + 0.0005 * it))
        h_new = project_feasible(np.clip(y - alpha * g, 0.0, 1.0), n)
        h_prev = h_curr.copy()
        h_curr = h_new
        if (it + 1) % 2000 == 0:
            c5_now = compute_c5(h_curr)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h_curr.copy()
    return best_h, best_c5

def struct_perturb(h, n, rng):
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for _ in range(5):
        f = rng.uniform(1.0, 12.0)
        p = rng.uniform(0.0, 2.0 * np.pi)
        a = rng.uniform(0.005, 0.035)
        pert += a * np.sin(2.0 * np.pi * f * x / n + p)
    pert += rng.normal(0, 0.005, n)
    return h + pert

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    FK = 2 * n - 1
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    rem = budget_s - (time.time() - t0)

    # Phase 2: Medium L-BFGS-B (T=4, K=full)

    # Phase 3: Fine L-BFGS-B (T=1.5, K=full)

    # Phase 4: Nesterov smooth subgradient (explore between basins)

        if c5 < c5_best: c5_best, h_best = c5, h

    # Phase 5: L-BFGS-B (T=1, K=400)

    # Phase 6: SLSQP with equality constraint
    rem = budget_s - (time.time() - t0)

        c5 = compute_c5(h)

    n_rest = 30
    for i in range(n_rest):
        rem = budget_s - (time.time() - t0)

            break

        elif i % 6 == 1:

            h_p = 0.5 * h_best + 0.5 * np.roll(h_best, sa)

            bs = n // 7
            for _ in range(2):
                bi = int(rng.integers(0, 7))

            h_p += rng.normal(0, 0.004, n)

        else:

        h_p = project_feasible(h_p, n)
        c5_p = compute_c5(h_p)
        if c5_p < c5_best + 0.006:

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
