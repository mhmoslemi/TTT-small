    dx = 2.0 / n
    center = n-1
    def obj(x):
        corr = np.correlate(x, 1.0-x, mode="full")
        return np.max(corr) * dx
    def jac(x):
        corr = np.correlate(x, 1.0-x, mode="full")
        mc = np.max(corr)
        top_idx = np.argpartition(corr, -K)[-K:]
        top_vals = corr[top_idx]
        w = (top_vals / max(mc, 1e-15)) ** 8
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.ones(K) / K
        else:
            w /= w_sum
        grad = np.zeros(n)
        for i in range(K):
            grad += w[i] * shift_grad(x, n, top_idx[i]-center)
            # Note: shift_grad returns the gradient for the shift (with dx=2/n) incorporated
        return grad
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}]
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
    h_opt = res.x
    h_opt = project_feasible(h_opt, n)
    return h_opt
