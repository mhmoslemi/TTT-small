        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, idx):
    s = idx - (n - 1)
    j = np.arange(n)
    g = np.zeros(n)
    if s >= 0:
        m1 = j >= s
        m2 = (j + s) < n
        g[m1] += (1.0 - h[j[m1] - s])
        g[m2] -= h[j[m2] + s]
    else:
        ss = -s
        m1 = (j + ss) < n
        m2 = j >= ss
        g[m1] += (1.0 - h[j[m1] + ss])
        g[m2] -= h[j[m2] - ss]
    return g

def sms_obj(x, n, T, K):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    return (m + np.log(e.sum()) / T) * (2.0 / n)

def sms_jac(x, n, T, K):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for wi, ci in zip(w, idx):
        if wi > 1e-12:
            g += wi * shift_grad(x, n, ci)
    return g * dx

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: sms_obj(x, n, T, K), h, method='SLSQP',
                       jac=lambda x: sms_jac(x, n, T, K),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except:
        return h.copy()

    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    return project_feasible(np.interp(x_new, x_old, h), n_new)

        h_cur = h_new

            if c1 < c5a:
                c5a, ha = c1, h_cur.copy()
    return ha, c5a

    h280b = upsample(h160, 280)
    for T, K, mi in [(2, 8, 2000), (0.5, 6, 3000)]:
        if time.time() > t2:
            break
        h280b = slsqp_opt(h280b, 280, T, K, mi, 1e-14)
        h280b = project_feasible(h280b, 280)
    c5b = compute_c5(h280b)
    if c5b < c5_best:
        c5_best, h_best = c5b, h280b.copy()

    # Phase 3: Nesterov subgradient at n
    t3 = t0 + budget_s * 0.78
    hs, c5s = nesterov_opt(h_best, n, t3, rng, K=15, lr0=0.25)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs

    # Phase 4: Fine SLSQP refinement
    t4 = t0 + budget_s * 0.88
    for T, K, mi in [(1, 8, 3000), (0.3, 6, 4000)]:
        if time.time() > t4:
            break
        hn = slsqp_opt(h_best, n, T, K, mi, 1e-15)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 5: Diversity restarts
    for pi in range(50):
        if time.time() > t_end - 25:
            break
        hp = perturb(h_best, n, rng, pi)
        hn = slsqp_opt(hp, n, 1, 8, 1500, 1e-14)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Final polish
    if time.time() < t_end - 8:
        hn = slsqp_opt(h_best, n, 0.3, 5, 5000, 1e-16)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
