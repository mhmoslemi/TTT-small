import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def smooth_val_grad(x, n, T, K):
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")
    if K is not None and K < len(corr):
        top_idx = np.argpartition(corr, -K)[-K:]
        tv = corr[top_idx]
        sh = T * (tv - tv.max())
        e = np.exp(sh)
        sw = e / e.sum()
        val = np.dot(sw, tv) * dx
        grad = np.zeros(n)
        for i in range(K):
            if sw[i] > 1e-12:
                grad += sw[i] * shift_grad(x, n, top_idx[i] - center)
        return val, grad * dx
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    val = (mc + np.log(e.sum()) / T) * dx
    grad = np.zeros(n)
    for oi in range(len(corr)):
        if sw[oi] > 1e-12:
            grad += sw[oi] * shift_grad(x, n, oi - center)
    return val, grad * dx

def nesterov_opt(h_init, n, t0, t_end, max_iters=600000):
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    T_start, T_end = 40.0, 1.5
    K_min, K_max = 3, 20
    for it in range(max_iters):
        if time.time() - t0 >= t_end:
            break
        frac = min((it + 1) / max(1, max_iters), 1.0)
        T = T_start * (T_end / T_start) ** (frac ** 0.6)
        K = max(K_min, int(K_min + (K_max - K_min) * frac * frac))
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        _, g = smooth_val_grad(h_y, n, T, K)
        alpha = 0.55 / (1.0 + 0.0015 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 8000 == 0:
            c = compute_c5(h_sg)
            if c < c5_best:
                c5_best = c
                h_best = h_sg.copy()
    c = compute_c5(h_sg)
    if c < c5_best:
        c5_best = c
        h_best = h_sg.copy()
    return h_best, c5_best

def slsqp_topk(h_init, n, K, maxiter=4000):
    dx = 2.0 / n
    center = n - 1
    target = n / 2.0

    def obj(x):
        c, _ = smooth_val_grad(x, n, 2.0, K)
        return c

    def jac(x):
        _, g = smooth_val_grad(x, n, 2.0, K)
        return g

    try:
        res = minimize(obj, h_init, method='SLSQP', jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-17})
        return res.x
    except Exception:
        return h_init.copy()

def lbfgs_smooth(h_init, n, T, K, maxiter=3000):
    dx = 2.0 / n
    target = n / 2.0

    def obj_pen(x):
        v, _ = smooth_val_grad(x, n, T, K)
        viol = x.sum() - target
        return v + 5e3 * viol * viol

    def jac_pen(x):
        _, g = smooth_val_grad(x, n, T, K)
        viol = x.sum() - target
        return g + 1e4 * viol * np.ones(n)

    try:
        res = minimize(obj_pen, h_init, method='L-BFGS-B', jac=jac_pen,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'gtol': 1e-16})
        return res.x
    except Exception:
        return h_init.copy()

def interleaved_opt(h_init, n, t0, t_end, n_blocks=12):

    c5_best = compute_c5(h)
    block_time = (t_end - t0) / n_blocks
    for blk in range(n_blocks):
        te_block = t0 + (blk + 1) * block_time
        if time.time() - t0 >= t_end:
            break
        h, c5_best = nesterov_opt(h, n, t0, te_block - 5, max(50000, int(200000 * block_time / 90)))
        h = project_feasible(h, n)
        if time.time() - t0 >= t_end - 10:
            break
        K = 3 + blk // 3
        h_new = slsqp_topk(h, n, K, 2500)
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < c5_best:
            c5_best = c
            h = h_new
    return h, c5_best

def transfer_and_opt(h_cur, n_cur, n_targ, t0, t_end, seed):

    h = np.zeros(n)
