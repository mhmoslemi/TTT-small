    for n_alt in n_tries:
        if time.time() > p4_end:
            break
        if n_alt == n:
            continue
        target_alt = n_alt / 2.0
        x_old = np.arange(len(h_best)) * (2.0 / len(h_best))
        x_new = np.linspace(0.0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = fix_constraint(h_alt, target_alt)
        c5_alt = compute_c5(h_alt)
        bounds_alt = [(0.0, 1.0)] * n_alt
        for T in [300, 3000, 20000]:
            if time.time() > p4_end:
                break
            try:
                h_new = slsqp_sm(h_alt, T, bounds_alt, 4000, 1e-17)
                h_new = fix_constraint(h_new, target_alt)
                c5_new = compute_c5(h_new)
                if c5_new < c5_alt:
                    c5_alt = c5_new
                    h_alt = h_new
            except Exception:
                pass
        for _ in range(6):
            if time.time() > p4_end:
                break
            corr_a = np.correlate(h_alt, 1.0 - h_alt, mode="full")
            worst_a = int(np.argmax(corr_a))
            try:
                h_new = slsqp_lag(h_alt, worst_a, bounds_alt, 2000, 1e-17)
                h_new = fix_constraint(h_new, target_alt)
                c5_new = compute_c5(h_new)
                if c5_new < c5_alt:
                    c5_alt = c5_new
                    h_alt = h_new
            except Exception:
                pass
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt
            target = n / 2.0
            bounds = [(0.0, 1.0)] * n

    # Phase 5: Final polish (13% budget)
    p5_end = t0 + budget_s - 3.0
    target = len(h_best) / 2.0
    n = len(h_best)
    bounds = [(0.0, 1.0)] * n
    for _ in range(25):
        if time.time() > p5_end:
            break
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst_f = int(np.argmax(corr_f))
        try:
            h_new = slsqp_lag(h_best, worst_f, bounds, 3000, 1e-20)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass
    h_best, c5_cd_f = coord_descent(h_best, target, 5, steps_cd)
    if c5_cd_f < c5_best:
        c5_best = c5_cd_f

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
