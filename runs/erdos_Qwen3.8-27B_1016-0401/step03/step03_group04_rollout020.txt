        sw = e / e.sum()
        grad = np.zeros(n)
        for ii in range(nc):
            if sw[ii] < 1e-12:
                continue
            grad += sw[ii] * shift_grad(h, n, ii - center)
    return val, grad * dx

        return smooth_val_grad(x, n, T, K)[0]
    def jac(x):
        return smooth_val_grad(x, n, T, K)[1]
    bounds = [(0.0, 1.0)] * n
    try:

            return h0
        res = minimize(obj, h0, method='L-BFGS-B', jac=jac,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol,
                                'gtol': tol * 0.05, 'maxls': 20,
                                'maxfun': maxiter * 2})

    except Exception:

def do_slsqp(h0, n, T, K, maxiter, tol, t0, tlim):

        return project_feasible(res.x, n) if time.time() - t0 <= tlim else h0
    except Exception:

def nesterov_subgrad(h0, n, t0, tlim, max_iters=200000):
    dx = 2.0 / n
    h_cur = h0.copy()
    h_prev = h0.copy()
    c5_best = compute_c5(h0)
    h_best = h0.copy()
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        beta = 2.0 / (it + 2.0)
        y = h_cur + beta * (h_cur - h_prev)
        y = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = shift_grad(y, n, shift) * dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        f_y = compute_c5(y)
        gap = max(f_y - compute_c5(h_best), 1e-8)
        alpha = min(2.0 * gap / (gn * gn + 1e-12), 0.2 / (1.0 + 0.001 * it))
        h_new = y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_cur.copy()
        h_cur = h_new
        c5_now = compute_c5(h_cur)
        if c5_now < c5_best:
            c5_best = c5_now
            h_best = h_cur.copy()
    return h_best, c5_best

    h2[h2 < thresh] = 0.0
    h2[h2 > 1.0 - thresh] = 1.0
    h2 = project_feasible(h2, n)
    return h2

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 5.0
    hi = float(y.max()) + 5.0

        mid = (lo + hi) / 2.0

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        if m1.any():
            grad[j[m1]] += (1.0 - h[j[m1] - shift])
        m2 = (j + shift) < n
        if m2.any():
            grad[j[m2]] -= h[j[m2] + shift]
    else:

        m1 = (j + s) < n
        if m1.any():

        m2 = j >= s
        if m2.any():
            grad[j[m2]] -= h[j[m2] - s]
    return grad

    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_new = do_slsqp(h_best, n, 0.3, 2, 8000, 1e-16, t0, t0 + rem - 20)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        h_new = do_lbfgsb(h_best, n, 0.2, 3, 12000, 1e-15, t0, t0 + rem - 10)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
