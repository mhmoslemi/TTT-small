import numpy as np
from scipy.optimize import minimize
import time

_P = {}

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project(y, n):
    t = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(50):
        m = (lo + hi) / 2.0
        s = np.clip(y - m, 0.0, 1.0).sum()
        if abs(s - t) < 1e-13:
            break
        if s > t:
            lo = m
        else:
            hi = m
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def sgrad(h, n, sh):
    j = np.arange(n)
    g = np.zeros(n)
    if sh >= 0:
        a, b = j >= sh, (j + sh) < n
        if a.any(): g[a] += 1.0 - h[j[a] - sh]
        if b.any(): g[b] -= h[j[b] + sh]
    else:
        s2, a, b = -sh, (j + s2) < n, j >= s2
        if a.any(): g[a] += 1.0 - h[j[a] + s2]
        if b.any(): g[b] -= h[j[b] - s2]
    return g

def _sv(x):
    n, T, K = _P['n'], _P['T'], _P['K']
    if time.time() > _P['tl']:
        _P['done'] = True
    c = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(c)
    ctr, dx, m = n - 1, 2.0 / n, len(c)
    if K < m:
        tk = np.argpartition(c, -K)[-K:]
        cs, mk = c[tk].copy(), np.max(c[tk])
    else:
        tk, cs, mk = np.arange(m), c.copy(), mc
    sh_ = T * (cs - mk)
    mx = np.max(sh_)
    e = np.exp(sh_ - mx)
    es = e.sum()
    val = (mk + (mx + np.log(es)) / T) * dx
    w = e / es
    g = np.zeros(n)
    for i in range(len(cs)):
        if w[i] < 1e-12:
            continue
        g += w[i] * sgrad(x, n, tk[i] - ctr)
    return val, g * dx

def _ob(x):
    return _sv(x)[0]

def _ja(x):
    return _sv(x)[1]

def _ct(x):
    return x.sum() - _P['n'] / 2.0

def _cj(x):
    return np.ones(_P['n'])

def lbfgsb(h, n, T, K, mi, tol, tl):
    _P['n'] = n
    _P['T'] = T
    _P['K'] = K
    _P['tl'] = tl
    _P['done'] = False
    bd = [(0.0, 1.0)] * n
    try:
        r = minimize(_ob, h.copy(), method='L-BFGS-B', jac=_ja, bounds=bd,
                     options={'maxiter': mi, 'ftol': tol, 'gtol': tol * 0.05,
                              'maxls': 20, 'maxfun': mi * 3})
        if _P['done']:
            return h
        return project(r.x, n)
    except Exception:
        return h.copy()

def slsqp(h, n, T, K, mi, tol, tl):
    _P['n'] = n
    _P['T'] = T
    _P['K'] = K
    _P['tl'] = tl
    _P['done'] = False
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _ct, 'jac': _cj}]
    try:
        r = minimize(_ob, h.copy(), method='SLSQP', jac=_ja, bounds=bd,
                     constraints=cs, options={'maxiter': mi, 'ftol': tol})
        if _P['done']:
            return h
        return project(r.x, n)
    except Exception:
