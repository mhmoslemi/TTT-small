import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        s = shift
        m1 = (j + s) < n
        g[m1] += (1.0 - h[m1 + s])
        m2 = j >= s
        g[m2] -= h[m2 - s]
    else:
        s = -shift
        m1 = j >= s
        g[m1] += (1.0 - h[m1 - s])
        m2 = (j + s) < n
        g[m2] -= h[m2 + s]
    return g

# Global state for SLSQP callbacks
_G_N = 0
_G_T = 1.0

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = float(np.max(corr))
    sh = _G_T * (corr - mc)
    return (mc + np.log(np.exp(sh).sum()) / _G_T) * (2.0 / _G_N)

def _jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = float(np.max(corr))
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-10:
            continue
        grad += w * shift_grad(x, _G_N, oi - center)
    return grad * dx

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
