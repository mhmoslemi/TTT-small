        corr = np.correlate(h, 1.0 - h, mode="full")
        center = n - 1
        top_p = np.argsort(corr)[-6:]
        for p in top_p:
            k = p - center
            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])
            rng.shuffle(ones)
            rng.shuffle(zeros)
            found = False
            for i in ones[:15]:
                if not found:
                    for j in zeros[:20]:
                        h[i] = 0.0; h[j] = 1.0
                        c5_new = compute_c5(h)
                        if c5_new < c5 - 1e-15:
                            c5 = c5_new; improved = True; found = True
                        else:
                            h[i] = 1.0; h[j] = 0.0
            if found:
                improved = True
                break
        if not improved:
            break
    return h, c5

def _multi_lag_slsqp(h, bounds, target, n_rounds, maxiter):
    n = len(h)
    c5 = compute_c5(h)
    for r in range(n_rounds):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top = np.argsort(corr)[-3:]
        for p in top:
            lag = int(p - (n - 1))
            try:
                h_new = _slsqp_lag(h, lag, bounds, maxiter, 1e-19)
                h_new = fix_constraint(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < c5 - 1e-17:
                    c5 = c5_new; h = h_new
            except Exception:
                pass
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    global _TARGET
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    _TARGET = target
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Multi-lag SLSQP cycling (35% budget)
    p1_end = t0 + budget_s * 0.35
    for outer in range(40):
        if elapsed() > p1_end:
            break
        h_best, c5_tmp = _multi_lag_slsqp(h_best, bounds, target, 3, 1500)
        if c5_tmp < c5_best - 1e-17:
            c5_best = c5_tmp

    # Phase 2: Soft-max SLSQP with multiple temperatures (20% budget)
    p2_end = t0 + budget_s * 0.55
    for T in [2000, 8000, 30000]:
        if elapsed() > p2_end:
            break
        try:
            h_new = _slsqp_sm(h_best, T, bounds, 4000, 1e-19)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-17:
                c5_best = c5_new; h_best = h_new
        except Exception:
            pass

    # Phase 3: Binary local search (20% budget)
    p3_end = t0 + budget_s * 0.75
    h_bin, c5_bin = _binary_swap_search(h_best, 8, rng)
    if c5_bin < c5_best:
        c5_best = c5_bin; h_best = h_bin
    # Polish binary solution with SLSQP
    if c5_bin <= c5_best + 1e-15:
        h_pol, c5_pol = _multi_lag_slsqp(h_bin, bounds, target, 5, 2000)
        if c5_pol < c5_best:
            c5_best = c5_pol; h_best = h_pol

    # Phase 4: Perturbation + re-optimize (15% budget)
    p4_end = t0 + budget_s * 0.90
    for trial in range(4):
        if elapsed() > p4_end:
            break
        h_pert = h_best + 0.02 * rng.standard_normal(n)
        h_pert = fix_constraint(h_pert, target)
        c5_pert = compute_c5(h_pert)
        if c5_pert < c5_best:
            c5_best = c5_pert; h_best = h_pert
        h_new2, c5_new2 = _multi_lag_slsqp(h_pert, bounds, target, 2, 1000)
        if c5_new2 < c5_best:
            c5_best = c5_new2; h_best = h_new2

    # Phase 5: Final aggressive polish (10% budget)
    p5_end = t0 + budget_s - 3.0
    for _ in range(8):
        if elapsed() > p5_end:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst = int(np.argmax(corr)) - (n - 1)
        try:
            h_new = _slsqp_lag(h_best, worst, bounds, 5000, 1e-20)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new; h_best = h_new
        except Exception:
            pass
    # Alternative: try second-worst lag
    corr = np.correlate(h_best, 1.0 - h_best, mode="full")
    top2 = np.argsort(corr)[-2:]
    for p in top2:
        if elapsed() > p5_end:
            break
        lag = int(p - (n - 1))
        try:
            h_new = _slsqp_lag(h_best, lag, bounds, 3000, 1e-20)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new; h_best = h_new
        except Exception:
            pass

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_constraint(h_best, target)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
