                j = int(order_g[jj])       # low grad → increase h
                if h[i] <= 0.01 or h[j] >= 0.99: continue
                trial = h.copy()
                trial[i] -= 0.01; trial[j] += 0.01
                trial = proj(trial, target)
                c_new = compute_c5(trial)
                if c_new < c5 - 1e-15:
                    h = trial; c5 = c_new; changed = True
                    break
            if changed: break
        if not changed: break
    return h, c5

def sturmian_start(n, alpha):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i * alpha) % 1.0 < 0.5 else 0.0
    return h
