from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def softmax_obj(x, n, T):
    dx = 2.0 / n
    corr = compute_corr(x)
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * dx

def softmax_grad(x, n, T):
    dx = 2.0 / n
    corr = compute_corr(x)
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-14:
            continue
        shift = oi - center
        if shift >= 0:
            idx = np.arange(shift, n)
            grad[idx] += w * (1.0 - x[idx - shift])
            idx2 = np.arange(0, n - shift)
            grad[idx2] -= w * x[idx2 + shift]
        else:
            s = -shift
            idx = np.arange(0, n - s)
            grad[idx] += w * (1.0 - x[idx + s])
            idx2 = np.arange(s, n)
            grad[idx2] -= w * x[idx2 - s]
    return grad * dx

def lbgfq_optimize(h, n, T, maxiter=5000):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return softmax_obj(x, n, T)
    def jac(x):
        return softmax_grad(x, n, T)
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jac,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except:
        return h.copy()

    center = n - 1
    h_cur = h.copy()
    c5_best = compute_c5(h_cur)
    h_best = h_cur.copy()
    step = 0.05

        mc = np.max(corr)
        T_local = 2.0

    cv = compute_c5(h_cur)

def swap_local_search(h, n, rng, n_swaps=5000, t0=None, time_budget=None):
    """Try random pair swaps that maintain the sum constraint."""
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    dx = 2.0 / n
    center = n - 1
    for _ in range(n_swaps):
        if t0 is not None and time.time() - t0 > time_budget:

        i = rng.integers(0, n)

        if i == j:

        # Try moving mass from i to j or vice versa

            delta = 0.01 * direction
            hi = h_cur[i] + delta
            hj = h_cur[j] - delta
            if hi < 0 or hi > 1 or hj < 0 or hj > 1:

            h_try = h_cur.copy()
            h_try[i] = hi

    return h_cur, c5_cur

    dx = 2.0 / n
    center = n - 1

        w = np.exp(3.0 * (corr[top_idx] - mv))
