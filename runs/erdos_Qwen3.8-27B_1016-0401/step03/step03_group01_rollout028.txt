                     constraints=cns, options={'maxiter': mi, 'ftol': ft})
        return r.x
    except Exception:
        return h.copy()

def bflip(h, n, t_end):
    c5 = compute_c5(h)
    improved = True
    passes = 0
    while improved and time.time() < t_end and passes < 200:
        improved = False
        passes += 1
        for i in range(n):
            if time.time() > t_end:
                break
            hf = h.copy()
            hf[i] = 1.0 - hf[i]
            hf = project_feasible(hf, n)
            c5n = compute_c5(hf)
            if c5n < c5 - 1e-12:
                h = hf
                c5 = c5n
                improved = True
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(280)
        h[::2] = 1.0
    n = len(h)
    h = project_feasible(h, n)
    c5b = compute_c5(h)

    # Phase 1: Coarse-to-fine SLSQP
    sched1 = [(10, 30, 2000), (5, 20, 3000), (2, 10, 4000), (1, 5, 5000), (0.5, 3, 5000)]
    for T, k, mi in sched1:
        if time.time() - t0 > budget_s - 250:
            break
        hn = slsqp(h, n, T, k, mi, 1e-17)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5b:
            c5b, h = c5, hn

    # Phase 2: Binary local search + polish
    rem = budget_s - (time.time() - t0)
    if rem > 180:
        hb, c5b2 = bflip(h, n, t0 + rem * 0.2)
        if c5b2 < c5b:
            c5b, h = c5b2, hb
        hn = slsqp(h, n, 1.0, 5, 4000, 1e-17)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5b:
            c5b, h = c5, hn
        # Second binary pass from polished
        rem2 = budget_s - (time.time() - t0)
        if rem2 > 100:
            hb2, c5b3 = bflip(h, n, t0 + rem2 * 0.15)
            if c5b3 < c5b:
                c5b, h = c5b3, hb2

    # Phase 3: Ultra-fine SLSQP polish
    for T, k in [(0.2, 3), (0.1, 2), (0.05, 1)]:
        if time.time() - t0 > budget_s - 120:
            break
        hn = slsqp(h, n, T, k, 5000, 1e-19)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5b:
            c5b, h = c5, hn

    # Phase 4: Try different resolutions
    rem = budget_s - (time.time() - t0)
    if rem > 250:
        for nf in [n * 3 // 4, n // 2, n * 5 // 4]:
            if time.time() - t0 > budget_s - 120:
                break
            if nf == n or nf < 60:
                continue
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0.0, 2.0, nf, endpoint=False)
            hl = np.interp(x_new, x_old, h)
            hl = project_feasible(hl, nf)
            for T2, k2 in [(3, 10), (1, 5)]:
                if time.time() - t0 > budget_s - 60:
                    break
                hn = slsqp(hl, nf, T2, k2, 3000, 1e-16)
                hn = project_feasible(hn, nf)
                c5 = compute_c5(hn)
                if c5 < c5b:
                    c5b, h, n, hl = c5, hn, nf, hn

    # Phase 5: Targeted perturbations + re-optimize
    rem = budget_s - (time.time() - t0)
    npert = max(0, min(int(rem / 35), 8))
    for pi in range(npert):
        if time.time() - t0 > budget_s - 10:
            break
        hp = h.copy()
        nf2 = int(rng.integers(2, 6))
        ix = rng.choice(n, size=nf2, replace=False)
        hp[ix] = 1.0 - hp[ix]
        hp = project_feasible(hp, n)
        hn = slsqp(hp, n, 2.0, 8, 2000, 1e-15)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5b:
            c5b, h = c5, hn

    # Grand final polish
    if time.time() - t0 < budget_s - 30:
        hn = slsqp(h, n, 0.1, 2, 4000, 1e-19)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5b:
            c5b, h = c5, hn

    h = project_feasible(h, n)
    c5f = min(compute_c5(h), c5b)
    return (h, c5f, n)
