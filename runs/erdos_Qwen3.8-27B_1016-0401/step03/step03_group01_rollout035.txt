def swap_delta(h, a, b, n):
    """Compute change in correlation when h[a]:1->0, h[b]:0->1"""
    # Returns a vector of length 2n-1 (corresponding to np.correlate 'full' output)
    # np.correlate 'full' index: index = k + (n-1) where k is the shift
    # c_full[idx] corresponds to shift s = idx - (n-1)
    
    delta = np.zeros(2*n - 1)
    offset = n - 1
    
    # h[a]: 1 -> 0
    # First term (h[a] as first factor): -(1 - h[a-s]) for s in [a-n+1, a]
    # Second term (h[a] as second factor via h[i-s]=h[a], i=a+s): +h[a+s] for s in [-a, n-a-1]
    
    for s in range(max(a-n+1, -(n-1)), min(a, n-1) + 1):
        if 0 <= a-s < n:
            delta[s + offset] -= (1.0 - h[a-s])
    for s in range(max(-a, -(n-1)), min(n-a-1, n-1) + 1):
        if 0 <= a+s < n:
            delta[s + offset] += h[a+s]
    
    # h[b]: 0 -> 1
    # First term: +(1-h[b-s]) for s in [b-n+1, b]
    # Second term: -h[b+s] for s in [-b, n-b-1]
    
    for s in range(max(b-n+1, -(n-1)), min(b, n-1) + 1):
        if 0 <= b-s < n:
            delta[s + offset] += (1.0 - h[b-s])
    for s in range(max(-b, -(n-1)), min(n-b-1, n-1) + 1):
        if 0 <= b+s < n:
            delta[s + offset] -= h[b+s]
    
    return delta

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_binary(h, n):
    """Project to binary with exactly n/2 ones"""
    target = n // 2
    h_bin = (h >= 0.5).astype(float)
    if h_bin.sum() > target:
        # Need to turn off some 1s
        ones = np.where(h_bin == 1)[0]
        # Turn off the ones with smallest h value
        order = np.argsort(h[ones])
        h_bin[ones[target:]] = 0
    elif h_bin.sum() < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        h_bin[zeros[:target - int(h_bin.sum())]] = 1
    return h_bin

def project_feasible(y, n):

        if abs(s - target) < 1e-14:

        if s > target:

        else:
            hi = mid

    """For each position a, compute the effect on correlation of flipping h[a]
    (i.e., h[a] -> 1-h[a]). Returns array of shape (n, 2n-1)."""
    effects = np.zeros((n, 2*n-1))
    offset = n - 1
    one_minus_h = 1.0 - h
    
    for a in range(n):
        # h[a] changes: the effect on c[s] = sum_i h[i]*(1-h[i-s])
        # As first factor (h[a]): contribution was h[a]*(1-h[a-s]) -> (1-h[a])*(1-h[a-s])
        #   change: (1-h[a])*(1-h[a-s]) - h[a]*(1-h[a-s]) = (1-2*h[a])*(1-h[a-s])
        #   valid when 0 <= a-s < n, i.e., a-n+1 <= s <= a
        # As second factor (h[i-s]=h[a], i=a+s): contribution was h[a+s]*(1-h[a]) -> h[a+s]*(1-(1-h[a]))
        #   change: h[a+s]*(h[a] - (1-h[a])) = h[a+s]*(2*h[a]-1)
        #   valid when 0 <= a+s < n, i.e., -a <= s <= n-a-1

        for s in range(s_lo, s_hi + 1):
            a_minus_s = a - s
