_G_T = 1.0

    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    sh = _G_T * (corr - mc)
    e = np.exp(sh)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 1.0

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)

        m1 = j >= shift

        grad[m2] -= h[j[m2] + shift]

        m2 = j >= s
