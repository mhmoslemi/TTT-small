def project_feasible(y, n):
    target = n / 2.0
    s = y.sum()
    if abs(s - target) < 1e-10 and y.min() >= -1e-10 and y.max() <= 1 + 1e-10:

def shift_grad(h, n, shift):
    grad = np.zeros(n)
    if shift == 0:
        return 2.0 * (h - 0.5)
    if shift > 0:
        nv = n - shift
        grad[:nv] += (1.0 - h[shift:])
        grad[shift:] -= (1.0 - h[:nv])
    else:
        s = -shift
        nv = n - s
        grad[s:] += (1.0 - h[:nv])
        grad[:nv] -= (1.0 - h[s:])
    return grad

def topk_subgrad(h, n, K):
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    n_corr = 2 * n - 1
    if K >= n_corr:
        idx = np.arange(n_corr)
        w = np.ones(n_corr) / n_corr
    else:
        top_idx = np.argpartition(corr, -K)[-K:]
        tv = corr[top_idx]
        w = tv / max(tv.sum(), 1e-15)
        # Boost weight on the max
        w = w ** 4
        w /= w.sum()
        idx = top_idx
    g = np.zeros(n)
    for i in range(len(idx)):
        g += w[i] * shift_grad(h, n, idx[i] - center)
    return g * dx, mc * dx

def slsqp_smooth(h, n, T, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    nc = 2 * n - 1
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq',
        'fun': lambda x: x.sum() - target,
        'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(nc):
            if sw[oi] < 1e-12:
                continue
            grad += sw[oi] * shift_grad(x, n, oi - center)
        return grad * dx
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

from scipy.optimize import minimize
import time

_SLQ_N = 0
_SLQ_TGT = 0.0

def _c_fun(x):
    return x.sum() - _SLQ_TGT

def _c_jac(x):
    return np.ones(_SLQ_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    if shift == 0:
        return 2.0 * (h - 0.5)
    grad = np.zeros(n)
    if shift > 0:
        nv = n - shift
        grad[:nv] += 1.0 - h[shift:]
        grad[shift:] -= 1.0 - h[:nv]
    else:
        s = -shift
        nv = n - s
        grad[s:] += 1.0 - h[:nv]
        grad[:nv] -= 1.0 - h[s:]
    return grad

def topk_subgrad(h, n, K):
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    nc = 2 * n - 1
    if K >= nc:
        idx = np.arange(nc)
        w = np.ones(nc) / nc
    else:
        idx = np.argpartition(corr, -K)[-K:]
        tv = corr[idx]
        w = (tv / max(tv.max(), 1e-15)) ** 6
        ws = w.sum()
        w = w / ws if ws > 1e-15 else np.ones(K) / K
    g = np.zeros(n)
    for i in range(len(idx)):
        g += w[i] * shift_grad(h, n, int(idx[i]) - center)
    return g * dx, float(np.max(corr) * dx)

def sgrad_loop(h0, n, t0, tmax, maxit=600000):
    h = h0.copy()
    hp = h0.copy()
    ha = h0.copy()
    cb = compute_c5(h)
    hb = h0.copy()
    K = 3
    a0 = 0.6
    ni = 0
    for it in range(maxit):
        if time.time() - t0 > tmax:
            break
        if it > 0 and it % 10000 == 0:
            if ni > 40000:
                K = min(K * 2, 100)
                a0 *= 0.6
                rng_r = np.random.default_rng(it)
                h = project_feasible(h + 0.015 * rng_r.normal(0, 1, n), n)
                ni = 0
            elif ni > 15000:
                K = min(K + 3, 80)
                ni = 0
        beta = (it + 1.0) / (it + 2.0) if it > 0 else 0.0
        hy = np.clip(h + beta * (h - hp), 0.0, 1.0)
        g, _ = topk_subgrad(hy, n, K)
        at = a0 / np.sqrt(float(it) + 1.0)
        hn = project_feasible(hy - at * g, n)
        hp = h
        h = hn
        ha += (hn - ha) / float(it + 1)
        if (it + 1) % 5000 == 0:

                cb = min(c1, c2)
                hb = h.copy() if c1 <= c2 else project_feasible(ha, n).copy()
                ni = 0
            else:
                ni += 5000
    cf = compute_c5(h)
    if cf < cb:
        cb, hb = cf, h.copy()
    return hb, cb

def slsqp_smooth(h, n, T, mxi, ftol):
    global _SLQ_N, _SLQ_TGT
    _SLQ_N, _SLQ_TGT = n, n / 2.0
    dx = 2.0 / n
    ctr = n - 1
    nc = 2 * n - 1
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]

    def obf(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        mc = c.max()
        e = np.exp(T * (c - mc))
        return (mc + np.log(e.sum()) / T) * dx

    def jcf(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        mc = c.max()
        e = np.exp(T * (c - mc))
        sw = e / e.sum()

            if sw[oi] > 1e-12:
                g += sw[oi] * shift_grad(x, n, oi - ctr)

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_TGT = 0.0
_SM_N = 0
_SM_T = 1.0
_SM_DX = 0.0
_SM_CTR = 0
_SM_NC = 0

def _cf(x):
    return x.sum() - _G_TGT

def _cj(x):
    return np.ones(_G_N)

def _so(x):
    c = np.correlate(x, 1.0 - x, mode="full")
    mc = c.max()
    e = np.exp(_SM_T * (c - mc))
    return (mc + np.log(e.sum()) / _SM_T) * _SM_DX

def _sj(x):
    c = np.correlate(x, 1.0 - x, mode="full")
    mc = c.max()
    e = np.exp(_SM_T * (c - mc))
    sw = e / e.sum()
    g = np.zeros(_SM_N)
    for oi in range(_SM_NC):
        if sw[oi] > 1e-12:
            g += sw[oi] * sgrad(x, _SM_N, oi - _SM_CTR)
    return g * _SM_DX

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def project(y, n):
    t = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(80):
        m = (lo + hi) / 2.0
        s = np.clip(y - m, 0.0, 1.0).sum()
        if abs(s - t) < 1e-13:
            break
        if s > t:
            lo = m
        else:
            hi = m
    return np.clip(y - m, 0.0, 1.0)

def sgrad(h, n, sh):
    if sh == 0:
        return 2.0 * (h - 0.5)
    g = np.zeros(n)
    if sh > 0:
        nv = n - sh
        g[:nv] += 1.0 - h[sh:]
        g[sh:] -= 1.0 - h[:nv]
    else:
        s = -sh
        nv = n - s
        g[s:] += 1.0 - h[:nv]
        g[:nv] -= 1.0 - h[s:]
    return g

def tsub(h, n, K):
    dx = 2.0 / n
    ctr = n - 1
    c = np.correlate(h, 1.0 - h, mode="full")
    nc = 2 * n - 1
    if K >= nc:
        idx = np.arange(nc)
        w = np.ones(nc) / nc
    else:
        idx = np.argpartition(c, -K)[-K:]
        tv = c[idx]
        w = (tv / max(tv.max(), 1e-15)) ** 6
        ws = w.sum()
        w = w / ws if ws > 1e-15 else np.ones(K) / K
    g = np.zeros(n)
    for i in range(len(idx)):
        g += w[i] * sgrad(h, n, int(idx[i]) - ctr)
    return g * dx

    hb = h0.copy()

    return hb, cb

def slsqp(h, n, T, mxi):
    global _G_N, _G_TGT, _SM_N, _SM_T, _SM_DX, _SM_CTR, _SM_NC
    _G_N = _SM_N = n
    _G_TGT = n / 2.0
    _SM_T = T
    _SM_DX = 2.0 / n
    _SM_CTR = n - 1
    _SM_NC = 2 * n - 1
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        r = minimize(_so, h, method='SLSQP', jac=_sj,
                     bounds=bnds, constraints=cons,
                     options={'maxiter': mxi, 'ftol': 1e-15})
        return r.x
    except Exception:
        return h.copy()
