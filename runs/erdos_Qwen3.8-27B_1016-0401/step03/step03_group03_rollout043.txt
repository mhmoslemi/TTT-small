    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0

        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:

        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

    sh = _G_T * (corr - mc)
    e = np.exp(sh)

        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        top_k = np.argsort(corr)[-4:]
        g = np.zeros(n)
        for ik in top_k:
            g += shift_grad_raw(h_cur, n, ik - center)
        g *= dx / len(top_k)
        pos_mask = g > 0.004
        neg_mask = g < -0.004
        if pos_mask.any() and neg_mask.any():
            pos_idx = np.where(pos_mask)[0]
            neg_idx = np.where(neg_mask)[0]
            improved = False
            for _ in range(6):
                ip = pos_idx[rng.integers(len(pos_idx))]
                inn = neg_idx[rng.integers(len(neg_idx))]
                delta = min(0.035, h_cur[ip], 1.0 - h_cur[inn])
                if delta < 0.001:
                    continue
                h_try = h_cur.copy()
                h_try[ip] -= delta
                h_try[inn] += delta
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < c5_cur - 1e-9:
                    h_cur = h_try
                    c5_cur = c5_try
                    improved = True
                    break
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Quick SLSQP warmup
    for T in [6.0, 3.0, 1.5, 0.8]:
        if time.time() - t0 > budget_s - 650:
            break
        h_new = do_slsqp(h_best, n, T, 6000, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Main Nesterov (adaptive K, longer)
    time_left = budget_s - (time.time() - t0) - 320
    if time_left > 20:
        h_sg, c5_sg = nesterov_opt(h_best, n, t0, time_left, K_start=6, K_end=15, max_iters=500000)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase 3: SLSQP fine polish (low temperature)
    for T in [0.6, 0.4, 0.25, 0.15, 0.1]:
        if time.time() - t0 > budget_s - 160:
            break
        h_new = do_slsqp(h_best, n, T, 15000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Local coordinate polish
    if time.time() - t0 < budget_s - 110:
        h_lp, c5_lp = local_polish(h_best, n, rng, steps=10000)
        if c5_lp < c5_best:
            c5_best = c5_lp
            h_best = h_lp

    # Phase 5: Second Nesterov burst (higher K)
    time_left = budget_s - (time.time() - t0) - 70
    if time_left > 20:
        h_sg2, c5_sg2 = nesterov_opt(h_best, n, t0, time_left, K_start=10, K_end=18, max_iters=200000)
        if c5_sg2 < c5_best:
            c5_best = c5_sg2
            h_best = h_sg2

    # Phase 6: Perturbation + SLSQP restarts
    for pi in range(12):
        if time.time() - t0 > budget_s - 25:
            break
        sigma = 0.002 + 0.004 * (pi % 4)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_new = do_slsqp(h_p, n, 1.0, 7000, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Final SLSQP polish
    if time.time() - t0 < budget_s - 8:
        h_new = do_slsqp(h_best, n, 0.12, 14000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
