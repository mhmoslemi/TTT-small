import time

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                if a <= 0: continue
                h[i] += a; d -= a
            else:
                a = min(-d, h[i])
                if a <= 0: continue
                h[i] -= a; d += a
            if abs(d) < 1e-12: break
    return h

def sm_obj(x, T, target):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted - np.max(shifted))
    return (mc + np.log(e.sum()) / T) * (2.0 / n) + 100.0 * (x.sum() - target) ** 2

        elif o > 0:

            grad[:n-p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n-p]

    grad += 200.0 * (x.sum() - target)
    return grad

    zero_pos = np.where(h == 0)[0].tolist()

        T = T_start * (T_end / T_start) ** frac

            break

        h[i], h[j] = h[j], h[i]

                h_best = h.copy()

            zero_pos.remove(j); zero_pos.append(i)

                zero_pos.remove(j); zero_pos.append(i)
            else:
                h[i], h[j] = h[j], h[i]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                if a <= 0:
                    continue
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                if a <= 0:
                    continue
                h[i] -= a
                d += a
            if abs(d) < 1e-12:
                break
    return h

def binary_fix(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    h = (h > 0.5).astype(float)
    s = int(h.sum())
    if s < target:
        cand = np.where(h == 0)[0]
        h[cand[:target - s]] = 1.0
    elif s > target:
        cand = np.where(h == 1)[0]
        h[cand[:s - target]] = 0.0
    return h

def sa_optimize(h, n_steps, T_lo, T_hi, rng):
    n = len(h)
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    one_pos = np.where(h == 1)[0].tolist()
    zero_pos = np.where(h == 0)[0].tolist()
    stuck = 0
    for step in range(n_steps):
        frac = step / max(n_steps - 1, 1)
        T = T_hi * (T_lo / T_hi) ** frac
        if T < 1e-10:
            T = 1e-10
        if not one_pos or not zero_pos:
            break
        i = one_pos[rng.integers(len(one_pos))]
        j = zero_pos[rng.integers(len(zero_pos))]
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new <= c5_cur:
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
                stuck = 0
            else:
                stuck += 1
            one_pos.remove(i)
            one_pos.append(j)
            zero_pos.remove(j)
            zero_pos.append(i)
        elif rng.random() < np.exp(-(c5_new - c5_cur) / T):
            c5_cur = c5_new
            one_pos.remove(i)
            one_pos.append(j)
            zero_pos.remove(j)
            zero_pos.append(i)
            stuck += 1
        else:
            h[i], h[j] = h[j], h[i]
        if stuck > 50000 and T > T_lo * 10:
            stuck = 0
            T = min(T * 2.0, 0.1)
    return h_best, c5_best

def lag_obj(x, li, n):
    return np.correlate(x, 1.0 - x, mode="full")[li] * (2.0 / n)

def lag_grad(x, li, n):
    o = li - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * x
    elif o > 0:
        g[o:] += 1.0 - x[:n - o]
        g[:n - o] += -x[o:]
    else:
        p = -o
        g[:n - p] += 1.0 - x[p:]
        g[p:] += -x[:n - p]
    return g * dx

def sm_obj(x, T, target):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh - sh.max())
    return (mc + np.log(e.sum()) / T) * (2.0 / n) + 50.0 * (x.sum() - target) ** 2

def sm_grad(x, T, target):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh - sh.max())
    sw = e / e.sum()
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx + 100.0 * (x.sum() - target)

def slsqp_refine(h, n, target, bounds, T, maxiter):
    res = minimize(sm_obj, h.copy(), args=(T, target), method='SLSQP',
                   jac=sm_grad, bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return fix_sum(res.x, target)

def slsqp_lag_refine(h, n, target, bounds, li, maxiter):
    cons = [dict(type='eq', fun=lambda x: x.sum() - target,
                 jac=lambda x: np.ones(len(x)))]
    res = minimize(lag_obj, h.copy(), args=(li, n), method='SLSQP',
                   jac=lag_grad, bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return fix_sum(res.x, target)

def make_kronecker(n, alpha):
    return (np.mod(np.arange(n, dtype=float) * alpha, 1.0) < 0.5).astype(float)

            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass
    # Try worst-lag focused
    if elapsed() < p2_end:
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst = int(np.argmax(corr))
        try:
            h_new = slsqp_lag_refine(h_best, n, target, bounds, worst, 3000)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 3: Binary push + more SA (15% budget)
    p3_end = t0 + budget_s - 15.0
    h_bin2 = binary_fix(h_best, target)
    c5_bin2 = compute_c5(h_bin2)
    if c5_bin2 < c5_best:
        c5_best = c5_bin2
        h_best = h_bin2
    while elapsed() < p3_end:
        n_steps = min(300000, max(10000, int((p3_end - time.time()) * 8000)))
        h_new, c5_new = sa_optimize(h_best, n_steps, T_lo * 0.5, T_hi * 0.3, rng)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 4: Final SLSQP polish (remaining)
    h_push = binary_fix(h_best, target)
    c5_push = compute_c5(h_push)
    if c5_push < c5_best:
        c5_best = c5_push
        h_best = h_push
    if elapsed() < t0 + budget_s - 5:
        try:
            h_new = slsqp_refine(h_best, n, target, bounds, 8000, 5000)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass
        # One more binary push
        h_bin3 = binary_fix(h_best, target)
        c5_bin3 = compute_c5(h_bin3)
        if c5_bin3 < c5_best:
            c5_best = c5_bin3
            h_best = h_bin3

    h_best = fix_sum(h_best, target)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
