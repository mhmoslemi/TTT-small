    return float(np.max(corr) * dx)

        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:

        else:
            hi = mid

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():

        if m2.any():
            g[m2] -= h[j[m2] + shift]

    c = corr[idx]
    m = c.max()
    e = np.exp(np.clip(T * (c - m), -500, 0))
    return (m + np.log(e.sum()) / T) * dx

def _smooth_max_jac(x, n, T, top_k):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    m = c.max()
    e = np.exp(np.clip(T * (c - m), -500, 0))
    w = e / e.sum()
    grad = np.zeros(n)
    for wi, ci in zip(w, idx):
        if wi > 1e-10:
            grad += wi * shift_grad(x, n, ci - (n - 1))
    return grad * dx

def _lbfgsb_opt(h, n, T, top_k, maxiter, lam=50.0):
    target = n / 2.0

    def obj(x):
        base = _smooth_max_obj(x, n, T, top_k)
        pen = lam * (x.sum() - target) ** 2
        return base + pen

    def jac(x):
        base_g = _smooth_max_jac(x, n, T, top_k)
        base_g += 2.0 * lam * (x.sum() - target)
        return base_g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jac, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-13, 'gtol': 1e-9})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def _slsqp_opt(h, n, T, top_k, maxiter, ftol):
    target = n / 2.0

    def obj(x):
        return _smooth_max_obj(x, n, T, top_k)

    def jac(x):
        return _smooth_max_jac(x, n, T, top_k)

    def cf(x):
        return x.sum() - target

    def cj(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def _cyclical_subgrad(h_init, n, t_end, rng, n_iters=200000):
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    for it in range(n_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        top5 = np.argsort(corr)[-5:]
        focus = top5[it % 5]
        s = focus - (n - 1)
        beta = float(it) / (it + 1.5)
        h_y = h_cur + beta * (h_cur - h_prev)
        g = shift_grad(h_y, n, s) * dx
        alpha = 0.35 / (1.0 + 0.004 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 15000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < c5_best:
                c5_best, h_best = c1, h_cur.copy()
            c2 = compute_c5(project_feasible(h_avg, n))
            if c2 < c5_best:
                c5_best, h_best = c2, project_feasible(h_avg, n).copy()
    return h_best, c5_best

def _perturb(h, n, rng, pi):
    mode = pi % 6
    hp = h.copy()
    if mode == 0:
        sig = 0.003 + 0.001 * (pi // 6)
        hp += rng.normal(0, sig, n)
    elif mode == 1:
        lb = rng.integers(0, n - 28)
        ln = rng.integers(8, 28)
        hp[lb:lb + ln] = 1.0 - hp[lb:lb + ln]
    elif mode == 2:
        hp = np.roll(hp, int(rng.integers(1, 10)))
    elif mode == 3:
        freq = rng.uniform(1.0, 7.0)
        hp += 0.008 * np.sin(2.0 * np.pi * freq * np.arange(n) / n)
    elif mode == 4:
        lb = rng.integers(0, n - 20)
        ln = rng.integers(10, 22)
        hp[lb:lb + ln] *= 0.85
    else:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        hp[[i, j]] = hp[[j, i]]
    return project_feasible(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n = 280
        h_best = np.clip(0.5 + 0.25 * np.cos(2.0 * np.pi * np.arange(n) / 37.0), 0.0, 1.0)
        h_best = project_feasible(h_best, n)

    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B coarse sweep (fast, 30% time)
    t1 = t0 + budget_s * 0.30
    T_sched = [(80, 40, 4000), (40, 35, 4000), (20, 30, 4000), (10, 25, 5000),
               (5, 20, 5000), (2, 16, 6000), (1, 12, 7000), (0.5, 10, 8000)]
    for T, tk, mi in T_sched:
        if time.time() > t1:
            break
        hn = _lbfgsb_opt(h_best, n, T, tk, mi, lam=50.0)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 2: SLSQP medium polish (15% time)
    t2 = t0 + budget_s * 0.45
    for T, tk, mi in [(8, 20, 4000), (4, 15, 5000), (2, 12, 6000)]:
        if time.time() > t2:
            break
        hn = _slsqp_opt(h_best, n, T, tk, mi, 1e-14)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 3: Cyclical subgradient (15% time)
    t3 = t0 + budget_s * 0.60
    hc, c5c = _cyclical_subgrad(h_best, n, t3, rng)
    if c5c < c5_best:
        c5_best, h_best = c5c, hc
    # Polish the cyclical result
    if time.time() < t3 + 30:
        hn = _slsqp_opt(h_best, n, 2, 12, 4000, 1e-14)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 4: Multi-perturbation + L-BFGS-B + SLSQP (25% time)
    t4 = t0 + budget_s * 0.85
    for pi in range(50):
        if time.time() > t4:
            break
        hp = _perturb(h_best, n, rng, pi)
        hn = _lbfgsb_opt(hp, n, 2, 15, 2000, lam=50.0)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn
            if time.time() < t4:
                hn2 = _slsqp_opt(h_best, n, 1, 10, 3000, 1e-14)
                hn2 = project_feasible(hn2, n)
                c52 = compute_c5(hn2)
                if c52 < c5_best:
                    c5_best, h_best = c52, hn2

    # Phase 5: Ultra-fine SLSQP polish (10% time)
    t5 = t0 + budget_s - 10
    for T, tk, mi in [(1, 10, 6000), (0.5, 8, 8000), (0.3, 6, 10000)]:
        if time.time() > t5:
            break
        hn = _slsqp_opt(h_best, n, T, tk, mi, 1e-16)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Final verification
    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
