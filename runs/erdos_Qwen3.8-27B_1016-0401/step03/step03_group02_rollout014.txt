import numpy as np
from scipy.optimize import minimize
import time

_G = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    order = np.argsort(h) if diff > 0 else np.argsort(-h)
    rem = abs(diff)
    for i in order:
        if diff > 0:
            a = min(rem, 1.0 - h[i])
            h[i] += a
        else:
            a = min(rem, h[i])
            h[i] -= a
        rem -= a
        if rem < 1e-14:
            break
    return h

def grad_lag(h, o):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] += 1.0 - h[:n-o]
        g[:n-o] += -h[o:]
    else:
        p = -o
        g[:n-p] += 1.0 - h[p:]
        g[p:] += -h[:n-p]
    return g * dx

def sm_obj(x, T):
    n = len(x)
    c = get_corr(x)
    mx = c.max()
    s = T * (c - mx)
    return (mx + np.log(np.exp(s).sum()) / T) * (2.0 / n)

def sm_grad(x, T):
    n = len(x)
    c = get_corr(x)
    mx = c.max()

    cons = [dict(type='eq', fun=eq_fun, jac=eq_jac)]

        res = minimize(sm_obj, h, args=(T,), method='SLSQP', jac=sm_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h

def slsqp_lag(h, bounds, maxiter):
    cons = [dict(type='eq', fun=eq_fun, jac=eq_jac)]
    try:

def binary_cd(h, target, n_sweeps):
    n = len(h)
    c5_cur = compute_c5(h)
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):

                c5_cur = c5_new

            break
