import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def slsqp_lse(h, n, T, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        return (mc + np.log(np.exp(sh).sum()) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(n_corr):
            w = sw[oi]
            if w < 1e-12:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return grad * dx
    def c_fun(x):
        return x.sum() - target
    def c_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def subgrad_exact(h, n, t0, remaining, n_iter=80000):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    c5_best = compute_c5(h)
    h_best = h.copy()
    step = 0.002
    K = 2
    for it in range(n_iter):
        if time.time() - t0 > remaining:
            break
        if it % 5000 == 0:
            K = 2 + (it // 10000) * 2
            K = min(K, 8)
        corr = np.correlate(h, 1.0 - h, mode="full")
        mc = np.max(corr)
        if K == 1:
            idx = int(np.argmax(corr))
            g = shift_grad(h, n, idx - center)
        else:
            top_idx = np.argpartition(corr, -K)[-K:]
            tv = corr[top_idx]
            w = (tv / max(mc, 1e-15))**40
            ws = w.sum()
            if ws < 1e-15:
                w = np.ones(K) / K
            else:
                w /= ws
            g = np.zeros(n)
            for i in range(K):
                g += w[i] * shift_grad(h, n, top_idx[i] - center)
        h_trial = project_feasible(h - step * g, n)
        c5t = compute_c5(h_trial)
        if c5t < c5_best - 1e-10:
            c5_best = c5t
            h_best = h_trial.copy()
            h = h_trial
            step = min(step * 1.01, 0.008)
        else:
            step *= 0.85
            if step < 1e-8:
                break
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(280)
        h0[::2] = 1.0
    n = len(h0)
    h0 = project_feasible(h0, n)
    c5_best = compute_c5(h0)
    h_best = h0.copy()

    # Phase 1: SLSQP with high-T log-sum-exp (accurate max approximation)
    T_sched = [100, 50, 25, 12, 8, 5, 3, 2, 1]
    for T in T_sched:
        if time.time() - t0 > budget_s - 500:
            break
        maxit = 6000 if T >= 12 else 9000
        h_new = slsqp_lse(h_best, n, T, maxit, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Subgradient refinement on exact objective
    h_sg, c5_sg = subgrad_exact(h_best, n, t0, budget_s - 200)
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg
    h_best = project_feasible(h_best, n)

    # Phase 3: Final SLSQP polish
    if time.time() - t0 < budget_s - 120:
        h_new = slsqp_lse(h_best, n, 1.0, 6000, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        h_new = slsqp_lse(h_best, n, 2.0, 4000, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Short subgradient pass after polish
    h_sg2, c5_sg2 = subgrad_exact(h_best, n, t0, budget_s - 100, n_iter=30000)
    if c5_sg2 < c5_best:
        c5_best = c5_sg2
        h_best = h_sg2
    h_best = project_feasible(h_best, n)

    # Phase 5: Try alternate resolutions
    if time.time() - t0 < budget_s - 100:
        for n2 in [240, 320, 360]:
            if time.time() - t0 > budget_s - 60:
                break
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0, 2.0, n2, endpoint=False)
            h_alt = np.interp(x_new, x_old, h_best)
            h_alt = project_feasible(h_alt, n2)
            for T2 in [20, 5, 2]:
                if time.time() - t0 > budget_s - 40:
                    break
                h_new = slsqp_lse(h_alt, n2, T2, 5000, 1e-16)
                h_new = project_feasible(h_new, n2)
                c5a = compute_c5(h_new)
                if c5a < c5_best:
                    c5_best = c5a
                    h_best = h_new
                    n = n2
            h_best = project_feasible(h_best, n)

    # Phase 6: Random restart perturbations
    if time.time() - t0 < budget_s - 60:
        for pi in range(8):
            if time.time() - t0 > budget_s - 30:
                break
            h_p = h_best + rng.normal(0, 0.005, n)
            h_p = project_feasible(h_p, n)
            h_new = slsqp_lse(h_p, n, 3.0, 3000, 1e-16)
            h_new = project_feasible(h_new, n)
            c5p = compute_c5(h_new)
            if c5p < c5_best:
                c5_best = c5p
                h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
