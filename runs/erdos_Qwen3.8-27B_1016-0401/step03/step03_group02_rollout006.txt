import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        cnt = int(mask.sum())
        amt = diff / cnt
        if diff > 0:
            amt = min(amt, (1.0 - h[mask]).min() * 0.99)
            h[mask] += amt
        else:
            amt = min(-amt, h[mask].min() * 0.99)
            h[mask] -= amt
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, lag, n):
    center = n - 1
    o = lag - center
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] += 1.0 - h[:n-o]
        g[:n-o] += -h[o:]
    else:
        p = -o
        g[:n-p] += 1.0 - h[p:]
        g[p:] += -h[:n-p]
    return g * dx

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
