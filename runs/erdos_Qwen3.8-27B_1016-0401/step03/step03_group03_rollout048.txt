def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 2.0
    hi = float(y.max()) + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if np.any(m1):
            grad[m1] += (1.0 - h[j[m1] - shift])
        if np.any(m2):
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if np.any(m1):
            grad[m1] += (1.0 - h[j[m1] + s])
        if np.any(m2):
            grad[m2] -= h[j[m2] - s]
    return grad

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _T) * (2.0 / _N)

    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _T * (corr - mc)
    e = np.exp(sh)

    center = _N - 1

    grad = np.zeros(_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-12:
            continue
        grad += w * shift_grad_raw(x, _N, oi - center)
    return grad * dx

                       bounds=bounds, constraints=cons,

        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    for pi in range(8):
        if time.time() > budget_s - 25:
            break
        sigma = 0.002 + 0.003 * (pi % 3)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_new = do_slsqp(h_p, n, 0.7, 6000, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Final SLSQP squeeze
    if time.time() < budget_s - 8:
        h_new = do_slsqp(h_best, n, 0.12, 15000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        # One more targeted polish on the result
        if time.time() < budget_s - 3:
            h_fp, c5_fp = targeted_polish(h_best, n, budget_s + t0 - 2)
            if c5_fp < c5_best:
                c5_best = c5_fp
                h_best = h_fp

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
