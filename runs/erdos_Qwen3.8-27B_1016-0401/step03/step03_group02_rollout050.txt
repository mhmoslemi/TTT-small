- -delta * h[j-o] for valid second index

The first term is a shifted version of (1-h), and the second is a shifted version of h.

```python
def all_flip_effects(h):
    """For each bit j, compute the effect of flipping it on the full correlation."""
    n = len(h)
    delta = 1.0 - 2.0 * h  # shape (n,)
    center = n - 1
    
    # For offset o >= 0:
    # First term at (j, o): valid if j <= n-1-o, contribution = delta[j] * (1-h[j+o])
    # Second term at (j, o): valid if j >= o, contribution = -delta[j] * h[j-o]
    
    # For offset o < 0 (let p = -o > 0):
    # First term: valid if j >= p, contribution = delta[j] * (1-h[j-p])
    # Second term: valid if j <= n-1-p, contribution = -delta[j] * h[j+p]
    
    # Let's compute the effect for all (j, o) pairs as a matrix, then sum over j
    # to get the correlation change for a single flip.
    # Actually, I want: for each j, the full vector of correlation changes.
    # That's an n × (2n-1) matrix. For n=340, that's 340 × 679 = ~231K values.
    
    # Alternative: just loop over j and compute the effect vector.
    effects = np.zeros((n, 2*n-1))
    one_minus_h = 1.0 - h
    
    for j in range(n):
        d = delta[j]
        # Offsets o >= 0
        for o in range(0, n):
            eff = 0.0
            if j <= n-1-o:
                eff += one_minus_h[j+o]
            if j >= o:
                eff -= h[j-o]
            effects[j, center+o] = d * eff
        # Offsets o < 0
        for p in range(1, n):
            eff = 0.0
            if j >= p:
                eff += one_minus_h[j-p]
            if j <= n-1-p:
                eff -= h[j+p]
            effects[j, center-p] = d * eff
    return effects
```

This is O(n²) which is fine for n up to ~1000.

```python
    return np.correlate(h, 1.0 - h, mode="full")
```

```python
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12: break
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                if a <= 0: continue
                h[i] += a; diff -= a
            else:
                a = min(-diff, h[i])
                if a <= 0: continue
                h[i] -= a; diff += a
            if abs(diff) < 1e-12: break
    return h

    n = len(h)
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    hb = np.zeros(n)
    hb[idx] = 1.0

    n = len(h)
    center = n - 1
    d = 1.0 - 2.0 * h
    omh = 1.0 - h
    eff = np.zeros((n, 2*n-1))
    for o in range(n):
        eff[:n-o, center+o] += d[:n-o] * omh[o:]
        eff[o:, center+o] -= d[o:] * h[:n-o]
    for p in range(1, n):
        eff[p:, center-p] += d[p:] * omh[:n-p]
        eff[:n-p, center-p] -= d[:n-p] * h[p:]
    return eff

        nm = np.max(corr[None, :] + e, axis=1) * dx

        h[j] ^= 1.0
        corr += e[j]

def slsqp_minmax(h, lags, maxiter=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        c = np.correlate(x, 1.0-x, mode="full") * dx
        T = 300.0
        mv = np.max(c)
        return mv + np.log(np.sum(np.exp(T*(c-mv)))) / T
    
    def grad(x):
        c = np.correlate(x, 1.0-x, mode="full") * dx
        T = 300.0
        mv = np.max(c)
        sw = np.exp(T*(c-mv)); sw /= sw.sum()
        g = np.zeros(n)
        center = n-1
        for idx, lag in enumerate(lags):
            w = sw[lag]
            if w < 1e-12: continue
            o = lag - center
            if o == 0:
                g += w*(1.0 - 2.0*x)
            elif o > 0:
                g[:n-o] += w*(1.0-x[o:]); g[o:] -= w*x[:n-o]
            else:
                p = -o
                g[p:] += w*(1.0-x[:n-p]); g[:n-p] -= w*x[p:]
        return g
    
    def eq(x): return x.sum() - target
    def eqj(x): return np.ones(n)
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=grad,
                      bounds=[(0,1)]*n,
                      constraints=[{'type':'eq','fun':eq,'jac':eqj}],
                      options={'maxiter':maxiter, 'ftol':1e-18})
        return res.x
    except:
        return h
```

corr[center+o] = sum_{i=0}^{n-1-o} h[i] * (1-h[i+o])

When h[j] changes by delta:
- If j is in the "first" position (i=j): need 0 ≤ j ≤ n-1-o
  Δ = delta * (1-h[j+o])
- If j is in the "second" position (i+o=j, so i=j-o): need 0 ≤ j-o ≤ n-1-o, i.e., o ≤ j ≤ n-1
  Δ = h[j-o] * (-(delta)) = -delta * h[j-o]

So the total effect at offset o for bit j:
effect[j, center+o] = delta[j] * (1-h[j+o]) [if 0≤j≤n-1-o] - delta[j] * h[j-o] [if o≤j≤n-1]

In vectorized form for all j at a fixed o:
- First term: j ranges from 0 to n-1-o
  eff[0:n-o, center+o] += d[0:n-o] * omh[o:n]
- Second term: j ranges from o to n-1
  eff[o:n, center+o] -= d[o:n] * h[0:n-o]

For offset -p (i.e., center-p) where p > 0:
corr[center-p] = sum_{i=p}^{n-1} h[i] * (1-h[i-p])

When h[j] changes by delta:
- If j is "first" (i=j): need p ≤ j ≤ n-1
  Δ = delta * (1-h[j-p])
- If j is "second" (i-p=j, i=j+p): need 0 ≤ j ≤ n-1-p
  Δ = -delta * h[j+p]

Vectorized:
- First term: j ranges from p to n-1
  eff[p:n, center-p] += d[p:n] * omh[0:n-p]
- Second term: j ranges from 0 to n-1-p
  eff[0:n-p, center-p] -= d[0:n-p] * h[p:n]

This looks correct! Let me verify with a small example.

n=3, h=[1,0,1], center=2
corr[2+(-1)] = corr[1] = sum_{i=1}^{2} h[i]*(1-h[i-1]) = h[1]*(1-h[0]) + h[2]*(1-h[1]) = 0*0 + 1*1 = 1

Using my formula for p=1:
- First term: j in [1,2], eff[j, 1] += d[j]*omh[j-1]

Total effect at j=1: 0 + (-1) = -1
Total effect at j=2: -1 + 0 = -1
Total effect at j=0: 0 + 0 = 0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                if a <= 0:
                    continue
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                if a <= 0:
                    continue
                h[i] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def binarize(h):
    n = len(h)
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def flip_effects(h):
    n = len(h)
    center = n - 1
    d = 1.0 - 2.0 * h
    omh = 1.0 - h
    eff = np.zeros((n, 2 * n - 1))
    for o in range(n):
        eff[:n-o, center+o] += d[:n-o] * omh[o:]
        eff[o:, center+o] -= d[o:] * h[:n-o]
    for p in range(1, n):
        eff[p:, center-p] += d[p:] * omh[:n-p]
        eff[:n-p, center-p] -= d[:n-p] * h[p:]
    return eff

def binary_ls(h, max_sweeps=200):
    n = len(h)
    dx = 2.0 / n
    h = binarize(h).copy()
    corr = compute_corr(h).astype(float)
    c5 = float(np.max(corr) * dx)
    for _ in range(max_sweeps):
        e = flip_effects(h)
        nm = (corr[None, :] + e)
        nm_max = np.max(nm, axis=1) * dx
        j = int(np.argmin(nm_max))
        if nm_max[j] >= c5 - 1e-14:
            break
        h[j] = 1.0 - h[j]
        corr = corr + e[j]
        c5 = float(nm_max[j])
    return h, c5

def slsqp_minmax(h, n_lags=5, maxiter=400):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    T = 400.0

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full") * dx
        mv = np.max(c)
        return mv + np.log(np.sum(np.exp(T * (c - mv)))) / T

    def grad(x):
        c = np.correlate(x, 1.0 - x, mode="full") * dx
        mv = np.max(c)
        sw = np.exp(T * (c - mv))
        sw /= sw.sum()
        g = np.zeros(n)
        center = n - 1
        for lag in range(2 * n - 1):
            w = sw[lag]
            if w < 1e-12:
                continue
            o = lag - center
            if o == 0:
                g += w * (1.0 - 2.0 * x)
            elif o > 0:
                g[:n-o] += w * (1.0 - x[o:])
                g[o:] -= w * x[:n-o]
            else:
                p = -o
                g[p:] += w * (1.0 - x[:n-p])
                g[:n-p] -= w * x[p:]
        return g

    def eq(x):
        return x.sum() - target

    def eqj(x):
        return np.ones(n)

    try:
        res = minimize(obj, h, method='SLSQP', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq, 'jac': eqj}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return res.x
    except Exception:
        return h

def coord_descent(h, n_sweeps=3):
    n = len(h)
    target = n / 2.0
    steps = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01]
    c5_cur = compute_c5(h)

                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur
