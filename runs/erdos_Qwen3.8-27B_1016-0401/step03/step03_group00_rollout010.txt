from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()

        else:

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

def top_k_grad(h, n, top_k, weights=None):
    """Compute gradient of weighted sum of top-k correlations."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-top_k:]
    if weights is None:
        weights = np.ones(top_k) / top_k
    g = np.zeros(n)
    for w, ci in zip(weights, idx):
        if w > 1e-12:
            g += w * shift_grad(h, n, ci - (n - 1))
    return g, idx, corr[idx]

def softmax_obj(h, n, T, top_k):
    """Softmax approximation of max over top-k correlations."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    val = (m + np.log(e.sum()) / T) * (2.0 / n)
    return val

def softmax_grad(h, n, T, top_k):
    """Gradient of softmax objective."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for wi, ci in zip(w, idx):
        if wi > 1e-12:
            g += wi * shift_grad(h, n, ci - (n - 1))
    return g * (2.0 / n)

_G = {}

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

        g += shift_grad(h, n, ci - (n - 1))

    return g
