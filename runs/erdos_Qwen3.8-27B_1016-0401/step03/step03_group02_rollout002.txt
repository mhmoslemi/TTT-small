def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_onto_constraint(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Distribute the difference
    if diff > 0:
        # Need to add: add to elements with most room (closest to 1)
        room = 1.0 - h
        # Add proportionally to room
        total_room = room.sum()
        if total_room > 0:
            h += diff * room / total_room
            h = np.clip(h, 0.0, 1.0)
    else:
        # Need to subtract: subtract from elements with most value
        total_val = h.sum()
        if total_val > 0:
            h += diff * h / total_val
            h = np.clip(h, 0.0, 1.0)
    # Fix any remaining difference
    diff = target - h.sum()
    for _ in range(100):
        if abs(diff) < 1e-14:

            h[idx] += add
            diff -= add
        else:
            idx = np.argmax(h)
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
    return h

            mask2[o:] = 1.0    # j in [o, n-1], i.e., j-o >= 0

            grad[:n-p] -= w * h[p:]           # -h[j+p] for j=0..n-p-1

def project(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    for _ in range(200):

            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break

            h[mask] += add * (1.0 - h[mask]) / (1.0 - h[mask]).sum()
        else:
            mask = h > 0.0
            if not mask.any():
                break
            sub = min(-diff, h[mask].sum())
            h[mask] += sub * h[mask] / h[mask].sum()
        diff = target - h.sum()
    # Final fix
    diff = target - h.sum()
    if diff > 1e-14:
        idx = np.argsort(1.0 - h)[::-1]

            a = min(diff, 1.0 - h[i])

            s = min(-diff, h[i])
            h[i] -= s
            diff += s
            if diff > -1e-14:
                break
    return h

    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted -= shifted.max()
    e = np.exp(shifted)

    cands = [(compute_c5(h_init), h_init)]
    cands.append((0, project(thue_morse(n0), tgt)))
    for f in [2, 3, 4, 5, 7, 11, 13]:
        h_s = (np.sin(np.arange(n0) * 2.0 * np.pi * f / n0) > 0).astype(float)
        cands.append((0, project(h_s, tgt)))
    for f1, f2 in [(2,3),(3,5),(2,7),(3,11),(5,13)]:
        x = np.arange(n0)
        h_c = ((np.sin(x*2*np.pi*f1/n0) + 0.7*np.sin(x*2*np.pi*f2/n0 + 0.5)) > 0).astype(float)
        cands.append((0, project(h_c, tgt)))

    # Score candidates
    scored = []
    for c in cands:
        h_c = c[1]
        c5_c = compute_c5(h_c)
        scored.append((c5_c, h_c))
    scored.sort(key=lambda x: x[0])

    # Optimize top candidates with projected gradient
    for _, h_c in scored[:6]:
        if elapsed() > budget_s * 0.35:
            break
        t_s = time.time()
        for T in [50, 200, 800, 3000]:
            if time.time() - t_s > 25:
                break
            h_c = proj_grad(h_c, T, 0.5, 800, tgt)
        h_c = project(h_c, tgt)
        c5_c = compute_c5(h_c)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()
        # Binarize and re-refine
        h_b = (h_c > 0.5).astype(float)
        h_b = project(h_b, tgt)
        h_b = proj_grad(h_b, 1000, 0.3, 400, tgt)
        c5_b = compute_c5(h_b)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Exhaustive pair-flip local search on best
    if elapsed() < budget_s * 0.75:
        h_s = (best_h > 0.5).astype(float)
        h_s = project(h_s, tgt)
        max_t = min(budget_s * 0.25, 200.0)
        h_s, c5_s = pair_search(h_s, tgt, max_t)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Also try pair search on second-best candidate
    if elapsed() < budget_s * 0.85 and len(scored) > 1:
        h_s2 = (scored[1][1] > 0.5).astype(float)
        h_s2 = project(h_s2, tgt)
        h_s2 = proj_grad(h_s2, 1000, 0.3, 300, tgt)
        h_s2 = (h_s2 > 0.5).astype(float)
        h_s2 = project(h_s2, tgt)
        h_s2, c5_s2 = pair_search(h_s2, tgt, min(budget_s * 0.1, 80.0))
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_s2.copy()

    # Multi-n exploration
    if elapsed() < budget_s * 0.9:
        for n_alt in [170, 200, 280, 400, 500]:
            if elapsed() > budget_s * 0.93:
                break
            tgt_a = n_alt / 2.0
            x_old = np.arange(len(best_h)) * (2.0 / len(best_h))
            x_new = np.arange(n_alt) * (2.0 / n_alt)
            h_a = np.interp(x_new, x_old, best_h)
            h_a = project(h_a, tgt_a)
            h_a = proj_grad(h_a, 1000, 0.4, 500, tgt_a)
            h_a = (h_a > 0.5).astype(float)
            h_a = project(h_a, tgt_a)
            c5_a = compute_c5(h_a)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_a.copy()

    # Final gradient polish
    if elapsed() < budget_s - 5:
        tgt_f = len(best_h) / 2.0
        best_h = proj_grad(best_h, 3000, 0.15, 500, tgt_f)
        best_h = project(best_h, tgt_f)
        c5_f = compute_c5(best_h)
        if c5_f < best_c5:
            best_c5 = c5_f

    best_h = project(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
