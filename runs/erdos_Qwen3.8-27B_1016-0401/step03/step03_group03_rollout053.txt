        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]

    dx = 2.0 / n
    center = n - 1
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    
    for step in range(max_steps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break

        ak = int(np.argmax(corr))
        shift = ak - center
        g = shift_grad(h_cur, n, shift)
        
        # Find best pair: i with g[i] > 0 (decrease) and j with g[j] < 0 (increase)
        # The change in the active shift value:
        # delta_corr = g[i]*(-delta) + g[j]*(+delta) = delta*(g[j] - g[i]) < 0 if g[i] > g[j]
        # We also need to check OTHER shifts don't increase too much
        pos_idx = np.where(g > 0.05)[0]
        neg_idx = np.where(g < -0.05)[0]
        if len(pos_idx) == 0 or len(neg_idx) == 0:
            break
        
        # Sample a few candidate pairs
        best_improve = 0
        best_i, best_j, best_d = -1, -1, 0
        n_try = min(5, len(pos_idx), len(neg_idx))
        for _ in range(n_try):
            i = pos_idx[rng.integers(len(pos_idx))]
            j = neg_idx[rng.integers(len(neg_idx))]
            # Max feasible delta
            d_max = min(h_cur[i], 1.0 - h_cur[j])
            d = min(0.1, d_max)
            h_try = h_cur.copy()
            h_try[i] -= d
            h_try[j] += d
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            improve = c5_cur - c5_try
            if improve > best_improve:
                best_improve = improve
                best_i, best_j, best_d = i, j, d
        
        if best_improve > 1e-12:
            h_cur[best_i] -= best_d
            h_cur[best_j] += best_d
            h_cur = np.clip(h_cur, 0.0, 1.0)
            h_cur = project_feasible(h_cur, n)
            c5_cur = compute_c5(h_cur)
        elif step > 100:
            # Try larger random perturbations
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                d = rng.uniform(0.01, 0.05)
                h_try = h_cur.copy()
                if h_try[i] > d:
                    h_try[i] -= d
                    h_try[j] = min(1.0, h_try[j] + d)
                    h_try = project_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < c5_cur:
                        h_cur = h_try
                        c5_cur = c5_try
    return h_cur, c5_cur

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

_G_N = 280
_G_T = 1.0

def _full_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(sh))) / _G_T) * (2.0 / _G_N)

def _full_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        if sw[oi] < 1e-13:
            continue
        grad += sw[oi] * shift_grad(x, _G_N, oi - center)
    return grad * dx

def _full_cons_eq(x):
    return x.sum() - _G_N / 2.0

def _full_cons_eq_jac(x):
    return np.ones(_G_N)

def do_slsqp_full(h, n, T, maxiter, ftol):
    global _G_N, _G_T
    _G_N = n
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _full_cons_eq, 'jac': _full_cons_eq_jac}]
    try:
        res = minimize(_full_obj, h, method='SLSQP', jac=_full_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x, 0.0, 1.0)
    except:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

_G_N = 280
_G_T = 1.0

def _full_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(sh))) / _G_T) * (2.0 / _G_N)

def _full_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        if sw[oi] < 1e-13:
            continue
        grad += sw[oi] * shift_grad(x, _G_N, oi - center)
    return grad * dx

def _cons_eq(x):
    return x.sum() - _G_N / 2.0

def _cons_eq_jac(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter, ftol):
    global _G_N, _G_T
    _G_N = n
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    try:
        res = minimize(_full_obj, h, method='SLSQP', jac=_full_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def h_from_fourier(coeffs, n, K):
    i = np.arange(n)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        h += coeffs[2*(k-1)] * np.cos(2.0 * np.pi * k * i / n)
        h += coeffs[2*(k-1)+1] * np.sin(2.0 * np.pi * k * i / n)
    return h

def fourier_coeffs(h, n, K):
    i = np.arange(n)

    return coeffs

def pair_descent(h, n, rng, max_steps, deadline):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for _ in range(max_steps):
        if time.time() > deadline:
            break

        ak = int(np.argmax(corr))
        shift = ak - (n - 1)
        g = shift_grad(h_cur, n, shift)
        pos_idx = np.where(g > 0.03)[0]
        neg_idx = np.where(g < -0.03)[0]
        if len(pos_idx) < 1 or len(neg_idx) < 1:
            break
        best_imp = 0.0
        best_trip = None
        n_try = min(10, len(pos_idx), len(neg_idx))

            ii = pos_idx[rng.integers(len(pos_idx))]

            ht = h_cur.copy()
            ht[ii] -= d
