import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def _set_state(n):
    global _g_n, _g_target
    _g_n = n
    _g_target = n / 2.0

def _c_eq_fun(x):
    return x.sum() - _g_target

def _c_eq_jac(x):
    return np.ones(_g_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        v1 = j + shift < n
        grad[v1] += (1.0 - h[j[v1] + shift])
        v2 = j >= shift
        grad[v2] -= h[j[v2] - shift]
    else:
        s = -shift
        v1 = j >= s
        grad[v1] += (1.0 - h[j[v1] - s])
        v2 = j + s < n
        grad[v2] -= h[j[v2] + s]
    return grad

def _sls_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * dx

def _sls_jac(x, n, T):
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    grad = np.zeros(n)
    for oi in range(n_corr):
        w = sw[oi]
        if w < 1e-12:
            continue
        grad += w * shift_grad(x, n, oi - center)
    return grad * dx

def _topk_obj(x, n, K):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    top_idx = np.argpartition(corr, -K)[-K:]
    top_vals = corr[top_idx]
    # Power-8 weighted max of top-K
    w = (top_vals / max(mc, 1e-15)) ** 8
    ws = w.sum()
    if ws < 1e-15:
        return mc * dx
    return np.dot(w, top_vals) / ws * dx

def _topk_jac(x, n, K):
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    top_idx = np.argpartition(corr, -K)[-K:]
    top_vals = corr[top_idx]
    w = (top_vals / max(mc, 1e-15)) ** 8
    ws = w.sum()
    if ws < 1e-15:
        w = np.ones(K) / K
    else:
        w = w / ws
    grad = np.zeros(n)
    for i in range(K):
        grad += w[i] * shift_grad(x, n, top_idx[i] - center)
    return grad * dx

def slsqp_smooth(h0, n, T, maxiter, ftol):
    _set_state(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_eq_fun, 'jac': _c_eq_jac}]
    try:
        res = minimize(_sls_obj, h0, method='SLSQP', jac=_sls_jac,
                       args=(n, T), bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0.copy()

def slsqp_topk(h0, n, K, maxiter, ftol):
    _set_state(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_eq_fun, 'jac': _c_eq_jac}]
    try:
        res = minimize(_topk_obj, h0, method='SLSQP', jac=_topk_jac,
                       args=(n, K), bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0.copy()

def nesterov_bt(h_init, n, t_start, time_limit, max_iters=400000):
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    alpha = 0.3
    K = 3
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        f_y = mc * dx
        
        if K <= 1:
            idx = int(np.argmax(corr))
            g = shift_grad(h_y, n, idx - center)
        else:
            top_idx = np.argpartition(corr, -K)[-K:]
            top_vals = corr[top_idx]
            w = top_vals / max(mc, 1e-15)
            w = w / w.sum()
            g = np.zeros(n)
            for i in range(K):
                g += w[i] * shift_grad(h_y, n, top_idx[i] - center)
        
        # Backtracking line search
        f_new = f_y
        h_new = h_y.copy()
        for _ls in range(15):

            f_try = compute_c5(h_try)

_g_n = 0
_g_target = 0.0

def _set_state(n):
    global _g_n, _g_target
    _g_n = n
    _g_target = n / 2.0

def _c_eq_fun(x):
    return x.sum() - _g_target

def _c_eq_jac(x):
    return np.ones(_g_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        v1 = j + shift < n
        grad[v1] += (1.0 - h[j[v1] + shift])
        v2 = j >= shift
        grad[v2] -= h[j[v2] - shift]
    else:
        s = -shift
        v1 = j >= s
        grad[v1] += (1.0 - h[j[v1] - s])
        v2 = j + s < n
        grad[v2] -= h[j[v2] + s]
    return grad

def _sls_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * dx

def _sls_jac(x, n, T):
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    grad = np.zeros(n)
    for oi in range(n_corr):
        w = sw[oi]
        if w < 1e-12:
            continue
        grad += w * shift_grad(x, n, oi - center)
    return grad * dx

def _topk_obj(x, n, K):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    top_idx = np.argpartition(corr, -K)[-K:]
    top_vals = corr[top_idx]
    w = (top_vals / max(mc, 1e-15)) ** 8
    ws = w.sum()
    if ws < 1e-15:
        return mc * dx
    return np.dot(w, top_vals) / ws * dx

def _topk_jac(x, n, K):
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    top_idx = np.argpartition(corr, -K)[-K:]
    top_vals = corr[top_idx]
    w = (top_vals / max(mc, 1e-15)) ** 8
    ws = w.sum()
    if ws < 1e-15:
        w = np.ones(K) / K
    else:
        w = w / ws
    grad = np.zeros(n)
    for i in range(K):
        grad += w[i] * shift_grad(x, n, top_idx[i] - center)
    return grad * dx

def slsqp_smooth(h0, n, T, maxiter, ftol):
    _set_state(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_eq_fun, 'jac': _c_eq_jac}]

                       args=(n, T), bounds=bounds, constraints=cons,
