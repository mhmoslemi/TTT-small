from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(np.floor(y.min())) - 1.0
    hi = float(np.ceil(y.max())) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j < (n - shift)
        if np.any(m1):
            grad[m1] += (1.0 - h[j[m1] + shift])
        m2 = j >= shift
        if np.any(m2):
            grad[m2] -= h[j[m2] - shift]
    else:
        s = -shift
        m1 = j >= s
        if np.any(m1):
            grad[m1] += (1.0 - h[j[m1] - s])
        m2 = j < (n - s)
        if np.any(m2):
            grad[m2] -= h[j[m2] + s]
    return grad

def slsqp_smooth(h_init, n, T, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        s = e.sum()
        if s < 1e-300:
            return mc * dx
        return (mc + np.log(s) / T) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        e_sum = e.sum()
        if e_sum < 1e-300:
            return np.zeros(n)
        sw = e / e_sum
        grad = np.zeros(n)
        for oi in range(n_corr):
            w = sw[oi]
            if w < 1e-12:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return grad * dx

    def cons_fun(x):
        return x.sum() - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h_init, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init.copy()

def binary_local_search(h_bin, n, max_sweeps=50):
    """Local search on binary solutions: flip pairs to improve C5."""

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(np.floor(y.min())) - 1.0
    hi = float(np.ceil(y.max())) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j < (n - shift)
        if np.any(m1):
            grad[m1] += (1.0 - h[j[m1] + shift])
        m2 = j >= shift
        if np.any(m2):
            grad[m2] -= h[j[m2] - shift]
    else:
        s = -shift
        m1 = j >= s
        if np.any(m1):
            grad[m1] += (1.0 - h[j[m1] - s])
        m2 = j < (n - s)
        if np.any(m2):
            grad[m2] -= h[j[m2] + s]
    return grad

def slsqp_smooth(h_init, n, T, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        s = e.sum()
        if s < 1e-300:
            return mc * dx
        return (mc + np.log(s) / T) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        e_sum = e.sum()
        if e_sum < 1e-300:
            return np.zeros(n)
        sw = e / e_sum
        grad = np.zeros(n)
        for oi in range(n_corr):
            w = sw[oi]
            if w < 1e-12:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return grad * dx

    def cons_fun(x):
        return x.sum() - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h_init, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init.copy()

    t_start = time.time()
    # Round to binary
    h_bin = (h_cont > 0.5).astype(float)
    # Fix to exactly n/2 ones
    n_ones = int(h_bin.sum())

        ones_idx = np.where(h_bin == 1)[0]

            h_bin[ones_idx[sort_order[i]]] = 0
    elif n_ones < n // 2:

        sort_order = np.argsort(h_cont[zeros_idx])[::-1]  # most "uncertain" first
        for i in range(n // 2 - n_ones):
            h_bin[zeros_idx[sort_order[i]]] = 1

    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit_s:

        improved = False
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]

                h_bin[b] = 1

                    ones = np.where(h_bin == 1)[0]

                    h_bin[b] = 0

def nesterov_step(h_init, n, iterations, time_limit):
    t_start = time.time()
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    
    for it in range(iterations):
        if time.time() - t_start > time_limit:
            break
        frac = it / max(iterations, 1)
        K = max(1, int(1 + 20 * frac * frac))
        beta = float(it) / (it + 2.5) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        
        if K <= 1:
            idx = int(np.argmax(corr))
            g = shift_grad(h_y, n, idx - center) * dx
        else:
            top_idx = np.argpartition(corr, -K)[-K:]
            top_vals = corr[top_idx]
            w = (top_vals / max(mc, 1e-15))**6
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.ones(K) / K
            else:
                w /= w_sum
            g = np.zeros(n)
            for i in range(K):
                g += w[i] * shift_grad(h_y, n, top_idx[i] - center)
            g *= dx
        
        alpha = 0.5 / (1.0 + 0.001 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        
        h_prev = h_sg
        h_sg = h_new
        
        if (it + 1) % 5000 == 0:
            c5c = compute_c5(h_sg)
            if c5c < c5_best:
                c5_best = c5c
                h_best = h_sg.copy()
    
    c5c = compute_c5(h_sg)
    if c5c < c5_best:
        c5_best = c5c
        h_best = h_sg.copy()

def binary_sa(h_cont, n, time_limit_s=20, start_temp=0.5, end_temp=0.01):
    t_start = time.time()
    h = (h_cont > 0.5).astype(float)
    # Fix count
    n_ones = int(h.sum())
    if n_ones != n // 2:
        if n_ones > n // 2:
            ones = np.where(h == 1)[0]
            excess = n_ones - n // 2
            rng_sa = np.random.default_rng(hash(n) % 10000)
            to_remove = rng_sa.choice(ones, excess, replace=False)
            h[to_remove] = 0
        else:
            zeros = np.where(h == 0)[0]
            deficit = n // 2 - n_ones
            rng_sa = np.random.default_rng(hash(n) % 10000)
            to_add = rng_sa.choice(zeros, deficit, replace=False)
            h[to_add] = 1
    
    dx = 2.0 / n
    best_corr = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
    best_h = h.copy()
    curr_corr = best_corr
    rng_sa = np.random.default_rng(42)
    
    while time.time() - t_start < time_limit_s:
        elapsed = time.time() - t_start
        temp = start_temp * (end_temp / start_temp) ** (elapsed / time_limit_s)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        a = ones[rng_sa.integers(len(ones))]
        b = zeros[rng_sa.integers(len(zeros))]
        
        h[a] = 0
        h[b] = 1
        corr = np.correlate(h, 1.0 - h, mode="full")
        new_corr = float(np.max(corr))
        delta = new_corr - curr_corr
        
        if delta < 0 or rng_sa.random() < np.exp(-delta / (temp * dx)):
            curr_corr = new_corr
            if new_corr < best_corr:
                best_corr = new_corr
                best_h = h.copy()
        else:
            h[a] = 1
            h[b] = 0
    
    return best_h, best_corr * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(np.floor(y.min())) - 1.0
    hi = float(np.ceil(y.max())) + 1.0
    mid = (lo + hi) / 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j < (n - shift)
        if np.any(m1):
            grad[m1] += (1.0 - h[j[m1] + shift])
        m2 = j >= shift
        if np.any(m2):
            grad[m2] -= h[j[m2] - shift]
    else:
        s = -shift
        m1 = j >= s
        if np.any(m1):
            grad[m1] += (1.0 - h[j[m1] - s])
        m2 = j < (n - s)
        if np.any(m2):
            grad[m2] -= h[j[m2] + s]
    return grad

def slsqp_smooth(h_init, n, T, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        s = e.sum()
        if s < 1e-300:
            return mc * dx
        return (mc + np.log(s) / T) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        e_sum = e.sum()
        if e_sum < 1e-300:
            return np.zeros(n)
        sw = e / e_sum
        grad = np.zeros(n)
        for oi in range(n_corr):
            w = sw[oi]
            if w < 1e-12:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return grad * dx

    def cons_fun(x):
        return x.sum() - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h_init, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init.copy()

def binary_greedy(h_cont, n, max_sweeps=50, time_limit_s=25):
    t_s = time.time()
    h = (h_cont > 0.5).astype(float)
    n_ones = int(h.sum())
    if n_ones != n // 2:
        if n_ones > n // 2:
            idx = np.where(h == 1)[0]
            order = np.argsort(h_cont[idx])[::-1]
            for i in range(n_ones - n // 2):
                h[idx[order[i]]] = 0
        else:
            idx = np.where(h == 0)[0]
            order = np.argsort(h_cont[idx])[::-1]
            for i in range(n // 2 - n_ones):
                h[idx[order[i]]] = 1
    best_corr = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
    for _ in range(max_sweeps):
        if time.time() - t_s > time_limit_s:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for a in ones:
            for b in zeros:
                h[a] = 0
                h[b] = 1
                nm = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
                if nm < best_corr:
                    best_corr = nm
                    improved = True
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                else:
                    h[a] = 1
                    h[b] = 0
        if not improved:
            break
    return h, best_corr * (2.0 / n)

def binary_sa(h_cont, n, time_limit_s=20, seed=42):
    t_s = time.time()
    rng = np.random.default_rng(seed)
    h = (h_cont > 0.5).astype(float)
    n_ones = int(h.sum())
    if n_ones != n // 2:
        if n_ones > n // 2:
            idx = np.where(h == 1)[0]
            rem = rng.choice(idx, n_ones - n // 2, replace=False)
            h[rem] = 0
        else:
            idx = np.where(h == 0)[0]
            add = rng.choice(idx, n // 2 - n_ones, replace=False)
            h[add] = 1
    dx = 2.0 / n
    best_corr = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
    best_h = h.copy()
    curr_corr = best_corr
    t0 = time.time()
    while time.time() - t_s < time_limit_s:
        fr = min((time.time() - t_s) / time_limit_s, 1.0)
        temp = 0.8 * (1.0 - fr) * dx + 0.01 * dx
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        h[a] = 0
        h[b] = 1
        nc = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
        d = nc - curr_corr
        if d < 0 or rng.random() < np.exp(-d / temp):
            curr_corr = nc
            if nc < best_corr:
                best_corr = nc
                best_h = h.copy()
        else:
            h[a] = 1
            h[b] = 0
    return best_h, best_corr * dx

def nesterov_run(h_init, n, iterations, time_limit):
    t_s = time.time()
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_b = compute_c5(h_init)
    h_b = h_init.copy()
    for it in range(iterations):
        if time.time() - t_s > time_limit:
            break
        fr = it / max(iterations, 1)
        K = max(1, int(1 + 30 * fr * fr))
        beta = float(it) / (it + 2.5) if it > 0 else 0.0
        h_y = np.clip(h_sg + beta * (h_sg - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        if K <= 1:
            idx = int(np.argmax(corr))
            g = shift_grad(h_y, n, idx - center) * dx
        else:
            ti = np.argpartition(corr, -K)[-K:]
            tv = corr[ti]
            w = (tv / max(mc, 1e-15)) ** 6
            ws = w.sum()
            if ws < 1e-15:
                w = np.ones(K) / K
            else:
                w /= ws
            g = np.zeros(n)
            for i in range(K):
                g += w[i] * shift_grad(h_y, n, ti[i] - center)
            g *= dx
        alpha = 0.5 / (1.0 + 0.001 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 5000 == 0:
            c = compute_c5(h_sg)
            if c < c5_b:
                c5_b = c
                h_b = h_sg.copy()
    c = compute_c5(h_sg)
    if c < c5_b:
        c5_b = c
        h_b = h_sg.copy()
    return h_b, c5_b

def interp_h(h, n_old, n_new):
    xo = np.arange(n_old) * (2.0 / n_old)
    xn = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def make_start(n, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    h[::2] = 1.0
    mask = rng.random(n) < 0.08
    h[mask] = 1.0 - h[mask]
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:

        h_best = make_start(280, seed)
    n = len(h_best)

    # Phase 1: SLSQP coarse
    for T in [16, 8]:
        if time.time() - t0 > budget_s - 200:
            break
        hn = slsqp_smooth(h_best, n, T, 3000, 1e-14)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best = c5
            h_best = hn

            h_best = hb + 0.005 * rng.normal(0, 1, n)
            h_best = project_feasible(h_best, n)

    # Phase 3: SLSQP medium
    for T in [6, 3]:
        if time.time() - t0 > budget_s - 180:
            break
        hn = slsqp_smooth(h_best, n, T, 4000, 1e-14)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best = c5
            h_best = hn

    # Phase 4: Nesterov
    if time.time() - t0 < budget_s - 150:
        hn, cn = nesterov_run(h_best, n, 80000, 50)
        if cn < c5_best:
            c5_best = cn
            h_best = hn

    # Phase 5: Binary SA (escape local optima)
    if time.time() - t0 < budget_s - 120:
        for sa_seed in [42, 123, 999]:
            if time.time() - t0 > budget_s - 100:

            hb, cb = binary_sa(h_best, n, 15, sa_seed)

                c5_best = cb
                h_best = hb.copy()

    # Phase 6: SLSQP fine from best binary
    if time.time() - t0 < budget_s - 80:
        h_soft = h_best.copy()
        h_soft = np.where(h_soft > 0.99, 0.99, h_soft)
        h_soft = np.where(h_soft < 0.01, 0.01, h_soft)
        h_soft = project_feasible(h_soft, n)
        for T in [3, 1]:
            if time.time() - t0 > budget_s - 40:

            c5 = compute_c5(hn)
            if c5 < c5_best:
                c5_best = c5
                h_best = hn

    # Phase 7: Multi-resolution
    if time.time() - t0 < budget_s - 120:
        for n2 in [200, 240, 320, 400]:
            if time.time() - t0 > budget_s - 60:
                break
            ha = interp_h(h_best, n, n2)
            ha = project_feasible(ha, n2)
            hn = slsqp_smooth(ha, n2, 8, 2000, 1e-13)
            hn = project_feasible(hn, n2)
            c5 = compute_c5(hn)
            if c5 < c5_best:
                c5_best = c5
                h_best = hn
                n = n2
            if time.time() - t0 < budget_s - 40:

                if cb2 < c5_best:
                    c5_best = cb2
                    h_best = hb2
                    n = n2

    # Phase 8: Final binary + polish
    if time.time() - t0 < budget_s - 20:
        hb, cb = binary_greedy(h_best, n, 30, 8)
        if cb < c5_best:
            c5_best = cb
            h_best = hb
        h_best = project_feasible(h_best, n)
        c5_f = compute_c5(h_best)
        if c5_f < c5_best:
            c5_best = c5_f

    h_best = np.clip(h_best, 0.0, 1.0)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
