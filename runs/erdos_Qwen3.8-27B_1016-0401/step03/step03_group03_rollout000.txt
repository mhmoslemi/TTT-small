import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_TARGET = 0.0
_G_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-12:
            continue
        grad += w * shift_grad_raw(x, _G_N, oi - center)
    return grad * dx

def _g_cons_fun(x):
    return x.sum() - _G_TARGET

def _g_cons_jac(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter, ftol):
    global _G_N, _G_TARGET, _G_T
    _G_N = n
    _G_TARGET = n / 2.0
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_cons_fun, 'jac': _g_cons_jac}]
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def nesterov_opt(h_init, n, t0, time_budget, K=8, max_iters=400000):
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    best_time = time.time()
    L = 1.0
    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        beta = (1.0 + np.sqrt(1.0 + 4.0 * L * L)) / 2.0
        h_y = h_sg + (beta - 1.0) / (1.0 + beta) * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top_idx = np.argsort(corr)[-K:]
        mv = corr[top_idx].max()
        w = np.exp(6.0 * (corr[top_idx] - mv))
        w /= w.sum()
        g = np.zeros(n)
        for wk, ik in zip(w, top_idx):
            shift = ik - center
            g += wk * shift_grad_raw(h_y, n, shift)
        g *= dx
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        alpha = 1.0 / (L + 0.5)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        resid = np.linalg.norm(h_new - h_y) / (alpha + 1e-15)
        if resid > 0.99 * L:
            L = max(L, resid * 0.5)
        else:
            L = max(0.3 * L + 0.1, resid)
        h_prev = h_sg
        h_sg = h_new
        it_cost = 1
        if (it + 1) % 5000 == 0:
            cv = compute_c5(h_sg)
            if cv < c5_best - 1e-9:
                c5_best = cv
                h_best = h_sg.copy()
                best_time = time.time()
    cv = compute_c5(project_feasible(h_sg, n))
    pf = project_feasible(h_sg, n)
    if cv < c5_best:
        c5_best = cv
        h_best = pf
    return h_best, c5_best

def blocking_local(h, n, rng, steps=12000):
    dx = 2.0 / n
    center = n - 1
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)

        top3 = np.argsort(corr)[-3:]
        g = np.zeros(n)

            shift = ik - center
            g += shift_grad_raw(h_cur, n, shift)
        g *= dx
        gi = int(np.argmax(g))

        c5_try = compute_c5(h_try)
        if c5_try < c5_cur - 1e-10:
            h_cur = h_try
            c5_cur = c5_try
    return h_cur, c5_cur
