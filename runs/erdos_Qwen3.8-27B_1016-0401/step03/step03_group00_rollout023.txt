def grad_shift(h, n, s):
    """Gradient of correlate shift s for h vs 1-h."""
    g = np.zeros(n)
    if s >= 0:
        # Terms where h[i] contributes: i = 0..n-1-s
        # h[i] * (1-h[i+s]): dh[i]/dh[i] = 1-h[i+s], dh[i]/dh[i+s] = -h[i]
        idx1 = np.arange(0, n - s)
        g[idx1] += (1.0 - h[idx1 + s])
        idx2 = np.arange(s, n)
        g[idx2] -= h[idx2 - s]
    else:
        ss = -s
        idx1 = np.arange(0, n - ss)
        g[idx1] += (1.0 - h[idx1 + ss])
        idx2 = np.arange(ss, n)
        g[idx2] -= h[idx2 - ss]
    return g

    """L-BFGS-B optimization of soft-max of top-K shifts."""
    def obj_f(x):

        idx = np.argpartition(corr, len(corr) - K)[len(corr) - K:]
        c = corr[idx]
        m = c.max()
        return (m + T * np.log(np.exp((c - m) / T).sum())) * (2.0 / n)
    
    def obj_g(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argpartition(corr, len(corr) - K)[len(corr) - K:]
        c = corr[idx]
        m = c.max()

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def grad_shift(h, n, s):
    g = np.zeros(n)
    if s >= 0:
        idx1 = np.arange(0, n - s)
        g[idx1] += (1.0 - h[idx1 + s])
        idx2 = np.arange(s, n)
        g[idx2] -= h[idx2 - s]
    else:
        ss = -s
        idx1 = np.arange(0, n - ss)
        g[idx1] += (1.0 - h[idx1 + ss])
        idx2 = np.arange(ss, n)
        g[idx2] -= h[idx2 - ss]
    return g

def nesterov_opt(h_init, n, t_end, rng, top_k=15, lr0=0.01, momentum=1.8):

    h_cur = h_init.copy()

    c5a = compute_c5(h_init)
    ha = h_init.copy()
    
    for it in range(500000):
        if time.time() > t_end:
            break

def slsqp_fine(h, n, T, K, maxiter, t_end):
    """Fine SLSQP with soft-max."""
    N = n
    TGT = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argpartition(corr, len(corr) - K)[len(corr) - K:]
        c = corr[idx]
        m = c.max()
        return (m + T * np.log(np.exp((c - m) / T).sum())) * (2.0 / N)
    
    def obj_g(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argpartition(corr, len(corr) - K)[len(corr) - K:]
        c = corr[idx]
        m = c.max()
        e = np.exp((c - m) / T)
        w = e / e.sum()
        g = np.zeros(N)
        for wi, ii in zip(w, idx):
            if wi > 1e-10:
                g += wi * grad_shift(x, N, ii - (N - 1))
        g *= (2.0 / N)
        return g
    
    def con_f(x):
        return x.sum() - TGT
    
    def con_g(x):
        return np.ones(N)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_g, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_f, 'jac': con_g}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except:
        return h.copy()

                h_bb[order[i]] = 1.0
    return h_bb

def bangbang_sa(h_init, n, t_end, rng, T0=0.002, T1=0.00005, cool=0.9998):
    h_bb = round_bangbang(h_init, n).copy()
    c5_cur = compute_c5(h_bb)
    c5_best = c5_cur
    h_best = h_bb.copy()
    T = T0
    for it in range(500000):
        if time.time() > t_end:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if h_bb[i] == h_bb[j]:
            continue
        h_bb[i], h_bb[j] = h_bb[j], h_bb[i]
        c5_new = compute_c5(h_bb)
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_bb.copy()
        else:
            h_bb[i], h_bb[j] = h_bb[j], h_bb[i]
        T = max(T * cool, T1)
    return h_best, c5_best

def bangbang_hill(h_init, n, t_end, rng, iters=200000):
    h_bb = round_bangbang(h_init, n).copy()
    c5_cur = compute_c5(h_bb)
    for it in range(iters):
        if time.time() > t_end:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if h_bb[i] == h_bb[j]:
            continue
        h_bb[i], h_bb[j] = h_bb[j], h_bb[i]
        c5_new = compute_c5(h_bb)
        if c5_new < c5_cur - 1e-14:
            c5_cur = c5_new
        else:
            h_bb[i], h_bb[j] = h_bb[j], h_bb[i]
    return h_bb, c5_cur

def slsqp_fine(h, n, T, K, maxiter):
    N = n
    TGT = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argpartition(corr, len(corr) - K)[len(corr) - K:]
        c = corr[idx]
        m = c.max()
        return (m + T * np.log(np.exp((c - m) / T).sum())) * (2.0 / N)
    def obj_g(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argpartition(corr, len(corr) - K)[len(corr) - K:]
        c = corr[idx]
        m = c.max()
        e = np.exp((c - m) / T)
        w = e / e.sum()
        g = np.zeros(N)
        for wi, ii in zip(w, idx):
            if wi > 1e-10:
                g += wi * grad_shift(x, N, ii - (N - 1))
        g *= (2.0 / N)
        return g
    def con_f(x):
        return x.sum() - TGT
    def con_g(x):
        return np.ones(N)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_g, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_f, 'jac': con_g}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Nesterov continuous optimization
    t1 = t0 + budget_s * 0.25
    hs, c5s = nesterov_opt(h_best, n, t1, rng, top_k=15, lr0=0.012, momentum=1.5)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs

    # Phase 2: Bang-bang simulated annealing
    t2 = t0 + budget_s * 0.50
    h_bb, c5_bb = bangbang_sa(h_best, n, t2, rng, T0=0.003, T1=0.00005, cool=0.9997)
    if c5_bb < c5_best:
        c5_best, h_best = c5_bb, h_bb.copy()

    # Phase 3: Hill climbing on bang-bang (exploit)
    t3 = t0 + budget_s * 0.65
    h_bb, c5_bb = bangbang_hill(h_best, n, t3, np.random.default_rng(seed + 7), iters=300000)
    if c5_bb < c5_best:
        c5_best, h_best = c5_bb, h_bb.copy()

    # Phase 4: Un-round with Nesterov
    t4 = t0 + budget_s * 0.80
    hs, c5s = nesterov_opt(h_best, n, t4, rng, top_k=12, lr0=0.006, momentum=1.3)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs

    # Phase 5: Final bang-bang hill climb
    t5 = t0 + budget_s * 0.90
    h_bb, c5_bb = bangbang_hill(h_best, n, t5, np.random.default_rng(seed + 13), iters=200000)
    if c5_bb < c5_best:
        c5_best, h_best = c5_bb, h_bb.copy()

    # Phase 6: SLSQP polish
    t6 = t0 + budget_s - 10
    for T, K, mi in [(4, 10, 400), (2, 8, 600), (1, 6, 800)]:
        if time.time() > t6:
            break
        hn = slsqp_fine(h_best, n, T, K, mi)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
