    dx = 2.0 / n
    center = n - 1
    nc = 2 * n - 1
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sm = np.max(sh)
        e = np.exp(sh - sm)
        return (mc + (np.log(e.sum()) + sm) / T) * dx
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sm = np.max(sh)
        e = np.exp(sh - sm)
        sw = e / e.sum()

            if sw[oi] > 1e-12:
                g += sw[oi] * grad_shift(x, n, oi - center)
        return g * dx
