    # Δ(o) = (h[b+o] - h[a+o]) + (h[b-o] - h[a-o]) for valid o
    delta = np.zeros(2*n-1)
    center = n-1
    # For o from -(n-1) to n-1:
    # h[a+o] valid when 0 <= a+o <= n-1, i.e., -a <= o <= n-1-a
    # h[b+o] valid when 0 <= b+o <= n-1
    # h[a-o] valid when 0 <= a-o <= n-1, i.e., a-n+1 <= o <= a

    # h[a-o]

    # h[b-o]

        if not improved:
            break
    return h, c5_cur

def perturb_and_optimize(h, target, bounds, rng, T, maxiter):
    n = len(h)
    h_p = h + rng.normal(0, 0.03, n)
    h_p = fix_sum(h_p, target)
    h_p = _slsqp(h_p, T, bounds, maxiter, 1e-17)
    h_p = fix_sum(h_p, target)
    return h_p

def run(seed=42, budget_s=1100, **kwargs):
    global _TARGET
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(340)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    _TARGET = target
    h_best = fix_sum(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: SLSQP with increasing T on initial (35% budget)
    p1_end = t0 + budget_s * 0.35
    for T in [3000, 10000, 30000, 60000]:
        if elapsed() > p1_end - 20:
            break
        h_new = _slsqp(h_best, T, bounds, 4000, 1e-19)
        h_new = fix_sum(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-16:
            c5_best = c5_new
            h_best = h_new
        else:
            h_best = h_new  # still use as next starting point

    # Phase 2: Perturbation restarts (20% budget)
    p2_end = t0 + budget_s * 0.55
    for restart in range(8):
        if elapsed() > p2_end - 15:
            break
        h_p = perturb_and_optimize(h_best, target, bounds, rng, 20000, 2500)
        h_p = fix_sum(h_p, target)
        c5_p = compute_c5(h_p)
        if c5_p < c5_best - 1e-16:
            c5_best = c5_p
            h_best = h_p
        # Also try from parent with different perturbation scale
        h_p2 = h_best + rng.normal(0, 0.08, n)
        h_p2 = fix_sum(h_p2, target)
        h_p2 = _slsqp(h_p2, 15000, bounds, 2000, 1e-18)
        h_p2 = fix_sum(h_p2, target)
        c5_p2 = compute_c5(h_p2)
        if c5_p2 < c5_best - 1e-16:
            c5_best = c5_p2
            h_best = h_p2

    # Phase 3: Alternative n values (25% budget)
    p3_end = t0 + budget_s * 0.80
    for n_alt in [320, 336, 348, 360, 380]:
        if elapsed() > p3_end - 30:
            break
        target_alt = n_alt / 2.0
        x_old = np.arange(len(h_best)) * (2.0 / len(h_best))
        x_new = np.linspace(0.0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best, period=2.0)
        h_alt = fix_sum(h_alt, target_alt)
        bounds_alt = [(0.0, 1.0)] * n_alt
        c5_alt = compute_c5(h_alt)
        for T_alt in [10000, 40000]:
            if elapsed() > p3_end - 25:
                break
            h_new = _slsqp(h_alt, T_alt, bounds_alt, 2500, 1e-18)
            h_new = fix_sum(h_new, target_alt)
            c5_new = compute_c5(h_new)
            if c5_new < c5_alt:
                h_alt = h_new
                c5_alt = c5_new
        h_alt, c5_cd = coord_descent(h_alt, target_alt, 2)
        if c5_cd < c5_alt:
            c5_alt = c5_cd
            h_alt = h_alt
        if c5_alt < c5_best - 1e-16:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt
            target = n / 2.0
            bounds = [(0.0, 1.0)] * n

    # Phase 4: Coordinate descent + binary push (10% budget)
    p4_end = t0 + budget_s * 0.90
    target = len(h_best) / 2.0
    n = len(h_best)
    bounds = [(0.0, 1.0)] * n
    for _ in range(4):
        if elapsed() > p4_end:
            break
        h_best, c5_cd = coord_descent(h_best, target, 3)
        if c5_cd < c5_best:
            c5_best = c5_cd
        # Binary push: push values toward 0 or 1
        for j in range(n):
            if h_best[j] > 0.7:
                trial = h_best.copy()
                trial[j] = 1.0
                trial = fix_sum(trial, target)
                c5_t = compute_c5(trial)
                if c5_t < c5_best - 1e-15:
                    h_best = trial
                    c5_best = c5_t
            elif h_best[j] < 0.3:
                trial = h_best.copy()
                trial[j] = 0.0
                trial = fix_sum(trial, target)
                c5_t = compute_c5(trial)
                if c5_t < c5_best - 1e-15:
                    h_best = trial
                    c5_best = c5_t

    # Phase 5: Final high-precision SLSQP (remaining budget)
    p5_end = t0 + budget_s - 4.0
    for T_final in [80000, 150000]:
        if elapsed() > p5_end:
            break
        h_new = _slsqp(h_best, T_final, bounds, 5000, 1e-20)
        h_new = fix_sum(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new
    h_best, c5_cd_final = coord_descent(h_best, target, 2)
    if c5_cd_final < c5_best:
        c5_best = c5_cd_final

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
