def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):

            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def lse_obj_grad(h, n, T, k):
    """Log-sum-exp of top-k correlations, with gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = np.max(topk)
    shifted = T * (topk - mc)
    exps = np.exp(shifted)
    logsum = np.log(exps.sum()) / T
    obj = (mc + logsum) * dx
    
    sh = exps / exps.sum()
    center = n - 1
    grad = np.zeros(n)
    j = np.arange(n)
    for oi in range(mk):
        w = sh[oi] * dx
        if w < 1e-14:
            continue
        shift = idx[oi] - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += w * (1.0 - h[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= w * h[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += w * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * h[j[m2] - s2]
    return obj, grad

def slsqp(h, n, T, k, maxiter, ftol):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    res = minimize(lambda x: lse_obj_grad(x, n, T, k)[0], h,
                   method='SLSQP',
                   jac=lambda x: lse_obj_grad(x, n, T, k)[1],
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

    sh = T * (topk - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * dx

def lse_grad(h, n, T, k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = topk.max()
    sh = T * (topk - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    j = np.arange(n)
    for oi in range(mk):
        w = sw[oi] * dx
        if w < 1e-14:
            continue
        shift = idx[oi] - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += w * (1.0 - h[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= w * h[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += w * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * h[j[m2] - s2]
    return grad

class ObjWrap:
    def __init__(self, n, T, k):
        self.n, self.T, self.k = n, T, k
    def f(self, x):
        return lse_obj(x, self.n, self.T, self.k)
    def g(self, x):
        return lse_grad(x, self.n, self.T, self.k)

class ConWrap:
    def __init__(self, n):
        self.t = n / 2.0
        self.v = np.ones(n)
    def f(self, x):
        return x.sum() - self.t
    def g(self, x):
        return self.v

def do_slsqp(h, n, T, k, maxiter, ftol):
    try:
        ow = ObjWrap(n, T, k)
        cw = ConWrap(n)
        res = minimize(ow.f, h.copy(), method='SLSQP', jac=ow.g,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cw.f, 'jac': cw.g}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def subgrad_step(h, n, t_end, max_iter=50000, lr0=0.015):
    dx = 2.0 / n
    h_cur = h.copy()
    c5_best = compute_c5(h_cur)
    h_best = h_cur.copy()
    lr = lr0
    mom = np.zeros(n)
    beta = 0.9
    for it in range(max_iter):
        if time.time() > t_end:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        k3 = 3
        idx3 = np.argpartition(corr, -k3)[-k3:]
        cvals = corr[idx3]
        mc = cvals.max()
        diff = cvals - mc
        wts = np.exp(20.0 * diff)
        wts /= wts.sum()
        center = n - 1
        j = np.arange(n)
        grad = np.zeros(n)
        for oi in range(k3):
            w = wts[oi] * dx
            if w < 1e-14:
                continue
            shift = idx3[oi] - center
            if shift >= 0:
                m1 = j >= shift
                grad[m1] += w * (1.0 - h_cur[j[m1] - shift])
                m2 = (j + shift) < n
                grad[m2] -= w * h_cur[j[m2] + shift]
            else:
                s2 = -shift
                m1 = (j + s2) < n
                grad[m1] += w * (1.0 - h_cur[j[m1] + s2])
                m2 = j >= s2
                grad[m2] -= w * h_cur[j[m2] - s2]
        mom = beta * mom - lr * grad
        h_new = h_cur + mom
        h_new = project_feasible(h_new, n)
        h_cur = h_new
        lr *= 0.9998
        if (it + 1) % 5000 == 0:
            c5 = compute_c5(h_cur)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_cur.copy()
    return h_best, c5_best

def make_periodic(p, pattern, n):
    h = np.tile(pattern, n // p + 1)[:n]
    return project_feasible(h, n)

def perturb(h, n, rng, scale=0.03):
    h_p = h.copy()
    r = rng.random(n)
    h_p += scale * (r - 0.5)
    return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP with diminishing temperature
    schedule = [(32, 12, 6000), (16, 10, 6000), (8, 8, 6000),
                (4, 6, 7000), (2, 5, 7000), (1, 4, 8000),
                (0.5, 3, 8000), (0.25, 2, 9000), (0.1, 2, 10000)]
    for T, k, mi in schedule:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        h_new = do_slsqp(h_best, n, T, k, mi, 1e-19)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Subgradient / momentum method
    rem = budget_s - (time.time() - t0)
    if rem > 200:
        h_sg, c5_sg = subgrad_step(h_best, n, t0 + rem * 0.35)
        h_sg = project_feasible(h_sg, n)
        c5_sg = compute_c5(h_sg)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase 3: More SLSQP polish after subgradient
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        for T, k in [(2, 4), (1, 3), (0.5, 2)]:
            rem = budget_s - (time.time() - t0)
            if rem < 50:
                break
            h_new = do_slsqp(h_best, n, T, k, 6000, 1e-20)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 4: Periodic exploration
    rem = budget_s - (time.time() - t0)
    if rem > 150:
        for p in [6, 8, 10, 12, 14, 16, 20, 24]:
            rem = budget_s - (time.time() - t0)
            if rem < 40:
                break
            for trial in range(3):
                bits = rng.integers(0, 2, p).astype(float)
                if abs(bits.sum() - p / 2.0) > p / 4:
                    continue
                h_p = make_periodic(p, bits, n)
                h_new = do_slsqp(h_p, n, 4, 8, 1500, 1e-14)
                h_new = project_feasible(h_new, n)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new

    # Phase 5: Perturbation + SLSQP restarts
    rem = budget_s - (time.time() - t0)
    n_perts = max(0, int(rem / 45))
    for pi in range(n_perts):
        if time.time() - t0 > budget_s - 20:
            break
        scale = 0.02 * (1 + 0.1 * pi)
        h_p = perturb(h_best, n, rng, scale)
        h_new = do_slsqp(h_p, n, 2, 5, 2000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Try different n
    rem = budget_s - (time.time() - t0)
    if rem > 200:
        for n_try in [200, 320, 420]:
            rem = budget_s - (time.time() - t0)
            if rem < 80 or n_try == n:
                continue
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_hi = np.interp(x_new, x_old, h_best)
            h_hi = project_feasible(h_hi, n_try)
            c5_hi = compute_c5(h_hi)
            best_here = max(c5_hi, c5_best)
            best_vec = h_best if c5_best < c5_hi else h_hi.copy()
            best_n = n if c5_best < c5_hi else n_try
            for T, k in [(8, 10), (2, 5), (0.5, 3)]:
                rem = budget_s - (time.time() - t0)
                if rem < 30:
                    break
                h_new = do_slsqp(best_vec, best_n, T, k, 4000, 1e-18)
                h_new = project_feasible(h_new, best_n)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
                    n = best_n

    # Phase 7: Final ultra-precise polish
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        for T, k in [(0.5, 3), (0.25, 2), (0.1, 2)]:
            rem = budget_s - (time.time() - t0)
            if rem < 20:
                break
            h_new = do_slsqp(h_best, n, T, k, 8000, 1e-20)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
