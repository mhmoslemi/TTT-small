    j = np.arange(n)
    g = np.zeros(n)
    if sh >= 0:
        m1 = j >= sh

            g[m1] += 1.0 - h[j[m1] - sh]
        m2 = (j + sh) < n
        if m2.any():

        m2 = j >= s
