from scipy.optimize import minimize
import time

_target = 0.0

def _cons_fun(x):
    return x.sum() - _target

def _cons_jac(x):
    return np.ones(len(x))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def _smooth_obj(x, n, T, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = np.clip(sh, -50, 50)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / T) * dx

def _smooth_jac(x, n, T, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh = np.clip(sh, -50, 50)
    e = np.exp(sh)
    sw = e / e.sum()
    center = n - 1
    n_corr = 2 * n - 1
    grad = np.zeros(n)
    for oi in range(n_corr):
        w = sw[oi]
        if w < 1e-10:
            continue
        grad += w * shift_grad(x, n, oi - center)
    return grad * dx

def _topk_obj(x, n, K, p, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return np.max(corr) * dx

def _topk_jac(x, n, K, p, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    center = n - 1
    if K <= 1:
        idx = int(np.argmax(corr))
        return shift_grad(x, n, idx - center) * dx
    top_idx = np.argpartition(corr, -K)[-K:]
    top_vals = corr[top_idx]
    w = (top_vals / max(mc, 1e-15)) ** p
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.ones(K) / K
    else:
        w /= w_sum
    grad = np.zeros(n)
    for i in range(K):
        grad += w[i] * shift_grad(x, n, top_idx[i] - center)
    return grad * dx

def slsqp_smooth_run(h, n, T, maxiter, ftol):
    _target = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        def obj_wrapped(x):
            return _smooth_obj(x, n, T, dx)
        def jac_wrapped(x):
            return _smooth_jac(x, n, T, dx)
        res = minimize(obj_wrapped, h.copy(), method='SLSQP', jac=jac_wrapped,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def slsqp_topk_run(h, n, K, p, maxiter, ftol):
    _target = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        def obj_wrapped(x):
            return _topk_obj(x, n, K, p, dx)
        def jac_wrapped(x):
            return _topk_jac(x, n, K, p, dx)
        res = minimize(obj_wrapped, h.copy(), method='SLSQP', jac=jac_wrapped,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def projected_grad_search(h, n, t0, budget_s, T_init, T_min, max_iters):
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    h_cur = h.copy()
    c5_best = compute_c5(h_cur)
    h_best = h_cur.copy()
    T = T_init
    stall = 0
    alpha = 0.05
    for it in range(max_iters):
        if time.time() - t0 > budget_s:
            break
        if it % 200 == 0 and T > T_min:
            T = max(T * 0.97, T_min)
            alpha = max(0.005, 0.05 / (1.0 + 0.005 * it))
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh = np.clip(sh, -50, 50)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(n_corr):
            w = sw[oi]
            if w < 5e-9:
                continue
            grad += w * shift_grad(h_cur, n, oi - center)
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        alpha_bt = alpha
        improved = False
        for ls in range(25):
            h_try = h_cur - alpha_bt * grad
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_best - 1e-14:
                h_cur = h_try
                c5_best = c5_try
                h_best = h_cur.copy()
                improved = True
                break
            alpha_bt *= 0.5
            if alpha_bt < 1e-8:
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break
    return h_best, c5_best

def coord_polish(h, n, t0, budget_s):
    h_cur = h.copy()
    c5_best = compute_c5(h_cur)
    rng = np.random.default_rng(2024)
    for rnd in range(200):
        if time.time() - t0 > budget_s:
            break
        any_imp = False
        for _ in range(500):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            for direction in [1, -1]:
                if direction == 1:
                    mx = min(h_cur[i], 1.0 - h_cur[j])
                else:
                    mx = min(h_cur[j], 1.0 - h_cur[i])
                if mx < 1e-12:
                    continue
                for frac in [0.05, 0.15, 0.4, 0.7, 0.95]:
                    d = direction * mx * frac
                    if abs(d) < 1e-13:
                        continue
                    h_cur[i] += d
                    h_cur[j] -= d
                    if h_cur[i] < -1e-9 or h_cur[i] > 1 + 1e-9 or h_cur[j] < -1e-9 or h_cur[j] > 1 + 1e-9:
                        h_cur[i] -= d
                        h_cur[j] += d
                        continue
                    h_cur[i] = float(np.clip(h_cur[i], 0.0, 1.0))
                    h_cur[j] = float(np.clip(h_cur[j], 0.0, 1.0))
                    c5_try = compute_c5(h_cur)
                    if c5_try < c5_best - 1e-13:
                        c5_best = c5_try
                        any_imp = True
                    else:
                        h_cur[i] -= d
                        h_cur[j] += d
            if time.time() - t0 > budget_s:
                break
        if not any_imp:
            break
    h_cur = project_feasible(h_cur, n)
    return h_cur, compute_c5(h_cur)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP descending T
    for T in [16, 8, 4, 2]:
        if time.time() - t0 > budget_s - 250:
            break
        h_new = slsqp_smooth_run(h_best, n, T, 6000, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Projected gradient with line search
    if time.time() - t0 < budget_s - 200:
        h_pg, c5_pg = projected_grad_search(h_best, n, t0, budget_s - 200, 3.0, 0.2, 150000)
        if c5_pg < c5_best:
            c5_best = c5_pg
            h_best = h_pg

    # Phase 3: Fine SLSQP
    for T in [1.0, 0.5, 0.3]:
        if time.time() - t0 > budget_s - 150:
            break
        h_new = slsqp_smooth_run(h_best, n, T, 8000, 1e-20)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Top-K SLSQP for binding constraints
    for K, p in [(2, 12), (3, 10), (5, 8)]:
        if time.time() - t0 > budget_s - 100:
            break
        h_new = slsqp_topk_run(h_best, n, K, p, 6000, 1e-20)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Coordinate descent polish
    if time.time() - t0 < budget_s - 40:
        h_cd, c5_cd = coord_polish(h_best, n, t0, budget_s - 40)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    # Phase 6: Final SLSQP polish
    if time.time() - t0 < budget_s - 5:
        h_new = slsqp_smooth_run(h_best, n, 0.2, 10000, 1e-20)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
