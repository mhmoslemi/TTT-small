import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def nesterov_run(h_init, n, t0, time_limit, max_iters=500000):
    """Nesterov with adaptive K and symmetry."""
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        frac = min(float(it) / max(max_iters, 1), 1.0)
        K = max(1, int(1 + 40 * frac * frac))
        
        momentum = 0.5 if it < 10 else (float(it - 1) / (float(it) + 1.5))
        h_y = h_sg + momentum * (h_sg - h_prev)
        
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = float(np.max(corr))
        
        if K == 1:
            idx = int(np.argmax(corr))
            g = shift_grad(h_y, n, idx - center) * dx
        else:
            top_idx = np.argpartition(corr, -K)[-K:]
            top_vals = corr[top_idx]
            w = (top_vals / max(mc, 1e-15)) ** 10
            ws = w.sum()
            if ws < 1e-15:
                w = np.ones(K) / K
            else:
                w /= ws
            g = np.zeros(n)
            for i in range(K):
                if w[i] > 1e-8:
                    g += w[i] * shift_grad(h_y, n, top_idx[i] - center)
            g *= dx

        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        
        h_prev = h_sg
        h_sg = h_new
        h_avg += (h_new - h_avg) / (float(it + 1))

            c5c = compute_c5(h_sg)

def game_iter(h_init, n, t0, time_limit, n_outer=100):

    h = h_init.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()

            break

        g_norm = (g - g.mean()) / (g.std() + 1e-15)
        q = np.exp(temperature * g_norm)
        q = q / q.sum()

        for inner in range(30):
            if time.time() - t0 > time_limit:
                break
            # Compute weighted gradient using only significant q

                wk = q[k_idx]

                    grad += wk * shift_grad(h, n, k_idx - center)
            grad *= dx
            
            alpha = 0.8 / (1.0 + 0.05 * inner)

            h = project_feasible(h, n)

            c5_best = c5_sym
            h_best = h_sym
            h = h_best.copy()
    
    return h_best, c5_best

def slsqp_polish(h, n, T, maxiter, ftol):
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(n_corr):
            w = sw[oi]
            if w < 1e-10:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return grad * dx
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    stall = 0
    last_best = c5_best

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        frac = min(float(it) / max(max_iters, 1), 1.0)
        K = max(1, int(1 + k_max * frac ** 1.5))
        mu = 0.5 if it < 5 else (float(it - 1) / (float(it) + 1.5))
        h_y = h_sg + mu * (h_sg - h_prev)

        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = float(np.max(corr))
        if K == 1:
            idx = int(np.argmax(corr))
            g = shift_grad(h_y, n, idx - center) * dx
        else:
            top_idx = np.argpartition(corr, -K)[-K:]
            tv = corr[top_idx]
            w = (tv / max(mc, 1e-15)) ** 12
            ws = w.sum()
            if ws < 1e-15:
                w = np.ones(K) / K
            else:
                w /= ws
            g = np.zeros(n)
            for i in range(K):
                if w[i] > 1e-8:
                    g += w[i] * shift_grad(h_y, n, top_idx[i] - center)
            g *= dx

        alpha = 0.5 / (1.0 + 0.0008 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        h_avg += (h_new - h_avg) / (float(it + 1))

        if (it + 1) % 4000 == 0:
            h_p = project_feasible(h_avg, n)
            c5a = compute_c5(h_p)
            c5c = compute_c5(h_sg)
            if c5a < c5_best:
                c5_best = c5a
                h_best = h_p.copy()
                stall = 0
            elif c5c < c5_best:
                c5_best = c5c
                h_best = h_sg.copy()
                stall = 0
            else:
                stall += 1
            if stall >= 8:
                stall = 0
                h_sg = h_best.copy()
                h_prev = h_best.copy()
                h_avg = h_best.copy()
                noise = 0.02 * np.random.standard_normal(n)
                h_sg = project_feasible(h_sg + noise, n)

    return h_best, c5_best

def mw_game(h_init, n, t0, time_limit, n_outer=120):
    h = h_init.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    center = n - 1
    n_shifts = 2 * n - 1
    dx = 2.0 / n
    q = np.ones(n_shifts) / n_shifts
    corr = np.correlate(h, 1.0 - h, mode="full")
    g = corr * dx
    temp0 = 8.0

    for outer in range(n_outer):
        if time.time() - t0 > time_limit:
            break
        temp = temp0 / (1.0 + 0.03 * outer)
        g_norm = (g - g.mean()) / (g.std() + 1e-15)
        q = np.exp(temp * g_norm)
        qs = q.sum()
        if qs > 1e-10:
            q /= qs
        else:
            q = np.ones(n_shifts) / n_shifts

        n_inner = 40
        for inner in range(n_inner):
            if time.time() - t0 > time_limit:
                break
            active = np.where(q > 2e-6)[0]
            if len(active) == 0:
                break
            grad = np.zeros(n)
            for k_idx in active:
                wk = q[k_idx]
                if wk > 1e-7:
                    grad += wk * shift_grad(h, n, k_idx - center)
            grad *= dx
            alpha = 1.0 / (1.0 + 0.1 * inner)
            h = h - alpha * grad
            h = project_feasible(h, n)

        corr = np.correlate(h, 1.0 - h, mode="full")
        g = corr * dx
        c5_curr = float(np.max(g))
        if c5_curr < c5_best:
            c5_best = c5_curr
            h_best = h.copy()
        h_rev = h_best[::-1].copy()
        h_sym = (h_best + h_rev) / 2.0
        h_sym = project_feasible(h_sym, n)
        c5_sym = compute_c5(h_sym)
        if c5_sym < c5_best:
            c5_best = c5_sym
            h_best = h_sym
            h = h_best.copy()

    return h_best, c5_best

def slsqp_polish(h, n, T, maxiter, ftol):
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        sw = e / e.sum()
        grad = np.zeros(n)
        for oi in range(n_corr):
            w = sw[oi]
            if w < 1e-10:
                continue
            grad += w * shift_grad(x, n, oi - center)
        return grad * dx

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def topk_slsqp(h, n, K, maxiter, ftol):
    dx = 2.0 / n
    center = n - 1
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return float(np.max(corr)) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = float(np.max(corr))
        top_idx = np.argpartition(corr, -K)[-K:]
        tv = corr[top_idx]
        w = (tv / max(mc, 1e-15)) ** 8
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(K) / K
        else:
            w /= ws
        grad = np.zeros(n)
        for i in range(K):
            grad += w[i] * shift_grad(x, n, top_idx[i] - center)
        return grad * dx

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def try_resolution(h_src, n_src, n_new, t0, time_limit, seed):
    x_old = np.arange(n_src) * (2.0 / n_src)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_alt = np.interp(x_new, x_old, h_src)
    h_alt = project_feasible(h_alt, n_new)
    c5_alt = compute_c5(h_alt)
    if n_new <= 400:
        h_new = nesterov_fast(h_alt, n_new, t0, time_limit, 60000, 25)
        h_new = project_feasible(h_new, n_new)
        c5_alt = compute_c5(h_new)
    return h_new, c5_alt, n_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        n0 = 280
        h_best = np.zeros(n0)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Symmetry average
    h_sym = (h_best + h_best[::-1]) / 2.0
    h_sym = project_feasible(h_sym, n)
    c5s = compute_c5(h_sym)
    if c5s < c5_best:
        c5_best = c5s
        h_best = h_sym

    # Phase 1: Fast Nesterov (50% time)
    t1 = t0 + 0.50 * budget_s
    h1, c5_1 = nesterov_fast(h_best, n, t0, t1, 500000, 40)
    if c5_1 < c5_best:
        c5_best = c5_1
        h_best = h1
    h_best = project_feasible(h_best, n)
    h_sym = (h_best + h_best[::-1]) / 2.0
    h_sym = project_feasible(h_sym, n)
    if compute_c5(h_sym) < c5_best:
        c5_best = compute_c5(h_sym)
        h_best = h_sym

    # Phase 2: MW Game (20% time)
    t2 = t0 + 0.70 * budget_s
    h2, c5_2 = mw_game(h_best, n, t0, t2, 100)
    if c5_2 < c5_best:
        c5_best = c5_2
        h_best = h2
    h_best = project_feasible(h_best, n)

    # Phase 3: SLSQP polish (15% time)
    t3 = t0 + 0.85 * budget_s
    for T_val in [6, 3, 2]:
        if time.time() - t0 > t3:
            break
        h3 = slsqp_polish(h_best, n, T_val, 1500, 1e-15)
        h3 = project_feasible(h3, n)
        c5_3 = compute_c5(h3)
        if c5_3 < c5_best:
            c5_best = c5_3
            h_best = h3

    # Phase 4: TopK SLSQP (8% time)
    t4 = t0 + 0.93 * budget_s
    for K_val in [5, 10, 15]:
        if time.time() - t0 > t4:
            break
        h4 = topk_slsqp(h_best, n, K_val, 2500, 1e-15)
        h4 = project_feasible(h4, n)
        c5_4 = compute_c5(h4)
        if c5_4 < c5_best:
            c5_best = c5_4
            h_best = h4

    # Phase 5: Final Nesterov + symmetry (5% time)
    t5 = budget_s - 3.0
    h5, c5_5 = nesterov_fast(h_best, n, t0, t5, 40000, 20)
    if c5_5 < c5_best:
        c5_best = c5_5
        h_best = h5
    h_best = project_feasible(h_best, n)
    h_sym = (h_best + h_best[::-1]) / 2.0
    h_sym = project_feasible(h_sym, n)
    c5f = compute_c5(h_sym)
    if c5f < c5_best:
        c5_best = c5f
        h_best = h_sym

    # Phase 6: Try alternate resolution if time remains
    elapsed = time.time() - t0
    if elapsed < budget_s - 120:
        for n_try in [n // 2, n * 2]:
            if n_try < 60 or n_try > 1400:
                continue
            if time.time() - t0 > budget_s - 60:
                break
            h_alt, c5_alt, n_alt = try_resolution(h_best, n, n_try, t0, budget_s - 30, seed)
            if c5_alt < c5_best:
                c5_best = c5_alt
                h_best = h_alt
                n = n_alt

    h_best = project_feasible(h_best, n)
    c5_best = min(c5_best, compute_c5(h_best))
    return (h_best, c5_best, n)
