        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-14: break
            if d > 0:
                a = min(d, 1.0 - h[i])
                if a > 0: h[i] += a; d -= a
            else:
                s = min(-d, h[i])
                if s > 0: h[i] -= s; d += s
    return h

def _cf(x): return x.sum() - _TS
def _cj(x): return np.ones(len(x))

def sm_f(x, T):
    c = np.correlate(x, 1.0 - x, mode="full")
    mx = c.max()
    return (mx + np.log(np.sum(np.exp(T * (c - mx)))) / T) * (2.0 / len(x))

def sm_g(x, T):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode="full")
    s = T * (c - c.max())
    w = np.exp(s); w /= w.sum()
    dx = 2.0 / n; g = np.zeros(n); ctr = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14: continue
        o = k - ctr
        if o == 0: g += wk * (1 - 2 * x)
        elif o > 0:
            g[o:] += wk * (1 - x[:n - o]); g[:n - o] -= wk * x[o:]
        else:
            p = -o; g[:n - p] += wk * (1 - x[p:]); g[p:] -= wk * x[:n - p]
    return g * dx

def opt_slq(h, T, tgt, mi=2000):
    global _TS
    _TS = tgt
    n = len(h)
    try:
        r = minimize(sm_f, h, args=(T,), method='SLSQP', jac=sm_g,
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                     options={'maxiter': mi, 'ftol': 1e-18})
        return np.clip(r.x, 0, 1)
    except:
        return h

def coord(h, tgt, sw=3):
    n = len(h); c5 = compute_c5(h)
    dts = [0.005, -0.005, 0.01, -0.01, 0.002, -0.002, 0.02, -0.02]
    for _ in range(sw):
        imp = False
        for i in range(n):
            for dt in dts:
                t = h.copy(); t[i] = np.clip(t[i] + dt, 0, 1)
                t = fix_sum(t, tgt)
                c = compute_c5(t)
                if c < c5 - 1e-15: h, c5, imp = t, c, True; break
            if h[i] > 0.5:
                t = h.copy(); t[i] = 1.0; t = fix_sum(t, tgt)
                c = compute_c5(t)
                if c < c5 - 1e-15: h, c5, imp = t, c, True
            elif h[i] < 0.5:
                t = h.copy(); t[i] = 0.0; t = fix_sum(t, tgt)
                c = compute_c5(t)
                if c < c5 - 1e-15: h, c5, imp = t, c, True
        if not imp: break
    return h, c5

def subgrad(h, tgt, niter=3000, a0=0.005):
    n = len(h); h = h.copy().astype(float); bc = compute_c5(h)
    for t in range(niter):
        c = np.correlate(h, 1.0 - h, mode="full")
        k = int(np.argmax(c)); o = k - (n - 1)
        g = np.zeros(n)
        if o == 0: g[:] = 1.0 - 2.0 * h
        elif o > 0:
            g[o:] += 1.0 - h[:n - o]; g[:n - o] += -h[o:]
        else:
            p = -o; g[:n - p] += 1.0 - h[p:]; g[p:] += -h[:n - p]
        g *= 2.0 / n
        a = a0 / (1.0 + t * 0.0005)
        h2 = np.clip(h - a * g, 0, 1)
        h2 = fix_sum(h2, tgt)
        c2 = compute_c5(h2)
        if c2 < bc: h, bc = h2, c2
    return h, bc
