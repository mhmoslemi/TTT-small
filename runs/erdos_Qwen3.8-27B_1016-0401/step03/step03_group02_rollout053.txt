def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                if a <= 0:
                    continue
                h[i] += a
                d -= a
            else:
                s = min(-d, h[i])
                if s <= 0:
                    continue
                h[i] -= s
                d += s
            if abs(d) < 1e-14:
                break
    return h

    e = np.exp(T * (c - mc))
    return (mc + np.log(np.sum(e)) / T) * dx
