def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    """Gradient of correlation at given shift w.r.t. h"""
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]

        return g * dx

        return x.sum() - target

        return np.ones(n)

    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            g[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            g[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            g[m2] -= h[j[m2] - s]
    return g

    dx = 2.0 / _G_N
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = float(np.max(corr))
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    return (mc + float(np.log(e.sum())) / _G_T) * dx

def _g_jac(x):
    dx = 2.0 / _G_N
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = float(np.max(corr))
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / float(e.sum())
    center = _G_N - 1
    g = np.zeros(_G_N)
    for oi in range(len(sw)):
        if sw[oi] > 1e-12:
            g += sw[oi] * shift_grad(x, _G_N, oi - center)
    return g * dx

def _g_cfun(x):
    return x.sum() - _G_N / 2.0

def _g_cjac(x):
    return np.ones(_G_N)

    _G_N = n
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_cfun, 'jac': _g_cjac}]
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

    center = n - 1
    h_cur = h.copy()
    h_prev = h.copy()
    c5_best = compute_c5(h_cur)
    h_best = h_cur.copy()
    for it in range(iters):
        if time.time() - t0 > budget:
            break
        beta = min(0.95, 0.7 + 0.0001 * it) if it > 5 else 0.0
        h_y = h_cur + beta * (h_cur - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        # Get top K
        K_eff = min(K, len(corr))
        top_idx = np.argpartition(corr, -K_eff)[-K_eff:]
        mv = corr[top_idx].max()
        w = np.exp(4.0 * (corr[top_idx] - mv))
        w /= w.sum()
        g = np.zeros(n)
        for wk, ik in zip(w, top_idx):
            g += wk * shift_grad(h_y, n, int(ik) - center)
        g *= dx
        alpha = 0.06 / (1.0 + 0.001 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_cur
        h_cur = h_new
        if (it + 1) % 8000 == 0:
            cv = compute_c5(h_cur)
            if cv < c5_best:
                c5_best = cv
                h_best = h_cur.copy()
    cv = compute_c5(h_cur)
    if cv < c5_best:
        c5_best = cv
        h_best = h_cur.copy()
    return h_best, c5_best

def coord_polish(h, n, rng, steps):
