        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

_G_N = 0
_G_T = 1.0
_G_K = 10

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_G_T * (c - m))
    return (m + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_G_T * (c - m))
    w = e / e.sum()
    grad = np.zeros(_G_N)
    for wi, ci in zip(w, idx):
        if wi > 1e-10:
            grad += wi * shift_grad(x, _G_N, ci - (_G_N - 1))
    return grad * (2.0 / _G_N)

def _g_cf(x):
    return x.sum() - _G_N / 2.0

def _g_cj(x):
    return np.ones(_G_N)

def _slsqp(h, n, T, top_k, maxiter, ftol):
    global _G_N, _G_T, _G_K
    _G_N, _G_T, _G_K = n, float(T), min(top_k, 2 * n - 1)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _g_cf, 'jac': _g_cj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def _adam_phase(h_init, n, t_end, rng, top_k=12, T_start=48.0, lr=0.015):
    dx = 2.0 / n
    h = h_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    b1, b2, eps_adam = 0.82, 0.998, 1e-8
    T = T_start
    for it in range(300000):
        if time.time() > t_end:
            break
        hc = np.clip(h, 0.0, 1.0)
        corr = np.correlate(hc, 1.0 - hc, mode="full")
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        mval = c.max()
        T = max(T * 0.9992, 0.8)
        e = np.exp(T * (c - mval))
        w = e / e.sum()
        g = np.zeros(n)
        for wi, ci in zip(w, idx):
            if wi > 1e-7:
                g += wi * shift_grad(hc, n, ci - (n - 1))
        g *= dx / top_k
        t = it + 1
        m[:] = b1 * m + (1.0 - b1) * g
        v[:] = b2 * v + (1.0 - b2) * g**2
        m_hat = m / (1.0 - b1**t)
        v_hat = v / (1.0 - b2**t)
        step = lr / (np.sqrt(v_hat) + eps_adam)
        alpha = 1.0
        for _bt in range(6):
            h_try = project_feasible(hc - alpha * step, n)
            ct = np.correlate(h_try, 1.0 - h_try, mode="full")
            if np.exp(T * (ct[idx] - mval)).sum() <= e.sum() * 1.002:
                break
            alpha *= 0.5
        h = h_try
        if (it + 1) % 4000 == 0:
            c5 = compute_c5(h)
            if c5 < c5_best:
                c5_best, h_best = c5, h.copy()
    return h_best, c5_best

def _nesterov(h_init, n, t_end, rng, top_k=10):
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5a = compute_c5(h_init)
    ha = h_init.copy()
    for it in range(500000):
        if time.time() > t_end:
            break
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h_y, n, ci - (n - 1))
        g *= dx / top_k
        alpha = 0.4 / (1.0 + 0.003 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 6000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < c5a:
                c5a, ha = c1, h_cur.copy()
            c2 = compute_c5(project_feasible(h_avg, n))
            if c2 < c5a:
                c5a, ha = c2, project_feasible(h_avg, n).copy()
    return ha, c5a

def _perturbed(h_best, n, rng, pi):
    mode = pi % 5
    if mode == 0:
        sig = 0.003 + 0.002 * (pi // 5) % 4
        return project_feasible(h_best + rng.normal(0, sig, n), n)
    elif mode == 1:
        h_p = h_best.copy()
        lb = rng.integers(0, max(1, n - 30))
        ln = rng.integers(8, 30)
        h_p[lb:lb+ln] = 1.0 - h_p[lb:ln]
        return project_feasible(h_p, n)
    elif mode == 2:
        h_p = h_best.copy()
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        h_p[[i, j]] = h_p[[j, i]]
        return project_feasible(h_p, n)
    elif mode == 3:
        h_p = h_best.copy()
        lb = rng.integers(0, max(1, n - 40))
        ln = rng.integers(10, 40)
        h_p[lb:lb+ln] = rng.uniform(0, 1, ln)
        return project_feasible(h_p, n)
    else:
        h_p = h_best + rng.normal(0, 0.008, n)
        h_p[rng.integers(0, n, 3)] = 1.0
        return project_feasible(h_p, n)

def _local_refine(h, n, t_end):
    dx = 2.0 / n
    c5 = compute_c5(h)
    improved = True
    while improved and time.time() < t_end:
        improved = False
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr)) - (n - 1)
        sg = shift_grad(h, n, k_star)
        for trial in range(5):
            h_try = h.copy()
            idx_move = np.argsort(-np.abs(sg))[:8]
            for idx in idx_move:
                if sg[idx] > 0:
                    h_try[idx] -= 0.002
                else:
                    h_try[idx] += 0.002
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5 - 1e-10:
                h = h_try
                c5 = c5_try
                improved = True
                break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n0 = 280
        h_best = np.zeros(n0)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse-to-fine
    t1 = t0 + budget_s * 0.22
    for T, tk, mi in [(32, 25, 1500), (12, 15, 2000), (4, 10, 3000)]:
        if time.time() > t1:
            break
        hn = _slsqp(h_best, n, T, tk, mi, 1e-14)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 2: ADAM (more iterations, faster per-step)
    t2 = t0 + budget_s * 0.45
    hs, c5s = _adam_phase(h_best, n, t2, rng, top_k=12, T_start=48.0, lr=0.015)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs

    # Phase 3: Nesterov
    t3 = t0 + budget_s * 0.60
    hs, c5s = _nesterov(h_best, n, t3, rng, top_k=10)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs

    # Phase 4: SLSQP fine
    t4 = t0 + budget_s * 0.72
    for T, tk, mi in [(4, 10, 3000), (2, 8, 4000), (1, 6, 5000)]:
        if time.time() > t4:
            break
        hn = _slsqp(h_best, n, T, tk, mi, 1e-15)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 5: Perturbation + SLSQP
    for pi in range(50):
        if time.time() > budget_s - 40:
            break
        hp = _perturbed(h_best, n, rng, pi)
        hn = _slsqp(hp, n, 2, 8, 2000, 1e-14)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 6: Local refinement on critical shifts
    if time.time() < budget_s - 15:
        h_best, c5_loc = _local_refine(h_best, n, budget_s - 10)
        if c5_loc < c5_best:
            c5_best = c5_loc

    # Phase 7: Final ultra-fine SLSQP
    if time.time() < budget_s - 5:
        hn = _slsqp(h_best, n, 0.5, 5, 4000, 1e-16)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
