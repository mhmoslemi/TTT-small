def lpp_val_grad(h, p):
    """Compute Lp norm of overlaps and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Use only positive part to avoid issues with negative values
    # Actually, overlaps should be non-negative for h in [0,1]
    corr_pos = np.maximum(corr, 0)  # safety
    gp = corr_pos ** p
    sp = gp.sum()
    val = (sp ** (1.0/p)) * dx
    # Gradient: val = (sp^(1/p)) * dx
    # dval/dh[j] = (1/p) * sp^(1/p - 1) * (sum_k p * corr[k]^(p-1) * d(corr[k])/dh[j]) * dx
    #           = sp^(1/p-1) * sum_k corr[k]^(p-1) * d(corr[k])/dh[j] * dx
    if p == 1:

        idx = np.argmax(corr)
        g = shift_grad(h, n, idx - (n-1)) * dx
        return val, g
    # Weight: w_k = corr[k]^(p-1)

    w_norm = sp ** (1.0/p - 1.0)

    center = n - 1

        if w[k] < 1e-15:

        grad += w[k] * shift_grad(h, n, k - center)
    grad *= w_norm * dx
    return val, grad

    return grad

def smooth_max_val_grad(h, n, T, K):
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    if K < len(corr):
        topk = np.argpartition(corr, -K)[-K:]
        cs = corr[topk]
        mc = np.max(cs)
        sh = T * (cs - mc)
        e = np.exp(sh)
        val = (mc + np.log(e.sum()) / T) * dx
        sw = e / e.sum()
        center = n - 1
        grad = np.zeros(n)
        for ii in range(len(sw)):
            w = sw[ii]
            if w > 1e-15:
                grad += w * shift_grad(h, n, topk[ii] - center)
        return val, grad * dx
    else:
        mc = np.max(corr)
        sh = T * (corr - mc)
        e = np.exp(sh)
        val = (mc + np.log(e.sum()) / T) * dx
        sw = e / e.sum()
        center = n - 1
        grad = np.zeros(n)
        for oi in range(len(sw)):
            w = sw[oi]
            if w > 1e-15:
                grad += w * shift_grad(h, n, oi - center)
        return val, grad * dx

def lpp_val_grad(h, n, p):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cp = np.maximum(corr, 0.0)
    gp = cp ** p
    sp = gp.sum()
    val = (sp ** (1.0 / p)) * dx
    if p <= 1.0:
        idx = int(np.argmax(corr))
        g = shift_grad(h, n, idx - (n - 1)) * dx
        return val, g
    w = cp ** (p - 1)
    w_norm = sp ** (1.0 / p - 1.0)
    grad = np.zeros(n)
    j = np.arange(n)
    center = n - 1
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-18:
            continue
        s = k - center
        if s >= 0:
            m1 = j >= s
            m2 = (j + s) < n
            if m1.any():
                grad[m1] += wk * (1.0 - h[j[m1] - s])
            if m2.any():
                grad[m2] -= wk * h[j[m2] + s]
        else:
            ss = -s
            m1 = (j + ss) < n
            m2 = j >= ss
            if m1.any():
                grad[m1] += wk * (1.0 - h[j[m1] + ss])
            if m2.any():
                grad[m2] -= wk * h[j[m2] - ss]
    return val, grad * w_norm * dx

_nlb_flag = [False]

def _lbf_obj(x, n, T, K, t0, tlim):
    if time.time() - t0 > tlim:
        _nlb_flag[0] = True
    v, _ = smooth_max_val_grad(x, n, T, K)
    return v

def _lbf_grad(x, n, T, K, t0, tlim):
    if time.time() - t0 > tlim:
        _nlb_flag[0] = True
    _, g = smooth_max_val_grad(x, n, T, K)
    return g

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):
    _nlb_flag[0] = False
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lbf_obj, h, method='L-BFGS-B', jac=_lbf_grad,
                       args=(n, T, K, t0, tlim),
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol * 0.1,
                                'maxls': 30, 'maxfun': maxiter * 4})
        if _nlb_flag[0]:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def do_slsqp(h, n, T, K, maxiter, tol, t0, tlim):
    _nlb_flag[0] = False
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: _lbf_obj(x, n, T, K, t0, tlim), h,
                       method='SLSQP', jac=lambda x: _lbf_grad(x, n, T, K, t0, tlim),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if _nlb_flag[0]:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coord_descent(h, n, max_passes, t0, tlim):
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    deltas = [0.002, 0.005, 0.01, 0.02, 0.05, -0.002, -0.005, -0.01, -0.02, -0.05]
    for p in range(max_passes):
        if time.time() - t0 > tlim:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > tlim:
                break
            best_d = 0.0
            for d in deltas:
                hv = h_cur[j] + d
                if hv < 0 or hv > 1:
                    continue
                hp = h_cur.copy()
                hp[j] = hv
                c5_try = compute_c5(hp)
                if c5_try < c5_cur - 1e-15:
                    c5_cur = c5_try
                    best_d = d
                    improved = True
            if best_d != 0:
                h_cur[j] += best_d
        h_cur = project_feasible(h_cur, n)
        c5_cur = compute_c5(h_cur)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        if not improved:
            break
    return best_h, best_c5

def nesterov_subgrad(h_init, n, t0, tlim, max_iters=300000):
    dx = 2.0 / n
    h_curr = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    f_star = 0.37
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        beta = it / (it + 2.0)
        y = h_curr + beta * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = shift_grad(y, n, shift) * dx
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        f_y = compute_c5(y)
        gap = max(f_y - f_star, 1e-7)
        alpha = min(2.0 * gap / (gn * gn + 1e-12), 0.2 / (1.0 + 0.001 * it))
        h_new = project_feasible(y - alpha * g, n)
        h_prev = h_curr.copy()
        h_curr = h_new
        if (it + 1) % 2000 == 0:
            c5_now = compute_c5(h_curr)
            if c5_now < c5_best:
                c5_best = c5_now
                h_best = h_curr.copy()
    return h_best, c5_best

def perturb_h(h, n, rng, mode):
    x = np.arange(n, dtype=float)
    if mode == 0:
        pert = np.zeros(n)
        for _ in range(4):
            f = rng.uniform(1, 10)
            p = rng.uniform(0, 2 * np.pi)
            a = rng.uniform(0.005, 0.03)
            pert += a * np.sin(2 * np.pi * f * x / n + p)
        pert += rng.normal(0, 0.005, n)
    elif mode == 1:
        sig = rng.uniform(0.003, 0.015)
        pert = rng.normal(0, sig, n)
    elif mode == 2:
        bs = max(1, n // 10)
        bi = int(rng.integers(0, 10))
        pert = np.zeros(n)
        pert[bi * bs:(bi + 1) * bs] = rng.uniform(-0.5, 0.5, bs)
        pert += rng.normal(0, 0.003, n)
    elif mode == 3:
        pert = np.zeros(n)
        for _ in range(int(rng.integers(3, 12))):
            idx = int(rng.integers(0, n))
            pert[idx] = rng.uniform(-0.3, 0.3)
    else:
        pert = rng.normal(0, 0.01, n)
    return h + pert

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0

    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B coarse-to-fine (soft-max)
    schedules = [(64, 4000), (32, 4000), (16, 5000), (8, 6000), (4, 7000), (2, 8000), (1.0, 10000), (0.5, 12000)]
    for T, mi in schedules:
        rem = budget_s - (time.time() - t0)
        if rem < 100:
            break
        K = min(150, 2 * n - 1)
        h_new = do_lbfgsb(h_best, n, T, K, mi, 1e-14, t0, budget_s - 50)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Lp-norm optimization (differentiable, different from soft-max)
    for p in [4, 8, 16, 32]:
        rem = budget_s - (time.time() - t0)
        if rem < 120:
            break
        _nlb_flag[0] = False
        def _lpp_obj(x):
            if time.time() - t0 > tlim_lpp:
                _nlb_flag[0] = True
            v, _ = lpp_val_grad(x, n, p)
            return v
        def _lpp_grad(x):
            if time.time() - t0 > tlim_lpp:
                _nlb_flag[0] = True
            _, g = lpp_val_grad(x, n, p)
            return g
        tlim_lpp = budget_s - 40
        _nlb_flag[0] = False
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(_lpp_obj, h_best, method='L-BFGS-B', jac=_lpp_grad,
                           bounds=bounds,
                           options={'maxiter': 4000, 'ftol': 1e-14, 'gtol': 1e-12,
                                    'maxls': 30, 'maxfun': 12000})
            if not _nlb_flag[0]:
                h_new = project_feasible(res.x, n)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
        except Exception:
            pass

    # Phase 3: Nesterov subgradient for global refinement
    rem = budget_s - (time.time() - t0)
    if rem > 250:
        h_sg, c5_sg = nesterov_subgrad(h_best, n, t0, t0 + rem - 200, 200000)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase 4: SLSQP with equality constraint
    for T in [3, 1, 0.5]:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        h_new = do_slsqp(h_best, n, T, 80, 3000, 1e-15, t0, budget_s - 30)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Diverse restarts
    n_rest = 80
    for pi in range(n_rest):
        rem = budget_s - (time.time() - t0)
        if rem < 20:
            break
        mode = pi % 5
        h_p = perturb_h(h_best, n, rng, mode)
        h_p = project_feasible(h_p, n)
        T_r = [8, 4, 2, 1, 0.5][pi % 5]
        h_new = do_lbfgsb(h_p, n, T_r, 100, 2000, 1e-13, t0, budget_s - 10)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Coordinate descent polish (incremental)
    rem = budget_s - (time.time() - t0)
    n_passes = max(1, int(rem / 30))
    if n_passes > 0:
        h_cd, c5_cd = coord_descent(h_best, n, n_passes, t0, budget_s - 15)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    # Phase 7: Final L-BFGS-B + SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h_new = do_lbfgsb(h_best, n, 1.0, 50, 8000, 1e-15, t0, budget_s - 20)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    rem = budget_s - (time.time() - t0)
    if rem > 50:
        h_new = do_slsqp(h_best, n, 1, 50, 4000, 1e-15, t0, t0 + rem - 10)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 8: Final coordinate descent
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        h_cd2, c5_cd2 = coord_descent(h_best, n, max(1, int(rem / 25)), t0, budget_s - 5)
        if c5_cd2 < c5_best:
            c5_best = c5_cd2
            h_best = h_cd2

    # Phase 9: Try lower resolution if time remains
    rem = budget_s - (time.time() - t0)
    if rem > 300:
        for n_alt in [200, 240, 320]:
            rem = budget_s - (time.time() - t0)
            if rem < 200:
                break
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.linspace(0, 2.0, n_alt, endpoint=False)
            h_alt = np.interp(x_new, x_old, h_best)
            h_alt = project_feasible(h_alt, n_alt)
            c5_alt = compute_c5(h_alt)
            for T, mi in [(8, 3000), (2, 4000), (1, 5000)]:
                rem = budget_s - (time.time() - t0)
                if rem < 60:
                    break
                h_new = do_lbfgsb(h_alt, n_alt, T, 100, mi, 1e-13, t0, budget_s - 30)
                c5 = compute_c5(h_new)
                if c5 < c5_alt:
                    c5_alt = c5
                    h_alt = h_new
            if c5_alt < c5_best:
                c5_best = c5_alt
                h_best = h_alt
                n = n_alt

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
