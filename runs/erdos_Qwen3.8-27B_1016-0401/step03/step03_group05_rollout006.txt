    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def diverse_starts(n, rng):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    h[::2] = 1.0
    starts.append(h)
    for freq in [1, 2, 3]:
        h = 0.5 + 0.45 * np.sin(np.pi * freq * x)
        starts.append(h)
    h = 0.5 + 0.25 * np.sin(2 * np.pi * x) + 0.15 * np.sin(5 * np.pi * x)
    starts.append(h)
    h = np.zeros(n)
    h[:n // 3] = 1.4
    h[2 * n // 3:] = 0.6
    starts.append(h)
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    remaining = lambda: budget_s - (time.time() - t0)

    # Phase 1: SLSQP smooth max polish (coarse to fine T)
    for T in [16, 8, 4, 2]:
        if remaining() < 120:
            break
        tb = min(50.0, remaining() * 0.12)
        h_new = slsqp_smooth(h_best, n, T, 4000, 1e-18, tb)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Nesterov subgradient refinement
    if remaining() > 250:
        tb = min(200.0, remaining() * 0.25)
        h_sg, c5_sg = nesterov_refine(h_best, n, tb)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg
        h_best = project_feasible(h_best, n)

    # Phase 3: SLSQP top-K focused polish
    for K in [3, 5, 8]:
        if remaining() < 80:
            break
        tb = min(35.0, remaining() * 0.1)
        h_new = slsqp_topk(h_best, n, K, 5000, 1e-18, tb)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Very fine smooth-max SLSQP
    for T in [1.0, 0.5]:
        if remaining() < 70:
            break
        tb = min(30.0, remaining() * 0.1)
        h_new = slsqp_smooth(h_best, n, T, 6000, 1e-19, tb)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Diverse starts with quick top-K optimization
    if remaining() > 120:
        for s in diverse_starts(n, rng)[:5]:
            if remaining() < 50:
                break
            s = project_feasible(s, n)
            tb = min(25.0, remaining() * 0.08)
            h_new = slsqp_topk(s, n, 5, 2000, 1e-16, tb)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 6: Resolution exploration
    if remaining() > 100:
        for n2 in [n + 40, n - 40, n * 4 // 3, n * 6 // 5]:
            if remaining() < 50:
                break
            if n2 < 64 or n2 > 500:
                continue
            if n2 % 2 != 0:
                n2 += 1
            h_alt = interpolate_h(h_best, n, n2)
            h_alt = project_feasible(h_alt, n2)
            tb = min(25.0, remaining() * 0.08)
            h_new = slsqp_topk(h_alt, n2, 5, 3000, 1e-16, tb)
            h_new = project_feasible(h_new, n2)
            c5_alt = compute_c5(h_new)
            if c5_alt < c5_best:
                c5_best = c5_alt
                h_best = h_new
                n = n2

    # Final short Nesterov polish
    if remaining() > 60:
        tb = min(40.0, remaining() * 0.5)
        h_sg, c5_sg = nesterov_refine(h_best, n, tb, max_iters=100000)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, n)
