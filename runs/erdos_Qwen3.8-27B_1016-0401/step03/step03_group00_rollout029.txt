    C = np.correlate(h, 1.0-h, mode="full") * dx
    c5_cur = C.max()
    
    for rnd in range(10000):
        if time.time() > t_end:
            break

        if L >= n//4:
            continue
        # Random block swap
        j = rng.integers(0, n - L)
        m = rng.integers(0, n - L)
        if abs(j-m) < L:
            continue
        # Check it's a valid swap (complementary blocks)
        block_j = h[j:j+L]
        block_m = h[m:m+L]
        if block_j.sum() != L - block_m.sum():
            continue  # Sum constraint would be violated

            c5_cur = c5_new
        else:
            # Undo
            h[j:j+L] = block_j
            h[m:m+L] = block_m
    
    return h, c5_cur

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

_GN = 0
_GT = 0.0
_TT = 1.0
_TK = 20

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[- _TK:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_TT * (c - m))
    return (m + np.log(e.sum()) / _TT) * (2.0 / _GN)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[- _TK:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_TT * (c - m))
    w = e / e.sum()
    grad = np.zeros(_GN)
    for wi, ci in zip(w, idx):
        if wi > 1e-10:
            grad += wi * shift_grad(x, _GN, ci - (_GN - 1))
    return grad * (2.0 / _GN)

def _g_cf(x):
    return x.sum() - _GT

def _g_cj(x):
    return np.ones(_GN)

def slsqp_refine(h, n, T, top_k, maxiter, ftol):
    global _GN, _GT, _TT, _TK
    _GN = n
    _GT = n / 2.0
    _TT = float(T)
    _TK = min(top_k, 2 * n - 1)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _g_cf, 'jac': _g_cj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def binary_hill_climb(h, n, top_k, t_end, rng, max_rounds=500):
    dx = 2.0 / n
    h = (h >= 0.5).astype(float)
    while h.sum() > n / 2:
        idx = np.where(h == 1)[0]
        h[rng.integers(len(idx))] = 0
    while h.sum() < n / 2:
        idx = np.where(h == 0)[0]
        h[rng.integers(len(idx))] = 1
    C = np.correlate(h, 1.0 - h, mode="full") * dx
    c5_cur = C.max()
    tau = 0.006
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        if rnd % 40 == 39:
            tau *= 0.93
        top_idx = np.argsort(C)[-top_k:]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        new_max = np.full((len(zeros), len(ones)), -np.inf)
        for si in top_idx:
            s = si - (n - 1)
            js = zeros + s
            jsm = zeros - s
            f = np.zeros(len(zeros))
            v1 = (js >= 0) & (js < n)
            v2 = (jsm >= 0) & (jsm < n)
            f[v1] += 1.0 - h[js[v1]]
            f[v2] -= h[jsm[v2]]
            ms = ones + s
            msm = ones - s
            g = np.zeros(len(ones))
            v3 = (ms >= 0) & (ms < n)
            v4 = (msm >= 0) & (msm < n)
            g[v3] -= 1.0 - h[ms[v3]]
            g[v4] += h[msm[v4]]
            dm = f[:, None] + g[None, :]
            new_max = np.maximum(new_max, C[si] + dm * dx)
        min_val = new_max.min()
        bi, bo = np.unravel_index(new_max.argmin(), new_max.shape)
        delta = min_val - c5_cur
        accept = delta < -1e-12
        if not accept and delta < 10 * tau:
            accept = rng.random() < np.exp(-delta / max(tau, 1e-15))
        if accept:
            h[zeros[bi]] = 1.0
            h[ones[bo]] = 0.0
            C = np.correlate(h, 1.0 - h, mode="full") * dx
            c5_cur = C.max()
    return h, c5_cur

def block_swap_search(h, n, t_end, rng, max_rounds=8000):
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode="full") * dx
    c5_cur = C.max()
    sizes = [2, 3, 4, 5, 6, 8, 10, 14]
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        L = int(sizes[rng.integers(len(sizes))])
        if L * 2 >= n:
            continue
        j = int(rng.integers(0, n - L))
        m = int(rng.integers(0, n - L))
        if abs(j - m) < L:
            continue
        bj = h[j:j+L].copy()
        bm = h[m:m+L].copy()
        if abs(bj.sum() - (L - bm.sum())) > 1e-8:
            continue
        h[j:j+L] = bm
        h[m:m+L] = bj
        Cn = np.correlate(h, 1.0 - h, mode="full") * dx
        c5n = Cn.max()
        if c5n < c5_cur - 1e-12:
            C = Cn
            c5_cur = c5n
        else:
            h[j:j+L] = bj
            h[m:m+L] = bm
    return h, c5_cur
