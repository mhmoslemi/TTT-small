    h_sg = h.copy()
    h_prev = h.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    for it in range(n_iter):
        if time.time() - t0 > t0 + budget:
            break
        beta = float(it) / (it + 4.0) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top = np.argsort(corr)[-K:]
        mv = corr[top].max()
        w = np.exp(8.0 * (corr[top] - mv))
        w /= w.sum()
        g = np.zeros(n)
        for wk, ik in zip(w, top):
            g += wk * sg_raw(h_y, n, ik - (n - 1))
        g *= dx
        alpha = 0.07 / (1.0 + 0.003 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 10000 == 0:
            cv = compute_c5(h_sg)
            if cv < c5_best:
                c5_best = cv
                h_best = h_sg.copy()
    cv = compute_c5(project_feasible(h_sg, n))
    if cv < c5_best:
        c5_best = cv
        h_best = project_feasible(h_sg, n).copy()
    return h_best, c5_best

def balancer(h, n, n_rounds=300):
    """Iteratively balance the top overlaps by projecting gradient of worst shift."""
    h_c = h.copy()
    c5_cur = compute_c5(h_c)
    h_best = h_c.copy()
    c5_best = c5_cur
    alpha = 0.03
    for rnd in range(n_rounds):
        corr = np.correlate(h_c, 1.0 - h_c, mode="full")
        ak = int(np.argmax(corr))
        sh = ak - (n - 1)
        g = sg_raw(h_c, n, sh) * (2.0 / n)
        h_new = project_feasible(h_c - alpha * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h_c = h_new
            c5_cur = c5_new
            alpha = min(0.08, alpha * 1.02)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h_c.copy()
        else:
            alpha *= 0.7
    return h_best, c5_best

def coord_polish(h, n, rng, steps=2000):
    h_c = h.copy()
    c5_c = compute_c5(h_c)
    for _ in range(steps):
        corr = np.correlate(h_c, 1.0 - h_c, mode="full")
        ak = int(np.argmax(corr))
        sh = ak - (n - 1)
        g = sg_raw(h_c, n, sh) * (2.0 / n)
        pm = np.where(g > 0.004)[0]
        nm = np.where(g < -0.004)[0]
        if len(pm) == 0 or len(nm) == 0:
            break
        ip = pm[rng.integers(len(pm))]
        inm = nm[rng.integers(len(nm))]
        d = min(0.015, h_c[ip], 1.0 - h_c[inm])
        h_try = h_c.copy()
        h_try[ip] -= d
        h_try[inm] += d
        h_try = project_feasible(h_try, n)
        c5t = compute_c5(h_try)
        if c5t < c5_c - 1e-10:
            h_c = h_try
            c5_c = c5t
    return h_c, c5_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse-to-fine
    for T in [32, 16, 8, 4, 2, 1, 0.5]:
        if time.time() - t0 > budget_s - 420:
            break
        h_new = s_solve(h_best, T, 8000, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 2: Nesterov multi-shift
    h_ng, c5_ng = nesterov_multi(h_best, n, t0, 180, K=4)
    if c5_ng < c5_best:
        c5_best, h_best = c5_ng, h_ng

    # Phase 3: SLSQP fine polish
    for T in [0.5, 0.3, 0.15, 0.1, 0.05, 0.03]:
        if time.time() - t0 > budget_s - 200:
            break
        h_new = s_solve(h_best, T, 12000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 4: Balancer - targeted reduction of worst shift
    if time.time() - t0 < budget_s - 80:
        h_bal, c5_bal = balancer(h_best, n, n_rounds=500)
        if c5_bal < c5_best:
            c5_best, h_best = c5_bal, h_bal

    # Phase 5: Coordinate polish
    if time.time() - t0 < budget_s - 50:
        h_cp, c5_cp = coord_polish(h_best, n, rng, steps=3000)
        if c5_cp < c5_best:
            c5_best, h_best = c5_cp, h_cp

    # Phase 6: Final SLSQP
    if time.time() - t0 < budget_s - 15:
        h_new = s_solve(h_best, 0.02, 20000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new

    # Phase 7: Micro-perturbations
    for pi in range(15):
        if time.time() - t0 > budget_s - 3:
            break
        sig = 0.0008 + 0.0012 * (pi % 4)
        h_p = project_feasible(h_best + rng.normal(0, sig, n), n)
        h_new = s_solve(h_p, 0.05, 8000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best, h_best = c5, h_new
        # Also try balancer from perturbed point
        if time.time() - t0 < budget_s - 3:
            h_b2, c5_b2 = balancer(h_best, n, n_rounds=100)
            if c5_b2 < c5_best:
                c5_best, h_best = c5_b2, h_b2

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
