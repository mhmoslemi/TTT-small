        c = compute_c5(t)
        if c < c5 - 1e-15:
            h, c5 = t, c
    return h, c5

def do_coord(h, tgt, sweeps=2):
    n = len(h)
    c5 = compute_c5(h)
    ds = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.018, -0.018]
    for _ in range(sweeps):
        imp = False
        for j in range(n):
            for dt in ds:
                v = h[j] + dt
                if v < 0 or v > 1:
                    continue
                t = h.copy()
                t[j] = v
                t = project(t, tgt)
                c = compute_c5(t)
                if c < c5 - 1e-15:
                    h, c5 = t, c
                    imp = True
                    break
        if not imp:
            break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    global _TARGET
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280).astype(float)
        h_best[::2] = 1.0
    n = len(h_best)
    tgt = n / 2.0
    _TARGET = tgt
    h_best = project(h_best, tgt)
    c5_best = compute_c5(h_best)
    deadline = t0 + budget_s - 4.0

    # Phase 1: SLSQP on top lags (18%)
    p1_end = t0 + budget_s * 0.18
    for _ in range(20):
        if time.time() > p1_end:
            break
        cr = get_corrs(h_best)
        top = np.argsort(cr)[-4:]
        for li in top:
            if time.time() > p1_end:
                break
            h_try = do_slsqp(h_best, li, n, 500)
            h_try = project(h_try, tgt)
            c = compute_c5(h_try)
            if c < c5_best - 1e-16:
                h_best, c5_best = h_try, c

    # Phase 2: Projected gradient with top-k gradient (48%)
    p2_end = t0 + budget_s * 0.66
    T = 80.0
    lr = 0.015
    step_c = 0
    best_no_improve = 0
    while time.time() < p2_end:
        c5_cur = compute_c5(h_best)
        if c5_cur < c5_best:
            c5_best = c5_cur
            best_no_improve = 0
        else:
            best_no_improve += 1

        g = sm_grad(h_best, T, n)
        c5_new = c5_cur
        h_new = h_best
        lr_try = lr
        for _ in range(30):
            h_trial = h_best - lr_try * g
            h_trial = project(h_trial, tgt)
            c_trial = compute_c5(h_trial)
            if c_trial < c5_cur - 1e-16:
                h_new = h_trial
                c5_new = c_trial
                break
            lr_try *= 0.4
        if c5_new < c5_cur - 1e-16:
            h_best = h_new
            lr = min(lr * 1.1, 0.08)
        else:
            lr = lr * 0.6
            # Try top-k gradient as alternative
            g2 = topk_grad(h_best, n, 5)
            gnorm = np.linalg.norm(g2)
            if gnorm > 1e-10:
                lr_try = lr * 0.5
                h_trial = h_best - lr_try * g2
                h_trial = project(h_trial, tgt)
                c_trial = compute_c5(h_trial)
                if c_trial < c5_cur - 1e-16:
                    h_best = h_trial
                    lr = min(lr_try * 1.1, 0.06)

        step_c += 1
        if step_c % 60 == 0:
            h_best, c5_bp = do_binary(h_best, tgt, 0.05)
            if c5_bp < c5_best:
                c5_best = c5_bp
        if step_c % 120 == 0:
            h_best, c5_cd = do_coord(h_best, tgt, 1)
            if c5_cd < c5_best:
                c5_best = c5_cd
        if step_c % 400 == 0:
            T = min(T * 2.0, 8000.0)
        if best_no_improve > 400:
            h_pert = h_best.copy()
            idx = rng.integers(0, n, 12)
            h_pert[idx] += rng.uniform(-0.06, 0.06, 12)
            h_pert = project(h_pert, tgt)
            c5_p = compute_c5(h_pert)
            if c5_p < c5_best * 1.005:
                h_best = h_pert
                best_no_improve = 0

    # Phase 3: SLSQP refinement (14%)
    p3_end = t0 + budget_s * 0.80
    for _ in range(15):
        if time.time() > p3_end:
            break
        cr = get_corrs(h_best)
        top = np.argsort(cr)[-3:]
        for li in top:
            if time.time() > p3_end:
                break
            h_try = do_slsqp(h_best, li, n, 900)
            h_try = project(h_try, tgt)
            c = compute_c5(h_try)
            if c < c5_best - 1e-16:
                h_best, c5_best = h_try, c

    # Phase 4: Multi-n exploration (5%)
    p4_end = t0 + budget_s * 0.92
    for n_alt in [n - 20, n + 20, n - 40, n + 40]:
        if time.time() > p4_end:
            break
        if n_alt < 100 or n_alt > 600:
            continue
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = project(h_alt, n_alt / 2.0)
        c5_alt = compute_c5(h_alt)
        tgt_alt = n_alt / 2.0
        for _ in range(4):
            if time.time() > p4_end:
                break
            cr_a = get_corrs(h_alt)
            li = int(np.argmax(cr_a))
            h_try = do_slsqp(h_alt, li, n_alt, 400)
            h_try = project(h_try, tgt_alt)
            c = compute_c5(h_try)
            if c < c5_alt - 1e-16:
                h_alt, c5_alt = h_try, c
        h_alt, c5_cd = do_coord(h_alt, tgt_alt, 2)
        if c5_cd < c5_alt:
            c5_alt = c5_cd
        h_alt, c5_bp = do_binary(h_alt, tgt_alt, 0.04)
        if c5_bp < c5_alt:
            c5_alt = c5_bp
        if c5_alt < c5_best:
            h_best, c5_best = h_alt, c5_alt
            n, tgt = n_alt, tgt_alt

    # Phase 5: Final polish (8%)
    h_best, c5_cd = do_coord(h_best, tgt, 4)
    if c5_cd < c5_best:
        c5_best = c5_cd
    h_best, c5_bp = do_binary(h_best, tgt, 0.03)
    if c5_bp < c5_best:
        c5_best = c5_bp
    if time.time() < deadline:
        cr = get_corrs(h_best)
        li = int(np.argmax(cr))
        h_try = do_slsqp(h_best, li, n, 1500)
        h_try = project(h_try, tgt)
        c = compute_c5(h_try)
        if c < c5_best:
            h_best, c5_best = h_try, c
        h_best, c5_cd2 = do_coord(h_best, tgt, 2)
        if c5_cd2 < c5_best:
            c5_best = c5_cd2

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
