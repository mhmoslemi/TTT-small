import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

_gN = 0
_gT = 1.0
_gK = 20

def _ls_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_gK:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_gT * (c - m))
    return (m + np.log(e.sum()) / _gT) * (2.0 / _gN)

def _ls_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_gK:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_gT * (c - m))
    w = e / e.sum()
    grad = np.zeros(_gN)
    for wi, ci in zip(w, idx):
        if wi > 1e-10:
            grad += wi * shift_grad(x, _gN, ci - (_gN - 1))
    return grad * (2.0 / _gN)

def _eq(x):
    return x.sum() - _gN / 2.0

def _eq_j(x):
    return np.ones(_gN)

def _slsqp(h, n, T, top_k, maxiter):
    global _gN, _gT, _gK
    _gN = n
    _gT = T
    _gK = min(top_k, 2 * n - 1)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_ls_obj, h, method='SLSQP', jac=_ls_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eq, 'jac': _eq_j}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return res.x
    except Exception:
        return h.copy()

def _norm_p_pen(x, n, p, mu):
    corr = np.abs(np.correlate(x, 1.0 - x, mode="full"))
    val = (np.power(corr, p).sum() / len(corr)) ** (1.0 / p) * (2.0 / n)
    return val + mu * (x.sum() - n / 2.0) ** 2

def _norm_p_pen_j(x, n, p, mu):
    corr = np.abs(np.correlate(x, 1.0 - x, mode="full"))
    base = (np.power(corr, p).sum() / len(corr)) ** (1.0 / p)
    w = np.power(corr, p - 1)
    w = w / w.sum()
    grad = np.zeros(n)
    for wi, ci in zip(w, range(len(corr))):
        if wi > 1e-12:
            grad += wi * shift_grad(x, n, ci - (n - 1))
    grad *= (2.0 / n) * base / (corr[[ci for ci in range(len(corr))]].max() + 1e-15)
    grad *= 1.0
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def _lbfgs(h, n, T, top_k, maxiter, mu):
    global _gN, _gT, _gK
    _gN = n
    _gT = T
    _gK = min(top_k, 2 * n - 1)
    def obj(x):
        return _ls_obj(x) + mu * (x.sum() - n / 2.0) ** 2
    def j(x):
        return _ls_jac(x) + 2.0 * mu * (x.sum() - n / 2.0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=j, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return res.x
    except Exception:
        return h.copy()

def _nesterov_adapt(h_init, n, t_end, k_start=20, k_end=6):
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5a = compute_c5(h_init)
    ha = h_init.copy()
    total_it = 400000
    for it in range(total_it):
        if time.time() > t_end:
            break
        beta = float(it) / (it + 1.5)
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        fraction = min(1.0, it / (0.5 * total_it))
        k_now = max(k_end, int(k_start - (k_start - k_end) * fraction))
        idx = np.argsort(corr)[-k_now:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h_y, n, ci - (n - 1))
        g *= dx / k_now
        alpha = 0.4 / (1.0 + 0.0008 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 5000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < c5a:
                c5a, ha = c1, h_cur.copy()
            h_p = project_feasible(h_avg, n)
            c2 = compute_c5(h_p)
            if c2 < c5a:
                c5a, ha = c2, h_p.copy()
    return ha, c5a

def _adam(h_init, n, t_end, lr=0.01, b1=0.9, b2=0.999, k=12):
    dx = 2.0 / n
    h_cur = h_init.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    c5a = compute_c5(h_init)
    ha = h_init.copy()
    for it in range(200000):
        if time.time() > t_end:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        idx = np.argsort(corr)[-k:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h_cur, n, ci - (n - 1))
        g *= dx / k
        m_vec = b1 * m_vec + (1 - b1) * g
        v_vec = b2 * v_vec + (1 - b2) * g * g
        m_hat = m_vec / (1 - b1 ** (it + 1))
        v_hat = v_vec / (1 - b2 ** (it + 1))
        h_new = project_feasible(h_cur - lr * m_hat / (np.sqrt(v_hat) + 1e-8), n)
        h_cur = h_new
        if (it + 1) % 5000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < c5a:
                c5a, ha = c1, h_cur.copy()
    return ha, c5a

def _perturb(h_best, n, rng, pi):
    mode = pi % 5
    if mode == 0:
        sig = 0.004 + 0.002 * (pi // 5) % 4
        return project_feasible(h_best + rng.normal(0, sig, n), n)
    elif mode == 1:
        h_p = h_best.copy()
        lb = int(rng.integers(0, max(1, n - 30)))
        ln = int(rng.integers(5, 30))
        h_p[lb:lb + ln] = 1.0 - h_p[lb:lb + ln]
        return project_feasible(h_p, n)
    elif mode == 2:
        h_p = h_best.copy()
        freq = int(rng.integers(2, max(3, n // 4)))
        phase = rng.uniform(0, 6.2832)
        sig = 0.006
        t = np.arange(n)
        h_p += sig * np.sin(2.0 * np.pi * freq * t / n + phase)
        return project_feasible(h_p, n)
    elif mode == 3:
        h_p = h_best.copy()
        k = n // 4
        i1, i2 = [int(x) for x in rng.choice(n, 2, replace=False)]
        h_p[i1:i1 + k], h_p[i2:i2 + k] = h_p[i2:i2 + k].copy(), h_p[i1:i1 + k].copy()
        return project_feasible(h_p, n)
    else:
        h_p = h_best.copy()
        center = int(rng.integers(n // 4, 3 * n // 4))
        width = int(rng.integers(max(2, n // 10), n // 4))
        amp = rng.uniform(-0.08, 0.08)
        t = np.arange(n)
        h_p += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
        return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n = 280
        h_best = np.zeros(n)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse-to-medium
    t1 = t0 + budget_s * 0.22
    for T, tk, mi in [(32, 30, 1500), (8, 20, 2500), (2, 12, 3500)]:
        if time.time() > t1:
            break
        hn = _slsqp(h_best, n, T, tk, mi)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 2: L-BFGS-B (fast, many iterations)
    t2 = t0 + budget_s * 0.35
    for T, tk, mi, mu in [(4, 15, 5000, 20.0), (1, 10, 8000, 50.0)]:
        if time.time() > t2:
            break
        hn = _lbfgs(h_best, n, T, tk, mi, mu)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 3: Nesterov with adaptive K
    t3 = t0 + budget_s * 0.52

    if c5s < c5_best:

    # Phase 4: SLSQP fine
    t4 = t0 + budget_s * 0.80
    for T, tk, mi in [(1, 10, 5000), (0.5, 8, 7000), (0.3, 6, 8000), (0.5, 10, 6000)]:

        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:

    # Phase 6: Adam for fine-tuning

        if c5a < c5_best:

    # Phase 7: Final ultra-fine SLSQP
    if time.time() < budget_s - 10:

        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
