import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def binary_push(h, n, strength=0.3):
    """Push values towards 0 or 1."""
    h_copy = h.copy()
    # Values above 0.5 push towards 1, below towards 0
    above = h_copy > 0.5
    below = ~above
    h_copy[above] = h_copy[above] + strength * (1.0 - h_copy[above])
    h_copy[below] = h_copy[below] - strength * h_copy[below]
    return project_feasible(h_copy, n)

def topk_smooth_obj(x, n, T, k):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = np.max(topk)
    sh = T * (topk - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    obj = (mc + np.log(e.sum()) / T) * (2.0 / n)
    return obj, idx, sw

def topk_smooth_grad(x, n, T, k):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = np.max(topk)
    sh = T * (topk - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    j = np.arange(n)
    for oi in range(mk):
        w = sw[oi]
        if w < 1e-12:
            continue
        shift = idx[oi] - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += w * (1.0 - x[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= w * x[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += w * (1.0 - x[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * x[j[m2] - s2]
    grad *= 2.0 / n
    return grad

def slsqp_opt(h, n, T, k, maxiter, ftol, t_end):
    target = n / 2.0
    def obj(x):
        v, _, _ = topk_smooth_obj(x, n, T, k)
        return v
    def jac(x):
        return topk_smooth_grad(x, n, T, k)
    def cf(x):
        return x.sum() - target
    def cj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

                    v2 = h_work[wi + shift]

                        # Flip the one that's more extreme

                            h_work[wi] = 1.0 - h_work[wi]
                        else:
                            h_work[wi + shift] = 1.0 - h_work[wi + shift]
                    else:

                        else:

        c5 = compute_c5(h_work)

  c = sum_{i=s2}^{n-1} h[i-s2] * (1-h[i])

To minimize c:
- For shift >= 0: we want h[i+shift] ≈ h[i]. Where h[i+shift]=1 and h[i]=0, the contribution is 1. Where h[i+shift]=0 and h[i]=1, contribution is 0.
  Actually: h[i+shift]*(1-h[i]) is maximized when h[i+shift]=1 and h[i]=0.
  
- For shift < 0 (s2 = -shift): h[i-s2]*(1-h[i]) is maximized when h[i-s2]=1 and h[i]=0.

So in both cases, the "bad" pairs are where the "source" value is 1 and the "destination" value is 0.

For the surgery:
- shift >= 0: pairs (i, i+shift). Bad when h[i+shift]=1, h[i]=0.
  Fix: set h[i+shift] → 0 or h[i] → 1.
- shift < 0, s2=-shift: pairs (i-s2, i). Bad when h[i-s2]=1, h[i]=0.
  Fix: set h[i-s2] → 0 or h[i] → 1.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 3.0
    hi = y.max() + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def binary_push(h, n, strength=0.2):
    hc = h.copy()
    above = hc > 0.5
    below = ~above
    hc[above] += strength * (1.0 - hc[above])
    hc[below] -= strength * hc[below]
    return project_feasible(hc, n)

def topk_obj(x, n, T, k):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = np.max(topk)
    sh = T * (topk - mc)
    e = np.exp(sh)
    obj = (mc + np.log(e.sum()) / T) * (2.0 / n)
    return obj, idx, e / e.sum()

def topk_grad(x, n, T, k):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = np.max(topk)
    sh = T * (topk - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    j = np.arange(n)
    for oi in range(mk):
        w = sw[oi]
        if w < 1e-13:
            continue
        shift = idx[oi] - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += w * (1.0 - x[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= w * x[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += w * (1.0 - x[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * x[j[m2] - s2]
    grad *= 2.0 / n
    return grad

def slsqp(h, n, T, k, maxiter, t_end):
    target = n / 2.0
    def obj(x):
        return topk_obj(x, n, T, k)[0]
    def jac(x):
        return topk_grad(x, n, T, k)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
        return res.x
    except Exception:
        return h.copy()

def fista(h_init, n, t_end, lr0=0.02):
    dx = 2.0 / n
    h_y = h_init.copy()
    h_z = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    t_mom = 1.0
    lr = lr0
    stall = 0
    for it in range(300000):
        if time.time() > t_end:
            break
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        k3 = 3
        idx3 = np.argpartition(corr, -k3)[-k3:]
        grad = np.zeros(n)
        center = n - 1
        j = np.arange(n)
        for oi in idx3:
            shift = oi - center
            if shift >= 0:
                m1 = j >= shift
                grad[m1] += (1.0 - h_y[j[m1] - shift])
                m2 = (j + shift) < n
                grad[m2] -= h_y[j[m2] + shift]
            else:
                s2 = -shift
                m1 = (j + s2) < n
                grad[m1] += (1.0 - h_y[j[m1] + s2])
                m2 = j >= s2
                grad[m2] -= h_y[j[m2] - s2]
        grad *= dx / k3
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        h_z = h_y + ((t_mom - 1.0) / t_new) * (h_y - h_z)
        h_y = h_z - lr * grad
        h_y = project_feasible(h_y, n)
        t_mom = t_new
        lr *= 0.99995
        if (it + 1) % 3000 == 0:
            h_y = binary_push(h_y, n, 0.08)
            c5 = compute_c5(h_y)
            if c5 < c5_best - 1e-8:
                c5_best = c5
                h_best = h_y.copy()
                stall = 0
            else:
                stall += 1
            if stall > 5:
                h_z = h_y.copy()
                t_mom = 1.0
                lr = lr0 * 0.5
                stall = 0
    return h_best, c5_best

def perturb(h, n, rng):
    h_p = h.copy()
    pt = rng.integers(0, 7)
    if pt == 0:
        h_p = np.roll(h_p, int(rng.integers(1, 6)))
    elif pt == 1:
        st = rng.integers(0, max(1, n - 12))
        ln = rng.integers(3, min(18, n - st))
        h_p[st:st+ln] = 1.0 - h_p[st:st+ln]
    elif pt == 2:
        freq = rng.integers(1, 15)
        amp = rng.uniform(0.01, 0.06)
        x = np.arange(n) * 2.0 / n
        h_p += amp * np.sin(np.pi * freq * x + rng.uniform(0, 6.28))
    elif pt == 3:
        h_p += rng.uniform(0.003, 0.025) * rng.normal(0, 1, n)
    elif pt == 4:
        b = rng.integers(4, 20)
        i = rng.integers(0, max(1, n - b))
        jj = rng.integers(0, max(1, n - b))
        tmp = h_p[i:i+b].copy()
        h_p[i:i+b] = h_p[jj:jj+b]
        h_p[jj:jj+b] = tmp
    elif pt == 5:
        fs = np.fft.rfft(h_p - 0.5)
        for _ in range(3):
            fs[rng.integers(0, len(fs))] *= rng.normal(1.0, 0.15)
        h_p = np.fft.irfft(fs, n=n) + 0.5
    else:
        mask = rng.random(n) < 0.08
        idx = np.where(mask)[0]
        if len(idx) > 0:
            h_p[idx] = rng.random(len(idx))
    return project_feasible(h_p, n)
