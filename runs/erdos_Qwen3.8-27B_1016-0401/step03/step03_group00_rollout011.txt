def coord_descent(h, n, n_rounds=5, t_end=None):

    h = h.copy()

    for r in range(n_rounds):

            break
        improved = False

            if t_end is not None and time.time() > t_end:

            old = h[i]
            # Try a few values around current
            candidates = np.clip([old + d for d in [-0.1, -0.05, -0.01, 0.0, 0.01, 0.05, 0.1]], 0, 1)
            best_c5 = compute_c5(h)

            for cv in candidates:
                h[i] = cv

                    best_val = cv

            if best_val != old:

        if not improved:
            break

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

# Global state for callbacks
_NS = {'n': 0, 'dx': 1.0, 'T': 1.0, 'tk': 10, 'target': 0.0}

def _obj(x):
    h = np.clip(x, 0.0, 1.0)
    n = _NS['n']
    corr = np.correlate(h, 1.0 - h, mode="full")
    T = _NS['T']
    tk = _NS['tk']
    idx = np.argsort(corr)[-tk:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    return (m + np.log(e.sum()) / T) * _NS['dx']

def _jac(x):
    h = np.clip(x, 0.0, 1.0)
    n = _NS['n']
    corr = np.correlate(h, 1.0 - h, mode="full")
    T = _NS['T']
    tk = _NS['tk']
    idx = np.argsort(corr)[-tk:]
    c = corr[idx]
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for wi, ci in zip(w, idx):
        if wi > 1e-10:
            g += wi * shift_grad(h, n, ci - (n - 1))
    return g * _NS['dx']

def _eq(x):
    return x.sum() - _NS['target']

def _eq_j(x):
    return np.ones(_NS['n'])

def do_slsqp(h, n, T, tk, maxiter, ftol):
    _NS['n'] = n
    _NS['dx'] = 2.0 / n
    _NS['T'] = float(T)
    _NS['tk'] = min(int(tk), 2 * n - 1)
    _NS['target'] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h, method='SLSQP', jac=_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eq, 'jac': _eq_j}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def do_nesterov(h_init, n, t_end, rng, top_k=8, max_iter=300000):
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5a = compute_c5(h_init)
    ha = h_init.copy()
    for it in range(max_iter):
        if time.time() > t_end:
            break
        beta = float(it) / (it + 1.5)
        h_y = h_cur + beta * (h_cur - h_prev)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h_y, n, ci - (n - 1))
        g *= dx / top_k
        # Adaptive: reduce step if C5 is increasing
        alpha = 0.4 / (1.0 + 0.0015 * it)
        h_cand = project_feasible(h_y - alpha * g, n)
        # Simple line search: if it gets worse, reduce step
        c_new = compute_c5(h_cand)
        if c_new > c5a * 1.01 and it > 100:
            alpha *= 0.5
            h_cand = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_cand
        h_avg += (h_cand - h_avg) / (it + 1)
        if (it + 1) % 3000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < c5a:
                c5a, ha = c1, h_cur.copy()
            h_av = project_feasible(h_avg, n)
            c2 = compute_c5(h_av)
            if c2 < c5a:
                c5a, ha = c2, h_av.copy()
    return ha, c5a

        hp = project_feasible(hp, n)
    hn = do_slsqp(hp, n, T, tk, maxiter, 1e-15)
    hn = project_feasible(hn, n)
    c5 = compute_c5(hn)
