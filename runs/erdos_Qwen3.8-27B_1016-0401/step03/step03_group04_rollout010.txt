    j = np.arange(n)

        h2 = project_feasible(h2, n2)
        c5 = compute_c5(h2)
        if c5 < c5_best - 1e-6:
            c5_best = c5
            h_best = h2
            n = n2
            h_new = lbfgsb_opt(h_best, n, 1.0, 40, 4000, 1e-15, dl - 5)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
