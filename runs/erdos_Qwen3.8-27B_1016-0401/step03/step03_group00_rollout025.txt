        h_try = h_cur.copy()
        h_try[i], h_try[j] = h_cur[j], h_cur[i]
        c5_try = compute_c5(h_try)
        if c5_try < c5_cur:
            h_cur, c5_cur = h_try, c5_try
            if c5_cur < c5_best:
                c5_best, h_best = c5_cur, h_cur.copy()
            continue
        for d in transfers:
            if h_cur[i] >= d and h_cur[j] <= 1.0 - d:
                h_try2 = h_cur.copy()
                h_try2[i] -= d
                h_try2[j] += d
                c5_t2 = compute_c5(h_try2)
                if c5_t2 < c5_cur:
                    h_cur, c5_cur = h_try2, c5_t2
                    if c5_cur < c5_best:
                        c5_best, h_best = c5_cur, h_cur.copy()
                    break
            if h_cur[j] >= d and h_cur[i] <= 1.0 - d:
                h_try3 = h_cur.copy()
                h_try3[j] -= d
                h_try3[i] += d
                c5_t3 = compute_c5(h_try3)
                if c5_t3 < c5_cur:
                    h_cur, c5_cur = h_try3, c5_t3
                    if c5_cur < c5_best:
                        c5_best, h_best = c5_cur, h_cur.copy()
                    break
        h_try4 = h_cur.copy()
        h_try4[i] = 1.0 - h_cur[i]
        h_try4 = project_feasible(h_try4, n)
        c5_t4 = compute_c5(h_try4)
        if c5_t4 < c5_cur:
            h_cur, c5_cur = h_try4, c5_t4
            if c5_cur < c5_best:
                c5_best, h_best = c5_cur, h_cur.copy()
    return h_best, c5_best

def _perturb_focused(h, n, rng, corr, pi):
    top = np.argsort(corr)[-15:]
    mode = pi % 3
    if mode == 0:
        sig = 0.001 + 0.003 * (pi % 5)
        return project_feasible(h + rng.normal(0, sig, n), n)
    elif mode == 1:
        hp = h.copy()
        k = top[pi % len(top)]
        shift = k - (n - 1)
        if shift >= 0:
            mask = np.arange(n) >= shift
            idxs = np.where(mask)[0]
            swap_i = idxs[pi % len(idxs)]
            swap_j = (swap_i + shift) % n
            hp[swap_i], hp[swap_j] = hp[swap_j], hp[swap_i]
        return project_feasible(hp, n)
    else:
        hp = h.copy()
        lb = pi % (n - 20)
        ln = 5 + (pi * 7) % 15
        hp[lb:lb + ln] = 1.0 - hp[lb:lb + ln]
        return project_feasible(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n0 = 280
        h_best = np.zeros(n0)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse-to-fine
    t1 = t0 + budget_s * 0.28
    for T, tk, mi in [(64, 30, 1500), (16, 20, 2500), (4, 15, 3500),
                       (1, 10, 4500), (0.5, 8, 5000)]:
        if time.time() > t1:
            break
        hn = _slsqp(h_best, n, T, tk, mi, 1e-15)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 2: Subgradient descent (new)
    t2 = t0 + budget_s * 0.45
    hs, c5s = _subgrad_descent(h_best, n, t2, top_k=10, max_iter=30000)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs

    # Phase 3: Very tight SLSQP
    t3 = t0 + budget_s * 0.65
    for T, tk, mi in [(1, 8, 4000), (0.5, 6, 6000), (0.3, 5, 8000),
                       (0.2, 4, 10000), (0.1, 3, 12000)]:
        if time.time() > t3:
            break
        hn = _slsqp(h_best, n, T, tk, mi, 1e-16)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 4: Swap local search (new)
    t4 = t0 + budget_s * 0.80
    hl, c5l = _swap_search(h_best, n, t4, max_iter=80000)
    if c5l < c5_best:
        c5_best, h_best = c5l, hl

    # Phase 5: Focused perturbations + SLSQP
    t5 = t0 + budget_s * 0.93
    for pi in range(40):
        if time.time() > t5:
            break
        corr_cur = np.correlate(h_best, 1.0 - h_best, mode="full")
        hp = _perturb_focused(h_best, n, rng, corr_cur, pi)
        hn = _slsqp(hp, n, 0.3, 5, 3000, 1e-15)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 6: Final ultra-fine polish
    if time.time() < t0 + budget_s - 15:
        hn = _slsqp(h_best, n, 0.1, 3, 10000, 1e-17)
        hn = project_feasible(hn, n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn
        # Final micro-swap
        hm, c5m = _swap_search(h_best, n, t0 + budget_s - 5, max_iter=20000)
        if c5m < c5_best:
            c5_best, h_best = c5m, hm

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
