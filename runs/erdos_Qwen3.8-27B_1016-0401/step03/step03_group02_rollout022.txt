    res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,

    return np.clip(res.x, 0.0, 1.0)

def local_search(h, t_end, max_sweeps=500):
    n = len(h)
    h = h.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    for _ in range(max_sweeps):
        if time.time() > t_end: break
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h > 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0: break
        improved = False
        for i in zeros:
            for j in ones:
                if i == j: continue
                h[i], h[j] = 1.0, 0.0
                c5_new = compute_c5(h)
                if c5_new < c5_cur - 1e-15:
                    c5_cur = c5_new
                    improved = True
                    if c5_cur < c5_best:
                        c5_best = c5_cur
                        h_best = h.copy()
                else:
                    h[i], h[j] = 0.0, 1.0
        if not improved: break
    return h_best, c5_best

def local_search_sa(h, t_end, rng, max_iters=200000):
    n = len(h)
    h = h.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    T0, T1 = 0.001, 0.00002
    it = 0
    while time.time() < t_end and it < max_iters:
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h > 0.5)[0]
        if len(zeros) < 1 or len(ones) < 1: break
        i = zeros[rng.integers(len(zeros))]
        j = ones[rng.integers(len(ones))]
        if i == j: it += 1; continue
        h[i], h[j] = 1.0, 0.0
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        T = T0 * (T1 / T0) ** (it / max_iters)
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5_cur = c5_new
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
        it += 1
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(340); h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: SLSQP on top lags (12% budget)
    p1 = t0 + budget_s * 0.12
    for _ in range(20):
        if time.time() > p1: break
        corr = compute_corr(h_best)
        top = np.argsort(corr)[-3:]
        for lag in top:
            if time.time() > p1: break
            try:
                h_new = slsqp_lag(h_best, lag, bounds, 400, 1e-16)
                h_new = fix_constraint(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-16:
                    c5_best = c5_new; h_best = h_new
            except Exception: pass

    # Phase 2: Binarize + greedy local search (30% budget)
    p2 = t0 + budget_s * 0.42
    h_bin = binarize(h_best, n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < c5_best: c5_best = c5_bin; h_best = h_bin.copy()
    h_ls, c5_ls = local_search(h_best, p2, max_sweeps=400)
    if c5_ls < c5_best: c5_best = c5_ls; h_best = h_ls.copy()

    # Phase 3: SLSQP from near-binary + re-binarize cycle (25%)
    p3 = t0 + budget_s * 0.67
    for cycle in range(15):
        if time.time() > p3: break
        corr = compute_corr(h_best)
        top = np.argsort(corr)[-4:]
        for lag in top:
            if time.time() > p3: break
            try:
                h_new = slsqp_lag(h_best, lag, bounds, 600, 1e-17)
                h_new = fix_constraint(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-16:
                    c5_best = c5_new; h_best = h_new
            except Exception: pass
        h_b2 = binarize(h_best, n)
        h_s2, c5_s2 = local_search(h_b2, min(time.time() + budget_s*0.02, p3), 80)
        if c5_s2 < c5_best: c5_best = c5_s2; h_best = h_s2.copy()

    # Phase 4: Simulated annealing on current binary (15%)
    p4 = t0 + budget_s * 0.82
    h_sa, c5_sa = local_search_sa(h_best, p4, rng, max_iters=150000)
    if c5_sa < c5_best: c5_best = c5_sa; h_best = h_sa.copy()
    h_g, c5_g = local_search(h_best, min(time.time() + budget_s*0.03, p4), 200)
    if c5_g < c5_best: c5_best = c5_g; h_best = h_g.copy()

    # Phase 5: Multi-n exploration (10%)
    p5 = t0 + budget_s * 0.92
    for n_alt in [n - 20, n + 20, n + 40, 400]:
        if time.time() > p5: break
        if n_alt < 100 or n_alt == n: continue
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = fix_constraint(h_alt, n_alt / 2.0)
        b_alt = [(0.0, 1.0)] * n_alt
        for _ in range(4):
            if time.time() > p5: break
            ca = compute_corr(h_alt)
            wl = int(np.argmax(ca))
            try:
                h_an = slsqp_lag(h_alt, wl, b_alt, 400, 1e-15)
                h_an = fix_constraint(h_an, n_alt / 2.0)
                c5_an = compute_c5(h_an)
                c5_a = compute_c5(h_alt)
                if c5_an < c5_a: h_alt = h_an
            except Exception: pass
        h_ab = binarize(h_alt, n_alt)
        h_ab, c5_ab = local_search(h_ab, min(time.time() + budget_s*0.015, p5), 60)
        if c5_ab < c5_best:
            c5_best = c5_ab; h_best = h_ab
            n = n_alt; target = n / 2.0; bounds = [(0.0, 1.0)] * n

    # Phase 6: Final SLSQP polish (8%)
    p6 = t0 + budget_s - 3.0
    for _ in range(25):
        if time.time() > p6: break
        corr = compute_corr(h_best)
        worst = int(np.argmax(corr))
        try:
            h_new = slsqp_lag(h_best, worst, bounds, 1000, 1e-19)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new; h_best = h_new
        except Exception: pass
    h_dl, c5_dl = local_search(h_best, min(time.time() + budget_s * 0.02, p6), 100)
    if c5_dl < c5_best: c5_best = c5_dl; h_best = h_dl.copy()

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
