    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * (2.0 / n)

def sm_grad(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    # Only process lags with significant weight
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < 1e-10:
            continue
        if o == 0:

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                if a > 1e-15:
                    h[i] += a
                    diff -= a
            else:
                s = min(-diff, h[i])
                if s > 1e-15:
                    h[i] -= s
                    diff += s
            if abs(diff) < 1e-14:
                break
    return h

def eq_fun(x):
    return x.sum() - _TARGET

def eq_jac(x):
    return np.ones(len(x))

def sm_obj(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shim = T * (corr - mc)
    shim = np.clip(shim, -500, 500)
    return (mc + np.log(np.sum(np.exp(shim))) / T) * (2.0 / n)

def sm_grad(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shim = T * (corr - mc)
    shim = np.clip(shim, -500, 500)
    e = np.exp(shim)
    sw = e / np.sum(e)
    center = n - 1
    dx = 2.0 / n
    grad = np.zeros(n)
    thresh = 1e-9
    for o in range(-(n-1), n):
        w = sw[center + o]
        if w < thresh:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n-o])
            grad[:n-o] += -w * x[o:]
        else:
            p = -o
            grad[:n-p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n-p]
    return grad * dx

def slsqp_sm(h, T, bounds, maxiter, ftol):
    global _TARGET
    _TARGET = len(h) / 2.0
    res = minimize(sm_obj, h, args=(T,), method='SLSQP',
                  jac=sm_grad, bounds=bounds,
                  constraints=[dict(type='eq', fun=eq_fun, jac=eq_jac)],
                  options={'maxiter': maxiter, 'ftol': ftol})
    return np.clip(res.x, 0.0, 1.0)

def slsqp_lag(h, lag_idx, bounds, maxiter, ftol):
    global _TARGET
    _TARGET = len(h) / 2.0

    def lag_obj(x):
        n = len(x)
        return np.correlate(x, 1.0 - x, mode="full")[lag_idx] * (2.0 / n)

    def lag_grad(x):
        n = len(x)
        o = lag_idx - (n - 1)
        dx = 2.0 / n
        g = np.zeros(n)
        if o == 0:
            g[:] = 1.0 - 2.0 * x
        elif o > 0:
            g[o:] += 1.0 - x[:n-o]
            g[:n-o] += -x[o:]
        else:
            p = -o
            g[:n-p] += 1.0 - x[p:]
            g[p:] += -x[:n-p]
        return g * dx

    res = minimize(lag_obj, h, method='SLSQP', jac=lag_grad,
                  bounds=bounds,
                  constraints=[dict(type='eq', fun=eq_fun, jac=eq_jac)],
                  options={'maxiter': maxiter, 'ftol': ftol})
    return np.clip(res.x, 0.0, 1.0)

def subgrad_step(h, lr, top_k=20):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    grad = np.zeros(n)
    for idx in top_idx:
        o = idx - center
        if o == 0:
            grad += 1.0 - 2.0 * h
        elif o > 0:
            grad[o:] += 1.0 - h[:n-o]
            grad[:n-o] += -h[o:]
        else:
            p = -o
            grad[:n-p] += 1.0 - h[p:]
            grad[p:] += -h[:n-p]
    grad *= dx / len(top_idx)
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_constraint(h_new, n / 2.0)

def coord_descent(h, n_sweeps, steps):
    n = len(h)
    target = n / 2.0
    c5_cur = compute_c5(h)
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            best_t = None
            best_c = c5_cur
            for dt in steps:
                trial = h.copy()
                trial[j] = np.clip(trial[j] + dt, 0.0, 1.0)
                trial = fix_constraint(trial, target)
                c5_n = compute_c5(trial)
                if c5_n < best_c - 1e-15:
                    best_c = c5_n
                    best_t = trial
            if best_t is not None:
                h = best_t
                c5_cur = best_c
                improved = True
        if not improved:
            break
    return h, c5_cur

def binary_push(h, thresh=0.05):
    n = len(h)
    target = n / 2.0
    c5_cur = compute_c5(h)
    for j in range(n):
        done = False
        for tv in ([1.0] if h[j] > thresh else []):
            trial = h.copy()
            trial[j] = tv
            trial = fix_constraint(trial, target)
            if compute_c5(trial) < c5_cur - 1e-14:
                h = trial
                c5_cur = compute_c5(h)
                done = True
        if not done:
            for tv in ([0.0] if h[j] < 1.0 - thresh else []):
                trial = h.copy()
                trial[j] = tv
                trial = fix_constraint(trial, target)
                if compute_c5(trial) < c5_cur - 1e-14:
                    h = trial
                    c5_cur = compute_c5(h)
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    global _TARGET
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(340)
        h_best[::2] = 1.0
    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: SLSQP smooth max, T schedule (35% budget)
    p1_end = t0 + budget_s * 0.35
    temps = [100, 300, 1000, 3000, 10000, 30000, 80000, 200000]
    for T in temps:
        if elapsed() > p1_end:
            break
        maxit = max(200, int(800 * (100 / max(T, 100)) ** 0.5))
        for _ in range(2):
            if elapsed() > p1_end:
                break
            try:
                h_new = slsqp_sm(h_best, T, bounds, maxit, 1e-16)
                h_new = fix_constraint(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-15:
                    c5_best = c5_new
                    h_best = h_new.copy()
            except Exception:
                pass

    # Phase 2: Subgradient + periodic SLSQP (25% budget)
    p2_end = t0 + budget_s * 0.60
    lr = 0.005
    no_improve = 0
    for it in range(20000):
        if elapsed() > p2_end:
            break
        h_new = subgrad_step(h_best, lr, top_k=15)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-15:
            c5_best = c5_new
            h_best = h_new
            no_improve = 0
        else:
            no_improve += 1
        lr = max(0.0005, lr * 0.9995)
        if it % 500 == 499:
            try:
                h_new2 = slsqp_sm(h_best, max(1000, int(2000 / (1 + no_improve/100))), bounds, 400, 1e-15)
                h_new2 = fix_constraint(h_new2, target)
                c5_new2 = compute_c5(h_new2)
                if c5_new2 < c5_best - 1e-15:
                    c5_best = c5_new2
                    h_best = h_new2
                    no_improve = 0
            except Exception:
                pass

    # Phase 3: Lag cycling SLSQP + coord descent + binary push (25% budget)
    p3_end = t0 + budget_s * 0.85
    for cycle in range(30):
        if elapsed() > p3_end:
            break
        # Optimize worst 3 lags
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        top3 = np.argsort(corr)[-3:]
        for li in top3:
            if elapsed() > p3_end:
                break
            try:
                h_new = slsqp_lag(h_best, li, bounds, 600, 1e-17)
                h_new = fix_constraint(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-16:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass
        # Fine coordinate descent
        steps_fine = [0.005, -0.005, 0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005]
        h_best, c5_cd = coord_descent(h_best, 2, steps_fine)
        if c5_cd < c5_best:
            c5_best = c5_cd
        # Binary push
        h_best, c5_bp = binary_push(h_best, 0.03)
        if c5_bp < c5_best:
            c5_best = c5_bp
        # Smooth max polish
        try:
            h_new = slsqp_sm(h_best, 20000, bounds, 500, 1e-17)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-16:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 4: Multi-n + final polish (15% budget)
    p4_end = t0 + budget_s - 5.0
    n_tries = [336, 344, 348, 320, 360]
    for n_alt in n_tries:
        if elapsed() > p4_end:
            break
        if n_alt == n:
            continue
        tgt_a = n_alt / 2.0
        x_old = np.arange(len(h_best)) * (2.0 / len(h_best))
        x_new = np.linspace(0.0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = fix_constraint(h_alt, tgt_a)
        c5_alt = compute_c5(h_alt)
        bnd_a = [(0.0, 1.0)] * n_alt
        for T in [2000, 10000, 50000]:
            if elapsed() > p4_end:
                break
            try:
                h_new = slsqp_sm(h_alt, T, bnd_a, 500, 1e-16)
                h_new = fix_constraint(h_new, tgt_a)
                c5_new = compute_c5(h_new)
                if c5_new < c5_alt - 1e-16:
                    c5_alt = c5_new
                    h_alt = h_new
            except Exception:
                pass
        h_alt, c5_a = coord_descent(h_alt, 2, [0.003, -0.003, 0.001, -0.001])
        if c5_a < c5_alt:
            c5_alt = c5_a
        h_alt, c5_bp_a = binary_push(h_alt, 0.03)
        if c5_bp_a < c5_alt:
            c5_alt = c5_bp_a
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt
            target = n / 2.0
            bounds = [(0.0, 1.0)] * n

    # Final polish
    target = len(h_best) / 2.0
    n = len(h_best)
    bounds = [(0.0, 1.0)] * n
    while elapsed() < t0 + budget_s - 3.0:
        try:
            h_new = slsqp_sm(h_best, 100000, bounds, 1500, 1e-19)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-17:
                c5_best = c5_new
                h_best = h_new
            else:
                break
        except Exception:
            break
        h_best, c5_cd_f = coord_descent(h_best, 2, [0.001, -0.001, 0.0005, -0.0005])
        if c5_cd_f < c5_best:
            c5_best = c5_cd_f

    c5_final = min(compute_c5(h_best), c5_best)
    h_best = fix_constraint(h_best, len(h_best) / 2.0)
    return (h_best, c5_final, len(h_best))
