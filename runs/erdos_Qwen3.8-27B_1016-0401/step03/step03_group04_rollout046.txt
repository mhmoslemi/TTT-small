def compute_c5(h):

    dx = 2.0 / n

    return float(np.max(corr) * dx)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m = j >= shift
        if m.any(): g[m] += (1.0 - h[j[m] - shift])
        m2 = (j + shift) < n
        if m2.any(): g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m = (j + s) < n
        if m.any(): g[m] += (1.0 - h[j[m] + s])
        m2 = j >= s
        if m2.any(): g[m2] -= h[j[m2] - s]
    return g

    lo, hi = -2.0, 3.0

        s = np.clip(y - mid, 0.0, 1.0).sum()
