from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    """Gradient of correlation at a given shift."""
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

def weighted_grad(h, n, lags, weights):
    """Compute gradient of weighted sum of correlations."""
    g = np.zeros(n)
    for lag, w in zip(lags, weights):
        if w > 1e-12:
            g += w * shift_grad(h, n, lag)
    g *= (2.0 / n)
    return g

def backtracking_step(h, g, n, max_iter=30):
    """Backtracking line search with projection."""
    c5_cur = compute_c5(h)
    alpha = 0.1
    best_h = h.copy()
    best_c5 = c5_cur
    for _ in range(max_iter):
        h_try = project_feasible(h - alpha * g, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
        alpha *= 0.7
        if alpha < 1e-8:
            break
    return best_h, best_c5

    h_cur = h.copy()
    c5_best = compute_c5(h_cur)
    h_best = h_cur.copy()
    count = 0

        corr = compute_all_corr(h_cur)

        top_idx = np.argsort(corr)[-K_base:]
        c_top = corr[top_idx]
        c_max = c_top.max()
        c_min_top = c_top.min()
        
        # If spread is small, use more lags
        K = K_base
        if c_max - c_min_top < 0.01:

            top_idx = np.argsort(corr)[-K:]
            c_top = corr[top_idx]

        # Step with adaptive alpha

        if c5_new < c5_best:

def pair_flip_search(h, n, t_end, rng, n_trials=5000):
    """Local search: flip pairs of values to improve C5."""
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    h_best = h_cur.copy()
    c5_best = c5_cur

            break
        # Pick two random indices

        j = rng.integers(0, n)

        # Flip: h[i] -> 1-h[i], h[j] -> 1-h[j]

    """SLSQP with log-sum-exp smoothing."""
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        m = c.max()
        e = np.exp(T * (c - m))
        return (m + np.log(e.sum()) / T) * (2.0 / n)
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        m = c.max()
        e = np.exp(T * (c - m))
        w = e / e.sum()
        grad = np.zeros(n)
        for wi, ci in zip(w, idx):
            if wi > 1e-10:
                grad += wi * shift_grad(x, n, ci - (n - 1))
        return grad * (2.0 / n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            g[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        if m1.any():
            g[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            g[m2] -= h[j[m2] - s]
    return g

def weighted_grad(h, n, lags, weights):
    g = np.zeros(n)
    for lag, w in zip(lags, weights):
        if w > 1e-12:
            g += w * shift_grad(h, n, lag)
    return g * (2.0 / n)

def adaptive_gradient(h, n, t_end, rng, K_base=25, momentum_decay=0.9):
    h_cur = h.copy()
    c5_best = compute_c5(h_cur)
    h_best = h_cur.copy()
    v = np.zeros(n)
    stall = 0
    lr = 0.08
    for it in range(200000):
        if time.time() > t_end:
            break
        corr = compute_all_corr(h_cur)
        top_idx = np.argsort(corr)[-K_base:]
        c_top = corr[top_idx]
        c_max = c_top.max()
        if it % 200 == 0:
            gap = c_max - c_top.min()
            if gap < 0.008:
                K = min(2 * K_base, len(corr))
                top_idx = np.argsort(corr)[-K:]
                c_top = corr[top_idx]
                c_max = c_top.max()
        T = 15.0
        w = np.exp(T * (c_top - c_max))
        w /= w.sum()
        g = weighted_grad(h_cur, n, top_idx - (n - 1), w)
        v = momentum_decay * v - lr * g
        h_new = project_feasible(h_cur + v, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-14:
            h_cur = h_new
            stall = 0
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
        else:
            stall += 1
            if stall > 80:
                lr *= 0.85
                if lr < 1e-5:
                    h_cur = project_feasible(h_best + rng.normal(0, 0.003, n), n)
                    v = np.zeros(n)
                    lr = 0.05
                stall = 0
        if it % 4000 == 0 and it > 0:
            c5_chk = compute_c5(h_cur)
            if c5_chk < c5_best:
                c5_best = c5_chk
                h_best = h_cur.copy()
    return h_best, c5_best

def pair_flip_search(h, n, t_end, rng, n_trials=30000):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    h_best = h_cur.copy()
    c5_best = c5_cur
    corr_cur = compute_all_corr(h_cur)
    top_set = set(np.argsort(corr_cur)[-25:].tolist())
    for trial in range(n_trials):
        if time.time() > t_end:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        h_try = h_cur.copy()
        h_try[i] = 1.0 - h_try[i]
        h_try[j] = 1.0 - h_try[j]
        c5_try = compute_c5(h_try)
        if c5_try < c5_cur - 1e-14:
            h_cur = h_try
            c5_cur = c5_try
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h_try.copy()
    return h_best, c5_best

def slsqp_polish(h, n, T, top_k, maxiter, ftol):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        m = c.max()
        e = np.exp(T * (c - m))
        return (m + np.log(e.sum()) / T) * (2.0 / n)
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        m = c.max()
        e = np.exp(T * (c - m))
        w = e / e.sum()
        grad = np.zeros(n)
        for wi, ci in zip(w, idx):
            if wi > 1e-10:
                grad += wi * shift_grad(x, n, ci - (n - 1))
        return grad * (2.0 / n)
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def fine_gradient(h, n, t_end):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for it in range(50000):
        if time.time() > t_end:

        top_idx = np.argsort(corr)[-12:]
        c_top = corr[top_idx]
        T = 40.0
        w = np.exp(T * (c_top - c_top.max()))
        w /= w.sum()
        g = weighted_grad(h_cur, n, top_idx - (n - 1), w)
        alpha = 0.02 / (1.0 + 0.005 * it)
        h_new = project_feasible(h_cur - alpha * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-15:
            h_cur = h_new
            c5_cur = c5_new
        else:
            alpha *= 0.5
            h_try = project_feasible(h_cur - alpha * g, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-15:
                h_cur = h_try
                c5_cur = c5_try
    return h_cur, c5_cur
