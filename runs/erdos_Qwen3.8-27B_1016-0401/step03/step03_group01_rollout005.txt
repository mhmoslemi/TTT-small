import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def lse_objective(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # log-sum-exp for numerical stability
    mc = np.max(corr)
    val = mc + np.log(np.sum(np.exp(corr - mc))) / beta
    return val * dx

def lse_gradient(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(corr - mc)
    w /= w.sum()
    # gradient of sum w_k * corr_k w.r.t. h
    grad = np.zeros(n)
    center = n - 1
    j = np.arange(n)
    for ki in range(len(corr)):
        wk = w[ki]
        if wk < 1e-12:
            continue
        shift = ki - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += wk * (1.0 - h[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= wk * h[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += wk * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= wk * h[j[m2] - s2]
    grad *= dx
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def lse_obj_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sh = corr - mc
    ex = np.exp(sh)
    tot = ex.sum()
    obj = (mc + np.log(tot) / beta) * dx
    w = ex / tot
    grad = np.zeros(n)
    center = n - 1
    j = np.arange(n)
    nz = np.where(w > 1e-8)[0]
    for ki in nz:
        wk = w[ki]
        s = ki - center
        if s >= 0:
            m1 = j >= s
            grad[m1] += wk * (1.0 - h[j[m1] - s])
            m2 = (j + s) < n
            grad[m2] -= wk * h[j[m2] + s]
        else:
            s2 = -s
            m1 = (j + s2) < n
            grad[m1] += wk * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= wk * h[j[m2] - s2]
    grad *= dx
    return obj, grad

def fista_opt(h_init, n, t_end, beta_init=3.0, beta_max=200.0, n_phase=20000):
    h_y = h_init.copy().astype(float)
    h_z = h_init.copy().astype(float)
    t_mom = 1.0
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    it = 0
    while time.time() < t_end:
        frac = min(1.0, it / n_phase)
        beta = beta_init * (beta_max / beta_init) ** frac
        obj, grad = lse_obj_grad(h_y, n, beta)
        lr = min(2.0, 4.0 * n / (2.0 * max(1.0, beta)))
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        mom = (t_mom - 1.0) / t_new
        h_z = h_y + mom * (h_y - h_z)
        h_y = project_feasible(h_z - lr * grad, n)
        t_mom = t_new
        it += 1
        if it % 2000 == 0:
            c5 = compute_c5(h_y)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_y.copy()
    return h_best, c5_best

def slsqp_opt(h, n, beta, maxiter=3000):
    target = n / 2.0
    def _obj(x):
        return lse_obj_grad(x, n, beta)[0]
    def _jac(x):
        return lse_obj_grad(x, n, beta)[1]
    def _cf(x):
        return x.sum() - target
    res = minimize(_obj, h, method='SLSQP', jac=_jac,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _cf, 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return res.x

def gen_start(n, rng, ptype):
    h = np.zeros(n)
    if ptype == 0:
        i = 0
        v = 1.0
        while i < n:

            h[i:i+bl] = v
            i += bl
            v = 1.0 - v
    elif ptype == 1:

        h[:] = tm
    elif ptype == 2:

    elif ptype == 3:

        h[:k] = 1.0
        h[n//2:n//2+k] = 1.0
        h[n//2+k+1:] = rng.random(n - n//2 - k - 1) < 0.2
    else:

        for _ in range(20):

            elif base.sum() > p / 2:
                base[rng.integers(p)] = 0.0
            else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def _ones_fn(x):
    return np.ones_like(x)

def lse_obj_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    ex = np.exp(corr - mc)
    tot = ex.sum()
    obj = (mc + np.log(tot) / beta) * dx
    w = ex / tot
    grad = np.zeros(n)
    center = n - 1
    j = np.arange(n)
    nz = np.where(w > 1e-8)[0]
    for ki in nz:
        wk = w[ki]
        s = ki - center
        if s >= 0:
            m1 = j >= s
            grad[m1] += wk * (1.0 - h[j[m1] - s])
            m2 = (j + s) < n
            grad[m2] -= wk * h[j[m2] + s]
        else:
            s2 = -s
            m1 = (j + s2) < n
            grad[m1] += wk * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= wk * h[j[m2] - s2]
    grad *= dx
    return obj, grad

    h_y = h_init.copy().astype(float)
    h_z = h_init.copy().astype(float)

    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    it = 0
    while time.time() < t_end:
        frac = min(1.0, it / n_phase)
        beta = beta_init * (beta_max / beta_init) ** frac
        obj, grad = lse_obj_grad(h_y, n, beta)
        lr = min(2.0, 4.0 * n / (2.0 * max(1.0, beta)))
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0

        it += 1

    return h_best, c5_best

def slsqp_opt(h, n, beta, maxiter=3000):
    target = n / 2.0
    h_c = h.copy()
    def _obj(x):
        return lse_obj_grad(x, n, beta)[0]
    def _jac(x):
        return lse_obj_grad(x, n, beta)[1]
    def _cf(x):
        return x.sum() - target
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _ones_fn}]
    res = minimize(_obj, h_c, method='SLSQP', jac=_jac,
                   bounds=[(0.0, 1.0)] * n, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return res.x
