                a, b = i, i + shift
                diff = h_new[b] - h_new[a]
                adj = alpha * 0.5 * diff
                h_new[a] += adj
                h_new[b] -= adj
        else:
            s2 = -shift
            for i in range(n - s2):

                b = i
                diff = h_new[b] - h_new[a]
                adj = alpha * 0.5 * diff
                h_new[a] += adj
                h_new[b] -= adj
    
    return np.clip(h_new, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

_OPT_N = 0
_OPT_T = 1.0
_OPT_K = 5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n * 0.5
    lo, hi = -3.0, 4.0
    mid = 0.5
    for _ in range(120):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-16:
            break
    return np.clip(y - mid, 0.0, 1.0)

def _grad_shift(h, shift):
    n = len(h)
    j = np.arange(n)
    g = np.zeros(n)
    if 0 <= shift < n:
        m1 = j >= shift
        g[m1] += (1.0 - h[j[m1] - shift])
        m2 = (j + shift) < n
        g[m2] -= h[j[m2] + shift]
    elif -n < shift < 0:
        s2 = -shift
        m1 = (j + s2) < n
        g[m1] += (1.0 - h[j[m1] + s2])
        m2 = j >= s2
        g[m2] -= h[j[m2] - s2]
    return g

def _sm_obj(x, T, k):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = topk.max()
    sh = T * (topk - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    obj = (mc + np.log(e.sum()) / T) * dx
    grad = np.zeros(n)
    center = n - 1
    for oi in range(mk):
        w = sw[oi]
        if w > 1e-12:
            shift = int(idx[oi] - center)
            grad += w * _grad_shift(x, shift)
    grad *= dx
    return obj, grad

def _obj_fn(x):
    return _sm_obj(x, _OPT_T, _OPT_K)[0]

def _jac_fn(x):
    return _sm_obj(x, _OPT_T, _OPT_K)[1]

def _cf(x):
    return x.sum() - _OPT_N * 0.5

def _cj(x):
    return np.ones(_OPT_N)

def slsqpopt(h, T, k, maxiter, ftol):
    global _OPT_N, _OPT_T, _OPT_K
    n = len(h)
    _OPT_N, _OPT_T, _OPT_K = n, T, k
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_obj_fn, h.copy(), method='SLSQP', jac=_jac_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def fista_exact(h, n, t_end, lr0=0.015, n_active=2):
    dx = 2.0 / n
    hy = h.copy()
    hz = h.copy()
    t = 1.0
    lr = lr0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stagnant = 0
    
    for it in range(500000):
        if time.time() > t_end:
            break
        corr = np.correlate(hy, 1.0 - hy, mode="full")
        na = min(n_active, len(corr))
        idx_na = np.argpartition(corr, -na)[-na:]
        grad = np.zeros(n)
        center = n - 1
        for oi in idx_na:
            shift = int(oi - center)
            grad += _grad_shift(hy, shift)
        grad *= dx / na
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) * 0.5
        beta = (t - 1.0) / t_new
        hz = hy + beta * (hy - hz)
        hy = hz - lr * grad
        hy = project_feasible(hy, n)
        t = t_new
        
        if (it + 1) % 5000 == 0:
            c5 = compute_c5(hy)
            if c5 < best_c5 - 1e-8:
                stagnant = 0
            else:
                stagnant += 1
            if c5 < best_c5:
                best_c5 = c5
                best_h = hy.copy()
            if stagnant > 10:
                lr *= 0.5
                stagnant = 0
    
    return best_h, best_c5

def lag_repair(h, n, alpha=0.08, n_lags=3):
    corr = np.correlate(h, 1.0 - h, mode="full")
    na = min(n_lags, len(corr))
    idx_na = np.argpartition(corr, -na)[-na:]
    h_new = h.copy()
    center = n - 1
    for oi in idx_na:
        shift = int(oi - center)
        if 0 < shift < n:
            diff = h_new[shift:] - h_new[:n - shift]
            h_new[:n - shift] += alpha * 0.5 * diff
            h_new[shift:] -= alpha * 0.5 * diff
        elif -n < shift < 0:
            s2 = -shift
            diff = h_new[:n - s2] - h_new[s2:]
            h_new[s2:] += alpha * 0.5 * diff
            h_new[:n - s2] -= alpha * 0.5 * diff
    return np.clip(h_new, 0.0, 1.0)

        h_p[s2:s2 + b] = tmp
    else:
        h_p = np.roll(h_p, rng.integers(-3, 4))
    return project_feasible(h_p, n)
