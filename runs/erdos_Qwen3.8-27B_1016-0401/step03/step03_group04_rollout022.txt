def lag_grad(h, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        ia = np.arange(m)
        ib = ia + lag
        grad[ia] += (1.0 - h[ib])
        grad[ib] -= h[ia]
    else:
        m = n + lag
        ia = np.arange(-lag, n)
        ib = np.arange(m)
        grad[ia] += (1.0 - h[ib])
        grad[ib] -= h[ia]
    return grad

def sm_val_grad(x, n, T, K):

    corr = np.correlate(x, 1.0 - x, mode="full")
    center = n - 1
    if K < len(corr):
        topk = np.argpartition(corr, -K)[-K:]
    else:
        topk = np.arange(len(corr))
    c = corr[topk]
    mc = c.max()
    s = T * (c - mc)
    s -= s.max()
    e = np.exp(s)
    w = e / e.sum()
    val = (mc + np.log(e.sum()) / T) * dx

    return val, grad * dx

        res = minimize(_obj, h.copy(), method='SLSQP', jac=_jac,
                       args=(n, T, K, tlim), bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if not _FLAG[0]:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def outer_approx(h_start, n, tlim, base_K=30, max_iters=500):
    dx = 2.0 / n
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    center = n - 1
    stalls = 0
    for it in range(max_iters):
        if time.time() > tlim:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        K_cur = min(base_K + it * 2, 2 * n - 1)
        topk = np.argpartition(corr, -K_cur)[-K_cur:]
        h_var = cvx.Variable(n)
        t_var = cvx.Variable()
        cons = [h_var >= 0, h_var <= 1, cvx.sum(h_var) == n / 2.0]
        for k_idx in topk:
            lag = k_idx - center
            f_cur = corr[k_idx] * dx
            g = lag_grad(h_cur, n, lag) * dx
            cons.append(f_cur + g @ (h_var - h_cur) <= t_var)
        prob = cvx.Problem(cvx.Minimize(t_var), cons)
        solved = False
        for solver in [cvx.CLARABEL, cvx.ECOS, cvx.SCS]:
            try:
                prob.solve(solver=solver)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
        if not solved or h_var.value is None:
            break
        h_new = project_feasible(np.asarray(h_var.value).flatten(), n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new
            stalls = 0
        else:
            stalls += 1
            if stalls > 30:
                break
            h_cur = 0.7 * h_cur + 0.3 * h_new
    return best_h, best_c5

def subgrad_direct(h, n, tlim, max_iters=80000):
    dx = 2.0 / n
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    center = n - 1
    for it in range(max_iters):
        if time.time() > tlim:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        topk = np.argpartition(corr, -3)[-3:]
        c_top = corr[topk]
        mc = c_top.max()
        T = 30.0
        s = T * (c_top - mc)
        s -= s.max()
        e = np.exp(s)
        w = e / e.sum()
        g = np.zeros(n)
        for i in range(3):
            if w[i] > 1e-12:
                g += w[i] * lag_grad(h_cur, n, topk[i] - center)
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        f_cur = compute_c5(h_cur)
        alpha = 5.0 / (gn * (gn + 1e-8) * (1.0 + 0.0005 * it))
        h_trial = h_cur - alpha * g
        h_trial = project_feasible(h_trial, n)
        f_trial = compute_c5(h_trial)
        if f_trial > f_cur:
            alpha *= 0.3
            h_trial = project_feasible(h_cur - alpha * g, n)
        h_cur = h_trial
        if (it + 1) % 2000 == 0:
            c5_now = compute_c5(h_cur)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h_cur.copy()
    c5_end = compute_c5(h_cur)
    if c5_end < best_c5:
        best_c5 = c5_end
        best_h = h_cur.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n0 = 200
        h_best = np.zeros(n0)
        h_best[::2] = 1.0
        h_best = project_feasible(h_best, n0)

    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B with T schedule
    sched = [(32, 3000), (16, 4000), (8, 5000), (4, 6000), (2, 8000), (1, 10000)]
    for T, mi in sched:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        h_new = do_lbfgsb(h_best, n, T, 80, mi, 1e-15, t0 + rem - 40)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Outer approximation (LP-based)
    for K_val in [20, 40, 60, 100]:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        h_oo, c5_oo = outer_approx(h_best, n, t0 + rem - 40, base_K=K_val, max_iters=300)
        if c5_oo < c5_best:
            c5_best = c5_oo
            h_best = h_oo

    # Phase 3: SLSQP polish
    for T in [2, 1, 0.5]:
        rem = budget_s - (time.time() - t0)
        if rem < 60:
            break
        h_new = do_slsqp(h_best, n, T, 50, 4000, 1e-16, t0 + rem - 25)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Direct subgradient on actual max
    rem = budget_s - (time.time() - t0)
    if rem > 100:
        h_sg, c5_sg = subgrad_direct(h_best, n, t0 + rem - 50, 60000)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase 5: Outer approximation round 2 (post-optimization)
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        h_oo2, c5_oo2 = outer_approx(h_best, n, t0 + rem - 40, base_K=80, max_iters=400)
        if c5_oo2 < c5_best:
            c5_best = c5_oo2
            h_best = h_oo2

    # Phase 6: Final L-BFGS-B + SLSQP
    for T in [1, 0.5]:
        rem = budget_s - (time.time() - t0)
        if rem < 50:
            break
        h_new = do_lbfgsb(h_best, n, T, 50, 15000, 1e-16, t0 + rem - 25)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_new = do_slsqp(h_best, n, 1, 50, 6000, 1e-17, t0 + rem - 10)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 7: Small diverse restarts
    n_rest = 30
    for pi in range(n_rest):
        rem = budget_s - (time.time() - t0)
        if rem < 25:
            break
        if pi % 3 == 0:
            sig = 0.002 + 0.003 * (pi % 3)
            h_p = h_best + rng.normal(0, sig, n)
        elif pi % 3 == 1:
            x_arr = np.arange(n)
            h_p = h_best + 0.015 * np.sin(2 * np.pi * rng.uniform(1.5, 5) * x_arr / n)
        else:
            h_p = h_best.copy()
            bs = max(1, n // 12)
            bi = int(rng.integers(0, n // bs))
            h_p[bi * bs:(bi + 1) * bs] += rng.uniform(-0.04, 0.04)
        h_p = project_feasible(h_p, n)
        h_new = do_lbfgsb(h_p, n, 2, 80, 2500, 1e-14, t0 + rem - 10)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        rem = budget_s - (time.time() - t0)
        if rem < 40:
            break
        h_oo_r, c5_oo_r = outer_approx(h_new, n, t0 + rem - 15, base_K=40, max_iters=100)
        if c5_oo_r < c5_best:
            c5_best = c5_oo_r
            h_best = h_oo_r

    # Phase 8: If time remains, try higher resolution
    rem = budget_s - (time.time() - t0)
    if rem > 250:
        for n_try in [n + 40, n + 80]:
            rem = budget_s - (time.time() - t0)
            if rem < 230:
                break
            x_old = np.linspace(0, 2.0, n, endpoint=False)
            x_new = np.linspace(0, 2.0, n_try, endpoint=False)
            h_int = np.interp(x_new, x_old, h_best)
            h_int = project_feasible(h_int, n_try)
            c5_i = compute_c5(h_int)
            if c5_i > c5_best + 0.01:
                continue
            h_try = h_int
            for T, mi in [(8, 3000), (2, 5000)]:
                rem = budget_s - (time.time() - t0)
                if rem < 100:
                    break
                h_try = do_lbfgsb(h_try, n_try, T, 80, mi, 1e-15, t0 + rem - 60)
            rem = budget_s - (time.time() - t0)
            if rem > 80:
                h_try, c5_oo_hi = outer_approx(h_try, n_try, t0 + rem - 40, base_K=50, max_iters=200)
            else:
                c5_oo_hi = compute_c5(h_try)
            if c5_oo_hi < c5_best:
                c5_best = c5_oo_hi
                h_best = h_try
                n = n_try

    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    if c5_final < c5_best:
        c5_best = c5_final
    return (h_best, c5_best, n)
