import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target):
    def sum_at(lam):
        return np.sum(np.clip(h + lam, 0.0, 1.0))
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if sum_at(mid) < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h + lam, 0.0, 1.0)

def sm_obj(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * (2.0 / n)

def sm_grad(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    shifted = T * (corr - mc)
    e = np.exp(shifted)
    sw = e / np.sum(e)
    center = n - 1
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for o in range(-(n - 1), n):
        w = sw[center + o]
        if w < 1e-14:
            continue
        if o == 0:
            grad += w * (1.0 - 2.0 * x)
        elif o > 0:
            grad[o:] += w * (1.0 - x[:n - o])
            grad[:n - o] += -w * x[o:]
        else:
            p = -o
            grad[:n - p] += w * (1.0 - x[p:])
            grad[p:] += -w * x[:n - p]
    return grad * dx_val

def subgrad_nesterov(h, target, n_iters, T_list, step, momentum, t0, tlimit):
    n = len(h)
    h_cur = h.astype(float).copy()
    v = np.zeros(n)
    best = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for it in range(n_iters):
        if time.time() > tlimit:
            break
        T = T_list[min(it // 100, len(T_list) - 1)]
        y = h_cur + v
        g = sm_grad(y, T)
        v = momentum * v - step * g
        h_new = project(h_cur + v, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best = h_new.copy()
        h_cur = h_new
    return best, best_c5

def lag_obj(x, lag_idx):
    n = len(x)
    return float(np.correlate(x, 1.0 - x, mode="full")[lag_idx] * (2.0 / n))

def lag_grad(x, lag_idx):
    n = len(x)
    o = lag_idx - (n - 1)
    dx_val = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * x
    elif o > 0:
        g[o:] += 1.0 - x[:n - o]
        g[:n - o] += -x[o:]
    else:
        p = -o
        g[:n - p] += 1.0 - x[p:]
        g[p:] += -x[:n - p]
    return g * dx_val

def _slsqp_lag(h, lag_idx, target, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    def _ef(x):
        return x.sum() - target
    def _ej(x):
        return np.ones(n)
    cons = [dict(type='eq', fun=_ef, jac=_ej)]
    res = minimize(lag_obj, h, args=(lag_idx,), method='SLSQP',
                   jac=lag_grad, bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return project(res.x, target)

def binary_push(h, target, thr):
    n = len(h)
    c5_cur = compute_c5(h)
    for j in range(n):
        trial = h.copy()
        if h[j] > 1.0 - thr:
            trial[j] = 1.0
        elif h[j] < thr:
            trial[j] = 0.0
        else:
            continue
        trial = project(trial, target)
        c5_new = compute_c5(trial)
        if c5_new < c5_cur - 1e-15:
            h = trial
            c5_cur = c5_new
    return h, c5_cur

def coord_descent(h, target, n_sweeps):
    n = len(h)
    c5_cur = compute_c5(h)
    dts = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            for dt in dts:
                trial = h.copy()
                trial[j] = np.clip(trial[j] + dt, 0.0, 1.0)
                trial = project(trial, target)
                c5_new = compute_c5(trial)
                if c5_new < c5_cur - 1e-15:
                    h = trial
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur
