        if abs(diff) < 1e-13:
            break
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:

                if add <= 0: continue
                h[i] += add; diff -= add
            else:
                sub = min(-diff, h[i])
                if sub <= 0: continue

def proj_constraint(h, target):
    """Project onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(h, 0, 1).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for i in order:
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add; diff -= add
        else:
            sub = min(-diff, h[i])
            h[i] -= sub; diff += sub
        if abs(diff) < 1e-14: break
    return h

def grad_max_lag(h):
    """Gradient of max_k correlate(h,1-h)[k]"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    o = k_star - (n - 1)  # actual offset
    g = np.zeros(n)
    if o == 0:
        g[:] = 1.0 - 2.0 * h
    elif o > 0:
        g[:n-o] += 1.0 - h[o:]
        g[o:] += -h[:n-o]
    else:

        g[p:] += 1.0 - h[:n-p]
        g[:n-p] += -h[p:]

    n = len(h)
    h_curr = h.copy()
    h_prev = h.copy()
    t_k = 1.0
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr)

            c5_cur = c5_new
            h = h_new
        else:
            break
    # FISTA
    if time.time() < end:
        for alpha in [0.02, 0.05, 0.01]:
            if time.time() > end: break
            h_f, c5_f = fista(h, target, n_iter=3000, alpha=alpha)
            if c5_f < c5_cur:
                c5_cur = c5_f
                h = h_f
    # Binary + coord
    if time.time() < end:
        h, c5_b = binary_flip(h, n_rounds=8)
        if c5_b < c5_cur: c5_cur = c5_b
        h, c5_cd = coord_descent(h, n_sweeps=4)
        if c5_cd < c5_cur: c5_cur = c5_cd
    # Final SLSQP
    if time.time() < end:
        for _ in range(20):
            if time.time() > end: break
            h_new = slsqp_worst(h, target, maxiter=5000, ftol=1e-22)
            h_new = fix_sum(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h = h_new
            else:
                break
    h, c5_cd2 = coord_descent(h, n_sweeps=3)
    if c5_cd2 < c5_cur: c5_cur = c5_cd2
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.zeros(340)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = fix_sum(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: Intensive SLSQP on worst lag (30%)
    p1_end = t0 + budget_s * 0.30
    for it in range(50):
        if time.time() > p1_end: break
        h_new = slsqp_worst(h_best, target, maxiter=4000, ftol=1e-21)
        h_new = fix_sum(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-15:
            c5_best = c5_new
            h_best = h_new
        else:
            if it > 5: break

    # Phase 2: FISTA with varying alpha (15%)
    p2_end = t0 + budget_s * 0.45
    for alpha in [0.03, 0.015, 0.05, 0.01, 0.08]:
        if time.time() > p2_end: break
        h_f, c5_f = fista(h_best, target, n_iter=5000, alpha=alpha)
        if c5_f < c5_best - 1e-15:
            c5_best = c5_f
            h_best = h_f

    # Phase 3: Binary flip + coord descent (10%)
    p3_end = t0 + budget_s * 0.55
    for _ in range(30):
        if time.time() > p3_end: break
        h_best, c5_b = binary_flip(h_best, n_rounds=6)
        if c5_b < c5_best: c5_best = c5_b
        h_best, c5_cd = coord_descent(h_best, n_sweeps=3)
        if c5_cd < c5_best: c5_best = c5_cd

    # Phase 4: Multi-n exploration (25%)
    p4_end = t0 + budget_s * 0.80
    n_candidates = [n]
    for dn in [-8, -4, 4, 8, 12, -12, 16, -16, 20, -20]:
        n_candidates.append(n + dn)
    n_candidates = sorted(set(nn for nn in n_candidates if nn >= 200))
    for n_alt in n_candidates:
        if time.time() > p4_end: break
        if n_alt == n: continue
        h_alt = resample(h_best, n_alt)
        target_alt = n_alt / 2.0
        c5_alt = compute_c5(h_alt)
        # Quick optimize
        for _ in range(15):
            if time.time() > p4_end: break
            h_new = slsqp_worst(h_alt, target_alt, maxiter=2000, ftol=1e-19)
            h_new = fix_sum(h_new, target_alt)
            c5_new = compute_c5(h_new)
            if c5_new < c5_alt - 1e-15:
                c5_alt = c5_new
                h_alt = h_new
            else:
                break
        h_alt, c5_b = binary_flip(h_alt, n_rounds=5)
        if c5_b < c5_alt: c5_alt = c5_b
        h_alt, c5_cd = coord_descent(h_alt, n_sweeps=3)
        if c5_cd < c5_alt: c5_alt = c5_cd
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n = n_alt
            target = n / 2.0

    # Phase 5: Final polish (15%)
    p5_end = t0 + budget_s - 3.0
    n = len(h_best)
    target = n / 2.0
    for _ in range(30):
        if time.time() > p5_end: break
        h_new = slsqp_worst(h_best, target, maxiter=6000, ftol=1e-23)
        h_new = fix_sum(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-15:
            c5_best = c5_new
            h_best = h_new
        else:
            break
    h_best, c5_cd = coord_descent(h_best, n_sweeps=5)
    if c5_cd < c5_best: c5_best = c5_cd
    h_best, c5_b = binary_flip(h_best, n_rounds=4)
    if c5_b < c5_best: c5_best = c5_b

    # Final SLSQP polish
    if time.time() < p5_end:
        h_new = slsqp_worst(h_best, target, maxiter=8000, ftol=1e-24)
        h_new = fix_sum(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    h_best = fix_sum(h_best, len(h_best) / 2.0)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
