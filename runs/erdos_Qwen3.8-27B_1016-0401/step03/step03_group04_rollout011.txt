            #   => k ranges from n-1-i to n-1+i

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = -2.0
    hi = 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5*(lo+hi), 0.0, 1.0)

def smooth_obj_grad(x, n, T, K):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    center = n - 1
    nl = len(corr)
    if K < nl:
        idx_k = np.argpartition(corr, -K)[-K:]
        c_sel = corr[idx_k]
    else:
        idx_k = np.arange(nl)
        c_sel = corr
    mc = c_sel.max()
    sh = T * (c_sel - mc)
    e = np.exp(sh)
    val = (mc + np.log(e.sum()) / T) * dx
    sw = e / e.sum()
    grad = np.zeros(n)
    for ii in range(len(sw)):
        w = sw[ii]
        if w < 1e-9:
            continue
        s = idx_k[ii] - center
        i_arr = np.arange(n)
        idx1 = i_arr + s
        v1 = (idx1 >= 0) & (idx1 < n)
        if v1.any():
            grad[v1] += w * (1.0 - x[idx1[v1]])
        idx2 = i_arr - s
        v2 = (idx2 >= 0) & (idx2 < n)
        if v2.any():
            grad[v2] -= w * x[idx2[v2]]
    return val, grad * dx

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        if time.time() > tlim:
            return smooth_obj_grad(x, n, T, K)[0]
        return smooth_obj_grad(x, n, T, K)[0]
    def jac(x):
        return smooth_obj_grad(x, n, T, K)[1]
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jac, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': tol, 'maxls': 20, 'maxfun': maxiter*2})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def coord_descend(h, n, max_sweeps=200, t0=0, tlim=999999):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() > tlim:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Alternate direction

        else:

        for i in indices:

            hi1 = min(2*n-2, n-1+i)
            lo2 = max(0, i-n+1)
            hi2 = min(2*n-2, i+n-1)
            g = np.zeros(2*n-1)
            if lo1 <= hi1:
                ks = np.arange(lo1, hi1+1)
                idx = i + ks - n + 1
                g[lo1:hi1+1] += 1.0 - h[idx]
            if lo2 <= hi2:
                ks = np.arange(lo2, hi2+1)
                idx = i - ks + n - 1
                g[lo2:hi2+1] -= h[idx]
            lo_t = -h[i]
            hi_t = 1.0 - h[i]
            if hi_t - lo_t < 1e-12:

def perturb(h, n, rng, mode=0):
    if mode == 0:
        return h + rng.normal(0, 0.01, n)
    elif mode == 1:
        x = np.arange(n, dtype=float)
        p = np.zeros(n)
        for _ in range(4):
            f = rng.uniform(1, 6)
            ph = rng.uniform(0, 2*np.pi)
            a = rng.uniform(0.01, 0.05)
            p += a * np.sin(2*np.pi*f*x/n + ph)
        return h + p
    else:
        # Block flip
        hp = h.copy()
        bs = max(1, n//10)
        bi = int(rng.integers(0, n//bs))
        hp[bi*bs:(bi+1)*bs] = 1.0 - hp[bi*bs:(bi+1)*bs]
        return hp + rng.normal(0, 0.003, n)
