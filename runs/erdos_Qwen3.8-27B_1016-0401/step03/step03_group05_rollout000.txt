    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def soft_max_grad(corr, T, n, center, thresh=1e-9):
    mc = np.max(corr)
    ea = T * (corr - mc)
    w = np.exp(ea - ea.max())
    w_sum = w.sum()
    sw = w / w_sum
    n_corr = len(corr)
    n = len(corr)  # not needed, using n from closure
    grad = np.zeros(n)
    for oi in range(n_corr):
        if sw[oi] < thresh:
            continue
        grad += sw[oi] * shift_grad(corr, 0, 0)  # placeholder, replaced below
    return grad, sw

def lbfgs_smooth(h_init, n, T, mu, maxiter, t0, time_limit):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        ea = T * (corr - mc)
        smax = ea.max()
        lse = mc + (smax + np.log(np.exp(ea - smax).sum())) / T
        return lse * dx + mu * (x.sum() - target) ** 2

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        ea = T * (corr - mc)
        w = np.exp(ea - ea.max())
        w_sum = w.sum()
        sw = w / w_sum
        x_cur = x
        grad = np.zeros(n)
        for oi in range(n_corr):
            if sw[oi] < 1e-9:
                continue
            grad += sw[oi] * shift_grad(x_cur, n, oi - center)
        grad *= dx
        grad += 2.0 * mu * (x.sum() - target)
        return grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_init, method='L-BFGS-B', jac=jac,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-17, 'gtol': 1e-9})
        return res.x
    except Exception:
        return h_init.copy()

def nesterov_polish(h_init, n, t0, time_limit, n_iters=400000):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    h = h_init.copy()
    h_mom = h_init.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    t_end = t0 + time_limit

    for it in range(n_iters):
        if time.time() > t_end:
            break
        frac = min(it / n_iters, 1.0)
        K = 1 + int(4 * frac * frac)

        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = h + beta * (h - h_mom)
        h_y = np.clip(h_y, 0.0, 1.0)

        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)

        if K <= 1:
            idx = int(np.argmax(corr))
            g = shift_grad(h_y, n, idx - center) * dx
        else:
            top_idx = np.argpartition(corr, -K)[-K:]
            top_vals = corr[top_idx]
            w = (top_vals / max(mc, 1e-15)) ** 10
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.ones(K) / K
            else:
                w /= w_sum
            g = np.zeros(n)
            for i in range(K):
                g += w[i] * shift_grad(h_y, n, top_idx[i] - center)
            g *= dx

        alpha = 0.4 / (1.0 + 0.003 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_mom = h
        h = h_new

        if (it + 1) % 10000 == 0:
            c = compute_c5(h)
            if c < c5_best:
                c5_best = c
                h_best = h.copy()
    return h_best, c5_best

def structured_perturb(h, n, rng, pi):
    mode = pi % 5
    x = np.arange(n) * 2.0 / n
    if mode == 0:
        f = 0.5 + 0.5 * (pi % 4)
        h_p = h + 0.015 * np.sin(2 * np.pi * f * x + rng.uniform(0, 6.28))
    elif mode == 1:
        f = 0.2 + 0.15 * (pi % 6)
        h_p = h + 0.01 * np.cos(4 * np.pi * f * x) * np.sin(2 * np.pi * f * x)
    elif mode == 2:
        block = max(n // 9, 10)
        h_p = h.copy()
        for b in range(0, n, block):
            h_p[b:b+block] += rng.uniform(-0.02, 0.02)
    elif mode == 3:
        h_p = h + rng.normal(0, 0.005 + 0.005 * (pi % 3), n)
    else:
        sig = 0.003 * (1 + pi % 4)
        h_p = h + rng.normal(0, sig, n)
    h_p = project_feasible(h_p, n)
    return h_p

def upsample_h(h_old, n_old, n_new):
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B with decreasing T (fast local optimization)
    for T in [32, 16, 8, 4]:
        if time.time() - t0 > budget_s - 500:
            break
        mu = 10.0
        maxit = 4000 if T >= 16 else 6000
        h_new = lbfgs_smooth(h_best, n, T, mu, maxit, t0, 120)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        # Reduce penalty and do another pass
        if T >= 8 and time.time() - t0 < budget_s - 400:
            h_new2 = lbfgs_smooth(h_best, n, T, 1.0, 3000, t0, 80)
            h_new2 = project_feasible(h_new2, n)
            c5 = compute_c5(h_new2)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new2

    # Phase 2: Nesterov fine polish (long, focused)
    time_left = budget_s - (time.time() - t0)
    nest_budget = min(time_left - 250, 400)
    if nest_budget > 50:
        h_sg, c5_sg = nesterov_polish(h_best, n, t0, nest_budget, 500000)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg
        h_best = project_feasible(h_best, n)

    # Phase 3: L-BFGS-B fine polish with very small T
    for T in [2.0, 1.0]:
        if time.time() - t0 > budget_s - 250:
            break
        h_new = lbfgs_smooth(h_best, n, T, 50.0, 5000, t0, 80)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Multi-restart with perturbations + L-BFGS-B
    for pi in range(25):
        if time.time() - t0 > budget_s - 150:
            break
        h_p = structured_perturb(h_best, n, rng, pi)
        h_new = lbfgs_smooth(h_p, n, 8.0, 10.0, 3000, t0, 30)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Try different resolutions
    elapsed = time.time() - t0
    if elapsed < budget_s - 120:
        for n2 in [200, 240, n + 40, n + 80]:
            if n2 < 80 or n2 > 600:
                continue
            if time.time() - t0 > budget_s - 60:
                break
            h_alt = upsample_h(h_best, n, n2)
            h_alt = project_feasible(h_alt, n2)
            c5_alt = compute_c5(h_alt)
            for T2 in [16, 4]:
                if time.time() - t0 > budget_s - 40:
                    break
                h_ne = lbfgs_smooth(h_alt, n2, T2, 10.0, 4000, t0, 50)
                h_ne = project_feasible(h_ne, n2)
                c5_alt = compute_c5(h_ne)
                if c5_alt < c5_best:
                    c5_best = c5_alt
                    h_best = h_ne.copy()
                    n = n2

    # Phase 6: Final Nesterov micro-polish
    remaining = budget_s - (time.time() - t0) - 10
    if remaining > 30:
        h_final, c5_fin = nesterov_polish(h_best, n, t0, remaining, 200000)
        if c5_fin < c5_best:
            c5_best = c5_fin
            h_best = h_final

    # Final projection and evaluation
    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    c5_final = min(c5_final, c5_best)
    return (h_best, c5_final, n)
