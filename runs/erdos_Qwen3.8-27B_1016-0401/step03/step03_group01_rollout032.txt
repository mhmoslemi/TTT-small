    dx = 2.0 / n

def slsqp_opt(h, n, T, k, maxiter, ftol):
    target = n / 2.0
    def obj(x):
        return topk_obj_grad(x, n, T, k)[0]
    def jac(x):
        return topk_obj_grad(x, n, T, k)[1]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def fista_opt(h_init, n, t_end, lr0=0.015, topk=5):
    dx = 2.0 / n
    h_y = h_init.copy()
    h_z = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    t_mom = 1.0
    lr = lr0
    j = np.arange(n)
    center = n - 1
    it = 0
    while time.time() < t_end:
        if it % 1000 == 0:
            c5 = compute_c5(h_y)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_y.copy()
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        kk = min(topk, len(corr))
        idx = np.argpartition(corr, -kk)[-kk:]
        grad = np.zeros(n)
        for oi in idx:
            shift = oi - center
            if shift >= 0:
                m1 = j >= shift
                grad[m1] += (1.0 - h_y[j[m1] - shift])
                m2 = (j + shift) < n
                grad[m2] -= h_y[j[m2] + shift]
            else:
                s2 = -shift
                m1 = (j + s2) < n
                grad[m1] += (1.0 - h_y[j[m1] + s2])
                m2 = j >= s2
                grad[m2] -= h_y[j[m2] - s2]
        grad *= dx / kk
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        h_z = h_y + beta * (h_y - h_z)
        h_y = h_z - lr * grad
        h_y = project_feasible(h_y, n)
        t_mom = t_new
        lr *= 0.99995
        it += 1
    c5 = compute_c5(h_y)
    if c5 < c5_best:
        c5_best = c5
        h_best = h_y.copy()
    return h_best, c5_best

def shift_focused_perturb(h, n, rng):
    """Perturb specifically to reduce worst shifts."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    idx = np.argsort(corr)[-3:]
    h_p = h.copy()
    j = np.arange(n)
    for oi in idx:
        shift = oi - center
        # Find where h and (1-h shifted) overlap most, reduce there
        if shift >= 0:
            m1 = j >= shift
            overlap = h[j[m1]] * (1.0 - h[j[m1] - shift])
        else:
            s2 = -shift
            m1 = (j + s2) < n
            overlap = h[j[m1] + s2] * (1.0 - h[j[m1] - s2])
        # Perturb the top overlap locations
        top_idx = np.argsort(overlap)[-max(3, n // 20):]
        amp = rng.uniform(0.02, 0.06)
        positions = j[m1[top_idx]] if shift >= 0 else j[m1[top_idx]] + s2
        for pos in positions[:10]:
            if 0 <= pos < n:
                h_p[pos] -= amp
    # Also add small random noise
    h_p += rng.uniform(0.002, 0.008) * rng.normal(0, 1, n)
    return h_p

def structured_init(n, rng):
    x = np.arange(n) * 2.0 / n
    stype = rng.integers(0, 4)
    if stype == 0:
        period = rng.integers(6, max(7, n // 15))
        h = np.zeros(n)
        for i in range(n):
            if (i // period) % 2 == 0:
                h[i] = 1.0
        h += rng.uniform(-0.15, 0.15, n)
    elif stype == 1:
        f = rng.uniform(1.0, n / 4.0)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
    elif stype == 2:
        f1 = rng.uniform(1, n / 8)
        f2 = f1 * rng.uniform(1.5, 3.0)
        ph1, ph2 = rng.uniform(0, 2*np.pi, 2)
        h = 0.5 + 0.25 * np.sin(2*np.pi*f1*x + ph1) + 0.2 * np.sin(2*np.pi*f2*x + ph2)
    else:
        h = rng.random(n)
        kern = np.ones(7) / 7.0
        h = np.convolve(h, kern, mode='same')
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP cascade on main candidate (n=280)
    t_p1 = t0 + budget_s * 0.30
    sched1 = [(256, 40), (64, 20), (16, 8), (4, 4), (1.0, 2), (0.3, 1), (0.1, 1)]
    for i, (T, k) in enumerate(sched1):
        if time.time() > t_p1 - 5:
            break
        remaining = t_p1 - time.time()
        mi = int(min(8000, max(500, remaining * 0.5 / (len(sched1) - 1 - i) + 500)))
        h_new = slsqp_opt(h_best, n, T, k, mi, 1e-19)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: FISTA with top-5 gradient
    t_p2 = t0 + budget_s * 0.45
    if time.time() < t_p2:
        h_f, c5_f = fista_opt(h_best, n, t_p2, lr0=0.012, topk=5)
        h_f = project_feasible(h_f, n)
        c5_f = compute_c5(h_f)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f

    # Phase 3: SLSQP polish after FISTA
    t_p3 = t0 + budget_s * 0.55
    for T, k in [(1.0, 2), (0.3, 1), (0.1, 1)]:
        if time.time() > t_p3 - 5:
            break
        h_new = slsqp_opt(h_best, n, T, k, 5000, 1e-19)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Try n=140 (coarser resolution might give lower C5)
    t_p4 = t0 + budget_s * 0.70
    n_alt = n // 2
    if n_alt >= 60 and time.time() < t_p4:
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = project_feasible(h_alt, n_alt)
        c5_alt = compute_c5(h_alt)
        if time.time() < t_p4:
            for T, k in [(16, 10), (4, 4), (1, 2), (0.3, 1)]:
                if time.time() > t_p4 - 5:
                    break
                h_new = slsqp_opt(h_alt, n_alt, T, k, 4000, 1e-18)
                h_new = project_feasible(h_new, n_alt)
                c5 = compute_c5(h_new)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = h_new
                    n = n_alt

    # Phase 5: Shift-focused perturbations + local opt
    t_p5 = t0 + budget_s * 0.90
    n_perts = max(2, int((t_p5 - time.time()) / 12))
    for pi in range(n_perts):
        if time.time() > t_p5 - 5:
            break
        h_p = shift_focused_perturb(h_best, n, rng)
        h_p = project_feasible(h_p, n)
        # Quick SLSQP
        h_new = slsqp_opt(h_p, n, 8, 4, 1500, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
            # Follow-up with tighter T
            if time.time() < t_p5 - 5:
                h_new2 = slsqp_opt(h_new, n, 0.5, 1, 3000, 1e-18)
                h_new2 = project_feasible(h_new2, n)
                c5_2 = compute_c5(h_new2)
                if c5_2 < c5_best:
                    c5_best = c5_2
                    h_best = h_new2
        # Also try small random perturbation
        elif time.time() < t_p5 - 8:
            h_r = h_best + rng.uniform(0.001, 0.004) * rng.normal(0, 1, n)
            h_r = project_feasible(h_r, n)
            h_new_r = slsqp_opt(h_r, n, 2, 2, 2000, 1e-17)
            h_new_r = project_feasible(h_new_r, n)
            c5_r = compute_c5(h_new_r)
            if c5_r < c5_best:
                c5_best = c5_r
                h_best = h_new_r

    # Phase 6: Structured starts at current n (different topologies)
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        for si in range(min(3, int(rem / 20))):
            if time.time() > t0 + budget_s - 15:
                break
            h_s = structured_init(n, rng)
            # Blend with current best
            h_s = 0.5 * h_s + 0.5 * h_best
            h_s = project_feasible(h_s, n)
            h_new = slsqp_opt(h_s, n, 4, 4, 2000, 1e-16)
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_new

    # Phase 7: Final aggressive polish
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h_new = slsqp_opt(h_best, n, 0.1, 1, 6000, 1e-20)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        if time.time() > t0 + budget_s - 20:
            pass
        else:
            h_new2 = slsqp_opt(h_best, n, 0.05, 1, 4000, 1e-20)
            h_new2 = project_feasible(h_new2, n)
            c5_2 = compute_c5(h_new2)
            if c5_2 < c5_best:
                c5_best = c5_2
                h_best = h_new2

    # Final validation
    h_best = project_feasible(h_best, n)
    assert h_best.min() >= -1e-10 and h_best.max() <= 1.0 + 1e-10
    assert abs(h_best.sum() - n / 2.0) < 0.01
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
