import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_gradient(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

def topk_grad(h, n, top_k, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-top_k:]
    c = corr[idx]
    m = c.max()
    w = np.exp(temp * (c - m))
    w /= w.sum()
    g = np.zeros(n)
    for wi, ci in zip(w, idx):
        if wi > 1e-12:
            g += wi * shift_gradient(h, n, ci - (n - 1))
    return g * dx

def nesterov(h_init, n, t_end, rng, top_k=10, T=5.0, lr0=0.4, decay=0.0008):
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5a = compute_c5(h_init)
    ha = h_init.copy()
    for it in range(2000000):
        if time.time() > t_end - 0.5:
            break
        beta = it / (it + 1.5) if it > 0 else 0.0
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        g = topk_grad(h_y, n, top_k, T)
        alpha = lr0 / (1.0 + decay * it)
        # Backtracking
        for _ in range(8):
            cand = project_feasible(h_y - alpha * g, n)
            c5c = compute_c5(cand)
            if c5c <= c5a + 1e-10 or alpha < 1e-6:
                break
            alpha *= 0.5
        h_prev = h_cur
        h_cur = cand
        h_avg += (cand - h_avg) / (it + 1)
        if (it + 1) % 3000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < c5a:
                c5a, ha = c1, h_cur.copy()
            c2 = compute_c5(project_feasible(h_avg, n))
            if c2 < c5a:
                c5a, ha = c2, project_feasible(h_avg, n).copy()
    return ha, c5a

_S_N = 0
_S_TGT = 0.0
_S_T = 1.0
_S_K = 8

def _s_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-(_S_K):]
    c = corr[idx]
    m = c.max()
    e = np.exp(_S_T * (c - m))
    return (m + np.log(e.sum() + 1e-30) / _S_T) * (2.0 / _S_N)

def _s_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-(_S_K):]
    c = corr[idx]
    m = c.max()
    e = np.exp(_S_T * (c - m))
    w = e / e.sum()
    grad = np.zeros(_S_N)
    for wi, ci in zip(w, idx):
        if wi > 1e-12:
            grad += wi * shift_gradient(x, _S_N, ci - (_S_N - 1))
    return grad * (2.0 / _S_N)

def _s_cf(x):
    return x.sum() - _S_TGT

def _s_cj(x):
    return np.ones(_S_N)

def slsqp(h, n, T, top_k, maxiter, ftol):
    global _S_N, _S_TGT, _S_T, _S_K
    _S_N = n
    _S_TGT = n / 2.0
    _S_T = float(T)
    _S_K = min(top_k, 2 * n - 1)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_s_obj, h, method='SLSQP', jac=_s_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _s_cf, 'jac': _s_cj}],
                       options={'maxiter': maxiter, 'ftol': ftol, 'eps': 1e-7})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def perturb(hb, n, rng, pi):
    mode = pi % 6
    if mode == 0:
        sig = 0.002 + 0.0015 * (pi // 6) % 5
        return project_feasible(hb + rng.normal(0, sig, n), n)
    elif mode == 1:
        h_p = hb.copy()
        lb = int(rng.integers(0, max(1, n - 25)))
        ln = int(rng.integers(4, 18))
        h_p[lb:lb+ln] = 1.0 - h_p[lb:lb+ln]
        return project_feasible(h_p, n)
    elif mode == 2:
        h_p = hb.copy()
        center = int(rng.integers(5, n - 5))
        width = int(rng.integers(4, 12))
        amp = rng.uniform(-0.08, 0.08)
        lo = max(0, center - width)
        hi = min(n, center + width + 1)
        x = np.arange(lo, hi)
        h_p[lo:hi] += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
        return project_feasible(h_p, n)
    elif mode == 3:
        h_p = hb.copy()
        k = int(rng.integers(2, n // 5))
        amp = rng.uniform(0.004, 0.015)
        ph = rng.uniform(0, 2 * np.pi)
        h_p += amp * np.sin(2 * np.pi * k * np.arange(n) / n + ph)
        return project_feasible(h_p, n)
    elif mode == 4:
        h_p = hb.copy()
        idx = int(rng.integers(5, n - 5))
        ln = int(rng.integers(2, 6))
        h_p[idx:idx+ln] = rng.uniform(0, 1, ln)
        return project_feasible(h_p, n)
    else:
        h_p = hb.copy()
        i1 = int(rng.integers(0, n))
        i2 = int(rng.integers(0, n))
        if i1 != i2:
            h_p[[i1, i2]] = h_p[[i2, i1]]
        return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n0 = 280
        h_best = np.zeros(n0)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Coarse Nesterov (larger top_k, higher temp)
    t1 = t0 + budget_s * 0.20
    hs, c5s = nesterov(h_best, n, t1, rng, top_k=20, T=10.0, lr0=0.5, decay=0.001)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs

    # Phase 2: SLSQP medium
    t2 = t0 + budget_s * 0.38
    for T, tk, mi in [(8, 15, 2500), (4, 10, 3500), (2, 8, 4500)]:
        if time.time() > t2:
            break
        hn = project_feasible(slsqp(h_best, n, T, tk, mi, 1e-14), n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 3: Fine Nesterov (longer, smaller top_k)
    t3 = t0 + budget_s * 0.60
    hs, c5s = nesterov(h_best, n, t3, rng, top_k=10, T=3.0, lr0=0.35, decay=0.0006)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs

    # Phase 4: SLSQP fine
    t4 = t0 + budget_s * 0.72
    for T, tk, mi in [(1, 6, 5000), (0.5, 5, 7000), (0.3, 4, 8000)]:
        if time.time() > t4:
            break
        hn = project_feasible(slsqp(h_best, n, T, tk, mi, 1e-15), n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 5: Ultra-fine Nesterov
    t5 = t0 + budget_s * 0.82
    hs, c5s = nesterov(h_best, n, t5, rng, top_k=5, T=1.0, lr0=0.2, decay=0.0004)
    if c5s < c5_best:
        c5_best, h_best = c5s, hs

    # Phase 6: Multi-restart perturbation + optimize
    t6 = t0 + budget_s * 0.95
    for pi in range(50):
        if time.time() > t6:
            break
        hp = perturb(h_best, n, rng, pi)
        hn = project_feasible(slsqp(hp, n, 2, 8, 2000, 1e-13), n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn

    # Phase 7: Final polish
    if time.time() < budget_s - 5:
        t_end = budget_s - 2
        hn = project_feasible(slsqp(h_best, n, 0.3, 4, 6000, 1e-16), n)
        c5 = compute_c5(hn)
        if c5 < c5_best:
            c5_best, h_best = c5, hn
        if time.time() < t_end - 10:
            hs, c5s = nesterov(h_best, n, t_end, rng, top_k=4, T=0.5, lr0=0.15, decay=0.0003)
            if c5s < c5_best:
                c5_best, h_best = c5s, hs

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
