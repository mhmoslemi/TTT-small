```python
def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-12:

    """Compute gradient of log-sum-exp approximation of max correlation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Use log-sum-exp for stability
    corr_max = corr.max()
    exp_corr = np.exp((corr - corr_max) * T)
    weights = exp_corr / exp_corr.sum()
    obj = (corr_max + np.log(exp_corr.sum()) / T) * dx

            grad[m] += w * (1.0 - h[j[m] + s2])
```

```python
    """One sweep of block coordinate descent."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    
    for i in range(n):
        # Compute D_k: coefficient of h[i] in each corr_k
        # h[i] appears in corr at shift k when:
        # - as first factor: valid when 0 <= i < n-|k| and 0 <= i+k < n
        #   D contribution: (1 - h[i+k]) if i+k valid
        # - as second factor: valid when 0 <= i-k < n and 0 <= i < n-|k|  
        #   D contribution: -h[i-k] if i-k valid
        
        # For shift k (correlation index = center + k):
        # corr_k = sum_{p: 0<=p<n, 0<=p+k<n} h[p]*(1-h[p+k])
        # h[i] appears when p=i (as first) or p+k=i i.e. p=i-k (as second)
        
        D = np.zeros(2*n-1)
        # As first factor: h[i]*(1-h[i+k]) for shifts k where 0<=i+k<n
        for k in range(-i, n-i):  # k such that 0 <= i+k < n
            # Also need 0 <= i < n - |k|? No, the overlap condition for corr at shift k
            # is: sum over p = 0 to n-1-|k|
            # So we need: 0 <= i <= n-1-|k|
            if 0 <= i <= n-1-abs(k):
                idx = center + k
                D[idx] += (1.0 - h[i+k] if 0 <= i+k < n else 0.0)
        
        # As second factor: h[i-k]*(1-h[i]) for shifts k where 0<=i-k<n
        for k in range(max(-n, i-n), min(n, n-i)):
            # h[i-k] is the first factor, h[i] is the second
            # shift k: corr_k = sum h[p]*(1-h[p+k]), h[i] is (1-h[p+k]) where p+k=i, p=i-k
            # Need 0 <= i-k < n and 0 <= i <= n-1-|k|
            if 0 <= i-k < n and 0 <= i <= n-1-abs(k):
                idx = center + k
                D[idx] += -h[i-k]
        
        # C_k = corr_k - h[i] * D_k
        C = corr - h[i] * D
        
        # Minimize max_k (C_k + t * D_k) over t in [0,1]
```

```python
    center = n - 1

            k_min = max(-i, -(n-1-i))  # from 0<=i+k<n: k >= -i; from 0<=i<=n-1-|k|: |k|<=n-1-i
            k_max = min(n-1-i, n-1-i)  # from 0<=i+k<n: k <= n-1-i; from |k|<=n-1-i

                    D[center + k] += (1.0 - h[i + k])
            # h[i] as second factor: h[i-k]*(1-h[i]) for valid k
            # Need 0 <= i-k < n and 0 <= i <= n-1-|k|
            k_min2 = max(-(n-1-i), -(n-1-i))  # from 0<=i-k<n: k <= i; from |k|<=n-1-i
            k_max2 = min(i, n-1-i)  # from 0<=i-k<n: k <= i (k > -n which is always true); from |k|
            for k in range(k_min2, k_max2 + 1):
                if 0 <= i - k < n and 0 <= i <= n - 1 - abs(k):
                    D[center + k] += -h[i - k]
            
            C = corr - h[i] * D
            # Minimize max(C + t*D) over t in [0,1]
            t_grid = np.linspace(0, 1, n_inner)
            vals = np.max(C + t_grid[:, None] * D[None, :], axis=1)
            h[i] = t_grid[np.argmin(vals)]
    
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid

def bcd_optimize(h, n, n_sweeps=50, t_grid_size=20, t_end=None):
    center = n - 1
    for sweep in range(n_sweeps):
        if t_end and time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        for i in range(n):
            # Compute D (coefficient of h[i] in each corr)
            D = np.zeros(2 * n - 1)
            # First factor: h[i]*(1-h[i+k])
            k_lo = max(-i, -(n-1-i))
            k_hi = n-1-i
            if k_lo <= k_hi:
                ks = np.arange(k_lo, k_hi+1)
                D[center+ks] += (1.0 - h[i+ks])
            # Second factor: h[i-k]*(1-h[i]) → coefficient -h[i-k]
            k_lo2 = max(i-n+1, -(n-1-i))
            k_hi2 = min(i, n-1-i)
            if k_lo2 <= k_hi2:
                ks2 = np.arange(k_lo2, k_hi2+1)
                D[center+ks2] += -h[i-ks2]
            
            C = corr - h[i]*D
            active = np.abs(D) > 1e-15
            if active.sum() == 0:
                continue
            Da = D[active]
            Ca = C[active]
            tg = np.linspace(0, 1, t_grid_size)
            vals = (Ca[None,:] + tg[:,None]*Da[None,:]).max(axis=1)
            h[i] = tg[vals.argmin()]
    return h

def slsqp_polish(h, n, T, k_top, maxiter):
    dx = 2.0 / n
    center = n - 1
    j = np.arange(n)
    
    def obj(x):
        corr = np.correlate(x, 1.0-x, mode="full")
        mk = min(k_top, len(corr))
        idx = np.argpartition(corr, len(corr)-mk)[-mk:]
        topk = corr[idx]
        mc = topk.max()
        sh = T * (topk - mc)
        expv = np.exp(sh)
        sw = expv / expv.sum()
        return (mc + np.log(expv.sum())/T) * dx
    
    def jac(x):
        corr = np.correlate(x, 1.0-x, mode="full")
        mk = min(k_top, len(corr))
        idx = np.argpartition(corr, len(corr)-mk)[-mk:]
        topk = corr[idx]
        mc = topk.max()
        sh = T * (topk - mc)
        expv = np.exp(sh)
        sw = expv / expv.sum()
        grad = np.zeros(n)
        for oi in range(mk):
            w = sw[oi]
            if w < 1e-11:
                continue
            shift = idx[oi] - center
            if shift >= 0:
                m1 = j >= shift
                grad[m1] += w*(1.0 - x[j[m1]-shift])
                m2 = (j+shift) < n
                grad[m2] -= w*x[j[m2]+shift]
            else:
                s2 = -shift
                m1 = (j+s2) < n
                grad[m1] += w*(1.0 - x[j[m1]+s2])
                m2 = j >= s2
                grad[m2] -= w*x[j[m2]-s2]
        grad *= dx
        return grad
    
    def cf(x):
        return x.sum() - n/2.0
    def cj(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)]*n
    cons = [{'type':'eq', 'fun':cf, 'jac':cj}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter':maxiter, 'ftol':1e-15})
        return res.x
    except:
        return h.copy()

def make_random_balanced(n, rng):
    h = (rng.random(n) < 0.5).astype(float)
    ones = h.sum()
    target = n//2
    if ones > target:
        idx = np.where(h==1)[0]
        flip = rng.choice(idx, int(ones-target), replace=False)
        h[flip] = 0
    elif ones < target:
        idx = np.where(h==0)[0]
        flip = rng.choice(idx, int(target-ones), replace=False)
        h[flip] = 1

def make_thue_morse(n):
    # Thue-Morse sequence
    h = np.zeros(n)
    for i in range(n):
        # Count bits
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        h[i] = c % 2
    return h
```

```python
def bcd_update(h, corr, n, top_m=50, t_grid_size=30):
    center = n - 1
    # Keep top_m shifts
    top_idx = np.argpartition(corr, -top_m)[-top_m:]
    top_c = corr[top_idx]
    top_shifts = top_idx - center  # actual shift values
    
    for i in range(n):
        D = np.zeros(top_m)
        C = np.zeros(top_m)
        for m_idx, k in enumerate(top_shifts):
            d = 0.0
            # As first factor
            if 0 <= i+k < n and 0 <= i <= n-1-abs(k):
                d += (1.0 - h[i+k])
            # As second factor
            if 0 <= i-k < n and 0 <= i <= n-1-abs(k):
                d += -h[i-k]
            D[m_idx] = d
            C[m_idx] = top_c[m_idx] - h[i] * d
        
        # Minimize max(C + t*D) over t in [0,1]
        tg = np.linspace(0, 1, t_grid_size)
        vals = (C[None,:] + tg[:,None]*D[None,:]).max(axis=1)
        h[i] = tg[vals.argmin()]
    
    return h
```
