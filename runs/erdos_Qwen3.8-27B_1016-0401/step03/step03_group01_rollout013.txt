import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    if y.sum() == target and y.min() >= 0 and y.max() <= 1:
        return y.copy()
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def smooth_topk(h, n, T, k):
    """Compute smooth max of top-k correlation values and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = np.max(topk)
    sh = T * (topk - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    obj = (mc + np.log(e.sum()) / T) * dx
    center = n - 1
    grad = np.zeros(n)
    j = np.arange(n)
    for oi in range(mk):
        w = sw[oi]
        if w < 1e-14:
            continue
        shift = idx[oi] - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += w * (1.0 - h[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= w * h[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += w * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * h[j[m2] - s2]
    grad *= dx
    return obj, grad

def projected_gradient(h0, n, T, k, n_iter, t_end, lr0=0.005):
    """FISTA with smooth top-k objective."""
    target = n / 2.0
    h_y = h0.copy()
    h_z = h0.copy()
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    t = 1.0
    lr = lr0
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        # FISTA forward-backward
        h_y = h_z + ((t - 1) / (t + 1)) * (h_z - h_y) if it > 0 else h_y.copy()
        
        _, grad = smooth_topk(h_y, n, T, k)
        h_y = h_y - lr * grad
        h_y = project_feasible(h_y, n)
        
        t = (1 + np.sqrt(1 + 4 * t * t)) / 2
        h_z = h_y.copy()
        
        if (it + 1) % 500 == 0:
            c5 = compute_c5(h_y)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_y.copy()
        
        # Decay learning rate slightly
        lr *= 0.99995
    
    c5 = compute_c5(h_y)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_y.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 1.0, y.max() + 1.0
    mid = (lo + hi) / 2.0
    for _ in range(80):
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
        mid = (lo + hi) / 2.0
    return np.clip(y - mid, 0.0, 1.0)

def grad_topk(h, n, T, k):
    """Return (smooth_obj, gradient) for top-k smooth max of correlation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = np.max(topk)
    sh = T * (topk - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    obj = (mc + np.log(e.sum()) / T) * dx
    center = n - 1
    grad = np.zeros(n)
    j = np.arange(n)
    for oi in range(mk):
        w = sw[oi]
        if w < 1e-13:
            continue
        shift = int(idx[oi]) - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += w * (1.0 - h[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= w * h[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += w * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * h[j[m2] - s2]
    grad *= dx
    return obj, grad

def fista_anneal(h0, n, t_end, n_per_stage=2000):
    """FISTA with T-annealing: start smooth, end sharp."""
    stages = [(50, 20, 0.01), (10, 10, 0.005), (4, 5, 0.003), (1, 3, 0.001), (0.5, 1, 0.0005)]
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for T, k, lr0 in stages:
        if time.time() > t_end - 5:
            break
        h_y = h.copy()
        h_z = h.copy()
        t = 1.0
        lr = lr0
        n_it = n_per_stage
        for it in range(n_it):
            if time.time() > t_end:
                break
            if it > 0:
                h_y = h_z + ((t - 1) / (t + 1)) * (h_z - h_y)
            _, grad = grad_topk(h_y, n, T, k)
            h_y = h_y - lr * grad
            h_y = project_feasible(h_y, n)
            t = (1 + np.sqrt(1 + 4 * t * t)) / 2.0
            h_z = h_y.copy()
            lr *= 0.99997
        h = h_y
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def slsqp_refine(h0, n, T, k, maxiter, ftol):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        return grad_topk(x, n, T, k)[0]
    def jac(x):
        return grad_topk(x, n, T, k)[1]
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0.copy()

def gen_perturb(h, n, rng, strength=1.0):
    """Generate a perturbed version of h."""
    h_p = h.copy()
    ptype = rng.integers(0, 8)
    if ptype == 0:
        # Random block swap
        b = rng.integers(5, max(6, n // 4))
        s1 = rng.integers(0, n - b)
        s2 = rng.integers(0, n - b)
        if s1 != s2:
            tmp = h_p[s1:s1+b].copy()
            h_p[s1:s1+b] = h_p[s2:s2+b]
            h_p[s2:s2+b] = tmp
    elif ptype == 1:
        # Frequency perturbation
        freq = rng.integers(1, 12)
        amp = rng.uniform(0.02, 0.08) * strength
        ph = rng.uniform(0, 2*np.pi)
        x = np.arange(n) * 2.0 / n
        h_p += amp * np.sin(np.pi * freq * x + ph)
    elif ptype == 2:
        # Small Gaussian noise
        h_p += rng.uniform(0.01, 0.04) * strength * rng.normal(0, 1, n)
    elif ptype == 3:
        # Block complement
        start = rng.integers(0, n - 10)
        ln = rng.integers(4, min(20, n - start))
        h_p[start:start+ln] = 1.0 - h_p[start:start+ln]
    elif ptype == 4:
        # Shift
        s = rng.integers(1, 5)
        h_p = np.roll(h_p, int(s))
    elif ptype == 5:
        # Random subset reassignment
        mask = rng.random(n) < 0.1
        cnt = int(mask.sum())
        if cnt > 0:
            h_p[mask] = rng.random(cnt)
    elif ptype == 6:
        # Piecewise: split into 3, shuffle
        t1, t2 = sorted(rng.choice(np.arange(10, n-10), size=2).tolist())
        segs = [h_p[:t1], h_p[t1:t2], h_p[t2:]]
        order = rng.permutation(3).tolist()
        h_p = np.concatenate([segs[order[0]], segs[order[1]], segs[order[2]]])
    else:
        # Ramp perturbation
        i1 = rng.integers(0, n-10)
        i2 = rng.integers(i1+5, n)
        ramp = np.linspace(0, rng.uniform(-0.05, 0.05)*strength, i2-i1)
        h_p[i1:i2] += ramp
    return h_p

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.tile([1.0, 0.0], 140)
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    
    # Phase 0: Quick FISTA on initial
    t_phase1 = t0 + budget_s * 0.35
    h_f, c5_f = fista_anneal(h_best, n, t_phase1, n_per_stage=3000)
    c5_f = compute_c5(project_feasible(h_f, n))
    h_f = project_feasible(h_f, n)
    if c5_f < c5_best:
        c5_best = c5_f
        h_best = h_f
    
    # Phase 1: Diverse population search
    t_phase2 = t0 + budget_s * 0.70
    n_cand = 20
    per_cand_time = (t_phase2 - time.time()) / max(1, n_cand)
    for ci in range(n_cand):
        if time.time() > t_phase2 - 5:
            break
        h_p = gen_perturb(h_best, n, rng, strength=rng.uniform(0.5, 1.5))
        h_p = project_feasible(h_p, n)
        c5_p = compute_c5(h_p)
        # Quick FISTA (shorter than initial)
        t_cand_end = min(time.time() + per_cand_time * 0.8, t_phase2 - 5)
        h_f, c5_f = fista_anneal(h_p, n, t_cand_end, n_per_stage=1500)
        h_f = project_feasible(h_f, n)
        c5_f = compute_c5(h_f)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f
    
    # Phase 2: SLSQP refinement
    t_phase3 = t0 + budget_s * 0.90
    for T, k in [(2, 5), (1, 3), (0.5, 2), (0.25, 1)]:
        if time.time() > t_phase3 - 10:
            break
        rem = t_phase3 - time.time()
        mi = int(min(3000, rem * 3))
        h_new = slsqp_refine(h_best, n, T, k, mi, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    
    # Phase 3: Final FISTA polish + more perturbations
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_f, c5_f = fista_anneal(h_best, n, t0 + budget_s - 10, n_per_stage=5000)
        h_f = project_feasible(h_f, n)
        c5_f = compute_c5(h_f)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f
    
    # Final: one more SLSQP push if time allows
    if time.time() < t0 + budget_s - 15:
        h_new = slsqp_refine(h_best, n, 0.5, 1, 2000, 1e-19)

        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
    
    h_best = project_feasible(h_best, n)
    c5_best = min(c5_best, compute_c5(h_best))
    return (h_best, c5_best, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 1.0, y.max() + 1.0
    mid = (lo + hi) / 2.0
    for _ in range(80):
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
        mid = (lo + hi) / 2.0
    return np.clip(y - mid, 0.0, 1.0)

def grad_topk(h, n, T, k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = np.max(topk)
    sh = T * (topk - mc)
    mx = sh.max()
    e = np.exp(sh - mx)
    sw = e / e.sum()
    obj = (mc + (mx + np.log(e.sum())) / T) * dx
    center = n - 1
    grad = np.zeros(n)
    j = np.arange(n)
    for oi in range(mk):
        w = sw[oi]
        if w < 1e-13:
            continue
        shift = int(idx[oi]) - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += w * (1.0 - h[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= w * h[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += w * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * h[j[m2] - s2]
    grad *= dx
    return obj, grad

def fista_stage(h0, n, T, k, n_iter, t_end, lr0):
    h_y = h0.copy()
    h_z = h0.copy()
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    t = 1.0
    lr = lr0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        if it > 0:
            h_y = h_z + ((t - 1.0) / (t + 1.0)) * (h_y - h_z)
        _, grad = grad_topk(h_y, n, T, k)
        h_y = h_y - lr * grad
        h_y = project_feasible(h_y, n)
        t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        h_z = h_y.copy()
        lr *= 0.99998
        if (it + 1) % 1000 == 0:
            c5 = compute_c5(h_y)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_y.copy()
    c5 = compute_c5(h_y)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_y.copy()
    return best_h, best_c5

def fista_anneal(h0, n, t_end, n_per_stage=2500):
    stages = [(60, 20, 0.012), (15, 10, 0.006), (5, 5, 0.003), (1.5, 3, 0.0015), (0.5, 1, 0.0005)]
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for T, k, lr0 in stages:
        if time.time() > t_end - 3:
            break
        h, c5 = fista_stage(h, n, T, k, n_per_stage, t_end, lr0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def _obj_w(x, n, T, k):
    return grad_topk(x, n, T, k)[0]

def _jac_w(x, n, T, k):
    return grad_topk(x, n, T, k)[1]

def _eq_fun(x, n):
    return x.sum() - n / 2.0

def _eq_jac(x, n):
    return np.ones(len(x))

def slsqp_run(h0, n, T, k, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_obj_w, h0, method='SLSQP', jac=_jac_w,
                       args=(n, T, k), bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0.copy()

def gen_perturb(h, n, rng):
    h_p = h.copy()
    ptype = rng.integers(0, 7)
    if ptype == 0:
        b = rng.integers(5, max(6, n // 4))
        s1, s2 = rng.integers(0, n - b, size=2)
        if s1 != s2:
            t = h_p[s1:s1+b].copy()
            h_p[s1:s1+b] = h_p[s2:s2+b]
            h_p[s2:s2+b] = t
    elif ptype == 1:
        f = rng.integers(1, 14)
        a = rng.uniform(0.02, 0.07)
        p = rng.uniform(0, 2 * np.pi)
        x = np.arange(n) * 2.0 / n
        h_p += a * np.sin(np.pi * f * x + p)
    elif ptype == 2:
        h_p += rng.uniform(0.01, 0.035) * rng.normal(0, 1, n)
    elif ptype == 3:
        st = rng.integers(0, n - 12)
        ln = rng.integers(4, min(18, n - st))
        h_p[st:st+ln] = 1.0 - h_p[st:st+ln]
    elif ptype == 4:
        s = rng.integers(1, 5)
        h_p = np.roll(h_p, int(s))
    elif ptype == 5:
        t1, t2 = sorted(rng.choice(np.arange(15, n-15), size=2).tolist())
        segs = [h_p[:t1], h_p[t1:t2], h_p[t2:]]
        ord_ = rng.permutation(3).tolist()
        h_p = np.concatenate([segs[ord_[0]], segs[ord_[1]], segs[ord_[2]]])
    else:
        mask = rng.random(n) < 0.08
        cnt = int(mask.sum())
        if cnt > 0:
            h_p[mask] = rng.random(cnt)
    return h_p

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.tile([1.0, 0.0], 140)
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: FISTA anneal on initial (40% budget)
    t1 = t0 + budget_s * 0.40
    h_f, c5_f = fista_anneal(h_best, n, t1, 3000)
    h_f = project_feasible(h_f, n)
    c5_f = compute_c5(h_f)
    if c5_f < c5_best:
        c5_best = c5_f
        h_best = h_f

    # Phase 2: Diverse population search (30% budget)
    t2 = t0 + budget_s * 0.72
    n_cand = 18
    for ci in range(n_cand):
        if time.time() > t2 - 10:
            break
        h_p = gen_perturb(h_best, n, rng)
        h_p = project_feasible(h_p, n)
        per_time = (t2 - time.time()) / max(1, n_cand - ci) * 0.85
        t_ce = min(time.time() + per_time, t2 - 3)
        h_f, c5_f = fista_anneal(h_p, n, t_ce, 1200)
        h_f = project_feasible(h_f, n)
        c5_f = compute_c5(h_f)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f

    # Phase 3: SLSQP refinement (15% budget)
    t3 = t0 + budget_s * 0.90
    for T, k in [(3, 8), (1, 3), (0.5, 2), (0.3, 1)]:
        if time.time() > t3 - 8:
            break
        rem = t3 - time.time()
        mi = int(min(2500, max(500, rem * 2.5)))
        h_new = slsqp_run(h_best, n, T, k, mi, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Final FISTA polish + one more SLSQP
    rem = budget_s - (time.time() - t0) - 15
    if rem > 20:
        h_f, c5_f = fista_anneal(h_best, n, t0 + budget_s - 30, 4000)
        h_f = project_feasible(h_f, n)
        c5_f = compute_c5(h_f)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f
    if time.time() < t0 + budget_s - 20:
        h_new = slsqp_run(h_best, n, 0.5, 1, 1500, 1e-19)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Try higher resolution if time allows
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        n2 = n * 2
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_hi = np.interp(x_new, x_old, h_best)
        h_hi = project_feasible(h_hi, n2)
        c5_hi = compute_c5(h_hi)
        if c5_hi < c5_best + 0.01:
            h_f, c5_f = fista_anneal(h_hi, n2, t0 + budget_s - 20, 1500)
            h_f = project_feasible(h_f, n2)
            c5_f = compute_c5(h_f)
            if c5_f < c5_best:
                c5_best = c5_f
                h_best = h_f
                n = n2

    h_best = project_feasible(h_best, n)
    c5_best = min(c5_best, compute_c5(h_best))
    return (h_best, c5_best, n)
