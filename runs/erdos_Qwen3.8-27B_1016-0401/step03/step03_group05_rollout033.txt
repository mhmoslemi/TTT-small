```python
def coord_optimize(h, i, lam, n):
    # Compute a_{i,s} and B_{i,s} for all s
    # a_{i,s} = (1 - h[i+s]) * valid(i+s) - h[i-s] * valid(i-s)
    # B_{i,s} = corr[s] - h[i] * a_{i,s}
    
    s_range = np.arange(-(n-1), n)  # shifts
    # For each shift s, compute coefficient of h[i]
    # h[i+s] valid if 0 <= i+s < n
    # h[i-s] valid if 0 <= i-s < n
    
    valid_plus = (i + s_range >= 0) & (i + s_range < n)
    valid_minus = (i - s_range >= 0) & (i - s_range < n)
    
    h_plus = np.zeros(2*n-1)
    h_minus = np.zeros(2*n-1)
    h_plus[valid_plus] = h[i + s_range[valid_plus]]
    h_minus[valid_minus] = h[i - s_range[valid_minus]]
    
    a = (1 - h_plus) * valid_plus - h_minus * valid_minus
    
    corr = np.correlate(h, 1-h, mode="full")
    B = corr - h[i] * a
    
    # Find t* = argmin_{t in [0,1]} max_s(B_s + t*a_s) + lam*t
    
    # Bisection on the slope
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = B + mid * a
        active = np.argmax(vals)
        slope = a[active] + lam
        if slope > 0:
            hi = mid
        else:
            lo = mid
    t_star = (lo + hi) / 2
    t_star = np.clip(t_star, 0, 1)
    return t_star
```
