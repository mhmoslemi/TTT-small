import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    # Returns gradient of corr_shift with respect to h
    # corr_shift = sum_j h[j+s]*(1-h[j]) where s=shift
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        # (1-h[j-shift]) for j>=shift, -h[j+shift] for j+shift<n
        m1 = j >= shift
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        m2 = (j + shift) < n
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        # (1-h[j+s]) for j+s<n, -h[j-s] for j>=s
        m1 = (j + s) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        m2 = j >= s
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def get_c5_and_grad(x, n, T, K):
    corr = np.correlate(x, 1.0 - x, mode="full")
    nc = len(corr)
    center = n - 1
    if K < nc:
        topk = np.argpartition(corr, -K)[-K:]
    else:
        topk = np.arange(nc)
    c_sel = corr[topk.copy()]
    mc = np.max(c_sel)
    sh = T * (c_sel - mc)
    e = np.exp(sh)
    es = e.sum()
    val = (mc + np.log(es) / T) * (2.0 / n)
    sw = e / es
    grad = np.zeros(n)
    for ii in range(len(sw)):
        w = sw[ii]
        if w > 1e-15:
            grad += w * shift_grad(x, n, int(topk[ii]) - center)
    return val, grad * (2.0 / n)

def active_set_opt(h, n, K=5, maxiter=5000, tol=1e-14, t0=None, tlim=None):
    """Minimize the average of top-K correlations."""
    current = h.copy()
    for it in range(maxiter):
        if t0 and time.time() - t0 > tlim:
            break
        corr = np.correlate(current, 1.0 - current, mode="full")
        topk = np.argpartition(corr, -K)[-K:]
        shifts = topk - (n - 1)
        
        # L-BFGS-B on the average of these K correlations
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode="full")
            return np.mean(c[topk]) * (2.0 / n)
        def grad_fn(x):
            c = np.correlate(x, 1.0 - x, mode="full")
            g = np.zeros(n)
            for s in shifts:
                g += shift_grad(x, n, s)
            return g / K * (2.0 / n)
        
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj, current, method='L-BFGS-B', jac=grad_fn,
                       bounds=bounds, options={'maxiter': 500, 'ftol': tol, 'gtol': tol*0.1})
        current = project_feasible(res.x, n)
        
        # Check convergence
        corr_new = np.correlate(current, 1.0 - current, mode="full")
        if np.max(corr_new) - np.mean(corr_new[topk]) < 1e-8:
            break
    return current

def nesterov(h_init, n, t0, tlim, max_iters=400000):
    dx = 2.0 / n
    h_curr = h_init.copy().astype(float)
    h_prev = h_init.copy().astype(float)
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    f_star = c5_best - 0.001  # slight optimism
    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        beta = it / (it + 2.5)
        y = h_curr + beta * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        g = shift_grad(y, n, shift) * dx
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        gap = max(compute_c5(y) - f_star, 1e-8)
        alpha = min(2.0 * gap / (gn * gn + 1e-12), 0.3 / (1.0 + 0.001 * it))
        h_new = project_feasible(y - alpha * g, n)
        h_prev = h_curr.copy()
        h_curr = h_new
        if (it + 1) % 2000 == 0:
            c5_now = compute_c5(h_curr)
            if c5_now < c5_best:
                c5_best = c5_now
                h_best = h_curr.copy()
                f_star = c5_best - 0.0005
    return h_best, c5_best
