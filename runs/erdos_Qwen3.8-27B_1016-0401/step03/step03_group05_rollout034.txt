```python
def jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = T * (corr - mc)
    e = np.exp(sh)
    w = e / e.sum()
    
    # A[i, k] = (1-x[i+s_k]) * vf[i,k] - x[i-s_k] * vs[i,k]
    # = vf[i,k] * (1-x[i+s_k]) - vs[i,k] * x[i-s_k]
    
    # First part: vf[i,k] * (1-x[i+s_k])
    # fi has shape (n, n_corr), vf has shape (n, n_corr)
    first = vf.astype(float) * (1.0 - x[fi])  # x indexed by fi where valid
    # But x[fi] is out of bounds where fi < 0 or fi >= n
    # Need to handle this
    fi_safe = np.clip(fi, 0, n-1)
    first = vf.astype(float) * (1.0 - x[fi_safe])
    
    second = vs.astype(float) * x[si_safe]  # similar
    si_safe = np.clip(si, 0, n-1)
    second = vs.astype(float) * x[si_safe]
    
    A = first - second
    grad = (dx / T) * (A @ w)
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def coord_descent(h_init, n, deadline, rng):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    s_vals = np.arange(-(n - 1), n, dtype=np.float64)
    k_vals = (s_vals + center).astype(int)
    i_idx = np.arange(n)[:, None]
    s_b = s_vals[None, :]
    vf = (i_idx + s_b >= 0) & (i_idx + s_b < n)
    vs = (i_idx - s_b >= 0) & (i_idx - s_b < n)
    fi = np.clip(i_idx + s_b, 0, n - 1).astype(int)
    si = np.clip(i_idx - s_b, 0, n - 1).astype(int)
    vmask = vf | vs
    t_coarse = np.linspace(0.0, 1.0, 51)

    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_best = float(np.max(corr) * dx)
    h_best = h.copy()
    lam = 0.0
    sweep = 0

    while time.time() < deadline:
        for i in rng.permutation(n):
            a_i = np.zeros(2 * n - 1)
            a_i[vf[i]] += (1.0 - h[fi[i][vf[i]]])
            a_i[vs[i]] -= h[si[i][vs[i]]]
            mask = vmask[i] & (np.abs(a_i) > 1e-14)
            if not mask.any():
                continue
            a_arr = a_i[mask]
            k_arr = k_vals[mask]
            b_arr = corr[k_arr] - a_arr * h[i]
            mu = lam / dx
            lines = a_arr[:, None] * t_coarse[None, :] + b_arr[:, None]
            f_vals = np.max(lines, axis=0) + mu * t_coarse
            best_local = t_coarse[np.argmin(f_vals)]
            eps = 0.02
            t_fin = np.linspace(max(0.0, best_local - eps), min(1.0, best_local + eps), 31)
            lines_r = a_arr[:, None] * t_fin[None, :] + b_arr[:, None]
            f_r = np.max(lines_r, axis=0) + mu * t_fin
            new_val = t_fin[np.argmin(f_r)]
            delta = new_val - h[i]
            if abs(delta) > 1e-14:
                h[i] = new_val
                corr[k_arr] += a_arr * delta

        cur_sum = h.sum()
        lam += 0.4 * (cur_sum - target)
        lam = np.clip(lam, -5.0, 5.0)
        sweep += 1
        if sweep % 40 == 0:
            c5_cur = float(np.max(corr) * dx)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
        if sweep % 300 == 0:
            h = project_feasible(h, n)
            corr = np.correlate(h, 1.0 - h, mode="full")
            lam = 0.0

    return h_best, c5_best

def slsqp_smooth(h_init, n, T, maxiter, ftol):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    s_vals = np.arange(n_corr) - center
    i_idx = np.arange(n)[:, None]
    s_b = s_vals[None, :]
    vf = (i_idx + s_b >= 0) & (i_idx + s_b < n)
    vs = (i_idx - s_b >= 0) & (i_idx - s_b < n)
    fi = np.clip(i_idx + s_b, 0, n - 1).astype(int)
    si = np.clip(i_idx - s_b, 0, n - 1).astype(int)
    vf_f = vf.astype(np.float64)
    vs_f = vs.astype(np.float64)

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(c)
        sh = T * (c - mc)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / T) * dx

    def jac(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(c)
        sh = T * (c - mc)
        e = np.exp(sh)
        w = e / e.sum()
        A = vf_f * (1.0 - x[fi]) - vs_f * x[si]
        return (dx / T) * (A @ w)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_init, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init.copy()

def perturb_h(h, n, rng, pi):
    x = np.arange(n) * 2.0 / n
    mode = pi % 5
    if mode == 0:
        f = 0.5 + 0.3 * (pi % 4)
        return h + 0.015 * np.sin(2 * np.pi * f * x + 0.5)
    elif mode == 1:
        f = 0.2 + 0.15 * (pi % 6)
        return h + 0.01 * np.cos(4 * np.pi * f * x)
    elif mode == 2:
        blk = n // 9
        hp = h.copy()
        for b in range(9):
            lo_b = b * blk
            hi_b = min(lo_b + blk, n)
            hp[lo_b:hi_b] += rng.uniform(-0.02, 0.02)
        return hp
    elif mode == 3:
        return h + rng.normal(0, 0.008, n)
    else:
        idx = rng.choice(n, size=n // 5, replace=False)
        hp = h.copy()
        hp[idx] = 1.0 - hp[idx]
        return hp

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP smooth max (coarse T to fine T)
    for T in [16, 8, 4]:
        if time.time() - t0 > budget_s - 500:
            break
        h_new = slsqp_smooth(h_best, n, T, 5000, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Coordinate descent (main workhorse)
    deadline_cd = min(t0 + budget_s - 250, time.time() + (budget_s * 0.45))
    h_cd, c5_cd = coord_descent(h_best, n, deadline_cd, rng)
    h_cd = project_feasible(h_cd, n)
    c5_cd = compute_c5(h_cd)
    if c5_cd < c5_best:
        c5_best = c5_cd
        h_best = h_cd

    # Phase 3: SLSQP fine polish after CD
    for T in [8, 4, 2]:
        if time.time() - t0 > budget_s - 150:
            break
        h_new = slsqp_smooth(h_best, n, T, 6000, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Another CD pass
    if time.time() - t0 < budget_s - 100:
        deadline_cd2 = t0 + budget_s - 80
        h_cd2, c5_cd2 = coord_descent(h_best, n, deadline_cd2, rng)
        h_cd2 = project_feasible(h_cd2, n)
        c5_cd2 = compute_c5(h_cd2)
        if c5_cd2 < c5_best:
            c5_best = c5_cd2
            h_best = h_cd2

    # Phase 5: Perturbation restarts
    remaining = budget_s - (time.time() - t0)
    n_restart = min(20, max(1, int(remaining / 40)))
    for pi in range(n_restart):
        if time.time() - t0 > budget_s - 30:
            break
        h_p = perturb_h(h_best, n, rng, pi)
        h_p = project_feasible(h_p, n)
        h_new = slsqp_smooth(h_p, n, 8, 2000, 1e-17)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Final fine polish
    if time.time() - t0 < budget_s - 15:
        h_new = slsqp_smooth(h_best, n, 4, 3000, 1e-18)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(c5_best, compute_c5(h_best))
    return (h_best, c5_final, n)
```
