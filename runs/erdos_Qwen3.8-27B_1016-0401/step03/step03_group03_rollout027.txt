```python
_G_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0

        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()

            break
        if s > target:
            lo = mid
        else:

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = (j - shift) >= 0
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

    sh = _G_T * (corr - mc)
    e = np.exp(sh)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_TARGET = 0.0
_G_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-12:
            continue
        grad += w * shift_grad_raw(x, _G_N, oi - center)
    return grad * dx

def _g_cons_fun(x):
    return x.sum() - _G_TARGET

def _g_cons_jac(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter, ftol):
    global _G_N, _G_TARGET, _G_T
    _G_N = n
    _G_TARGET = n / 2.0
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_cons_fun, 'jac': _g_cons_jac}]
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def nesterov_adaptive(h_init, n, t0, time_budget, K=3, max_iters=400000):
    """Nesterov with adaptive step size and temperature."""
    dx = 2.0 / n
    center = n - 1
    h_m = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    
    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        T = max(0.5, 8.0 * np.exp(-it / 30000.0))
        
        beta = min(0.85, 0.4 + 0.5 * it / (it + 5.0))
        h_y = h_m + beta * (h_m - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top_idx = np.argpartition(corr, -K)[-K:]
        mv = corr[top_idx].max()
        w = np.exp(T * (corr[top_idx] - mv))
        w /= w.sum()
        
        g = np.zeros(n)
        for wk, ik in zip(w, top_idx):
            if wk < 1e-10:
                continue
            shift = ik - center
            g += wk * shift_grad_raw(h_y, n, shift)
        g *= dx
        
        alpha = 0.12 / (1.0 + 0.001 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_m
        h_m = h_new
        
        if (it + 1) % 8000 == 0:
            cv = compute_c5(h_m)
            if cv < c5_best:
                c5_best = cv
                h_best = h_m.copy()
    
    cv = compute_c5(project_feasible(h_m, n))
    if cv < c5_best:
        c5_best = cv
        h_best = project_feasible(h_m, n).copy()
    return h_best, c5_best

def systematic_polish(h, n, rng, t0, time_limit, max_rounds=200):
    """Greedy pair-move local search."""
    dx = 2.0 / n
    center = n - 1
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        topK = np.argpartition(corr, -4)[-4:]
        
        g = np.zeros(n)
        for ik in topK:
            shift = ik - center
            g += shift_grad_raw(h_cur, n, shift)
        g /= len(topK)
        
        pos = np.where(g > 0.003)[0]
        neg = np.where(g < -0.003)[0]
        
        if len(pos) == 0 or len(neg) == 0:
            # Fallback: random perturb
            h_try = project_feasible(h_cur + rng.normal(0, 0.005, n), n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur:
                h_cur = h_try
                c5_cur = c5_try
            continue
        
        # Try top candidates
        pos_cands = pos[np.argsort(-g[pos])[:min(8, len(pos))]]
        neg_cands = neg[np.argsort(g[neg])[:min(8, len(neg))]]
        
        improved = False
        for i in pos_cands:
            if improved:
                break
            for j in neg_cands:
                for delta in [0.015, 0.03, 0.05, 0.08]:
                    if delta > h_cur[i] or delta > 1.0 - h_cur[j]:

                    h_try = h_cur.copy()
                    h_try[i] -= delta
                    h_try[j] += delta
                    h_try = project_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < c5_cur - 1e-12:
                        h_cur = h_try
                        c5_cur = c5_try
                        improved = True
                        break
        if not improved:
            # Try random moves
            for _ in range(50):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue

                if delta > h_cur[i] or delta > 1.0 - h_cur[j]:
                    continue

                h_try[i] -= delta
                h_try[j] += delta
                h_try = np.clip(h_try, 0, 1)
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < c5_cur - 1e-12:
                    h_cur = h_try
                    c5_cur = c5_try
                    break
    
    return h_cur, c5_cur
```

```python
import numpy as np
from scipy.optimize import minimize

_G_N = 0
_G_TARGET = 0.0
_G_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad_raw(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        grad[m1] += (1.0 - h[j[m1] - shift])
        grad[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        grad[m1] += (1.0 - h[j[m1] + s])
        grad[m2] -= h[j[m2] - s]
    return grad

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _G_T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _G_N - 1
    dx = 2.0 / _G_N
    grad = np.zeros(_G_N)
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-12:
            continue
        grad += w * shift_grad_raw(x, _G_N, oi - center)
    return grad * dx

def _g_cons_fun(x):
    return x.sum() - _G_TARGET

def _g_cons_jac(x):
    return np.ones(_G_N)

def do_slsqp(h, n, T, maxiter, ftol):
    global _G_N, _G_TARGET, _G_T
    _G_N = n
    _G_TARGET = n / 2.0
    _G_T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_cons_fun, 'jac': _g_cons_jac}]
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def nesterov_adaptive(h_init, n, t0, time_budget, K=3, max_iters=300000):
    dx = 2.0 / n
    center = n - 1
    h_m = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break

        beta = min(0.85, 0.4 + 0.45 * it / (it + 5.0))

        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top_idx = np.argpartition(corr, -K)[-K:]

        w /= w.sum()
```
