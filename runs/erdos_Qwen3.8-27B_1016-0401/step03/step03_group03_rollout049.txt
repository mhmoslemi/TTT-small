    for T in [64, 32, 16, 8, 4]:
        if time.time() - t0 > budget_s - 580:
            break
        h_new = do_slsqp(h_best, n, T, 6000, 1e-13)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Nesterov with momentum reset
    h_nes, c5_nes = nesterov_opt(h_best, n, t0, budget_s - 400,
                                  K=10, lr0=0.12, lr_decay=0.002,
                                  max_iters=600000, reset_period=50000)
    if c5_nes < c5_best:
        c5_best = c5_nes
        h_best = h_nes
    else:
        h_best = project_feasible(h_best, n)

    # Phase 3: Fine SLSQP
    for T in [2.0, 1.0, 0.7, 0.5, 0.3, 0.2]:
        if time.time() - t0 > budget_s - 280:
            break
        h_new = do_slsqp(h_best, n, T, 12000, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: Binary-continuous hybrid cycles
    for cycle in range(4):
        if time.time() - t0 > budget_s - 180:
            break
        h_bin = project_binary(h_best, n)
        h_new = do_slsqp(h_bin, n, 1.0, 10000, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new
        h_p = h_bin + rng.normal(0, 0.04, n)
        h_p = project_feasible(h_p, n)
        h_new = do_slsqp(h_p, n, 2.0, 8000, 1e-14)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Block coordinate polish
    if time.time() - t0 < budget_s - 100:
        h_bp, c5_bp = block_polish(h_best, n, rng, t0, budget_s - 30,
                                    block_size=25, n_passes=20)
        if c5_bp < c5_best:
            c5_best = c5_bp
            h_best = h_bp

    # Phase 6: Multi-restart perturbations
    for pi in range(35):
        if time.time() - t0 > budget_s - 20:
            break
        sigma = 0.003 + 0.005 * (pi % 5)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_new = do_slsqp(h_p, n, 3.0, 4000, 1e-13)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Final SLSQP polish
    if time.time() - t0 < budget_s - 8:
        h_new = do_slsqp(h_best, n, 0.2, 15000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    return (h_best, c5_final, n)
