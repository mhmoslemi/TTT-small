def shift_grad(h, n, shift):
    j = np.arange(n)
    grad = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] - shift])
        if m2.any():
            grad[m2] -= h[j[m2] + shift]
    else:

        m2 = j >= s
        if m1.any():
            grad[m1] += (1.0 - h[j[m1] + s])
        if m2.any():
            grad[m2] -= h[j[m2] - s]
    return grad

def active_grad(h, n, K=5):
    """Compute gradient using top-K shifts (average if tied)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    topk = np.argsort(corr)[-K:]
    mc = np.max(corr)

    w = np.exp(-10.0 * gaps / max(mc, 1e-10))

        grad += w[ii] * shift_grad(h, n, topk[ii] - center)
    return grad, float(mc) * (2.0 / n)

    h_prev = h_init.copy()
    y = h_curr.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()

    for it in range(max_iters):
        if time.time() - t0 > tlim:
            break
        beta = it / (it + 2.5)

        return v, g

        return x.sum() - n/2.0
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(y.min()) - 3.0
    hi = float(y.max()) + 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m = j >= shift
        g[m] += (1.0 - h[j[m] - shift])
        m2 = (j + shift) < n
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m = (j + s) < n
        g[m] += (1.0 - h[j[m] + s])
        m2 = j >= s
        g[m2] -= h[j[m2] - s]
    return g

def smooth_max(h, n, T, K):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = float(np.max(corr))
    if K < len(corr):
        idx = np.argpartition(corr, -K)[-K:]
    else:
        idx = np.arange(len(corr))
    cs = corr[idx]
    mcs = float(np.max(cs))
    sh = T * (cs - mcs)
    e = np.exp(sh)
    ess = e.sum()
    val = (mcs + np.log(ess) / T) * (2.0 / n)
    sw = e / ess
    center = n - 1
    g = np.zeros(n)
    for ii in range(len(idx)):
        if sw[ii] > 1e-14:
            g += sw[ii] * shift_grad(h, n, idx[ii] - center)
    return val, g * (2.0 / n)

def sg_topk(h, n, K):
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    mc = float(np.max(corr))
    if K < len(corr):
        idx = np.argpartition(corr, -K)[-K:]
    else:
        idx = np.arange(len(corr))
    cs = corr[idx]
    mcs = float(np.max(cs))
    w = np.exp(-30.0 * (mcs - cs) / max(mcs, 1e-10))
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-10:
            g += w[ii] * shift_grad(h, n, idx[ii] - center)
    return g, mc * (2.0 / n)

def do_lbfgsb(h, n, T, K, maxiter, tol, t0, tlim):
    tp = [False]
    def f(x):
        if time.time() - t0 > tlim: tp[0] = True
        return smooth_max(x, n, T, K)[0]
    def jf(x):
        if time.time() - t0 > tlim: tp[0] = True
        return smooth_max(x, n, T, K)[1]
    try:
        r = minimize(f, h, method='L-BFGS-B', jac=jf,
                     bounds=[(0.0,1.0)]*n,
                     options={'maxiter':maxiter,'ftol':tol,'gtol':tol*0.05,'maxls':15})
        if tp[0]: return h
        return project_feasible(r.x, n)
    except Exception:
        return h.copy()

def do_slsqp(h, n, T, K, maxiter, tol, t0, tlim):
    tp = [False]
    def jf(x):
        if time.time() - t0 > tlim: tp[0] = True
        v, g = smooth_max(x, n, T, K)
        return v, g
    def cu(x): return x.sum() - n/2.0
    def cuj(x): return np.ones(n)
    try:
        r = minimize(jf[0] if False else lambda x: jf(x)[0], h, method='SLSQP',
                     jac=lambda x: jf(x)[1], bounds=[(0.0,1.0)]*n,
                     constraints=[{'type':'eq','fun':cu,'jac':cuj}],
                     options={'maxiter':maxiter,'ftol':tol})
        if tp[0]: return h
        return project_feasible(r.x, n)
    except Exception:
        return h.copy()

def nesterov_sg(h0, n, t0, tlim, K=3, maxit=200000):
    dx = 2.0/n
    hc, hp = h0.copy(), h0.copy()
    best, hb = compute_c5(h0), h0.copy()
    f_lb = 0.375
    for it in range(maxit):
        if time.time() - t0 > tlim: break
        mu = it/(it+2.5)
        y = np.clip(hc + mu*(hc-hp), 0.0, 1.0)
        gr, fy = sg_topk(y, n, K)
        g = gr * dx
        gn = np.linalg.norm(g)
        if gn < 1e-14: break
        gap = max(fy - f_lb, 1e-7)
        a = min(2*gap/(gn*gn+1e-12), 0.5/(1+0.003*it))
        hn = project_feasible(y - a*g, n)
        hp, hc = hc.copy(), hn
        if (it+1)%5000==0:
            fv = compute_c5(hc)
            if fv < best: best, hb = fv, hc.copy()
    fv = compute_c5(hc)
    if fv < best: best, hb = fv, hc.copy()
    return hb, best

        p += rng.normal(0, 0.005, n)
    elif mode == 1:
        p = h + rng.normal(0, rng.uniform(0.002, 0.015), n)
    elif mode == 2:
        p = h.copy()
        bs = max(1, n//int(rng.integers(4,10)))
        for _ in range(int(rng.integers(1,3))):
            st = int(rng.integers(0, max(1,n-bs)))
            p[st:st+bs] = 1.0 - p[st:st+bs]
        p += rng.normal(0, 0.003, n)
    else:
        p = 1.0 - h
        p += rng.normal(0, 0.01, n)
    return project_feasible(p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    
    # Phase 1: L-BFGS-B annealing
    for T, mi in [(32,2500),(16,3500),(8,4000),(4,5000),(2,6000),(1,7000)]:
        rem = budget_s - (time.time()-t0)
        if rem < 80: break
        hn = do_lbfgsb(h_best, n, T, 120, mi, 1e-14, t0, budget_s-40)
        c5 = compute_c5(hn)
        if c5 < c5_best: c5_best, h_best = c5, hn
    
    # Phase 2: Nesterov exact
    for K in [2, 4]:
        rem = budget_s - (time.time()-t0)
        if rem < 60: break
        hs, cb = nesterov_sg(h_best, n, t0, t0+min(rem-30, 180), K, 120000)
        if cb < c5_best: c5_best, h_best = cb, hs
    
    # Phase 3: SLSQP
    for T in [4, 2, 1]:
        rem = budget_s - (time.time()-t0)
        if rem < 50: break
        hn = do_slsqp(h_best, n, T, 80, 2500, 1e-15, t0, budget_s-15)
        c5 = compute_c5(hn)
        if c5 < c5_best: c5_best, h_best = c5, hn
    
    # Phase 4: Fine L-BFGS-B
    for T in [0.5, 1.0]:
        rem = budget_s - (time.time()-t0)
        if rem < 40: break
        hn = do_lbfgsb(h_best, n, T, 60, 6000, 1e-15, t0, budget_s-15)
        c5 = compute_c5(hn)
        if c5 < c5_best: c5_best, h_best = c5, hn
    
    # Phase 5: Nesterov again
    rem = budget_s - (time.time()-t0)
    if rem > 60:
        hs, cb = nesterov_sg(h_best, n, t0, t0+rem-20, 2, 80000)
        if cb < c5_best: c5_best, h_best = cb, hs
    
    # Phase 6: Diverse restarts
    for pi in range(80):

        if rem < 10: break
