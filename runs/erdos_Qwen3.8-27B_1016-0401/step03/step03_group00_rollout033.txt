```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

def gradient_c5(h, n, top_k=5):
    """Compute weighted gradient of C5 using top-k shifts."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-top_k:]
    g = np.zeros(n)
    for s in idx:
        shift = s - (n - 1)
        g += shift_grad(h, n, shift)
    return g / top_k, corr

def projected_gradient_step(h, n, t_end):
    """Custom projected gradient with backtracking."""
    dx = 2.0 / n
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    max_iter = 1000
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        # Compute gradient
        g, corr = gradient_c5(h_cur, n, top_k=3)
        # Backtracking line search
        delta = 0.1
        for _ in range(20):
            h_trial = project_feasible(h_cur - delta * g, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-10:
                break
            delta /= 2.0
        else:
            break  # No improvement possible with this gradient
        
        h_cur = h_trial
        best_c5 = c5_trial
        best_h = h_trial.copy()
    
    return best_h, best_c5

def lp_trust_region(h, n, m_shifts=20, t_end=None):
    """Iterative LP with trust region."""
    dx = 2.0 / n
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    
    for outer in range(50):
        if t_end is not None and time.time() > t_end:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        top_idx = np.argsort(corr)[-m_shifts:]
        
        # Build LP
        m = len(top_idx)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)
        
        for i, s in enumerate(top_idx):
            shift = s - (n - 1)
            g = shift_grad(h_cur, n, shift)
            c_k = corr[s]
            A_ub[i, :n] = dx * g
            A_ub[i, n] = -1.0
            b_ub[i] = dx * (g @ h_cur - c_k)
        
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                break
            h_lp = res.x[:n]
        except Exception:
            break
        
        # Trust region: try β = 1, 0.5, 0.25, ...
        for beta in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_trial = project_feasible(h_cur + beta * (h_lp - h_cur), n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-12:
                h_cur = h_trial
                best_c5 = c5_trial
                best_h = h_trial.copy()
                break
        else:
            # No improvement, try smaller
            break
    
    return best_h, best_c5

def slsqp_optimize(h, n, T, top_k, maxiter, ftol):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Use soft-max objective
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(c)[-top_k:]
        cv = c[idx]
        m = cv.max()
        e = np.exp(T * (cv - m))
        return (m + np.log(e.sum()) / T) * dx
    
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(c)[-top_k:]
        cv = c[idx]
        m = cv.max()
        e = np.exp(T * (cv - m))
        w = e / e.sum()
        g = np.zeros(n)
        for wi, si in zip(w, idx):
            if wi > 1e-12:
                sh = si - (n - 1)
                g += wi * shift_grad(x, n, sh)
        return g * dx
    
    def con_fun(x):
        return x.sum() - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

def lp_trust_region(h, n, m_shifts, t_end):
    dx = 2.0 / n
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    for outer in range(80):
        if time.time() > t_end:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        top_idx = np.argsort(corr)[-m_shifts:]
        m = len(top_idx)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)
        for i, s in enumerate(top_idx):
            shift = s - (n - 1)
            g = shift_grad(h_cur, n, shift)
            c_k = corr[s]
            A_ub[i, :n] = dx * g
            A_ub[i, n] = -1.0
            b_ub[i] = dx * (g @ h_cur - c_k)
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if not res.success:
                break
            h_lp = res.x[:n]
        except Exception:
            break
        improved = False
        for beta in [1.0, 0.7, 0.5, 0.3, 0.15]:
            h_trial = project_feasible(h_cur + beta * (h_lp - h_cur), n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-13:
                h_cur = h_trial
                best_c5 = c5_trial
                best_h = h_trial.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_c5

def grad_backtrack(h, n, t_end, top_k=3):
    dx = 2.0 / n
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    for it in range(5000):
        if time.time() > t_end:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for s in idx:
            g += shift_grad(h_cur, n, s - (n - 1))
        g /= top_k
        delta = 0.5 / (1.0 + 0.001 * it)
        for _ in range(25):
            h_trial = project_feasible(h_cur - delta * g, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-12:
                break
            delta *= 0.6
        else:
            if delta < 1e-10:
                break
            h_trial = h_cur.copy()
            c5_trial = best_c5
        h_cur = h_trial
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
    return best_h, best_c5

def binary_local_search(h, n, t_end):
    dx = 2.0 / n
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    j = np.arange(n)
    improved = True
    while improved and time.time() < t_end:
        improved = False
        for i in range(0, n, 3):
            if time.time() > t_end:
                break
            for seg_len in [1, 2, 3]:
                if i + seg_len > n:
                    continue
                h_trial = h_cur.copy()
                h_trial[i:i+seg_len] = 1.0 - h_trial[i:i+seg_len]
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < best_c5 - 1e-10:
                    h_cur = h_trial
                    best_c5 = c5_t
                    best_h = h_cur.copy()
                    improved = True
                    break
    return best_h, best_c5

def slsqp_opt(h, n, T, top_k, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(c)[-top_k:]
        cv = c[idx]
        m = cv.max()
        e = np.exp(T * (cv - m))
        return (m + np.log(e.sum()) / T) * dx
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(c)[-top_k:]
        cv = c[idx]
        m = cv.max()
        e = np.exp(T * (cv - m))
        w = e / e.sum()
        g = np.zeros(n)
        for wi, si in zip(w, idx):
            if wi > 1e-12:
                g += wi * shift_grad(x, n, si - (n - 1))
        return g * dx
    def cf(x):
        return x.sum() - n / 2.0
    def cj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()
```
