    it = 0
    while time.time() < t_end:
        beta = 2.0 / (it + 2.0)
        h_y = np.clip(h_cur + beta * (h_cur - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h_y, n, ci - (n - 1))
        g *= dx / top_k
        alpha = lr0 / (1.0 + decay * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        it += 1
        if (it + 1) % 3000 == 0:
            c = compute_c5(h_cur)
            if c < best_c:
                best_c, best_h = c, h_cur.copy()
    c = compute_c5(h_cur)
    if c < best_c:
        best_c, best_h = c, h_cur.copy()
    return best_h, best_c

def _active_refine(h, n, t_end, rng, n_shifts=4):
    best_c = compute_c5(h)
    best_h = h.copy()
    h_cur = h.copy()
    dx = 2.0 / n
    it = 0
    while time.time() < t_end:
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        idx = np.argsort(corr)[-n_shifts:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h_cur, n, ci - (n - 1))
        g *= dx / n_shifts
        alpha = 0.002 / (1.0 + 0.0002 * it)
        h_new = project_feasible(h_cur - alpha * g, n)
        h_cur = h_new
        it += 1
        if (it + 1) % 2000 == 0:
            c = compute_c5(h_cur)
            if c < best_c:
                best_c, best_h = c, h_cur.copy()
    c = compute_c5(h_cur)
    if c < best_c:
        best_c, best_h = c, h_cur.copy()
    return best_h, best_c

def _perturb_struct(h, n, rng, mode):
    h_p = h.copy()
    if mode == 0:
        h_p += rng.normal(0, 0.002, n)
    elif mode == 1:
        lb = rng.integers(0, max(1, n - 30))
        ln = rng.integers(8, 25)
        h_p[lb:lb + ln] = 1.0 - h_p[lb:lb + ln]
    elif mode == 2:
        i, j = rng.integers(0, n, 2)
        h_p[[i, j]] = h_p[[j, i]]
    elif mode == 3:
        freq = rng.integers(2, 6)
        phase = rng.uniform(0, 2 * np.pi)
        h_p += 0.003 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    else:
        h_p += rng.normal(0, 0.004, n)
        h_p[rng.integers(0, n)] = 1.0
    return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n0 = 280
        h_best = np.zeros(n0)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Nesterov wide top_k (explore landscape) - 28%
    t1 = t0 + budget_s * 0.28
    hb, c1 = _nesterov(h_best, n, t1, rng, top_k=25, lr0=0.5, decay=0.0004)
    if c1 < c5_best:
        c5_best, h_best = c1, hb

    # Phase 2: SLSQP medium temperature (build structure) - 22%
    t2 = t0 + budget_s * 0.50
    for T, tk, mi in [(12, 20, 2500), (6, 15, 3000), (3, 12, 3000)]:
        if time.time() > t2:
            break
        hn = _slsqp(h_best, n, T, tk, mi, 1e-14)
        hn = project_feasible(hn, n)
        c = compute_c5(hn)
        if c < c5_best:
            c5_best, h_best = c, hn

    # Phase 3: Nesterov narrow top_k (exploit) - 20%
    t3 = t0 + budget_s * 0.70
    hb, c3 = _nesterov(h_best, n, t3, rng, top_k=8, lr0=0.25, decay=0.001)
    if c3 < c5_best:
        c5_best, h_best = c3, hb

    # Phase 4: SLSQP fine temperature (polish) - 12%
    t4 = t0 + budget_s * 0.82
    for T, tk, mi in [(1.5, 10, 4000), (0.7, 8, 4000)]:
        if time.time() > t4:
            break
        hn = _slsqp(h_best, n, T, tk, mi, 1e-15)
        hn = project_feasible(hn, n)
        c = compute_c5(hn)
        if c < c5_best:
            c5_best, h_best = c, hn

    # Phase 5: Active-set micro-refine - 6%
    t5 = t0 + budget_s * 0.88
    if time.time() < t5:
        hb, c5r = _active_refine(h_best, n, t5, rng, n_shifts=4)
        if c5r < c5_best:
            c5_best, h_best = c5r, hb

    # Phase 6: Perturbation restarts + SLSQP - remaining time
    restart_count = 0
    while time.time() < budget_s - 8:
        mode = restart_count % 5
        hp = _perturb_struct(h_best, n, rng, mode)
        hn = _slsqp(hp, n, 1.0, 8, 2000, 1e-13)
        hn = project_feasible(hn, n)
        c = compute_c5(hn)
        if c < c5_best:
            c5_best, h_best = c, hn
        restart_count += 1
        if restart_count % 3 == 0:
            hb, c_r = _active_refine(h_best, n, min(time.time() + 5, budget_s - 8), rng, 3)
            if c_r < c5_best:
                c5_best, h_best = c_r, hb

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
