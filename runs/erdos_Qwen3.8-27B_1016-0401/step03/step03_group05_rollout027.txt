import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 3.0
    hi = y.max() + 3.0
    mid = (lo + hi) / 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

def slsqp_weighted(h_s, n, idx_list, w_list, maxiter, timeout_s):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    t0 = time.time()

    def obj(x):
        total = 0.0
        for ii, w in zip(idx_list, w_list):
            s = int(ii) - center
            total += w * np.dot(x, 1.0 - np.roll(x, s))
        return total * dx

    def jac(x):
        if time.time() - t0 > timeout_s:
            raise Exception("t/o")
        g = np.zeros(n)
        for ii, w in zip(idx_list, w_list):
            s = int(ii) - center
            g += w * shift_grad(x, n, s)
        return g * dx

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                    'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_s, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return res.x
    except Exception:
        return h_s.copy()

def slsqp_lse(h_s, n, T, K, maxiter, timeout_s):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    t0 = time.time()

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        if K < n_corr:
            idx = np.argpartition(corr, -K)[-K:]
            c = corr[idx]
        else:
            c = corr
        m = c.max()
        s = T * (c - m)
        e = np.exp(s)
        return (m + np.log(e.sum()) / T) * dx

    def jac(x):
        if time.time() - t0 > timeout_s:
            raise Exception("t/o")
        corr = np.correlate(x, 1.0 - x, mode="full")
        if K < n_corr:
            idx = np.argpartition(corr, -K)[-K:]
            c = corr[idx]
            shifts = idx - center
        else:
            c = corr
            shifts = np.arange(n_corr) - center
        m = c.max()
        s = T * (c - m)
        e = np.exp(s)
        w = e / e.sum()
        g = np.zeros(n)
        for i in range(len(w)):
            if w[i] > 1e-10:
                g += w[i] * shift_grad(x, n, int(shifts[i]))
        return g * dx

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                    'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_s, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return res.x
    except Exception:
        return h_s.copy()

def nesterov_refine(h0, n, t_limit, n_iters=200000):
    target = n / 2.0
    dx = 2.0 / n
    center = n - 1
    h_y = h0.copy()
    h_prev = h0.copy()
    h_avg = h0.copy()
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    for it in range(n_iters):
        if time.time() > t_limit:
            break
        beta = (it / (it + 1.5)) if it > 0 else 0.0
        h_y = h_y + beta * (h_y - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        K = min(30, max(1, 30 * it // n_iters + 1))
        top_idx = np.argpartition(corr, -K)[-K:]
        top_vals = corr[top_idx]
        mc = top_vals.max()
        w = (top_vals / max(mc, 1e-15)) ** 6
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(K) / K
        else:
            w /= ws
        g = np.zeros(n)
        for i in range(K):
            g += w[i] * shift_grad(h_y, n, int(top_idx[i]) - center)
        g *= dx
        alpha = 0.4 / (1.0 + 0.005 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_y.copy()
        h_y = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        if (it + 1) % 20000 == 0:
            ha = project_feasible(h_avg, n)
            c = compute_c5(ha)
            if c < best_c5:
                best_c5 = c
                best_h = ha.copy()
    return best_h, best_c5
