    return float(np.max(corr) * dx)

def fix_constraint(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                d = min(diff, 1.0 - h[i])
            else:
                d = min(-diff, h[i])
            if d <= 0:
                continue
            h[i] += d if diff > 0 else -d
            diff -= d if diff > 0 else -d
            if abs(diff) < 1e-14:
                break
    return h

def lag_obj(x, li):
    n = len(x)
    return np.correlate(x, 1.0 - x, mode="full")[li] * (2.0 / n)

        c = np.correlate(h, 1.0 - h, mode="full")
        w = int(np.argmax(c))
        h2 = do_slsqp(h, w, 1200, 1e-18)
        h2 = fix_constraint(h2, tgt)
        c2 = compute_c5(h2)
        if c2 < c5_best:
            c5_best = c2
            h = h2
            h_best = h.copy()
        h, cg = grad_cycle(h, 3, tgt, n_top=5)
        if cg < c5_best:
            c5_best = cg
            h_best = h.copy()
    h, c5bp = binary_push(h, 0.02, tgt)
    if c5bp < c5_best:
        c5_best = c5bp
        h_best = h.copy()

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
