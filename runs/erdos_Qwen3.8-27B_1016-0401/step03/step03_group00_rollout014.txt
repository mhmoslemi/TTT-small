        e = np.exp(T * (c - m))

        return np.ones(n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def grad_shift_correct(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j <= (n - 1 - shift)
        m2 = j >= shift
        if np.any(m1):
            g[m1] += (1.0 - h[j[m1] + shift])
        if np.any(m2):
            g[m2] -= h[j[m2] - shift]
    else:
        s = -shift
        m1 = j >= s
        m2 = j <= (n - 1 - s)
        if np.any(m1):
            g[m1] += (1.0 - h[j[m1] - s])
        if np.any(m2):
            g[m2] -= h[j[m2] + s]
    return g

def slsqp_opt(h, n, T, top_k, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        m = c.max()
        e = np.exp(T * (c - m))
        es = e.sum()
        if es < 1e-300:
            return m * dx
        return (m + np.log(es) / T) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
        m = c.max()
        e = np.exp(T * (c - m))
        es = e.sum()
        if es < 1e-300:
            es = 1.0
        w = e / es
        grad = np.zeros(n)
        for wi, ci in zip(w, idx):
            if wi > 1e-12:
                grad += wi * grad_shift_correct(x, n, ci - (n - 1))
        return grad * dx
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def nesterov_run(h_init, n, t_end, top_k, T_smooth, lr0, decay):
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    best = h_init.copy()
    best_c5 = compute_c5(h_init)
    lr = lr0
    mu = 0.9
    for it in range(20000000):
        if time.time() > t_end:
            break
        h_y = np.clip(h_cur + mu * (h_cur - h_prev), 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        c_top = corr[idx]
        m = c_top.max()
        e = np.exp(T_smooth * (c_top - m))
        es = e.sum()
        if es < 1e-300:
            w = np.ones(top_k) / top_k
        else:
            w = e / es
        g = np.zeros(n)
        for wi, ci in zip(w, idx):
            if wi > 1e-12:
                g += wi * grad_shift_correct(h_y, n, ci - (n - 1))
        g *= dx / top_k
        h_new = project_feasible(h_y - lr * g, n)
        h_prev = h_cur
        h_cur = h_new
        lr *= decay
        if (it + 1) % 3000 == 0:
            c5 = compute_c5(h_cur)
            if c5 < best_c5:
                best_c5 = c5
                best = h_cur.copy()
    return best, best_c5

def swap_local_search(h_input, n, t_end, rng, n_trials=500000):
    best = h_input.copy()
    best_c5 = compute_c5(h_input)
    h = h_input.copy()
    for it in range(n_trials):
        if time.time() > t_end:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n0 = 280
        h_best = np.zeros(n0)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse-to-fine with corrected gradient
    t1 = t0 + budget_s * 0.28
    for T, tk, mi in [(40, 25, 1500), (15, 20, 2000), (6, 15, 2500),
                       (2, 10, 3500), (1, 8, 4000), (0.5, 6, 5000)]:
        if time.time() > t1:
            break
        h_try = slsqp_opt(h_best, n, T, tk, mi, 1e-15)
        c5 = compute_c5(h_try)
        if c5 < c5_best:
            c5_best, h_best = c5, h_try

    # Phase 2: Nesterov exact worst-shift
    t2 = t0 + budget_s * 0.45
    h_ns, c5_ns = nesterov_run(h_best, n, t2, top_k=1, T_smooth=1.0,
                               lr0=0.04, decay=0.999995)
    if c5_ns < c5_best:
        c5_best, h_best = c5_ns, h_ns

    # Phase 3: Nesterov smoothed top-5
    t3 = t0 + budget_s * 0.58
    h_ns2, c5_ns2 = nesterov_run(h_best, n, t3, top_k=5, T_smooth=40.0,
                                 lr0=0.025, decay=0.99999)
    if c5_ns2 < c5_best:
        c5_best, h_best = c5_ns2, h_ns2

    # Phase 4: SLSQP fine
    t4 = t0 + budget_s * 0.72
    for T, tk, mi in [(3, 10, 3000), (1, 6, 5000), (0.5, 4, 7000)]:
        if time.time() > t4:
            break
        h_try = slsqp_opt(h_best, n, T, tk, mi, 1e-16)
        c5 = compute_c5(h_try)
        if c5 < c5_best:
            c5_best, h_best = c5, h_try

    # Phase 5: Swap local search
    t5 = t0 + budget_s * 0.88
    h_sw, c5_sw = swap_local_search(h_best, n, t5, rng)
    if c5_sw < c5_best:
        c5_best, h_best = c5_sw, h_sw

    # Phase 6: Final SLSQP + swap
    if time.time() < budget_s - 15:
        h_try = slsqp_opt(h_best, n, 1, 4, 3000, 1e-16)
        c5 = compute_c5(h_try)
        if c5 < c5_best:
            c5_best, h_best = c5, h_try
    if time.time() < budget_s - 5:

        if c5_sw2 < c5_best:

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
