import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 0.01
    hi = y.max() + 0.01
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def topk_grad(h, n, K, p):
    """Compute weighted top-K subgradient. Vectorized."""
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    if K >= 2*n-1:
        top_idx = np.arange(2*n-1)
    else:
        top_idx = np.argpartition(corr, -K)[-K:]
    top_vals = corr[top_idx]
    w = (top_vals / max(top_vals.max(), 1e-15)) ** p
    ws = w.sum()
    if ws > 1e-15:
        w /= ws
    else:
        w = np.full(len(top_idx), 1.0/len(top_idx))
    
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        wi = w[i]
        if wi < 1e-12:
            continue
        s = top_idx[i] - center
        if s >= 0:
            gap = n - s
            if gap > 0:
                grad[:gap] += wi * (1.0 - h[s:])
            if s < n:
                grad[s:] -= wi * h[:gap]
        else:
            t = -s
            gap = n - t
            if gap > 0:
                grad[t:] += wi * (1.0 - h[:gap])
            if t < n:
                grad[:gap] -= wi * h[t:]
    return grad * (2.0/n)

def fast_subgrad(h_init, n, t0, t_end, K0=2, K1=12, p=16, beta=0.93):
    """Nesterov momentum subgradient with adaptive parameters."""
    h = h_init.copy()
    h_prev = h.copy()
    v = np.zeros(n)
    c5_best = compute_c5(h)
    h_best = h.copy()
    lr = 0.5
    it = 0
    eval_interval = 10000
    
    while time.time() < t_end:
        frac = min(it / 200000, 1.0)
        K_cur = int(K0 + (K1 - K0) * frac * frac)
        
        # Nesterov extrapolation
        beta_cur = it / (it + 1.5) if it > 0 else 0.0
        y = h + beta_cur * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        g = topk_grad(y, n, K_cur, p)
        
        # Step
        h_new = y - lr * g
        h_new = project_feasible(h_new, n)
        
        h_prev = h
        h = h_new
        # Decay step size slowly
        lr *= 0.99995
        
        it += 1
        if it % eval_interval == 0:
            c5_cur = compute_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
    
    return h_best, c5_best

def bitwise_refine(h, n, t0, t_end):

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    s0 = np.clip(y, 0.0, 1.0)
    if abs(s0.sum() - target) < 1e-10:
        return s0
    lo = s0.min() - 0.01
    hi = s0.max() + 0.01
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def topk_grad(h, n, K, p):
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    if K >= 2 * n - 1:
        top_idx = np.arange(2 * n - 1)
    else:
        top_idx = np.argpartition(corr, -K)[-K:]
    tv = corr[top_idx]
    w = (tv / max(tv.max(), 1e-15)) ** p
    ws = w.sum()
    if ws > 1e-15:
        w /= ws
    else:
        w = np.full(len(top_idx), 1.0 / len(top_idx))
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        wi = w[i]
        if wi < 1e-12:
            continue
        s = top_idx[i] - center
        if s >= 0:
            gap = n - s
            if gap > 0:
                grad[:gap] += wi * (1.0 - h[s:])
            if s < n:
                grad[s:] -= wi * h[:gap]
        else:
            t = -s
            gap = n - t
            if gap > 0:
                grad[t:] += wi * (1.0 - h[:gap])
            if t < n:
                grad[:gap] -= wi * h[t:]
    return grad * (2.0 / n)

def nesterov_subgrad(h_init, n, t0, t_end, K0=2, K1=15, p=16, beta0=0.0):
    h = h_init.copy()
    h_prev = h.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    lr = 0.4
    it = 0
    stall = 0
    eval_every = 5000
    
    while time.time() < t_end:

        lr *= 0.99998
        it += 1

            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
                stall = 0
            else:
                stall += eval_every
            eval_every = min(eval_every * 1.1, 20000)
    
    return h_best, c5_best

def coord_refine(h_init, n, t0, t_end, seed=42):

        c5_best = c5_1
        h_best = h1.copy()

    # Phase 2: Coordinate refinement (coarse steps)
    t2 = t0 + budget_s * 0.50
    h2, c5_2 = coord_refine(h_best, n, t0, t2, seed, [0.05, 0.1, 0.15, 0.2, 0.03])
    if c5_2 < c5_best:
        c5_best = c5_2
        h_best = h2.copy()

    # Phase 3: SLSQP polish (smooth softmax, low T for accuracy)
    t3 = t0 + budget_s * 0.68
    if time.time() < t3:
        h3 = slsqp_polish(h_best, n, T=8, maxiter=3000, t_end=t3)
        h3 = project_feasible(h3, n)
        c5_3 = compute_c5(h3)
        if c5_3 < c5_best:
            c5_best = c5_3
            h_best = h3.copy()

    # Phase 4: Second subgradient round (finer, higher p)
    t4 = t0 + budget_s * 0.82
    h4, c5_4 = nesterov_subgrad(h_best, n, t0, t4, K0=3, K1=20, p=32, lr_init=0.2)
    if c5_4 < c5_best:
        c5_best = c5_4
        h_best = h4.copy()

    # Phase 5: Fine coordinate refinement
    t5 = t0 + budget_s * 0.95
    h5, c5_5 = coord_refine(h_best, n, t0, t5, seed + 7, [0.01, 0.02, 0.04, 0.01, 0.03])
    if c5_5 < c5_best:
        c5_best = c5_5
        h_best = h5.copy()

    # Phase 6: Final SLSQP micro-polish if time allows
    if time.time() < t0 + budget_s * 0.98:
        h6 = slsqp_polish(h_best, n, T=4, maxiter=2000, t_end=t0 + budget_s * 0.99)
        h6 = project_feasible(h6, n)
        c5_6 = compute_c5(h6)
        if c5_6 < c5_best:
            c5_best = c5_6
            h_best = h6.copy()

    h_best = project_feasible(h_best, n)
    c5_final = compute_c5(h_best)
    c5_best = min(c5_best, c5_final)
    return (h_best, c5_best, n)
