def corr_and_grad(h, lag_idx):
    """Compute correlation at specific lag and its gradient."""
    n = len(h)
    o = lag_idx - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if o == 0:
        corr = np.sum(h * (1 - h))
        g[:] = 1.0 - 2.0 * h
    elif o > 0:

        g[o:] += 1.0 - h[:n-o]
        g[:n-o] += -h[o:]
    else:
        p = -o
        corr = np.sum(h[:n-p] * (1 - h[p:]))
        g[:n-p] += 1.0 - h[p:]
        g[p:] += -h[:n-p]

    corr = np.correlate(h, 1.0 - h, mode="full")
