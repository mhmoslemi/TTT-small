from scipy.optimize import minimize
import time

_G_N = 0
_G_TGT = 0.0
_G_T = 1.0
_G_K = 20
_G_IDX = None
_G_W = None

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_G_K:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_G_T * (c - m))
    _G_W = e / e.sum()
    _G_IDX = idx
    return (m + np.log(e.sum()) / _G_T) * (2.0 / _G_N)

def _g_jac(x):
    grad = np.zeros(_G_N)
    for wi, ci in zip(_G_W, _G_IDX):
        if wi > 1e-10:
            grad += wi * shift_grad(x, _G_N, ci - (_G_N - 1))
    return grad * (2.0 / _G_N)

def _g_cfun(x):
    return x.sum() - _G_TGT

def _g_cjac(x):
    return np.ones(_G_N)

def _slsqp(h, n, T, top_k, maxiter, ftol):
    global _G_N, _G_TGT, _G_T, _G_K, _G_IDX, _G_W
    _G_N = n
    _G_TGT = n / 2.0
    _G_T = float(T)
    _G_K = min(top_k, 2 * n - 1)
    _G_IDX = None
    _G_W = None
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_g_obj, h, method='SLSQP', jac=_g_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _g_cfun, 'jac': _g_cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def _nesterov(h_init, n, t_end, rng, top_k=8):
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5a = compute_c5(h_init)
    ha = h_init.copy()
    it = 0
    while time.time() < t_end:
        beta = float(it) / (it + 1.5) if it > 0 else 0.0
        h_y = h_cur + beta * (h_cur - h_prev)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h_y, n, ci - (n - 1))
        g *= dx / top_k
        alpha = 1.0 / (1.0 + 0.005 * it)
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        h_avg += (h_new - h_avg) / (it + 1)
        it += 1
        if it % 4000 == 0:
            c1 = compute_c5(h_cur)
            if c1 < c5a:
                c5a, ha = c1, h_cur.copy()
            c2 = compute_c5(project_feasible(h_avg, n))
            if c2 < c5a:
                c5a, ha = c2, project_feasible(h_avg, n).copy()
    return ha, c5a

def _coord_descent(h, n, t_end, rng):
    c5 = compute_c5(h)
    best = h.copy()
    steps_desc = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    passes = 0
    max_stalls = 3
    stall = 0
    while time.time() < t_end:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            corr = np.correlate(best, 1.0 - best, mode="full")
            top_idx = np.argsort(corr)[-5:]
            g = np.zeros(n)
            for ci in top_idx:
                g += shift_grad(best, n, ci - (n - 1))
            g /= len(top_idx)
            direction = -np.sign(g[i]) if abs(g[i]) > 1e-12 else 1.0
            for s in steps_desc:
                mag = s * direction
                h_try = best.copy()
                h_try[i] = np.clip(best[i] + mag, 0.0, 1.0)
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < c5 - 5e-9:
                    best = h_try
                    c5 = c5_try
                    improved = True
                    break
            if improved:
                break
        passes += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= max_stalls:
                break
    return best, c5

def _perturb(h, n, rng, pi):
    mode = pi % 5

        idx = rng.integers(0, n, size=8)
        hp[idx] = rng.uniform(0, 1, size=8)
        return project_feasible(hp, n)
