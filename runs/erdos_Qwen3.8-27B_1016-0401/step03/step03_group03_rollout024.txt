from scipy.optimize import minimize

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    hi = float(y.max()) + 2.0

            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def get_shift_grad(h, n, shift):
    grad = np.zeros(n)
    if shift >= 0:
        idx = np.arange(n - shift)
        grad[idx] += (1.0 - h[idx + shift])
        idx2 = np.arange(shift, n)
        grad[idx2] -= h[idx2 - shift]
    else:
        s = -shift
        idx = np.arange(n - s)
        grad[idx] += (1.0 - h[idx + s])
        idx2 = np.arange(s, n)
        grad[idx2] -= h[idx2 - s]
    return grad

# Global state for SLSQP callbacks
_n_global = 0
_T_global = 1.0

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _T_global * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _T_global) * (2.0 / _n_global)

def _jac(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = np.max(corr)
    sh = _T_global * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _n_global - 1
    dx = 2.0 / _n_global
    grad = np.zeros(_n_global)
    # Only use shifts with significant weight
    for oi in range(len(sw)):
        w = sw[oi]
        if w < 1e-10:
            continue
        grad += w * get_shift_grad(x, _n_global, oi - center)
    return grad * dx

def _cons_fun(x):
    return x.sum() - _n_global / 2.0

def _cons_jac(x):
    return np.ones(_n_global)

def solve_slsqp(h, n, T, maxiter=10000, ftol=1e-15):
    global _n_global, _T_global
    _n_global = n
    _T_global = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 2.0
    mid = (lo + hi) / 2.0
    for _ in range(128):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, shift):
    n = len(h)
    grad = np.zeros(n)
    if shift >= 0:
        i = np.arange(n - shift)
        grad[i] += (1.0 - h[i + shift])
        i2 = np.arange(shift, n)
        grad[i2] -= h[i2 - shift]
    else:
        s = -shift
        i = np.arange(n - s)
        grad[i] += (1.0 - h[i + s])
        i2 = np.arange(s, n)
        grad[i2] -= h[i2 - s]
    return grad

def obj_fn(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = float(np.max(corr))
    sh = _T * (corr - mc)
    e = np.exp(sh)
    return (mc + np.log(e.sum()) / _T) * (2.0 / _N)

def jac_fn(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = float(np.max(corr))
    sh = _T * (corr - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    center = _N - 1
    dx = 2.0 / _N
    grad = np.zeros(_N)
    for oi in range(len(sw)):
        if sw[oi] < 1e-10:
            continue
        grad += sw[oi] * shift_grad(x, oi - center)
    return grad * dx

def _cons_eq(x):
    return x.sum() - _N / 2.0

def _cons_eq_jac(x):
    return np.ones(_N)

def run_slsqp(h, n, T, maxiter, ftol):
    global _N, _T
    _N = n
    _T = float(T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', jac=jac_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def nesterov_phase(h_init, n, t0, time_budget, K=8, max_iters=500000):
    dx = 2.0 / n
    center = n - 1
    h_sg = h_init.copy()
    h_prev = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        beta = float(it) / (it + 3.0) if it > 0 else 0.0
        h_y = h_sg + beta * (h_sg - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        top_idx = np.argsort(corr)[-K:]
        mv = float(corr[top_idx].max())
        w = np.exp(8.0 * (corr[top_idx] - mv))
        w /= w.sum()
        g = np.zeros(n)
        for wk, ik in zip(w, top_idx):
            g += wk * shift_grad(h_y, int(ik) - center)
        g *= dx
        alpha = 0.08 / (1.0 + 0.003 * it)
        h_new = h_y - alpha * g
        h_new = project_feasible(h_new, n)
        h_prev = h_sg
        h_sg = h_new
        if (it + 1) % 8000 == 0:
            cv = compute_c5(h_sg)
            if cv < c5_best:
                c5_best = cv
                h_best = h_sg.copy()
    cv = compute_c5(h_sg)
    if cv < c5_best:
        c5_best = cv
        h_best = h_sg.copy()
    return h_best, c5_best

def sa_local_search(h, n, rng, t0, time_budget, n_iter=20000):
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    T_sa = 0.002
    for it in range(n_iter):
        if time.time() - t0 > time_budget:
            break
        T_sa *= 0.9997
        i1 = rng.integers(0, n)
        i2 = rng.integers(0, n)
        if i1 == i2:
            continue
        delta = rng.uniform(0.005, 0.04)
        direction = 1.0 if h_cur[i1] < h_cur[i2] else -1.0
        mov = min(delta, abs(h_cur[i1] - h_cur[i2]),
                  h_cur[i1] if direction < 0 else 1.0 - h_cur[i1],
                  h_cur[i2] if direction > 0 else 1.0 - h_cur[i2])
        if mov < 1e-6:
            continue
        h_try = h_cur.copy()
        h_try[i1] -= direction * mov
        h_try[i2] += direction * mov
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < c5_cur or (rng.random() < np.exp(-(c5_try - c5_cur) / max(T_sa, 1e-9))):
            h_cur = h_try
            c5_cur = c5_try
    return h_cur, c5_cur

def equalize_top(h, n, M=5, t0, time_budget):
    center = n - 1
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for _ in range(50):
        if time.time() - t0 > time_budget:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        top_idx = np.argsort(corr)[-M:]
        top_vals = corr[top_idx]
        spread = top_vals.max() - top_vals.min()
        if spread < 1e-6:
            break
        g = np.zeros(n)
        for ik in top_idx:
            shift = int(ik) - center
            g += (top_vals.max() - corr[ik]) * shift_grad(h_cur, shift)
        g -= np.mean(g)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        step = min(0.01, spread / (g_norm * dx + 1e-12)) * dx
        h_new = h_cur - step * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h_cur = h_new
            c5_cur = c5_new
    return h_cur, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(280)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP coarse-to-fine with extended low-T range
    temps = [256, 128, 64, 32, 16, 8, 4, 2, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12]
    for T in temps:
        if time.time() - t0 > budget_s - 550:
            break
        h_new = run_slsqp(h_best, n, T, 10000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: Nesterov with higher K and exp(8) weighting
    h_sg, c5_sg = nesterov_phase(h_best, n, t0, budget_s - 450, K=8, max_iters=400000)
    if c5_sg < c5_best:
        c5_best = c5_sg
        h_best = h_sg
    else:
        h_best = project_feasible(h_best, n)

    # Phase 3: Equalize top correlations
    if time.time() - t0 < budget_s - 350:
        h_eq, c5_eq = equalize_top(h_best, n, M=6, t0=t0, time_budget=budget_s - 340)
        if c5_eq < c5_best:
            c5_best = c5_eq
            h_best = h_eq

    # Phase 4: Fine SLSQP polish (very low T)
    for T in [0.08, 0.05, 0.03, 0.02]:
        if time.time() - t0 > budget_s - 250:
            break
        h_new = run_slsqp(h_best, n, T, 15000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: SA local search
    if time.time() - t0 < budget_s - 120:
        h_sa, c5_sa = sa_local_search(h_best, n, rng, t0, budget_s - 110, n_iter=15000)
        if c5_sa < c5_best:
            c5_best = c5_sa
            h_best = h_sa

    # Phase 6: Multi-restart perturbations + SLSQP
    for pi in range(12):
        if time.time() - t0 > budget_s - 40:
            break
        sigma = 0.002 + 0.003 * (pi % 4)
        h_p = h_best + rng.normal(0, sigma, n)
        h_p = project_feasible(h_p, n)
        h_new = run_slsqp(h_p, n, 0.5, 6000, 1e-15)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Final SLSQP polish
    if time.time() - t0 < budget_s - 15:
        h_new = run_slsqp(h_best, n, 0.05, 12000, 1e-16)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
