import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def topk_obj_grad(x, n, T, k):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mk = min(k, len(corr))
    idx = np.argpartition(corr, -mk)[-mk:]
    topk = corr[idx]
    mc = topk.max()
    sh = T * (topk - mc)
    e = np.exp(sh)
    sw = e / e.sum()
    obj = (mc + np.log(e.sum()) / T) * (2.0 / n)
    center = n - 1
    grad = np.zeros(n)
    j = np.arange(n)
    for oi in range(mk):
        w = sw[oi]
        if w < 1e-12:
            continue
        shift = idx[oi] - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += w * (1.0 - x[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= w * x[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += w * (1.0 - x[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * x[j[m2] - s2]
    grad *= 2.0 / n
    return obj, grad

def _obj_wrap(x, n, T, k):
    return topk_obj_grad(x, n, T, k)[0]

def _jac_wrap(x, n, T, k):
    return topk_obj_grad(x, n, T, k)[1]

def slsqp_step(h, n, T, k, maxiter, t_end):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda z: z.sum() - target, 'jac': lambda z: np.ones(n)}]
    # Can't use lambdas per rules - use a different approach
    if time.time() > t_end:
        return h.copy()
    try:
        res = minimize(
            lambda z: topk_obj_grad(z, n, T, k)[0],
            h, method='SLSQP',
            jac=lambda z: topk_obj_grad(z, n, T, k)[1],
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda z: z.sum() - target,
                          'jac': lambda z: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-20})
        return res.x
    except Exception:
        return h.copy()

def slsqp_step2(h, n, T, k, maxiter, t_end):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    if time.time() > t_end:
        return h.copy()
    class _Obj:
        def __init__(self, nn, TT, kk):
            self.nn = nn; self.TT = TT; self.kk = kk
        def __call__(self, x):
            return topk_obj_grad(x, self.nn, self.TT, self.kk)[0]
        def jac(self, x):
            return topk_obj_grad(x, self.nn, self.TT, self.kk)[1]
    class _Con:
        def __init__(self, tgt):
            self.tgt = tgt
        def fun(self, x):
            return x.sum() - self.tgt
        def jac(self, x):
            return np.ones(x.shape)
    obj = _Obj(n, T, k)
    con = _Con(target)
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj.jac, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con.fun, 'jac': con.jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return res.x
    except Exception:
        return h.copy()

def fista_opt(h_init, n, t_end, lr0=0.04):
    dx = 2.0 / n
    h_y = h_init.copy()
    h_z = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    t_mom = 1.0
    lr = lr0
    j = np.arange(n)
    center = n - 1
    for it in range(80000):
        if time.time() > t_end:
            break
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        ka = min(6, len(corr))
        idx = np.argpartition(corr, -ka)[-ka:]
        tv = corr[idx]
        sw = np.exp(3.0 * (tv - tv.max()))
        sw /= sw.sum()
        grad = np.zeros(n)
        for oi in range(ka):
            w = sw[oi]
            shift = idx[oi] - center
            if shift >= 0:
                m1 = j >= shift
                grad[m1] += w * (1.0 - h_y[j[m1] - shift])
                m2 = (j + shift) < n
                grad[m2] -= w * h_y[j[m2] + shift]
            else:
                s2 = -shift
                m1 = (j + s2) < n
                grad[m1] += w * (1.0 - h_y[j[m1] + s2])
                m2 = j >= s2
                grad[m2] -= w * h_y[j[m2] - s2]
        grad *= dx
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        h_z = h_y + beta * (h_y - h_z)
        h_y = project_feasible(h_z - lr * grad, n)
        t_mom = t_new
        lr *= 0.99997
        if (it + 1) % 3000 == 0:
            c5 = compute_c5(h_y)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_y.copy()
    c5_f = compute_c5(h_y)
    if c5_f < c5_best:
        c5_best = c5_f
        h_best = h_y.copy()
    return h_best, c5_best

def pair_transfer(h, n, t_end, max_rounds=3000):
    dx = 2.0 / n
    target = n / 2.0
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    h_best = h_cur.copy()
    c5_best = c5_cur
    j = np.arange(n)
    center = n - 1
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        top3 = np.argpartition(corr, -3)[-3:]
        pk = top3[np.argmax(corr[top3])]
        shift = pk - center
        if shift >= 0:
            m1 = j >= shift
            xs = j[m1]
            contrib = h_cur[xs] * (1.0 - h_cur[xs - shift])
        else:
            s2 = -shift
            m1 = (j + s2) < n
            xs = j[m1]
            contrib = h_cur[xs] * (1.0 - h_cur[xs + s2])
        if len(contrib) == 0:
            break
        top_idx = np.argsort(contrib)[-8:]
        found = False
        for ti in top_idx:
            xi = xs[ti]
            oi2 = xi - shift if shift >= 0 else xi + s2
            if oi2 < 0 or oi2 >= n:
                continue
            for d in (0.02, 0.05, 0.1, 0.2):
                if h_cur[xi] < d or h_cur[oi2] > 1.0 - d:
                    continue
                h_t = h_cur.copy()
                h_t[xi] -= d
                h_t[oi2] += d
                c5_t = compute_c5(h_t)
                if c5_t < c5_cur - 1e-12:
                    h_cur = h_t
                    c5_cur = c5_t
                    if c5_cur < c5_best:
                        c5_best = c5_cur
                        h_best = h_cur.copy()
                    found = True
                    break
            if found:
                break
        if not found and (rnd + 1) % 200 == 0:
            break
    return h_best, c5_best

def freq_perturb(h, n, rng):
    H = fft(h - h.mean())
    fc = rfftfreq(n)
    cut = rng.uniform(0.2, 0.9)
    mask = np.zeros_like(H, dtype=float)
    pos = np.where(fc >= 0)[0]
    mask[pos] = (np.abs(fc[pos]) <= cut * 0.5).astype(float)
    neg = np.where(fc < 0)[0]
    mask[neg] = (np.abs(fc[neg]) <= cut * 0.5).astype(float)
    h_f = ifft(H * mask).real + h.mean()
    return np.clip(h_f, 0.0, 1.0)

def rfftfreq(n):
    return np.fft.rffreq(n)
