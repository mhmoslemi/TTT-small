    x_cur = h[i]
    center = n - 1
    ms = np.arange(2 * n - 1)
    b = i + ms - center
    a = i - ms + center
    valid_b = (b >= 0) & (b < n)
    valid_a = (a >= 0) & (a < n)
    bv = np.clip(b, 0, n-1)
    av = np.clip(a, 0, n-1)
