                   options={'maxiter': maxiter, 'ftol': ftol})

def slsqp_sm(h, T, bounds, maxiter=3000, ftol=1e-19):
    global _TARGET
    _TARGET = len(h) / 2.0
    target = len(h) / 2.0
    cons = [dict(type='eq', fun=_eq_fun, jac=_eq_jac)]

    def sm_obj(x):
        return lbfgs_sm_obj(x, T, 0.0, target)

    def sm_grad(x):
        return lbfgs_sm_grad(x, T, 0.0, target)

    res = minimize(sm_obj, h, method='SLSQP', jac=sm_grad,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return np.clip(res.x, 0.0, 1.0)

def lbfgs_sm(h, T, lam, target, bounds, maxiter=5000):
    res = minimize(lbfgs_sm_obj, h, args=(T, lam, target), method='L-BFGS-B',
                   jac=lbfgs_sm_grad, bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-19, 'gtol': 1e-12,
                            'maxls': 50})
    return np.clip(res.x, 0.0, 1.0)

def coord_descent_fine(h, n_sweeps=3, base_step=0.005):
    n = len(h)
    target = n / 2.0
    c5_cur = compute_c5(h)
    for sweep in range(n_sweeps):
        improved = False
        steps = [base_step * (0.6 ** k) for k in range(5)]
        steps += [-s for s in steps]
        for j in range(n):
            best_dt = 0
            best_val = c5_cur
            for dt in steps:
                nv = h[j] + dt
                if nv <= 0.0 or nv >= 1.0:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = fix_constraint(trial, target)
                c5_new = compute_c5(trial)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_dt = dt
            if best_dt != 0:
                h = h.copy()
                h[j] = np.clip(h[j] + best_dt, 0.0, 1.0)
                h = fix_constraint(h, target)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def binary_push(h, threshold=0.02):
    n = len(h)
    target = n / 2.0
    c5_cur = compute_c5(h)
    imp = False
    for j in range(n):
        if h[j] > 1.0 - threshold:
            trial = h.copy()
            trial[j] = 1.0
        elif h[j] < threshold:
            trial = h.copy()
            trial[j] = 0.0
        else:
            continue
        trial = fix_constraint(trial, target)
        c5_new = compute_c5(trial)
        if c5_new < c5_cur - 1e-15:
            h = trial
            c5_cur = c5_new
            imp = True
    return h, c5_cur, imp

def run(seed=42, budget_s=1100, **kwargs):
    global _TARGET
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.zeros(340)
        h_best[::2] = 1.0

    n = len(h_best)
    target = n / 2.0
    h_best = fix_constraint(h_best, target)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Quick SLSQP on top-3 lags (8%)
    p1_end = t0 + budget_s * 0.08
    for _ in range(5):
        if elapsed() > p1_end:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode="full")
        top3 = np.argsort(corr)[-3:]
        for lag in top3:
            if elapsed() > p1_end:
                break
            try:
                h_new = slsqp_lag(h_best, int(lag), bounds, 500, 1e-19)
                h_new = fix_constraint(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-16:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    # Phase 2: L-BFGS-B smooth max with penalty annealing (25%)
    p2_end = t0 + budget_s * 0.33
    sched = [(150, 1e-5), (400, 1e-4), (1000, 1e-3), (3000, 1e-2),
             (8000, 0.1), (20000, 1.0), (50000, 10.0)]
    for T, lam in sched:
        if elapsed() > p2_end:
            break
        try:
            h_new = lbfgs_sm(h_best, T, lam, target, bounds, 5000)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-16:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 3: SLSQP smooth max (17%)
    p3_end = t0 + budget_s * 0.50
    for T in [500, 2000, 8000, 30000, 100000]:
        if elapsed() > p3_end:
            break
        try:
            h_new = slsqp_sm(h_best, T, bounds, 4000, 1e-19)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-16:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass

    # Phase 4: Fine coord descent + binary push + SLSQP cycles (25%)
    p4_end = t0 + budget_s * 0.75
    for it in range(20):
        if elapsed() > p4_end:
            break
        h_best, c5_cd = coord_descent_fine(h_best, 2, 0.004)
        if c5_cd < c5_best:
            c5_best = c5_cd
        h_best, c5_bp, imp = binary_push(h_best, 0.015)
        if c5_bp < c5_best:
            c5_best = c5_bp
        if not imp:
            corr_b = np.correlate(h_best, 1.0 - h_best, mode="full")
            top2 = np.argsort(corr_b)[-2:]
            for lag in top2:
                if elapsed() > p4_end:
                    break
                try:
                    h_new = slsqp_lag(h_best, int(lag), bounds, 600, 1e-19)
                    h_new = fix_constraint(h_new, target)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_best:
                        c5_best = c5_new
                        h_best = h_new
                except Exception:
                    pass
        # Periodic L-BFGS-B refresh
        if it % 4 == 3:
            try:
                h_new = lbfgs_sm(h_best, 5000, 0.1, target, bounds, 3000)
                h_new = fix_constraint(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass

    # Phase 5: Nearby n exploration (10%)
    p5_end = t0 + budget_s * 0.85
    n_cur = len(h_best)
    for dn in [-6, -4, 4, 6, -10, 10]:
        n_alt = n_cur + dn
        if n_alt < 20 or elapsed() > p5_end:
            continue
        x_old = np.arange(n_cur) * (2.0 / n_cur)
        x_new = np.linspace(0.0, 2.0, n_alt, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best, period=2.0)
        target_alt = n_alt / 2.0
        h_alt = fix_constraint(h_alt, target_alt)
        c5_alt = compute_c5(h_alt)
        bounds_alt = [(0.0, 1.0)] * n_alt
        for _ in range(3):
            if elapsed() > p5_end:
                break
            corr_a = np.correlate(h_alt, 1.0 - h_alt, mode="full")
            worst_a = int(np.argmax(corr_a))
            try:
                h_new = slsqp_lag(h_alt, worst_a, bounds_alt, 400, 1e-17)
                h_new = fix_constraint(h_new, target_alt)
                c5_new = compute_c5(h_new)
                if c5_new < c5_alt - 1e-16:
                    c5_alt = c5_new
                    h_alt = h_new
            except Exception:
                pass
        h_alt, c5_cd_a = coord_descent_fine(h_alt, 2, 0.004)
        if c5_cd_a < c5_alt:
            c5_alt = c5_cd_a
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_alt
            n_cur = n_alt

    # Phase 6: Final polish (15%)
    p6_end = t0 + budget_s - 3.0
    n = len(h_best)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    for _ in range(12):
        if elapsed() > p6_end:
            break
        corr_f = np.correlate(h_best, 1.0 - h_best, mode="full")
        worst_f = int(np.argmax(corr_f))
        try:
            h_new = slsqp_lag(h_best, worst_f, bounds, 2000, 1e-20)
            h_new = fix_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass
    h_best, c5_cd_f = coord_descent_fine(h_best, 4, 0.002)
    if c5_cd_f < c5_best:
        c5_best = c5_cd_f
    h_best, c5_bp_f, _ = binary_push(h_best, 0.008)
    if c5_bp_f < c5_best:
        c5_best = c5_bp_f

    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, len(h_best))
