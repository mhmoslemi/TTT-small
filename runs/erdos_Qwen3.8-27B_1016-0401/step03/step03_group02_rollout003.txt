import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        # Spread adjustment uniformly where possible
        if diff > 0:
            room = 1.0 - h
            total_room = room.sum()
            if total_room < 1e-15:
                break
            add = np.minimum(diff, room * diff / max(total_room, 1e-30))
            h += add
        else:
            avail = h
            total_avail = avail.sum()
            if total_avail < 1e-15:
                break
            sub = np.minimum(-diff, avail * (-diff) / max(total_avail, 1e-30))
            h -= sub
    return np.clip(h, 0.0, 1.0)

def project(h, target):
    """Project onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(h, 0.0, 1.0)
    return fix_sum(h, target)

def grad_cjk(h, k, n):
    """Gradient of corr_k * dx with respect to h"""
    dx = 2.0 / n
    g = np.zeros(n)
    o = k - (n - 1)  # Convert from "full" correlation index to offset
    if o == 0:
        g[:] = 1.0 - 2.0 * h
    elif o > 0:
        g[o:] += 1.0 - h[:n-o]
        g[:n-o] += -h[o:]
    else:
        p = -o
        g[:n-p] += 1.0 - h[p:]
        g[p:] += -h[:n-p]
    return g * dx

def subgrad_minimax(h, target, top_k=20, temperature=50):
    """Compute subgradient of max_k f_k(h) using softmax over top-k"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    mc = vals.max()
    # Softmax weights over top-k
    idx = np.argsort(vals)[-top_k:]
    w = np.zeros(len(vals))
    sw = np.exp(temperature * (vals[idx] - mc))
    sw /= sw.sum()
    w[idx] = sw
    # Subgradient
    g = np.zeros(n)
    for i_k in idx:
        if w[i_k] < 1e-14:
            continue
        g += w[i_k] * grad_cjk(h, i_k, n)
    return g, mc

def subgrad_step(h, target, n_iters, alpha0=0.01, alpha_decay=0.995):
    """Subgradient descent with projection"""
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_cur)
    alpha = alpha0
    for it in range(n_iters):
        g, c5_cur = subgrad_minimax(h_cur, target)
        h_new = h_cur - alpha * g
        h_new = project(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        # Only accept if not too much worse (allow some exploration)
        if c5_new < c5_cur + 0.001:
            h_cur = h_new
        else:
            h_cur = project(h_cur - 0.5 * alpha * g, target)
        alpha *= alpha_decay
    return best_h, best_c5

def slsqp_smoothmax(h, T, target, maxiter=2000):
    """SLSQP with log-sum-exp smoothed max"""
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = corr.max()
        s = T * (corr - mc)
        return (mc + np.log(s.exp().sum()) / T) * dx
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = corr.max()
        s = T * (corr - mc)
        e = s.exp()
        sw = e / e.sum()
        center = n - 1
        g = np.zeros(n)
        for o in range(-(n-1), n):
            w = sw[center + o]
            if w < 1e-13:
                continue
            if o == 0:
                g += w * (1.0 - 2.0 * x)
            elif o > 0:
                g[o:] += w * (1.0 - x[:n-o])
                g[:n-o] += -w * x[o:]
            else:
                p = -o
                g[:n-p] += w * (1.0 - x[p:])
                g[p:] += -w * x[:n-p]
        return g * dx
    
    def eqfun(x):
        return x.sum() - target
    
    def eqjac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [dict(type='eq', fun=eqfun, jac=eqjac)]
    
    res = minimize(obj, h, method='SLSQP', jac=grad, bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
    return np.clip(res.x, 0.0, 1.0)
