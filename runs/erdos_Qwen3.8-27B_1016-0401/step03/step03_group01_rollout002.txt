import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2, 0.0, 1.0)

def binary_local_search(h_bin, n, t_end, seed=0):
    rng = np.random.default_rng(seed)
    h = h_bin.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = 0.01
    for it in range(500000):
        if time.time() > t_end:

        if it % 5000 == 0:
            T *= 0.999
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]

        c5_new = compute_c5(h)
        if c5_new < c5 or rng.random() < np.exp(-(c5_new - c5) / max(T, 1e-10)):
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[a], h[b] = 1.0, 0.0
    return best_h, best_c5

def smooth_max_grad(h, n, T):
    """Log-sum-exp smooth max of correlation, with gradient."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    cm = np.max(corr)
    sh = T * (corr - cm)
    e = np.exp(sh)
    sw = e / e.sum()
    obj = (cm + np.log(e.sum()) / T) * (2.0 / n)
    # Gradient
    grad = np.zeros(n)
    center = n - 1
    j = np.arange(n)
    for oi in range(len(corr)):
        w = sw[oi]
        if w < 1e-8:
            continue
        shift = oi - center
        if shift >= 0:
            m = j >= shift
            grad[m] += w * (1.0 - h[j[m] - shift])
            m2 = (j + shift) < n
            grad[m2] -= w * h[j[m2] + shift]
        else:
            s2 = -shift
            m = (j + s2) < n
            grad[m] += w * (1.0 - h[j[m] + s2])
            m2 = j >= s2
            grad[m2] -= w * h[j[m2] - s2]
    grad *= 2.0 / n
    return obj, grad

def slsqp_opt(h, n, T, maxiter, ftol=1e-14):
    target = n / 2.0
    def obj(x):
        return smooth_max_grad(x, n, T)[0]
    def jac(x):
        return smooth_max_grad(x, n, T)[1]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def projected_gradient(h, n, T, t_end, lr0=0.01):
    """Projected gradient with backtracking."""
    h_y = h.copy()
    dx = 2.0 / n
    tgt = n / 2.0

def make_binary_pattern(n, ptype, rng):
    h = np.zeros(n)
    if ptype == 0:  # alternating
        h[::2] = 1.0
    elif ptype == 1:  # block
        h[:n//2] = 1.0
    elif ptype == 2:  # periodic with period 3
        h[::3] = 1.0
        h[1::3] = 1.0  # no, that's all 1s
        # period 3: 110 repeated
        for i in range(n):
            h[i] = 1.0 if (i % 3 < 2) else 0.0
    elif ptype == 3:  # periodic with period 5
        for i in range(n):
            h[i] = 1.0 if (i % 5 < 2) else 0.0  # Not exactly half
        # Adjust to get exactly n/2 ones
    elif ptype == 4:  # random binary
        idx = rng.permutation(n)
        h[idx[:n//2]] = 1.0
    elif ptype == 5:  # Walsh-like
        for i in range(n):
            # Count trailing zeros of i
            v = i
            c = 0
            while v % 2 == 0 and v > 0:
                c += 1
                v //= 2
            h[i] = 1.0 if c % 2 == 0 else 0.0
    # Fix sum constraint
    s = h.sum()
    if abs(s - n/2) > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        diff = n/2 - s
        if diff > 0:
            for idx in ones[:int(diff)]:
                h[idx] = 0.0
            for idx in zeros[:int(diff)]:
                h[idx] = 1.0
        else:
            for idx in zeros[:int(-diff)]:
                h[idx] = 1.0
            for idx in ones[:int(-diff)]:
                h[idx] = 0.0
    return h.astype(float)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_and_grad(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    cm = np.max(corr)
    sh = T * (corr - cm)
    e = np.exp(sh)
    sw = e / e.sum()
    obj = (cm + np.log(e.sum()) / T) * (2.0 / n)
    grad = np.zeros(n)
    center = n - 1
    j = np.arange(n)
    active = np.where(sw > 1e-5)[0]
    for oi in active:
        w = sw[oi]
        shift = oi - center
        if shift >= 0:
            m1 = j >= shift
            grad[m1] += w * (1.0 - h[j[m1] - shift])
            m2 = (j + shift) < n
            grad[m2] -= w * h[j[m2] + shift]
        else:
            s2 = -shift
            m1 = (j + s2) < n
            grad[m1] += w * (1.0 - h[j[m1] + s2])
            m2 = j >= s2
            grad[m2] -= w * h[j[m2] - s2]
    grad *= 2.0 / n
    return obj, grad

def slsqp_opt(h, n, T, maxiter, ftol=1e-14):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        return smooth_max_and_grad(x, n, T)[0]
    def jac(x):
        return smooth_max_and_grad(x, n, T)[1]
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def binary_sa(h_bin, n, t_end, seed=0):
    rng = np.random.default_rng(seed)
    h = h_bin.copy()
    c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), c5
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = 0.02
    count = 0
    while time.time() < t_end:
        if count % 3000 == 0:
            T *= 0.995
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        h[a], h[b] = 0.0, 1.0
        c5_new = compute_c5(h)
        if c5_new <= c5 or rng.random() < np.exp(-(c5_new - c5) / max(T, 1e-12)):
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[a], h[b] = 1.0, 0.0
        count += 1
    return best_h, best_c5

def make_pattern(n, ptype, rng):
    h = np.zeros(n)
    if ptype == 0:
        h[::2] = 1.0
    elif ptype == 1:
        h[:n // 2] = 1.0
    elif ptype == 2:
        for i in range(n):
            h[i] = 1.0 if (i % 3 < 2) else 0.0
    elif ptype == 3:
        for i in range(n):
            h[i] = 1.0 if (i % 5 < 3) else 0.0
    elif ptype == 4:
        idx = rng.permutation(n)
        h[idx[:n // 2]] = 1.0
    else:
        for i in range(n):
            v, c = i, 0
            while v % 2 == 0 and v > 0:
                c += 1
                v //= 2
            h[i] = 1.0 if c % 2 == 0 else 0.0
    diff = int(round(n / 2 - h.sum()))
    if diff > 0:
        z = np.where(h == 0)[0]

    elif diff < 0:
        o = np.where(h == 1)[0]
        if len(o) >= -diff:
            h[rng.choice(o, -diff, replace=False)] = 0.0
    return h
