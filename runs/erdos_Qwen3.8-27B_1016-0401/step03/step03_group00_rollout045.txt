    h_prev = h.copy()
    f_prev = compute_c5(h_cur)
    h_best = h_cur.copy()
    f_best = f_prev

    g_prev = None

            break

        if it > 0:

            alpha = 0.5 / (1.0 + 0.01 * it)

        g_prev = g.copy()

        h_prev = h_cur
        h_cur = h_new

            if f_cur < f_best:
                f_best = f_cur
                h_best = h_cur.copy()

    if f_final < f_best:
        f_best = f_final
        h_best = h_cur.copy()

    return x.sum() - _G_TGT

    _G_K = min(top_k, 2 * n - 1)

def nesterov_direct(h_init, n, t_end, top_k=8):
    """Direct subgradient with Nesterov momentum and adaptive step."""
    dx = 2.0 / n
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    h_best = h_cur.copy()
    f_best = compute_c5(h_cur)
    
    for it in range(300000):

        # Nesterov extrapolation
        if it > 0:
            beta = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / (it + 1.0)
            h_y = h_cur + beta * (h_cur - h_prev)
            h_y = np.clip(h_y, 0.0, 1.0)
        else:
            h_y = h_cur
        
        # Gradient of top-k shifts (unsmoothed)
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for ci in idx:
            g += shift_grad(h_y, n, ci - (n - 1))
        g *= dx / top_k
        
        # Adaptive step size (decay with annealing)
        alpha = 1.0 / (2.0 + 0.005 * it)
        
        h_new = project_feasible(h_y - alpha * g, n)
        h_prev = h_cur
        h_cur = h_new
        
        if (it + 1) % 5000 == 0:
            f_cur = compute_c5(h_cur)
            if f_cur < f_best:
                f_best = f_cur
                h_best = h_cur.copy()
    
    f_final = compute_c5(h_cur)
    if f_final < f_best:
        f_best = f_final
        h_best = h_cur.copy()
    return h_best, f_best

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def shift_grad(h, n, shift):
    j = np.arange(n)
    g = np.zeros(n)
    if shift >= 0:
        m1 = j >= shift
        m2 = (j + shift) < n
        g[m1] += (1.0 - h[j[m1] - shift])
        g[m2] -= h[j[m2] + shift]
    else:
        s = -shift
        m1 = (j + s) < n
        m2 = j >= s
        g[m1] += (1.0 - h[j[m1] + s])
        g[m2] -= h[j[m2] - s]
    return g

_GN = 0
_GTGT = 0.0
_GT = 1.0
_GK = 10

def _obj_lse(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_GK:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_GT * (c - m))
    return (m + np.log(e.sum()) / _GT) * (2.0 / _GN)

def _jac_lse(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    idx = np.argsort(corr)[-_GK:]
    c = corr[idx]
    m = c.max()
    e = np.exp(_GT * (c - m))
    w = e / e.sum()
    grad = np.zeros(_GN)
    for wi, ci in zip(w, idx):
        if wi > 1e-10:
            grad += wi * shift_grad(x, _GN, ci - (_GN - 1))
    return grad * (2.0 / _GN)
