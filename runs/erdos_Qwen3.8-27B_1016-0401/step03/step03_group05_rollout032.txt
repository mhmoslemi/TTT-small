def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def grad_shift(h, n, s):
    """Gradient of corr[s] w.r.t. h"""
    g = np.zeros(n)
    if s >= 0:
        # corr = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
        for i in range(n - s):
            g[i] += (1.0 - h[i + s])
            g[i + s] -= h[i]
    else:
        sp = -s
        for i in range(n - sp):
            g[i + sp] += (1.0 - h[i])
            g[i] -= h[i + sp]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def grad_shift(h, n, s):
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        g[:m] += 1.0 - h[s:]
        g[s:] -= h[:m]
    else:
        sp = -s
        m = n - sp
        g[sp:] += 1.0 - h[:m]
        g[:m] -= h[sp:]
    return g

def fista_optimize(h_init, n, t0, budget_s):
    """FISTA with adaptive log-sum-exp smoothing"""
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    h_y = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    t_k = 1.0
    it = 0
    max_iters = 500000
    
    while it < max_iters:
        if time.time() - t0 > budget_s:
            break
        
        frac = min(it / (0.7 * max_iters), 1.0)
        T = 40.0 * (3.0 / 40.0) ** frac  # 40 -> 3
        
        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh = np.clip(sh, -40, 0)
        e = np.exp(sh)
        sw = e / e.sum()
        
        # Top-K active set
        K = max(8, int(30 * (1 - frac) + 8 * frac))
        if K < n_corr:
            top_idx = np.argpartition(sw, -K)[-K:]
            top_w = sw[top_idx]
            wsum = top_w.sum()
            if wsum > 0:
                top_w /= wsum
            else:
                top_w = np.ones(K) / K
        else:
            top_idx = np.arange(n_corr)
            top_w = sw
        
        grad = np.zeros(n)
        for ki in range(len(top_idx)):
            if top_w[ki] < 1e-9:
                continue
            s = top_idx[ki] - center
            grad += top_w[ki] * grad_shift(h_y, n, s)
        grad *= dx
        
        # Lipschitz estimate
        L = 4.0 * dx  # rough
        
        # FISTA
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h_update = h_y + (t_k - 1.0) / t_next * (h_y - h_prev)
        h_update = np.clip(h_update, 0.0, 1.0)
        
        # Gradient step
        h_new = h_update - (1.0 / L) * grad
        h_new = project_feasible(h_new, n)
        
        h_prev = h_y
        h_y = h_new
        t_k = t_next
        h_avg += (h_new - h_avg) / (it + 1.0)
        it += 1
        
        if it % 10000 == 0:
            c5_a = compute_c5(project_feasible(h_avg, n))
            if c5_a < c5_best:
                c5_best = c5_a
                h_best = h_avg.copy()
            c5_y = compute_c5(h_y)
            if c5_y < c5_best:
                c5_best = c5_y
                h_best = h_y.copy()
        
        # Momentum reset if objective increased
        if it % 5000 == 0:
            if compute_c5(h_y) > c5_best * 1.001:
                t_k = 1.0
                h_prev = h_y.copy()
    
    c5_final = min(compute_c5(h_best), c5_best)
    return h_best, c5_final

def slsqp_topk(h_init, n, K, maxiter, ftol, t0, budget_s):
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        top_idx = np.argpartition(corr, -K)[-K:] if K < n_corr else np.arange(n_corr)
        top_vals = corr[top_idx]
        p = 20
        return (np.sum(np.maximum(top_vals, 1e-15)**p) / K)**(1.0/p) * dx
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        top_idx = np.argpartition(corr, -K)[-K:] if K < n_corr else np.arange(n_corr)
        top_vals = corr[top_idx]
        p = 20
        w = np.maximum(top_vals, 1e-15) ** (p-1)
        wsum = w.sum()
        if wsum < 1e-15:
            w = np.ones(K) / K
        else:
            w /= wsum
        grad = np.zeros(n)
        for ki in range(len(top_idx)):
            if w[ki] < 1e-10:
                continue
            s = top_idx[ki] - center
            grad += w[ki] * grad_shift(x, n, s)
        return grad * dx
    
    def cons_fun(x):
        return x.sum() - target
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h_init, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        return res.x
    except Exception:
        return h_init.copy()

def make_periodic(n, period, fill_ratio):
    """Create periodic 0/1 pattern"""
    h = np.zeros(n)
    ones_per_period = max(1, int(period * fill_ratio))
    for i in range(n):
        if (i % period) < ones_per_period:
            h[i] = 1.0
    return h

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)


def grad_shift(h, n, s):
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        g[:m] += 1.0 - h[s:]
        g[s:] -= h[:m]
    else:
        sp = -s
        m = n - sp
        g[sp:] += 1.0 - h[:m]
        g[:m] -= h[sp:]
    return g


def sum_constraint(x):
    return x.sum() - len(x) / 2.0


def sum_constraint_jac(x):
    return np.ones(len(x))


class SmoothObj:
    def __init__(self, n, T, K):
        self.n = n
        self.T = T
        self.K = K
        self.dx = 2.0 / n
        self.center = n - 1
        self.n_corr = 2 * n - 1

    def __call__(self, x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = self.T * (corr - mc)
        sh = np.clip(sh, -40.0, 0.0)
        e = np.exp(sh)
        return (mc + np.log(e.sum()) / self.T) * self.dx

    def grad(self, x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = self.T * (corr - mc)
        sh = np.clip(sh, -40.0, 0.0)
        e = np.exp(sh)
        sw = e / e.sum()
        K = min(self.K, self.n_corr)
        top_idx = np.argpartition(sw, -K)[-K:]
        top_w = sw[top_idx]
        wsum = top_w.sum()
        if wsum > 1e-15:
            top_w = top_w / wsum
        else:
            top_w = np.ones(K) / K
        g = np.zeros(self.n)
        for ki in range(len(top_idx)):
            if top_w[ki] < 1e-9:
                continue
            s = top_idx[ki] - self.center
            g += top_w[ki] * grad_shift(x, self.n, s)
        return g * self.dx


def slsqp_run(h_init, n, T, K, maxiter, ftol, t0, budget_s):
    fun = SmoothObj(n, T, K)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
    try:
        res = minimize(fun, h_init, method='SLSQP', jac=fun.grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        if time.time() - t0 > budget_s:
            return h_init.copy()
        return res.x
    except Exception:
        return h_init.copy()


def fista_run(h_init, n, t0, budget_s):
    dx = 2.0 / n
    center = n - 1
    n_corr = 2 * n - 1
    h_y = h_init.copy()
    h_prev = h_init.copy()
    h_avg = h_init.copy()
    c5_best = compute_c5(h_init)
    h_best = h_init.copy()
    t_k = 1.0
    it = 0
    max_iters = 600000

    while it < max_iters:
        if time.time() - t0 > budget_s:
            break
        frac = min(it / (0.8 * max_iters), 1.0)
        T = 50.0 * (2.0 / 50.0) ** frac

        corr = np.correlate(h_y, 1.0 - h_y, mode="full")
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh = np.clip(sh, -40.0, 0.0)
        e = np.exp(sh)
        sw = e / e.sum()

        K = max(6, int(25 * (1.0 - frac) + 6 * frac))
        if K < n_corr:
            top_idx = np.argpartition(sw, -K)[-K:]
            top_w = sw[top_idx]
            wsum = top_w.sum()
            if wsum > 1e-15:
                top_w = top_w / wsum
            else:
                top_w = np.ones(K) / K
        else:
            top_idx = np.arange(n_corr)
            top_w = sw

        grad = np.zeros(n)
        for ki in range(len(top_idx)):
            if top_w[ki] < 1e-9:
                continue
            s = top_idx[ki] - center
            grad += top_w[ki] * grad_shift(h_y, n, s)
        grad *= dx

        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        L = 6.0 * dx * (1.0 + T * 0.1)
        step = min(1.0 / L, 2.0 / (T + 1.0))

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h_mom = h_y + ((t_k - 1.0) / t_next) * (h_y - h_prev)
        h_mom = np.clip(h_mom, 0.0, 1.0)

        h_new = h_mom - step * grad
        h_new = project_feasible(h_new, n)

        h_prev = h_y
        h_y = h_new
        t_k = t_next
        h_avg += (h_new - h_avg) / (it + 1.0)
        it += 1

        if it % 8000 == 0:
            c5_a = compute_c5(project_feasible(h_avg, n))
            if c5_a < c5_best:
                c5_best = c5_a
                h_best = h_avg.copy()
            c5_y = compute_c5(h_y)
            if c5_y < c5_best:
                c5_best = c5_y
                h_best = h_y.copy()
            if c5_best > compute_c5(h_init) * 0.995:
                if time.time() - t0 > budget_s * 0.3:
                    break
        if it % 4000 == 0 and c5_best > 0.382:
            t_k = 1.0
            h_prev = h_y.copy()

    return h_best, c5_best


def perturb_h(h, rng, pi, n):
    x = np.arange(n) * (2.0 / n)
    mode = pi % 5
    if mode == 0:
        f = 1.0 + 0.5 * (pi % 4)
        h_p = h + 0.02 * np.sin(2.0 * np.pi * f * x + rng.uniform(0, 6.28))
    elif mode == 1:
        f = 0.5 + 0.3 * (pi % 6)
        h_p = h + 0.015 * np.cos(2.0 * np.pi * f * x + rng.uniform(0, 6.28))
    elif mode == 2:
        nb = 5 + (pi % 4)
        h_p = h.copy()
        blen = n // nb
        for b in range(nb):
            lo_i = b * blen
            hi_i = min(lo_i + blen, n)
            h_p[lo_i:hi_i] += rng.uniform(-0.025, 0.025)
    elif mode == 3:
        h_p = h + rng.normal(0, 0.008 + 0.004 * (pi % 3), n)
    else:
        idx = rng.choice(n, size=n // 10, replace=False)
        h_p = h.copy()
        h_p[idx] += rng.uniform(-0.05, 0.05, len(idx))
    return h_p


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float)
    else:
        n = 280
        h_best = np.zeros(n)
        h_best[::2] = 1.0
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP medium polish
    if time.time() - t0 < budget_s - 500:
        h_new = slsqp_run(h_best, n, 12.0, 15, 3000, 1e-14, t0, budget_s - 450)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 2: FISTA long run
    if time.time() - t0 < budget_s - 350:
        h_f, c5_f = fista_run(h_best, n, t0, budget_s - 300)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f
        h_best = project_feasible(h_best, n)

    # Phase 3: SLSQP tight polish (high T for sharp max)
    if time.time() - t0 < budget_s - 200:
        h_new = slsqp_run(h_best, n, 4.0, 8, 4000, 1e-15, t0, budget_s - 180)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 4: SLSQP very tight
    if time.time() - t0 < budget_s - 120:
        h_new = slsqp_run(h_best, n, 2.0, 5, 3000, 1e-16, t0, budget_s - 100)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 5: Perturbation + local search cycles
    for pi in range(20):
        if time.time() - t0 > budget_s - 80:
            break
        h_p = perturb_h(h_best, rng, pi, n)
        h_p = project_feasible(h_p, n)
        h_new = slsqp_run(h_p, n, 6.0, 10, 1500, 1e-14, t0, budget_s - 60)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_new

    # Phase 6: Try higher resolution
    elapsed = time.time() - t0
    if elapsed < budget_s - 130:
        n2 = n + 48
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_alt = np.interp(x_new, x_old, h_best)
        h_alt = project_feasible(h_alt, n2)
        h_new = slsqp_run(h_alt, n2, 8.0, 12, 3000, 1e-14, t0, budget_s - 80)
        h_new = project_feasible(h_new, n2)
        c5_alt = compute_c5(h_new)
        if c5_alt < c5_best:
            c5_best = c5_alt
            h_best = h_new
            n = n2
        if time.time() - t0 < budget_s - 60:
            h_new2 = slsqp_run(h_alt, n2, 3.0, 6, 2000, 1e-15, t0, budget_s - 40)
            h_new2 = project_feasible(h_new2, n2)
            c5_alt2 = compute_c5(h_new2)
            if c5_alt2 < c5_best:
                c5_best = c5_alt2
                h_best = h_new2
                n = n2

    h_best = project_feasible(h_best, n)
    c5_final = min(compute_c5(h_best), c5_best)
    return (h_best, c5_final, n)
