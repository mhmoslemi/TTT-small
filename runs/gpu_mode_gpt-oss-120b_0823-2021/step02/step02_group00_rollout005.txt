"""
custom_kernel – “outgoing” TriMul (AlphaFold3)

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (FP32, manual – cheap & fast).
2️⃣  Linear projections + gating are done in FP16 with torch.nn.functional.linear.
3️⃣  Optional binary/float mask is applied *in‑place* by a tiny Triton kernel
    (this satisfies the “at least one kernel” requirement).
4️⃣  The cubic contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is reshaped to a batched matrix‑multiply (B·D , N , N) and performed with
    torch.bmm which on Hopper uses Tensor‑cores (FP16→FP32 accumulation).
5️⃣  The result is returned to FP32, a second LayerNorm is applied, multiplied
    by the output‑gate and finally projected back to the original dimension.
All heavy arithmetic is in FP16/FP32; the final tensor is FP32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton mask kernel (applies the same mask to two tensors)
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    a_ptr,          # left  tensor   (B, N, N, H)
    b_ptr,          # right tensor   (B, N, N, H)
    mask_ptr,       # mask (B, N, N) broadcasted over H
    B, N, H,
    stride_az, stride_ay, stride_ax, stride_ah,   # strides for a
    stride_bz, stride_by, stride_bx, stride_bh,   # strides for b
    stride_mz, stride_my, stride_mx,              # strides for mask
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)

    # linear index into the flattened (B,N,N) mask volume
    mask_idx = offs
    mask_m = mask_idx < B * N * N

    # compute (z,y,x) coordinates of the mask element
    z = mask_idx // (N * N)
    rem = mask_idx % (N * N)
    y = rem // N
    x = rem % N

    # load mask value (scalar) and broadcast over H
    mask_val = tl.load(mask_ptr + z * stride_mz + y * stride_my + x * stride_mx,
                       mask=mask_m, other=0.0)

    # broadcast mask to H dimension
    h_offs = tl.arange(0, H)

    # load left/right slices (shape H)
    a = tl.load(a_ptr + z * stride_az + y * stride_ay + x * stride_ax + h_offs * stride_ah,
                mask=mask_m, other=0.0)
    b = tl.load(b_ptr + z * stride_bz + y * stride_by + x * stride_bx + h_offs * stride_bh,
                mask=mask_m, other=0.0)

    # apply mask
    a = a * mask_val
    b = b * mask_val

    # store back
    tl.store(a_ptr + z * stride_az + y * stride_ay + x * stride_ax + h_offs * stride_ah,
             a, mask=mask_m)
    tl.store(b_ptr + z * stride_bz + y * stride_by + x * stride_bx + h_offs * stride_bh,
             b, mask=mask_m)


def _apply_mask_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """
    left, right : (B,N,N,H) fp16 contiguous
    mask        : (B,N,N) fp16 (or bool) contiguous
    """
    B, N, _, H = left.shape
    assert right.shape == (B, N, N, H)
    assert mask.shape == (B, N, N)

    # make sure everything is fp16 contiguous
    left = left.contiguous()
    right = right.contiguous()
    mask = mask.to(left.dtype).contiguous()

    # strides in elements (not bytes)
    a_z, a_y, a_x, a_h = left.stride()
    b_z, b_y, b_x, b_h = right.stride()
    m_z, m_y, m_x = mask.stride()

    total = B * N * N
    BLOCK = 128
    grid = (triton.cdiv(total, BLOCK),)

    apply_mask_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        mask.data_ptr(),
        B, N, H,
        a_z, a_y, a_x, a_h,
        b_z, b_y, b_x, b_h,
        m_z, m_y, m_x,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return left, right


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        mask         : torch.Tensor or None [B, N, N] (float32/bool) – optional
        weights      : dict of leaf weight tensors as used in the reference code
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (manual, FP32)
    # ------------------------------------------------------------------
    eps = 1e-5
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                      # still FP32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections & gates (FP16)
    # ------------------------------------------------------------------
    def _to_fp16(t): return t.to(torch.float16)

    # projection weights: (hidden_dim, dim)
    left_proj_w  = _to_fp16(weights["left_proj.weight"])
    right_proj_w = _to_fp16(weights["right_proj.weight"])
    # gate weights: (hidden_dim, dim)
    left_gate_w  = _to_fp16(weights["left_gate.weight"])
    right_gate_w = _to_fp16(weights["right_gate.weight"])
    out_gate_w   = _to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Mask (optional) – Triton kernel
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure mask is broadcastable and on same device
        mask = mask.to(device)
        left, right = _apply_mask_triton(left, right, mask)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – still FP16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction: batch matrix‑multiply via torch.bmm
    #    left  : (B, N, N, H) -> (B*H, N, N)
    #    right : (B, N, N, H) -> (B*H, N, N)   (need transpose on last two)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_bmm  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_bmm = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # bmm in FP16, accumulation in FP32 (torch does this automatically)
    out_bmm = torch.bmm(left_bmm, right_bmm)   # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_bmm.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)                # back to FP32

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + linear projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)        # broadcast over hidden_dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                # (B,N,N,dim)

    return out
```

**Explanation of decisions**

* **Mask handling** – a dedicated Triton kernel (`apply_mask_kernel`) touches each `(b,i,j)` location once and multiplies both the `left` and `right` tensors by the mask value. This satisfies the requirement that *some* work is performed on the GPU with Triton while keeping the kernel extremely lightweight.

* **Heavy N³ contraction** – reshaping the problem to a batched GEMM `(B·hidden_dim, N, N) × (B·hidden_dim, N, N)ᵀ` lets cuBLAS/Tensor‑cores on the H200 execute the dominant cubic work orders of magnitude faster than a hand‑written Triton GEMM. This pattern is well‑known in AlphaFold‑style code and complies with the “you may keep most of the computation in PyTorch” rule.

* **Precision** – all linear / gating operations are performed in FP16 for bandwidth‑efficiency, while the two LayerNorms and the final projection stay in FP32 for numerical stability, matching the reference implementation.

* **No extra fall‑backs** – the code either runs the Triton mask kernel (if a mask is present) or skips it; there is no hidden Python‑only fallback that would hide kernel failures.
