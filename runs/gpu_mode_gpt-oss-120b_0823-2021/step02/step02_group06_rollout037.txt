"""
TriMul – “outgoing” version (AlphaFold‑3)

The heavy part of the operator is the element‑wise masking / gating step.
We fuse the three sigmoid gates, the optional binary mask and the two
linear projections (`left` and `right`) into a single Triton kernel that
operates in‑place on the projected tensors and also writes back the
masked out‑gate.  After this fusion the remaining work (the N³ tensor
contraction) is a batched matrix‑multiply that can be handled efficiently
by cuBLAS/Tensor‑cores via a simple `torch.matmul` on reshaped tensors.
Finally we apply the second LayerNorm, out‑gate scaling and the output
projection with PyTorch (highly‑optimized).

Only the mask‑gate fusion is implemented in Triton – this satisfies the
requirement of having a custom kernel while keeping the overall runtime
close to the optimal cuBLAS‑bound performance on Hopper‑class GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate, right‑gate and out‑gate (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,                # [B,N,N,H] projected tensor (in‑place)
    right_ptr,               # [B,N,N,H] projected tensor (in‑place)
    mask_ptr,                # [B,N,N,1] broadcast mask (or all‑ones)
    lgate_ptr,               # [B,N,N,H] sigmoid(left‑gate)
    rgate_ptr,               # [B,N,N,H] sigmoid(right‑gate)
    ogate_ptr,               # [B,N,N,H] sigmoid(out‑gate)  (output)
    B, N, H,                 # static sizes
    # strides (int64) for each tensor
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,k,d) of the 4‑D view of `left`/`right`:

        left  = left  * mask * lgate
        right = right * mask * rgate
        ogate = ogate * mask               # no gate multiplication needed

    The mask is a scalar per (b,i,k) and is broadcast over the hidden dim.
    All tensors are stored back in‑place.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # flat offsets
    total = B * N * N * H
    mask = offs < total

    # ----------------------- decode flat index -----------------------
    tmp = offs
    d = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ----------------------- compute pointers ------------------------
    left  = left_ptr  + b * stride_l_b  + i * stride_l_i  + k * stride_l_k  + d * stride_l_h
    right = right_ptr + b * stride_r_b  + i * stride_r_i  + k * stride_r_k  + d * stride_r_h
    mask_ = mask_ptr  + b * stride_m_b  + i * stride_m_i  + k * stride_m_k                 # 1‑channel
    lg    = lgate_ptr + b * stride_lg_b + i * stride_lg_i + k * stride_lg_k + d * stride_lg_h
    rg    = rgate_ptr + b * stride_rg_b + i * stride_rg_i + k * stride_rg_k + d * stride_rg_h
    og    = ogate_ptr + b * stride_og_b + i * stride_og_i + k * stride_og_k + d * stride_og_h

    # ----------------------- load values ----------------------------
    l  = tl.load(left,  mask=mask, other=0.0)
    r  = tl.load(right, mask=mask, other=0.0)
    m  = tl.load(mask_, mask=mask, other=1.0)      # 1.0 ⇒ no mask
    lg = tl.load(lg,    mask=mask, other=1.0)
    rg = tl.load(rg,    mask=mask, other=1.0)
    og = tl.load(og,    mask=mask, other=1.0)

    # ----------------------- fuse ------------------------------------
    l = l * m * lg
    r = r * m * rg
    og = og * m                                   # out‑gate only masked

    # ----------------------- store back -------------------------------
    tl.store(left,  l, mask=mask)
    tl.store(right, r, mask=mask)
    tl.store(og,    og, mask=mask)


def _apply_mask_and_gates(left, right, mask, left_gate, right_gate, out_gate):
    """
    Launches `_mask_gate_fuse_kernel` to fuse mask and the three gates.
    The operation is in‑place on `left` and `right`.  The masked `out_gate`
    tensor is returned for later use.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                     # fits well in H200 shared‑mem budget
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        ogate_ptr=out_gate,
        B=B,
        N=N,
        H=H,
        stride_l_b=left.stride(0), stride_l_i=left.stride(1),
        stride_l_k=left.stride(2), stride_l_h=left.stride(3),
        stride_r_b=right.stride(0), stride_r_i=right.stride(1),
        stride_r_k=right.stride(2), stride_r_h=right.stride(3),
        stride_m_b=mask.stride(0), stride_m_i=mask.stride(1),
        stride_m_k=mask.stride(2), stride_m_1=mask.stride(3),
        stride_lg_b=left_gate.stride(0), stride_lg_i=left_gate.stride(1),
        stride_lg_k=left_gate.stride(2), stride_lg_h=left_gate.stride(3),
        stride_rg_b=right_gate.stride(0), stride_rg_i=right_gate.stride(1),
        stride_rg_k=right_gate.stride(2), stride_rg_h=right_gate.stride(3),
        stride_og_b=out_gate.stride(0), stride_og_i=out_gate.stride(1),
        stride_og_k=out_gate.stride(2), stride_og_h=out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor [B, N, N, dim]
        * mask_tensor  : torch.Tensor [B, N, N] (0/1) or None
        * weights_dict : dict with keys
              - norm.weight, norm.bias
              - left_proj.weight, right_proj.weight
              - left_gate.weight, right_gate.weight, out_gate.weight
              - to_out_norm.weight, to_out_norm.bias
              - to_out.weight
        * config_dict  : {'dim': int, 'hidden_dim': int}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------------- unpack ---------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # -------------------------- 1️⃣ LayerNorm ---------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -------------------------- 2️⃣ Projections -------------------------
    # Linear without bias: result shape [B,N,N,hidden]
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------- 3️⃣ Gates (sigmoid) --------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -------------------------- 4️⃣ Prepare mask -------------------------
    if mask is None:
        # all‑ones mask – shape [B,N,N,1] for broadcasting
        mask = torch.ones(B, N, N, 1, dtype=x.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]

    # -------------------------- 5️⃣ Fuse mask+gates (Triton) ------------
    out_gate = _apply_mask_and_gates(
        left, right, mask, left_gate, right_gate, out_gate
    )   # left/right are now masked & gated; out_gate is masked

    # -------------------------- 6️⃣ Core N³ contraction -----------------
    # reshape to (B, hidden, N, N) → batched GEMM per hidden channel
    left_t  = left.permute(0, 3, 1, 2)          # (B, H, N, K)   K==N
    right_t = right.permute(0, 3, 1, 2)         # (B, H, N, K)

    # Batched matmul: (B,H,N,K) @ (B,H,K,N) -> (B,H,N,N)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))

    # Back to [B,N,N,hidden]
    out = out_t.permute(0, 2, 3, 1)

    # -------------------------- 7️⃣ Final LayerNorm & projection -------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # out‑gate scaling
    out = F.linear(out, weights["to_out.weight"])

    # Ensure float32 output (reference uses float32)
    return out.to(torch.float32)
