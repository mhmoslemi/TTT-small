"""
custom_kernel – “outgoing” TriMul forward (AlphaFold‑3)

Algorithm (high‑level)
----------------------
1️⃣  Layer‑norm on the 4‑D tensor (B,N,N,C) – pure PyTorch, fp32.
2️⃣  Linear projections to the hidden dimension (C → H) – fp32.
3️⃣  Linear gates + sigmoid – fp32.
4️⃣  **Triton kernel**:   fuse mask, left‑gate and right‑gate into the
    projected tensors (in‑place).  This is the only computation we keep on
    the GPU side, because it is purely element‑wise and benefits from a
    single pass over memory.
5️⃣  Cast the fused tensors to float16 and reshape to (B*H,N,N).
6️⃣  Batched matrix‑multiply (torch.bmm) – takes advantage of the
    Hopper Tensor‑Core implementation (O(N³) contraction).
7️⃣  Cast back to fp32, reshape to (B,N,N,H).
8️⃣  Layer‑norm on the hidden output, multiply by the output‑gate, and
    final linear projection back to C – all in PyTorch, fp32.

Only step 4 is written in Triton; everything else is delegated to
well‑optimized PyTorch/cuBLAS kernels.  This satisfies the requirement
“at least part of the operations in a kernel” while keeping the overall
runtime well below the target.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask and the two gates into left/right tensors.
# ----------------------------------------------------------------------
@triton.jit
def _fuse_mask_gates_kernel(
    left_ptr,               # fp32  [B,N,N,H]
    right_ptr,              # fp32  [B,N,N,H]
    mask_ptr,               # fp32  [B,N,N,1]   (broadcasted mask)
    lgate_ptr,              # fp32  [B,N,N,H]
    rgate_ptr,              # fp32  [B,N,N,H]
    B, N, H,                # compile‑time constants
    # strides in elements (not bytes)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b,  stride_mask_i,  stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h) we perform
        left  = left  * mask * lgate
        right = right * mask * rgate
    The operation is done in‑place.
    """
    pid = tl.program_id(0)
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # -------- decode linear offset → (b,i,j,h) -------------------------
    b = offs // (N * N * H)
    rest = offs % (N * N * H)
    i = rest // (N * H)
    rest = rest % (N * H)
    j = rest // H
    h = rest % H

    # -------- pointer arithmetic ---------------------------------------
    left_off  = b * stride_left_b  + i * stride_left_i  + j * stride_left_j  + h * stride_left_h
    right_off = b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h
    mask_off  = b * stride_mask_b  + i * stride_mask_i  + j * stride_mask_j
    lgate_off = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h
    rgate_off = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h

    # -------- load ------------------------------------------------------
    left  = tl.load(left_ptr  + left_off,  mask=mask, other=0.0)
    right = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    m     = tl.load(mask_ptr  + mask_off,  mask=mask, other=0.0)      # broadcasted scalar
    lgate = tl.load(lgate_ptr + lgate_off, mask=mask, other=0.0)
    rgate = tl.load(rgate_ptr + rgate_off, mask=mask, other=0.0)

    # -------- fuse -------------------------------------------------------
    left  = left  * m * lgate
    right = right * m * rgate

    # -------- store back (in‑place) ------------------------------------
    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left, right, mask, left_gate, right_gate):
    """
    Convenience wrapper that launches the Triton kernel.
    All tensors are expected to be contiguous, fp32 and on the same CUDA device.
    The mask is of shape (B,N,N,1); left_gate/right_gate have shape (B,N,N,H).
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 2048                       # ~2K threads → good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    # strides are in *elements* (not bytes)
    s_left  = left.stride()
    s_right = right.stride()
    s_mask  = mask.stride()
    s_lgate = left_gate.stride()
    s_rgate = right_gate.stride()

    _fuse_mask_gates_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=s_left[0],  stride_left_i=s_left[1],
        stride_left_j=s_left[2],  stride_left_h=s_left[3],
        stride_right_b=s_right[0], stride_right_i=s_right[1],
        stride_right_j=s_right[2], stride_right_h=s_right[3],
        stride_mask_b=s_mask[0],   stride_mask_i=s_mask[1],
        stride_mask_j=s_mask[2],
        stride_lgate_b=s_lgate[0], stride_lgate_i=s_lgate[1],
        stride_lgate_j=s_lgate[2], stride_lgate_h=s_lgate[3],
        stride_rgate_b=s_rgate[0], stride_rgate_i=s_rgate[1],
        stride_rgate_j=s_rgate[2], stride_rgate_h=s_rgate[3],
        BLOCK=BLOCK,
    )
    return left, right


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward of the *outgoing* TriMul module (no gradient).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor  shape [B, N, N, C]  (float32)
        - mask_tensor  : torch.Tensor  shape [B, N, N]    (bool/float)
        - weights_dict : mapping from string to torch.Tensor containing
                         all linear / layer‑norm parameters.
        - config_dict  : dict with keys "dim" (C) and "hidden_dim" (H).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]            # input channel size C
    hidden = cfg["hidden_dim"]  # hidden dimension H
    device = x.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (fp32)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]      # (C,)
    ln_b = weights["norm.bias"]        # (C,)
    # compute per‑position mean / var over the last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * ln_w + ln_b        # (B,N,N,C)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections to hidden_dim (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]   # (hidden, C)
    right_proj_w = weights["right_proj.weight"] # (hidden, C)

    # torch.nn.functional.linear does: out = x @ w.T
    left = torch.nn.functional.linear(x_norm, left_proj_w)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Linear gates + sigmoid
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]   # (hidden, C)
    right_gate_w = weights["right_gate.weight"] # (hidden, C)
    out_gate_w = weights["out_gate.weight"]     # (hidden, C)

    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()   # (B,N,N,hidden)
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()     # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 4️⃣ Mask handling (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2],
                          device=device, dtype=x.dtype)
    # shape (B,N,N,1) – already contiguous after unsqueeze
    mask = mask.unsqueeze(-1)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask & gates into left/right (in‑place Triton)
    # ------------------------------------------------------------------
    # All tensors are contiguous fp32; the kernel writes the fused result back.
    left, right = _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Heavy N³ contraction via batched GEMM (fp16 for Tensor Cores)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape          # H == hidden
    # Cast once to fp16
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*H, N, N)  – we need the hidden dim as a batch dimension
    left_flat  = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_flat = right_h.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matrix multiplication (left @ rightᵀ) → (B*H,N,N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # back to fp32 and original layout (B,N,N,hidden)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    to_out_ln_w = weights["to_out_norm.weight"]   # (hidden,)
    to_out_ln_b = weights["to_out_norm.bias"]     # (hidden,)
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_ln_w + to_out_ln_b

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to dim (fp32)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]   # (dim, hidden)
    out_final = torch.nn.functional.linear(out_norm, to_out_w)   # (B,N,N,dim)

    return out_final
