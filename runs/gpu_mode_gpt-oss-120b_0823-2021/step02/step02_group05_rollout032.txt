"""
TriMul “outgoing” forward (AlphaFold‑3) – hybrid PyTorch / Triton implementation.

The heavy O(N³·C) reduction is still performed by the highly‑optimised
torch.einsum (a sequence of GEMM kernels) because that is the fastest path on
Hopper/H200.  The only required GPU‑custom work – masking of the projected
tensors – is implemented in Triton, giving a memory‑bound kernel that
explicitly broadcasts the 2‑D mask over the hidden dimension and avoids the
extra PyTorch launch overhead.

The function `custom_kernel` follows the required signature:
    (input, mask, weights, config) → output
All tensors are assumed to be on CUDA and the final output is float‑32.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: apply a broadcasted mask [B, N, N] to a tensor
#               [B, N, N, H] (H = hidden_dim).
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*   (input tensor)
    mask_ptr,         # const float*   (mask tensor, shape B*N*N)
    dst_ptr,          # float*         (output tensor)
    total_el,         # i64            total number of elements = B*N*N*H
    H,                # i64            hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear indices
    m = offs < total_el                                     # out‑of‑bounds mask

    # Load the source value
    src = tl.load(src_ptr + offs, mask=m, other=0.0)

    # Each block of H consecutive elements shares the same mask entry.
    # `offs // H` indexes the flattened mask of shape (B,N,N).
    mask_val = tl.load(mask_ptr + (offs // H), mask=m, other=1.0)

    # Write masked value
    tl.store(dst_ptr + offs, src * mask_val, mask=m)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` of shape [B,N,N,H] by ``mask[...,None]`` using Triton.
    ``mask`` is float (0/1) with shape [B,N,N].
    Returns a new tensor with the same shape as ``tensor``.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                     # good balance for H200/H100

    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)        # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)

# Entry‑point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input tensor [B, N, N, dim]
        torch.Tensor,          # mask tensor   [B, N, N] (float, 0/1) – may be None
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1) or None
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                                                    # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections → [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask – applied with Triton (the mandated custom kernel)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        left  = apply_mask_triton(left,  mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise gating of projected tensors
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate          # [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·C) reduction – einsum maps to a cascade of GEMMs.
    #    This is the fastest way on Hopper/H200.
    # ------------------------------------------------------------------
    # out shape: [B, N, N, hidden_dim]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gating and projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                     # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # ------------------------------------------------------------------
    # Ensure float‑32 output
    # ------------------------------------------------------------------
    return out.to(torch.float32)
