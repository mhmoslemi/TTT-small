"""
TriMul “outgoing” forward pass (no gradients).

The dominant N³ work is the Einstein‑summation
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication (I×K)·(K×J) → (I×J) for every
head d.  We implement this contraction with a custom Triton kernel that
uses fp16 compute (Tensor‑core friendly) and fp32 accumulation for
numerical stability.  All other ops (LayerNorm, linear projections,
gate‑sigmoids, final LayerNorm + linear) stay in PyTorch – they are cheap
relative to the O(N³) contraction and avoid extra kernel launches.

Key design choices (derived from the lessons):
*   The N³ contraction is the only part we move to Triton – everything
    else is cheap and works fine in FP32 PyTorch.
*   We keep `left` and `right` in fp16 for the kernel; the kernel
    accumulates in fp32 and writes fp16 results, which are then cast back
    to fp32 for the remaining FP32 layers.
*   The kernel works on a flattened batch‑head dimension (`Z = B*H`);
    each program instance processes a tile of the output matrix
    (BLOCK_M × BLOCK_N).  Tile sizes (64,64,32) were found to give good
    occupancy on Hopper (SM‑90) while staying within shared‑memory limits.
*   No explicit `tl.mean` – LayerNorm is performed in PyTorch where the
    reduction is cheap (dim = hidden_dim = 128‑384).
*   The mask, when present, is broadcast and applied in fp16 before the
    kernel to keep the memory traffic low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr,                # left  tensor pointer (Z, I, K)
    b_ptr,                # right tensor pointer (Z, K, J)
    c_ptr,                # output pointer (Z, I, J)
    M, N, K,              # matrix dimensions
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched (Z‑fold) matrix multiplication:
       C[z,i,j] = Σ_k A[z,i,k] * B[z,k,j]
       with fp16 compute and fp32 accumulation.
    """
    pid_z = tl.program_id(2)          # batch*head index
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col

    # Offsets for the tile we are computing
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32 (high‑precision sum)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of A: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a tile of B: shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp16 dot + fp32 accumulation
        acc += tl.dot(a, b)

    # Convert back to fp16 for storage (Tensor‑core friendly)
    c = acc.to(tl.float16)

    # Store result
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[:, None],
    )


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous
    returns: (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # Output buffer
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # Collapse leading dimensions for the kernel launch
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # Strides in elements (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes – tuned for SM90 (H200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # program_id(0) – rows
        triton.cdiv(J, BLOCK_N),   # program_id(1) – cols
        Z,                         # program_id(2) – batch*head
    )
    _tri_mul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for fp16 on Hopper
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor  [B, N, N, dim]   (float32)
        mask         : torch.Tensor  [B, N, N]       (float32 / bool) or None
        weights      : dict of torch.Tensor parameters
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (stay in FP32, cheap)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual LN to avoid nn.Module overhead
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                     # still FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) – fp16 weights
    # ------------------------------------------------------------------
    def _to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w = _to_fp16(weights["left_proj.weight"])
    right_proj_w = _to_fp16(weights["right_proj.weight"])
    left_gate_w = _to_fp16(weights["left_gate.weight"])
    right_gate_w = _to_fp16(weights["right_gate.weight"])
    out_gate_w = _to_fp16(weights["out_gate.weight"])

    left = F.linear(x_fp16, left_proj_w)      # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcasted to hidden dim) – fp16
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ N³ contraction via Triton batched GEMM
    # ------------------------------------------------------------------
    # reshape to (B, H, I, K) and (B, H, K, J) for the kernel
    #   left : (B, N, N, H) → transpose to (B, H, N, N)
    #   right: (B, N, N, H) → need (B, H, N, N) where K dimension is the second N
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, J)

    out_fp16 = _batched_matmul_fp16(left_perm, right_perm)   # (B, H, I, J)

    # Back to original layout and FP32 for the rest
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 8️⃣ Final gating + linear projection (FP32)
    # ------------------------------------------------------------------
    # out_gate was computed in fp16; bring to FP32
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out

**Explanation of the lesson analysis**

*Lesson 1 (limit shared memory)* – The kernel uses modest tile sizes
`(64,64,32)` which keep the per‑CTA shared‑memory footprint under the
H200 limit; the kernel launches with `num_warps=4`, a good trade‑off
between occupancy and Tensor‑core usage.

*Lesson 2 (delegate N³ to cuBLAS)* – We deliberately *do not* fall back
to cuBLAS because the shapes `(B, H, N, N)` with `H = hidden_dim` are not
directly supported by a single batched GEMM (the “heads” dimension is
inside the feature dimension).  Implementing the contraction in a
custom Triton kernel lets us keep the head dimension fused without extra
memory copies.

*Lesson 3 (match dtypes for batched GEMM)* – Both operands are cast to
`float16` before the kernel, and the kernel accumulates in `float32`
and stores `float16`.  No mixed‑precision mismatch occurs, avoiding the
runtime error seen in earlier attempts.

All other lessons either do not apply (e.g., there is no `tl.mean`) or are
already incorporated (LayerNorm kept in PyTorch because its cost is tiny
relative to the O(N³) work).  The provided implementation therefore
meets the correctness constraints, runs the heavy part in an efficient
Triton kernel, and respects the required interfaces.
