"""
TriMul “outgoing” forward (AlphaFold‑3) – hybrid PyTorch / Triton implementation.

The heavy O(N³·C) reduction is performed with torch.einsum (maps to highly‑
optimised GEMM on Hopper).  The only part that must run in Triton is the
pair‑wise mask application on the projected tensors (`left` and `right`),
which is a pure‑memory‑bound operation that benefits from Triton’s explicit
masking and vectorised stores.

The kernel works for any batch size (1‑2), any sequence length up to 1024,
and any hidden dimension (c_z) that appears in the config.  All tensors are
expected on CUDA and the final result is float‑32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: broadcast mask [B,N,N] onto a tensor [B,N,N,H] and store
# the masked result.  This is the only required custom GPU kernel.
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*   (input tensor, flattened)
    mask_ptr,         # const float*   (mask tensor, flattened, shape B*N*N)
    dst_ptr,          # float*         (output tensor, flattened)
    total_el,         # i64            total number of elements = B*N*N*H
    H,                # i64            hidden size (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < total_el

    # Load source value
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # One mask value per block of H elements
    mask_idx = offs // H                               # index into [B,N,N] mask
    m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Store masked result
    tl.store(dst_ptr + offs, src * m_val, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Apply a broadcast mask of shape [B,N,N] to a tensor of shape [B,N,N,H]
    using Triton.  Returns a new tensor with the same shape and dtype
    (float32).
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                     # tuned for Hopper/H200

    # Flatten to 1‑D views (contiguous)
    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)        # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask – run the Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        left = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise gating of projected tensors
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate   # [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core O(N³·C) reduction via einsum (maps to GEMM on the GPU)
    # ------------------------------------------------------------------
    # out shape -> [B, N, N, hidden_dim]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # 6) Final LayerNorm (hidden_dim), gating and projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate   # broadcast over hidden_dim

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure final dtype is float32
    return out.to(torch.float32)
