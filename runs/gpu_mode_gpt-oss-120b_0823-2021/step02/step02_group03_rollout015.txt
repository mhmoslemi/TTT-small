"""
TriMul (outgoing) – fast forward pass
------------------------------------
* LayerNorm, linear projections and the three gating layers are kept in
  PyTorch – they are cheap compared with the O(N³) contraction.
* The heavy contraction

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

  is a batched matrix‑matrix multiplication (B·H independent GEMMs) and is
  performed with a custom Triton kernel.  Each program‑ID works on one
  (batch, hidden) tile and on a 2‑D tile of the (i, j) output matrix.
  The kernel tiles the K‑dimension to keep the shared‑memory footprint
  small, accumulates in FP32 for numerical stability and writes the result
  back as FP16.  The final result is then converted back to FP32, LayerNorm‑
  ed, gated and projected to the original `dim`.

* Optional masking is applied *before* the kernel (broadcasted over the
  hidden dimension) – the kernel therefore only sees dense tensors.
* The implementation works for any `B∈{1,2}`, any `N` up to 1024 and any
  hidden dimension (`hidden_dim`) that fits the weight tensors.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM for each (batch, hidden) slice
# -------------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # const fp16 *  [B, H, N, N]
    right_ptr,         # const fp16 *  [B, H, N, N]
    out_ptr,           #      fp16 *  [B, H, N, N]
    B, N, H,           # int32 sizes
    # strides of left  (B,H,N,N)
    stride_lbh, stride_lhd, stride_lhn, stride_lhk,
    # strides of right (B,H,N,N)
    stride_rbh, stride_rhd, stride_rhn, stride_rhk,
    # strides of out   (B,H,N,N)
    stride_obh, stride_obd, stride_obn, stride_obk,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a tile of the output:

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    Each program-id maps to a unique (b, h, tile_i, tile_j) triplet.
    """
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1️⃣ decode pid → (b, h, tile_i, tile_j)
    # --------------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_M)          # tiles along the i‑axis
    num_j = tl.cdiv(N, BLOCK_N)          # tiles along the j‑axis
    tile_j = pid % num_j
    pid //= num_j
    tile_i = pid % num_i
    pid //= num_i
    h = pid % H
    b = pid // H

    # --------------------------------------------------------------
    # 2️⃣ compute global offsets for this tile
    # --------------------------------------------------------------
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (higher precision than the input fp16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3️⃣ loop over K dimension in blocks of size BLOCK_K
    # --------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, h, i, k]   → shape (BLOCK_M, BLOCK_K)
        l_ptr = (
            left_ptr
            + b * stride_lbh
            + h * stride_lhd
            + offs_i[:, None] * stride_lhn
            + offs_k[None, :] * stride_lhk
        )
        l = tl.load(l_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[b, h, j, k] needs to be transposed → (BLOCK_K, BLOCK_N)
        r_ptr = (
            right_ptr
            + b * stride_rbh
            + h * stride_rhd
            + offs_k[:, None] * stride_rhk
            + offs_j[None, :] * stride_rhn
        )
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # dot‑product over K
        acc += tl.dot(l, r)                     # (M,K) @ (K,N) → (M,N)

    # --------------------------------------------------------------
    # 4️⃣ write the result back (FP16)
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_obh
        + h * stride_obd
        + offs_i[:, None] * stride_obn
        + offs_j[None, :] * stride_obk
    )
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        * x          – (B, N, N, dim)  torch.float32
        * mask       – (B, N, N)       torch.float32 or bool  (may be None)
        * weights    – dictionary that contains all linear‑layer weight/bias
                       tensors exactly as named in the reference PyTorch model
        * config     – {"dim": dim, "hidden_dim": hidden_dim}

    Returns
    -------
    torch.Tensor
        (B, N, N, dim)  torch.float32  – the result of the TriMul forward pass
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm + linear projections (cheap, stay in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 2️⃣ Mask (if present) – broadcast over hidden channel
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)          # (B,N,N,1)
        left  = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 3️⃣ Gating (sigmoid) – also cheap
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ Reshape for the kernel: (B,N,N,H) → (B,H,N,N)   and cast to fp16
    # -----------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)          # (B,H,N,N) fp16

    # -----------------------------------------------------------------
    # 5️⃣ Triton launch – tile sizes tuned for H200 shared‑memory budget
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # total program IDs = B * H * ceil(N/BLOCK_M) * ceil(N/BLOCK_N)
    total_tiles = (
        B * hidden_dim *
        triton.cdiv(N, BLOCK_M) *
        triton.cdiv(N, BLOCK_N)
    )
    grid = (total_tiles,)

    _trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        # strides left
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        # strides right
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        # strides out
        out_fp16.stride(0), out_fp16.stride(1),
        out_fp16.stride(2), out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Post‑kernel: back‑to (B,N,N,hidden), LayerNorm, gating, final linear
    # -----------------------------------------------------------------
    out = out_fp16.to(dtype).permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    out = out * out_gate               # broadcast over N,N
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,dim)

    return out
