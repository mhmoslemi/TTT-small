"""
TriMul “outgoing” forward (AlphaFold3)

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32, manual for speed).
2. Linear projections → hidden_dim and three gating projections
   (all cast to FP16 for the heavy work).
3. Optional mask is applied to the projected tensors.
4. Gating (sigmoid) is applied.
5. Core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is executed as a batched matrix‑multiply (N×N)·(N×N) → (N×N) for every
   (batch, hidden) pair using a Triton kernel.  The kernel works in FP16,
   accumulates in FP32 and writes FP16.
6. Convert to FP32, apply a second LayerNorm, multiply by the output‑gate
   and finish with a linear projection back to the original dimension.

The Triton kernel tiles the (i,j) output plane (BLOCK_M×BLOCK_N) and reduces
over k in tiles of size BLOCK_K.  Tile sizes (64,64,32) are tuned for Hopper
(fp16, 4‑warp).  All strides are passed in element units.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched mat‑mul (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_matmul_kernel(
    a_ptr,            # left  tensor  (B*H, I, K)
    b_ptr,            # right tensor  (B*H, K, J)
    c_ptr,            # output tensor (B*H, I, J)
    M, N, K,          # extents
    stride_az, stride_am, stride_ak,   # a strides (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)          # batch*head dimension
    pid_m = tl.program_id(1)          # tile along M (i)
    pid_n = tl.program_id(2)          # tile along N (j)

    # start offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds handling
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile: (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16×FP16 → FP32 accumulation
        acc += tl.dot(a, b)

    # write result (FP32 → FP16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left : (B, H, I, K)   FP16, contiguous
    right: (B, H, K, J)   FP16, contiguous
    returns (B, H, I, J)  FP16
    """
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2, "inner dims must match"

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse batch*head for the kernel launch
    batch = B * H
    a = left.reshape(batch, I, K)
    b = right.reshape(batch, K, J)
    c = out.reshape(batch, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tile sizes – empirically good for Hopper fp16
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,                               # program id 0  (z)
        triton.cdiv(I, BLOCK_M),             # program id 1  (m)
        triton.cdiv(J, BLOCK_N),             # program id 2  (n)
    )
    _batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # fp16 friendly
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass – PyTorch wrapper that calls the kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward of the outgoing TriMul operator.
    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B,N,N,dim]    float32
        mask         : torch.Tensor [B,N,N] or None (bool/float)
        weights      : dict of leaf tensors:
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        config       : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor [B,N,N,dim] float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (FP32)
    # ------------------------------------------------------------------
    wn = weights["norm.weight"]
    bn = weights["norm.bias"]
    # mean / var across dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + 1e-5)
    x_norm = x_norm * wn + bn                     # still FP32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections & gating (cast to FP16)
    # ------------------------------------------------------------------
    def _fp16(t): return t.to(torch.float16)

    # projection weights: (hidden_dim, dim)
    left_w  = _fp16(weights["left_proj.weight"])
    right_w = _fp16(weights["right_proj.weight"])

    # gate weights: (hidden_dim, dim)
    left_gate_w  = _fp16(weights["left_gate.weight"])
    right_gate_w = _fp16(weights["right_gate.weight"])
    out_gate_w   = _fp16(weights["out_gate.weight"])

    x16 = x_norm.to(torch.float16)                # (B,N,N,dim)

    left  = F.linear(x16, left_w)                 # (B,N,N,hidden)
    right = F.linear(x16, right_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x16, left_gate_w).sigmoid()
    right_gate = F.linear(x16, right_gate_w).sigmoid()
    out_gate   = F.linear(x16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction via Triton batched GEMM
    #   left : (B,N,N,hidden)  -> (B, hidden, N, N)
    #   right: (B,N,N,hidden)  -> (B, hidden, N, N) then transpose K,J
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _batched_matmul_fp16(left_t, right_t)  # (B, H, I, J)

    # reshape back to (B,N,N,hidden) and to FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    w_out_norm = weights["to_out_norm.weight"]
    b_out_norm = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * w_out_norm + b_out_norm

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + linear projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over dim
    w_out = weights["to_out.weight"]
    out = F.linear(out, w_out)                      # (B,N,N,dim)

    return out
