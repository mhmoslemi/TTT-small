"""
TriMul “outgoing” forward pass (AlphaFold3).

*   LayerNorm, linear projections and gating are performed in PyTorch
    (they are cheap compared to the N³ contraction).

*   The costly contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is realised as a batched matrix multiplication
        (B·D, N, N) @ (B·D, N, N)^T
    and executed by a small Triton kernel that
        – works on fp16 for Tensor‑Core speed,
        – accumulates in fp32 for numerical stability,
        – tiles the M×N output with 64×64 blocks and a 32‑wide K‑reduction.
    The kernel is launched with a three‑dimensional grid:
        (tile_i, tile_j, batch·head) → one program per (i‑block, j‑block) of a
        particular (batch, hidden‑dim) slice.

*   After the GEMM we reshape back to the original layout, apply a second
    LayerNorm, an output gate and a final linear projection – all in torch
    (float32).

The implementation works for all shapes listed in the specification,
supports optional masks, and returns a float32 tensor of shape
[bs, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,                     # left  tensor  (Z, I, K)
    b_ptr,                     # right tensor  (Z, K, J)
    c_ptr,                     # output tensor (Z, I, J)
    M, N, K,                   # matrix sizes
    stride_az, stride_am, stride_ak,   # a strides (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)               # batch * hidden_dim index
    pid_m = tl.program_id(0)               # tile row
    pid_n = tl.program_id(1)               # tile col

    # ---------------------------- tile offsets -------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------- K‑reduction ---------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load a tile (M×K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load b tile (K×N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fused dot‑product, acc in fp32
        acc += tl.dot(a, b)

    # write result (cast back to fp16 for storage)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM used by the TriMul contraction.

    Arguments
    ----------
    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous

    Returns
    -------
    out   : (B, H, I, J)   fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # output allocation
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse leading dimensions for the kernel launch
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling tuned for Hopper (SM‑90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # program_id(0) – rows
        triton.cdiv(J, BLOCK_N),   # program_id(1) – cols
        Z,                         # program_id(2) – batch*head
    )

    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor   [B, N, N, dim]   (float32)
        mask         : torch.Tensor   [B, N, N]        (float32/bool) or None
        weights      : dict of parameter tensors
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]   (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (keep in FP32)
    # ------------------------------------------------------------------
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    mean = inp.mean(dim=-1, keepdim=True)
    var = inp.var(dim=-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + eps)
    x = x * w_norm + b_norm                        # [B,N,N,dim] FP32

    # ------------------------------------------------------------------
    # 2. Linear projections (cast to FP16 for the heavy part)
    # ------------------------------------------------------------------
    def _to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w  = _to_fp16(weights["left_proj.weight"])    # (hidden, dim)
    right_proj_w = _to_fp16(weights["right_proj.weight"])

    left_gate_w  = _to_fp16(weights["left_gate.weight"])
    right_gate_w = _to_fp16(weights["right_gate.weight"])
    out_gate_w   = _to_fp16(weights["out_gate.weight"])

    # keep final LN / linear in FP32
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # fp16 copy of the normalized tensor
    x_fp16 = x.to(torch.float16)

    # projections (B,N,N,hidden)
    left  = F.linear(x_fp16, left_proj_w)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask may be bool or float; convert to fp16 and expand
        mask_fp16 = mask.to(torch.float16, non_blocking=True).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton batched GEMM (the N³ contraction)
    #    left  : (B, hidden, N, N)   –> (B, hidden, I, K)
    #    right : (B, hidden, N, N)   –> (B, hidden, K, J)  with K = N, J = N
    # ------------------------------------------------------------------
    # we need (B, hidden, I, K) and (B, hidden, K, J)
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _triton_batched_gemm(left_t, right_t)   # (B, H, I, J) fp16

    # reshape back to (B, N, N, dim) and cast to FP32 for the remaining ops
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm
    # ------------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7. Output gate + final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, to_out_w)                     # (B,N,N,dim) FP32

    return out
