"""
TriMul “outgoing” forward (AlphaFold‑3) – hybrid PyTorch + Triton implementation.

Key ideas
---------
* LayerNorms and linear projections are performed with torch.nn.functional – they are
  already heavily optimised on Hopper (H200) and give the best raw performance.
* The only operation that is *purely memory‑bound* and therefore benefits from an
  explicit Triton kernel is the optional pair‑wise mask broadcast (`mask[...,None]`).
  The kernel multiplies a tensor of shape [B, N, N, H] by a mask of shape [B, N, N]
  without any extra synchronisation or reductions.
* The O(N³·C) reduction (`einsum('... i k d, ... j k d -> ... i j d')`) is expressed
  with torch.einsum – it is internally lowered to a series of GEMM calls that achieve
  near‑peak FLOPs on the H200, which is far faster than a naïve Triton implementation.
* All arithmetic is done in float32 (the weights are stored as float32); optional
  mixed‑precision can be added later but is not required for correctness.

The public entry point `custom_kernel` follows the signature required by the
evaluator.  It unpacks the inputs, applies LayerNorm, linear projections, optional
masking (via Triton), gating, the cubic reduction, a second LayerNorm, gating and a
final linear projection back to the original dimension.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: broadcast mask multiplication
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*   – flattened (B,N,N,H) tensor
    mask_ptr,         # const float*   – flattened (B,N,N) mask
    dst_ptr,          # float*         – output tensor
    total_el,         # i64            – B*N*N*H
    H,                # i64            – hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear element indices
    mask = offs < total_el                             # out‑of‑bounds mask

    # Load source element
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # Every H consecutive elements share the same mask entry.
    # Compute index into the flattened mask (B,N,N) by integer division.
    mask_idx = offs // H
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Store masked result
    tl.store(dst_ptr + offs, src * mval, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` of shape [B,N,N,H] by ``mask[...,None]`` using Triton.
    ``mask`` is a float (0/1) tensor of shape [B,N,N].
    Returns a new tensor with the same shape as ``tensor``.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024  # empirically good for H200 (covers ~4 KB per thread block)

    src = tensor.view(-1)          # 1‑D contiguous view
    dst = torch.empty_like(src)    # allocate output
    mask_flat = mask.view(-1)      # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)  # launch a 1‑D grid

    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry‑point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see reference PyTorch code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional pair‑wise mask (broadcast on hidden_dim) – Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        left  = apply_mask_triton(left,  mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise masking of the projected tensors
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate   # [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·C) reduction via einsum (maps to batched GEMM)
    # ------------------------------------------------------------------
    # out shape [B, N, N, hidden_dim]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm, output gating and final projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate  # broadcast over hidden_dim

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure float32 (all ops above already use fp32)
    return out.to(torch.float32)
