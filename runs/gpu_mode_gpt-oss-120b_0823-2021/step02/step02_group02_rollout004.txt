"""
TriMul “outgoing” operator (AlphaFold3) – mixed PyTorch / Triton implementation.

The expensive part of the forward pass is the 4‑D tensor contraction

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

which is a batched matrix‑multiply over the pairwise dimension *k*.  
We reshape the tensors to **[B·H, N, N]**, launch a single Triton kernel that
computes a tiled GEMM (BLOCK_M × BLOCK_N tiles, iterating over K in
BLOCK_K chunks) with FP16 arithmetic and FP32 accumulation, and finally
reshape the result back to the original layout.  All remaining linear
layers, LayerNorms and gating are performed in regular PyTorch – they are
trivial compared with the O(N³) contraction.

The kernel works for any batch size `B∈{1,2}`, sequence length `N` up to
1024, and hidden dimension `H` (named *hidden_dim* in the config).  It
handles an optional binary mask (broadcasted on the hidden dimension) and
produces a float‑32 output tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM for the contraction
# -------------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr,          # const fp16*  [B*H, N, N]
    right_ptr,         # const fp16*  [B*H, N, N]   (will be transposed)
    out_ptr,           #       fp16*  [B*H, N, N]
    BxH, N,            # integer sizes
    stride_l_batch,    # stride over the (B*H) dimension of left
    stride_l_row,      # stride over the row (i)   of left
    stride_l_col,      # stride over the col (k)   of left
    stride_r_batch,    # stride over the (B*H) dimension of right
    stride_r_row,      # stride over the row (j)   of right
    stride_r_col,      # stride over the col (k)   of right
    stride_o_batch,    # stride over the (B*H) dimension of out
    stride_o_row,      # stride over the row (i)   of out
    stride_o_col,      # stride over the col (j)   of out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a tile (BLOCK_M × BLOCK_N) of the matrix product
        out[bh, i, j] = Σ_k left[bh, i, k] * right[bh, j, k]
    for a given (batch*head) index ``bh``.
    The right operand is accessed transposed so that the inner product is a
    regular dot product over the K dimension.
    """
    pid = tl.program_id(0)                     # linear id over (bh, tile_i, tile_j)

    # ---------------------------------------------------------
    # decode pid → (bh, tile_i, tile_j)
    # ---------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    tile_j = pid % num_j
    pid //= num_j
    tile_i = pid % num_i
    pid //= num_i
    bh = pid                                    # 0 ≤ bh < BxH

    # ---------------------------------------------------------
    # start indices for this tile
    # ---------------------------------------------------------
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (better numeric stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------
    # loop over K dimension in chunks of BLOCK_K
    # ---------------------------------------------------------
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[bh, i, k]  → shape (M, K)
        l_ptr = (left_ptr
                 + bh * stride_l_batch
                 + offs_i[:, None] * stride_l_row
                 + offs_k[None, :] * stride_l_col)
        l = tl.load(l_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[bh, j, k] (we need its transpose k‑j)
        # right is stored as (bh, row=j, col=k); we load (k, j)
        r_ptr = (right_ptr
                 + bh * stride_r_batch
                 + offs_k[:, None] * stride_r_col   # k dimension
                 + offs_j[None, :] * stride_r_row)  # j dimension
        r_t = tl.load(r_ptr,
                      mask=mask_k[:, None] & mask_j[None, :],
                      other=0.0).to(tl.float32)

        # dot‑product over K
        acc += tl.dot(l, r_t)                     # (M,K) @ (K,N) → (M,N)

    # ---------------------------------------------------------
    # write the tile back
    # ---------------------------------------------------------
    o_ptr = (out_ptr
             + bh * stride_o_batch
             + offs_i[:, None] * stride_o_row
             + offs_j[None, :] * stride_o_col)
    tl.store(o_ptr,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,
        torch.Tensor,
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask         : [B, N, N] (bool/float) or None
        - weights      : dict of all linear / layer‑norm weights (float32)
        - config       : dict with ``dim`` and ``hidden_dim`` keys
    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(dtype))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(dtype))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(dtype))
    )
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Reshape to [B*H, N, N] and cast to fp16 for the kernel
    # -----------------------------------------------------------------
    # (B,N,N,H) → (B,N,N,hidden) already; we need (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # output buffer
    out_fp16 = torch.empty_like(left_fp16)         # (B, H, N, N)

    # -----------------------------------------------------------------
    # 6️⃣ Triton GEMM launch
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # total number of (batch*head) combinations
    BH = B * hidden_dim

    # grid size = (BH * tiles_i * tiles_j)
    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    total_tiles = BH * grid_i * grid_j
    grid = (total_tiles,)

    _trimul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        BH,
        N,
        # strides for left  [B*H, N, N]
        left_fp16.stride(0),   # batch*head stride
        left_fp16.stride(2),   # row stride (i)
        left_fp16.stride(3),   # col stride (k)
        # strides for right (will be read transposed)
        right_fp16.stride(0),
        right_fp16.stride(2),  # row stride (j) – used after transposition
        right_fp16.stride(3),  # col stride (k)
        # strides for output
        out_fp16.stride(0),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Reshape back to original layout and remaining PyTorch ops
    # -----------------------------------------------------------------
    # (B, H, N, N) → (B, N, N, H)
    out = out_fp16.to(dtype).permute(0, 2, 3, 1).contiguous()

    # final LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply output gate (broadcast over N,N)
    out = out * out_gate

    # linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
