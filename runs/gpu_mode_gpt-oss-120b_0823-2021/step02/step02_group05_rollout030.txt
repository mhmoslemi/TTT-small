"""
TriMul “outgoing” forward (AlphaFold‑3) – fast Triton‑augmented implementation.

The heavy O(N³·C) reduction is performed with torch.einsum (maps to
high‑performance GEMM on Hopper).  The only part we accelerate with
Triton is the optional pair‑wise mask application on the projected tensors.
The kernel fuses the broadcast of the mask (shape [B,N,N]) over the hidden
dimension H and performs a pure element‑wise multiply – a memory‑bound
operation that Triton can execute with full‑warp stores and no Python
overhead.

All other ops (LayerNorm, Linear, sigmoid gates, final LayerNorm and
output projection) are left to PyTorch for numerical stability and to
reuse the highly‑optimised kernels already present in the CUDA stack.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: apply a broadcast mask of shape [B,N,N] to a tensor
# of shape [B,N,N,H].  The mask is 0/1 (float) and is broadcast on the
# last dimension.  The kernel is launched with a 1‑D grid that covers the
# total number of elements.
# ----------------------------------------------------------------------
@triton.jit
def _mask_apply_kernel(
    src_ptr,          # const float*   (input tensor)
    mask_ptr,         # const float*   (mask tensor, flattened B*N*N)
    dst_ptr,          # float*         (output tensor)
    total_el,         # i64            total number of elements = B*N*N*H
    H,                # i64            hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)           # linear indices
    mask = offs < total_el                              # bounds check

    # Load source values (float32)
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # One mask value per block of H consecutive elements
    mask_idx = offs // H
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Write masked result
    tl.store(dst_ptr + offs, src * mval, mask=mask)


def _apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` of shape [B,N,N,H] element‑wise by ``mask[...,None]``
    using a fused Triton kernel.  ``mask`` is a float (0/1) tensor of shape
    [B,N,N].  Returns a new tensor with the same shape as ``tensor``.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                                 # good trade‑off for H200

    # Flatten to 1‑D views – no extra copy for input
    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)                    # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    _mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry‑point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )                                            # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections → [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask – apply with Triton (the only GPU‑custom part)
    # ------------------------------------------------------------------
    if not nomask:
        left  = _apply_mask_triton(left,  mask)
        right = _apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise gating of projected tensors
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate                     # [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·C) reduction via einsum (maps to fast GEMM)
    # ------------------------------------------------------------------
    # out shape: [B, N, N, hidden_dim]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gating and output projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                         # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure float32 output (all ops already use float32)
    return out.to(torch.float32)
