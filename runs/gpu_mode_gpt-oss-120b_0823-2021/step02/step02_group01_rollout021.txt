"""
TriMul “outgoing” forward (no gradients).

The expensive N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM in Triton:
    left   : (B, H, I, K)   where H = hidden_dim, I = N, K = N
    right  : (B, H, K, J)   where J = N
    out    : (B, H, I, J)   → later reshaped back to (B,N,N,dim)

All lightweight ops (LayerNorm, linear projections, sigmoids, final
LayerNorm + linear) stay in PyTorch.  Mixed‑precision is used for the
core GEMM (fp16 compute, fp32 accumulation, fp16 write‑back) and the
result is cast back to fp32 for the remaining steps.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (I×K)·(K×J) → (I×J) for many heads (hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    a_ptr,                 # left  tensor  (Z, I, K)
    b_ptr,                 # right tensor  (Z, K, J)
    c_ptr,                 # output tensor (Z, I, J)
    M, N, K,               # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a  (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b  (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c  (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # program IDs – 3‑D grid (M‑tile, N‑tile, batch*head)
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)          # row tile index
    pid_n = tl.program_id(1)          # col tile index
    pid_z = tl.program_id(2)          # batch*head index

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (avoid overflow for long reductions)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # reduction over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile   shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile  shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot‑product accumulation (fp32)
        acc += tl.dot(a, b)

    # store result (back to fp16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM used inside TriMul.

    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous
    returns (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # allocate output
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten leading dimensions so kernel sees a 3‑D tensor
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides (in elements, not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling tuned for Hopper – fits in shared memory & uses Tensor‑cores
    BLOCK_M = 64   # rows per block
    BLOCK_N = 64   # cols per block
    BLOCK_K = 32   # reduction block

    grid = (
        triton.cdiv(I, BLOCK_M),   # program_id(0)
        triton.cdiv(J, BLOCK_N),   # program_id(1)
        Z,                         # program_id(2)
    )
    batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for fp16 on SM90
        num_stages=2,              # keep shared‑mem pressure low
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch orchestration + Triton kernel)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]        (float32)
        mask         : torch.Tensor [B, N, N] or None   (float32 or bool)
        weights      : dict of model parameters (float32)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (keep in fp32 for numerical stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                      # shape (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (convert to fp16 for the heavy part)
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = weights["to_out_norm.weight"]   # stay fp32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # fp16 copy of the normalized input
    x_fp16 = x.to(torch.float16)

    # project
    left = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: (B,N,N) → (B,N,N,1) → broadcast over hidden_dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton batched GEMM (the N³ contraction)
    #    left  : (B,N,N,hidden) → (B, H, I, K)
    #    right : (B,N,N,hidden) → (B, H, K, J)  (note the transpose of axes 2 & 3)
    # ------------------------------------------------------------------
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, J)

    out_fp16 = triton_batched_gemm(left_t, right_t)   # (B, H, I, J) fp16

    # reshape back to (B,N,N,dim) and cast to fp32 for the remaining layers
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm
    # ------------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7. Final gating and linear projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, to_out_w)

    return out
