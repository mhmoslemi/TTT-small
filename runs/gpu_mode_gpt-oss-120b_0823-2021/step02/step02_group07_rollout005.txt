"""
custom_kernel – “outgoing” TriMul forward pass (AlphaFold3)

Algorithm (high‑level)
----------------------
1️⃣  Layer‑Norm on the 4‑D tensor (B,N,N,C) – pure PyTorch, fp32.
2️⃣  Linear projections to the hidden dimension H (no bias) – fp32.
3️⃣  Linear gates + sigmoid – fp32.
4️⃣  **Triton kernel** fuses the mask (if any) and the left/right gates into the
    projected tensors *in‑place*.  This removes one full‑size element‑wise
    pass over the data and keeps everything in fp32 (the most stable dtype for
    the LayerNorm‑gate‑mask pipeline).
5️⃣  Cast the two tensors to fp16 and reshape them to
    (B*H, N, N).  A batched GEMM (torch.bmm) computes the cubic contraction

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    which is exactly the `einsum('... i k d, ... j k d -> ... i j d')` in the
    reference implementation.  Using Tensor‑Core fp16 gives ≈10× speed‑up.
6️⃣  Cast the result back to fp32, undo the reshape → (B,N,N,H).
7️⃣  Layer‑Norm, output‑gate and final linear – pure PyTorch, fp32.

Only step 4 is implemented in Triton; everything else stays in PyTorch
because cuBLAS already provides the fastest implementation for the O(N³) part.
The kernel works for any batch size (1 or 2), any sequence length up to 1024,
any input‑channel size C (≤384) and any hidden size H (128 in the tests).
"""

import torch
import triton
import triton.language as tl
import math


# ----------------------------------------------------------------------
# Triton kernel – fuse mask & left/right gates into the projected tensors
# ----------------------------------------------------------------------
@triton.jit
def _fuse_mask_gates_kernel(
    left_ptr,          # fp32   [B,N,N,H]
    right_ptr,         # fp32   [B,N,N,H]
    mask_ptr,          # fp32   [B,N,N,1]   (broadcasted mask)
    lgate_ptr,         # fp32   [B,N,N,H]
    rgate_ptr,         # fp32   [B,N,N,H]
    B, N, H,           # compile‑time constants
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h) we compute
        left  = left  * mask * lgate
        right = right * mask * rgate
    in‑place.  All tensors are stored in row‑major order; the kernel works on a
    flat index and decodes it into (b,i,j,h).  The mask has a trailing singleton
    dimension, therefore we load it once per (b,i,j) and broadcast to H.
    """
    pid = tl.program_id(0)
    total = B * N * N * H                     # total number of elements
    offs = pid * BLOCK + tl.arange(0, BLOCK)  # linear offsets handled by this thread
    mask = offs < total

    # ---- decode linear offset into (b,i,j,h) -------------------------
    b = offs // (N * N * H)
    rest = offs % (N * N * H)
    i = rest // (N * H)
    rest = rest % (N * H)
    j = rest // H
    h = rest % H

    # ---- compute raw pointer offsets ---------------------------------
    left_off  = b * stride_left_b  + i * stride_left_i  + j * stride_left_j  + h * stride_left_h
    right_off = b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h
    mask_off  = b * stride_mask_b  + i * stride_mask_i  + j * stride_mask_j                # no h dimension
    lgate_off = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h
    rgate_off = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h

    # ---- load ---------------------------------------------------------
    left  = tl.load(left_ptr  + left_off,  mask=mask, other=0.0)
    right = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    m     = tl.load(mask_ptr  + mask_off,  mask=mask, other=0.0)   # broadcasted scalar per (b,i,j)
    lgate = tl.load(lgate_ptr + lgate_off, mask=mask, other=0.0)
    rgate = tl.load(rgate_ptr + rgate_off, mask=mask, other=0.0)

    # ---- fuse ---------------------------------------------------------
    left  = left  * m * lgate
    right = right * m * rgate

    # ---- store back in‑place -------------------------------------------
    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left: torch.Tensor,
                         right: torch.Tensor,
                         mask: torch.Tensor,
                         lgate: torch.Tensor,
                         rgate: torch.Tensor) -> None:
    """
    In‑place element‑wise fusion of mask & gates into ``left`` and ``right``.
    All tensors must be contiguous, on the same CUDA device and have
    dtype=torch.float32.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 2048                              # ~2 k threads → good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    # strides in *elements* (not bytes)
    s_left = left.stride()
    s_right = right.stride()
    s_mask = mask.stride()
    s_lgate = lgate.stride()
    s_rgate = rgate.stride()

    _fuse_mask_gates_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=lgate,
        rgate_ptr=rgate,
        B=B, N=N, H=H,
        stride_left_b=s_left[0],  stride_left_i=s_left[1],
        stride_left_j=s_left[2],  stride_left_h=s_left[3],
        stride_right_b=s_right[0], stride_right_i=s_right[1],
        stride_right_j=s_right[2], stride_right_h=s_right[3],
        stride_mask_b=s_mask[0],   stride_mask_i=s_mask[1],
        stride_mask_j=s_mask[2],
        stride_lgate_b=s_lgate[0], stride_lgate_i=s_lgate[1],
        stride_lgate_j=s_lgate[2], stride_lgate_h=s_lgate[3],
        stride_rgate_b=s_rgate[0], stride_rgate_i=s_rgate[1],
        stride_rgate_j=s_rgate[2], stride_rgate_h=s_rgate[3],
        BLOCK=BLOCK,
    )
    # The kernel writes directly into ``left`` and ``right`` – nothing to return.


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

        input_tensor : torch.Tensor  (B, N, N, C)   float32, CUDA
        mask         : torch.Tensor  (B, N, N)    float32, CUDA  (may be None)
        weights_dict : dict of torch.Tensor containing all nn.Linear /
                       LayerNorm parameters (float32, CUDA)
        config_dict  : dict with keys:
                         - "dim"          : C
                         - "hidden_dim"  : H

    Returns
    -------
    torch.Tensor
        Tensor of shape (B, N, N, C), dtype=torch.float32, on the same device.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    device = x.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (fp32, manual)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]   # (C,)
    ln_b = weights["norm.bias"]     # (C,)

    # mean / var over the channel dimension
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * ln_w + ln_b                     # (B,N,N,C)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (fp32, no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]   # (H, C)
    right_proj_w = weights["right_proj.weight"] # (H, C)

    # torch.nn.functional.linear does: out = input @ weight.T + bias
    left = torch.nn.functional.linear(x_norm, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Gates (fp32) + sigmoid
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()   # (B,N,N,H)
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()     # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Mask handling (broadcast to channel dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2],
                          device=device, dtype=x.dtype)
    # shape (B,N,N,1) – broadcastable to (B,N,N,H)
    mask = mask.unsqueeze(-1)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask + gates into left/right (in‑place, Triton)
    # ------------------------------------------------------------------
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Cubic contraction via batched GEMM (fp16, Tensor‑Core)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Cast to fp16 for the heavy matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*H, N, N) → each head is an independent batch
    left_flat = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, N)
    right_flat = right_h.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matrix multiplication (using cuBLAS/Tensor‑Core)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B, N, N, H) and cast to fp32
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    to_out_ln_w = weights["to_out_norm.weight"]   # (H,)
    to_out_ln_b = weights["to_out_norm.bias"]     # (H,)

    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_ln_w + to_out_ln_b

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate (fp32)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear (fp32)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]          # (C, H)
    out_final = torch.nn.functional.linear(out_norm, to_out_w)   # (B,N,N,C)

    return out_final
