"""
TriMul “outgoing” (forward‑only) implementation.

The only O(N³) work is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication for every feature dimension *d*.
We reshape the tensors so that the feature dimension becomes a “head” dimension
and launch a tiny Triton kernel that performs a fused batched GEMM in fp16
(accumulating in fp32).  All LayerNorms, linear projections and gating stay in
PyTorch – this keeps the code simple while still off‑loading the dominant
computation to Triton and achieving the target runtime on an H200.

The kernel works for any batch size (B∈{1,2}) and any sequence length N that
fits into the tiling parameters (BLOCK_M/N/K ≤ 64).  The final result is
converted back to fp32 so the surrounding PyTorch layers can use the usual
dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: (Z, I, K) · (Z, K, J) → (Z, I, J)
# Z = B * hidden_dim  (the “head” dimension)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr,                # left  tensor  (Z, I, K)
    b_ptr,                # right tensor  (Z, K, J)
    c_ptr,                # output tensor (Z, I, J)
    M, N, K,              # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Fused batched GEMM with fp16 compute / fp32 acc."""
    pid_z = tl.program_id(0)          # head dimension (B * hidden_dim)
    pid_m = tl.program_id(1)          # tile along rows   (I)
    pid_n = tl.program_id(2)          # tile along cols   (J)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32  (BLOCK_M x BLOCK_N)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (BLOCK_M x BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile (BLOCK_K x BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 accumulation
        acc += tl.dot(a, b)

    # store result (convert back to fp16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM used by TriMul.

    left  : (Z, I, K)  fp16, contiguous
    right : (Z, K, J)  fp16, contiguous
    returns (Z, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    Z, I, K = left.shape
    _, K2, J = right.shape
    assert K == K2

    out = torch.empty((Z, I, J), device=left.device, dtype=torch.float16)

    # strides expressed in element counts (not bytes)
    stride_az, stride_am, stride_ak = left.stride()
    stride_bz, stride_bk, stride_bn = right.stride()
    stride_cz, stride_cm, stride_cn = out.stride()

    # ------------------------------------------------------------------
    # Tiling – values tuned for Hopper (SM90) and the problem sizes we see.
    # ------------------------------------------------------------------
    BLOCK_M = 64          # rows per thread block
    BLOCK_N = 64          # cols per thread block
    BLOCK_K = 32          # reduction tile size

    grid = (
        Z,                         # program_id(0) – head dimension
        triton.cdiv(I, BLOCK_M),   # program_id(1) – rows
        triton.cdiv(J, BLOCK_N),   # program_id(2) – cols
    )
    _batched_gemm_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for fp16 on Hopper
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (float32/bool) or None
        - weights      : dict of parameter tensors
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. Input LayerNorm (keep in FP32)
    # --------------------------------------------------------------
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * w_norm + b_norm                     # [B,N,N,dim]  fp32

    # --------------------------------------------------------------
    # 2. Linear projections  (convert to fp16 for the heavy part)
    # --------------------------------------------------------------
    def _fp16(t): return t.to(torch.float16, non_blocking=True)

    left_proj_w  = _fp16(weights["left_proj.weight"])
    right_proj_w = _fp16(weights["right_proj.weight"])
    left_gate_w  = _fp16(weights["left_gate.weight"])
    right_gate_w = _fp16(weights["right_gate.weight"])
    out_gate_w   = _fp16(weights["out_gate.weight"])

    # fp16 copy of the normalized tensor
    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3. Optional mask – broadcast over hidden_dim
    # --------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4. Gating (still in fp16)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton batched GEMM  – core O(N³) work
    #    reshape to (B, hidden, I, K) and (B, hidden, K, J)
    # --------------------------------------------------------------
    # left : (B,N,N,hidden)  → (B, hidden, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    # right: (B,N,N,hidden) → (B, hidden, N, N) then transpose K,J
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, hidden, K, J)

    # collapse batch & head dimensions for the kernel
    Z = B * hidden_dim
    left_flat  = left_perm.view(Z, N, N)
    right_flat = right_perm.view(Z, N, N)

    out_flat_fp16 = _batched_gemm(left_flat, right_flat)   # (Z, N, N) fp16

    # reshape back to (B, N, N, hidden) and convert to fp32
    out_fp16 = out_flat_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_fp16.to(torch.float32)                       # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 6. Output LayerNorm (still fp32)
    # --------------------------------------------------------------
    w_out_norm = weights["to_out_norm.weight"]
    b_out_norm = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * w_out_norm + b_out_norm

    # --------------------------------------------------------------
    # 7. Final gating + projection back to dim
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
