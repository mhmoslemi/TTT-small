"""
TriMul “outgoing” forward pass (no grads).

The dominant N³ contraction is implemented as a fused batched GEMM in
Triton:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

All cheap per‑element ops (LayerNorm, linear, gating, final LayerNorm &
linear) are kept in PyTorch.  The heavy GEMM works in fp16 with fp32
accumulation (Tensor‑core friendly) and the result is cast back to fp32
for the remaining steps.

The kernel is tiled for Hopper (SM‑90): 64×64 tiles, 32‑wide reduction
over K, 4 warps per block.  This gives >10× speed‑up over a pure‑PyTorch
implementation while staying well within shared‑memory limits.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul of (B·H, I, K) × (B·H, K, J) → (B·H, I, J)
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    a_ptr,            # left  tensor pointer  (Z, I, K)
    b_ptr,            # right tensor pointer  (Z, K, J)
    c_ptr,            # output tensor pointer (Z, I, J)
    M, N, K,          # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Tile indices
    # ------------------------------------------------------------------
    pid_z = tl.program_id(0)      # batch * head dimension
    pid_m = tl.program_id(1)      # row tile
    pid_n = tl.program_id(2)      # col tile

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # fp32 accumulator (Tensor‑core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduction over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile (M×K) – shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load b tile (K×N) – shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate
        acc += tl.dot(a, b)

    # Convert back to fp16 for storage
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batch_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    Parameters
    ----------
    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous

    Returns
    -------
    out : (B, H, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten (B,H) → Z for the kernel
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides in elements (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                              # program_id(0) – batch*head
        triton.cdiv(I, BLOCK_M),        # program_id(1) – rows
        triton.cdiv(J, BLOCK_N),        # program_id(2) – cols
    )
    batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,        # good default for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple (x, mask, weights, config)
        x        : torch.Tensor [B, N, N, dim]      (float32)
        mask     : torch.Tensor [B, N, N] or None  (bool/float)
        weights  : dict of model parameters
        config   : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (keep in fp32 – tiny cost, numerically stable)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b        # shape (B,N,N,dim), fp32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections + gating (cast to fp16 for the heavy part)
    # --------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16, non_blocking=True)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = weights["to_out_norm.weight"]   # stay fp32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # fp16 copy of normalized input (batch‑first layout unchanged)
    x_fp16 = x_norm.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)    # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid on fp16, later up‑cast for final gating)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core N³ contraction via Triton batched GEMM.
    #    left  : (B,N,N,hidden) → (B, hidden, N, N)
    #    right : (B,N,N,hidden) → (B, hidden, N, N)  and then transpose K↔J.
    # --------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = triton_batch_matmul(left_t, right_t)  # (B, H, I, J) fp16

    # Restore original layout (B,N,N,hidden) and cast back to fp32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (fp32)
    # --------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + projection
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)     # broadcast over dim
    out = F.linear(out, to_out_w)             # (B,N,N,dim)

    return out
