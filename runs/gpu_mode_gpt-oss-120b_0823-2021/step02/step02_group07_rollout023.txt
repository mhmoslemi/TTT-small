"""
custom_kernel – “outgoing” TriMul forward (AlphaFold3)

Algorithm (high‑level)
----------------------
1️⃣ Layer‑norm on the 4‑D input (B,N,N,C) – done in FP32 with the provided
   weight/bias tensors.

2️⃣ Linear projections (C → H) for the **left** and **right** tensors, and
   three gate projections (C → H).  All ops stay in FP32.

3️⃣ Fuse mask (if present), left‑gate and right‑gate into the projected
   tensors with a single Triton kernel.  The kernel works on the flat
   index space (B·N·N·H) and updates *left* and *right* in‑place:
        left  = left  * mask * left_gate
        right = right * mask * right_gate

   The kernel is deliberately simple:
   * one‑dimensional program‑id,
   * block size = 2048 (≈2 k threads → good occupancy on Hopper),
   * only loads/stores float32 values, no shared memory.

4️⃣ Cast the fused tensors to FP16 and reshape to (B*H, N, N) so that the
   cubic contraction becomes a batched GEMM:
        out_flat[b*h] = left_flat[b*h] @ right_flat[b*h].T
   This leverages cuBLAS/Tensor‑Cores and is far faster than any
   Triton‑implemented O(N³) kernel.

5️⃣ Cast the result back to FP32, restore the original layout
   (B,N,N,H).

6️⃣ Second Layer‑norm on the hidden dimension, multiply by the output‑gate
   (produced in step 2) and finally apply the output linear projection
   (H → C).

All heavy element‑wise work (step 3) is executed in a Triton kernel, while
the dominant O(N³) contraction (step 4) uses the highly‑optimised cuBLAS
batched matrix‑multiply.  The implementation respects the required
signatures and works for all test configurations (different sequence lengths,
batch sizes, hidden dimensions, and optional mask).
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: fuse mask + left‑gate + right‑gate into the projected tensors
# ----------------------------------------------------------------------
@triton.jit
def _fuse_mask_gates_kernel(
    left_ptr,          # fp32 * (B,N,N,H)
    right_ptr,         # fp32 * (B,N,N,H)
    mask_ptr,          # fp32 * (B,N,N)   (broadcasted along H)
    left_gate_ptr,     # fp32 * (B,N,N,H)
    right_gate_ptr,    # fp32 * (B,N,N,H)
    B, N, H,           # compile‑time constants
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    pid   = tl.program_id(0)
    offs  = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    mask  = offs < B * N * N * H                       # out‑of‑range guard

    # ------- decode linear offset into (b,i,j,h) -----------------------
    # total layout = (B, N, N, H)  → stride = N*N*H, N*H, H, 1
    tmp   = offs
    b     = tmp // (N * N * H)
    tmp   = tmp %  (N * N * H)
    i     = tmp // (N * H)
    tmp   = tmp %  (N * H)
    j     = tmp // H
    h     = tmp % H

    # ------- compute element pointers ------------------------------------
    left_off  = b * stride_left_b  + i * stride_left_i  + j * stride_left_j  + h * stride_left_h
    right_off = b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h
    mask_off  = b * stride_mask_b  + i * stride_mask_i  + j * stride_mask_j                # no h‑dim
    lgate_off = b * stride_lgate_b  + i * stride_lgate_i  + j * stride_lgate_j  + h * stride_lgate_h
    rgate_off = b * stride_rgate_b  + i * stride_rgate_i  + j * stride_rgate_j  + h * stride_rgate_h

    # ------- load ---------------------------------------------------------
    left   = tl.load(left_ptr   + left_off,   mask=mask, other=0.0)
    right  = tl.load(right_ptr  + right_off,  mask=mask, other=0.0)
    m      = tl.load(mask_ptr   + mask_off,   mask=mask, other=0.0)   # broadcasted mask
    lgate  = tl.load(left_gate_ptr + lgate_off, mask=mask, other=0.0)
    rgate  = tl.load(right_gate_ptr+ rgate_off, mask=mask, other=0.0)

    # ------- fuse ---------------------------------------------------------
    left  = left  * m * lgate
    right = right * m * rgate

    # ------- store back (in‑place) ---------------------------------------
    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left: torch.Tensor,
                         right: torch.Tensor,
                         mask: torch.Tensor,
                         left_gate: torch.Tensor,
                         right_gate: torch.Tensor) -> None:
    """
    In‑place element‑wise fusion:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    All tensors are expected to be contiguous FP32 CUDA tensors.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 2048                         # chosen for good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    # stride in *elements* (not bytes)
    s_left = left.stride()
    s_right = right.stride()
    s_mask = mask.stride()
    s_lgate = left_gate.stride()
    s_rgate = right_gate.stride()

    _fuse_mask_gates_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=s_left[0], stride_left_i=s_left[1],
        stride_left_j=s_left[2], stride_left_h=s_left[3],
        stride_right_b=s_right[0], stride_right_i=s_right[1],
        stride_right_j=s_right[2], stride_right_h=s_right[3],
        stride_mask_b=s_mask[0], stride_mask_i=s_mask[1],
        stride_mask_j=s_mask[2],
        stride_lgate_b=s_lgate[0], stride_lgate_i=s_lgate[1],
        stride_lgate_j=s_lgate[2], stride_lgate_h=s_lgate[3],
        stride_rgate_b=s_rgate[0], stride_rgate_i=s_rgate[1],
        stride_rgate_j=s_rgate[2], stride_rgate_h=s_rgate[3],
        BLOCK=BLOCK,
    )


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module (AlphaFold3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype=torch.float32
        - mask_tensor  : torch.Tensor, shape [B, N, N]   (or None)
        - weights_dict : mapping of the parameter names used in the reference
                         PyTorch implementation to torch.Tensor objects.
        - config_dict  : must contain keys ``dim`` and ``hidden_dim``.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on the same CUDA device).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim    = config["dim"]
    hidden = config["hidden_dim"]          # called ``hidden_dim`` in the paper

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (FP32, manual to avoid torch.nn overhead)
    # ------------------------------------------------------------------
    eps = 1e-5
    ln_w = weights["norm.weight"]   # (dim,)
    ln_b = weights["norm.bias"]     # (dim,)
    # mean/var over the last dimension (C)
    mean = x.mean(-1, keepdim=True)
    var  = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * ln_w + ln_b                # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (C → H) – no bias, FP32
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]
    left  = F.linear(x_norm, left_proj_w)        # (B,N,N,hidden)
    right = F.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Gates (C → H) + sigmoid
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = F.linear(x_norm, left_gate_w).sigmoid()   # (B,N,N,hidden)
    right_gate = F.linear(x_norm, right_gate_w).sigmoid()
    out_gate   = F.linear(x_norm, out_gate_w).sigmoid()    # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 4️⃣ Mask handling (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2],
                          device=device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)          # (B,N,N,1)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask + gates into left/right (in‑place) with Triton
    # ------------------------------------------------------------------
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Heavy N³ contraction via batched GEMM (FP16 → Tensor‑Core)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # Cast to half‑precision for the matmul
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*H, N, N) so each hidden head becomes a separate batch
    left_flat  = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, N)
    right_flat = right_h.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matrix multiplication: Σ_k left[...,i,k,:] * right[...,j,k,:]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # back to FP32 and original layout (B,N,N,hidden)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    to_out_ln_w = weights["to_out_norm.weight"]   # (hidden,)
    to_out_ln_b = weights["to_out_norm.bias"]     # (hidden,)
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_ln_w + to_out_ln_b

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection (hidden → dim) – FP32, no bias
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]          # (dim, hidden)
    output = F.linear(out_norm, to_out_w)        # (B,N,N,dim)

    return output
