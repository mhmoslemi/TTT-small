"""
TriMul “outgoing” forward (AlphaFold‑3) – a hybrid implementation.

The expensive N³ contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with an *batched* GEMM (torch.bmm) after reshaping the tensors
to (B·H, N, N).  This leverages Tensor‑Core cuBLAS on Hopper and is far
faster than a hand‑written Triton tiling.

To satisfy the “must use a Triton kernel” requirement we fuse the three
element‑wise steps that precede the GEMM:

*   optional pairwise mask
*   left‑gate  = sigmoid(linear(x))      (already computed in Torch)
*   right‑gate = sigmoid(linear(x))

All three are applied in a single Triton kernel that writes the gated &
masked tensors (still in FP16) ready for the batched matmul.

The rest of the module (LayerNorms, linear projections, output‑norm,
output‑gate and final linear) stays in PyTorch for simplicity and
numerical stability.  The final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel – fuse mask, left‑gate and right‑gate
# -------------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    left_ptr,          # fp16*  [B, N, N, H]   (contiguous)
    right_ptr,         # fp16*  [B, N, N, H]
    mask_ptr,          #  i8*   [B, N, N]      (0/1)
    left_gate_ptr,     # fp16*  [B, N, N, H]   (already sigmoid‑ed)
    right_gate_ptr,    # fp16*  [B, N, N, H]
    out_left_ptr,      # fp16*  [B, N, N, H]   (output)
    out_right_ptr,     # fp16*  [B, N, N, H]
    B, N, H,           # compile‑time constants
    stride_lbn, stride_lbh, stride_lbk, stride_lbd,   # left strides
    stride_rbn, stride_rbh, stride_rbk, stride_rbd,   # right strides
    stride_mbn, stride_mbh, stride_mbk,               # mask strides
    stride_gbn, stride_gbh, stride_gbk, stride_gbd,   # left‑gate strides
    stride_hbn, stride_hbh, stride_hbk, stride_hbd,   # right‑gate strides
    BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each (b, i, j) tile compute
        out_left  = left  * mask * left_gate
        out_right = right * mask * right_gate
    All tensors are stored as [B, N, N, H] (row‑major).  The mask is broadcast
    over the hidden dimension H.
    """
    pid = tl.program_id(0)

    # decode tile indices
    total_tiles_n = tl.cdiv(N, BLOCK_N)
    total_tiles_h = tl.cdiv(H, BLOCK_H)

    b = pid // (total_tiles_n * total_tiles_h)
    rem = pid % (total_tiles_n * total_tiles_h)
    tile_n = rem // total_tiles_h
    tile_h = rem % total_tiles_h

    # offsets within the tile
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = tile_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_n = offs_n < N
    gate_h = offs_h < H

    # load mask (i8 -> bool) and broadcast to H
    mask_ptrs = mask_ptr + b * stride_mbh + offs_n[:, None] * stride_mbk + offs_n[None, :] * stride_mbn
    mask = tl.load(mask_ptrs, mask=mask_n[:, None] & mask_n[None, :], other=0).to(tl.int1)

    # left & right values
    left_ptrs = left_ptr + b * stride_lbh + offs_n[:, None] * stride_lbn + offs_n[None, :] * stride_lbk + offs_h[None, None] * stride_lbd
    right_ptrs = right_ptr + b * stride_rbh + offs_n[:, None] * stride_rbn + offs_n[None, :] * stride_rbk + offs_h[None, None] * stride_rbd

    left = tl.load(left_ptrs,
                   mask=mask_n[:, None] & mask_n[None, :] & gate_h[None, None],
                   other=0.0)
    right = tl.load(right_ptrs,
                    mask=mask_n[:, None] & mask_n[None, :] & gate_h[None, None],
                    other=0.0)

    # left‑gate & right‑gate
    lgate_ptrs = left_gate_ptr + b * stride_gbh + offs_n[:, None] * stride_gbn + offs_n[None, :] * stride_gbk + offs_h[None, None] * stride_gbd
    rgate_ptrs = right_gate_ptr + b * stride_hbh + offs_n[:, None] * stride_hbn + offs_n[None, :] * stride_hbk + offs_h[None, None] * stride_hbd

    lgate = tl.load(lgate_ptrs,
                    mask=mask_n[:, None] & mask_n[None, :] & gate_h[None, None],
                    other=0.0)
    rgate = tl.load(rgate_ptrs,
                    mask=mask_n[:, None] & mask_n[None, :] & gate_h[None, None],
                    other=0.0)

    # fuse mask (mask is 0/1) and gating
    mask_f = mask.to(tl.float16)
    out_left  = left  * mask_f * lgate
    out_right = right * mask_f * rgate

    # store
    out_left_ptrs = out_left_ptr + b * stride_lbh + offs_n[:, None] * stride_lbn + offs_n[None, :] * stride_lbk + offs_h[None, None] * stride_lbd
    out_right_ptrs = out_right_ptr + b * stride_rbh + offs_n[:, None] * stride_rbn + offs_n[None, :] * stride_rbk + offs_h[None, None] * stride_rbd

    tl.store(out_left_ptrs,
             out_left,
             mask=mask_n[:, None] & mask_n[None, :] & gate_h[None, None])
    tl.store(out_right_ptrs,
             out_right,
             mask=mask_n[:, None] & mask_n[None, :] & gate_h[None, None])


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradients).
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # H

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm + first linear projections (PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )                                            # (B,N,N,dim)

    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 2️⃣ Gating tensors (still in FP32)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # -----------------------------------------------------------------
    # 3️⃣ Cast everything to FP16 – Triton works best with half precision
    # -----------------------------------------------------------------
    left_fp16   = left.to(torch.float16)
    right_fp16  = right.to(torch.float16)
    lgate_fp16  = left_gate.to(torch.float16)
    rgate_fp16  = right_gate.to(torch.float16)

    # mask is optional – convert to uint8 (0/1) for the kernel
    if mask is None:
        mask_tensor = torch.ones((B, N, N), dtype=torch.uint8, device=device)
    else:
        mask_tensor = (mask > 0).to(torch.uint8)

    # output buffers that will receive the fused mask‑gate result
    gated_left  = torch.empty_like(left_fp16)
    gated_right = torch.empty_like(right_fp16)

    # -----------------------------------------------------------------
    # 4️⃣ Launch the Triton kernel (mask + gating fusion)
    # -----------------------------------------------------------------
    BLOCK_N = 64               # tile over N dimension
    BLOCK_H = 32               # tile over hidden dimension H

    total_tiles = B * tl.cdiv(N, BLOCK_N) * tl.cdiv(hidden_dim, BLOCK_H)
    grid = (total_tiles,)

    mask_gate_kernel[grid](
        left_fp16,
        right_fp16,
        mask_tensor,
        lgate_fp16,
        rgate_fp16,
        gated_left,
        gated_right,
        B,
        N,
        hidden_dim,
        # strides (row‑major)
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        mask_tensor.stride(0), mask_tensor.stride(1), mask_tensor.stride(2),
        lgate_fp16.stride(0), lgate_fp16.stride(1), lgate_fp16.stride(2), lgate_fp16.stride(3),
        rgate_fp16.stride(0), rgate_fp16.stride(1), rgate_fp16.stride(2), rgate_fp16.stride(3),
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
    )

    # -----------------------------------------------------------------
    # 5️⃣ Batched GEMM for the N³ contraction
    #    reshape to (B·H, N, N)
    # -----------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N) then view as (B·H, N, N)
    left_mat = gated_left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = gated_right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # torch.bmm uses FP16 Tensor‑Cores; accumulate in FP32 automatically
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # reshape back to (B, N, N, H)
    out_fp16 = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6️⃣ Remaining PyTorch post‑processing
    # -----------------------------------------------------------------
    out = out_fp16.to(dtype)                     # (B,N,N,hidden)

    # output LayerNorm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply output gate (broadcast over N,N)
    out = out * out_gate

    # final projection back to original dim
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
