"""
TriMul “outgoing” forward (AlphaFold‑3) – Triton‑accelerated mask.

The heavy O(N³·C) reduction is performed with a batched matrix‑multiply
(`torch.bmm`).  After reshaping the projected tensors the operation becomes
a standard GEMM (left @ rightᵀ) which is already highly‑optimised on Hopper
(H200).  The only custom GPU work is the optional mask‑application, which is
implemented as a simple element‑wise multiply in a Triton kernel that
broadcasts a [B,N,N] mask over the hidden dimension.

The implementation follows the reference PyTorch code but fuses the mask
with the projected tensors and replaces `einsum` with a batched GEMM to
reach the target ~700 µs runtime.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: apply a broadcast mask of shape [B, N, N] to a tensor
# of shape [B, N, N, H].  The mask is 0/1 (float) and is broadcast on the
# last dimension.
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,        # const float*  input tensor (flattened)
    mask_ptr,       # const float*  mask tensor   (flattened B*N*N)
    dst_ptr,        # float*        output tensor (flattened)
    total_el,       # i64           total number of elements = B*N*N*H
    H,              # i64           hidden dim (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < total_el                             # OOB mask

    # Load the source value
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # Every consecutive H elements share the same mask entry.
    # Integer division by H gives the index into the flattened mask.
    mask_val = tl.load(mask_ptr + (offs // H), mask=mask, other=1.0)

    # Store masked value
    tl.store(dst_ptr + offs, src * mask_val, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` of shape [B,N,N,H] by ``mask[...,None]`` using Triton.
    ``mask`` is float (0/1) with shape [B,N,N].
    Returns a new tensor with the same shape as ``tensor``.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                                 # good balance for H200

    src = tensor.view(-1)                        # flatten
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)                    # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)
    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry‑point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last (dim) channel
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                                 # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask – Triton implementation
    # ------------------------------------------------------------------
    if not nomask:
        left  = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise gating of projected tensors
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(
        x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(
        x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate      # [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·C) reduction – batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    -> reshape to (B*hidden_dim, N, N) and use torch.bmm
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # merge batch and channel dimensions
    left_flat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)

    # GEMM: left @ rightᵀ   (rightᵀ has shape (B*H,N,N) -> (B*H,N,N))
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H,N,N)

    # reshape back to [B,N,N,hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Final LayerNorm (hidden_dim), gating and output projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                       # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)               # guarantee float32 output
