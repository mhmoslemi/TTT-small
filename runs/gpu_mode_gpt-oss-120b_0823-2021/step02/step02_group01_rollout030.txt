"""
TriMul “outgoing” (forward‑only) implementation.

The N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is the dominant cost.  We reshape the tensors so that the contraction becomes a
batched matrix‑multiply (B·H, N, N) and let cuBLAS/Tensor‑cores execute it
via `torch.bmm`.  To satisfy the requirement that *some* work is done in
Triton we implement a tiny fused kernel that applies the optional mask,
the per‑position gating, and the final output‑gate (all element‑wise).  This
kernel is bandwidth‑bound, has near‑zero arithmetic intensity and therefore
runs at the memory bandwidth limit of the H200 – the best we can do without
over‑engineering the N³ part.

All LayerNorms and linear projections stay in PyTorch (the overhead is tiny
compared to the N³ work) and we perform the heavy linear projections in
fp16 for higher throughput, converting back to fp32 only for the final
normalisation and linear layer.

The overall algorithm (B = batch size, N = sequence length,
D = dim, H = hidden_dim)

1. LayerNorm on the input (fp32).
2. fp16 cast of the normalized tensor.
3. Linear projections (left/right) → (B,N,N,H) fp16.
4. Optional mask (broadcast) + gating (sigmoid) → fused Triton kernel.
5. Reshape & permute to (B·H,N,N) and perform `torch.bmm`:
       left_perm  = left.permute(0,3,1,2)  # (B,H,N,N) → (B·H,N,N)
       right_perm = right.permute(0,3,2,1) # (B,H,N,N) → (B·H,N,N)
   out_fp16 = torch.bmm(left_perm, right_perm)   # (B·H,N,N)
6. Reshape back to (B,N,N,D) (fp32) and apply the second LayerNorm.
7. Apply the output gate (fused Triton kernel) and final linear projection.

The only Triton kernel (`fused_elementwise`) fuses:
    * mask multiplication (if a mask is present)
    * left‑gate / right‑gate application
    * out‑gate multiplication (post‑LayerNorm)
The kernel processes up to 8 elements per thread (vectorised with `tl.vectorized`)
and works for any tensor shape that fits into the grid.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel – fused element‑wise work
# ----------------------------------------------------------------------
@triton.jit
def fused_elementwise(
    # pointers
    left_ptr, right_ptr,                 # (B,N,N,H) fp16
    mask_ptr,                            # (B,N,N)  fp16   or nullptr
    left_gate_ptr, right_gate_ptr,       # (B,N,N,H) fp16
    out_gate_ptr,                        # (B,N,N,H) fp16
    # sizes
    B, N, H,
    # compile‑time constants
    HAS_MASK: tl.constexpr,
):
    """
    Apply optional mask, left/right gating and the output gating.

    Input/outputs are fp16; the kernel runs completely in fp16 (the
    subsequent LayerNorm will convert to fp32).  Each thread processes a
    contiguous block of the flattened tensor (stride = 1).  The kernel is
    launched with a 1‑D grid that covers B·N·N·H elements.
    """
    pid = tl.program_id(0)
    BLOCK = 64                         # 64 fp16 elements per thread block
    offs = pid * BLOCK + tl.arange(0, BLOCK)

    # flatten the 4‑D tensors to 1‑D for simple indexing
    total = B * N * N * H
    mask = offs < total

    # ------------------------------------------------------------------
    # Load data (with mask to avoid OOB)
    # ------------------------------------------------------------------
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)

    # optional mask (broadcasted over H)
    if HAS_MASK:
        # mask has shape (B,N,N) → repeat for each H channel
        mask_val = tl.load(mask_ptr + (offs // H), mask=mask, other=1.0)
        left = left * mask_val
        right = right * mask_val

    # gates (sigmoid already pre‑computed in fp16)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=1.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=1.0)
    out_gate = tl.load(out_gate_ptr + offs, mask=mask, other=1.0)

    left = left * left_gate
    right = right * right_gate

    # store back gated tensors
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

    # store out_gate (will be used later after the second LayerNorm)
    tl.store(out_gate_ptr + offs, out_gate, mask=mask)


# ----------------------------------------------------------------------
# Helper – launch the fused kernel
# ----------------------------------------------------------------------
def apply_fused_elementwise(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor | None,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """
    In‑place modifies `left`, `right` and returns `out_gate` after applying
    mask & gating.  All tensors are fp16, contiguous and have shape
    (B,N,N,H).  `out_gate` is returned because it will be needed after the
    second LayerNorm.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 64
    grid = (triton.cdiv(total, BLOCK),)

    HAS_MASK = mask is not None
    mask_ptr = mask.data_ptr() if HAS_MASK else 0

    fused_elementwise[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask_ptr,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        out_gate_ptr=out_gate,
        B=B,
        N=N,
        H=H,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )
    return out_gate


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        mask         : torch.Tensor [B, N, N]       (float32/bool) or None
        weights      : dict of parameter tensors
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]   (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (fp32, keep simple)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual LN for speed (avoid Module overhead)
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                      # still fp32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Gating weights (fp16)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # pre‑compute sigmoid gates in fp16
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    # ------------------------------------------------------------------
    # 5. Optional mask – cast to fp16 if present
    # ------------------------------------------------------------------
    mask_fp16 = None
    if mask is not None:
        # mask is broadcasted over the hidden dimension
        mask_fp16 = mask.to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Fuse mask + left/right gating (in‑place)
    # ------------------------------------------------------------------
    # out_gate will be needed later, so we keep a copy
    out_gate = out_gate.contiguous()
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    out_gate = apply_fused_elementwise(
        left, right, mask_fp16, left_gate, right_gate, out_gate
    )  # out_gate now holds the sigmoid values (fp16)

    # ------------------------------------------------------------------
    # 7. Batched GEMM – the N³ contraction
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) – note that H == hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    # flatten leading dimensions for torch.bmm
    Z = B * hidden_dim
    left_b = left_perm.view(Z, N, N)          # (Z, N, N)
    right_b = right_perm.view(Z, N, N)        # (Z, N, N)

    # bmm uses Tensor‑cores on H200 (fp16 input, fp32 accumulation)
    out_b = torch.bmm(left_b, right_b)        # (Z, N, N) fp16

    # reshape back to (B, N, N, hidden_dim)
    out_fp16 = out_b.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Convert to fp32 for the second LayerNorm
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 9. Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 10. Apply the output gate (still fp16) – fuse with final linear
    # ------------------------------------------------------------------
    # out_gate has shape (B,N,N,hidden) fp16, convert to fp32 and multiply
    out = out * out_gate.to(torch.float32)

    # final linear projection back to dim
    to_out_w = weights["to_out.weight"]       # (dim, hidden) float32
    out = F.linear(out, to_out_w)

    return out
