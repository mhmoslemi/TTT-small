"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (high‑level)
----------------------
1️⃣  Layer‑norm the input (FP32, manual for speed – no extra sync).
2️⃣  Four linear projections (left, right, three gates) are performed in
    FP16, because the dominant work is a large 3‑D contraction.
3️⃣  Optional mask is applied to the left/right projections.
4️⃣  Sigmoid gating is applied.
5️⃣  Core O(N³) contraction  
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]  
    is realized as a *batched* matrix‑multiply:
        left  : (B, H, I, K)   = (B, hidden_dim, N, N)
        right : (B, H, K, J)   = (B, hidden_dim, N, N)ᵀ
    Each (batch,head) pair is independent, so a single Triton kernel
    computes many GEMMs in parallel.  The kernel works in FP16, accumulates
    in FP32 and writes FP16 results.
6️⃣  Result is reshaped back to (B,N,N,hidden_dim) and cast to FP32.
7️⃣  Second Layer‑norm (FP32) + output‑gate (FP32) + final linear projection
    (FP32) give the final tensor of shape (B,N,N,dim).

Design choices (why this is fast on Hopper/H200)
------------------------------------------------
* The contraction is the only O(N³) work; everything else is cheap.
* By fusing the contraction into one Triton kernel we avoid the overhead of
  launching many small cuBLAS calls and we keep all data on‑chip.
* Tile sizes (BLOCK_M = BLOCK_N = 64, BLOCK_K = 32) were chosen because they
  fit well into the shared‑memory budget of SM‑90 and let the hardware use
  Tensor‑cores (fp16×fp16→fp32).
* Strides are passed explicitly – no assumptions about tensor contiguity
  beyond the required layout (B, H, I, K) / (B, H, K, J).
* All other ops stay in PyTorch (LayerNorm, gating, final linear) because
  they are memory‑bound and already highly optimized.
* The kernel runs over a 3‑D grid: (tile‑rows, tile‑cols, batch*head).  This
  matches the “batched GEMM” pattern used in the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched GEMM (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,               # left  tensor  (B*H, I, K)   fp16
    b_ptr,               # right tensor  (B*H, K, J)   fp16
    c_ptr,               # output tensor (B*H, I, J)   fp16
    M, N, K,             # extents
    stride_az, stride_am, stride_ak,   # a strides (batch, i, k)
    stride_bz, stride_bk, stride_bn,   # b strides (batch, k, j)
    stride_cz, stride_cm, stride_cn,   # c strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # program IDs – one program per (tile_row, tile_col, batch_head)
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)          # tile along M dimension
    pid_n = tl.program_id(1)          # tile along N dimension
    pid_z = tl.program_id(2)          # batch*head index

    # --------------------------------------------------------------
    # tile start offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over reduction dimension K in tiles
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile : (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile : (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp16×fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # write result (cast back to fp16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous
    returns: (B, H, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # allocate output
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse batch*head dimension so the kernel sees a 3‑D tensor
    batch = B * H
    a = left.view(batch, I, K)
    b = right.view(batch, K, J)
    c = out.view(batch, I, J)

    # strides in *elements* (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tile sizes – empirically good on Hopper/SM‑90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),          # rows
        triton.cdiv(J, BLOCK_N),          # cols
        batch,                            # one program per (batch,head)
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # fp16 friendly on SM‑90
    )
    return out


# ----------------------------------------------------------------------
# Main entry point – the “outgoing” TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask         : torch.Tensor [B, N, N] or None (bool/float)
        weights      : dict of leaf weight tensors
        config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dim (FP32, manual)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    # mean / var over dim axis
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gating (cast to fp16 for the heavy part)
    # --------------------------------------------------------------
    # helper – cast weight once (they are stored as FP32)
    def _fp16(t): return t.to(torch.float16)

    # projection matrices: (hidden, dim)
    left_proj_w   = _fp16(weights["left_proj.weight"])
    right_proj_w  = _fp16(weights["right_proj.weight"])

    # gate matrices: (hidden, dim)
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])

    # cast activations to fp16
    x_fp16 = x.to(torch.float16)           # (B,N,N,dim)

    # linear projections
    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)  # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden)
    # --------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction: batched GEMM using Triton
    # --------------------------------------------------------------
    # We need tensors shaped (B, hidden, N, N) for the kernel.
    # left : (B,N,N,hidden) -> (B, hidden, N, N)
    left_t  = left.permute(0, 3, 1, 2).contiguous()
    # right : (B,N,N,hidden) -> (B, hidden, N, N) and then transpose K↔J
    right_t = right.permute(0, 3, 2, 1).contiguous()

    # fp16 batched GEMM (B*hidden, N, N) × (B*hidden, N, N)ᵀ → (B*hidden, N, N)
    out_fp16 = _batched_gemm_fp16(left_t, right_t)   # (B, hidden, N, N)

    # back to (B,N,N,hidden) and FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP32)
    # --------------------------------------------------------------
    to_out_ln_w = weights["to_out_norm.weight"]
    to_out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * to_out_ln_w + to_out_ln_b

    # --------------------------------------------------------------
    # 7️⃣ Output gating & final linear (both FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over hidden
    to_out_w = weights["to_out.weight"]            # (dim, hidden)
    out = F.linear(out, to_out_w)                  # (B,N,N,dim)

    return out
```

**Explanation of the lessons applied**

| Lesson | Relevance | Action taken |
|--------|-----------|--------------|
| 1 Limit shared memory | Needed for the large N³ kernel | Tile sizes (64×64×32) keep shared‑memory < 64 KB per SM on SM‑90. |
| 2 Delegate N³ to cuBLAS | Would be fast, but the spec requires “at least part of the ops in a kernel”. Using cuBLAS only would defeat the purpose. | Implemented the contraction as a custom Triton GEMM kernel (still a batched matmul). |
| 3 Match dtypes for batched GEMM | Mixing fp16/fp32 in the kernel caused runtime errors. | All inputs to the kernel are fp16, accumulation in fp32, output cast back to fp16. |
| Stride handling | Previous failures came from assuming 3‑dim strides. | Strides are extracted from the *collapsed* 3‑D view (`a.stride()`, `b.stride()`, `c.stride()`). |
| Mask handling | Mask may be absent. | Guarded with `if mask is not None`. |

The final implementation keeps the heavyweight O(N³) work inside a tightly‑tuned Triton kernel while the surrounding LayerNorms, gating and final projection stay in highly‑optimized PyTorch. This satisfies the correctness, performance and “must‑use‑Triton” requirements.
