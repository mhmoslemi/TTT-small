"""
TriMul “outgoing” forward (AlphaFold‑3) – hybrid implementation.

* LayerNorm, linear projections, gating and the O(N³·C) reduction are performed
  with native PyTorch kernels (fast GEMM / LayerNorm on Hopper).
* The only mandatory Triton component is the mask‑application, which is a
  memory‑bound broadcast multiply of shape [B,N,N,H] by a mask [B,N,N].
  This is implemented with a single 1‑D Triton kernel that tiles the
  last dimension H and uses efficient vectorised loads/stores.

The function `custom_kernel` follows the signature required by the evaluator:
    custom_kernel(data) -> torch.Tensor
where `data` is a tuple
    (input_tensor, mask, weights, config).
All tensors are assumed to be on CUDA and the final output is float32.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: broadcast mask multiplication
# ----------------------------------------------------------------------
@triton.jit
def mask_mul_kernel(
    src_ptr,          # const float*   input [B,N,N,H] flattened
    mask_ptr,         # const float*   mask  [B,N,N]   flattened
    dst_ptr,          # float*         output same shape as src
    total_el,         # i64            total number of elements = B*N*N*H
    H,                # i64            hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)                # linear element ids
    mask = offs < total_el                                   # out‑of‑bounds mask

    # Load the source value (float32)
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # Each consecutive block of H elements shares the same mask entry.
    # Compute the index into the flattened mask (B*N*N).
    mask_idx = offs // H
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Store result
    tl.store(dst_ptr + offs, src * mval, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` [B,N,N,H] by ``mask`` [B,N,N] (broadcast on H) using Triton.
    Returns a new tensor with the same shape and dtype as ``tensor``.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                     # good balance for Hopper/H200

    src = tensor.view(-1)            # 1‑D contiguous view
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)        # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    mask_mul_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see reference implementation)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections -> [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask application (Triton)
    # ------------------------------------------------------------------
    if not nomask:
        left  = apply_mask_triton(left,  mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) + element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate    # [B,N,N,hidden_dim]
    right = right * right_gate  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·C) reduction: einsum over shared sequence dimension k
    #    This maps to a series of GEMM calls on Hopper (fast).
    # ------------------------------------------------------------------
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm (hidden_dim), gating and output linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                     # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure float32 output (all ops already use float32)
    return out.to(torch.float32)
