"""
TriMul “outgoing” operator (AlphaFold3) – mixed PyTorch / Triton implementation
-----------------------------------------------------------------------------

The dominant O(N³) contraction

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

is performed with a single batched GEMM (`torch.bmm`) that lets the
cuBLAS/Tensor‑Core stack on the Hopper H200 handle the heavy work
efficiently.  The remaining element‑wise work (masking, gating and the final
linear projection) is fused into a tiny Triton kernel that:

* applies the optional binary mask,
* multiplies left/right by their respective sigmoid gates,
* multiplies the final output by the output‑gate before the last linear
  projection.

All tensors are kept in FP16 inside the kernel; the final result is returned
as FP32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: element‑wise fusion of mask, gating and final output‑gate
# -------------------------------------------------------------------------
@triton.jit
def trimul_fuse_kernel(
    out_ptr,          # fp16*  [B, H, N, N]   – result of bmm, already transposed to [B,H,N,N]
    out_gate_ptr,    # fp16*  [B, 1, N, N]   – broadcasted output gate
    mask_ptr,        # i1*   [B, N, N]      – optional mask, 0/1 (may be NULL)
    left_gate_ptr,   # fp16*  [B, H, N, N]   – left gate after sigmoid
    right_gate_ptr,  # fp16*  [B, H, N, N]   – right gate after sigmoid
    B, N, H,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    stride_gate_b, stride_gate_h, stride_gate_i, stride_gate_j,
    stride_mask_b, stride_mask_i, stride_mask_j,
    HAS_MASK: tl.constexpr,
):
    """
    For every element (b, h, i, j):
        out = out * left_gate * right_gate
        if mask exists: out = out * mask
        out = out * out_gate
    All ops are in FP16; the kernel writes the fused result back in‑place.
    """
    pid = tl.program_id(0)
    # linear id → (b, h, i, j)
    num_hw = H * N * N
    b = pid // num_hw
    tmp = pid % num_hw
    h = tmp // (N * N)
    tmp = tmp % (N * N)
    i = tmp // N
    j = tmp % N

    # offsets
    out_off = (b * stride_out_b + h * stride_out_h +
               i * stride_out_i + j * stride_out_j)
    lg_off  = (b * stride_gate_b + h * stride_gate_h +
               i * stride_gate_i + j * stride_gate_j)
    rg_off  = (b * stride_gate_b + h * stride_gate_h +
               i * stride_gate_i + j * stride_gate_j)   # same shape as left_gate
    og_off  = (b * stride_gate_b + 0 * stride_gate_h +
               i * stride_gate_i + j * stride_gate_j)   # out_gate broadcast

    out_val = tl.load(out_ptr + out_off).to(tl.float32)
    lg_val  = tl.load(left_gate_ptr + lg_off).to(tl.float32)
    rg_val  = tl.load(right_gate_ptr + rg_off).to(tl.float32)
    og_val  = tl.load(out_gate_ptr + og_off).to(tl.float32)

    out_val = out_val * lg_val * rg_val

    if HAS_MASK:
        mask_off = (b * stride_mask_b + i * stride_mask_i + j * stride_mask_j)
        m = tl.load(mask_ptr + mask_off).to(tl.float32)   # 0.0 or 1.0
        out_val = out_val * m

    out_val = out_val * og_val
    tl.store(out_ptr + out_off, out_val.to(tl.float16))


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input, mask, weights, config)
            - input : [B, N, N, dim]   (float32)
            - mask  : [B, N, N]        (bool/float, may be None)
            - weights: dictionary containing all linear / LN parameters
            - config : {"dim": int, "hidden_dim": int}
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    dtype = torch.float32

    # -----------------------------------------------------------------
    # 1. LayerNorm + linear projections (still in PyTorch, cheap)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )
    # shape: (B,N,N,hidden)
    left = F.linear(x, weights["left_proj.weight"].to(dtype))
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 2. Gating values (sigmoid) – keep in FP16 for the kernel
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(dtype))
    ).to(torch.float16)
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(dtype))
    ).to(torch.float16)
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(dtype))
    ).to(torch.float16)          # shape (B,N,N,hidden)

    # -----------------------------------------------------------------
    # 3. Reshape to [B*H, N, N] and run batched GEMM
    #    left  : (B, H, N, N) → (B*H, N, N)
    #    right : (B, H, N, N) → (B*H, N, N) and transpose last two dims
    # -----------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    BxH, _, _ = left_fp16.shape
    # Batched matrix multiplication (FP16×FP16→FP16, accumulate in FP32)
    out_fp16 = torch.empty_like(left_fp16)        # (B,H,N,N)
    out_fp16 = torch.bmm(
        left_fp16.view(B * hidden_dim, N, N),
        right_fp16.view(B * hidden_dim, N, N).transpose(1, 2)
    ).view(B, hidden_dim, N, N)

    # -----------------------------------------------------------------
    # 4. Fuse mask, gates and output‑gate with a tiny Triton kernel
    # -----------------------------------------------------------------
    has_mask = mask is not None
    if has_mask:
        mask = mask.to(torch.int8)               # 0 / 1 integer mask
    else:
        # create a dummy pointer (won’t be dereferenced)
        mask = torch.empty(0, dtype=torch.int8, device=x.device)

    # out_gate needs to be broadcast to (B,H,N,N) – take first hidden dim slice
    out_gate_broadcast = out_gate.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    # launch kernel
    total_elements = B * hidden_dim * N * N
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    trimul_fuse_kernel[grid](
        out_fp16,
        out_gate_broadcast,
        mask,
        left_gate,
        right_gate,
        B, N, hidden_dim,
        out_fp16.stride(0), out_fp16.stride(1), out_fp16.stride(2), out_fp16.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),  # dummy (same shape)
        mask.stride(0) if has_mask else 0,
        mask.stride(1) if has_mask else 0,
        mask.stride(2) if has_mask else 0,
        HAS_MASK=1 if has_mask else 0,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 5. Back to (B,N,N,hidden) layout and remaining PyTorch ops
    # -----------------------------------------------------------------
    out = out_fp16.to(dtype).permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # final linear back to original dim
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
