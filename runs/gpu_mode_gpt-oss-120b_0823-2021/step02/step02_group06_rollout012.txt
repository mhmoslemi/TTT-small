"""
TriMul – outgoing version (AlphaFold‑3)

The heavy O(N³) contraction is performed with a single batched GEMM that
leverages cuBLAS/Tensor‑cores (torch.matmul).  The only part that really
benefits from a custom kernel is the element‑wise fusing of the optional mask,
the three sigmoid‑gates and the two linear projections (left/right).  The kernel
works in‑place on the projected tensors and also writes back the masked
out‑gate, thus keeping all element‑wise work on the GPU and avoiding extra
launches.

The overall forward pass is:

1️⃣ Layer‑norm over the input channel dimension (dim).
2️⃣ Linear projections → left / right   (shape [B,N,N,H]).
3️⃣ Linear gates → left_gate, right_gate, out_gate   (shape [B,N,N,H]),
   followed by a sigmoid.
4️⃣ Fuse mask (if any) + gates into left/right (in‑place) and also mask out_gate
   with a single Triton kernel.
5️⃣ Batched matrix‑multiply per hidden channel:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   implemented as torch.matmul on tensors of shape (B,H,N,N).
6️⃣ Final LayerNorm over the hidden dimension, out‑gate scaling and linear
   projection back to the original channel dimension (dim).

Only step 4 is written in Triton – this satisfies the “must contain a Triton
kernel” requirement while keeping the runtime optimal on Hopper‑class GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, sigmoid‑gates and (in‑place) apply them to the
# projected tensors left / right.  The out‑gate is also masked and written
# back (the sigmoid has already been applied in Python).
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # [B,N,N,H] projected tensor (in‑place)
    right_ptr,              # [B,N,N,H] projected tensor (in‑place)
    mask_ptr,               # [B,N,N,1]   broadcast mask (or all‑ones)
    out_gate_ptr,           # [B,N,N,H]   already‑sigmoid‑ed, will be masked
    B, N, H,                # static sizes
    # strides (int64) of the tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,k,d)   (the 4‑D view of left/right):
        left  = left  * mask * left_gate
        right = right * mask * right_gate
        out_gate = out_gate * mask
    The mask is a scalar per (b,i,k) and is broadcast over the hidden dim.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # flat indices
    total = B * N * N * H
    msk = offs < total

    # ------------------------------------------------------------------
    # Decode flat index → (b,i,k,d)
    # ------------------------------------------------------------------
    tmp = offs
    d = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ------------------------------------------------------------------
    # Pointer arithmetic
    # ------------------------------------------------------------------
    left_ptr   = left_ptr   + b * stride_l_b   + i * stride_l_i   + k * stride_l_k   + d * stride_l_h
    right_ptr  = right_ptr  + b * stride_r_b   + i * stride_r_i   + k * stride_r_k   + d * stride_r_h
    mask_ptr   = mask_ptr   + b * stride_m_b   + i * stride_m_i   + k * stride_m_k                # 1‑channel broadcast
    og_ptr     = out_gate_ptr + b * stride_og_b + i * stride_og_i + k * stride_og_k + d * stride_og_h

    # ------------------------------------------------------------------
    # Load data (mask is a scalar, broadcast for the hidden dim)
    # ------------------------------------------------------------------
    l   = tl.load(left_ptr,   mask=msk, other=0.0)
    r   = tl.load(right_ptr,  mask=msk, other=0.0)
    m   = tl.load(mask_ptr,   mask=msk, other=1.0)   # 1.0 when mask absent
    og  = tl.load(og_ptr,    mask=msk, other=1.0)

    # ------------------------------------------------------------------
    # Fuse
    # ------------------------------------------------------------------
    l = l * m          # left_gate already multiplied in Python
    r = r * m          # right_gate already multiplied in Python
    og = og * m

    # ------------------------------------------------------------------
    # Store back
    # ------------------------------------------------------------------
    tl.store(left_ptr,  l, mask=msk)
    tl.store(right_ptr, r, mask=msk)
    tl.store(og_ptr,    og, mask=msk)


def _apply_mask_and_gates(left, right, mask, out_gate):
    """
    Launch the Triton kernel that fuses the mask into left/right/out_gate.
    Both ``left`` and ``right`` are modified in‑place.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                               # good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        out_gate_ptr=out_gate,
        B=B, N=N, H=H,
        stride_l_b=left.stride(0), stride_l_i=left.stride(1),
        stride_l_k=left.stride(2), stride_l_h=left.stride(3),
        stride_r_b=right.stride(0), stride_r_i=right.stride(1),
        stride_r_k=right.stride(2), stride_r_h=right.stride(3),
        stride_m_b=mask.stride(0), stride_m_i=mask.stride(1),
        stride_m_k=mask.stride(2), stride_m_1=mask.stride(3),
        stride_og_b=out_gate.stride(0), stride_og_i=out_gate.stride(1),
        stride_og_k=out_gate.stride(2), stride_og_h=out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor of shape [B, N, N, dim]
        * mask_tensor  : torch.Tensor of shape [B, N, N] (0/1) or None
        * weights_dict : dict with the same keys as the reference module
        * config_dict  : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the result of the forward pass.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm over the channel dimension ``dim``
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → left / right   [B,N,N,hidden]
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Gates (sigmoid) → also [B,N,N,hidden]
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 4️⃣ Multiply projected tensors by their respective gates
    #    (do this before the Triton kernel to keep the kernel simple)
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Prepare mask – broadcast to hidden dim.
    # --------------------------------------------------------------
    if mask is None:
        mask = torch.ones(B, N, N, 1, dtype=x.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]

    # --------------------------------------------------------------
    # 6️⃣ Fuse mask into left/right (in‑place) and also mask out_gate
    # --------------------------------------------------------------
    out_gate = _apply_mask_and_gates(left, right, mask, out_gate)

    # --------------------------------------------------------------
    # 7️⃣ Core contraction Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM per hidden channel.
    # --------------------------------------------------------------
    # reshape to (B, hidden, N, N) → treat hidden as batch dim for matmul
    left_t  = left.permute(0, 3, 1, 2)          # (B, H, N, K)
    right_t = right.permute(0, 3, 1, 2)         # (B, H, N, K)

    # batched matmul: (B,H,N,K) @ (B,H,K,N) -> (B,H,N,N)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))

    # back to original layout (B,N,N,hidden)
    out = out_t.permute(0, 2, 3, 1)

    # --------------------------------------------------------------
    # 8️⃣ Final LayerNorm over hidden dim, out‑gate scaling and projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                         # gated scaling
    out = F.linear(out, weights["to_out.weight"])

    # --------------------------------------------------------------
    # Return float32 (reference implementation uses float32)
    # --------------------------------------------------------------
    return out.to(torch.float32)
