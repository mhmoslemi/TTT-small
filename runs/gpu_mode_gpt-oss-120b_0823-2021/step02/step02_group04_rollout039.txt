"""
TriMul – outgoing variant (AlphaFold‑3)

This implementation follows the reference PyTorch code but moves the expensive
element‑wise work (mask‑fusion + three sigmoid gates) into a single Triton
kernel.  All other linear layers and the O(N³) contraction are performed with
high‑performance PyTorch / cuBLAS kernels, which on Hopper are far faster than
hand‑written Triton GEMMs.  The overall algorithm:

0️⃣ LayerNorm over the last dimension (dim) – PyTorch.
1️⃣ Bias‑free linear projections → left / right (shape B,N,N,hdim).
2️⃣ Linear projections for the three gates (pre‑sigmoid) → same shape.
3️⃣ Triton kernel fuses:
       left  = left  * mask * sigmoid(left_gate)
       right = right * mask * sigmoid(right_gate)
       out_gate = sigmoid(out_gate)                (in‑place)
   The mask is broadcasted over the hidden dimension.
4️⃣ Core contraction (batched GEMM):
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched matmul after a permute to (B,hdim,N,N).
5️⃣ Final LayerNorm → element‑wise out‑gate scaling → linear projection back to
   the original dim (all in PyTorch).

The kernel works for float32 and bfloat16 tensors, any batch size {1,2},
any sequence length up to 1024 and arbitrary hidden dimensions (including the
different c / c_z values used by AlphaFold‑3.  The output is always float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: mask + three sigmoid gates → left, right, out_gate (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,               # [B,N,N,H]   float*
    mask_ptr,                          # [B,N,N,1]   float* (broadcasted)
    lgate_ptr, rgate_ptr, ogate_ptr,  # [B,N,N,H]   float* (pre‑sigmoid)
    B, N, H,
    # strides (int64)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k, stride_mask_1,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    stride_ogate_b, stride_ogate_i, stride_ogate_k, stride_ogate_h,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,k,d):
        left  = left  * mask * sigmoid(lgate)
        right = right * mask * sigmoid(rgate)
        ogate = sigmoid(ogate)               (stored back in place)
    The mask is broadcasted over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear index
    total = B * N * N * H
    mask = offs < total

    # ---- decode linear index into (b,i,k,d) ----
    idx = offs
    d = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # ---- compute element pointers ----
    left   = left_ptr   + b * stride_left_b   + i * stride_left_i   + k * stride_left_k   + d * stride_left_h
    right  = right_ptr  + b * stride_right_b  + i * stride_right_i  + k * stride_right_k  + d * stride_right_h
    mask_p = mask_ptr   + b * stride_mask_b   + i * stride_mask_i   + k * stride_mask_k   # 1‑channel
    lgate  = lgate_ptr  + b * stride_lgate_b  + i * stride_lgate_i  + k * stride_lgate_k  + d * stride_lgate_h
    rgate  = rgate_ptr  + b * stride_rgate_b  + i * stride_rgate_i  + k * stride_rgate_k  + d * stride_rgate_h
    ogate  = ogate_ptr  + b * stride_ogate_b  + i * stride_ogate_i  + k * stride_ogate_k  + d * stride_ogate_h

    # ---- load ----
    l  = tl.load(left,  mask=mask, other=0.0)
    r  = tl.load(right, mask=mask, other=0.0)
    m  = tl.load(mask_p, mask=mask, other=1.0)          # 1.0 → no mask
    lg = tl.load(lgate, mask=mask, other=0.0)
    rg = tl.load(rgate, mask=mask, other=0.0)
    og = tl.load(ogate, mask=mask, other=0.0)

    # ---- sigmoid (fast, accurate enough for inference) ----
    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))
    og = 1.0 / (1.0 + tl.exp(-og))

    # ---- fuse ----
    l = l * m * lg
    r = r * m * rg

    # ---- store back ----
    tl.store(left,  l, mask=mask)
    tl.store(right, r, mask=mask)
    tl.store(ogate, og, mask=mask)


def _apply_mask_and_gates(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    lgate: torch.Tensor,
    rgate: torch.Tensor,
    ogate: torch.Tensor,
) -> torch.Tensor:
    """
    Launches the Triton kernel that updates ``left`` and ``right`` in‑place and
    returns ``ogate`` already passed through the sigmoid.
    All tensors are (B,N,N,H) except ``mask`` which is (B,N,N,1).
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                          # 1 K threads → good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left, right,
        mask,
        lgate, rgate, ogate,
        B, N, H,
        # strides
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2), mask.stride(3),
        lgate.stride(0), lgate.stride(1), lgate.stride(2), lgate.stride(3),
        rgate.stride(0), rgate.stride(1), rgate.stride(2), rgate.stride(3),
        ogate.stride(0), ogate.stride(1), ogate.stride(2), ogate.stride(3),
        BLOCK=BLOCK,
    )
    return ogate  # now contains sigmoid(out_gate)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32 or bf16)
        - mask_tensor  : torch.Tensor of shape [B, N, N] or None
        - weights_dict : mapping with keys
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 0️⃣ Input LayerNorm (over dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,dim)

    # --------------------------------------------------------------
    # 1️⃣ Linear projections for left/right (bias‑free)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 2️⃣ Gate projections (pre‑sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"])
    right_gate = F.linear(x, weights["right_gate.weight"])
    out_gate   = F.linear(x, weights["out_gate.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Prepare mask (broadcast to hidden dimension)
    # --------------------------------------------------------------
    if mask is None:
        mask4d = torch.ones((B, N, N, 1), dtype=x.dtype, device=device)
    else:
        mask4d = mask.unsqueeze(-1).to(x.dtype)       # (B,N,N,1)

    # --------------------------------------------------------------
    # 4️⃣ Fuse mask + sigmoid(gates) into left/right (in‑place) and
    #    compute sigmoid(out_gate) → stored back in out_gate
    # --------------------------------------------------------------
    out_gate = _apply_mask_and_gates(left, right, mask4d,
                                    left_gate, right_gate, out_gate)

    # --------------------------------------------------------------
    # 5️⃣ Core contraction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM after permuting to (B,hidden,N,N)
    # --------------------------------------------------------------
    # (B,N,N,hidden) → (B,hidden,N,N)
    left_t  = left.permute(0, 3, 1, 2)          # (B, hidden, N, K)
    right_t = right.permute(0, 3, 1, 2)         # (B, hidden, N, K)

    # batched matrix multiplication (uses cuBLAS Tensor‑cores)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))   # (B, hidden, N, N)

    # back to original layout
    out = out_t.permute(0, 2, 3, 1)            # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 6️⃣ Final LayerNorm → out‑gate scaling → linear projection back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                      # element‑wise gate
    out = F.linear(out, weights["to_out.weight"])

    # guarantee float32 output as required by the spec
    return out.to(torch.float32)
