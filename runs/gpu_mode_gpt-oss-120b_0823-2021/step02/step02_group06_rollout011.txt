"""
custom_kernel – Triton‑accelerated forward pass of the “outgoing” TriMul
operator (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm over the input channel dimension (dim) – PyTorch.
2. Linear projections (no bias) → left / right tensors of shape
   [B, N, N, H] where H = hidden_dim – PyTorch.
3. Compute three sigmoidal gates (left_gate, right_gate, out_gate) – PyTorch.
4. Fuse mask (optional), the three gates and the projected tensors in a
   single Triton kernel:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
        out_gate = out_gate * mask            (mask broadcast over H)
   The kernel works in‑place on ``left`` and ``right`` and writes the masked
   ``out_gate`` back.
5. Compute the core tensor‑contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched matrix‑multiply per hidden channel:
        left  → (B, H, N, K)   with K = N
        right → (B, H, N, K)
        out   = torch.matmul(left, right.transpose(-1,-2))
   This uses cuBLAS/Tensor‑cores (fast on H200).
6. Final LayerNorm over the hidden dimension, multiply by the masked out_gate
   and project back to the original dim – PyTorch.
The function returns a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask + three gates into left/right (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                # [B,N,N,H] projected tensors
    mask_ptr,                           # [B,N,N,1] broadcast mask (or all‑ones)
    lgate_ptr, rgate_ptr, ogate_ptr,    # [B,N,N,H] sigmoid‑gates
    B, N, H,
    # strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """In‑place mask + gate fusion.

    For each element (b,i,k,d):
        left  = left  * mask * lgate
        right = right * mask * rgate
        out_gate = ogate * mask
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # flat element index
    total = B * N * N * H
    mask = offs < total

    # ----- decode flat index into (b,i,k,d) --------------------------
    tmp = offs
    d = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ----- compute pointers -------------------------------------------
    left   = left_ptr   + b * stride_l_b   + i * stride_l_i   + k * stride_l_k   + d * stride_l_h
    right  = right_ptr  + b * stride_r_b   + i * stride_r_i   + k * stride_r_k   + d * stride_r_h
    mask_  = mask_ptr   + b * stride_m_b   + i * stride_m_i   + k * stride_m_k                # scalar per (b,i,k)
    lgate  = lgate_ptr  + b * stride_lg_b  + i * stride_lg_i  + k * stride_lg_k  + d * stride_lg_h
    rgate  = rgate_ptr  + b * stride_rg_b  + i * stride_rg_i  + k * stride_rg_k  + d * stride_rg_h
    ogate  = ogate_ptr  + b * stride_og_b  + i * stride_og_i  + k * stride_og_k  + d * stride_og_h

    # ----- load --------------------------------------------------------
    l = tl.load(left,  mask=mask, other=0.0)
    r = tl.load(right, mask=mask, other=0.0)
    m = tl.load(mask_, mask=mask, other=1.0)          # 1.0 when mask absent
    lg = tl.load(lgate, mask=mask, other=1.0)
    rg = tl.load(rgate, mask=mask, other=1.0)
    og = tl.load(ogate, mask=mask, other=1.0)

    # ----- fuse ---------------------------------------------------------
    l = l * m * lg
    r = r * m * rg
    og = og * m

    # ----- store --------------------------------------------------------
    tl.store(left,  l, mask=mask)
    tl.store(right, r, mask=mask)
    tl.store(ogate, og, mask=mask)


def _apply_mask_and_gates(left, right, mask, lgate, rgate, ogate):
    """
    Launch _mask_gate_fuse_kernel.
    All tensors must be contiguous and on the same CUDA device.
    Returns the (already‑modified) out_gate tensor.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                     # good for H200 (fits in 64 KB shared)
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=lgate,
        rgate_ptr=rgate,
        ogate_ptr=ogate,
        B=B,
        N=N,
        H=H,
        stride_l_b=left.stride(0), stride_l_i=left.stride(1),
        stride_l_k=left.stride(2), stride_l_h=left.stride(3),
        stride_r_b=right.stride(0), stride_r_i=right.stride(1),
        stride_r_k=right.stride(2), stride_r_h=right.stride(3),
        stride_m_b=mask.stride(0), stride_m_i=mask.stride(1),
        stride_m_k=mask.stride(2), stride_m_1=mask.stride(3),
        stride_lg_b=lgate.stride(0), stride_lg_i=lgate.stride(1),
        stride_lg_k=lgate.stride(2), stride_lg_h=lgate.stride(3),
        stride_rg_b=rgate.stride(0), stride_rg_i=rgate.stride(1),
        stride_rg_k=rgate.stride(2), stride_rg_h=rgate.stride(3),
        stride_og_b=ogate.stride(0), stride_og_i=ogate.stride(1),
        stride_og_k=ogate.stride(2), stride_og_h=ogate.stride(3),
        BLOCK=BLOCK,
    )
    return ogate   # now contains masked sigmoid


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of TriMul (outgoing) with a fused Triton kernel.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (fp32/bf16)
        - mask_tensor  : torch.Tensor [B, N, N] (0/1) or None
        - weights_dict : dict with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict  : {"dim": int, "hidden_dim": int}
    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → left / right  [B,N,N,hidden]
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4️⃣ Prepare mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(B, N, N, 1, dtype=x.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(dtype=x.dtype, device=device)   # [B,N,N,1]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask + gates into left/right (in‑place) via Triton
    # ------------------------------------------------------------------
    out_gate = _apply_mask_and_gates(
        left, right, mask, left_gate, right_gate, out_gate
    )   # left/right now contain masked+gated values

    # ------------------------------------------------------------------
    # 6️⃣ Core contraction (batched GEMM per hidden channel)
    #    left  : (B,N,N,hidden) → (B, hidden, N, K)
    #    right : (B,N,N,hidden) → (B, hidden, N, K)
    #    out   = left @ right.T   → (B, hidden, N, N)
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2)          # (B, H, N, K)
    right_t = right.permute(0, 3, 1, 2)         # (B, H, N, K)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))   # (B, H, N, N)

    # back to (B,N,N,hidden)
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7️⃣ Final LayerNorm, out‑gate scaling and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                       # gated scaling
    out = F.linear(out, weights["to_out.weight"])

    # Ensure float32 output as reference implementation expects
    return out.to(torch.float32)
