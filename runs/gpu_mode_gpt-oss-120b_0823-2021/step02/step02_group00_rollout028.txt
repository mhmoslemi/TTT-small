"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (FP32, manual for speed).
2️⃣  Four linear projections (left/right + three gates) are performed in FP16.
3️⃣  Optional mask is applied to the left/right hidden tensors.
4️⃣  Gating (sigmoid) is applied.
5️⃣  The core N³ contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is a batched matrix‑multiply of shape (B·H, N, N)·(B·H, N, N)ᵀ → (B·H, N, N).
    It is executed by a small Triton GEMM kernel that works in FP16,
    accumulates in FP32 and writes FP16.
6️⃣  The result is reshaped back to (B,N,N,dim) and a second Layer‑norm,
    output‑gate and final linear projection (FP32) are applied.

The Triton kernel tiles the I‑ and J‑dimensions with BLOCK_M = BLOCK_N = 64
and the reduction K‑dimension with BLOCK_K = 32 – a good fit for Hopper’s
Tensor‑cores and the 128‑dim hidden size used in the tests.
All other work (norms, linear layers, masking, sigmoid) stays in PyTorch
so we keep the implementation simple while still off‑loading the dominant
O(B·N³·hidden) work to the GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched GEMM (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,               # left  tensor   (batch, I, K)   fp16
    b_ptr,               # right tensor   (batch, K, J)   fp16
    c_ptr,               # output tensor  (batch, I, J)   fp16
    M, N, K,             # matrix extents
    stride_az, stride_am, stride_ak,   # a strides (batch, i, k)
    stride_bz, stride_bk, stride_bn,   # b strides (batch, k, j)
    stride_cz, stride_cm, stride_cn,   # c strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    #  program IDs – each (i‑tile, j‑tile, batch) is a separate program
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)          # tile row   (I)
    pid_n = tl.program_id(1)          # tile col   (J)
    pid_z = tl.program_id(2)          # batch index (B·H)

    # --------------------------------------------------------------
    #  tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    #  reduction over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load tiles of A and B
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16 × FP16 → FP32 accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    #  store result (convert back to FP16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _triton_batch_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous
    returns: (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse the batch*head dimension – the kernel expects a 3‑D tensor
    batch = B * H
    a = left.view(batch, I, K)
    b = right.view(batch, K, J)
    c = out.view(batch, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tile sizes tuned for Hopper (FP16, Tensor‑cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),          # rows
        triton.cdiv(J, BLOCK_N),          # cols
        batch,                            # one program per (B·H)
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                     # fp16‑friendly on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward – PyTorch + Triton
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask         : torch.Tensor or None, shape [B, N, N] (float or bool)
        - weights      : dict of leaf weight tensors (see docstring above)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim], dtype float32.
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]          # = c_z in the paper

    # --------------------------------------------------------------
    # 1️⃣  Input LayerNorm (FP32, manual)
    # --------------------------------------------------------------
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    # mean / var over the last dimension (dim)
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_ln = (x - mean) / torch.sqrt(var + 1e-5)
    x_ln = x_ln * w_norm + b_norm       # still FP32

    # --------------------------------------------------------------
    # 2️⃣  Linear projections & gates (FP16 for speed)
    # --------------------------------------------------------------
    def _fp16(t): return t.to(torch.float16)

    # projection matrices: (hidden, dim)
    w_left  = _fp16(weights["left_proj.weight"])
    w_right = _fp16(weights["right_proj.weight"])

    # gate matrices: (hidden, dim)
    w_lgate = _fp16(weights["left_gate.weight"])
    w_rgate = _fp16(weights["right_gate.weight"])
    w_ogate = _fp16(weights["out_gate.weight"])

    x_f16 = x_ln.to(torch.float16)      # (B,N,N,dim)

    left  = F.linear(x_f16, w_left)    # (B,N,N,hidden)
    right = F.linear(x_f16, w_right)

    # --------------------------------------------------------------
    # 3️⃣  Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        # mask shape -> (B,N,N,1) to broadcast over hidden
        m = mask.to(torch.float16).unsqueeze(-1)
        left  = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣  Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_f16, w_lgate).sigmoid()
    right_gate = F.linear(x_f16, w_rgate).sigmoid()
    out_gate   = F.linear(x_f16, w_ogate).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣  Heavy contraction via Triton GEMM
    #     left  : (B,N,N,hidden)   -> (B,hidden,N,N)  (permute to B,H,I,K)
    #     right : (B,N,N,hidden)   -> (B,hidden,N,N)  (permute to B,H,K,J)
    # --------------------------------------------------------------
    #   We view hidden as “heads” (H) because the kernel expects a batch×head
    #   dimension.  hidden is always 128 in the tests, which fits nicely.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # batched matrix multiplication in FP16
    out_fp16 = _triton_batch_gemm(left_perm, right_perm)   # (B, H, I, J)

    # back to (B, N, N, hidden) and convert to FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣  Output LayerNorm (FP32)
    # --------------------------------------------------------------
    w_outnorm = weights["to_out_norm.weight"]
    b_outnorm = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out_ln = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out_ln = out_ln * w_outnorm + b_outnorm

    # --------------------------------------------------------------
    # 7️⃣  Output gate + final linear projection (FP32)
    # --------------------------------------------------------------
    out_gated = out_ln * out_gate.to(torch.float32)     # broadcast over dim
    w_out = weights["to_out.weight"]                    # (dim, hidden)
    out_final = F.linear(out_gated, w_out)              # (B,N,N,dim)

    return out_final
