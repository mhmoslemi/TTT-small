"""
TriMul “outgoing” forward pass (no gradients).

The expensive N³ contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is performed as a batched GEMM with a small custom Triton kernel
(batched mat‑mul over the hidden dimension d).  All other
operations (LayerNorm, linear projections, gating, final LayerNorm
and linear) stay in PyTorch – they are cheap compared to the
N³ core and avoid the extra bookkeeping required for a full‑kernel
implementation.

The kernel works on fp16 tensors (fp32 accumulation) and returns
fp16 results that are cast back to fp32 for the remaining
PyTorch steps.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (Z, I, K) @ (Z, K, J) -> (Z, I, J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,                # left  matrix pointer   (Z, I, K)
    b_ptr,                # right matrix pointer   (Z, K, J)
    c_ptr,                # output matrix pointer  (Z, I, J)
    M, N, K,              # matrix sizes
    stride_az, stride_am, stride_ak,   # a strides (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # tile identifiers
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)          # row tile
    pid_n = tl.program_id(1)          # col tile
    pid_z = tl.program_id(2)          # batch‑head tile (Z)

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (reduce error)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension in BLOCK_K chunks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile: shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 dot‑product accumulation
        acc += tl.dot(a, b)

    # store result as fp16 (Tensor‑cores produce fp16 output)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[:, None],
    )


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the N³ core.

    Parameters
    ----------
    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous

    Returns
    -------
    out : (B, H, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # allocate output
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten the first two dimensions: Z = B*H
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides (in elements, not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Heuristic block sizes that fit well on Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # program_id(0)
        triton.cdiv(J, BLOCK_N),   # program_id(1)
        Z,                         # program_id(2)
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]       (float32/bool) or None
        - weights      : dict of parameter tensors
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. Input LayerNorm (keep in FP32)
    # --------------------------------------------------------------
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_ln = (x - mean) / torch.sqrt(var + eps)
    x_ln = x_ln * w_norm + b_norm                     # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2. Cast to FP16 for the heavy part
    # --------------------------------------------------------------
    x_fp16 = x_ln.to(torch.float16)

    # Linear projections (bias‑free)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left = F.linear(x_fp16, left_proj_w)              # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        # mask: [B,N,N] → [B,N,N,1] → fp16
        m = mask.to(torch.float16).unsqueeze(-1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction (batched GEMM)
    #    left : (B,N,N,hidden)  → (B,hidden,N,N) → (B,hidden,I= N, K= N)
    #    right: (B,N,N,hidden)  → (B,hidden,N,N) → (B,hidden,K= N, J= N)
    # --------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, J)

    out_fp16 = _batched_gemm_fp16(left_perm, right_perm)   # (B, H, I, J)

    # back to original layout and FP32 for the rest of the pipeline
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,dim)

    # --------------------------------------------------------------
    # 6. Output LayerNorm (FP32)
    # --------------------------------------------------------------
    w_out_ln = weights["to_out_norm.weight"]
    b_out_ln = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * w_out_ln + b_out_ln

    # --------------------------------------------------------------
    # 7. Final gating + projection
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)               # gating
    out = F.linear(out, weights["to_out.weight"])       # final linear
    return out
