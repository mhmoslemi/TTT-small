"""
TriMul (outgoing) – forward only
--------------------------------
1. Layer‑norm on the last dimension (dim) – PyTorch (highly tuned).
2. Two linear projections → left / right  (B,N,N,hidden) – PyTorch.
3. Three “gate” linear layers + sigmoid – PyTorch.
4. Fuse *mask* (optional) and the three gates into left/right/out_gate
   with a single Triton kernel.  The kernel works in‑place on ``left`` and
   ``right`` and also writes the masked ``out_gate`` back.
5. Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM for every hidden channel:
        left  : (B, hidden, N, K)   (K = N)
        right : (B, hidden, N, K)
   We use torch.matmul (cuBLAS/Tensor‑cores) because it is far faster than a
   hand‑written N³ Triton loop on Hopper‑class GPUs.
6. Final LayerNorm over the hidden dimension, out‑gate scaling and a final
   linear projection back to ``dim`` – PyTorch.
The only mandatory Triton part is step 4; everything else stays in
PyTorch to keep the implementation simple and fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse optional mask, left‑gate, right‑gate and out‑gate.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,          # [B,N,N,H] projected tensors
    mask_ptr,                     # [B,N,N,1] broadcast mask (or all‑ones)
    lgate_ptr, rgate_ptr, ogate_ptr,   # [B,N,N,H] sigmoid‑gates
    B, N, H,                      # dimensions
    # strides (int64) of the 6 tensors above
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,k,d)  (the 4‑D view of left/right):
        left  = left  * mask * lgate
        right = right * mask * rgate
        out_gate[b,i,k,d] = ogate * mask
    All ops are element‑wise, no synchronization required.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # flat offsets
    total = B * N * N * H
    mask = offs < total

    # --------------------------------------------------------------
    # Decode flat index → (b,i,k,d)
    # --------------------------------------------------------------
    tmp = offs
    d = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # --------------------------------------------------------------
    # Pointer arithmetic
    # --------------------------------------------------------------
    left  = left_ptr  + b * stride_l_b  + i * stride_l_i  + k * stride_l_k  + d * stride_l_h
    right = right_ptr + b * stride_r_b  + i * stride_r_i  + k * stride_r_k  + d * stride_r_h
    mask_ = mask_ptr  + b * stride_m_b  + i * stride_m_i  + k * stride_m_k               # scalar per (b,i,k)
    lg    = lgate_ptr + b * stride_lg_b + i * stride_lg_i + k * stride_lg_k + d * stride_lg_h
    rg    = rgate_ptr + b * stride_rg_b + i * stride_rg_i + k * stride_rg_k + d * stride_rg_h
    og    = ogate_ptr + b * stride_og_b + i * stride_og_i + k * stride_og_k + d * stride_og_h

    # --------------------------------------------------------------
    # Load (mask is broadcast → scalar)
    # --------------------------------------------------------------
    l = tl.load(left,  mask=mask, other=0.0)
    r = tl.load(right, mask=mask, other=0.0)
    m = tl.load(mask_, mask=mask, other=1.0)          # 1.0 if mask absent
    lgv = tl.load(lg, mask=mask, other=1.0)
    rgv = tl.load(rg, mask=mask, other=1.0)
    ogv = tl.load(og, mask=mask, other=1.0)

    # --------------------------------------------------------------
    # Fuse
    # --------------------------------------------------------------
    l = l * m * lgv
    r = r * m * rgv
    ogv = ogv * m

    # --------------------------------------------------------------
    # Store back
    # --------------------------------------------------------------
    tl.store(left,  l,  mask=mask)
    tl.store(right, r,  mask=mask)
    tl.store(og,    ogv, mask=mask)


def _apply_mask_and_gates(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """
    Launches ``_mask_gate_fuse_kernel``.
    ``left`` and ``right`` are modified in‑place.
    Returns the masked ``out_gate`` (also written back in‑place).
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                                   # fits into 64 KB SMEM on H200
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        ogate_ptr=out_gate,
        B=B,
        N=N,
        H=H,
        stride_l_b=left.stride(0), stride_l_i=left.stride(1),
        stride_l_k=left.stride(2), stride_l_h=left.stride(3),
        stride_r_b=right.stride(0), stride_r_i=right.stride(1),
        stride_r_k=right.stride(2), stride_r_h=right.stride(3),
        stride_m_b=mask.stride(0), stride_m_i=mask.stride(1),
        stride_m_k=mask.stride(2), stride_m_1=mask.stride(3),
        stride_lg_b=left_gate.stride(0), stride_lg_i=left_gate.stride(1),
        stride_lg_k=left_gate.stride(2), stride_lg_h=left_gate.stride(3),
        stride_rg_b=right_gate.stride(0), stride_rg_i=right_gate.stride(1),
        stride_rg_k=right_gate.stride(2), stride_rg_h=right_gate.stride(3),
        stride_og_b=out_gate.stride(0), stride_og_i=out_gate.stride(1),
        stride_og_k=out_gate.stride(2), stride_og_h=out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor  [B, N, N, dim]   (float32 / bf16)
        * mask_tensor  : torch.Tensor  [B, N, N] or None
        * weights_dict : dict with keys matching the reference implementation
        * config_dict  : dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor  [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over ``dim``
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections → left / right  (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gates + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4️⃣ Prepare mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(B, N, N, 1, dtype=x.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask + gates into left/right (in‑place) via Triton.
    #    The kernel also writes the masked out_gate back.
    # ------------------------------------------------------------------
    out_gate = _apply_mask_and_gates(
        left, right, mask, left_gate, right_gate, out_gate
    )   # left/right now contain masked & gated values

    # ------------------------------------------------------------------
    # 6️⃣ Core N³ contraction – batched GEMM per hidden channel.
    #    Shape: (B, hidden, N, N)
    # ------------------------------------------------------------------
    # Permute to (B, hidden, N, N) for batched matmul
    left_t  = left.permute(0, 3, 1, 2)    # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2)   # (B, H, N, N)

    # Batched matmul: (B,H,N,N) @ (B,H,N,N).transpose(-1,-2) → (B,H,N,N)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))

    # Back to original layout (B,N,N,hidden)
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7️⃣ Final LayerNorm, out‑gate scaling, linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                         # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])  # (B,N,N,dim)

    # Cast to float32 as required by the reference implementation
    return out.to(torch.float32)
