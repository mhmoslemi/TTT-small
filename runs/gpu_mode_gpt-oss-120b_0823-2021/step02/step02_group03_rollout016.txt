"""
custom_kernel – TriMul “outgoing” operator (AlphaFold‑3).

The tensor shape is (B, N, N, D) where D is the model dimension.
The forward pass consists of

1️⃣  Layer‑norm over the last dimension.
2️⃣  Four linear projections (left/right values and three gates).
3️⃣  Optional mask that is broadcast over the hidden dimension H.
4️⃣  Element‑wise gating (sigmoid).
5️⃣  The expensive contraction  

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    for every batch‑element *b* and hidden slice *h*.
    This is a batched matrix‑multiply (B·H) × (N,N) @ (N,N)ᵀ.
    We implement it with a handcrafted Triton kernel that tiles the
    i‑ and j‑axes (BLOCK_M × BLOCK_N) and iterates over k in
    BLOCK_K‑sized tiles.  Operands are cast to fp16 for bandwidth,
    accumulation is done in fp32, and the result is written back as fp16.

6️⃣  Layer‑norm, gating and final linear to recover the original
    dimensionality D.

All heavy arithmetic stays on the GPU; only cheap PyTorch ops are used
outside the kernel.  The kernel works for any contiguous layout because
strides are passed explicitly.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: one (batch, hidden) tile of the outer‑product
# -------------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # const fp16  [B, H, N, N]
    right_ptr,         # const fp16  [B, H, N, N]
    out_ptr,           #   mut fp16  [B, H, N, N]
    B, N, H,           # int32 sizes
    # strides for left  (B,H,N,N)
    stride_lbh, stride_lhd, stride_lhn, stride_lhk,
    # strides for right (B,H,N,N)
    stride_rbh, stride_rhd, stride_rhn, stride_rhk,
    # strides for out   (B,H,N,N)
    stride_obh, stride_obd, stride_obn, stride_obk,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a tile of
        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
    The kernel processes a single (b, h) pair and a (BLOCK_M, BLOCK_N)
    tile of the (i, j) output matrix.
    """
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Decompose program id  →  (b, h, tile_i, tile_j)
    # --------------------------------------------------------------
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)

    tile_j = pid % num_tiles_j
    pid //= num_tiles_j
    tile_i = pid % num_tiles_i
    pid //= num_tiles_i
    h = pid % H
    b = pid // H

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, h, i, k]   shape (M, K)
        l_ptr = (
            left_ptr
            + b * stride_lbh
            + h * stride_lhd
            + offs_i[:, None] * stride_lhn
            + offs_k[None, :] * stride_lhk
        )
        l = tl.load(l_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[b, h, j, k]   we need the transpose → (K, N)
        r_ptr = (
            right_ptr
            + b * stride_rbh
            + h * stride_rhd
            + offs_k[:, None] * stride_rhk
            + offs_j[None, :] * stride_rhn
        )
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # dot product over K
        acc += tl.dot(l, r)          # (M, K) @ (K, N) → (M, N)

    # --------------------------------------------------------------
    # Store the tile
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_obh
        + h * stride_obd
        + offs_i[:, None] * stride_obn
        + offs_j[None, :] * stride_obk
    )
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data = (x, mask, weights, config)
        x            : (B, N, N, D)      torch.float32
        mask         : (B, N, N)         torch.bool / float32   (or None)
        weights      : dict of all linear / LN parameters (already on device)
        config       : {"dim": D, "hidden_dim": H}

    Returns
    -------
    out : (B, N, N, D)   torch.float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm over the model dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (left, right) – cheap, stay in PyTorch
    # -----------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)          # (B,N,N,1)
        left  = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – also cheap
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Rearrange to (B, H, N, N) and cast to fp16 for the kernel
    # -----------------------------------------------------------------
    # (B,N,N,H) → (B,H,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)   # (B, H, N, N) fp16

    # -----------------------------------------------------------------
    # 6️⃣ Triton launch
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # total program IDs = B * H * ceil(N/BLOCK_M) * ceil(N/BLOCK_N)
    grid = (
        triton.cdiv(B * hidden_dim *
                   triton.cdiv(N, BLOCK_M) *
                   triton.cdiv(N, BLOCK_N), 1),
    )

    _trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        out_fp16.stride(0), out_fp16.stride(1),
        out_fp16.stride(2), out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Back‑to (B,N,N,H) → LayerNorm → gating → final linear
    # -----------------------------------------------------------------
    out = out_fp16.to(dtype).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    out = out * out_gate                      # broadcast over (N,N)
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,D)

    return out
