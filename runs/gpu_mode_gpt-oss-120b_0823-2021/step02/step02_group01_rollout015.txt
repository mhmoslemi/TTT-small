"""
TriMul “outgoing” forward pass (no gradients).

The expensive contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is performed as a batched matrix‑multiply (I×K)·(K×J) → (I×J) with a
custom Triton kernel.  All lightweight work (LayerNorm, linear layers,
gating, final LayerNorm + linear) stays in PyTorch.  Mixed‑precision
(fp16 compute / fp32 accumulation) is used for the GEMM – the result is
converted back to fp32 before the remaining post‑processing.

The kernel tiles the multiplication along the M‑ and N‑axes with
BLOCK_M = BLOCK_N = 64 and along the reduction axis with BLOCK_K = 32,
which fits well into the shared‑memory budget of Hopper (SM‑90) and
exposes Tensor‑Core‑friendly shapes.  Three program‑ids are used:
  pid_z – batch * hidden‑dim (treated as one “head”)
  pid_m – tile of the output rows   (I dimension)
  pid_n – tile of the output columns (J dimension)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,            # left  matrix pointer   (Z, I, K)
    b_ptr,            # right matrix pointer   (Z, K, J)
    c_ptr,            # output matrix pointer  (Z, I, J)
    M, N, K,          # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: (Z, I, K) @ (Z, K, J) → (Z, I, J)"""

    pid_z = tl.program_id(0)          # batch * hidden_dim
    pid_m = tl.program_id(1)          # row tile
    pid_n = tl.program_id(2)          # col tile

    # -------------------  tile offsets  -----------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (better numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------  reduction over K  -------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile  (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 accumulation
        acc += tl.dot(a, b)

    # Store fp16 result (Tensor‑Core friendly)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul core contraction.
    Arguments
    ----------
    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous
    Returns
    -------
    out   : (B, H, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # output buffer
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten (B,H) → Z for the kernel
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides in element units (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tiling – empirically good for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (Z,
            triton.cdiv(I, BLOCK_M),
            triton.cdiv(J, BLOCK_N))

    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # good default for fp16 on H200
        num_stages=2,         # fits into shared memory
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        mask         : torch.Tensor [B, N, N]      (float32/bool) or None
        weights      : dict of model parameters
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (keep in FP32)
    # --------------------------------------------------------------
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var  = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * w_norm + b_norm                       # shape (B,N,N,dim), fp32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (FP16 for the heavy part)
    # --------------------------------------------------------------
    def _fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w  = _fp16(weights["left_proj.weight"])
    right_proj_w = _fp16(weights["right_proj.weight"])
    left_gate_w  = _fp16(weights["left_gate.weight"])
    right_gate_w = _fp16(weights["right_gate.weight"])
    out_gate_w   = _fp16(weights["out_gate.weight"])

    # Cast normalized input to fp16 once
    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        # mask: [B,N,N] → [B,N,N,1]
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Gating (still FP16)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction via the Triton kernel
    #    left  : (B,N,N,hidden) → (B,hidden,N,N) → (B,hidden,I,K)
    #    right : (B,N,N,hidden) → (B,hidden,N,N) → (B,hidden,K,J)
    # --------------------------------------------------------------
    # Move hidden dim to the second axis and make the tensors contiguous
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _batched_gemm_fp16(left_t, right_t)   # (B, H, I, J) fp16

    # Restore the original layout and cast back to fp32 for the rest
    out = out_fp16.permute(0, 2, 3, 1).contiguous()   # (B,N,N,dim)
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32)
    # --------------------------------------------------------------
    w_out_ln = weights["to_out_norm.weight"]
    b_out_ln = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * w_out_ln + b_out_ln

    # --------------------------------------------------------------
    # 7️⃣ Final gating + projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over dim
    out = F.linear(out, weights["to_out.weight"])

    return out
```

**Explanation of the design choices**

* The heavy \(N^{3}\) contraction is the only part moved to Triton.  
  All other operations are cheap enough to stay in PyTorch, avoiding
  unnecessary kernel launches and keeping the implementation simple.

* The kernel works on **fp16** operands but accumulates in **fp32**,
  matching the pattern that gives the best throughput on Hopper while
  preserving numerical stability for the subsequent LayerNorm steps.

* Tiling parameters (`BLOCK_M = BLOCK_N = 64`, `BLOCK_K = 32`) were
  selected because they:
  - fit into the shared‑memory budget of SM‑90,
  - produce multiples of 8‑wide Tensor‑Core tiles,
  - give good occupancy with `num_warps = 4`.

* The kernel is launched with three program‑ids:
  `pid_z` iterates over the flattened `(B * hidden_dim)` dimension,
  while `pid_m` / `pid_n` tile the output rows and columns respectively.
  This layout mirrors the classic batched GEMM pattern and lets Triton
  generate efficient load/store instructions.

* All weight tensors are loaded once and cast to the appropriate dtype
  outside the kernel, eliminating per‑element casting overhead.

* The final result is returned in **float32**, matching the expected
  datatype of the surrounding model.
