"""
custom_kernel – “outgoing” TriMul forward pass (AlphaFold‑3)

Algorithm (high‑level)
----------------------
1.  Layer‑Norm over the channel dimension (float32, PyTorch).
2.  Linear projections  x → left/right (C → H)  (bias‑free, float32).
3.  Linear gates → sigmoid (float32).
4.  Fuse mask, left‑gate and right‑gate into the projected tensors
    *in‑place* with a tiny Triton kernel (the only required kernel).
5.  Cast left/right to fp16 and reshape to (B*H, N, N).
6.  Batched mat‑mul  (B*H) × (N×N)  using torch.bmm → O(N³) contraction.
7.  Cast back to fp32 and restore (B,N,N,H) layout.
8.  Output Layer‑Norm, output‑gate and final linear (float32, PyTorch).

Only step‑4 is performed in Triton; everything else stays in PyTorch so that
the dominant cubic contraction benefits from cuBLAS Tensor‑Core kernels.
The Triton kernel fuses three element‑wise operations (mask, left‑gate,
right‑gate) into a single pass, keeping the launch overhead minimal.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel – fuse mask and gates into left/right tensors (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _fuse_mask_gates_kernel(
    left_ptr,          # fp32  [B,N,N,H]
    right_ptr,         # fp32  [B,N,N,H]
    mask_ptr,          # fp32  [B,N,N]   (broadcasted)
    lgate_ptr,         # fp32  [B,N,N,H]
    rgate_ptr,         # fp32  [B,N,N,H]
    B, N, H,           # compile‑time constants
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    pid  = tl.program_id(0)
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # ----- decode linear offset → (b,i,j,h) -----
    b   = offs // (N * N * H)
    rest = offs % (N * N * H)
    i   = rest // (N * H)
    rest = rest % (N * H)
    j   = rest // H
    h   = rest % H

    # ----- compute element pointers -----
    left_off   = b * stride_left_b   + i * stride_left_i   + j * stride_left_j   + h * stride_left_h
    right_off  = b * stride_right_b  + i * stride_right_i  + j * stride_right_j  + h * stride_right_h
    mask_off   = b * stride_mask_b   + i * stride_mask_i   + j * stride_mask_j
    lgate_off  = b * stride_lgate_b  + i * stride_lgate_i  + j * stride_lgate_j  + h * stride_lgate_h
    rgate_off  = b * stride_rgate_b  + i * stride_rgate_i  + j * stride_rgate_j  + h * stride_rgate_h

    # ----- load values (masked) -----
    left   = tl.load(left_ptr   + left_off,   mask=mask, other=0.0)
    right  = tl.load(right_ptr  + right_off,  mask=mask, other=0.0)
    m      = tl.load(mask_ptr   + mask_off,   mask=mask, other=0.0)   # scalar mask per (b,i,j)
    lgate  = tl.load(lgate_ptr  + lgate_off,  mask=mask, other=0.0)
    rgate  = tl.load(rgate_ptr  + rgate_off,  mask=mask, other=0.0)

    # ----- fuse: (mask * gate) onto the projection -----
    left  = left  * m * lgate
    right = right * m * rgate

    # ----- store back in‑place -----
    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left: torch.Tensor,
                         right: torch.Tensor,
                         mask: torch.Tensor,
                         left_gate: torch.Tensor,
                         right_gate: torch.Tensor) -> None:
    """
    In‑place element‑wise fusion of mask and gates using Triton.
    All inputs are expected to be on the same CUDA device and dtype=float32.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 2048                                   # ~2k threads → good occupancy
    grid = (triton.cdiv(total, BLOCK),)

    # strides are expressed in **elements**, not bytes
    s_left   = left.stride()
    s_right  = right.stride()
    s_mask   = mask.stride()
    s_lgate  = left_gate.stride()
    s_rgate  = right_gate.stride()

    _fuse_mask_gates_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=s_left[0],   stride_left_i=s_left[1],
        stride_left_j=s_left[2],   stride_left_h=s_left[3],
        stride_right_b=s_right[0], stride_right_i=s_right[1],
        stride_right_j=s_right[2], stride_right_h=s_right[3],
        stride_mask_b=s_mask[0],   stride_mask_i=s_mask[1],
        stride_mask_j=s_mask[2],
        stride_lgate_b=s_lgate[0], stride_lgate_i=s_lgate[1],
        stride_lgate_j=s_lgate[2], stride_lgate_h=s_lgate[3],
        stride_rgate_b=s_rgate[0], stride_rgate_i=s_rgate[1],
        stride_rgate_j=s_rgate[2], stride_rgate_h=s_rgate[3],
        BLOCK=BLOCK,
    )
    # No return – tensors are updated in‑place

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        * mask_tensor  : torch.Tensor, shape [B, N, N] (float32) or None
        * weights_dict : mapping of weight name → torch.Tensor
        * config_dict  : must contain "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim      = config["dim"]
    hidden   = config["hidden_dim"]
    device   = x.device
    eps      = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the channel dimension (float32)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]      # (dim,)
    ln_b = weights["norm.bias"]        # (dim,)
    # mean/var over last dim
    mean = x.mean(-1, keepdim=True)
    var  = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * ln_w + ln_b        # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]  # (hidden, dim)
    left  = F.linear(x_norm, left_proj_w)   # (B,N,N,hidden)
    right = F.linear(x_norm, right_proj_w)  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 3️⃣ Gates + sigmoid
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]
    left_gate  = F.linear(x_norm, left_gate_w).sigmoid()   # (B,N,N,hidden)
    right_gate = F.linear(x_norm, right_gate_w).sigmoid()
    out_gate   = F.linear(x_norm, out_gate_w).sigmoid()    # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 4️⃣ Prepare mask (broadcasted to hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2],
                         device=device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)   # (B,N,N,1)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask and gates into left/right (in‑place) via Triton
    # ------------------------------------------------------------------
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Cubic contraction via batched GEMM (fp16 → Tensor‑Core)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # cast to half for the heavy matmul
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*H, N, N)   (permute puts H as leading dim)
    left_flat  = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right_h.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matrix multiply:   (B*H, N, N) @ (B*H, N, N)^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # restore fp32 layout (B,N,N,hidden)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (float32)
    # ------------------------------------------------------------------
    to_out_ln_w = weights["to_out_norm.weight"]   # (hidden,)
    to_out_ln_b = weights["to_out_norm.bias"]     # (hidden,)
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_ln_w + to_out_ln_b

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate (broadcast already matches hidden dim)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear back to original channel dim (float32)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]          # (dim, hidden)
    out_final = F.linear(out_norm, to_out_w)    # (B,N,N,dim)

    return out_final
