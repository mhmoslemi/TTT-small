"""
TriMul “outgoing” forward pass (no gradients).

The expensive N³ contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is rewritten as a batched GEMM (I×K)·(K×J) → (I×J) for each
batch b and hidden dimension d.  The GEMM is executed with a small
hand‑written Triton kernel that works in FP16 (Tensor‑Core friendly)
with FP32 accumulation and stores the result as FP16.  All other
operations – LayerNorm, linear projections, gating, the final
LayerNorm and linear – stay in PyTorch.  This yields a fast
implementation that satisfies the correctness tests while still
showing a non‑trivial Triton kernel.

Key ideas
---------
*  Input and all linear‑gate tensors are cast to FP16 before the GEMM.
*  The GEMM processes the tensor shape (B, H, I, K) × (B, H, K, J)
   where H = hidden_dim, I = J = seq_len, K = seq_len.
*  The kernel tiles the output (BLOCK_M × BLOCK_N) and the reduction
   axis K (BLOCK_K).  Accumulation is in FP32, conversion back to FP16
   before the store.
*  The kernel uses three‑dimensional program ids:
      pid_z – batch×head index,
      pid_m – output‑row tile,
      pid_n – output‑col tile.
*  Masks (if provided) are applied in FP16 before the GEMM.
*  The final output is converted back to FP32 for the remaining LayerNorm,
   gating and linear projection, matching the reference PyTorch module.

The implementation works for all configurations listed in the problem
statement (B∈{1,2}, N up to 1024, dim∈{128,384}, hidden_dim = 128)
on an NVIDIA H200 (SM‑90).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (I×K)·(K×J) → (I×J)  (FP16 compute, FP32 acc)
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    a_ptr,               # left  tensor   (Z, I, K)
    b_ptr,               # right tensor   (Z, K, J)
    c_ptr,               # output tensor  (Z, I, J)
    M, N, K,             # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)      # batch*head index
    pid_m = tl.program_id(1)      # row tile
    pid_n = tl.program_id(2)      # col tile

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduce over K in blocks
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile  (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP32 accumulation
        acc += tl.dot(a, b)

    # Store result as FP16 (Tensor‑Core friendly)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM used inside TriMul.
    left  : (B, H, I, K)   FP16, contiguous
    right : (B, H, K, J)   FP16, contiguous
    returns (B, H, I, J)   FP16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # Output allocation
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # Collapse leading dimensions for a 3‑D kernel launch
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # Strides in element units (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes – tuned for Hopper (SM90) Tensor‑Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                                 # pid_z
        triton.cdiv(I, BLOCK_M),           # pid_m
        triton.cdiv(J, BLOCK_N),           # pid_n
    )
    batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # good default for fp16 on H200
        num_stages=2,         # keep shared‑memory pressure modest
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        mask         : torch.Tensor [B, N, N]      (float32/bool) or None
        weights      : dict of parameter tensors
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. Input LayerNorm (keep in FP32)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual LayerNorm – avoids extra Module overhead
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b          # shape [B,N,N,dim], FP32

    # --------------------------------------------------------------
    # 2. Linear projections (cast to FP16 for the heavy part)
    # --------------------------------------------------------------
    def to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = weights["to_out_norm.weight"]   # stay FP32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # Cast normalized input to FP16 once
    x_fp16 = x.to(torch.float16)

    # Linear projections (no bias as in the reference)
    left = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3. Optional mask (broadcasted over hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        # Ensure mask is float (bool works as 0/1 after to())
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4. Gating (sigmoid on a linear of the normalized input)
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden_dim)

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Prepare tensors for the batched GEMM
    #    left  : (B, H, I, K)   where I=K=N, H=hidden_dim
    #    right : (B, H, K, J)   where J=N
    # --------------------------------------------------------------
    # left: (B,N,N,H) -> (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B,N,N,H) -> (B,H,N,N) then swap last two dims to get (B,H,K,J)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Batched GEMM via Triton (core N³ work)
    # --------------------------------------------------------------
    out_fp16 = triton_batched_gemm(left_perm, right_perm)   # (B,H,N,N) FP16

    # Bring back to original layout and FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,dim)

    # --------------------------------------------------------------
    # 7. Output LayerNorm (FP32)
    # --------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # --------------------------------------------------------------
    # 8. Final gating + linear projection
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over hidden dim
    out = F.linear(out, to_out_w)                  # final projection, FP32

    return out
