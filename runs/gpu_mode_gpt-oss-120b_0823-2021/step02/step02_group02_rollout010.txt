"""
TriMul “outgoing” operator (AlphaFold3) – fast forward implementation.

*   The LayerNorms and Linear projections are kept in PyTorch (they are cheap
    compared to the O(N³) contraction).
*   The only “heavy” work moved to Triton is the element‑wise fusion of
    masking + gating for the left/right projected tensors.  This saves an
    extra memory round‑trip and lets us keep the O(N³) contraction in a
    single batched GEMM that cuBLAS/Tensor‑cores execute extremely efficiently
    on Hopper (H200).
*   After gating, the tensors are reshaped to (B*H, N, N) and a
    torch.bmm‑based matmul implements

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    which is exactly the `einsum('... i k d, ... j k d -> ... i j d', …)` in the
    reference code.
*   The result is brought back to the original layout, Layer‑normed,
    gated by `out_gate` and linearly projected to the model dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: fuse mask (optional) and two gates into left/right tensors
# -------------------------------------------------------------------------
@triton.jit
def fuse_mask_gate_kernel(
    left_ptr,          # float32*  [B, N, N, H]
    right_ptr,         # float32*  [B, N, N, H]
    mask_ptr,          # uint8*   [B, N, N]  (0/1 mask)   – can be nullptr
    left_gate_ptr,     # float32* [B, N, N, H]
    right_gate_ptr,    # float32* [B, N, N, H]
    out_left_ptr,      # float32* [B, N, N, H]
    out_right_ptr,     # float32* [B, N, N, H]
    B, N, H,           # sizes (int)
    stride_lbn, stride_lbh, stride_lbk, stride_lbh1,   # left  strides
    stride_rbn, stride_rbh, stride_rbk, stride_rbh1,   # right strides
    stride_mbn, stride_mbh,                            # mask  strides
    stride_gbn, stride_gbh,                           # gate  strides
    has_mask: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For every element (b, i, j, h) compute

        out_left  = left  * mask * left_gate
        out_right = right * mask * right_gate

    (`mask` is broadcasted over the hidden dimension).
    All tensors are stored row‑major, i.e. the last dimension is the fastest
    moving one.
    """
    pid = tl.program_id(0)

    # linear index of the element within the flat view B*N*N*H
    total = B * N * N * H
    idx = pid * BLOCK + tl.arange(0, BLOCK)
    mask = idx < total

    # decode linear index to (b, i, j, h)
    h = idx % H
    idx //= H
    j = idx % N
    idx //= N
    i = idx % N
    b = idx // N

    # offsets for the four tensors
    left_off   = (b * stride_lbh  + i * stride_lbn  + j * stride_lbk  + h * stride_lbh1)
    right_off  = (b * stride_rbh  + i * stride_rbn  + j * stride_rbk  + h * stride_rbh1)
    left_g_off = (b * stride_gbh  + i * stride_gbn  + j * stride_gbn  + h * stride_gbn)
    right_g_off= (b * stride_gbh  + i * stride_gbn  + j * stride_gbn  + h * stride_gbn)

    # load values
    l = tl.load(left_ptr  + left_off,   mask=mask, other=0.0)
    r = tl.load(right_ptr + right_off,  mask=mask, other=0.0)

    lg = tl.load(left_gate_ptr  + left_g_off,  mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + right_g_off, mask=mask, other=0.0)

    if has_mask:
        m = tl.load(mask_ptr + (b * stride_mbh + i * stride_mbn + j * stride_mbn), mask=mask, other=0.0)
        m = m.to(tl.float32)          # mask is uint8 -> float32
    else:
        m = 1.0

    out_l = l * lg * m
    out_r = r * rg * m

    tl.store(out_left_ptr  + left_off,  out_l, mask=mask)
    tl.store(out_right_ptr + right_off, out_r, mask=mask)


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input tensor [B, N, N, dim]
        torch.Tensor,          # mask tensor   [B, N, N]   (or None)
        Dict[str, torch.Tensor],
        Dict[str, Any]
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. LayerNorm + linear projections (PyTorch – cheap)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 2. Gating (produced in FP32)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # -----------------------------------------------------------------
    # 3. Fuse mask + gating with a tiny Triton kernel
    # -----------------------------------------------------------------
    # allocate contiguous tensors for the fused results
    left_fused  = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    # kernel launch configuration
    BLOCK = 1024                     # number of elements per program
    total_elems = B * N * N * hidden_dim
    num_blocks = (total_elems + BLOCK - 1) // BLOCK

    # strides (row‑major) for the 4‑D tensors
    l_strides = left.stride()       # (B,N,N,hidden) -> (sB,sN,sN,sh)
    r_strides = right.stride()
    g_strides = left_gate.stride()  # same layout as left/right

    mask_ptr = mask if mask is not None else torch.empty(0, device=device, dtype=torch.uint8)

    fuse_mask_gate_kernel[(num_blocks,)](
        left_ptr          = left,
        right_ptr         = right,
        mask_ptr          = mask_ptr,
        left_gate_ptr     = left_gate,
        right_gate_ptr    = right_gate,
        out_left_ptr      = left_fused,
        out_right_ptr     = right_fused,
        B                 = B,
        N                 = N,
        H                 = hidden_dim,
        stride_lbn        = l_strides[2],
        stride_lbh        = l_strides[0],
        stride_lbk        = l_strides[1],
        stride_lbh1       = l_strides[3],
        stride_rbn        = r_strides[2],
        stride_rbh        = r_strides[0],
        stride_rbk        = r_strides[1],
        stride_rbh1       = r_strides[3],
        stride_mbn        = mask.stride(1) if mask is not None else 0,
        stride_mbh        = mask.stride(0) if mask is not None else 0,
        stride_gbn        = g_strides[2],
        stride_gbh        = g_strides[0],
        has_mask          = mask is not None,
        BLOCK             = BLOCK,
    )

    # -----------------------------------------------------------------
    # 4. Batched GEMM: O(N³) contraction
    #    left_fused/right_fused : [B, N, N, H] -> [B*H, N, N]
    # -----------------------------------------------------------------
    left_b = left_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_b = right_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # (B*H, N, N) @ (B*H, N, N).transpose(1,2)  -> (B*H, N, N)
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_b.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # -----------------------------------------------------------------
    # 5. Post‑processing in PyTorch (LayerNorm, output gate, final linear)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                       # broadcast over N,N
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
