"""
custom_kernel – outgoing TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the input (FP32, manual for speed).
2️⃣  Linear projections + gating are computed in FP16.
3️⃣  Optional mask is applied (broadcast over the hidden dimension).
4️⃣  The core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is a batched matrix‑multiply (N×N)·(N×N) → (N×N) for every
    (batch, hidden) pair.  It is performed by a tiny Triton kernel that
    tiles the I‑, J‑ and K‑dimensions (BLOCK_M/ BLOCK_N = 64, BLOCK_K = 32)
    and accumulates in FP32 → FP16 output.
5️⃣  The result is reshaped back to (B,N,N,dim) and a second LayerNorm
    (FP32) is applied.
6️⃣  Output‑gate (sigmoid) multiplies the tensor and a final linear
    projection restores the original dimensionality.

Only the N³ contraction (step 4) runs in Triton; everything else stays in
PyTorch which keeps the implementation simple, numerically stable and
portable.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def _triton_tri_mul_kernel(
    a_ptr,            # left  tensor  (batch*head, I, K)   fp16
    b_ptr,            # right tensor  (batch*head, K, J)   fp16
    c_ptr,            # out   tensor  (batch*head, I, J)   fp16
    M, N, K,          # matrix sizes
    stride_az, stride_am, stride_ak,   # a strides (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)            # batch*head index
    pid_m = tl.program_id(1)            # tile row   (I)
    pid_n = tl.program_id(2)            # tile col   (J)

    # ---- tile start offsets ------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (better numeric stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- reduction over K ---------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A : (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B : (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot‑product (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # ---- write result -------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous
    returns (B, H, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # Collapse batch*head because the kernel expects a 3‑D tensor
    a = left.reshape(B * H, I, K)
    b = right.reshape(B * H, K, J)
    c = out.reshape(B * H, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes that fit well on Hopper (fp16, 4‑warps)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                     # pid_z  – one program per (batch, head)
        triton.cdiv(I, BLOCK_M),  # pid_m
        triton.cdiv(J, BLOCK_N),  # pid_n
    )
    _triton_tri_mul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # good default for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask         : torch.Tensor [B, N, N] or None (bool/float)
        - weights      : dict of leaf weight tensors
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]          # = hidden_dim in the paper

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual – cheap and avoids an extra kernel)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + 1e-5)
    x_norm = x_norm * ln_w + ln_b                     # still FP32

    # ------------------------------------------------------------------
    # 2️⃣ Projections & gates – work in FP16 for speed
    # ------------------------------------------------------------------
    # cast once
    def to_fp16(t): return t.to(torch.float16)

    left_proj_w  = to_fp16(weights["left_proj.weight"])   # (hidden, dim)
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w  = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w   = to_fp16(weights["out_gate.weight"])

    x_fp16 = x_norm.to(torch.float16)                     # (B,N,N,dim)

    # linear projections (no bias as in the reference)
    left  = F.linear(x_fp16, left_proj_w)                # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ optional mask (broadcast over hidden)
    # ------------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)          # (B,N,N,1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4️⃣ gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ core contraction – Triton batched GEMM
    #    left : (B,N,N,hidden)   →  (B, hidden, N, N)
    #    right: (B,N,N,hidden)   →  (B, hidden, N, N)  but we need (K,J) order
    # ------------------------------------------------------------------
    # bring hidden dimension to the second axis (acts like “head”)
    left_t  = left.permute(0, 3, 1, 2).contiguous()      # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()     # (B, H, K, J)

    out_fp16 = _batched_matmul_triton(left_t, right_t)   # (B, H, I, J)

    # reshape back to (B, N, N, dim) and cast to FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ second LayerNorm (FP32)
    # ------------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 7️⃣ output gate & final linear projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)   # broadcast over dim
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
