"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (high‑level)
----------------------
1️⃣  Layer‑norm the 4‑D input (FP32, manual – cheap because dim ≤ 384).
2️⃣  Four linear projections + three gating projections are performed in FP16
    (no bias, bias‑free linear layers as in the reference model).
3️⃣  Optional binary mask is applied to the left/right hidden tensors.
4️⃣  Gating (sigmoid) is multiplied into the projected tensors.
5️⃣  Core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is a batched matrix‑multiply over the “k” dimension.
    We launch a Triton kernel that treats each (batch, hidden‑dim) slice as an
    independent GEMM   (I = N, K = N, J = N).  The kernel works in fp16,
    accumulates in fp32 and writes fp16.
6️⃣  Convert the result back to FP32, apply a second LayerNorm, multiply by the
    output‑gate, and finish with a final linear projection to the original
    dimension.

The kernel is deliberately simple and tuned for Hopper/SM‑90:
    BLOCK_M = BLOCK_N = 64   (tile size in the I/J dimensions)
    BLOCK_K = 32             (tile size in the reduction dimension)
    4 warps, fp16 arithmetic → Tensor‑cores are fully utilised.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched GEMM (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,            # left  tensor  (B*H, I, K)
    b_ptr,            # right tensor  (B*H, K, J)
    c_ptr,            # output tensor (B*H, I, J)
    M, N, K,          # extents
    stride_az, stride_am, stride_ak,   # a strides (batch, i, k)
    stride_bz, stride_bk, stride_bn,   # b strides (batch, k, j)
    stride_cz, stride_cm, stride_cn,   # c strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    #  Tile identifiers
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)          # tile along M (rows)
    pid_n = tl.program_id(1)          # tile along N (cols)
    pid_z = tl.program_id(2)          # batch*head index

    # --------------------------------------------------------------
    #  Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    #  Accumulator (FP32 for numerical stability)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    #  Loop over K in BLOCK_K sized chunks
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile  (BLOCK_M × BLOCK_K)  – fp16
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile  (BLOCK_K × BLOCK_N) – fp16
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Multiply‑accumulate (fp16 × fp16 → fp32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    #  Store the result (back to fp16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left   : (B, H, I, K)   fp16, contiguous
    right  : (B, H, K, J)   fp16, contiguous
    returns: (B, H, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # output allocation
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse batch*head dimension – kernel expects 3‑D tensors
    batch = B * H
    a = left.view(batch, I, K)
    b = right.view(batch, K, J)
    c = out.view(batch, I, J)

    # strides in elements (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tile sizes – tuned for Hopper (FP16, Tensor‑cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # rows
        triton.cdiv(J, BLOCK_N),   # cols
        batch,                     # one program per (batch,head)
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # optimal on SM‑90 for fp16
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor (bool/float) or None, shape [B, N, N]
        weights      : dict of leaf weight tensors (see reference code)
        config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack & basic constants
    # --------------------------------------------------------------
    inp, mask, w, cfg = data
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]
    dev = inp.device

    # --------------------------------------------------------------
    # 2️⃣ first LayerNorm (FP32, manual)
    # --------------------------------------------------------------
    ln_w = w["norm.weight"]
    ln_b = w["norm.bias"]
    # mean / var over the last dimension ``dim``
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                      # still FP32

    # --------------------------------------------------------------
    # 3️⃣ cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # linear projection weights: shape (hidden, dim)
    left_proj_w   = w["left_proj.weight"].to(torch.float16)
    right_proj_w  = w["right_proj.weight"].to(torch.float16)

    # gate weights: shape (hidden, dim)
    left_gate_w   = w["left_gate.weight"].to(torch.float16)
    right_gate_w  = w["right_gate.weight"].to(torch.float16)
    out_gate_w    = w["out_gate.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ project & gate (FP16)
    # --------------------------------------------------------------
    # (B,N,N,hidden) = (B,N,N,dim) @ (hidden,dim).T
    left  = F.linear(x_fp16, left_proj_w)   # no bias
    right = F.linear(x_fp16, right_proj_w)

    # mask handling – mask is broadcast over the hidden dimension
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * m
        right = right * m

    # gates (sigmoid)
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ core contraction via Triton GEMM
    #    left  : (B,N,N,hidden) → (B, hidden, N, N)   (i,k) layout
    #    right : (B,N,N,hidden) → (B, hidden, N, N)   (j,k) layout
    # --------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # Batched GEMM: (B*H, N, N) · (B*H, N, N)ᵀ  →  (B*H, N, N)
    out_fp16 = _triton_batched_gemm(left_t, right_t)   # (B, H, N, N)

    # Back‑to original layout (B, N, N, hidden) and FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ second LayerNorm (FP32)
    # --------------------------------------------------------------
    ln2_w = w["to_out_norm.weight"]
    ln2_b = w["to_out_norm.bias"]
    mean2 = out.mean(-1, keepdim=True)
    var2 = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean2) / torch.sqrt(var2 + 1e-5)
    out = out * ln2_w + ln2_b

    # --------------------------------------------------------------
    # 7️⃣ final gating + linear projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)        # broadcast over hidden
    out = F.linear(out, w["to_out.weight"])      # (B,N,N,dim)

    return out
