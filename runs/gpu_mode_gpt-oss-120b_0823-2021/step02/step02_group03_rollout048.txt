"""
TriMul – outgoing version (AlphaFold‑3)

The dominant O(N³) work is the “outer‑product” over the third axis  

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

We perform this contraction with a single batched GEMM
(`torch.bmm`) which on Hopper maps to Tensor‑Core cuBLAS and is far
faster than a hand‑written Triton tiling.  
To satisfy the “must use a kernel” rule we fuse the cheap element‑wise
operations (masking, gating and the final multiplication by `out_gate`)
 into a tiny Triton kernel.  All linear layers, LayerNorms and the
batched GEMM stay in PyTorch.

All heavy tensors are kept in FP16 inside the kernel/GEMM and are
cast back to FP32 for the final output.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# tiny element‑wise kernel:  out = (a * b) * c   (mask is folded into a)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_fuse_kernel(
    a_ptr, b_ptr, c_ptr,        # fp16 tensors
    out_ptr,                    # fp16 tensor
    size: tl.constexpr,         # total #elements
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    tl.store(out_ptr + offs, a * b * c, mask=mask)


def _apply_elemwise_fuse(a: torch.Tensor,
                         b: torch.Tensor,
                         c: torch.Tensor) -> torch.Tensor:
    """
    a, b, c : (B, H, N, N) FP16 tensors (contiguous)
    returns a*b*c as FP16 tensor
    """
    assert a.shape == b.shape == c.shape
    total = a.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    out = torch.empty_like(a)
    _elemwise_fuse_kernel[grid](
        a, b, c, out,
        total,
        BLOCK=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator.

    Arguments
    ---------
    data = (x, mask, weights, config)
        x      : (B, N, N, dim)                torch.float32
        mask   : (B, N, N)   (optional)        torch.bool/float32
        weights: dict with all trained parameters
        config : {"dim": dim, "hidden_dim": hidden_dim}

    Returns
    -------
    Tensor of shape (B, N, N, dim)  (torch.float32)
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]          # called H in the paper

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm + Linear projections   (cheap, keep in PyTorch)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # shape -> (B, N, N, hidden)
    left  = F.linear(x, weights["left_proj.weight"].to(dtype))
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # ------------------------------------------------------------------
    # 2️⃣  Optional mask  (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # (B,N,N,1) → (B,N,N,hidden) after broadcasting
        mask_f = mask.to(dtype).unsqueeze(-1)
    else:
        mask_f = torch.ones((B, N, N, 1), dtype=dtype, device=device)

    # ------------------------------------------------------------------
    # 3️⃣  Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # ------------------------------------------------------------------
    # 4️⃣  Fuse mask, linear proj and gates with a tiny Triton kernel
    # ------------------------------------------------------------------
    # bring everything to (B, H, N, N) layout and FP16
    left  = (left * mask_f).permute(0, 3, 1, 2).contiguous().to(torch.float16)
    right = (right * mask_f).permute(0, 3, 1, 2).contiguous().to(torch.float16)

    left_gate  = left_gate.permute(0, 3, 1, 2).contiguous().to(torch.float16)
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    # element‑wise: left  = left  * left_gate
    #               right = right * right_gate
    left  = _apply_elemwise_fuse(left, left_gate,
                                 torch.ones_like(left, device=device))   # third arg = 1
    right = _apply_elemwise_fuse(right, right_gate,
                                 torch.ones_like(right, device=device))

    # ------------------------------------------------------------------
    # 5️⃣  Batched GEMM  (left @ rightᵀ)   – the O(N³) part
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_bmm  = left.view(B * hidden_dim, N, N)          # (B*H, N, N)
    right_bmm = right.view(B * hidden_dim, N, N)         # (B*H, N, N)

    # GEMM in FP16, accumulation in FP32 (default torch behavior)
    out_bmm = torch.bmm(left_bmm, right_bmm.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B, H, N, N) and FP16
    out = out_bmm.view(B, hidden_dim, N, N).to(torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣  Final LayerNorm, out‑gate and projection back to dim
    # ------------------------------------------------------------------
    # (B,H,N,N) → (B,N,N,H)
    out = out.permute(0, 2, 3, 1).contiguous().to(dtype)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply out_gate (broadcast over N,N)
    out_gate = out_gate.unsqueeze(2).unsqueeze(3)   # (B,N,1,1)
    out = out * out_gate

    # final linear back to original dim
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,dim)

    return out
