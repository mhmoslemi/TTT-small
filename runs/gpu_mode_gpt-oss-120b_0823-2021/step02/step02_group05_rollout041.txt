"""
TriMul “outgoing” forward (AlphaFold‑3) – GPU accelerated version.

The implementation follows the reference PyTorch module but moves the most
memory‑bound part (the optional pair‑wise mask application) to a small Triton
kernel.  All other operations (LayerNorm, linear projections, gating,
the O(N³·C) reduction, final LayerNorm and output projection) are kept in
PyTorch because they map to highly‑optimised GEMM/LayerNorm kernels on Hopper.

The kernel `mask_apply_kernel` multiplies a tensor of shape
    [B, N, N, H]  (H = hidden_dim)
by a broadcast mask of shape [B, N, N] (0/1 values).  The multiplication is
performed element‑wise across the last dimension, which is exactly the work
required by the “mask” step in the reference code.

The entry‑point `custom_kernel` performs:
1. LayerNorm on the input (dim‑dimensional).
2. Linear projections → left/right (hidden_dim).
3. Optional mask (applied with Triton).
4. Gating (sigmoid) and element‑wise gating of left/right.
5. Core reduction `out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]`
   implemented with `torch.einsum` (maps to a series of GEMM kernels).
6. Final LayerNorm, out‑gate gating and output projection back to dim.

All tensors are expected on CUDA and the final result is float‑32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: apply a broadcast mask of shape [B, N, N] to a tensor
# of shape [B, N, N, H].
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*   (input tensor)
    mask_ptr,         # const float*   (mask tensor, flattened B*N*N)
    dst_ptr,          # float*         (output tensor)
    total_el,         # i64            total number of elements = B*N*N*H
    H,                # i64            hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear element indices
    mask = offs < total_el                             # OOB guard

    # Load source value
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # Each contiguous block of H elements shares the same mask entry.
    # Compute index into the flattened mask (shape B*N*N) = offs // H
    mask_idx = offs // H
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Store masked result
    tl.store(dst_ptr + offs, src * mval, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` of shape [B, N, N, H] by ``mask[...,None]`` using Triton.
    ``mask`` is float (0/1) with shape [B, N, N].
    Returns a new tensor with the same shape as ``tensor``.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                     # tuned for H200 (large shared‑mem, high bandwidth)

    # flatten to 1‑D views (contiguous)
    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)        # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry‑point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                                            # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → shape [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask – apply with Triton (the only custom kernel)
    # ------------------------------------------------------------------
    if not nomask:
        left  = apply_mask_triton(left,  mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise gating of projected tensors
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(
        x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(
        x, weights["out_gate.weight"]).sigmoid()

    left  = left  * left_gate                     # [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·hidden_dim) reduction (einsum ↔ series of GEMMs)
    # ------------------------------------------------------------------
    # out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # 6. Final LayerNorm, out‑gate gating and output projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                         # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure float32 output (already is)
    return out.to(torch.float32)
