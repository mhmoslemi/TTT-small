"""
custom_kernel for the “outgoing” TriMul operator (AlphaFold‑3).

- Performs LayerNorm, linear projections and the O(N³·C) reduction with
  PyTorch (the reduction is mapped to highly‑optimised GEMM kernels on
  Hopper, which is faster than a naïve Triton implementation).
- The only mandatory Triton part is the mask‑application, which is a
  pure‑memory‑bound element‑wise multiplication that benefits from
  explicit vectorised loads/stores.
- The implementation supports both the masked and the nomask case,
  arbitrary batch size (1‑2), sequence length (128‑1024) and hidden
  dimensions (128‑384).  All tensors are expected on CUDA and the
  final output is float‑32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: apply a broadcast mask of shape [B, N, N] to a tensor
# of shape [B, N, N, H].  The mask is 0/1 (float) and is broadcast on the
# last dimension.  This kernel is the only required Triton component.
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*   (input tensor, flattened)
    mask_ptr,         # const float*   (mask tensor, flattened, shape B*N*N)
    dst_ptr,          # float*         (output tensor, flattened)
    total_el,         # i64            total number of elements = B*N*N*H
    H,                # i64            hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear indices inside the block
    mask = offs < total_el                     # out‑of‑bounds guard

    # Load the source value (float32)
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # The mask is shared by H consecutive elements.
    # Compute index into the flattened mask (B*N*N) and load it.
    mask_idx = offs // H
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Store masked result
    tl.store(dst_ptr + offs, src * mval, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` of shape [B, N, N, H] by ``mask[...,None]`` using
    a custom Triton kernel.  ``mask`` must be float (0/1) with shape [B, N, N].
    Returns a new tensor with the same shape and dtype as ``tensor``.
    """
    B, N, _, H = tensor.shape
    total_el = B * N * N * H
    BLOCK = 1024                     # reasonable for H200 (≈4 KB per block)

    # Flatten everything to 1‑D contiguous views.
    src = tensor.reshape(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.reshape(-1)     # [B,N,N]

    grid = (triton.cdiv(total_el, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total_el,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry‑point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Linear projections (no bias) → shape [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Optional mask – apply with Triton (mandatory custom kernel)
    # --------------------------------------------------------------
    if not nomask:
        left = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # --------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise gating of projected tensors
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"]
    ).sigmoid()                     # [B,N,N,hidden_dim]
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"]
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Core O(N³·C) reduction via batched GEMM (einsum)
    # --------------------------------------------------------------
    #   out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)
    # shape: [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 6) Final LayerNorm, output gating and projection back to dim
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                         # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure float32 (all ops already use float32)
    return out.to(torch.float32)
