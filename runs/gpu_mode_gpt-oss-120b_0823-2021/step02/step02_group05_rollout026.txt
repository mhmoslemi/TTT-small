"""
TriMul “outgoing” forward (AlphaFold‑3) with a Triton accelerator.

The heavy O(N³·C) reduction is left to Torch’s highly‑optimised GEMM
(einsum → a sequence of batched GEMMs on Hopper).  
The only part we move to Triton is the optional mask‑application on the
projected tensors (left/right).  This memory‑bound step benefits from
explicit vectorised loads/stores and a single kernel launch, satisfying the
requirement that *some* work be performed in Triton while keeping the
overall implementation simple and fast.

All tensors are assumed to be on CUDA, float32.  The final output is also
float32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: broadcast a [B,N,N] mask onto a [B,N,N,H] tensor.
# Each block processes BLOCK elements (linearised).  H (the hidden dim)
# is a compile‑time constant to enable the //‑division that maps an
# element index to its mask index.
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*   (input tensor, flattened)
    mask_ptr,         # const float*   (mask tensor, flattened [B,N,N])
    dst_ptr,          # float*         (output tensor, flattened)
    total_el,         # i64            total number of elements = B*N*N*H
    H,                # i64            hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid   = tl.program_id(0)
    offs  = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < total_el

    # Load the source value (if out‑of‑bounds load 0.0 – will be masked later)
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # The mask is shared by H consecutive elements along the last dim.
    # Integer division gives the mask element that corresponds to this
    # position in the flattened [B,N,N] tensor.
    mask_idx = offs // H
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Write masked value
    tl.store(dst_ptr + offs, src * mval, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Apply ``mask[...,None]`` (shape [B,N,N,1]) to ``tensor`` (shape [B,N,N,H])
    using the Triton kernel above.  Returns a new tensor with the same shape.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                     # good occupancy on H200

    src = tensor.view(-1)            # flatten
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)        # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry point used by the evaluator.
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    nomask      = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                                                   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (Triton)
    # ------------------------------------------------------------------
    if not nomask:
        left  = apply_mask_triton(left,  mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) + element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate       # [B, N, N, hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·C) reduction via einsum (batched GEMM)
    #     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gating, and projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                           # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure float32 output (already the case)
    return out.to(torch.float32)
