"""
custom_kernel – “outgoing” TriMul forward (AlphaFold3)

Algorithm
---------
1.  Layer‑Norm on the 4‑D tensor (B,N,N,C)                       (fp32)
2.  Linear projections to the hidden dimension H                (fp32)
3.  Compute three gates, apply sigmoid                         (fp32)
4.  Fuse mask, left‑gate and right‑gate into the projected
    tensors in‑place with a tiny Triton kernel                (fp32)
5.  Cast the fused tensors to float16 and reshape to
    (B*H, N, N) – each hidden channel becomes a “head”.
6.  Batched GEMM (torch.bmm) computes the cubic contraction
    out_h[b*h, i, j] = Σ_k left_h[b*h, i, k] * right_h[b*h, j, k]
    (Tensor‑Core, fp16, cuBLAS)
7.  Cast back to fp32, restore the original layout
8.  Layer‑Norm + output‑gate + final linear projection          (fp32)

Only the element‑wise masking / gating step is written in Triton;
the dominant O(N³) work is delegated to cuBLAS which is optimal on
H200/H100.  The implementation works for any batch size (1‑2),
sequence length up to 1024 and arbitrary dim / hidden_dim
(e.g. 128, 384 …) and correctly handles missing or random masks.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask and both gates into left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp32,   [B,N,N,H]
    right_ptr,         # fp32,   [B,N,N,H]
    mask_ptr,          # fp32,   [B,N,N]   (broadcasted mask)
    lgate_ptr,         # fp32,   [B,N,N,H]
    rgate_ptr,         # fp32,   [B,N,N,H]
    B, N, H,           # compile‑time constants
    # strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    total = B * N * N * H
    mask = offs < total

    # decode linear offset → (b, i, j, h)
    b = (offs // (N * N * H))
    rest = offs % (N * N * H)
    i = (rest // (N * H))
    rest = rest % (N * H)
    j = (rest // H)
    h = rest % H

    # pointer arithmetic (convert to element offsets)
    l_off = b * stride_l_b + i * stride_l_i + j * stride_l_j + h * stride_l_h
    r_off = b * stride_r_b + i * stride_r_i + j * stride_r_j + h * stride_r_h
    m_off = b * stride_m_b + i * stride_m_i + j * stride_m_j
    lg_off = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + h * stride_lg_h
    rg_off = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j + h * stride_rg_h

    # loads (mask is broadcasted, other tensors are per‑channel)
    l   = tl.load(left_ptr  + l_off,  mask=mask, other=0.0)
    r   = tl.load(right_ptr + r_off, mask=mask, other=0.0)
    m   = tl.load(mask_ptr  + m_off,  mask=mask, other=0.0)
    lg  = tl.load(lgate_ptr + lg_off, mask=mask, other=0.0)
    rg  = tl.load(rgate_ptr + rg_off, mask=mask, other=0.0)

    # fuse mask and gates (sigmoid already applied in Python)
    l = l * m * lg
    r = r * m * rg

    # store back in‑place
    tl.store(left_ptr  + l_off, l, mask=mask)
    tl.store(right_ptr + r_off, r, mask=mask)


def _fuse_mask_and_gates(left, right, mask, left_gate, right_gate):
    """In‑place element‑wise mask/gate fusion via Triton.

    All tensors are expected to be fp32 and on the same CUDA device.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 2048                                   # ~2 k threads → good occupancy
    grid = (triton.cdiv(total, BLOCK),)

    # strides in *elements* (not bytes)
    s_l = left.stride()
    s_r = right.stride()
    s_m = mask.stride()
    s_lg = left_gate.stride()
    s_rg = right_gate.stride()

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_l_b=s_l[0], stride_l_i=s_l[1],
        stride_l_j=s_l[2], stride_l_h=s_l[3],
        stride_r_b=s_r[0], stride_r_i=s_r[1],
        stride_r_j=s_r[2], stride_r_h=s_r[3],
        stride_m_b=s_m[0], stride_m_i=s_m[1],
        stride_m_j=s_m[2],
        stride_lg_b=s_lg[0], stride_lg_i=s_lg[1],
        stride_lg_j=s_lg[2], stride_lg_h=s_lg[3],
        stride_rg_b=s_rg[0], stride_rg_i=s_rg[1],
        stride_rg_j=s_rg[2], stride_rg_h=s_rg[3],
        BLOCK=BLOCK,
    )
    return left, right


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N]   (float32) or None
        - weights_dict : mapping of weight name → torch.Tensor (all fp32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on the same CUDA device)
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    device = x.device
    eps = 1e-5

    # -------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (fp32)
    # -------------------------------------------------
    ln_w = weights["norm.weight"]   # (dim,)
    ln_b = weights["norm.bias"]     # (dim,)
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * ln_w + ln_b                     # (B,N,N,dim)

    # -------------------------------------------------
    # 2️⃣ Linear projections (fp32, bias‑free)
    # -------------------------------------------------
    left_proj_w = weights["left_proj.weight"]   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_w)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x_norm, right_proj_w)

    # -------------------------------------------------
    # 3️⃣ Gates + sigmoid (fp32)
    # -------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()

    # -------------------------------------------------
    # 4️⃣ Prepare mask (broadcast to channel dim)
    # -------------------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2],
                          device=device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)                     # (B,N,N,1)

    # -------------------------------------------------
    # 5️⃣ Fuse mask and gates in‑place (Triton)
    # -------------------------------------------------
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # -------------------------------------------------
    # 6️⃣ Batched GEMM for the cubic contraction (fp16)
    # -------------------------------------------------
    B, N, _, H = left.shape
    # Cast to half‑precision for Tensor‑Core acceleration
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*H, N, N)
    left_flat = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_flat = right_h.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matmul: each head contracts over the 3rd dimension
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H,N,N)

    # back to fp32, restore (B,N,N,hidden)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # -------------------------------------------------
    # 7️⃣ Output LayerNorm (fp32)
    # -------------------------------------------------
    to_out_ln_w = weights["to_out_norm.weight"]
    to_out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_ln_w + to_out_ln_b

    # -------------------------------------------------
    # 8️⃣ Apply output gate
    # -------------------------------------------------
    out_norm = out_norm * out_gate

    # -------------------------------------------------
    # 9️⃣ Final linear projection back to dim (fp32)
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"]          # (dim, hidden)
    out_final = torch.nn.functional.linear(out_norm, to_out_w)   # (B,N,N,dim)

    return out_final
