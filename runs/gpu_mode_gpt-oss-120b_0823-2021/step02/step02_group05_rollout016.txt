"""
TriMul “outgoing” forward (AlphaFold‑3) – Triton‑accelerated mask application.

The kernel only masks the projected tensors (left/right) because that is the
only pure‑memory‑bound step that benefits from explicit vectorisation.
All arithmetic (LayerNorm, linear layers, gating, the O(N³·C) reduction,
final LayerNorm and projection) is performed with highly‑optimised
PyTorch kernels (which on Hopper map to fast GEMM/LayerNorm kernels).
The implementation works for any batch‑size (1‑2), sequence length up to
1024 and hidden dimensions 128/384 as required by the test suite.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: broadcast‑mask a 4‑D tensor [B,N,N,H] with a 3‑D mask
# [B,N,N] (float 0/1).  The mask is applied element‑wise on the last
# dimension without any extra memory copies.
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*  – input tensor (flattened)
    mask_ptr,         # const float*  – mask tensor (flattened)
    dst_ptr,          # float*        – output tensor (flattened)
    total_el,         # i64           – total number of elements B*N*N*H
    H,                # i64           – size of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)      # linear indices for this block
    mask = offs < total_el                         # out‑of‑bounds guard

    # Load source values (float32)
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # Compute index of the corresponding mask element:
    # each consecutive H entries share the same mask entry → floor division
    mask_idx = offs // H
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Store masked result
    tl.store(dst_ptr + offs, src * mask_val, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` of shape [B,N,N,H] by ``mask[...,None]`` using Triton.

    Parameters
    ----------
    tensor : torch.Tensor   – contiguous CUDA tensor, float32
    mask   : torch.Tensor   – contiguous CUDA tensor, float32 (0/1)

    Returns
    -------
    torch.Tensor with the same shape as ``tensor``.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                                 # good occupancy on H200

    # Flatten for 1‑D kernel launch
    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)                    # shape [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry‑point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : (input_tensor, mask, weights, config)
        input_tensor : [B, N, N, dim]   float32 CUDA
        mask         : [B, N, N]       float32 CUDA (0/1); may be omitted when
                                        config["nomask"] is True.
        weights      : dict of model parameters (see reference code)
        config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                       optional ``nomask`` (bool, default True).

    Returns
    -------
    output : [B, N, N, dim]  float32 CUDA
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                                            # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask – apply with Triton (the only custom GPU kernel)
    # ------------------------------------------------------------------
    if not nomask:
        left  = apply_mask_triton(left,  mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate       # [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·C) reduction via einsum → fast GEMM on Hopper
    # ------------------------------------------------------------------
    # ``left``  : [B, i, k, h]
    # ``right`` : [B, j, k, h]
    # result    : [B, i, j, h]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # 6. Final LayerNorm, gating and output linear projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                              # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure float32 output (all ops are float32 already)
    return out.to(torch.float32)
