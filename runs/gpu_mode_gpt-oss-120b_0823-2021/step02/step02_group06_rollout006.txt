--------------------
The “outgoing” TriMul operator consists of three clear phases  

1. **Layer‑norm + linear projections** – cheap element‑wise work, done with
   PyTorch’s highly‑optimised kernels.  
2. **Mask & gate fusion** – the only part that benefits from a custom kernel.
   For every element *(b,i,k,d)* we compute  

   ```
   left  = left  * mask * sigmoid(left_gate)
   right = right * mask * sigmoid(right_gate)
   out_gate = sigmoid(out_gate) * mask
   ```

   All three tensors are updated in‑place, so only one kernel launch is needed.
3. **Tensor contraction** – a batched matrix‑multiply over the *k* dimension
   (`out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`).  
   This is simply a GEMM for every hidden channel *d* and is delegated to
   cuBLAS/torch.matmul, which on Hopper (H200) is far faster than any Triton
   tiling we could write.

The final steps (second layer‑norm, out‑gate scaling and the output linear
projection) are again performed with PyTorch.

The kernel works for both “masked” and “nomask” cases (the mask is broadcast to
the hidden dimension).  It supports `float32` and `bfloat16` tensors and works
for any batch size ≤ 2, sequence length ≤ 1024 and hidden dimension = 128.

```python
"""
custom_kernel – Triton‑accelerated forward pass of the “outgoing” TriMul
operator (AlphaFold‑3).  The kernel fuses the optional mask, the three sigmoid
gates and the two projected tensors (left/right) into a single pass over the
data.  All other work (layer‑norms, linear projections, the N³ contraction) is
performed with PyTorch which already uses cuBLAS/Tensor‑cores on H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate, right‑gate and out‑gate.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr, mask_ptr,
    lgate_ptr, rgate_ptr, ogate_ptr,
    B, N, H,
    # stride arguments (int64)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """
    In‑place fusion:
        left  = left  * mask * lgate
        right = right * mask * rgate
        out_gate = ogate * mask          (out_gate already holds sigmoid values)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)      # flat ids
    total = B * N * N * H
    mask = offs < total

    # ----- decode flat index → (b,i,k,d) -------------------------------
    tmp = offs
    d = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ----- compute pointers -------------------------------------------
    left_ptr   = left_ptr   + b * stride_l_b   + i * stride_l_i   + k * stride_l_k   + d * stride_l_h
    right_ptr  = right_ptr  + b * stride_r_b   + i * stride_r_i   + k * stride_r_k   + d * stride_r_h
    mask_ptr   = mask_ptr   + b * stride_m_b   + i * stride_m_i   + k * stride_m_k                     # 1‑channel
    lg_ptr     = lgate_ptr  + b * stride_lg_b  + i * stride_lg_i  + k * stride_lg_k  + d * stride_lg_h
    rg_ptr     = rgate_ptr  + b * stride_rg_b  + i * stride_rg_i  + k * stride_rg_k  + d * stride_rg_h
    og_ptr     = ogate_ptr  + b * stride_og_b  + i * stride_og_i  + k * stride_og_k  + d * stride_og_h

    # ----- load ---------------------------------------------------------
    l  = tl.load(left_ptr,  mask=mask, other=0.0)
    r  = tl.load(right_ptr, mask=mask, other=0.0)
    m  = tl.load(mask_ptr,  mask=mask, other=1.0)   # 1.0 for nomask case
    lg = tl.load(lg_ptr,    mask=mask, other=1.0)
    rg = tl.load(rg_ptr,    mask=mask, other=1.0)
    og = tl.load(og_ptr,    mask=mask, other=1.0)

    # ----- fuse ---------------------------------------------------------
    l  = l * m * lg
    r  = r * m * rg
    og = og * m          # out_gate only needs mask

    # ----- store --------------------------------------------------------
    tl.store(left_ptr,  l, mask=mask)
    tl.store(right_ptr, r, mask=mask)
    tl.store(og_ptr,    og, mask=mask)


def _apply_mask_and_gates(left, right, mask, left_gate, right_gate, out_gate):
    """
    Launch the Triton kernel defined above.
    All tensors are (B, N, N, H) except *mask* which is (B,N,N,1).
    The kernel updates ``left`` and ``right`` in‑place and returns the
    masked ``out_gate`` tensor (also updated in‑place).
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                       # 1024 works well on H200 (≈4 KB per thread)
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        ogate_ptr=out_gate,
        B=B,
        N=N,
        H=H,
        stride_l_b=left.stride(0), stride_l_i=left.stride(1),
        stride_l_k=left.stride(2), stride_l_h=left.stride(3),
        stride_r_b=right.stride(0), stride_r_i=right.stride(1),
        stride_r_k=right.stride(2), stride_r_h=right.stride(3),
        stride_m_b=mask.stride(0), stride_m_i=mask.stride(1),
        stride_m_k=mask.stride(2), stride_m_1=mask.stride(3),
        stride_lg_b=left_gate.stride(0), stride_lg_i=left_gate.stride(1),
        stride_lg_k=left_gate.stride(2), stride_lg_h=left_gate.stride(3),
        stride_rg_b=right_gate.stride(0), stride_rg_i=right_gate.stride(1),
        stride_rg_k=right_gate.stride(2), stride_rg_h=right_gate.stride(3),
        stride_og_b=out_gate.stride(0), stride_og_i=out_gate.stride(1),
        stride_og_k=out_gate.stride(2), stride_og_h=out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor, shape [B, N, N, dim]
        * mask_tensor  : torch.Tensor, shape [B, N, N]   or None
        * weights_dict : dict with the same keys as the reference module
        * config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm over dim
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4️⃣ prepare mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(B, N, N, 1, dtype=x.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(x.dtype)   # (B,N,N,1)

    # ------------------------------------------------------------------
    # 5️⃣ fuse mask & gates via Triton (in‑place on left/right, out_gate is updated)
    # ------------------------------------------------------------------
    out_gate = _apply_mask_and_gates(
        left, right, mask, left_gate, right_gate, out_gate
    )

    # ------------------------------------------------------------------
    # 6️⃣ core contraction Σ_k left[b,i,k,d] * right[b,j,k,d]
    #     Implemented as a batched GEMM per hidden channel.
    # ------------------------------------------------------------------
    # reshape to (B, hidden, N, N) for batched matmul
    left_t  = left.permute(0, 3, 1, 2)          # (B, H, N, K)   K == N
    right_t = right.permute(0, 3, 1, 2)         # (B, H, N, K)
    # (B,H,N,N) = (B,H,N,K) @ (B,H,K,N)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))
    # back to (B,N,N,hidden)
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7️⃣ final LayerNorm, out‑gate scaling and output projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # gated scaling
    out = F.linear(out, weights["to_out.weight"])

    # Output must be float32 (reference uses float32)
    return out.to(torch.float32)
```
