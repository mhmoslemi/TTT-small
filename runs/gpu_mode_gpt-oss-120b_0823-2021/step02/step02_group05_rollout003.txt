"""
TriMul “outgoing” forward (AlphaFold‑3) – hybrid implementation.

*   LayerNorm, linear layers and the O(N³·C) reduction are performed with
    PyTorch (the reduction becomes a series of highly‑optimised GEMM calls on
    Hopper, which is faster than a naïve Triton implementation).
*   The only mandatory Triton component is the **pair‑wise mask application**,
    which is a memory‑bound operation that benefits from explicit vectorised
    loads/stores and avoids the extra Python overhead of broadcasting the mask.
*   All tensors are expected to be on a CUDA device (float32).  The final
    output is also float32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# Triton kernel: multiply a 4‑D tensor [B,N,N,H] by a broadcast mask
#               of shape [B,N,N] (float 0/1).  The mask is applied
#               element‑wise on the last dimension.
# --------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*   – input tensor flattened
    mask_ptr,         # const float*   – mask flattened (B*N*N)
    dst_ptr,          # float*         – output tensor flattened
    total_el,         # i64            – total number of elements = B*N*N*H
    H,                # i64            – hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < total_el                            # OOB mask

    # Load source value
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # One mask value per block of H consecutive elements
    mask_idx = offs // H
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Store masked result
    tl.store(dst_ptr + offs, src * mval, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Apply ``mask[...,None]`` to ``tensor`` using Triton.

    Parameters
    ----------
    tensor : torch.Tensor
        Shape [B, N, N, H], float32, CUDA.
    mask : torch.Tensor
        Shape [B, N, N], float32 (0/1), CUDA.

    Returns
    -------
    torch.Tensor
        Same shape as ``tensor`` with the mask applied.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                        # good default for H200
    grid = (triton.cdiv(total, BLOCK),)

    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)           # [B,N,N]

    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# --------------------------------------------------------------
# Entry point – called by the evaluator
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (bias‑free) → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask – apply with Triton (the required custom kernel)
    # ------------------------------------------------------------------
    if not nomask:
        left  = apply_mask_triton(left,  mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # Gating (sigmoid) – also bias‑free linear layers
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate           # [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·C) reduction via einsum (maps to efficient GEMM on Hopper)
    # ------------------------------------------------------------------
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Final LayerNorm, gating and projection back to original dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                         # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
