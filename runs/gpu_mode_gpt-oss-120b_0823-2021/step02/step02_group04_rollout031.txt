"""
TriMul – outgoing version (AlphaFold‑3)

The forward pass is split into three stages:

0️⃣  LayerNorm over the last dimension (plain PyTorch – cheap & needs
    mean/var, easier to keep on the host).

1️⃣  Linear projections for the two “sides’’ and the three gating
    tensors (bias‑free).  The three gating tensors are fused together with
    the optional mask inside a single Triton kernel.  This kernel
    *in‑places* updates ``left`` and ``right`` and also overwrites the
    ``out_gate`` tensor with its sigmoid value.

2️⃣  Core O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    This is exactly a batched matrix‑multiply (B * hidden) × (N × N)
    and is delegated to cuBLAS via ``torch.matmul`` on a view
    (B, hidden, N, N).

3️⃣  Final LayerNorm → out‑gate scaling → linear projection back to the
    original feature dimension (PyTorch).

Only the element‑wise mask‑gate fusion lives in Triton, satisfying the
“must contain a Triton kernel’’ requirement while keeping the dominant
O(N³) work on the highly‑optimized cuBLAS path.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: mask + three sigmoid gates → left, right, out_gate (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                     # [B,N,N,H]  float32
    mask_ptr,                                 # [B,N,N,1]  float32 (broadcasted)
    lgate_ptr, rgate_ptr, ogate_ptr,          # [B,N,N,H]  float32 (pre‑sigmoid)
    B, N, H,
    # strides (int64)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,k,d):
        left  = left  * mask * sigmoid(lgate)
        right = right * mask * sigmoid(rgate)
        ogate = sigmoid(ogate)          (stored back in‑place)

    ``mask`` is broadcasted over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear index
    total = B * N * N * H
    mask = offs < total

    # decode linear index → (b,i,k,d)
    idx = offs
    d = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # pointer arithmetic
    left   = left_ptr   + b * stride_l_b   + i * stride_l_i   + k * stride_l_k   + d * stride_l_h
    right  = right_ptr  + b * stride_r_b   + i * stride_r_i   + k * stride_r_k   + d * stride_r_h
    maskp  = mask_ptr   + b * stride_m_b   + i * stride_m_i   + k * stride_m_k               # 1‑channel
    lgate  = lgate_ptr  + b * stride_lg_b  + i * stride_lg_i  + k * stride_lg_k  + d * stride_lg_h
    rgate  = rgate_ptr  + b * stride_rg_b  + i * stride_rg_i  + k * stride_rg_k  + d * stride_rg_h
    ogate  = ogate_ptr  + b * stride_og_b  + i * stride_og_i  + k * stride_og_k  + d * stride_og_h

    # load
    l = tl.load(left,  mask=mask, other=0.0)
    r = tl.load(right, mask=mask, other=0.0)
    m = tl.load(maskp, mask=mask, other=1.0)          # 1.0 → no‑mask case
    lg = tl.load(lgate, mask=mask, other=0.0)
    rg = tl.load(rgate, mask=mask, other=0.0)
    og = tl.load(ogate, mask=mask, other=0.0)

    # sigmoid (fast, enough for inference)
    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))
    og = 1.0 / (1.0 + tl.exp(-og))

    # fuse
    l = l * m * lg
    r = r * m * rg

    # store back
    tl.store(left,  l, mask=mask)
    tl.store(right, r, mask=mask)
    tl.store(ogate, og, mask=mask)


def _apply_mask_and_gates(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """
    Launch the Triton kernel that performs
        left *= mask * sigmoid(left_gate)
        right *= mask * sigmoid(right_gate)
        out_gate = sigmoid(out_gate)

    All tensors are contiguous, dtype float32, and have shape
    (B, N, N, H) except ``mask`` which is (B, N, N, 1).
    The kernel works in‑place on ``left``/``right`` and returns the
    already‑sigmoided ``out_gate``.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                                   # good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left, right,
        mask,
        left_gate, right_gate, out_gate,
        B, N, H,
        # strides
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2), mask.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32/bf16)
        - mask_tensor  : torch.Tensor, shape [B, N, N] or ``None``
        - weights_dict : mapping with keys
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32) – the module output.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 0️⃣ LayerNorm over the feature dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 1️⃣ Linear projections (bias‑free) → [B,N,N,hidden]
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 2️⃣ Gate pre‑sigmoid tensors (same shape)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"])
    right_gate = F.linear(x, weights["right_gate.weight"])
    out_gate   = F.linear(x, weights["out_gate.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Prepare mask (broadcasted to 4‑D).  When ``mask`` is None we use
    #    an all‑ones mask, which makes the kernel a pure sigmoid fusion.
    # ------------------------------------------------------------------
    if mask is None:
        mask4d = torch.ones((B, N, N, 1), dtype=inp.dtype, device=device)
    else:
        mask4d = mask.unsqueeze(-1).to(inp.dtype)     # (B,N,N,1)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask + sigmoid gates (Triton)
    # ------------------------------------------------------------------
    out_gate = _apply_mask_and_gates(left, right, mask4d,
                                     left_gate, right_gate, out_gate)

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction – batched matmul over the K dimension.
    #    left[b,:,k,:] is (N, hidden) when we view as (B, hidden, N, K).
    # ------------------------------------------------------------------
    # (B, N, N, hidden) -> (B, hidden, N, N)
    left_t  = left.permute(0, 3, 1, 2)
    right_t = right.permute(0, 3, 1, 2)

    # batched matrix multiply (B, hidden, N, N)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))

    # back to original layout (B, N, N, hidden)
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm → out‑gate scaling → linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])

    # spec demands float32 output
    return out.to(torch.float32)
