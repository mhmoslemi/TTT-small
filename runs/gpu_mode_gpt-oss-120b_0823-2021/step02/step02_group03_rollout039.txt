"""
TriMul (outgoing) – fast forward only.

The dominant O(N³) contraction is a batched matrix‑matrix multiplication
```
out[b, h] = left[b, h] @ right[b, h].T          # (N,N) @ (N,N) -> (N,N)
```
where *h* indexes the hidden dimension (`hidden_dim`).  On Hopper the
cuBLAS/Tensor‑core path (`torch.bmm`) is far faster than any hand‑tuned
Triton GEMM, so we let PyTorch perform that work.

What we still need to accelerate with Triton is the *mask‑gate fusion* that
pre‑processes the two projected tensors before the batched GEMM.  The kernel
loads the projected values, multiplies by the (optional) pairwise mask,
applies three sigmoid‑gates and writes the result back – all in FP16 while
accumulating in FP32 to avoid under‑flow.  The rest of the pipeline
(LayerNorm, final linear, etc.) stays in PyTorch because it is cheap compared
to the N³ step.

The overall flow is therefore

```
x ──► LayerNorm
│
├─► Linear (left_proj)  ──►
│                         │
│                 ──► mask‑gate‑fusion kernel (left_fp16)
│
├─► Linear (right_proj) ──►
│                         │
│                 ──► mask‑gate‑fusion kernel (right_fp16)
│
└─► Linear (gate) → sigmoid (left_gate, right_gate, out_gate)

left_fp16, right_fp16  –► reshape to (B*H, N, N)
                         torch.bmm(left, right.transpose(1,2))
                         → out_fp16 (B*H, N, N)

out_fp16 –► reshape back → (B,N,N,hidden)
           ► LayerNorm → gating → final Linear → (B,N,N,dim)
```

Only the mask‑gate‑fusion kernel is written in Triton; everything else uses
high‑performance PyTorch primitives.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: fuse mask (if present) and three sigmoid gates
# -------------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    proj_ptr,          # *const fp16   [B, N, N, H]   (projected tensor)
    out_ptr,           # *mut   fp16   [B, N, N, H]   (result after mask+gate)
    mask_ptr,          # *const fp16   [B, N, N, 1]   (optional, can be nullptr)
    gate_ptr,          # *const fp16   [B, N, N, H]   (sigmoid‑gate for this proj)
    B, N, H,           # tensor sizes
    stride_pb, stride_pn, stride_pk, stride_ph,   # proj strides
    stride_ob, stride_on, stride_ok, stride_oh,   # out  strides
    stride_mb, stride_mn, stride_mk,               # mask strides (ignored if mask_ptr==0)
    stride_gb, stride_gn, stride_gk, stride_gh,   # gate strides
    BLOCK: tl.constexpr,
):
    """
    out[b, i, j, h] = proj[b,i,j,h] * mask[b,i,j,0] * gate[b,i,j,h]

    All tensors are stored in row‑major (C) order.
    """
    pid = tl.program_id(0)

    # each program works on a contiguous BLOCK‑size chunk of the (i,j) plane
    total = B * N * N
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # decode linear offset → (b,i,j)
    b = offs // (N * N)
    rem = offs % (N * N)
    i = rem // N
    j = rem % N

    # load projection
    proj_off = (
        b * stride_pb
        + i * stride_pn
        + j * stride_pk
    )
    # (H,) vector for this (b,i,j)
    proj = tl.load(proj_ptr + proj_off + tl.arange(0, H) * stride_ph,
                   mask=mask[:, None],
                   other=0.0).to(tl.float32)

    # load mask if present
    if mask_ptr != 0:
        mask_off = b * stride_mb + i * stride_mn + j * stride_mk
        m = tl.load(mask_ptr + mask_off,
                    mask=mask,
                    other=1.0).to(tl.float32)   # broadcast scalar
    else:
        m = 1.0

    # load gate (already sigmoid‑ed in PyTorch)
    gate_off = (
        b * stride_gb
        + i * stride_gn
        + j * stride_gk
        + tl.arange(0, H) * stride_gh
    )
    g = tl.load(gate_ptr + gate_off,
                mask=mask[:, None],
                other=0.0).to(tl.float32)

    # fused multiplication
    out = proj * m * g

    # store
    out_off = (
        b * stride_ob
        + i * stride_on
        + j * stride_ok
    )
    tl.store(out_ptr + out_off + tl.arange(0, H) * stride_oh,
             out.to(tl.float16),
             mask=mask[:, None])


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : (x, mask, weights, config)
        x        – (B, N, N, dim)    float32
        mask     – (B, N, N) or None float32/bool (0/1)
        weights  – dict with all linear / layer‑norm parameters
        config   – {"dim": dim, "hidden_dim": hidden_dim}

    Returns
    -------
    Tensor of shape (B, N, N, dim)  (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    ######################################################################
    # 1️⃣ LayerNorm on the input (cheap)
    ######################################################################
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    ######################################################################
    # 2️⃣ Linear projections (left, right) – keep in FP32 for now
    ######################################################################
    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    ######################################################################
    # 3️⃣ Gates (sigmoid) – also in FP32
    ######################################################################
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(dtype))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(dtype))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(dtype))
    )

    ######################################################################
    # 4️⃣ Fuse mask + gate with a tiny Triton kernel (operates in FP16)
    ######################################################################
    #   We bring the tensors to (B,N,N,hidden) → fp16 and launch a 1‑D kernel
    #   that works on blocks of the flattened (B,N,N) axis.
    #   This eliminates a separate Python loop and keeps the whole pipeline
    #   on the device.
    ######################################################################
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    left_gate_fp16  = left_gate.to(torch.float16)
    right_gate_fp16 = right_gate.to(torch.float16)

    # allocate fused outputs
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    # mask handling – convert to fp16 and add a dummy last dim
    if mask is None:
        mask_ptr = 0
        mask_fp16 = torch.tensor([], dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        mask_ptr = mask_fp16.data_ptr()

    # launch parameters
    BLOCK = 1024  # 1‑D block size; fits comfortably in registers
    total_elements = B * N * N
    grid = (triton.cdiv(total_elements, BLOCK),)

    # left side
    _mask_gate_fusion_kernel[grid](
        left_fp16,
        left_fused,
        mask_ptr,
        left_gate_fp16,
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        left_fused.stride(0), left_fused.stride(1),
        left_fused.stride(2), left_fused.stride(3),
        mask_fp16.stride(0) if mask_ptr != 0 else 0,
        mask_fp16.stride(1) if mask_ptr != 0 else 0,
        mask_fp16.stride(2) if mask_ptr != 0 else 0,
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        BLOCK=BLOCK,
    )

    # right side
    _mask_gate_fusion_kernel[grid](
        right_fp16,
        right_fused,
        mask_ptr,
        right_gate_fp16,
        B, N, hidden_dim,
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        right_fused.stride(0), right_fused.stride(1),
        right_fused.stride(2), right_fused.stride(3),
        mask_fp16.stride(0) if mask_ptr != 0 else 0,
        mask_fp16.stride(1) if mask_ptr != 0 else 0,
        mask_fp16.stride(2) if mask_ptr != 0 else 0,
        right_gate_fp16.stride(0), right_gate_fp16.stride(1),
        right_gate_fp16.stride(2), right_gate_fp16.stride(3),
        BLOCK=BLOCK,
    )

    ######################################################################
    # 5️⃣ Batched GEMM via torch.bmm (fast Tensor‑core path)
    ######################################################################
    # reshape to (B*H, N, N) for bmm
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # (B*H, N, N) @ (B*H, N, N).T  →  (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    ######################################################################
    # 6️⃣ Final LayerNorm, out‑gate and linear projection back to dim
    ######################################################################
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    out = out * out_gate  # broadcast over N,N
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,dim)

    return out
