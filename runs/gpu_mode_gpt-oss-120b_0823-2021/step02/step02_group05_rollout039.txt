"""
TriMul “outgoing” forward (AlphaFold‑3) – hybrid implementation.

*   LayerNorm, linear projections and the O(N³·C) reduction are carried out
    with native PyTorch operators (they map to highly‑optimised cuBLAS/WMMA
    kernels on Hopper and dominate the runtime).

*   The only mandatory Triton component is the pair‑wise mask‑application
    on the projected tensors (`left` and `right`).  This is a pure‑memory
    bound operation that benefits from Triton’s explicit masking and vectorised
    stores and satisfies the requirement that *some* work be performed on the
    GPU with a custom kernel.

The function `custom_kernel` follows the exact API required by the evaluator.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: broadcast mask [B,N,N] onto tensor [B,N,N,H]
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*   (input tensor, flattened)
    mask_ptr,         # const float*   (mask tensor, flattened, shape B*N*N)
    dst_ptr,          # float*         (output tensor, flattened)
    total_el,         # i64            total number of elements = B*N*N*H
    H,                # i64            hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < total_el                             # bounds guard

    # Load source value
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # Each consecutive H elements share the same mask entry.
    # Integer division truncates, giving index into mask of shape (B,N,N)
    mask_val = tl.load(mask_ptr + (offs // H), mask=mask, other=1.0)

    # Store masked result
    tl.store(dst_ptr + offs, src * mask_val, mask=mask)


def triton_apply_mask(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` of shape [B,N,N,H] by ``mask[...,None]`` using Triton.

    Parameters
    ----------
    tensor : torch.Tensor
        Input tensor (float32, CUDA) – will NOT be modified in‑place.
    mask   : torch.Tensor
        Binary mask (float32, CUDA) of shape [B,N,N].

    Returns
    -------
    torch.Tensor
        Masked tensor with the same shape as ``tensor`` (float32, CUDA).
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                      # empirically good for H200

    # Flatten to 1‑D views for the kernel
    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)          # shape (B*N*N,)

    grid = (triton.cdiv(total, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Main entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1, CUDA)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------
    # Functional layer_norm – weight/bias supplied by the model.
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask – apply with Triton (the mandated custom kernel)
    # --------------------------------------------------------------
    if not nomask:
        left = triton_apply_mask(left, mask)
        right = triton_apply_mask(right, mask)

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) + element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate      # [B,N,N,hidden_dim]
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·C) reduction: einsum over the shared “k” dimension.
    #    This maps to a sequence of GEMM calls that are already
    #    highly‑optimised on Hopper, so we keep it in PyTorch.
    # --------------------------------------------------------------
    # out shape : [B, N, N, hidden_dim]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # --------------------------------------------------------------
    # 6️⃣ Final LayerNorm (hidden_dim), gating and projection back to dim
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                                   # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure the output is float32 (all ops already use float32)
    return out.to(torch.float32)
