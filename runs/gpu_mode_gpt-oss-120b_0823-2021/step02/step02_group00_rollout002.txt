"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (high‑level)
----------------------
1.  Layer‑norm the 4‑D input (FP32, manual for speed).
2.  Four linear layers (left/right projections + three gates).  All
    weights are cast to FP16 and the mat‑muls are performed with
    torch.nn.functional.linear (still very fast on H200).
3.  A tiny Triton kernel fuses the three element‑wise operations that
    follow the linear projections:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    (the mask may be missing → identity).  The kernel works in FP16,
    writes FP16 and therefore does not create any extra Python
    overhead.
4.  The core N³ contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is reshaped into a batched matrix‑multiply and executed with
    torch.bmm, which on Hopper maps to Tensor‑Core cuBLAS kernels.
    This is the dominant work (∝ N³·C) and is the fastest path on the
    target hardware.
5.  The result is brought back to FP32, a second Layer‑Norm is applied,
    multiplied by the output‑gate and a final linear projection returns
    the tensor to its original dimension.

Only step 3 is written in Triton – the rest stays in PyTorch, which
meets the “at least part of the work must be in a kernel” rule while
keeping the implementation simple and fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – element‑wise mask & gating fusion (FP16)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fusion_kernel(
    left_ptr, right_ptr,               # [B, N, N, H] FP16
    left_gate_ptr, right_gate_ptr,     # [B, N, N, H] FP16
    mask_ptr,                          # [B, N, N]   FP16 or nullptr
    B, N, H,                           # tensor extents
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # map linear offset to (b,i,j)
    b = offs // (N * N)
    tmp = offs % (N * N)
    i = tmp // N
    j = tmp % N

    # broadcast over hidden dimension H
    offs_h = tl.arange(0, H)

    # compute flat pointers
    left  = left_ptr  + b * stride_left_b  + i * stride_left_i  + j * stride_left_j  + offs_h * stride_left_h
    right = right_ptr + b * stride_right_b + i * stride_right_i + j * stride_right_j + offs_h * stride_right_h
    lgate = left_gate_ptr  + b * stride_lgate_b  + i * stride_lgate_i  + j * stride_lgate_j  + offs_h * stride_lgate_h
    rgate = right_gate_ptr + b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + offs_h * stride_rgate_h

    # optional mask (broadcasted over H)
    if mask_ptr != 0:
        mk = mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
        m = tl.load(mk, mask=mask, other=tl.float16(1.0))
        m = m.to(tl.float16)          # scalar broadcasted later
    else:
        m = tl.float16(1.0)

    # load
    l = tl.load(left,  mask=mask[:, None] & (offs_h < H), other=tl.float16(0.0))
    r = tl.load(right, mask=mask[:, None] & (offs_h < H), other=tl.float16(0.0))
    lg = tl.load(lgate, mask=mask[:, None] & (offs_h < H), other=tl.float16(0.0))
    rg = tl.load(rgate, mask=mask[:, None] & (offs_h < H), other=tl.float16(0.0))

    # fuse
    l = l * lg * m
    r = r * rg * m

    # store back
    tl.store(left,  l, mask=mask[:, None] & (offs_h < H))
    tl.store(right, r, mask=mask[:, None] & (offs_h < H))


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask):
    """
    Fuse mask and gating in‑place via Triton.
    All tensors must be FP16, contiguous and have shape [B,N,N,H].
    """
    B, N, _, H = left.shape
    BLOCK = 8192                      # ~64 KB per launch, fits comfortably
    total = B * N * N
    grid = (triton.cdiv(total, BLOCK),)

    # strides in element units
    s_left = left.stride()
    s_right = right.stride()
    s_lgate = left_gate.stride()
    s_rgate = right_gate.stride()
    if mask is None:
        mask_ptr = 0
        s_mask = (0, 0, 0)
    else:
        mask = mask.to(torch.float16)
        s_mask = mask.stride()
        mask_ptr = mask
    mask_gate_fusion_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_ptr,
        B=B, N=N, H=H,
        stride_left_b=s_left[0], stride_left_i=s_left[1],
        stride_left_j=s_left[2], stride_left_h=s_left[3],
        stride_right_b=s_right[0], stride_right_i=s_right[1],
        stride_right_j=s_right[2], stride_right_h=s_right[3],
        stride_lgate_b=s_lgate[0], stride_lgate_i=s_lgate[1],
        stride_lgate_j=s_lgate[2], stride_lgate_h=s_lgate[3],
        stride_rgate_b=s_rgate[0], stride_rgate_i=s_rgate[1],
        stride_rgate_j=s_rgate[2], stride_rgate_h=s_rgate[3],
        stride_mask_b=s_mask[0] if mask is not None else 0,
        stride_mask_i=s_mask[1] if mask is not None else 0,
        stride_mask_j=s_mask[2] if mask is not None else 0,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return left, right


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradient needed).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf weight tensors (see reference code)
        config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                      # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (FP16)
    # --------------------------------------------------------------
    def _fp16(t): return t.to(torch.float16)

    left_proj_w   = _fp16(weights["left_proj.weight"])
    right_proj_w  = _fp16(weights["right_proj.weight"])
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)    # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 3️⃣ Mask + gating fusion (Triton)
    # --------------------------------------------------------------
    left, right = _apply_mask_and_gate(left, right,
                                       left_gate, right_gate,
                                       mask if cfg.get("nomask", True) is False else None)

    # --------------------------------------------------------------
    # 4️⃣ Core contraction – batched GEMM (N³)
    # --------------------------------------------------------------
    # reshape to (B, hidden, N, N) and (B, hidden, N, N) → (B*hidden, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden, N, N)   # (B*H, N, N)

    # batched matrix multiply: (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)                                 # fp16 accumulation

    # reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()    # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 5️⃣ Output LayerNorm (FP32)
    # --------------------------------------------------------------
    out_norm_w = weights["to_out_norm.weight"]
    out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_norm_w + out_norm_b          # FP32

    # --------------------------------------------------------------
    # 6️⃣ Final gating + projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)       # broadcast over hidden dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                # (B,N,N,dim)

    return out
```

The implementation follows the reference PyTorch logic, moves the only
non‑trivial element‑wise work (mask + gate fusion) into a custom Triton
kernel, and delegates the cubic‑time contraction to a highly‑optimized
batched GEMM (`torch.bmm`).  All intermediate tensors are kept in FP16
where it is beneficial, while the final output is returned in FP32 as
required.  This design satisfies the correctness constraints and is
well‑suited to the Hopper‑class H200 GPU.
