"""
TriMul “outgoing” forward pass (AlphaFold‑3 style)

Algorithm
---------
1. Layer‑norm the input (FP32, fused manually – cheap compared to N³ work).
2. Four linear projections + three gate‑linear layers are executed in FP16.
   * left  = x·W_left   ,  right = x·W_right
   * left_gate  = σ(x·W_lgate) , right_gate = σ(x·W_rgate) , out_gate = σ(x·W_ogate)
   * left  ← left  * left_gate      (element‑wise)
   * right ← right * right_gate
   * optional mask (broadcast over hidden_dim) is applied after projection.
3. Core contraction  
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]          (∑ over k)
   is a batched matrix‑multiply for every hidden‑dim slice *d*.
   We fuse the hidden‑dim loop into the kernel: each program processes a
   (batch, head) pair where *head* == hidden_dim (the “d” axis).  The kernel
   computes (N×N)·(N×N) → (N×N) for one head at a time, using FP16 arithmetic
   and FP32 accumulation.
   Tile sizes are tuned for Hopper/SM‑90:
        BLOCK_M = BLOCK_N = 64      # i/j tiles
        BLOCK_K = 32                # reduction‑k tile
   4 warps are used (fp16 friendly) and shared‑memory usage stays well
   below the limit.
4. The result is reshaped back to (B,N,N,hidden_dim) and cast to FP32.
5. Second Layer‑norm (FP32), multiplication by out_gate (broadcast), and a
   final linear projection back to the original dimension (FP32).

Only the N³ contraction is performed in Triton; everything else stays in
PyTorch for simplicity and to keep the launch overhead minimal.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for a single hidden‑dim slice
#   (B, H, I, K) · (B, H, K, J) → (B, H, I, J)
# ----------------------------------------------------------------------
@triton.jit
def _triton_outgoing_matmul(
    a_ptr,          # left  : (batch, head, I, K)   fp16
    b_ptr,          # right : (batch, head, K, J)   fp16
    c_ptr,          # out   : (batch, head, I, J)   fp16
    I, J, K,
    stride_az, stride_ah, stride_ai, stride_ak,   # a strides
    stride_bz, stride_bh, stride_bk, stride_bj,   # b strides
    stride_cz, stride_ch, stride_ci, stride_cj,   # c strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)          # batch*head index
    pid_m = tl.program_id(1)          # tile row (i)
    pid_n = tl.program_id(2)          # tile col (j)

    # offsets of the tile we are working on
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < I
    mask_n = offs_n < J

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile : (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile : (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot‑product, accumulate in FP32
        acc += tl.dot(a, b)

    # store result (back to FP16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_outgoing_batch_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous
    returns: (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # view as (batch*z, ...) for the kernel
    batch_dim = B * H
    a = left.view(batch_dim, I, K)
    b = right.view(batch_dim, K, J)
    c = out.view(batch_dim, I, J)

    # strides in elements (not bytes)
    stride_az, stride_ai, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bj = b.stride()
    stride_cz, stride_ci, stride_cj = c.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_dim,                # z dimension = batch * head
        triton.cdiv(I, BLOCK_M),  # rows
        triton.cdiv(J, BLOCK_N),  # cols
    )

    _triton_outgoing_matmul[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        I=I,
        J=J,
        K=K,
        stride_az=stride_az,
        stride_ah=0,  # unused because we folded head into batch_dim
        stride_ai=stride_ai,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bh=0,
        stride_bk=stride_bk,
        stride_bj=stride_bj,
        stride_cz=stride_cz,
        stride_ch=0,
        stride_ci=stride_ci,
        stride_cj=stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full forward: PyTorch + Triton
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask         : torch.Tensor [B, N, N] or None (float/bool)
        - weights      : dict of leaf tensors (see docstring above)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual – cheap)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    # mean / var over last dimension (dim)
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gates (cast to FP16 for speed)
    # --------------------------------------------------------------
    # convert once
    def to_fp16(t): return t.to(torch.float16)

    left_proj_w   = to_fp16(weights["left_proj.weight"])   # (hidden_dim, dim)
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = weights["to_out_norm.weight"]         # keep FP32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]              # (dim, hidden_dim)

    x_fp16 = x.to(torch.float16)                          # (B,N,N,dim)

    # Linear projections (no bias)
    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction – Triton batched matmul over hidden_dim slices
    #    left : (B, hidden_dim, N, N)   (i,k,d) → (B,H,I,K)
    #    right: (B, hidden_dim, N, N)   (j,k,d) → (B,H,K,J) after permute
    # --------------------------------------------------------------
    # permute so that hidden_dim becomes the "head" dimension expected by the
    # kernel.  The layout after permute is contiguous for the GEMM.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _triton_outgoing_batch_matmul(left_perm, right_perm)   # (B,H,I,J)

    # reshape back to (B,N,N,hidden_dim) and up‑cast to FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP32)
    # --------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    # --------------------------------------------------------------
    # 7️⃣ Output gating + final projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over dim
    out = F.linear(out, to_out_w)                  # (B,N,N,dim)

    return out
