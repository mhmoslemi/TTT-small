The implementation follows the reference PyTorch module but moves the **O(N³) contraction** to a custom Triton kernel.  
All other operations (layer‑norms, linear projections, gating, final‑norm, output projection) are kept in PyTorch because they are cheap compared to the contraction and benefit from the highly‑optimised cuBLAS kernels.

The algorithm
1. **Layer‑norm** of the input `x` (FP32, manual for speed).  
2. **Linear projections** (`left`, `right`) and **gates** (`left_gate`, `right_gate`, `out_gate`) are performed in **FP16**.  
3. Optional **mask** is applied to the projected tensors.  
4. **Gating** (sigmoid) is applied.  
5. **Core contraction**  
   \[
   out_{b,i,j,d}= \sum_{k=0}^{N-1} left_{b,i,k,d}\;right_{b,j,k,d}
   \]
   is evaluated as a batched matrix‑multiply for every hidden‑dimension `d`.  
   For each batch‑head pair we treat `left` and `right` as two matrices of shape `(N,N)` and compute their product with a Triton GEMM (`(N×N)·(N×N) → (N×N)`).  
   The kernel works in **FP16**, accumulates in **FP32**, and writes back FP16.  
6. Convert the result back to FP32, apply the **output‑layer‑norm**, multiply by the **output‑gate**, and finish with the final linear projection (FP32).  

The kernel is deliberately simple and tuned for Hopper (SM‑90):
* Tile sizes `BLOCK_M = BLOCK_N = 64`, `BLOCK_K = 32` (fits in shared memory, enough arithmetic intensity for Tensor‑Cores).  
* 4 warps per block – a good default for fp16 GEMM on H200.

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel : batched matrix‑multiply (I×K)·(K×J) → (I×J)
# --------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,            # left  matrix   (batch, I, K)
    b_ptr,            # right matrix   (batch, K, J)
    c_ptr,            # output matrix  (batch, I, J)
    M, N, K,          # extents
    stride_az, stride_am, stride_ak,   # a strides (batch, i, k)
    stride_bz, stride_bk, stride_bn,   # b strides (batch, k, j)
    stride_cz, stride_cm, stride_cn,   # c strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs – tile indices and the collapsed batch dimension
    pid_m = tl.program_id(0)          # row tile
    pid_n = tl.program_id(1)          # col tile
    pid_z = tl.program_id(2)          # batch (B*H)

    # start offsets of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds handling
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis K in BLOCK_K‑sized chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # a tile : (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # b tile : (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot‑product (fp16×fp16 → fp32) and accumulate
        acc += tl.dot(a, b)

    # write result (convert back to fp16 for storage)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous
    returns: (B, H, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse batch*head so the kernel sees a 3‑D tensor
    batch_dim = B * H
    a = left.view(batch_dim, I, K)
    b = right.view(batch_dim, K, J)
    c = out.view(batch_dim, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),      # rows
        triton.cdiv(J, BLOCK_N),      # cols
        batch_dim,                    # one block per (batch,head)
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # fp16‑friendly on Hopper
    )
    return out


# --------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool / float)
        weights      : dict of leaf weight tensors (see docstring)
        config       : dict with keys "dim", "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # mean / var over the last dimension (dim)
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * norm_w + norm_b    # still FP32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections & gating (cast to FP16)
    # ------------------------------------------------------------------
    # cast weights once
    def fp16(t): return t.to(torch.float16)

    left_proj_w   = fp16(weights["left_proj.weight"])   # (hidden_dim, dim)
    right_proj_w  = fp16(weights["right_proj.weight"])
    left_gate_w   = fp16(weights["left_gate.weight"])
    right_gate_w  = fp16(weights["right_gate.weight"])
    out_gate_w    = fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)   # (B, N, N, dim)

    left  = F.linear(x_fp16, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        # mask broadcast over hidden_dim
        m = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction with Triton
    #    left  : (B, N, N, hidden) → (B, H, I, K)
    #    right : (B, N, N, hidden) → (B, H, K, J)
    #    where H = hidden_dim, I = J = N, K = N
    # ------------------------------------------------------------------
    # first bring the hidden dimension to the second axis (batch‑head)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right needs to be (B, H, K, J) where K=N, J=N but transposed
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    # Batch‑GEMM in fp16 → fp16 (accumulate in fp32)
    out_fp16 = _batched_gemm_fp16(left_perm, right_perm)   # (B, H, N, N)

    # back to (B, N, N, dim) and FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + linear projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)     # broadcast over dim
    to_out_w = weights["to_out.weight"]        # (dim, dim)
    out = F.linear(out, to_out_w)              # (B, N, N, dim)

    return out
```

**Key points**

* Only the **N³ contraction** is off‑loaded to Triton; everything else uses highly‑optimised PyTorch kernels.  
* The kernel works on fused `(batch, hidden)` dimension (`B*H`) so we get a true **batched GEMM** without any extra Python loops.  
* All intermediate tensors that the kernel touches are in **fp16** to maximise Tensor‑Core throughput, while accumulations are in **fp32** to keep numerical quality.  
* The final result is returned in **float32**, matching the reference implementation.
