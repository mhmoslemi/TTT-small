"""
custom_kernel – “outgoing” TriMul forward pass (AlphaFold‑3)

Algorithm
---------
1. Layer‑Norm on the 4‑D tensor (B,N,N,C)            → fp32
2. Linear projections (C → H)                       → fp32
3. Linear gates + sigmoid                           → fp32
4. Triton kernel fuses:  mask, left‑gate, right‑gate
   (in‑place) into the projected tensors            → fp32
5. Cast the two projected tensors to fp16 and
   reshape to (B*H, N, N)                           → fp16
6. Batched GEMM (torch.bmm) over the N dimension,
   i.e. out_h = left_h @ right_hᵀ                  → fp16 (Tensor‑Core)
7. Cast result back to fp32 and restore layout      → fp32
8. Output Layer‑Norm + output‑gate                  → fp32
9. Final linear projection (H → C)                 → fp32
The dominant O(N³) contraction is performed by cuBLAS (torch.bmm);
the only custom Triton kernel fuses the element‑wise mask and gate
operations, which yields a noticeable speed‑up on the H200.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel – fuse mask + left/right gates into the projected tensors
# ----------------------------------------------------------------------
@triton.jit
def _fuse_mask_gate_kernel(
    left_ptr,          # fp32 [B,N,N,H]
    right_ptr,         # fp32 [B,N,N,H]
    mask_ptr,          # fp32 [B,N,N,1]  (broadcasted mask)
    left_gate_ptr,     # fp32 [B,N,N,H]
    right_gate_ptr,    # fp32 [B,N,N,H]
    # static sizes
    B, N, H,           # compile‑time constants
    # strides (in *elements*, not bytes)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h) perform
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    in‑place.  The kernel works on a flat view of the 4‑D tensors.
    """
    pid = tl.program_id(0)
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # ---- decode linear index into (b,i,j,h) -------------------------
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem = rem % (N * H)
    j = rem // H
    h = rem % H

    # ---- compute raw pointers ---------------------------------------
    left_off   = b * stride_left_b   + i * stride_left_i   + j * stride_left_j   + h * stride_left_h
    right_off  = b * stride_right_b  + i * stride_right_i  + j * stride_right_j  + h * stride_right_h
    mask_off   = b * stride_mask_b   + i * stride_mask_i   + j * stride_mask_j
    lgate_off  = b * stride_lgate_b  + i * stride_lgate_i  + j * stride_lgate_j  + h * stride_lgate_h
    rgate_off  = b * stride_rgate_b  + i * stride_rgate_i  + j * stride_rgate_j  + h * stride_rgate_h

    # ---- load --------------------------------------------------------
    left  = tl.load(left_ptr   + left_off,   mask=mask, other=0.0)
    right = tl.load(right_ptr  + right_off,  mask=mask, other=0.0)
    m     = tl.load(mask_ptr   + mask_off,   mask=mask, other=0.0)   # broadcasted mask (scalar)
    lgate = tl.load(left_gate_ptr + lgate_off, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr+ rgate_off, mask=mask, other=0.0)

    # ---- fuse ---------------------------------------------------------
    left  = left  * m * lgate
    right = right * m * rgate

    # ---- write back (in‑place) ----------------------------------------
    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left, right, mask, left_gate, right_gate):
    """
    Wrapper that launches the Triton kernel defined above.
    All tensors are fp32 and reside on the same CUDA device.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 2048                     # ~2 k threads → good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    s_left = left.stride()
    s_right = right.stride()
    s_mask = mask.stride()
    s_lgate = left_gate.stride()
    s_rgate = right_gate.stride()

    _fuse_mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=s_left[0], stride_left_i=s_left[1],
        stride_left_j=s_left[2], stride_left_h=s_left[3],
        stride_right_b=s_right[0], stride_right_i=s_right[1],
        stride_right_j=s_right[2], stride_right_h=s_right[3],
        stride_mask_b=s_mask[0], stride_mask_i=s_mask[1],
        stride_mask_j=s_mask[2],
        stride_lgate_b=s_lgate[0], stride_lgate_i=s_lgate[1],
        stride_lgate_j=s_lgate[2], stride_lgate_h=s_lgate[3],
        stride_rgate_b=s_rgate[0], stride_rgate_i=s_rgate[1],
        stride_rgate_j=s_rgate[2], stride_rgate_h=s_rgate[3],
        BLOCK=BLOCK,
    )
    return left, right


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N]  (float32) or None
        - weights_dict : mapping from weight name to torch.Tensor
        - config_dict  : {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) on the same device as input.
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, C = x.shape
    H = config["hidden_dim"]                  # hidden_dim == hidden

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (manual, fp32) on the last dimension
    # ------------------------------------------------------------------
    eps = 1e-5
    ln_w = weights["norm.weight"]   # (C,)
    ln_b = weights["norm.bias"]     # (C,)
    mean = x.mean(-1, keepdim=True)               # (B,N,N,1)
    var  = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)   # (B,N,N,C)
    x_norm = x_norm * ln_w + ln_b                  # broadcasted

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (C → H) – no bias
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]    # (H, C)
    right_proj_w = weights["right_proj.weight"]   # (H, C)
    left = F.linear(x_norm, left_proj_w)          # (B,N,N,H)
    right = F.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Gates (C → H) + sigmoid
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_w))   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Prepare mask (broadcast to the hidden dimension)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(B, N, N, device=device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)      # (B,N,N,1) – same dtype as x (fp32)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask + gates into left/right (in‑place) via Triton
    # ------------------------------------------------------------------
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Heavy N³ contraction – batched GEMM with Tensor Cores
    # ------------------------------------------------------------------
    # Cast to fp16 for maximum throughput
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*H, N, N) for torch.bmm
    left_flat  = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_flat = right_h.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matrix multiplication: (B*H,N,N) @ (B*H,N,N)^T -> (B*H,N,N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # back to fp32 and original layout (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    to_out_ln_w = weights["to_out_norm.weight"]   # (H,)
    to_out_ln_b = weights["to_out_norm.bias"]     # (H,)
    eps_o = 1e-5
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps_o)
    out_norm = out_norm * to_out_ln_w + to_out_ln_b

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate (already computed on hidden dim)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection (H → C) – no bias
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]          # (C, H)
    out_final = F.linear(out_norm, to_out_w)     # (B,N,N,C)

    return out_final
