"""
TriMul “outgoing” forward kernel (AlphaFold‑3).

The heavy O(N³·C) reduction is kept in PyTorch (torch.einsum maps to a
sequence of highly‑optimised GEMM kernels on Hopper).  The only
explicit Triton part is the element‑wise mask application, which is
memory‑bound and benefits from Triton’s vectorised stores and
broadcast‑mask handling.  All linear projections, LayerNorms and gating
remain in PyTorch to keep the implementation simple and numerically
identical to the reference model.

The entry‑point `custom_kernel` receives:
    (input_tensor, mask, weights, config)

and returns the tensor `[B, N, N, dim]` produced by the “outgoing” TriMul
operator.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: multiply a 4‑D tensor [B,N,N,H] by a broadcast mask
# [B,N,N] (mask is float 0/1).  One thread processes a contiguous BLOCK
# of elements; the mask index is obtained by integer‑division by H.
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*  (input tensor, flattened)
    mask_ptr,         # const float*  (mask tensor, flattened B*N*N)
    dst_ptr,          # float*        (output tensor, flattened)
    total_el,         # i64           total number of elements = B*N*N*H
    H,                # i64           size of the last dimension (hidden)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    mask = offs < total_el                            # OOB guard

    # load the source values
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # each group of H consecutive elements shares the same mask entry
    mask_idx = offs // H
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # write masked result
    tl.store(dst_ptr + offs, src * mask_val, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Apply ``mask[...,None]`` to ``tensor`` (both on CUDA) using Triton.
    The mask is 0/1 float of shape [B,N,N]; tensor is [B,N,N,H].
    Returns a new tensor with the same shape and dtype as ``tensor``.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                                       # good occupancy on H200

    # flatten for 1‑D indexing
    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)                         # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]       (float32 0/1)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                                                  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias) → shape [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask – apply with Triton (the only GPU‑custom part)
    # ------------------------------------------------------------------
    if not nomask:
        left  = apply_mask_triton(left,  mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # Gating (sigmoid) and element‑wise gating of projected tensors
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate          # [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·C) reduction via Einstein summation.
    # torch.einsum is lowered to a batch of GEMMs which are highly
    # optimised on the H200, giving the best practical performance.
    # ------------------------------------------------------------------
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Final LayerNorm (hidden_dim), gating and projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                                           # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # ensure float32 output (all ops already use float32)
    return out.to(torch.float32)
