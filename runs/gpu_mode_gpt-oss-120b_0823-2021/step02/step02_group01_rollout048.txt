"""
TriMul “outgoing” forward pass (no grads).

The costly contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched GEMM for every head d:
    left  : (B, D, I, K)   (I → i, K → k)
    right : (B, D, J, K)   (J → j, K → k)  → transpose -> (B, D, K, J)

We implement this GEMM with a single Triton kernel (fp16 compute, fp32
accumulation) and keep all the remaining LayerNorm / linear / gating
operations in PyTorch (they are cheap compared with the N³ work).

The kernel tiles the output (M‑rows × N‑cols) with BLOCK_M×BLOCK_N tiles and
reduces over K with BLOCK_K.  fp16 is used for loads and the dot product,
while the accumulator is fp32 to retain accuracy.  After the reduction the
result is cast back to fp16, stored, and later converted to fp32 for the
final LayerNorm and linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (I×K)·(K×J) → (I×J)  for all heads D
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    a_ptr,                # left  (Z, I, K)   fp16
    b_ptr,                # right (Z, K, J)   fp16
    c_ptr,                # out   (Z, I, J)   fp16
    M, N, K,              # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)   # batch*head dimension
    pid_m = tl.program_id(1)   # rows
    pid_n = tl.program_id(2)   # cols

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # fp32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load a tile  (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load b tile  (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # multiply‑accumulate
        acc += tl.dot(a, b)

    # write result (fp16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul core.

    left  : (B, D, I, K)   fp16, contiguous
    right : (B, D, K, J)   fp16, contiguous
    returns out : (B, D, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, D, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # allocate output
    out = torch.empty((B, D, I, J), device=left.device, dtype=torch.float16)

    # flatten the leading dimensions so the kernel sees a 3‑D tensor
    Z = B * D
    a = left.reshape(Z, I, K)
    b = right.reshape(Z, K, J)
    c = out.reshape(Z, I, J)

    # strides (in element counts, not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tiling parameters – tuned for Hopper (SM90) fp16 tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                            # program_id(0) → batch*head
        triton.cdiv(I, BLOCK_M),     # program_id(1) → rows
        triton.cdiv(J, BLOCK_N),     # program_id(2) → cols
    )
    batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                 # works well for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : tuple (x, mask, weights, config)
        x          : torch.Tensor [B, N, N, dim]   (float32)
        mask       : torch.Tensor [B, N, N] or None (float32/bool)
        weights    : dict of model parameters (all float32)
        config     : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm  (keep in fp32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b                      # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16) – no bias in the reference code
    # ------------------------------------------------------------------
    # cast once
    def to_fp16(t): return t.to(torch.float16, non_blocking=True)
    left_proj_w   = to_fp16(weights["left_proj.weight"])   # (dim, hidden)
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_w      = weights["to_out.weight"]               # final proj stays fp32

    # fp16 copy of normalized input
    x_fp16 = x_norm.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)    # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Mask (optional)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating (fp16)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton
    #    left  : (B,N,N,hidden) → (B,hidden,N,N) → (B,hidden,I,K)
    #    right : (B,N,N,hidden) → (B,hidden,N,N) → (B,hidden,K,J)
    # ------------------------------------------------------------------
    # move hidden dim to axis 1 and make tensors contiguous for the kernel
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # batched GEMM in fp16, result (B, H, I, J) fp16
    out_fp16 = triton_batched_gemm(left_perm, right_perm)

    # back to original layout and fp32 for the rest
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7. Final gating + linear projection (fp32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, to_out_w)          # (B,N,N,dim)

    return out
```

**Explanation of the design choices**

1. **Kernel vs. PyTorch** – The only truly cubic‑time piece is the `einsum`.  
   By reshaping the tensors we turn it into a batched matrix‑multiply that can be
   executed efficiently with a custom Triton kernel (fp16 compute, fp32
   accumulation). All cheaper ops (layer‑norms, linear gates, final projection)
   stay in PyTorch to avoid unnecessary kernel launch overhead.

2. **Mixed‑precision** – Input / weight tensors are originally fp32.  
   We cast the heavy‑path tensors to fp16 before the GEMM (the kernel loads fp16
   and uses Tensor‑cores). Accumulation is fp32 to retain numerical stability.
   The result is cast back to fp32 for the final LayerNorm and linear layers,
   matching the reference behavior.

3. **Mask handling** – When a mask is present we broadcast it to the hidden
   dimension and apply it in fp16 before gating, exactly like the reference
   implementation (`left = left * mask; right = right * mask`).

4. **Tiling parameters** – `BLOCK_M = BLOCK_N = 64` and `BLOCK_K = 32` give
   full utilisation of Hopper Tensor‑cores while staying well under shared‑memory
   limits. `num_warps=4` is a good default for fp16 on SM‑90.

5. **Strides** – The kernel receives element‑wise strides (`tensor.stride()`) and
   therefore works for any contiguous layout produced by the `permute(...).contiguous()`
   calls.

6. **Correctness** – The sequence of operations (norm → proj → mask → gates →
   contraction → out‑norm → out‑gate → final linear) reproduces the reference
   PyTorch `TriMul` forward exactly (up to floating‑point rounding differences
   caused by the fp16 compute, which are negligible for the required
   correctness tolerance).

The function `custom_kernel` can be called directly with the tuple
`(input_tensor, mask, weights, config)` and will return the processed tensor
with shape `[B, N, N, dim]` in `float32`.
