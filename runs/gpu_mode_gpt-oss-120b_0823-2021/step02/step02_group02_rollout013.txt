"""
TriMul “outgoing” operator (AlphaFold‑3) – mixed PyTorch / Triton implementation.

The expensive contraction

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

is performed by a custom Triton kernel that tiles the (i,j) plane
with BLOCK_M × BLOCK_N tiles and iterates over k in BLOCK_K chunks.
All arithmetic is carried out in FP16 with FP32 accumulation,
then the result is cast back to FP32 for the remaining
LayerNorm / gating / linear operations that stay in PyTorch.

Only the core 4‑D batched‑gemm is in Triton – everything else
(LayerNorm, linear layers, sigmoid gates, final projection) is left
to Torch‑CUDA which already uses highly‑optimized kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM over the pairwise dimension
# -------------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,          # const fp16*   [B, H, N, N]   (row‑major)
    right_ptr,         # const fp16*   [B, H, N, N]
    out_ptr,           #       fp16*   [B, H, N, N]
    B, N, H,           # int32 sizes
    # strides (int64)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_j, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute one (b, h) tile of size BLOCK_M × BLOCK_N of the output:
        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    The kernel is launched with a 1‑D grid where each program id encodes
    (b, h, tile_i, tile_j).
    """
    pid = tl.program_id(0)

    # -----------------------------------------------------------------
    # decode pid -> (b, h, tile_i, tile_j)
    # -----------------------------------------------------------------
    n_tiles_i = tl.cdiv(N, BLOCK_M)
    n_tiles_j = tl.cdiv(N, BLOCK_N)

    tile_j = pid % n_tiles_j
    pid //= n_tiles_j
    tile_i = pid % n_tiles_i
    pid //= n_tiles_i
    h = pid % H
    b = pid // H

    # -----------------------------------------------------------------
    # coordinate offsets for this tile
    # -----------------------------------------------------------------
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # loop over K dimension in BLOCK_K chunks
    # -----------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, h, i, k]  -> (M, K)
        l_ptr = (left_ptr
                 + b * stride_l_b
                 + h * stride_l_h
                 + offs_i[:, None] * stride_l_i
                 + offs_k[None, :] * stride_l_k)
        l = tl.load(l_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[b, h, j, k]  -> need (K, N) (i.e. transpose over (j,k))
        r_ptr = (right_ptr
                 + b * stride_r_b
                 + h * stride_r_h
                 + offs_k[:, None] * stride_r_k
                 + offs_j[None, :] * stride_r_j)
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # dot‑product over K
        acc += tl.dot(l, r)          # (M,K) @ (K,N) -> (M,N)

    # -----------------------------------------------------------------
    # write back result (FP16) with proper mask
    # -----------------------------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_o_b
                + h * stride_o_h
                + offs_i[:, None] * stride_o_i
                + offs_j[None, :] * stride_o_j)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper that follows the exact API required by the test harness
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ---------
    data : (input, mask, weights, config)
        input   – tensor of shape [B, N, N, dim] (float32)
        mask    – tensor of shape [B, N, N]   (bool/float, may be None)
        weights – dict with all linear / norm parameters
        config  – dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack arguments
    # -----------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (bias=False) → shape (B, N, N, hidden_dim)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden_dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # (B, N, N, 1)
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers) – same shape as left/right
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Rearrange to (B, H, N, N) and cast to FP16 for the kernel
    # -----------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output tensor in FP16, same layout as left/right
    out_fp16 = torch.empty_like(left_fp16)

    # -----------------------------------------------------------------
    # 6) Launch the Triton kernel
    # -----------------------------------------------------------------
    # tuning parameters – chosen to fit comfortably in shared memory on H200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # grid dimensions: one program per (b, h, tile_i, tile_j)
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_progs = B * hidden_dim * grid_m * grid_n
    grid = (total_progs,)

    _trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        # strides for left tensor
        left_fp16.stride(0),   # B
        left_fp16.stride(1),   # H
        left_fp16.stride(2),   # i (row)
        left_fp16.stride(3),   # k (col)
        # strides for right tensor
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(3),  # note: we need stride over j (second N)
        right_fp16.stride(2),  # stride over k
        # strides for output tensor
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7) Convert back to (B, N, N, hidden_dim) and continue in PyTorch
    # -----------------------------------------------------------------
    out = out_fp16.to(torch.float32).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # LayerNorm over hidden_dim
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # apply output gate (broadcast over the N×N plane)
    out = out * out_gate

    # final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out
