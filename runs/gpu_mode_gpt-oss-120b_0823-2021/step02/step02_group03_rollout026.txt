"""
TriMul – outgoing version (AlphaFold‑3)

The expensive part of the forward pass is the three‑dimensional contraction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

For each hidden channel *d* this is a batched matrix‑matrix multiplication
(left_h) @ (right_h)ᵀ of shape (N,N)·(N,N) → (N,N).  The implementation
splits the work into three stages:

1️⃣  Cheap per‑element work (LayerNorm, linear projections, sigmoid gates)
    stays in PyTorch – these operations are memory‑bound and extremely fast.

2️⃣  The contraction is carried out with a **single Triton kernel** that
    performs a tiled GEMM for every (batch, hidden) pair.  The kernel works
    in FP16 for bandwidth, accumulates in FP32, and writes back FP16.
    Tiling parameters (BLOCK_M, BLOCK_N, BLOCK_K) are chosen to fit comfortably
    in the shared‑memory budget of an NVIDIA H200 while giving good occupancy.

3️⃣  The result is reshaped back to (B, N, N, dim) and the final
    LayerNorm / gating / linear projection are performed again in PyTorch.

The kernel processes one tile of the output matrix (i‑range × j‑range) for a
single (batch, hidden) slice.  No synchronization across tiles is required
because each tile works on disjoint output elements.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM over the “k” dimension for every hidden channel
# -------------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_gemm_kernel(
    left_ptr,               # const fp16 *, shape (B, H, N, N)
    right_ptr,              # const fp16 *, shape (B, H, N, N)
    out_ptr,                #       fp16 *, shape (B, H, N, N)
    B, N, H,                # int32 scalars
    stride_lb, stride_lh, stride_ln, stride_lk,   # left strides
    stride_rb, stride_rh, stride_rn, stride_rk,   # right strides
    stride_ob, stride_oh, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a single (batch, hidden) tile:
        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
    """
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Decode pid → (b, h, tile_i, tile_j)
    # --------------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_M)          # tiles along i
    num_j = tl.cdiv(N, BLOCK_N)          # tiles along j
    tile_j = pid % num_j
    pid //= num_j
    tile_i = pid % num_i
    pid //= num_i
    h = pid % H
    b = pid // H

    # --------------------------------------------------------------
    # Offsets for this tile
    # --------------------------------------------------------------
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over k in BLOCK_K sized chunks
    # --------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, h, i, k]   (M x K)
        l_ptr = (
            left_ptr
            + b * stride_lb
            + h * stride_lh
            + offs_i[:, None] * stride_ln
            + offs_k[None, :] * stride_lk
        )
        l = tl.load(l_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[b, h, j, k] needs transposition → (K x N)
        r_ptr = (
            right_ptr
            + b * stride_rb
            + h * stride_rh
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_rn
        )
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # GEMM over the K dimension
        acc += tl.dot(l, r)          # (M,K) @ (K,N) → (M,N)

    # --------------------------------------------------------------
    # Write back the tile
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + h * stride_oh
        + offs_i[:, None] * stride_on
        + offs_j[None, :] * stride_ok
    )
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ----------
    data = (x, mask, weights, config)
        x          : (B, N, N, dim)            float32
        mask       : (B, N, N)  (bool/float)  or ``None``
        weights    : dict of model parameters (all float32)
        config     : {"dim": dim, "hidden_dim": hidden_dim}

    Returns
    -------
    out : (B, N, N, dim)    float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1️⃣  LayerNorm + Linear projections (cheap, stay in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 2️⃣  Apply optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 3️⃣  Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4️⃣  Rearrange to (B, H, N, N) and cast to fp16 for the kernel
    # -----------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)   # (B, H, N, N) fp16

    # -----------------------------------------------------------------
    # 5️⃣  Triton launch – tiled GEMM over K
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # number of program instances = B * H * ceil(N/BM) * ceil(N/BN)
    grid = (
        triton.cdiv(B * hidden_dim *
                   triton.cdiv(N, BLOCK_M) *
                   triton.cdiv(N, BLOCK_N), 1),
    )

    _trimul_outgoing_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        out_fp16.stride(0), out_fp16.stride(1),
        out_fp16.stride(2), out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6️⃣  Post‑kernel: reshape, LayerNorm, gating, final linear
    # -----------------------------------------------------------------
    # (B, H, N, N) -> (B, N, N, H)
    out = out_fp16.to(dtype).permute(0, 2, 3, 1).contiguous()

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate           # broadcast over N,N
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,dim)

    return out
