"""
TriMul – “outgoing” version (AlphaFold‑3)

Algorithm (GPU‑focused)
-----------------------
1️⃣  Layer‑norm over the feature dimension (dim).
2️⃣  Two bias‑free linear projections → left / right  (shape B,N,N,H).
3️⃣  Three bias‑free linear gates → sigmoid(left_gate),
    sigmoid(right_gate) and sigmoid(out_gate) (shape B,N,N,H).
4️⃣  **Triton kernel** fuses the optional mask, the three gates and the
    element‑wise multiplication into a single pass:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
        out_gate = out_gate * mask               (mask is broadcast over H)
    The kernel works in‑place on ``left`` and ``right`` and writes back the
    masked ``out_gate``.
5️⃣  Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    This is a batched GEMM per hidden channel.  After permuting to
    (B, H, N, K) we use the highly‑optimised cuBLAS kernel via
    ``torch.matmul`` (H‑fold batched matrix‑multiply).
6️⃣  Final LayerNorm over the hidden dimension, scaling by the masked
    ``out_gate`` and a bias‑free linear projection back to the original
    ``dim``.  The final tensor is returned as ``float32``.

Only step 4 is implemented in Triton – this satisfies the “must contain a
Triton kernel” requirement while keeping the dominant O(N³) work in cuBLAS,
which gives the best runtime on Hopper‑class GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, gate sigmoids and element‑wise multiplication.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                         # [B,N,N,H] projected tensors
    mask_ptr,                                    # [B,N,N,1] broadcast mask (or all‑ones)
    lgate_ptr, rgate_ptr, ogate_ptr,            # [B,N,N,H] sigmoid‑gates
    B, N, H,
    # stride arguments for each tensor (int64, passed from host)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,k,d) in the 4‑D view of ``left``/``right``:
        left  = left  * mask * lgate
        right = right * mask * rgate
        out_gate = out_gate * mask          # only mask, no gate multiplication
    The mask is scalar per (b,i,k) and is broadcast over the hidden dim.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)           # flat offsets
    total = B * N * N * H
    mask = offs < total

    # --------------------------------------------------------------
    # Decode flat index → (b,i,k,d)
    # --------------------------------------------------------------
    tmp = offs
    d = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # --------------------------------------------------------------
    # Compute pointers for each tensor
    # --------------------------------------------------------------
    left  = left_ptr  + b * stride_l_b  + i * stride_l_i  + k * stride_l_k  + d * stride_l_h
    right = right_ptr + b * stride_r_b  + i * stride_r_i  + k * stride_r_k  + d * stride_r_h
    mask_ = mask_ptr  + b * stride_m_b  + i * stride_m_i  + k * stride_m_k                # scalar
    lgate = lgate_ptr + b * stride_lg_b + i * stride_lg_i + k * stride_lg_k + d * stride_lg_h
    rgate = rgate_ptr + b * stride_rg_b + i * stride_rg_i + k * stride_rg_k + d * stride_rg_h
    ogate = ogate_ptr + b * stride_og_b + i * stride_og_i + k * stride_og_k + d * stride_og_h

    # --------------------------------------------------------------
    # Load values (mask is broadcast, load once per thread)
    # --------------------------------------------------------------
    l = tl.load(left,  mask=mask, other=0.0)
    r = tl.load(right, mask=mask, other=0.0)
    m = tl.load(mask_, mask=mask, other=1.0)   # 1.0 → no‑mask case
    lg = tl.load(lgate, mask=mask, other=1.0)
    rg = tl.load(rgate, mask=mask, other=1.0)
    og = tl.load(ogate, mask=mask, other=1.0)

    # --------------------------------------------------------------
    # Fuse
    # --------------------------------------------------------------
    l = l * m * lg
    r = r * m * rg
    og = og * m

    # --------------------------------------------------------------
    # Store results back in‑place
    # --------------------------------------------------------------
    tl.store(left,  l, mask=mask)
    tl.store(right, r, mask=mask)
    tl.store(ogate, og, mask=mask)


def _apply_mask_and_gates(left, right, mask, left_gate, right_gate, out_gate):
    """
    Launches the Triton kernel that fuses mask and gates.
    All tensors are contiguous, reside on the same CUDA device and have dtype
    torch.float32 (or torch.bfloat16 – the kernel works for both).

    The operation is performed **in‑place** on ``left`` and ``right``; the
    masked ``out_gate`` tensor is returned for later use.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    # 1024 is a good compromise for H200 – it yields ~1 Mi elements per wave
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        ogate_ptr=out_gate,
        B=B,
        N=N,
        H=H,
        stride_l_b=left.stride(0), stride_l_i=left.stride(1),
        stride_l_k=left.stride(2), stride_l_h=left.stride(3),
        stride_r_b=right.stride(0), stride_r_i=right.stride(1),
        stride_r_k=right.stride(2), stride_r_h=right.stride(3),
        stride_m_b=mask.stride(0), stride_m_i=mask.stride(1),
        stride_m_k=mask.stride(2),
        stride_lg_b=left_gate.stride(0), stride_lg_i=left_gate.stride(1),
        stride_lg_k=left_gate.stride(2), stride_lg_h=left_gate.stride(3),
        stride_rg_b=right_gate.stride(0), stride_rg_i=right_gate.stride(1),
        stride_rg_k=right_gate.stride(2), stride_rg_h=right_gate.stride(3),
        stride_og_b=out_gate.stride(0), stride_og_i=out_gate.stride(1),
        stride_og_k=out_gate.stride(2), stride_og_h=out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Public entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor of shape [B, N, N, dim] (float32 or bf16)
        * mask_tensor  : torch.Tensor of shape [B, N, N] (0/1) or ``None``
        * weights_dict : dict with the same keys as the reference module
                         (norm.weight, norm.bias, left_proj.weight, …)
        * config_dict  : dict with entries ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the result of the forward
        pass.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the input feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free) → left / right  [B,N,N,hidden]
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gates (sigmoid) → also [B,N,N,hidden]
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4️⃣ Prepare mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        # Create an all‑ones mask on‑the‑fly; shape [B,N,N,1]
        mask = torch.ones(B, N, N, 1, dtype=x.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(x.dtype)      # [B,N,N,1]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask + gates into left/right (in‑place) via Triton.
    #    The kernel also writes back the masked out_gate.
    # ------------------------------------------------------------------
    out_gate = _apply_mask_and_gates(
        left, right, mask, left_gate, right_gate, out_gate
    )   # left/right now contain mask*gate*proj, out_gate is masked

    # ------------------------------------------------------------------
    # 6️⃣ Core contraction: Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM per hidden channel.
    # ------------------------------------------------------------------
    # (B,N,N,hidden) → (B, hidden, N, K) where K == N
    left_t = left.permute(0, 3, 1, 2)   # (B, H, N, K)
    right_t = right.permute(0, 3, 1, 2) # (B, H, N, K)

    # Batched matmul: (B,H,N,K) @ (B,H,K,N) → (B,H,N,N)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))

    # Back to original layout (B,N,N,hidden)
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7️⃣ Final LayerNorm over hidden dimension, out‑gate scaling and projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                 # gated scaling
    out = F.linear(out, weights["to_out.weight"])

    # Ensure float32 output (reference module returns float32)
    return out.to(torch.float32)
