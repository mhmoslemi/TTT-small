"""
TriMul “outgoing” (AlphaFold‑3) – fast forward pass.

The expensive step is a batched matrix‑multiply over the pairwise dimension

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

where `left`/`right` are the gated projections of shape
[B, H, N, N].  The kernel tiling strategy (BLOCK_M × BLOCK_N tiles,
BLOCK_K‑wise accumulation) computes this contraction in FP16 with
FP32 accumulation and writes the result back as FP16.  All other
operations (LayerNorm, linear projections, gating, final LayerNorm
and linear) are performed with PyTorch for simplicity and numerical
stability.

The implementation works for any batch size (≤2), sequence length
(≤1024) and hidden dimension (≤384) required by the test‑suite.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel – batched GEMM for a single hidden slice
# -------------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,          # const float16*  [B, H, N, N]
    right_ptr,         # const float16*  [B, H, N, N]
    out_ptr,           #       float16*  [B, H, N, N]
    B, N, H,           # tensor sizes (int)
    # strides (int) – left
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    # strides (int) – right
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    # strides (int) – out
    stride_o_b, stride_o_h, stride_o_i, stride_o_k,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a tile of
        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    The left operand is stored row‑major (i,k) while the right operand is
    accessed as (j,k).  We load a (M,K) tile from ``left`` and a (K,N) tile
    from ``right`` (by transposing the last two dims on‑the‑fly) and perform
    a dot product.  Accumulation is done in FP32 for accuracy.
    """
    pid = tl.program_id(0)                     # linear id over (b, h, tile_i, tile_j)

    # -------------------------------------------------
    # decode pid → (b, h, tile_i, tile_j)
    # -------------------------------------------------
    num_tile_i = tl.cdiv(N, BLOCK_M)
    num_tile_j = tl.cdiv(N, BLOCK_N)

    tile_j = pid % num_tile_j
    pid //= num_tile_j
    tile_i = pid % num_tile_i
    pid //= num_tile_i
    h = pid % H
    b = pid // H

    # -------------------------------------------------
    # start offsets for this tile
    # -------------------------------------------------
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # loop over K dimension in BLOCK_K steps
    # -------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, h, i, k]   → (M, K)
        l_ptr = (left_ptr
                 + b   * stride_l_b
                 + h   * stride_l_h
                 + offs_i[:, None] * stride_l_i
                 + offs_k[None, :] * stride_l_k)
        l = tl.load(l_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[b, h, j, k] → need (K, N) → we load (k, j)
        r_ptr = (right_ptr
                 + b   * stride_r_b
                 + h   * stride_r_h
                 + offs_k[:, None] * stride_r_k
                 + offs_j[None, :] * stride_r_i)   # note stride_r_i = stride over “i” (the second N)
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # (M,K) @ (K,N) → (M,N)
        acc += tl.dot(l, r)

    # -------------------------------------------------
    # write result (cast back to FP16)
    # -------------------------------------------------
    out_ptrs = (out_ptr
                + b   * stride_o_b
                + h   * stride_o_h
                + offs_i[:, None] * stride_o_i
                + offs_j[None, :] * stride_o_k)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data :
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask         : [B, N, N]        (bool/float, may be None)
        - weights      : dict of torch.Parameters (no bias in linear layers)
        - config       : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    ########################################################################
    # 1. LayerNorm + linear projections (pure PyTorch)
    ########################################################################
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # linear projections (bias=False)
    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    ########################################################################
    # 2. Optional mask
    ########################################################################
    if mask is not None:
        # mask shape (B,N,N,1) – broadcast over hidden dim
        mask_ = mask.unsqueeze(-1).to(dtype)
        left = left * mask_
        right = right * mask_

    ########################################################################
    # 3. Gating (sigmoid)
    ########################################################################
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left = left * left_gate
    right = right * right_gate

    ########################################################################
    # 4. Rearrange to [B, H, N, N] and cast to fp16 for the kernel
    ########################################################################
    # (B,N,N,H) → (B,H,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)   # shape (B, H, N, N), fp16

    ########################################################################
    # 5. Triton kernel launch
    ########################################################################
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # grid = total number of (b, h) tiles × tile_i × tile_j
    grid_bh = B * hidden_dim
    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    total_tiles = grid_bh * grid_i * grid_j
    grid = (total_tiles,)

    _trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        # strides for left tensor
        left_fp16.stride(0),   # B
        left_fp16.stride(1),   # H
        left_fp16.stride(2),   # i (row)
        left_fp16.stride(3),   # k (col)
        # strides for right tensor
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        # strides for output tensor
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    ########################################################################
    # 6. Back‑to original layout + final LayerNorm & linear
    ########################################################################
    # (B, H, N, N) → (B, N, N, H)
    out = out_fp16.to(dtype).permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply output gate (broadcast over N,N)
    out = out * out_gate

    # final projection back to `dim`
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
