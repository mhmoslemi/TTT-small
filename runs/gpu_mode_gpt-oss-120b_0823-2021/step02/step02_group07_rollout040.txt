"""
custom_kernel – “outgoing” TriMul forward (AlphaFold3)

Algorithm
---------
1️⃣ Layer‑Norm on the 4‑D tensor (B,N,N,C) – pure PyTorch, fp32.
2️⃣ Linear projection (C → H) for *left* and *right* tensors – fp32.
3️⃣ Linear gates → sigmoid – fp32.
4️⃣ **Triton kernel** fuses:
        *mask* (if present)  ×  *gate*  ×  *proj*
   into the left/right tensors **in‑place**.  This is the only
   compute‑heavy region that benefits from the GPU’s massive
   parallelism.
5️⃣ Cast the fused tensors to fp16 and reshape to
   (B*H, N, N).  A batched GEMM (torch.bmm) performs the
   N³ contraction (the einsum) using Tensor‑Cores – the
   dominant cost of the module.
6️⃣ Layer‑Norm, output‑gate and final linear (fp32) – PyTorch.

Only the element‑wise fusing step is written in Triton; everything
else stays in PyTorch because cuBLAS already gives near‑optimal
throughput for the cubic contraction.
"""
import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ------------------------------------------------------------------
# Triton kernel – fuse mask, left‑gate and right‑gate into the projected
# tensors.  All tensors are contiguous in fp32.
# ------------------------------------------------------------------
@triton.jit
def _fuse_mask_gates_kernel(
    left_ptr,          #  [B,N,N,H]  fp32
    right_ptr,         #  [B,N,N,H]  fp32
    mask_ptr,          #  [B,N,N,1]  fp32  (broadcasted mask)
    left_gate_ptr,     #  [B,N,N,H]  fp32
    right_gate_ptr,    #  [B,N,N,H]  fp32
    B, N, H,           # compile‑time constants
    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h) compute
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    in‑place.
    """
    pid = tl.program_id(0)
    total = B * N * N * H                # number of scalar elements
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total                  # out‑of‑bounds guard

    # decode linear offset → (b,i,j,h)
    b = offs // (N * N * H)
    rest = offs % (N * N * H)
    i = rest // (N * H)
    rest = rest % (N * H)
    j = rest // H
    h = rest % H

    # compute flat addresses
    left_off  = b * stride_left_b  + i * stride_left_i  + j * stride_left_j  + h * stride_left_h
    right_off = b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h
    mask_off  = b * stride_mask_b  + i * stride_mask_i  + j * stride_mask_j
    lgate_off = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h
    rgate_off = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h

    # loads (mask is broadcasted, other = 0.0 for safety)
    left   = tl.load(left_ptr   + left_off,   mask=mask, other=0.0)
    right  = tl.load(right_ptr  + right_off,  mask=mask, other=0.0)
    m      = tl.load(mask_ptr   + mask_off,   mask=mask, other=0.0)
    lgate  = tl.load(left_gate_ptr + lgate_off, mask=mask, other=0.0)
    rgate  = tl.load(right_gate_ptr+ rgate_off, mask=mask, other=0.0)

    # fuse
    left  = left  * m * lgate
    right = right * m * rgate

    # store back in‑place
    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left, right, mask, left_gate, right_gate):
    """
    Call the Triton kernel on contiguous fp32 tensors.
    The tensors are expected to have shape (B,N,N,H) except *mask*
    which is (B,N,N,1) (broadcasted).
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 2048                     # enough threads for good occupancy
    grid = (triton.cdiv(total, BLOCK),)

    # element strides (in *elements*, not bytes)
    s_left   = left.stride()
    s_right  = right.stride()
    s_mask   = mask.stride()
    s_lgate  = left_gate.stride()
    s_rgate  = right_gate.stride()

    _fuse_mask_gates_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=s_left[0],   stride_left_i=s_left[1],
        stride_left_j=s_left[2],   stride_left_h=s_left[3],
        stride_right_b=s_right[0], stride_right_i=s_right[1],
        stride_right_j=s_right[2], stride_right_h=s_right[3],
        stride_mask_b=s_mask[0],   stride_mask_i=s_mask[1],
        stride_mask_j=s_mask[2],
        stride_lgate_b=s_lgate[0], stride_lgate_i=s_lgate[1],
        stride_lgate_j=s_lgate[2], stride_lgate_h=s_lgate[3],
        stride_rgate_b=s_rgate[0], stride_rgate_i=s_rgate[1],
        stride_rgate_j=s_rgate[2], stride_rgate_h=s_rgate[3],
        BLOCK=BLOCK,
    )
    return left, right


# ------------------------------------------------------------------
# Main entry‑point required by the evaluation harness
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32, same device as input)
    """
    x, mask, weights, config = data
    dim = config["dim"]            # input channel dimension C
    hidden = config["hidden_dim"]  # projection dimension H
    device = x.device
    eps = 1e-5

    # -------------------------------------------------
    # 1️⃣ LayerNorm over the last dim (fp32)
    # -------------------------------------------------
    ln_w = weights["norm.weight"]   # (C,)
    ln_b = weights["norm.bias"]     # (C,)
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * ln_w + ln_b                     # (B,N,N,C)

    # -------------------------------------------------
    # 2️⃣ Linear projections (C → H) – no bias
    # -------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (H, C)
    right_proj_w = weights["right_proj.weight"]  # (H, C)

    # torch.nn.functional.linear expects (out_features, in_features)
    left = F.linear(x_norm, left_proj_w)   # (B,N,N,H)
    right = F.linear(x_norm, right_proj_w) # (B,N,N,H)

    # -------------------------------------------------
    # 3️⃣ Gates + sigmoid (C → H)
    # -------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = F.linear(x_norm, left_gate_w).sigmoid()   # (B,N,N,H)
    right_gate = F.linear(x_norm, right_gate_w).sigmoid()
    out_gate   = F.linear(x_norm, out_gate_w).sigmoid()   # (B,N,N,H)

    # -------------------------------------------------
    # 4️⃣ Broadcast mask (if missing treat as all‑ones)
    # -------------------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2],
                          device=device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)            # (B,N,N,1)

    # -------------------------------------------------
    # 5️⃣ Fuse mask + gates into left/right (Triton)
    # -------------------------------------------------
    # ensure tensors are contiguous for the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask.contiguous()
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # -------------------------------------------------
    # 6️⃣ Batched GEMM – Tensor‑Core friendly (fp16)
    # -------------------------------------------------
    B, N, _, H = left.shape
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*H, N, N) -> each head is an independent matrix
    left_flat = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right_h.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # out_flat[b*H + h, i, j] = Σ_k left_flat[b*H + h, i, k] * right_flat[b*H + h, j, k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # back to fp32 and original layout (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # -------------------------------------------------
    # 7️⃣ Output LayerNorm (fp32)
    # -------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]   # (H,)
    out_ln_b = weights["to_out_norm.bias"]     # (H,)

    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * out_ln_w + out_ln_b

    # -------------------------------------------------
    # 8️⃣ Apply output gate (element‑wise)
    # -------------------------------------------------
    out_norm = out_norm * out_gate          # (B,N,N,H)

    # -------------------------------------------------
    # 9️⃣ Final linear (H → C) – no bias
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"]     # (C, H)
    out_final = F.linear(out_norm, to_out_w)   # (B,N,N,C)

    return out_final
