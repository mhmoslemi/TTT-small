"""
TriMul – outgoing version (AlphaFold‑3)

The kernel fuses the element‑wise mask and the three sigmoid gates
(left, right, out) into the projected tensors.  All other
operations (LayerNorm, linear projections, the N³ contraction and the
final LayerNorm + projection) are performed with PyTorch because
cuBLAS/Tensor‑cores give the best performance on Hopper GPUs.

Algorithm
---------
1. LayerNorm on the input (dim‑wise) – PyTorch.
2. Linear projections (bias‑free) → left/right (B,N,N,hidden).
3. Linear projections for the three gates (pre‑sigmoid) → same shape.
4. Triton kernel:
       left   ← left   * mask * sigmoid(left_gate)
       right  ← right  * mask * sigmoid(right_gate)
       out_gate ← sigmoid(out_gate)      (stored in‑place)
   The mask is broadcasted over the hidden dimension.
5. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched matrix multiply after permuting to
   (B,hidden,N,N) → (B*hidden, N, N).
6. Final LayerNorm, element‑wise scaling by out_gate and linear projection
   back to dim – PyTorch.
7. Return float32 tensor of shape (B,N,N,dim).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: mask + 3 sigmoid gates → left, right, out_gate (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,               # [B,N,N,H]  float32 (proj tensors)
    mask_ptr,                           # [B,N,N,1]  float32 (broadcasted)
    lgate_ptr, rgate_ptr, ogate_ptr,   # [B,N,N,H]  float32 (pre‑sigmoid)
    B, N, H,                            # shape parameters
    # stride arguments (int64)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k, stride_mask_1,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    stride_ogate_b, stride_ogate_i, stride_ogate_k, stride_ogate_h,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,k,d) (the two “pair” indices are i and k,
    the hidden dim is d):
        left   = left   * mask * sigmoid(lgate)
        right  = right  * mask * sigmoid(rgate)
        ogate  = sigmoid(ogate)          # stored back in‑place
    ``mask`` is broadcasted over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear index
    total = B * N * N * H
    mask = offs < total

    # ---- decode linear index to (b,i,k,d) --------------------------------
    idx = offs
    d = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # ---- compute pointers --------------------------------------------------
    left   = left_ptr   + b * stride_left_b   + i * stride_left_i   + k * stride_left_k   + d * stride_left_h
    right  = right_ptr  + b * stride_right_b  + i * stride_right_i  + k * stride_right_k  + d * stride_right_h
    mask_p = mask_ptr   + b * stride_mask_b   + i * stride_mask_i   + k * stride_mask_k                     # 1‑channel
    lgate  = lgate_ptr  + b * stride_lgate_b  + i * stride_lgate_i  + k * stride_lgate_k  + d * stride_lgate_h
    rgate  = rgate_ptr  + b * stride_rgate_b  + i * stride_rgate_i  + k * stride_rgate_k  + d * stride_rgate_h
    ogate  = ogate_ptr  + b * stride_ogate_b  + i * stride_ogate_i  + k * stride_ogate_k  + d * stride_ogate_h

    # ---- load --------------------------------------------------------------
    l  = tl.load(left,   mask=mask, other=0.0)
    r  = tl.load(right,  mask=mask, other=0.0)
    m  = tl.load(mask_p, mask=mask, other=1.0)   # 1.0 → no mask
    lg = tl.load(lgate,  mask=mask, other=0.0)
    rg = tl.load(rgate,  mask=mask, other=0.0)
    og = tl.load(ogate,  mask=mask, other=0.0)

    # ---- sigmoid (numeric stable) -----------------------------------------
    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))
    og = 1.0 / (1.0 + tl.exp(-og))

    # ---- fuse ---------------------------------------------------------------
    l = l * m * lg
    r = r * m * rg

    # ---- store back ---------------------------------------------------------
    tl.store(left,  l, mask=mask)
    tl.store(right, r, mask=mask)
    tl.store(ogate, og, mask=mask)


def _apply_mask_and_gates(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """
    Launches ``_mask_gate_fuse_kernel`` which updates ``left`` and ``right`` in‑place
    and returns the sigmoided ``out_gate`` tensor.
    All tensors are (B,N,N,H) except ``mask`` which is (B,N,N,1).
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                                   # good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left,
        right,
        mask,
        left_gate,
        right_gate,
        out_gate,
        B,
        N,
        H,
        # strides
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2), mask.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate  # now contains sigmoid(out_gate)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor  [B, N, N, dim]  (float32 / bf16)
        - mask_tensor  : torch.Tensor  [B, N, N]  or ``None``
        - weights_dict : mapping with keys
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------------
    # 0️⃣ Input LayerNorm (dim‑wise)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 1️⃣ Linear projections (bias‑free) → left / right  [B,N,N,hidden]
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 2️⃣ Gate projections (pre‑sigmoid) → same shape
    # --------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"])
    right_gate = F.linear(x, weights["right_gate.weight"])
    out_gate   = F.linear(x, weights["out_gate.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Prepare mask (broadcast to hidden dim)
    # --------------------------------------------------------------
    if mask is None:
        mask4d = torch.ones((B, N, N, 1), dtype=input_tensor.dtype, device=device)
    else:
        mask4d = mask.unsqueeze(-1).to(input_tensor.dtype)

    # --------------------------------------------------------------
    # 4️⃣ Fuse mask + sigmoid(gates) into left/right (in‑place) and
    #     sigmoid(out_gate) – Triton kernel
    # --------------------------------------------------------------
    out_gate = _apply_mask_and_gates(left, right, mask4d,
                                     left_gate, right_gate, out_gate)

    # --------------------------------------------------------------
    # 5️⃣ Core contraction:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Perform as a batched GEMM on (B*hidden, N, N)
    # --------------------------------------------------------------
    # reshape to (B*hidden, N, N)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, K)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)  # (B*H, N, K)

    # batched matmul: (B*H, N, N) * (B*H, N, N)^T  → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1)

    # --------------------------------------------------------------
    # 6️⃣ Final LayerNorm, out‑gate scaling, linear projection back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # element‑wise gate
    out = F.linear(out, weights["to_out.weight"])

    # --------------------------------------------------------------
    # 7️⃣ Ensure float32 output as required by the spec
    # --------------------------------------------------------------
    return out.to(torch.float32)
