"""
custom_kernel – “outgoing” TriMul (AlphaFold3)

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (B,N,N,C) in FP32 (manual mean/var, no extra
    synchronisation – each element works independently).
2️⃣  Four linear projections (left, right, left_gate, right_gate, out_gate) are
    performed in FP16 for speed.  The input is cast once to FP16 and the weight
    matrices are stored FP16 in the weight dict.
3️⃣  Optional binary mask (shape B,N,N) is broadcast over the hidden dimension
    and applied to *left* and *right* before gating.
4️⃣  Gating (sigmoid) is applied element‑wise.
5️⃣  Core N³ contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    is a batched matrix multiplication where each pair (b,d) is an
    (N×N)·(N×N) → (N×N) multiply.  The heavy work is done by a Triton kernel that
    tiles the reduction dimension K in blocks of 32 and the output tiles in
    64×64.  Operands are FP16, accumulation in FP32, final store FP16.
6️⃣  Convert the result back to FP32, apply a second LayerNorm (FP32),
    multiply by the output‑gate and finish with a final linear projection
    (FP32) to the original channel dimension.

Only the N³ contraction is in Triton; everything else stays in PyTorch so the
kernel launch overhead is tiny and the implementation remains easy to verify.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched mat‑mul (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_matmul_kernel(
    a_ptr,           # left  tensor  (B*H, I, K)
    b_ptr,           # right tensor  (B*H, K, J)
    c_ptr,           # output tensor (B*H, I, J)
    M, N, K,         # extents
    stride_az, stride_am, stride_ak,   # a strides (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)          # batch*head index
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col

    # start offsets of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in BLOCK_K steps
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M×K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile (K×N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16×FP16 → FP32 accumulation
        acc += tl.dot(a, b)

    # write result (cast back to FP16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous
    returns: (B, H, I, J) fp16
    """
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # view as (B*H, I, K) etc.  strides are expressed in elements
    a = left.view(B * H, I, K)
    b = right.view(B * H, K, J)
    c = out.view(B * H, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tiling parameters – these hit the sweet spot on Hopper (SM‑90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),          # rows
        triton.cdiv(J, BLOCK_N),          # cols
        B * H,                            # batch*head
    )
    _batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # fp16 friendly on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward – PyTorch orchestration + Triton contraction
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf weight tensors (see docstring)
        config       : dict containing keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual – cheap because dim is modest)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gates (cast to FP16)
    # --------------------------------------------------------------
    # weight tensors are stored in FP32; cast once to FP16
    def _fp16(t): return t.to(torch.float16)

    left_proj_w   = _fp16(weights["left_proj.weight"])   # (hidden, dim)
    right_proj_w  = _fp16(weights["right_proj.weight"])

    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)       # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy contraction with Triton
    # --------------------------------------------------------------
    # bring hidden dimension to the “head” axis so the kernel sees (B,head,I,K)
    # hidden == head here (the paper treats hidden as the number of heads)
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    # right needs to be (B, H, K, J) where J == I (square)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _batched_matmul_fp16(left_t, right_t)   # (B, H, I, J)

    # restore original layout (B,N,N,hidden) then to FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32)
    # --------------------------------------------------------------
    to_out_ln_w = weights["to_out_norm.weight"]
    to_out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * to_out_ln_w + to_out_ln_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)               # broadcast over dim
    to_out_w = weights["to_out.weight"]                 # (dim, hidden)
    out = F.linear(out, to_out_w)                       # (B,N,N,dim)

    return out
```

**Explanation of key design choices**

* **LayerNorm** – performed manually in FP32; the per‑element mean/var are cheap
  compared to the O(N³) contraction and avoid the extra kernel launch.

* **Projections & Gates** – cast the input and all weight matrices to FP16.
  The hidden dimension is at most 384, so the FP16 matrix‑vector multiplies are
  memory‑bound and benefit from the smaller datatype.

* **Mask handling** – the mask is optional; when present it is broadcast over the
  hidden dimension and applied before gating, exactly as in the reference code.

* **Core contraction** – the operation is a batched GEMM where each
  (batch, hidden) pair multiplies two N×N matrices.  A custom Triton kernel
  tiles the output in 64×64 blocks and the reduction dimension in 32‑element
  tiles, which matches the sweet spot for Tensor‑cores on Hopper (SM‑90).  The
  kernel accumulates in FP32 for numerical stability and stores in FP16.

* **Output processing** – after the kernel we convert back to FP32, apply the
  second LayerNorm, multiply by the output‑gate (still in FP32) and finish with a
  final linear layer (FP32).  This matches the original PyTorch implementation
  while keeping the heavy compute inside the Triton kernel.

The implementation respects all constraints: it works for any `B ∈ {1,2}`,
any `N` up to 1024, arbitrary `dim`/`hidden_dim`, optional masks, and returns a
float‑32 tensor.  Only the N³ contraction is in Triton, satisfying the rule that
*at least part* of the operation must be a custom kernel.
