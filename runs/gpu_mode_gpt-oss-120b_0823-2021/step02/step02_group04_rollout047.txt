"""
TriMul – outgoing version (AlphaFold‑3)

The kernel only handles the element‑wise “mask + gate” fusion that is the
bottleneck for a pure‑Python implementation.  All other heavy‑weight
operations (LayerNorm, bias‑free linear projections, the N³ contraction and the
final projection) are performed with native PyTorch which already uses
high‑performance cuBLAS/Tensor‑Core kernels on Hopper.

Pipeline
---------
0️⃣ Layer‑norm over the last dimension (dim)                – PyTorch
1️⃣ Linear projections (left/right) and three gate
    projections (pre‑sigmoid)                               – PyTorch
2️⃣ Triton kernel                                            – mask × sigmoid(gates)
   * left  ← left  * mask * sigmoid(left_gate)
   * right ← right * mask * sigmoid(right_gate)
   * out_gate ← sigmoid(out_gate)   (stored in‑place)
3️⃣ Core contraction (out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d])
   – reshaped to (B, H, N, K) and executed with torch.matmul   – PyTorch
4️⃣ Final LayerNorm, out‑gate scaling and linear projection   – PyTorch
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate, right‑gate and out‑gate
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # [B,N,N,H]   float32
    right_ptr,         # [B,N,N,H]   float32
    mask_ptr,          # [B,N,N,1]   float32 (broadcasted)
    lgate_ptr, rgate_ptr, ogate_ptr,   # [B,N,N,H] pre‑sigmoid
    B, N, H,
    # strides (int64)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """In‑place fusion:
          left  = left  * mask * sigmoid(left_gate)
          right = right * mask * sigmoid(right_gate)
          out_gate = sigmoid(out_gate)
    The mask is broadcasted over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear index
    total = B * N * N * H
    mask = offs < total

    # ----- decode linear index (b,i,k,d) ---------------------------------
    idx = offs
    d = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # ----- pointer arithmetic --------------------------------------------
    left   = left_ptr   + b * stride_l_b   + i * stride_l_i   + k * stride_l_k   + d * stride_l_h
    right  = right_ptr  + b * stride_r_b   + i * stride_r_i   + k * stride_r_k   + d * stride_r_h
    mask_  = mask_ptr   + b * stride_m_b   + i * stride_m_i   + k * stride_m_k                # 1‑channel
    lgate  = lgate_ptr  + b * stride_lg_b  + i * stride_lg_i  + k * stride_lg_k  + d * stride_lg_h
    rgate  = rgate_ptr  + b * stride_rg_b  + i * stride_rg_i  + k * stride_rg_k  + d * stride_rg_h
    ogate  = ogate_ptr  + b * stride_og_b  + i * stride_og_i  + k * stride_og_k  + d * stride_og_h

    # ----- load -----------------------------------------------------------
    l_val = tl.load(left,  mask=mask, other=0.0)
    r_val = tl.load(right, mask=mask, other=0.0)
    m_val = tl.load(mask_, mask=mask, other=1.0)        # 1.0 when mask is omitted
    lg_val = tl.load(lgate, mask=mask, other=0.0)
    rg_val = tl.load(rgate, mask=mask, other=0.0)
    og_val = tl.load(ogate, mask=mask, other=0.0)

    # ----- sigmoid (numerically stable, fast) ----------------------------
    lg_sig = 1.0 / (1.0 + tl.exp(-lg_val))
    rg_sig = 1.0 / (1.0 + tl.exp(-rg_val))
    og_sig = 1.0 / (1.0 + tl.exp(-og_val))

    # ----- fuse -----------------------------------------------------------
    l_val = l_val * m_val * lg_sig
    r_val = r_val * m_val * rg_sig

    # ----- store back -----------------------------------------------------
    tl.store(left,  l_val, mask=mask)
    tl.store(right, r_val, mask=mask)
    tl.store(ogate, og_sig, mask=mask)


def _mask_gate_fuse(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """
    Launches the Triton kernel that:
        * multiplies mask & sigmoid(left_gate) into ``left`` (in‑place)
        * multiplies mask & sigmoid(right_gate) into ``right`` (in‑place)
        * sigmoid‑s the ``out_gate`` tensor (in‑place, also returned)

    All tensors are 4‑D with shape (B, N, N, H) except ``mask`` which is
    (B, N, N, 1).  dtype must be float32.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    # 1024 works well on Hopper for the typical hidden size (≤384)
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left, right,
        mask,
        left_gate, right_gate, out_gate,
        B, N, H,
        # strides
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2), mask.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate  # already sigmoid‑ed


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        • input_tensor : torch.Tensor  [B, N, N, dim]   (float32 / bf16)
        • mask_tensor  : torch.Tensor  [B, N, N]  or None
        • weights_dict : mapping with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        • config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 0️⃣ Layer‑norm over the ``dim`` dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 1️⃣ Linear projections (bias‑free) → [B, N, N, hidden]
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 2️⃣ Gate pre‑sigmoid tensors
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"])
    right_gate = F.linear(x, weights["right_gate.weight"])
    out_gate   = F.linear(x, weights["out_gate.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Broadcast mask to 4‑D (or use ones)
    # ------------------------------------------------------------------
    if mask is None:
        mask4d = torch.ones((B, N, N, 1), dtype=x.dtype, device=device)
    else:
        mask4d = mask.unsqueeze(-1).to(x.dtype)   # (B,N,N,1)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask & sigmoid‑gates (Triton)
    # ------------------------------------------------------------------
    out_gate = _mask_gate_fuse(left, right, mask4d, left_gate, right_gate, out_gate)

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched matrix multiply (cuBLAS/Tensor‑Core)
    # ------------------------------------------------------------------
    # reshape to (B, hidden, N, K)
    left_t  = left.permute(0, 3, 1, 2)          # (B, H, N, K)
    right_t = right.permute(0, 3, 1, 2)         # (B, H, N, K)

    # batched GEMM on the K dimension
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))   # (B, H, N, N)

    # back to original layout
    out = out_t.permute(0, 2, 3, 1)            # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm → out‑gate scaling → linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                       # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])

    # Ensure the required output dtype
    return out.to(torch.float32)
