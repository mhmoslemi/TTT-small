"""
TriMul – outgoing version (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm on the input channel dim                     (torch)
2️⃣  Linear projections → left / right (B,N,N,H)            (torch)
3️⃣  Linear gates → left_gate, right_gate, out_gate         (torch‑sigmoid)
4️⃣  Triton kernel fuses:
        • optional mask (broadcast over H)
        • left‑gate & right‑gate applied in‑place to left/right
        • out_gate masked (mask‑only, no gating)
   This is the only non‑trivial region that must run on the GPU.
5️⃣  Core contraction  Σ_k left[b,i,k,d] * right[b,j,k,d]   → (B,N,N,H)
    Implemented as a batched GEMM per hidden channel (cuBLAS) which is
    far faster than a hand‑written N³ kernel.
6️⃣  Final LayerNorm on the hidden dim, out‑gate scaling and
    linear projection back to the original dim                (torch)

Only step 4 is written in Triton; everything else uses highly‑optimized
PyTorch kernels.  The Triton kernel works on an arbitrary batch size,
sequence length and hidden dimension, and supports the presence or absence
of a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate, right‑gate and out‑gate.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] projected tensors
    mask_ptr,                          # [B,N,N,1] broadcast mask (or all‑ones)
    lgate_ptr, rgate_ptr, ogate_ptr,   # [B,N,N,H] sigmoid‑gates
    B, N, H,
    # strides (int64) of the tensors above
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """
    In‑place fuse mask and gates.
        left  = left  * mask * lgate
        right = right * mask * rgate
        out_gate = out_gate * mask                (mask only, gate already applied)
    The mask is a scalar per (b,i,k) and is broadcast over H.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # flat offsets
    total = B * N * N * H
    active = offs < total

    # ------------------------------------------------------------------
    # Decode linear index → (b,i,k,d)
    # ------------------------------------------------------------------
    idx = offs
    d = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # ------------------------------------------------------------------
    # Pointer arithmetic
    # ------------------------------------------------------------------
    left  = left_ptr   + b * stride_l_b   + i * stride_l_i   + k * stride_l_k   + d * stride_l_h
    right = right_ptr  + b * stride_r_b   + i * stride_r_i   + k * stride_r_k   + d * stride_r_h
    mask  = mask_ptr   + b * stride_m_b   + i * stride_m_i   + k * stride_m_k                 # 1‑channel
    lg    = lgate_ptr  + b * stride_lg_b  + i * stride_lg_i  + k * stride_lg_k  + d * stride_lg_h
    rg    = rgate_ptr  + b * stride_rg_b  + i * stride_rg_i  + k * stride_rg_k  + d * stride_rg_h
    og    = ogate_ptr  + b * stride_og_b  + i * stride_og_i  + k * stride_og_k  + d * stride_og_h

    # ------------------------------------------------------------------
    # Load (mask is broadcast, so load a scalar once)
    # ------------------------------------------------------------------
    l  = tl.load(left,  mask=active, other=0.0)
    r  = tl.load(right, mask=active, other=0.0)
    m  = tl.load(mask,  mask=active, other=1.0)   # 1.0 when mask is omitted
    lg = tl.load(lg,    mask=active, other=1.0)
    rg = tl.load(rg,    mask=active, other=1.0)
    og = tl.load(og,    mask=active, other=1.0)

    # ------------------------------------------------------------------
    # Fuse
    # ------------------------------------------------------------------
    l = l * m * lg
    r = r * m * rg
    og = og * m                     # out_gate only needs the mask

    # ------------------------------------------------------------------
    # Store back
    # ------------------------------------------------------------------
    tl.store(left,  l,  mask=active)
    tl.store(right, r,  mask=active)
    tl.store(og,    og, mask=active)


def _apply_mask_and_gates(left, right, mask, left_gate, right_gate, out_gate):
    """
    Launch the Triton kernel to fuse mask & gates.  Operates in‑place on
    `left` and `right`; returns the masked `out_gate`.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    # Empirically good for Hopper‑class GPUs – 1024 threads gives good occupancy
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        ogate_ptr=out_gate,
        B=B,
        N=N,
        H=H,
        stride_l_b=left.stride(0), stride_l_i=left.stride(1),
        stride_l_k=left.stride(2), stride_l_h=left.stride(3),
        stride_r_b=right.stride(0), stride_r_i=right.stride(1),
        stride_r_k=right.stride(2), stride_r_h=right.stride(3),
        stride_m_b=mask.stride(0), stride_m_i=mask.stride(1),
        stride_m_k=mask.stride(2), stride_m_1=mask.stride(3),
        stride_lg_b=left_gate.stride(0), stride_lg_i=left_gate.stride(1),
        stride_lg_k=left_gate.stride(2), stride_lg_h=left_gate.stride(3),
        stride_rg_b=right_gate.stride(0), stride_rg_i=right_gate.stride(1),
        stride_rg_k=right_gate.stride(2), stride_rg_h=right_gate.stride(3),
        stride_og_b=out_gate.stride(0), stride_og_i=out_gate.stride(1),
        stride_og_k=out_gate.stride(2), stride_og_h=out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Public entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor of shape [B, N, N, dim] (float32/bf16)
        * mask_tensor  : torch.Tensor of shape [B, N, N] with 0/1 or ``None``.
        * weights_dict : mapping with the same keys as the reference
                         implementation (LayerNorm weight/bias, Linear weights,
                         etc.).
        * config_dict  : dictionary containing ``dim`` and ``hidden_dim``.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the forward result.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (over dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections → left/right  (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gates (sigmoid) → same shape
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4️⃣ Prepare mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(B, N, N, 1, dtype=x.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask & gates via Triton (in‑place on left/right)
    # ------------------------------------------------------------------
    out_gate = _apply_mask_and_gates(
        left, right, mask, left_gate, right_gate, out_gate
    )   # left/right are now masked & gated; out_gate holds masked gate

    # ------------------------------------------------------------------
    # 6️⃣ Core contraction Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Reshape to (B*hidden, N, N) and use torch.bmm (cuBLAS) – fast.
    # ------------------------------------------------------------------
    # (B,N,N,hidden) → (B*hidden, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # Batched matmul: (B*H, N, N) @ (B*H, N, N)^T → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore layout (B,N,N,hidden)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7️⃣ Final LayerNorm on hidden dim, out‑gate scaling and projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # gated scaling
    out = F.linear(out, weights["to_out.weight"])

    # Cast back to float32 as required by the reference implementation
    return out.to(torch.float32)
