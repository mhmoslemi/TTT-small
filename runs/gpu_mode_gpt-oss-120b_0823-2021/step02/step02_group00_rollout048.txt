"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  LayerNorm on the last channel (FP32, manual for speed and to avoid an extra
    Triton sync).
2️⃣  Four linear projections (left/right + three gates).  The projection
    weights are stored in FP16 and the intermediate tensors are kept in FP16.
3️⃣  Optional mask is applied to the left/right tensors (broadcast over the
    hidden dimension).
4️⃣  Gating – sigmoid on the three gate tensors.
5️⃣  Core contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] • right[b,j,k,d]      (d = hidden_dim)
    is a batched matrix‑multiply:
        (B, hidden, N, N)   ×   (B, hidden, N, N)ᵀ   →   (B, hidden, N, N)
    Implemented with a custom Triton kernel that tiles the (i,j) plane
    (BLOCK_M × BLOCK_N) and reduces over k (BLOCK_K).  Operands are FP16,
    accumulation is FP32 and the result is written back as FP16.
6️⃣  Convert the result back to (B,N,N,dim) FP32, apply a second LayerNorm,
    multiply by the output‑gate, and finally a linear projection back to
    the original dimension.

The kernel is deliberately simple, tuned for Hopper/SM‑90:
    BLOCK_M = BLOCK_N = 64    # tiles in the output (i,j) plane
    BLOCK_K = 32               # tile in reduction dimension
    num_warps = 4 (FP16 friendly)
All strides are passed in element units (not bytes) to avoid extra work
inside the kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched (B, H, N, N) × (B, H, N, N).T → (B, H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def batch_tri_mul_kernel(
    left_ptr,      # (B*H, N, K)   K = N
    right_ptr,     # (B*H, N, K)   K = N
    out_ptr,       # (B*H, N, N)
    N, K,          # matrix extents (K == N)
    stride_lz, stride_lm, stride_lk,   # left strides (batch, i, k)
    stride_rz, stride_rk, stride_rn,   # right strides (batch, k, n)
    stride_oz, stride_om, stride_on,   # out strides   (batch, i, n)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_z = tl.program_id(2)          # batch*head

    # start offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left:  (batch, i, k)
        a = tl.load(
            left_ptr
            + pid_z * stride_lz
            + offs_m[:, None] * stride_lm
            + offs_k[None, :] * stride_lk,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right: (batch, k, n)
        b = tl.load(
            right_ptr
            + pid_z * stride_rz
            + offs_k[:, None] * stride_rk
            + offs_n[None, :] * stride_rn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16 × FP16 → FP32 accumulation
        acc += tl.dot(a, b)

    # store result as FP16
    c = acc.to(tl.float16)
    tl.store(
        out_ptr
        + pid_z * stride_oz
        + offs_m[:, None] * stride_om
        + offs_n[None, :] * stride_on,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def triton_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, N, K)   FP16, contiguous
    right : (B, H, N, K)   FP16, contiguous   (K == N)
    returns: (B, H, N, N) FP16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, N, K = left.shape
    assert K == N and right.shape == (B, H, N, K)

    out = torch.empty((B, H, N, N), device=left.device, dtype=torch.float16)

    # collapse batch*head dimension for the kernel
    batch = B * H
    l = left.reshape(batch, N, K)
    r = right.reshape(batch, N, K)
    o = out.reshape(batch, N, N)

    # element‑wise strides (not bytes)
    stride_lz, stride_lm, stride_lk = l.stride()
    stride_rz, stride_rk, stride_rn = r.stride()
    stride_oz, stride_om, stride_on = o.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch,                     # batch*head
    )
    batch_tri_mul_kernel[grid](
        left_ptr=l,
        right_ptr=r,
        out_ptr=o,
        N=N,
        K=K,
        stride_lz=stride_lz,
        stride_lm=stride_lm,
        stride_lk=stride_lk,
        stride_rz=stride_rz,
        stride_rk=stride_rk,
        stride_rn=stride_rn,
        stride_oz=stride_oz,
        stride_om=stride_om,
        stride_on=stride_on,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        - weights      : dict of leaf tensors
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape          # dim == config["dim"]
    hidden_dim = config["hidden_dim"]          # == c_z in the paper

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm on last channel (FP32)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # mean / var over channel dimension
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * norm_w + norm_b                    # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gating (cast to FP16)
    # --------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16)

    # projection matrices: (hidden_dim, dim)
    left_proj_w  = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])

    # gate matrices: (hidden_dim, dim)
    left_gate_w  = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w   = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)               # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)     # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        # mask shape → (B,N,N,1) and broadcast over hidden_dim
        m = mask.to(torch.float16).unsqueeze(-1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction – Triton batched GEMM
    # --------------------------------------------------------------
    # reshape to (B, hidden, N, N) for the kernel
    left_t  = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_t = right.permute(0, 3, 2, 1).contiguous()   # (B, H, N, N)

    # batched matmul (FP16, accum FP32)
    out_fp16 = triton_tri_mul(left_t, right_t)        # (B, H, N, N)

    # back to (B, N, N, dim) and FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)           # broadcast over dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                    # (B,N,N,dim)

    return out
```

**Explanation of lesson relevance**

*Lesson 1 (shared‑memory limits)* – The kernel uses modest tile sizes
`64 × 64 × 32`, which keep shared memory well below the H200 limit; no extra
stages are needed.

*Lesson 2 (delegate N³ to cuBLAS)* – While a pure `torch.bmm` would be fast,
the problem statement requires that **some part** of the contraction be
implemented in Triton.  We therefore keep the dominant O(N³) work in a
custom kernel but still use the same algorithmic structure as a batched GEMM,
which is essentially what cuBLAS would do.

*Lesson 3 (matching dtypes)* – Both operands of the kernel are cast to
`float16` before the multiply; accumulation is performed in `float32` and the
result is written back as `float16`.  This matches the recommendation and
avoids type‑mismatch errors.

The rest of the pipeline (LayerNorms, gating, final projection) stays in
PyTorch because they are cheap compared with the O(N³) contraction and would
add unnecessary kernel‑launch overhead.  The overall implementation satisfies
the specification, supports all listed configurations, and provides a fast
forward‑only TriMul operator.
