The implementation follows the reference PyTorch module but moves the **O(N³) contraction**

```
out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
```

to a custom Trit‑on kernel.  
All other ops (layer‑norm, linear projections, gating, final layer‑norm and projection) are kept in PyTorch because they are cheap compared to the N³ work.  
The kernel treats each `(batch, head)` pair as an independent matrix‑multiply  

```
L   : (H, N, K)   where H = B * hidden_dim
Rᵀ : (H, N, K)   (right is transposed on the last two axes)
out : (H, N, N)
```

and computes `out = L @ Rᵀ` with fp16 inputs, fp32 accumulation and fp16 write‑back.  
Tiling is 64 × 64 with a reduction tile of 32 (tuned for Hopper/SM90).  

```python
"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1. Layer‑norm the input (FP32, manual).
2. Linear projections (left/right) and three gates (left, right, out) – all in FP16.
3. Optional mask is applied to the projected tensors.
4. Gating (sigmoid) is applied.
5. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM in a Triton kernel:
       left_perm  : (B, H, N, K)  = left.permute(0,3,1,2)
       right_perm : (B, H, K, N)  = right.permute(0,3,2,1)
   The kernel computes (B*H, N, K) @ (B*H, K, N) → (B*H, N, N) in FP16/FP32.
6. Convert back to (B,N,N,dim) FP32, apply a second LayerNorm.
7. Multiply by the output‑gate and project back to the original dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched matrix multiplication (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr,            # left  matrix   (batch, I, K)
    b_ptr,            # right matrix   (batch, K, J)
    c_ptr,            # output matrix  (batch, I, J)
    M, N, K,          # matrix extents
    stride_az, stride_am, stride_ak,   # a strides (batch, i, k)
    stride_bz, stride_bk, stride_bn,   # b strides (batch, k, j)
    stride_cz, stride_cm, stride_cn,   # c strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1‑D program IDs (tile indices)
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)          # tile along M (rows)
    pid_n = tl.program_id(1)          # tile along N (cols)
    pid_z = tl.program_id(2)          # batch * head index

    # ------------------------------------------------------------------
    # 2‑D offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # 3‑D accumulator (FP32 for numerical stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4‑loop over K dimension (reduction)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile : (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile : (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate (fp16·fp16 → fp32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # 5‑store result (down‑cast to fp16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous
    returns: (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # Collapse batch & head into a single dimension for the kernel
    batch = B * H
    a = left.view(batch, I, K)
    b = right.view(batch, K, J)
    c = out.view(batch, I, J)

    # Strides in element units (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes – good defaults for SM90 fp16 GEMM
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),          # rows
        triton.cdiv(J, BLOCK_N),          # cols
        batch,                            # one program per (batch,head)
    )
    _batched_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # fp16 friendly on Hopper
    )
    return out


# ----------------------------------------------------------------------
# Helper: manual LayerNorm (FP32) – avoids extra memory copies
# ----------------------------------------------------------------------
def _layer_norm_fp32(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps=1e-5):
    # x : (..., C)  float32
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Entry point – the required `custom_kernel` function
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]  (float32)
        - mask         : torch.Tensor of shape [B, N, N] or None (float/bool)
        - weights      : dict of leaf weight tensors
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = _layer_norm_fp32(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # still FP32, shape (B,N,N,dim)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gates – convert to fp16 for speed
    # --------------------------------------------------------------
    # we keep a single view of x in fp16
    x_fp16 = x.to(torch.float16)

    # projection weights: (hidden_dim, dim)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    # gate weights: (hidden_dim, dim)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # linear projections
    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction via Triton batched GEMM
    # --------------------------------------------------------------
    # reshape to (B, H, I, K) where H = hidden_dim
    left_perm  = left.permute(0, 3, 1, 2).contiguous()    # (B, H, I, K)
    # right must be transposed on the last two dims to match the einsum
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, H, K, J)

    # batched matmul in fp16
    out_fp16 = _batched_matmul_fp16(left_perm, right_perm)   # (B, H, I, J)

    # back to (B, N, N, dim) and FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP32)
    # --------------------------------------------------------------
    out = _layer_norm_fp32(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 7️⃣ Output gating and final projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)   # broadcast over dim
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
```
