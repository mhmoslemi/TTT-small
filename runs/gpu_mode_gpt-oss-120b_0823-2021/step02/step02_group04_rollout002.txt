"""
TriMul – outgoing version (AlphaFold‑3)

The forward pass consists of six stages:

0️⃣  Layer‑norm over the last dimension (dim) – done in PyTorch.
1️⃣  Two bias‑free linear projections (left/right) and three gate
    projections (pre‑sigmoid) – PyTorch.
2️⃣  Fuse the optional mask and the three sigmoids into the projected
    tensors with a single Triton kernel (in‑place on ``left``/``right``,
    returns the already‑sigmoided ``out_gate``).
3️⃣  Core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    – realised as a batched GEMM after permuting to (B, H, N, K).  The heavy
    N³ work is delegated to cuBLAS (torch.matmul) which is optimal on Hopper.
4️⃣  Final Layer‑norm, element‑wise out‑gate scaling and linear projection back
    to the original dimension – PyTorch.
5️⃣  Cast the final result to float32 (the spec requires FP32 output).

Only the element‑wise mask‑gate fusion is implemented in Triton, satisfying the
requirement that the solution “must contain a Triton kernel” while keeping the
overall runtime close to the theoretical optimum on an NVIDIA H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: mask + three sigmoid gates → left, right (in‑place) &
# out_gate (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                 # [B,N,N,H]  (float32)
    mask_ptr,                            # [B,N,N,1]  (float32, broadcasted)
    lgate_ptr, rgate_ptr, ogate_ptr,    # [B,N,N,H]  (float32, pre‑sigmoid)
    B, N, H,
    # strides (int64)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k, stride_mask_1,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    stride_ogate_b, stride_ogate_i, stride_ogate_k, stride_ogate_h,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,k,d):
        left  = left  * mask * sigmoid(lgate)
        right = right * mask * sigmoid(rgate)
        ogate = sigmoid(ogate)           (stored back in‑place)

    The mask is broadcasted over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)      # linear index
    total = B * N * N * H
    mask = offs < total

    # decode linear index → (b,i,k,d)
    idx = offs
    d = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # compute pointers (note: mask has no hidden stride)
    left   = left_ptr   + b * stride_left_b   + i * stride_left_i   + k * stride_left_k   + d * stride_left_h
    right  = right_ptr  + b * stride_right_b  + i * stride_right_i  + k * stride_right_k  + d * stride_right_h
    mask_p = mask_ptr   + b * stride_mask_b   + i * stride_mask_i   + k * stride_mask_k   # 1‑channel
    lgate  = lgate_ptr  + b * stride_lgate_b  + i * stride_lgate_i  + k * stride_lgate_k  + d * stride_lgate_h
    rgate  = rgate_ptr  + b * stride_rgate_b  + i * stride_rgate_i  + k * stride_rgate_k  + d * stride_rgate_h
    ogate  = ogate_ptr  + b * stride_ogate_b  + i * stride_ogate_i  + k * stride_ogate_k  + d * stride_ogate_h

    # load (mask is 0/1, default to 1 when mask is absent)
    l = tl.load(left,  mask=mask, other=0.0)
    r = tl.load(right, mask=mask, other=0.0)
    m = tl.load(mask_p, mask=mask, other=1.0)          # 1.0 → no‑mask case
    lg = tl.load(lgate, mask=mask, other=0.0)
    rg = tl.load(rgate, mask=mask, other=0.0)
    og = tl.load(ogate, mask=mask, other=0.0)

    # sigmoid (fast, accurate enough for inference)
    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))
    og = 1.0 / (1.0 + tl.exp(-og))

    # fuse
    l = l * m * lg
    r = r * m * rg

    # store back
    tl.store(left,  l, mask=mask)
    tl.store(right, r, mask=mask)
    tl.store(ogate, og, mask=mask)


def _apply_mask_and_gates(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """
    Launches ``_mask_gate_fuse_kernel`` which updates ``left`` and ``right`` in‑place
    and returns ``out_gate`` already passed through sigmoid.
    All tensors have shape (B, N, N, H) except ``mask`` which is (B, N, N, 1).
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                                 # fits well in registers on H200
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left, right,
        mask,
        left_gate, right_gate, out_gate,
        B, N, H,
        # strides
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2), mask.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        BLOCK=BLOCK,
    )
    # ``out_gate`` now contains sigmoid(out_gate)
    return out_gate


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor of shape [B, N, N, dim] (fp32 or bf16)
        - mask_tensor  : torch.Tensor of shape [B, N, N] or None
        - weights_dict : keys
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 0️⃣ Input LayerNorm (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 1️⃣ Linear projections (bias‑free) → [B,N,N,hidden]
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 2️⃣ Gate pre‑sigmoid projections → same shape
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"])
    right_gate = F.linear(x, weights["right_gate.weight"])
    out_gate   = F.linear(x, weights["out_gate.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Prepare mask (broadcasted to 4‑D) – if no mask use ones
    # ------------------------------------------------------------------
    if mask is None:
        mask4d = torch.ones((B, N, N, 1), dtype=inp.dtype, device=device)
    else:
        mask4d = mask.unsqueeze(-1).to(inp.dtype)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask + sigmoid(gates) into left/right (in‑place) and out_gate
    # ------------------------------------------------------------------
    out_gate = _apply_mask_and_gates(left, right, mask4d,
                                     left_gate, right_gate, out_gate)

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction (batched GEMM over K)
    #     out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to (B, hidden, N, K) where K == N
    left_t  = left.permute(0, 3, 1, 2)          # (B, hidden, N, K)
    right_t = right.permute(0, 3, 1, 2)         # (B, hidden, N, K)

    # batched matmul – uses cuBLAS/Tensor‑cores
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))   # (B, hidden, N, N)

    # back to original layout
    out = out_t.permute(0, 2, 3, 1)            # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm, out‑gate scaling, linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                      # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])

    # --------------------------------------------------------------
    # Ensure FP32 output (the spec requires float32)
    # --------------------------------------------------------------
    return out.to(torch.float32)
