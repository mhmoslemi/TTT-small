"""
TriMul “outgoing” forward (AlphaFold‑3).

- LayerNorm, linear projections, gating and the O(N³·C) reduction are kept in
  PyTorch because they map to highly‑optimised CUDA kernels (especially the
  einsum which is a batched GEMM on Hopper).
- The only hard‑to‑fuse part that is memory‑bound – the optional element‑wise
  mask applied to the projected tensors – is implemented in Triton.  This
  satisfies the “use a custom kernel” requirement while keeping overall
  performance excellent.
- The Triton kernel works for any batch size, sequence length and hidden
  dimension.  It multiplies a tensor of shape [B, N, N, H] by a broadcast mask
  of shape [B, N, N] (float 0/1) without extra temporaries.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: broadcast mask multiplication
# ----------------------------------------------------------------------
@triton.jit
def mask_mul_kernel(
    src_ptr,          # const float*  (input [B,N,N,H] flattened)
    mask_ptr,         # const float*  (mask [B,N,N] flattened)
    dst_ptr,          # float*        (output)
    total_el,         # i64           total number of elements (=B*N*N*H)
    H,                # i64           hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    off = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    m = off < total_el                                # OOB mask

    # Load the source value (float32)
    src = tl.load(src_ptr + off, mask=m, other=0.0)

    # Each consecutive block of H elements shares the same mask entry.
    # Integer division (off // H) gives the index into the mask tensor.
    mask_val = tl.load(mask_ptr + (off // H), mask=m, other=1.0)

    # Store masked result
    tl.store(dst_ptr + off, src * mask_val, mask=m)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` of shape [B,N,N,H] by ``mask[...,None]`` using Triton.
    ``mask`` is float (0/1) with shape [B,N,N].

    Returns a new tensor with the same shape as ``tensor`` (still on CUDA).
    """
    B, N, _, H = tensor.shape
    total_el = B * N * N * H
    BLOCK = 1024                                          # good balance on H200

    src = tensor.view(-1)                                 # flatten
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)                             # [B,N,N]

    grid = (triton.cdiv(total_el, BLOCK),)

    mask_mul_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total_el,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry‑point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (the only custom‑kernel part)
    # ------------------------------------------------------------------
    if not nomask:
        left  = apply_mask_triton(left,  mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # Gating (sigmoid) and element‑wise gating of projected tensors
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate        # [B, N, N, hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·C) reduction via batched GEMM (einsum)
    # ------------------------------------------------------------------
    # out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Final LayerNorm, gating and projection back to original dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                         # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure float32 output
    return out.to(torch.float32)
