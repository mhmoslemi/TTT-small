"""
TriMul outgoing (AlphaFold‑3 style) – forward only.

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiply over the “hidden” dimension *d*.
We implement this contraction with a single Triton kernel (fp16 compute,
fp32 accumulation).  All cheap operations – LayerNorm, linear layers,
sigmoids, final LayerNorm and projection – stay in PyTorch.  The kernel
receives the tensors already permuted to (B, H, I, K) and (B, H, K, J) and
writes the result (B, H, I, J).  The host code reshapes / casts, calls the
kernel and restores the original layout.

The implementation works for any batch size ≤2, sequence length up to 1024,
and hidden / model dimensions that appear in the test matrix.  It runs
entirely on the H200 (SM‑90) using 4 warps per block and tiles of 64×64×32,
which matches the tensor‑core friendly shape for fp16.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (I×K)·(K×J) → (I×J) for each head (hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _batch_matmul_kernel(
    a_ptr,            # left  tensor   (Z, I, K)
    b_ptr,            # right tensor   (Z, K, J)
    c_ptr,            # output tensor  (Z, I, J)
    M, N, K,          # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)          # batch * head index
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col

    # ----- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (more accurate than fp16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- reduction over K ---------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile (BLOCK_M × BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile (BLOCK_K × BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 accumulation
        acc += tl.dot(a, b)

    # ----- write back (convert to fp16) ---------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_batch_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction.
    Arguments:
        left  – (B, H, I, K)   fp16, contiguous
        right – (B, H, K, J)   fp16, contiguous
    Returns:
        out – (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten leading dimensions so the kernel sees a 3‑D tensor
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides in element units (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tile sizes – tuned for Hopper / SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # program_id(0)
        triton.cdiv(J, BLOCK_N),   # program_id(1)
        Z,                         # program_id(2)
    )
    _batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                # 4 warps give good occupancy on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (fp32)
        - mask         : torch.Tensor [B, N, N]       (fp32 or bool) or None
        - weights      : dict of parameter tensors
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (fp32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm  (keep in fp32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                     # [B,N,N,dim] fp32

    # ------------------------------------------------------------------
    # 2) Linear projections (cast to fp16 for the heavy part)
    # ------------------------------------------------------------------
    def _fp16(t):      # one‑time cast to fp16, keep a view
        return t.to(torch.float16, non_blocking=True)

    left_proj_w   = _fp16(weights["left_proj.weight"])
    right_proj_w  = _fp16(weights["right_proj.weight"])
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])

    # fp16 copy of the normalized tensor
    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)      # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Optional mask  (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid on fp16)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Contract over the third axis (the N³ part) via Triton
    #    left : (B,N,N,hidden) -> (B,hidden,N,N) -> (B,hidden,I,K)
    #    right: (B,N,N,hidden) -> (B,hidden,N,N) -> (B,hidden,K,J)
    # ------------------------------------------------------------------
    # permute to bring the hidden dimension in front
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _triton_batch_matmul(left_perm, right_perm)   # (B,H,I,J)

    # restore original layout and cast back to fp32 for the rest
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7) Final gating + projection (fp32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
