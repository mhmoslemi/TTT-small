"""
custom_kernel – “outgoing” TriMul forward pass (AlphaFold‑3)

Algorithm (high‑level)
----------------------
1️⃣  Layer‑Norm on the 4‑D input  (B,N,N,C) – fp32, manual.
2️⃣  Linear projections  (C → H)   – fp32, bias‑free.
3️⃣  Linear gates + sigmoid       – fp32.
4️⃣  **Triton kernel** fuses mask, left‑gate and right‑gate into the
    projected tensors (in‑place, fp32).  This eliminates three separate
    element‑wise passes and is the only region that must run on the GPU
    with Triton.
5️⃣  Cast the fused tensors to float16 and reshape to
    (B*H, N, N).  The cubic contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is an Einstein‑sum that is exactly a batched matrix‑multiply:
        out_flat = bmm(left_flat, right_flat.transpose(1,2))
    Using torch.bmm (cuBLAS/Tensor‑Core) is far faster than a hand‑written
    N³ Triton kernel.
6️⃣  Cast the result back to fp32, restore the original layout
    (B,N,N,H).
7️⃣  Output Layer‑Norm, output‑gate and final linear projection – fp32.

Only step‑4 is implemented in Triton; the rest stays in PyTorch because
they are either cheap (norm, linear) or already heavily optimized
(batched GEMM).  The kernel is deliberately simple, works for any
batch‑size (1‑2), any sequence length and any hidden dimension, and
handles the “no‑mask’’ case by treating a missing mask as all‑ones.
"""

import torch
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel – fuse mask and gates into left/right tensors (in‑place)
# ------------------------------------------------------------------
@triton.jit
def _fuse_mask_gate_kernel(
    left_ptr,          # fp32  [B,N,N,H]
    right_ptr,         # fp32  [B,N,N,H]
    mask_ptr,          # fp32  [B,N,N]  (broadcasted on H)
    lgate_ptr,         # fp32  [B,N,N,H]
    rgate_ptr,         # fp32  [B,N,N,H]

    B, N, H,           # compile‑time constants
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,

    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offset
    mask = offs < B * N * N * H                       # out‑of‑bounds guard

    # ---------- decode linear offset → (b,i,j,h) ----------
    b = offs // (N * N * H)
    rest = offs % (N * N * H)
    i = rest // (N * H)
    rest = rest % (N * H)
    j = rest // H
    h = rest % H

    # ---------- pointer arithmetic ----------
    left_off  = b * stride_left_b  + i * stride_left_i  + j * stride_left_j  + h * stride_left_h
    right_off = b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h
    mask_off  = b * stride_mask_b  + i * stride_mask_i  + j * stride_mask_j
    lgate_off = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h
    rgate_off = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h

    # ---------- loads ----------
    left  = tl.load(left_ptr  + left_off,  mask=mask, other=0.0)
    right = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    m     = tl.load(mask_ptr  + mask_off,  mask=mask, other=0.0)   # broadcasted scalar per (b,i,j)
    lgate = tl.load(lgate_ptr + lgate_off, mask=mask, other=0.0)
    rgate = tl.load(rgate_ptr + rgate_off, mask=mask, other=0.0)

    # ---------- fuse ----------
    left  = left  * m * lgate
    right = right * m * rgate

    # ---------- store back (in‑place) ----------
    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left, right, mask, left_gate, right_gate):
    """
    In‑place element‑wise fusion:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    All tensors are fp32, shape:
        left/right : (B,N,N,H)
        mask       : (B,N,N)   (will be unsqueezed to (...,1) before call)
        gates      : (B,N,N,H)
    """
    B, N, _, H = left.shape
    total = B * N * N * H

    BLOCK = 2048                     # ~2 k threads → good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    s_left  = left.stride()
    s_right = right.stride()
    s_mask  = mask.stride()
    s_lgate = left_gate.stride()
    s_rgate = right_gate.stride()

    _fuse_mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=s_left[0],  stride_left_i=s_left[1],
        stride_left_j=s_left[2],  stride_left_h=s_left[3],
        stride_right_b=s_right[0], stride_right_i=s_right[1],
        stride_right_j=s_right[2], stride_right_h=s_right[3],
        stride_mask_b=s_mask[0],   stride_mask_i=s_mask[1],
        stride_mask_j=s_mask[2],
        stride_lgate_b=s_lgate[0], stride_lgate_i=s_lgate[1],
        stride_lgate_j=s_lgate[2], stride_lgate_h=s_lgate[3],
        stride_rgate_b=s_rgate[0], stride_rgate_i=s_rgate[1],
        stride_rgate_j=s_rgate[2], stride_rgate_h=s_rgate[3],
        BLOCK=BLOCK,
    )
    return left, right


# ------------------------------------------------------------------
# Entry point required by the evaluation harness
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape (B, N, N, C), dtype=torch.float32
        - mask_tensor  : torch.Tensor, shape (B, N, N)  (may be None)
        - weights_dict : mapping of str → torch.Tensor containing all
                         LayerNorm, Linear and gate weights (all fp32)
        - config_dict  : {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor
        Tensor of shape (B, N, N, C), dtype=torch.float32, on the same device.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]           # C
    hidden = cfg["hidden_dim"] # H
    device = x.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (fp32)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]   # (C,)
    ln_b = weights["norm.bias"]     # (C,)
    # mean & var over last dim
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * ln_w + ln_b                       # (B,N,N,C)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (C → hidden) – bias‑free
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]   # (hidden, C)
    right_proj_w = weights["right_proj.weight"] # (hidden, C)

    # torch.nn.functional.linear does: out = x @ w.t()
    left = torch.nn.functional.linear(x_norm, left_proj_w)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x_norm, right_proj_w) # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 3️⃣ Gates (C → hidden) + sigmoid
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()   # (B,N,N,hidden)
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 4️⃣ Mask handling – broadcast to hidden dim
    # ------------------------------------------------------------------
    if mask is None:
        # all‑ones mask (no masking)
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2],
                         device=device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)        # (B,N,N,1)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask & gates into left/right (in‑place, Triton)
    # ------------------------------------------------------------------
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Batched GEMM (fp16) for the O(N³) contraction
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden
    # Cast to float16 – Tensor‑Core GEMM is ~10× faster
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*H, N, N)
    left_flat = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_flat = right_h.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matrix multiply: Σ_k left[...,i,k] * right[...,j,k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H,N,N)

    # back to fp32 and original layout (B,N,N,hidden)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    to_out_ln_w = weights["to_out_norm.weight"]  # (hidden,)
    to_out_ln_b = weights["to_out_norm.bias"]    # (hidden,)
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_ln_w + to_out_ln_b

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate (fp32)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to dim (hidden → dim) – bias‑free
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]          # (dim, hidden)
    out_final = torch.nn.functional.linear(out_norm, to_out_w)   # (B,N,N,dim)

    return out_final
