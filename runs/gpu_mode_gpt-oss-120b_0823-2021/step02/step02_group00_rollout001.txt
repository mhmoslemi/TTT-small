"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (FP32, manual for speed).
2️⃣ Four linear projections (left, right, three gates) are performed in FP16.
3️⃣ Optional mask is applied to the left/right tensors.
4️⃣ Gating (sigmoid) is applied, producing gated left/right tensors.
5️⃣ Core O(N³) contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is evaluated as a batch of matrix‑multiplications:
        (B, H, I, K)  ×  (B, H, K, J) → (B, H, I, J)
   where H = hidden_dim, I = J = N, K = N.
   The multiplication is done with a small Triton kernel that
   tiles the I/J dimensions (64×64) and the reduction dimension K (32).
   Operands are FP16, accumulation is FP32, result is stored as FP16.
6️⃣ The FP16 result is reshaped back to (B,N,N,hidden_dim) → FP32,
   followed by a second Layer‑norm, output‑gate multiplication and the
   final linear projection to the original `dim`.

Only the heavy contraction (step 5) lives in Triton – the rest is
implemented in PyTorch for readability and because their cost is
negligible compared with the N³ work.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel – batched GEMM   (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,            # left  (batch, I, K)  FP16
    b_ptr,            # right (batch, K, J)  FP16
    c_ptr,            # out   (batch, I, J)  FP16
    M, N, K,          # matrix extents
    stride_az, stride_am, stride_ak,   # a strides (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_z = tl.program_id(2)          # batch*head dimension

    # start offsets of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile  (M × K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile  (K × N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # accumulate: FP16×FP16 → FP32
        acc += tl.dot(a, b)

    # write back as FP16
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left   : (B, H, I, K)  FP16  contiguous
    right  : (B, H, K, J)  FP16  contiguous
    returns: (B, H, I, J) FP16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse batch*head dimension so each program works on a 2‑D matmul
    batch = B * H
    a = left.reshape(batch, I, K)
    b = right.reshape(batch, K, J)
    c = out.reshape(batch, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tile sizes tuned for Hopper fp16 tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # rows
        triton.cdiv(J, BLOCK_N),   # cols
        batch,                     # one program per (batch,head)
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # fp16 friendly on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of leaf weight tensors
        - config       : dict with keys ``dim`` and ``hidden_dim``
    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm (FP32, manual)
    # --------------------------------------------------------------
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * w_norm + b_norm                 # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gates (FP16)
    # --------------------------------------------------------------
    to_fp16 = lambda t: t.to(torch.float16)

    # projection weights: (hidden, dim)
    left_w   = to_fp16(weights["left_proj.weight"])
    right_w  = to_fp16(weights["right_proj.weight"])
    # gate weights: (hidden, dim)
    left_gate_w  = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w   = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                     # (B,N,N,dim)

    left  = F.linear(x_fp16, left_w)                # (B,N,N,hidden)
    right = F.linear(x_fp16, right_w)

    # --------------------------------------------------------------
    # 3️⃣ Mask (if any)
    # --------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)    # (B,N,N,1)
        left  = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy contraction – Triton batched GEMM
    # --------------------------------------------------------------
    # reshape to (B, hidden, I, K) and (B, hidden, K, J)
    # I = J = K = N
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _batched_gemm(left_t, right_t)         # (B, H, I, J)

    # back to (B, N, N, hidden) then FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Second Layer‑norm (FP32)
    # --------------------------------------------------------------
    w_out_norm = weights["to_out_norm.weight"]
    b_out_norm = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * w_out_norm + b_out_norm

    # --------------------------------------------------------------
    # 7️⃣ Output gate + final linear projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)       # broadcast over hidden dim
    w_out = weights["to_out.weight"]             # (dim, hidden)
    out = F.linear(out, w_out)                   # (B,N,N,dim)

    return out
