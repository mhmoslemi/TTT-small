"""
TriMul “outgoing” forward pass (no gradients).

The expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication over the “hidden” dimension `d`.
We implement it with a single Triton kernel that computes
    (B, H, I, K) @ (B, H, K, J) → (B, H, I, J)
using fp16 compute (Tensor‑cores) and fp32 accumulation.
All remaining ops (LayerNorm, linear projections, gating, final
LayerNorm+linear) stay in PyTorch – they are tiny compared with the
N³ work and do not benefit from a custom kernel.

The kernel tiles the output in blocks of (M,N) and reduces over K in
chunks of BLOCK_K.  It works for any batch size (B∈{1,2}) and any
hidden dimension (H = hidden_dim, up‑to 768 in the tests).  The
implementation is fully generic, supports an optional mask (broadcast
to the hidden dimension) and returns float32 output as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM  (B*H, I, K) @ (B*H, K, J) → (B*H, I, J)
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    a_ptr,            # left   : (Z, I, K)
    b_ptr,            # right  : (Z, K, J)
    c_ptr,            # output : (Z, I, J)
    M, N, K,          # matrix sizes
    stride_az, stride_am, stride_ak,   # strides of a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides of b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides of c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)          # batch*head dimension
    pid_m = tl.program_id(0)          # tile row  (M)
    pid_n = tl.program_id(1)          # tile col  (N)

    # ------------------------------------------------------------------
    # block offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (better numeric stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile  (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 accumulation
        acc += tl.dot(a, b)

    # write back (store as fp16, later cast to fp32 in Python)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM used by TriMul.

    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous

    Returns
    -------
    out : (B, H, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # output tensor
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten leading dimensions for the kernel
    Z = B * H
    a = left.reshape(Z, I, K)
    b = right.reshape(Z, K, J)
    c = out.reshape(Z, I, J)

    # strides in element counts (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling – works well on Hopper/SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # program_id(0)
        triton.cdiv(J, BLOCK_N),   # program_id(1)
        Z,                         # program_id(2)
    )

    batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # good default for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)

        input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        mask         : torch.Tensor [B, N, N]       (float32/bool) or None
        weights      : dict of parameter tensors
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]  (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (keep in FP32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b          # [B,N,N,dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections + gating (switch to FP16 for the heavy part)
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    # projection weights (FP16)
    w_left  = to_fp16(weights["left_proj.weight"])
    w_right = to_fp16(weights["right_proj.weight"])
    # gating weights (FP16)
    w_lgate = to_fp16(weights["left_gate.weight"])
    w_rgate = to_fp16(weights["right_gate.weight"])
    w_outgate = to_fp16(weights["out_gate.weight"])

    # final output norm/linear stay FP32
    w_to_out_norm = weights["to_out_norm.weight"]
    b_to_out_norm = weights["to_out_norm.bias"]
    w_to_out = weights["to_out.weight"]

    # cast normalized input to FP16 for the matrix ops
    x_fp16 = x_norm.to(torch.float16)

    # projections (B,N,N,hidden)
    left  = F.linear(x_fp16, w_left)   # no bias in the reference impl
    right = F.linear(x_fp16, w_right)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B,N,N] → [B,N,N,1] → fp16
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, w_lgate).sigmoid()
    right_gate = F.linear(x_fp16, w_rgate).sigmoid()
    out_gate   = F.linear(x_fp16, w_outgate).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ contraction using Triton
    # ------------------------------------------------------------------
    # reshape to (B, H, I, K) and (B, H, K, J) where
    #   I = N, J = N, K = hidden_dim, H = dim (the original channel dimension)
    # After projection we have: left/right shape = [B,N,N,hidden]
    #   we need to treat `hidden` as the new head dimension H.
    # For the outgoing version, the multiplication is:
    #   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # This is equivalent to a batched matmul over (i,k) x (j,k) for each d.
    # We therefore transpose the last two axes of `right`.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # Batched GEMM via Triton (returns fp16)
    out_fp16 = triton_batched_gemm(left_perm, right_perm)   # (B,H,I,J)

    # Back to original layout: (B,N,N,dim) in FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm
    # ------------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * w_to_out_norm + b_to_out_norm

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + linear projection (as in reference)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)   # out_gate was computed in FP16
    out = F.linear(out, w_to_out)            # final projection, no bias

    return out
