"""
custom_kernel – outgoing TriMul (AlphaFold3)

Algorithm (high‑level)
----------------------
1️⃣  Layer‑norm over the last dimension (fp32, pure PyTorch – cheap).
2️⃣  Linear projections (left/right) and linear gates (left/right/out) –
    also done in PyTorch (fp32, bias‑free).
3️⃣  Fuse mask, left‑gate and right‑gate into the projected tensors *in‑place*
    with a tiny Triton kernel (element‑wise work, fully vectorised).
4️⃣  Cast the fused tensors to float16 and reshape them to
    (B*H, N, N).  The cubic contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    becomes a batched matrix‑multiply:
       out_flat = bmm(left_flat, right_flatᵀ)
    which is executed by cuBLAS/Tensor‑Cores (very fast on Hopper).
5️⃣  Reshape back to (B,N,N,H) and perform a second Layer‑norm, the
    output‑gate, and the final linear projection (all fp32, PyTorch).

Only the mask‑gate‑fusion is written in Triton – this gives a noticeable
speed‑up while keeping the implementation simple and portable across the
various test configurations.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask and the two gates into left/right tensors.
# ----------------------------------------------------------------------
@triton.jit
def _fuse_mask_gates_kernel(
    left_ptr,          # fp32 [B, N, N, H]
    right_ptr,         # fp32 [B, N, N, H]
    mask_ptr,          # fp32 [B, N, N]   (broadcasted mask)
    lgate_ptr,         # fp32 [B, N, N, H]
    rgate_ptr,         # fp32 [B, N, N, H]
    B, N, H,           # compile‑time ints
    # strides in *elements* (not bytes)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear ids
    mask = offs < (B * N * N * H)

    # ---- decode linear offset into (b,i,j,h) -------------------------
    # total size = B * N * N * H
    b = (offs // (N * N * H))
    rest = offs % (N * N * H)
    i = (rest // (N * H))
    rest = rest % (N * H)
    j = (rest // H)
    h = rest % H

    # ---- compute element pointers ------------------------------------
    left_off  = b * stride_left_b  + i * stride_left_i  + j * stride_left_j  + h * stride_left_h
    right_off = b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h
    mask_off  = b * stride_mask_b  + i * stride_mask_i  + j * stride_mask_j
    lgate_off = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h
    rgate_off = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h

    # ---- load --------------------------------------------------------
    left  = tl.load(left_ptr  + left_off,  mask=mask, other=0.0)
    right = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    m     = tl.load(mask_ptr  + mask_off,  mask=mask, other=0.0)   # mask is scalar per (b,i,j)
    lgate = tl.load(lgate_ptr + lgate_off, mask=mask, other=0.0)
    rgate = tl.load(rgate_ptr + rgate_off, mask=mask, other=0.0)

    # ---- fuse ---------------------------------------------------------
    left  = left  * m * lgate
    right = right * m * rgate

    # ---- store back (in‑place) ---------------------------------------
    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left, right, mask, left_gate, right_gate):
    """
    In‑place element‑wise fusion:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    All tensors are fp32 and have shape [B, N, N, H] (mask is [B,N,N,1]).
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 2048                         # ~2 k threads → good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    # strides in *elements* (torch returns strides in elements)
    s_left  = left.stride()
    s_right = right.stride()
    s_mask  = mask.stride()
    s_lgate = left_gate.stride()
    s_rgate = right_gate.stride()

    _fuse_mask_gates_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=s_left[0],   stride_left_i=s_left[1],
        stride_left_j=s_left[2],   stride_left_h=s_left[3],
        stride_right_b=s_right[0], stride_right_i=s_right[1],
        stride_right_j=s_right[2], stride_right_h=s_right[3],
        stride_mask_b=s_mask[0],   stride_mask_i=s_mask[1],
        stride_mask_j=s_mask[2],
        stride_lgate_b=s_lgate[0], stride_lgate_i=s_lgate[1],
        stride_lgate_j=s_lgate[2], stride_lgate_h=s_lgate[3],
        stride_rgate_b=s_rgate[0], stride_rgate_i=s_rgate[1],
        stride_rgate_j=s_rgate[2], stride_rgate_h=s_rgate[3],
        BLOCK=BLOCK,
    )
    return left, right


# ----------------------------------------------------------------------
# Main entry point (the harness calls this)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradient required).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask         : torch.Tensor, shape [B, N, N]   (or None)
        - weights_dict : dict of torch.Tensors, containing all linear /
                         layer‑norm parameters (bias‑free linear layers)
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32, on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    device = x.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm on input (fp32, pure PyTorch)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]   # (dim,)
    ln_b = weights["norm.bias"]     # (dim,)
    # manual layer‑norm because we need the exact same epsilon as the reference
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * ln_w + ln_b               # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias, fp32)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]
    left  = torch.nn.functional.linear(x_norm, left_proj_w)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣  Gates + sigmoid (fp32)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]
    left_gate  = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 4️⃣  Prepare mask (broadcast to channel dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2],
                          device=device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)                     # (B,N,N,1)

    # ------------------------------------------------------------------
    # 5️⃣  Triton fusion: mask & gates → left/right (in‑place)
    # ------------------------------------------------------------------
    left, right = _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 6️⃣  Batched GEMM (N³ contraction) – fp16 for speed
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # cast to fp16 (Tensor‑Core friendly)
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # reshape to (B*H, N, N) so that each head is a separate batch
    left_flat  = left_f16.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_flat = right_f16.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matrix multiplication: Σ_k left[...,i,k] * right[...,j,k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))    # (B*H,N,N)

    # back to fp32 and original layout (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣  Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    to_out_ln_w = weights["to_out_norm.weight"]
    to_out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_ln_w + to_out_ln_b

    # ------------------------------------------------------------------
    # 8️⃣  Apply output gate
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9️⃣  Final linear projection back to dim (fp32)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]          # (dim, hidden)
    out_final = torch.nn.functional.linear(out_norm, to_out_w)   # (B,N,N,dim)

    return out_final
```

**Explanation of design choices**

* **Why a tiny Triton kernel?**  
  The only truly element‑wise work that cannot be fused efficiently in PyTorch
  is the combination of the pairwise mask with two separate gate tensors.
  Performing this with a custom kernel removes an extra memory traversal and
  avoids an intermediate allocation, giving a noticeable speed‑up without
  complicating the dominant cubic contraction.

* **Why batched `torch.bmm` for the contraction?**  
  The einsum `out[...,i,k,d] * right[...,j,k,d]` is mathematically a batch of
  matrix multiplications for each hidden dimension `d`.  cuBLAS/Tensor‑Cores
  on Hopper perform this operation orders of magnitude faster than any
  pure‑Triton 3‑dimensional tiling we could write, and the reshape/re‑reshape
  overhead is negligible compared with the O(N³) work.

* **Mixed‑precision handling**  
  Both projected tensors are cast to `float16` before the batched GEMM, which
  lets the H200 exploit Tensor‑Core lanes.  The result is immediately up‑cast
  to `float32` because the surrounding layer‑norms and gates expect fp32
  precision (matching the reference implementation).

* **Correctness guarantees**  
  All linear layers are bias‑free as in the reference, layer‑norm epsilons are
  identical, and the mask is broadcast correctly (`mask[...,None]`).  The
  final output dtype and shape match the PyTorch reference exactly.

* **Flexibility**  
  The kernel works for any `B ∈ {1,2}`, `N` up to 1024, `dim` ∈ {128,384},
  `hidden_dim` ∈ {128,384,768}, and also handles the case where `mask` is
  `None` (treated as all‑ones).  It therefore satisfies all test configurations
  described in the problem statement.
