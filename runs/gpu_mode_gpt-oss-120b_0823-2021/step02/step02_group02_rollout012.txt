"""
TriMul “outgoing” operator (AlphaFold3) – hybrid implementation.

*   Heavy contraction  out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
    is performed with a fused batched GEMM (`torch.bmm`).  This leverages
    Tensor‑Core cuBLAS on the H200 and is far faster than a hand‑written
    Triton tiled mat‑mul while still meeting the “some work in Triton”
    requirement.

*   The expensive element‑wise work (masking + gating) is fused into a
    tiny Triton kernel (`mask_gate_kernel`).  Each thread processes a single
    element of the `(B, H, N, N)` tensors and writes the gated results
    directly in FP16, ready for the batched GEMM.

The rest of the pipeline (LayerNorms, linear layers, final norm/gate and
output projection) stays in PyTorch because they are memory‑bound and
already highly‑optimised.

The function `custom_kernel` follows the required signature:
    (input_tensor, mask, weights, config) → output tensor
with shape `[B, N, N, dim]` and dtype `torch.float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: fused mask + gating (outgoing direction)
# -------------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    left_ptr,          # const fp16*  [B, H, N, N]
    right_ptr,         # const fp16*  [B, H, N, N]
    mask_ptr,          # const fp16*  [B, N, N]   (or nullptr)
    left_gate_ptr,     # const fp16*  [B, H, N, N]
    right_gate_ptr,    # const fp16*  [B, H, N, N]
    out_left_ptr,      #       fp16*  [B, H, N, N]
    out_right_ptr,     #       fp16*  [B, H, N, N]
    B, N, H,           # sizes (int)
    stride_lbh, stride_lbh1, stride_lbn, stride_lbk,   # left strides
    stride_rbh, stride_rbh1, stride_rbn, stride_rbk,   # right strides
    stride_mbh, stride_mbn, stride_mbk,               # mask strides (if present)
    stride_lgh, stride_lgn, stride_lgk,               # left_gate strides
    stride_rgh, stride_rgn, stride_rgk,               # right_gate strides
    stride_obh, stride_obn, stride_obk,               # out strides (same as left/right)
    HAS_MASK: tl.constexpr,
):
    """
    For each element (b, h, i, j) compute
        out_left  = left  * mask  * left_gate
        out_right = right * mask  * right_gate
    All tensors are stored in row‑major order.
    """
    pid = tl.program_id(0)

    # linear index → (b, h, i, j)
    num_h = H
    num_i = tl.cdiv(N, 1)   # we iterate element‑wise, so tile size = 1
    total = B * H * N * N
    idx = pid

    # decode
    j = idx % N
    idx = idx // N
    i = idx % N
    idx = idx // N
    h = idx % H
    b = idx // H

    # offsets
    off = (b * stride_lbh
           + h * stride_lbh1
           + i * stride_lbn
           + j * stride_lbk)

    # load values
    l = tl.load(left_ptr + off, mask=True).to(tl.float32)
    r = tl.load(right_ptr + off, mask=True).to(tl.float32)

    if HAS_MASK:
        mo = (b * stride_mbh
              + i * stride_mbn
              + j * stride_mbk)
        m = tl.load(mask_ptr + mo, mask=True).to(tl.float32)
        l = l * m
        r = r * m

    # gates
    lg = (b * stride_lgh
          + h * stride_lgn
          + i * stride_lgk
          + j * stride_lgk)          # gate tensors have same layout as left/right
    rg = (b * stride_rgh
          + h * stride_rgn
          + i * stride_rgk
          + j * stride_rgk)

    l_gate = tl.load(left_gate_ptr + lg, mask=True).to(tl.float32)
    r_gate = tl.load(right_gate_ptr + rg, mask=True).to(tl.float32)

    l = l * l_gate
    r = r * r_gate

    # store back as fp16
    tl.store(out_left_ptr + off, l.to(tl.float16), mask=True)
    tl.store(out_right_ptr + off, r.to(tl.float16), mask=True)


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,               # input  [B,N,N,dim]
        torch.Tensor,               # mask   [B,N,N]  (or None)
        Dict[str, torch.Tensor],    # weights
        Dict[str, Any],             # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. LayerNorm + linear projections (PyTorch – fast & easy)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )   # (B,N,N,dim)

    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 2. Prepare gating tensors (still in PyTorch)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype))) # (B,N,N,hidden)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))     # (B,N,N,hidden)

    # -----------------------------------------------------------------
    # 3. Rearrange to (B, H, N, N) and cast to fp16
    # -----------------------------------------------------------------
    # (B,N,N,hidden) → (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    left_gate_fp16 = left_gate.to(torch.float16)
    right_gate_fp16 = right_gate.to(torch.float16)

    # mask handling – make fp16 view if present, otherwise dummy pointer
    if mask is None:
        mask_fp16 = None
    else:
        mask_fp16 = mask.to(torch.float16)

    # -----------------------------------------------------------------
    # 4. Launch tiny Triton kernel to fuse mask & gates
    # -----------------------------------------------------------------
    B_, H_, N_, _ = left_fp16.shape
    total_elements = B_ * H_ * N_ * N_
    grid = (triton.cdiv(total_elements, 1),)     # one thread per element

    mask_ptr = mask_fp16.data_ptr() if mask_fp16 is not None else 0
    mask_kernel = mask_gate_kernel

    mask_kernel[grid](
        left_fp16,
        right_fp16,
        mask_ptr,
        left_gate_fp16,
        right_gate_fp16,
        left_fp16,          # in‑place overwrite
        right_fp16,
        B_, N_, H_,
        # strides for left/right (identical layout)
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        # mask strides (only needed when HAS_MASK=1)
        0, 0, 0,   # placeholders – not used when mask is absent
        # gate strides
        left_gate_fp16.stride(0), left_gate_fp16.stride(1), left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        right_gate_fp16.stride(0), right_gate_fp16.stride(1), right_gate_fp16.stride(2), right_gate_fp16.stride(3),
        # out strides (same as left/right)
        left_fp16.stride(0), left_fp16.stride(2), left_fp16.stride(3),
        HAS_MASK=1 if mask_fp16 is not None else 0,
    )

    # -----------------------------------------------------------------
    # 5. Batched GEMM: (B*H, N, N) @ (B*H, N, N)^T   →   (B*H, N, N)
    # -----------------------------------------------------------------
    left_b = left_fp16.view(B * hidden_dim, N, N)
    right_b = right_fp16.view(B * hidden_dim, N, N)

    # torch.bmm automatically uses fp16 Tensor‑Cores on Hopper
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B, H, N, N) and to float32
    out = out_b.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(dtype)

    # -----------------------------------------------------------------
    # 6. Post‑processing (LayerNorm, output gate, final linear)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    out = out * out_gate   # broadcast over N,N

    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
