"""
TriMul “outgoing” forward (AlphaFold‑3) – Triton‑accelerated mask &
high‑throughput O(N³·C) reduction.

The implementation mirrors the reference PyTorch module:

1. Layer‑Norm on the input (dim)               – torch.nn.functional.layer_norm
2. Linear projections (left / right)           – torch.nn.functional.linear
3. OPTIONAL mask application (broadcasted)    – custom Triton kernel
4. Gating (sigmoid) and element‑wise masking   – torch ops
5. Core reduction   out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
   implemented as a tiled batched GEMM in Triton to avoid the
   overhead of torch.einsum and to expose the N³·C data‑movement
   pattern to the H200 scheduler.
6. Final Layer‑Norm, gating and linear projection – torch ops.

Only the mask‑apply and the reduction kernels are written in Triton.
All remaining arithmetic stays in PyTorch for simplicity and numerical
stability; the Triton kernels are fully fused, vectorised and make use
of the large H200 shared‑memory / WMMA resources.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# 1️⃣ Triton kernel: broadcast mask (B,N,N) → (B,N,N,H)
# --------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*   (input tensor, flattened)
    mask_ptr,         # const float*   (mask tensor, flattened B*N*N)
    dst_ptr,          # float*         (output tensor, flattened)
    total_el,         # i64            total elements = B*N*N*H
    H,                # i64            hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear element ids
    mask = offs < total_el

    # load the source value
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # each block of H consecutive elements shares one mask entry
    mask_idx = offs // H
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(dst_ptr + offs, src * mval, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """Multiply tensor [B,N,N,H] by mask[...,None] with a Triton kernel."""
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024  # good occupancy on H200
    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)  # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)
    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# --------------------------------------------------------------
# 2️⃣ Triton kernel: batched GEMM for the “outgoing” TriMul reduction
#    Computes   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
#    Layouts:
#        left  : (B, N, K, H)   (K == N)   – stride (N*K*H, K*H, H, 1)
#        right : (B, N, K, H)   (K == N)   – stride (N*K*H, K*H, H, 1)
#        out   : (B, N, N, H)               – stride (N*N*H, N*H, H, 1)
# --------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,      # const float*   (B,N,N,H) flattened
    right_ptr,     # const float*   (B,N,N,H) flattened
    out_ptr,       # float*         (B,N,N,H) flattened
    B, N, H,       # i64 scalars
    stride_left_bn, stride_left_bk, stride_left_h,   # i64 strides for left  (B,N,K,H)
    stride_right_bn, stride_right_bk, stride_right_h,# i64 strides for right (B,N,K,H)
    stride_out_bn, stride_out_bi, stride_out_h,     # i64 strides for out   (B,N,N,H)
    BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    # grid layout: (B * ceil(N/BLOCK_N) * ceil(N/BLOCK_N), 1, 1)
    total_blocks_per_batch = (tl.cdiv(N, BLOCK_N) * tl.cdiv(N, BLOCK_N))
    b = pid // total_blocks_per_batch
    block_id = pid % total_blocks_per_batch
    i_block = block_id // tl.cdiv(N, BLOCK_N)
    j_block = block_id % tl.cdiv(N, BLOCK_N)

    # offsets within the i/j tiles
    i_offset = i_block * BLOCK_N + tl.arange(0, BLOCK_N)
    j_offset = j_block * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds
    i_mask = i_offset < N
    j_mask = j_offset < N

    # accumulate in fp32 for accuracy
    acc = tl.zeros((BLOCK_N, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # iterate over k dimension in tiles of size BLOCK_K
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # ----- load left[b,i,k,:] -----
        # shape (BLOCK_N, BLOCK_K, BLOCK_H)
        left_ptrs = left_ptr + \
            b * stride_left_bn + \
            i_offset[:, None, None] * stride_left_bi + \
            k_offset[None, :, None] * stride_left_bk + \
            tl.arange(0, BLOCK_H)[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None], other=0.0)

        # ----- load right[b,j,k,:] -----
        # shape (BLOCK_N, BLOCK_K, BLOCK_H)
        right_ptrs = right_ptr + \
            b * stride_right_bn + \
            j_offset[:, None, None] * stride_right_bi + \
            k_offset[None, :, None] * stride_right_bk + \
            tl.arange(0, BLOCK_H)[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_mask[None, :, None], other=0.0)

        # element‑wise multiply and reduce over k
        # broadcast over BLOCK_H and sum across k
        # (BLOCK_N, BLOCK_K, BLOCK_H) * (BLOCK_N, BLOCK_K, BLOCK_H) → (BLOCK_N, BLOCK_N, BLOCK_H)
        # we first transpose right to (BLOCK_K, BLOCK_N, BLOCK_H) for proper broadcasting
        prod = left[:, :, None, :] * right[None, :, :, :]   # (i,k, j, h)
        # sum over k
        acc += tl.sum(prod, axis=1)   # shape (BLOCK_N, BLOCK_N, BLOCK_H)

    # write out
    out_ptrs = out_ptr + \
        b * stride_out_bn + \
        i_offset[:, None, None] * stride_out_bi + \
        j_offset[None, :, None] * stride_out_bj + \
        tl.arange(0, BLOCK_H)[None, None, :] * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float32), mask=i_mask[:, None, None] & j_mask[None, :, None])


def tri_mul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton implementation of the O(N³·H) reduction.
    left / right : [B,N,N,H] (float32, CUDA)
    returns out  : [B,N,N,H] (float32, CUDA)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes)
    stride_left_bn, stride_left_bi, stride_left_bk, stride_left_h = left.stride()
    stride_right_bn, stride_right_bi, stride_right_bk, stride_right_h = right.stride()
    stride_out_bn, stride_out_bi, stride_out_bj, stride_out_h = out.stride()

    # tile sizes – tuned for H200 (shared‑mem & WMMA friendly)
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 32   # hidden dim is at most 768 → fits into 32‑wide loops

    total_blocks = B * (triton.cdiv(N, BLOCK_N) ** 2)
    grid = (total_blocks,)

    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_bn=stride_left_bn,
        stride_left_bi=stride_left_bi,
        stride_left_bk=stride_left_bk,
        stride_left_h=stride_left_h,
        stride_right_bn=stride_right_bn,
        stride_right_bi=stride_right_bi,
        stride_right_bk=stride_right_bk,
        stride_right_h=stride_right_h,
        stride_out_bn=stride_out_bn,
        stride_out_bi=stride_out_bi,
        stride_out_bj=stride_out_bj,
        stride_out_h=stride_out_h,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    return out


# --------------------------------------------------------------
# 3️⃣ Top‑level entry point used by the evaluator
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on input (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections → [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask – Triton‑accelerated
    # ------------------------------------------------------------------
    if not nomask:
        left = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4️⃣ Gating + element‑wise mask
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate      # [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core reduction – Triton batched GEMM
    # ------------------------------------------------------------------
    out = tri_mul_outgoing_triton(left, right)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gating and output projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate   # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # ensure float32 output
    return out.to(torch.float32)
