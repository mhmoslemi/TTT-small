"""
TriMul – outgoing version (AlphaFold‑3)

The kernel fuses three element‑wise operations that are cheap but memory‑bound:
 *   mask broadcast (or all‑ones)
 *   three sigmoid‑gates
 *   multiplication of the left/right projected tensors by the mask and
   their respective gates.

All other heavy work (layer‑norms, linear projections and the N³ contraction)
remains in PyTorch where cuBLAS/Tensor‑cores give the best throughput on
Hopper GPUs.  The overall pipeline is therefore:

0️⃣  LayerNorm on the input (PyTorch, bias‑free)
1️⃣  Linear projections → left/right (hidden dim) and three gate tensors
2️⃣  Triton kernel → in‑place: left = left * mask * sigmoid(left_gate)
                      right = right * mask * sigmoid(right_gate)
                      out_gate = sigmoid(out_gate)   (stored back)
3️⃣  Batched GEMM:   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
                     (implemented as torch.matmul on tensors of shape
                     (B, hidden, N, N))
4️⃣  Final LayerNorm → scale by out_gate → linear projection back to dim
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: mask + sigmoid(gates) → left, right, out_gate (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                     # [B,N,N,H]  float32
    mask_ptr,                                 # [B,N,N,1]  float32 (broadcasted)
    lgate_ptr, rgate_ptr, ogate_ptr,          # [B,N,N,H]  float32 (pre‑sigmoid)
    B, N, H,
    # strides (int64)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k, stride_mask_1,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    stride_ogate_b, stride_ogate_i, stride_ogate_k, stride_ogate_h,
    BLOCK: tl.constexpr,
):
    """Fuse mask, three sigmoid‑gates and the element‑wise multiplication.

    The mask is broadcasted over the hidden dimension.  The kernel updates
    ``left`` and ``right`` in‑place and stores the sigmoided ``out_gate`` back
    into ``ogate_ptr``.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear index
    total = B * N * N * H
    active = offs < total

    # ---- decode linear index (b,i,k,d) ---------------------------------
    idx = offs
    d = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # ---- compute pointers ------------------------------------------------
    left   = left_ptr   + b * stride_left_b   + i * stride_left_i   + k * stride_left_k   + d * stride_left_h
    right  = right_ptr  + b * stride_right_b  + i * stride_right_i  + k * stride_right_k  + d * stride_right_h
    mask   = mask_ptr   + b * stride_mask_b   + i * stride_mask_i   + k * stride_mask_k                 # 1‑channel
    lgate  = lgate_ptr  + b * stride_lgate_b  + i * stride_lgate_i  + k * stride_lgate_k  + d * stride_lgate_h
    rgate  = rgate_ptr  + b * stride_rgate_b  + i * stride_rgate_i  + k * stride_rgate_k  + d * stride_rgate_h
    ogate  = ogate_ptr  + b * stride_ogate_b  + i * stride_ogate_i  + k * stride_ogate_k  + d * stride_ogate_h

    # ---- load -----------------------------------------------------------
    l = tl.load(left,  mask=active, other=0.0)
    r = tl.load(right, mask=active, other=0.0)
    m = tl.load(mask,  mask=active, other=1.0)        # 1.0 when no mask
    lg = tl.load(lgate, mask=active, other=0.0)
    rg = tl.load(rgate, mask=active, other=0.0)
    og = tl.load(ogate, mask=active, other=0.0)

    # ---- sigmoid (fast, numerically stable enough for inference) -------
    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))
    og = 1.0 / (1.0 + tl.exp(-og))

    # ---- fuse -----------------------------------------------------------
    l = l * m * lg
    r = r * m * rg

    # ---- store ----------------------------------------------------------
    tl.store(left,  l, mask=active)
    tl.store(right, r, mask=active)
    tl.store(ogate, og, mask=active)   # out_gate overwritten with sigmoid
    

def _apply_mask_and_gates(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """
    Launch ``_mask_gate_fuse_kernel``.  All tensors are ``float32`` and have
    shape (B, N, N, H) except ``mask`` which is (B, N, N, 1).  The kernel updates
    ``left`` and ``right`` in‑place and returns the sigmoided ``out_gate``.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                     # good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left, right,
        mask,
        left_gate, right_gate, out_gate,
        B, N, H,
        # strides
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2), mask.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate   # already sigmoided


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, dim] (float32 / bf16)
        - mask_tensor  : torch.Tensor [B, N, N]   or ``None``
        - weights_dict : mapping with keys
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 0️⃣ Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 1️⃣ Linear projections (bias‑free) → left/right (hidden)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 2️⃣ Gates (pre‑sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"])
    right_gate = F.linear(x, weights["right_gate.weight"])
    out_gate   = F.linear(x, weights["out_gate.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Broadcast mask (or use all‑ones) to 4‑D
    # ------------------------------------------------------------------
    if mask is None:
        mask4d = torch.ones((B, N, N, 1), dtype=x.dtype, device=device)
    else:
        mask4d = mask.unsqueeze(-1).to(x.dtype)   # (B,N,N,1)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask + sigmoid(gates) into left/right (in‑place) and out_gate
    # ------------------------------------------------------------------
    out_gate = _apply_mask_and_gates(left, right, mask4d, left_gate,
                                     right_gate, out_gate)   # out_gate now sigmoided

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction: batched GEMM over the K dimension
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden, N, K) for torch.matmul
    left_t  = left.permute(0, 3, 1, 2)          # (B, hidden, N, K)
    right_t = right.permute(0, 3, 1, 2)         # (B, hidden, N, K)

    # (B, hidden, N, N) = (B, hidden, N, K) @ (B, hidden, K, N)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))

    # Back to (B, N, N, hidden)
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm → out‑gate scaling → linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                # element‑wise scaling by sigmoid(out_gate)
    out = F.linear(out, weights["to_out.weight"])

    # Output must be float32 as required by the spec
    return out.to(torch.float32)
