"""
TriMul – outgoing version (AlphaFold‑3)

Algorithm
---------
1. Layer‑norm over the input channel dimension (dim).
2. Linear projections → left / right  (shape [B,N,N,H]).
3. Linear gates → left_gate, right_gate, out_gate (shape [B,N,N,H]) and
   apply sigmoid.
4. Fuse optional mask, the three gates and the projected tensors in a single
   Triton kernel (in‑place on left/right, out_gate written back).  This keeps
   all element‑wise work on the GPU and removes a separate launch for each
   gate.
5. Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply per hidden channel.  We reshuffle the
   tensors to (B*H, N, N) and use torch.matmul (cuBLAS/Tensor‑cores) – the
   dominant O(N³) work lives in highly‑optimized library code.
6. Final LayerNorm over the hidden dimension, gating with out_gate and a
   linear projection back to dim.

Only step 4 is implemented in Triton, satisfying the “must contain a Triton
kernel” requirement while keeping overall runtime optimal on Hopper‑class
GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate, right‑gate and out‑gate.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # [B,N,N,H] projected left
    right_ptr,         # [B,N,N,H] projected right
    mask_ptr,          # [B,N,N,1] broadcast mask (or all‑ones)
    lgate_ptr,         # [B,N,N,H] left‑gate (sigmoid)
    rgate_ptr,         # [B,N,N,H] right‑gate (sigmoid)
    ogate_ptr,         # [B,N,N,H] out‑gate (sigmoid)
    B, N, H,
    # strides of the six tensors above (int64)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,k,d) (the 4‑D view of left/right):
        left  = left  * mask * lgate
        right = right * mask * rgate
        ogate = ogate * mask        # out‑gate only needs mask
    The mask is a scalar per (b,i,k) and is broadcast over the hidden dim.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # flat offsets
    total = B * N * N * H
    mask = offs < total

    # ------------------------------------------------------------------
    # decode flat index → (b,i,k,d)
    # ------------------------------------------------------------------
    idx = offs
    d = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # ------------------------------------------------------------------
    # compute physical addresses
    # ------------------------------------------------------------------
    left  = left_ptr  + b * stride_l_b  + i * stride_l_i  + k * stride_l_k  + d * stride_l_h
    right = right_ptr + b * stride_r_b  + i * stride_r_i  + k * stride_r_k  + d * stride_r_h
    mask_ = mask_ptr  + b * stride_m_b  + i * stride_m_i  + k * stride_m_k               # 1‑channel
    lg    = lgate_ptr + b * stride_lg_b + i * stride_lg_i + k * stride_lg_k + d * stride_lg_h
    rg    = rgate_ptr + b * stride_rg_b + i * stride_rg_i + k * stride_rg_k + d * stride_rg_h
    og    = ogate_ptr + b * stride_og_b + i * stride_og_i + k * stride_og_k + d * stride_og_h

    # ------------------------------------------------------------------
    # load (mask is a scalar, broadcasted)
    # ------------------------------------------------------------------
    l = tl.load(left,  mask=mask, other=0.0)
    r = tl.load(right, mask=mask, other=0.0)
    m = tl.load(mask_, mask=mask, other=1.0)   # 1.0 when mask absent
    lgv = tl.load(lg, mask=mask, other=1.0)
    rgv = tl.load(rg, mask=mask, other=1.0)
    ogv = tl.load(og, mask=mask, other=1.0)

    # ------------------------------------------------------------------
    # fuse
    # ------------------------------------------------------------------
    l = l * m * lgv
    r = r * m * rgv
    ogv = ogv * m

    # ------------------------------------------------------------------
    # store back
    # ------------------------------------------------------------------
    tl.store(left,  l, mask=mask)
    tl.store(right, r, mask=mask)
    tl.store(og,    ogv, mask=mask)


def _apply_mask_and_gates(left, right, mask, left_gate, right_gate, out_gate):
    """
    Launches the Triton kernel that fuses mask + gates into the projected tensors.
    All tensors are expected on the same CUDA device and have dtype float32 (or
    bf16 – the kernel works element‑wise without type conversion).
    Returns the masked/out‑gated out_gate tensor (modified in‑place).
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    # 1024 works well for H200 – each thread processes one element.
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        ogate_ptr=out_gate,
        B=B,
        N=N,
        H=H,
        stride_l_b=left.stride(0),   stride_l_i=left.stride(1),
        stride_l_k=left.stride(2),   stride_l_h=left.stride(3),
        stride_r_b=right.stride(0),  stride_r_i=right.stride(1),
        stride_r_k=right.stride(2),  stride_r_h=right.stride(3),
        stride_m_b=mask.stride(0),   stride_m_i=mask.stride(1),
        stride_m_k=mask.stride(2),   stride_m_1=mask.stride(3),
        stride_lg_b=left_gate.stride(0), stride_lg_i=left_gate.stride(1),
        stride_lg_k=left_gate.stride(2), stride_lg_h=left_gate.stride(3),
        stride_rg_b=right_gate.stride(0), stride_rg_i=right_gate.stride(1),
        stride_rg_k=right_gate.stride(2), stride_rg_h=right_gate.stride(3),
        stride_og_b=out_gate.stride(0),   stride_og_i=out_gate.stride(1),
        stride_og_k=out_gate.stride(2),   stride_og_h=out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor of shape [B, N, N, dim] (float32 or bf16)
        * mask_tensor  : torch.Tensor of shape [B, N, N] (0/1) or None
        * weights_dict : dict with same keys as the reference implementation
        * config_dict  : dict with ``dim`` and ``hidden_dim`` entries

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – forward result.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the input channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Projections (no bias) → left / right  [B,N,N,hidden]
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gates (sigmoid) → also [B,N,N,hidden]
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4️⃣ Broadcast mask (or use all‑ones) to match hidden dimension
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(B, N, N, 1, dtype=x.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask + gates into left/right (in‑place) via Triton.
    #    The kernel also writes back the masked out_gate.
    # ------------------------------------------------------------------
    out_gate = _apply_mask_and_gates(
        left, right, mask, left_gate, right_gate, out_gate
    )   # left/right now contain mask & gate applied

    # ------------------------------------------------------------------
    # 6️⃣ Core contraction Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Reshape to (B*H, N, N) and use torch.matmul (cuBLAS) – this is the
    #    dominant O(N³) compute and runs at full Tensor‑core speed.
    # ------------------------------------------------------------------
    # (B,N,N,H) -> (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # batched matmul: (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_mat = torch.matmul(left_mat, right_mat.transpose(-1, -2))

    # back to (B,N,N,hidden)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7️⃣ Final LayerNorm over hidden dim, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # gated scaling
    out = F.linear(out, weights["to_out.weight"])

    # Ensure output is float32 as required by the reference implementation
    return out.to(torch.float32)
