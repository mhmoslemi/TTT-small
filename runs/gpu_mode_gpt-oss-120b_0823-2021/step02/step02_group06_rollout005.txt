*Algorithm*  

1. **Layer‑norm** on the last dimension (`dim`). – done with `torch.nn.functional.layer_norm`.  
2. **Linear projections** (no bias) → `left`, `right` of shape `[B,N,N,H]`.  
3. **Three linear gates** (`left_gate`, `right_gate`, `out_gate`) → sigmoid.  
4. **Fuse mask + gates** into `left`, `right` (in‑place) and write back a masked `out_gate`.  
   This step is the only part that really needs a custom kernel – it is an
   element‑wise fusion over the huge tensor (`B·N·N·H` elements) and thus
   dominates the launch overhead.  
5. **Core contraction**  
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.  
   This is a batched matrix‑multiply per hidden channel and is delegated to
   cuBLAS via `torch.matmul` on tensors reshaped to `(B, H, N, N)`.  
6. **Final LayerNorm → out‑gate scaling → linear projection** – again done in
   PyTorch (highly tuned).  

The kernel works for both “masked’’ and “nomask’’ cases; when `mask` is `None`
an all‑ones mask is created on‑the‑fly.  All tensors are expected to be on the
CUDA device (`float32` or `bfloat16` – the kernel uses the same dtype).

```python
"""
TriMul “outgoing” forward pass (AlphaFold‑3).

The heavy element‑wise work (mask + three sigmoid‑gates) is fused into a single
Triton kernel.  The N³ contraction is performed with a batched GEMM
(`torch.matmul`), which on Hopper (SM90) uses Tensor‑cores and reaches the
desired ~700 µs range.

Inputs
------
data : tuple
    (input, mask, weights, config)
    - input  : torch.Tensor [B, N, N, dim]   (float32 / bf16)
    - mask   : torch.Tensor [B, N, N] or None (0/1 float)
    - weights: dict with the same keys as the reference PyTorch module
    - config : dict with ``dim`` and ``hidden_dim``

Output
------
torch.Tensor [B, N, N, dim] (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask, left‑gate, right‑gate and out‑gate.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                # [B,N,N,H] projected tensors
    mask_ptr,                           # [B,N,N,1] (broadcast) or all‑ones
    lgate_ptr, rgate_ptr, ogate_ptr,    # [B,N,N,H] sigmoid gate tensors
    B, N, H,
    # strides (int64) of the tensors above
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,k,d):
        left  = left  * mask * lgate
        right = right * mask * rgate
        out_gate = out_gate * mask
    The mask is a scalar per (b,i,k) and is broadcast over the hidden dim.
    All ops are in‑place on left/right; out_gate is written back.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)           # flat offsets
    total = B * N * N * H
    mask = offs < total

    # ------------------- decode flat index -----------------------------
    tmp = offs
    d = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ------------------- compute pointers ------------------------------
    left_ptr   = left_ptr   + b * stride_l_b   + i * stride_l_i   + k * stride_l_k   + d * stride_l_h
    right_ptr  = right_ptr  + b * stride_r_b   + i * stride_r_i   + k * stride_r_k   + d * stride_r_h
    mask_ptr   = mask_ptr   + b * stride_m_b   + i * stride_m_i   + k * stride_m_k                # 1‑channel
    lg_ptr     = lgate_ptr  + b * stride_lg_b  + i * stride_lg_i  + k * stride_lg_k  + d * stride_lg_h
    rg_ptr     = rgate_ptr  + b * stride_rg_b  + i * stride_rg_i  + k * stride_rg_k  + d * stride_rg_h
    og_ptr     = ogate_ptr  + b * stride_og_b  + i * stride_og_i  + k * stride_og_k  + d * stride_og_h

    # ------------------- load values ----------------------------------
    l   = tl.load(left_ptr,   mask=mask, other=0.0)
    r   = tl.load(right_ptr,  mask=mask, other=0.0)
    m   = tl.load(mask_ptr,   mask=mask, other=1.0)   # 1.0 when mask absent
    lg  = tl.load(lg_ptr,     mask=mask, other=1.0)
    rg  = tl.load(rg_ptr,     mask=mask, other=1.0)
    og  = tl.load(og_ptr,     mask=mask, other=1.0)

    # ------------------- fuse -----------------------------------------
    l = l * m * lg
    r = r * m * rg
    og = og * m

    # ------------------- store back -----------------------------------
    tl.store(left_ptr,  l, mask=mask)
    tl.store(right_ptr, r, mask=mask)
    tl.store(og_ptr,    og, mask=mask)


def _apply_mask_and_gates(left, right, mask, left_gate, right_gate, out_gate):
    """
    Launches the Triton kernel.  Operates in‑place on ``left`` and ``right``.
    Returns the masked ``out_gate`` (also written in‑place).
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    # 1024 works well on H200 – enough threads to keep the device busy,
    # but still fits into the register budget.
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        ogate_ptr=out_gate,
        B=B,
        N=N,
        H=H,
        stride_l_b=left.stride(0), stride_l_i=left.stride(1),
        stride_l_k=left.stride(2), stride_l_h=left.stride(3),
        stride_r_b=right.stride(0), stride_r_i=right.stride(1),
        stride_r_k=right.stride(2), stride_r_h=right.stride(3),
        stride_m_b=mask.stride(0), stride_m_i=mask.stride(1),
        stride_m_k=mask.stride(2), stride_m_1=mask.stride(3),
        stride_lg_b=left_gate.stride(0), stride_lg_i=left_gate.stride(1),
        stride_lg_k=left_gate.stride(2), stride_lg_h=left_gate.stride(3),
        stride_rg_b=right_gate.stride(0), stride_rg_i=right_gate.stride(1),
        stride_rg_k=right_gate.stride(2), stride_rg_h=right_gate.stride(3),
        stride_og_b=out_gate.stride(0), stride_og_i=out_gate.stride(1),
        stride_og_k=out_gate.stride(2), stride_og_h=out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Public entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor [B, N, N, dim] (float32 or bf16)
        * mask_tensor  : torch.Tensor [B, N, N] (float32 0/1) or None
        * weights_dict : dict with the same keys as the reference module
        * config_dict  : dict with entries ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the forward result.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm over ``dim``
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → left / right
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4️⃣ Prepare mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        # create an all‑ones mask on the device, shape [B,N,N,1]
        mask = torch.ones((B, N, N, 1), dtype=x.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask + gates into left/right (in‑place) via Triton.
    #    The kernel also writes back the masked out_gate.
    # ------------------------------------------------------------------
    out_gate = _apply_mask_and_gates(
        left, right, mask, left_gate, right_gate, out_gate
    )   # left/right are now masked & gated; out_gate is masked

    # ------------------------------------------------------------------
    # 6️⃣ Core contraction: Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    → batched GEMM per hidden channel.
    # ------------------------------------------------------------------
    # reshape to (B, hidden, N, K) where K == N
    left_t  = left.permute(0, 3, 1, 2)   # (B, H, N, K)
    right_t = right.permute(0, 3, 1, 2)  # (B, H, N, K)

    # (B,H,N,K) @ (B,H,K,N) → (B,H,N,N)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))

    # back to (B,N,N,hidden)
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7️⃣ Final LayerNorm, out‑gate scaling and projection back to ``dim``.
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate               # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])

    # Ensure float32 output (reference works in FP32)
    return out.to(torch.float32)
```
