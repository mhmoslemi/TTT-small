"""
TriMul “outgoing” forward (AlphaFold‑3) – hybrid PyTorch / Triton implementation.

Key ideas
---------
* All heavy linear / LayerNorm work is done with PyTorch’s highly‑optimised CUDA
  kernels (the O(N³·C) reduction is a batched GEMM that already runs at near‑
  peak on Hopper).
* The only required custom GPU work is the optional pair‑wise mask multiplication,
  which is memory‑bound and benefits from Triton’s explicit masking and vectorised
  stores.  A single kernel handles both the left and right projected tensors in
  one pass, reducing kernel launch overhead.
* The code works for any batch size (1‑2), sequence length up to 1024, and
  arbitrary `dim` / `hidden_dim` values seen in the test matrix.
* All tensors are expected on the CUDA device and the final output is `float32`.

The entry‑point `custom_kernel` follows the signature required by the evaluator.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: apply a broadcast mask of shape [B, N, N] to two tensors
# of shape [B, N, N, H] (the left‑projected and right‑projected tensors).
# Both tensors are processed in a single kernel to lower launch overhead.
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_two_kernel(
    left_ptr,            # const float*  (left projected tensor)
    right_ptr,           # const float*  (right projected tensor)
    mask_ptr,            # const float*  (mask tensor, shape B*N*N)
    left_out_ptr,        # float*        (masked left)
    right_out_ptr,       # float*        (masked right)
    total_el,            # i64          total elements = B*N*N*H
    H,                   # i64          hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    # linear index of the element we are working on
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_el

    # Load the two source values
    left_val  = tl.load(left_ptr  + offs, mask=mask, other=0.0)
    right_val = tl.load(right_ptr + offs, mask=mask, other=0.0)

    # Every H consecutive elements share the same mask entry.
    # Compute the index into the flattened mask (shape B*N*N).
    mask_idx = offs // H
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Apply mask and store
    tl.store(left_out_ptr  + offs, left_val  * mask_val, mask=mask)
    tl.store(right_out_ptr + offs, right_val * mask_val, mask=mask)


def apply_mask_triton_two(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Multiply ``left`` and ``right`` (both [B,N,N,H]) by ``mask[...,None]`` using a
    single Triton kernel.  Returns the two masked tensors.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                     # chosen for good occupancy on H200

    # flatten everything – Triton works with 1‑D pointers
    left_flat  = left.view(-1)
    right_flat = right.view(-1)
    mask_flat  = mask.view(-1)       # [B,N,N]

    left_out  = torch.empty_like(left_flat)
    right_out = torch.empty_like(right_flat)

    grid = (triton.cdiv(total, BLOCK),)

    mask_apply_two_kernel[grid](
        left_ptr   = left_flat,
        right_ptr  = right_flat,
        mask_ptr   = mask_flat,
        left_out_ptr  = left_out,
        right_out_ptr = right_out,
        total_el   = total,
        H          = H,
        BLOCK      = BLOCK,
    )
    return left_out.view_as(left), right_out.view_as(right)


# ----------------------------------------------------------------------
# Entry‑point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    # Using functional layer_norm keeps the implementation simple and fast.
    # weight and bias are provided by the model.
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )                                            # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias) → [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask – apply with a fused Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        left, right = apply_mask_triton_two(left, right, mask)

    # ------------------------------------------------------------------
    # Gating (sigmoid) and element‑wise gating of projected tensors
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate       # [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·C) reduction: batched GEMM via torch.einsum.
    # This maps to a series of highly‑optimised GEMM kernels on Hopper.
    # ------------------------------------------------------------------
    # out shape: [B, N, N, hidden_dim]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # Final LayerNorm (hidden_dim), gating and output projection back to dim
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # broadcast over hidden_dim

    # Linear projection back to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure float32 output
    return out.to(torch.float32)
