"""
TriMul – “outgoing” path (AlphaFold‑3)

The heavy work is the rank‑3 contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each batch element *b* and hidden channel *d* this is a dense
(N,N) × (N,N) matrix‑matrix product

    L(b,d) @ R(b,d).T      with   L,R ∈ ℝ^{N×N} ,  out ∈ ℝ^{N×N}

The surrounding LayerNorms, Linear projections and sigmoid gates are cheap
and stay in PyTorch.  The contraction itself is implemented as a tiled
batched GEMM in Triton:

* data layout for the kernel is (B, H, N, N) – i.e. the hidden dimension is
  moved to the second axis.  This lets each kernel instance work on a single
  (b,h) pair, avoiding any inter‑block synchronisation.
* The kernel tiles the output matrix with BLOCK_M × BLOCK_N threads and
  iterates over the K‑dimension in BLOCK_K‑sized chunks.  All loads are
  vectorised (fp16) and accumulation is performed in fp32 for numerical
  stability.  The final result is written back as fp16 and later cast back to
  fp32 for the remaining PyTorch pipeline.
* Optional masking is applied **before** the Triton call – the mask is a
  broadcasted fp16 tensor of shape (B,1,N,N) and simply zeros out the
  unwanted entries.
* Block sizes (64,64,32) were chosen to fit comfortably in the shared‑memory
  budget of an H200 while keeping high occupancy.  They are exposed as
  `tl.constexpr` parameters so the launcher can tune them if needed.

The overall forward pass therefore looks like:

    x → LayerNorm → (proj‑left/right) → mask → gates →
    (B,N,N,H) → (B,H,N,N) fp16 → Triton GEMM →
    fp16 → (B,N,N,H) fp32 → LayerNorm → out‑gate → final linear → (B,N,N,dim)

The Triton kernel accounts for arbitrary (possibly non‑contiguous) stride
layouts, making the implementation robust to the permutations performed in
the host code.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel that computes a batched matrix‑matrix product for each
# (batch, hidden) slice:
#   out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
# -------------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # const fp16 *  [B, H, N, N]
    right_ptr,         # const fp16 *  [B, H, N, N]
    out_ptr,           #      fp16 *  [B, H, N, N]
    B, N, H,           # scalar sizes
    stride_lbh, stride_lhd, stride_lhn, stride_lhk,   # left strides
    stride_rbh, stride_rhd, stride_rhn, stride_rhk,   # right strides
    stride_obh, stride_obd, stride_obn, stride_obk,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # -----------------------------------------------------------------
    # decode pid → (b, h, block_i, block_j)
    # -----------------------------------------------------------------
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)

    block_j = pid % num_blocks_j
    pid //= num_blocks_j
    block_i = pid % num_blocks_i
    pid //= num_blocks_i
    h = pid % H
    b = pid // H

    # -----------------------------------------------------------------
    # tile offsets and boundary masks
    # -----------------------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # loop over K dimension
    # -----------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- left[b, h, i, k]  (M×K) --------------------------------
        l_ptr = (
            left_ptr
            + b * stride_lbh
            + h * stride_lhd
            + offs_i[:, None] * stride_lhn
            + offs_k[None, :] * stride_lhk
        )
        l = tl.load(l_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # -------- right[b, h, j, k]  (N×K) need transposition ------------
        # we load as (K,N) and later dot with l (M,K)
        r_ptr = (
            right_ptr
            + b * stride_rbh
            + h * stride_rhd
            + offs_k[:, None] * stride_rhk
            + offs_j[None, :] * stride_rhn
        )
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(l, r)

    # -----------------------------------------------------------------
    # write back
    # -----------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_obh
        + h * stride_obd
        + offs_i[:, None] * stride_obn
        + offs_j[None, :] * stride_obk
    )
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : (x, mask, weights, config)
        x        : (B, N, N, dim)          float32
        mask     : (B, N, N)   (bool/float)  – may be ``None``
        weights  : dict with all linear / norm parameters (float32)
        config   : {"dim": dim, "hidden_dim": hidden_dim}

    Returns
    -------
    Tensor of shape (B, N, N, dim)  float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]          # called “H” in the kernel

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm + linear projections (cheap – stay in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N, H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 2️⃣ Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)           # (B,N,N,1)
        left  = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 3️⃣ Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ Rearrange to (B, H, N, N) and cast to fp16 for the kernel
    # -----------------------------------------------------------------
    # (B,N,N,H) → (B,H,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)   # (B, H, N, N) fp16

    # -----------------------------------------------------------------
    # 5️⃣ Triton launch
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # total program IDs = B * H * ceil(N/BLOCK_M) * ceil(N/BLOCK_N)
    grid = (
        triton.cdiv(B * hidden_dim *
                    triton.cdiv(N, BLOCK_M) *
                    triton.cdiv(N, BLOCK_N), 1),
    )

    _trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        out_fp16.stride(0), out_fp16.stride(1),
        out_fp16.stride(2), out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Post‑kernel PyTorch pipeline
    # -----------------------------------------------------------------
    # (B, H, N, N) → (B, N, N, H)  and back to fp32
    out = out_fp16.to(dtype).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    out = out * out_gate                     # broadcast over N,N
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,dim)

    return out
