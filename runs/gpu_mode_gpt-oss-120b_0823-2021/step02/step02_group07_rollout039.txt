```python
"""
custom_kernel – outgoing TriMul (AlphaFold3) forward pass

Algorithm
---------
1.  Layer‑norm over the last dimension (FP32, pure‑torch – cheap).
2.  Linear projections (x → left/right) and the three gate‑projections
    (left_gate, right_gate, out_gate).  All are bias‑free linear layers.
3.  Fuse mask, left‑gate and right‑gate into the projected tensors with a
    single Triton kernel (in‑place, FP32).  This removes three separate
    element‑wise passes.
4.  Cast the fused tensors to FP16 and reshape to (B*H, N, N) so that the
    cubic contraction becomes a batched matrix‑multiply:
        out_h[b*h, i, j] = Σ_k left_h[b*h, i, k] * right_h[b*h, j, k]
    The heavy O(N³) work is delegated to cuBLAS/Tensor‑Core via
    torch.bmm, which is optimal on Hopper (H200).
5.  Cast the result back to FP32 and restore the original layout.
6.  Output‑layer‑norm (FP32) + output‑gate (already computed) + final linear.
7.  Return the tensor of shape [B, N, N, dim] (float32).

Only step‑3 is implemented in Triton – the rest stays in PyTorch because
they are bandwidth‑bound and much cheaper than the N³ contraction.
The kernel is deliberately simple: it loads a flat index, decodes the
(b,i,j,h) coordinates, applies the mask and the two sigmoid gates and
stores the fused values back in‑place.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel – fuse mask and the two gates into left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def _fuse_mask_gates_kernel(
    left_ptr,          # fp32 * [B,N,N,H]
    right_ptr,         # fp32 * [B,N,N,H]
    mask_ptr,          # fp32 * [B,N,N,1]   (broadcasted mask)
    left_gate_ptr,     # fp32 * [B,N,N,H]
    right_gate_ptr,    # fp32 * [B,N,N,H]
    B, N, H,           # compile‑time constants
    # stride in *elements* (not bytes)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear element offset
    mask = offs < (B * N * N * H)                     # bounds guard

    # ------- decode linear offset (b,i,j,h) ---------------------------
    b = offs // (N * N * H)
    rest = offs % (N * N * H)
    i = rest // (N * H)
    rest = rest % (N * H)
    j = rest // H
    h = rest % H

    # ------- pointer arithmetic ---------------------------------------
    left_off   = b * stride_left_b   + i * stride_left_i   + j * stride_left_j   + h * stride_left_h
    right_off  = b * stride_right_b  + i * stride_right_i  + j * stride_right_j  + h * stride_right_h
    mask_off   = b * stride_mask_b   + i * stride_mask_i   + j * stride_mask_j
    lgate_off  = b * stride_lgate_b  + i * stride_lgate_i  + j * stride_lgate_j  + h * stride_lgate_h
    rgate_off  = b * stride_rgate_b  + i * stride_rgate_i  + j * stride_rgate_j  + h * stride_rgate_h

    # ------- load ----------------------------------------------------
    left   = tl.load(left_ptr   + left_off,   mask=mask, other=0.0)
    right  = tl.load(right_ptr  + right_off,  mask=mask, other=0.0)
    m      = tl.load(mask_ptr   + mask_off,   mask=mask, other=0.0)   # broadcasted mask scalar
    lgate  = tl.load(left_gate_ptr + lgate_off, mask=mask, other=0.0)
    rgate  = tl.load(right_gate_ptr+ rgate_off, mask=mask, other=0.0)

    # ------- fuse -----------------------------------------------------
    left  = left  * m * lgate
    right = right * m * rgate

    # ------- store back (in‑place) ------------------------------------
    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left: torch.Tensor,
                         right: torch.Tensor,
                         mask: torch.Tensor,
                         left_gate: torch.Tensor,
                         right_gate: torch.Tensor) -> None:
    """
    In‑place fusion of mask, left_gate and right_gate into ``left`` and ``right``.
    All tensors are FP32 and have shape [B, N, N, H] (mask: [B,N,N,1]).
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 2048                                 # ≈ 2 k threads → good occupancy
    grid = (triton.cdiv(total, BLOCK),)

    # element‑stride (in *elements*)
    s_l = left.stride()
    s_r = right.stride()
    s_m = mask.stride()
    s_lg = left_gate.stride()
    s_rg = right_gate.stride()

    _fuse_mask_gates_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=s_l[0], stride_left_i=s_l[1],
        stride_left_j=s_l[2], stride_left_h=s_l[3],
        stride_right_b=s_r[0], stride_right_i=s_r[1],
        stride_right_j=s_r[2], stride_right_h=s_r[3],
        stride_mask_b=s_m[0], stride_mask_i=s_m[1],
        stride_mask_j=s_m[2],
        stride_lgate_b=s_lg[0], stride_lgate_i=s_lg[1],
        stride_lgate_j=s_lg[2], stride_lgate_h=s_lg[3],
        stride_rgate_b=s_rg[0], stride_rgate_i=s_rg[1],
        stride_rgate_j=s_rg[2], stride_rgate_h=s_rg[3],
        BLOCK=BLOCK,
    )
    # No return – left/right are modified in‑place


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N]   (may be None)
        - weights_dict : mapping of all linear / ln parameters
        - config_dict  : {"dim": int, "hidden_dim": int}
    Returns
    -------
    torch.Tensor : shape [B, N, N, dim], dtype float32, on the same CUDA device.
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]          # = H in the paper

    # -------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (fp32)
    # -------------------------------------------------
    eps = 1e-5
    ln_w = weights["norm.weight"]   # (dim,)
    ln_b = weights["norm.bias"]     # (dim,)
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * ln_w + ln_b                     # (B,N,N,dim)

    # -------------------------------------------------
    # 2️⃣ Linear projections (no bias) – FP32
    # -------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3️⃣ Gates + sigmoid – FP32
    # -------------------------------------------------
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # -------------------------------------------------
    # 4️⃣ Prepare mask (broadcast to channel dim)
    # -------------------------------------------------
    if mask is None:
        mask = torch.ones(B, N, N, device=device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)                     # (B,N,N,1)

    # -------------------------------------------------
    # 5️⃣ Fuse mask & gates into left/right with Triton (in‑place)
    # -------------------------------------------------
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # -------------------------------------------------
    # 6️⃣ Heavy cubic contraction via batched GEMM (FP16)
    # -------------------------------------------------
    # Cast to fp16 for Tensor‑Core acceleration
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*H, N, N)   –  H == hidden
    left_flat = left_h.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H,N,N)
    right_flat = right_h.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # batched matmul: Σ_k left[...,i,k] * right[...,j,k]
    # right is transposed on the last two axes
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))      # (B*H,N,N)

    # back to FP32 and original layout [B,N,N,hidden]
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # -------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32)
    # -------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    eps_o = 1e-5
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps_o)
    out_norm = out_norm * out_ln_w + out_ln_b

    # -------------------------------------------------
    # 8️⃣ Apply output gate (already sigmoid‑ed)
    # -------------------------------------------------
    out_norm = out_norm * out_gate

    # -------------------------------------------------
    # 9️⃣ Final linear projection back to dim (FP32)
    # -------------------------------------------------
    final = F.linear(out_norm, weights["to_out.weight"])   # (B,N,N,dim)

    return final
```

**Key points**

* Only the element‑wise “mask × gate” fusion is done in Triton – this is the
  part that benefits from kernel‑level parallelism and eliminates three separate
  passes over the massive `[B,N,N,H]` tensor.
* The dominant cubic operation is performed with a batched `torch.bmm`
  on FP16 tensors, letting cuBLAS/Tensor‑Cores on the H200 handle the
  O(N³) workload efficiently.
* All LayerNorms and linear layers remain in PyTorch (FP32) – they are
  bandwidth‑bound and far cheaper than the N³ contraction.
* The implementation respects the required signature, supports the
  “no‑mask” case, works for any `dim` / `hidden_dim` allowed by the test
  matrix, and always returns a float‑32 tensor on the original device.
