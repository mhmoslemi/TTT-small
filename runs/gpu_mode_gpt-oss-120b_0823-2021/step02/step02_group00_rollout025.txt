"""
TriMul “outgoing” operator – forward pass (no gradients).

Algorithm
---------
1️⃣ Layer‑norm the input (FP32, manual for speed).
2️⃣ Linear projections + gating (cast to FP16).
3️⃣ Optional mask is applied to the projected tensors.
4️⃣ The core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix‑multiply (N×N)·(N×N) → (N×N) for each
   (batch, hidden‑dim) pair.  It is performed by a tiny Triton kernel
   that tiles the I/J dimensions (BLOCK_M×BLOCK_N) and reduces over K
   (BLOCK_K).  Operands are FP16, accumulation is FP32, result stored as FP16.
5️⃣ Convert back to FP32, second LayerNorm, output‑gate, and final linear
   projection to the original dimensionality.

The kernel is deliberately simple and tuned for Hopper (SM‑90):
    BLOCK_M = BLOCK_N = 64
    BLOCK_K = 32
    num_warps = 4   (good FP16 performance on H200)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched mat‑mul (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_matmul_kernel(
    a_ptr,            # left  matrix (batch, I, K)
    b_ptr,            # right matrix (batch, K, J)
    c_ptr,            # output matrix (batch, I, J)
    M, N, K,          # extents
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ----- program IDs -------------------------------------------------
    pid_m = tl.program_id(0)          # tile along M
    pid_n = tl.program_id(1)          # tile along N
    pid_z = tl.program_id(2)          # batch * hidden_dim index

    # ----- offsets ------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- reduction over K --------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M×K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile (K×N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16×FP16 → FP32 accumulation
        acc += tl.dot(a, b)

    # ----- write result -------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _triton_batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix‑multiply used in TriMul.

    left  : (B, H, I, K)   FP16, contiguous
    right : (B, H, K, J)   FP16, contiguous
    returns: (B, H, I, J)  FP16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse batch*head for a 3‑D kernel launch
    batch = B * H
    a = left.view(batch, I, K)
    b = right.view(batch, K, J)
    c = out.view(batch, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),          # tiles over I
        triton.cdiv(J, BLOCK_N),          # tiles over J
        batch,                            # one program per (B,H)
    )
    _batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # FP16‑friendly on Hopper/H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask         : torch.Tensor or None, shape [B, N, N] (float or bool)
        - weights      : dict of leaf tensors (see reference implementation)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual)
    # --------------------------------------------------------------
    eps = 1e-5
    norm_w = weights["norm.weight"]          # (dim,)
    norm_b = weights["norm.bias"]            # (dim,)
    # compute mean / var over the last dimension
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                  # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Project & gate (cast to FP16 for the heavy ops)
    # --------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16)

    # linear weights are stored as (out_features, in_features) in PyTorch
    left_proj_w   = to_fp16(weights["left_proj.weight"])   # (hidden_dim, dim)
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)       # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        # make mask broadcastable over hidden_dim
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction – Triton batched GEMM
    # --------------------------------------------------------------
    # reshape to (B, H, I, K) and (B, H, K, J)
    # here H == hidden_dim, I == N, K == N, J == N
    left_tiles  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_tiles = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _triton_batched_matmul(left_tiles, right_tiles)   # (B, H, I, J)

    # back to (B, N, N, dim)
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)        # broadcast over dim
    to_out_w = weights["to_out.weight"]           # (dim, dim)
    out = F.linear(out, to_out_w)                # (B,N,N,dim)

    return out
