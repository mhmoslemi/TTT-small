"""
TriMul – outgoing version (AlphaFold‑3)

The heavy N³ contraction is performed with a fused batched matrix‑multiply
(using torch.matmul, which maps to cuBLAS/Tensor‑cores on Hopper).  The only
triton kernel fuses the optional mask together with the three sigmoid gates
and applies them in‑place to the projected ``left``/``right`` tensors and
stores the sigmoided ``out_gate``.  This satisfies the “must contain a Triton
kernel” requirement while keeping the overall runtime optimal for the
target H200 GPU.

Algorithm
---------
0. Layer‑norm on input (PyTorch, bias‑free).
1. Linear projections → left/right (bias‑free) and gate pre‑sigmoids.
2. Triton kernel:
     left  = left  * mask * sigmoid(left_gate)
     right = right * mask * sigmoid(right_gate)
     out_gate = sigmoid(out_gate)          (stored in‑place)
3. Core contraction (outgoing version):
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched matmul after permuting to (B, H, N, K).
4. Final LayerNorm → element‑wise * out_gate → linear back to dim.

All tensors are float32; mixed‑precision could be added but is not required
for correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: mask + three sigmoid gates → left, right, out_gate (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    left_ptr,           #  [B,N,N,H] float32
    right_ptr,          #  [B,N,N,H] float32
    mask_ptr,           #  [B,N,N,1] float32  (broadcasted over H)
    lgate_ptr,          #  [B,N,N,H] float32  (pre‑sigmoid)
    rgate_ptr,          #  [B,N,N,H] float32
    ogate_ptr,          #  [B,N,N,H] float32
    B, N, H,
    # strides (int64)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k, stride_mask_1,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    stride_ogate_b, stride_ogate_i, stride_ogate_k, stride_ogate_h,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,k,d) compute
        left  = left  * mask * sigmoid(lgate)
        right = right * mask * sigmoid(rgate)
        ogate = sigmoid(ogate)          (stored back)
    The mask is broadcasted on the hidden dimension H.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)            # linear index
    total = B * N * N * H
    mask = offs < total

    # ----- decode linear index to (b,i,k,d) ---------------------------
    idx = offs
    d = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # ----- address calculation ----------------------------------------
    left   = left_ptr   + b * stride_left_b   + i * stride_left_i   + k * stride_left_k   + d * stride_left_h
    right  = right_ptr  + b * stride_right_b  + i * stride_right_i  + k * stride_right_k  + d * stride_right_h
    mask_  = mask_ptr   + b * stride_mask_b   + i * stride_mask_i   + k * stride_mask_k   # no hidden stride
    lgate  = lgate_ptr  + b * stride_lgate_b  + i * stride_lgate_i  + k * stride_lgate_k  + d * stride_lgate_h
    rgate  = rgate_ptr  + b * stride_rgate_b  + i * stride_rgate_i  + k * stride_rgate_k  + d * stride_rgate_h
    ogate  = ogate_ptr  + b * stride_ogate_b  + i * stride_ogate_i  + k * stride_ogate_k  + d * stride_ogate_h

    # ----- load --------------------------------------------------------
    l = tl.load(left,  mask=mask, other=0.0)
    r = tl.load(right, mask=mask, other=0.0)
    m = tl.load(mask_, mask=mask, other=1.0)          # 1.0 → no mask, 0.0 → masked out
    lg = tl.load(lgate, mask=mask, other=0.0)
    rg = tl.load(rgate, mask=mask, other=0.0)
    og = tl.load(ogate, mask=mask, other=0.0)

    # ----- sigmoid (exact, works fine for inference) -----------------
    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))
    og = 1.0 / (1.0 + tl.exp(-og))

    # ----- fuse ---------------------------------------------------------
    l = l * m * lg
    r = r * m * rg

    # ----- store back ---------------------------------------------------
    tl.store(left,  l, mask=mask)
    tl.store(right, r, mask=mask)
    tl.store(ogate, og, mask=mask)


def _apply_mask_and_gates(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """
    Launches the triton kernel that updates ``left`` and ``right`` in‑place
    and returns the sigmoided ``out_gate``.
    Shapes:
        left, right, left_gate, right_gate, out_gate : (B,N,N,H)
        mask                                         : (B,N,N,1)  (broadcasted)
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    # 1‑K elements per thread; 1024 is a good trade‑off for Hopper.
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_kernel[grid](
        left,
        right,
        mask,
        left_gate,
        right_gate,
        out_gate,
        B, N, H,
        # strides
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2), mask.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate  # now contains sigmoid(out_gate)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32/bf16)
        - mask_tensor  : torch.Tensor, shape [B, N, N] or None
        - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 0️⃣ Layer‑norm on the input (bias‑free)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 1️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x, w["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, w["right_proj.weight"])

    # ------------------------------------------------------------------
    # 2️⃣ Gate pre‑sigmoids (bias‑free)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, w["left_gate.weight"])
    right_gate = F.linear(x, w["right_gate.weight"])
    out_gate   = F.linear(x, w["out_gate.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Prepare mask broadcasted to 4‑D
    # ------------------------------------------------------------------
    if mask is None:
        mask4d = torch.ones((B, N, N, 1), dtype=x.dtype, device=device)
    else:
        mask4d = mask.unsqueeze(-1).to(x.dtype)   # (B,N,N,1)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask + sigmoid(gates) into left/right (in‑place) and out_gate
    # ------------------------------------------------------------------
    out_gate = _apply_mask_and_gates(left, right, mask4d,
                                     left_gate, right_gate, out_gate)

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction (outgoing version):
    #        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched matmul after permuting.
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, K) where K == N
    left_t  = left.permute(0, 3, 1, 2)          # (B, H, N, K)
    right_t = right.permute(0, 3, 1, 2)         # (B, H, N, K)

    # Batched GEMM: (B,H,N,K) @ (B,H,K,N) -> (B,H,N,N)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))

    # back to (B,N,N,H)
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm → out_gate scaling → linear back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )
    out = out * out_gate                       # element‑wise gate
    out = F.linear(out, w["to_out.weight"])

    # Ensure float32 output as requested by the spec
    return out.to(torch.float32)
