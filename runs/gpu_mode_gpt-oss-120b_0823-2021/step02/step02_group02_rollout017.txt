Below is a self‑contained implementation of the **outgoing** TriMul operator that follows the reference PyTorch code but moves the expensive `einsum` contraction to a **triton‑accelerated batched GEMM**.  

Key ideas  

| Step | How it is done |
|------|----------------|
| **1️⃣ LayerNorm + linear / gating** | Done in pure PyTorch (fast, uses cuBLAS). |
| **2️⃣ Masking** | Simple element‑wise multiplication, still in PyTorch. |
| **3️⃣ Rearrange** | From `[B,N,N,H]` → `[B·H,N,N]` (contiguous) – required layout for the kernel. |
| **4️⃣ Triton kernel** | Computes `out = left @ rightᵀ` for each `(b·h)` slice. The kernel tiles the `M×N` output with `BLOCK_M×BLOCK_N` blocks and accumulates over `K` in `BLOCK_K`‑wide chunks. FP16 arithmetic with FP32 accumulation gives Tensor‑Core‑friendly precision. |
| **5️⃣ Post‑processing** | Back‑to original layout, `LayerNorm`, output‑gate, final linear – all in PyTorch. |

The kernel is deliberately **minimal** (no shared memory, no `tl.dot` overloads) and works for any `B ∈ {1,2}`, `N ∈ {128,…,1024}`, `H = hidden_dim` (typically 128) and any `dim` (128 or 384).  
All strides are passed explicitly, so the kernel works for tensors with arbitrary memory layout.

```python
"""
Outgoing TriMul (AlphaFold3) – fast Triton implementation.

The heavy contraction
    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
is evaluated as a batched matrix‑multiply
    out_slice = left_slice @ right_slice.T
for each (b, h) pair.  The kernel tiles the output with
BLOCK_M × BLOCK_N blocks and accumulates over the K‑dimension
in BLOCK_K steps, using fp16 inputs and fp32 accumulation.
All surrounding LayerNorm / Linear / gating operations stay in
regular PyTorch (cuBLAS), which is already highly optimized.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM for a single (b, h) slice
# -------------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr,          # const float16*  [B*H, N, N]
    right_ptr,         # const float16*  [B*H, N, N]
    out_ptr,           #       float16*  [B*H, N, N]
    B_H, N,            # sizes
    stride_l_batch, stride_l_m, stride_l_k,   # left strides  (batch, row, col)
    stride_r_batch, stride_r_m, stride_r_k,   # right strides (batch, row, col)
    stride_o_batch, stride_o_m, stride_o_k,   # out strides   (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a (BLOCK_M × BLOCK_N) tile of out = left @ rightᵀ
    for a single batch‑hidden index (b·h).  The kernel is launched with a
    1‑D grid where each program id corresponds to a distinct (batch, tile_i,
    tile_j) triple.
    """
    pid = tl.program_id(0)

    # decode linear pid → (batch_idx, tile_i, tile_j)
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    tile_j = pid % num_tiles_n
    pid //= num_tiles_n
    tile_i = pid % num_tiles_m
    batch_idx = pid // num_tiles_m

    # start offsets for this tile
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[batch_idx, i, k]   → (M, K)
        l_ptr = (left_ptr
                 + batch_idx * stride_l_batch
                 + offs_i[:, None] * stride_l_m
                 + offs_k[None, :] * stride_l_k)
        l = tl.load(l_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[batch_idx, j, k]   → we need its transpose (k, N)
        r_ptr = (right_ptr
                 + batch_idx * stride_r_batch
                 + offs_k[:, None] * stride_r_k   # k dimension first
                 + offs_j[None, :] * stride_r_m)  # j dimension second (original col)
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # accumulate: (M,K) @ (K,N) → (M,N)
        acc += tl.dot(l, r)

    # write result back as fp16
    out_ptrs = (out_ptr
                + batch_idx * stride_o_batch
                + offs_i[:, None] * stride_o_m
                + offs_j[None, :] * stride_o_k)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Input:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # H

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm + linear projections (PyTorch, fp32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # Linear layers (bias=False)
    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # --------------------------------------------------------------
    # 2️⃣ Mask (optional)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 3️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4️⃣ Rearrange to [B*H, N, N] and cast to fp16 for the kernel
    # --------------------------------------------------------------
    # (B,N,N,H) → (B,H,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    # merge batch & hidden dimensions
    left = left.view(B * hidden_dim, N, N).to(torch.float16)
    right = right.view(B * hidden_dim, N, N).to(torch.float16)

    out_fp16 = torch.empty_like(left)   # shape (B*H, N, N), fp16

    # --------------------------------------------------------------
    # 5️⃣ Triton kernel launch
    # --------------------------------------------------------------
    # Tunable tile sizes – chosen to preserve enough shared memory on H200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # grid = number of (batch*hidden) * tiles_i * tiles_j
    num_tiles_m = triton.cdiv(N, BLOCK_M)
    num_tiles_n = triton.cdiv(N, BLOCK_N)
    total_programs = (B * hidden_dim) * num_tiles_m * num_tiles_n
    grid = (total_programs,)

    trimul_gemm_kernel[grid](
        left,
        right,
        out_fp16,
        B * hidden_dim,
        N,
        # strides of left
        left.stride(0),   # batch*hidden stride
        left.stride(1),   # row stride (i)
        left.stride(2),   # col stride (k)
        # strides of right
        right.stride(0),
        right.stride(1),
        right.stride(2),
        # strides of output
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6️⃣ Reshape back to (B,N,N,H) and continue in PyTorch
    # --------------------------------------------------------------
    out = out_fp16.to(dtype).view(B, hidden_dim, N, N)      # (B,H,N,N)
    out = out.permute(0, 2, 3, 1).contiguous()              # (B,N,N,H)

    # to_out_norm (LayerNorm over hidden dim)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # output gate (broadcast over N,N)
    out = out * out_gate

    # final linear back to dim
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
```

### Why this satisfies the requirements

* **No gradients** – The code never calls `torch.autograd.grad` or defines backward passes, matching the “forward‑only” requirement.
