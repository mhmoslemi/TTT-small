"""
custom_kernel – Triton‑accelerated “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1.  Layer‑norm over the input feature dimension (dim) – done with torch.nn.functional.
2.  Linear projections (no bias) → left, right tensors of shape [B,N,N,H].
3.  Three gate projections → left_gate, right_gate, out_gate (sigmoid).
4.  Fuse optional mask, the three sigmoid‑gates and the two projected tensors
    in a single Triton kernel:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
        out_gate = out_gate * mask                (no hidden‑dim multiplication)
    The kernel works in‑place on `left` and `right` and writes the masked
    `out_gate` back.
5.  Core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    This is a batched matrix‑multiply per hidden channel.  We reshape the
    tensors to (B*H, N, N) and call torch.bmm – cuBLAS/Tensor‑cores handle the
    O(N³) work efficiently on Hopper.
6.  Final LayerNorm over the hidden dimension, scale by the masked out_gate
    and linear projection back to `dim`.

Only step‑4 requires a custom Triton kernel; the remaining work stays in
high‑performance PyTorch.  The kernel is deliberately simple, uses a 1‑D grid
and a flat‑index decode, and is launched with a large block size (1024) that
matches the memory‑bandwidth characteristics of the H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask + three gates into left/right (in‑place) &
#               write back masked out_gate.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,          # [B,N,N,H] projected tensors
    mask_ptr,                     # [B,N,N,1] broadcast mask (or all‑ones)
    lgate_ptr, rgate_ptr, ogate_ptr,   # [B,N,N,H] sigmoid gates
    B, N, H,                      # static sizes
    # strides (int64) of the tensors above
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """
    For each element (b, i, k, d) in left/right:
        left   = left   * mask * left_gate
        right  = right  * mask * right_gate
        out_gate = out_gate * mask          (no d‑wise multiplication)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # flat offsets
    total = B * N * N * H
    mask = offs < total

    # ------- decode flat index into (b,i,k,d) -------------------------
    tmp = offs
    d = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ------- compute element pointers ---------------------------------
    left_ptr   = left_ptr   + b * stride_l_b   + i * stride_l_i   + k * stride_l_k   + d * stride_l_h
    right_ptr  = right_ptr  + b * stride_r_b   + i * stride_r_i   + k * stride_r_k   + d * stride_r_h
    mask_ptr   = mask_ptr   + b * stride_m_b   + i * stride_m_i   + k * stride_m_k                # 1‑channel
    lg_ptr     = lgate_ptr  + b * stride_lg_b  + i * stride_lg_i  + k * stride_lg_k  + d * stride_lg_h
    rg_ptr     = rgate_ptr  + b * stride_rg_b  + i * stride_rg_i  + k * stride_rg_k  + d * stride_rg_h
    og_ptr     = ogate_ptr  + b * stride_og_b  + i * stride_og_i  + k * stride_og_k  + d * stride_og_h

    # ------- load ----------------------------------------------------
    l  = tl.load(left_ptr,   mask=mask, other=0.0)
    r  = tl.load(right_ptr,  mask=mask, other=0.0)
    m  = tl.load(mask_ptr,   mask=mask, other=1.0)   # 1.0 -> no mask
    lg = tl.load(lg_ptr,     mask=mask, other=1.0)
    rg = tl.load(rg_ptr,     mask=mask, other=1.0)
    og = tl.load(og_ptr,     mask=mask, other=1.0)

    # ------- fuse ----------------------------------------------------
    l  = l * m * lg
    r  = r * m * rg
    og = og * m

    # ------- store ---------------------------------------------------
    tl.store(left_ptr,   l, mask=mask)
    tl.store(right_ptr,  r, mask=mask)
    tl.store(og_ptr,     og, mask=mask)


def _apply_mask_and_gates(left, right, mask, left_gate, right_gate, out_gate):
    """
    Launches the Triton kernel that fuses mask + gates.
    Operates in‑place on `left` and `right`; returns the masked `out_gate`.
    All tensors must be contiguous and on the same device.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                       # fits well on H200
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        ogate_ptr=out_gate,
        B=B,
        N=N,
        H=H,
        stride_l_b=left.stride(0), stride_l_i=left.stride(1),
        stride_l_k=left.stride(2), stride_l_h=left.stride(3),
        stride_r_b=right.stride(0), stride_r_i=right.stride(1),
        stride_r_k=right.stride(2), stride_r_h=right.stride(3),
        stride_m_b=mask.stride(0), stride_m_i=mask.stride(1),
        stride_m_k=mask.stride(2), stride_m_1=mask.stride(3),
        stride_lg_b=left_gate.stride(0), stride_lg_i=left_gate.stride(1),
        stride_lg_k=left_gate.stride(2), stride_lg_h=left_gate.stride(3),
        stride_rg_b=right_gate.stride(0), stride_rg_i=right_gate.stride(1),
        stride_rg_k=right_gate.stride(2), stride_rg_h=right_gate.stride(3),
        stride_og_b=out_gate.stride(0), stride_og_i=out_gate.stride(1),
        stride_og_k=out_gate.stride(2), stride_og_h=out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor [B, N, N, dim]  (float32 / bf16)
        * mask_tensor  : torch.Tensor [B, N, N] or None (0/1 float)
        * weights_dict : dict with keys matching the reference module
            (norm.weight, norm.bias, left_proj.weight, right_proj.weight,
             left_gate.weight, right_gate.weight, out_gate.weight,
             to_out_norm.weight, to_out_norm.bias, to_out.weight)
        * config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – forward output.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Input layer‑norm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → left / right  [B,N,N,hidden]
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gate projections + sigmoid  → [B,N,N,hidden]
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4️⃣ Prepare mask (broadcast to hidden dim).  If mask is None we use
    #    a tensor of ones; otherwise we unsqueeze to [B,N,N,1].
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(B, N, N, 1, dtype=x.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(x.dtype)   # now [B,N,N,1]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask & gates with a Triton kernel (in‑place on left/right)
    # ------------------------------------------------------------------
    out_gate = _apply_mask_and_gates(
        left, right, mask, left_gate, right_gate, out_gate
    )   # left/right are masked+gated, out_gate is masked

    # ------------------------------------------------------------------
    # 6️⃣ Core contraction: Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Reshape to (B*H, N, N) and use torch.bmm (cuBLAS) – the dominant cost.
    # ------------------------------------------------------------------
    # shape -> (B, N, N, H)  ->  (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # batched matmul: (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7️⃣ Final LayerNorm over hidden dim, out‑gate scaling, and projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # Ensure float32 output (reference uses float32)
    return out.to(torch.float32)
