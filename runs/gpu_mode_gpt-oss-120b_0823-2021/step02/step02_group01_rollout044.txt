"""
TriMul “outgoing” forward pass (no gradients).

The only O(N³) work – the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
– is performed with a fused batched GEMM written in Triton.
All other lightweight ops (LayerNorm, Linear, gating, final LayerNorm/Linear)
stay in PyTorch so we keep the implementation simple and numerically stable.

Input/Output layout:
    x              : (B,N,N,Dim)           float32
    mask (optional): (B,N,N)               bool/float32
    out            : (B,N,N,Dim)           float32
The hidden dimension is `Hidden = config["hidden_dim"]` (always a multiple of 32
on the test set).  For speed we cast the tensors that participate in the
batched GEMM to fp16, accumulate in fp32, and finally cast the result back to
fp32 for the remaining LayerNorm / Linear steps.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply (I×K)·(K×J) → (I×J)
#   left  : (Z, I, K)   fp16
#   right : (Z, K, J)   fp16
#   out   : (Z, I, J)   fp16  (accumulated in fp32)
# ----------------------------------------------------------------------
@triton.jit
def _batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1) tile indices
    # ------------------------------------------------------------------
    pid_z = tl.program_id(0)          # batch * hidden
    pid_m = tl.program_id(1)          # rows  (i)
    pid_n = tl.program_id(2)          # cols  (j)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 2) reduction over K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile   (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile  (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 accumulation
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # 3) write result (store as fp16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batch_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction.
    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous
    returns out : (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten leading dimensions so the kernel sees a 3‑D tensor
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # stride in element units (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes – tuned for Hopper SM90 (mix of compute & memory)
    BLOCK_M = 64   # rows  per tile
    BLOCK_N = 64   # cols  per tile
    BLOCK_K = 32   # reduction tile

    grid = (
        Z,                               # program_id(0) – batch*head
        triton.cdiv(I, BLOCK_M),         # program_id(1) – rows
        triton.cdiv(J, BLOCK_N),         # program_id(2) – cols
    )
    _batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                     # good default for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, Dim] (float32)
        mask         : torch.Tensor [B, N, N]   (bool/float32) or None
        weights      : dict of parameter tensors (all float32)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, Dim] (float32)
    """
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, D = x.shape
    H = cfg["hidden_dim"]          # hidden dimension (often == D)

    eps = 1e-5

    # --------------------------------------------------------------
    # 1) Input LayerNorm (keep in fp32 for numerical stability)
    # --------------------------------------------------------------
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_ln = (x - mean) / torch.sqrt(var + eps)
    x_ln = x_ln * w_norm + b_norm                     # (B,N,N,D)

    # --------------------------------------------------------------
    # 2) Linear projections (cast to fp16 for the heavy GEMM)
    # --------------------------------------------------------------
    def _fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w = _fp16(weights["left_proj.weight"])      # (Hidden, D)
    right_proj_w = _fp16(weights["right_proj.weight"])    # (Hidden, D)

    left_gate_w = _fp16(weights["left_gate.weight"])
    right_gate_w = _fp16(weights["right_gate.weight"])
    out_gate_w = _fp16(weights["out_gate.weight"])

    # fp16 version of the normalized input
    x_fp16 = x_ln.to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # (B,N,N,Hidden)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4) Gating (still fp16)
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()    # (B,N,N,Hidden)

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Triton batched GEMM  (contract over the middle N dimension)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    #   reshape to (B, H, I, K) and (B, H, K, J)
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, J)

    out_fp16 = _batch_matmul(left_t, right_t)       # (B, H, I, J) fp16

    # back to (B, N, N, Hidden) and cast to fp32 for the rest
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6) Output LayerNorm
    # --------------------------------------------------------------
    w_out_ln = weights["to_out_norm.weight"]
    b_out_ln = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out_ln = (out - mean_o) / torch.sqrt(var_o + eps)
    out_ln = out_ln * w_out_ln + b_out_ln

    # --------------------------------------------------------------
    # 7) Final gating + linear projection back to Dim
    # --------------------------------------------------------------
    out_gated = out_ln * out_gate.to(torch.float32)
    to_out_w = weights["to_out.weight"]               # (Dim, Hidden)
    out_final = F.linear(out_gated, to_out_w)         # (B,N,N,Dim)

    return out_final
