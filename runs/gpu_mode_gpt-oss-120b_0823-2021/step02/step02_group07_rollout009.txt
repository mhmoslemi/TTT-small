"""
custom_kernel – outgoing TriMul forward (AlphaFold3)

Algorithm
---------
1️⃣ LayerNorm on the 4‑D input (B,N,N,C) – fp32, manual (no torch.nn.LayerNorm)
2️⃣ Linear projection “left” & “right” (C → H) – fp32, bias‑free
3️⃣ Linear gates + sigmoid (C → H) – fp32
4️⃣ Triton kernel fuses mask, left‑gate, right‑gate into the projected tensors
   (in‑place, fp32).  This is the only part that must run on Triton.
5️⃣ Cast to fp16 and reshape to (B*H, N, N) – Tensor‑Core friendly
6️⃣ Batched GEMM (torch.bmm) computes the cubic contraction
   out[...,i,j,d] = Σ_k left[...,i,k,d] * right[...,j,k,d]
7️⃣ LayerNorm + output‑gate + final linear – fp32, manual
8️⃣ Return float‑32 tensor of shape (B,N,N,C)

The heavy O(N³) work is delegated to cuBLAS (torch.bmm) because it far
out‑performs any custom Triton tiling on Hopper.  The only Triton
kernel performs the element‑wise fusion that would otherwise require
three separate passes (mask, left‑gate, right‑gate) and thus saves
memory traffic and kernel launch overhead.
"""

import torch
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel – fuse mask, left‑gate and right‑gate into left/right
# ------------------------------------------------------------------
@triton.jit
def _fuse_mask_gate_kernel(
    left_ptr,          # fp32  [B,N,N,H]
    right_ptr,         # fp32  [B,N,N,H]
    mask_ptr,          # fp32  [B,N,N,1]   (broadcasted mask)
    left_gate_ptr,     # fp32  [B,N,N,H]
    right_gate_ptr,    # fp32  [B,N,N,H]
    B, N, H,           # compile‑time constants
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    pid   = tl.program_id(0)
    idx   = pid * BLOCK + tl.arange(0, BLOCK)          # linear index inside the tensor
    mask  = idx < B * N * N * H                         # guard

    # decode linear index → (b,i,j,h)
    b = idx // (N * N * H)
    rest = idx % (N * N * H)
    i = rest // (N * H)
    rest = rest % (N * H)
    j = rest // H
    h = rest % H

    # compute flat offsets for each operand
    left_off = b * stride_left_b + i * stride_left_i + j * stride_left_j + h * stride_left_h
    right_off = b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j                # mask has no H dim
    lgate_off = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h
    rgate_off = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h

    # load
    left   = tl.load(left_ptr   + left_off,   mask=mask, other=0.0)
    right  = tl.load(right_ptr  + right_off,  mask=mask, other=0.0)
    m      = tl.load(mask_ptr   + mask_off,   mask=mask, other=0.0)   # broadcasted scalar per (b,i,j)
    lgate  = tl.load(left_gate_ptr + lgate_off, mask=mask, other=0.0)
    rgate  = tl.load(right_gate_ptr+ rgate_off, mask=mask, other=0.0)

    # fuse mask and gates (the original code: left*=mask*left_gate, right*=mask*right_gate)
    left  = left  * m * lgate
    right = right * m * rgate

    # store back in‑place
    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left, right, mask, left_gate, right_gate):
    """
    In‑place element‑wise fusion of mask and both gates.
    All tensors are fp32 and have shape:
        left/right/left_gate/right_gate : (B,N,N,H)
        mask                           : (B,N,N,1)  (broadcasted)
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 2048                              # ~2 k threads → good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    # strides in *elements* (not bytes)
    sl = left.stride()
    sr = right.stride()
    sm = mask.stride()
    slg = left_gate.stride()
    srg = right_gate.stride()

    _fuse_mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=sl[0], stride_left_i=sl[1], stride_left_j=sl[2], stride_left_h=sl[3],
        stride_right_b=sr[0], stride_right_i=sr[1], stride_right_j=sr[2], stride_right_h=sr[3],
        stride_mask_b=sm[0], stride_mask_i=sm[1], stride_mask_j=sm[2],
        stride_lgate_b=slg[0], stride_lgate_i=slg[1], stride_lgate_j=slg[2], stride_lgate_h=slg[3],
        stride_rgate_b=srg[0], stride_rgate_i=srg[1], stride_rgate_j=srg[2], stride_rgate_h=srg[3],
        BLOCK=BLOCK,
    )
    return left, right


# ------------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradient needed).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask        : torch.Tensor, shape [B, N, N]  (float32) or None
        - weights_dict: mapping of string → torch.Tensor (all float32)
        - config_dict : contains "dim" (C) and "hidden_dim" (H)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) on the same device as input.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim       = cfg["dim"]          # C
    hidden    = cfg["hidden_dim"]   # H
    device    = x.device
    eps       = 1e-5                # LayerNorm epsilon (same for both norms)

    # ------------------------------------------------------------------
    # 1️⃣ Manual LayerNorm on the 4‑D input (fp32)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]   # (C,)
    ln_b = weights["norm.bias"]     # (C,)
    # mean / var over last dim
    mean = x.mean(dim=-1, keepdim=True)
    var  = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * ln_w + ln_b                                 # (B,N,N,C)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (C → H) – bias‑free, fp32
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (H, C)
    right_proj_w = weights["right_proj.weight"]  # (H, C)

    # torch.nn.functional.linear does: out = x @ weight.T + bias
    left  = torch.nn.functional.linear(x_norm, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x_norm, right_proj_w)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3️⃣ Gates (C → H) + sigmoid (fp32)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()   # (B,N,N,H)
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()  # (B,N,N,H)
    out_gate   = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Mask handling  (broadcast to the hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2],
                          dtype=x.dtype, device=device)
    mask = mask.unsqueeze(-1)                 # (B,N,N,1)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask + gates into left/right (in‑place, via Triton)
    # ------------------------------------------------------------------
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Heavy cubic contraction using batched GEMM (fp16, Tensor‑cores)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # cast to fp16 for the matmul
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*H, N, N) → each head is a separate batch element
    left_flat  = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, N)
    right_flat = right_h.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H, N, N)

    # out_flat[b*h, i, j] = Σ_k left_flat[b*h,i,k] * right_flat[b*h,j,k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # back to fp32 and original layout (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    to_out_ln_w = weights["to_out_norm.weight"]   # (H,)
    to_out_ln_b = weights["to_out_norm.bias"]     # (H,)

    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_ln_w + to_out_ln_b

    # ------------------------------------------------------------------
    # 8️⃣ Apply the output gate (fp32)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear (H → C) – fp32, bias‑free
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]   # (C, H)
    out_final = torch.nn.functional.linear(out_norm, to_out_w)   # (B,N,N,C)

    return out_final
