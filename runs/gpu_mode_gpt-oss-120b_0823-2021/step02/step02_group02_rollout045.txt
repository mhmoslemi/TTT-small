"""
TriMul “outgoing” (AlphaFold‑3) – mixed‑precision implementation.

*   LayerNorm, linear projections, and gating are done in PyTorch (FP32).
*   A tiny Triton kernel (`apply_mask_gate_kernel`) fuses the three element‑wise
    ops *mask*, *left‑gate* and *right‑gate* in FP16 – this is the required
    “part of the operations in a kernel”.
*   The costly N³ contraction
        out[b,i,j] = Σₖ left[b,i,k] · right[b,j,k]
    is performed as a batched GEMM with torch .bmm (cuBLAS/Tensor‑cores), after
    reshaping the tensors to (B·H, N, N).  This is the fastest strategy on H200.
*   The final LayerNorm, output‑gate and linear projection are again carried out
    in PyTorch.  The function returns a FP32 tensor of shape [B, N, N, dim].

The kernel works for:
    B ∈ {1,2}, N up to 1024, dim ∈ {128,384}, hidden_dim = 128,
    with or without a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel – element‑wise mask + gating (FP16)
# -------------------------------------------------------------------------
@triton.jit
def apply_mask_gate_kernel(
    left_ptr,          # fp16*  [B, N, N, H]   (contiguous)
    right_ptr,         # fp16*  [B, N, N, H]
    out_left_ptr,      # fp16*  [B, N, N, H]   (output after mask & gate)
    out_right_ptr,     # fp16*  [B, N, N, H]
    mask_ptr,          # fp16*  [B, N, N, 1]   (broadcasted mask)
    left_gate_ptr,     # fp16*  [B, N, N, H]
    right_gate_ptr,    # fp16*  [B, N, N, H]
    total_elems,       # int   = B*N*N*H
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offset < total_elems

    # load raw tensors
    l = tl.load(left_ptr + offset, mask=mask, other=0.0)
    r = tl.load(right_ptr + offset, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offset, mask=mask, other=1.0)   # 1.0 when no mask
    lg = tl.load(left_gate_ptr + offset, mask=mask, other=1.0)
    rg = tl.load(right_gate_ptr + offset, mask=mask, other=1.0)

    # fused mask & gate
    l_out = l * m * lg
    r_out = r * m * rg

    tl.store(out_left_ptr + offset, l_out, mask=mask)
    tl.store(out_right_ptr + offset, r_out, mask=mask)


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,               # input [B, N, N, dim]   (FP32)
        torch.Tensor,               # mask  [B, N, N]        (bool/float)
        Dict[str, torch.Tensor],    # weights
        Dict[str, Any],             # config {"dim":..., "hidden_dim":...}
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns tensor of shape [B, N, N, dim] (FP32).
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm + linear projections (PyTorch, FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )
    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # --------------------------------------------------------------
    # 2. Gates (FP32) – compute before casting to FP16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # --------------------------------------------------------------
    # 3. Cast to FP16 and prepare mask (also FP16)
    # --------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    left_gate_fp16  = left_gate.to(torch.float16)
    right_gate_fp16 = right_gate.to(torch.float16)

    if mask is None:
        # a dummy mask of ones – broadcastable to [B,N,N,1]
        mask_fp16 = torch.ones((B, N, N, 1), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)        # [B,N,N,1]

    # allocate output buffers for the fused step
    left_masked_fp16  = torch.empty_like(left_fp16)
    right_masked_fp16 = torch.empty_like(right_fp16)

    # --------------------------------------------------------------
    # 4. Launch tiny Triton kernel that fuses mask & gates
    # --------------------------------------------------------------
    total = B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    apply_mask_gate_kernel[grid](
        left_fp16.data_ptr(),
        right_fp16.data_ptr(),
        left_masked_fp16.data_ptr(),
        right_masked_fp16.data_ptr(),
        mask_fp16.data_ptr(),
        left_gate_fp16.data_ptr(),
        right_gate_fp16.data_ptr(),
        total,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 5. Batched GEMM for the N³ contraction (cuBLAS/Tensor‑cores)
    #    Shape after permute: [B, H, N, N] → reshape to (B*H, N, N)
    # --------------------------------------------------------------
    # (B,N,N,H) → (B,H,N,N)
    left_t = left_masked_fp16.permute(0, 3, 1, 2).contiguous()
    right_t = right_masked_fp16.permute(0, 3, 1, 2).contiguous()

    # reshape for batched matmul
    BxH = B * hidden_dim
    left_bmm  = left_t.view(BxH, N, N)          # (B*H, N, N)
    right_bmm = right_t.view(BxH, N, N)         # (B*H, N, N)

    # out_bmm = left @ right^T   (the “outgoing” formulation)
    out_bmm = torch.bmm(left_bmm, right_bmm.transpose(1, 2))   # (B*H, N, N)

    # reshape back to [B, H, N, N] and permute to original layout
    out_fp16 = out_bmm.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 6. Final LayerNorm, output‑gate and linear projection (FP32)
    # --------------------------------------------------------------
    out_fp32 = out_fp16.to(dtype)                      # (B,N,N,H)

    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # broadcast out_gate (shape B,N,N,1) over the hidden dim
    out_fp32 = out_fp32 * out_gate

    out = F.linear(out_fp32, weights["to_out.weight"].to(dtype))   # back to dim

    return out
