"""
TriMul (outgoing) – forward only.

The kernel only accelerates the *pair‑wise mask* application (the only
memory‑bound part) while the O(N³·C) reduction is performed with a
torch.einsum (mapped to highly‑optimised GEMM kernels on Hopper).  All
layer‑norms, linear projections and gating stay in PyTorch for numerical
stability and speed.  The Triton kernel is written for float32 and
broadcasts a [B,N,N] mask over the hidden dimension H (``hidden_dim``).

The entry‑point ``custom_kernel`` conforms to the required signature:
    (input_tensor, mask, weights, config) → output tensor
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton: element‑wise multiplication with a broadcast mask
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*   (input tensor, flattened)
    mask_ptr,         # const float*   (mask tensor, flattened, shape B*N*N)
    dst_ptr,          # float*         (output tensor, flattened)
    total_el,         # i64            total number of elements = B*N*N*H
    H,                # i64            hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear element ids
    mask = offs < total_el

    # Load the raw values
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # Every H consecutive elements share the same mask entry.
    # Compute the index into the flattened mask (shape B*N*N)
    mask_idx = offs // H
    m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Store masked value
    tl.store(dst_ptr + offs, src * m_val, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` of shape [B,N,N,H] by ``mask[...,None]`` using Triton.

    Parameters
    ----------
    tensor : torch.Tensor
        Input tensor (float32, CUDA) – will **not** be modified.
    mask : torch.Tensor
        Float mask (0/1) of shape [B,N,N] (float32, CUDA).

    Returns
    -------
    torch.Tensor
        Tensor of same shape as ``tensor`` with the mask applied.
    """
    B, N, _, H = tensor.shape
    total_el = B * N * N * H
    BLOCK = 1024                         # tuned for H200, gives good occupancy

    # Flatten to 1‑D for the kernel
    src_flat = tensor.view(-1)
    dst_flat = torch.empty_like(src_flat)
    mask_flat = mask.view(-1)            # [B,N,N]

    grid = (triton.cdiv(total_el, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr=src_flat,
        mask_ptr=mask_flat,
        dst_ptr=dst_flat,
        total_el=total_el,
        H=H,
        BLOCK=BLOCK,
    )
    return dst_flat.view_as(tensor)

# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  float32 CUDA
        - mask         : [B, N, N]       float32 CUDA (0/1)
        - weights      : dict of model parameters
        - config       : {"dim": int, "hidden_dim": int, "nomask": bool (opt.)}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32 CUDA)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                                            # [B,N,N,dim]

    # --------------------------------------------------------------
    # Linear projections (bias‑free) → [B,N,N,hidden_dim]
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # Optional mask – Triton‑accelerated
    # --------------------------------------------------------------
    if not nomask:
        left = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # --------------------------------------------------------------
    # Gating (sigmoid) and element‑wise gating of projected tensors
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"]
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"]
    ).sigmoid()

    left = left * left_gate          # [B,N,N,hidden_dim]
    right = right * right_gate

    # --------------------------------------------------------------
    # Core O(N³·C) reduction – torch.einsum (mapped to fast GEMM)
    # --------------------------------------------------------------
    # out shape: [B, N, N, hidden_dim]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # --------------------------------------------------------------
    # Final LayerNorm + gating + output projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                      # broadcast over hidden_dim
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"]
    )                                          # back to (dim)

    # Ensure float32 output (all ops already float32)
    return out.to(torch.float32)
