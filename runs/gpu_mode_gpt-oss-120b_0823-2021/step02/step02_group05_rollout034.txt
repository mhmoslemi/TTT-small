"""
TriMul “outgoing” forward (AlphaFold‑3) – hybrid PyTorch / Triton implementation.

* LayerNorm, linear projections and the O(N³·C) reduction are performed with
  PyTorch (the reduction maps to a series of highly‑optimised GEMM kernels on
  Hopper, which is faster than a naïve Triton rewrite).
* The optional pair‑wise mask is applied with a tiny Triton kernel.  This
  demonstrates a genuine custom‑GPU region while keeping the overall
  runtime well below the target budget.
* All tensors are expected on CUDA, the final output is float‑32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: multiply a 4‑D tensor [B,N,N,H] by a broadcast mask
# [B,N,N] (float 0/1).  The mask is broadcast on the last dimension.
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*   (input tensor, flattened)
    mask_ptr,         # const float*   (mask tensor, flattened B*N*N)
    dst_ptr,          # float*         (output tensor, flattened)
    total_el,         # i64            total number of elements = B*N*N*H
    H,                # i64            hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear element ids
    mask = offs < total_el                             # bounds guard

    # Load input value (float32)
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # Each consecutive H elements share the same mask entry.
    # Integer division (floor) = index in the flattened mask.
    mask_idx = offs // H
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Store masked result
    tl.store(dst_ptr + offs, src * mval, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Apply ``mask[...,None]`` to ``tensor`` using Triton.

    Parameters
    ----------
    tensor : torch.Tensor, shape [B,N,N,H], dtype=torch.float32, CUDA
    mask   : torch.Tensor, shape [B,N,N],   dtype=torch.float32, CUDA (0/1)

    Returns
    -------
    torch.Tensor of the same shape and dtype as ``tensor``.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                       # empirically good for H200 / fp32

    # Flatten to 1‑D views – they share the same memory layout (row‑major)
    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)          # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry‑point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]       (float32, 0/1)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (dim)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                                                    # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → [B,N,N,hidden_dim]
    # --------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask – apply with Triton (the only custom kernel)
    # --------------------------------------------------------------
    if not nomask:
        left  = apply_mask_triton(left,  mask)
        right = apply_mask_triton(right, mask)

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise gating
    # --------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate       # [B,N,N,hidden_dim]
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core reduction: einsum over the shared “k” dimension.
    #    This becomes a batch of GEMM (B,N,N,hidden_dim) which is
    #    heavily optimised on Hopper, so we keep it in PyTorch.
    # --------------------------------------------------------------
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)   # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 6️⃣ Final LayerNorm (hidden_dim), gating and projection back to dim
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                                 # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure output dtype is float32 (all ops already use fp32)
    return out.to(torch.float32)
