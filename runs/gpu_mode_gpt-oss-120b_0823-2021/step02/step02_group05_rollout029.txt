"""
TriMul (outgoing) forward – fast Triton‑accelerated implementation.

The kernel only handles the *mask‑application* step, which is the only
memory‑bound part that benefits from explicit vectorised loads/stores.
All the compute‑heavy pieces (LayerNorm, linear projections, gating,
and the O(N³·C) reduction) are performed with PyTorch – on Hopper the
einsum is translated into a sequence of highly‑optimised GEMM kernels
that are faster than a naïve Triton triple‑loop.

The implementation follows the reference PyTorch module exactly:
1. Layer‑norm on the input (dim).
2. Linear projections → left / right (hidden_dim).
3. Optional mask (broadcasted on the last dim) applied with Triton.
4. Gating (sigmoid) and element‑wise multiplication.
5. Reduction  out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]   (einsum).
6. Layer‑norm on the hidden dimension, gating, final linear projection
   back to dim.

The function `custom_kernel` receives a tuple
    (input_tensor, mask, weights, config)
and returns the processed tensor of shape [bs, seq_len, seq_len, dim].
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel – element‑wise multiplication of a [B,N,N,H] tensor with
# a broadcast mask of shape [B,N,N].  The mask is stored as float32 0/1.
# The kernel works on a flattened view and therefore needs the hidden
# dimension H to locate the corresponding mask entry.
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*   (input tensor)
    mask_ptr,         # const float*   (mask tensor, flattened B*N*N)
    dst_ptr,          # float*         (output tensor)
    total_el,         # i64            total number of elements = B*N*N*H
    H,                # i64            hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    mask = offs < total_el                            # out‑of‑bounds guard

    # Load the source value (float32)
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # Each consecutive H elements belong to the same (b,i,j) location,
    # therefore the mask index is floor(offs / H).
    mask_idx = offs // H
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Store the masked result
    tl.store(dst_ptr + offs, src * mval, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Apply a broadcast mask of shape [B,N,N] to a tensor of shape
    [B,N,N,H] using the Triton kernel defined above.
    Returns a new tensor with the same shape and dtype as ``tensor``.
    """
    B, N, _, H = tensor.shape
    total_el = B * N * N * H
    BLOCK = 1024                     # 1024 works well on Hopper/H200

    # Flatten tensors to 1‑D
    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)        # [B,N,N]

    grid = (triton.cdiv(total_el, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total_el,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see reference module)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free) → left / right   [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden_dim) – Triton path
    # ------------------------------------------------------------------
    if not nomask:
        left  = apply_mask_triton(left,  mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate      # [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·C) reduction via einsum → [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    # The einsum is equivalent to a batched GEMM over the shared k‑axis.
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm (hidden_dim), gating and linear projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                       # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure the result is float32 (all ops already use float32)
    return out.to(torch.float32)
