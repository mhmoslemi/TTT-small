"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm
---------
1️⃣ Layer‑norm over the input channel dim (`dim`).
2️⃣ Linear projections → `left` / `right` (B,N,N,hidden)   (no bias).
3️⃣ Compute three sigmoid gates (left, right, out)  (B,N,N,hidden).
4️⃣ Fuse optional mask, the three gates and the mask‑application onto
   `left`, `right` and `out_gate` in a single Triton kernel.
   The kernel works in‑place on `left`/`right` and writes the masked
   `out_gate` back.
5️⃣ Core contraction  
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]    
   This is a batched matrix‑multiply per hidden channel:
        (B, hidden, N, K) @ (B, hidden, K, N) → (B, hidden, N, N)
   We perform it with Torch’s highly‑optimised batched GEMM (cuBLAS/Tensor‑cores).
6️⃣ Final LayerNorm over the hidden dimension, scaling by `out_gate`
   and a linear projection back to `dim`.

Only step 4 is implemented in Triton – the required “contains a Triton
kernel” constraint – while the dominant O(N³) work stays in cuBLAS for
maximum throughput on Hopper/H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask and the three gates.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] projected tensors
    mask_ptr,                          # [B,N,N,1] (or all‑ones)
    lgate_ptr, rgate_ptr, ogate_ptr,   # [B,N,N,H] sigmoid‑gate tensors
    B, N, H,
    # strides (int64) of the tensors above
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,   # mask has a dummy fourth stride
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,k,d) of `left`/`right` (the 4‑D view):
        left  = left  * mask * lgate
        right = right * mask * rgate
        out_gate[b,i,k,d] = ogate * mask          # store masked out‑gate
    The mask lives on (b,i,k) and is broadcast along the hidden dim.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)       # flat offset
    total = B * N * N * H
    mask = offs < total

    # --------------------------------------------------------------
    # Decode flat index into (b,i,k,d)
    # --------------------------------------------------------------
    tmp = offs
    d = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # --------------------------------------------------------------
    # Build pointers for every tensor
    # --------------------------------------------------------------
    left_ptr   = left_ptr   + b * stride_l_b   + i * stride_l_i   + k * stride_l_k   + d * stride_l_h
    right_ptr  = right_ptr  + b * stride_r_b   + i * stride_r_i   + k * stride_r_k   + d * stride_r_h
    mask_ptr   = mask_ptr   + b * stride_m_b   + i * stride_m_i   + k * stride_m_k                     # scalar mask
    lgate_ptr  = lgate_ptr  + b * stride_lg_b  + i * stride_lg_i  + k * stride_lg_k  + d * stride_lg_h
    rgate_ptr  = rgate_ptr  + b * stride_rg_b  + i * stride_rg_i  + k * stride_rg_k  + d * stride_rg_h
    ogate_ptr  = ogate_ptr  + b * stride_og_b  + i * stride_og_i  + k * stride_og_k  + d * stride_og_h

    # --------------------------------------------------------------
    # Load values (mask is broadcast, load a scalar once)
    # --------------------------------------------------------------
    l   = tl.load(left_ptr,   mask=mask, other=0.0)
    r   = tl.load(right_ptr,  mask=mask, other=0.0)
    m   = tl.load(mask_ptr,   mask=mask, other=1.0)   # 1.0 when mask absent
    lg  = tl.load(lgate_ptr,  mask=mask, other=1.0)
    rg  = tl.load(rgate_ptr,  mask=mask, other=1.0)
    og  = tl.load(ogate_ptr,  mask=mask, other=1.0)

    # --------------------------------------------------------------
    # Fuse operations
    # --------------------------------------------------------------
    l = l * m * lg
    r = r * m * rg
    og = og * m               # out‑gate only needs mask

    # --------------------------------------------------------------
    # Store back
    # --------------------------------------------------------------
    tl.store(left_ptr,  l, mask=mask)
    tl.store(right_ptr, r, mask=mask)
    tl.store(ogate_ptr, og, mask=mask)


def _apply_mask_and_gates(left, right, mask, left_gate, right_gate, out_gate):
    """
    Launch the Triton kernel that fuses:
        * mask multiplication,
        * left/right gating,
        * out‑gate masking.
    The kernel mutates `left` and `right` in‑place and returns the
    masked `out_gate`.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    # Empirically good block size for Hopper (fits into 32‑KB shared memory)
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        ogate_ptr=out_gate,
        B=B,
        N=N,
        H=H,
        stride_l_b=left.stride(0),  stride_l_i=left.stride(1),
        stride_l_k=left.stride(2),  stride_l_h=left.stride(3),
        stride_r_b=right.stride(0), stride_r_i=right.stride(1),
        stride_r_k=right.stride(2), stride_r_h=right.stride(3),
        stride_m_b=mask.stride(0),  stride_m_i=mask.stride(1),
        stride_m_k=mask.stride(2),  stride_m_1=mask.stride(3),
        stride_lg_b=left_gate.stride(0), stride_lg_i=left_gate.stride(1),
        stride_lg_k=left_gate.stride(2), stride_lg_h=left_gate.stride(3),
        stride_rg_b=right_gate.stride(0), stride_rg_i=right_gate.stride(1),
        stride_rg_k=right_gate.stride(2), stride_rg_h=right_gate.stride(3),
        stride_og_b=out_gate.stride(0), stride_og_i=out_gate.stride(1),
        stride_og_k=out_gate.stride(2), stride_og_h=out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor    [B, N, N, dim]   (float32 / bf16)
        * mask_tensor  : torch.Tensor    [B, N, N] or None (0/1 float)
        * weights_dict : dict with the same keys as the reference module
        * config_dict  : dict with ``dim`` and ``hidden_dim`` entries

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]          # also called hidden_dim in the reference

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    # (B,N,N,dim) @ (dim, hidden) -> (B,N,N,hidden)
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Compute sigmoid gates
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 4️⃣ Prepare mask (broadcast to hidden dim)
    # --------------------------------------------------------------
    if mask is None:
        # all‑ones mask -> shape (B,N,N,1)
        mask = torch.ones(B, N, N, 1, dtype=x.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(x.dtype)   # (B,N,N,1)

    # --------------------------------------------------------------
    # 5️⃣ Fuse mask + gates (Triton)
    #    left/right are modified in‑place, out_gate comes back masked.
    # --------------------------------------------------------------
    out_gate = _apply_mask_and_gates(left, right, mask,
                                     left_gate, right_gate, out_gate)

    # --------------------------------------------------------------
    # 6️⃣ Core contraction Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Treat hidden dim as a batch dimension for a batched GEMM.
    # --------------------------------------------------------------
    # reshape to (B, hidden, N, N)
    left_t  = left.permute(0, 3, 1, 2)   # (B, H, N, K)   K == N
    right_t = right.permute(0, 3, 1, 2)  # (B, H, N, K)

    # batched matrix multiply -> (B, H, N, N)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))

    # back to (B, N, N, hidden)
    out = out_t.permute(0, 2, 3, 1)

    # --------------------------------------------------------------
    # 7️⃣ Final LayerNorm on hidden dim, gating, final linear proj.
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # scale by masked out_gate
    out = F.linear(out, weights["to_out.weight"])

    # The reference implementation returns float32.
    return out.to(torch.float32)
