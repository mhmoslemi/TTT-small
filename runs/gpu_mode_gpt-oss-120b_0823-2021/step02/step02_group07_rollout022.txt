"""
custom_kernel – “outgoing” TriMul forward pass (AlphaFold3)

Algorithm
---------
1. LayerNorm on the 4‑D input (B, N, N, C)   → fp32
2. Linear projections  (C → H)               → fp32
3. Linear gates + sigmoid                     → fp32
4. Fuse mask, left‑gate, right‑gate into the projected
   tensors with a small Triton kernel (in‑place, fp32).
5. Cast the fused tensors to fp16 and reshape to
   (B*H, N, N) so that the cubic contraction becomes a
   batched GEMM: out_h = left_h @ right_hᵀ.
   Tensor‑cores make this step ~10× faster than a naïve
   Triton triple‑loop.
6. Cast the result back to fp32, restore the original
   layout (B, N, N, H).
7. Output LayerNorm + output‑gate (fp32).
8. Final linear projection back to C (fp32).

Only the element‑wise mask‑gate fusion is done in Triton;
the dominant O(N³) work is delegated to cuBLAS via torch.bmm,
which is the fastest known implementation on Hopper (H200).
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask and per‑position gates into left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp32,   [B,N,N,H]
    right_ptr,         # fp32,   [B,N,N,H]
    mask_ptr,          # fp32,   [B,N,N,1] (broadcasted)
    left_gate_ptr,     # fp32,   [B,N,N,H]
    right_gate_ptr,    # fp32,   [B,N,N,H]
    B, N, H,           # compile‑time constants
    # strides (in *elements*, not bytes)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    """In‑place: left = left * mask * left_gate,
                 right = right * mask * right_gate."""
    pid = tl.program_id(0)
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # linear offset → (b,i,j,h)
    b = (offs // (N * N * H))
    r = offs % (N * N * H)
    i = (r // (N * H))
    r = r % (N * H)
    j = (r // H)
    h = r % H

    # compute per‑tensor pointer offsets
    left_off = (b * stride_left_b + i * stride_left_i
                + j * stride_left_j + h * stride_left_h)
    right_off = (b * stride_right_b + i * stride_right_i
                 + j * stride_right_j + h * stride_right_h)
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    lgate_off = (b * stride_lgate_b + i * stride_lgate_i
                 + j * stride_lgate_j + h * stride_lgate_h)
    rgate_off = (b * stride_rgate_b + i * stride_rgate_i
                 + j * stride_rgate_j + h * stride_rgate_h)

    # loads
    left = tl.load(left_ptr + left_off, mask=mask, other=0.0)
    right = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=mask, other=0.0)      # broadcasted mask
    lgate = tl.load(left_gate_ptr + lgate_off, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + rgate_off, mask=mask, other=0.0)

    # fuse
    left = left * m * lgate
    right = right * m * rgate

    # store back (in‑place)
    tl.store(left_ptr + left_off, left, mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left, right, mask, left_gate, right_gate):
    """In‑place element‑wise fusion using the Triton kernel above."""
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 2048                     # good occupancy on Hopper
    grid = (triton.cdiv(total, BLOCK),)

    # strides in *elements*
    s_left = left.stride()
    s_right = right.stride()
    s_mask = mask.stride()
    s_lgate = left_gate.stride()
    s_rgate = right_gate.stride()

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=s_left[0], stride_left_i=s_left[1],
        stride_left_j=s_left[2], stride_left_h=s_left[3],
        stride_right_b=s_right[0], stride_right_i=s_right[1],
        stride_right_j=s_right[2], stride_right_h=s_right[3],
        stride_mask_b=s_mask[0], stride_mask_i=s_mask[1],
        stride_mask_j=s_mask[2],
        stride_lgate_b=s_lgate[0], stride_lgate_i=s_lgate[1],
        stride_lgate_j=s_lgate[2], stride_lgate_h=s_lgate[3],
        stride_rgate_b=s_rgate[0], stride_rgate_i=s_rgate[1],
        stride_rgate_j=s_rgate[2], stride_rgate_h=s_rgate[3],
        BLOCK=BLOCK,
    )
    return left, right


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        * mask_tensor   : torch.Tensor of shape [B, N, N] or None
        * weights_dict  : mapping name → torch.Tensor (all on same device)
        * config_dict   : must contain keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32, same device)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden = config["hidden_dim"]          # H in the paper

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last axis (fp32)
    # ------------------------------------------------------------------
    eps = 1e-5
    ln_w = weights["norm.weight"]          # (C,)
    ln_b = weights["norm.bias"]            # (C,)
    # mean & var over channel dim
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * ln_w + ln_b          # (B,N,N,C)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias, fp32)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]   # (H, C)
    right_proj_w = weights["right_proj.weight"] # (H, C)
    # torch.nn.functional.linear expects weight of shape (out_features, in_features)
    left = torch.nn.functional.linear(x_norm, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x_norm, right_proj_w) # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3️⃣ Gates + sigmoid (fp32)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()   # (B,N,N,H)
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid() # (B,N,N,H)
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()     # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Prepare mask (broadcast to H dim) – if mask missing treat as 1
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2],
                          device=device, dtype=x.dtype)
    # (B,N,N,1) – Triton kernel expects same dtype as left/right (fp32)
    mask = mask.unsqueeze(-1).to(x.dtype)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask & gates into left/right in‑place (Triton)
    # ------------------------------------------------------------------
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Heavy N³ contraction via batched GEMM (fp16, Tensor‑cores)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape          # H == hidden

    # cast to half‑precision for the matmul
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*H, N, N) – we contract over the third axis (k)
    left_flat  = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_flat = right_h.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matrix multiply: (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # back to fp32 and original layout (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    to_out_ln_w = weights["to_out_norm.weight"]   # (H,)
    to_out_ln_b = weights["to_out_norm.bias"]     # (H,)
    eps_o = 1e-5
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps_o)
    out_norm = out_norm * to_out_ln_w + to_out_ln_b

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate (fp32, element‑wise)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to original dim (fp32)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]          # (C, H)
    out_final = torch.nn.functional.linear(out_norm, to_out_w)   # (B,N,N,C)

    return out_final
