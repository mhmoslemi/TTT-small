"""
TriMul “outgoing” forward (AlphaFold‑3) – hybrid PyTorch + Triton implementation.

The heavy O(N³·C) reduction (einsum over the shared “k’’ axis) is kept in
PyTorch because on Hopper it maps to a series of highly‑optimised GEMM kernels.
The only part that is forced to run in Triton is the optional pair‑wise mask
application on the projected tensors (left/right).  This memory‑bound step
benefits from Triton’s explicit vectorised loads/stores and masking while keeping
the rest of the pipeline simple and fast.

All tensors are expected on CUDA (float32).  The final output is also float32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: multiply a [B,N,N,H] tensor by a broadcast mask [B,N,N].
# The mask is 0/1 (float) and is broadcast on the last dimension.
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*   (input tensor)
    mask_ptr,         # const float*   (mask tensor, shape B*N*N)
    dst_ptr,          # float*         (output tensor)
    total_el,         # i64            total elements = B*N*N*H
    H,                # i64            hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)            # linear element ids
    mask = offs < total_el                               # OOB guard

    # Load the source value
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # Each H‑wide slice shares the same mask entry.
    # Integer division gives the index into the flattened mask.
    mask_val = tl.load(mask_ptr + (offs // H), mask=mask, other=1.0)

    # Store masked value
    tl.store(dst_ptr + offs, src * mask_val, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` of shape [B,N,N,H] by ``mask[...,None]`` using Triton.
    ``mask`` is float (0/1) with shape [B,N,N].
    Returns a new tensor with the same shape as ``tensor``.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                     # good trade‑off for Hopper/H200

    # flatten to 1‑D views
    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)        # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry‑point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                                            # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections → shape [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask – Triton accelerated
    # ------------------------------------------------------------------
    if not nomask:
        left  = apply_mask_triton(left,  mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise gating of projected tensors
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate           # [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·C) reduction: einsum over the shared “k’’ dimension.
    #    This maps to a sequence of GEMM calls that are already optimal on
    #    Hopper (no need for a custom Triton implementation).
    # ------------------------------------------------------------------
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm (hidden_dim), gating and projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                     # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure float32 output (all ops use float32)
    return out.to(torch.float32)
