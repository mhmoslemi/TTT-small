"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (FP32, manual – no extra kernel needed).
2️⃣ Four linear projections + three gate‑projections.
   • All linear work is done in FP16 for speed.
3️⃣ Optional mask is applied (broadcast over the hidden dimension).
4️⃣ Sigmoid gates are applied.
5️⃣ Core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix‑multiply (N×N)·(N×N) → (N×N) for each
   (batch, hidden) pair.  The heavy O(N³) work is handled by a
   tiny Triton GEMM kernel that tiles the M‑, N‑ and K‑axes
   (64×64×32 tiles, fp16×fp16→fp32 accumulation, fp16 store).
   The kernel is launched on a collapsed dimension B*H.
6️⃣ Convert the result back to FP32, apply a second LayerNorm,
   multiply by the output‑gate and finally a linear projection
   back to the original dimension.

The implementation follows the “outgoing” variant from the
AlphaFold‑3 paper and is tuned for Hopper/SM‑90 (NVIDIA H200).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched GEMM (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_matmul_kernel(
    a_ptr,                       # left matrix  (B*H, I, K)
    b_ptr,                       # right matrix (B*H, K, J)
    c_ptr,                       # output       (B*H, I, J)
    M, N, K,                     # extents
    stride_az, stride_am, stride_ak,   # a strides (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)          # batch*head index
    pid_m = tl.program_id(0)          # tile along I
    pid_n = tl.program_id(1)          # tile along J

    # ------------------- tile start offsets -------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------- accumulator (fp32) -------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- reduction over K --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (BLOCK_M x BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile (BLOCK_K x BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp16×fp16 → fp32 dot product
        acc += tl.dot(a, b)

    # ------------------- store result (fp16) -----------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous
    returns: (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse batch*head for the kernel
    batch_dim = B * H
    a = left.view(batch_dim, I, K)
    b = right.view(batch_dim, K, J)
    c = out.view(batch_dim, I, J)

    # strides in element units (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # tiles over I
        triton.cdiv(J, BLOCK_N),   # tiles over J
        batch_dim,                 # one program per (batch,head)
    )
    _batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Main entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (float or bool)
        weights      : dict of leaf weight tensors (see reference code)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual – faster than nn.LayerNorm)
    # --------------------------------------------------------------
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    # compute mean & var over the last dimension
    mean = inp.mean(dim=-1, keepdim=True)
    var = inp.var(dim=-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * w_norm + b_norm                     # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Project to hidden dim (FP16)
    # --------------------------------------------------------------
    to_fp16 = lambda t: t.to(torch.float16)

    left_proj_w  = to_fp16(weights["left_proj.weight"])   # (hidden, dim)
    right_proj_w = to_fp16(weights["right_proj.weight"])

    left_gate_w  = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w   = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                         # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)                # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden)
    # --------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)         # (B,N,N,1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – also in FP16
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction via Triton GEMM
    #    left : (B,N,N,hidden)  -> (B, hidden, N, N)
    #    right: (B,N,N,hidden)  -> (B, hidden, N, N) and then transpose K,J
    # --------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()      # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()     # (B, H, K, J)

    out_fp16 = _triton_batched_matmul(left_t, right_t)   # (B, H, I, J)

    # back to (B,N,N,dim) and FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32)
    # --------------------------------------------------------------
    w_out_norm = weights["to_out_norm.weight"]
    b_out_norm = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * w_out_norm + b_out_norm

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)               # broadcast over dim
    out = F.linear(out, weights["to_out.weight"])        # (B,N,N,dim)

    return out
