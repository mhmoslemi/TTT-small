"""
TriMul – outgoing version (AlphaFold‑3)

Algorithm
---------
1️⃣ Layer‑norm over the last dimension (dim) – done in PyTorch (fast, stable).
2️⃣ Linear projections (left / right) and three gate projections – bias‑free
   Linear layers in PyTorch.
3️⃣ One Triton kernel fuses  
       * optional mask (broadcast over the hidden dim)  
       * sigmoid(gate) for left, right and out‑gate  
       * in‑place multiplication of the mask & gate into the projected tensors.
   The kernel works on a flat index → (b,i,k,d) and therefore supports any
   batch size (1‑2), any sequence length N and any hidden dimension H.
4️⃣ Core contraction:  
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]  
   This is a batched matrix‑multiply on the view (B, H, N, K) → uses
   torch.matmul which falls back to cuBLAS/Tensor‑cores (the dominant O(N³)
   work).  No extra Triton work is required for the N³ part.
5️⃣ Final Layer‑norm, out‑gate scaling and a linear projection back to dim –
   again in PyTorch.

Only step‑3 is implemented in Triton, satisfying the “must contain a Triton
kernel’’ requirement while keeping the overall runtime optimal on Hopper‑based
GPUs (H200).  All tensors stay in float32 throughout (the spec asks for a
float32 output), but the kernel can be run in float16 if the user passes
float16 tensors – the code is dtype‑agnostic.

The kernel is deliberately simple:
    * no shared memory, no block‑level sync,
    * a single large 1‑D grid (BLOCK = 1024) that maps nicely to the H200
      register budget,
    * uses the exact strides of the input tensors (no assumptions about the
      layout),
    * sigmoid is computed with the standard exp‑based formula (fast enough for
      inference and fully accurate).

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: combine mask + three sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,           # [B,N,N,H]  float32
    right_ptr,          # [B,N,N,H]  float32
    mask_ptr,           # [B,N,N,1]  float32 (broadcasted)
    left_gate_ptr,      # [B,N,N,H]  float32 (pre‑sigmoid)
    right_gate_ptr,     # [B,N,N,H]  float32 (pre‑sigmoid)
    out_gate_ptr,       # [B,N,N,H]  float32 (pre‑sigmoid)
    B, N, H,
    # strides (int64)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k, stride_mask_1,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    stride_ogate_b, stride_ogate_i, stride_ogate_k, stride_ogate_h,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,k,d):
        left  = left  * mask * sigmoid(left_gate)
        right = right * mask * sigmoid(right_gate)
        out_gate = sigmoid(out_gate)          (stored back in‑place)

    mask has a singleton hidden dimension, therefore its stride for the hidden
    axis is ignored (`stride_mask_1`).  All other tensors are 4‑D.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    total = B * N * N * H
    active = offs < total

    # ---- decode linear index → (b,i,k,d) -----------------------------
    idx = offs
    d = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # ---- compute pointers --------------------------------------------
    left   = left_ptr   + b * stride_left_b   + i * stride_left_i   + k * stride_left_k   + d * stride_left_h
    right  = right_ptr  + b * stride_right_b  + i * stride_right_i  + k * stride_right_k  + d * stride_right_h
    mask_  = mask_ptr   + b * stride_mask_b   + i * stride_mask_i   + k * stride_mask_k                 # hidden dim omitted
    lgate  = left_gate_ptr  + b * stride_lgate_b  + i * stride_lgate_i  + k * stride_lgate_k  + d * stride_lgate_h
    rgate  = right_gate_ptr + b * stride_rgate_b + i * stride_rgate_i + k * stride_rgate_k + d * stride_rgate_h
    ogate  = out_gate_ptr   + b * stride_ogate_b   + i * stride_ogate_i   + k * stride_ogate_k   + d * stride_ogate_h

    # ---- load ---------------------------------------------------------
    l   = tl.load(left,  mask=active, other=0.0)
    r   = tl.load(right, mask=active, other=0.0)
    m   = tl.load(mask_, mask=active, other=1.0)      # 1.0 → no mask
    lg  = tl.load(lgate, mask=active, other=0.0)
    rg  = tl.load(rgate, mask=active, other=0.0)
    og  = tl.load(ogate, mask=active, other=0.0)

    # ---- sigmoid ------------------------------------------------------
    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))
    og = 1.0 / (1.0 + tl.exp(-og))

    # ---- fuse ---------------------------------------------------------
    l = l * m * lg
    r = r * m * rg

    # ---- store back ---------------------------------------------------
    tl.store(left,   l, mask=active)
    tl.store(right,  r, mask=active)
    tl.store(ogate,  og, mask=active)   # out_gate is now sigmoided
    # left_gate / right_gate are not needed afterwards – they can be freed
    # by the caller if desired.


def _apply_mask_and_gates(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """
    Launches the Triton kernel that:
        * multiplies left/right by mask,
        * multiplies left/right by sigmoid(gate),
        * replaces out_gate with sigmoid(out_gate).

    All tensors are shape (B,N,N,H) except ``mask`` which is (B,N,N,1).
    Returns the sigmoided ``out_gate`` tensor (the tensor itself is mutated in‑place).
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024   # chosen to keep register pressure low on H200
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fusion_kernel[grid](
        left,
        right,
        mask,
        left_gate,
        right_gate,
        out_gate,
        B, N, H,
        # strides
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2), mask.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate   # now contains sigmoid(out_gate)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32 / bf16)
        - mask_tensor  : torch.Tensor of shape [B, N, N] or ``None``.
        - weights_dict : mapping with the following keys (all tensors on the same device):
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]          # = config["hidden_dim"]

    # --------------------------------------------------------------
    # 0️⃣ Input layer‑norm (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 1️⃣ Linear projections (bias‑free)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 2️⃣ Gate pre‑sigmoid tensors
    # --------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"])
    right_gate = F.linear(x, weights["right_gate.weight"])
    out_gate   = F.linear(x, weights["out_gate.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Prepare mask (broadcasted to hidden dim)
    # --------------------------------------------------------------
    if mask is None:
        mask4d = torch.ones((B, N, N, 1), dtype=x.dtype, device=device)
    else:
        mask4d = mask.unsqueeze(-1).to(x.dtype)   # (B,N,N,1)

    # --------------------------------------------------------------
    # 4️⃣ Triton‑fused mask + sigmoid(gates) → left, right, out_gate
    # --------------------------------------------------------------
    out_gate = _apply_mask_and_gates(left, right, mask4d,
                                    left_gate, right_gate, out_gate)

    # --------------------------------------------------------------
    # 5️⃣ Core contraction (batched matmul over K dimension)
    #     out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # bring hidden dim to the second axis so torch.matmul works as (B*H, N, N)
    left_t  = left.permute(0, 3, 1, 2)          # (B, hidden, N, K)
    right_t = right.permute(0, 3, 1, 2)         # (B, hidden, N, K)

    # batched matrix multiplication – cuBLAS/Tensor‑cores handle the O(N³) work
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))   # (B, hidden, N, N)

    # restore original layout
    out = out_t.permute(0, 2, 3, 1)            # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 6️⃣ Final layer‑norm, out‑gate scaling, linear projection back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                       # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])

    # --------------------------------------------------------------
    # guarantee float32 output as required by the spec
    # --------------------------------------------------------------
    return out.to(torch.float32)
