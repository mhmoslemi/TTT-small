"""
TriMul “outgoing” (AlphaFold‑3) – hybrid implementation.

*   A tiny Triton kernel fuses the mask, the three gating tensors and the
    linear projections (`left` and `right`).  This kernel touches every element
    of the 4‑D tensors `[B, N, N, H]` once and therefore benefits from the
    massive parallelism of the H200.
*   The expensive N³ contraction

        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

    is performed with a single batched GEMM (`torch.bmm`) after reshaping the
    tensors to `(B*H, N, N)`.  cuBLAS on Hopper delivers FP16 Tensor‑Core
    throughput that is far faster than a hand‑written Triton tiling for the
    typical sizes (N up to 1024, H = hidden_dim = 128).

All data are kept in FP16 inside the kernel and the GEMM; the final result is
converted back to FP32, Layer‑Normed, gated and projected to the original
dimension `dim`.  No gradients are required (forward‑only).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel – element‑wise mask & gating fusion
# -------------------------------------------------------------------------
@triton.jit
def fuse_kernel(
    left_ptr,          # fp16*  [B, N, N, H]
    right_ptr,         # fp16*  [B, N, N, H]
    left_gate_ptr,     # fp16*  [B, N, N, H]
    right_gate_ptr,    # fp16*  [B, N, N, H]
    mask_ptr,          # fp8?   [B, N, N]   (will be cast to fp16 inside)
    B, N, H,           # sizes (int)
    stride_l0, stride_l1, stride_l2, stride_l3,   # left strides
    stride_r0, stride_r1, stride_r2, stride_r3,   # right strides
    stride_g0, stride_g1, stride_g2, stride_g3,   # gate strides
    stride_m0, stride_m1, stride_m2,              # mask strides
    BLOCK: tl.constexpr,
):
    """
    For every element (b, i, j, h) compute

        left  = left  * mask * left_gate
        right = right * mask * right_gate

    The mask is broadcasted over the hidden dimension `H`.
    """
    pid = tl.program_id(0)
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # decode linear index into (b,i,j,h)
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    i = (tmp // N) % N
    b = tmp // (N * N)

    # compute pointers
    left_off  = (b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3)
    right_off = (b * stride_r0 + i * stride_r1 + j * stride_r2 + h * stride_r3)
    lg_off    = (b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3)
    rg_off    = (b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3)  # same layout
    m_off     = (b * stride_m0 + i * stride_m1 + j * stride_m2)

    # load (mask is fp32 -> cast)
    m = tl.load(mask_ptr + m_off, mask=mask, other=0.0).to(tl.float16)   # [B,N,N]
    lg = tl.load(left_gate_ptr + lg_off, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + rg_off, mask=mask, other=0.0)

    l = tl.load(left_ptr + left_off, mask=mask, other=0.0)
    r = tl.load(right_ptr + right_off, mask=mask, other=0.0)

    l = l * m * lg
    r = r * m * rg

    tl.store(left_ptr  + left_off,  l, mask=mask)
    tl.store(right_ptr + right_off, r, mask=mask)


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input [B,N,N,dim]
        torch.Tensor,          # mask   [B,N,N]   (or None)
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # == H

    # -----------------------------------------------------------------
    # 1. LayerNorm + linear projections (performed in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left  = F.linear(x, weights["left_proj.weight"].to(dtype))     # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # gating tensors (pre‑sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))   # (B,N,N,H)

    # -----------------------------------------------------------------
    # 2. Fuse mask & gating with a tiny Triton kernel
    # -----------------------------------------------------------------
    # ensure contiguous layout [B,N,N,H]
    left  = left.contiguous()
    right = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    left_fp16  = left .to(torch.float16)
    right_fp16 = right.to(torch.float16)
    lg_fp16    = left_gate .to(torch.float16)
    rg_fp16    = right_gate.to(torch.float16)

    if mask is not None:
        mask_fp16 = mask.to(torch.float16).contiguous()
    else:
        # a mask of all ones – no effect
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # launch kernel (one thread per element)
    BLOCK = 128
    total_elements = B * N * N * hidden_dim
    grid = (triton.cdiv(total_elements, BLOCK),)

    fuse_kernel[grid](
        left_fp16,
        right_fp16,
        lg_fp16,
        rg_fp16,
        mask_fp16,
        B,
        N,
        hidden_dim,
        # strides for left
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        left_fp16.stride(3),
        # strides for right
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        # strides for gates
        lg_fp16.stride(0),
        lg_fp16.stride(1),
        lg_fp16.stride(2),
        lg_fp16.stride(3),
        # strides for mask (no hidden dim)
        mask_fp16.stride(0),
        mask_fp16.stride(1),
        mask_fp16.stride(2),
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 3. Batched GEMM (cuBLAS) – the N³ contraction
    # -----------------------------------------------------------------
    # reshape to (B*H, N, N) and multiply left @ rightᵀ
    left_bh = left_fp16.view(B * hidden_dim, N, N)
    right_bh = right_fp16.view(B * hidden_dim, N, N)

    # torch.bmm works with fp16 on Hopper (Tensor‑Core)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))   # (B*H, N, N)

    # reshape back to [B, N, N, H]
    out = out_bh.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype)   # FP32 for the next LayerNorm

    # -----------------------------------------------------------------
    # 4. Post‑processing (LayerNorm, output gate, final linear)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # out_gate has shape [B,N,N,H] – broadcast over the (i,j) pair dimensions
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
