"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3) forward pass.

The implementation keeps the heavyweight O(N³·C) reduction in PyTorch
(einsum → a series of highly‑optimised GEMMs on Hopper) and moves the
mask‑application into Triton, which is the only memory‑bound part that
benefits from explicit vectorised loads/stores.  All linear layers,
LayerNorms and sigmoids are performed with the native PyTorch kernels,
ensuring numerical correctness and maximal utilisation of the H200
tensor‑cores.

The kernel `mask_apply_kernel` multiplies a tensor of shape
[B, N, N, H] by a broadcast mask [B, N, N] (float 0/1) without any Python
looping, using a 1‑D grid and a simple stride‑based address calculation.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: broadcast mask multiplication
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*   (input tensor, flattened)
    mask_ptr,         # const float*   (mask tensor, flattened, shape B*N*N)
    dst_ptr,          # float*         (output tensor, flattened)
    total_el,         # i64            total elements = B*N*N*H
    H,                # i64            hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear element indices
    mask = offs < total_el                             # out‑of‑bounds guard

    # load source element
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # each consecutive block of H elements shares the same mask entry
    mask_idx = offs // H
    m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # store masked value
    tl.store(dst_ptr + offs, src * m_val, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` of shape [B, N, N, H] by ``mask[...,None]`` using Triton.
    ``mask`` is float (0/1) with shape [B, N, N] and resides on the same device.
    Returns a new tensor with the same shape and dtype as ``tensor``.
    """
    B, N, _, H = tensor.shape
    total_el = B * N * N * H
    BLOCK = 1024                     # empirically good for Hopper/H200

    # flatten to 1‑D for the kernel
    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)        # [B, N, N]

    grid = (triton.cdiv(total_el, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total_el,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input tensor [B, N, N, dim]
        torch.Tensor,          # mask tensor  [B, N, N] (float0/1) or None
        Dict[str, torch.Tensor],  # weights dict
        Dict[str, Any],            # config dict
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32 0/1) or None
        - weights      : dict with the following keys
            * norm.weight, norm.bias
            * left_proj.weight, right_proj.weight
            * left_gate.weight, right_gate.weight, out_gate.weight
            * to_out_norm.weight, to_out_norm.bias
            * to_out.weight
        - config : dict containing at least ``dim`` and ``hidden_dim``.
            Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied with Triton)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape: [B, N, N]; broadcasted onto last dim
        left = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"]
    ).sigmoid()                                   # [B,N,N,hidden_dim]
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"]
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·C) reduction via einsum (maps to GEMM on Hopper)
    # ------------------------------------------------------------------
    # out shape: [B, N, N, hidden_dim]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm, output gating and projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate  # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure the result is float32 (already is, but keep explicit)
    return out.to(torch.float32)
