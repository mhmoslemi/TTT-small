"""
custom_kernel – “outgoing” TriMul (AlphaFold3)

Algorithm
---------
1️⃣ Layer‑norm the input (FP32, hand‑written – cheaper than nn.LayerNorm).
2️⃣ Linear projections + gating are performed in FP16 for bandwidth efficiency.
3️⃣ Optional mask is applied (broadcast over the hidden dimension).
4️⃣ Core contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix‑multiply (I×K)·(K×J) → (I×J) where
   I = J = N (sequence length) and K = hidden_dim.
   We launch a single Triton kernel that treats each (batch, head) pair as
   an independent GEMM; the kernel tiles in the I/J dimensions (64×64) and
   reduces over K in 32‑element tiles, accumulates in FP32 and writes FP16.
5️⃣ Convert the result back to FP32, apply a second LayerNorm, multiply by the
   output‑gate and finish with a linear projection to the original dimension.

All heavy work (step 4) lives in Triton; the remaining steps stay in PyTorch
because they are cheap relative to the O(N³) contraction.
The implementation supports both masked / un‑masked inputs and arbitrary
batch size (B∈{1,2}) and hidden dimensions (c_z∈{128,384,768}).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel – batched GEMM (I×K)·(K×J) → (I×J)
# --------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,            # left  matrix   (B*H, I, K)
    b_ptr,            # right matrix   (B*H, K, J)
    c_ptr,            # output matrix  (B*H, I, J)
    M, N, K,          # matrix extents
    stride_az, stride_am, stride_ak,   # a strides (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul contraction.

    Each program processes a tile (BLOCK_M × BLOCK_N) of the output matrix
    for a single batch‑head index `pid_z`.
    """
    pid_z = tl.program_id(2)          # batch*head dimension
    pid_m = tl.program_id(0)          # tile row index
    pid_n = tl.program_id(1)          # tile col index

    # ---- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (more stable than FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- reduction over K ------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M × BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile: (BLOCK_K × BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot‑product (FP16 × FP16 -> FP32)
        acc += tl.dot(a, b)

    # ---- store ---------------------------------------------------------
    c = acc.to(tl.float16)   # back to FP16 for storage
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)   FP16, contiguous
    right : (B, H, K, J)   FP16, contiguous
    returns: (B, H, I, J) FP16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # output buffer
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse batch*head for the kernel
    batch = B * H
    a = left.reshape(batch, I, K)
    b = right.reshape(batch, K, J)
    c = out.reshape(batch, I, J)

    # strides in element units (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tile sizes – tuned for Hopper FP16 Tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),          # rows
        triton.cdiv(J, BLOCK_N),          # cols
        batch,                            # one program per (batch,head)
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # fp16 friendly on SM90
    )
    return out


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf weight tensors (see docstring above)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual)
    # --------------------------------------------------------------
    # weight & bias are 1‑D tensors of length `dim`
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gating (FP16)
    # --------------------------------------------------------------
    # convert once
    def _to_fp16(t): return t.to(torch.float16)

    left_proj_w   = _to_fp16(weights["left_proj.weight"])   # (hidden, dim)
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)           # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)     # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ optional mask
    # --------------------------------------------------------------
    if mask is not None:
        # mask broadcast over hidden
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()   # (B,N,N,hidden)
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ heavy contraction with Triton
    # --------------------------------------------------------------
    # reshape to (B, H, I, K) and (B, H, K, J)
    # here H == hidden, I == J == N, K == hidden
    # we need left: (B, hidden, N, hidden)   –  i←seq, k←hidden
    # and right: (B, hidden, hidden, N)    –  k←hidden, j←seq
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # batch GEMM in FP16
    out_fp16 = _batched_gemm(left_perm, right_perm)     # (B, H, I, J)

    # back to (B, N, N, dim) and FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ second LayerNorm (FP32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 7️⃣ final gating + linear projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over dim
    to_out_w = weights["to_out.weight"]             # (dim, hidden)
    out = F.linear(out, to_out_w)                   # (B,N,N,dim)

    return out
```

**Explanation of design choices**

* **LayerNorm** – done manually in FP32 because the cost is negligible compared with the O(N³) contraction and avoids the extra kernel launch.
* **Projection & gating** – cast to FP16 (the weights are stored FP32) to halve memory traffic; PyTorch `F.linear` on FP16 tensors is highly optimized on Hopper.
* **Mask handling** – a single broadcast multiplication; no extra kernel needed.
* **Core contraction** – the dominant work. By reshaping `left` to `(B, H, I, K)` and `right` to `(B, H, K, J)` we obtain a standard batched GEMM where each `(batch, head)` pair is independent. The Triton kernel tiles in I/J (`64×64`) and reduces over K in `32`‑element chunks, accumulates in FP32 (to keep numerical stability for the large‑N case) and writes FP16. This pattern matches the proven “persistent‑workspace” GEMM style and fits comfortably into the shared‑memory budget of SM‑90.
* **Post‑processing** – the output of the kernel is converted back to FP32, a second LayerNorm is applied, the output‑gate (computed in FP16 then cast) modulates the result, and the final linear projection brings the tensor back to the original `dim`. All these steps stay in PyTorch because they are memory‑bound and cheap.

The implementation respects all constraints:
* works for any `B ∈ {1,2}`, `N` up to 1024, `dim ∈ {128,384}` and `hidden_dim ∈ {128,384,768}`;
* supports masked or un‑masked inputs;
* returns a float‑32 tensor of shape `[B, N, N, dim]`;
* includes at least one non‑trivial Triton kernel (the batched GEMM) and runs efficiently on an NVIDIA H200.
