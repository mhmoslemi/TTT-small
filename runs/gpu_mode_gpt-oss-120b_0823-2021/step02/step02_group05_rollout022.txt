"""
TriMul “outgoing” forward (AlphaFold‑3) – fast Triton mask apply.

The kernel only handles the element‑wise multiplication of the projected
tensors (`left` and `right`) by the optional pair‑wise mask of shape
[batch, seq_len, seq_len].  All remaining ops (LayerNorm, linear projections,
gating, the O(N³·C) reduction, final LayerNorm and output projection) are done
with native PyTorch which already maps to highly‑optimised CUDA GEMM kernels.
The mask‑apply kernel is memory‑bound and benefits from Triton’s explicit
vectorised loads/stores and masking, satisfying the “some work must be in
Triton” requirement while keeping the implementation simple and portable.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: multiply a 4‑D tensor [B,N,N,H] by a broadcast mask
#               [B,N,N,1] (mask is 0/1 float).  The mask is applied along the
#               last dimension, i.e. each block of `H` consecutive elements
#               shares the same mask value.
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*   (input tensor)
    mask_ptr,         # const float*   (mask tensor, shape B*N*N)
    dst_ptr,          # float*         (output tensor)
    total_el,         # i64            total number of elements = B*N*N*H
    H,                # i64            hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    # linear index for this thread
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    # out‑of‑bounds guard (required for the last partial block)
    mask = offs < total_el

    # Load source value
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # Each contiguous `H` elements correspond to one mask entry.
    # Integer division by H yields the flattened mask index.
    mask_idx = offs // H
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Write the masked result
    tl.store(dst_ptr + offs, src * mask_val, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Apply ``mask`` of shape [B,N,N] to ``tensor`` of shape [B,N,N,H] using
    Triton.  Returns a new tensor with the same shape as ``tensor``.
    """
    B, N, _, H = tensor.shape
    total_el = B * N * N * H
    BLOCK = 1024                     # good occupancy on Hopper/H200

    # Flatten to 1‑D for the kernel
    src_flat = tensor.view(-1)
    dst_flat = torch.empty_like(src_flat)
    mask_flat = mask.view(-1)        # [B,N,N]

    grid = (triton.cdiv(total_el, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr=src_flat,
        mask_ptr=mask_flat,
        dst_ptr=dst_flat,
        total_el=total_el,
        H=H,
        BLOCK=BLOCK,
    )
    return dst_flat.view_as(tensor)


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32 0/1)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )                                             # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections → [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (the part executed in Triton)
    # ------------------------------------------------------------------
    if not nomask:
        left = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise mask of the projected tensors
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate          # [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·C) reduction via einsum (maps to many GEMMs)
    #       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    out = torch.einsum("b i k d, b j k d -> b i j d", left, right)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm (hidden_dim), gating and output projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                               # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure float32 output (all ops use float32 already)
    return out.to(torch.float32)
