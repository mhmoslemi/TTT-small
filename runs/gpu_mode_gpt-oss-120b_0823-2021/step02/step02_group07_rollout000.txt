"""
custom_kernel – “outgoing” TriMul forward pass (AlphaFold3)

Algorithm
---------
1️⃣  Layer‑Norm on the 4‑D tensor (B,N,N,C) – fp32, manual to avoid extra kernels.  
2️⃣  Linear projections   (C → H) – fp32, bias‑free.  
3️⃣  Linear gates + sigmoid – fp32.  
4️⃣  Fuse mask, left‑gate and right‑gate into the projected tensors with a tiny
    Triton kernel (in‑place, fp32).  
5️⃣  Cast the fused tensors to fp16 and reshape to (B*H, N, N).  
6️⃣  Batched GEMM (torch.bmm) performs the O(N³) contraction
    `out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h]`
    using Tensor‑Cores.  
7️⃣  Cast back to fp32 and restore the original layout.  
8️⃣  Output LayerNorm + output‑gate (fp32).  
9️⃣  Final linear projection back to C (fp32).  

Only the element‑wise fusion (step 4) lives in Triton; the dominant cubic
contraction is delegated to cuBLAS via `torch.bmm`, which is the fastest
approach on Hopper (sm_90).  The implementation works for both masked and
unmasked inputs, any batch size ≤ 2, and the dimensionalities required by the
test suite.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask and the two gates into the projected tensors
# ----------------------------------------------------------------------
@triton.jit
def _fuse_mask_gates_kernel(
    left_ptr,         # fp32 [B,N,N,H]
    right_ptr,        # fp32 [B,N,N,H]
    mask_ptr,         # fp32 [B,N,N]   (broadcasted mask)
    lgate_ptr,        # fp32 [B,N,N,H]
    rgate_ptr,        # fp32 [B,N,N,H]
    B, N, H,          # compile‑time constants
    # strides in *elements* (not bytes)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total

    # ----- decode linear offset → (b,i,j,h) ---------------------------
    b = offs // (N * N * H)
    rest = offs % (N * N * H)
    i = rest // (N * H)
    rest = rest % (N * H)
    j = rest // H
    h = rest % H

    # ----- compute element pointers ------------------------------------
    left_off  = b * stride_left_b  + i * stride_left_i  + j * stride_left_j  + h * stride_left_h
    right_off = b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h
    mask_off  = b * stride_mask_b  + i * stride_mask_i  + j * stride_mask_j
    lgate_off = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h
    rgate_off = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h

    # ----- load ---------------------------------------------------------
    left  = tl.load(left_ptr  + left_off,  mask=valid, other=0.0)
    right = tl.load(right_ptr + right_off, mask=valid, other=0.0)
    m     = tl.load(mask_ptr  + mask_off,  mask=valid, other=0.0)   # scalar mask per (b,i,j)
    lgate = tl.load(lgate_ptr + lgate_off, mask=valid, other=0.0)
    rgate = tl.load(rgate_ptr + rgate_off, mask=valid, other=0.0)

    # ----- fuse ---------------------------------------------------------
    left  = left  * m * lgate
    right = right * m * rgate

    # ----- store back (in‑place) ----------------------------------------
    tl.store(left_ptr  + left_off,  left,  mask=valid)
    tl.store(right_ptr + right_off, right, mask=valid)


def _fuse_mask_and_gates(left: torch.Tensor,
                         right: torch.Tensor,
                         mask: torch.Tensor,
                         left_gate: torch.Tensor,
                         right_gate: torch.Tensor) -> None:
    """
    In‑place element‑wise fusion:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    All tensors are fp32 and have the shapes required by the kernel.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 2048                                   # ~2 k threads → good occupancy
    grid = (triton.cdiv(total, BLOCK),)

    # strides in *elements* (not bytes)
    s_left = left.stride()
    s_right = right.stride()
    s_mask = mask.stride()
    s_lgate = left_gate.stride()
    s_rgate = right_gate.stride()

    _fuse_mask_gates_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=s_left[0], stride_left_i=s_left[1],
        stride_left_j=s_left[2], stride_left_h=s_left[3],
        stride_right_b=s_right[0], stride_right_i=s_right[1],
        stride_right_j=s_right[2], stride_right_h=s_right[3],
        stride_mask_b=s_mask[0], stride_mask_i=s_mask[1],
        stride_mask_j=s_mask[2],
        stride_lgate_b=s_lgate[0], stride_lgate_i=s_lgate[1],
        stride_lgate_j=s_lgate[2], stride_lgate_h=s_lgate[3],
        stride_rgate_b=s_rgate[0], stride_rgate_i=s_rgate[1],
        stride_rgate_j=s_rgate[2], stride_rgate_h=s_rgate[3],
        BLOCK=BLOCK,
    )
    # No return – the operation is in‑place.


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, C], float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (or None)
        - weights_dict : mapping of all linear / layer‑norm parameters
        - config_dict  : {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C], float32, on the same CUDA device.
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]                # C
    hidden = config["hidden_dim"]      # H
    device = x.device
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣  LayerNorm over the channel dimension (manual, fp32)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]      # (C,)
    ln_b = weights["norm.bias"]        # (C,)
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * ln_w + ln_b      # (B,N,N,C)

    # --------------------------------------------------------------
    # 2️⃣  Linear projections (C → H) – bias‑free, fp32
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]   # (H, C)
    right_proj_w = weights["right_proj.weight"] # (H, C)

    left = torch.nn.functional.linear(x_norm, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x_norm, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣  Gates (fp32) + sigmoid
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]   # (H, C)
    right_gate_w = weights["right_gate.weight"] # (H, C)
    out_gate_w = weights["out_gate.weight"]     # (H, C)

    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()   # (B,N,N,H)
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()     # (B,N,N,H)

    # --------------------------------------------------------------
    # 4️⃣  Mask handling – broadcast to channel dim
    # --------------------------------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2], device=device,
                         dtype=x.dtype)
    mask = mask.unsqueeze(-1)            # (B,N,N,1)

    # --------------------------------------------------------------
    # 5️⃣  Fuse mask + gates into left/right (in‑place) via Triton
    # --------------------------------------------------------------
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)
    # left/right are now already multiplied by mask & respective gates

    # --------------------------------------------------------------
    # 6️⃣  Heavy N³ contraction – batched GEMM (fp16, Tensor‑Cores)
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # cast to fp16 for the matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*H, N, N) where each “head” contracts independently
    left_flat = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, N)
    right_flat = right_h.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matrix multiplication: left @ rightᵀ  → (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # back to fp32 and original layout (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 7️⃣  Output LayerNorm (fp32)
    # --------------------------------------------------------------
    to_out_ln_w = weights["to_out_norm.weight"]   # (H,)
    to_out_ln_b = weights["to_out_norm.bias"]     # (H,)
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_ln_w + to_out_ln_b

    # --------------------------------------------------------------
    # 8️⃣  Apply output gate (fp32)
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate

    # --------------------------------------------------------------
    # 9️⃣  Final linear projection back to C (fp32)
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"]          # (C, H)
    out_final = torch.nn.functional.linear(out_norm, to_out_w)  # (B,N,N,C)

    return out_final
