"""
custom_kernel – “outgoing” TriMul forward pass (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑Norm over the channel dimension (fp32, pure PyTorch – cheap).
2️⃣  Linear projections (fp32, bias‑free) → left, right.
3️⃣  Linear gates (fp32) + sigmoid.
4️⃣  **Triton kernel** fuses:
        * mask (optional, broadcasted on the channel dim)
        * left‑gate & right‑gate
        * the two projected tensors
   The kernel writes the fused tensors back in‑place (still fp32).
5️⃣  Cast to fp16 and reshape to (B·H, N, N) so that the cubic contraction becomes a
    batched matrix‑multiply.
6️⃣  `torch.bmm` (Tensor‑Core enabled) computes the Einstein‑sum
        out[..., i, j, d] = Σₖ left[..., i, k, d] * right[..., j, k, d]
7️⃣  Layer‑Norm, output‑gate and final linear (all fp32, PyTorch).

Only step 4 is on‑device; everything else is cheap PyTorch and lets us use the
highly‑optimized cuBLAS batched GEMM for the O(N³) work, which is the bottleneck
on the Hopper‑class H200 GPU.
"""

import torch
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel – fuse mask, left‑gate and right‑gate into the projected tensors
# ------------------------------------------------------------------
@triton.jit
def _fuse_mask_gates_kernel(
    left_ptr,          # [B,N,N,H] fp32
    right_ptr,         # [B,N,N,H] fp32
    mask_ptr,          # [B,N,N,1] fp32   (broadcasted mask)
    left_gate_ptr,     # [B,N,N,H] fp32
    right_gate_ptr,    # [B,N,N,H] fp32
    B, N, H,           # compile‑time constants
    # strides (in elements, i.e. not bytes)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    """In‑place:  left = left * mask * left_gate
                 right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # ----- decode linear offset → (b,i,j,h) -----------------------
    b = offs // (N * N * H)
    rest = offs % (N * N * H)
    i = rest // (N * H)
    rest = rest % (N * H)
    j = rest // H
    h = rest % H

    # ----- pointer arithmetic ------------------------------------
    left_off   = b * stride_left_b   + i * stride_left_i   + j * stride_left_j   + h * stride_left_h
    right_off  = b * stride_right_b  + i * stride_right_i  + j * stride_right_j  + h * stride_right_h
    mask_off   = b * stride_mask_b   + i * stride_mask_i   + j * stride_mask_j                # no h‑dim
    lgate_off  = b * stride_lgate_b  + i * stride_lgate_i  + j * stride_lgate_j  + h * stride_lgate_h
    rgate_off  = b * stride_rgate_b  + i * stride_rgate_i  + j * stride_rgate_j  + h * stride_rgate_h

    # ----- loads -------------------------------------------------
    left   = tl.load(left_ptr   + left_off,   mask=mask,   other=0.0)
    right  = tl.load(right_ptr  + right_off,  mask=mask,   other=0.0)
    m      = tl.load(mask_ptr   + mask_off,   mask=mask,   other=0.0)   # broadcasted 0/1
    lgate  = tl.load(left_gate_ptr + lgate_off, mask=mask, other=0.0)
    rgate  = tl.load(right_gate_ptr+ rgate_off, mask=mask, other=0.0)

    # ----- fuse --------------------------------------------------
    left  = left  * m * lgate
    right = right * m * rgate

    # ----- stores (in‑place) ------------------------------------
    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left, right, mask, left_gate, right_gate):
    """
    Calls the above kernel to fuse mask and both gates into ``left`` and ``right``
    (both tensors are modified in‑place).  All tensors are expected to be
    contiguous ``float32`` on the same CUDA device.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 2048                     # a few thousand threads → good occupancy
    grid = (triton.cdiv(total, BLOCK),)

    s_left = left.stride()
    s_right = right.stride()
    s_mask = mask.stride()
    s_lgate = left_gate.stride()
    s_rgate = right_gate.stride()

    _fuse_mask_gates_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=s_left[0], stride_left_i=s_left[1],
        stride_left_j=s_left[2], stride_left_h=s_left[3],
        stride_right_b=s_right[0], stride_right_i=s_right[1],
        stride_right_j=s_right[2], stride_right_h=s_right[3],
        stride_mask_b=s_mask[0], stride_mask_i=s_mask[1],
        stride_mask_j=s_mask[2],
        stride_lgate_b=s_lgate[0], stride_lgate_i=s_lgate[1],
        stride_lgate_j=s_lgate[2], stride_lgate_h=s_lgate[3],
        stride_rgate_b=s_rgate[0], stride_rgate_i=s_rgate[1],
        stride_rgate_j=s_rgate[2], stride_rgate_h=s_rgate[3],
        BLOCK=BLOCK,
    )
    return left, right


# ------------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ---------- unpack ------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    device = x.device
    eps = 1e-5

    # ---------- 1️⃣ LayerNorm (FP32) ----------------------------------
    ln_w = weights["norm.weight"]      # (dim,)
    ln_b = weights["norm.bias"]        # (dim,)
    # mean/var over channel dimension
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * ln_w + ln_b               # (B,N,N,dim)

    # ---------- 2️⃣ Linear projections --------------------------------
    left_proj_w = weights["left_proj.weight"]   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"] # (hidden, dim)

    left = torch.nn.functional.linear(x_norm, left_proj_w)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x_norm, right_proj_w)

    # ---------- 3️⃣ Gates + sigmoid -----------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()   # (B,N,N,hidden)
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()     # (B,N,N,hidden)

    # ---------- 4️⃣ Mask handling -------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2],
                          device=device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)                # (B,N,N,1)

    # ---------- 5️⃣ Fuse mask + gates (Triton) -----------------------
    # All tensors are contiguous float32; the kernel works in‑place.
    left, right = _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ---------- 6️⃣ Batched GEMM (N³ contraction) --------------------
    B, N, _, H = left.shape                      # H == hidden
    # Cast to fp16 for Tensor‑Core speed‑up
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # (B, H, N, N) → (B*H, N, N)
    left_flat  = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right_h.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # out_flat  : (B*H, N, N) = left_flat @ right_flatᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # back to fp32 and original layout (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ---------- 7️⃣ Output LayerNorm (FP32) ---------------------------
    to_out_ln_w = weights["to_out_norm.weight"]   # (hidden,)
    to_out_ln_b = weights["to_out_norm.bias"]     # (hidden,)

    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_ln_w + to_out_ln_b

    # ---------- 8️⃣ Apply output gate --------------------------------
    out_norm = out_norm * out_gate

    # ---------- 9️⃣ Final linear (FP32) -------------------------------
    to_out_w = weights["to_out.weight"]           # (dim, hidden)
    out_final = torch.nn.functional.linear(out_norm, to_out_w)   # (B,N,N,dim)

    return out_final
```

**Explanation of the performance choices**

* The only compute‑intensive part, the element‑wise multiplication by mask and
  gates, is done with a single Triton kernel (`_fuse_mask_gates_kernel`).  This
  eliminates three separate PyTorch kernels and avoids extra memory traffic.

* The cubic contraction is reduced to a batched matrix‑multiply (`torch.bmm`);
  on Hopper GPUs this invokes cuBLAS‑TensorCore kernels that are far faster than
  any hand‑written Triton GEMM for the large `N` values in the test suite.

* All Layer‑Norms and linear layers stay in FP32 to preserve numerical stability
  (the reference implementation does the same) while the heavy GEMM runs in
  FP16 for maximal throughput.

* The kernel works for both masked and un‑masked inputs; if `mask` is `None` we
  synthesize an all‑ones mask.

* The code respects the required launch convention, uses element‑wise strides
  taken from the tensors, and works for all configurations listed in the
  specification.
