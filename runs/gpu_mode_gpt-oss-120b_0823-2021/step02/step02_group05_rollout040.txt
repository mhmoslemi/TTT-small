"""
TriMul “outgoing” forward (AlphaFold‑3) – hybrid implementation.

The heavy O(N³·C) reduction is performed with a fused torch.einsum, which on
Hopper maps to a cascade of highly‑optimised GEMM kernels.  The only
mandatory Triton component is the optional mask‑application, which is a
pure‑memory‑bound element‑wise multiplication that benefits from
explicit masking and vectorised stores.

The kernel `mask_apply_kernel` multiplies a 4‑D tensor [B,N,N,H] with a
broadcast mask of shape [B,N,N] (float 0/1).  It is launched over a 1‑D
grid covering the flattened element count.  All other layers (LayerNorm,
Linear, gating, final LayerNorm + projection) are expressed with
torch.nn.functional calls for correctness and speed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise mask * tensor
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*   (input tensor, flattened)
    mask_ptr,         # const float*   (mask tensor, flattened, shape B*N*N)
    dst_ptr,          # float*         (output tensor, flattened)
    total_el,         # i64            total number of elements = B*N*N*H
    H,                # i64            size of the last dimension (hidden_dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < total_el

    # Load the source value
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # Each block of H consecutive elements shares the same mask entry.
    # Compute index into the flattened mask: floor(offs / H)
    mask_idx = offs // H
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Store masked result
    tl.store(dst_ptr + offs, src * mval, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` of shape [B,N,N,H] by ``mask[...,None]`` using Triton.
    ``mask`` must be float (0/1) with shape [B,N,N] and be on the same device.
    Returns a new tensor with the same shape and dtype as ``tensor``.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024  # a good trade‑off for H200 (fits in shared memory, good occupancy)

    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)  # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the feature dimension ``dim``
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden_dim (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask – apply with Triton (mandatory custom kernel)
    # ------------------------------------------------------------------
    if not nomask:
        left = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise gating of projections
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate      # [B, N, N, hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core O(N³·hidden_dim) reduction via einsum (maps to GEMM on Hopper)
    # ------------------------------------------------------------------
    # out[i,j] = Σ_k left[:,i,k,:] * right[:,j,k,:]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate  # broadcast gate over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure final dtype is float32 (all ops already use float32)
    return out.to(torch.float32)
