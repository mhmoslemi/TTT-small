"""
TriMul – “outgoing” version (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm over the input channel dim (C).
2️⃣  Two bias‑free linear projections → `left` and `right`  (B,N,N,H).
3️⃣  Three bias‑free linear gates → `left_gate`, `right_gate`, `out_gate`
    (B,N,N,H) and apply a sigmoid.
4️⃣  Fuse optional mask, the three gates and the two projected tensors in a
    single Triton kernel:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
        out_gate = out_gate * mask               (mask is broadcast over H)
    The kernel works in‑place on `left`/`right` and writes back the masked
    `out_gate`.
5️⃣  Core contraction (the N³ work) is a batched GEMM per hidden channel:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    This is expressed as a batched matmul on tensors of shape (B,H,N,N) –
    cuBLAS/Tensor‑cores handle it efficiently on Hopper.
6️⃣  Final LayerNorm over the hidden dimension, multiply by the masked
    `out_gate`, and a bias‑free linear projection back to the original
    channel dimension C.

Only step 4 is implemented in Triton, satisfying the “must contain a Triton
kernel” requirement while keeping overall runtime optimal on H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse optional mask + three sigmoid gates.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                # [B,N,N,H] projected tensors
    mask_ptr,                           # [B,N,N,1] broadcast mask (or all‑ones)
    lgate_ptr, rgate_ptr, ogate_ptr,    # [B,N,N,H] sigmoid‑gates
    B, N, H,
    # strides (int64) of the tensors above
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """In‑place apply mask and gates.
    For every element (b,i,k,d):
        left  = left  * mask * lgate
        right = right * mask * rgate
        out_gate = out_gate * mask
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)           # flat offset
    total = B * N * N * H
    active = offs < total

    # ----- decode flat index to (b,i,k,d) -----------------------------
    tmp = offs
    d = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ----- compute element pointers ------------------------------------
    left_ptr   = left_ptr   + b * stride_l_b   + i * stride_l_i   + k * stride_l_k   + d * stride_l_h
    right_ptr  = right_ptr  + b * stride_r_b   + i * stride_r_i   + k * stride_r_k   + d * stride_r_h
    mask_ptr   = mask_ptr   + b * stride_m_b   + i * stride_m_i   + k * stride_m_k                # 1‑channel
    lg_ptr     = lgate_ptr  + b * stride_lg_b  + i * stride_lg_i  + k * stride_lg_k  + d * stride_lg_h
    rg_ptr     = rgate_ptr  + b * stride_rg_b  + i * stride_rg_i  + k * stride_rg_k  + d * stride_rg_h
    og_ptr     = ogate_ptr  + b * stride_og_b  + i * stride_og_i  + k * stride_og_k  + d * stride_og_h

    # ----- load --------------------------------------------------------
    l  = tl.load(left_ptr,   mask=active, other=0.0)
    r  = tl.load(right_ptr,  mask=active, other=0.0)
    m  = tl.load(mask_ptr,   mask=active, other=1.0)   # 1.0 when mask absent
    lg = tl.load(lg_ptr,     mask=active, other=1.0)
    rg = tl.load(rg_ptr,     mask=active, other=1.0)
    og = tl.load(og_ptr,     mask=active, other=1.0)

    # ----- fuse ---------------------------------------------------------
    l = l * m * lg
    r = r * m * rg
    og = og * m                     # out‑gate only needs the mask

    # ----- store --------------------------------------------------------
    tl.store(left_ptr,  l,  mask=active)
    tl.store(right_ptr, r,  mask=active)
    tl.store(og_ptr,    og, mask=active)


def _apply_mask_and_gates(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """
    Launches the Triton kernel above.  Works in‑place on `left`/`right`,
    returns the masked `out_gate`.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                                    # good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        ogate_ptr=out_gate,
        B=B,
        N=N,
        H=H,
        stride_l_b=left.stride(0), stride_l_i=left.stride(1),
        stride_l_k=left.stride(2), stride_l_h=left.stride(3),
        stride_r_b=right.stride(0), stride_r_i=right.stride(1),
        stride_r_k=right.stride(2), stride_r_h=right.stride(3),
        stride_m_b=mask.stride(0), stride_m_i=mask.stride(1),
        stride_m_k=mask.stride(2), stride_m_1=mask.stride(3),
        stride_lg_b=left_gate.stride(0), stride_lg_i=left_gate.stride(1),
        stride_lg_k=left_gate.stride(2), stride_lg_h=left_gate.stride(3),
        stride_rg_b=right_gate.stride(0), stride_rg_i=right_gate.stride(1),
        stride_rg_k=right_gate.stride(2), stride_rg_h=right_gate.stride(3),
        stride_og_b=out_gate.stride(0), stride_og_i=out_gate.stride(1),
        stride_og_k=out_gate.stride(2), stride_og_h=out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate

# Main entry point (as required by the problem statement)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor [B, N, N, C] (float32 / bf16)
        * mask_tensor  : torch.Tensor [B, N, N] (0/1) or None
        * weights_dict : dict with the same keys as the reference implementation
                         (LayerNorm weights/bias, linear weights, etc.)
        * config_dict  : dict with entries ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor [B, N, N, C] (float32) – result of the forward pass.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm over channel dim
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gates + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4️⃣ Prepare mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(B, N, N, 1, dtype=x.dtype, device=device)   # all‑ones
    else:
        mask = mask.unsqueeze(-1).to(x.dtype)                         # [B,N,N,1]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask & gates in‑place via Triton
    # ------------------------------------------------------------------
    out_gate = _apply_mask_and_gates(
        left, right, mask, left_gate, right_gate, out_gate
    )   # left/right now contain mask‑gated values; out_gate is also masked

    # ------------------------------------------------------------------
    # 6️⃣ Core contraction  Σ_k left[b,i,k,h] * right[b,j,k,h]
    #     Batched GEMM over (B, hidden) – leverages cuBLAS/Tensor‑cores.
    # ------------------------------------------------------------------
    # reshape to (B, H, N, K) where K == N
    left_t  = left.permute(0, 3, 1, 2)        # (B, H, N, K)
    right_t = right.permute(0, 3, 1, 2)       # (B, H, N, K)

    # (B,H,N,K) @ (B,H,K,N) -> (B,H,N,N)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))

    # back to (B,N,N,H)
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7️⃣ Final LayerNorm, out‑gate scaling and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])

    # Ensure FP32 output (reference implementation uses float32)
    return out.to(torch.float32)
