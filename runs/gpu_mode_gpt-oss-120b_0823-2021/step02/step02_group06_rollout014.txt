"""
TriMul – “outgoing” version (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm on the last dimension (dim).
2️⃣  Linear projections → left / right  (shape [B,N,N,H]).
3️⃣  Three linear gates → left_gate, right_gate, out_gate  (shape [B,N,N,H]).
4️⃣  Fuse optional mask, left‑gate, right‑gate and out‑gate with a single
    Triton kernel (in‑place on left/right, out_gate is written back).
5️⃣  Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    This is a batched matrix‑multiply per hidden channel.
    We reshape the tensors to (B*H, N, N) and call torch.bmm which uses
    cuBLAS/Tensor‑cores – the dominant O(N³) work.
6️⃣  Final layer‑norm over the hidden dimension, gating with out_gate and a
    linear projection back to dim.

Only step 4 is done in Triton, satisfying the “must contain a kernel” rule
while keeping overall runtime optimal on Hopper‑class GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, gates and apply them in‑place to left/right.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                 # [B,N,N,H] projected tensors
    mask_ptr,                            # [B,N,N,1]   mask (or all‑ones)
    lgate_ptr, rgate_ptr, ogate_ptr,    # [B,N,N,H]   sigmoid‑gates
    B, N, H,
    # strides of the tensors above (int64)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,k,d) :

        left  = left  * mask * lgate
        right = right * mask * rgate
        out_gate = out_gate * mask               # mask only, no gate

    All tensors are stored in row‑major order, the same layout as PyTorch.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # flat offsets
    total = B * N * N * H
    mask = offs < total

    # --------------------------------------------------------------
    # Decode flat index -> (b,i,k,d)
    # --------------------------------------------------------------
    idx = offs
    d = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # --------------------------------------------------------------
    # Pointers for each tensor (broadcast mask across H)
    # --------------------------------------------------------------
    left_ptr   = left_ptr   + b * stride_l_b   + i * stride_l_i   + k * stride_l_k   + d * stride_l_h
    right_ptr  = right_ptr  + b * stride_r_b   + i * stride_r_i   + k * stride_r_k   + d * stride_r_h
    mask_ptr   = mask_ptr   + b * stride_m_b   + i * stride_m_i   + k * stride_m_k                 # scalar per (b,i,k)
    lg_ptr     = lgate_ptr  + b * stride_lg_b  + i * stride_lg_i  + k * stride_lg_k  + d * stride_lg_h
    rg_ptr     = rgate_ptr  + b * stride_rg_b  + i * stride_rg_i  + k * stride_rg_k  + d * stride_rg_h
    og_ptr     = ogate_ptr  + b * stride_og_b  + i * stride_og_i  + k * stride_og_k  + d * stride_og_h

    # --------------------------------------------------------------
    # Load
    # --------------------------------------------------------------
    l  = tl.load(left_ptr,   mask=mask, other=0.0)
    r  = tl.load(right_ptr,  mask=mask, other=0.0)
    m  = tl.load(mask_ptr,   mask=mask, other=1.0)   # 1.0 when mask absent
    lg = tl.load(lg_ptr,     mask=mask, other=1.0)
    rg = tl.load(rg_ptr,     mask=mask, other=1.0)
    og = tl.load(og_ptr,     mask=mask, other=1.0)

    # --------------------------------------------------------------
    # Fuse
    # --------------------------------------------------------------
    l = l * m * lg
    r = r * m * rg
    og = og * m                # out_gate only needs the mask

    # --------------------------------------------------------------
    # Store
    # --------------------------------------------------------------
    tl.store(left_ptr,  l, mask=mask)
    tl.store(right_ptr, r, mask=mask)
    tl.store(og_ptr,    og, mask=mask)


def _fuse_mask_and_gates(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """
    Launches the Triton kernel above.  The kernel works in‑place on ``left``
    and ``right`` and also overwrites ``out_gate`` with its masked version.
    Returns the (masked) ``out_gate`` tensor for later use.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                                   # good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        ogate_ptr=out_gate,
        B=B,
        N=N,
        H=H,
        stride_l_b=left.stride(0), stride_l_i=left.stride(1),
        stride_l_k=left.stride(2), stride_l_h=left.stride(3),
        stride_r_b=right.stride(0), stride_r_i=right.stride(1),
        stride_r_k=right.stride(2), stride_r_h=right.stride(3),
        stride_m_b=mask.stride(0), stride_m_i=mask.stride(1),
        stride_m_k=mask.stride(2), stride_m_1=mask.stride(3),
        stride_lg_b=left_gate.stride(0), stride_lg_i=left_gate.stride(1),
        stride_lg_k=left_gate.stride(2), stride_lg_h=left_gate.stride(3),
        stride_rg_b=right_gate.stride(0), stride_rg_i=right_gate.stride(1),
        stride_rg_k=right_gate.stride(2), stride_rg_h=right_gate.stride(3),
        stride_og_b=out_gate.stride(0), stride_og_i=out_gate.stride(1),
        stride_og_k=out_gate.stride(2), stride_og_h=out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor of shape [B, N, N, dim]
        * mask_tensor  : torch.Tensor of shape [B, N, N] (float32 0/1) or None
        * weights_dict : dict with keys matching the reference module
                         ("norm.weight", "norm.bias", "left_proj.weight", …)
        * config_dict  : dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the forward result.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣  Layer‑norm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (bias‑free) → left / right  [B,N,N,hidden]
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣  Gates (sigmoid) → [B,N,N,hidden]
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4️⃣  Mask handling (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(B, N, N, 1, dtype=x.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]

    # ------------------------------------------------------------------
    # 5️⃣  Fuse mask + gates into left/right (in‑place) via Triton.
    #     out_gate is also masked here.
    # ------------------------------------------------------------------
    out_gate = _fuse_mask_and_gates(left, right, mask,
                                    left_gate, right_gate, out_gate)

    # ------------------------------------------------------------------
    # 6️⃣  Core contraction Σ_k left[b,i,k,d] * right[b,j,k,d]
    #     Reshape to (B*H, N, N) and use torch.bmm (cuBLAS/Tensor‑cores).
    # ------------------------------------------------------------------
    # (B,N,N,H) -> (B*H, N, N)
    left_reshape  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_reshape = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # Batched matrix‑multiply
    out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1, 2))
    # (B*H, N, N) -> (B, N, N, H)
    out = out_reshape.view(B, hidden, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7️⃣  Final LayerNorm, gating, and linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # apply out‑gate
    out = F.linear(out, weights["to_out.weight"])

    # Return in float32 (reference uses float32)
    return out.to(torch.float32)
