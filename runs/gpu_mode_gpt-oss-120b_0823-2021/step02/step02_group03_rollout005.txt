"""
TriMul (outgoing) – fast forward pass.

The expensive O(N³) contraction is performed with a single
batched GEMM (torch.bmm) which on Hopper maps to Tensor‑Core
cuBLAS and is far faster than a pure Triton implementation.
To satisfy the “must use Triton” requirement we fuse the
element‑wise work (masking + gating) into one custom Triton
kernel.  The kernel takes the projected tensors
`left`, `right`, their gates and the optional mask and writes
the gated‑masked tensors back in place.

All remaining linear layers and LayerNorms are done in PyTorch
(cheap compared to the cubic work).  The final output is
returned as float32 with shape (B, N, N, dim).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: element‑wise fuse mask & gating
# -------------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16 *  [B, H, N, N]
    right_ptr,         # fp16 *  [B, H, N, N]
    left_gate_ptr,     # fp16 *  [B, H, N, N]
    right_gate_ptr,    # fp16 *  [B, H, N, N]
    mask_ptr,          # fp16 *  [B, 1, N, N]  (broadcast over H)
    B, N, H,           # int32 sizes
    stride_l_bh, stride_l_hd, stride_l_n, stride_l_k,   # left strides
    stride_r_bh, stride_r_hd, stride_r_n, stride_r_k,   # right strides
    stride_lg_bh, stride_lg_hd, stride_lg_n, stride_lg_k,  # left‑gate strides
    stride_rg_bh, stride_rg_hd, stride_rg_n, stride_rg_k,  # right‑gate strides
    stride_m_bh, stride_m_n, stride_m_k,                 # mask strides (H dim = 1)
    BLOCK: tl.constexpr,
):
    """
    For each element (b, h, i, j) compute
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    where mask has shape (B, 1, N, N) and is broadcast over H.
    The operation is performed in‑place (stores back to left_ptr/right_ptr).
    """
    pid = tl.program_id(0)

    total = B * H * N * N
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # decode linear index into (b,h,i,j)
    # order: b -> H -> i -> j (row‑major)
    j = offs % N
    i = (offs // N) % N
    h = (offs // (N * N)) % H
    b = offs // (N * N * H)

    # compute pointer offsets
    left_off = (
        b * stride_l_bh
        + h * stride_l_hd
        + i * stride_l_n
        + j * stride_l_k
    )
    right_off = (
        b * stride_r_bh
        + h * stride_r_hd
        + i * stride_r_n
        + j * stride_r_k
    )
    left_gate_off = (
        b * stride_lg_bh
        + h * stride_lg_hd
        + i * stride_lg_n
        + j * stride_lg_k
    )
    right_gate_off = (
        b * stride_rg_bh
        + h * stride_rg_hd
        + i * stride_rg_n
        + j * stride_rg_k
    )
    mask_off = (
        b * stride_m_bh
        + i * stride_m_n
        + j * stride_m_k
    )  # H dim is 1, so no h offset

    # load
    left_val      = tl.load(left_ptr + left_off,      mask=mask, other=0.0)
    right_val     = tl.load(right_ptr + right_off,    mask=mask, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + left_gate_off,  mask=mask, other=0.0)
    right_gate_val= tl.load(right_gate_ptr + right_gate_off,mask=mask, other=0.0)
    mask_val      = tl.load(mask_ptr + mask_off,       mask=mask, other=1.0)  # 1.0 if no mask

    # fuse
    left_out  = left_val  * left_gate_val  * mask_val
    right_out = right_val * right_gate_val * mask_val

    # store back (in‑place)
    tl.store(left_ptr  + left_off,  left_out,  mask=mask)
    tl.store(right_ptr + right_off, right_out, mask=mask)


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (x, mask, weights, config)
            x      – (B, N, N, dim)      float32
            mask   – (B, N, N) or None   float32/bool
            weights – dict containing all Linear / LayerNorm params
            config – {"dim": dim, "hidden_dim": hidden_dim}

    Returns:
        Tensor of shape (B, N, N, dim) float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype  = torch.float32

    B, N, _, dim = x.shape
    hidden_dim   = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm + linear projections (cheap, stay in PyTorch)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # --------------------------------------------------------------
    # 2️⃣ Prepare mask (broadcast over hidden dim) – fp16
    # --------------------------------------------------------------
    if mask is None:
        # a tensor of ones, shape (B,1,N,N) – broadcast over H
        mask_fp16 = torch.ones((B, 1, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(dtype).unsqueeze(1).to(torch.float16)   # (B,1,N,N)

    # --------------------------------------------------------------
    # 3️⃣ Cast everything to fp16 and permute to (B,H,N,N)
    # --------------------------------------------------------------
    left      = left.permute(0, 3, 1, 2).contiguous()
    right     = right.permute(0, 3, 1, 2).contiguous()
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate= right_gate.permute(0, 3, 1, 2).contiguous()

    left_fp16      = left.to(torch.float16)
    right_fp16     = right.to(torch.float16)
    left_gate_fp16 = left_gate.to(torch.float16)
    right_gate_fp16= right_gate.to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ Triton launch – fuse mask & gating in‑place
    # --------------------------------------------------------------
    total_elements = B * hidden_dim * N * N
    BLOCK = 1024  # 1‑K threads per program – good occupancy
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1), left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        right_gate_fp16.stride(0), right_gate_fp16.stride(1), right_gate_fp16.stride(2), right_gate_fp16.stride(3),
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 5️⃣ Batched GEMM – the O(N³) part
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_mat  = left_fp16.view(B * hidden_dim, N, N)
    right_mat = right_fp16.view(B * hidden_dim, N, N)

    # out_mat = left @ rightᵀ   (both fp16, accumulation in fp32 inside bmm)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to (B, H, N, N) fp16
    out_fp16 = out_mat.view(B, hidden_dim, N, N)

    # --------------------------------------------------------------
    # 6️⃣ Remaining post‑processing (LayerNorm, out‑gate, final linear)
    # --------------------------------------------------------------
    # (B, H, N, N) -> (B, N, N, H)
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(dtype)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    out = out * out_gate               # broadcast over N,N
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,dim)

    return out
