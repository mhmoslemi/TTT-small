"""
TriMul (outgoing) – fast forward pass
------------------------------------

The expensive part of the TriMul forward pass is the contraction

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

i.e. a batched matrix‑matrix multiply for every (batch, hidden) slice:
    (N,N) @ (N,N)^T → (N,N)

We keep the cheap LayerNorm / Linear / gating work in PyTorch (they are
memory‑bound and negligible compared with the O(N³) work) and implement the
batched GEMM in Triton.  The kernel works on a single (b, h) tile and
tiles the i‑ and j‑dimensions (M×N) while looping over k in blocks.  All
loads are in FP16, the accumulation is FP32 for numerical stability and the
result is written back as FP16.  The host wrapper reshapes the tensors to
the layout expected by the kernel, launches it, then finishes the
post‑processing (LayerNorm, gating, final linear) in PyTorch.

The implementation follows the “standard” tiling strategy that fits easily
into the ~64 KB shared‑memory budget of an H200 and gives high occupancy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM for a single hidden slice
# -------------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # const fp16  [B, H, N, N]
    right_ptr,         # const fp16  [B, H, N, N]
    out_ptr,           #      fp16  [B, H, N, N]
    B, N, H,           # int32 scalars
    # strides for left  (B,H,N,N)
    stride_lbh, stride_lhd, stride_lhn, stride_lhk,
    # strides for right (B,H,N,N)
    stride_rbh, stride_rhd, stride_rhn, stride_rhk,
    # strides for out   (B,H,N,N)
    stride_obh, stride_obd, stride_obn, stride_obk,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
    for a single (b, h) pair.  The tile covers
        i ∈ [block_i*BM, block_i*BM+BM)
        j ∈ [block_j*BN, block_j*BN+BN)
    """
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1) decode pid → (b, h, block_i, block_j)
    # --------------------------------------------------------------
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)

    block_j = pid % num_blocks_j
    pid //= num_blocks_j
    block_i = pid % num_blocks_i
    pid //= num_blocks_i
    h = pid % H
    b = pid // H

    # --------------------------------------------------------------
    # 2) compute the concrete offsets for i/j
    # --------------------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) loop over k dimension in tiles
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, h, i, k]  (M×K)
        l_ptr = (
            left_ptr
            + b * stride_lbh
            + h * stride_lhd
            + offs_i[:, None] * stride_lhn
            + offs_k[None, :] * stride_lhk
        )
        l = tl.load(l_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[b, h, j, k] needs to be transposed → (K×N)
        r_ptr = (
            right_ptr
            + b * stride_rbh
            + h * stride_rhd
            + offs_k[:, None] * stride_rhk
            + offs_j[None, :] * stride_rhn
        )
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # dot‑product over K
        acc += tl.dot(l, r)          # (M,K) @ (K,N) → (M,N)

    # --------------------------------------------------------------
    # 4) write back the result (FP16)
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_obh
        + h * stride_obd
        + offs_i[:, None] * stride_obn
        + offs_j[None, :] * stride_obk
    )
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper exposing the same API as the PyTorch reference
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config) where
        * x          – input tensor (B, N, N, dim)  dtype torch.float32
        * mask       – (B, N, N)  bool/float32 or None
        * weights    – dict with the same keys as the reference module
        * config     – {"dim": dim, "hidden_dim": hidden_dim}

    Returns
    -------
    torch.Tensor
        (B, N, N, dim)  dtype torch.float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) LayerNorm + Linear projections (cheap – keep in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 2) Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)          # (B,N,N,1)
        left  = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 3) Gating (sigmoid) – also cheap
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4) Prepare tensors for the Triton kernel
    #    layout: (B, H, N, N)  and cast to FP16
    # -----------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)        # (B,H,N,N) fp16

    # -----------------------------------------------------------------
    # 5) Triton launch
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # total program ids = B * H * ceil(N/BM) * ceil(N/BN)
    num_i = triton.cdiv(N, BLOCK_M)
    num_j = triton.cdiv(N, BLOCK_N)
    total_progs = B * hidden_dim * num_i * num_j
    grid = (total_progs,)

    _trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        out_fp16.stride(0), out_fp16.stride(1),
        out_fp16.stride(2), out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6) Post‑processing (LayerNorm, gating, final linear) – back to (B,N,N,hidden)
    # -----------------------------------------------------------------
    out = out_fp16.to(dtype).permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    out = out * out_gate                     # broadcast over N,N
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,dim)

    return out
