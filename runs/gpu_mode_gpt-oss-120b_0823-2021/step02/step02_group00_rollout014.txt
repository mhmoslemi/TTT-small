We implement the *outgoing* version of the TriMul operator (AlphaFold‑3).  
The forward pass consists of:

1. **Layer‑norm** of the input tensor (FP32, handcrafted for speed).  
2. **Linear projections** (`left_proj`, `right_proj`) and **gates** (`left_gate`, `right_gate`, `out_gate`) – everything cast to **float16**.  
3. **Optional mask** applied to the projected tensors.  
4. **Element‑wise gating** (`sigmoid`).  
5. **Main contraction**  

\[
\text{out}[b,i,j,d]=\sum_{k} \text{left}[b,i,k,d]\;\text{right}[b,j,k,d]
\]

   This is a batched matrix‑multiply of shape  
   `(B, H, I, K)·(B, H, K, J) → (B, H, I, J)` where `H = hidden_dim`.  
   We launch a custom Triton kernel that performs a tiled GEMM
   (`I × K` by `K × J`).  
   - Tile sizes: `BLOCK_M = BLOCK_N = 64`, `BLOCK_K = 32` (good for Hopper‑90).  
   - Accumulate in **fp32**, store in **fp16**.  
   - One program per `(batch * hidden_dim)` slab.

6. **Second Layer‑norm**, **output‑gate**, and final **linear projection** back to the original `dim` (FP32).

Only the heavy O(N³) contraction is executed in Triton; everything else stays in PyTorch for readability and correctness.

```python
"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Fast forward pass:
1️⃣ LayerNorm on input (FP32, manual).
2️⃣ Linear projections & gates → FP16.
3️⃣ Optional mask.
4️⃣ Gating (sigmoid).
5️⃣ Core contraction Σ_k left[b,i,k,d] * right[b,j,k,d] performed by a
   tiled Triton GEMM (batch‑matmul) in FP16 (accumulate FP32).
6️⃣ Second LayerNorm, output‑gate, final linear projection (FP32).

The Triton kernel processes each (batch, hidden_dim) slab as an independent
matrix‑multiply (N×N)·(N×N) → (N×N).  Tile sizes are tuned for Hopper/SM‑90.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_matmul_kernel(
    a_ptr,            # left  tensor   (B*H, I, K)
    b_ptr,            # right tensor   (B*H, K, J)
    c_ptr,            # output tensor  (B*H, I, J)
    M, N, K,          # matrix extents
    stride_az, stride_am, stride_ak,   # a strides (batch, i, k)
    stride_bz, stride_bk, stride_bn,   # b strides (batch, k, j)
    stride_cz, stride_cm, stride_cn,   # c strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile along I
    pid_n = tl.program_id(1)          # tile along J
    pid_z = tl.program_id(2)          # batch * hidden_dim index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M×K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (K×N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16×FP16 → FP32 accumulation
        acc += tl.dot(a, b)

    # Write result (back to FP16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left : (B, H, I, K)   FP16, contiguous
    right: (B, H, K, J)   FP16, contiguous
    returns (B, H, I, J)  FP16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # Output allocation
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # Collapse batch*head for the kernel
    batch = B * H
    a = left.reshape(batch, I, K)
    b = right.reshape(batch, K, J)
    c = out.reshape(batch, I, J)

    # Strides in elements
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes – tuned for Hopper (SM‑90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),          # rows
        triton.cdiv(J, BLOCK_N),          # cols
        batch,                            # batch*head
    )
    _batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # good for FP16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask         : torch.Tensor or None, shape [B, N, N] (float or bool)
        - weights      : dict of leaf weight tensors
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32, manual)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    # mean / var over last dim (=dim)
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + 1e-5)
    x_norm = x_norm * ln_w + ln_b          # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gating (cast to FP16)
    # --------------------------------------------------------------
    def _fp16(t): return t.to(torch.float16)

    left_proj_w   = _fp16(weights["left_proj.weight"])
    right_proj_w  = _fp16(weights["right_proj.weight"])
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    x_fp16 = x_norm.to(torch.float16)      # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction with Triton
    # --------------------------------------------------------------
    # reorder to (B, H, I, K) for left and (B, H, K, J) for right
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, J)

    out_fp16 = _batched_matmul_fp16(left_t, right_t)   # (B, H, I, J)

    # back to (B, N, N, dim) and FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP32)
    # --------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    # --------------------------------------------------------------
    # 7️⃣ Output gate + final linear projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over dim
    out = F.linear(out, to_out_w)                   # (B,N,N,dim)

    return out
```
