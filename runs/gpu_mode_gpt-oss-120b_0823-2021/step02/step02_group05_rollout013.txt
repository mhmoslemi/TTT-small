"""
custom_kernel – “outgoing” TriMul forward (AlphaFold‑3)

* LayerNorm, all linear projections, gating and the final O(N³·C) reduction are
  performed with PyTorch (torch.nn.functional).  On Hopper the einsum
  internally dispatches to a series of highly‑optimised GEMM kernels which is
  far faster than a naïve Triton implementation for the N³ term.

* The only part that is mandatory to run in Triton is the optional pair‑wise
  mask application on the projected tensors (`left` and `right`).  This is a
  pure‑memory‑bound operation that benefits from Triton’s explicit mask handling
  and vectorised stores.

The kernel works for any batch size (1‑2), sequence length up to 1024 and
channel dimensions 128/384.  The final output is always float‑32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# Triton kernel: element‑wise multiplication of a 4‑D tensor
# [B, N, N, H] by a broadcast mask [B, N, N, 1].
# --------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*   (input tensor)
    mask_ptr,         # const float*   (mask tensor, shape B*N*N)
    dst_ptr,          # float*         (output tensor)
    total_el,         # i64            total number of elements = B*N*N*H
    H,                # i64            hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear element ids
    mask = offs < total_el                            # OOB guard

    # Load a block of the source tensor
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # For each block of H consecutive elements the mask index is the same.
    # Integer division is performed via floor division on the integer type.
    mask_idx = offs // H
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Store the masked value
    tl.store(dst_ptr + offs, src * mask_val, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` of shape [B, N, N, H] by ``mask[..., None]`` using Triton.
    ``mask`` is a float tensor containing 0/1 values of shape [B, N, N].
    Returns a new tensor with the same shape as ``tensor``.
    """
    B, N, _, H = tensor.shape
    total_elements = B * N * N * H
    BLOCK = 1024                      # empirically good for H200

    # Flatten to 1‑D views – required because Triton kernels work on
    # contiguous 1‑D memory ranges.
    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)         # [B, N, N]

    grid = (triton.cdiv(total_elements, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total_elements,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# --------------------------------------------------------------
# Entry‑point used by the evaluator.
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections → shape [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (the only Triton‑accelerated part)
    # ------------------------------------------------------------------
    if not nomask:
        left  = apply_mask_triton(left,  mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4️⃣  Gating (sigmoid) and element‑wise gating of projected tensors
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate      # [B, N, N, hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Core O(N³·hidden_dim) reduction.
    #    On Hopper the einsum maps to a sequence of GEMM kernels which is
    #    far more efficient than a hand‑written Triton kernel for the cubic
    #    term.
    # ------------------------------------------------------------------
    # out shape: [B, N, N, hidden_dim]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # 6️⃣  Final LayerNorm, output gating and projection back to ``dim``
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                     # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure float32 output (all ops already use float32 on H200)
    return out.to(torch.float32)
