"""
custom_kernel: “outgoing” TriMul forward (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (B,N,N,D) – done in PyTorch (tiny tensor).
2️⃣  Linear projections `left_proj`, `right_proj` and three gates
    (`left_gate`, `right_gate`, `out_gate`).  All weights are cast to
    fp16 and the activations are kept in fp16 for the heavy contraction.
3️⃣  Optional mask is applied (broadcasted over the hidden dimension).
4️⃣  Gated tensors are multiplied in the N‑dimension:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    The contraction is a batched matrix‑multiply for each head `h`.
    We implement it with a fused Triton kernel that computes
    (B·H, I, K) @ (B·H, K, J) → (B·H, I, J) using fp16 compute,
    fp32 accumulation and fp16 store.
5️⃣  The result is permuted back to (B,N,N,D), cast to fp32 and passed
    through a second Layer‑norm, a final gating, and a linear projection
    – all in PyTorch.

The kernel fuses the three‑dimensional contraction (the O(N³) part) and
remains the only place where Triton is used, which gives a large speed‑up
while keeping the surrounding code simple and numerically stable.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel (fp16 compute, fp32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,                # left   (Z, I, K)   fp16
    b_ptr,                # right  (Z, K, J)   fp16
    c_ptr,                # out    (Z, I, J)   fp16
    M, N, K,              # matrix dimensions
    stride_az, stride_am, stride_ak,   # a strides
    stride_bz, stride_bk, stride_bn,   # b strides
    stride_cz, stride_cm, stride_cn,   # c strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)          # batch*head dimension
    pid_m = tl.program_id(1)          # tile‑row   (I)
    pid_n = tl.program_id(2)          # tile‑col   (J)

    # ------------------------------------------------------------------
    #  tile coordinates
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (better numeric stability for the sum over K)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  reduction over K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)                     # fp16·fp16 → fp32

    # store result as fp16
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for tensors of shape (B, H, I, K) and (B, H, K, J) → (B, H, I, J)

    Both inputs must be contiguous fp16 tensors.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten leading dimensions for the kernel
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides in element units (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling – tuned for Hopper / SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                                      # pid_z
        triton.cdiv(I, BLOCK_M),               # pid_m
        triton.cdiv(J, BLOCK_N),               # pid_n
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor  [B, N, N, dim]   (float32)
        mask         : torch.Tensor  [B, N, N]       (float32/bool) or None
        weights      : dict of torch.Tensor, same keys as the
                       reference PyTorch implementation.
        config       : dict with keys "dim" and "hidden_dim".

    Returns
    -------
    torch.Tensor  [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    H = config["hidden_dim"]          # hidden dimension (called “hidden_dim”)

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm  (keep in fp32 – tensor is small)
    # ------------------------------------------------------------------
    gamma = weights["norm.weight"]
    beta  = weights["norm.bias"]
    mean = x.mean(-1, keepdim=True)
    var  = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * gamma + beta                       # [B,N,N,dim] fp32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy path
    # ------------------------------------------------------------------
    x_fp16 = x_norm.to(torch.float16)

    # linear weights (no bias in the reference)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Projections
    # ------------------------------------------------------------------
    # shape: (B,N,N,H)
    left  = F.linear(x_fp16, left_proj_w)          # fp16
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5. Gates (sigmoid in fp16)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction via Triton (batched GEMM)
    #    left : (B, N, N, H) → (B, H, N, N)   (i,k) order
    #    right: (B, N, N, H) → (B, H, N, N)   (j,k) order -> transpose K,J
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _triton_batched_gemm(left_perm, right_perm)   # (B,H,N,N) fp16

    # back to original layout and fp32 for the rest of the pipeline
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 8. Final gating + linear projection (still fp32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over last dim
    out = F.linear(out, weights["to_out.weight"])
    return out
