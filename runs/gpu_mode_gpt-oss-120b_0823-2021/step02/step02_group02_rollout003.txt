"""
TriMul “outgoing” forward pass (AlphaFold3).

* LayerNorm, linear projections and the final output‑linear are kept in PyTorch
  (they are cheap compared with the N³ contraction).
* The expensive element‑wise work – mask application and the three sigmoidal
  gates – is fused into a single Triton kernel (`trimul_fuse_kernel`).  This
  reduces the number of global memory passes over the 4‑D tensors.
* The N³ contraction itself is reduced to a batched matrix‑multiply:
        out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
  after we transpose the second operand (right) we can call
  `torch.bmm` which on H200 uses Tensor‑cores (FP16) and is far faster than a
  hand‑written tiled kernel.
* All arithmetic inside the kernel is FP16 →FP32 accumulation →FP16 store.
  The final result is cast back to FP32, LayerNorm‑normalised, gated and
  projected to the original `dim`.

The function `custom_kernel` follows the required signature:
    (input, mask, weights, config) → output tensor [B, N, N, dim] (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask & three gates (left, right, out)
# ----------------------------------------------------------------------
@triton.jit
def trimul_fuse_kernel(
    # tensors
    left_ptr,       # float16*  [B, N, N, H]   (projected, before gating)
    right_ptr,      # float16*  [B, N, N, H]
    left_gate_ptr,  # float16*  [B, N, N, H]   (linear → sigmoid)
    right_gate_ptr, # float16*  [B, N, N, H]
    out_gate_ptr,   # float16*  [B, N, N, H]   (same shape as left/right)

    mask_ptr,       # uint8*   [B, N, N]  (0/1 mask)  – may be nullptr
    # sizes
    B, N, H,
    # strides (row‑major) – passed from the host
    stride_left_B, stride_left_N1, stride_left_N2, stride_left_H,
    stride_right_B, stride_right_N1, stride_right_N2, stride_right_H,
    stride_lgate_B, stride_lgate_N1, stride_lgate_N2, stride_lgate_H,
    stride_rgate_B, stride_rgate_N1, stride_rgate_N2, stride_rgate_H,
    stride_ogate_B, stride_ogate_N1, stride_ogate_N2, stride_ogate_H,
    stride_mask_B, stride_mask_N1, stride_mask_N2,
    # compile‑time constants
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h) computes
        left  = left  * mask * sigmoid(left_gate)
        right = right * mask * sigmoid(right_gate)
    and also writes the sigmoid(out_gate) to a separate buffer (used later).
    All tensors are stored as FP16; accumulation is performed in FP32.
    """
    pid = tl.program_id(0)                     # linear thread id over all elements
    total = B * N * N * H
    if pid >= total:
        return

    # decode linear index
    h = pid % H
    pid //= H
    j = pid % N
    pid //= N
    i = pid % N
    b = pid // N

    # offsets
    off_left = (b * stride_left_B +
                i * stride_left_N1 +
                j * stride_left_N2 +
                h * stride_left_H)
    off_right = (b * stride_right_B +
                 i * stride_right_N1 +
                 j * stride_right_N2 +
                 h * stride_right_H)
    off_lgate = (b * stride_lgate_B +
                 i * stride_lgate_N1 +
                 j * stride_lgate_N2 +
                 h * stride_lgate_H)
    off_rgate = (b * stride_rgate_B +
                 i * stride_rgate_N1 +
                 j * stride_rgate_N2 +
                 h * stride_rgate_H)
    off_ogate = (b * stride_ogate_B +
                 i * stride_ogate_N1 +
                 j * stride_ogate_N2 +
                 h * stride_ogate_H)

    # load values (FP16 → FP32)
    left   = tl.load(left_ptr + off_left).to(tl.float32)
    right  = tl.load(right_ptr + off_right).to(tl.float32)
    l_gate = tl.load(left_gate_ptr + off_lgate).to(tl.float32)
    r_gate = tl.load(right_gate_ptr + off_rgate).to(tl.float32)
    o_gate = tl.load(out_gate_ptr + off_ogate).to(tl.float32)

    # optional mask (uint8 → 0/1 float)
    if HAS_MASK:
        off_mask = (b * stride_mask_B +
                    i * stride_mask_N1 +
                    j * stride_mask_N2)
        m = tl.load(mask_ptr + off_mask).to(tl.float32)   # 0.0 or 1.0
        left  *= m
        right *= m

    # sigmoid (approx via built‑in)
    l_gate = tl.sigmoid(l_gate)
    r_gate = tl.sigmoid(r_gate)
    o_gate = tl.sigmoid(o_gate)

    # apply gates
    left  = left  * l_gate
    right = right * r_gate

    # store back (FP32 → FP16)
    tl.store(left_ptr  + off_left , left .to(tl.float16))
    tl.store(right_ptr + off_right, right.to(tl.float16))
    tl.store(out_gate_ptr + off_ogate, o_gate.to(tl.float16))   # keep for later


# ----------------------------------------------------------------------
# Host wrapper
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ----------
    data :
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask         : [B, N, N]        (uint8 or bool) or None
        - weights      : dict of all linear / norm parameters
        - config       : {'dim': int, 'hidden_dim': int}

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension (called hidden_dim in the paper)

    # ------------------------------------------------------------------
    # 1. LayerNorm + first linear projections (PyTorch)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(input_tensor.dtype),
        bias=weights["norm.bias"].to(input_tensor.dtype),
    )   # (B,N,N,dim)

    # Linear projections → (B,N,N,H)   (bias=False)
    left = F.linear(x, weights["left_proj.weight"].to(x.dtype))
    right = F.linear(x, weights["right_proj.weight"].to(x.dtype))

    # linear layers for the three gates (still FP32)
    left_gate  = F.linear(x, weights["left_gate.weight"].to(x.dtype))
    right_gate = F.linear(x, weights["right_gate.weight"].to(x.dtype))
    out_gate   = F.linear(x, weights["out_gate.weight"].to(x.dtype))

    # ------------------------------------------------------------------
    # 2. Cast everything to FP16 for the kernel (memory‑bandwidth bound)
    # ------------------------------------------------------------------
    left_fp16       = left.to(torch.float16).contiguous()
    right_fp16      = right.to(torch.float16).contiguous()
    left_gate_fp16  = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()
    out_gate_fp16   = out_gate.to(torch.float16).contiguous()

    if mask is not None:
        mask_uint8 = (mask > 0).to(torch.uint8).contiguous()
    else:
        mask_uint8 = torch.empty(0, device=device, dtype=torch.uint8)   # dummy

    # ------------------------------------------------------------------
    # 3. Launch Triton kernel – fuse mask & sigmoid gates
    # ------------------------------------------------------------------
    BLOCK = 128                                     # one thread per element
    total_elements = B * N * N * H
    grid = (triton.cdiv(total_elements, BLOCK),)

    trimul_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        out_gate_fp16,
        mask_uint8,
        B, N, H,
        # strides for left
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        # strides for right
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        # strides for left_gate
        left_gate_fp16.stride(0), left_gate_fp16.stride(1), left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        # strides for right_gate
        right_gate_fp16.stride(0), right_gate_fp16.stride(1), right_gate_fp16.stride(2), right_gate_fp16.stride(3),
        # strides for out_gate (stores sigmoid(out_gate))
        out_gate_fp16.stride(0), out_gate_fp16.stride(1), out_gate_fp16.stride(2), out_gate_fp16.stride(3),
        # mask strides
        mask_uint8.stride(0) if mask is not None else 0,
        mask_uint8.stride(1) if mask is not None else 0,
        mask_uint8.stride(2) if mask is not None else 0,
        HAS_MASK=mask is not None,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 4. Rearrange to (B, H, N, N) for the batched GEMM
    # ------------------------------------------------------------------
    # (B,N,N,H) → (B,H,N,N)
    left_bhnn  = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_bhnn = right_fp16.permute(0, 3, 1, 2).contiguous()
    out_gate_bhnn = out_gate_fp16.permute(0, 3, 1, 2).contiguous()   # sigmoid(out_gate)

    # cast to FP16 (already) and reshape for torch.bmm
    left_flat  = left_bhnn.view(B * H, N, N)          # (B*H, N, N)
    right_flat = right_bhnn.view(B * H, N, N).transpose(1, 2)   # (B*H, N, N) with K dimension transposed

    # ------------------------------------------------------------------
    # 5. Batched matrix multiply (Tensor‑core fast path)
    # ------------------------------------------------------------------
    out_flat = torch.bmm(left_flat, right_flat)      # (B*H, N, N) in FP16
    out_bhnn = out_flat.view(B, H, N, N)             # back to 4‑D

    # ------------------------------------------------------------------
    # 6. Back‑to original layout and post‑processing (PyTorch)
    # ------------------------------------------------------------------
    # (B,H,N,N) → (B,N,N,H)
    out = out_bhnn.permute(0, 2, 3, 1).contiguous().to(input_tensor.dtype)   # (B,N,N,H) FP32

    # final LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(input_tensor.dtype),
        bias=weights["to_out_norm.bias"].to(input_tensor.dtype),
    )

    # apply output gate (sigmoid already computed in kernel)
    out_gate = out_gate_bhnn.permute(0, 2, 3, 1).contiguous().to(input_tensor.dtype)
    out = out * out_gate

    # linear back to original dim
    out = F.linear(out, weights["to_out.weight"].to(input_tensor.dtype))

    return out
