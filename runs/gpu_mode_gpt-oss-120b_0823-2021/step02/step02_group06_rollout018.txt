"""
custom_kernel – Triton‑accelerated forward of the outgoing TriMul operator.

Algorithm
---------
1. Layer‑norm over the last dimension (dim)          – PyTorch (fast).
2. Linear projections (left/right)                  – PyTorch.
3. Compute three gate tensors and apply sigmoid   – PyTorch.
4. Optional mask (broadcast) is fused together with the
   three gates inside a single Triton kernel.  The kernel
   works in‑place on `left` and `right` and also writes the
   masked `out_gate`.
5. Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix‑multiply per hidden channel:
        (B, H, N, K) @ (B, H, N, K)ᵀ → (B, H, N, N)
   performed with `torch.matmul` (cuBLAS/Tensor‑cores).
6. Final LayerNorm, out‑gate scaling and linear projection – PyTorch.

Only step‑4 is implemented in Triton; the rest uses highly‑optimized
PyTorch kernels, which is the best trade‑off for the H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask + three sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # [B,N,N,H]
    right_ptr,         # [B,N,N,H]
    mask_ptr,          # [B,N,N,1]  (or all‑ones)
    lgate_ptr,         # [B,N,N,H]
    rgate_ptr,         # [B,N,N,H]
    ogate_ptr,         # [B,N,N,H]
    B, N, H,
    # strides (int64) of each tensor
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,k,d):
        left  = left  * mask * lgate
        right = right * mask * rgate
        out_gate = out_gate * mask
    The mask is scalar per (b,i,k) and is broadcast over the hidden dim.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # flat offset into left/right
    total = B * N * N * H
    mask = offs < total

    # ------------------- decode flat index -------------------
    tmp = offs
    d = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ------------------- compute pointers -------------------
    left_ptr   = left_ptr   + b * stride_l_b   + i * stride_l_i   + k * stride_l_k   + d * stride_l_h
    right_ptr  = right_ptr  + b * stride_r_b   + i * stride_r_i   + k * stride_r_k   + d * stride_r_h
    mask_ptr   = mask_ptr   + b * stride_m_b   + i * stride_m_i   + k * stride_m_k                # 1‑channel
    lg_ptr     = lgate_ptr  + b * stride_lg_b  + i * stride_lg_i  + k * stride_lg_k  + d * stride_lg_h
    rg_ptr     = rgate_ptr  + b * stride_rg_b  + i * stride_rg_i  + k * stride_rg_k  + d * stride_rg_h
    og_ptr     = ogate_ptr  + b * stride_og_b  + i * stride_og_i  + k * stride_og_k  + d * stride_og_h

    # ------------------- load -------------------------------------------------
    l  = tl.load(left_ptr,   mask=mask, other=0.0)
    r  = tl.load(right_ptr,  mask=mask, other=0.0)
    m  = tl.load(mask_ptr,   mask=mask, other=1.0)   # 1.0 when mask absent
    lg = tl.load(lg_ptr,     mask=mask, other=1.0)
    rg = tl.load(rg_ptr,     mask=mask, other=1.0)
    og = tl.load(og_ptr,     mask=mask, other=1.0)

    # ------------------- fuse --------------------------------------------------
    l = l * m * lg
    r = r * m * rg
    og = og * m                 # out‑gate only needs mask

    # ------------------- store -------------------------------------------------
    tl.store(left_ptr,  l, mask=mask)
    tl.store(right_ptr, r, mask=mask)
    tl.store(og_ptr,    og, mask=mask)


def _apply_mask_and_gates(left, right, mask, lgate, rgate, out_gate):
    """
    Launches the Triton kernel above. All tensors are on the same device.
    Returns the (already masked) out_gate tensor.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                                 # good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=lgate,
        rgate_ptr=rgate,
        ogate_ptr=out_gate,
        B=B, N=N, H=H,
        stride_l_b=left.stride(0), stride_l_i=left.stride(1),
        stride_l_k=left.stride(2), stride_l_h=left.stride(3),
        stride_r_b=right.stride(0), stride_r_i=right.stride(1),
        stride_r_k=right.stride(2), stride_r_h=right.stride(3),
        stride_m_b=mask.stride(0), stride_m_i=mask.stride(1),
        stride_m_k=mask.stride(2), stride_m_1=mask.stride(3),
        stride_lg_b=lgate.stride(0), stride_lg_i=lgate.stride(1),
        stride_lg_k=lgate.stride(2), stride_lg_h=lgate.stride(3),
        stride_rg_b=rgate.stride(0), stride_rg_i=rgate.stride(1),
        stride_rg_k=rgate.stride(2), stride_rg_h=rgate.stride(3),
        stride_og_b=out_gate.stride(0), stride_og_i=out_gate.stride(1),
        stride_og_k=out_gate.stride(2), stride_og_h=out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor [B, N, N, dim] (float32/bf16)
        * mask_tensor  : torch.Tensor [B, N, N] (float32 0/1) or None
        * weights_dict : dict with keys exactly as in the reference PyTorch
                         implementation (norm.weight, norm.bias, left_proj.weight,
                         right_proj.weight, left_gate.weight, right_gate.weight,
                         out_gate.weight, to_out_norm.weight,
                         to_out_norm.bias, to_out.weight)
        * config_dict  : {'dim': int, 'hidden_dim': int}
    Returns
    -------
    torch.Tensor
        Tensor [B, N, N, dim] – forward result (float32).
    """
    # ------------------- unpack -------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------- 1️⃣ LayerNorm over dim -------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------- 2️⃣ Linear projections (no bias) -------------------
    # shape -> [B, N, N, hidden]
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------- 3️⃣ Gates (sigmoid) -------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------- 4️⃣ Prepare mask -------------------
    if mask is None:
        # broadcast mask of all ones
        mask = torch.ones(B, N, N, 1, dtype=x.dtype, device=device)
    else:
        # mask shape -> [B, N, N, 1] for broadcasting
        mask = mask.unsqueeze(-1).to(x.dtype)

    # ------------------- 5️⃣ Fuse mask + gates with Triton -------------------
    out_gate = _apply_mask_and_gates(
        left, right, mask, left_gate, right_gate, out_gate
    )   # left/right now hold masked * gated values; out_gate is masked

    # ------------------- 6️⃣ Core contraction (batched GEMM) -------------------
    # reshape to (B, hidden, N, N) → (B, hidden, N, K) where K == N
    left_t  = left.permute(0, 3, 1, 2)   # (B, H, N, K)
    right_t = right.permute(0, 3, 1, 2)  # (B, H, N, K)

    # (B,H,N,K) @ (B,H,K,N)  -> (B,H,N,N)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))

    # back to original layout (B,N,N,hidden)
    out = out_t.permute(0, 2, 3, 1)

    # ------------------- 7️⃣ Final LayerNorm, out‑gate scaling, projection -------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # apply out‑gate
    out = F.linear(out, weights["to_out.weight"])

    # ensure float32 output as in reference
    return out.to(torch.float32)
