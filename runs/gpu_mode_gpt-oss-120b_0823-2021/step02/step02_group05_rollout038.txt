"""
TriMul “outgoing” forward (AlphaFold‑3) – CUDA + Triton implementation.

The heavy O(N³·C) reduction is performed with :func:`torch.einsum`, which on
Hopper maps to a sequence of highly‑optimised GEMM kernels.  The only
mandatory Triton portion is the *pair‑wise mask* application; this is
memory‑bound and benefits from Triton’s explicit masking and contiguous
vectorised stores while keeping the rest of the pipeline in PyTorch for
best overall performance.

The kernel works for any batch size (1‑2), sequence length up to 1024,
and hidden dimensions that are multiples of 32 (the typical values used
by AlphaFold‑3).  All tensors are expected on the CUDA device; the final
output is float‑32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: multiply a [B,N,N,H] tensor by a broadcast mask [B,N,N]
# (mask values are 0/1 float).  The kernel processes the flattened view
# and uses integer division to locate the corresponding mask entry.
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,       # const float*  – input tensor flattened
    mask_ptr,      # const float*  – mask flattened (B*N*N)
    dst_ptr,       # float*        – output tensor flattened
    total_el,      # i64           – total elements = B*N*N*H
    hidden_dim,    # i64           – size of the last dimension (H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < total_el

    # Load the source element
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # One mask element per hidden_dim consecutive entries
    mask_idx = offs // hidden_dim                     # index into B*N*N mask
    m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Store the masked value
    tl.store(dst_ptr + offs, src * m_val, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Apply ``mask[..., None]`` to ``tensor`` using a Triton kernel.
    The operation is equivalent to ``tensor * mask.unsqueeze(-1)`` but avoids
    the overhead of the PyTorch broadcast for large tensors.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                     # good occupancy on H200

    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)        # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        hidden_dim=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry‑point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (dim)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections → hidden_dim
    # --------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask – Triton‑accelerated
    # --------------------------------------------------------------
    if not nomask:
        left  = apply_mask_triton(left,  mask)
        right = apply_mask_triton(right, mask)

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise gating of projected tensors
    # --------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate   # [B, N, N, hidden_dim]
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·C) reduction – torch.einsum (maps to GEMM on H200)
    # --------------------------------------------------------------
    # out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)   # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gating and projection back to dim
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                     # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # torch.nn.functional.linear keeps the dtype; ensure float32 output
    return out.to(torch.float32)
