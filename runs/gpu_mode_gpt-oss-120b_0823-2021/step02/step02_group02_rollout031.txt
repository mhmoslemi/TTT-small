Below is a self‑contained implementation of the **outgoing** `TriMul` operator.  
Only the cubic‑complex contraction  

```
out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
```

is executed in Triton – everything else (layer‑norms, linear layers, masking,
gating, the final “to‑out” projection) stays in regular PyTorch because they are
trivial compared with the `N³` work.

The kernel is a classic **batched GEMM** with the extra hidden dimension `H`
(= `hidden_dim`).  We treat the data as a 4‑D tensor `[B, H, N, N]` in row‑major
order and compute a block of the result `[M, N]` at a time while iterating over
the reduction dimension `K` in tiles.  The implementation uses:

* **FP16 arithmetic** for the loads and the dot product (Tensor‑core friendly).  
* **FP32 accumulation** (`tl.float32`) to avoid precision loss.  
* **Tiling parameters** `BLOCK_M = 64`, `BLOCK_N = 64`, `BLOCK_K = 32` – they fit
  comfortably in the shared‑memory budget of an H200 and give good occupancy.

All strides are passed explicitly from the host so that the kernel works for
any memory layout (`contiguous` or not).  The kernel does **not** compute any
statistics, so no synchronization or `tl.mean` is required.

The host‑side wrapper

```python
custom_kernel(data)
```

* unpacks the tuple `(input, mask, weights, config)`,
* performs the LayerNorms, linear projections and gating in PyTorch,
* reshapes the tensors to `[B, H, N, N]` and casts them to `float16`,
* launches the Triton kernel,
* reshapes the result back to `[B, N, N, dim]`,
* applies the final LayerNorm, output‑gate and linear projection.

The wrapper returns a **float‑32** tensor exactly matching the reference
implementation.

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -------------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# -------------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,            # const fp16*  [B, H, N, N]
    right_ptr,           # const fp16*  [B, H, N, N]
    out_ptr,             #       fp16*  [B, H, N, N]
    B, N, H,             # int sizes
    stride_lbh, stride_lhh, stride_lhn, stride_lhk,   # left strides
    stride_rbh, stride_rhh, stride_rhn, stride_rhk,   # right strides
    stride_obh, stride_ohh, stride_ohn, stride_ohk,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a tile of
        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    left/right/out are stored as [B, H, N, N] (row‑major).  The kernel processes
    one (b, h) pair and a (BLOCK_M × BLOCK_N) tile of the (i, j) matrix.
    Accumulation is performed in FP32, the final result is stored as FP16.
    """
    pid = tl.program_id(0)                     # linear id over (b, h, tile_m, tile_n)

    # -------------------------------------------------
    # Decode pid -> (b, h, tile_m, tile_n)
    # -------------------------------------------------
    num_h = H
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    tile_n = pid % num_n
    pid //= num_n
    tile_m = pid % num_m
    pid //= num_m
    h = pid % num_h
    b = pid // num_h

    # -------------------------------------------------
    # Offsets for this tile
    # -------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Loop over K dimension in BLOCK_K chunks
    # -------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, h, i, k] => (M, K)
        l_ptr = (left_ptr
                 + b * stride_lbh
                 + h * stride_lhh
                 + offs_m[:, None] * stride_lhn
                 + offs_k[None, :] * stride_lhk)
        l = tl.load(l_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[b, h, j, k] is accessed as (K, N) because we need the transpose
        r_ptr = (right_ptr
                 + b * stride_rbh
                 + h * stride_rhh
                 + offs_k[:, None] * stride_rhk
                 + offs_n[None, :] * stride_rhn)
        r_t = tl.load(r_ptr,
                      mask=mask_k[:, None] & mask_n[None, :],
                      other=0.0).to(tl.float32)

        # (M, K) @ (K, N) -> (M, N)
        acc += tl.dot(l, r_t)

    # -------------------------------------------------
    # Write the tile back (FP16)
    # -------------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_obh
                + h * stride_ohh
                + offs_m[:, None] * stride_ohn
                + offs_n[None, :] * stride_ohk)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask         : [B, N, N]       (bool / float)
        - weights      : dict of model parameters (float32)
        - config       : dict with "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. LayerNorm + Linear projections (PyTorch, FP32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 2. Optional mask (broadcast over the hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 3. Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Reshape to [B, H, N, N] and cast to fp16 for the kernel
    # -----------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)   # shape [B, H, N, N], fp16

    # -----------------------------------------------------------------
    # 5. Launch Triton kernel
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # grid dimensions: one program per (b, h, tile_m, tile_n)
    grid = (
        B * hidden_dim * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),
    )

    trimul_outgoing_kernel[grid](
        # pointers
        left_fp16,
        right_fp16,
        out_fp16,
        # sizes
        B,
        N,
        hidden_dim,
        # strides (row‑major)
        left_fp16.stride(0),   # stride over B
        left_fp16.stride(1),   # stride over H
        left_fp16.stride(2),   # stride over N (i dimension)
        left_fp16.stride(3),   # stride over N (k dimension)
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Convert back to original layout, final LN, gate and linear
    # -----------------------------------------------------------------
    # out_fp16: [B, H, N, N] -> [B, N, N, H]
    out = out_fp16.to(dtype).permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden)

    # to_out_norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply output gate (broadcast over N,N)
    out = out * out_gate

    # final linear back to dim
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
```
