"""
TriMul – outgoing version (AlphaFold‑3)

Algorithm
---------
1️⃣ Layer‑norm over the feature dimension (dim) – PyTorch (fast, already
    vectorised).
2️⃣ Linear projections (bias‑free) → left, right   (B,N,N,hidden) – PyTorch.
3️⃣ Gate pre‑activations (left_gate, right_gate, out_gate) – PyTorch.
4️⃣ Fuse mask + sigmoid(gates) into left/right (in‑place) and sigmoid
   out_gate with a single Triton kernel.  This is the *must‑have* Triton
   part.
5️⃣ Core contraction (out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]).
   The N³ work is delegated to cuBLAS via a batched GEMM on
   (B, hidden, N, N) tensors – the fastest path on Hopper.
6️⃣ Final Layer‑norm, out‑gate scaling and linear projection back to dim –
   PyTorch.

The kernel only touches element‑wise ops, keeping register pressure low and
allowing the huge matrix‑multiply to run at full Tensor‑core throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – mask + three sigmoid gates → left, right (in‑place) &
# out_gate (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr, right_ptr,                    # [B,N,N,H]   float32
    mask_ptr,                               # [B,N,N,1]   float32 (broadcasted)
    lgate_ptr, rgate_ptr, ogate_ptr,       # [B,N,N,H]   float32 (pre‑sigmoid)
    B, N, H,
    # strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k, stride_mask_1,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    stride_ogate_b, stride_ogate_i, stride_ogate_k, stride_ogate_h,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,k,d) do
        left  = left  * mask * sigmoid(lgate)
        right = right * mask * sigmoid(rgate)
        ogate = sigmoid(ogate)          (stored back in‑place)
    mask is a scalar (0/1) per (b,i,k) and is broadcast over d.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear index
    total = B * N * N * H
    mask = offs < total

    # decode linear index → (b,i,k,d)
    idx = offs
    d = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # compute pointers
    left   = left_ptr   + b * stride_left_b   + i * stride_left_i   + k * stride_left_k   + d * stride_left_h
    right  = right_ptr  + b * stride_right_b  + i * stride_right_i  + k * stride_right_k  + d * stride_right_h
    mask_p = mask_ptr   + b * stride_mask_b   + i * stride_mask_i   + k * stride_mask_k   # no hidden stride
    lgate  = lgate_ptr  + b * stride_lgate_b  + i * stride_lgate_i  + k * stride_lgate_k  + d * stride_lgate_h
    rgate  = rgate_ptr  + b * stride_rgate_b  + i * stride_rgate_i  + k * stride_rgate_k  + d * stride_rgate_h
    ogate  = ogate_ptr  + b * stride_ogate_b  + i * stride_ogate_i  + k * stride_ogate_k  + d * stride_ogate_h

    # load values
    l = tl.load(left,  mask=mask, other=0.0)
    r = tl.load(right, mask=mask, other=0.0)
    m = tl.load(mask_p, mask=mask, other=1.0)          # 1.0 → no‑mask, 0.0 → masked‑out
    lg = tl.load(lgate, mask=mask, other=0.0)
    rg = tl.load(rgate, mask=mask, other=0.0)
    og = tl.load(ogate, mask=mask, other=0.0)

    # sigmoid – use the exact expression (fast enough for inference)
    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))
    og = 1.0 / (1.0 + tl.exp(-og))

    # fuse
    l = l * m * lg
    r = r * m * rg

    # store back
    tl.store(left,  l, mask=mask)
    tl.store(right, r, mask=mask)
    tl.store(ogate, og, mask=mask)


def _apply_mask_gate(left, right, mask, lgate, rgate, ogate):
    """
    Launches the Triton kernel defined above. All tensors are
    contiguous, shape (B,N,N,H) except ``mask`` which is (B,N,N,1).
    The function returns ``ogate`` after it has been sigmoid‑ed in‑place.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                     # good trade‑off on H200
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fusion_kernel[grid](
        left, right,
        mask,
        lgate, rgate, ogate,
        B, N, H,
        # strides
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2), mask.stride(3),
        lgate.stride(0), lgate.stride(1), lgate.stride(2), lgate.stride(3),
        rgate.stride(0), rgate.stride(1), rgate.stride(2), rgate.stride(3),
        ogate.stride(0), ogate.stride(1), ogate.stride(2), ogate.stride(3),
        BLOCK=BLOCK,
    )
    return ogate


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, dim] (float32 / bf16)
        - mask_tensor  : torch.Tensor [B, N, N] or None
        - weights_dict : keys
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 0️⃣ Input LayerNorm (over dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 1️⃣ Linear projections (bias‑free)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])    # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 2️⃣ Gate pre‑activations (still linear, no bias)
    # --------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"])
    right_gate = F.linear(x, weights["right_gate.weight"])
    out_gate   = F.linear(x, weights["out_gate.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Prepare mask (broadcast to the hidden dim)
    # --------------------------------------------------------------
    if mask is None:
        mask4d = torch.ones((B, N, N, 1), dtype=x.dtype, device=device)
    else:
        mask4d = mask.unsqueeze(-1).to(x.dtype)        # (B,N,N,1)

    # --------------------------------------------------------------
    # 4️⃣ Fuse mask + sigmoid(gates) with Triton (in‑place)
    # --------------------------------------------------------------
    out_gate = _apply_mask_gate(left, right, mask4d, left_gate, right_gate, out_gate)

    # --------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM (cuBLAS/Tensor‑cores)
    #     out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # reshape to (B, hidden, N, K) where K == N
    left_t  = left.permute(0, 3, 1, 2)          # (B, hidden, N, N)
    right_t = right.permute(0, 3, 1, 2)         # (B, hidden, N, N)

    # batched matmul: (B, hidden, N, N) @ (B, hidden, N, N)^T
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))   # (B, hidden, N, N)

    # back to original layout (B,N,N,hidden)
    out = out_t.permute(0, 2, 3, 1)

    # --------------------------------------------------------------
    # 6️⃣ Final LayerNorm → out‑gate scaling → linear projection back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                       # element‑wise gate
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    # --------------------------------------------------------------
    # Ensure output dtype matches specification (float32)
    # --------------------------------------------------------------
    return out.to(torch.float32)
