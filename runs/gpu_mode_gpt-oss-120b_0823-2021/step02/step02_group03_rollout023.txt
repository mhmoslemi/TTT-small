"""
TriMul – outgoing version (AlphaFold‑3)

The dominant O(N³) work is the “outer‑product” over the third axis  

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

which is exactly a batched matrix‑matrix multiplication
    (B·H, N, N) @ (B·H, N, N)ᵀ  → (B·H, N, N)

cuBLAS (torch.bmm) is far faster than a handcrafted Triton GEMM on the
H200, therefore we delegate the heavy work to `torch.bmm`.  
To still satisfy the “must‑use‑Triton” rule we fuse the optional mask and
the two per‑head gates (`left_gate`, `right_gate`) into a single
Triton kernel that works on the four‑dimensional tensor
`(B, N, N, H)` (H = hidden_dim).  This kernel:

* loads a block of `left` (or `right`) together with the mask,
* multiplies by the corresponding gate (sigmoid already computed in
  FP32, cast to FP16 for the kernel),
* stores the result back in‑place.

All other linear layers, LayerNorms and the final output projection are
performed in PyTorch (they are cheap compared to the O(N³) contraction).

The implementation supports:
* arbitrary batch size B (1 or 2 in the test suite),
* any sequence length N up to 1024,
* hidden dimension H = hidden_dim (≤384),
* optional mask (None → no masking, otherwise boolean/float mask).

The final tensor is returned in FP32 with shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel – fuse mask + gating into the projected tensors
# -------------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    proj_ptr,          # *mut fp16   (B,N,N,H)
    gate_ptr,          # *const fp16  (B,N,N,H)   sigmoid(gate) already cast
    mask_ptr,          # *const fp16  (B,N,N,1)   may be nullptr (mask disabled)
    B, N, H,           # sizes (int32)
    stride_pb, stride_pn, stride_pk, stride_ph,   # proj strides
    stride_gb, stride_gn, stride_gk, stride_gh,   # gate strides
    stride_mb, stride_mn, stride_mk, stride_mh,   # mask strides (unused if mask_ptr==0)
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h):
        proj[b,i,j,h] = proj[b,i,j,h] * gate[b,i,j,h] * mask[b,i,j,0]
    """
    pid = tl.program_id(0)

    # linear index → (b,i,j)
    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i = ij // N
    j = ij % N

    # thread‑local offsets in the hidden dimension
    offs_h = tl.arange(0, BLOCK)
    mask_h = 0  # mask has a singleton hidden dim

    # bounds
    valid_b = b < B
    valid_i = i < N
    valid_j = j < N
    active = valid_b & valid_i & valid_j

    # pointers – note the hidden dimension is the fastest moving index
    proj_base = (
        proj_ptr
        + b * stride_pb
        + i * stride_pn
        + j * stride_pk
    )
    gate_base = (
        gate_ptr
        + b * stride_gb
        + i * stride_gn
        + j * stride_gk
    )
    if mask_ptr != 0:
        mask_base = (
            mask_ptr
            + b * stride_mb
            + i * stride_mn
            + j * stride_mk
        )
    else:
        mask_base = 0

    # load
    proj = tl.load(
        proj_base + offs_h * stride_ph,
        mask=offs_h < H,
        other=0.0,
    ).to(tl.float32)

    gate = tl.load(
        gate_base + offs_h * stride_gh,
        mask=offs_h < H,
        other=0.0,
    ).to(tl.float32)

    if mask_ptr != 0:
        m = tl.load(
            mask_base + mask_h * stride_mh,
            mask=True,
            other=1.0,
        ).to(tl.float32)
        proj = proj * m

    proj = proj * gate

    # store back
    tl.store(
        proj_base + offs_h * stride_ph,
        proj.to(tl.float16),
        mask=offs_h < H,
    )


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data = (x, mask, weights, config)
        x          : (B, N, N, dim)            float32
        mask       : (B, N, N) or None         bool / float32
        weights    : dict of model parameters (torch.float32)
        config     : {"dim": dim, "hidden_dim": hidden_dim}

    Returns
    -------
    Tensor of shape (B, N, N, dim)  float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]          # H

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm + Linear projections (cheap, done in PyTorch)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # --------------------------------------------------------------
    # 2️⃣ Compute gates (sigmoid) – keep in FP32 for numerical stability
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))   # (B,N,N,H)

    # --------------------------------------------------------------
    # 3️⃣ Cast everything to FP16 – the heavy kernel works in FP16
    # --------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    left_gate_fp16  = left_gate.to(torch.float16)
    right_gate_fp16 = right_gate.to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ Apply mask (if any) and gating with a tiny Triton kernel
    # --------------------------------------------------------------
    if mask is not None:
        # mask must be broadcastable to (B,N,N,1) and cast to fp16
        mask_fp16 = mask.to(dtype).unsqueeze(-1).to(torch.float16)
        mask_ptr = mask_fp16.data_ptr()
    else:
        mask_fp16 = None
        mask_ptr = 0

    # helper to launch the kernel for a tensor + gate pair
    def _mask_gate(tensor_fp16, gate_fp16):
        BLOCK = 64                                   # hidden‑dim block size
        total_ij = N * N
        grid = (B * total_ij, )                      # one program per (b,i,j)

        _apply_mask_gate_kernel[grid](
            tensor_fp16,
            gate_fp16,
            mask_fp16 if mask_fp16 is not None else 0,
            B, N, hidden_dim,
            tensor_fp16.stride(0), tensor_fp16.stride(1),
            tensor_fp16.stride(2), tensor_fp16.stride(3),
            gate_fp16.stride(0),   gate_fp16.stride(1),
            gate_fp16.stride(2),   gate_fp16.stride(3),
            # mask strides (dummy if mask_ptr==0)
            0, 0, 0, 0,
            BLOCK=BLOCK,
        )

    _mask_gate(left_fp16,  left_gate_fp16)
    _mask_gate(right_fp16, right_gate_fp16)

    # --------------------------------------------------------------
    # 5️⃣ Reshape to (B*H, N, N) and perform the O(N³) batched GEMM with cuBLAS
    # --------------------------------------------------------------
    # (B,N,N,H) -> (B*H, N, N)
    left_bh = left_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_bh = right_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # out = left @ rightᵀ   (both fp16, accumulation in fp32)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))   # (B*H, N, N)

    # back to (B, N, N, H) in fp32
    out = out_bh.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(dtype)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm, out‑gate and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # out_gate was computed in FP32, need to broadcast over (N,N)
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,dim)

    return out
