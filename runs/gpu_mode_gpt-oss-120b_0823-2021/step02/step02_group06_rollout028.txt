"""
TriMul – “outgoing” version (AlphaFold‑3)

Algorithm (high‑level)
----------------------
1️⃣  Layer‑norm on the input channel dimension (dim).
2️⃣  Two bias‑free linear projections → left / right   [B,N,N,H].
3️⃣  Three bias‑free linear gates → left_gate, right_gate, out_gate  [B,N,N,H].
4️⃣  Fuse (optional) mask, left‑gate, right‑gate and out‑gate in a single
    Triton kernel.  The kernel works **in‑place** on `left` and `right` and
    writes the masked sigmoid back to `out_gate`.  This eliminates extra
    Python launches and keeps all element‑wise work on the GPU.
5️⃣  Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    This is a batched matrix‑multiply per hidden channel.  By reshaping we
    obtain (B*H, N, N) matrices and call the highly‑optimized cuBLAS kernel
    via `torch.bmm`.
6️⃣  Final LayerNorm over the hidden dimension, out‑gate scaling and a linear
    projection back to the original `dim`.

Only step 4 is implemented in Triton – the rest uses PyTorch which already
leverages Tensor‑cores on Hopper (H200).  The implementation works for any
batch size ≤2, sequence length up to 1024 and arbitrary `dim` / `hidden_dim`
as required by the test suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: mask + gate fusion
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                     # [B,N,N,H] projected tensors
    mask_ptr,                                 # [B,N,N,1] (or all‑ones)
    lgate_ptr, rgate_ptr, ogate_ptr,          # [B,N,N,H] sigmoid gates
    B, N, H,
    # strides of the tensors above (int64)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """
    In‑place fusion:
        left  = left  * mask * lgate
        right = right * mask * rgate
        out_gate = out_gate * mask          (sigmoid already applied)
    The mask is a scalar per (b,i,k) and broadcast over the hidden dim.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # flat offsets
    total = B * N * N * H
    mask = offs < total

    # ---------- decode (b,i,k,d) from flat index ----------
    tmp = offs
    d = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ---------- pointer arithmetic ----------
    left_ptr  = left_ptr  + b * stride_l_b  + i * stride_l_i  + k * stride_l_k  + d * stride_l_h
    right_ptr = right_ptr + b * stride_r_b  + i * stride_r_i  + k * stride_r_k  + d * stride_r_h
    mask_ptr  = mask_ptr  + b * stride_m_b  + i * stride_m_i  + k * stride_m_k               # scalar mask
    lg_ptr    = lgate_ptr + b * stride_lg_b + i * stride_lg_i + k * stride_lg_k + d * stride_lg_h
    rg_ptr    = rgate_ptr + b * stride_rg_b + i * stride_rg_i + k * stride_rg_k + d * stride_rg_h
    og_ptr    = ogate_ptr + b * stride_og_b + i * stride_og_i + k * stride_og_k + d * stride_og_h

    # ---------- loads ----------
    l   = tl.load(left_ptr,  mask=mask, other=0.0)
    r   = tl.load(right_ptr, mask=mask, other=0.0)
    m   = tl.load(mask_ptr,  mask=mask, other=1.0)      # 1.0 when no mask
    lg  = tl.load(lg_ptr,    mask=mask, other=1.0)
    rg  = tl.load(rg_ptr,    mask=mask, other=1.0)
    og  = tl.load(og_ptr,    mask=mask, other=1.0)

    # ---------- fuse ----------
    l = l * m * lg
    r = r * m * rg
    og = og * m                     # out‑gate only needs mask

    # ---------- stores ----------
    tl.store(left_ptr,  l, mask=mask)
    tl.store(right_ptr, r, mask=mask)
    tl.store(og_ptr,    og, mask=mask)


def _apply_mask_and_gates(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """
    Launches the Triton kernel above.  Works in‑place on `left`/`right` and
    returns the masked `out_gate`.  All tensors must be contiguous and on the
    same CUDA device.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    # 1024 is a good launch size for Hopper; it fits in a single wave.
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        ogate_ptr=out_gate,
        B=B,
        N=N,
        H=H,
        stride_l_b=left.stride(0), stride_l_i=left.stride(1), stride_l_k=left.stride(2), stride_l_h=left.stride(3),
        stride_r_b=right.stride(0), stride_r_i=right.stride(1), stride_r_k=right.stride(2), stride_r_h=right.stride(3),
        stride_m_b=mask.stride(0), stride_m_i=mask.stride(1), stride_m_k=mask.stride(2), stride_m_1=mask.stride(3),
        stride_lg_b=left_gate.stride(0), stride_lg_i=left_gate.stride(1), stride_lg_k=left_gate.stride(2), stride_lg_h=left_gate.stride(3),
        stride_rg_b=right_gate.stride(0), stride_rg_i=right_gate.stride(1), stride_rg_k=right_gate.stride(2), stride_rg_h=right_gate.stride(3),
        stride_og_b=out_gate.stride(0), stride_og_i=out_gate.stride(1), stride_og_k=out_gate.stride(2), stride_og_h=out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor of shape [B, N, N, dim] (float32 / bf16)
        * mask_tensor  : torch.Tensor of shape [B, N, N] (0/1) or ``None``.
        * weights_dict : dict with the same keys as the reference PyTorch
                         implementation (LayerNorm weights/bias, linear weights,
                         etc.).
        * config_dict  : dict with entries ``dim`` and ``hidden_dim``.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the forward result.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    H = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections → left / right  (bias‑free)
    # --------------------------------------------------------------
    # These are shape [B,N,N,H]
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Gates + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 4️⃣ Prepare mask (broadcast over the hidden dim)
    # --------------------------------------------------------------
    if mask is None:
        # all‑ones mask → shape [B,N,N,1]
        mask = torch.ones(B, N, N, 1, dtype=x.dtype, device=device)
    else:
        # expand last dim for broadcasting
        mask = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]

    # --------------------------------------------------------------
    # 5️⃣ Fuse mask + gates (Triton)
    # --------------------------------------------------------------
    out_gate = _apply_mask_and_gates(
        left, right, mask, left_gate, right_gate, out_gate
    )   # left/right are now masked & gated; out_gate is masked

    # --------------------------------------------------------------
    # 6️⃣ Core N³ contraction via batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) for a single bmm per channel
    left_flat  = left.reshape(B * H, N, N)
    right_flat = right.reshape(B * H, N, N)

    # batched matmul → (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B,N,N,H)
    out = out_flat.reshape(B, H, N, N).permute(0, 2, 3, 1)   # (B,N,N,H)

    # --------------------------------------------------------------
    # 7️⃣ Final LayerNorm, out‑gate scaling and output projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # gated scaling
    out = F.linear(out, weights["to_out.weight"])

    # Ensure float32 output (reference uses float32)
    return out.float()
