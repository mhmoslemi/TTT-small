"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward only)
------------------------
1️⃣ Layer‑norm the 4‑D input (FP32, manual for speed).
2️⃣ Four linear projections (left/right + three gates) are done in FP16.
3️⃣ Optional pair‑wise mask is applied to the left/right tensors.
4️⃣ Gating (sigmoid) is applied element‑wise.
5️⃣ Core O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is executed by a tiny Triton kernel that treats each (batch, dim) pair as an
   independent matrix‑multiply (N×N)·(N×N) → (N×N).  The kernel works in
   FP16, accumulates in FP32 and writes FP16.
6️⃣ Convert back to FP32, apply a second LayerNorm, multiply by the output‑gate,
   and finish with a linear projection back to the original dimension.

The Triton kernel is deliberately simple and tuned for Hopper (SM‑90):
    BLOCK_M = BLOCK_N = 64   # tile size in i/j dimensions
    BLOCK_K = 32             # tile size in the reduction (k) dimension
    num_warps = 4            # good default for FP16 on H200
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched matrix‑multiply (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_matmul_kernel(
    a_ptr,            # left  matrix   (batch, I, K)
    b_ptr,            # right matrix   (batch, K, J)
    c_ptr,            # output matrix  (batch, I, J)
    M, N, K,          # extents
    stride_az, stride_am, stride_ak,   # a strides  (z, i, k)
    stride_bz, stride_bk, stride_bn,   # b strides  (z, k, n)
    stride_cz, stride_cm, stride_cn,   # c strides  (z, i, n)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile along M
    pid_n = tl.program_id(1)          # tile along N
    pid_z = tl.program_id(2)          # batch‑head index

    # start offsets of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (good precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction axis K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile: (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16 × FP16 → FP32 dot
        acc += tl.dot(a, b)

    # write result (cast back to FP16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiply used inside TriMul.

    left  : (B, H, I, K)    FP16, contiguous
    right : (B, H, K, J)    FP16, contiguous
    returns: (B, H, I, J)   FP16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse batch*head dimension for the kernel
    batch = B * H
    a = left.view(batch, I, K)
    b = right.view(batch, K, J)
    c = out.view(batch, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),          # tiles over I
        triton.cdiv(J, BLOCK_N),          # tiles over J
        batch,                            # one program per (B,H)
    )
    _batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                     # fp16‑friendly for H200
    )
    return out


# ----------------------------------------------------------------------
# Main entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask         : torch.Tensor or None, shape [B, N, N] (float or bool)
        - weights      : dict of leaf tensors (see code below)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    device = x.device

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (manual, FP32)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    # mean / var over the last dim
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + 1e-5)
    x_norm = x_norm * ln_w + ln_b                     # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gating (FP16)
    # --------------------------------------------------------------
    def _fp16(t): return t.to(torch.float16)

    # projection weights (hidden_dim, dim)
    w_left  = _fp16(weights["left_proj.weight"])
    w_right = _fp16(weights["right_proj.weight"])
    # gate weights (hidden_dim, dim)
    w_lgate = _fp16(weights["left_gate.weight"])
    w_rgate = _fp16(weights["right_gate.weight"])
    w_ogate = _fp16(weights["out_gate.weight"])

    x16 = x_norm.to(torch.float16)                    # (B,N,N,dim)

    left  = F.linear(x16, w_left)                     # (B,N,N,hidden_dim)
    right = F.linear(x16, w_right)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)      # (B,N,N,1)
        left  = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x16, w_lgate).sigmoid()
    right_gate = F.linear(x16, w_rgate).sigmoid()
    out_gate   = F.linear(x16, w_ogate).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Contraction via Triton batched‑matmul
    #    left  : (B, N, N, hidden)   -> (B, H, I, K)
    #    right : (B, N, N, hidden)   -> (B, H, K, J)
    # --------------------------------------------------------------
    #   H == hidden_dim, I == J == N, K == N
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _triton_batched_matmul(left_t, right_t)   # (B, H, I, J)

    # reshape back to (B, N, N, dim) and cast to FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)            # broadcast over dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                     # (B,N,N,dim)

    return out
