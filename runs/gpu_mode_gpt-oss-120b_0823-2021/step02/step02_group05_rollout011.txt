"""
TriMul “outgoing” forward (AlphaFold‑3) – Triton‑accelerated mask.
The heavy O(N³·C) reduction stays in PyTorch (leveraging Hopper’s GEMM
optimisations), while the optional pair‑wise mask is applied with a custom
Triton kernel that streams the data‑wise mask efficiently.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication of a 4‑D tensor [B,N,N,H]
# with a broadcast mask [B,N,N] (mask is 0/1 float).  The kernel works on
# a flat view of the tensors; each contiguous block of H elements shares
# the same mask entry.
# ----------------------------------------------------------------------
@triton.jit
def _mask_apply_kernel(
    src_ptr,          # const float*  (input tensor)
    mask_ptr,         # const float*  (mask tensor, shape B*N*N)
    dst_ptr,          # float*        (output tensor)
    total_el,         # i64           total number of elements = B*N*N*H
    H,                # i64           hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < total_el                            # OOB guard

    # Load source values
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # Index into mask: each group of H consecutive elements uses the same mask entry.
    mask_idx = offs // H
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Store masked result
    tl.store(dst_ptr + offs, src * mval, mask=mask)


def _apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` of shape [B,N,N,H] by ``mask[...,None]`` using Triton.
    ``mask`` is float (0/1) with shape [B,N,N].  Returns a fresh tensor.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                                      # good occupancy on H200

    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)                         # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    _mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry‑point: forward pass of the outgoing TriMul operator.
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]       (float32, 0/1)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (int);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm on the input channel dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=1e-5,
    )                                                   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections → [B,N,N,hidden]
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, w["left_proj.weight"])
    right = torch.nn.functional.linear(x, w["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional pair‑wise mask (Triton‑accelerated)
    # ------------------------------------------------------------------
    if not nomask:
        left  = _apply_mask_triton(left,  mask)
        right = _apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # Gating (sigmoid) – also linear, no bias
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, w["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, w["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, w["out_gate.weight"]).sigmoid()

    left  = left * left_gate     # [B,N,N,hidden]
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·C) reduction – leveraged by PyTorch’s einsum → GEMM ladder
    # ------------------------------------------------------------------
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Final LayerNorm, gating and projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                                     # broadcast over hidden
    out = torch.nn.functional.linear(out, w["to_out.weight"])

    # Ensure float32 output (all ops are in float32 already)
    return out.to(torch.float32)
