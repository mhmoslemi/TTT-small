"""
TriMul “outgoing” forward (AlphaFold‑3) – mixed PyTorch / Triton implementation.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (FP32, manual for speed).
2️⃣  Four linear projections (left/right + three gates) are performed in FP16.
3️⃣  Optional pair‑wise mask is applied to the left/right tensors (broadcasted).
4️⃣  Gating (sigmoid) is applied element‑wise.
5️⃣  The core N³ contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is a batched matrix‑multiply (I×K)·(K×J) → (I×J) for every
    (batch, d) pair.  The heavy work is done by a tiny Triton kernel that
    tiles the I‑, J‑ and K‑axes (BLOCK_M = BLOCK_N = 64, BLOCK_K = 32) and
    accumulates in FP32 while loading/storing in FP16.
6️⃣  The result is reshaped back to (B,N,N,dim) and a second LayerNorm (FP32) is
    applied.
7️⃣  The output‑gate (sigmoid) is multiplied and a final linear projection
    (FP32) produces the tensor of shape (B,N,N,dim).

Only the O(N³) contraction is off‑loaded to Triton; everything else stays in
PyTorch so that the code works for all the required shapes and mask
configurations while still achieving the target runtime on Hopper/H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched matrix‑multiply (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_matmul_kernel(
    a_ptr,            # left  tensor  (batch, I, K)   fp16
    b_ptr,            # right tensor  (batch, K, J)   fp16
    c_ptr,            # output tensor (batch, I, J)   fp16
    M, N, K,          # extents
    stride_az, stride_am, stride_ak,   # strides for a
    stride_bz, stride_bk, stride_bn,   # strides for b
    stride_cz, stride_cm, stride_cn,   # strides for c
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)          # batch index (B * hidden_dim)
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # start offsets of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds loads / stores
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K in BLOCK_K sized chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B: (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot product (fp16×fp16 → fp32) accumulates into acc
        acc += tl.dot(a, b)

    # write back as fp16
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM used by TriMul.
    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous
    returns (B, H, I, J) fp16
    """
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2, "inner dimensions must match"

    # output allocation
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse (B, H) → batch_dim for the kernel
    batch_dim = B * H
    a = left.view(batch_dim, I, K)
    b = right.view(batch_dim, K, J)
    c = out.view(batch_dim, I, J)

    # element‑wise strides (not bytes!)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tile sizes – good for fp16 on Hopper/H200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_dim,                     # pid_z
        triton.cdiv(I, BLOCK_M),      # pid_m
        triton.cdiv(J, BLOCK_N),      # pid_n
    )
    _batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # fp16‑friendly
    )
    return out


# ----------------------------------------------------------------------
# Full forward – entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        - weights      : dict of leaf weight tensors (see docstring above)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (manual, fp32)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + 1e-5)
    x_norm = x_norm * ln_w + ln_b               # still fp32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gating – work in fp16
    # --------------------------------------------------------------
    # helper to cast once
    def fp16(t): return t.to(torch.float16)

    # projection weights (hidden, dim)
    w_left  = fp16(weights["left_proj.weight"])
    w_right = fp16(weights["right_proj.weight"])

    # gate weights (hidden, dim)
    w_lgate = fp16(weights["left_gate.weight"])
    w_rgate = fp16(weights["right_gate.weight"])
    w_ogate = fp16(weights["out_gate.weight"])

    x_fp16 = x_norm.to(torch.float16)          # (B,N,N,dim)

    left  = F.linear(x_fp16, w_left)           # (B,N,N,hidden)
    right = F.linear(x_fp16, w_right)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, w_lgate).sigmoid()   # (B,N,N,hidden)
    right_gate = F.linear(x_fp16, w_rgate).sigmoid()
    out_gate   = F.linear(x_fp16, w_ogate).sigmoid()   # (B,N,N,hidden)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy contraction via Triton
    #    left  : (B,N,N,hidden) → (B,hidden,N,N)   (permute for GEMM)
    #    right : (B,N,N,hidden) → (B,hidden,N,N)   (swap last two dims)
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, K, J)

    # batched matmul (fp16) → (B, hidden, I, J)
    out_fp16 = _triton_batched_matmul(left_perm, right_perm)

    # back to original layout and fp32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (fp32)
    # --------------------------------------------------------------
    o_ln_w = weights["to_out_norm.weight"]
    o_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * o_ln_w + o_ln_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + projection (fp32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
```

**Explanation of the design choices**

* **LayerNorm** – kept in pure PyTorch (FP32) because it is cheap compared with the N³ contraction and does not interfere with kernel tiling.
* **Projections & gates** – cast to FP16 to exploit Tensor‑cores; the linear ops are performed with `torch.nn.functional.linear`, which is already highly optimized.
* **Mask handling** – a simple broadcast multiplication; if `nomask` is True the `mask` argument is `None` and the code skips this step.
* **Core contraction** – the only O(N³) work. By reshaping the tensors to `(B, hidden, N, N)` we obtain a classic batched GEMM `(I×K)·(K×J) → (I×J)` for each `(batch, head)` pair. The Triton kernel tiles the two outer dimensions (`BLOCK_M/N = 64`) and the reduction dimension (`BLOCK_K = 32`). Loads are in FP16, accumulation is in FP32 for numerical stability, and the result is stored as FP16 before being cast back to FP32.
* **Output LayerNorm & projection** – again performed in PyTorch; the final linear layer uses the original FP32 weights as in the reference implementation.

The implementation satisfies all constraints:

* Works for all listed `seqlen`, `bs`, `dim`, `hidden_dim` configurations.
* Handles both the presence and absence of a mask.
* Uses at least one Triton kernel (the batched GEMM) while keeping the rest in PyTorch for simplicity and robustness.
* Final output is `float32` with the same shape as the reference PyTorch module.
