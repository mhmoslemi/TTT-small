"""
TriMul “outgoing” forward pass (AlphaFold‑3 style).

* 4‑D input   : (B, N, N, D)        – D = dim
* LayerNorm   : performed in PyTorch (small‑ish tensor)
* Linear proj : 4 × Linear (no bias) → hidden_dim
* Mask        : optional, broadcasted to hidden_dim
* Gating      : sigmoid(Linear) and element‑wise multiplication
* Core        : out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
                (the N³ contraction).  This is implemented as a
                batched matrix multiplication with a custom Triton kernel
                (fp16 compute, fp32 accumulation) to exploit Tensor‑cores.
* Output      : LayerNorm → gating → Linear (all in PyTorch).

The kernel works on tensors of shape (Z, I, K) @ (Z, K, J) where
Z = B * hidden_dim, I = N, J = N, K = N.  The result has shape (Z, I, J)
and is later reshaped back to (B, N, N, dim).  All heavy arithmetic is
performed in fp16, while the final output is returned in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel (I×K)·(K×J) → (I×J)  –  fp16 compute / fp32 acc
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    a_ptr,            # left matrix   (Z, I, K)
    b_ptr,            # right matrix  (Z, K, J)
    c_ptr,            # output matrix (Z, I, J)
    M, N, K,          # matrix sizes
    stride_az, stride_am, stride_ak,   # a strides (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)          # batch*head index
    pid_m = tl.program_id(1)          # tile row   (I)
    pid_n = tl.program_id(2)          # tile col   (J)

    # ------------------------------------------------------------------
    # 0‑D offsets for this tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (higher precision for the sum over K)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 1‑D reduction over K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile  (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 accumulation
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # 2‑D store (convert back to fp16 for the following PyTorch layers)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM via the Triton kernel above.

    Parameters
    ----------
    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous

    Returns
    -------
    out : (B, H, I, J) fp16   (still on GPU)
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # output allocation
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # combine the two leading dimensions → Z
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides in element units (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # ------------------------------------------------------------------
    # Tile size – tuned for Hopper (SM90) – keep shared‑mem < 64 KB
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                                 # program_id(0) – batch*head
        triton.cdiv(I, BLOCK_M),           # program_id(1) – rows
        triton.cdiv(J, BLOCK_N),           # program_id(2) – cols
    )
    batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,            # good default for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch orchestration + Triton core)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]    (float32)
        mask         : torch.Tensor [B, N, N]        (float32/bool) or None
        weights      : dict of parameter tensors
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm  (keep in float32 – tensor is small)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                     # (B,N,N,dim) fp32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # linear weights (no bias) – cast once
    def _fp16(t): return t.to(torch.float16, non_blocking=True)

    left_proj_w   = _fp16(weights["left_proj.weight"])
    right_proj_w  = _fp16(weights["right_proj.weight"])
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])

    # ------------------------------------------------------------------
    # 3. Projections
    # ------------------------------------------------------------------
    left  = F.linear(x_fp16, left_proj_w)       # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)      # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 4. Optional mask  (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5. Gating
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core N³ contraction → Triton batched GEMM
    #    left : (B,N,N,hidden)   → (B, hidden, N, N)
    #    right: (B,N,N,hidden)   → (B, hidden, N, N) and then transpose K,J
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = triton_batched_gemm(left_perm, right_perm)   # (B, H, I, J) fp16

    # back to original layout and fp32 for the rest of the pipeline
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm (still fp32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 8. Final gating + projection (all fp32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # out_gate still fp16 → cast
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
