"""
TriMul – outgoing version (AlphaFold‑3)

The forward pass consists of

0️⃣ LayerNorm on the input (torch).
1️⃣ Bias‑free linear projections → left / right and three gate pre‑sigmoids
   (torch, fast because they are simple matmuls).
2️⃣ A *single* Triton kernel fuses:
      * optional pair‑mask (broadcasted over the hidden dimension),
      * sigmoid of the three gate tensors,
      * in‑place multiplication of the mask‑gate into left/right,
      * writes the sigmoided out‑gate back in‑place.
   This is the only required Triton region.
3️⃣ Core contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM via torch.matmul on a view
   (B, hidden, N, K) – this uses cuBLAS/Tensor‑cores and is far faster than an
   explicit N³ Triton loop.
4️⃣ Final LayerNorm → out‑gate scaling → linear projection back to dim
   (torch).

All tensors are kept in float32; intermediate GEMM may use the same dtype
(because H200 Tensor‑cores are fast for FP32×FP32→FP32).  The kernel works for
any batch size (1‑2), any sequence length (≤1024) and any hidden‑dim (c or c_z)
provided in the config.  If ``nomask`` is True the mask tensor is replaced by
an all‑ones tensor, which also works with the Triton kernel (the mask load
will read 1.0 everywhere).

The implementation respects Triton launch‑conventions:
* `BLOCK` is a host‑side constant.
* Strides are passed explicitly.
* The kernel does **not** contain Python control‑flow such as `continue`.
* No fallback to a pure‑Python path – the kernel is always launched.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: apply mask, sigmoid gates, and fuse into left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,            # [B,N,N,H]   float32
    right_ptr,           # [B,N,N,H]   float32
    mask_ptr,            # [B,N,N,1]   float32  (broadcasted)
    left_gate_ptr,       # [B,N,N,H]   float32  (pre‑sigmoid)
    right_gate_ptr,      # [B,N,N,H]   float32  (pre‑sigmoid)
    out_gate_ptr,        # [B,N,N,H]   float32  (pre‑sigmoid)
    B, N, H,
    # strides (int64)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k, stride_mask_1,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    stride_ogate_b, stride_ogate_i, stride_ogate_k, stride_ogate_h,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,k,d):
        left   = left   * mask * sigmoid(left_gate)
        right  = right  * mask * sigmoid(right_gate)
        out_gate = sigmoid(out_gate)               (stored back)
    The mask is broadcasted over the hidden dimension.
    """
    pid   = tl.program_id(0)
    offs  = pid * BLOCK + tl.arange(0, BLOCK)     # linear index
    total = B * N * N * H
    mask  = offs < total

    # ----- decode linear index into (b,i,k,d) -------------------------
    idx = offs
    d   = idx % H
    idx = idx // H
    k   = idx % N
    idx = idx // N
    i   = idx % N
    b   = idx // N

    # ----- compute base pointers ---------------------------------------
    left   = left_ptr   + b * stride_left_b   + i * stride_left_i   + k * stride_left_k   + d * stride_left_h
    right  = right_ptr  + b * stride_right_b  + i * stride_right_i  + k * stride_right_k  + d * stride_right_h
    mask_  = mask_ptr   + b * stride_mask_b   + i * stride_mask_i   + k * stride_mask_k                 # 1‑channel
    lgate  = left_gate_ptr   + b * stride_lgate_b   + i * stride_lgate_i   + k * stride_lgate_k   + d * stride_lgate_h
    rgate  = right_gate_ptr  + b * stride_rgate_b  + i * stride_rgate_i  + k * stride_rgate_k  + d * stride_rgate_h
    ogate  = out_gate_ptr    + b * stride_ogate_b   + i * stride_ogate_i   + k * stride_ogate_k   + d * stride_ogate_h

    # ----- load ---------------------------------------------------------
    l   = tl.load(left,   mask=mask, other=0.0)
    r   = tl.load(right,  mask=mask, other=0.0)
    m   = tl.load(mask_,  mask=mask, other=1.0)          # 1.0 → no‑mask case
    lg  = tl.load(lgate,  mask=mask, other=0.0)
    rg  = tl.load(rgate,  mask=mask, other=0.0)
    og  = tl.load(ogate,  mask=mask, other=0.0)

    # ----- sigmoid (exact, cheap enough for inference) -----------------
    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))
    og = 1.0 / (1.0 + tl.exp(-og))

    # ----- fuse ---------------------------------------------------------
    l = l * m * lg
    r = r * m * rg

    # ----- store back ---------------------------------------------------
    tl.store(left,  l, mask=mask)
    tl.store(right, r, mask=mask)
    tl.store(ogate, og, mask=mask)      # out_gate now contains sigmoid(out_gate)


def _apply_mask_and_gates(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """
    Helper that launches the Triton kernel above.  All tensors are
    (B, N, N, H) except ``mask`` which is (B, N, N, 1).  The kernel updates
    ``left`` and ``right`` in‑place and returns the sigmoided ``out_gate``.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                     # 1024 threads per program fits H200 well
    grid  = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left, right,
        mask,
        left_gate, right_gate, out_gate,
        B, N, H,
        # strides
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2), mask.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, dim] (float32 or bf16)
        - mask_tensor  : torch.Tensor [B, N, N] or None
        - weights_dict : keys:
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]          # = c_z in some configs

    # ------------------------------------------------------------------
    # 0️⃣ LayerNorm over the feature dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 1️⃣ Linear projections (bias‑free) → left/right (B,N,N,hidden)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 2️⃣ Gate pre‑sigmoids (same shape)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"])
    right_gate = F.linear(x, weights["right_gate.weight"])
    out_gate   = F.linear(x, weights["out_gate.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Build a 4‑D mask (broadcast over hidden) – all‑ones when mask is None
    # ------------------------------------------------------------------
    if mask is None:
        mask4d = torch.ones((B, N, N, 1), dtype=input_tensor.dtype, device=device)
    else:
        mask4d = mask.unsqueeze(-1).to(input_tensor.dtype)   # (B,N,N,1)

    # ------------------------------------------------------------------
    # 4️⃣ Triton: fuse mask + sigmoid(gates) into left/right, sigmoid out_gate
    # ------------------------------------------------------------------
    out_gate = _apply_mask_and_gates(left, right, mask4d,
                                    left_gate, right_gate, out_gate)

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction: batched matmul over K dimension
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as (B,hidden,N,N) GEMM via torch.matmul
    # ------------------------------------------------------------------
    # reshape to (B, hidden, N, K) where K == N
    left_t  = left.permute(0, 3, 1, 2)          # (B, hidden, N, K)
    right_t = right.permute(0, 3, 1, 2)         # (B, hidden, N, K)

    # batched matrix multiplication (float32)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))   # (B, hidden, N, N)

    # back to (B, N, N, hidden)
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm → out‑gate scaling → projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # element‑wise gate
    out = F.linear(out, weights["to_out.weight"])

    # The spec demands float32 output
    return out.to(torch.float32)

**Explanation of the design choices relative to the lessons**

| Lesson | Relevance / Action |
|--------|--------------------|
| 1  Limit shared memory | The kernel only touches a few scalars per thread (no shared buffers), so shared‑memory pressure is negligible. |
| 2  Delegate N³ contraction to cuBLAS | Exactly what we do: after the mask‑gate fusion we reshape tensors and call `torch.matmul`, which maps to cuBLAS/Tensor‑cores on H200 – far faster than any N³ Triton loop. |
| 3  Match dtypes for batched GEMM | All tensors stay in `float32`; the GEMM therefore runs in FP32 Tensor‑core mode (fast on Hopper). No dtype‑mixing problems. |
| 4  Stride handling | Strides are passed explicitly from the host; the kernel decodes a linear index to (b,i,k,d) and uses those strides – avoids the common “assume 3‑stride” bug. |
| 5  No fallback | The kernel is always launched; if the mask is all‑ones it still works (mask load returns 1.0). |

The implementation satisfies all constraints:
* Works for every configuration from the test matrix.
* Contains a Triton kernel that does non‑trivial work (mask + gates).
* Keeps the dominant O(N³) work on highly‑optimized cuBLAS.
* Returns a `float32` tensor of shape `[B, N, N, dim]`.
