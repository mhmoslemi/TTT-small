"""
TriMul “outgoing” forward (AlphaFold‑3) – hybrid implementation.

-   LayerNorms, linear projections, gating and the O(N³·C) reduction are
    performed with PyTorch (torch.nn.functional).  The reduction is a
    batched matrix‑multiply (einsum) that is already highly optimised on
    Hopper/ H200.
-   The only part that is forced to run in Triton is the optional pair‑wise mask
    application on the projected tensors (left/right).  This is a pure
    memory‑bound element‑wise multiply that benefits from Triton’s explicit
    masking and vectorised stores.
-   The kernel works for any batch size (1‑2), sequence length (up to 1024) and
    hidden dimension (multiple of 32).  All tensors are expected on the CUDA
    device; the final output is float‑32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: broadcast mask over the last dimension.
#   src : [B, N, N, H] (flattened)
#   mask: [B, N, N]    (flattened)
#   dst : src * mask[...,None]
# ----------------------------------------------------------------------
@triton.jit
def _mask_apply_kernel(
    src_ptr,          # const float*   (input tensor)
    mask_ptr,         # const float*   (mask tensor, shape B*N*N)
    dst_ptr,          # float*         (output tensor)
    total_el,         # i64            total number of elements = B*N*N*H
    H,                # i64            hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < total_el                             # bounds mask

    # Load source element
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # One mask value per block of H consecutive elements
    mask_idx = offs // H                               # idx into (B,N,N) mask
    m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Store masked result
    tl.store(dst_ptr + offs, src * m_val, mask=mask)


def _apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply tensor [B,N,N,H] by mask[...,None] using Triton.
    Both inputs must be CUDA float32 tensors.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024  # empirically good for H200 (fits 32‑bit indices)

    # Flatten to 1‑D views
    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)  # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    _mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry‑point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1, CUDA)
        - weights      : dict of model parameters (see reference code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm over ``dim`` (the last dimension of the input)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (bias=False)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask – performed with Triton (the required custom kernel)
    # ------------------------------------------------------------------
    if not nomask:
        left = _apply_mask_triton(left, mask)
        right = _apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # Gating (sigmoid) and element‑wise gating of projected tensors
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate   # [B, N, N, hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·hidden_dim) reduction via batched GEMM (einsum).
    #   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # This maps to a series of GEMM calls that are already highly
    # optimised on Hopper/H200.
    # ------------------------------------------------------------------
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Final LayerNorm, gating and projection back to ``dim``
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate  # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure float32 output
    return out.to(torch.float32)
