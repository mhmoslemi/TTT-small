"""
TriMul “outgoing” forward (AlphaFold‑3) – Triton‑accelerated mask application.

The heavy O(N³·C) reduction is performed by a single torch.einsum,
which on Hopper maps to highly‑optimised GEMM kernels.
The only custom Triton region is the broadcast mask multiplication on the
projected tensors (left/right).  This memory‑bound operation profits from
explicit vectorised loads/stores and avoids a PyTorch element‑wise kernel.

The implementation follows the reference PyTorch module exactly:
  * LayerNorm on the input (dim)
  * Linear projections (left/right) and gates (left_gate, right_gate, out_gate)
  * Optional mask (broadcast over the hidden dimension) – Triton kernel
  * Element‑wise gating of left/right
  * Reduction   out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
  * LayerNorm on the hidden dimension, out‑gate gating and final linear projection
All tensors are assumed to be on CUDA and the final output is float‑32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: broadcast mask multiplication
#   tensor:  [B, N, N, H]  (float32)
#   mask  :  [B, N, N]     (float32, 0/1)
#   out   :  tensor * mask[...,None]
# ----------------------------------------------------------------------
@triton.jit
def mask_mul_kernel(
    src_ptr,          # const float*  (input tensor)
    mask_ptr,         # const float*  (mask tensor, flattened B*N*N)
    dst_ptr,          # float*        (output tensor)
    total_el,         # i64           total number of elements = B*N*N*H
    H,                # i64           hidden dimension (last dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)           # linear indices
    mask = offs < total_el

    # Load source values
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # The mask is the same for each block of H consecutive elements.
    # Compute index into the flattened mask.
    mask_idx = offs // H
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Store the masked result
    tl.store(dst_ptr + offs, src * mval, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` of shape [B,N,N,H] by ``mask[...,None]`` using Triton.
    ``mask`` is float (0/1) with shape [B,N,N].
    Returns a new tensor with the same shape as ``tensor``.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                         # good occupancy on H200

    # 1‑D contiguous views
    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)            # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    mask_mul_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_flat,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor, shape [B, N, N]   (float32, 0/1)
        - weights      : dict of model parameters (see code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (ints);
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask – Triton implementation
    # ------------------------------------------------------------------
    if not nomask:
        left = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # Gates (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate   # [B, N, N, hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·C) reduction – einsum maps to a stack of GEMMs on Hopper
    # ------------------------------------------------------------------
    # out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Final hidden‑dim LayerNorm, out‑gate gating and projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                                 # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure float32 output (all ops use float32)
    return out.to(torch.float32)
