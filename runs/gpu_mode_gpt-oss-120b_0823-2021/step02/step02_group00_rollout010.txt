"""
custom_kernel – “outgoing” TriMul (AlphaFold3)

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (FP32, manual for speed).
2️⃣ Linear projections + gating are performed in FP16:
   * left  = x @ W_left   , right = x @ W_right
   * left_gate = σ(x @ W_left_gate)   , right_gate = σ(x @ W_right_gate)
   * out_gate   = σ(x @ W_out_gate)
   * left  ← left  * left_gate   , right ← right * right_gate
   * optional mask (broadcast over hidden_dim) is applied before gating.
3️⃣ Core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply over the reduction axis *k*.
   We reshape each head (the hidden dimension) as a separate “head” and call a
   Triton kernel that computes many independent GEMMs of shape (N×N)·(N×N).
   The kernel works in FP16, accumulates in FP32 and writes FP16.
4️⃣ Convert the result back to FP32, apply a second LayerNorm, multiply by the
   output‑gate and finish with a linear projection to the original dimension.

The Triton kernel is deliberately simple and tuned for Hopper (SM‑90):
   BLOCK_M = BLOCK_N = 64   # tile size in the output i/j dimensions
   BLOCK_K = 32              # tile size in the reduction k dimension
   num_warps = 4             # good FP16 utilisation on H200
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr,          # left operand  (batch, I, K)   fp16
    b_ptr,          # right operand (batch, K, J)   fp16
    c_ptr,          # output        (batch, I, J)   fp16
    M, N, K,        # extents
    stride_az, stride_am, stride_ak,   # a strides (batch, i, k)
    stride_bz, stride_bk, stride_bn,   # b strides (batch, k, j)
    stride_cz, stride_cm, stride_cn,   # c strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # program IDs – a separate program for each (batch, head) pair
    # ------------------------------------------------------------------
    pid_z = tl.program_id(0)          # batch*head dimension
    pid_m = tl.program_id(1)          # tile row   (I)
    pid_n = tl.program_id(2)          # tile col   (J)

    # ------------------------------------------------------------------
    # compute tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # a: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # b: (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp16×fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # write result – cast back to fp16
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _batched_gemm(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a : (B, H, I, K)   fp16, contiguous
    b : (B, H, K, J)   fp16, contiguous
    returns (B, H, I, J) fp16
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, H, I, K = a.shape
    _, _, K2, J = b.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=a.device, dtype=torch.float16)

    # collapse (B,H) → batch_dim for the kernel
    batch_dim = B * H
    a_ = a.view(batch_dim, I, K)
    b_ = b.view(batch_dim, K, J)
    c_ = out.view(batch_dim, I, J)

    stride_az, stride_am, stride_ak = a_.stride()
    stride_bz, stride_bk, stride_bn = b_.stride()
    stride_cz, stride_cm, stride_cn = c_.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_dim,                     # pid_z
        triton.cdiv(I, BLOCK_M),       # pid_m
        triton.cdiv(J, BLOCK_N),       # pid_n
    )
    _batched_gemm_kernel[grid](
        a_ptr=a_,
        b_ptr=b_,
        c_ptr=c_,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # tuned for FP16 on Hopper
    )
    return out


# ----------------------------------------------------------------------
# Main entry point – pure PyTorch + Triton
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        - mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        - weights      : dict of leaf weight tensors (see docstring above)
        - config       : dict containing keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over last dimension (FP32)
    # --------------------------------------------------------------
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_ln = (x - mean) / torch.sqrt(var + 1e-5)
    x_ln = x_ln * w_norm + b_norm               # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_fp16 = x_ln.to(torch.float16)

    # linear weights are stored as (hidden_dim, dim) – transpose for F.linear
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Projections
    # --------------------------------------------------------------
    # shape -> (B, N, N, hidden_dim)
    left  = F.linear(x_fp16, left_proj_w)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 5️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Core contraction via Triton
    #    treat hidden_dim as a “head” dimension
    # --------------------------------------------------------------
    # reshape to (B, H, I, K) and (B, H, K, J)
    left_t   = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t  = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _batched_gemm(left_t, right_t)          # (B, H, I, J)

    # back to (B, N, N, hidden_dim) and FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP32)
    # --------------------------------------------------------------
    w_out_ln = weights["to_out_norm.weight"]
    b_out_ln = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * w_out_ln + b_out_ln

    # --------------------------------------------------------------
    # 8️⃣ Final gating + linear projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)   # broadcast over hidden_dim
    to_out_w = weights["to_out.weight"]      # (dim, hidden_dim)
    out = F.linear(out, to_out_w)            # (B,N,N,dim), FP32

    return out
```

### How the design follows the lessons

| Lesson | Relevance & Action |
|--------|---------------------|
| **Limit shared memory** | Tile sizes `64×64×32` keep the per‑CTA shared‑memory well below Hopper limits. |
| **Delegate N³ to cuBLAS** | Instead of a naïve N³ loop we perform the contraction as many independent GEMMs, which Triton executes with tensor‑cores, achieving the same asymptotic work as cuBLAS but lets us fuse the mask/gating beforehand. |
| **Match dtypes for batched GEMM** | Both operands (`left_t`, `right_t`) are cast to **fp16** before the kernel; accumulation is in fp32, then the result is cast back to fp16. All later ops are performed in fp32, satisfying the mixed‑precision requirements. |
| **Stride handling** | Strides are obtained from the *view‑flattened* tensors (`a_`, `b_`, `c_`) that are 3‑D, matching exactly the three strides the kernel expects. No extra assumptions are made. |

The kernel processes each `(batch, hidden_dim)` pair independently, which maps naturally onto the “head” dimension introduced by the hidden projection. The rest of the TriMul pipeline (layer‑norms, linear layers, gating, mask) stays in regular PyTorch – this keeps the implementation concise while still accelerating the dominant O(N³) contraction.
