"""
custom_kernel – “outgoing” TriMul (AlphaFold3)

Algorithm
---------
1️⃣ Layer‑norm the input (FP32, manual for speed).
2️⃣ Linear projections + gating are performed in FP16.
3️⃣ Optional mask (broadcast over hidden_dim) is applied in FP16.
4️⃣ The core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix‑multiply over the last three dimensions.
   We launch a Triton kernel that treats every (batch, dim) pair as an
   independent GEMM  (N×N)·(N×N) → (N×N) in FP16, accumulating in FP32.
   Tile sizes (BLOCK_M, BLOCK_N, BLOCK_K) = (64,64,32) are tuned for
   Hopper/SM‑90 Tensor‑cores.
5️⃣ The result is cast back to FP32, a second LayerNorm is applied,
   multiplied by the output‑gate and finally projected back to the
   original dimension with a FP32 linear layer.

Only the O(N³) contraction is written in Triton; all other ops stay in
PyTorch for simplicity and to keep the implementation robust.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr,            # left  : (B*H, I, K)
    b_ptr,            # right : (B*H, K, J)
    c_ptr,            # out   : (B*H, I, J)
    M, N, K,          # matrix extents
    stride_az, stride_am, stride_ak,   # a strides (batch, i, k)
    stride_bz, stride_bk, stride_bn,   # b strides (batch, k, j)
    stride_cz, stride_cm, stride_cn,   # c strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    #  program IDs
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)          # tile along M
    pid_n = tl.program_id(1)          # tile along N
    pid_z = tl.program_id(2)          # batch*head index

    # --------------------------------------------------------------
    #  tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    #  accumulator (FP32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    #  reduction over K
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A : (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B : (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot – fp16×fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    #  write result (cast back to fp16 for storage)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous
    returns (B, H, I, J)   fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse batch*head for the kernel launch
    batch = B * H
    a = left.view(batch, I, K)
    b = right.view(batch, K, J)
    c = out.view(batch, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),          # rows
        triton.cdiv(J, BLOCK_N),          # cols
        batch,                            # one program per (batch,head)
    )
    _batched_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
#  Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        mask         : torch.Tensor or None, shape [B, N, N] (float or bool)
        weights      : dict of leaf tensors (see docstring above)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim], dtype torch.float32
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (fp32, manual)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    # mean / var over last dim
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still fp32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections + gates (fp16)
    # --------------------------------------------------------------
    # cast once
    def _fp16(t): return t.to(torch.float16)

    left_proj_w   = _fp16(weights["left_proj.weight"])      # (hidden_dim, dim)
    right_proj_w  = _fp16(weights["right_proj.weight"])
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])
    to_out_w      = weights["to_out.weight"]               # final proj stays fp32

    x_fp16 = x.to(torch.float16)                            # (B,N,N,dim)

    # projections
    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ heavy contraction with Triton
    #    shape juggling so that hidden_dim becomes the “head” dimension H
    # --------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()    # (B, H, I, K)
    # right needs (B, H, K, J) where J == N
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, H, K, J)

    # batched GEMM in fp16, result (B, H, N, N) in fp16
    out_fp16 = _batched_gemm(left_perm, right_perm)

    # back to (B, N, N, H) and fp32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ second LayerNorm (fp32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 7️⃣ output gate + final linear (fp32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over dim
    out = F.linear(out, to_out_w)                   # (B,N,N,dim)

    return out
```

**Explanation of design choices**

* The O(N³) contraction dominates runtime; we off‑load it to a single Triton kernel that performs a batched GEMM for every (batch, hidden_dim) pair.  
* Tile sizes (64×64×32) match the Hopper Tensor‑core sweet‑spot, keep shared‑memory modest and give high occupancy with `num_warps=4`.  
* All LayerNorms and linear layers that are not on the critical path remain in PyTorch – they are cheap compared with N³ work and avoid extra kernel‑launch overhead.  
* Mixed‑precision: the projection/gating stages use FP16 for bandwidth, while the two LayerNorms and the final linear stay in FP32 for numerical stability.  
* Optional mask handling is done in FP16; broadcasting avoids an extra kernel.  

The implementation satisfies all constraints, runs the core contraction in Triton, and returns a FP32 tensor of shape `[B, N, N, dim]`.
