"""
TriMul “outgoing” kernel (AlphaFold‑3).

The module consists of
1️⃣ LayerNorm on the 4‑D input.
2️⃣ Linear projections + sigmoid gates (no bias).
3️⃣ Optional pair‑wise mask.
4️⃣ The expensive pair‑wise product
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   –‑ we keep this part in PyTorch because it is a true dense
   N³ contraction that is already well‑optimised by torch.compile
   on Hopper.  The rest of the work (layer‑norm, gating, masking)
   is moved to a tiny Triton kernel so that the launch overhead is
   negligible while still demonstrating a non‑trivial fusion.

The kernel works for any batch size (≤2), sequence length N (≤1024)
and hidden dimension Dᵤ (the “hidden_dim” argument, typically 128).
All tensors are expected to be on the current CUDA device and the
final output is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
#  Triton kernel: layer‑norm + linear projections + gates + masking
# ----------------------------------------------------------------------
@triton.jit
def trimul_fwd_kernel(
        # pointers
        x_ptr,               # [B, N, N, C]  input
        out_left_ptr,        # [B, N, N, H]  left projection (pre‑gate)
        out_right_ptr,       # [B, N, N, H]  right projection (pre‑gate)
        out_left_gate_ptr,   # [B, N, N, H]  left gate   (sigmoid)
        out_right_gate_ptr,  # [B, N, N, H]  right gate  (sigmoid)
        out_out_gate_ptr,    # [B, N, N, H]  out gate    (sigmoid)
        # weight & bias tensors (flattened for simplicity)
        norm_w_ptr, norm_b_ptr,
        left_w_ptr, right_w_ptr,
        left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,
        # strides (in elements)
        stride_xb, stride_xi, stride_xj, stride_xc,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_ri, stride_rk, stride_rh,
        stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
        stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
        stride_og_b, stride_og_i, stride_og_k, stride_og_h,
        # meta‑parameters
        B: tl.constexpr, N: tl.constexpr, C: tl.constexpr, H: tl.constexpr,
        BLOCK_B: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_C: tl.constexpr, BLOCK_H: tl.constexpr,
        EPS: tl.constexpr = 1e-6
):
    """
    For each block of (b, i, j) computes
        * LayerNorm over the last dimension `C`
        * Linear projections `left`, `right`
        * Sigmoid gates
    The results are written to separate output buffers that will later be
    consumed by the PyTorch part of the operator.
    """
    pid = tl.program_id(0)

    # ---- grid layout -------------------------------------------------
    #  pid enumerates blocks over batch and the two spatial dims:
    #  pid = b * (N/BLOCK_N)² + i_block * (N/BLOCK_N) + j_block
    num_i = tl.cdiv(N, BLOCK_N)
    num_j = tl.cdiv(N, BLOCK_N)

    b   = pid // (num_i * num_j)
    tmp = pid % (num_i * num_j)
    i0  = (tmp // num_j) * BLOCK_N
    j0  = (tmp %  num_j) * BLOCK_N

    # guard against out‑of‑range batches
    if b >= B:
        return

    # ---- offsets ------------------------------------------------------
    offs_i = i0 + tl.arange(0, BLOCK_N)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ---- load the [B,N,N,C] slice ------------------------------------
    #   shape (BLOCK_N, BLOCK_N, C)
    x_ptrs = x_ptr + \
        b * stride_xb + \
        offs_i[:, None] * stride_xi + \
        offs_j[None, :] * stride_xj

    # load whole C‑vector for each (i,j) – we will perform a fused
    # layer‑norm + two linear layers + three gates.
    x = tl.load(x_ptrs + tl.arange(0, C)[None, None, :] * stride_xc,
                mask=mask_i[:, None, None] & mask_j[None, :, None],
                other=0.0)          # (M,N,C)

    # ---- LayerNorm ----------------------------------------------------
    # compute mean & variance across C (per (b,i,j) element)
    mean = tl.reduce(x, 'c', tl.sum) / C
    var  = tl.reduce((x - mean) * (x - mean), 'c', tl.sum) / C
    x_norm = (x - mean) / tl.sqrt(var + EPS)

    # ---- Linear projection: left = x_norm @ left_w.T  (no bias) -----
    # weights are stored as (H, C) → we treat them as column‑major.
    left_w = tl.load(left_w_ptr + tl.arange(0, H)[:, None] * C + tl.arange(0, C)[None, :],
                     mask=tl.arange(0, H)[:, None] < H,
                     other=0.0)               # (H,C)
    left = tl.dot(x_norm, left_w)                # (M,N,H)

    # ---- Linear projection: right ------------------------------------
    right_w = tl.load(right_w_ptr + tl.arange(0, H)[:, None] * C + tl.arange(0, C)[None, :],
                      mask=tl.arange(0, H)[:, None] < H,
                      other=0.0)               # (H,C)
    right = tl.dot(x_norm, right_w)              # (M,N,H)

    # ---- Gates (sigmoid) ---------------------------------------------
    def sigmoid(x):
        return 1.0 / (1.0 + tl.exp(-x))

    lg_w = tl.load(left_gate_w_ptr + tl.arange(0, H)[:, None] * C + tl.arange(0, C)[None, :],
                    other=0.0)
    left_gate = sigmoid(tl.dot(x_norm, lg_w))

    rg_w = tl.load(right_gate_w_ptr + tl.arange(0, H)[:, None] * C + tl.arange(0, C)[None, :],
                    other=0.0)
    right_gate = sigmoid(tl.dot(x_norm, rg_w))

    og_w = tl.load(out_gate_w_ptr + tl.arange(0, H)[:, None] * C + tl.arange(0, C)[None, :],
                    other=0.0)
    out_gate = sigmoid(tl.dot(x_norm, og_w))

    # ---- write results -------------------------------------------------
    #  All buffers have the same layout as the input: [B,N,N,H]
    out_left_ptrs  = out_left_ptr  + b * stride_lb + offs_i[:,None] * stride_li + offs_j[None,:] * stride_lk
    out_right_ptrs = out_right_ptr + b * stride_rb + offs_i[:,None] * stride_ri + offs_j[None,:] * stride_rk
    out_lg_ptrs    = out_left_gate_ptr  + b * stride_lg_b + offs_i[:,None] * stride_lg_i + offs_j[None,:] * stride_lg_k
    out_rg_ptrs    = out_right_gate_ptr + b * stride_rg_b + offs_i[:,None] * stride_rg_i + offs_j[None,:] * stride_rg_k
    out_og_ptrs    = out_out_gate_ptr   + b * stride_og_b + offs_i[:,None] * stride_og_i + offs_j[None,:] * stride_og_k

    # store (M,N,H)
    tl.store(out_left_ptrs  + tl.arange(0, H)[None, None, :] * stride_lh,
             left,    mask=mask_i[:,None,None] & mask_j[None,:,None])
    tl.store(out_right_ptrs + tl.arange(0, H)[None, None, :] * stride_rh,
             right,   mask=mask_i[:,None,None] & mask_j[None,:,None])
    tl.store(out_lg_ptrs    + tl.arange(0, H)[None, None, :] * stride_lg_h,
             left_gate,  mask=mask_i[:,None,None] & mask_j[None,:,None])
    tl.store(out_rg_ptrs    + tl.arange(0, H)[None, None, :] * stride_rg_h,
             right_gate, mask=mask_i[:,None,None] & mask_j[None,:,None])
    tl.store(out_og_ptrs    + tl.arange(0, H)[None, None, :] * stride_og_h,
             out_gate,   mask=mask_i[:,None,None] & mask_j[None,:,None])


# ----------------------------------------------------------------------
#  Python wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the TriMul “outgoing” operator.
    Only the LayerNorm → projections → gating stage is off‑loaded to
    Triton; the N³ contraction is performed with a regular
    torch.einsum (which on Hopper is already heavily optimized).

    Args
    ----
    data : (input, mask, weights, config)
        input   – [B, N, N, C]   float32 on CUDA
        mask    – [B, N, N]      (bool / 0‑1) on CUDA
        weights – dict with the exact keys used in the reference model
        config  – dict with at least ``dim`` (C) and ``hidden_dim`` (H)

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, C = x.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    #   Allocate intermediate buffers (float32)
    # ------------------------------------------------------------------
    left      = torch.empty((B, N, N, H), dtype=torch.float32, device=device)
    right     = torch.empty((B, N, N, H), dtype=torch.float32, device=device)
    left_gate = torch.empty((B, N, N, H), dtype=torch.float32, device=device)
    right_gate= torch.empty((B, N, N, H), dtype=torch.float32, device=device)
    out_gate  = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    #   Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_B = 1                     # B is at most 2 → we keep it simple
    BLOCK_N = 64                    # 64‑wide tiles give good occupancy on H200
    BLOCK_C = C                    # we load the whole channel dim at once
    BLOCK_H = 64                    # hidden dim is 128 → two tiles

    grid = (triton.cdiv(B, BLOCK_B) *
            triton.cdiv(N, BLOCK_N) *
            triton.cdiv(N, BLOCK_N),)

    trimul_fwd_kernel[grid](
        # pointers
        x,
        left, right,
        left_gate, right_gate, out_gate,
        # weight pointers (flattened)
        weights["norm.weight"].contiguous(),
        weights["norm.bias"].contiguous(),
        weights["left_proj.weight"].contiguous(),
        weights["right_proj.weight"].contiguous(),
        weights["left_gate.weight"].contiguous(),
        weights["right_gate.weight"].contiguous(),
        weights["out_gate.weight"].contiguous(),
        # strides (in elements)
        x.stride(0), x.stride(1), x.stride(2), x.stride(3),
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        # meta‑params
        B, N, C, H,
        BLOCK_B=BLOCK_B, BLOCK_N=BLOCK_N,
        BLOCK_C=BLOCK_C, BLOCK_H=BLOCK_H,
        # optional mask handling – we embed it after the kernel
    )

    # ------------------------------------------------------------------
    #   Apply the (optional) pair mask *after* projection
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape: [B,N,N] → broadcast over hidden dim
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    #   Gates
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    #   N³ contraction – torch.einsum (fp16 for speed, back‑to‑fp32)
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.bfloat16)
    right_fp16 = right.to(torch.bfloat16)

    out = torch.einsum('b i k d, b j k d -> b i j d', left_fp16, right_fp16)
    out = out.to(torch.float32)                # back to FP32

    # ------------------------------------------------------------------
    #   Output LayerNorm + out‑gate + final linear
    # ------------------------------------------------------------------
    # to_out_norm (LayerNorm over hidden_dim)
    out = F.layer_norm(out,
                       normalized_shape=(H,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    out = out * out_gate                       # broadcast over hidden dim

    # final linear projection back to C
    out = torch.nn.functional.linear(out,
                                     weights["to_out.weight"])

    return out
