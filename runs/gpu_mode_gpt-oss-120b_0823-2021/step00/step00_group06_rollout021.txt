"""
TriMul “outgoing” implementation.

*   LayerNorm, Linear projections and gating are carried out with
    regular torch kernels (they are cheap compared to the O(N³) work).
*   The core operation  

        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    is a batched matrix‑multiply for every hidden dimension *d*.
    We fuse the reduction over *k* for a block of output rows (i) and
    columns (j) and a block of hidden channels (d) in a single Triton
    kernel.  Each iteration of the inner *k*‑loop loads a tile of the
    two input tensors, performs a per‑channel matrix multiply with
    `tl.dot` and atomically accumulates the result in FP32.  The final
    result is stored back in FP32.

The kernel is launched with a three‑dimensional grid:
    (batch, ceil(N/BM), ceil(N/BN))

The implementation supports any batch size ≤2, sequence length up to
1024 and hidden dimensions that are multiples of the channel block
size (here 32).  Mixed‑precision (FP16) can be added later – the
reference implementation keeps everything in FP32 for numerical
stability.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# ----------------------------------------------------------------------
# Triton kernel: batched matmul over the “k” dimension for every hidden
# channel block.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               #   [B, N, N, H]  left tensor (masked & gated)
    right_ptr,              #   [B, N, N, H]  right tensor (masked & gated)
    out_ptr,                #   [B, N, N, H]  output
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,  # rows   of the output (i‑axis)
    BLOCK_N: tl.constexpr,  # cols   of the output (j‑axis)
    BLOCK_K: tl.constexpr,  # reduction axis (k‑axis)
    BLOCK_H: tl.constexpr,  # hidden‑channel block size
):
    # --------------------------------------------------------------
    # 1) Identify the tile (batch, i‑block, j‑block) this program works on
    # --------------------------------------------------------------
    pid = tl.program_id(0)
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    b = pid // (num_i * num_j)
    rem = pid % (num_i * num_j)
    pid_i = rem // num_j
    pid_j = rem % num_j

    # --------------------------------------------------------------
    # 2) Offsets for the current tile
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # hidden‑channel offsets (same for every tile)
    offs_h = tl.arange(0, BLOCK_H)

    # ----------------------------------------------------------------
    # 3) Accumulator for the tile (M × N × H)
    # ----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ----------------------------------------------------------------
    # 4) Loop over the reduction axis k in blocks
    # ----------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers to the current k‑tile of left and right
        left_tile = left_ptr + \
            b * stride_left_b + \
            offs_i[:, None, None] * stride_left_i + \
            offs_k[None, :, None] * stride_left_k + \
            offs_h[None, None, :] * stride_left_h

        right_tile = right_ptr + \
            b * stride_right_b + \
            offs_j[:, None, None] * stride_right_j + \
            offs_k[None, :, None] * stride_right_k + \
            offs_h[None, None, :] * stride_right_h

        # load (M×K×H) and (N×K×H) tiles, mask out‑of‑bounds entries
        left = tl.load(
            left_tile,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )
        right = tl.load(
            right_tile,
            mask=mask_j[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )

        # ----------------------------------------------------------------
        # 5) For each hidden channel slice compute a (M×K)·(K×N) matmul
        # ----------------------------------------------------------------
        for h in range(0, BLOCK_H):
            a = left[:, :, h]   # (M, K)
            b = right[:, :, h]  # (N, K) -> we need (K, N)
            # tl.dot expects (M,K) @ (K,N) → (M,N)
            acc[:, :, h] += tl.dot(a, b, trans_b=True)

    # ----------------------------------------------------------------
    # 6) Write the accumulated tile back to global memory
    # ----------------------------------------------------------------
    out_tile = out_ptr + \
        b * stride_out_b + \
        offs_i[:, None, None] * stride_out_i + \
        offs_j[None, :, None] * stride_out_j + \
        offs_h[None, None, :] * stride_out_h

    tl.store(
        out_tile,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None],
    )


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : Tensor[bs, N, N, dim]    (float32, on CUDA)
        - mask         : Tensor[bs, N, N]         (bool or 0/1, on CUDA)
        - weights      : dict of weight tensors
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    Tensor[bs, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1) LayerNorm + Linear projections + gates (pure torch)
    # --------------------------------------------------------------
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask (if supplied)
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # Gates (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(
        x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(
        x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(
        x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 2) Triton kernel: core O(N³) reduction
    # --------------------------------------------------------------
    # Strides for the 4‑D tensors
    stride_left = left.stride()
    stride_right = right.stride()
    # left: [B,N,N,H]
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = stride_left
    # right: [B,N,N,H] (note j‑axis is the second N)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = stride_right

    # Output placeholder
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=device)
    stride_out = out.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = stride_out

    # Blocking parameters (tuned for Hopper, can be adjusted)
    BLOCK_M = 64   # rows of i‑axis
    BLOCK_N = 64   # cols of j‑axis
    BLOCK_K = 32   # reduction axis
    BLOCK_H = 32   # hidden‑channel block (hidden_dim is a multiple of 32)

    # Grid dimensions: (batch * ceil(N/BM) * ceil(N/BN))
    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid = (bs * grid_i * grid_j,)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # --------------------------------------------------------------
    # 3) Final Linear + LayerNorm + out‑gate (torch)
    # --------------------------------------------------------------
    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    mean = out.mean(-1, keepdim=True)
    var = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean) / torch.sqrt(var + 1e-5)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # apply out‑gate and final projection
    out_norm = out_norm * out_gate
    final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return final
