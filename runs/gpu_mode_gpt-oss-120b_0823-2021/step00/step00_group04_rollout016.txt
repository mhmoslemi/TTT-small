"""
TriMul “outgoing” implementation.

The expensive part of the forward pass is the 4‑D einsum

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which can be seen as a batched matrix multiplication for each
(hidden) channel *h* :

    out_h = left_h @ right_h.T                (shape B×N×N)

The kernel below computes this product tile‑wise with
Triton (BLOCK_M × BLOCK_N tiles, looping over K).  All
normalisation, linear projections and gates are left in PyTorch
(they are cheap compared with the N³ work).  Mixed‑precision
(FP16) is used inside the kernel; the final result is cast back to
float32 so the external API matches the reference implementation.
"""

import torch
import triton
import triton.language as tl
from torch import nn, einsum


# ----------------------------------------------------------------------
# Triton kernel: one kernel instance per (batch, hidden) pair.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # problem size
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for a single (b, h) pair.
    """
    # ------------------------------------------------------------------
    # 1) identify this (batch, hidden) pair
    # ------------------------------------------------------------------
    pid = tl.program_id(0)                 # combined batch*head dimension
    pid_i = tl.program_id(1)                # tile over i
    pid_j = tl.program_id(2)                # tile over j

    b = pid // H                           # batch index
    h = pid % H                            # hidden dimension index

    # ------------------------------------------------------------------
    # 2) tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) loop over K dimension in BLOCK_K steps
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,i,k,h]  -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
            + h * stride_lh
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right[b,j,k,h] we need the transpose (k, j) for dot
        # right_T[k, j] -> shape (BLOCK_K, BLOCK_N)
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_rj
            + h * stride_rh
        )
        right_t = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # matrix multiplication over K (reduce)
        acc += tl.dot(left, right_t)        # (M,K) @ (K,N) -> (M,N)

    # ------------------------------------------------------------------
    # 4) write out
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
        + h * stride_oh
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _tri_mul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, H]   (float16)
    returns out : [B, N, N, H]    (float16)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)                     # FP16 buffer

    # strides in elements (not bytes)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # tile sizes – tuned for Hopper/SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                                   # batch*head dimension
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _trimul_einsum_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # good default for fp16 on Hopper
    )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments
    ----------
    data : tuple (input, mask, weights, config)
        input  – torch.Tensor [B, N, N, dim]        (float32)
        mask   – torch.Tensor [B, N, N] (optional) (float32/bool)
        weights – dict of Parameter tensors
        config  – dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1) LayerNorm on the input (dim)
    # --------------------------------------------------------------
    #   We implement LayerNorm manually to avoid an extra kernel launch.
    eps = 1e-5
    gamma = weights["norm.weight"]
    beta = weights["norm.bias"]
    # mean & var over last dimension
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, keepdim=True, unbiased=False)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x_norm * gamma + beta               # shape [B,N,N,dim]

    # --------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # --------------------------------------------------------------
    #   left_proj, right_proj : (dim → hidden) per element
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Apply mask if present
    # --------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                 # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4) Compute gates (sigmoid) and apply
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"]
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"]
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Core N³ multiplication – Triton kernel (FP16 inside)
    # --------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # kernel produces fp16; cast back to fp32 for the remaining ops
    out_fp16 = _tri_mul_einsum(left_fp16, right_fp16)
    out = out_fp16.to(torch.float32)

    # --------------------------------------------------------------
    # 6) Output LayerNorm + gate + final linear
    # --------------------------------------------------------------
    # to_out_norm (hidden -> hidden)
    eps2 = 1e-5
    gamma2 = weights["to_out_norm.weight"]
    beta2 = weights["to_out_norm.bias"]
    mean2 = out.mean(-1, keepdim=True)
    var2 = out.var(-1, keepdim=True, unbiased=False)
    out_norm = (out - mean2) / torch.sqrt(var2 + eps2)
    out = out_norm * gamma2 + beta2

    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
