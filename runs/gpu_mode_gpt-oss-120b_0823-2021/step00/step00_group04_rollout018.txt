"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left`/`right` have shape [B,N,N,H] (H = hidden_dim).
We keep the pre‑ and post‑processing (LayerNorm, Linear, gating) in
pure PyTorch (they are highly optimized) and accelerate the
triple‑product with a Triton kernel that tiles the (i,j) plane and the
hidden dimension.  The kernel computes a blocked matrix‑multiply‑like
accumulation over the k‑axis and writes the result back in FP32.
The final output is brought back to torch and the remaining
LayerNorm/Linear are applied.

The kernel works for any batch size (1‑2), sequence length up to 1024,
and hidden dimensions that are a multiple of the block size (the
default 32 works for all allowed hidden sizes).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel – computes the 4‑D einsum
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # block sizes (compile‑time constants)
    BLOCK_M: tl.constexpr,  # tile over i
    BLOCK_N: tl.constexpr,  # tile over j
    BLOCK_H: tl.constexpr,  # tile over hidden dimension
    BLOCK_K: tl.constexpr,  # tile over reduction axis k
):
    """
    Compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a blocked triple loop.
    """
    pid = tl.program_id(0)

    # total number of tiles in the (i,j) grid (per batch)
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    # decode program id → (b, tile_i, tile_j)
    b = pid // (num_m * num_n)
    tmp = pid % (num_m * num_n)
    tile_i = tmp // num_n
    tile_j = tmp % num_n

    # thread‑local indices
    i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    h = tl.arange(0, BLOCK_H)

    # masks for out‑of‑bounds handling
    i_mask = i < N
    j_mask = j < N
    h_mask = h < H

    # accumulator: (BLOCK_M, BLOCK_N, BLOCK_H)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # loop over the reduction axis k
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # left[b,i,k,h]
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + k[None, :, None] * stride_l_k
            + h[None, None, :] * stride_l_h,
            mask=i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )
        # right[b,j,k,h]
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + j[:, None, None] * stride_r_j
            + k[None, :, None] * stride_r_k
            + h[None, None, :] * stride_r_h,
            mask=j_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )
        # accumulate element‑wise product
        acc += left * right

    # write result
    out = acc
    tl.store(
        out_ptr
        + b * stride_o_b
        + i[:, None, None] * stride_o_i
        + j[None, :, None] * stride_o_j
        + h[None, None, :] * stride_o_h,
        out,
        mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
    )


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Args:
        data: (input, mask, weights, config)
            - input:  [B, N, N, D]  (float32)
            - mask :  [B, N, N]    (float32 or bool)
            - weights: dict with all linear / layer‑norm parameters
            - config : dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension (projected)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim = D)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    # fused layer‑norm (torch has an efficient implementation)
    x = F.layer_norm(input_tensor, (D,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – shape [B,N,N,H]
    # ------------------------------------------------------------------
    def linear4d(x, w):
        # x: [B,N,N,D] , w: [H,D]  (weight is transposed compared to nn.Linear)
        # we perform (x @ w.T)
        return torch.einsum("bnid,hd->bnih", x, w)

    left = linear4d(x, weights["left_proj.weight"].to(device))
    right = linear4d(x, weights["right_proj.weight"].to(device))

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.einsum("bnid,hd->bnih", x, weights["left_gate.weight"].to(device))
    )
    right_gate = torch.sigmoid(
        torch.einsum("bnid,hd->bnih", x, weights["right_gate.weight"].to(device))
    )
    out_gate = torch.sigmoid(
        torch.einsum("bnid,hd->bnih", x, weights["out_gate.weight"].to(device))
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the core einsum
    # ------------------------------------------------------------------
    # allocate output tensor
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # launch configuration – block sizes tuned for H200
    BLOCK_M = 64   # i tile
    BLOCK_N = 64   # j tile
    BLOCK_H = 32   # hidden tile (must divide H; H∈{128,384,768})
    BLOCK_K = 32   # reduction tile

    # number of program instances
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (B * grid_m * grid_n,)

    trimul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_h=stride_l_h,
        stride_r_b=stride_r_b,
        stride_r_j=stride_r_j,
        stride_r_k=stride_r_k,
        stride_r_h=stride_r_h,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_h=stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=weights["to_out_norm.weight"].to(device),
                       bias=weights["to_out_norm.bias"].to(device))
    out = out * out_gate
    # final projection back to D
    out = torch.einsum("bnih,dh->bnid", out, weights["to_out.weight"].to(device))

    return out
