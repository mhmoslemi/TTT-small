"""
TriMul (outgoing) – Triton‑accelerated core

The expensive part of the TriMul forward pass is the
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched 3‑D tensor contraction (N³·C work).
The kernel below tiles the (i,j) output plane and iterates over
k in blocks, accumulating in float32.  All surrounding
operations (LayerNorm, linear projections, gating, final
LayerNorm + Linear) are kept in PyTorch – they are cheap
compared to the N³ reduction and avoid extra code‑gen
complexity.  The kernel works for any batch size ≤2,
any sequence length N up to 1024 and any hidden dimension
(usually 128).  The final tensor is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------
# Triton kernel: batched N³ reduction over the hidden dimension
# -------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const float*  [B, N, N, H]
    right_ptr,         # const float*  [B, N, N, H]
    out_ptr,           #       float*  [B, N, N, H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]"""

    pid = tl.program_id(0)                     # linear program id over (b, i_block, j_block)
    num_b = B
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)

    b = pid // (num_i * num_j)
    rem = pid % (num_i * num_j)
    i_block = rem // num_j
    j_block = rem % num_j

    # base offsets for this tile
    i_offset = i_block * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_block * BLOCK_J + tl.arange(0, BLOCK_J)
    h_offset = tl.arange(0, BLOCK_H)

    # masks for out‑of‑bounds
    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_h = h_offset < H

    # compute flat offsets (will add k‑offset later inside the loop)
    left_base = (
        b * stride_left_b
        + i_offset[:, None, None] * stride_left_i
        + 0 * stride_left_k      # k will be added in the loop
        + h_offset[None, None, :] * stride_left_h
    )
    right_base = (
        b * stride_right_b
        + 0 * stride_right_j      # j will be added in the loop
        + j_offset[None, :, None] * stride_right_j
        + 0 * stride_right_k      # k added later
        + h_offset[None, None, :] * stride_right_h
    )
    out_base = (
        b * stride_out_b
        + i_offset[:, None, None] * stride_out_i
        + j_offset[None, :, None] * stride_out_j
        + h_offset[None, None, :] * stride_out_h
    )

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over k dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left[b,i,k,:]  -> shape (BLOCK_I, BLOCK_K, BLOCK_H)
        left_ptrs = left_base + k_offset[None, :, None] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :] & mask_h[None, None, :], other=0.0)
        # load right[b,j,k,:] -> shape (BLOCK_J, BLOCK_K, BLOCK_H)
        right_ptrs = right_base + k_offset[None, :, None] * stride_right_k
        right = tl.load(right_ptrs, mask=mask_j[None, :] & mask_k[:, None] & mask_h[None, None, :], other=0.0)

        # transpose right to (BLOCK_K, BLOCK_J, BLOCK_H) for dot
        right = tl.trans(right, (1, 0, 2))

        # multiply‑accumulate over k
        # left: (I, K, H), right: (K, J, H) → we need sum_k left*right
        # We perform a dot per hidden slice H
        for hh in range(0, BLOCK_H):
            # shape (I, K) * (K, J) -> (I, J)
            acc[:, :, hh] += tl.dot(left[:, :, hh], right[:, :, hh])

        # advance pointers (implicit via k_offset recomputed each iter)
    # store result
    out_ptrs = out_base
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])


# -------------------------------------------------------------
# Host‑side entry point
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the outgoing TriMul operator.
    Arguments
    ---------
    data : (input, mask, weights, config)
        input  – [B, N, N, dim]   (float32)
        mask   – [B, N, N]        (float32 or bool)
        weights – dict with all linear / layer‑norm parameters
        config – {'dim': int, 'hidden_dim': int}
    Returns
    -------
    out : torch.Tensor  [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ---------- LayerNorm ----------
    def layer_norm(x, weight, bias, eps=1e-5):
        # x: [..., dim]
        mean = x.mean(-1, keepdim=True)
        var = ((x - mean) ** 2).mean(-1, keepdim=True)
        return (x - mean) * torch.rsqrt(var + eps) * weight + bias

    x = layer_norm(
        input_tensor,
        weights["norm.weight"].to(device),
        weights["norm.bias"].to(device),
    )

    # ---------- Linear projections ----------
    # Linear without bias: y = x @ W.T
    left = F.linear(x, weights["left_proj.weight"].to(device))          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # ---------- Apply mask (if present) ----------
    if mask is not None:
        mask3d = mask.unsqueeze(-1)                                 # [B,N,N,1]
        left = left * mask3d
        right = right * mask3d

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device)))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device)))

    left = left * left_gate
    right = right * right_gate

    # ---------- Triton reduction ----------
    # allocate output tensor (float32)
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – chosen to fit H200 shared memory & occupancy
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32          # hidden dim is a multiple of 32 in all test configs
    BLOCK_K = 32

    grid = (triton.cdiv(B * N * N, BLOCK_I * BLOCK_J),)   # one program per (b,i_block,j_block)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Post‑process ----------
    # to_out_norm (LayerNorm over hidden)
    out = layer_norm(
        out,
        weights["to_out_norm.weight"].to(device),
        weights["to_out_norm.bias"].to(device),
    )
    out = out * out_gate

    # final linear back to dim
    out = F.linear(out, weights["to_out.weight"].to(device))   # [B,N,N,dim]

    return out.to(torch.float32)
