"""
TriMul (outgoing) – Triton‑accelerated core.

The heavy part of TriMul is the 4‑D “einsum”:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where   left/right = (x·W)∘σ(x·W_gate)  and the mask is already applied.
We keep all preprocessing (LayerNorm, Linear, Gate, mask) in PyTorch
(because they are highly optimized there) and off‑load only the
reduction over the sequence dimension *k* to a Triton kernel.

The kernel works on a tile (BLOCK_M × BLOCK_N) of the (i,j) plane,
and on a tile (BLOCK_H) of the channel dimension *d*.
For each tile we accumulate over *k* in blocks of size BLOCK_K.
All arithmetic is performed in float32; the final tensor is returned
as float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -------------------------------------------------------------
# Triton kernel
# -------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,          #  [B, N, N, H]   left  (i,k,d)
    right_ptr,         #  [B, N, N, H]   right (j,k,d)
    out_ptr,           #  [B, N, N, H]   output (i,j,d)

    # sizes
    B, N, H,

    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute one tile of the outgoing TriMul reduction.
    Each program instance processes:
        * one batch   (pid_b)
        * a tile of i (pid_m)
        * a tile of j (pid_n)
        * a tile of channel d (pid_h)
    """
    pid_b = tl.program_id(0)          # batch index
    pid_m = tl.program_id(1)          # i‑tile index
    pid_n = tl.program_id(2)          # j‑tile index
    pid_h = tl.program_id(3)          # channel‑tile index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    d_offset = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    # mask for out‑of‑bounds
    i_mask = i_offset < N
    j_mask = j_offset < N
    d_mask = d_offset < H

    # ------------------------------------------------------------------
    # Pointers to the beginning of the tile (without k dimension yet)
    # ------------------------------------------------------------------
    left_tile_base  = left_ptr  + pid_b * stride_left_b \
                                   + i_offset[:, None, None] * stride_left_i \
                                   + 0 * stride_left_k \
                                   + d_offset[None, None, :] * stride_left_h
    right_tile_base = right_ptr + pid_b * stride_right_b \
                                   + j_offset[:, None, None] * stride_right_j \
                                   + 0 * stride_right_k \
                                   + d_offset[None, None, :] * stride_right_h

    # Accumulator – float32, initialise to 0
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduction over k
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(0, num_k_tiles):
        # k range for this tile
        k_offset = tk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask   = k_offset < N

        # Load left: shape (M, K, H)
        left_ptrs  = left_tile_base + k_offset[None, :, None] * stride_left_k
        left = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :], other=0.0)

        # Load right: shape (N, K, H)
        right_ptrs = right_tile_base + k_offset[None, :, None] * stride_right_k
        right = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :], other=0.0)

        # For every channel in the tile we perform a small matrix‑multiply:
        #   acc[...,h] += left[...,h] @ right[...,h].T
        #   (left : M×K, right : N×K)
        # Use tl.dot which reduces over the K dimension.
        for h in range(0, BLOCK_H):
            # Guard against the last tile being partially out‑of‑bounds
            if not d_mask[h]:
                continue
            left_h  = left[:, :, h]          # (M, K)
            right_h = right[:, :, h]         # (N, K)

            # tl.dot expects (M, K) @ (K, N) → (M, N)
            # right_h needs to be transposed.
            acc[:, :, h] += tl.dot(left_h, right_h, trans_b=True)   # (M,N)

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + pid_b * stride_out_b \
                        + i_offset[:, None, None] * stride_out_i \
                        + j_offset[None, :, None] * stride_out_j \
                        + d_offset[None, None, :] * stride_out_h

    tl.store(out_ptrs,
             acc.to(tl.float32),
             mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :])

# -------------------------------------------------------------
# Host‑side helper
# -------------------------------------------------------------
def launch_trimul_kernel(
    left: torch.Tensor,
    right: torch.Tensor,
    *,
    BLOCK_M: int = 64,
    BLOCK_N: int = 64,
    BLOCK_K: int = 32,
    BLOCK_H: int = 32,
) -> torch.Tensor:
    """
    Calls the Triton kernel and returns the accumulated tensor.
    Shapes:
        left , right : (B, N, N, H)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)

    # Strides in elements (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(H, BLOCK_H),
    )
    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,               # fine‑tune for Hopper
    )
    return out

# -------------------------------------------------------------
# Main entry point required by the evaluator
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = torch.float32

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # manual layer‑norm: (x - mean) / sqrt(var+eps) * w + b
    eps = 1e-5
    mean = input_tensor.mean(-1, keepdim=True)
    var  = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) + mask
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]         # (hidden_dim, dim)

    # shape after projection: (B,N,N,hidden_dim)
    left  = F.linear(x, left_proj_w)   # bias = None
    right = F.linear(x, right_proj_w)

    # mask handling
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(dtype)          # (B,N,N,1)
        left  = left * mask_unsq
        right = right * mask_unsq

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid) and application
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core TriMul reduction (Triton)
    # ------------------------------------------------------------------
    # The kernel expects float32; we already are in float32.
    out = launch_trimul_kernel(left, right,
                               BLOCK_M=64, BLOCK_N=64, BLOCK_K=32, BLOCK_H=32)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]   # (dim, hidden_dim)

    # LayerNorm on hidden_dim
    out_mean = out.mean(-1, keepdim=True)
    out_var  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - out_mean) / torch.sqrt(out_var + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # apply output gate (broadcast over hidden_dim)
    out = out * out_gate

    # final linear back to dim
    out = F.linear(out, to_out_w)   # bias is None

    return out.to(dtype)
