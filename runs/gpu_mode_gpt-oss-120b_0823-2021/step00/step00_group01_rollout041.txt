"""
TriMul "outgoing" forward pass (AlphaFold3).

The heavy part is the pairwise multiplicative sum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We rewrite it as a batched matrix multiply:
    left_  : (B, N, N, D) → (B*D, N, K)   with K = N
    right_ : (B, N, N, D) → (B*D, N, K)   (same layout)
    out_   : (B*D, N, N) = left_ @ right_.transpose(-2,-1)

The matrix multiply is performed by a custom Triton kernel that
supports arbitrary batch size (B*D) and works in FP16 (the
intermediate accumulation uses FP32 for accuracy).  All remaining
operations (LayerNorm, Linear, gating, final projection) are kept in
PyTorch for simplicity and to avoid excessive kernel launches.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------
# Triton batched GEMM (left @ right^T)  –  FP16 inputs, FP32 accumulation
# --------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    batch, M, N, K,                    # sizes
    stride_ab, stride_ak,              # a: (batch, M, K)
    stride_bb, stride_bk,              # b: (batch, N, K)  (we read as transposed)
    stride_cb, stride_cm,              # c: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,
):
    pid = tl.program_id(0)
    # --------------------------------------------------------------
    # 1) map program id to batch and row‐tile
    # --------------------------------------------------------------
    batch_id = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_grid = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))

    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # 2D tiling (m,n) within one batch
    pid_m = pid_grid // num_pid_n
    pid_n = pid_grid %  num_pid_n

    # --------------------------------------------------------------
    # 3) offsets for this tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers with batch offset
    a_batch_ptr = a_ptr + batch_id * stride_ab
    b_batch_ptr = b_ptr + batch_id * stride_bb
    c_batch_ptr = c_ptr + batch_id * stride_cb

    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_ak + offs_k[None, :] * 1)
    # we need B^T, i.e. we read b as (N, K) but dot with k dimension,
    # so we load b with stride for K then N.
    b_ptrs = b_batch_ptr + (offs_n[:, None] * stride_bk + offs_k[None, :] * 1)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & (offs_k[None, :] < K), other=0.0).to(tl.float32)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * 1        # stride_ak == 1 for contiguous K dim
        b_ptrs += BLOCK_K * 1

    # write back
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cb
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------------
# Helper to launch the kernel
# --------------------------------------------------------------------
def batched_gemm(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a: (B*D, M, K)  FP16
    b: (B*D, N, K)  FP16  (will be used as transposed)
    returns: (B*D, M, N) FP16
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, N, _ = b.shape
    c = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # stride layout (batch, row, col)
    stride_ab = a.stride(0)
    stride_ak = a.stride(1)
    stride_bb = b.stride(0)
    stride_bk = b.stride(1)
    stride_cb = c.stride(0)
    stride_cm = c.stride(1)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    GROUP_M = 8

    grid = (triton.cdiv(B * M * N, BLOCK_M * BLOCK_N),)  # one program per tile across batches
    # Alternative tiling: iterate over batch inside kernel (as above)
    # compute total programs = B * ceil(M/BLOCK_M) * ceil(N/BLOCK_N)
    grid = (
        B * triton.cdiv(M, BLOCK_M) * triton.cdiv(N, BLOCK_N),
    )

    batched_gemm_kernel[grid](
        a,
        b,
        c,
        B,
        M,
        N,
        K,
        stride_ab,
        stride_ak,
        stride_bb,
        stride_bk,
        stride_cb,
        stride_cm,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_M=GROUP_M,
    )
    return c

# --------------------------------------------------------------------
# Main custom kernel entry point
# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
            - mask         : torch.Tensor [B, N, N]      (bool or 0/1)
            - weights      : dict of raw weight tensors
            - config       : dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ----------------------------------------------------------------
    # 1) LayerNorm over last dim
    # ----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(input_tensor.dtype),
        bias=weights["norm.bias"].to(input_tensor.dtype),
    )  # (B,N,N,dim)

    # ----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ----------------------------------------------------------------
    # shape: (..., dim) @ (hidden, dim).T  → (..., hidden)
    left = F.linear(
        x,
        weights["left_proj.weight"].to(input_tensor.dtype),  # (hidden, dim)
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(input_tensor.dtype),
    )  # (B,N,N,hidden)

    # ----------------------------------------------------------------
    # 3) Apply mask (if any)
    # ----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)          # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------
    # 4) Gates (sigmoid)
    # ----------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(
            x,
            weights["left_gate.weight"].to(input_tensor.dtype),
        )
    )
    right_gate = torch.sigmoid(
        F.linear(
            x,
            weights["right_gate.weight"].to(input_tensor.dtype),
        )
    )
    out_gate = torch.sigmoid(
        F.linear(
            x,
            weights["out_gate.weight"].to(input_tensor.dtype),
        )
    )
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # 5) Prepare for GEMM: reshape to (B*hidden, N, N)
    #    left  : (B, N, N, hidden) → (B*hidden, N, N)
    #    right : same, will be used transposed inside kernel
    # ----------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left = left.view(B * hidden, N, N).to(torch.float16)
    right = right.view(B * hidden, N, N).to(torch.float16)

    # ----------------------------------------------------------------
    # 6) Batched matmul (left @ right^T) via Triton
    # ----------------------------------------------------------------
    out_flat = batched_gemm(left, right)  # (B*hidden, N, N), fp16

    # ----------------------------------------------------------------
    # 7) Restore original shape (B, N, N, hidden)
    # ----------------------------------------------------------------
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # ----------------------------------------------------------------
    # 8) to_out_norm + gating + final linear (back to dim)
    # ----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(input_tensor.dtype),
        bias=weights["to_out_norm.bias"].to(input_tensor.dtype),
    )
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(input_tensor.dtype),  # (dim, hidden)
    )
    return out.to(torch.float32)
