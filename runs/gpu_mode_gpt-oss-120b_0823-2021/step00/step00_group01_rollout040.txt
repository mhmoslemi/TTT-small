"""
TriMul “outgoing” kernel (AlphaFold‑3).

The heavy part of the forward pass is the einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape (B,N,N,H) (H = hidden_dim).  
For each channel *d* the operation is a rank‑1 update (outer product)
over the *k* dimension:

    out_d  +=  left_k_d.T @ right_k_d          (shape M×N)

We implement this in Triton.  The surrounding LayerNorms, linear
projections, gates and the final linear layer are kept in PyTorch – they
are cheap compared with the O(B·N³·H) work of the einsum.  The Triton
kernel processes a tile of the (i, j) plane (BLOCK_M × BLOCK_N) for a
single batch element and a single hidden channel at a time, accumulating
over the k‑dimension in blocks (BLOCK_K).  The kernel is launched with a
3‑D grid (batch, i‑tile, j‑tile) and supports any batch size ≤2,
sequence length up to 1024 and hidden_dim up to 384.

All arithmetic inside the kernel is in float32; the final result is cast
back to float32 after the surrounding PyTorch post‑processing.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# --------------------------------------------------------------
# Triton kernel: one channel (hidden dim) per launch
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const float*  [B, N, N, H]
    right_ptr,         # const float*  [B, N, N, H]
    out_ptr,           #       float*  [B, N, N, H]
    B, N, H,
    stride_bl, stride_bi, stride_bk, stride_bd,   # left strides
    stride_rl, stride_ri, stride_rk, stride_rd,   # right strides
    stride_ol, stride_oi, stride_ok, stride_od,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute one hidden channel of the TriMul outer‑product einsum.
    One thread block processes a tile (BLOCK_M, BLOCK_N) of the (i,j)
    plane for a single batch element and a single hidden channel.
    """
    pid_b = tl.program_id(0)           # batch dimension
    pid_i = tl.program_id(1)           # i‑tile
    pid_j = tl.program_id(2)           # j‑tile

    # offsets for the current tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    k0 = 0

    # range masks (handle tail)
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # each thread block works on a single hidden channel `h`
    # The launch code will vary `h` in the outer Python loop.
    h = tl.program_id(3)               # hidden channel index

    # compute base pointers for this batch & channel
    left_base  = left_ptr  + pid_b * stride_bl + h * stride_bd
    right_base = right_ptr + pid_b * stride_rl + h * stride_rd
    out_base   = out_ptr   + pid_b * stride_ol + h * stride_od

    # accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        # k‑range for this iteration
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left[i,k]  shape (M, K)
        left_ptrs  = left_base \
                     + (offs_i[:, None] * stride_bi) \
                     + (offs_k[None, :] * stride_bk)
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # load right[j,k] shape (N, K)
        right_ptrs = right_base \
                     + (offs_j[:, None] * stride_ri) \
                     + (offs_k[None, :] * stride_rk)
        right_tile = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :],
                             other=0.0)

        # rank‑1 update: sum_k left[i,k] * right[j,k]
        # this is a matrix multiplication of (M×K) @ (N×K).T
        acc += tl.dot(left_tile, right_tile)   # (M,N)

        k0 += BLOCK_K

    # store the accumulated tile
    out_ptrs = out_base \
               + (offs_i[:, None] * stride_oi) \
               + (offs_j[None, :] * stride_ok)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Helper: launch Triton kernel for all hidden channels
# --------------------------------------------------------------
def triton_trimul_outgoing(left: torch.Tensor, right: torch.Tensor,
                           mask: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, H]   (float32)
    mask         : [B, N, N]      (float32, 0/1)

    Returns:
        out : [B, N, N, H] (float32)
    """
    B, N, _, H = left.shape
    device = left.device

    # apply mask (broadcast on hidden dim)
    left = left * mask.unsqueeze(-1)
    right = right * mask.unsqueeze(-1)

    out = torch.empty_like(left)

    # launch parameters (tuned for H200)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (B,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            H)                     # extra dimension for hidden channel

    left_ptr = left
    right_ptr = right
    out_ptr = out

    # stride extraction (torch stride order is (B,N,N,H))
    stride_bl, stride_bi, stride_bk, stride_bd = left.stride()
    stride_rl, stride_ri, stride_rk, stride_rd = right.stride()
    stride_ol, stride_oi, stride_ok, stride_od = out.stride()

    trimul_outgoing_kernel[grid](
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_bl, stride_bi, stride_bk, stride_bd,
        stride_rl, stride_ri, stride_rk, stride_rd,
        stride_ol, stride_oi, stride_ok, stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


# --------------------------------------------------------------
# Full TriMul forward (PyTorch + Triton)
# --------------------------------------------------------------
class TriMulFast(nn.Module):
    def __init__(self, dim: int, hidden_dim: int):
        super().__init__()
        self.dim = dim
        self.hidden_dim = hidden_dim

        self.norm = nn.LayerNorm(dim, elementwise_affine=True)
        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim, elementwise_affine=True)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        # x : [B,N,N,dim]   mask : [B,N,N]
        x = self.norm(x)

        left = self.left_proj(x)          # [B,N,N,H]
        right = self.right_proj(x)        # [B,N,N,H]

        left = left * mask.unsqueeze(-1)
        right = right * mask.unsqueeze(-1)

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        # ----- Triton accelerated outer‑product einsum -----
        out = triton_trimul_outgoing(left, right, mask)   # [B,N,N,H]

        # post‑processing (still in PyTorch)
        out = self.to_out_norm(out)
        out = out * out_gate
        out = self.to_out(out)
        return out.float()


# --------------------------------------------------------------
# Entrypoint required by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator with a Triton‑accelerated
    core einsum.  The function receives a tuple
        (input_tensor, mask, weights, config)
    and returns the processed tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]

    # ------------------------------------------------------------------
    # Build the module and inject the provided weights (no gradients needed)
    # ------------------------------------------------------------------
    model = TriMulFast(dim, hidden_dim).to(input_tensor.device)

    # linear / layer‑norm weights (all tensors are already on the correct device)
    model.norm.weight.copy_(weights["norm.weight"])
    model.norm.bias.copy_(weights["norm.bias"])
    model.left_proj.weight.copy_(weights["left_proj.weight"])
    model.right_proj.weight.copy_(weights["right_proj.weight"])
    model.left_gate.weight.copy_(weights["left_gate.weight"])
    model.right_gate.weight.copy_(weights["right_gate.weight"])
    model.out_gate.weight.copy_(weights["out_gate.weight"])
    model.to_out_norm.weight.copy_(weights["to_out_norm.weight"])
    model.to_out_norm.bias.copy_(weights["to_out_norm.bias"])
    model.to_out.weight.copy_(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # Forward – everything runs on the GPU; the heavy part is inside Triton
    # ------------------------------------------------------------------
    output = model(input_tensor, mask)
    return output
