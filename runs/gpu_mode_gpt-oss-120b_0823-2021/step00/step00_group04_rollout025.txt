"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The expensive part of TriMul is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each batch b and hidden channel d this is a standard matrix
multiplication:
    out_{b,d} = left_{b,d} @ right_{b,d}.T          (N×N) @ (N×N) → (N×N)

The kernel below performs a batched, block‑wise matmul with
    A : (B, N, N, D)        – left
    B : (B, N, N, D)        – right
    C : (B, N, N, D)        – output

All LayerNorms, Linear projections, and gates are kept in PyTorch (they are
memory‑bound and cheap compared with the N³ work).  Only the N³‑scale
einsum is off‑loaded to Triton.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for each hidden channel (D)
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # sizes
    B, N, D,
    # strides (flattened, see launch code)
    stride_Ab, stride_Ai, stride_Ak, stride_Ad,
    stride_Bb, stride_Bj, stride_Bk, stride_Bd,
    stride_Cb, stride_Ci, stride_Cj, stride_Cd,
    # block sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C[b,i,j,d] = Σ_k A[b,i,k,d] * B[b,j,k,d]   (B,N,N,D)."""
    pid_b = tl.program_id(0)          # batch index
    pid_d = tl.program_id(1)          # hidden‑channel index
    pid_mn = tl.program_id(2)         # tile index for (i,j)

    # tile coordinates
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid_mn // num_n
    pid_n = pid_mn %  num_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers for this batch / channel
    a_base = a_ptr + pid_b * stride_Ab + pid_d * stride_Ad
    b_base = b_ptr + pid_b * stride_Bb + pid_d * stride_Bd
    c_base = c_ptr + pid_b * stride_Cb + pid_d * stride_Cd

    # initialise accumulator in fp32 (the most stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A[b,i,k,d]   shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_m[:, None] * stride_Ai
                         + offs_k[None, :] * stride_Ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # load B[b,j,k,d]   shape (BLOCK_N, BLOCK_K)
        b_ptrs = b_base + (offs_n[:, None] * stride_Bj
                         + offs_k[None, :] * stride_Bk)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # dot‑product accumulation
        acc += tl.dot(a, b)          # (M,K)·(N,K)ᵀ → (M,N)

    # write back C[b,i,j,d]
    c_ptrs = c_base + (offs_m[:, None] * stride_Ci
                     + offs_n[None, :] * stride_Cj)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: launch the Triton kernel
# ----------------------------------------------------------------------
def trimul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right:  (B, N, N, D)   float16 or float32
    returns:     (B, N, N, D)   same dtype as inputs (we keep fp16 inside kernel)
    """
    B, N, _, D = left.shape
    device = left.device

    # ensure fp16 inside kernel for bandwidth, accumulate in fp32
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out = torch.empty_like(left_fp16)

    # strides as 1‑D offsets (in elements, not bytes)
    stride_Ab, stride_Ai, stride_Ak, stride_Ad = left_fp16.stride()
    stride_Bb, stride_Bj, stride_Bk, stride_Bd = right_fp16.stride()
    stride_Cb, stride_Ci, stride_Cj, stride_Cd = out.stride()

    # block sizes – tuned for Hopper (SM90)  (64‑128 work well)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B,                                 # batch dimension
        D,                                 # hidden‑channel dimension
        triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N)   # tiles over (i,j)
    )

    trimul_einsum_kernel[grid](
        left_fp16, right_fp16, out,
        B, N, D,
        stride_Ab, stride_Ai, stride_Ak, stride_Ad,
        stride_Bb, stride_Bj, stride_Bk, stride_Bd,
        stride_Cb, stride_Ci, stride_Cj, stride_Cd,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,   # enough to keep SM busy
    )
    return out.to(left.dtype)          # convert back to original dtype


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]          torch.float32
            mask         : [B, N, N]               torch.float32 (0/1) or bool
            weights      : dict of torch tensors
            config       : dict with keys  dim, hidden_dim

    Returns:
        out : torch.Tensor  [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------
    # manual LayerNorm to avoid extra memory copies
    eps = 1e-5
    weight = weights["norm.weight"]
    bias   = weights["norm.bias"]
    # (B,N,N,dim) -> mean/var over dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * weight + bias            # shape (B,N,N,dim)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    # shape: (dim, hidden_dim)
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = torch.einsum("...d,dh->...h", x, left_proj_w)   # (B,N,N,hidden)
    right = torch.einsum("...d,dh->...h", x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Apply mask (if present)
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # (B,N,N,1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.einsum("...d,dh->...h", x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.einsum("...d,dh->...h", x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.einsum("...d,dh->...h", x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core TriMul einsum – Triton kernel
    # --------------------------------------------------------------
    out = trimul_einsum(left, right)   # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear
    # --------------------------------------------------------------
    # to_out_norm
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    eps2 = 1e-5
    mean2 = out.mean(dim=-1, keepdim=True)
    var2  = ((out - mean2) ** 2).mean(dim=-1, keepdim=True)
    out = (out - mean2) / torch.sqrt(var2 + eps2)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate

    # final linear (hidden -> dim)
    to_out_w = weights["to_out.weight"]
    out = torch.einsum("...h,hd->...d", out, to_out_w)   # (B,N,N,dim)

    return out.to(torch.float32)   # guarantee float32 output
