"""
TriMul “outgoing” kernel (AlphaFold3).

The expensive part is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden dimension `d` this is a dense matrix multiplication:
    (L_d)  : (N, K)   with K = N
    (R_d)  : (N, K)   (note the second index is K)
    out_d  = L_d @ R_d.T          → (N, N)

We fuse the three‑dimensional batch (B, D) into the Triton launch grid and
perform a block‑wise GEMM using fp16 arithmetic (the model works in bfloat16
or fp16).  All other operations (LayerNorm, Linear, gating, final
LayerNorm/Linear) are kept in PyTorch – they are cheap compared with the
O(B·N³·D) contraction.

The kernel returns a float‑32 tensor as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -----------------------------------------------------------
# Triton kernel for the core contracted multiplication
# -----------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # sizes
    B, N, D,
    # strides (in elements, not bytes)
    stride_bl, stride_bi, stride_bk, stride_bd,   # left  [b,i,k,d]
    stride_br, stride_bj, stride_bk_r, stride_bd_r,  # right [b,j,k,d]
    stride_bo, stride_bo_i, stride_bo_j, stride_bd_o,  # out   [b,i,j,d]
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_K: tl.constexpr,
    # other constants
    EPS: tl.constexpr = 1e-6,
):
    pid = tl.program_id(axis=0)          # batch * D tiles
    pid_b = pid // (D // BLOCK_D)        # batch index
    pid_d = (pid % (D // BLOCK_D)) * BLOCK_D

    # block offsets
    offs_m = tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute absolute row / col / d indices
    i = tl.program_id(axis=1) * BLOCK_M + offs_m   # i dimension
    j = tl.program_id(axis=2) * BLOCK_N + offs_n   # j dimension
    d = pid_d + tl.arange(0, BLOCK_D)               # hidden dim slice

    # mask out‑of‑bounds
    mask_i = i < N
    mask_j = j < N
    mask_d = d < D

    # pointers for the current tile
    left_tile_ptr  = left_ptr  + pid_b * stride_bl \
                                   + i[:,None,None] * stride_bi \
                                   + offs_k[None,:,None] * stride_bk \
                                   + d[None,None,:] * stride_bd

    right_tile_ptr = right_ptr + pid_b * stride_br \
                                   + j[None,:,None] * stride_bj \
                                   + offs_k[None,:,None] * stride_bk_r \
                                   + d[None,None,:] * stride_bd_r

    # accumulator in FP32 for better accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K
        cur_k = k_off + offs_k
        cur_mask_k = cur_k < N

        # load tiles (if out‑of‑bounds fill with 0)
        L = tl.load(left_tile_ptr + cur_k * stride_bk,
                    mask=mask_i[:,None,None] & cur_mask_k[None,:,None] & mask_d[None,None,:],
                    other=0.0).to(tl.float32)          # (M,K,D)
        R = tl.load(right_tile_ptr + cur_k * stride_bk_r,
                    mask=mask_j[None,:,None] & cur_mask_k[None,:,None] & mask_d[None,None,:],
                    other=0.0).to(tl.float32)          # (N,K,D)

        # inner product over K
        # we need Σ_k L[i,k,d] * R[j,k,d]  -> dot along K
        acc += tl.dot(L, R, trans_b=False)          # (M,N,D)

    # write back
    out_tile_ptr = out_ptr + pid_b * stride_bo \
                               + i[:,None,None] * stride_bo_i \
                               + j[None,:,None] * stride_bo_j \
                               + d[None,None,:] * stride_bd_o

    tl.store(out_tile_ptr,
             acc.to(tl.float32),
             mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_d[None,None,:])

# -----------------------------------------------------------
# Helper: launch wrapper for the kernel
# -----------------------------------------------------------
def trimul_gemm(left: torch.Tensor, right: torch.Tensor):
    """
    left, right : [B, N, N, D]   (float16 / bfloat16)
    returns      : [B, N, N, D]   (float32)
    """
    B, N, _, D = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in elements (not bytes)
    stride_bl, stride_bi, stride_bk, stride_bd = left.stride()
    stride_br, stride_bj, stride_bk_r, stride_bd_r = right.stride()
    stride_bo, stride_bo_i, stride_bo_j, stride_bd_o = out.stride()

    # block sizes – tuned for Hopper SM90 (shared‑mem 64KB)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_D = 32          # hidden dim tile
    BLOCK_K = 32

    # grid dimensions
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid_bd = triton.cdiv(D, BLOCK_D)
    grid = (B * grid_bd, grid_m, grid_n)

    trimul_gemm_kernel[grid](
        left, right, out,
        B, N, D,
        stride_bl, stride_bi, stride_bk, stride_bd,
        stride_br, stride_bj, stride_bk_r, stride_bd_r,
        stride_bo, stride_bo_i, stride_bo_j, stride_bd_o,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_D=BLOCK_D, BLOCK_K=BLOCK_K,
        num_warps=4,               # 4 warps works well for these sizes
    )
    return out

# -----------------------------------------------------------
# Full TriMul forward (most ops in PyTorch, core contraction in Triton)
# -----------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            input_tensor : [B, N, N, dim]   (float32)
            mask         : [B, N, N]        (bool / 0‑1 float)
            weights      : dict of parameter tensors
            config       : dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]

    # ---------- 1. LayerNorm ----------
    # LayerNorm over the last dim (dim)
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ---------- 2. Linear projections ----------
    # shape: [B,N,N,hidden]
    left = F.linear(x, weights["left_proj.weight"].to(device))          # no bias
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # ---------- 3. Apply mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)           # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- 4. Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device)))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device)))   # used later

    left = left * left_gate
    right = right * right_gate

    # ---------- 5. Core contraction (Triton) ----------
    # Cast to fp16/bfloat16 for faster matmul; keep a copy in FP32 for final norm
    left_fp = left.to(torch.bfloat16)
    right_fp = right.to(torch.bfloat16)

    # out shape [B,N,N,hidden] in FP32
    out = trimul_gemm(left_fp, right_fp)   # Triton kernel

    # ---------- 6. to_out_norm ----------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ---------- 7. Apply out_gate ----------
    out = out * out_gate

    # ---------- 8. Final linear ----------
    out = F.linear(out, weights["to_out.weight"].to(device))

    return out  # float32
