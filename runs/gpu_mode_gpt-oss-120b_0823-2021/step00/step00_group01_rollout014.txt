"""
TriMul “outgoing” implementation.

The heavy N³ interaction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a custom Triton kernel that tiles the (i,j) pair
and the hidden dimension `d`.  The kernel accumulates a block of
output values while looping over the `k` dimension in chunks of
`BLOCK_K`.  All surrounding linear / layer‑norm / gating operations
are kept in PyTorch – they are cheap compared with the N³ term and
avoid the need for a second reduction on `d`.

The final tensor is returned in float‑32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const float*   [B, N, N, H]
    right_ptr,         # const float*   [B, N, N, H]
    out_ptr,           # float*         [B, N, N, H]
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_D: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1) Decode a 3‑D index (batch, i‑tile, j‑tile) from pid
    # --------------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    b = pid // (num_i * num_j)
    pid_rem = pid % (num_i * num_j)
    i_tile = pid_rem // num_j
    j_tile = pid_rem % num_j

    # --------------------------------------------------------------
    # 2) Compute the physical offsets for the tile we are responsible for
    # --------------------------------------------------------------
    i_idx = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_idx = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    d_idx = tl.arange(0, BLOCK_D)

    # Masks for out‑of‑bounds rows / columns / channels
    mask_i = i_idx < N
    mask_j = j_idx < N
    mask_d = d_idx < H

    # Initialise the accumulator in fp32 (the kernel works in fp16 for
    # better throughput but accumulates in fp32 to keep numerical quality)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) Loop over k in blocks of BLOCK_K
    # --------------------------------------------------------------
    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        k_idx = k_start * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # ------------------------------------------------------------------
        # Load left[b, i, k, d]  → (BLOCK_M, BLOCK_K, BLOCK_D)
        # Load right[b, j, k, d] → (BLOCK_N, BLOCK_K, BLOCK_D)
        # ------------------------------------------------------------------
        left = tl.load(
            left_ptr
            + b * stride_lb
            + i_idx[:, None, None] * stride_li
            + k_idx[None, :, None] * stride_lk
            + d_idx[None, None, :] * stride_lh,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )
        right = tl.load(
            right_ptr
            + b * stride_rb
            + j_idx[:, None, None] * stride_rj
            + k_idx[None, :, None] * stride_rk
            + d_idx[None, None, :] * stride_rh,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # ------------------------------------------------------------------
        # Element‑wise product & reduction over k
        # ------------------------------------------------------------------
        # left * right   → (BLOCK_M, BLOCK_N, BLOCK_K, BLOCK_D)
        # we sum over k, leaving (BLOCK_M, BLOCK_N, BLOCK_D)
        prod = tl.cast(left, tl.float32) * tl.cast(right, tl.float32)
        # sum over the K axis
        acc += tl.sum(prod, axis=2)

    # --------------------------------------------------------------
    # 4) Write the block back
    # --------------------------------------------------------------
    out = tl.cast(acc, tl.float16)  # store in fp16, model expects fp16 after einsum
    tl.store(
        out_ptr
        + b * stride_ob
        + i_idx[:, None, None] * stride_oi
        + j_idx[None, :, None] * stride_oj
        + d_idx[None, None, :] * stride_oh,
        out,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


# ----------------------------------------------------------------------
# Helper: launch the kernel
# ----------------------------------------------------------------------
def trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]  (float16 or float32)
    returns out  : [B, N, N, H]  (float16)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    device = left.device

    # Cast to fp16 for the kernel (the reference implementation does the einsum
    # in bf16, fp16 is a safe drop‑in on Hopper)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out = torch.empty_like(left_fp16)

    # strides in element units
    stride_lb, stride_li, stride_lk, stride_lh = left_fp16.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # tile sizes – experimentally good for H200 (128‑thread warps)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_D = 32   # hidden dim is a multiple of 32 (128, 384)
    BLOCK_K = 32

    total_tiles = B * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N)
    grid = (total_tiles,)

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_D=BLOCK_D,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # 128 threads per warp → 4 warps = 512 threads
    )
    return out.to(torch.float32)   # the caller will immediately cast back to fp32


# ----------------------------------------------------------------------
# Full forward pass (PyTorch for everything except the N³ reduction)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : tuple (input, mask, weights, config)
        input  : torch.Tensor [B, N, N, dim]   (float32)
        mask   : torch.Tensor [B, N, N]        (bool / float)
        weights: dict of torch.Tensor – model parameters
        config : dict with keys "dim", "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the last dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # manual LayerNorm (no need for autograd)
    eps = 1e-5
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – we keep them in fp32 for stability
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"] # [hidden_dim, dim]

    left  = torch.einsum("bijkd,hd->bijkh", x, left_proj_w)   # [B,N,N,hidden]
    right = torch.einsum("bijkd,hd->bijkh", x, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Apply mask (if any) – mask is broadcasted over the hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating networks (sigmoid) – all linear, no bias
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(torch.einsum("bijkd,hd->bijkh", x, left_gate_w))
    right_gate = torch.sigmoid(torch.einsum("bijkd,hd->bijkh", x, right_gate_w))
    out_gate   = torch.sigmoid(torch.einsum("bijkd,hd->bijkh", x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) N³ reduction – Triton kernel
    # ------------------------------------------------------------------
    # left/right are [B,N,N,hidden]; we feed them to the kernel which
    # returns the same shape in fp16, then cast back to fp32.
    out = trimul_outgoing_triton(left, right)   # [B,N,N,hidden] fp32

    # ------------------------------------------------------------------
    # 6) Final LayerNorm, gating and projection back to dim
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # LayerNorm over hidden dimension
    mean_h = out.mean(dim=-1, keepdim=True)
    var_h  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_h) / torch.sqrt(var_h + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to original dim
    out = torch.einsum("bijkh,dh->bijkd", out, to_out_w)

    return out.to(torch.float32)
