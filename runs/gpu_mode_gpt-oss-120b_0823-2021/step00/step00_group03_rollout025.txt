"""
TriMul “outgoing” kernel (AlphaFold3).

We keep the cheap per‑element ops (LayerNorm, Linear projections,
gates, final LayerNorm + Linear) in PyTorch and offload the
N‑wise reduction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

to a tiny Triton kernel.  Each program processes a single (b,i,j)
triplet and accumulates a vector of size `hidden_dim` in registers.
The kernel is memory‑bound only on the (k)‑reduction loop and
achieves high occupancy on H200 because it works on float32 data,
uses a single warp per program and needs only
`BLOCK_H` threads (default 64) for the hidden dimension.
The final output is cast back to float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: one program per (batch, i, j) computes the hidden_dim
# vector by reducing over the k‑dimension.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const float*  [B,N,N,H]
    right_ptr,         # const float*  [B,N,N,H]
    out_ptr,           # float*        [B,N,N,H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_H: tl.constexpr,
):
    # --------------------------------------------------------------
    # Identify (b,i,j) handled by this program.
    # --------------------------------------------------------------
    pid = tl.program_id(0)
    total = B * N * N
    pid = tl.where(pid < total, pid, 0)          # guard for excess threads
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # --------------------------------------------------------------
    # Offsets for the hidden dimension.
    # --------------------------------------------------------------
    offs_d = tl.arange(0, BLOCK_H)
    mask_d = offs_d < H

    acc = tl.zeros((BLOCK_H,), dtype=tl.float32)

    # --------------------------------------------------------------
    # Reduce over k.
    # --------------------------------------------------------------
    for k in range(0, N):
        left_off = (left_ptr
                    + b * stride_left_b
                    + i * stride_left_i
                    + k * stride_left_k
                    + offs_d * stride_left_h)
        right_off = (right_ptr
                     + b * stride_right_b
                     + j * stride_right_j
                     + k * stride_right_k
                     + offs_d * stride_right_h)

        l = tl.load(left_off, mask=mask_d, other=0.0)
        r = tl.load(right_off, mask=mask_d, other=0.0)
        acc += l * r

    # --------------------------------------------------------------
    # Write result.
    # --------------------------------------------------------------
    out_off = (out_ptr
               + b * stride_out_b
               + i * stride_out_i
               + j * stride_out_j
               + offs_d * stride_out_h)
    tl.store(out_off, acc, mask=mask_d)


# ----------------------------------------------------------------------
# Host entry point.
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float16/float32)
        - mask         : [B, N, N]        (bool or {0,1})
        - weights      : dict of all Linear/LayerNorm params
        - config       : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # Helper to apply a Linear (bias‑free) with pre‑loaded weight.
    # --------------------------------------------------------------
    def linear_no_bias(x, w):
        # x: [..., dim]   w: [hidden_dim, dim]
        return torch.nn.functional.linear(x, w)

    # --------------------------------------------------------------
    # LayerNorm (no bias if not present)
    # --------------------------------------------------------------
    def layer_norm(x, weight, bias):
        return F.layer_norm(
            x,
            normalized_shape=(dim if x.shape[-1] == dim else hidden_dim,),
            weight=weight,
            bias=bias,
            eps=1e-5,
        )

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"].to(torch.float32),
        bias=weights["norm.bias"].to(torch.float32),
    )

    # ------------------------------------------------------------------
    # 2) Projections
    # ------------------------------------------------------------------
    left = linear_no_bias(x, weights["left_proj.weight"].to(torch.float32))
    right = linear_no_bias(x, weights["right_proj.weight"].to(torch.float32))

    # ------------------------------------------------------------------
    # 3) Mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        linear_no_bias(x, weights["left_gate.weight"].to(torch.float32))
    )
    right_gate = torch.sigmoid(
        linear_no_bias(x, weights["right_gate.weight"].to(torch.float32))
    )
    out_gate = torch.sigmoid(
        linear_no_bias(x, weights["out_gate.weight"].to(torch.float32))
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton reduction (the costly N^3 operation)
    # ------------------------------------------------------------------
    # Allocate output buffer
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Compute strides for the kernel
    stride_left = left.stride()   # (B, N, N, H)
    stride_right = right.stride()
    stride_out = out.stride()

    BLOCK_H = 64                     # fits 128‑dim hidden easily (2 passes if needed)

    grid = (triton.cdiv(B * N * N, 1),)   # one program per (b,i,j)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_h=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_j=stride_right[1],
        stride_right_k=stride_right[2],
        stride_right_h=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_h=stride_out[3],
        BLOCK_H=BLOCK_H,
        num_warps=4,                     # enough parallelism per program
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing: hidden → dim
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"].to(torch.float32),
        bias=weights["to_out_norm.bias"].to(torch.float32),
    )
    out = out * out_gate                         # broadcasted over (i,j)
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float32),
        bias=None,
    )

    return out.to(torch.float32)
