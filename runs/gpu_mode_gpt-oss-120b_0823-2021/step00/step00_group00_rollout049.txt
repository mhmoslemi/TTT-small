"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply for every channel *d*.  This file
computes the surrounding LayerNorm / Linear / gating logic in PyTorch
(and therefore re‑uses the supplied FP32 weights) and off‑loads the
contraction to a fused Triton kernel.  The kernel works on blocks of the
(i, j) plane and a small vector of the channel dimension, reducing over the
sequence dimension *k* in tiles.  The final result is cast back to FP32.

The kernel is written for Hopper (sm_90) but contains no device‑specific
intrinsics – it will also run on earlier GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton kernel: batched reduction over the sequence dimension
# --------------------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # meta‑parameters
    B, N, D,
    # strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,   # tile size on i dimension
    BLOCK_N: tl.constexpr,   # tile size on j dimension
    BLOCK_D: tl.constexpr,   # tile size on channel dimension
    BLOCK_K: tl.constexpr,   # tile size on reduction dimension k
):
    """ Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] """
    pid_b = tl.program_id(2)          # batch dimension
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile

    # offsets for the current tile
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d_off = tl.arange(0, BLOCK_D)

    # mask for out‑of‑bounds on i / j
    i_mask = i_off < N
    j_mask = j_off < N

    # allocate accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over k tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # -----------------
        # load left tile
        # shape: (BLOCK_M, BLOCK_K, BLOCK_D)
        # -----------------
        left_ptrs = (
            left_ptr
            + pid_b * stride_left_b
            + i_off[:, None, None] * stride_left_i
            + k_off[None, :, None] * stride_left_k
            + d_off[None, None, :] * stride_left_d
        )
        left = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None], other=0.0)

        # -----------------
        # load right tile
        # shape: (BLOCK_N, BLOCK_K, BLOCK_D)
        # -----------------
        right_ptrs = (
            right_ptr
            + pid_b * stride_right_b
            + j_off[:, None, None] * stride_right_j
            + k_off[None, :, None] * stride_right_k
            + d_off[None, None, :] * stride_right_d
        )
        right = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_mask[None, :, None], other=0.0)

        # -----------------
        # reduction over k for every d
        # we perform a dot for each channel slice
        # -----------------
        for d in range(0, BLOCK_D):
            # (BLOCK_M, BLOCK_K) dot (BLOCK_N, BLOCK_K)^T -> (BLOCK_M, BLOCK_N)
            acc[:, :, d] += tl.dot(left[:, :, d], right[:, :, d].T)

    # -----------------
    # store result
    # -----------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_out_b
        + i_off[:, None, None] * stride_out_i
        + j_off[None, :, None] * stride_out_j
        + d_off[None, None, :] * stride_out_d
    )
    tl.store(out_ptrs, acc.to(tl.float32), mask=i_mask[:, None, None] & j_mask[None, :, None])


# --------------------------------------------------------------
# Host‑side entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args:
        data: (input, mask, weights, config)
            - input : [B, N, N, dim]    (float32, on CUDA)
            - mask  : [B, N, N]           (bool/float, on CUDA)
            - weights: dict of model weights (FP32 tensors)
            - config : {"dim": int, "hidden_dim": int}

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, _, _ = input_tensor.shape
    device = input_tensor.device
    dtype = torch.float32

    # --------------------------------------------------
    # 1) LayerNorm + Linear + gating – done in PyTorch
    # --------------------------------------------------
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask (optional)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 2) Triton kernel for the contraction Σ_k left[...,k,d] * right[...,k,d]
    # ------------------------------------------------------------------
    # Prepare contiguous tensors (ensure stride layout needed by kernel)
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Tile sizes – tuned for Hopper (shared‑memory friendly)
    BLOCK_M = 32   # i dimension
    BLOCK_N = 32   # j dimension
    BLOCK_D = 16   # channel dimension (must divide hidden_dim)
    BLOCK_K = 32   # reduction dimension

    # Grid definition (3‑D)
    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B,                                # batch
    )

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_d=stride_left_d,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_d=stride_right_d,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_d=stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_D=BLOCK_D,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------
    # 3) Output post‑processing (LayerNorm + gating + final linear)
    # --------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out.to(dtype)   # guarantee FP32 output
