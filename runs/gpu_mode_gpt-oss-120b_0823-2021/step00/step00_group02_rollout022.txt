"""
TriMul “outgoing” implementation.

The heavy N³ part is the pair‑wise reduction

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

We keep the surrounding LayerNorm / Linear / gating logic in PyTorch
(because they are cheap) and off‑load the N³ reduction to a Triton kernel.
The kernel processes the i‑j plane in 2‑D tiles, iterating over the
k‑dimension in blocks and accumulating in float32.  The hidden dimension
(`hidden_dim`) is split into chunks of 32 elements (configurable) that are
handled with `tl.dot`.  The final result is cast back to float16 for the
subsequent LayerNorm / Linear which run in FP32, guaranteeing the required
float‑32 output.

The kernel works for any batch size (B=1/2) and sequence length up to
1024, hidden_dim = 128 (or a multiple of the tile size).  Masks are applied
outside the kernel in PyTorch – the kernel therefore only sees dense tensors.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ------------------------------------------------------------
# Triton kernel: pair‑wise reduction over the k‑dimension
# ------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,          # [B, N, N, H]  (i, k, h)
    right_ptr,         # [B, N, N, H]  (j, k, h)
    out_ptr,           # [B, N, N, H]  (i, j, h)
    B, N, H,           # static sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, CHUNK_H: tl.constexpr):
    """
    Compute out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]

    * The i‑j plane is tiled with BLOCK_I × BLOCK_J threads.
    * The reduction over k is performed in blocks of size BLOCK_K.
    * The hidden dimension H is processed in chunks of CHUNK_H using tl.dot.
    """
    pid = tl.program_id(0)
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)

    i_tile = pid // num_j          # tile index along i
    j_tile = pid %  num_j          # tile index along j

    # Offsets for the current tile
    offs_i = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # accumulator: (BLOCK_I, BLOCK_J, CHUNK_H)  in float32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J, CHUNK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k dimension in blocks
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.arange(0, BLOCK_K)
        offs_k = k_start + cur_k
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load a chunk of the hidden dimension for left and right.
        # We load CHUNK_H consecutive channels at a time.
        # ------------------------------------------------------------------
        for h_start in range(0, H, CHUNK_H):
            cur_h = tl.arange(0, CHUNK_H)
            offs_h = h_start + cur_h

            # left:  [i, k, h]  (broadcast over batch)
            left_ptrs = (
                left_ptr
                + offs_i[:, None, None] * stride_left_i
                + offs_k[None, :, None] * stride_left_k
                + offs_h[None, None, :] * stride_left_h
                + tl.zeros([1, 1, 1], dtype=tl.int64) * stride_left_b
            )
            left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0)

            # right: [j, k, h]
            right_ptrs = (
                right_ptr
                + offs_j[:, None, None] * stride_right_j
                + offs_k[None, :, None] * stride_right_k
                + offs_h[None, None, :] * stride_right_h
                + tl.zeros([1, 1, 1], dtype=tl.int64) * stride_right_b
            )
            right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0)

            # ------------------------------------------------------------------
            # Compute dot product over the k‑axis for each (i,j) pair and each h‑chunk.
            # left  : (BLOCK_I, BLOCK_K, CHUNK_H)
            # right : (BLOCK_J, BLOCK_K, CHUNK_H)
            # We need Σₖ left[i,k,h] * right[j,k,h]  ->  (BLOCK_I, BLOCK_J, CHUNK_H)
            # This is done by iterating over k inside the block and accumulating.
            # ------------------------------------------------------------------
            for kk in range(BLOCK_K):
                kmask = mask_k[kk]
                if kmask == 0:
                    continue

                l_vec = left[:, kk, :]          # (BLOCK_I, CHUNK_H)
                r_vec = right[:, kk, :]         # (BLOCK_J, CHUNK_H)

                # expand dimensions so broadcasting works:
                # l_vec[:, None, :]  -> (BLOCK_I, 1, CHUNK_H)
                # r_vec[None, :, :]  -> (1, BLOCK_J, CHUNK_H)
                acc += l_vec[:, None, :] * r_vec[None, :, :]

    # ------------------------------------------------------------------
    # Write back the accumulated result in float16 (the surrounding
    # layers will cast back to float32 anyway).
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + tl.arange(0, CHUNK_H)[None, None, :] * stride_out_h
        + tl.zeros([1, 1, 1], dtype=tl.int64) * stride_out_b
    )
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None, None] & mask_j[None, :, None])


# ------------------------------------------------------------
# Host side helper to launch the kernel
# ------------------------------------------------------------
def launch_trimul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]  (float16)
    returns out : [B, N, N, H]  (float16)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32          # fits well in shared memory
    CHUNK_H = 32           # hidden_dim is a multiple of 32 (128)

    grid = (triton.cdiv(N, BLOCK_I) * triton.cdiv(N, BLOCK_J),)

    trimul_einsum_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K, CHUNK_H=CHUNK_H,
    )
    return out


# ------------------------------------------------------------
# Complete custom kernel entry point
# ------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Args
    ----
    data : (input, mask, weights, config)
        input  : [B, N, N, dim]   (float32)
        mask   : [B, N, N]        (float32 or bool)
        weights: dict of all param tensors (float32)
        config : {'dim': int, 'hidden_dim': int, ...}

    Returns
    -------
    output : [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]

    # --------------------------------------------------------
    # 1) LayerNorm on dim
    # --------------------------------------------------------
    # LayerNorm: (x - mean) / sqrt(var + eps) * w + b
    eps = 1e-5
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    x = input_tensor
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + eps)
    x = x * w_norm + b_norm

    # --------------------------------------------------------
    # 2) Linear projections (no bias) – keep in float16 for speed
    # --------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"]        # [hidden_dim, dim]

    # (B,N,N,dim) @ (dim,hidden_dim) -> (B,N,N,hidden_dim)
    left = torch.nn.functional.linear(x, left_proj_w).to(torch.float16)
    right = torch.nn.functional.linear(x, right_proj_w).to(torch.float16)

    # --------------------------------------------------------
    # 3) Apply mask (if present)
    # --------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------
    # 4) Gating (sigmoid) – stay in float32 for stability, then cast
    # --------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate.to(torch.float16)
    right = right * right_gate.to(torch.float16)

    # --------------------------------------------------------
    # 5) Core N³ reduction via Triton
    # --------------------------------------------------------
    out = launch_trimul_einsum(left, right)          # still float16

    # --------------------------------------------------------
    # 6) Post‑processing (LayerNorm, gating, final linear)
    # --------------------------------------------------------
    out = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # LayerNorm on hidden_dim
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # Apply output gate
    out = out * out_gate

    # Final linear back to dim
    out = torch.nn.functional.linear(out, to_out_w)

    return out
