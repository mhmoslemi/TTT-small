"""
TriMul “outgoing” implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input (`x`) over the last dimension (dim).
2. Linear projections + gates (all done in FP32 Torch for simplicity):
        left  = (x @ W_left ) * mask * sigmoid(x @ W_lgate)
        right = (x @ W_right) * mask * sigmoid(x @ W_rgate)
3. The expensive part is the pair‑wise inner‑product over the third axis:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is exactly a batched matrix‑multiply:
        out_d = left_d   @ right_d.T   for every hidden channel d.
   We reshape the tensors to (B*D, N, N) and launch a Triton GEMM that
   multiplies A (M×K) with B (N×K) (i.e. B is treated transposed).
4. Apply the final LayerNorm, out‑gate and output linear (again in Torch).
5. Return FP32 tensor of shape [B, N, N, dim].

Only the N³ inner‑product is off‑loaded to Triton; everything else stays in
Torch to keep the implementation short while still giving a large speed‑up.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton batched GEMM :  C = A @ B^T   (A: (M,K), B: (N,K))
# --------------------------------------------------------------------------- #
@triton.jit
def batched_gemm_T_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A strides: (batch, M, K)
    stride_bm, stride_bn, stride_bk,  # B strides: (batch, N, K)
    stride_cm, stride_cn,          # C strides: (batch, M, N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)                     # batch index
    # pointers for this batch
    a_batch = a_ptr + pid * stride_am
    b_batch = b_ptr + pid * stride_bm
    c_batch = c_ptr + pid * stride_cm

    # block offsets
    offs_m = tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute tile coordinates
    block_m = tl.program_id(1)
    block_n = tl.program_id(2)

    # global offsets
    offs_m = block_m * BLOCK_M + offs_m
    offs_n = block_n * BLOCK_N + offs_n

    # mask for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulators
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        cur_k = k_base + offs_k
        cur_k = tl.where(cur_k < K, cur_k, 0)        # safe guard

        # load A : (M, K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am +
                            cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None], other=0.0)

        # load B : (N, K)   (we will dot with B, not B^T)
        b_ptrs = b_batch + (offs_n[:, None] * stride_bn +
                            cur_k[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_n[:, None], other=0.0)

        # dot‑product over K (both are (M,K) and (N,K))
        acc += tl.dot(a, b)

    # write back C : (M,N)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm +
                        offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------------------- #
# Helper: fused LayerNorm (FP32) – simple implementation
# --------------------------------------------------------------------------- #
def layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]  weight/bias: (C,)
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias

# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass with a Triton‑accelerated inner product.
    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor: [B, N, N, dim]   (float32)
            mask        : [B, N, N]        (float32 or bool)
            weights     : dict of torch tensors (all fp32)
            config      : dict containing "dim" and "hidden_dim"
    Returns:
        out: torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]          # = H

    # ------------------------------------------------------------------- #
    # 1) LayerNorm over dim
    # ------------------------------------------------------------------- #
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 2) Linear projections (no bias)  :  FP32
    # ------------------------------------------------------------------- #
    # shape: (B,N,N,dim) @ (dim,hidden) -> (B,N,N,hidden)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3) Apply mask (if present) – mask may be bool or fp32
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)        # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------- #
    # 4) Gates
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5) Core N³ operation – use Triton
    #    We compute for each hidden channel d:
    #        out_d = left_d @ right_d.T
    # ------------------------------------------------------------------- #
    # convert to fp16 for kernel (still accurate enough)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*hidden, N, N)
    left_flat = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_flat = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # strides (batch, M, K) etc.
    a_ptr = left_flat
    b_ptr = right_flat
    c_ptr = out_flat

    stride_am = left_flat.stride(0)
    stride_ak = left_flat.stride(2)   # K stride
    stride_bm = right_flat.stride(0)
    stride_bn = right_flat.stride(1)
    stride_bk = right_flat.stride(2)
    stride_cm = out_flat.stride(0)
    stride_cn = out_flat.stride(2)

    # launch parameters – tuned for Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden,                                     # batch dimension
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_T_kernel[grid](
        a_ptr, b_ptr, c_ptr,
        N, N, N,
        stride_am, stride_ak,
        stride_bm, stride_bn, stride_bk,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B,N,N,hidden]
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------- #
    # 6) Final LayerNorm + out_gate + output linear
    # ------------------------------------------------------------------- #
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
