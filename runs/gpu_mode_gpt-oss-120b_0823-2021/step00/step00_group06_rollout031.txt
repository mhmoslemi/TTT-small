"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Layer‑norm + three linear projections (left, right, three gates) are
   performed with regular PyTorch – they are cheap compared to the O(N³)
   core.
2. The core operation is
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix‑multiply for every hidden dimension d:
        out_d[b] = left_d[b] @ right_d[b].transpose(-1,-2)
   (left_d and right_d have shape (N,N) for each d).
3. A custom Triton kernel computes the batched GEMM for a single
   (batch, hidden‑dim) pair.  The host code launches the kernel for all
   B·hidden_dim pairs, keeping the rest of the work in PyTorch.
4. The result is normalized, gated and projected back to the original
   feature dimension with PyTorch nn.LayerNorm / Linear layers.

The kernel works for any batch size ≤2, any sequence length up to 1024,
and hidden dimensions that are a multiple of 32 (the block size).  It
produces a float‑32 tensor, but internally uses float‑16 for speed on
Hopper (SM90) GPUs.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for one (batch, hidden_dim) slice
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    # pointers
    L_ptr,               # left  (B,N,N,HD)
    R_ptr,               # right (B,N,N,HD)
    O_ptr,               # out   (B,N,N,HD)
    # problem size
    N: tl.constexpr,
    K: tl.constexpr,
    # strides (as 1‑D offsets)
    stride_L_batch, stride_L_i, stride_L_k,
    stride_R_batch, stride_R_j, stride_R_k,
    stride_O_batch, stride_O_i, stride_O_j,
    # block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute:
        O[b,i,j] = Σₖ L[b,i,k] * R[b,j,k]      (R is transposed on‑the‑fly)
    For a *single* (batch, hidden‑dim) pair.
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    # offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds (needed for the last tile)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the start of the tile for this (b,hd)
    L_tile_ptr = L_ptr + offs_m[:, None] * stride_L_i + tl.arange(0, BLOCK_K)[None, :] * stride_L_k
    R_tile_ptr = R_ptr + offs_n[None, :] * stride_R_j + tl.arange(0, BLOCK_K)[:, None] * stride_R_k

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kp in range(0, tl.cdiv(K, BLOCK_K)):
        # load a BLOCK_M×BLOCK_K tile from L and a BLOCK_K×BLOCK_N tile from R
        L = tl.load(L_tile_ptr, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - kp * BLOCK_K), other=0.0).to(tl.float16)
        R = tl.load(R_tile_ptr, mask=mask_n[None, :] & (tl.arange(0, BLOCK_K) < K - kp * BLOCK_K), other=0.0).to(tl.float16)

        # matrix‑multiply the two tiles (R is already transposed)
        acc += tl.dot(L, R, out=acc, allow_tf32=False)

        # advance pointers to next K‑tile
        L_tile_ptr += BLOCK_K * stride_L_k
        R_tile_ptr += BLOCK_K * stride_R_k

    # store the result (cast back to float16 -> float32 accumulator already)
    O_tile_ptr = O_ptr + offs_m[:, None] * stride_O_i + offs_n[None, :] * stride_O_j
    tl.store(O_tile_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: launch the kernel for the whole tensor
# ----------------------------------------------------------------------
def launch_trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right:  [B, N, N, HD]   (float16)
    returns:    [B, N, N, HD]   (float16)
    """
    B, N, _, HD = left.shape
    K = N                     # sum over the third dimension

    # output allocation (float16 – later cast to float32)
    out = torch.empty_like(left)

    # kernel block sizes – chosen for Hopper (SM90)  (must divide HD)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # strides (as scalar offsets, multiplied by element size later)
    stride_L = left.stride()   # (batch, i, k, d)
    stride_R = right.stride()
    stride_O = out.stride()

    # Convert strides from elements to bytes (required by Triton)
    elem_sz = left.element_size()   # =2 for float16
    stride_L_batch = stride_L[0] * elem_sz
    stride_L_i     = stride_L[1] * elem_sz
    stride_L_k     = stride_L[2] * elem_sz
    stride_R_batch = stride_R[0] * elem_sz
    stride_R_j     = stride_R[2] * elem_sz   # note: right tensor axes (B,N,N,HD) -> j is second N
    stride_R_k     = stride_R[1] * elem_sz   # we need k dimension (the middle N)
    stride_O_batch = stride_O[0] * elem_sz
    stride_O_i     = stride_O[1] * elem_sz
    stride_O_j     = stride_O[2] * elem_sz

    # launch one kernel per (batch, hidden_dim) pair
    for b in range(B):
        for hd in range(HD):
            L_ptr = left.data_ptr() + b * stride_L_batch + hd * left.stride(3) * elem_sz
            R_ptr = right.data_ptr() + b * stride_R_batch + hd * right.stride(3) * elem_sz
            O_ptr = out.data_ptr() + b * stride_O_batch + hd * out.stride(3) * elem_sz

            grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
            trimul_gemm_kernel[grid](
                L_ptr, R_ptr, O_ptr,
                N, K,
                stride_L_batch, stride_L_i, stride_L_k,
                stride_R_batch, stride_R_j, stride_R_k,
                stride_O_batch, stride_O_i, stride_O_j,
                BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
                num_warps=4,
            )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton core)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]        (float32)
            - mask         : [B, N, N]            (bool/float)
            - weights      : dict of parameter tensors
            - config       : {"dim": int, "hidden_dim": int}
    Returns:
        output: [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]

    # ------------------ 1. LayerNorm & linear projections (PyTorch) ------------------
    # Build lightweight modules using the supplied weight tensors.
    # Biases are present only for the LayerNorms.
    norm = nn.LayerNorm(dim, elementwise_affine=True).to(input_tensor.dtype).to(input_tensor.device)
    norm.weight.data.copy_(weights["norm.weight"])
    norm.bias.data.copy_(weights["norm.bias"])

    left_proj = nn.Linear(dim, hidden_dim, bias=False, dtype=input_tensor.dtype).to(input_tensor.device)
    left_proj.weight.data.copy_(weights["left_proj.weight"])
    right_proj = nn.Linear(dim, hidden_dim, bias=False, dtype=input_tensor.dtype).to(input_tensor.device)
    right_proj.weight.data.copy_(weights["right_proj.weight"])

    left_gate = nn.Linear(dim, hidden_dim, bias=False, dtype=input_tensor.dtype).to(input_tensor.device)
    left_gate.weight.data.copy_(weights["left_gate.weight"])
    right_gate = nn.Linear(dim, hidden_dim, bias=False, dtype=input_tensor.dtype).to(input_tensor.device)
    right_gate.weight.data.copy_(weights["right_gate.weight"])
    out_gate = nn.Linear(dim, hidden_dim, bias=False, dtype=input_tensor.dtype).to(input_tensor.device)
    out_gate.weight.data.copy_(weights["out_gate.weight"])

    to_out_norm = nn.LayerNorm(hidden_dim, elementwise_affine=True).to(input_tensor.dtype).to(input_tensor.device)
    to_out_norm.weight.data.copy_(weights["to_out_norm.weight"])
    to_out_norm.bias.data.copy_(weights["to_out_norm.bias"])

    to_out = nn.Linear(hidden_dim, dim, bias=False, dtype=input_tensor.dtype).to(input_tensor.device)
    to_out.weight.data.copy_(weights["to_out.weight"])

    # Normalization
    x = norm(input_tensor)                                 # [B,N,N,dim]

    # Linear projections
    left = left_proj(x)                                     # [B,N,N,hd]
    right = right_proj(x)                                   # [B,N,N,hd]

    # Apply mask (if present).  mask is broadcasted on the hidden dim.
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)           # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # Gating
    left_gate_val = left_gate(x).sigmoid()
    right_gate_val = right_gate(x).sigmoid()
    out_gate_val = out_gate(x).sigmoid()

    left = left * left_gate_val
    right = right * right_gate_val

    # ------------------ 2. Core O(N³) multiplication (Triton) ------------------
    # Cast to float16 for the kernel (GPU native) – the surrounding code stays fp32.
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Triton kernel computes: out = left @ right^T   (per hidden dim)
    out_fp16 = launch_trimul_gemm(left_fp16, right_fp16)   # [B,N,N,hd] float16

    out = out_fp16.to(torch.float32)                       # back to fp32

    # ------------------ 3. Post‑processing (PyTorch) ------------------
    out = to_out_norm(out)
    out = out * out_gate_val
    out = to_out(out)                                      # [B,N,N,dim]

    return out
