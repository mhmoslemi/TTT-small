"""
TriMul “outgoing” kernel (AlphaFold3)

We compute
    out[b,i,j,d] = Σ_k  left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are the masked‑and‑gated projections of the input.
The O(N³·D) contraction is performed with a custom Triton kernel that
tiles the (i,j) plane and iterates over the reduction dimension `k`.
All other operations (LayerNorm, Linear, gates) are kept in PyTorch for
readability.  The kernel works in mixed‑precision: `left`/`right` are
stored as FP16, the accumulation is FP32 and the final result is
returned as FP32.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# --------------------------------------------------------------
# Triton kernel
# --------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,       # tensors
    B, N, D,                            # static sizes
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides  (b, i, k, d)
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides (b, j, k, d)
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides   (b, i, j, d)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Compute a block of the output tensor.
    Grid layout: (batch, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    """
    b = tl.program_id(0)          # batch dimension
    pid_m = tl.program_id(1)      # block row   (i)
    pid_n = tl.program_id(2)      # block col   (j)

    # ----------------------------------------------------------
    # Offsets and masks for i, j, k, d
    # ----------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator : (M, N, D)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ----------------------------------------------------------
    # Loop over the reduction dimension k
    # ----------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k):
        k_start = kb * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---------- left block : (M, K, D) ----------
        # pointer = base + b*stride_lb + i*stride_li + k*stride_lk + d*stride_ld
        left_ptrs = left_ptr + \
            b * stride_lb + \
            offs_i[:, None, None] * stride_li + \
            offs_k[None, :, None] * stride_lk + \
            tl.arange(0, BLOCK_D)[None, None, :] * stride_ld

        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None],
                       other=0.0).to(tl.float16)

        # ---------- right block (transposed) : (K, N, D) ----------
        # we need right[j,k,d] → we load as (K, N, D) by swapping j/k strides
        right_ptrs = right_ptr + \
            b * stride_rb + \
            offs_k[:, None, None] * stride_rk + \
            offs_j[None, :, None] * stride_rj + \
            tl.arange(0, BLOCK_D)[None, None, :] * stride_rd

        right = tl.load(right_ptrs,
                        mask=mask_k[:, None, None] & mask_j[None, :, None],
                        other=0.0).to(tl.float16)

        # ---------- inner product for each d ----------
        # Loop over D in chunks of BLOCK_D (max 32) – unrolled by the compiler
        for d in range(0, BLOCK_D):
            l = left[:, :, d]          # (M, K)
            r = right[:, :, d]         # (K, N)
            acc[:, :, d] += tl.dot(l, r)   # (M, N)

    # ----------------------------------------------------------
    # Store the result
    # ----------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_ob + \
        offs_i[:, None, None] * stride_oi + \
        offs_j[None, :, None] * stride_oj + \
        tl.arange(0, BLOCK_D)[None, None, :] * stride_od

    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] &
                  mask_j[None, :, None])


# --------------------------------------------------------------
# Helper: fused LayerNorm (PyTorch) – we keep it in PyTorch
# --------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            input   – torch.Tensor [B, N, N, dim]   (float32, on CUDA)
            mask    – torch.Tensor [B, N, N]        (bool/float, on CUDA)
            weights – dict of pretrained parameters
            config  – dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"]
    )

    # ------------------------------------------------------
    # 2. Linear projections (no bias) – keep in FP32 then cast
    # ------------------------------------------------------
    # shape: [B,N,N,hidden]
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"]
    )
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"]
    )

    # ------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"]
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"]
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------
    # 5. Cast to FP16 for the heavy contraction
    # ------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------
    # 6. Allocate output buffer (FP32)
    # ------------------------------------------------------
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # ------------------------------------------------------
    # 7. Launch Triton kernel
    # ------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_D = 32                     # hidden ≤ 128 → 4 iterations max

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # stride helpers (int64 tensors)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    trimul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------
    # 8. Post‑processing (LayerNorm, gate, final linear)
    # ------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"]
    )
    out = out * out_gate          # broadcast over (B,N,N,hidden)

    # final linear back to `dim`
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"]
    )

    return out.to(torch.float32)
