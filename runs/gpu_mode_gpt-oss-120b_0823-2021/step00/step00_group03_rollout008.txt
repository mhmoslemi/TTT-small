"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Layer‑Norm on the input (PyTorch – cheap and easy to fuse).
2. Linear projections + sigmoid gates (PyTorch – dense mat‑muls are already
   highly optimized on the H200).
3. Apply the mask (if any) and the gates → the tensors *left* and *right*
   have shape   [B, N, N, H]  with H = hidden_dim.
4. The cubic “einsum”  
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is the only O(N³·H) part.  A custom Triton kernel computes it:
   * each program works on a tile (BLOCK_I × BLOCK_J) of the (i,j) plane;
   * the reduction over k is performed by a tiled loop;
   * the hidden dimension H is kept inside the accumulator (vectorised on
     tl.dot) – this yields good utilisation on Hopper (SM90) because the
     dot‑product unit works on 128‑bit (4×fp32) lanes.
5. The result is Layer‑Norm → gate → final linear projection (again in PyTorch).

Only the cubic reduction is off‑loaded to Triton; everything else stays in
PyTorch which keeps the implementation short, correct and easy to
debug.  The final tensor is returned in float32 as required.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel for the cubic reduction (the einsum)
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,               # const float*  [B, N, N, H]  (b,i,k,d)
    right_ptr,              # const float*  [B, N, N, H]  (b,j,k,d)
    out_ptr,                #       float*  [B, N, N, H]  (b,i,j,d)
    B, N, H,
    stride_left_bn, stride_left_ik, stride_left_id,      # strides for left
    stride_right_bn, stride_right_jk, stride_right_jd,   # strides for right
    stride_out_bn, stride_out_ij, stride_out_id,         # strides for out
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute
          out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    Launch grid = (ceil(N/BLOCK_I), ceil(N/BLOCK_J), B)
    """
    pid_i = tl.program_id(0)          # tile along i
    pid_j = tl.program_id(1)          # tile along j
    pid_b = tl.program_id(2)          # batch dimension

    # base indices for this tile
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    b = pid_b

    # ranges for i,j inside the tile (will be masked at the boundaries)
    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ----- accumulator: shape (BLOCK_I, BLOCK_J, H) -----------------
    acc = tl.zeros((BLOCK_I, BLOCK_J, H), dtype=tl.float32)

    # loop over k in tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tile_k in range(0, num_k_tiles):
        k_base = tile_k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------------------------------------------------------------
        # Load left  : [B, i, k, d]  -> shape (BLOCK_I, BLOCK_K, H)
        # Load right : [B, j, k, d]  -> shape (BLOCK_J, BLOCK_K, H)
        # -------------------------------------------------------------
        # left pointer arithmetic
        left_ptrs = left_ptr \
                    + b * stride_left_bn \
                    + offs_i[:, None, None] * stride_left_ik \
                    + offs_k[None, :, None] * stride_left_id \
                    + tl.arange(0, H)[None, None, :] * 1

        right_ptrs = right_ptr \
                    + b * stride_right_bn \
                    + offs_j[:, None, None] * stride_right_jk \
                    + offs_k[None, :, None] * stride_right_jd \
                    + tl.arange(0, H)[None, None, :] * 1

        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0)
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0)

        # left: (I,K,H)   right: (J,K,H)
        # we need Σ_k left[i,k,d] * right[j,k,d]
        # Compute with a dot over the K dimension for every (i,j,d)
        #   acc[i,j,d] += Σ_k left[i,k,d] * right[j,k,d]
        # Triton does not have a direct batched‑dot, so we do an explicit loop
        # over K (the inner tile is tiny – BLOCK_K is typically 32).
        for kk in range(0, BLOCK_K):
            k_ok = kk < N - k_base
            if k_ok:
                # shape (I, H)
                l = left[:, kk, :]          # (I, H)
                # shape (J, H)
                r = right[:, kk, :]         # (J, H)

                # outer product over i and j, keep H as the last dim
                # acc[i,j,d] += l[i,d] * r[j,d]
                # we can use broadcasting: (I,1,H) * (1,J,H) -> (I,J,H)
                acc += l[:, None, :] * r[None, :, :]

    # -------------------------------------------------------------
    # Write back
    # -------------------------------------------------------------
    out_ptrs = out_ptr \
               + b * stride_out_bn \
               + offs_i[:, None, None] * stride_out_ij \
               + offs_j[None, :, None] * stride_out_id \
               + tl.arange(0, H)[None, None, :] * 1

    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None])

# ----------------------------------------------------------------------
# Helper: layer‑norm that works on the last dimension (same as nn.LayerNorm)
# ----------------------------------------------------------------------
def layer_norm(tensor: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # tensor: [..., C]
    mean = tensor.mean(dim=-1, keepdim=True)
    var = tensor.var(dim=-1, unbiased=False, keepdim=True)
    norm = (tensor - mean) * torch.rsqrt(var + eps)
    return norm * weight + bias

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module with a Triton kernel for the
    cubic reduction.  All tensors are assumed to reside on the CUDA device.
    """
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]
    assert D == config["dim"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input (float32)
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"].to(input_tensor.dtype),
        bias=weights["norm.bias"].to(input_tensor.dtype),
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – we do them with torch.einsum for speed
    # ------------------------------------------------------------------
    # shape: [B,N,N,H]
    left = torch.einsum("b i j d, h d -> b i j h", x, weights["left_proj.weight"])
    right = torch.einsum("b i j d, h d -> b i j h", x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Mask (if present) – broadcast over the hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) – also applied in-place
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.einsum("b i j d, h d -> b i j h", x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.einsum("b i j d, h d -> b i j h", x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.einsum("b i j d, h d -> b i j h", x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cubic reduction via Triton
    # ------------------------------------------------------------------
    # allocate output
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=input_tensor.device)

    # Triton launch parameters
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        B,
    )

    # compute strides (contiguous tensors → simple pattern)
    stride_left = left.stride()   # (BN, iN, kN, dN)
    stride_right = right.stride()
    stride_out = out.stride()

    trimul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_bn=stride_left[0],
        stride_left_ik=stride_left[1],
        stride_left_id=stride_left[2],
        stride_right_bn=stride_right[0],
        stride_right_jk=stride_right[1],
        stride_right_jd=stride_right[2],
        stride_out_bn=stride_out[0],
        stride_out_ij=stride_out[1],
        stride_out_id=stride_out[2],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output processing (layer‑norm, gating, final linear)
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"].to(out.dtype),
        bias=weights["to_out_norm.bias"].to(out.dtype),
    )
    out = out * out_gate

    # final linear (hidden -> dim)
    out = torch.einsum("b i j h, d h -> b i j d", out, weights["to_out.weight"])

    return out
