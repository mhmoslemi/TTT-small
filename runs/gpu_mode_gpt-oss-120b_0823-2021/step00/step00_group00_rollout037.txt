"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the bilinear reduction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batch of hidden‑dim matrix multiplications (N×N)·(N×N) → (N×N).
We fuse this reduction with a custom Triton kernel that performs a
blocked GEMM on the flattened tensor of shape (B*H, N, N) where H = hidden_dim.
All other operations (LayerNorm, Linear, gating, final LayerNorm/Linear)
are kept in PyTorch for simplicity.

The kernel works in FP16 (which is the native type of the model) and
produces FP32 results that are later fed to the remaining torch ops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matmul (A @ B) for tensors of shape (B*H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def trilu_trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    outer, N,                           # outer = B*H, matrix size N
    stride_am, stride_ak, stride_an,    # a strides (outer,N,N)
    stride_bk, stride_bn, stride_bm,    # b strides (outer,N,N) (B,K,M) order for GEMM
    stride_cm, stride_cn,               # c strides (outer,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # each program works on one (outer, block_M, block_N) tile
    outer_id = pid // tl.cdiv(N, BLOCK_M)          # which (B*H) we are on
    tile_id   = pid % tl.cdiv(N, BLOCK_M)          # which tile in M dimension
    block_m = tile_id * BLOCK_M
    block_n = (pid % tl.cdiv(N, BLOCK_N)) * BLOCK_N   # actually we use 2‑D grid later

    # compute start offsets
    offs_m = block_m + tl.arange(0, BLOCK_M)
    offs_n = block_n + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointer to the start of this block in A and B
    a_ptrs = a_ptr + outer_id * stride_am + offs_m[:, None] * stride_ak + tl.arange(0, BLOCK_K)[None, :] * stride_an
    b_ptrs = b_ptr + outer_id * stride_bm + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K
        cur_k = tl.minimum(tl.full((1,), BLOCK_K, dtype=tl.int32), N - k_off)
        a = tl.load(a_ptrs + k_off * stride_an, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < cur_k), other=0.0)
        b = tl.load(b_ptrs + k_off * stride_bk, mask=(tl.arange(0, BLOCK_K)[:, None] < cur_k) & mask_n[None, :], other=0.0)
        # promote to fp16→fp32 inside dot
        acc += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_an
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c_ptrs = c_ptr + outer_id * stride_cm + offs_m[:, None] * stride_cn + offs_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side helper to launch the kernel
# ----------------------------------------------------------------------
def trilu_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a, b : (B*H, N, N) float16 tensors, contiguous
    returns c : (B*H, N, N) float32
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, N, _ = a.shape
    c = torch.empty_like(a, dtype=torch.float32)

    # strides in elements (not bytes) for Triton
    stride_am, stride_ak, stride_an = a.stride()
    stride_bm, stride_bk, stride_bn = b.stride()
    stride_cm, stride_cn = c.stride()

    # block sizes – tuned for Hopper, can be changed if needed
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (B_H * grid_m * grid_n,)

    trilu_trimul_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        outer=B_H,
        N=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_an=stride_an,
        stride_bm=stride_bm,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    Only the O(N³) reduction is executed in Triton.
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim) – torch implementation
    # ------------------------------------------------------------------
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # mean / var per element over dim
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias          # shape (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) – still in torch (lightweight)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"])).to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Prepare tensors for the Triton GEMM:
    #    left  : (B,N,N,hidden)  -> (B*hidden, N, N)
    #    right : (B,N,N,hidden)  -> (B*hidden, N, N) with K dimension transposed
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()           # (B, hidden, N, N)
    right = right.permute(0, 3, 2, 1).contiguous()         # (B, hidden, N, N)

    B_H = B * hidden_dim
    left_flat = left.view(B_H, N, N)
    right_flat = right.view(B_H, N, N)

    # ------------------------------------------------------------------
    # 6) Triton batched matmul: out_flat = left_flat @ right_flat  (fp16→fp32)
    # ------------------------------------------------------------------
    out_flat = trilu_matmul_fp16(left_flat, right_flat)   # (B*hidden, N, N) float32

    # ------------------------------------------------------------------
    # 7) Reshape back to (B,N,N,hidden)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 8) Final LayerNorm + gating + projection (torch)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # LayerNorm over hidden dim
    mean2 = out.mean(-1, keepdim=True)
    var2 = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean2) / torch.sqrt(var2 + eps)
    out = out * to_out_norm_weight + to_out_norm_bias

    out = out * out_gate.to(torch.float32)                # gating (broadcast)

    # final linear back to dim
    out = F.linear(out, weights["to_out.weight"]).to(torch.float32)

    return out
