"""
TriMul (outgoing) – fast Triton implementation
================================================
The forward pass consists of

1️⃣ Layer‑norm on the 4‑D input                (torch)
2️⃣ Two linear projections + sigmoid gates     (torch, bias‑free)
3️⃣ Element‑wise mask & gate multiplication
4️⃣ Core “tri‑multiplicative” reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply for every hidden channel *d*.
   It is the only O(N³) part and is implemented in Triton.
5️⃣ Output Layer‑norm, gating & final linear   (torch)

The Triton kernel computes a tile of the output tensor
(i‑block  ×  j‑block) for a set of hidden channels.
It loops over the reduction dimension *k* in blocks, accumulates the
dot‑product with `tl.dot` (float16 → float32 accumulator) and finally
stores the result (float32).  All other operations stay in PyTorch so
that the implementation works for any batch size, sequence length and
hidden dimension allowed by the test matrix.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel:   out = Σ_k left[...,i,k,d] * right[...,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,          # [B, N, N, H]  (B,N,K,H) –‑> K is the reduction axis
    right_ptr,         # [B, N, N, H]  (B,N,K,H)
    out_ptr,           # [B, N, N, H]
    # strides
    stride_lb, stride_lk, stride_lh,  # left  strides (B,N,K,H)
    stride_rb, stride_rk, stride_rh,  # right strides (B,N,K,H)
    stride_ob, stride_oi, stride_oj, stride_oh,  # out strides (B,N,N,H)
    # dimensions
    B, N, H,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(axis=0)                # linear id over (b, i_tile, j_tile, h_tile)
    b   = pid // (tl.cdiv(N, BLOCK_I) * tl.cdiv(N, BLOCK_J) * tl.cdiv(H, BLOCK_H))
    tmp = pid % (tl.cdiv(N, BLOCK_I) * tl.cdiv(N, BLOCK_J) * tl.cdiv(H, BLOCK_H))
    i_tile = tmp // (tl.cdiv(N, BLOCK_J) * tl.cdiv(H, BLOCK_H))
    tmp    = tmp % (tl.cdiv(N, BLOCK_J) * tl.cdiv(H, BLOCK_H))
    j_tile = tmp // tl.cdiv(H, BLOCK_H)
    h_tile = tmp % tl.cdiv(H, BLOCK_H)

    # block offsets
    i_off = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)
    h_off = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)

    # mask out‑of‑bounds
    i_mask = i_off < N
    j_mask = j_off < N
    h_mask = h_off < H

    # initialise accumulator (float32)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over k in BLOCK_K steps
    for k_start in range(0, N, BLOCK_K):
        k_off = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # -------- load left[b, i, k, h]  --------
        # left layout: (B, N_i, N_k, H)
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + i_off[:, None, None] * stride_lk   # i dim
            + k_off[None, :, None] * stride_lk   # k dim (same stride as i because contiguous)
            + h_off[None, None, :] * stride_lh
        )
        left = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :] & h_mask[None, None, :],
                       other=0.0).to(tl.float32)

        # -------- load right[b, j, k, h] --------
        # right layout: (B, N_j, N_k, H)
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + j_off[:, None, None] * stride_rk   # j dim
            + k_off[None, :, None] * stride_rk   # k dim
            + h_off[None, None, :] * stride_rh
        )
        right = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_mask[None, :] & h_mask[None, None, :],
                        other=0.0).to(tl.float32)

        # -------- accumulate dot over k ----------
        # acc[i,j,h] += Σ_k left[i,k,h] * right[j,k,h]
        # we can use tl.dot on the k‑axis for each (i,j,h) triple.
        # Reshape to (BLOCK_I*BLOCK_J, BLOCK_K) and (BLOCK_J*BLOCK_I, BLOCK_K) not needed –
        # simply perform element‑wise multiply & sum.
        acc += tl.dot(left, right)   # (BLOCK_I, BLOCK_J, BLOCK_H) because dot works on last dim

    # -------- store result ----------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + i_off[:, None, None, None] * stride_oi
        + j_off[None, :, None, None] * stride_oj
        + h_off[None, None, :, None] * stride_oh
    )
    tl.store(out_ptrs, acc.to(tl.float32),
             mask=i_mask[:, None, None, None] & j_mask[None, :, None, None] & h_mask[None, None, :, None])

# ----------------------------------------------------------------------
# Helper – launch Triton kernel
# ----------------------------------------------------------------------
def triton_trimul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, H]  (float16 or float32)
    returns out  : [B, N, N, H]  (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in elements (not bytes)
    stride_lb, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # --------------------------------------------------------------
    # Tile sizes – tuned for Hopper (sm_90) and the given dimensions.
    # --------------------------------------------------------------
    BLOCK_I = 64   # along seq dim i
    BLOCK_J = 64   # along seq dim j
    BLOCK_K = 32   # reduction block
    BLOCK_H = 32   # hidden channel block (must divide hidden_dim or be padded)

    total_blocks_i = triton.cdiv(N, BLOCK_I)
    total_blocks_j = triton.cdiv(N, BLOCK_J)
    total_blocks_h = triton.cdiv(H, BLOCK_H)

    total_programs = B * total_blocks_i * total_blocks_j * total_blocks_h
    grid = (total_programs,)

    triton.runtime.driver.active.get_current_context().set_active_device(left.device.index)

    trimul_outgoing_kernel[grid](
        left_ptr        = left,
        right_ptr       = right,
        out_ptr         = out,
        stride_lb       = stride_lb,
        stride_lk       = stride_lk,
        stride_lh       = stride_lh,
        stride_rb       = stride_rb,
        stride_rk       = stride_rk,
        stride_rh       = stride_rh,
        stride_ob       = stride_ob,
        stride_oi       = stride_oi,
        stride_oj       = stride_oj,
        stride_oh       = stride_oh,
        B               = B,
        N               = N,
        H               = H,
        BLOCK_I         = BLOCK_I,
        BLOCK_J         = BLOCK_J,
        BLOCK_K         = BLOCK_K,
        BLOCK_H         = BLOCK_H,
    )
    return out

# ----------------------------------------------------------------------
# Full custom kernel (entry point)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul (outgoing) module.
    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask         : [B, N, N]       (float32 or bool)
        - weights      : dict of torch.Tensor (already on device)
        - config       : dict with keys  dim, hidden_dim
    Returns
    -------
    output : torch.Tensor  [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    device = input_tensor.device
    dtype = torch.float32

    # ------------------- 1) LayerNorm on dim -----------------------
    # weight / bias are provided
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------- 2) Linear projections --------------------
    left = F.linear(
        x,
        weight=weights["left_proj.weight"],   # (hidden_dim, dim)
        bias=None,
    )
    right = F.linear(
        x,
        weight=weights["right_proj.weight"],
        bias=None,
    )

    # 3) Apply mask (if any)
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # 4) Gates
    left_gate = torch.sigmoid(
        F.linear(x, weight=weights["left_gate.weight"], bias=None)
    )
    right_gate = torch.sigmoid(
        F.linear(x, weight=weights["right_gate.weight"], bias=None)
    )
    out_gate = torch.sigmoid(
        F.linear(x, weight=weights["out_gate.weight"], bias=None)
    )

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # 5) Core TriMul – Triton kernel (use float16 for speed)
    # ----------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    out = triton_trimul_outgoing(left_h, right_h)     # float32 result

    # ----------------------------------------------------------------
    # 6) Output layer‑norm, gating and final linear
    # ----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weight=weights["to_out.weight"], bias=None)

    return out.to(dtype)
