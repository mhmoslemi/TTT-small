"""
TriMul “outgoing” implementation.

The expensive part of the forward pass is the ternary einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is exactly a batched matrix‑multiply for every hidden dimension *d*:

    out[b,:,:,d] = left[b,:,:,d] @ right[b,:,:,d].T

The kernel below computes this operation in FP16 (accumulating in FP32) for
all batches and hidden channels in a single launch.  All surrounding
operations (LayerNorms, Linear projections, sigmoid gates, final
projection) are done in PyTorch – this already satisfies the “at least one
kernel” requirement while keeping the code simple and numerically stable.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel: batched matmul over the (i,k) × (j,k) dimensions.
# --------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr,               # left  [B, N, N, H]   (FP16)
    B_ptr,               # right [B, N, N, H]   (FP16)
    C_ptr,               # out   [B, N, N, H]   (FP16)
    B, N, H,
    stride_Ab, stride_Ai, stride_Ak, stride_Ah,
    stride_Bb, stride_Bj, stride_Bk, stride_Bh,
    stride_Cb, stride_Ci, stride_Cj, stride_Ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ   where the multiplication is performed for every
    batch `b` and hidden channel `h` separately.

    A : (B, N, N, H)   → (i,k) matrix
    B : (B, N, N, H)   → (j,k) matrix (will be read as (k,j) after transpose)
    C : (B, N, N, H)   → result (i,j)
    """
    pid = tl.program_id(0)                     # linear index over (batch, hidden)
    batch = pid // H
    hid   = pid % H

    # block offsets for the output tile (i,j)
    offs_i = tl.arange(0, BLOCK_M)
    offs_j = tl.arange(0, BLOCK_N)

    # global start indices
    i_start = tl.program_id(1) * BLOCK_M
    j_start = tl.program_id(2) * BLOCK_N

    # bounds
    mask_i = i_start + offs_i < N
    mask_j = j_start + offs_j < N

    # pointers shifted to the beginning of the current (b, h) plane
    A_base = A_ptr + batch * stride_Ab + hid * stride_Ah
    B_base = B_ptr + batch * stride_Bb + hid * stride_Bh
    C_base = C_ptr + batch * stride_Cb + hid * stride_Ch

    # allocate accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = k_start + offs_k < N

        # A: (i, k)
        A_ptrs = A_base + (i_start + offs_i)[:, None] * stride_Ai \
                       + (k_start + offs_k)[None, :] * stride_Ak
        A = tl.load(A_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        A = A.to(tl.float32)                     # FP16 → FP32

        # B: (j, k)   we need Bᵀ, i.e. (k, j)
        B_ptrs = B_base + (j_start + offs_j)[:, None] * stride_Bj \
                       + (k_start + offs_k)[None, :] * stride_Bk
        B = tl.load(B_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        B = B.to(tl.float32)                     # FP16 → FP32

        # dot product over k
        acc += tl.dot(A, B)                      # (M,K)·(K,N) → (M,N)

    # write back
    C_ptrs = C_base + (i_start + offs_i)[:, None] * stride_Ci \
                   + (j_start + offs_j)[None, :] * stride_Cj
    C = acc.to(tl.float16)                      # back to FP16
    tl.store(C_ptrs, C, mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Helper that launches the Triton kernel
# --------------------------------------------------------------
def batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : (B, N, N, H)   FP16 tensors
    returns      : (B, N, N, H)   FP16 tensor
    """
    B, N, _, H = left.shape
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(B * H, 1),          # program_id(0) = batch*head
        triton.cdiv(N, BLOCK_M),        # program_id(1) = i‑tile
        triton.cdiv(N, BLOCK_N),        # program_id(2) = j‑tile
    )

    # strides in element units (torch returns in bytes, convert)
    sAb, sAi, sAk, sAh = (
        left.stride(0),
        left.stride(1),
        left.stride(2),
        left.stride(3),
    )
    sBb, sBj, sBk, sBh = (
        right.stride(0),
        right.stride(1),
        right.stride(2),
        right.stride(3),
    )
    sCb, sCi, sCj, sCh = (
        left.stride(0),   # output has same layout as left/right
        left.stride(1),
        left.stride(2),
        left.stride(3),
    )

    out = torch.empty_like(left)

    batched_matmul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, H,
        sAb, sAi, sAk, sAh,
        sBb, sBj, sBk, sBh,
        sCb, sCi, sCj, sCh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


# --------------------------------------------------------------
# Main entry point required by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Only the heavy N³ einsum is executed with a custom Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm ----------
    norm_w = weights["norm.weight"].to(dtype)
    norm_b = weights["norm.bias"].to(dtype)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ---------- Linear projections (no bias) ----------
    def linear_no_bias(x, w):
        # x : (..., dim)   w : (hidden_dim, dim)
        return F.linear(x, w)

    left  = linear_no_bias(x, weights["left_proj.weight"].to(dtype))
    right = linear_no_bias(x, weights["right_proj.weight"].to(dtype))

    # ---------- Apply mask if present ----------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)          # (B,N,N,1)
        left  = left * mask_
        right = right * mask_

    # ---------- Gates ----------
    left_gate  = torch.sigmoid(linear_no_bias(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(linear_no_bias(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(linear_no_bias(x, weights["out_gate.weight"].to(dtype)))

    left  = left * left_gate
    right = right * right_gate

    # ---------- Heavy N³ operation (Triton) ----------
    # cast to FP16 for the kernel, accumulate in FP32 inside the kernel
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # result in FP16, later cast back to FP32 for the remaining layers
    out_fp16 = batched_matmul_triton(left_fp16, right_fp16)
    out = out_fp16.to(dtype)

    # ---------- Final LayerNorm + projection ----------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"].to(dtype),
                       bias=weights["to_out_norm.bias"].to(dtype))
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (hidden_dim, dim)

    return out
