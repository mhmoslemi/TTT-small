"""
TriMul “outgoing” implementation (forward only).

The heavy part is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We keep all pre‑ and post‑processing (LayerNorm, Linear, gating) in
torch for simplicity and launch a Triton kernel that performs the
reduction over **k**.  The kernel works on a 2‑D grid of (i,j) tiles
and reduces a tiled **k** dimension in a loop, accumulating a
(B, TILE_I, TILE_J, D) buffer in FP32.  The final result is stored as
float32 (the model uses float32 for the final output, although the
kernel can operate in mixed precision).

The kernel is written for arbitrary batch size B (1 or 2), sequence
length N (up to 1024) and hidden dimension H (typically 128).  All
tensors are expected to be contiguous in memory; strides are passed
explicitly to keep the code generic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr,               # const float*  [B,N,N,H]  left  (i,k)
    right_ptr,              # const float*  [B,N,N,H]  right (j,k)
    out_ptr,                # float*        [B,N,N,H]  output (i,j)
    B, N, H,                # int64 scalars
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    The kernel is launched with a 2‑D grid (i‑tile, j‑tile) and loops
    over k‑tiles.
    """
    pid_i = tl.program_id(0)          # tile index along i
    pid_j = tl.program_id(1)          # tile index along j
    pid_b = tl.program_id(2)          # batch index (grid‑dim 2)

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    b = pid_b

    i_range = i_base + tl.arange(0, BLOCK_I)
    j_range = j_base + tl.arange(0, BLOCK_J)
    d_range = tl.arange(0, BLOCK_D)

    i_mask = i_range < N
    j_mask = j_range < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k dimension
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K
        k_range = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # ------------------------------------------------------------------
        # load left[b,i,k,d]  – shape (I, K, D)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_batch
            + i_range[:, None, None] * stride_left_i
            + k_range[None, :, None] * stride_left_k
            + d_range[None, None, :] * stride_left_d
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None, None] & k_mask[None, :, None],
            other=0.0,
        )   # (I, K, D)

        # ------------------------------------------------------------------
        # load right[b,j,k,d] – shape (J, K, D)
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_right_batch
            + j_range[:, None, None] * stride_right_j
            + k_range[None, :, None] * stride_right_k
            + d_range[None, None, :] * stride_right_d
        )
        right = tl.load(
            right_ptrs,
            mask=j_mask[:, None, None] & k_mask[None, :, None],
            other=0.0,
        )   # (J, K, D)

        # ------------------------------------------------------------------
        # element‑wise product and reduction over k
        # left  : (I, K, D) → broadcast to (I, J, K, D)
        # right : (J, K, D) → broadcast to (I, J, K, D)
        # ------------------------------------------------------------------
        prod = left[:, None, :, :] * right[None, :, :, :]   # (I, J, K, D)
        acc += tl.sum(prod, axis=2)                       # sum over K → (I, J, D)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_batch
        + i_range[:, None, None] * stride_out_i
        + j_range[None, :, None] * stride_out_j
        + d_range[None, None, :] * stride_out_d
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float32),
        mask=i_mask[:, None, None] & j_mask[None, :, None],
    )


# ----------------------------------------------------------------------
# Host‑side helper
# ----------------------------------------------------------------------
def _launch_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: [B,N,N,H] (float32)
    returns out:   [B,N,N,H] (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # --------------------------------------------------------------
    # strides (in elements, not bytes)
    # --------------------------------------------------------------
    s_left = left.stride()   # (B,N,N,H)
    s_right = right.stride()
    s_out = out.stride()

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    BLOCK_D = 32   # H ≤ 384, 32 works (will process ceil(H/32) tiles)

    grid = (
        triton.cdiv(N, BLOCK_I),       # i‑tiles
        triton.cdiv(N, BLOCK_J),       # j‑tiles
        B,                             # batch dimension
    )

    _trimul_einsum_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],   # note: right stride over j is the same as left stride over i
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full custom kernel (the entry point required by the evaluation harness)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Fast forward‑only TriMul (outgoing) implementation.

    Arguments
    ---------
    data : (input, mask, weights, config)
        input  – Tensor [B, N, N, dim] (float32)
        mask   – Tensor [B, N, N]      (bool or float, broadcastable)
        weights – dict with all Linear/LayerNorm parameters
        config  – dict containing "dim" and "hidden_dim"

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    def linear(x, w):
        # x: [..., dim] , w: [hidden, dim]
        return torch.nn.functional.linear(x, w)   # -> [..., hidden]

    left = linear(x, weights["left_proj.weight"])
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core TriMul einsum – Triton
    # ------------------------------------------------------------------
    # left/right are B,N,N,hidden
    out = _launch_triton(left, right)   # float32 result, shape [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Post‑process: LayerNorm, out_gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # -> [B,N,N,dim]

    return out
