"""
TriMul “outgoing” kernel (AlphaFold3).

The module first applies LayerNorm, linear projections and sigmoid gates in
regular PyTorch (these are cheap compared to the N³ reduction).  The expensive
part

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is executed in Triton.  For each hidden dimension ``d`` the operation is a
matrix‑multiply between the slice ``left[...,d]`` (shape B×N×N) and the
transpose of ``right[...,d]``.  The kernel loops over the reduction dimension
``k`` in blocks, accumulates the partial results and finally writes the
(B, i, j, d) block to global memory.

The implementation:
* supports arbitrary batch size (1‑2) and sequence length up to 1024;
* works with both masked and un‑masked inputs;
* uses mixed‑precision (FP16 for the heavy reduction) but returns FP32;
* runs on Hopper (SM‑90) with typical block sizes tuned for the H200.
"""

import torch
import triton
import triton.language as tl
from torch import nn

# ----------------------------------------------------------------------------- #
# Triton kernel that computes the N³ reduction for a block of (i, j) pairs.
# ----------------------------------------------------------------------------- #
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, H,                                     # problem size
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    for a tile (i∈[i0,i0+BLOCK_M), j∈[j0,j0+BLOCK_N), h∈[h0,h0+BLOCK_H).
    """
    pid_b = tl.program_id(0)          # batch
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Accumulator: (BLOCK_M, BLOCK_N, BLOCK_H)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # Loop over reduction dimension k in blocks
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kk * BLOCK_K

        # ----- load a tile of left (M x K x H) -----
        offs_i = i0 + tl.arange(0, BLOCK_M)
        offs_k = k0 + tl.arange(0, BLOCK_K)
        offs_h = tl.arange(0, BLOCK_H)

        mask_i = offs_i < N
        mask_k = offs_k < N

        left_ptrs = (
            left_ptr
            + pid_b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_h[None, None, :] * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )  # (M, K, H) in fp16 -> promote later

        # ----- load a tile of right (N x K x H) -----
        offs_j = j0 + tl.arange(0, BLOCK_N)
        mask_j = offs_j < N

        right_ptrs = (
            right_ptr
            + pid_b * stride_right_b
            + offs_j[:, None, None] * stride_right_j
            + offs_k[None, :, None] * stride_right_k
            + offs_h[None, None, :] * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )  # (N, K, H)

        # ----- accumulate over k -----
        # For each hidden channel h we perform a matrix‑multiply:
        #   (M,K) * (N,K)^T  -> (M,N)
        # Using tl.dot which sums over K.
        for h in range(0, BLOCK_H):
            left_h = left[:, :, h]   # (M, K)
            right_h = right[:, :, h] # (N, K)

            # dot with transposed B (i.e. sum over K)
            out_h = tl.dot(left_h, right_h, trans_b=True)  # (M, N)

            acc[:, :, h] += out_h

    # ----- write result -----
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N

    out_ptrs = (
        out_ptr
        + pid_b * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_h[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None],
    )


# ----------------------------------------------------------------------------- #
# Helper functions (LayerNorm & Linear) – stay in PyTorch for simplicity
# ----------------------------------------------------------------------------- #
def _layer_norm(x, weight, bias, eps=1e-5):
    # x: (..., C)
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


def _linear(x, weight):
    # weight: (out_features, in_features)
    return torch.nn.functional.linear(x, weight)  # bias already folded into model


# ----------------------------------------------------------------------------- #
# Entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    assert hidden_dim % 32 == 0, "hidden_dim should be multiple of BLOCK_H (32)."

    # --------------------------------------------------------------------- #
    # 1. LayerNorm + Linear projections (all in torch, fp32)
    # --------------------------------------------------------------------- #
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,dim)

    left = _linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = _linear(x, weights["right_proj.weight"])

    # mask handling (optional)
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # gates
    left_gate = torch.sigmoid(_linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(_linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(_linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 2. Triton kernel for the N³ reduction
    # --------------------------------------------------------------------- #
    # Cast to fp16 for the heavy reduction
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # Block sizes – tuned for Hopper
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 32

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_h=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_j=stride_right[1],
        stride_right_k=stride_right[2],
        stride_right_h=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_h=stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # --------------------------------------------------------------------- #
    # 3. Final LayerNorm, gating and output projection (torch)
    # --------------------------------------------------------------------- #
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate  # broadcast over hidden dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
