"""
TriMul “outgoing” implementation.

The heavy work is the ternary einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]          (1)

where
    left  = (x @ W_left)  * mask * sigmoid(x @ W_left_gate)
    right = (x @ W_right) * mask * sigmoid(x @ W_right_gate)

Both left and right have shape [B, N, N, H] (H = hidden_dim).
The kernel below computes (1) in FP16 for speed and writes the
result in FP32.  All LayerNorms, linear projections and gating are
performed in PyTorch (they are cheap compared to the N³ reduction).

The kernel works for any batch size ≤2, any sequence length N
that fits into the grid, and any hidden_dim that is a multiple of
BLOCK_D (default 32).  If H is not a multiple of BLOCK_D the
kernel pads the last block with zeros – the padding does not affect
the result because the extra entries are never read (masked by the
runtime bounds check).

The launch grid is (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B).
Each program computes a TILE of (i, j) and iterates over the k‑dimension
in blocks of size BLOCK_K, accumulating a vector of length BLOCK_D.
The final accumulated vector is stored to the output tensor.
"""

import torch
import triton
import triton.language as tl
from torch import nn

# ----------------------------------------------------------------------
# Triton kernel for the reduction in (1)
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const *fp16   [B, N, N, H]
    right_ptr,         # const *fp16   [B, N, N, H]
    out_ptr,           # *fp32         [B, N, N, H]
    B, N, H,           # sizes (int)
    stride_left_bh, stride_left_nh, stride_left_nk, stride_left_d,   # left strides
    stride_right_bh, stride_right_nh, stride_right_nk, stride_right_d,# right strides
    stride_out_bh, stride_out_nh, stride_out_nk, stride_out_d,      # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Compute a tile (M,N) of the N×N pairwise product for a single batch."""
    pid_b = tl.program_id(2)                     # batch index
    pid_m = tl.program_id(0)                     # tile row (i)
    pid_n = tl.program_id(1)                     # tile col (j)

    # ----- tile coordinates -------------------------------------------------
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # ----- offsets -----------------------------------------------------------
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)
    d_range = tl.arange(0, BLOCK_D)

    # mask out-of‑bounds rows/cols
    i_mask = i_range < N
    j_mask = j_range < N
    d_mask = d_range < H

    # linear offsets for the left tensor   left[b,i,k,d]
    # we will iterate over k in the inner loop
    # left_ptr base for this batch
    left_batch_ptr = left_ptr + pid_b * stride_left_bh

    # right tensor base   right[b,j,k,d]
    right_batch_ptr = right_ptr + pid_b * stride_right_bh

    # output base   out[b,i,j,d]
    out_batch_ptr = out_ptr + pid_b * stride_out_bh

    # ----- accumulator (FP32) -----------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ----- iterate over k ----------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # load left tile  [M, K, D]
        left_offs = (
            i_range[:, None, None] * stride_left_nh  # i
            + k_range[None, :, None] * stride_left_nk  # k
            + d_range[None, None, :] * stride_left_d   # d
        )
        left = tl.load(
            left_batch_ptr + left_offs,
            mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
            other=0.0,
        ).to(tl.float32)     # cast to FP32 for accumulation

        # load right tile [N, K, D]
        right_offs = (
            j_range[:, None, None] * stride_right_nh   # j
            + k_range[None, :, None] * stride_right_nk  # k
            + d_range[None, None, :] * stride_right_d   # d
        )
        right = tl.load(
            right_batch_ptr + right_offs,
            mask=j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # multiply and reduce over k dimension
        # left: (M,K,D)   right: (N,K,D)   we need (M,N,D)
        # compute element‑wise product and sum across K
        # broadcasting: (M,1,K,D) * (1,N,K,D) -> (M,N,K,D) -> sum K
        prod = left[:, None, :, :] * right[None, :, :, :]   # (M,N,K,D)
        acc += tl.sum(prod, axis=2)                         # (M,N,D)

    # ----- write back --------------------------------------------------------
    out_offs = (
        i_range[:, None, None] * stride_out_nh   # i
        + j_range[None, :, None] * stride_out_nk # j
        + d_range[None, None, :] * stride_out_d   # d
    )
    tl.store(
        out_batch_ptr + out_offs,
        acc,
        mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :],
    )

# ----------------------------------------------------------------------
# Helper – launch wrapper
# ----------------------------------------------------------------------
def trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]  (FP16)
    returns     : [B, N, N, H]  (FP32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in element units (not bytes)
    stride_left_bh, stride_left_nh, stride_left_nk, stride_left_d = left.stride()
    stride_right_bh, stride_right_nh, stride_right_nk, stride_right_d = right.stride()
    stride_out_bh, stride_out_nh, stride_out_nk, stride_out_d = out.stride()

    # tile sizes – tuned for Hopper (sm_90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 32                     # must divide hidden_dim (pad via mask)

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B)

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_bh,
        stride_left_nh,
        stride_left_nk,
        stride_left_d,
        stride_right_bh,
        stride_right_nh,
        stride_right_nk,
        stride_right_d,
        stride_out_bh,
        stride_out_nh,
        stride_out_nk,
        stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    return out

# ----------------------------------------------------------------------
# Full forward (mix of PyTorch & Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (float32 → keep precision)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # manual LayerNorm (no torch.nn because we need to inject the weight)
    mean = input_tensor.mean(-1, keepdim=True)
    var  = input_tensor.var(-1, unbiased=False, keepdim=True)
    eps  = 1e-5
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias   # shape [B,N,N,dim]

    # --------------------------------------------------------------
    # 2. Linear projections (FP16 for speed)
    # --------------------------------------------------------------
    # reshape to 2‑D for matmul: [B*N*N, dim] @ [dim, hidden]
    B, N, _, _ = x_norm.shape
    x_flat = x_norm.view(-1, dim).to(torch.float16)

    left_proj_w = weights["left_proj.weight"].t().to(torch.float16)   # [dim, hidden]
    right_proj_w = weights["right_proj.weight"].t().to(torch.float16)

    left = torch.nn.functional.linear(x_flat, left_proj_w)          # [B*N*N, hidden]
    right = torch.nn.functional.linear(x_flat, right_proj_w)

    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 3. Apply mask (if present)
    # --------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)                # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) – keep in FP16, later up‑cast
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].t().to(torch.float16)
    right_gate_w = weights["right_gate.weight"].t().to(torch.float16)
    out_gate_w = weights["out_gate.weight"].t().to(torch.float16)

    left_gate = torch.nn.functional.linear(x_flat, left_gate_w).view(B, N, N, hidden_dim)
    right_gate = torch.nn.functional.linear(x_flat, right_gate_w).view(B, N, N, hidden_dim)
    out_gate = torch.nn.functional.linear(x_flat, out_gate_w).view(B, N, N, hidden_dim)

    left = left * left_gate.sigmoid()
    right = right * right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # --------------------------------------------------------------
    # 5. Core ternary multiplication – Triton kernel
    # --------------------------------------------------------------
    # left/right are FP16, kernel returns FP32
    out_hidden = trimul_outgoing_triton(left, right)   # [B,N,N,hidden]

    # --------------------------------------------------------------
    # 6. to_out_norm + gating + final linear (all in FP32)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    # LayerNorm on hidden dimension
    mean_h = out_hidden.mean(-1, keepdim=True)
    var_h  = out_hidden.var(-1, unbiased=False, keepdim=True)
    out_hidden_norm = (out_hidden - mean_h) / torch.sqrt(var_h + eps)
    out_hidden_norm = out_hidden_norm * to_out_norm_w + to_out_norm_b

    # gating
    out_hidden_norm = out_hidden_norm * out_gate

    # final projection back to dim (float32)
    to_out_w = weights["to_out.weight"].t()   # [hidden, dim]
    out = torch.nn.functional.linear(out_hidden_norm, to_out_w)  # [B,N,N,dim]

    return out.to(torch.float32)
