"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy work of the module is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiply per channel `d`
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].transpose(-2,-1)

The kernel below performs this multiplication in Triton.
All LayerNorms, linear projections and gates are kept in PyTorch (they are
memory‑bound and cheap compared with the O(N³·D) contraction).  The kernel
receives the already‑projected, masked and gated tensors `left` and `right`
and writes the result into `out`.  The final linear layer and output‑norm are
also done in PyTorch; the kernel therefore satisfies the requirement of
“at least part of the operations in a kernel”.

The implementation:
* Tiles the output matrix (i‑j plane) with BLOCK_M × BLOCK_N threads.
* Loops over the reduction dimension `k` in BLOCK_K‑sized chunks.
* Loads `left` and `right` in FP16 (the reference uses bfloat16) and
  accumulates in FP32 for numerical stability.
* The kernel is launched with a third grid dimension that flattens
  (batch, channel) → pid_z, so each (batch, d) pair works on its own
  independent matrix multiply.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # const *fp16   [B, N, N, D]
    right_ptr,              # const *fp16   [B, N, N, D]
    out_ptr,                #       *fp32   [B, N, N, D]
    B, N, D,                # int32 scalars
    stride_left_bn, stride_left_nk, stride_left_kd,
    stride_right_bn, stride_right_nk, stride_right_kd,
    stride_out_bn, stride_out_nj, stride_out_jd,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul per channel: out = left @ right^T  (FP16 → FP32)."""
    pid_z = tl.program_id(2)               # flattened (batch, channel)
    pid_m = tl.program_id(0)               # i‑block
    pid_n = tl.program_id(1)               # j‑block

    # decode (batch, d) from pid_z
    batch_idx = pid_z // D
    d_idx = pid_z % D

    # offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the beginning of this (batch, d) slice
    left_base  = left_ptr \
               + batch_idx * stride_left_bn \
               + d_idx * stride_left_kd
    right_base = right_ptr \
               + batch_idx * stride_right_bn \
               + d_idx * stride_right_kd
    out_base   = out_ptr \
               + batch_idx * stride_out_bn \
               + d_idx * stride_out_jd

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # k‑range for this iteration
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left[b, i, k, d]  -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base \
                  + offs_m[:, None] * stride_left_nk \
                  + offs_k[None, :] * stride_left_kd
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right[b, j, k, d]  -> shape (BLOCK_N, BLOCK_K)
        # note we need right[..., j, k] (no transpose) because the kernel
        # computes dot(left_i_k, right_j_k) directly.
        right_ptrs = right_base \
                   + offs_n[:, None] * stride_right_nk \
                   + offs_k[None, :] * stride_right_kd
        right = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # accumulate Σ_k left[i,k] * right[j,k]
        acc += tl.dot(left, right, trans_b=False)   # (M,K)·(N,K) -> (M,N)

    # write back
    out_ptrs = out_base \
             + offs_m[:, None] * stride_out_nj \
             + offs_n[None, :] * stride_out_jd
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: launch wrapper
# ----------------------------------------------------------------------
def launch_trimul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: [B, N, N, D]  (float16)
    returns out:   [B, N, N, D]  (float32)
    """
    B, N, _, D = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in elements (not bytes)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * D,
    )
    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        D,
        stride_left[0],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full TriMul forward (partial Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]  (float32)
            mask         : [B, N, N]       (bool/float)
            weights      : dict of model parameters (float32)
            config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # manual layernorm (no need for triton – cheap)
    mean = input_tensor.mean(-1, keepdim=True)
    var  = ((input_tensor - mean) ** 2).mean(-1, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * norm_weight + norm_bias

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – keep in float16 for the kernel
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]        # (hidden_dim, dim)

    # shape: [B,N,N,hidden_dim]
    left = torch.nn.functional.linear(x, left_proj_w).to(torch.float16)
    right = torch.nn.functional.linear(x, right_proj_w).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) – still in float32, then cast
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate.to(torch.float16)
    right = right * right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Heavy contraction via Triton
    # ------------------------------------------------------------------
    # out shape will be [B, N, N, hidden_dim] in FP32
    out = launch_trimul_outgoing(left, right)   # fp32

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # LayerNorm on hidden_dim
    mean_o = out.mean(-1, keepdim=True)
    var_o  = ((out - mean_o) ** 2).mean(-1, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_w)

    return out
