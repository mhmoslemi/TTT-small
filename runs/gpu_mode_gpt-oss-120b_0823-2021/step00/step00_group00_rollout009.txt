"""
TriMul “outgoing” implementation.

The heavy work is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix‑multiply per hidden channel:
    out[b,:,:,d] = left[b,:,:,d] @ right[b,:,:,d].T
The kernel below performs this multiplication in FP16 with FP32 accumulation.
All surrounding LayerNorm / Linear / gating work is kept in PyTorch for
readability and to avoid excessive kernel complexity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: batched matmul over the hidden dimension.
# -------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    B, N, H,                             # batch, seq_len, hidden_dim
    stride_a_b, stride_a_n, stride_a_h,  # a: (b, i, k, h)
    stride_b_b, stride_b_n, stride_b_h,  # b: (b, j, k, h)  (note k is last dim)
    stride_c_b, stride_c_i, stride_c_j,  # c: (b, i, j, h)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # pid maps to (batch, hidden) pair
    batch_id = pid // H
    hidden_id = pid % H

    # block offsets for the output matrix (i,j)
    offs_m = tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)

    # compute global offsets
    a_ptrs = a_ptr + batch_id * stride_a_b \
                  + hidden_id * stride_a_h \
                  + offs_m[:, None] * stride_a_n \
                  + 0 * stride_a_h   # k offset added inside loop
    b_ptrs = b_ptr + batch_id * stride_b_b \
                  + hidden_id * stride_b_h \
                  + 0 * stride_b_n \
                  + offs_n[None, :] * stride_b_n  # k offset added inside loop

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        # load A block (i,k)
        a = tl.load(
            a_ptrs + k_offset * stride_a_h,
            mask=(offs_m[:, None] < N) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < N),
            other=0.0,
        ).to(tl.float16)

        # load B block (j,k) – note we need the transpose, so we load (k,j)
        b = tl.load(
            b_ptrs + k_offset * stride_b_h,
            mask=(k_offset + tl.arange(0, BLOCK_K)[:, None] < N) & (offs_n[None, :] < N),
            other=0.0,
        ).to(tl.float16)

        # dot: sum over k (inner dim)
        acc += tl.dot(a, b, trans_b=True)   # trans_b swaps the k dimension

    # write back (convert to fp16 then up‑cast later)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + batch_id * stride_c_b \
                    + hidden_id * stride_c_i * N \
                    + offs_m[:, None] * stride_c_i \
                    + offs_n[None, :] * stride_c_j
    tl.store(
        c_ptrs,
        c,
        mask=(offs_m[:, None] < N) & (offs_n[None, :] < N),
    )

# -------------------------------------------------------------
# Host‑side helper to launch the kernel
# -------------------------------------------------------------
def batch_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : (B, N, N, H)  FP16
    returns out : (B, N, N, H)   FP16
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes)
    stride_a_b, stride_a_n, stride_a_k, stride_a_h = left.stride()
    stride_b_b, stride_b_n, stride_b_k, stride_b_h = right.stride()
    stride_c_b, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # launch one program per (batch, hidden)
    total = B * H
    grid = (total,)

    # reasonable block sizes for Hopper (fp16 compute)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    batch_matmul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_a_b, stride_a_n, stride_a_h,
        stride_b_b, stride_b_n, stride_b_h,
        stride_c_b, stride_c_i, stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out

# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul block.

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor  [B, N, N, dim]   (float32)
        mask         : torch.Tensor  [B, N, N]        (float32 or bool)
        weights      : dict of torch.Tensor for every linear / norm weight/bias
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor  [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ---------- LayerNorm ----------
    # x = (x - mean) / sqrt(var+eps) * weight + bias
    # PyTorch implementation (fast, already fused)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    # Linear w/out bias: y = x @ w.T
    # w shape: (hidden, dim)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Convert to fp16 for the heavy matmul ----------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ---------- Triton batched matmul ----------
    # out_tmp shape: [B,N,N,hidden]  (fp16)
    out_tmp = batch_matmul(left_fp16, right_fp16)

    # Back to fp32 for the rest of the pipeline
    out = out_tmp.to(torch.float32)

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
