"""
TriMul “outgoing” implementation.

The heavy part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply for every hidden channel *d* :

    out[b,:,:,d] = left[b,:,:,d] @ right[b,:,:,d].T

The kernel below performs this GEMM in pure Triton.  All LayerNorms,
Linear projections and sigmoids are kept in PyTorch (they are cheap
compared with the N³ work).  The kernel is launched with a three‑dim
grid (batch, hidden, block‑i, block‑j) and uses fp32 accumulation while
reading inputs as fp16 (the model weights are stored in fp16 to save
bandwidth).  The final tensor is cast back to fp32 so the surrounding
code can continue with float32‑only ops.

Only the einsum is off‑loaded to Triton – this satisfies the “must
implement at least part of the operations in a kernel” rule while still
giving a large speed‑up on H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton kernel: out = left @ right.transpose(-1,-2)  (per batch & channel)
# --------------------------------------------------------------

@triton.jit
def trimul_einsum_kernel(
    left_ptr,               # *float16   [B, N, N, H]
    right_ptr,              # *float16   [B, N, N, H]
    out_ptr,                # *float32   [B, N, N, H]
    B, N, H,
    stride_lb0, stride_lb1, stride_lb2, stride_lb3,   # left strides
    stride_rb0, stride_rb1, stride_rb2, stride_rb3,   # right strides
    stride_ob0, stride_ob1, stride_ob2, stride_ob3,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    One program computes a TILE_M × TILE_N output block for a single
    (batch, hidden) pair.
    """
    pid = tl.program_id(0)

    # decompose pid -> batch, hidden, tile_i, tile_j
    num_hidden = H
    tiles_i = tl.cdiv(N, BLOCK_M)
    tiles_j = tl.cdiv(N, BLOCK_N)
    per_batch = num_hidden * tiles_i * tiles_j

    batch = pid // per_batch
    pid_rem = pid % per_batch
    hidden = pid_rem // (tiles_i * tiles_j)
    pid_rem = pid_rem % (tiles_i * tiles_j)
    tile_i = pid_rem // tiles_j
    tile_j = pid_rem % tiles_j

    # block coordinates
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers to the start of the (batch,hidden) slice
    left_batch = left_ptr + batch * stride_lb0 + hidden * stride_lb3
    right_batch = right_ptr + batch * stride_rb0 + hidden * stride_rb3
    out_batch = out_ptr + batch * stride_ob0 + hidden * stride_ob3

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + offs_k
        mask_k = k_offset < N

        # left[b,i,k,h]   -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_batch + offs_i[:, None] * stride_lb1 + k_offset[None, :] * stride_lb2
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right[b,j,k,h]  -> we need it transposed for GEMM:
        #   we load right as (BLOCK_K, BLOCK_N) = right[b,k,j,h]
        right_ptrs = right_batch + k_offset[:, None] * stride_rb2 + offs_j[None, :] * stride_rb1
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # GEMM on the tile
        acc += tl.dot(left, right)          # (M,K) @ (K,N) -> (M,N)

    # write back
    out_ptrs = out_batch + offs_i[:, None] * stride_ob1 + offs_j[None, :] * stride_ob2
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Host side helper
# --------------------------------------------------------------

def triton_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : [B, N, N, H]  fp16
    returns    : [B, N, N, H]  fp32
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in elements (not bytes)
    stride_lb0, stride_lb1, stride_lb2, stride_lb3 = left.stride()
    stride_rb0, stride_rb1, stride_rb2, stride_rb3 = right.stride()
    stride_ob0, stride_ob1, stride_ob2, stride_ob3 = out.stride()

    # tile sizes – tuned for Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    tiles_i = triton.cdiv(N, BLOCK_M)
    tiles_j = triton.cdiv(N, BLOCK_N)
    total_programs = B * H * tiles_i * tiles_j
    grid = (total_programs,)

    trimul_einsum_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_lb0, stride_lb1, stride_lb2, stride_lb3,
        stride_rb0, stride_rb1, stride_rb2, stride_rb3,
        stride_ob0, stride_ob1, stride_ob2, stride_ob3,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                # good default for fp16 GEMM
    )
    return out


# --------------------------------------------------------------
# Full custom kernel entry point
# --------------------------------------------------------------

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    TriMul “outgoing” forward pass with the einsum part in Triton.

    Args
    ----
    data : (input, mask, weights, config)
        input  – torch.Tensor  [B, N, N, dim] (float32)
        mask   – torch.Tensor  [B, N, N]    (float32 or bool)
        weights – dict of all Linear/LayerNorm parameters (fp16)
        config  – dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor  [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----------------------------------------------------------
    # 1. LayerNorm (float32) – we use torch.nn.functional.layer_norm
    # ----------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias   = weights["norm.bias"].to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ----------------------------------------------------------
    # 2. Linear projections (no bias)
    # ----------------------------------------------------------
    # store as fp16 for bandwidth
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)   # (dim, hidden)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)    # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)

    # ----------------------------------------------------------
    # 3. Apply mask (if any) – broadcast on hidden dim
    # ----------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)      # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------
    # 4. Gates (sigmoid)
    # ----------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 5. Triton einsum (core N³ work)
    # ----------------------------------------------------------
    # Cast to fp16 for the kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out = triton_einsum(left_fp16, right_fp16)          # [B,N,N,hidden] fp32

    # ----------------------------------------------------------
    # 6. Output projection
    # ----------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate                                 # gating after norm

    to_out_w = weights["to_out.weight"].to(torch.float32)   # (hidden, dim)
    out = F.linear(out, to_out_w)                       # final [B,N,N,dim]

    return out
