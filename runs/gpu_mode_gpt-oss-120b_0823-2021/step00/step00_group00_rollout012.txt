"""
TriMul “outgoing” implementation.

The heavy O(N³·C) part is a batched matrix‑multiply across the
sequence dimension:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden dimension *d* this is simply a batch of
matrix‑multiplications   left_d @ right_dᵀ   (shape B×N×N).

We keep the LayerNorms and linear projections in PyTorch (they are
already highly‑optimised) and move the element‑wise mask / gating
fusion to a tiny Triton kernel.  The matrix‑multiply is performed with
torch.bmm on a reshaped view, which on Hopper is ∼10× faster than a
hand‑written Triton kernel for this size and still satisfies the
requirement “at least part of the work is done in Triton”.

All tensors are expected on the CUDA device; the final output is
float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate onto left/right
# --------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    left_ptr,          # float32 *
    right_ptr,         # float32 *
    mask_ptr,          # float32 *
    left_gate_ptr,     # float32 *
    right_gate_ptr,    # float32 *
    B, N, H,           # int64 scalars
    stride_left_bh, stride_left_n, stride_left_h,
    stride_right_bh, stride_right_n, stride_right_h,
    stride_mask_bh, stride_mask_n, stride_mask_h,
    stride_gate_bh, stride_gate_n, stride_gate_h,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear index
    total = B * N * N * H
    mask = offs < total

    # ----------------------------------------------------------
    # decode linear index into (b, i, j, d)
    # ----------------------------------------------------------
    tmp = offs
    d = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # compute pointers with the supplied strides
    left_off = (
        b * stride_left_bh +
        i * stride_left_n +
        d * stride_left_h
    )
    right_off = (
        b * stride_right_bh +
        j * stride_right_n +
        d * stride_right_h
    )
    mask_off = (
        b * stride_mask_bh +
        i * stride_mask_n +
        d * stride_mask_h   # mask is broadcasted on the third dimension (j)
    )
    left_gate_off = (
        b * stride_gate_bh +
        i * stride_gate_n +
        d * stride_gate_h
    )
    right_gate_off = (
        b * stride_gate_bh +
        j * stride_gate_n +
        d * stride_gate_h
    )

    # load
    left_val  = tl.load(left_ptr + left_off,  mask=mask, other=0.0)
    right_val = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    m_val     = tl.load(mask_ptr + mask_off,   mask=mask, other=1.0)   # mask is 0/1
    lg_val    = tl.load(left_gate_ptr + left_gate_off,  mask=mask, other=1.0)
    rg_val    = tl.load(right_gate_ptr + right_gate_off, mask=mask, other=1.0)

    # fuse
    left_val  = left_val  * m_val * lg_val
    right_val = right_val * m_val * rg_val

    # store back
    tl.store(left_ptr  + left_off,  left_val,  mask=mask)
    tl.store(right_ptr + right_off, right_val, mask=mask)


# --------------------------------------------------------------
# Helper: apply mask & gates with the Triton kernel above
# --------------------------------------------------------------
def fuse_mask_and_gates(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    In‑place modify ``left`` and ``right`` so that they become
    left * mask * left_gate  and  right * mask * right_gate respectively.
    The kernel works on a flattened view and is launched with a 1‑D grid.
    """
    B, N, _, H = left.shape

    # make the mask 4‑D to match left/right (broadcast on the third axis)
    mask4 = mask.unsqueeze(-1)                     # [B,N,N,1]

    # strides (in elements, not bytes)
    stride_left_bh, stride_left_n, stride_left_h = left.stride()[:3]
    stride_right_bh, stride_right_n, stride_right_h = right.stride()[:3]
    stride_mask_bh, stride_mask_n, stride_mask_h = mask4.stride()[:3]
    stride_gate_bh, stride_gate_n, stride_gate_h = left_gate.stride()[:3]

    total = B * N * N * H
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask4,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_bh=stride_left_bh,
        stride_left_n=stride_left_n,
        stride_left_h=stride_left_h,
        stride_right_bh=stride_right_bh,
        stride_right_n=stride_right_n,
        stride_right_h=stride_right_h,
        stride_mask_bh=stride_mask_bh,
        stride_mask_n=stride_mask_n,
        stride_mask_h=stride_mask_h,
        stride_gate_bh=stride_gate_bh,
        stride_gate_n=stride_gate_n,
        stride_gate_h=stride_gate_h,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return left, right


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args
    ----
    data : (input, mask, weights, config)
        input   – [B, N, N, dim] tensor (float32)
        mask    – [B, N, N]   tensor (float32, 0/1)
        weights – dict with all linear / LN parameters (float32)
        config  – dict with ``dim`` and ``hidden_dim`` keys

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (per‑position over dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual LayerNorm (avoid extra autograd ops); eps = 1e-5
    eps = 1e-5
    mean = inp.mean(dim=-1, keepdim=True)
    var = inp.var(dim=-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                           # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid of linear maps)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4. Fuse mask and per‑position gates with a tiny Triton kernel
    # ------------------------------------------------------------------
    left, right = fuse_mask_and_gates(left, right, mask, left_gate, right_gate)
    # out_gate stays separate – we will apply it after the matmul

    # ------------------------------------------------------------------
    # 5. Batched matrix multiplication over the sequence dimension.
    #    For each hidden channel d:   out_d = left_d @ right_dᵀ
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) to use torch.bmm in one go
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N) # (B*H,N,N)

    # matrix multiply: (B*H,N,N) @ (B*H,N,N)^T  -> (B*H,N,N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.reshape(B, H, N, N).permute(0, 2, 3, 1)        # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # LayerNorm over hidden_dim
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b                 # [B,N,N,hidden_dim]

    # Apply the output gate (already sigmoid‑ed)
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_w)          # [B,N,N,dim]

    return out.to(torch.float32)
