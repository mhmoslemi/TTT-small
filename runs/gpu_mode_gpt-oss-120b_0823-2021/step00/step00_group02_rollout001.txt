"""
TriMul “outgoing” implementation.

*   Layer‑norm, linear projections and gating are done with plain
    PyTorch (they are cheap compared to the N³ core).
*   The only heavy part – the 4‑D einsum
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
  – is computed by a custom Triton kernel that performs a batched
    matrix‑multiply for every hidden channel *d*.  For a fixed
    (b,d) the operation is exactly a matrix product
        (N,N) left[b,:,:,d] @ (N,N) right[b,:,:,d].T
  (the transpose is handled inside the kernel).
*   The kernel works for arbitrary batch size B (≤2), sequence length N
    (up to 1024) and hidden dimension H (the “hidden_dim” argument).
    It uses 32×32 tiles and accumulates over the K‑dimension in blocks.
*   All tensors are expected to be on CUDA; the final output is returned
    as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul per hidden channel (N × N) × (N × N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # *float16* left  [B, N, N, H]
    right_ptr,              # *float16* right [B, N, N, H]
    out_ptr,                # *float32* out   [B, N, N, H]

    B, N, H,
    # Strides for left   (b,i,k,d)
    stride_lb, stride_li, stride_lk, stride_lh,
    # Strides for right  (b,j,k,d)  (note j instead of i)
    stride_rb, stride_rj, stride_rk, stride_rh,
    # Strides for out    (b,i,j,d)
    stride_ob, stride_oi, stride_oj, stride_oh,

    BLOCK_M: tl.constexpr,   # tile size in i
    BLOCK_N: tl.constexpr,   # tile size in j
    BLOCK_K: tl.constexpr,   # tile size in k
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""

    pid0 = tl.program_id(0)               # combined batch*hidden index
    pid1 = tl.program_id(1)               # block i
    pid2 = tl.program_id(2)               # block j

    # decode batch and hidden channel
    b = pid0 // H
    h = pid0 % H

    # ------------------------------------------------------------------
    # offsets
    # ------------------------------------------------------------------
    i_off = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑range tiles
    i_mask = i_off < N
    j_mask = j_off < N

    # pointer base for this (b,h)
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh

    # allocate accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over k dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # left[i,k]   -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + \
                    i_off[:, None] * stride_li + \
                    k_off[None, :] * stride_lk
        left_tile = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # right[j,k]  -> we need right_T[k,j] => load right[j,k] and treat as (k,j)
        # stride for j then k, we will transpose via load order
        right_ptrs = right_base + \
                     j_off[:, None] * stride_rj + \
                     k_off[None, :] * stride_rk
        # note: we load (j,k) but later we will use it as (k,j) via transpose
        right_tile = tl.load(right_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # multiply‑accumulate: (i,k) * (j,k) summed over k  → need (i,k)·(j,k)ᵀ
        # right_tile is (j,k); we transpose it to (k,j) for dot
        acc += tl.dot(left_tile, right_tile, trans_b=True)   # (M,N)

    # ------------------------------------------------------------------
    # write result
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
               b * stride_ob + \
               i_off[:, None] * stride_oi + \
               j_off[None, :] * stride_oj + \
               h * stride_oh
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim] (float32)
            - mask         : [B, N, N] (float32 or bool)
            - weights      : dict of model weights (float32)
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. Layer‑norm on the input (dim) – use fused torch implementation
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) – produce [B,N,N,hidden]
    # --------------------------------------------------------------
    # reshape to 2‑D for matmul, keep float16 for speed
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)
    left  = torch.nn.functional.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 3. Apply mask (if present) – broadcast on hidden dim
    # --------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)            # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(
        torch.nn.functional.linear(x_flat, weights["left_gate.weight"])
    ).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, weights["right_gate.weight"])
    ).view(B, N, N, hidden_dim)
    out_gate   = torch.sigmoid(
        torch.nn.functional.linear(x_flat, weights["out_gate.weight"])
    ).view(B, N, N, hidden_dim)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core einsum via Triton – compute out[b,i,j,d]
    # --------------------------------------------------------------
    # Cast to half for the kernel (better throughput) – kernel accumulates in FP32
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out = torch.empty_like(left_h, dtype=torch.float32, device=device)   # [B,N,N,H]

    # strides (in elements) for the 4‑D tensors
    stride_lb, stride_li, stride_lk, stride_lh = left_h.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right_h.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # launch configuration
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    grid = (
        B * hidden_dim,                         # program_id(0) – batch * hidden
        triton.cdiv(N, BLOCK_M),                # program_id(1) – i tiles
        triton.cdiv(N, BLOCK_N),                # program_id(2) – j tiles
    )
    tri_mul_kernel[grid](
        left_h,
        right_h,
        out,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6. Output post‑processing (LayerNorm, gating, final linear)
    # --------------------------------------------------------------
    # to_out_norm (hidden_dim)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # final linear back to dim
    out_flat = out.view(-1, hidden_dim)                     # (B*N*N, hidden)
    out = torch.nn.functional.linear(
        out_flat,
        weights["to_out.weight"],
    ).view(B, N, N, dim)

    return out
