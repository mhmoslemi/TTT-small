"""
tri_mul_triton.py
~~~~~~~~~~~~~~~~~

Implementation of the *outgoing* TriMul operator (AlphaFold‑3) with a
Triton kernel that fuses the O(N³) ``einsum('... i k d, ... j k d -> ... i j d')``
into a tiled matrix‑multiply‑like kernel.  All LayerNorms and linear
projections are kept in PyTorch (they are cheap compared with the N³ term).
The kernel processes a (batch, i‑tile, j‑tile, hidden) block and iterates over
the k‑dimension in tiles, accumulating the element‑wise product per hidden
channel.

The final output is cast back to float32 as required by the reference
implementation.
"""
import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict

# -------------------------------------------------------------------------
# Triton kernel
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    # pointers
    left_ptr,                # [B, N, N, H]  left tensor   (i, k, h)
    right_ptr,               # [B, N, N, H]  right tensor  (j, k, h)
    out_ptr,                 # [B, N, N, H]  output tensor (i, j, h)

    # sizes
    B, N, H,

    # strides (flattened NHW layout)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,   # i‑tile
    BLOCK_N: tl.constexpr,   # j‑tile
    BLOCK_K: tl.constexpr,   # k‑tile
    BLOCK_H: tl.constexpr,   # hidden‑channel tile
):
    """ Compute
            out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
        in a tiled fashion.
    """
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # ---- compute base offsets for this program ---------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)          # (BLOCK_M,)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)          # (BLOCK_N,)
    offs_h = tl.arange(0, BLOCK_H)                           # (BLOCK_H,)

    # bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # pointers to the start of the current i/j block (no k yet)
    left_base = (left_ptr
                 + pid_b * stride_left_b
                 + offs_i[:, None, None] * stride_left_i
                 + offs_h[None, None, :] * stride_left_h)   # (M,1,H)

    right_base = (right_ptr
                  + pid_b * stride_right_b
                  + offs_j[:, None, None] * stride_right_j
                  + offs_h[None, None, :] * stride_right_h) # (N,1,H)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ---- loop over k -----------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(0, num_k_tiles):
        offs_k = tk * BLOCK_K + tl.arange(0, BLOCK_K)          # (BLOCK_K,)
        mask_k = offs_k < N

        # load left tile  [i, k, h]   -> (M, K, H)
        left_ptrs = (left_base
                     + offs_k[None, :, None] * stride_left_k)  # broadcast k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :] & mask_h[None, None, :], other=0.0)

        # load right tile [j, k, h]   -> (N, K, H)
        right_ptrs = (right_base
                      + offs_k[None, :, None] * stride_right_k)  # broadcast k
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :] & mask_h[None, None, :], other=0.0)

        # ---- inner product over K for each hidden channel -----------------------
        # We cannot call tl.dot on 3‑D tensors, so we iterate over the hidden
        # channel tile.  BLOCK_H is small (≤ 32) and the loop is fully
        # unrolled by the Triton compiler.
        for dh in range(0, BLOCK_H):
            # mask for the current channel
            m_h = mask_h[dh]

            # (M, K) dot (N, K) → (M, N)
            l = left[:, :, dh]     # (M, K)
            r = right[:, :, dh]    # (N, K)
            # note: tl.dot expects (M,K) @ (K,N) – we need the transpose of r.
            acc[:, :, dh] += tl.dot(l, r, trans_b=True) * m_h

    # ---- write result -----------------------------------------------------------
    out_ptrs = (out_ptr
                + pid_b * stride_out_b
                + offs_i[:, None, None] * stride_out_i
                + offs_j[None, :, None] * stride_out_j
                + offs_h[None, None, :] * stride_out_h)

    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None, None] &
                                            mask_j[None, :, None] &
                                            mask_h[None, None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

        * input_tensor : torch.Tensor  [B, N, N, dim]   (float32)
        * mask         : torch.Tensor  [B, N, N]        (float32 or bool)
        * weights      : dict of torch.Tensor containing all linear / LN weights
        * config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. PyTorch part – layernorms & linear projections (fast enough)
    # -----------------------------------------------------------------
    # layernorm on the last dimension
    norm = nn.LayerNorm(dim, elementwise_affine=True).to(device)
    norm.weight.data = weights["norm.weight"]
    norm.bias.data   = weights["norm.bias"]

    x = norm(input_tensor)                       # [B,N,N,dim]

    # linear projections (no bias)
    def linear(t, w):
        # t: [B,N,N,dim] , w: [hidden, dim] (Transposed compared to nn.Linear)
        return torch.einsum('bnid,hd->bnih', t, w)   # result: [B,N,N,hidden]

    left  = linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = linear(x, weights["right_proj.weight"])

    # apply mask if present
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # gates
    left_gate  = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 2. Triton kernel for the N³ einsum
    # -----------------------------------------------------------------
    # ensure contiguous & appropriate dtype (use fp16 inside kernel for speed)
    left  = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    out   = torch.empty_like(left, dtype=torch.float16, device=device)

    # strides (flattened)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b,   stride_out_i,   stride_out_j,   stride_out_h   = out.stride()

    # tile sizes – tuned for Hopper (sm_90)
    BLOCK_M = 64   # i‑tile
    BLOCK_N = 64   # j‑tile
    BLOCK_K = 32   # k‑tile
    BLOCK_H = 32   # hidden‑channel tile (hidden_dim is a multiple of 32)

    grid = (B,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
        num_warps=8,  # Hopper can handle many warps
    )

    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 3. Post‑processing (LayerNorm + out‑gate + final linear)
    # -----------------------------------------------------------------
    to_out_norm = nn.LayerNorm(hidden_dim, elementwise_affine=True).to(device)
    to_out_norm.weight.data = weights["to_out_norm.weight"]
    to_out_norm.bias.data   = weights["to_out_norm.bias"]

    to_out = nn.Linear(hidden_dim, dim, bias=False).to(device)
    to_out.weight.data = weights["to_out.weight"]

    out = to_out_norm(out)
    out = out * out_gate
    out = to_out(out)                       # [B,N,N,dim]

    return out
