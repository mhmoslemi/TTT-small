"""
TriMul “outgoing” implementation.

We perform the heavy N³ part (the einsum over the k‑dimension) with a
Triton kernel.  The kernel computes, for every hidden channel d :

        out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which is exactly a batched matrix multiplication L_d @ R_dᵀ
with L_d,R_d ∈ ℝ^{N×N}.  The kernel fuses the B×D hidden‑channel
dimension into a single batch dimension so that a single launch
covers all (b,d) pairs.  All LayerNorm / linear projections are
performed in PyTorch (they are cheap compared with the N³ work) and the
final result is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------
# Triton kernel: batched matmul (L @ Rᵀ) for each hidden channel
# -------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    # pointers
    left_ptr,          # *[B, N, N, Dh]  (row‑major)
    right_ptr,         # *[B, N, N, Dh]
    out_ptr,           # *[B, N, N, Dh]
    # sizes
    B, N, Dh,
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
    for a tile of (i,j) and a single (b,d) pair.
    """
    pid = tl.program_id(0)          # batch dimension = b * Dh + d
    pid_m = tl.program_id(1)        # i‑tile
    pid_n = tl.program_id(2)        # j‑tile

    # decode b and d from the flattened batch id
    b = pid // Dh
    d = pid % Dh

    # compute starting offsets of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for boundary handling
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the first element of the current (b,d) matrix pair
    left_base  = left_ptr  + b * stride_left_b  + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_b   + d * stride_out_d

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k‑dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # offsets for the k‑tile
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load L block: shape (BLOCK_M, BLOCK_K)
        l_ptr = left_base \
                + offs_m[:, None] * stride_left_i \
                + offs_k[None, :] * stride_left_k
        L = tl.load(l_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load R block (need transpose → we load as (j,k) then treat as (k,j))
        r_ptr = right_base \
                + offs_n[:, None] * stride_right_j \
                + offs_k[None, :] * stride_right_k
        R = tl.load(r_ptr, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # multiply‑accumulate
        acc += tl.dot(L, R)          # (M,K)·(K,N) → (M,N)

    # write result
    out_ptrs = out_base \
               + offs_m[:, None] * stride_out_i \
               + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------
# Helper: launch the Triton kernel
# -------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, Dh]   (float32)
    returns out   : [B, N, N, Dh]   (float32)
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, Dh = left.shape

    out = torch.empty_like(left)

    # strides in number of elements (not bytes)
    s_b, s_i, s_k, s_d = left.stride()
    rs_b, rs_j, rs_k, rs_d = right.stride()
    os_b, os_i, os_j, os_d = out.stride()

    # tile sizes – chosen for H200 (SM90) 128‑thread warps
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * Dh,                              # flattened (b,d) batch
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, Dh,
        s_b, s_i, s_k, s_d,
        rs_b, rs_j, rs_k, rs_d,
        os_b, os_i, os_j, os_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# -------------------------------------------------------------
# Full custom kernel entry‑point
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments
    ----------
    data : (input, mask, weights, config)
        input  – [B, N, N, dim]   (float32)
        mask   – [B, N, N]        (float32 or bool)
        weights – dict with all layer‑norm / linear weights & biases
        config – dict with keys: dim, hidden_dim
    Returns
    -------
    out : torch.Tensor   [B, N, N, dim]   (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ---------------------------------------------------------
    # LayerNorm: (x - mean) / sqrt(var+eps) * w + b
    eps = 1e-5
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    # compute mean / var across dim (last axis) keeping batch dims
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * w_norm + b_norm

    # ---------------------------------------------------------
    # 2) Linear projections (no bias) – keep in float32
    # ---------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------
    # 3) Apply mask (if present)
    # ---------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------------------------------------------------------
    # 4) Gating (sigmoid)
    # ---------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 5) Core N³ multiplication – Triton kernel
    # ---------------------------------------------------------
    # left/right are [B,N,N,hidden] in float32 already
    out_hidden = tri_mul_triton(left, right)   # shape [B,N,N,hidden]

    # ---------------------------------------------------------
    # 6) to_out_norm (LayerNorm over hidden dim) + gating
    # ---------------------------------------------------------
    w_outnorm = weights["to_out_norm.weight"]
    b_outnorm = weights["to_out_norm.bias"]
    mean_h = out_hidden.mean(dim=-1, keepdim=True)
    var_h = out_hidden.var(dim=-1, unbiased=False, keepdim=True)
    out_hidden = (out_hidden - mean_h) / torch.sqrt(var_h + eps)
    out_hidden = out_hidden * w_outnorm + b_outnorm

    out_hidden = out_hidden * out_gate          # broadcast over hidden dim

    # ---------------------------------------------------------
    # 7) Final linear projection back to dim
    # ---------------------------------------------------------
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out.to(torch.float32)
