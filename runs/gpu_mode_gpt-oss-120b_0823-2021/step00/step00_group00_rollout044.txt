"""
TriMul “outgoing” kernel (AlphaFold3)

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape (B, N, N, H) (H = hidden_dim).
For each hidden channel *d* this is a matrix‑multiply
    (N×N) @ (N×N)ᵀ → (N×N).

The kernel below tiles the i‑ and j‑ dimensions (BLOCK_M × BLOCK_N) and
accumulates over the k‑dimension in blocks (BLOCK_K).  A small inner
loop processes a few hidden channels at a time (BLOCK_D).  All
operations are carried out in fp16 for speed; the final tensor is
converted back to fp32 on the host.

Only the core reduction is done in Triton – layer‑norms, linear
projections and gating are left to regular PyTorch which is cheap
compared to the N³ work.
"""

import torch
import triton
import triton.language as tl
from torch import nn

# -------------------------------------------------------------
# Triton kernel: batched matmul over the “k” dimension for several
# hidden channels at once.
# -------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # tensor sizes
    B, N, H,
    # strides (row‑major)
    stride_lb, stride_lk, stride_lh,
    stride_rb, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)                       # linear id over (b, i_tile, j_tile)
    num_b = B
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    # decode program id
    b   = pid // (num_i * num_j)
    rem = pid % (num_i * num_j)
    i_tile = rem // num_j
    j_tile = rem %  num_j

    # start offsets for the block
    i_start = i_tile * BLOCK_M
    j_start = j_tile * BLOCK_N

    # mask for out‑of‑bounds tiles
    mask_i = i_start + tl.arange(0, BLOCK_M) < N
    mask_j = j_start + tl.arange(0, BLOCK_N) < N

    # accumulator for each hidden sub‑block (fp32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        mask_k = k_start + tl.arange(0, BLOCK_K) < N

        # pointers for the current k‑slice
        left_offset  = (b * stride_lb
                        + (i_start) * stride_lk
                        + (k_start) * stride_lh)
        right_offset = (b * stride_rb
                        + (j_start) * stride_rk
                        + (k_start) * stride_rh)

        # load left : (M, K, D)  and right : (N, K, D)
        left  = tl.load(left_ptr + left_offset
                        + tl.arange(0, BLOCK_M)[:, None, None] * stride_lk
                        + tl.arange(0, BLOCK_K)[None, :, None] * stride_lh
                        + tl.arange(0, BLOCK_D)[None, None, :] * stride_lh,
                        mask=mask_i[:, None, None] & mask_k[None, :, None],
                        other=0.0).to(tl.float32)

        right = tl.load(right_ptr + right_offset
                        + tl.arange(0, BLOCK_N)[:, None, None] * stride_rk
                        + tl.arange(0, BLOCK_K)[None, :, None] * stride_rh
                        + tl.arange(0, BLOCK_D)[None, None, :] * stride_rh,
                        mask=mask_j[:, None, None] & mask_k[None, :, None],
                        other=0.0).to(tl.float32)

        # perform multiplication‑accumulation over K
        # we treat the D dimension as a batch of independent matmuls
        for d in range(0, BLOCK_D):
            # (M, K) dot (N, K)^T -> (M, N)
            a = left[:, :, d]          # (M,K)
            b = right[:, :, d]         # (N,K)
            # tl.dot expects (M,K) and (K,N); we transpose the second operand
            acc[:, :, d] += tl.dot(a, b, trans_b=True)

    # write back
    out_offset = (b * stride_ob
                  + i_start * stride_oi
                  + j_start * stride_oj)
    tl.store(out_ptr + out_offset
             + tl.arange(0, BLOCK_M)[:, None, None] * stride_oi
             + tl.arange(0, BLOCK_N)[None, :, None] * stride_oj
             + tl.arange(0, BLOCK_D)[None, None, :] * stride_oh,
             acc.to(tl.float16),   # store fp16, later cast to fp32 in Python
             mask=mask_i[:, None, None] & mask_j[None, :, None])


# -------------------------------------------------------------
# Host‑side helper
# -------------------------------------------------------------
def _launch_trimul_kernel(left, right):
    """
    left / right : torch.Tensor of shape (B,N,N,H) in fp16
    returns out   : torch.Tensor of shape (B,N,N,H) in fp16
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides (in elements)
    stride_lb, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # tile sizes – tuned for Hopper (SM90)  (fits in shared memory)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8                     # must divide hidden_dim

    assert H % BLOCK_D == 0, "hidden_dim must be multiple of BLOCK_D"

    grid = (B * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),)

    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, H,
        stride_lb, stride_lk, stride_lh,
        stride_rb, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
        num_warps=4,  # enough to hide latency
    )
    return out


# -------------------------------------------------------------
# Full custom kernel (mixes PyTorch and Triton)
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            input  – torch.Tensor [B, N, N, dim] (float32)
            mask   – torch.Tensor [B, N, N]      (float32 or bool)
            weights – dict of torch.Tensor parameters (float32)
            config  – dict with keys:
                      * dim        – input channel dimension
                      * hidden_dim – hidden dimension (must be multiple of 8)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. LayerNorm (input)
    # -----------------------------------------------------------------
    eps = 1e-5
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # fused layernorm in fp32 → fp16
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + eps
    )
    x = x * norm_w + norm_b          # shape [B,N,N,dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias) – keep in fp16 for speed
    # -----------------------------------------------------------------
    x16 = x.to(torch.float16)

    left = torch.nn.functional.linear(
        x16, weights["left_proj.weight"].to(torch.float16)
    )  # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(
        x16, weights["right_proj.weight"].to(torch.float16)
    )

    # -----------------------------------------------------------------
    # 3. Apply mask (if present) – broadcast on hidden dim
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x16, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x16, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x16, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core N³ reduction – Triton kernel
    # -----------------------------------------------------------------
    # left / right are [B,N,N,hidden_dim] fp16
    out_hidden = _launch_trimul_kernel(left, right)   # fp16

    # -----------------------------------------------------------------
    # 6. Post‑processing (LayerNorm, gating, final linear)
    # -----------------------------------------------------------------
    out_hidden = out_hidden.to(torch.float32)

    # to_out_norm (LayerNorm over hidden_dim)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean = out_hidden.mean(-1, keepdim=True)
    var = out_hidden.var(-1, unbiased=False, keepdim=True)
    out_hidden = (out_hidden - mean) / torch.sqrt(var + eps)
    out_hidden = out_hidden * to_out_norm_w + to_out_norm_b

    out_hidden = out_hidden * out_gate.to(torch.float32)

    # final projection back to dim
    out = torch.nn.functional.linear(
        out_hidden, weights["to_out.weight"].to(torch.float32)
    )  # shape [B,N,N,dim]

    return out
