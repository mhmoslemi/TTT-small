"""
Optimized “outgoing” TriMul kernel (AlphaFold‑3 style).

The heavy N³ work is the pairwise inner‑product
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

We keep all cheap ops (layer‑norms, linear layers, sigmoids, final
projection) in PyTorch and fuse only the N³ reduction in a single
Triton kernel.  The kernel tiles the (i,j) plane with
BLOCK_M×BLOCK_N threads and reduces over K in blocks of BLOCK_K.
Hidden dimension (d) is processed as a vector of length `hidden_dim`
using a `tl.dot` on each slice – this yields full‑fidelity fp32
accumulators while keeping the inner‑loop bandwidth‑limited.

The kernel works for arbitrary batch size (B≤2), sequence length N
(up to 1024 in the tests) and hidden dimension (always a multiple of 32
in the reference configs).  All tensors are expected to reside on the
CUDA device already.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched reduction over the “k” dimension.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,          # [B,N,N,HD]   left projected & gated
    right_ptr,         # [B,N,N,HD]   right projected & gated
    out_ptr,           # [B,N,N,HD]   result
    # strides (in elements, not bytes)
    stride_bl, stride_il, stride_kl, stride_dl,
    stride_br, stride_jr, stride_kr, stride_dr,
    stride_bo, stride_io, stride_jo, stride_do,
    # sizes
    B, N, HD,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]"""

    pid = tl.program_id(0)                # 1‑D grid over (i,j) tiles
    pid_batch = tl.program_id(1)          # batch dimension

    # ------------------------------------------------------------------
    # 1) tile indices for the (i,j) output plane
    # ------------------------------------------------------------------
    num_i_tiles = tl.cdiv(N, BLOCK_M)
    i_tile = pid // num_i_tiles           # row tile
    j_tile = pid %  num_i_tiles           # col tile

    i_base = i_tile * BLOCK_M
    j_base = j_tile * BLOCK_N

    # ------------------------------------------------------------------
    # 2) offsets within the tile (including a vector dim)
    # ------------------------------------------------------------------
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)                     # reduction axis
    offs_d = tl.arange(0, BLOCK_D)                     # part of hidden dim

    # masks for out‑of‑bounds tiles
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # flatten the hidden dimension into blocks of size BLOCK_D
    # (HD is always a multiple of BLOCK_D in the provided configs)
    # ------------------------------------------------------------------
    # 3) accumulate into a fp32 buffer
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4) reduction loop over k
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        # actual k size for this iteration
        cur_k = tl.where(k_start + offs_k < N, 1, 0)

        # Pointers to the current K‑slice of left/right
        left_tile_ptr = left_ptr + \
            pid_batch * stride_bl + \
            (offs_i[:, None, None] * stride_il) + \
            ((k_start + offs_k)[None, :, None] * stride_kl) + \
            (offs_d[None, None, :] * stride_dl)

        right_tile_ptr = right_ptr + \
            pid_batch * stride_br + \
            ((k_start + offs_k)[:, None, None] * stride_kr) + \
            (offs_j[None, :, None] * stride_jr) + \
            (offs_d[None, None, :] * stride_dr)

        # Load tiles (masking k‑dimension)
        left = tl.load(left_tile_ptr, mask=mask_i[:,None,None] & cur_k[None,:,None], other=0.0)
        right = tl.load(right_tile_ptr, mask=cur_k[:,None,None] & mask_j[None,:,None], other=0.0)

        # ------------------------------------------------------------------
        # 5) dot‑product over the K‑block for each (i,j) pair and each d‑slice
        #    We unroll the hidden‑dim vector: for each d we compute
        #        Σₖ left[...,d] * right[...,d]
        #    This can be expressed as a batched matmul where the innermost
        #    dimension is BLOCK_D and we use tl.dot.
        # ------------------------------------------------------------------
        # left : (M, K, D)   right : (K, N, D)
        # we need (M, N, D) = Σₖ left[M,K,D] * right[K,N,D]
        # Perform a dot for each d slice.
        for d in range(0, BLOCK_D):
            # shape (M, K) ⋅ (K, N) -> (M, N)
            l = left[:, :, d]          # (M, K)
            r = right[:, :, d]         # (K, N)
            acc[:, :, d] += tl.dot(l, r)   # (M, N) accumulate

    # ------------------------------------------------------------------
    # 6) write result back
    # ------------------------------------------------------------------
    out_tile_ptr = out_ptr + \
        pid_batch * stride_bo + \
        (offs_i[:, None, None] * stride_io) + \
        (offs_j[None, :, None] * stride_jo) + \
        (offs_d[None, None, :] * stride_do)

    tl.store(out_tile_ptr,
             acc.to(tl.float32),      # keep fp32 precision
             mask=mask_i[:,None,None] & mask_j[None,:,None])

# ----------------------------------------------------------------------
# Helper for launching the Triton kernel
# ----------------------------------------------------------------------
def launch_triton_trimul(left: torch.Tensor,
                         right: torch.Tensor,
                         ) -> torch.Tensor:
    """
    left, right : [B, N, N, HD] (float32)
    returns out : [B, N, N, HD] (float32)
    """
    B, N, _, HD = left.shape
    assert right.shape == (B, N, N, HD)

    # tile sizes – chosen to fit into shared memory on H200
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64          # reduction tile; 64 is a good trade‑off on Hopper
    BLOCK_D = 32          # hidden dim block (HD is a multiple of 32)

    # strides (in elements)
    stride_bl, stride_il, stride_kl, stride_dl = left.stride()
    stride_br, stride_jr, stride_kr, stride_dr = right.stride()
    out = torch.empty_like(left)
    stride_bo, stride_io, stride_jo, stride_do = out.stride()

    # launch grid: (i_tiles * j_tiles, batch)
    grid = (triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N), B)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        stride_bl=stride_bl, stride_il=stride_il,
        stride_kl=stride_kl, stride_dl=stride_dl,
        stride_br=stride_br, stride_jr=stride_jr,
        stride_kr=stride_kr, stride_dr=stride_dr,
        stride_bo=stride_bo, stride_io=stride_io,
        stride_jo=stride_jo, stride_do=stride_do,
        B=B, N=N, HD=HD,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
        num_warps=4,  # fits well on H200
    )
    return out


# ----------------------------------------------------------------------
# Fully‑fused forward function (only the N³ part lives in Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   (torch.float32, on CUDA)
            - mask         : [B, N, N]        (torch.float32 or bool)
            - weights      : dict of parameters (torch tensors)
            - config       : dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ---------- 1) LayerNorm on the input ----------
    # norm weight/bias are 1‑D tensors of length dim
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_w + norm_b   # [B,N,N,dim]

    # ---------- 2) Linear projections (no bias) ----------
    # shape: (dim, hidden_dim)
    left_proj_w  = weights["left_proj.weight"].to(device)   # [hidden_dim, dim] in PyTorch Linear
    right_proj_w = weights["right_proj.weight"].to(device)
    # we need weight.T for matmul because PyTorch stores (out_features, in_features)
    left = torch.einsum('...c,hdc->...hd', x, left_proj_w)   # [B,N,N,hidden]
    right = torch.einsum('...c,hdc->...hd', x, right_proj_w)

    # ---------- 3) Apply mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- 4) Gated versions ----------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.einsum('...c,hdc->...hd', x, left_gate_w).sigmoid()
    right_gate = torch.einsum('...c,hdc->...hd', x, right_gate_w).sigmoid()
    out_gate   = torch.einsum('...c,hdc->...hd', x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- 5) N³ reduction in Triton ----------
    # left/right are [B,N,N,hidden] float32
    out = launch_triton_trimul(left, right)   # [B,N,N,hidden]

    # ---------- 6) Output post‑processing ----------
    # to_out_norm (LayerNorm over hidden_dim)
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    # compute mean/var over hidden dim
    mean = out.mean(-1, keepdim=True)
    var  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean) / torch.sqrt(var + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    # multiply by out_gate (shape broadcastable)
    out = out * out_gate

    # final linear projection back to dim (no bias)
    to_out_w = weights["to_out.weight"].to(device)   # [dim, hidden]
    out = torch.einsum('...h,dh->...d', out, to_out_w)   # [B,N,N,dim]

    return out.to(torch.float32)
