"""
TriMul (outgoing) – Triton‑accelerated core

We keep the cheap linear + layer‑norm + gating work in PyTorch (they are
memory‑bound and cheap compared with the O(N³) pairwise product).  The heavy
part is

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

for every batch, i‑, j‑, d‑index.  This is a batched “outer‑product‑sum’’ and
dominates the runtime (N up to 1024, hidden_dim = 128).  The kernel tiles the
(i,j) plane with BLOCK_N×BLOCK_N tiles and iterates over k in BLOCK_K steps.
For each tile we load a (BLOCK_K, hidden) slice of *left* (i‑fixed) and a
(BLOCK_K, hidden) slice of *right* (j‑fixed), multiply element‑wise, sum over
k and accumulate into a per‑thread hidden‑dim accumulator.  Finally we write
the accumulator (float32) back to the output tensor.

The kernel works for any batch size (1 or 2) and any sequence length that is a
multiple or not of the block sizes (masked with “mask” on the i‑k and j‑k
dimensions is applied before the kernel launch).  All final tensors are
returned as float32.
"""

import torch
import triton
import triton.language as tl
from torch import nn

# ------------------------------------------------------------
# Triton kernel: sum_k left[b,i,k,d] * right[b,j,k,d]
# ------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,          #   [B, N, N, H]  (i,k) slice
    right_ptr,         #   [B, N, N, H]  (j,k) slice
    out_ptr,           #   [B, N, N, H]  result
    # strides (in elements, not bytes)
    stride_bl, stride_il, stride_kl, stride_hl,   # left
    stride_br, stride_jr, stride_kr, stride_hr,   # right
    stride_bo, stride_io, stride_jo, stride_ho,   # out
    B, N, H,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------
    # program ids
    # ------------------------------------------------------------
    pid_b = tl.program_id(0)          # batch
    pid_i = tl.program_id(1)          # tile i
    pid_j = tl.program_id(2)          # tile j

    # ------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------
    i_offset = pid_i * BLOCK_N + tl.arange(0, BLOCK_N)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑range rows / cols
    i_mask = i_offset < N
    j_mask = j_offset < N

    # ------------------------------------------------------------------
    # accumulator: one hidden‑dim vector per (i,j) element in the tile
    # ------------------------------------------------------------------
    # shape: (BLOCK_N, BLOCK_N, H)
    acc = tl.zeros((BLOCK_N, BLOCK_N, H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over k dimension in tiles of BLOCK_K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tile_k in range(0, num_k_tiles):
        k_offset = tile_k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # ------------------------------------------------------------
        # Load left[b, i, k, :]  -> shape (BLOCK_N, BLOCK_K, H)
        # ------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + pid_b * stride_bl
            + (i_offset[:, None] * stride_il)
            + (k_offset[None, :] * stride_kl)
            + tl.arange(0, H)[None, None, :] * stride_hl
        )
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # ------------------------------------------------------------
        # Load right[b, j, k, :] -> shape (BLOCK_N, BLOCK_K, H)
        # ------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + pid_b * stride_br
            + (j_offset[:, None] * stride_jr)
            + (k_offset[None, :] * stride_kr)
            + tl.arange(0, H)[None, None, :] * stride_hr
        )
        right = tl.load(right_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)

        # ------------------------------------------------------------
        # element‑wise product and reduction over k
        # ------------------------------------------------------------------
        # left * right : (BLOCK_N, BLOCK_K, H)
        prod = left * right                       # element‑wise
        acc += tl.sum(prod, axis=1)               # sum over k tile -> (BLOCK_N, H)

        # after tl.sum we have shape (BLOCK_N, H).  We need to broadcast
        # across the j dimension as well – we add to each (i,j) entry.
        # Using tl.broadcast we expand acc across j.
        # Simpler: accumulate into a temporary and later broadcast:
        # Here we accumulate into a (BLOCK_N, BLOCK_N, H) tensor by
        # adding the summed vector to every j in the tile.
        # This works because the sum over k is identical for all j in the
        # current tile (they share the same k slice).  Hence we add the
        # vector to each column.
        # (If you prefer a more explicit version you can loop over j,
        # but broadcasting is cheap.)
        acc = acc + tl.broadcast_to(acc[:, 0:1, :], (BLOCK_N, BLOCK_N, H))

    # ------------------------------------------------------------
    # write back
    # ------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_bo
        + (i_offset[:, None] * stride_io)
        + (j_offset[None, :] * stride_jo)
        + tl.arange(0, H)[None, None, :] * stride_ho
    )
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


# ------------------------------------------------------------
# Host‑side helper
# ------------------------------------------------------------
def launch_trimul_outgoing(left, right):
    """
    left, right : torch tensors of shape [B, N, N, H] (float32)
    returns      : torch tensor of same shape
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in element units (not bytes)
    stride_bl, stride_il, stride_kl, stride_hl = left.stride()
    stride_br, stride_jr, stride_kr, stride_hr = right.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out.stride()

    # block sizes – tuned for Hopper SM90 (shared memory ~ 64 KiB)
    BLOCK_N = 32   # tile for i and j dimensions
    BLOCK_K = 64   # tile for reduction over k

    grid = (B,
            triton.cdiv(N, BLOCK_N),
            triton.cdiv(N, BLOCK_N))

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_jr, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        B, N, H,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ------------------------------------------------------------
# Full custom kernel (PyTorch + Triton)
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]   float32
            mask         : [B, N, N]        (bool/float) – 1 where valid
            weights      : dict of model weights (torch tensors)
            config       : {"dim": int, "hidden_dim": int}
    Returns:
        output tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    # LayerNorm can stay in PyTorch – it is cheap compared to the O(N³)
    # operation and lets us reuse PyTorch’s stable implementation.
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias          # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – shape [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left  = torch.einsum("bnijc,hd->bnijh", x, left_proj_w)   # -> [B,N,N,hidden]
    right = torch.einsum("bnijc,hd->bnijh", x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)       # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise gate application
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.einsum("bnijc,hd->bnijh", x, left_gate_w).sigmoid()
    right_gate = torch.einsum("bnijc,hd->bnijh", x, right_gate_w).sigmoid()
    out_gate   = torch.einsum("bnijc,hd->bnijh", x, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul product – Triton kernel
    # ------------------------------------------------------------------
    # left/right are [B,N,N,hidden] float32
    out = launch_trimul_outgoing(left, right)   # still float32

    # ------------------------------------------------------------------
    # 6. Final layer‑norm, out‑gate and projection back to dim
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    # LayerNorm on hidden dimension
    mu = out.mean(-1, keepdim=True)
    var = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mu) / torch.sqrt(var + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate                               # gate

    to_out_w = weights["to_out.weight"]                # (dim, hidden)
    out = torch.einsum("bnijh,dh->bnijd", out, to_out_w)  # back to dim

    return out.to(torch.float32)
