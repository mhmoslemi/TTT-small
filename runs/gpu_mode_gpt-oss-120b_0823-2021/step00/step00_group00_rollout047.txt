"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part of the forward pass is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is an N³ reduction with a hidden dimension `D` (usually 128).
The kernel below tiles the (i,j) plane and accumulates over k.
All LayerNorm / Linear / gating work is performed in PyTorch
(because they are cheap compared to the N³ term) while the
einsum is fully fused in Triton.

The kernel works for arbitrary batch size (≤2), sequence length N
(≤1024) and hidden dimension D (any multiple of 4).  The final
output is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel for the einsum reduction
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,          # const float*  [B, N, N, D]
    right_ptr,         # const float*  [B, N, N, D]
    out_ptr,           #       float*  [B, N, N, D]
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b,   stride_out_i,   stride_out_j,   stride_out_d,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Compute   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""
    pid = tl.program_id(0)                     # linear id over (b, i, j) tiles
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)

    b = pid // (num_i * num_j)                 # batch index
    rest = pid % (num_i * num_j)
    i_tile = rest // num_j                     # i‑tile index
    j_tile = rest % num_j                      # j‑tile index

    # Offsets within the tiled i/j blocks
    i_off = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)

    # Mask for tiles that exceed the actual size
    mask_i = i_off < N
    mask_j = j_off < N

    # Pointers to the start of the (i,j) tile for this batch
    left_tile_ptr  = left_ptr  + b * stride_left_b \
                                 + i_off[:, None, None] * stride_left_i \
                                 + tl.arange(0, BLOCK_K)[None, :, None] * stride_left_k \
                                 + tl.arange(0, BLOCK_D)[None, None, :] * stride_left_d
    right_tile_ptr = right_ptr + b * stride_right_b \
                                 + j_off[:, None, None] * stride_right_i \
                                 + tl.arange(0, BLOCK_K)[None, :, None] * stride_right_k \
                                 + tl.arange(0, BLOCK_D)[None, None, :] * stride_right_d

    # Initialise accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Load a BLOCK_K × BLOCK_D slice for every (i) and (j)
        left   = tl.load(left_tile_ptr,  mask=mask_i[:, None, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < N), other=0.0)
        right  = tl.load(right_tile_ptr, mask=mask_j[:, None, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < N), other=0.0)

        # Element‑wise product and sum over the reduction axis (k)
        # left, right : (BLOCK_I, BLOCK_K, BLOCK_D) and (BLOCK_J, BLOCK_K, BLOCK_D)
        # We need a broadcast over i and j → compute outer product via broadcasting
        # left[i,k,d] * right[j,k,d]  -> (BLOCK_I, BLOCK_J, BLOCK_K, BLOCK_D)
        prod = left[:, None, :, :] * right[None, :, :, :]   # (i, j, k, d)

        acc += tl.sum(prod, axis=2)                         # sum over k → (i, j, d)

        # Advance pointers to next k‑block
        left_tile_ptr  = left_tile_ptr  + BLOCK_K * stride_left_k
        right_tile_ptr = right_tile_ptr + BLOCK_K * stride_right_k

    # Write the result back
    out_tile_ptr = out_ptr + b * stride_out_b \
                            + i_off[:, None, None] * stride_out_i \
                            + j_off[None, :, None] * stride_out_j \
                            + tl.arange(0, BLOCK_D)[None, None, :] * stride_out_d

    tl.store(out_tile_ptr,
             acc.to(tl.float32),
             mask=mask_i[:, None, None] & mask_j[None, :, None])

# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass for the TriMul “outgoing” operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N]   (float32 or bool)
        - weights      : dict of parameter tensors
        - config       : dict with keys ``dim`` and ``hidden_dim``
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, cfg = data
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]
    device = inp.device

    # ---------- 1. LayerNorm ----------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    x = F.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)

    # ---------- 2. Linear projections ----------
    def linear(t, w):
        # w : [out, in]   (no bias)
        return F.linear(t, w)

    left  = linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = linear(x, weights["right_proj.weight"])

    # ---------- 3. Apply mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                # [B,N,N,1]
        left  = left  * mask_exp
        right = right * mask_exp

    # ---------- 4. Gating ----------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ---------- 5. Triton einsum ----------
    # ensure contiguous for pointer arithmetic
    left  = left.contiguous()
    right = right.contiguous()
    out   = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides in elements (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_d = right.stride()
    stride_out_b,   stride_out_i,   stride_out_j,   stride_out_d   = out.stride()

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_D = 4   # hidden is a multiple of 4 (128)

    grid = (triton.cdiv(B * N * N, BLOCK_I * BLOCK_J),)   # one program per (b,i,j) tile
    trimul_einsum_kernel[grid](
        left_ptr   = left,
        right_ptr  = right,
        out_ptr    = out,
        B = B, N = N, D = hidden,
        stride_left_b = stride_left_b,
        stride_left_i = stride_left_i,
        stride_left_k = stride_left_k,
        stride_left_d = stride_left_d,
        stride_right_b = stride_right_b,
        stride_right_i = stride_right_i,
        stride_right_k = stride_right_k,
        stride_right_d = stride_right_d,
        stride_out_b   = stride_out_b,
        stride_out_i   = stride_out_i,
        stride_out_j   = stride_out_j,
        stride_out_d   = stride_out_d,
        BLOCK_I = BLOCK_I,
        BLOCK_J = BLOCK_J,
        BLOCK_K = BLOCK_K,
        BLOCK_D = BLOCK_D,
    )

    # ---------- 6. Output post‑processing ----------
    out = F.layer_norm(out, (hidden,), weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])      # [B,N,N,dim]

    return out
