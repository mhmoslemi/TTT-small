"""
TriMul “outgoing” implementation.

The heavy part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batch of N×N matrix‑multiplications for each hidden channel *d*.
The kernel below performs a blocked GEMM over the (i,j) plane while
accumulating over *k*.  All other operations (layer‑norms, linear projections,
gates and the final output linear) are kept in PyTorch for simplicity.
The kernel works in fp16 for the two inputs and accumulates in fp32,
returning fp32 results that are later cast back to the model dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: blocked N×N matrix multiply for each hidden channel.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          #   [B, N, N, H]  (i,k,d)
    right_ptr,         #   [B, N, N, H]  (j,k,d)
    out_ptr,           #   [B, N, N, H]  (i,j,d)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_j, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)                # batch dim
    pid_m = tl.program_id(1)                # i‑block
    pid_n = tl.program_id(2)                # j‑block

    # Offsets for the current block
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for OOB rows/cols
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # Pointers to the start of the block (no hidden dim yet)
    left_base = left_ptr + pid_b * stride_l_b \
                + offs_i[:, None] * stride_l_i \
                + offs_k[None, :] * stride_l_k
    right_base = right_ptr + pid_b * stride_r_b \
                + offs_j[:, None] * stride_r_j \
                + offs_k[None, :] * stride_r_k

    # Accumulator for each hidden channel
    acc = tl.zeros((BLOCK_M, BLOCK_N, H), dtype=tl.float32)

    # Loop over k‑dim
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        # Load a block of left/right: shape (M, K, H) and (N, K, H)
        # We load the whole hidden dimension at once (contiguous in memory)
        left = tl.load(
            left_base + kk * BLOCK_K * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # (M, K, H) fp16
        right = tl.load(
            right_base + kk * BLOCK_K * stride_r_k,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )  # (N, K, H) fp16

        # Compute dot products for each hidden channel.
        # tl.dot works on 2‑D tensors; we broadcast over H.
        # The inner product is performed per channel by using tl.dot on the
        # K‑dimension and then broadcasting the result.
        # tl.dot returns fp32 when inputs are fp16, which is what we need.
        # Shape after dot: (M, N, H)
        acc += tl.dot(left, right, acc_dtype=tl.float32)

    # Write back, respecting OOB masks on i/j.
    out_ptrs = out_ptr + pid_b * stride_o_b \
               + offs_i[:, None] * stride_o_i \
               + offs_j[None, :] * stride_o_j
    # The hidden‑dim stride is contiguous, so we can store the whole vector.
    tl.store(
        out_ptrs,
        acc,
        mask=(mask_i[:, None] & mask_j[None, :]),
    )


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        output tensor of shape [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm and linear projections (keep in torch for simplicity)
    # --------------------------------------------------------------
    # norm
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # mask handling (optional)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 2. Triton kernel for the core einsum (outgoing)
    # --------------------------------------------------------------
    # Cast to fp16 for the kernel (H200 works well with fp16)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output
    out = torch.empty_like(left_fp16, dtype=torch.float32)

    # strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left_fp16.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_d = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # block sizes – tuned for Hopper (64 works well)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_d,
        stride_r_b, stride_r_j, stride_r_k, stride_r_d,
        stride_o_b, stride_o_i, stride_o_j, stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 3. Post‑processing (LayerNorm, gating, final linear)
    # --------------------------------------------------------------
    # to_out_norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
