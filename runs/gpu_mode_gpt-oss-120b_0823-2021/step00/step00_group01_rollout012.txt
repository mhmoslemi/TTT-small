"""
TriMul “outgoing” implementation.

The heavy part of the TriMul forward pass is the
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix‑multiply for every hidden dimension *d*:
    out[d] = left[d] @ right[d].T   (left,right : [B,N,K]).

The kernel below computes this triple‑product in Triton while the
surrounding linear, norm and gate operations stay in PyTorch.
All tensors are expected on the CUDA device, final output is float32.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict

# ------------------------------------------------------------
# Triton kernel: batched matmul for each hidden dimension
# ------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    a_ptr,                     # left   [B, N, K, D]
    b_ptr,                     # right  [B, N, K, D]
    c_ptr,                     # out    [B, N, N, D]
    # sizes
    B, N, K, D,
    # strides (in elements, not bytes)
    stride_ab, stride_ak, stride_ad,      # a: (B,N,K,D)
    stride_bb, stride_bk, stride_bd,      # b: (B,N,K,D)
    stride_cb, stride_cn, stride_cd,      # c: (B,N,N,D)
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,   # i‑tile size
    BLOCK_N: tl.constexpr,   # j‑tile size
    BLOCK_K: tl.constexpr,   # reduction tile size
    BLOCK_D: tl.constexpr,   # hidden‑dim tile size
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # decode program id -> batch, tile_i, tile_j
    # ------------------------------------------------------------------
    total_tiles_i = tl.cdiv(N, BLOCK_M)
    total_tiles_j = tl.cdiv(N, BLOCK_N)
    b = pid // (total_tiles_i * total_tiles_j)               # batch index
    pid_remaining = pid % (total_tiles_i * total_tiles_j)
    tile_i = pid_remaining // total_tiles_j
    tile_j = pid_remaining % total_tiles_j

    # ------------------------------------------------------------------
    # compute start offsets for this tile
    # ------------------------------------------------------------------
    i_offset = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d_offset = tl.arange(0, BLOCK_D)

    # masks for out‑of‑bounds
    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_d = d_offset < D

    # ------------------------------------------------------------------
    # pointers for the output tile (B,N,N,D)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + b * stride_cb
        + i_offset[:, None, None] * stride_cn          # i dimension
        + j_offset[None, :, None] * stride_cn          # j dimension (same stride as i because shape is [B,N,N,D])
        + d_offset[None, None, :] * stride_cd
    )

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over k
    # ------------------------------------------------------------------
    for k_outer in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_outer * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # load left:   [i,k,d]
        a_ptrs = (
            a_ptr
            + b * stride_ab
            + i_offset[:, None, None] * stride_ak
            + k_offset[None, :, None] * stride_ad
            + d_offset[None, None, :] * stride_ad
        )
        a = tl.load(a_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

        # load right:  [j,k,d]
        b_ptrs = (
            b_ptr
            + b * stride_bb
            + j_offset[:, None, None] * stride_bk
            + k_offset[None, :, None] * stride_bd
            + d_offset[None, None, :] * stride_bd
        )
        b_val = tl.load(b_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

        # multiply‑accumulate over k (broadcast d)
        # a: (M,K,D)  b: (N,K,D) -> we need sum_k a[i,k,d] * b[j,k,d]
        # expand a and b to (M,K,1,D) and (1,N,K,D) then use dot
        a_exp = a[:, :, None, :]          # (M,K,1,D)
        b_exp = b_val[None, :, :, :]      # (1,N,K,D)
        # element‑wise product and reduce over k
        prod = a_exp * b_exp              # (M,N,K,D)
        acc += tl.sum(prod, axis=1)       # sum over K  -> (M,N,D)

    # ------------------------------------------------------------------
    # write result
    # ------------------------------------------------------------------
    tl.store(
        c_ptrs,
        acc.to(tl.float16),   # store in fp16 to save bandwidth
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    )
# ------------------------------------------------------------

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the outgoing TriMul operator.
    The function follows the reference PyTorch implementation but
    computes the triple‑product with a custom Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm + linear projections + gates (PyTorch, cheap)
    # ------------------------------------------------------------------
    # norm
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias

    # projections
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hd]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask handling
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left  = left * mask_
        right = right * mask_

    # gates
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 2) Triton kernel for the N³ reduction (produces fp16)
    # ------------------------------------------------------------------
    # shapes & strides (in elements)
    B = bs
    K = N
    D = hidden_dim
    a = left  # [B,N,K,D]
    b = right # [B,N,K,D]

    # allocate output tensor (fp16, will be cast later)
    out = torch.empty((B, N, N, D), dtype=torch.float16, device=device)

    # kernel launch configuration
    BLOCK_M = 64   # tile for i dimension
    BLOCK_N = 64   # tile for j dimension
    BLOCK_K = 32   # reduction tile
    BLOCK_D = 32   # tile for hidden dim (must divide hidden_dim, pad with mask)

    grid = (
        triton.cdiv(B * N * N, BLOCK_M * BLOCK_N),  # one program per (b,i_tile,j_tile)
    )
    # compute strides (in elements)
    stride_ab, stride_ak, stride_ad = a.stride()
    stride_bb, stride_bk, stride_bd = b.stride()
    stride_cb, stride_cn, stride_cd = out.stride()

    trimul_outgoing_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=out,
        B=B, N=N, K=K, D=D,
        stride_ab=stride_ab,
        stride_ak=stride_ak,
        stride_ad=stride_ad,
        stride_bb=stride_bb,
        stride_bk=stride_bk,
        stride_bd=stride_bd,
        stride_cb=stride_cb,
        stride_cn=stride_cn,
        stride_cd=stride_cd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    torch.cuda.synchronize()

    out = out.to(torch.float32)   # cast back to fp32 for the remaining ops

    # ------------------------------------------------------------------
    # 3) post‑processing (LayerNorm, gate, final linear) – PyTorch
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
