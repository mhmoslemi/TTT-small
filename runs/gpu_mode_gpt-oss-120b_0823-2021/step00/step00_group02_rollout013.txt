"""
TriMul “outgoing” kernel (AlphaFold‑3).

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where   left  = (x·W_left)   * gate_left,
        right = (x·W_right)  * gate_right.

The implementation follows the classic “fusion” strategy:

*   LayerNorm, the four linear projections and the three sigmoid‑gates are
    performed in plain PyTorch (they are cheap compared to the N³ work).
*   The N³‑loop is off‑loaded to a Triton kernel.  Each program instance
    computes a single (batch, i, j) output vector of length ``hidden_dim``.
    It walks over the k‑dimension in a simple for‑loop, accumulating the
    element‑wise product in a register array (size ≤ 384, which comfortably
    fits into the register file of an H200 SM).

The kernel works for any batch size B (1‑2), any sequence length N
(128‑1024) and any hidden dimension H (≤ 384).  The final output is
converted back to float32 and passed through the final LayerNorm,
out‑gate and linear projection.

Running time is dominated by the N³ term; the kernel keeps the
memory traffic low by loading a single hidden‑dim vector from *left* and
*right* for each k and immediately multiplying‑accumulating, thus avoiding
any intermediate tensor allocations.

All tensors are assumed to already reside on the CUDA device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel for the N³ einsum
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,          # const float*  [B,N,N,H]
    right_ptr,         # const float*  [B,N,N,H]
    out_ptr,           #       float*  [B,N,N,H]
    B, N, H,           # int32 scalars
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_H: tl.constexpr):
    """Compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    one thread per (b,i,j) pair.
    """
    pid = tl.program_id(0)
    total = B * N * N
    # decode pid -> (b,i,j)
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # accumulator for the hidden dimension
    acc = tl.zeros((BLOCK_H,), dtype=tl.float32)

    # loop over k
    for k in range(0, N):
        # offsets for the hidden‑dim vector
        left_off  = b * stride_lb + i * stride_li + k * stride_lk + tl.arange(0, BLOCK_H) * stride_lh
        right_off = b * stride_rb + j * stride_rj + k * stride_rk + tl.arange(0, BLOCK_H) * stride_rh

        # load (H,) vectors
        l = tl.load(left_ptr + left_off, mask=True, other=0.0)
        r = tl.load(right_ptr + right_off, mask=True, other=0.0)

        # element‑wise multiply‑accumulate
        acc += l * r

    # store result
    out_off = b * stride_ob + i * stride_oi + j * stride_oj + tl.arange(0, BLOCK_H) * stride_oh
    tl.store(out_ptr + out_off, acc)


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Args
    ----
    data : (input, mask, weights, config)
        input   – torch.Tensor [B, N, N, dim]  (float32)
        mask    – torch.Tensor [B, N, N]      (float32 or bool)
        weights – dict with the exact keys used in the reference module
        config  – dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm and linear projections (PyTorch)
    # ------------------------------------------------------------------
    # layer‑norm on last dimension
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                 # [B,N,N,dim]

    # linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # apply mask if present (mask may be all‑ones)
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                           # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 2) Triton kernel for the N³ einsum
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # [B,N,N,H]

    # strides in elements (not bytes)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # launch one thread per (b,i,j)
    total_threads = B * N * N
    grid = (total_threads,)

    trimul_einsum_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_H=hidden_dim,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 3) Final LayerNorm, out‑gate and output projection (PyTorch)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])          # back to dim

    return out.to(torch.float32)
