"""
TriMul “outgoing” implementation.

Algorithm
---------
1. Layer‑norm + linear projections are cheap – we compute them in PyTorch.
2. The expensive part is the 3‑dimensional tensor contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]            (1)
   which is an einsum over the shared “k” axis.
3. We launch a Triton kernel that tiles the (i, j) plane (BLOCK_I×BLOCK_J)
   and iterates over k one element at a time (BLOCK_K = 1).  For each k we
   load a slice `left_i_d` of shape (BLOCK_I, D) and a slice `right_j_d`
   of shape (BLOCK_J, D) and accumulate their outer product
   `left_i_d[:,None,:] * right_j_d[None,:,:]` into an accumulator
   (I,J,D).  The accumulator is kept in float32 for numerical stability,
   while the loads use the input dtype (float16/bfloat16) for bandwidth.
4. After the loop finishes we store the result, apply the final
   LayerNorm, gating and output linear layer (again in PyTorch) and
   return a float‑32 tensor.

The kernel is deliberately simple (no shared‑memory tricks) but
exploits the massive parallelism of the H200: each program instance
computes a tile of the output matrix for a single batch element.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel for the core contraction (1)
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,               # *const float*, shape [B,N,N,D]
    right_ptr,              # *const float*, shape [B,N,N,D]
    out_ptr,                # *float*,      shape [B,N,N,D]
    B, N, D,
    stride_lb, stride_lk, stride_ld,   # strides for left   (b,i,k,d)
    stride_rb, stride_rk, stride_rd,   # strides for right  (b,j,k,d)
    stride_ob, stride_oi, stride_oj, stride_od,   # strides for out (b,i,j,d)
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
):
    # -------------------------------------------------------------
    # 1)  Program‑id decomposition
    # -------------------------------------------------------------
    pid = tl.program_id(0)
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)

    b   = pid // (num_i * num_j)
    rem = pid %  (num_i * num_j)
    i0  = rem // num_j          # i‑block index
    j0  = rem %  num_j          # j‑block index

    # -------------------------------------------------------------
    # 2)  Offsets for the current tile
    # -------------------------------------------------------------
    offs_i = i0 * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = j0 * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (I,J,D) in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, D), dtype=tl.float32)

    # -------------------------------------------------------------
    # 3)  Loop over the shared k‑dimension (one element at a time)
    # -------------------------------------------------------------
    for k in range(0, N):
        # load left slice  (I,D)
        left_ptrs = (left_ptr
                     + b * stride_lb
                     + offs_i[:, None] * stride_lk
                     + k * stride_ld
                     + tl.arange(0, D)[None, :])
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & (k < N),
                       other=0.0).to(tl.float32)          # (I,D)

        # load right slice (J,D)
        right_ptrs = (right_ptr
                      + b * stride_rb
                      + offs_j[:, None] * stride_rk
                      + k * stride_rd
                      + tl.arange(0, D)[None, :])
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & (k < N),
                        other=0.0).to(tl.float32)          # (J,D)

        # outer product over the D channel dimension
        # left  : (I, D) -> (I,1,D)
        # right : (J, D) -> (1,J,D)
        # result: (I,J,D)
        acc += left[:, None, :] * right[None, :, :]

    # -------------------------------------------------------------
    # 4)  Store the tile
    # -------------------------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_ob
                + offs_i[:, None, None] * stride_oi
                + offs_j[None, :, None] * stride_oj
                + tl.arange(0, D)[None, None, :])
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None])


# ----------------------------------------------------------------------
# Helper to perform LayerNorm without creating a nn.Module (weights &
# bias are given as tensors)
# ----------------------------------------------------------------------
def apply_layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor) -> torch.Tensor:
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    eps = 1e-5
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1)  LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = apply_layer_norm(
        input_tensor,
        weights["norm.weight"].to(dtype),
        weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2)  Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(dtype))          # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # ------------------------------------------------------------------
    # 3)  Apply mask if present
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                      # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4)  Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5)  Triton contraction
    # ------------------------------------------------------------------
    # Use fp16/BF16 for the kernel (the input tensors are already in that dtype)
    left_k = left.contiguous()
    right_k = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=left_k.dtype, device=device)

    # strides (in elements, not bytes) – torch gives them in element counts
    s_lb, s_lk, s_ld = left_k.stride()
    s_rb, s_rk, s_rd = right_k.stride()
    s_ob, s_oi, s_oj, s_od = out.stride()

    # Tile sizes – chosen to fit H200 shared memory / register pressure
    BLOCK_I = 64
    BLOCK_J = 64

    grid = (B * triton.cdiv(N, BLOCK_I) * triton.cdiv(N, BLOCK_J),)
    trimul_kernel[grid](
        left_k,
        right_k,
        out,
        B, N, hidden_dim,
        s_lb, s_lk, s_ld,
        s_rb, s_rk, s_rd,
        s_ob, s_oi, s_oj, s_od,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        num_warps=4,  # enough parallelism for H200
    )

    # ------------------------------------------------------------------
    # 6)  Post‑processing (LayerNorm, gating, final linear)
    # ------------------------------------------------------------------
    out = apply_layer_norm(
        out,
        weights["to_out_norm.weight"].to(dtype),
        weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    # Ensure final dtype is float32 as required
    return out.to(torch.float32)
