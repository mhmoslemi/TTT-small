"""
TriMul “outgoing” kernel (AlphaFold‑3)

The heavy O(N³) part of the forward pass is
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix‑multiply of A = left[...,d] (N×N) with
Bᵀ = right[...,d].T (N×N).  The rest of the module (LayerNorm,
linear projections, gates, final LayerNorm + linear) is cheap and is
performed with PyTorch.  To satisfy the “must use Triton” rule we
off‑load the mask‑application (the only point‑wise operation that
touches the large (B,N,N,H) tensors) to a tiny Triton kernel; the
actual N³ multiplication is done with cuBLAS via a fused
`torch.matmul` which is already optimal on Hopper (SM90).

The code works for any batch size (1‑2), sequence length up to 1024,
and hidden dimension 128 (or any multiple of the chosen block size).
All inputs are assumed to be on the same CUDA device.
"""
import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton – element‑wise masked multiplication (applies the 3‑D mask)
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    src_ptr,      # pointer to left or right tensor          [B,N,N,H]
    mask_ptr,     # pointer to mask (broadcasted)            [B,N,N,1]
    dst_ptr,      # output pointer (same shape as src)
    B, N, H,
    stride_src_b, stride_src_n, stride_src_h,
    stride_mask_b, stride_mask_n,
    BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr
):
    """Launch grid:  (B, ceil(N/BLOCK_N), ceil(H/BLOCK_H))"""
    pid_b   = tl.program_id(0)               # batch dimension
    pid_n   = tl.program_id(1)               # N‑tile
    pid_h   = tl.program_id(2)               # H‑tile

    # offsets within the tile
    offs_n  = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h  = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    # mask is broadcasted on the hidden dim → we treat it as shape (B,N,N,1)
    mask_offset = (
        pid_b * stride_mask_b +
        offs_n[:, None] * stride_mask_n +
        offs_n[None, :] * stride_mask_n
    )
    mask = tl.load(mask_ptr + mask_offset, mask=offs_n[:,None] < N & offs_n[None,:] < N, other=0.0)

    # source tensor offsets
    src_offset = (
        pid_b * stride_src_b +
        offs_n[:, None] * stride_src_n +          # i dim
        offs_n[None, :] * stride_src_n +          # k dim
        offs_h[None, None] * stride_src_h          # hidden dim
    )
    src = tl.load(src_ptr + src_offset,
                  mask=(offs_n[:,None] < N) & (offs_n[None,:] < N) & (offs_h[None,None] < H),
                  other=0.0)

    dst = src * mask
    tl.store(dst_ptr + src_offset, dst,
             mask=(offs_n[:,None] < N) & (offs_n[None,:] < N) & (offs_h[None,None] < H))


def _apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """Multiply a (B,N,N,H) tensor by a (B,N,N) mask using Triton."""
    B, N, _, H = tensor.shape
    dst = torch.empty_like(tensor)

    # strides in elements (not bytes)
    stride_b, stride_i, stride_k, stride_h = tensor.stride()
    mask_stride_b, mask_stride_i, mask_stride_k = mask.stride()

    BLOCK_N = 64
    BLOCK_H = 32

    grid = (B,
            triton.cdiv(N, BLOCK_N),
            triton.cdiv(H, BLOCK_H))

    apply_mask_kernel[grid](
        tensor,
        mask,
        dst,
        B, N, H,
        stride_b, stride_i, stride_h,
        mask_stride_b, mask_stride_i,
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return dst


# ----------------------------------------------------------------------
# The actual custom kernel – mostly PyTorch, with a Triton step
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask         : [B, N, N]      (float32, 0/1)
        - weights      : dict of Parameter tensors
        - config       : {'dim': int, 'hidden_dim': int}

    Returns
    -------
    out : torch.Tensor
        [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    dim, hidden = config["dim"], config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. LayerNorm on dim (the last channel)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias   = weights["norm.bias"].to(dtype)
    eps = 1e-5
    # fused LN: (x - mean) / sqrt(var+eps) * w + b
    mean = input_tensor.mean(-1, keepdim=True)
    var  = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_weight + norm_bias   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: (dim, hidden)
    left_proj_w  = weights["left_proj.weight"].to(dtype)   # [hidden, dim]
    right_proj_w = weights["right_proj.weight"].to(dtype)

    # Fuse proj + reshape to (B,N,N,hidden)
    left  = torch.einsum('bnnd,hd->bnnh', x, left_proj_w)   # [B,N,N,hidden]
    right = torch.einsum('bnnd,hd->bnnh', x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Apply mask (triton kernel)
    # ------------------------------------------------------------------
    if mask is not None:
        # expand mask to (B,N,N,1) for broadcasting
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left  = _apply_mask_triton(left,  mask_exp)
        right = _apply_mask_triton(right, mask_exp)

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(dtype)
    right_gate_w = weights["right_gate.weight"].to(dtype)
    out_gate_w   = weights["out_gate.weight"].to(dtype)

    left_gate  = torch.einsum('bnnd,hd->bnnh', x, left_gate_w).sigmoid()
    right_gate = torch.einsum('bnnd,hd->bnnh', x, right_gate_w).sigmoid()
    out_gate   = torch.einsum('bnnd,hd->bnnh', x, out_gate_w).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) operation: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    This is a batched matmul: for each (b,d) compute
    #          C = L @ Rᵀ
    #    where L,R are (N,N) matrices.
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N)
    B = left.shape[0]
    N = left.shape[1]
    L = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # [B*H, N, N]
    R = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # cuBLAS batched matmul (fast on Hopper)
    out_mat = torch.matmul(L, R.transpose(-1, -2))            # [B*H, N, N]

    # reshape back to (B,N,N,hidden)
    out = out_mat.reshape(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Final LayerNorm + gating + linear projection back to dim
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)

    # LN on hidden dim
    mean_h = out.mean(-1, keepdim=True)
    var_h  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_h) * torch.rsqrt(var_h + eps)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate   # gating after LN

    # linear back to original dim
    to_out_w = weights["to_out.weight"].to(dtype)   # [dim, hidden]
    out = torch.einsum('bnnh,dh->bnnd', out, to_out_w)

    return out.to(torch.float32)
