"""
tri_mul_triton.py

Implementation of the “outgoing” TriMul operator (AlphaFold‑3) where the heavy
pair‑wise multiplication‑sum (the einsum) is executed in a fused Triton
kernel.  All linear layers, LayerNorms and gating are left in PyTorch – they
are memory‑bound and cheap compared to the N³ hidden‑dim work.  The kernel
computes

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

for every batch `b`, sequence positions `i,j` and hidden dimension `d`.
The implementation tiles the `(i,j)` matrix with 2‑D blocks and iterates over
the `k` dimension in tiles, accumulating a per‑channel sum in fp32.
Mixed‑precision (fp16 for the inputs, fp32 accumulation) is used and the
final tensor is cast back to float32 to match the reference output.
"""
import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------------------- #
# Triton kernel: tiled N×N matrix where each element is a vector of length H
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_einsum_kernel(
    out_ptr,                # *[B, N, N, H]   output
    left_ptr,               # *[B, N, N, H]   left projection (masked & gated)
    right_ptr,              # *[B, N, N, H]   right projection (masked & gated)
    B, N, H,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_l_b,  stride_l_i,  stride_l_k,  stride_l_h,
    stride_r_b,  stride_r_j,  stride_r_k,  stride_r_h,
    BLOCK_M: tl.constexpr,  # tile size in i
    BLOCK_N: tl.constexpr,  # tile size in j
    BLOCK_K: tl.constexpr,  # tile size in k
    BLOCK_H: tl.constexpr   # tile size in hidden dim
):
    pid_b = tl.program_id(2)                     # batch dimension
    pid_i = tl.program_id(0)                     # i‑tile index
    pid_j = tl.program_id(1)                     # j‑tile index

    # ------------------------------------------------------------------- #
    # 1‑D offsets for the current tile
    # ------------------------------------------------------------------- #
    i_base = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    h_base = tl.arange(0, BLOCK_H)

    mask_i = i_base < N
    mask_j = j_base < N
    mask_h = h_base < H

    # ------------------------------------------------------------------- #
    #  Accumulator: shape (BLOCK_M, BLOCK_N, BLOCK_H) in fp32
    # ------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    #  Loop over the k‑dimension (the “summation” axis)
    # ------------------------------------------------------------------- #
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_base < N

        # --------------------------------------------------------------- #
        #  Load left:   [i, k, h]   (broadcast batch)
        # --------------------------------------------------------------- #
        left_ptrs = (
            left_ptr
            + pid_b * stride_l_b
            + i_base[:, None, None] * stride_l_i
            + k_base[None, :, None] * stride_l_k
            + h_base[None, None, :] * stride_l_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        # left is fp16 (or fp32); promote to fp32 for accumulation
        left = left.to(tl.float32)

        # --------------------------------------------------------------- #
        #  Load right:  [j, k, h]   (note the j index comes first)
        # --------------------------------------------------------------- #
        right_ptrs = (
            right_ptr
            + pid_b * stride_r_b
            + j_base[:, None, None] * stride_r_j
            + k_base[None, :, None] * stride_r_k
            + h_base[None, None, :] * stride_r_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        right = right.to(tl.float32)

        # --------------------------------------------------------------- #
        #  Element‑wise multiplication over the hidden dimension and sum
        #  across k (the inner reduction).  We keep the hidden dimension
        #  inside the accumulator, so we just add the product.
        # --------------------------------------------------------------- #
        # left : (M, K, H)   -> (M, K, H)
        # right: (N, K, H)   -> (N, K, H)
        # We need to multiply left[i,k,h] * right[j,k,h] and sum over k.
        # Expand dims to broadcast i↔j:
        left_exp = left[:, None, :, :]          # (M,1,K,H)
        right_exp = right[None, :, :, :]        # (1,N,K,H)
        prod = left_exp * right_exp              # (M,N,K,H)
        acc += tl.sum(prod, axis=2)               # sum over K → (M,N,H)

    # ------------------------------------------------------------------- #
    #  Store the result
    # ------------------------------------------------------------------- #
    out_ptrs = (
        out_ptr
        + pid_b * stride_out_b
        + i_base[:, None, None] * stride_out_i
        + j_base[None, :, None] * stride_out_j
        + h_base[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


# --------------------------------------------------------------------------- #
# Helper to launch the kernel for a single batch (the kernel itself loops over B)
# --------------------------------------------------------------------------- #
def launch_triton_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right:  [B, N, N, H] (fp16 or fp32)
    returns      : [B, N, N, H] (fp32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)

    # Tile sizes – tuned for Hopper (sm_90).  They fit into 64 KB shared memory.
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 32   # hidden dim is a multiple of 32 in all test configs

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B,                                # batch
    )

    trimul_einsum_kernel[grid](
        out_ptr=out,
        left_ptr=left,
        right_ptr=right,
        B=B,
        N=N,
        H=H,
        stride_out_b=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_h=out.stride(3),
        stride_l_b=left.stride(0),
        stride_l_i=left.stride(1),
        stride_l_k=left.stride(2),
        stride_l_h=left.stride(3),
        stride_r_b=right.stride(0),
        stride_r_j=right.stride(1),
        stride_r_k=right.stride(2),
        stride_r_h=right.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : (input, mask, weights, config)
        input   – Tensor [B, N, N, dim]  (float32)
        mask    – Tensor [B, N, N]       (bool/float, 1.0 for keep)
        weights – dict with all linear / LN parameters (float32)
        config  – dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------- #
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------- #
    norm = nn.LayerNorm(dim, elementwise_affine=False).to(device)
    norm.weight = nn.Parameter(weights["norm.weight"])
    norm.bias = nn.Parameter(weights["norm.bias"])

    x = norm(input_tensor)                # [B,N,N,dim]

    # ------------------------------------------------------------------- #
    # 2️⃣ Linear projections (no bias) – keep them in fp16 for speed
    # ------------------------------------------------------------------- #
    left_proj = nn.Linear(dim, hidden_dim, bias=False).to(device)
    right_proj = nn.Linear(dim, hidden_dim, bias=False).to(device)

    left_proj.weight = nn.Parameter(weights["left_proj.weight"])
    right_proj.weight = nn.Parameter(weights["right_proj.weight"])

    # cast to fp16 before matmul to improve bandwidth, store as fp16
    x_fp16 = x.to(torch.float16)
    left = left_proj(x_fp16)          # [B,N,N,hidden_dim]  fp16
    right = right_proj(x_fp16)        # [B,N,N,hidden_dim]  fp16

    # ------------------------------------------------------------------- #
    # 3️⃣ Apply mask (if present) – mask is broadcasted on hidden dim
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid) – keep in fp16, then back‑to‑fp16 multiply
    # ------------------------------------------------------------------- #
    left_gate = nn.Linear(dim, hidden_dim, bias=False).to(device)
    right_gate = nn.Linear(dim, hidden_dim, bias=False).to(device)
    out_gate = nn.Linear(dim, hidden_dim, bias=False).to(device)

    left_gate.weight = nn.Parameter(weights["left_gate.weight"])
    right_gate.weight = nn.Parameter(weights["right_gate.weight"])
    out_gate.weight = nn.Parameter(weights["out_gate.weight"])

    left_gate_val = left_gate(x_fp16).sigmoid()
    right_gate_val = right_gate(x_fp16).sigmoid()
    out_gate_val = out_gate(x_fp16).sigmoid()        # [B,N,N,hidden_dim]

    left = left * left_gate_val
    right = right * right_gate_val

    # ------------------------------------------------------------------- #
    # 5️⃣ Heavy einsum – GPU fusion via Triton
    # ------------------------------------------------------------------- #
    # The kernel returns fp32, matching the reference implementation.
    out = launch_triton_einsum(left, right)          # [B,N,N,hidden_dim] fp32

    # ------------------------------------------------------------------- #
    # 6️⃣ Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------- #
    to_out_norm = nn.LayerNorm(hidden_dim, elementwise_affine=False).to(device)
    to_out = nn.Linear(hidden_dim, dim, bias=False).to(device)

    to_out_norm.weight = nn.Parameter(weights["to_out_norm.weight"])
    to_out_norm.bias = nn.Parameter(weights["to_out_norm.bias"])
    to_out.weight = nn.Parameter(weights["to_out.weight"])

    out = to_out_norm(out)               # [B,N,N,hidden_dim] fp32
    out = out * out_gate_val             # gating in hidden dim
    out = to_out(out)                    # [B,N,N,dim] fp32

    return out
