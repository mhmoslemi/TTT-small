"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part is the triple‑wise contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are already masked and gated.
We fuse this N‑by‑N reduction over the shared dimension *k* with a
Triton block‑GEMM.  All surrounding LayerNorm/Linear/Gate work is kept in
PyTorch (fast‑vectorised and already fused by torch.compile).  The Triton
kernel is launched with a 3‑D grid (batch‑head, i‑block, j‑block) and
produces fp16 accumulators that are converted back to fp32 for the final
LayerNorm and output projection.

The implementation supports any batch size (1‑2), sequence length up to
1024 and hidden dimension 128 (as required by the tests).  Masks are
optional – if `mask` is `None` the kernel works on the un‑masked tensors.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel:   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,        # const dtype*   (B,N,N,HD)
    right_ptr,       # const dtype*   (B,N,N,HD)
    out_ptr,         #       dtype*   (B,N,N,HD)
    # sizes
    B, N, HD,
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    FP16: tl.constexpr,
):
    """Block‑wise GEMM over the shared `k` dimension."""
    pid_bhd = tl.program_id(0)                 # combined batch * hidden index
    pid_i   = tl.program_id(1)                 # i‑block
    pid_j   = tl.program_id(2)                 # j‑block

    # decode batch and head indices
    b = pid_bhd // HD
    d = pid_bhd %  HD

    # block start coordinates
    i_off = pid_i * BLOCK_M
    j_off = pid_j * BLOCK_N

    # mask for out‑of‑bounds tiles
    mask_i = i_off + tl.arange(0, BLOCK_M) < N
    mask_j = j_off + tl.arange(0, BLOCK_N) < N

    # allocate accumulators
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K
        cur_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # -------------------- load left --------------------
        # left[b, i, k, d]   shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            d * stride_left_d + \
            (i_off + tl.arange(0, BLOCK_M))[:, None] * stride_left_i + \
            cur_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # -------------------- load right -------------------
        # right[b, j, k, d]  shape (BLOCK_N, BLOCK_K)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            d * stride_right_d + \
            (j_off + tl.arange(0, BLOCK_N))[:, None] * stride_right_j + \
            cur_k[None, :] * stride_right_k
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0)

        # optional fp16 compute (FP16=False -> stay fp32)
        if FP16:
            left  = left.to(tl.float16)
            right = right.to(tl.float16)
            acc   = acc + tl.dot(left, right).to(tl.float32)
        else:
            acc   = acc + tl.dot(left, right)

    # write result
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        d * stride_out_d + \
        (i_off + tl.arange(0, BLOCK_M))[:, None] * stride_out_i + \
        (j_off + tl.arange(0, BLOCK_N))[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Helper that launches the kernel
# --------------------------------------------------------------
def trimul_outgoing_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : (B, N, N, HD)   dtype float16 or float32
    returns out : (B, N, N, HD)   dtype float32
    """
    B, N, _, HD = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in elements
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # block sizes (tuned for H200, N≤1024, HD=128)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    FP16 = int(left.dtype == torch.float16)  # compute in fp16 if possible

    grid = (
        B * HD,                         # program_id(0) : batch*head
        triton.cdiv(N, BLOCK_M),       # i‑tiles
        triton.cdiv(N, BLOCK_N),       # j‑tiles
    )
    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, HD=HD,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_d=stride_left_d,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_d=stride_right_d,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_d=stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        FP16=FP16,
        num_warps=4,
    )
    return out


# --------------------------------------------------------------
# Main entry point required by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  float32
        - mask         : [B, N, N]     (bool or 0/1) – may be None
        - weights      : dict of pre‑loaded parameters
        - config       : dict with keys ``dim`` and ``hidden_dim``
    Returns
    -------
    torch.Tensor
        [B, N, N, dim]  float32
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B,N,N,dim)  float32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    #   left  = x @ left_proj_weight.T
    #   right = x @ right_proj_weight.T
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask if present (mask shape [B,N,N] -> [...,1])
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)    # broadcast over hidden
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid) and gating of projected tensors
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction – Triton kernel (use fp16 for speed)
    # ------------------------------------------------------------------
    # Cast to fp16 before the kernel if the hardware supports it
    left_k = left.half()
    right_k = right.half()
    out = trimul_outgoing_torch(left_k, right_k)          # (B,N,N,hidden) float32

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])        # (B,N,N,dim)

    return out.float()
