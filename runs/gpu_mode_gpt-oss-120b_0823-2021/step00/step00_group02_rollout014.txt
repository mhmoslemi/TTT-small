"""
TriMul “outgoing” implementation.

The heavy N³ part is a batched “pairwise‑dot”:
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

We keep LayerNorms and Linear projections in PyTorch (they are cheap)
and accelerate the triple‑product with a Triton kernel that tiles over the
summation axis `k` and the hidden dimension `d`.  Each program instance
computes one (b,i,j) tile and reduces over k for all d in a vectorized
fashion.  Mixed‑precision is used inside the kernel (float16 loads,
float32 accumulation) and the final result is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes the N³ reduction for a single (b,i,j) tile.
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,            # *float16  [B,N,N,H]
    right_ptr,           # *float16  [B,N,N,H]
    out_ptr,             # *float32  [B,N,N,H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    """
    Each program computes output for a fixed (b,i,j) and all hidden channels.
    """
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # first sequence index
    pid_j = tl.program_id(2)          # second sequence index

    # offsets for the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)

    # accumulator in fp32
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    # loop over k in tiles of size BLOCK_K
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K) + k_offset

        # mask for tails
        mask_k = offs_k < N
        mask_h = offs_h < H
        mask = mask_k[:, None] & mask_h[None, :]

        # pointers for the current tile
        left_tile_ptr = (
            left_ptr
            + pid_b * stride_left_b
            + pid_i * stride_left_i
            + offs_k[:, None] * stride_left_k
            + offs_h[None, :] * stride_left_h
        )
        right_tile_ptr = (
            right_ptr
            + pid_b * stride_right_b
            + pid_j * stride_right_j
            + offs_k[:, None] * stride_right_k
            + offs_h[None, :] * stride_right_h
        )

        # load fp16, apply mask, zero‑fill out‑of‑bounds
        left = tl.load(left_tile_ptr, mask=mask, other=0.0).to(tl.float32)
        right = tl.load(right_tile_ptr, mask=mask, other=0.0).to(tl.float32)

        # reduce over k (axis 0)
        acc += tl.sum(left * right, axis=0)

    # store the result (fp32)
    out_tile_ptr = (
        out_ptr
        + pid_b * stride_out_b
        + pid_i * stride_out_i
        + pid_j * stride_out_j
        + offs_h * stride_out_h
    )
    tl.store(out_tile_ptr, acc, mask=offs_h < H)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass with a fused Triton kernel for the
    N³ reduction.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.float32 tensor [B, N, N, dim]
            - mask:          torch.float32 or bool tensor [B, N, N]
            - weights: dict of pretrained parameters
            - config:  dict with keys dim, hidden_dim

    Returns:
        torch.float32 tensor [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    dim, hidden = config["dim"], config["hidden_dim"]
    device = input_tensor.device
    eps = 1e-5

    # ---------- LayerNorm on the input ----------
    # torch.nn.functional.layer_norm expects normalized_shape and weight/bias
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]

    # ---------- Linear projections (no bias) ----------
    # shape: [B,N,N,hidden]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask ----------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Triton kernel for the N³ reduction ----------
    B, N, _, _ = left.shape
    # cast to fp16 for the kernel (good perf on Hopper)
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides (in elements, not bytes) for each tensor
    stride_left = [s // left_f16.element_size() for s in left_f16.stride()]   # (b,i,k,d)
    stride_right = [s // right_f16.element_size() for s in right_f16.stride()]  # (b,j,k,d)
    stride_out = [s // out.element_size() for s in out.stride()]               # (b,i,j,d)

    # Kernel launch parameters
    BLOCK_K = 32          # tile size along the summed dimension
    BLOCK_H = 32          # tile size for hidden (must divide hidden or be padded)

    grid = (B, N, N)      # one program per (b,i,j)

    trimul_kernel[grid](
        left_f16,
        right_f16,
        out,
        B, N, hidden,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # ---------- Final LayerNorm, gating and linear ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
