"""
TriMul “outgoing” implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input (`x`) with the provided LN weight/bias.
2. Linear projections (left/right) + their element‑wise gates (sigmoid).
   The mask (if present) is applied before gating.
3. The core N³ operation:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply over the sequence dimension N for
   every hidden‑channel `d`.  We flatten the (batch, hidden) dimensions
   and launch a custom Triton kernel that performs a tiled GEMM
   (no‑bias, no‑activation) accumulating in FP32 and storing the result
   in FP32.
4. Apply a second LayerNorm, multiply by the output‑gate and a final
   linear projection back to the original `dim`.

The Triton kernel computes a standard blocked GEMM:

   C = A @ Bᵀ   where
       A : (B*H, N, K)   (left)
       B : (B*H, N, K)   (right)
       C : (B*H, N, N)

The kernel is written for float16 inputs (to exploit Tensor‑cores) but
accumulates in float32 to keep numerical stability; the final result is
cast back to float32 before the remaining PyTorch post‑processing.
"""


import torch
import torch.nn as nn
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ  (A,B : (batch, N, K))
# --------------------------------------------------------------
@triton.jit
def batch_gemm_T_kernel(
    A,                     # *float16   (batch, M, K)
    B,                     # *float16   (batch, N, K)
    C,                     # *float32   (batch, M, N)  output
    batch, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bn, stride_Bk,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # which (batch*hidden) instance
    pid_m = tl.program_id(1)              # tile row
    pid_n = tl.program_id(2)              # tile col

    # offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the tile for this batch
    a_ptrs = A + pid_batch * stride_Ab + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = B + pid_batch * stride_Bb + offs_n[:, None] * stride_Bn + tl.arange(0, BLOCK_K)[None, :] * stride_Bk

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load blocks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K), other=0.0)

        # matmul: a (M,K)  *  bᵀ (N,K)ᵀ  -> (M,N)
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # write back
    c_ptrs = C + pid_batch * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Helper: launch the kernel
# --------------------------------------------------------------
def batch_gemm_T(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for tensors of shape (B, N, N, H) -> (B, N, N, H)
    but flattened to (B*H, N, N).  A and B must be float16 CUDA tensors.
    Returns float32 tensor of same shape.
    """
    assert A.is_cuda and B.is_cuda
    BATCH, N, _, H = A.shape
    assert A.shape == B.shape
    # flatten batch*hidden dimension
    A_f = A.permute(0, 3, 1, 2).contiguous().view(BATCH * H, N, N)   # (B*H, N, K)
    B_f = B.permute(0, 3, 1, 2).contiguous().view(BATCH * H, N, N)   # (B*H, N, K)

    batch, M, K = A_f.shape
    _, N, _ = B_f.shape

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch,
            triton.cdiv(M, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    C_f = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    batch_gemm_T_kernel[grid](
        A_f,
        B_f,
        C_f,
        batch,
        M,
        N,
        K,
        A_f.stride(0), A_f.stride(1), A_f.stride(2),
        B_f.stride(0), B_f.stride(1), B_f.stride(2),
        C_f.stride(0), C_f.stride(1), C_f.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    C = C_f.view(BATCH, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]   torch.float32 CUDA
            mask         : [B, N, N]       torch.float32 CUDA (or bool)
            weights      : dict of parameters (all torch.float32 CUDA)
            config       : dict with keys "dim", "hidden_dim"
    Returns
    -------
    out : torch.Tensor  [B, N, N, dim]  (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ---------- LayerNorm ----------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # compute mean & var across last dim (dim)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + 1e-5)
    x = x * norm_w + norm_b

    # ---------- Linear projections (no bias) ----------
    # shape: (dim, hidden)
    left_proj_w = weights["left_proj.weight"]          # (hidden, dim) in PyTorch Linear weight.T
    right_proj_w = weights["right_proj.weight"]
    left = torch.einsum('bijk,hd->bijkh', x, left_proj_w)   # [B,N,N,hidden]
    right = torch.einsum('bijk,hd->bijkh', x, right_proj_w)

    # ---------- Mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)                      # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gates ----------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.einsum('bijk,hd->bijkh', x, left_gate_w).sigmoid()
    right_gate = torch.einsum('bijk,hd->bijkh', x, right_gate_w).sigmoid()
    out_gate = torch.einsum('bijk,hd->bijkh', x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Core N³ multiplication (Triton) ----------
    # Convert to fp16 for the kernel (Tensor‑core friendly)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # output shape: [B,N,N,hidden]  (float32)
    out = batch_gemm_T(left_fp16, right_fp16)          # fp32 result

    # ---------- Post‑processing ----------
    # to_out_norm (LayerNorm over hidden dim)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]

    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out = (out - mean_o) * torch.rsqrt(var_o + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate

    # final linear projection back to dim
    to_out_w = weights["to_out.weight"]                # (dim, hidden)
    out = torch.einsum('bijkh,dh->bijkd', out, to_out_w)

    return out.contiguous()
