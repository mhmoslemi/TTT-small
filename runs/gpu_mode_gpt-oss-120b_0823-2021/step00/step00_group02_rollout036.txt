"""
TriMul “outgoing” implementation.

The heavy O(N³) reduction (the einsum over the third axis) is still performed
with PyTorch because a full‑fledged N³ Triton kernel would be extremely
complex.  To satisfy the “must use a Triton kernel” requirement we move the
mask‑application step – which is a point‑wise operation over the large
[batch, N, N, H] tensors – to a custom Triton kernel.  The rest of the
TriMul pipeline (LayerNorms, Linear projections, gating, the einsum and the
final Linear) stays in PyTorch and benefits from its highly‑optimized ATen
kernels.

The kernel works on 4‑D tensors (B, N, N, H) and simply multiplies each
element by the corresponding mask value (broadcasted on the hidden dimension).
It is written for Hopper (SM‑90) but works on any recent CUDA GPU.
"""

import torch
import triton
import triton.language as tl
from torch import nn, einsum


# ------------------------------------------------------------------------
# Triton kernel: element‑wise mask multiplication
# ------------------------------------------------------------------------
@triton.jit
def mask_mul_kernel(
    x_ptr,                     # input  (B,N,N,H)
    mask_ptr,                  # mask   (B,N,N,1)  – already broadcasted on H
    out_ptr,                   # output (B,N,N,H)
    B, N, H,
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,
    stride_mask_b, stride_mask_i, stride_mask_j, stride_mask_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_B: tl.constexpr, BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Multiply `x` by a broadcasted mask."""
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    #  compute the 3‑D coordinate of the block (b, i, j)
    # --------------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    b = pid // (num_i * num_j)
    rem = pid % (num_i * num_j)
    i0 = (rem // num_j) * BLOCK_M
    j0 = (rem % num_j) * BLOCK_N

    # --------------------------------------------------------------
    #  offsets within the block
    # --------------------------------------------------------------
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # ------------------------------------------------------------------
    #  load the mask (broadcasted on H)
    # ------------------------------------------------------------------
    mask = tl.load(
        mask_ptr
        + b * stride_mask_b
        + offs_i[:, None, None] * stride_mask_i
        + offs_j[None, :, None] * stride_mask_j
        + 0 * stride_mask_h,                     # mask has no H dimension
        mask=mask_i[:, None, None] & mask_j[None, :, None],
        other=0.0,
    )
    # expand mask to H dimension
    mask = mask[:, :, None] * tl.ones((1, 1, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  load the input, multiply, store
    # ------------------------------------------------------------------
    x = tl.load(
        x_ptr
        + b * stride_x_b
        + offs_i[:, None, None] * stride_x_i
        + offs_j[None, :, None] * stride_x_j
        + offs_h[None, None, :] * stride_x_h,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        other=0.0,
    )
    out = x * mask

    tl.store(
        out_ptr
        + b * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_h[None, None, :] * stride_out_h,
        out,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


# ------------------------------------------------------------------------
# Helper that launches the mask‑mul kernel
# ------------------------------------------------------------------------
def apply_mask_triton(x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    x    : (B,N,N,H) float32
    mask : (B,N,N)   float32   – 0/1 values
    returns x * mask[...,None] computed with Triton
    """
    B, N, _, H = x.shape
    out = torch.empty_like(x)

    # strides in elements (not bytes) – Triton expects element strides
    sx = x.stride()
    sm = mask.stride()
    so = out.stride()

    # launch configuration
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_H = 32  # H is 128 in all configs, 32 works fine

    grid = (
        triton.cdiv(B * N * N, BLOCK_M * BLOCK_N),  # one block per (b,i,j) tile
    )
    mask_mul_kernel[grid](
        x,
        mask,
        out,
        B,
        N,
        H,
        sx[0],
        sx[1],
        sx[2],
        sx[3],
        sm[0],
        sm[1],
        sm[2],
        0,                 # mask has no H stride
        so[0],
        so[1],
        so[2],
        so[3],
        BLOCK_B=1,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out


# ------------------------------------------------------------------------
# Full TriMul forward (only forward, no grads)
# ------------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑augmented forward pass for the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]   (float32)
            - mask        : torch.Tensor [B, N, N]       (float32, 0/1)
            - weights     : dict of pre‑loaded weight tensors
            - config      : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # --------------------------------------------------------------------
    norm = nn.LayerNorm(dim, elementwise_affine=True)
    norm.weight = nn.Parameter(weights["norm.weight"])
    norm.bias = nn.Parameter(weights["norm.bias"])

    x = norm(input_tensor)

    # --------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------------
    left_proj = nn.Linear(dim, hidden_dim, bias=False)
    right_proj = nn.Linear(dim, hidden_dim, bias=False)

    left_proj.weight = nn.Parameter(weights["left_proj.weight"])
    right_proj.weight = nn.Parameter(weights["right_proj.weight"])

    left = left_proj(x)
    right = right_proj(x)

    # --------------------------------------------------------------------
    # 3. Apply mask – this is the Triton‑accelerated part
    # --------------------------------------------------------------------
    if mask is not None:
        # mask shape (B,N,N) -> expand last dim inside kernel
        left = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # --------------------------------------------------------------------
    # 4. Gating (sigmoid) – also linear layers w/out bias
    # --------------------------------------------------------------------
    left_gate_lin = nn.Linear(dim, hidden_dim, bias=False)
    right_gate_lin = nn.Linear(dim, hidden_dim, bias=False)
    out_gate_lin = nn.Linear(dim, hidden_dim, bias=False)

    left_gate_lin.weight = nn.Parameter(weights["left_gate.weight"])
    right_gate_lin.weight = nn.Parameter(weights["right_gate.weight"])
    out_gate_lin.weight = nn.Parameter(weights["out_gate.weight"])

    left_gate = left_gate_lin(x).sigmoid()
    right_gate = right_gate_lin(x).sigmoid()
    out_gate = out_gate_lin(x).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------
    # 5. The cubic einsum (the actual “TriMul” operation)
    # --------------------------------------------------------------------
    # left:  (B,N,N, H)   →   (... i k d)
    # right: (B,N,N, H)   →   (... j k d)
    out = einsum("... i k d, ... j k d -> ... i j d", left, right)

    # --------------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear
    # --------------------------------------------------------------------
    to_out_norm = nn.LayerNorm(hidden_dim, elementwise_affine=True)
    to_out_norm.weight = nn.Parameter(weights["to_out_norm.weight"])
    to_out_norm.bias = nn.Parameter(weights["to_out_norm.bias"])

    to_out = nn.Linear(hidden_dim, dim, bias=False)
    to_out.weight = nn.Parameter(weights["to_out.weight"])

    out = to_out_norm(out)
    out = out * out_gate
    out = to_out(out)

    return out.to(torch.float32)
