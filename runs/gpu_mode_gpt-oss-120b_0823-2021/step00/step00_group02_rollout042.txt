"""
TriMul “outgoing” implementation.

We perform the inexpensive point‑wise parts (layer‑norm, linear projections,
gating, mask) with a tiny Triton kernel that fuses the element‑wise
multiplications.  The heavy N³ “einsum” is reduced to a batch of
matrix‑multiplications:
    out[b, i, j, h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
which is exactly
    out_h = left_h @ right_hᵀ                    (B‑batch, N×N matrix)
We therefore reshape the tensors so that each (batch,head) pair becomes a
single “batch” dimension and use a single torch.bmm (highly tuned cuBLAS) for the
N³ work.  The final normalisation, output‑gate and linear projection are
performed again in PyTorch.  All tensors stay on the CUDA device and the
final result is float‑32.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# -------------------------------------------------------------
# Triton kernel – fuses mask & gating element‑wise multiplications
# -------------------------------------------------------------
@triton.jit
def tri_mul_fuse_kernel(
    left_ptr,               # [B, N, N, H]   (float32)
    right_ptr,              # [B, N, N, H]   (float32)
    mask_ptr,               # [B, N, N, 1]   (float32) – broadcasted mask
    left_gate_ptr,          # [B, N, N, H]   (float32)
    right_gate_ptr,         # [B, N, N, H]   (float32)
    B, N, H,                # static sizes
    stride_left_bhn, stride_left_hn, stride_left_n, stride_left_c,
    stride_right_bhn, stride_right_hn, stride_right_n, stride_right_c,
    stride_mask_bhn, stride_mask_hn, stride_mask_n,
    stride_lgate_bhn, stride_lgate_hn, stride_lgate_n, stride_lgate_c,
    stride_rgate_bhn, stride_rgate_hn, stride_rgate_n, stride_rgate_c,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # linear thread id
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # indices this thread works on
    mask = offs < total

    # ----- decode linear offset into 4‑D indices -----
    # linear = ((b * N + i) * N + k) * H + h
    b = (offs // (N * N * H))
    rest = offs % (N * N * H)
    i = (rest // (N * H))
    rest = rest % (N * H)
    k = (rest // H)
    h = rest % H

    # compute pointer offsets
    left_off = (b * stride_left_bhn +
                i * stride_left_hn +
                k * stride_left_n +
                h * stride_left_c)
    right_off = (b * stride_right_bhn +
                 i * stride_right_hn +
                 k * stride_right_n +
                 h * stride_right_c)
    mask_off = (b * stride_mask_bhn +
                i * stride_mask_hn +
                k * stride_mask_n)                # mask has no channel dim
    lgate_off = (b * stride_lgate_bhn +
                 i * stride_lgate_hn +
                 k * stride_lgate_n +
                 h * stride_lgate_c)
    rgate_off = (b * stride_rgate_bhn +
                 i * stride_rgate_hn +
                 k * stride_rgate_n +
                 h * stride_rgate_c)

    # load values (mask broadcasted to channel dim)
    left  = tl.load(left_ptr + left_off , mask=mask, other=0.0)
    right = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_off , mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + lgate_off, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + rgate_off, mask=mask, other=0.0)

    # fused element‑wise ops
    left  = left  * m * lgate
    right = right * m * rgate

    # store back (in‑place)
    tl.store(left_ptr  + left_off , left ,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


# ----------------------------------------------------------------------
# Helper to launch the above kernel (one‑dimensional launch)
# ----------------------------------------------------------------------
def fuse_mask_and_gates(left, right, mask, left_gate, right_gate):
    """In‑place fuse mask & gates using Triton."""
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                     # 1 k threads per program – good for H200
    grid = (triton.cdiv(total, BLOCK),)

    # strides (in elements, not bytes) – match the layout of the contiguous tensors
    stride_left = left.stride()
    stride_right = right.stride()
    stride_mask = mask.stride()
    stride_lgate = left_gate.stride()
    stride_rgate = right_gate.stride()

    tri_mul_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        # left strides
        stride_left_bhn=stride_left[0],
        stride_left_hn=stride_left[1],
        stride_left_n=stride_left[2],
        stride_left_c=stride_left[3],
        # right strides
        stride_right_bhn=stride_right[0],
        stride_right_hn=stride_right[1],
        stride_right_n=stride_right[2],
        stride_right_c=stride_right[3],
        # mask strides (no channel dim)
        stride_mask_bhn=stride_mask[0],
        stride_mask_hn=stride_mask[1],
        stride_mask_n=stride_mask[2],
        # left‑gate strides
        stride_lgate_bhn=stride_lgate[0],
        stride_lgate_hn=stride_lgate[1],
        stride_lgate_n=stride_lgate[2],
        stride_lgate_c=stride_lgate[3],
        # right‑gate strides
        stride_rgate_bhn=stride_rgate[0],
        stride_rgate_hn=stride_rgate[1],
        stride_rgate_n=stride_rgate[2],
        stride_rgate_c=stride_rgate[3],
        BLOCK=BLOCK,
    )
    # kernel works in‑place, return tensors for convenience
    return left, right


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass with a tiny Triton kernel.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm ----------
    # (B, N, N, dim) -> (B, N, N, dim)
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    eps = 1e-5
    # manual LN to stay in fp32 (weights are fp32)
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b

    # ---------- Linear projections ----------
    # Linear layers without bias: y = x @ W.T
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_w)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_norm, right_proj_w)

    # ---------- Gates ----------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()

    # ---------- Mask handling ----------
    if mask is None:
        mask = torch.ones_like(mask, dtype=left.dtype) if False else None  # dummy
    mask = mask.unsqueeze(-1)                       # (B, N, N, 1)

    # ---------- Fuse mask & gates with Triton ----------
    # The kernel works in‑place on `left` and `right`
    fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ---------- N³ einsum via batched matmul ----------
    # reshape to (B*hidden_dim, N, N) and multiply with its transpose
    B, N, _, H = left.shape
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)      # (B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, N)

    # out_flat = left_flat @ right_flat.transpose(-2, -1)   # (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)   # (B, N, N, hidden_dim)

    # ---------- Output normalization ----------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    eps_o = 1e-5
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps_o)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # ---------- Apply output gate ----------
    out_norm = out_norm * out_gate

    # ---------- Final linear ----------
    to_out_w = weights["to_out.weight"]
    out_final = torch.nn.functional.linear(out_norm, to_out_w)   # (B, N, N, dim)

    return out_final.to(torch.float32)
