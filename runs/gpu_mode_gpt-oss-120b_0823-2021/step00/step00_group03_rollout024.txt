"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The heavy part of the forward pass is the 4‑D Einstein sum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape (B, N, N, H) (H = hidden_dim).
The term is a batched, per‑feature inner‑product over the third axis.
We implement this contraction with a Triton kernel that
* tiles the i‑ and j‑axes (BLOCK_M × BLOCK_N)
* tiles the reduction axis k (BLOCK_K)
* processes a small chunk of the hidden dimension (BLOCK_D) at a time
  using vectorised `tl.dot`.

All LayerNorms, Linear projections and sigmoid gates are left in
regular PyTorch – they are cheap compared with the O(B·N³·H) contraction.
The Triton kernel is launched once per batch element and once per
H‑tile, so the overall memory traffic stays O(B·N²·H) and we achieve
high arithmetic intensity on Hopper (SM‑90) GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------------
# Triton kernel : per‑batch, per‑hidden‑tile contraction
# --------------------------------------------------------------------
@triton.jit
def trimul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    # strides (B dimension is folded into the launch loop)
    stride_li, stride_lk, stride_ld,   # left  : (i, k, d)
    stride_ri, stride_rk, stride_rd,   # right : (i, k, d) – note: right uses i‑axis for “j”
    stride_oi, stride_oj, stride_od,   # out   : (i, j, d)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute
            out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]
       for a single batch element.
    """
    pid = tl.program_id(0)
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    i_base = (pid // num_j) * BLOCK_M
    j_base = (pid %  num_j) * BLOCK_N

    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)
    d = tl.arange(0, BLOCK_D)

    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32 (left/right may be FP16/BF16)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ----------------------------------------------------------------
    # reduction over k
    # ----------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        k_off = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left[i, k, d]   -> shape (M, K, D)
        left_offs = i[:, None, None] * stride_li \
                 + k_off[None, :, None] * stride_lk \
                 + d[None, None, :] * stride_ld
        left = tl.load(left_ptr + left_offs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None],
                       other=0.0)

        # right[j, k, d]  -> shape (N, K, D)
        right_offs = j[:, None, None] * stride_ri \
                  + k_off[None, :, None] * stride_rk \
                  + d[None, None, :] * stride_rd
        right = tl.load(right_ptr + right_offs,
                        mask=mask_j[:, None, None] & mask_k[None, :, None],
                        other=0.0)

        # broadcast to (M, N, K, D) and accumulate over K
        # (M,1,K,D) * (1,N,K,D) -> (M,N,K,D)
        prod = left[:, None, :, :] * right[None, :, :, :]
        acc += tl.sum(prod, axis=2)               # sum over K
    # ----------------------------------------------------------------
    # write result
    # ----------------------------------------------------------------
    out_offs = i[:, None, None] * stride_oi \
            + j[None, :, None] * stride_oj \
            + d[None, None, :] * stride_od
    tl.store(out_ptr + out_offs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None])


# --------------------------------------------------------------------
# Helper that launches the kernel for a whole batch
# --------------------------------------------------------------------
def trimul_contract(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : (B, N, N, H)  (dtype can be fp16/bf16)
    returns    : (B, N, N, H)  (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # tile size – tuned for Hopper, fits in shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 32                     # must divide H (we pad inside kernel)

    # strides for the three tensors (ignore batch stride – we loop over B)
    stride_li = left.stride(2)   # stride on k
    stride_lk = left.stride(1)   # stride on i   (actually N dimension)
    stride_ld = left.stride(3)   # stride on d
    stride_ri = right.stride(2)  # stride on k
    stride_rk = right.stride(1)  # stride on i (j‑axis)
    stride_rd = right.stride(3)  # stride on d
    stride_oi = out.stride(2)
    stride_oj = out.stride(1)
    stride_od = out.stride(3)

    for b in range(B):
        left_b  = left[b]
        right_b = right[b]
        out_b   = out[b]

        grid = (triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),)
        trimul_contract_kernel[grid](
            left_b, right_b, out_b,
            N, H,
            stride_li, stride_lk, stride_ld,
            stride_ri, stride_rk, stride_rd,
            stride_oi, stride_oj, stride_od,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
        )
    return out


# --------------------------------------------------------------------
# Main entry point required by the evaluation harness
# --------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module.
    Only the O(N³·H) contraction is executed on the GPU via Triton;
    everything else (LayerNorm, Linear, sigmoid) stays in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual LN to avoid extra memory allocation
    mean = input_tensor.mean(-1, keepdim=True)
    var  = input_tensor.var(-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x_norm = x_norm * norm_w + norm_b

    # ----------------------------------------------------------------
    # 2) Linear projections (no bias) – we keep them in FP16/BF16 for speed
    # ----------------------------------------------------------------
    # weight shapes: (hidden_dim, dim)
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left  = F.linear(x_norm, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_norm, right_proj_w)

    # ----------------------------------------------------------------
    # 3) Apply mask if present
    # ----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                      # (B,N,N,1)
        left  = left * mask
        right = right * mask

    # ----------------------------------------------------------------
    # 4) Gating (sigmoid) on the original normalized tensor
    # ----------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # 5) Core contraction – Triton
    # ----------------------------------------------------------------
    # Cast to a compact dtype for the kernel (fp16 works well on H200)
    left_ker  = left.to(torch.float16)
    right_ker = right.to(torch.float16)

    out = trimul_contract(left_ker, right_ker)       # (B,N,N,hidden_dim) FP32

    # ----------------------------------------------------------------
    # 6) Final LayerNorm, gating and output projection
    # ----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # LN on hidden dimension
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate                                   # broadcast over H

    # final linear (hidden_dim -> dim)
    out = F.linear(out, to_out_w)                          # (B,N,N,dim)

    return out.to(torch.float32)
