"""
TriMul “outgoing” kernel (alpha‑Fold3).

We keep the LayerNorm / Linear / gating in PyTorch (they are cheap compared to the
N³ reduction) and fuse the costly einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

into a Triton kernel.  The kernel iterates over the reduction dimension *k* in
tiles (BLOCK_K) and accumulates a vector of hidden channels (BLOCK_H).  The
launch grid covers (batch, i, j, hidden_tile).  All arithmetic is performed in
float32 and the final tensor is returned in float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: block‑wise sum‑over‑k for a tile of hidden dim
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # [B, N, N, H]  (i,k,d)
    right_ptr,         # [B, N, N, H]  (j,k,d)
    out_ptr,           # [B, N, N, H]  (i,j,d)

    B, N, H,                 # static sizes
    stride_l_bs, stride_l_i, stride_l_k, stride_l_h,
    stride_r_bs, stride_r_j, stride_r_k, stride_r_h,
    stride_o_bs, stride_o_i, stride_o_j, stride_o_h,

    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    # --------------------------------------------------------------
    # 1) map program id to (b,i,j, h_tile)
    # --------------------------------------------------------------
    pid = tl.program_id(0)                     # linear index over all blocks
    total_ij = N * N
    b   = pid // (total_ij * tl.cdiv(H, BLOCK_H))
    tmp = pid % (total_ij * tl.cdiv(H, BLOCK_H))
    ij  = tmp // tl.cdiv(H, BLOCK_H)
    h_tile = tmp % tl.cdiv(H, BLOCK_H)

    i = ij // N
    j = ij % N

    # offsets for hidden dimension within this tile
    offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # --------------------------------------------------------------
    # 2) allocate accumulator
    # --------------------------------------------------------------
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) loop over k in tiles
    # --------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        # k‑tile offsets
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # pointers into left[b,i,k,d] and right[b,j,k,d]
        left_ptrs  = left_ptr \
            + b   * stride_l_bs \
            + i   * stride_l_i \
            + cur_k[:, None] * stride_l_k \
            + offs_h[None, :] * stride_l_h

        right_ptrs = right_ptr \
            + b   * stride_r_bs \
            + j   * stride_r_j \
            + cur_k[:, None] * stride_r_k \
            + offs_h[None, :] * stride_r_h

        # load tiles (shape: [BLOCK_K, BLOCK_H])
        left_tile  = tl.load(left_ptrs,  mask=mask_k[:,None] & mask_h[None,:], other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_k[:,None] & mask_h[None,:], other=0.0)

        # reduction over k → dot product per hidden channel
        # tl.dot works on 1‑D vectors; we use element‑wise multiply + sum
        acc += tl.sum(left_tile * right_tile, axis=0)

    # --------------------------------------------------------------
    # 4) write back result
    # --------------------------------------------------------------
    out_ptrs = out_ptr \
        + b   * stride_o_bs \
        + i   * stride_o_i \
        + j   * stride_o_j \
        + offs_h * stride_o_h

    tl.store(out_ptrs, acc, mask=mask_h)


# --------------------------------------------------------------
# Host wrapper
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module with a fused Triton kernel.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]   torch.float32 (or fp16/bf16)
            mask         : [B, N, N]        torch.float32 (0/1)
            weights      : dict of torch tensors (see docstring above)
            config       : dict with keys "dim", "hidden_dim"

    Returns:
        out : torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim      = config["dim"]
    hidden   = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ----------------------------------------------------------
    # 1) LayerNorm on the last dim
    # ----------------------------------------------------------
    norm_w = weights["norm.weight"].to(dtype)
    norm_b = weights["norm.bias"].to(dtype)
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_w + norm_b

    # ----------------------------------------------------------
    # 2) Linear projections (no bias) and gating
    # ----------------------------------------------------------
    # shape helpers
    def linear(t, w):
        # t: [B,N,N,dim]  w: [hidden, dim]
        return F.linear(t, w)          # torch performs (t @ w.T)

    left  = linear(x, weights["left_proj.weight"].to(dtype))   # [B,N,N,hidden]
    right = linear(x, weights["right_proj.weight"].to(dtype))

    # mask (optional)
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)          # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    left_gate  = torch.sigmoid(linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(linear(x, weights["out_gate.weight"].to(dtype)))

    left  = left  * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 3) Triton reduction (einsum over k)
    # ----------------------------------------------------------
    # allocate output buffer in fp32 (final dtype)
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # launch configuration
    BLOCK_K = 64          # tile size along reduction dimension k
    BLOCK_H = 32          # tile size along hidden dimension
    grid = (
        triton.cdiv(B * N * N * triton.cdiv(hidden, BLOCK_H), 1),  # flatten all 4‑D blocks
    )

    # strides (in elements) for each tensor
    stride_l_bs, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_bs, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_bs, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # launch
    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_l_bs=stride_l_bs,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_h=stride_l_h,
        stride_r_bs=stride_r_bs,
        stride_r_j=stride_r_j,
        stride_r_k=stride_r_k,
        stride_r_h=stride_r_h,
        stride_o_bs=stride_o_bs,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_h=stride_o_h,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ----------------------------------------------------------
    # 4) Final LayerNorm, gating and linear projection back to dim
    # ----------------------------------------------------------
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * weights["to_out_norm.weight"].to(dtype) + weights["to_out_norm.bias"].to(dtype)
    out = out * out_gate

    # final linear (hidden -> dim)
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    # ensure float32 output
    return out.to(torch.float32)
