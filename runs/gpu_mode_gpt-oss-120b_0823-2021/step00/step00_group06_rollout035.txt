"""
TriMul “outgoing” kernel (AlphaFold3).

We compute
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where
    left  = (x @ W_left )  * mask * sigmoid(x @ W_left_gate)
    right = (x @ W_right)  * mask * sigmoid(x @ W_right_gate)

The heavy N³ part (the sum over *k*) is performed in a custom Triton
kernel.  All LayerNorms, linear projections and gating are kept in
torch (they are cheap compared to the N³ reduction).  The kernel works
on arbitrary batch‑size (1‑2), sequence length and hidden dimension
(≤ 384) and returns a float‑32 tensor ready for the final LayerNorm and
output linear.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched “tri‑mul” reduction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # sizes
    B, N, H,
    # strides
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].

    Grid: (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B)
    Each program works on a (BLOCK_M × BLOCK_N) tile of the (i,j) matrix
    for a single batch element.  The reduction over k is tiled with
    BLOCK_K.
    """
    pid_b = tl.program_id(2)                     # batch dimension
    pid_i = tl.program_id(0)                     # i‑tile
    pid_j = tl.program_id(1)                     # j‑tile

    # --- coordinate ranges -------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialise accumulator for **all** hidden channels
    acc = tl.zeros((BLOCK_M, BLOCK_N, H), dtype=tl.float32)

    # loop over k dimension
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # ------------------------------------------------------------------
        # Load left  : shape (BLOCK_M, BLOCK_K, H)
        # Load right : shape (BLOCK_N, BLOCK_K, H)   (note j‑first)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + pid_b * stride_lb
            + offs_i[:, None, None] * stride_li
            + cur_k[None, :, None] * stride_lk
            + tl.arange(0, H)[None, None, :] * stride_lh
        )
        right_ptrs = (
            right_ptr
            + pid_b * stride_rb
            + offs_j[:, None, None] * stride_rj
            + cur_k[None, :, None] * stride_rk
            + tl.arange(0, H)[None, None, :] * stride_rh
        )

        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0)
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0)

        # left  : (M,K,H)   right : (N,K,H)
        # we need Σ_k left[...,k,d] * right[...,k,d] per (i,j,d)
        # perform element‑wise product then sum over k
        prod = left * right                           # (M,N,K,H) broadcasted over K
        # sum over K
        acc += tl.sum(prod, axis=1)                    # (M,N,H)

    # ----------------------------------------------------------------------
    # Store result
    # ----------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_ob
        + offs_i[:, None, None] * stride_oi
        + offs_j[None, :, None] * stride_oj
        + tl.arange(0, H)[None, None, :] * stride_oh
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None])


# ----------------------------------------------------------------------
# Host side helper
# ----------------------------------------------------------------------
def launch_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H] (float32)
    returns out  : [B, N, N, H] (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes) for each dimension
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # block sizes – tuned for Hopper (sm90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B,
    )
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # good default for H200
    )
    return out


# ----------------------------------------------------------------------
# Full custom kernel (torch + triton)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : (input, mask, weights, config)
        input   – [B, N, N, dim]   float32 cuda tensor
        mask    – [B, N, N]        float32/bool cuda tensor
        weights – dict of pre‑loaded weight tensors
        config  – {'dim': int, 'hidden_dim': int}
    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # torch.nn.functional.layer_norm works on the last dimension
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    def linear_no_bias(x, w):
        # x : [..., dim]   w : [hidden_dim, dim]
        return torch.einsum('...d,hd->...h', x, w)

    left  = linear_no_bias(x, weights["left_proj.weight"])
    right = linear_no_bias(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(linear_no_bias(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_no_bias(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear_no_bias(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. The N³ reduction – our Triton kernel
    # ------------------------------------------------------------------
    # left/right now have shape [B,N,N,hidden_dim]
    out = launch_tri_mul(left, right)          # float32

    # ------------------------------------------------------------------
    # 6. Final LayerNorm + gating + linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    out = out * out_gate                       # broadcast over hidden dim
    # final linear back to dim
    out = torch.einsum('...h,dh->...d', out, weights["to_out.weight"])
    return out
