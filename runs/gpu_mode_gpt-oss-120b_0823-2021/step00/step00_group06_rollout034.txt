"""
TriMul “outgoing” kernel (AlphaFold‑3) – forward pass only.

Algorithm
---------
1.  Layer‑norm the 4‑D input (B,N,N,C)   →   x.
2.  Linear projections (no bias)          →   left, right   (B,N,N,H).
3.  Optional mask (broadcast on the last dim) is applied to left/right.
4.  Gated projections (sigmoid)            →   left_gate, right_gate, out_gate.
5.  left *= left_gate , right *= right_gate.
6.  Core operation (the only part in Triton):
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]      (H‑dim kept)
    This is a batched matrix‑multiply per hidden channel.
    Implemented as a Triton kernel that loops over k‑tiles and over the
    hidden dimension in small blocks (BLOCK_D).  Accumulators are float32.
7.  Layer‑norm over the hidden dimension, gated by out_gate,
    final linear to the original channel dimension.
8.  Return the result in float32.

Only the N³ “einsum” is executed on the GPU via Triton; all other
operations stay in PyTorch for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton kernel: batched per‑channel matrix multiply (outgoing TriMul)
# ---------------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left  [b, i, k, d]
    stride_lb, stride_li, stride_lk, stride_ld,
    # strides for right [b, j, k, d]
    stride_rb, stride_rj, stride_rk, stride_rd,
    # strides for out   [b, i, j, d]
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """ Compute out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] """
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # block along i
    pid_j = tl.program_id(2)          # block along j

    # base offsets for this block
    i_base = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_base < N
    j_mask = j_base < N

    # accumulator: (M,N,D)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over k tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k):
        k_base = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_base < N

        # ----- load left tile -----
        # shape (M, K, D)
        left_ptrs = (
            left_ptr
            + pid_b * stride_lb
            + i_base[:, None, None] * stride_li
            + k_base[None, :, None] * stride_lk
            + tl.arange(0, BLOCK_D)[None, None, :] * stride_ld
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None, None] & k_mask[None, :, None],
            other=0.0,
        )

        # ----- load right tile -----
        # shape (N, K, D)
        right_ptrs = (
            right_ptr
            + pid_b * stride_rb
            + j_base[:, None, None] * stride_rj
            + k_base[None, :, None] * stride_rk
            + tl.arange(0, BLOCK_D)[None, None, :] * stride_rd
        )
        right = tl.load(
            right_ptrs,
            mask=j_mask[:, None, None] & k_mask[None, :, None],
            other=0.0,
        )

        # ----- multiply‑accumulate for each hidden channel block -----
        # left : (M,K,BD)   right : (N,K,BD)
        # we compute, for each d in BD:
        #    acc[:, :, d] += left[:,:,d] @ right[:,:,d].T
        for d_off in range(0, BLOCK_D):
            l = left[:, :, d_off]          # (M,K)
            r = right[:, :, d_off]         # (N,K)
            # tl.dot supports (M,K) @ (K,N) via trans_b=True
            acc[:, :, d_off] += tl.dot(l, r, trans_b=True)

    # ----- store result -----
    out_ptrs = (
        out_ptr
        + pid_b * stride_ob
        + i_base[:, None, None] * stride_oi
        + j_base[None, :, None] * stride_oj
        + tl.arange(0, BLOCK_D)[None, None, :] * stride_od
    )
    tl.store(
        out_ptrs,
        acc,
        mask=i_mask[:, None, None] & j_mask[None, :, None],
    )

# ---------------------------------------------------------------------------
# Host entry point
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    N³ multiplication. All other layers (LayerNorm, Linear, gating) are
    performed with PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden = config["hidden_dim"]

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections (no bias) ----------
    # shape: (B,N,N,hidden)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)               # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Triton kernel for the core operation ----------
    # Output buffer
    out = torch.empty((bs, N, N, hidden), dtype=torch.float32, device=device)

    # Strides (in elements) for the tensors
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # Block sizes – tuned for Hopper (SM90)
    BLOCK_M = 64   # i dimension
    BLOCK_N = 64   # j dimension
    BLOCK_K = 32   # k tile
    BLOCK_D = 32   # hidden dim tile (hidden ≤ 384)

    grid = (
        bs,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=N,
        H=hidden,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_ld=stride_ld,
        stride_rb=stride_rb,
        stride_rj=stride_rj,
        stride_rk=stride_rk,
        stride_rd=stride_rd,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_od=stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ---------- Post‑processing ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
