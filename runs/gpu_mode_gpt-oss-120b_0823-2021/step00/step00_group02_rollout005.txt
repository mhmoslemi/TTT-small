"""
TriMul “outgoing” forward pass (no grads).

The heavy  N³  operation is
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]          (1)

where left/right are the projected, gated and masked tensors.
We implement (1) with a Triton kernel that blocks over the i‑ and j‑axes and
accumulates over k.  The surrounding linear layers, LayerNorms and sigmoids are
performed with PyTorch (they are cheap compared with the N³ work).  The kernel
uses float32 accumulation and writes a float32 result; the rest of the model
expects float32, so no cast is needed.

The kernel processes a block (BLOCK_M × BLOCK_N) of the output matrix for a
single batch element and a single hidden dimension slice (BLOCK_H).  For each
output element it sums the element‑wise product over the k‑dimension in blocks
of size BLOCK_K.  The implementation relies on `tl.dot`, which efficiently
computes a matrix‑matrix product for each hidden slice.
"""

import torch
import triton
import triton.language as tl
from torch import nn
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel implementing the N³ tensor contraction (1)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # *[B, N, N, H] left tensor
    right_ptr,           # *[B, N, N, H] right tensor
    out_ptr,             # *[B, N, N, H] output tensor
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)                     # linear index over (b, h_block, i_block, j_block)
    num_b = B
    num_h = tl.cdiv(H, BLOCK_H)
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    # de‑compose pid
    b = pid // (num_h * num_i * num_j)
    tmp = pid % (num_h * num_i * num_j)
    h_block = tmp // (num_i * num_j)
    tmp = tmp % (num_i * num_j)
    i_block = tmp // num_j
    j_block = tmp % num_j

    # offsets for the current tile
    offs_i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_h = h_block * BLOCK_H + tl.arange(0, BLOCK_H)

    # masks for out‑of‑bounds (only needed on the edges)
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # pointers to the beginning of the tile
    left_tile_ptr  = left_ptr \
        + b * stride_l_b \
        + offs_i[:, None, None] * stride_l_i \
        + offs_k[None, :, None] * stride_l_k \
        + offs_h[None, None, :] * stride_l_h

    right_tile_ptr = right_ptr \
        + b * stride_r_b \
        + offs_j[:, None, None] * stride_r_j \
        + offs_k[None, :, None] * stride_r_k \
        + offs_h[None, None, :] * stride_r_h

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # iterate over k‑blocks
    for kb in range(0, N, BLOCK_K):
        cur_k = kb + offs_k
        cur_k_mask = cur_k < N

        # ----- load left block -----
        # shape (M, K, H)
        left = tl.load(
            left_tile_ptr,
            mask=mask_i[:, None, None] & cur_k_mask[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # ----- load right block -----
        # shape (N, K, H)
        right = tl.load(
            right_tile_ptr,
            mask=mask_j[:, None, None] & cur_k_mask[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # compute contribution for each hidden slice using tl.dot
        # we need: out_ijh += Σ_k left[i,k,h] * right[j,k,h]
        # tl.dot works on 2‑D matrices, so we loop over h blocks
        for hh in range(0, BLOCK_H):
            if not mask_h[hh]:
                continue
            # (M, K) dot (N, K)^T  → (M, N)
            left_h  = left[:, :, hh]          # (M, K)
            right_h = right[:, :, hh]         # (N, K)
            # note: tl.dot expects (M, K) @ (K, N)
            prod = tl.dot(left_h, right_h, trans_b=True)   # (M, N)
            acc[:, :, hh] += prod

        # advance pointers to next k‑block
        left_tile_ptr  += BLOCK_K * stride_l_k
        right_tile_ptr += BLOCK_K * stride_r_k

    # ----- write result -----
    out_tile_ptr = out_ptr \
        + b * stride_o_b \
        + offs_i[:, None, None] * stride_o_i \
        + offs_j[None, :, None] * stride_o_j \
        + offs_h[None, None, :] * stride_o_h

    tl.store(
        out_tile_ptr,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


# ----------------------------------------------------------------------
# Helper functions (LayerNorm + Linear with pre‑loaded weights)
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor) -> torch.Tensor:
    """Fused LayerNorm (no eps handling – default eps=1e‑5)."""
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    return (x - mean) * torch.rsqrt(var + 1e-5) * weight + bias


def linear(x: torch.Tensor, weight: torch.Tensor) -> torch.Tensor:
    """Linear without bias: y = x @ weight.T   (weight: out_dim × in_dim)."""
    return torch.nn.functional.linear(x, weight)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: (input, mask, weights, config)
            - input  : [B, N, N, dim]  (float32)
            - mask   : [B, N, N]      (float32/bool) – 1 where valid
            - weights: dict of model parameters
            - config : dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm ----------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )

    # ---------- Projections ----------
    left  = linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = linear(x, weights["right_proj.weight"])

    # ---------- Apply mask (if any) ----------
    if mask is not None:
        mask_ext = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left  = left * mask_ext
        right = right * mask_ext

    # ---------- Gates ----------
    left_gate  = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ---------- Triton contraction ----------
    # allocate output buffer
    out = torch.empty_like(left)  # same shape [B,N,N,hidden_dim]

    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # launch configuration
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 32

    grid = (
        triton.cdiv(B, 1)
        * triton.cdiv(hidden_dim, BLOCK_H)
        * triton.cdiv(N, BLOCK_M)
        * triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # ---------- Output post‑processing ----------
    out = layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = linear(out, weights["to_out.weight"])   # back to [B,N,N,dim]

    return out.float()
