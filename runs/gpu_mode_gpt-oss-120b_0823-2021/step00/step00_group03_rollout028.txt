"""
TriMul “outgoing” implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm + linear projections + sigmoid‑gates – performed in PyTorch
   (the cost is negligible compared to the N³ interaction).
2. The core operation is
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is exactly a batched matrix‑multiply of the
        (i,k) matrix with the (j,k) matrix transposed.
   We reshape the tensors so that each (batch, d) pair becomes one
   independent matrix‑multiply and launch a Triton GEMM kernel.
3. The GEMM kernel follows the classic “persistent‑workspace” pattern
   (see Triton examples).  It accumulates fp32 results, stores fp16
   intermediate values for speed and finally converts back to fp32.
4. The final LayerNorm, out‑gate and output linear are again done in PyTorch
   and the result is returned as float32.

The kernel works for any batch size (1‑2), sequence length up to 1024
and hidden dimensions that are a multiple of 32 (the test config uses 128).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton GEMM kernel (batch‑gemm with fp16 inputs, fp32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    a_ptr,                # left  matrix   (B*H, M, K)   fp16
    b_ptr,                # right matrix   (B*H, N, K)   fp16   (note: K is NOT transposed!)
    c_ptr,                # output matrix  (B*H, M, N)   fp32
    batch_stride_a,       # stride to next (B*H) slice in a
    batch_stride_b,
    batch_stride_c,
    M, N, K,
    stride_am, stride_ak, # a matrix strides
    stride_bn, stride_bk, # b matrix strides (note: we will read b as (N,K))
    stride_cm, stride_cn, # c matrix strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)                     # linear id over (batch*H)
    pid_batch = pid // tl.cdiv(K, 1)           # not used, we treat pid as tile id
    batch_idx = pid

    # base pointers for this batch
    a_batch = a_ptr + batch_idx * batch_stride_a
    b_batch = b_ptr + batch_idx * batch_stride_b
    c_batch = c_ptr + batch_idx * batch_stride_c

    # grid decomposition
    pid_m = tl.program_id(1)   # rows (M)
    pid_n = tl.program_id(2)   # cols (N)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer arithmetic
    a_ptrs = a_batch + (offs_m[:, None] * stride_am
                       + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                       + offs_n[None, :] * stride_bn)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < K),
                    other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=(k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot product
        acc += tl.dot(a, b)          # fp16*fp16 -> fp32 accumulation

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write result
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm
                        + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the GEMM kernel
# ----------------------------------------------------------------------
def batch_gemm(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a : (B*H, M, K)  float16
    b : (B*H, N, K)  float16   (note: K not transposed!)
    returns c : (B*H, M, N) float32
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, M, K = a.shape
    _, N, _ = b.shape

    c = torch.empty((B_H, M, N), dtype=torch.float32, device=a.device)

    # strides (in elements)
    stride_am, stride_ak = a.stride()
    stride_bn, stride_bk = b.stride()
    stride_cm, stride_cn = c.stride()

    # batch stride = distance between consecutive (B*H) slices
    batch_stride_a = a.stride(0) * a.shape[1] * a.shape[2]
    batch_stride_b = b.stride(0) * b.shape[1] * b.shape[2]
    batch_stride_c = c.stride(0) * c.shape[1] * c.shape[2]

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B_H,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        batch_stride_a=batch_stride_a,
        batch_stride_b=batch_stride_b,
        batch_stride_c=batch_stride_c,
        M=M,
        N=N,
        K=K,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bn=stride_bn,
        stride_bk=stride_bk,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return c

# Full custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.
    data = (input_tensor, mask, weights, config)
    Returns: tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ---------- LayerNorm ----------
    norm_w = weights["norm.weight"].to(dtype)
    norm_b = weights["norm.bias"].to(dtype)
    # manual LN (no bias for speed)
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    eps = 1e-5
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b

    # ---------- Linear projections (FP16 for speed) ----------
    # weight shapes: (hidden_dim, dim)
    left_proj_w = weights["left_proj.weight"].to(dtype).t()   # (dim, hidden_dim)
    right_proj_w = weights["right_proj.weight"].to(dtype).t()
    left = torch.nn.functional.linear(x, left_proj_w)        # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_w)

    # ---------- Mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ---------- Gates ----------
    left_gate_w = weights["left_gate.weight"].to(dtype).t()
    right_gate_w = weights["right_gate.weight"].to(dtype).t()
    out_gate_w = weights["out_gate.weight"].to(dtype).t()

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core N³ interaction – fused batched GEMM via Triton
    # left  : (B,N,N,hidden)   -> (B*hidden, N, N)
    # right : (B,N,N,hidden)   -> (B*hidden, N, N)  but we need (j,k) -> transpose K
    # We therefore treat right as (B*hidden, N, K) where K = N and later use
    # the kernel that multiplies left (M= N) with right (N = N) over K = N.
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).reshape(B * hidden_dim, N, N)
    # we need right with K as the last dimension (j,k) -> we keep as is and
    # the kernel expects (B*H, N, K) where K is the shared dimension.
    right_fp16 = right.to(torch.float16).reshape(B * hidden_dim, N, N)

    # Triton kernel computes: out = left @ right^T   (since we want sum_k left[i,k] * right[j,k])
    # To achieve this we simply feed right unchanged and let the kernel multiply
    # a (M,K) with (N,K) (note: both share K). The kernel accumulates over K, giving the
    # desired result.
    out_fp32 = batch_gemm(left_fp16, right_fp16)          # (B*H, N, N)

    out = out_fp32.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,hidden)

    # ---------- Output LayerNorm ----------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ---------- Out‑gate ----------
    out = out * out_gate

    # ---------- Final linear ----------
    to_out_w = weights["to_out.weight"].to(dtype).t()   # (hidden, dim)
    out = torch.nn.functional.linear(out, to_out_w)    # (B,N,N,dim)

    return out.to(dtype)
