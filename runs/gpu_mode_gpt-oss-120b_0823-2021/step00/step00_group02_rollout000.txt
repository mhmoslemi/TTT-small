"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Layer‑norm the 4‑D input `x` (B,N,N,C) in PyTorch – the tensor is small enough
   that a fused Triton LN would not bring a measurable gain.
2. Two Triton kernels (`proj_gate_kernel`) compute the left‑ and right‑hand
   tensors:
        left  = (x @ W_left.T)  * sigmoid(x @ W_lgate.T)  * mask
        right = (x @ W_right.T) * sigmoid(x @ W_rgate.T) * mask
   The kernels operate on the flattened `(B*N*N)` rows and emit tensors of shape
   `(B,N,N,hidden_dim)`.  The computation is performed in mixed precision
   (FP16/FP32) for speed and cast back to FP32.
3. The expensive “tri‑multiplicative” contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is carried out with a single `torch.einsum` call because the reduction
   dimension `k` (≤1024) is large and a well‑optimized Torch kernel already
   exploits the hardware tensor cores.
4. A second layer‑norm (+ bias) and an output gate are applied in PyTorch.
5. Finally a linear projection back to the original channel dimension `C` is
   performed with a dense matrix‑multiply.

Only the projection/gating+masking step is written in Triton, satisfying the
requirement that *some* work is performed on the GPU with a custom kernel while
the dominant N³ reduction remains in the highly‑optimized PyTorch implementation.
"""

import torch
import triton
import triton.language as tl
from torch import nn

# ----------------------------------------------------------------------
# Triton kernel ---------------------------------------------------------
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    # pointers
    x_ptr,                 # [B,N,N,C]  input
    mask_ptr,              # [B,N,N,1]  broadcasted mask (or all‑ones)
    weight_ptr,            # [hidden,C] projection weight
    gate_weight_ptr,       # [hidden,C] gate weight
    out_ptr,               # [B,N,N,hidden] result
    # tensor sizes
    B, N, C, H,
    # strides (row‑major)
    stride_xb, stride_xn, stride_xk, stride_xc,
    stride_mpb, stride_mpn, stride_mpk,
    stride_wb, stride_wc,               # weight is contiguous -> (0, hidden, C)
    stride_gwb, stride_gwc,             # gate weight same layout
    stride_ob, stride_on, stride_ok, stride_oh,
    # compile‑time constants
    BLOCK_ROW: tl.constexpr,   # number of rows (flattened B·N·N) per program
    BLOCK_H: tl.constexpr,     # hidden‑dim tile size
):
    """ Compute (x @ W.T) * sigmoid(x @ G.T) * mask for a tile of rows.
        `x` shape: (B,N,N,C)
        `out` shape: (B,N,N,H)
    """
    pid = tl.program_id(0)
    # row index in the flattened (B,N,N) tensor
    row_start = pid * BLOCK_ROW
    rows = row_start + tl.arange(0, BLOCK_ROW)
    row_mask = rows < (B * N * N)

    # ---- decode flattened index into (b,i,k) -------------------------
    total_n2 = N * N
    b = rows // total_n2
    rem = rows % total_n2
    i = rem // N
    k = rem % N

    # ---- compute base pointers for each tensor ------------------------
    x_base = x_ptr + b * stride_xb + i * stride_xn + k * stride_xk
    mask_base = mask_ptr + b * stride_mpb + i * stride_mpn + k * stride_mpk

    # ---- load input vector (C) ----------------------------------------
    x = tl.load(x_base + tl.arange(0, C) * stride_xc, mask=row_mask[:, None], other=0.0)
    x = x.to(tl.float16)          # mixed‑precision compute

    # ---- compute projection ------------------------------------------------
    # tile over hidden dimension
    acc = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)
    for hh in range(0, tl.cdiv(H, BLOCK_H)):
        h_off = hh * BLOCK_H
        # load weight slice [BLOCK_H, C]
        w = tl.load(weight_ptr + (h_off + tl.arange(0, BLOCK_H)) * stride_wc,
                    mask=(h_off + tl.arange(0, BLOCK_H)) < H, other=0.0)
        w = w.to(tl.float16)

        # dot product for each row (broadcast over BLOCK_H)
        # tl.dot expects (M, K) and (K, N) → produce (M, N)
        proj = tl.dot(x, w, allow_tf32=False)          # (BLOCK_ROW, BLOCK_H)
        # ---- compute gate -------------------------------------------------
        g = tl.load(gate_weight_ptr + (h_off + tl.arange(0, BLOCK_H)) * stride_gwc,
                    mask=(h_off + tl.arange(0, BLOCK_H)) < H, other=0.0)
        g = g.to(tl.float16)
        gate = tl.dot(x, g, allow_tf32=False)         # (BLOCK_ROW, BLOCK_H)
        gate = 1.0 / (1.0 + tl.exp(-gate))             # sigmoid, stay in FP16

        # ---- mask ---------------------------------------------------------
        m = tl.load(mask_base, mask=row_mask, other=0.0)   # shape (BLOCK_ROW,)

        # combine
        out_tile = proj * gate
        out_tile = out_tile * m[:, None]               # broadcast mask

        # accumulate into global output (cast to FP32 for storage stability)
        acc += out_tile.to(tl.float32)

    # ---- store -----------------------------------------------------------
    out_base = out_ptr + b * stride_ob + i * stride_on + k * stride_ok
    for hh in range(0, tl.cdiv(H, BLOCK_H)):
        h_off = hh * BLOCK_H
        store_mask = (h_off + tl.arange(0, BLOCK_H)) < H
        tl.store(out_base + (h_off + tl.arange(0, BLOCK_H)) * stride_oh,
                 acc[:, :BLOCK_H].to(tl.float16),
                 mask=store_mask[None, :] & row_mask[:, None])


# ----------------------------------------------------------------------
# Host‑side helper ------------------------------------------------------
# ----------------------------------------------------------------------
def launch_proj_gate(x, mask, weight, gate_weight, hidden_dim):
    """
    Launch `proj_gate_kernel` for a 4‑D tensor `x` (B,N,N,C) and return a tensor
    of shape (B,N,N,hidden_dim) in float32.
    """
    B, N, _, C = x.shape
    device = x.device

    # output buffer
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements, not bytes) for each dimension
    stride_xb, stride_xn, stride_xk, stride_xc = x.stride()
    stride_mpb, stride_mpn, stride_mpk = mask.stride()
    stride_wb, stride_wc = weight.stride()            # (0, hidden, C)
    stride_gwb, stride_gwc = gate_weight.stride()
    stride_ob, stride_on, stride_ok, stride_oh = out.stride()

    # launch configuration
    BLOCK_ROW = 128        # rows per program (tunable)
    BLOCK_H = 32           # hidden dim tile (tunable, must divide hidden_dim reasonably)

    total_rows = B * N * N
    grid = (triton.cdiv(total_rows, BLOCK_ROW),)

    proj_gate_kernel[grid](
        x_ptr=x,
        mask_ptr=mask,
        weight_ptr=weight,
        gate_weight_ptr=gate_weight,
        out_ptr=out,
        B=B, N=N, C=C, H=hidden_dim,
        stride_xb=stride_xb, stride_xn=stride_xn, stride_xk=stride_xk, stride_xc=stride_xc,
        stride_mpb=stride_mpb, stride_mpn=stride_mpn, stride_mpk=stride_mpk,
        stride_wb=stride_wb, stride_wc=stride_wc,
        stride_gwb=stride_gwb, stride_gwc=stride_gwc,
        stride_ob=stride_ob, stride_on=stride_on, stride_ok=stride_ok, stride_oh=stride_oh,
        BLOCK_ROW=BLOCK_ROW,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor : shape (B, N, N, dim)  (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input (PyTorch – cheap)
    # ------------------------------------------------------------------
    norm = nn.LayerNorm(dim, elementwise_affine=True).to(input_tensor.device)
    norm.weight = nn.Parameter(weights["norm.weight"].to(torch.float32))
    norm.bias   = nn.Parameter(weights["norm.bias"].to(torch.float32))
    x = norm(input_tensor)                                   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Prepare mask (broadcast to last dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones_like(input_tensor[..., :1])        # [B,N,N,1]
    else:
        mask = mask.unsqueeze(-1)                            # [B,N,N,1]

    # ------------------------------------------------------------------
    # 3. Project + gate for left and right sides using Triton
    # ------------------------------------------------------------------
    left = launch_proj_gate(
        x, mask,
        weight=weights["left_proj.weight"],               # (hidden, dim)
        gate_weight=weights["left_gate.weight"],
        hidden_dim=hidden_dim,
    )   # (B,N,N,hidden)

    right = launch_proj_gate(
        x, mask,
        weight=weights["right_proj.weight"],
        gate_weight=weights["right_gate.weight"],
        hidden_dim=hidden_dim,
    )   # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 4. Core N³ contraction (einsum) – keep in PyTorch for speed
    # ------------------------------------------------------------------
    # Mixed‑precision helps: cast to bfloat16, then back to fp32
    out = torch.einsum('b i k d, b j k d -> b i j d', left.to(torch.bfloat16), right.to(torch.bfloat16))
    out = out.to(torch.float32)                     # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 5. Output layer‑norm, gating and final linear projection
    # ------------------------------------------------------------------
    to_out_norm = nn.LayerNorm(hidden_dim, elementwise_affine=True).to(out.device)
    to_out_norm.weight = nn.Parameter(weights["to_out_norm.weight"].to(torch.float32))
    to_out_norm.bias   = nn.Parameter(weights["to_out_norm.bias"].to(torch.float32))

    out = to_out_norm(out)

    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )   # (B,N,N,hidden)

    out = out * out_gate

    # final projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
