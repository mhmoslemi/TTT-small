"""
TriMul “outgoing” implementation.

Algorithm
---------
1. Layer‑norm the input (dim = C).
2. Linear projections → left/right (C → H) and three sigmoid gates.
3. Apply mask (if present) and the per‑gate multiplicative masks.
4. Compute the expensive N³ reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply for every hidden channel *h*.
   A Triton kernel tiles the (i,j) output plane (BLOCK_I × BLOCK_J) and
   reduces over k in small blocks (BLOCK_K).  The hidden dimension is
   processed in sub‑tiles (BLOCK_H) so that each thread block keeps the
   accumulator in registers (float32) and only writes once.
5. Layer‑norm on the H‑dimensional result, multiply by the output gate
   and a final linear projection back to C.

The kernel works for any batch size (1‑2), sequence length up to 1024 and
hidden dimension that is a multiple of BLOCK_H (we use 16, which divides 128).
All tensors are expected on the CUDA device; the final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the N³ reduction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # const * left  [B,N,N,H]
    right_ptr,              # const * right [B,N,N,H]
    out_ptr,                # * out [B,N,N,H]

    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    # program ids
    pid_b = tl.program_id(0)               # batch axis
    pid_i = tl.program_id(1)               # output i tile
    pid_j = tl.program_id(2)               # output j tile

    # start offsets for this tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, BLOCK_H)               # processed within the tile

    # mask out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # pointers with base offsets
    left_base  = left_ptr  + pid_b * stride_left_b
    right_base = right_ptr + pid_b * stride_right_b
    out_base   = out_ptr   + pid_b * stride_out_b

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over k dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left[ i, k, h ]  -> shape (BLOCK_I, BLOCK_K, BLOCK_H)
        left_ptrs = left_base \
                    + (offs_i[:, None, None] * stride_left_i) \
                    + (offs_k[None, :, None] * stride_left_k) \
                    + (offs_h[None, None, :] * stride_left_h)
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0).to(tl.float32)

        # load right[ j, k, h ]  -> shape (BLOCK_J, BLOCK_K, BLOCK_H)
        right_ptrs = right_base \
                     + (offs_j[:, None, None] * stride_right_j) \
                     + (offs_k[None, :, None] * stride_right_k) \
                     + (offs_h[None, None, :] * stride_right_h)
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0).to(tl.float32)

        # contract over k :  sum_k left[i,k,h] * right[j,k,h]
        # we broadcast over h and sum over k
        # left: (I, K, H), right: (J, K, H) -> need (I,J,H)
        # we compute via tl.dot for each h slice
        for h in range(0, BLOCK_H):
            l = left[:, :, h]          # (I, K)
            r = right[:, :, h]         # (J, K)
            # dot: (I,K) @ (J,K).T  => we need element‑wise product then sum over K
            # tl.dot works for (I,K) @ (K,J).  So transpose r.
            acc[:, :, h] += tl.dot(l, r, transb=True)   # (I,J)

    # write result
    out_ptrs = out_base \
               + (offs_i[:, None, None] * stride_out_i) \
               + (offs_j[None, :, None] * stride_out_j) \
               + (offs_h[None, None, :] * stride_out_h)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_ij[:, :, None] & (offs_h[None, None, :] < H))


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, C]  float32 on CUDA
        - mask         : [B, N, N]    (bool/float) on CUDA
        - weights      : dict of parameters (see docstring)
        - config       : dict, must contain "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]
    assert C == config["dim"]

    # ---------- LayerNorm ----------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (C,), weight=norm_w, bias=norm_b)

    # ---------- Linear projections ----------
    # Linear without bias: y = x @ W.T  (W shape H×C)
    left_proj_w  = weights["left_proj.weight"]          # (H, C)
    right_proj_w = weights["right_proj.weight"]         # (H, C)

    # reshape for matmul: combine batch & seq dims
    x_flat = x.reshape(B * N * N, C)                    # (B*N*N, C)
    left_flat  = torch.nn.functional.linear(x_flat, left_proj_w)   # (B*N*N, H)
    right_flat = torch.nn.functional.linear(x_flat, right_proj_w)  # (B*N*N, H)

    left  = left_flat.reshape(B, N, N, H)
    right = right_flat.reshape(B, N, N, H)

    # ---------- Mask ----------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)      # (B,N,N,1)
        left  = left * mask_
        right = right * mask_

    # ---------- Gates ----------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    # reuse x for gate projections
    x_flat = x.reshape(B * N * N, C)
    left_gate_flat  = torch.nn.functional.linear(x_flat, left_gate_w).sigmoid()
    right_gate_flat = torch.nn.functional.linear(x_flat, right_gate_w).sigmoid()
    out_gate_flat   = torch.nn.functional.linear(x_flat, out_gate_w).sigmoid()

    left_gate  = left_gate_flat.reshape(B, N, N, H)
    right_gate = right_gate_flat.reshape(B, N, N, H)
    out_gate   = out_gate_flat.reshape(B, N, N, H)

    left  = left * left_gate
    right = right * right_gate

    # ---------- Triton reduction ----------
    out = torch.empty_like(left)   # shape [B,N,N,H], dtype float32

    # launch parameters (tuned for H200)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 16                     # H is multiple of 16 (128)

    grid = (B,
            triton.cdiv(N, BLOCK_I),
            triton.cdiv(N, BLOCK_J))

    # stride calculations
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=H,
        stride_left_b=stride_left_b, stride_left_i=stride_left_i,
        stride_left_k=stride_left_k, stride_left_h=stride_left_h,
        stride_right_b=stride_right_b, stride_right_j=stride_right_j,
        stride_right_k=stride_right_k, stride_right_h=stride_right_h,
        stride_out_b=stride_out_b, stride_out_i=stride_out_i,
        stride_out_j=stride_out_j, stride_out_h=stride_out_h,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
        num_warps=4
    )

    # ---------- Output normalization & projection ----------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]          # (C, H)

    # LayerNorm on hidden dim
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # multiply by output gate
    out = out * out_gate

    # final linear projection back to C
    out_flat = out.reshape(B * N * N, H)
    out_proj = torch.nn.functional.linear(out_flat, to_out_w)   # (B*N*N, C)
    out = out_proj.reshape(B, N, N, C)

    return out
