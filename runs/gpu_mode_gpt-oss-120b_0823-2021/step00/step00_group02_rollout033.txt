"""
TriMul “outgoing” kernel (AlphaFold3).

We compute
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where
    left = (x @ W_left)   * mask * sigmoid(x @ W_left_gate)
    right = (x @ W_right) * mask * sigmoid(x @ W_right_gate)

The heavy O(N³·C) contraction is performed by a fused Triton kernel that
tiles the k‑dimension (BLOCK_K) and accumulates a BLOCK_M×BLOCK_N tile of the
output in FP32.  All other ops (LayerNorm, linear projections, gating,
output‑norm and final linear) are cheap and stay in PyTorch.

The kernel is launched with a 3‑D grid:
    dim0 – (batch * hidden_dim)   → each program works on a fixed (b, d)
    dim1 – tiles over the i‑axis (seq_len)
    dim2 – tiles over the j‑axis (seq_len)

The kernel returns a FP32 tensor of shape [B, N, N, C] which is then passed
through the final LayerNorm, gating and linear layer (all in torch) and
returned as float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel for the 4‑D contraction Σ_k left[...,i,k,d] * right[...,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_out_kernel(
    left_ptr,          # *const float *
    right_ptr,         # *const float *
    out_ptr,           # *float *
    B, N, H,           # int32 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute a tile of the outgoing TriMul contraction."""
    # program ids -------------------------------------------------------
    pid0 = tl.program_id(0)          # (b, d) linearised
    pid1 = tl.program_id(1)          # i‑tile
    pid2 = tl.program_id(2)          # j‑tile

    b = pid0 // H                    # batch index
    h = pid0 % H                     # hidden‑dim index

    # tile offsets
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k in BLOCK_K chunks
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers with proper strides
        left_offs = (
            b * stride_left_b + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        right_offs = (
            b * stride_right_b + h * stride_right_h
            + offs_j[:, None] * stride_right_j
            + offs_k[None, :] * stride_right_k
        )

        # load tiles
        left = tl.load(
            left_ptr + left_offs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        right = tl.load(
            right_ptr + right_offs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right needs to be transposed on the k‑axis to obtain
        # Σ_k left[i,k] * right[j,k]  =  left[i,:] · right[j,:]^T
        # right is (BLOCK_N, BLOCK_K) after transposition
        right_T = tl.trans(right, (1, 0))

        # dot product: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(left, right_T)

    # store the result
    out_offs = (
        b * stride_out_b + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_ptr + out_offs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper: fused LayerNorm (no bias for speed – bias is added afterwards)
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: [..., C]   weight/bias: (C,)
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator using a fused Triton kernel.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor:   [B, N, N, dim]   float32/float16 on CUDA
            - mask:           [B, N, N]        broadcastable, 0/1
            - weights: dict with all linear / norm parameters
            - config:  dict with keys "dim", "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dim (dim)
    # ------------------------------------------------------------------
    x = _layer_norm(
        x,
        weights["norm.weight"],
        weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) -> left/right
    # ------------------------------------------------------------------
    # we keep everything in FP32 for numeric safety before the kernel
    x_flat = x.reshape(-1, dim)                      # (B*N*N, dim)

    left = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 3. Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1)               # (B,N,N,1)
        left = left * mask_unsq
        right = right * mask_unsq

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden)
    )
    right_gate = torch.sigmoid(
        F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden)
    )
    out_gate = torch.sigmoid(
        F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden)
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton contraction Σ_k left[...,i,k,d] * right[...,j,k,d]
    # ------------------------------------------------------------------
    # allocate output buffer
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides (in element count) for the contiguous tensors
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – empirically good for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(B * hidden, 1),          # dim0: one program per (b, hidden)
        triton.cdiv(N, BLOCK_M),            # dim1: i‑tiles
        triton.cdiv(N, BLOCK_N),            # dim2: j‑tiles
    )

    _trimul_out_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear (all in torch)
    # ------------------------------------------------------------------
    out = _layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # final projection back to dim
    out_flat = out.reshape(-1, hidden)
    out = F.linear(out_flat, weights["to_out.weight"]).view(B, N, N, dim)

    # ensure float32 output as required
    return out.to(torch.float32)
