"""
TriMul “outgoing” implementation.
-------------------------------------------------
1. LayerNorm + four linear projections + three sigmoid gates are done in
   plain PyTorch (they are cheap compared to the O(N³) contraction).
2. The costly contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is performed by a fused Triton kernel.  The kernel tiles the i‑ and j‑axes
   (BLOCK_M × BLOCK_N) and iterates over k in blocks (BLOCK_K).  The hidden
   channel dimension `d` is processed in small chunks (BLOCK_D) to keep the
   register pressure low while still using the fast tl.dot primitive.
3. The kernel produces fp16 results that are converted back to fp32, then
   the final LayerNorm, gating and output linear are applied in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: blockwise contraction over the third dimension (k)
# ----------------------------------------------------------------------
@triton.jit
def trimul_contract_kernel(
    left_ptr,          # [B, N, N, D] left  (after proj & gate)
    right_ptr,         # [B, N, N, D] right (after proj & gate)
    out_ptr,           # [B, N, N, D] output
    B, N, D,
    stride_lb, stride_lk, stride_ld,   # left strides: b, i, d
    stride_rb, stride_rk, stride_rd,   # right strides: b, j, d
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides: b, i, j, d
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)

    # ----- decode linear program id into (b, i, j, d0) -----
    total_ij = N * N
    total = B * total_ij * (D // BLOCK_D)
    b   = (pid // (total_ij * (D // BLOCK_D))) % B
    ij  = (pid // (D // BLOCK_D)) % total_ij
    d0  = (pid % (D // BLOCK_D)) * BLOCK_D

    i = ij // N
    j = ij % N

    # ----- tile offsets -----
    offs_i = i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_d = d0 + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # ----- pointers with proper strides -----
    # left:   (b, i, k, d)
    left_base = left_ptr + b * stride_lb + i * stride_lk
    # right:  (b, j, k, d)
    right_base = right_ptr + b * stride_rb + j * stride_rk
    # out:    (b, i, j, d)
    out_base = out_ptr + b * stride_ob + i * stride_oi + j * stride_oj

    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ----- reduction over k -----
    for k_outer in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_outer * BLOCK_K
        k_valid = cur_k + offs_k < N

        # left[b,i,k,d]
        left_ptrs = left_base + cur_k * stride_ld + offs_k[None, None, :] * stride_ld + offs_d[None, None, :] * stride_ld
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & k_valid[None, None, :] & mask_d[None, None, :],
                       other=0.0)

        # right[b,j,k,d]
        right_ptrs = right_base + cur_k * stride_rd + offs_k[None, None, :] * stride_rd + offs_d[None, None, :] * stride_rd
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] & k_valid[None, None, :] & mask_d[None, None, :],
                        other=0.0)

        # dot over the k‑axis for every (i,j,d) tile
        # tl.dot operates on 2‑D matrices, so we reshape
        left_2d  = left.reshape((BLOCK_M, BLOCK_K * BLOCK_D))
        right_2d = right.reshape((BLOCK_N, BLOCK_K * BLOCK_D)).T   # (BLOCK_K*BLOCK_D, BLOCK_N)
        acc += tl.dot(left_2d, right_2d).reshape((BLOCK_M, BLOCK_N, BLOCK_D))

    # ----- store result -----
    out_ptrs = out_base + offs_d[None, None, :] * stride_od
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])


# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------- LayerNorm ---------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------- Linear projs ------------------------
    # (B,N,N,hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------- Masking ----------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                       # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------- Gates -------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------- Triton contraction -----------------
    # Cast to fp16 for the heavy reduction (speed‑memory trade‑off)
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float16, device=device)

    # strides (in elements, not bytes)
    stride_lb, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8   # hidden_dim is a multiple of 8 in all test configs

    total_programs = bs * N * N * (hidden_dim // BLOCK_D)
    grid = (total_programs,)

    trimul_contract_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        bs, N, hidden_dim,
        stride_lb, stride_lk, stride_ld,
        stride_rb, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    out = out.to(torch.float32)

    # --------------------------- Final LayerNorm & Linear ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
