"""
TriMul “outgoing” implementation.
* layer‑norms, linear projections and final post‑processing stay in PyTorch.
* the cubic‑cost contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
  is equivalent to a batched matrix‑multiply
        out = left @ right.transpose(-2,-1)
  for every hidden channel *d*.  A single Triton kernel computes many
  (batch, hidden) slices simultaneously, using classic tiling
  (BLOCK_M × BLOCK_N × BLOCK_K) and a fused dot‑product accumulation.
  The kernel works for arbitrary batch size, sequence length N and hidden
  dimension H (here H = hidden_dim).  All tensors are assumed to be on
  CUDA device and in float32; intermediate accumulation is float32 and the
  final result is stored in float32.
"""

import torch
import triton
import triton.language as tl
from torch import nn

# --------------------------------------------------------------
# Triton kernel: batched matmul with the second operand transposed
# --------------------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr,           # [B, N, N, H]   (contiguous in last dim)
    right_ptr,          # [B, N, N, H]
    out_ptr,            # [B, N, N, H]
    # problem size
    B, N, H,
    # strides (in elements, not bytes)
    stride_lb, stride_lk, stride_ld,   # left  stride for batch, row(i), col(k), hidden
    stride_rb, stride_rj, stride_rk, stride_rd,   # right stride for batch, row(j), col(k), hidden
    stride_ob, stride_oi, stride_oj, stride_od,   # out   stride for batch, row(i), col(j), hidden
    # tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)               # linear id over (batch, hidden)
    bid = pid // H                       # batch index
    hid = pid %  H                       # hidden channel index

    # tile offsets in the output matrix (M=row i, N=col j)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the beginning of the (batch, hidden) slice
    left_base  = left_ptr  + bid * stride_lb + hid * stride_ld
    right_base = right_ptr + bid * stride_rb + hid * stride_rd
    out_base   = out_ptr   + bid * stride_ob + hid * stride_od

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile of left:   shape (BLOCK_M, BLOCK_K)
        # left layout: (i, k)
        a_ptrs = left_base + offs_m[:, None] * stride_lk + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load a tile of right transposed: we need right[j, k] -> treat as (j,k)
        # To get right[j, k] we use stride_rj for row j and stride_rk for col k.
        # Since we need right.T we load (j, k) but later dot with a (i,k).
        b_ptrs = right_base + offs_n[:, None] * stride_rj + offs_k[None, :] * stride_rk
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # dot‑product over k dimension
        acc += tl.dot(a, b)

    # write result
    out_ptrs = out_base + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Helper: launch the kernel for a given tensor layout
# --------------------------------------------------------------
def launch_trimul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]  (float32, contiguous)
    returns out : same shape, computed as left @ right^T over the N dimension
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape

    out = torch.empty_like(left)

    # strides in elements (not bytes)
    stride_lb, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # tile sizes – chosen to fit H200 SMs and make good use of shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                        # one program per (batch, hidden)
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=H,
        stride_lb=stride_lb, stride_lk=stride_lk, stride_ld=stride_ld,
        stride_rb=stride_rb, stride_rj=stride_rj, stride_rk=stride_lk,  # note: right col stride same as left row stride
        stride_rd=stride_rd,
        stride_ob=stride_ob, stride_oi=stride_oi, stride_oj=stride_oj, stride_od=stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


# --------------------------------------------------------------
# Full custom kernel entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.
    Args
    ----
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : [B, N, N, dim]    (float32)
        mask         : [B, N, N]         (float32 or bool)
        weights      : dict of torch.Parameters (see docstring above)
        config       : {"dim": int, "hidden_dim": int}
    Returns
    -------
    out : torch.Tensor  [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    device = input_tensor.device
    dtype = torch.float32

    # ---------- LayerNorm ----------
    # Manual LN to avoid extra kernel launches; simple per‑element op.
    # x_norm = (x - mean) / sqrt(var + eps) * weight + bias
    eps = 1e-5
    x = input_tensor
    # mean & var over last dimension (dim)
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (x - mean) * torch.rsqrt(var + eps)
    x = x * weights["norm.weight"].view(1, 1, 1, -1) + weights["norm.bias"].view(1, 1, 1, -1)

    # ---------- Linear projections ----------
    # shape: [B, N, N, hidden_dim]
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- Mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ---------- Gates ----------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Core contraction (Triton) ----------
    # Compute out = left @ right^T over the N dimension, per hidden channel
    out = launch_trimul(left, right)   # [B, N, N, hidden_dim]

    # ---------- post‑processing ----------
    # LayerNorm on hidden dimension
    out = (out - out.mean(dim=-1, keepdim=True)) * torch.rsqrt(out.var(dim=-1, unbiased=False, keepdim=True) + eps)
    out = out * weights["to_out_norm.weight"].view(1, 1, 1, -1) + weights["to_out_norm.bias"].view(1, 1, 1, -1)

    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(dtype)
