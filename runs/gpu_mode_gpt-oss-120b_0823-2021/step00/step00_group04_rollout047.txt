"""
TriMul “outgoing” kernel (AlphaFold‑3).

The expensive part of the forward pass is the
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
for every batch * hidden channel.
For a fixed channel `d` this is exactly a matrix multiplication
    L_d @ R_dᵀ   (L_d,R_d ∈ ℝ^{N×N}).

The implementation below
* performs all LayerNorm / Linear / gating steps in native PyTorch,
  because they are cheap compared to the N³ contraction,
* launches a batched GEMM Triton kernel that computes
  `out = left @ right.transpose(-2,-1)` in FP16 (mixed precision) and
  casts the result back to FP32,
* finally applies the output LayerNorm, output gate and output Linear
  (again in PyTorch) to obtain a tensor of shape
  [B, N, N, dim] (float32).
The kernel works for any batch size (1‑2), sequence length up to 1024 and
hidden dimension that is a multiple of 32 (the usual case in AlphaFold‑3).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM:  out[b, d, i, j] = left[b, d, i, k] @ right[b, d, j, k]^T
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,  # batch, hidden, seq_len
    stride_lb, stride_lk, stride_lh, stride_lc,   # left strides (b, i, d, k)
    stride_rb, stride_rk, stride_rh, stride_rc,   # right strides (b, j, d, k)
    stride_ob, stride_oh, stride_oc,              # out strides (b, i, j, d)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,  # channels processed per program
):
    pid = tl.program_id(0)
    # layout of programs:
    # pid = ((b * num_h + h) * num_i + i) * num_j + j
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    num_h = tl.cdiv(H, BLOCK_H)

    b = pid // (num_h * num_i * num_j)
    tmp = pid % (num_h * num_i * num_j)
    h_idx = tmp // (num_i * num_j)
    tmp = tmp % (num_i * num_j)
    i_idx = tmp // num_j
    j_idx = tmp % num_j

    # offsets for this block
    i_off = i_idx * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = j_idx * BLOCK_N + tl.arange(0, BLOCK_N)
    h_off = h_idx * BLOCK_H + tl.arange(0, BLOCK_H)

    # mask for out‑of‑bounds
    i_mask = i_off < N
    j_mask = j_off < N
    h_mask = h_off < H

    # pointers to the start of the block
    left_base  = left_ptr  + b*stride_lb + h_off[:,None,None]*stride_lh + i_off[None,:,None]*stride_lc
    right_base = right_ptr + b*stride_rb + h_off[:,None,None]*stride_rh + j_off[None,:,None]*stride_rk

    # accumulate in fp32
    acc = tl.zeros((BLOCK_H, BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # load tiles
        # left:  (h, i, k)
        l = tl.load(left_base + k_off[None,None,:]*stride_lk,
                    mask = i_mask[:,None,None] & k_mask[None,None,:] & h_mask[:,None,None],
                    other = 0.0).to(tl.float32)
        # right: (h, j, k) we need the transposed k dimension -> we load as (h, j, k)
        r = tl.load(right_base + k_off[None,:,None]*stride_rk,
                    mask = j_mask[None,:,None] & k_mask[None,None,:] & h_mask[:,None,None],
                    other = 0.0).to(tl.float32)

        # GEMM on the k‑axis
        acc += tl.dot(l, r, out_dtype=tl.float32)   # (h,i,k)·(h,k,j) -> (h,i,j)

        # advance pointers
        left_base  += BLOCK_K * stride_lk
        right_base += BLOCK_K * stride_rk

    # store result
    out_ptrs = out_ptr + b*stride_ob + h_off[:,None,None]*stride_oh + i_off[None,:,None]*stride_oc + j_off[None,None,:]
    tl.store(out_ptrs, acc, mask = h_mask[:,None,None] & i_mask[None,:,None] & j_mask[None,None,:])

def batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right :  [B, N, N, H]  (float16)
    returns      :  [B, N, N, H]  (float32)
    """
    B, N, _, H = left.shape
    # output will be in FP32
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides (must be in elements, not bytes)
    stride_lb, stride_lc, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rk, stride_rj, stride_rh = right.stride()  # right is [B,N,N,H]
    # we need right indexed as (b, j, k, h) -> same layout as left but swapped axes 1&2
    # we will just treat right as is and use stride_rj for j, stride_rk for k.

    stride_ob, stride_oc, stride_oj, stride_oh = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 32   # hidden channels per thread block (must divide H)

    grid = (
        triton.cdiv(B, 1) *
        triton.cdiv(H, BLOCK_H) *
        triton.cdiv(N, BLOCK_M) *
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        left, right, out,
        B, H, N,
        stride_lb, stride_lk, stride_lh, stride_lc,
        stride_rb, stride_rk, stride_rh, stride_rj,
        stride_ob, stride_oh, stride_oc,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
        num_warps=4,  # enough for fp16 matmul on H200
    )
    return out

# ----------------------------------------------------------------------
# Full TriMul forward using the Triton GEMM above
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.FloatTensor [B, N, N, dim]
            - mask        : torch.FloatTensor [B, N, N]  (0/1)
            - weights     : dict of torch tensors (see problem statement)
            - config      : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.FloatTensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]

    # ---------- LayerNorm ----------
    # x = (x - mean) / sqrt(var+eps) * weight + bias
    eps = 1e-5
    weight = weights["norm.weight"]
    bias = weights["norm.bias"]
    # compute per‑sample mean/var over the last axis (dim)
    x = input_tensor
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x = x_norm * weight + bias

    # ---------- Linear projections ----------
    # Shapes: (B,N,N,dim) @ (dim, hidden)ᵀ => (B,N,N,hidden)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask (if provided) ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Heavy contraction (B,N,N,hidden) x (B,N,N,hidden).T ----------
    # Cast to fp16 for the GEMM, then back to fp32.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Triton GEMM produces fp32 output
    out = batched_gemm(left_fp16, right_fp16)   # [B,N,N,hidden] fp32

    # ---------- Output normalization and linear ----------
    # LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out.to(torch.float32)
