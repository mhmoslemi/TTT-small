"""
TriMul “outgoing” implementation.

The heavy part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply for every hidden dimension `d`.
We keep the LayerNorms / Linear projections in PyTorch (they are
memory‑bound and cheap) and off‑load the N³ reduction to a single
Triton kernel.  The kernel treats the hidden dimension in
BLOCK_H‑sized chunks, computes a blocked GEMM over (i,j) for each
chunk and accumulates in fp32.  All tensors are assumed to be on the
CUDA device and the final result is returned as float32.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict

# -------------------------------------------------------------
# Triton kernel : blocked “einsum” (outgoing tri‑multiplicative update)
# -------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,                # *float16   [B, N, N, H]
    right_ptr,               # *float16   [B, N, N, H]
    out_ptr,                 # *float32   [B, N, N, H]
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,   # tile size for i
    BLOCK_N: tl.constexpr,   # tile size for j
    BLOCK_K: tl.constexpr,   # tile size for k (reduction)
    BLOCK_H: tl.constexpr    # tile size for hidden dim
):
    # -----------------------------------------------------------------
    # program IDs
    #   axis 0 : tiling over (i,j) plane
    #   axis 1 : tiling over hidden dimension H
    # -----------------------------------------------------------------
    pid = tl.program_id(0)
    pid_h = tl.program_id(1)

    # block coordinates
    i_block = (pid % tl.cdiv(N, BLOCK_M)) * BLOCK_M
    j_block = (pid // tl.cdiv(N, BLOCK_M)) * BLOCK_N
    h_block = pid_h * BLOCK_H

    # -----------------------------------------------------------------
    # offsets within each block
    # -----------------------------------------------------------------
    offs_i = i_block + tl.arange(0, BLOCK_M)
    offs_j = j_block + tl.arange(0, BLOCK_N)
    offs_h = h_block + tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # -----------------------------------------------------------------
    # pointers – we keep the batch dimension implicit (stride 0)
    # -----------------------------------------------------------------
    left_base = left_ptr + \
        offs_i[:, None, None] * stride_l_i + \
        offs_h[None, None, :] * stride_l_h

    right_base = right_ptr + \
        offs_j[None, :, None] * stride_r_j + \
        offs_h[None, None, :] * stride_r_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # -----------------------------------------------------------------
    # reduction over k
    # -----------------------------------------------------------------
    for k_outer in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_outer * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load a block of left  : (M, K, H)
        a_ptr = left_base + \
                offs_k[None, :, None] * stride_l_k
        a = tl.load(a_ptr,
                    mask=mask_i[:, None, None] &
                         mask_k[None, :, None] &
                         mask_h[None, None, :],
                    other=0.0).to(tl.float32)

        # load a block of right : (K, N, H)
        b_ptr = right_base + \
                offs_k[:, None, None] * stride_r_k
        b = tl.load(b_ptr,
                    mask=mask_k[:, None, None] &
                         mask_j[None, :, None] &
                         mask_h[None, None, :],
                    other=0.0).to(tl.float32)

        # compute GEMM for every hidden slice in the block
        for hh in range(0, BLOCK_H):
            acc[:, :, hh] += tl.dot(a[:, :, hh], b[:, :, hh])

    # -----------------------------------------------------------------
    # write result (fp32)
    # -----------------------------------------------------------------
    out_ptrs = out_ptr + \
        offs_i[:, None, None] * stride_o_i + \
        offs_j[None, :, None] * stride_o_j + \
        offs_h[None, None, :] * stride_o_h

    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] &
                  mask_j[None, :, None] &
                  mask_h[None, None, :])


# -------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    The heavy N³ reduction is performed by `trimul_outgoing_kernel`.
    All other linear / norm / gate operations stay in PyTorch.
    """
    # ---------------------------------------------------------
    # unpack
    # ---------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    device = inp.device
    dtype = inp.dtype

    # -----------------------------------------------------------------
    # 1. LayerNorm on the last dimension
    # -----------------------------------------------------------------
    x = (inp - (weights["norm.weight"] * 0 + weights["norm.bias"]))  # dummy to keep graph
    # manual layernorm (no bias in weights dict for simplicity)
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)  →  FP16 for the kernel
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"]).to(torch.float16)
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"]).to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Apply mask (if present) – mask is bool / 0‑1 float
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid) – stays in FP32 then cast back
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(
        x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(torch.nn.functional.linear(
        x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(
        x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

    # -----------------------------------------------------------------
    # 5. Triton kernel: compute out = Σ_k left[...,i,k,d] * right[...,j,k,d]
    # -----------------------------------------------------------------
    out = torch.empty((B, N, N, hidden),
                      dtype=torch.float32, device=device)

    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tile sizes – tuned for Hopper (sm_90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 32

    grid = (triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),   # (i,j) tiles
            triton.cdiv(hidden, BLOCK_H))                     # hidden tiles

    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, hidden,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
        num_warps=4,                      # good default for H200
    )

    # -----------------------------------------------------------------
    # 6. Post‑processing in PyTorch
    # -----------------------------------------------------------------
    # LayerNorm over hidden dimension
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # gate & final linear
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
