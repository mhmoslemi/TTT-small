"""
TriMul “outgoing” forward pass (AlphaFold‑3).

The heavy part is the 3‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

for every batch *b* and hidden channel *d*.  We keep the cheap
operations (layer‑norms, linear projections and gating) in PyTorch and
only fuse the  N³ reduction with a Triton kernel.  The kernel works on
a 2‑D tile of the (i,j) plane (BLOCK_M × BLOCK_N) and reduces over
k in BLOCK_K‑sized chunks.  All arithmetic is performed in fp32 and the
final result is cast back to fp32 (the rest of the network also expects
fp32).

The kernel is launched once per hidden channel (the hidden dimension is
small – 128 – so the launch overhead is negligible) and therefore the
code stays simple while still giving a large speed‑up compared with the
pure‑Python einsum.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: one hidden channel at a time
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # *const float*   shape (B, N, N, H)
    right_ptr,         # *const float*   shape (B, N, N, H)
    out_ptr,           # *float*         shape (B, N, N, H)
    B, N, H,           # tensor sizes (compile‑time constants)
    stride_left_bn, stride_left_ck, stride_left_d,  # left strides
    stride_right_bn, stride_right_ck, stride_right_d,  # right strides
    stride_out_bn, stride_out_cn, stride_out_d,       # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[...,d] for a single hidden channel d."""
    pid_b = tl.program_id(2)                       # batch dimension
    pid_i = tl.program_id(0)                       # i‑tile
    pid_j = tl.program_id(1)                       # j‑tile

    # offsets in the i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # keep the current hidden channel as a compile‑time constant
    #   the host will launch one kernel per d (see custom_kernel)
    d = tl.program_id(3)   # the hidden‑channel dimension (0 … H‑1)

    # pointer to the beginning of this batch and channel
    left_base  = left_ptr  + pid_b * stride_left_bn \
                          + d * stride_left_d
    right_base = right_ptr + pid_b * stride_right_bn \
                          + d * stride_right_d
    out_base   = out_ptr   + pid_b * stride_out_bn \
                          + d * stride_out_d

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k in chunks of BLOCK_K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,i,k,d]   -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base \
                    + (offs_i[:, None] * stride_left_ck) \
                    + (offs_k[None, :] * stride_left_d)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right[b,j,k,d]  -> shape (BLOCK_N, BLOCK_K)
        # note the transposition w.r.t. k (we need k as the reduction axis)
        right_ptrs = right_base \
                     + (offs_j[:, None] * stride_right_ck) \
                     + (offs_k[None, :] * stride_right_d)
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # dot‑product over k for every (i,j) pair
        acc += tl.dot(left, right)                 # (BM,BN)

    # write back
    out_ptrs = out_base \
               + (offs_i[:, None] * stride_out_cn) \
               + (offs_j[None, :] * stride_out_d)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side helper that launches the kernel for all hidden channels
# ----------------------------------------------------------------------
def launch_trimul(out, left, right, mask, config):
    """
    out, left, right: torch tensors on CUDA with shape (B,N,N,H)
    mask:               torch tensor (B,N,N)   – 1 where the pair is kept
    config: dict with block sizes
    """
    B, N, _, H = left.shape
    device = left.device

    # ------------------------------------------------------------------
    # Strides in elements (not bytes); Triton works with element strides
    # ------------------------------------------------------------------
    stride_left = left.stride()   # (BN, CI, CK, D)
    stride_right = right.stride()
    stride_out = out.stride()

    # unpack strides
    stride_left_bn, stride_left_ck, stride_left_k, stride_left_d = stride_left
    stride_right_bn, stride_right_ck, stride_right_k, stride_right_d = stride_right
    stride_out_bn, stride_out_cn, stride_out_n, stride_out_d = stride_out

    BLOCK_M = config.get("BLOCK_M", 64)
    BLOCK_N = config.get("BLOCK_N", 64)
    BLOCK_K = config.get("BLOCK_K", 32)

    grid = (
        triton.cdiv(N, BLOCK_M),               # i‑tiles
        triton.cdiv(N, BLOCK_N),               # j‑tiles
        B,                                     # batch
        H                                      # hidden channel
    )

    # launch one kernel per hidden channel (the 4‑th program id)
    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_bn, stride_left_ck, stride_left_d,
        stride_right_bn, stride_right_ck, stride_right_d,
        stride_out_bn, stride_out_cn, stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        * mask_tensor  : torch.Tensor [B, N, N]      (float32/bool)
        * weights_dict : mapping of all linear / norm weights
        * config_dict  : {'dim':int, 'hidden_dim':int, …}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1) LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_w + norm_b                         # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2) Linear projections (no bias) – we use torch.nn.functional.linear
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = F.linear(x, left_proj_w)      # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3) Apply mask (if any) – mask shape [B,N,N] -> [B,N,N,1]
    # --------------------------------------------------------------
    if cfg.get("nomask", True):
        mask_f = torch.ones_like(mask, dtype=left.dtype, device=device).unsqueeze(-1)
    else:
        mask_f = mask.unsqueeze(-1).to(left.dtype)
    left  = left * mask_f
    right = right * mask_f

    # --------------------------------------------------------------
    # 4) Gating (sigmoid) – also linear without bias
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Triton contraction over k (the “TriMul” core)
    # --------------------------------------------------------------
    out = torch.empty_like(left)   # shape [B,N,N,hidden_dim], fp32

    # small block‑size configuration – works well on Hopper
    launch_cfg = {"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 32}
    launch_trimul(out, left, right, mask_f, launch_cfg)

    # --------------------------------------------------------------
    # 6) Post‑processing: LayerNorm(hidden_dim) → gate → final linear
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    # LayerNorm on hidden_dim (same manual formula as above)
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate                       # gating

    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                     # back to dim

    return out.to(torch.float32)
