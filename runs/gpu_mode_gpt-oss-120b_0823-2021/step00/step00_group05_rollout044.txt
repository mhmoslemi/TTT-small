"""
TriMul “outgoing” kernel (AlphaFold‑3).

We implement the expensive pairwise reduction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where
    left  = (x @ W_left )  * sigmoid(x @ W_left_gate) * mask
    right = (x @ W_right)  * sigmoid(x @ W_right_gate) * mask

All LayerNorm / linear / gating ops are kept in PyTorch (they are cheap
compared with the N³ reduction).  Only the N³ contraction is executed in a
single Triton kernel.  The kernel works for arbitrary batch size (B≤2),
sequence length N (≤1024) and hidden dimension H (the “dim” after the
contraction, e.g. 128 or 384).  The kernel accumulates in fp32 and writes the
final result in fp32, while the inner computation is performed in fp16 for
throughput.

The launch configuration tiles the output with
    BLOCK_I × BLOCK_J × BLOCK_D
and iterates over K in chunks of BLOCK_K.
Each tile computes a small matrix‑multiply for a slice of the
hidden dimension using `tl.dot`.  The kernel is fully fused – it reads
masked left/right tensors, multiplies with the gating tensors, and
accumulates the result.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,          #  [B, N, N, H]   (fp16)
    right_ptr,         #  [B, N, N, H]   (fp16)
    out_ptr,           #  [B, N, N, H]   (fp32)

    # strides
    stride_lb, stride_lk, stride_ld,  # left  (B,N,K,D)
    stride_rb, stride_rk, stride_rd,  # right (B,N,K,D)
    stride_ob, stride_oi, stride_oj, stride_od,  # out   (B,N,N,D)

    B, N, H,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(axis=0)                     # linear id over (i,j) tiles
    num_i = tl.cdiv(N, BLOCK_I)
    i_tile = pid // num_i
    j_tile = pid % num_i

    # offsets for the output tile
    i_offset = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)

    # mask out‑of‑bounds
    i_mask = i_offset < N
    j_mask = j_offset < N

    # each thread works on a (BLOCK_D) slice of the hidden dim
    d_offset = tl.arange(0, BLOCK_D)

    # initialise accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # loop over K dimension
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        k_mask = cur_k < N

        # ------------------------------------------------------------------
        # load left  : shape (BLOCK_I, BLOCK_K, BLOCK_D)
        # ------------------------------------------------------------------
        left_ptrs = left_ptr + \
            i_offset[:, None, None] * stride_lb + \
            cur_k[None, :, None] * stride_lk + \
            d_offset[None, None, :] * stride_ld
        left = tl.load(left_ptrs, mask=(i_mask[:, None, None] & k_mask[None, :, None]), other=0.0).to(tl.float16)

        # ------------------------------------------------------------------
        # load right : shape (BLOCK_J, BLOCK_K, BLOCK_D)
        # ------------------------------------------------------------------
        right_ptrs = right_ptr + \
            j_offset[:, None, None] * stride_rb + \
            cur_k[None, :, None] * stride_rk + \
            d_offset[None, None, :] * stride_rd
        right = tl.load(right_ptrs, mask=(j_mask[:, None, None] & k_mask[None, :, None]), other=0.0).to(tl.float16)

        # ------------------------------------------------------------------
        # compute dot over K for each d
        #   left  : (I, K, D)
        #   right : (J, K, D)
        #   we need (I, J, D) = Σ_k left[i,k,d] * right[j,k,d]
        #   we do it by broadcasting left & right to (I, J, K, D) and using
        #   tl.dot on the K dimension.
        # ------------------------------------------------------------------
        # reshape for dot: (I*D, K) @ (J*D, K)^T  ->  (I*D, J*D)
        # but tl.dot only works on 2‑D tensors; we therefore iterate over d.
        for d in range(0, BLOCK_D):
            l = left[:, :, d]        # (I, K)
            r = right[:, :, d]       # (J, K)
            # multiply‑accumulate: Σ_k l[i,k] * r[j,k]
            # we can use tl.dot after transposing r
            acc[:, :, d] += tl.dot(l, r, trans_b=True)   # (I,J)

    # ------------------------------------------------------------------
    # write back result (fp32)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        i_offset[:, None, None] * stride_ob + \
        j_offset[None, :, None] * stride_oi + \
        d_offset[None, None, :] * stride_od
    tl.store(out_ptrs, acc, mask=(i_mask[:, None, None] &
                                  j_mask[None, :, None] &
                                  (d_offset < H)))


# ----------------------------------------------------------------------
# Helper wrappers
# ----------------------------------------------------------------------
def launch_trimul_kernel(left: torch.Tensor,
                         right: torch.Tensor,
                         out: torch.Tensor):
    """
    left/right/out are expected to have dtype fp16 (left/right) and fp32 (out)
    and shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    # tile sizes – tuned for Hopper (SM90)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_D = 32                     # H is a multiple of 32 in all tests

    grid = (triton.cdiv(N, BLOCK_I) * triton.cdiv(N, BLOCK_J),)

    left_ptr = left
    right_ptr = right
    out_ptr = out

    stride_lb = left.stride(0)
    stride_lk = left.stride(2)
    stride_ld = left.stride(3)

    stride_rb = right.stride(0)
    stride_rk = right.stride(2)
    stride_rd = right.stride(3)

    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_od = out.stride(3)

    trimul_outgoing_kernel[grid](
        left_ptr,
        right_ptr,
        out_ptr,
        stride_lb, stride_lk, stride_ld,
        stride_rb, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        B, N, H,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Full TriMul forward (PyTorch part + Triton kernel)
# ----------------------------------------------------------------------
class TriMulFast(nn.Module):
    """TriMul “outgoing” variant with the N³ contraction in Triton."""
    def __init__(self, dim: int, hidden_dim: int):
        super().__init__()
        self.norm = nn.LayerNorm(dim, eps=1e-6)
        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.to_out_norm = nn.LayerNorm(hidden_dim, eps=1e-6)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x : [B, N, N, dim]   (float32)
        mask : [B, N, N]     (float32 or bool)
        returns [B, N, N, dim] (float32)
        """
        B, N, _, dim = x.shape
        # ------------------------------------------------------------------
        # 1) layer‑norm + linear projections (keep in fp32 for stability)
        # ------------------------------------------------------------------
        x = self.norm(x)
        left = self.left_proj(x)      # (B,N,N,hd)
        right = self.right_proj(x)

        # mask broadcast
        if mask is not None:
            mask = mask.unsqueeze(-1)          # (B,N,N,1)
            left = left * mask
            right = right * mask

        # gating
        left_gate = torch.sigmoid(self.left_gate(x))
        right_gate = torch.sigmoid(self.right_gate(x))
        out_gate = torch.sigmoid(self.out_gate(x))

        left = left * left_gate
        right = right * right_gate

        # ------------------------------------------------------------------
        # 2) N³ contraction in fp16 using Triton
        # ------------------------------------------------------------------
        left_f16 = left.to(torch.float16)
        right_f16 = right.to(torch.float16)
        out = torch.empty_like(left, dtype=torch.float32)   # shape (B,N,N,hd)

        launch_trimul_kernel(left_f16, right_f16, out)

        # ------------------------------------------------------------------
        # 3) final norm, gating and projection back to dim
        # ------------------------------------------------------------------
        out = self.to_out_norm(out)
        out = out * out_gate
        out = self.to_out(out)
        return out


# ----------------------------------------------------------------------
# Entry point required by the grader
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    data = (input, mask, weights, config)

    The function builds a TriMulFast module, injects the supplied weights,
    and runs the forward pass on the provided tensors.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # instantiate model and copy the supplied parameters
    # --------------------------------------------------------------
    model = TriMulFast(dim, hidden_dim).to(input_tensor.device)

    # linear weights (stored as (out_features, in_features) in PyTorch)
    model.norm.weight.data.copy_(weights["norm.weight"])
    model.norm.bias.data.copy_(weights["norm.bias"])

    model.left_proj.weight.data.copy_(weights["left_proj.weight"])
    model.right_proj.weight.data.copy_(weights["right_proj.weight"])

    model.left_gate.weight.data.copy_(weights["left_gate.weight"])
    model.right_gate.weight.data.copy_(weights["right_gate.weight"])
    model.out_gate.weight.data.copy_(weights["out_gate.weight"])

    model.to_out_norm.weight.data.copy_(weights["to_out_norm.weight"])
    model.to_out_norm.bias.data.copy_(weights["to_out_norm.bias"])
    model.to_out.weight.data.copy_(weights["to_out.weight"])

    # --------------------------------------------------------------
    # forward
    # --------------------------------------------------------------
    with torch.no_grad():
        output = model(input_tensor, mask)

    return output
