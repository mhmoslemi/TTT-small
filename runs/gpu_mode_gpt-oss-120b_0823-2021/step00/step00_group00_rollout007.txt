"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix‑multiply for every hidden channel *d*.
We implement this contraction with a Triton kernel that works on
(i,j) tiles while streaming over the k‑dimension.
All surrounding ops (LayerNorm, Linear, gates, final LayerNorm + Linear)
are kept in PyTorch – they are cheap compared to the N³ contraction.
The kernel returns a float‑32 tensor of shape (B, N, N, H) (H = hidden_dim).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM over the third axis (k) for each hidden dim.
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr,          # B,N,N,H  (i,k,d)
    right_ptr,         # B,N,N,H  (j,k,d)
    out_ptr,           # B,N,N,H  (i,j,d)
    # strides (in elements, not bytes)
    stride_bln, stride_ln, stride_lk, stride_ld,   # left strides
    stride_brn, stride_rn, stride_rk, stride_rd,   # right strides
    stride_obn, stride_on, stride_oj, stride_od,   # out strides
    # dimensions
    B, N, H,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,   # i‑tile
    BLOCK_N: tl.constexpr,   # j‑tile
    BLOCK_K: tl.constexpr,   # k‑chunk
    BLOCK_H: tl.constexpr,   # hidden‑dim chunk (<=H)
):
    pid = tl.program_id(0)
    # 1‑D grid: each program handles a tile of (i,j)
    pid_m = pid // tl.cdiv(N, BLOCK_N)          # i‑tile index
    pid_n = pid %  tl.cdiv(N, BLOCK_N)          # j‑tile index

    # Offsets for the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Guard masks (in case N is not a multiple of the block size)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Accumulator over k and hidden dim
    # We process H in chunks of BLOCK_H (default 32) to keep registers low.
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # Loop over k dimension in BLOCK_K steps
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k_start, BLOCK_K)
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # mask for k (partial last block)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load left: shape (M, K, H) -> (BLOCK_M, cur_k, BLOCK_H)
        # ------------------------------------------------------------------
        left_ptrs = left_ptr + \
            offs_m[:, None, None] * stride_lk + \
            offs_k[None, :, None] * stride_ln + \
            tl.arange(0, BLOCK_H)[None, None, :] * stride_ld + \
            tl.zero   # batch dimension handled by outer loop later
        left = tl.load(left_ptrs, mask=mask_m[:, None, None] & mask_k[None, :, None], other=0.0)

        # ------------------------------------------------------------------
        # Load right: shape (N, K, H) -> (BLOCK_N, cur_k, BLOCK_H)
        # Note: right is indexed as (j, k, d)
        # ------------------------------------------------------------------
        right_ptrs = right_ptr + \
            offs_n[:, None, None] * stride_rk + \
            offs_k[None, :, None] * stride_rn + \
            tl.arange(0, BLOCK_H)[None, None, :] * stride_rd + \
            tl.zero
        right = tl.load(right_ptrs, mask=mask_n[:, None, None] & mask_k[None, :, None], other=0.0)

        # ------------------------------------------------------------------
        # Contract over k (multiply‑accumulate)
        # left: (M, K, H)   right: (N, K, H) -> sum_k left[...,h] * right[...,h]
        # ------------------------------------------------------------------
        # broadcast left over N dim, right over M dim
        left_exp  = left[:, :, :, None]          # (M,K,H,1)
        right_exp = right[:, :, None, :]         # (N,K,1,H)
        # inner product over K
        prod = left_exp * right_exp               # (M,N,K,H)
        # sum over K
        acc += tl.sum(prod, axis=2)               # (M,N,H)

    # ------------------------------------------------------------------
    # Write back: need to handle hidden dim tiling
    # ------------------------------------------------------------------
    for h_start in range(0, H, BLOCK_H):
        cur_h = tl.minimum(H - h_start, BLOCK_H)
        offs_h = h_start + tl.arange(0, BLOCK_H)

        out_ptrs = out_ptr + \
            offs_m[:, None] * stride_on + \
            offs_n[None, :] * stride_oj + \
            offs_h[None, None, :] * stride_od + \
            tl.zero

        mask_h = offs_h < H
        tl.store(out_ptrs,
                 acc[:, :, :cur_h],
                 mask=mask_m[:, None] & mask_n[None, :] & mask_h[None, None, :])

# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator using a Triton kernel for
    the O(N³) contraction.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm ---------------------------------------------------
    # layer‑norm over the last dim (dim)
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), norm_w, norm_b)

    # ----- Linear projections (no bias) --------------------------------
    # Shapes: (dim, hidden_dim)
    left_proj_w  = weights["left_proj.weight"].to(device)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(device)

    # torch.nn.Linear expects (out_features, in_features)
    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,hd)
    right = torch.nn.functional.linear(x, right_proj_w)

    # ----- Apply mask --------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)       # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ----- Gates (sigmoid) ---------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton contraction: out = Σ_k left[...,i,k,d] * right[...,j,k,d]
    # ------------------------------------------------------------------
    # Prepare pointers and strides (in elements)
    left_ptr  = left.data_ptr()
    right_ptr = right.data_ptr()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)
    out_ptr = out.data_ptr()

    # strides (in elements, not bytes)
    stride_bln, stride_ln, stride_lk, stride_ld = left.stride()
    stride_brn, stride_rn, stride_rk, stride_rd = right.stride()
    stride_obn, stride_on, stride_oj, stride_od = out.stride()

    # Tile sizes – tuned for Hopper (sm_90). 32‑bit accumulation, float16 loads.
    BLOCK_M = 64   # tile over i
    BLOCK_N = 64   # tile over j
    BLOCK_K = 32   # k‑chunk streamed
    BLOCK_H = 32   # hidden‑dim chunk (hidden_dim is ≤ 384)

    # Grid: one program per (i_tile, j_tile) per batch
    total_tiles_i = triton.cdiv(N, BLOCK_M)
    total_tiles_j = triton.cdiv(N, BLOCK_N)
    grid = (total_tiles_i * total_tiles_j * B,)

    # Launch kernel for each batch separately (simple loop, batch dim small)
    for b in range(B):
        # Offsets to the start of the batch
        left_batch_ptr  = left_ptr  + b * stride_bln
        right_batch_ptr = right_ptr + b * stride_brn
        out_batch_ptr   = out_ptr   + b * stride_obn

        trimul_kernel[grid](
            left_batch_ptr,
            right_batch_ptr,
            out_batch_ptr,
            stride_bln, stride_ln, stride_lk, stride_ld,
            stride_brn, stride_rn, stride_rk, stride_rd,
            stride_obn, stride_on, stride_oj, stride_od,
            B, N, hidden_dim,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )

    # ----- Final processing (LayerNorm, gate, projection) ---------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    to_out_w      = weights["to_out.weight"].to(device)   # (dim, hidden_dim)

    out = F.layer_norm(out, (hidden_dim,), to_out_norm_w, to_out_norm_b)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)
    return out
