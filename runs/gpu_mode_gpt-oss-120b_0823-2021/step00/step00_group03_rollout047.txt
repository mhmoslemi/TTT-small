"""
TriMul “outgoing” kernel (AlphaFold‑3).

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where
    left  = (x @ W_left)  * sigmoid(x @ W_left_gate)  * mask
    right = (x @ W_right) * sigmoid(x @ W_right_gate) * mask

The kernel below computes the above reduction in pure Triton.
All LayerNorm / Linear / sigmoid / mask / final projection are done in
regular PyTorch for simplicity; only the O(N³·D) multiplication‑accumulate
is accelerated.  The kernel works for arbitrary batch size (B≤2),
sequence length N (≤1024) and hidden dimension H (≤384).  The
output is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:  out[b,i,j,h] += sum_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # static sizes
    # strides (B,N,K,H) for left/right and (B,N,N,H) for out
    stride_bl, stride_bi, stride_bk, stride_bh,
    stride_rl, stride_rj, stride_rk, stride_rh,
    stride_ol, stride_oi, stride_oj, stride_oh,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Compute the 4‑D “einsum” using a tiled reduction over K."""
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1) 2‑D block index (i_block, j_block) and batch index b
    # --------------------------------------------------------------
    blocks_i = tl.cdiv(N, BLOCK_I)
    blocks_j = tl.cdiv(N, BLOCK_J)
    total_blocks = B * blocks_i * blocks_j

    b = pid // (blocks_i * blocks_j)
    rem = pid % (blocks_i * blocks_j)
    i_block = rem // blocks_j
    j_block = rem % blocks_j

    # --------------------------------------------------------------
    # 2) Offsets inside the tiles
    # --------------------------------------------------------------
    i_offset = i_block * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_block * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i_offset < N
    j_mask = j_offset < N

    # accumulator: (BLOCK_I, BLOCK_J, BLOCK_H)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) Loop over K in tiles
    # --------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # -------- left[b, i, k, h] --------
        left_ptrs = (
            left_ptr
            + b * stride_bl
            + i_offset[:, None, None] * stride_bi
            + k_offset[None, :, None] * stride_bk
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_bh
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None, None] & k_mask[None, :, None],
            other=0.0,
        )  # (BI, BK, BH)

        # -------- right[b, j, k, h] --------
        right_ptrs = (
            right_ptr
            + b * stride_rl
            + j_offset[:, None, None] * stride_rj
            + k_offset[None, :, None] * stride_rk
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_rh
        )
        right = tl.load(
            right_ptrs,
            mask=j_mask[:, None, None] & k_mask[None, :, None],
            other=0.0,
        )  # (BJ, BK, BH)

        # -------- outer product over the K dimension --------
        # left:  (BI, BK, BH) -> (BI, 1, BK, BH)
        # right: (BJ, BK, BH) -> (1, BJ, BK, BH)
        # after broadcast multiply we sum over BK → (BI, BJ, BH)
        l_exp = left[:, None, :, :]          # (BI, 1, BK, BH)
        r_exp = right[None, :, :, :]         # (1, BJ, BK, BH)
        prod = l_exp * r_exp                 # (BI, BJ, BK, BH)
        acc += tl.sum(prod, axis=2)          # reduce over K

    # --------------------------------------------------------------
    # 4) Write back
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ol
        + i_offset[:, None, None] * stride_oi
        + j_offset[None, :, None] * stride_oj
        + tl.arange(0, BLOCK_H)[None, None, :] * stride_oh
    )
    tl.store(
        out_ptrs,
        acc,
        mask=i_mask[:, None, None] & j_mask[None, :, None],
    )


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]          # H (called hidden_dim in the paper)

    # ------------------------------------------------------------------
    # 1) LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual layer‑norm to stay on GPU without extra kernels
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * norm_w + norm_b

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) and gating
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden]
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # mask handling
    if mask is not None:
        mask = mask.unsqueeze(-1)            # (B,N,N,1)
        left = left * mask
        right = right * mask

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Triton kernel for the einsum reduction
    # ------------------------------------------------------------------
    # allocate output
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides (in elements) for the three tensors
    stride_bl, stride_bi, stride_bk, stride_bh = left.stride()
    stride_rl, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ol, stride_oi, stride_oj, stride_oh = out.stride()

    # tile sizes – tuned for H200 (shared memory ~ 64 KB)
    BLOCK_I = 64   # i‑dimension
    BLOCK_J = 64   # j‑dimension
    BLOCK_K = 32   # k‑reduction tile
    BLOCK_H = hidden  # we keep the whole hidden dim inside the tile

    # total number of program instances
    blocks_i = triton.cdiv(N, BLOCK_I)
    blocks_j = triton.cdiv(N, BLOCK_J)
    total_progs = B * blocks_i * blocks_j
    grid = (total_progs,)

    trimul_einsum_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden,
        stride_bl,
        stride_bi,
        stride_bk,
        stride_bh,
        stride_rl,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ol,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 4) Final LayerNorm, gating and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden,), weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out.to(torch.float32)
