"""
TriMul “outgoing” kernel (AlphaFold‑3)
------------------------------------
We implement the heavy N³ component

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are already projected, masked and gated.
`left` and `right` have shape   (B, N, N, H)   (H = hidden_dim).

The kernel tiles the (i,j) plane (BLOCK_I × BLOCK_J) and iterates over K
in blocks (BLOCK_K).  All three dimensions of the hidden channel are kept
as a vector so that a single `tl.dot` can accumulate a full H‑wide product.
The reduction is performed in FP32 regardless of the input precision; the
final result is written back as FP32.

The rest of the module (LayerNorms, linear projections, sigmoid gates and the
final linear) is left in plain PyTorch because they are memory‑bound and
already highly optimized.  Only the N³ contraction is moved to Triton,
which gives a large speed‑up on Hopper (SM‑90) GPUs.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # [B, N, N, H]   (i,k) layout
    right_ptr,         # [B, N, N, H]   (j,k) layout
    out_ptr,           # [B, N, N, H]   (i,j) result
    B, N, H,
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = Σₖ left[...,i,k,:] * right[...,j,k,:]"""

    pid_b = tl.program_id(2)                      # batch dimension
    pid_i = tl.program_id(0)                      # tile over i
    pid_j = tl.program_id(1)                      # tile over j

    # ----- coordinate ranges -------------------------------------------------
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    i_range = i_base + tl.arange(0, BLOCK_I)
    j_range = j_base + tl.arange(0, BLOCK_J)

    i_mask = i_range < N
    j_mask = j_range < N

    # accumulator: (BLOCK_I, BLOCK_J, H) in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J, H), dtype=tl.float32)

    # ----- loop over k -------------------------------------------------------
    for k_base in range(0, tl.cdiv(N, BLOCK_K)):
        k_range = k_base * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # Load left[b, i, k, :]   → shape (BLOCK_I, BLOCK_K, H)
        left_ptrs = (
            left_ptr
            + pid_b * stride_left_b
            + i_range[:, None, None] * stride_left_i
            + k_range[None, :, None] * stride_left_k
            + tl.arange(0, H)[None, None, :] * stride_left_h
        )
        left = tl.load(left_ptrs, mask=(i_mask[:, None] & k_mask[None, :])[:, :, None], other=0.0)

        # Load right[b, j, k, :]  → shape (BLOCK_J, BLOCK_K, H)
        right_ptrs = (
            right_ptr
            + pid_b * stride_right_b
            + j_range[:, None, None] * stride_right_j
            + k_range[None, :, None] * stride_right_k
            + tl.arange(0, H)[None, None, :] * stride_right_h
        )
        right = tl.load(right_ptrs, mask=(j_mask[:, None] & k_mask[None, :])[:, :, None], other=0.0)

        # Compute dot over K dimension for every (i,j) pair.
        #   left : (I, K, H)   → (I, K, 1, H)
        #   right: (J, K, H)   → (1, J, K, H)
        # Broadcast and multiply → (I, J, K, H)
        # Reduce over K.
        # Using tl.dot which expects (I*K, H) × (J*K, H) is not possible,
        # so we perform the reduction manually with an inner dot.
        # Because H is tiny (≤384) we can do a vector‑dot per (i,j,k).
        # tl.dot works on 1‑D vectors; we unroll over K with a for‑loop.

        for kk in range(0, BLOCK_K):
            k_valid = k_mask[kk]
            if k_valid:
                # (I, H) * (J, H) → (I, J, H) accumulation
                left_vec  = left[:, kk, :]   # (I, H)
                right_vec = right[:, kk, :]  # (J, H)

                # expand dimensions to broadcast
                left_vec  = left_vec[:, None, :]   # (I, 1, H)
                right_vec = right_vec[None, :, :]  # (1, J, H)

                acc += left_vec * right_vec        # (I, J, H)

    # ----- write back ---------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_out_b
        + i_range[:, None, None] * stride_out_i
        + j_range[None, :, None] * stride_out_j
        + tl.arange(0, H)[None, None, :] * stride_out_h
    )
    tl.store(out_ptrs, acc, mask=(i_mask[:, None, None] & j_mask[None, :, None])[:, :, None])


# ----------------------------------------------------------------------
# Helper – launch wrapper
# ----------------------------------------------------------------------
def launch_trimul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right :   [B, N, N, H]   (float16 / float32)
    returns       :   [B, N, N, H]   (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # tile sizes – tuned for Hopper (shared‑mem 64 KiB, 128‑bit loads)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_I),   # i‑tiles
        triton.cdiv(N, BLOCK_J),   # j‑tiles
        B,                         # batch
    )

    # strides in elements (not bytes)
    left_stride = left.stride()
    right_stride = right.stride()
    out_stride = out.stride()

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        left_stride[0], left_stride[1], left_stride[2], left_stride[3],
        right_stride[0], right_stride[1], right_stride[2], right_stride[3],
        out_stride[0], out_stride[1], out_stride[2], out_stride[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul.
    Arguments:
        data = (input_tensor, mask, weights_dict, config_dict)
    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over last dim (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual LN in FP32 for reproducibility
    mean = input_tensor.mean(-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (input_tensor - mean) / torch.rsqrt(var + 1e-5)
    x = x_norm * norm_w + norm_b

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – keep in FP16 to save bandwidth
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"], bias=None
    ).to(torch.float16)
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"], bias=None
    ).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Apply (optional) mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)          # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – stay in FP16
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"], bias=None
    ).sigmoid().to(torch.float16)
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"], bias=None
    ).sigmoid().to(torch.float16)
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"], bias=None
    ).sigmoid().to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. N³ contraction – Triton kernel
    # ------------------------------------------------------------------
    # left/right currently (B,N,N,hidden_dim) float16
    out = launch_trimul_outgoing(left, right)          # (B,N,N,hidden_dim) float32

    # ------------------------------------------------------------------
    # 6. to_out_norm (LayerNorm over hidden_dim)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    # LN in FP32
    mean_o = out.mean(-1, keepdim=True)
    var_o = ((out - mean_o) ** 2).mean(-1, keepdim=True)
    out = (out - mean_o) / torch.rsqrt(var_o + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7. Final gating & linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"], bias=None
    )  # result already float32

    return out
