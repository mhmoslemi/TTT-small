"""
TriMul "outgoing" forward pass implemented with a Triton kernel for the
heavy  N³  einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

The remaining operations (LayerNorms, linear projections, gates) are kept in
PyTorch for simplicity.  The kernel works with fp16 inputs and accumulates in
fp32, returning fp32 results which are then post‑processed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------
# Triton kernel: pairwise multiplication over the third dimension (k)
# --------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,          # *[B, N, N, H]  fp16
    right_ptr,         # *[B, N, N, H]  fp16
    out_ptr,           # *[B, N, N, H]  fp32
    B, N, H,
    stride_lbn, stride_lbk, stride_lbd,   # left strides: batch, i, k, d
    stride_rbn, stride_rbk, stride_rbd,   # right strides: batch, j, k, d
    stride_obn, stride_obi, stride_obj,   # out strides: batch, i, j, d
    BLOCK_M: tl.constexpr,               # tile size for i dimension
    BLOCK_N: tl.constexpr,               # tile size for j dimension
    BLOCK_K: tl.constexpr,               # tile size for k reduction
):
    pid = tl.program_id(0)                         # batch index
    pid_i = tl.program_id(1)                       # i‑tile index
    pid_j = tl.program_id(2)                       # j‑tile index

    # -------------------- offsets --------------------
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    # Mask out‑of‑range rows / cols
    i_mask = i_offset < N
    j_mask = j_offset < N

    # Pointers to the start of this (b,i,j) tile
    left_tile_ptr  = left_ptr  + pid * stride_lbn \
                     + i_offset[:, None, None] * stride_lbk \
                     + k_offset[None, :, None] * stride_lbd
    right_tile_ptr = right_ptr + pid * stride_rbn \
                     + j_offset[:, None, None] * stride_rbk \
                     + k_offset[None, :, None] * stride_rbd

    # Accumulator for the reduction over k
    acc = tl.zeros((BLOCK_M, BLOCK_N, H), dtype=tl.float32)

    # Loop over k in blocks
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        # Load left[b,i,k,:]  -> (M, K, H)
        left = tl.load(left_tile_ptr, mask=(i_mask[:, None, None] & (k_offset < N)), other=0.0)
        # Load right[b,j,k,:] -> (N, K, H)
        right = tl.load(right_tile_ptr, mask=(j_mask[:, None, None] & (k_offset < N)), other=0.0)

        # left: (M, K, H), right: (N, K, H)
        # We need Σ_k left[b,i,k,d] * right[b,j,k,d]
        # Compute dot product across k dimension for each d
        # Use tl.dot which reduces over the second axis
        for d in range(0, H, 1):
            l = left[:, :, d]   # (M, K)
            r = right[:, :, d]  # (N, K)
            # tl.dot expects (M,K) @ (K,N) -> (M,N)
            # Transpose r to (K,N)
            acc[:, :, d] += tl.dot(l, r, trans_b=True)

        # advance pointers
        left_tile_ptr  += BLOCK_K * stride_lbd
        right_tile_ptr += BLOCK_K * stride_rbd
        k_offset += BLOCK_K

    # -------------------- store result --------------------
    out_tile_ptr = out_ptr + pid * stride_obn \
                   + i_offset[:, None, None] * stride_obi \
                   + j_offset[None, :, None] * stride_obj

    # mask for store
    store_mask = i_mask[:, None, None] & j_mask[None, :, None]
    tl.store(out_tile_ptr, acc, mask=store_mask)


# --------------------------------------------------------------------
# Helper function to launch the kernel
# --------------------------------------------------------------------
def trimul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, H]  (fp16)
    returns out  : [B, N, N, H]  (fp32)
    """
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # strides (in elements)
    stride_lbn, stride_lbk, stride_lbd = left.stride()
    stride_rbn, stride_rbk, stride_rbd = right.stride()
    stride_obn, stride_obi, stride_obj = out.stride()

    # tile sizes – chosen to fit H200 shared memory / registers
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (B,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    trimul_einsum_kernel[grid](
        left, right, out,
        B, N, H,
        stride_lbn, stride_lbk, stride_lbd,
        stride_rbn, stride_rbk, stride_rbd,
        stride_obn, stride_obi, stride_obj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # empirically good for fp16 * fp16 -> fp32
    )
    return out


# --------------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul “outgoing” module.
    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]  (float32)
            - mask  : torch.Tensor [B, N, N]      (float32 or bool)
            - weights: dict of torch.Tensor containing all linear / LN params
            - config : dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim, hidden_dim = config["dim"], config["hidden_dim"]

    # ---------- LayerNorm on input ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float16)                     # cast to fp16 for speed

    # ---------- Linear projections (no bias) ----------
    # shapes: [B,N,N,hidden_dim]
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # ---------- Apply mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"])).to(torch.float32)  # keep fp32 for later mul

    left = left * left_gate
    right = right * right_gate

    # ---------- Heavy N³ einsum via Triton ----------
    # left/right are fp16, kernel returns fp32
    out = trimul_einsum(left, right)   # shape [B,N,N,hidden_dim], fp32

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ---------- Apply out gate ----------
    out = out * out_gate.unsqueeze(-1)   # broadcast over hidden_dim

    # ---------- Final linear back to dim ----------
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
