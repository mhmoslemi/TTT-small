"""
TriMul “outgoing” kernel (AlphaFold‑3).

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where
    left  = (x @ W_left)  * mask * σ(x @ W_left_gate)
    right = (x @ W_right) * mask * σ(x @ W_right_gate)

The surrounding LayerNorm, linear projections and gating are cheap
and are kept in PyTorch.  Only the N³ reduction is off‑loaded to a
custom Triton kernel.  The kernel works on a single hidden‑dim channel
(`d`) at a time, which matches the Hopper architecture (large registers
and 128‑bit loads).  It receives pointers to the already‑masked+gated
tensors `left` and `right` of shape

    [B, N, N, H]   (H = hidden_dim)

and writes `out` of the same shape.  The kernel is tiled over the
(`i`,`j`) plane with block sizes `BLOCK_M` and `BLOCK_N`.  The reduction
over `k` is performed in a loop with a tile size `BLOCK_K`.  All
accumulations use fp32 for numerical stability, while loads/stores are
fp16 (the model runs in mixed precision).  The final result is cast back
to fp32 by the caller.

The kernel is launched as:

    grid = (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))

so each program instance handles one `(b, d)` pair and a tile of the
output matrix `(i_tile, j_tile)`.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel for the core N³ einsum
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,          # *fp16  [B, N, N, H]
    right_ptr,         # *fp16  [B, N, N, H]
    out_ptr,           # *fp32  [B, N, N, H]
    B, N, H,
    stride_bl, stride_bi, stride_bk, stride_bd,   # left strides
    stride_rl, stride_ri, stride_rk, stride_rd,   # right strides
    stride_ol, stride_oi, stride_ok, stride_od,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program id gives (b_h, i_tile, j_tile)
    pid = tl.program_id(0)
    pid_bh = pid // (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_i  = (pid // tl.cdiv(N, BLOCK_N)) % tl.cdiv(N, BLOCK_M)
    pid_j  = pid % tl.cdiv(N, BLOCK_N)

    # decode batch and hidden dim from pid_bh
    b = pid_bh // H
    d = pid_bh % H

    # start offsets for this tile
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds tiles
    mask_i = i_offset < N
    mask_j = j_offset < N

    # pointers to the (b,d) slice
    left_base  = left_ptr  + b * stride_bl + d * stride_bd
    right_base = right_ptr + b * stride_rl + d * stride_rd
    out_base   = out_ptr   + b * stride_ol + d * stride_od

    # allocate accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left[b,i,k,d]  -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base \
            + (i_offset[:, None] * stride_bi) \
            + (k_offset[None, :] * stride_bk)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right[b,j,k,d] -> shape (BLOCK_N, BLOCK_K)
        right_ptrs = right_base \
            + (j_offset[:, None] * stride_ri) \
            + (k_offset[None, :] * stride_rk)
        # note: we need right transposed to match dot (M,N) x (K,N) -> (M,N)
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right is (BLOCK_N, BLOCK_K); we need (BLOCK_K, BLOCK_N)
        right_T = tl.trans(right)

        # accumulate product over k
        acc += tl.dot(left, right_T)   # (BLOCK_M, BLOCK_N)

    # write result
    out_ptrs = out_base \
        + (i_offset[:, None] * stride_oi) \
        + (j_offset[None, :] * stride_ok)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Helper: fused LayerNorm (fp32 -> fp32)
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: [..., C]  fp32
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    norm = (x - mean) / torch.sqrt(var + eps)
    return norm * weight + bias

# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    The heavy N³ reduction is performed by a Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------------------- 1. LayerNorm on the input ----------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"].to(torch.float32),
        bias=weights["norm.bias"].to(torch.float32),
    )   # [B,N,N,dim]  fp32

    # ---------------------- 2. Linear projections (fp32 -> fp16) ----------
    # W shapes: [dim, hidden_dim]
    left_proj = torch.nn.functional.linear(
        x, weights["left_proj.weight"].to(torch.float32), bias=None
    )
    right_proj = torch.nn.functional.linear(
        x, weights["right_proj.weight"].to(torch.float32), bias=None
    )
    # gating
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"].to(torch.float32), bias=None
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"].to(torch.float32), bias=None
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"].to(torch.float32), bias=None
    ).sigmoid()

    # apply mask if present
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)               # [B,N,N,1]
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # element‑wise gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    # cast to fp16 for the kernel (saves bandwidth)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ---------------------- 3. Triton N³ reduction -----------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements) for the 4‑D tensors
    # left  [B, N, N, H]  -> (b,i,k,d)
    stride_bl, stride_bi, stride_bk, stride_bd = left.stride()
    # right [B, N, N, H]  -> (b,j,k,d)
    stride_rl, stride_ri, stride_rk, stride_rd = right.stride()
    # out   [B, N, N, H]  -> (b,i,j,d)
    stride_ol, stride_oi, stride_ok, stride_od = out.stride()

    # block sizes – tuned for Hopper (shared‑memory friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(B * hidden_dim, 1),                 # one program per (b, d)
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_bl=stride_bl,
        stride_bi=stride_bi,
        stride_bk=stride_bk,
        stride_bd=stride_bd,
        stride_rl=stride_rl,
        stride_ri=stride_ri,
        stride_rk=stride_rk,
        stride_rd=stride_rd,
        stride_ol=stride_ol,
        stride_oi=stride_oi,
        stride_ok=stride_ok,
        stride_od=stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # Hopper can sustain many warps
    )

    # ---------------------- 4. Post‑processing (LayerNorm, gating) -----
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"].to(torch.float32),
        bias=weights["to_out_norm.bias"].to(torch.float32),
    )
    # apply out_gate (broadcast over i,j)
    out = out * out_gate

    # final linear projection back to original dim
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float32), bias=None
    )   # [B,N,N,dim]

    return out.to(torch.float32)
