"""
TriMul “outgoing” implementation.

The expensive part of the forward pass is

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]          (1)

where left/right have shape (B,N,N,H) and H = hidden_dim.
(1) is a reduction over the sequence dimension *k* and can be
performed with a Triton kernel that iterates over *k* in blocks,
accumulates the element‑wise product and finally writes the result.

All LayerNorm / Linear / gating operations are kept in PyTorch for
readability – they are cheap compared with the O(B·N³·H) work of (1).
The kernel works for any batch size (1‑2), any sequence length that fits
in memory and any hidden dimension that is a multiple of 32
(the internal block size we use).  The final output is cast back to
float‑32 as required by the specification.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ------------------------------------------------------------------- #
# Triton kernel that computes the reduction in (1)
# ------------------------------------------------------------------- #
@triton.jit
def trimul_kernel(
    left_ptr,          # const float*  [B,N,N,H]
    right_ptr,         # const float*  [B,N,N,H]
    out_ptr,           #       float*  [B,N,N,H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Each program instance computes a tile (i0:i0+BLOCK_I, j0:j0+BLOCK_J, :).
    The reduction over k is performed in a loop over blocks of size BLOCK_K.
    """
    pid = tl.program_id(0)

    # total number of (b,i_tile,j_tile) tiles
    tiles_i = tl.cdiv(N, BLOCK_I)
    tiles_j = tl.cdiv(N, BLOCK_J)
    tiles_per_batch = tiles_i * tiles_j

    b = pid // tiles_per_batch
    pid_tile = pid % tiles_per_batch
    i_tile = pid_tile // tiles_j
    j_tile = pid_tile % tiles_j

    # offsets within the tensors
    i_off = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i_off < N
    j_mask = j_off < N

    # allocate accumulator in float32 (better precision)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # ----- load left[b,i,k,:]  (I,K,H) -----
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i_off[:, None, None] * stride_left_i
            + k_off[None, :, None] * stride_left_k
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_left_h
        )
        left = tl.load(left_ptrs,
                       mask=i_mask[:, None, None] & k_mask[None, :, None],
                       other=0.0)          # (I,K,H)

        # ----- load right[b,j,k,:] (J,K,H) -----
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + j_off[:, None, None] * stride_right_j
            + k_off[None, :, None] * stride_right_k
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_right_h
        )
        right = tl.load(right_ptrs,
                        mask=j_mask[:, None, None] & k_mask[None, :, None],
                        other=0.0)          # (J,K,H)

        # we need left[i,k,h] * right[j,k,h] summed over k.
        # Expand dimensions so that broadcasting can happen over the k axis.
        # left:  (I, K,   1, H)
        # right: (1, K,   J, H)
        left_exp  = left[:, :, None, :]          # (I,K,1,H)
        right_exp = right[None, :, :, :]         # (1,K,J,H)

        prod = left_exp * right_exp               # (I,K,J,H)
        acc += tl.sum(prod, axis=1)                # sum over K → (I,J,H)

    # ----- write result -----
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i_off[:, None, None] * stride_out_i
        + j_off[None, :, None] * stride_out_j
        + tl.arange(0, BLOCK_H)[None, None, :] * stride_out_h
    )
    # store as float16 to save bandwidth, convert to fp32 later in PyTorch
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=i_mask[:, None, None] & j_mask[None, :, None])


# ------------------------------------------------------------------- #
# Helper – launch wrapper
# ------------------------------------------------------------------- #
def launch_trimum_kernel(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : torch.Tensor  [B,N,N,H]  (float32)
    returns out : torch.Tensor    [B,N,N,H]  (float32)
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float16)   # we store fp16 inside the kernel

    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – chosen to fit H200 shared memory comfortably
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_H = 32               # hidden_dim must be a multiple of 32

    total_tiles = B * triton.cdiv(N, BLOCK_I) * triton.cdiv(N, BLOCK_J)
    grid = (total_tiles,)

    trimul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
    )
    # cast back to float32 for the rest of the pipeline
    return out.to(torch.float32)


# ------------------------------------------------------------------- #
# Full forward pass – mixture of PyTorch (cheap ops) and Triton (expensive)
# ------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  float32
        - mask         : [B, N, N]      (bool / float)
        - weights      : dict of parameter tensors
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    output : torch.Tensor  [B, N, N, dim]  float32
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------- #
    # 1. LayerNorm on the last dimension
    # --------------------------------------------------------------- #
    # Manual LayerNorm because we have the weight / bias already.
    eps = 1e-5
    # shape (dim,)
    ln_w = weights["norm.weight"].to(input_tensor.dtype)
    ln_b = weights["norm.bias"].to(input_tensor.dtype)

    # compute mean / var over dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x_norm * ln_w + ln_b                               # [B,N,N,dim]

    # --------------------------------------------------------------- #
    # 2. Linear projections (no bias) – we keep them in fp32
    # --------------------------------------------------------------- #
    left_proj_w = weights["left_proj.weight"]            # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]

    left = torch.einsum("bnnd,hd->bnnhd", x, left_proj_w)   # shape (B,N,N,hidden_dim)
    right = torch.einsum("bnnd,hd->bnnhd", x, right_proj_w)

    # --------------------------------------------------------------- #
    # 3. Apply mask (if present)
    # --------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------- #
    # 4. Gating
    # --------------------------------------------------------------- #
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.einsum("bnnd,hd->bnnhd", x, left_gate_w).sigmoid()
    right_gate = torch.einsum("bnnd,hd->bnnhd", x, right_gate_w).sigmoid()
    out_gate = torch.einsum("bnnd,hd->bnnhd", x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # 5. Heavy reduction – Triton kernel
    # --------------------------------------------------------------- #
    # left/right are (B,N,N,hidden_dim) – we pass them directly.
    out = launch_trimum_kernel(left, right)            # (B,N,N,hidden_dim) float32

    # --------------------------------------------------------------- #
    # 6. Output projection + final LayerNorm + gating
    # --------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]                # (dim, hidden_dim)

    # LayerNorm on hidden_dim
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out = out_norm * to_out_norm_w + to_out_norm_b

    # Gate
    out = out * out_gate

    # Final linear
    output = torch.einsum("bnnhd,dh->bnnd", out, to_out_w)   # back to dim

    return output
