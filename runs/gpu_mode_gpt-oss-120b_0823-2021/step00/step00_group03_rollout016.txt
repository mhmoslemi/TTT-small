"""
TriMul “outgoing” forward pass (AlphaFold‑3).

The only non‑trivial operation is the 4‑D einsum

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where
    left  = Linear(dim→hdim)(norm(x)) * mask * sigmoid(Linear(dim→hdim)(x))
    right = Linear(dim→hdim)(norm(x)) * mask * sigmoid(Linear(dim→hdim)(x))

`left` and `right` are produced with regular torch linear layers (they are
tiny compared to the cubic contraction).  The cubic contraction is carried
out by a fused Triton kernel that tiles the three dimensions `(i, j, k)` and
processes the hidden dimension `h` in small chunks.  The kernel accumulates
in `float32` regardless of the compute precision (we load in `float16` for
bandwidth).  After the kernel we finish the pipeline in torch:
* Layer‑norm on the hidden dimension
* Gating with `out_gate`
* Final linear projection back to `dim`

The implementation works for any batch size (1‑2), any sequence length
up to 1024 and any hidden‑dimension that is a multiple of 32 (the reference
models use 128).  It also supports the optional mask (the mask is already
expanded to the hidden dimension before the kernel launch).
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel:  out[b, i, j, h] += Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,               # [B, N, N, H]   (float16)
    right_ptr,              # [B, N, N, H]   (float16)
    out_ptr,                # [B, N, N, H]   (float32) – write‑only
    # strides (in elements, not bytes)
    stride_bl_i, stride_bl_k, stride_bl_h,   # left  B,N,K,H
    stride_br_j, stride_br_k, stride_br_h,   # right B,N,K,H
    stride_bo_i, stride_bo_j, stride_bo_h,   # out   B,N,N,H
    # problem size
    B, N, H,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)                # batch index
    pid_i = tl.program_id(1)                # tile over i
    pid_j = tl.program_id(2)                # tile over j

    # ------------------------------------------------------------------
    # 1. compute the start offsets of the tile
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # coordinate vectors for the tile
    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, BLOCK_H)

    # mask out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # ------------------------------------------------------------------
    # 2. create accumulator (float32) for the whole tile
    #    shape: [BLOCK_I, BLOCK_J, BLOCK_H]
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3. loop over k tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # --------------------------------------------------------------
        # 3a. load left[b, i, k, h]   -> shape [BLOCK_I, BLOCK_K, BLOCK_H]
        # --------------------------------------------------------------
        left_ptrs = left_ptr + \
            pid_b * stride_bl_i * N * H + \
            offs_i[:, None, None] * stride_bl_i + \
            offs_k[None, :, None] * stride_bl_k + \
            offs_h[None, None, :] * stride_bl_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] &
                            mask_k[None, :, None] &
                            mask_h[None, None, :],
                       other=0.0).to(tl.float32)      # promote to fp32

        # --------------------------------------------------------------
        # 3b. load right[b, j, k, h]  -> shape [BLOCK_J, BLOCK_K, BLOCK_H]
        # --------------------------------------------------------------
        right_ptrs = right_ptr + \
            pid_b * stride_br_j * N * H + \
            offs_j[:, None, None] * stride_br_j + \
            offs_k[None, :, None] * stride_br_k + \
            offs_h[None, None, :] * stride_br_h
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] &
                             mask_k[None, :, None] &
                             mask_h[None, None, :],
                        other=0.0).to(tl.float32)

        # --------------------------------------------------------------
        # 3c. contract over k: for each h we need Σ_k left[i,k,h] * right[j,k,h]
        #     This can be expressed as a batch of dot products.
        # --------------------------------------------------------------
        # left : [I, K, H]   right : [J, K, H]
        # we compute for every (i, j, h):
        #   Σ_k left[i,k,h] * right[j,k,h]
        # We use tl.dot on the K dimension.
        for ih in range(0, BLOCK_H, 1):
            # slice the current hidden channel
            l = left[:, :, ih]       # [I, K]
            r = right[:, :, ih]      # [J, K]
            # tl.dot expects the reduction dimension as the last axis of the
            # first operand and the first axis of the second operand.
            # So we transpose r.
            dot = tl.dot(l, r.T)     # [I, J]
            acc[:, :, ih] += dot

    # ------------------------------------------------------------------
    # 4. write accumulator back to global memory (float32)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        pid_b * stride_bo_i * N * H + \
        offs_i[:, None, None] * stride_bo_i + \
        offs_j[None, :, None] * stride_bo_j + \
        offs_h[None, None, :] * stride_bo_h

    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] &
                  mask_j[None, :, None] &
                  mask_h[None, None, :])

# ----------------------------------------------------------------------
# Host side helper -----------------------------------------------------
# ----------------------------------------------------------------------
def _launch_trimul(out, left, right, BLOCK_I=64, BLOCK_J=64, BLOCK_K=64, BLOCK_H=32):
    """
    Launch `trimul_outgoing_kernel` for a single batch.
    All tensors must be on CUDA, left/right are float16, out is float32.
    """
    B, N, _, H = left.shape
    # strides in elements (not bytes)
    stride_bl_i, stride_bl_k, stride_bl_h = left.stride(0), left.stride(2), left.stride(3)
    stride_br_j, stride_br_k, stride_br_h = right.stride(0), right.stride(2), right.stride(3)
    stride_bo_i, stride_bo_j, stride_bo_h = out.stride(0), out.stride(2), out.stride(3)

    grid = (
        B,
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
    )
    trimul_outgoing_kernel[grid](
        left, right, out,
        stride_bl_i, stride_bl_k, stride_bl_h,
        stride_br_j, stride_br_k, stride_br_h,
        stride_bo_i, stride_bo_j, stride_bo_h,
        B, N, H,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
        num_warps=4,
    )

# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)

        * input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        * mask         : torch.Tensor [B, N, N]       (float32 or bool)
        * weights      : dict of torch.Tensor parameters
        * config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias   = weights["norm.bias"].to(device)
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) \
        / torch.sqrt(input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5)
    x = x * norm_weight + norm_bias                           # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # we keep the operations in float16 for the cubic contraction
    x_fp16 = x.to(torch.float16)

    left_proj_w  = weights["left_proj.weight"].to(device)      # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"].to(device)

    left  = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B,N,N,hdim]
    right = torch.nn.functional.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)            # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear layers on the *original* x)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    x_fp32 = x.to(torch.float32)        # gating works in fp32 for stability
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_fp32, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp32, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_fp32, out_gate_w))

    left  = left  * left_gate.to(torch.float16)
    right = right * right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Cubic contraction in Triton (output in fp32)
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)
    _launch_trimul(out, left, right,
                   BLOCK_I=64, BLOCK_J=64, BLOCK_K=64, BLOCK_H=32)

    # ------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm, output gate, final linear)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    to_out_w      = weights["to_out.weight"].to(device)

    # Layer‑norm on hidden dimension
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(out.var(-1, unbiased=False, keepdim=True) + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    # Apply output gate (broadcast over hidden dim)
    out = out * out_gate.to(torch.float32).unsqueeze(-1)   # [B,N,N,hdim]

    # Final linear back to `dim`
    out = torch.nn.functional.linear(out, to_out_w)       # [B,N,N,dim]

    return out
