"""
TriMul “outgoing” implementation.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,D) and apply the four linear layers
   (proj & gates) using the supplied weights (all ops stay in torch).
2. Mask the projected tensors and multiply by the sigmoid‑gates.
3. The expensive part is the 4‑D “einsum”

       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

   which we compute with a Triton kernel.  For each batch we launch a
   2‑D grid over blocks of `i` and `j`.  Inside the kernel we iterate over
   the sequence dimension `k` (scalar loop, BLOCK_K = 1) and accumulate the
   outer‑product of the `D`‑dimensional vectors.  This yields a per‑block
   accumulator of shape (BLOCK_I, BLOCK_J, D) in FP32.  After the loop the
   result is written back to the output tensor.

4. Finally apply the output Layer‑Norm, the output gate and the final
   linear projection (all in torch) and return a float‑32 tensor.

The kernel performs the only O(N³·D) work; everything else is cheap
torch ops, which gives a strong speed‑up on Hopper (sm_90) while keeping
the implementation simple and correct for all required shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel: core “einsum” reduction
# --------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,               # *float32   [B,N,N,D]
    right_ptr,              # *float32   [B,N,N,D]
    out_ptr,                # *float32   [B,N,N,D]
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
):
    """Compute  out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]   """

    pid_b = tl.program_id(0)                     # batch index
    pid_ij = tl.program_id(1)                    # combined (i,j) block index

    # number of blocks in each dimension
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)

    # current block coordinates
    i_block = pid_ij // num_j
    j_block = pid_ij % num_j

    # absolute indices for the block
    i = i_block * BLOCK_I + tl.arange(0, BLOCK_I)
    j = j_block * BLOCK_J + tl.arange(0, BLOCK_J)

    # masks for out‑of‑bounds threads
    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32   shape (BLOCK_I, BLOCK_J, D)
    acc = tl.zeros((BLOCK_I, BLOCK_J, D), dtype=tl.float32)

    # ----------------------------------------------------------
    # loop over the reduction dimension k (scalar loop: BLOCK_K = 1)
    # ----------------------------------------------------------
    for k in range(0, N):
        # pointers for the current k
        left_off = (
            pid_b * stride_left_b
            + i[:, None, None] * stride_left_i
            + k * stride_left_k
            + tl.arange(0, D)[None, None, :] * stride_left_d
        )
        right_off = (
            pid_b * stride_right_b
            + j[None, :, None] * stride_right_j
            + k * stride_right_k
            + tl.arange(0, D)[None, None, :] * stride_right_d
        )

        # load vectors of shape (BLOCK_I, D) and (BLOCK_J, D)
        left_vec = tl.load(left_ptr + left_off, mask=mask_i[:, None], other=0.0)   # (I, D)
        right_vec = tl.load(right_ptr + right_off, mask=mask_j[None, :], other=0.0) # (J, D)

        # broadcast to (I, J, D) and accumulate
        acc += left_vec[:, None, :] * right_vec[None, :, :]

    # ----------------------------------------------------------
    # write result back
    # ----------------------------------------------------------
    out_off = (
        pid_b * stride_out_b
        + i[:, None, None] * stride_out_i
        + j[None, :, None] * stride_out_j
        + tl.arange(0, D)[None, None, :] * stride_out_d
    )
    tl.store(out_ptr + out_off,
             acc,
             mask=(mask_i[:, None, None] & mask_j[None, :, None]))

# --------------------------------------------------------------
# Helper for layer‑norm without torch.nn.LayerNorm (uses torch ops)
# --------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """x: [..., C]  weight/bias: [C]  ->  normalized output (float32)"""
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias

# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args
    ----
    data : (input, mask, weights, config)
        input  : torch.Tensor[B, N, N, D]      (float32, on CUDA)
        mask   : torch.Tensor[B, N, N]         (float32/bool, on CUDA)
        weights: dict of tensors (all float32, on CUDA)
        config : dict with keys `dim` and `hidden_dim`

    Returns
    -------
    torch.Tensor[B, N, N, D]  (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension (c_z)

    # ------------------------------------------------------------------
    # 1. Layer‑norm + linear projections & gates (pure torch)
    # ------------------------------------------------------------------
    # norm
    x = layer_norm(input_tensor,
                   weights["norm.weight"],
                   weights["norm.bias"])

    # linear projections (bias‑free)
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # mask (optional)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # gates
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 2. Core TriMul via Triton
    # ------------------------------------------------------------------
    # allocate output buffer (FP32)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=input_tensor.device)

    # strides for pointer arithmetic
    # left:  (B,N,N,H)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    # right: (B,N,N,H)  note index order is (b, j, k, h) in the einsum
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    # out:   (B,N,N,H)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # launch parameters
    BLOCK_I = 32
    BLOCK_J = 32
    grid = (B, triton.cdiv(N, BLOCK_I) * triton.cdiv(N, BLOCK_J))

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_d=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_d=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_d=stride_out_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
    )

    # ------------------------------------------------------------------
    # 3. Output post‑processing (torch)
    # ------------------------------------------------------------------
    out = layer_norm(out,
                     weights["to_out_norm.weight"],
                     weights["to_out_norm.bias"])

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
