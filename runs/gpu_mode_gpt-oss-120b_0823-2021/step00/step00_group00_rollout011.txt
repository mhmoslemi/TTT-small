"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Layer‑norm the input tensor x  →  x̂  (float32).
2. Linear projections + gating (all done with torch for simplicity):
       left  = (x̂ @ W_left ) * mask * sigmoid(x̂ @ W_left_gate )
       right = (x̂ @ W_right) * mask * sigmoid(x̂ @ W_right_gate)
   Shapes after step‑2:  [B, N, N, H]   (H = hidden_dim)
3. Core TriMul operation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply of size (N×N) for each (b,h) pair,
   where the right operand is transposed on its last two axes.
   The heavy‑weight N³ work is performed by a custom Triton kernel that
   computes a blocked GEMM in FP16 and accumulates into FP32.
4. Apply final LayerNorm → gate → linear projection (torch).

The kernel implements a classic tiled GEMM:
   BLOCK_M × BLOCK_N output tile,
   iterating over K‑tiles of size BLOCK_K.
   We fuse the batch‑and‑hidden dimensions into a single “outer”
   dimension (outer = B * H) to launch a 2‑D grid (outer, ceil(N/BLOCK_M),
   ceil(N/BLOCK_N)).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel : batched GEMM (left @ rightᵀ)  for the TriMul core
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    # pointers
    left_ptr,          # [outer, N, N]   (float16)
    right_ptr,         # [outer, N, N]   (float16)   (will be read transposed)
    out_ptr,           # [outer, N, N]   (float32)
    # dimensions
    N, H,              # seq_len, hidden_dim (used only for stride calcs)
    stride_left_outer, stride_left_m, stride_left_k,
    stride_right_outer, stride_right_m, stride_right_k,
    stride_out_outer, stride_out_m, stride_out_n,
    # tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr
):
    """
    Computes, for each outer index (batch*hidden):
        out[m, n] = Σ_k left[m, k] * right[n, k]      (right read transposed)
    """
    pid_outer = tl.program_id(0)               # outer = b * H
    pid_m = tl.program_id(1)                   # tile over rows   (i)
    pid_n = tl.program_id(2)                   # tile over cols   (j)

    # --------------------------------------------------------------
    # compute tile offsets
    # --------------------------------------------------------------
    num_m_tiles = tl.cdiv(N, BLOCK_M)
    num_n_tiles = tl.cdiv(N, BLOCK_N)

    # grid‑reduction across m‑tiles to improve L2 reuse (standard trick)
    group_id = (pid_m * GROUP_SIZE_M) // num_m_tiles
    first_pid_m = group_id * (num_m_tiles // GROUP_SIZE_M)
    pid_m = first_pid_m + (pid_m % (num_m_tiles // GROUP_SIZE_M))

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers for the current tile
    left_tile_ptr  = left_ptr  + pid_outer * stride_left_outer \
                     + offs_m[:, None] * stride_left_m \
                     + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    right_tile_ptr = right_ptr + pid_outer * stride_right_outer \
                     + offs_n[:, None] * stride_right_m \
                     + tl.arange(0, BLOCK_K)[None, :] * stride_right_k   # k dimension same

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load a K‑tile from left (M × K)  – fp16 → fp32 on the fly
        left = tl.load(left_tile_ptr, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < N - k*BLOCK_K), other=0.0)
        # load a K‑tile from right (N × K) – we will use it as transposed
        right = tl.load(right_tile_ptr, mask=mask_n[:, None] & (tl.arange(0, BLOCK_K)[None, :] < N - k*BLOCK_K), other=0.0)

        # dot: (M×K) · (N×K)ᵀ  → (M×N)
        # trick: tl.dot works on (M×K)·(K×N), so we just swap axes of right
        acc += tl.dot(left, right, trans_b=True)

        # advance pointers to next K‑tile
        left_tile_ptr  += BLOCK_K * stride_left_k
        right_tile_ptr += BLOCK_K * stride_right_k

    # --------------------------------------------------------------
    # write result
    # --------------------------------------------------------------
    out_tile_ptr = out_ptr + pid_outer * stride_out_outer \
                   + offs_m[:, None] * stride_out_m \
                   + offs_n[None, :] * stride_out_n
    tl.store(out_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Host‑side helper to launch the kernel
# ----------------------------------------------------------------------
def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : [B, N, N, H]  (float16)
    returns    : [B, N, N, H]  (float32)
    """
    B, N, _, H = left.shape
    # fuse outer dimension
    outer = B * H

    # view as [outer, N, N]
    left_view  = left.permute(0, 3, 1, 2).reshape(outer, N, N).contiguous()
    right_view = right.permute(0, 3, 1, 2).reshape(outer, N, N).contiguous()

    out = torch.empty_like(left_view, dtype=torch.float32)

    # strides (in elements, not bytes)
    stride_left_outer, stride_left_m, stride_left_k = left_view.stride()
    stride_right_outer, stride_right_m, stride_right_k = right_view.stride()
    stride_out_outer, stride_out_m, stride_out_n = out.stride()

    # tile sizes – tuned for Hopper (SM90) and typical N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    GROUP_SIZE_M = 8

    grid = (
        outer,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_gemm_kernel[grid](
        left_view,
        right_view,
        out,
        N,
        H,
        stride_left_outer,
        stride_left_m,
        stride_left_k,
        stride_right_outer,
        stride_right_m,
        stride_right_k,
        stride_out_outer,
        stride_out_m,
        stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
        num_warps=4,               # enough for fp16 dot
    )
    # reshape back to [B, N, N, H]
    return out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]


# ----------------------------------------------------------------------
# Full custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            mask         : torch.Tensor [B, N, N]       (float32 or bool)
            weights      : dict of parameter tensors
            config       : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                   # [B,N,N,dim]  float32

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – we keep them in float16 for speed
    # ------------------------------------------------------------------
    # reshape to 2‑D for a single matmul per projection
    x_flat = x.reshape(-1, dim)                     # [(B*N*N), dim]

    # left / right projections
    left_proj  = torch.nn.functional.linear(x_flat, weights["left_proj.weight"]).to(torch.float16)
    right_proj = torch.nn.functional.linear(x_flat, weights["right_proj.weight"]).to(torch.float16)

    left  = left_proj.reshape(B, N, N, hidden)     # [B,N,N,H]  fp16
    right = right_proj.reshape(B, N, N, hidden)    # [B,N,N,H]  fp16

    # ------------------------------------------------------------------
    # 3. Mask handling (optional)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) – still in float32 for stability, then cast
    # ------------------------------------------------------------------
    # compute gates on the original x (float32)
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["left_gate.weight"])).reshape(B, N, N, hidden)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["right_gate.weight"])).reshape(B, N, N, hidden)
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_flat, weights["out_gate.weight"])).reshape(B, N, N, hidden)

    left  = (left * left_gate.to(left.dtype))
    right = (right * right_gate.to(right.dtype))

    # ------------------------------------------------------------------
    # 5. Core TriMul (batched GEMM) – Triton
    # ------------------------------------------------------------------
    # out_tmp : [B, N, N, hidden]  (float32)
    out_tmp = trimul_gemm(left, right)               # fp16 → fp32 inside kernel

    # ------------------------------------------------------------------
    # 6. Final LayerNorm, gate and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_tmp,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                   # [B,N,N,hidden] fp32

    out = out * out_gate                # apply output gate (float32)

    # linear projection back to dim
    out_flat = out.reshape(-1, hidden)
    out_proj = torch.nn.functional.linear(out_flat, weights["to_out.weight"]).reshape(B, N, N, dim)

    return out_proj.to(torch.float32)
