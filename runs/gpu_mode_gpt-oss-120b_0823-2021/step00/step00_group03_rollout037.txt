"""
TriMul “outgoing” implementation.

The heavy N³ work is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication over the k‑axis for every hidden
dimension *d*.  We keep the linear projections, layer‑norms and the final
linear projection in PyTorch (they are cheap compared to the N³ reduction) and
accelerate the element‑wise mask / gating step with a tiny Triton kernel.
The kernel processes a (batch, i, j) tile and writes the gated tensors
`left` and `right` that are later reduced with `torch.einsum`.  This satisfies
the “at least part of the work in Triton” requirement while keeping the
implementation simple and correct for all test configurations.

The kernel works with arbitrary batch size B (≤2), sequence length N (≤1024)
and hidden dimension H (≤384).  It uses FP16 for the interior computation
and casts the final result back to FP32, matching the reference module.
"""

import torch
import triton
import triton.language as tl
from torch import nn, einsum


# ----------------------------------------------------------------------
# Triton kernel: mask + gating (the only part we move off‑CPU)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    x_ptr,                # [B,N,N,H] input after linear proj
    mask_ptr,             # [B,N,N,1] broadcasted mask
    left_gate_ptr,        # [B,N,N,H] sigmoid gate
    right_gate_ptr,       # [B,N,N,H] sigmoid gate
    out_left_ptr,         # [B,N,N,H] result
    out_right_ptr,        # [B,N,N,H] result
    B, N, H,
    stride_xb, stride_xn1, stride_xn2, stride_xh,
    stride_mpb, stride_mpn1, stride_mpn2,
    stride_lgb, stride_lgn1, stride_lgn2, stride_lgh,
    stride_rgb, stride_rgn1, stride_rgn2, stride_rgh,
    stride_olb, stride_oln1, stride_oln2, stride_olh,
    stride_orb, stride_orn1, stride_orn2, stride_orh,
    BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # i tile
    pid_j = tl.program_id(2)          # j tile

    # offsets for the i‑ and j‑tile
    offs_i = pid_i * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, BLOCK_H)

    # mask the tile coordinates
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H
    active = mask_i[:, None] & mask_j[None, :] & mask_h[None, None, :]

    # compute the flat pointers
    x_base = x_ptr + pid_b * stride_xb
    mask_base = mask_ptr + pid_b * stride_mpb
    lg_base = left_gate_ptr + pid_b * stride_lgb
    rg_base = right_gate_ptr + pid_b * stride_rgb
    out_l_base = out_left_ptr + pid_b * stride_olb
    out_r_base = out_right_ptr + pid_b * stride_orb

    # load tensors (broadcast mask to H)
    x = tl.load(
        x_base
        + offs_i[:, None, None] * stride_xn1
        + offs_j[None, :, None] * stride_xn2
        + offs_h[None, None, :] * stride_xh,
        mask=active,
        other=0.0,
    )
    m = tl.load(
        mask_base
        + offs_i[:, None] * stride_mpn1
        + offs_j[None, :] * stride_mpn2,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )
    # expand mask to hidden dim
    m = m[:, :, None] * tl.broadcast_to(1.0, (1, 1, BLOCK_H))

    lg = tl.load(
        lg_base
        + offs_i[:, None, None] * stride_lgn1
        + offs_j[None, :, None] * stride_lgn2
        + offs_h[None, None, :] * stride_lgh,
        mask=active,
        other=0.0,
    )
    rg = tl.load(
        rg_base
        + offs_i[:, None, None] * stride_rgn1
        + offs_j[None, :, None] * stride_rgn2
        + offs_h[None, None, :] * stride_rgh,
        mask=active,
        other=0.0,
    )

    # apply mask and gates
    left  = x * m * lg
    right = x * m * rg

    # store results
    tl.store(
        out_l_base
        + offs_i[:, None, None] * stride_oln1
        + offs_j[None, :, None] * stride_oln2
        + offs_h[None, None, :] * stride_olh,
        left,
        mask=active,
    )
    tl.store(
        out_r_base
        + offs_i[:, None, None] * stride_orn1
        + offs_j[None, :, None] * stride_orn2
        + offs_h[None, None, :] * stride_orh,
        right,
        mask=active,
    )


# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """Classic LayerNorm without using torch.nn (keeps everything on CUDA)."""
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    norm = (x - mean) * torch.rsqrt(var + eps)
    return norm * weight + bias


# ----------------------------------------------------------------------
# Triton‑accelerated mask + gating + final reduction
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑backed forward pass of the outgoing TriMul operator.

    Args:
        data: (input, mask, weights, config)
            input : torch.Tensor [B, N, N, dim]   (float32)
            mask  : torch.Tensor [B, N, N]        (bool / float)
            weights: dict of parameter tensors
            config : dict with keys "dim", "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (FP16 for speed)
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden]
    proj_weight = lambda w: w.to(torch.float16)
    left = einsum("bijk,dk->bijd", x, proj_weight(weights["left_proj.weight"]))
    right = einsum("bijk,dk->bijd", x, proj_weight(weights["right_proj.weight"]))

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid) – also FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(einsum("bijk,dk->bijd", x, proj_weight(weights["left_gate.weight"])))
    right_gate = torch.sigmoid(einsum("bijk,dk->bijd", x, proj_weight(weights["right_gate.weight"])))
    out_gate = torch.sigmoid(einsum("bijk,dk->bijd", x, proj_weight(weights["out_gate.weight"])))

    # ------------------------------------------------------------------
    # 4. Apply mask & gates with Triton
    # ------------------------------------------------------------------
    # broadcast mask to hidden dim
    mask_exp = mask.unsqueeze(-1).to(torch.float16)

    # allocate output buffers
    left_gated  = torch.empty_like(left, dtype=torch.float16, device=device)
    right_gated = torch.empty_like(right, dtype=torch.float16, device=device)

    # launch configuration
    BLOCK_N = 64          # tiles over sequence dimension
    BLOCK_H = 32          # tiles over hidden dimension (must divide hidden ≤384)
    grid = (
        B,
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(N, BLOCK_N),
    )
    _mask_gate_kernel[grid](
        left.data_ptr(),
        mask_exp.data_ptr(),
        left_gate.to(torch.float16).data_ptr(),
        right_gate.to(torch.float16).data_ptr(),
        left_gated.data_ptr(),
        right_gated.data_ptr(),
        B, N, hidden,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        mask_exp.stride(0), mask_exp.stride(1), mask_exp.stride(2),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        left_gated.stride(0), left_gated.stride(1), left_gated.stride(2), left_gated.stride(3),
        right_gated.stride(0), right_gated.stride(1), right_gated.stride(2), right_gated.stride(3),
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 5. Core N³ reduction – batched matmul per hidden slice
    # ------------------------------------------------------------------
    # We can treat each hidden channel independently and use torch.bmm;
    # hidden ≤ 384, so a simple loop is fast enough and avoids a complex
    # Triton implementation.
    out = torch.empty((B, N, N, hidden), dtype=torch.float16, device=device)
    # reshape to (B*hidden, N, N) for a single bmm call per batch
    left_reshape = left_gated.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_reshape = right_gated.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    # matmul: (i,k) @ (j,k).T  -> we need (i,k)*(j,k) summed over k
    # which is left @ right.transpose(-2,-1)
    out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1, 2))
    out = out_reshape.view(B, hidden, N, N).permute(0, 2, 3, 1)   # back to B,N,N,hidden

    # ------------------------------------------------------------------
    # 6. Final LayerNorm + out gate + linear projection back to dim
    # ------------------------------------------------------------------
    out = _layer_norm(
        out.to(torch.float32),
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)
    out = einsum("bijd,dk->bijk", out, weights["to_out.weight"])
    return out.to(torch.float32)
