"""
TriMul "outgoing" forward pass with a fused Triton kernel.

The heavy part of the TriMul operation is the 4‑D Einstein sum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]                (1)

where
    left  = (x @ W_left )  * sigmoid(x @ W_left_gate) * mask
    right = (x @ W_right)  * sigmoid(x @ W_right_gate) * mask

All LayerNorms and Linear projections are kept in PyTorch (they are already
highly optimized).  The triple‑loop over *k* is replaced by a batched
matrix‑multiply kernel written in Triton, which processes a tile of (i,j)
for a small block of the hidden dimension *d* in FP16 and accumulates in
FP32.  The kernel is launched with a 4‑D grid (batch, i‑tile, j‑tile, d‑tile).

The final result is cast back to FP32, multiplied by the output gate and
passed through the final LayerNorm + Linear projection, matching the
reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# -------------------------------------------------------------------------
# Triton kernel for the core einsum (1)
# -------------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,                # [B, N, N, D]   left tensor
    right_ptr,               # [B, N, N, D]   right tensor
    out_ptr,                 # [B, N, N, D]   output tensor (FP32)
    B, N, D,
    stride_lb, stride_lk, stride_ld,   # strides for left  (B,N,K,D)
    stride_rb, stride_rj, stride_rd,   # strides for right (B,J,K,D)
    stride_ob, stride_oi, stride_oj,   # strides for out   (B,I,J,D)
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program identifiers
    pid_b = tl.program_id(0)                # batch dimension
    pid_i = tl.program_id(1)                # tile over i (first seq dim)
    pid_j = tl.program_id(2)                # tile over j (second seq dim)
    pid_d = tl.program_id(3)                # tile over hidden dim

    # offsets for this tile
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    d_base = pid_d * BLOCK_D

    # range indices
    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)
    offs_d = d_base + tl.arange(0, BLOCK_D)

    # masks for out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # accumulate in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # loop over k dimension in chunks of BLOCK_K
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left[b, i, k, d] -----
        # shape (BLOCK_I, BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr + \
            pid_b * stride_lb + \
            offs_i[:, None, None] * stride_lk + \
            offs_k[None, :, None] * stride_ld + \
            offs_d[None, None, :] * 1
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                       other=0.0).to(tl.float16)

        # ----- load right[b, j, k, d] -----
        # shape (BLOCK_J, BLOCK_K, BLOCK_D)
        right_ptrs = right_ptr + \
            pid_b * stride_rb + \
            offs_j[:, None, None] * stride_rj + \
            offs_k[None, :, None] * stride_rd + \
            offs_d[None, None, :] * 1
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                        other=0.0).to(tl.float16)

        # transpose right to (BLOCK_K, BLOCK_J, BLOCK_D) for dot product
        # we perform a dot over the K dimension for each d slice
        for d in range(0, BLOCK_D):
            if not mask_d[d]:
                continue
            # left[:, :, d]   : (BLOCK_I, BLOCK_K)
            # right[:, :, d]  : (BLOCK_J, BLOCK_K) -> need (BLOCK_K, BLOCK_J)
            l = left[:, :, d]            # fp16
            r = right[:, :, d]           # fp16
            # dot: (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J) -> (BLOCK_I, BLOCK_J)
            acc[:, :, d] += tl.dot(l, r, out_dtype=tl.float32)

    # ----- store result -----
    out_ptrs = out_ptr + \
        pid_b * stride_ob + \
        offs_i[:, None, None] * stride_oi + \
        offs_j[None, :, None] * stride_oj + \
        offs_d[None, None, :] * 1
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def triton_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out = Σ_k left[..., i, k, d] * right[..., j, k, d]
    using the Triton kernel above.
    Shapes: left/right   [B, N, N, D]   (float16)
    Returns: out           [B, N, N, D]   (float32)
    """
    B, N, _, D = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides (in elements)
    stride_lb, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj = out.stride()

    # tile sizes – chosen to fit H200 shared memory / registers
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_D = 32        # hidden dim typically <= 384
    BLOCK_K = 64        # good occupancy for N up to 1024

    grid = (
        B,
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        triton.cdiv(D, BLOCK_D),
    )
    trimul_einsum_kernel[grid](
        left,
        right,
        out,
        B, N, D,
        stride_lb, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rd,
        stride_ob, stride_oi, stride_oj,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_D=BLOCK_D,
        BLOCK_K=BLOCK_K,
    )
    return out


# -------------------------------------------------------------------------
# Full custom kernel entry point
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------- 1. LayerNorm on the last dimension ---------------------------------
    # manual LayerNorm using provided weight/bias (no bias for the inner LN)
    # input shape: [B,N,N,dim]
    eps = 1e-5
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)

    # Compute mean / var over dim (last axis)
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b          # shape [B,N,N,dim]

    # ------- 2. Linear projections (FP16 for speed) -----------------------------
    x_fp16 = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)   # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = torch.einsum('bnkd,hd->bnkh', x_fp16, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.einsum('bnkd,hd->bnkh', x_fp16, right_proj_w)

    # ------- 3. Apply mask ------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)            # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------- 4. Gates (sigmoid) -------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(torch.einsum('bnkd,hd->bnkh', x_fp16, left_gate_w))
    right_gate = torch.sigmoid(torch.einsum('bnkd,hd->bnkh', x_fp16, right_gate_w))
    out_gate = torch.sigmoid(torch.einsum('bnkd,hd->bnkh', x_fp16, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------- 5. Core einsum (fused kernel) --------------------------------------
    # left/right are [B,N,N,hidden_dim] in FP16
    out = triton_einsum(left, right)          # result FP32, shape [B,N,N,hidden_dim]

    # ------- 6. Output LayerNorm + gating ---------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    # LN over hidden_dim
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # gating (broadcast)
    out_norm = out_norm * out_gate.to(torch.float32)

    # ------- 7. Final linear projection -----------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)   # [dim, hidden_dim]
    out_fp16 = out_norm.to(torch.float16)
    final = torch.einsum('bnkh,dh->bndh', out_fp16, to_out_w)          # [B,N,N,dim]
    final = final.to(torch.float32)

    return final
