"""
TriMul outgoing kernel (AlphaFold3)

We compute
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right have already been projected, masked and gated.
The heavy O(N³·H) part is fused in Triton:
  * Block over i, j (BI,BJ) and over the reduction dimension k (BK).
  * Small tile over the hidden dimension (BH) to keep registers modest.
  * Inside the k‑tile we accumulate element‑wise products for each d.
All other operations (LayerNorm, Linear, gating, final norm & projection)
are performed in PyTorch for simplicity.
The kernel works for fp16 computation; the final result is cast back to fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: fused element‑wise product‑sum over k
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # shapes
    B, N, H,
    # strides (row‑major tensors)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # tile sizes (compile‑time constants)
    BI: tl.constexpr, BJ: tl.constexpr, BK: tl.constexpr, BH: tl.constexpr,
):
    # ----------------------------------------------------------
    #  program IDs – grid is (B, ceil(N/BI), ceil(N/BJ))
    # ----------------------------------------------------------
    b  = tl.program_id(0)                     # batch
    pid_i = tl.program_id(1)                  # block idx over i
    pid_j = tl.program_id(2)                  # block idx over j

    i0 = pid_i * BI
    j0 = pid_j * BJ

    # ----------------------------------------------------------
    #  masks for out‑of‑bounds threads
    # ----------------------------------------------------------
    off_i = i0 + tl.arange(0, BI)
    off_j = j0 + tl.arange(0, BJ)
    mask_i = off_i < N
    mask_j = off_j < N

    # accumulator: (BI, BJ, BH) in fp32
    acc = tl.zeros((BI, BJ, BH), dtype=tl.float32)

    # ----------------------------------------------------------
    #  loop over k‑tiles
    # ----------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BK)
    for tk in range(num_k_tiles):
        k0 = tk * BK
        off_k = k0 + tl.arange(0, BK)
        mask_k = off_k < N

        # ------------------------------------------------------------------
        #  load left[b, i, k, d]  → shape (BI, BK, BH)
        #  load right[b, j, k, d] → shape (BJ, BK, BH)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + off_i[:, None, None] * stride_left_i
            + off_k[None, :, None] * stride_left_k
            + tl.arange(0, BH)[None, None, :] * stride_left_h
        )
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + off_j[:, None, None] * stride_right_j
            + off_k[None, :, None] * stride_right_k
            + tl.arange(0, BH)[None, None, :] * stride_right_h
        )

        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & (tl.arange(0, BH)[None, None, :] < H),
            other=0.0,
            eviction_policy="evict_last",
        )   # (BI, BK, BH)  fp16
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & (tl.arange(0, BH)[None, None, :] < H),
            other=0.0,
            eviction_policy="evict_last",
        )   # (BJ, BK, BH)  fp16

        # ------------------------------------------------------------------
        #  accumulate over the reduction dimension k.
        #  we unroll the inner k‑loop (BK is small, e.g. 32)
        # ------------------------------------------------------------------
        for kk in range(BK):
            mask_kk = mask_k[kk]
            if mask_kk:
                l_vec = left[:, kk, :]          # (BI, BH)
                r_vec = right[:, kk, :]         # (BJ, BH)

                # broadcast to (BI, BJ, BH) and accumulate
                acc += tl.cast(l_vec[:, None, :], tl.float32) * tl.cast(r_vec[None, :, :], tl.float32)

    # ----------------------------------------------------------
    #  write result
    # ----------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + off_i[:, None, None] * stride_out_i
        + off_j[None, :, None] * stride_out_j
        + tl.arange(0, BH)[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & (tl.arange(0, BH)[None, None, :] < H),
    )


# --------------------------------------------------------------
# Host side entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            input  – torch.Tensor [B, N, N, dim]  (float32)
            mask   – torch.Tensor [B, N, N]       (float32 or bool)
            weights – dict of pre‑trained parameters
            config  – dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- linear / norm utilities ----------
    def linear(x, w, b=None):
        # x: [..., dim_in]   w: [dim_out, dim_in]
        return F.linear(x, w, b)

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- projections ----------
    left = linear(x, weights["left_proj.weight"])
    right = linear(x, weights["right_proj.weight"])

    if not nomask:
        mask_u = mask.unsqueeze(-1)                    # [B,N,N,1]
        left = left * mask_u
        right = right * mask_u

    # ---------- gates ----------
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    #  Triton kernel: compute out = Σ_k left[...,k,:] * right[...,k,:]
    # --------------------------------------------------------------
    # cast to fp16 for the heavy kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # strides (since tensors are contiguous, we can compute them directly)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_fp16.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_fp16.stride()

    # tile sizes – chosen to fit H200 SM resources
    BI, BJ, BK, BH = 32, 32, 32, 32   # BH must divide hidden_dim (we handle remainder via masks)

    grid = (B,
            triton.cdiv(N, BI),
            triton.cdiv(N, BJ))

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BI=BI, BJ=BJ, BK=BK, BH=BH,
    )

    # back to fp32
    out = out_fp16.to(torch.float32)

    # ---------- final norm, gating and projection ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = linear(out, weights["to_out.weight"])
    return out
