"""
TriMul “outgoing” implementation.

The heavy O(N³·H) contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is expressed as a batched matrix‑multiply per hidden channel:
    out_h = left_h @ right_h.T
where left_h and right_h are (B·N, N) matrices for each hidden dimension h.

A custom Triton kernel performs the per‑channel matmul, accumulating the
result in fp32.  All layer‑norms, linear projections and sigmoids are kept
in PyTorch (they are inexpensive compared with the N³ work).  The kernel
operates on contiguous tensors with shape
    left : (B, N, N, H)
    right: (B, N, N, H)
and writes the result into `out` with the same shape.
The final linear projection and output‑layer‑norm are also done in PyTorch.
The kernel is launched with a 2‑D grid (one block per (i‑block, j‑block))
and loops over the reduction dimension `k` in tiles.
"""

import torch
import triton
import triton.language as tl
from torch import nn

# ------------------------------------------------------------
# Triton kernel: per‑channel matmul (left @ rightᵀ)
# ------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # const *float32
    right_ptr,              # const *float32
    out_ptr,                # *float32
    B, N, H,                # batch size, seq len, hidden dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,  # tile size for i dimension
    BLOCK_N: tl.constexpr,  # tile size for j dimension
    BLOCK_K: tl.constexpr,  # tile size for reduction k
):
    pid_i = tl.program_id(0)      # i‑tile index
    pid_j = tl.program_id(1)      # j‑tile index
    pid_b = tl.program_id(2)      # batch index

    # ---- compute start offsets for this tile ---------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    b_off   = pid_b

    # ---- offsets within the tensors -------------------------------------------
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds tiles
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointer arithmetic helpers
    left_base  = left_ptr  + b_off * stride_left_b
    right_base = right_ptr + b_off * stride_right_b
    out_base   = out_ptr   + b_off * stride_out_b

    # ---- allocate accumulator: (BLOCK_M, BLOCK_N, H) -------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, H), dtype=tl.float32)

    # ---- loop over k ---------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        # load a tile of left: shape (BLOCK_M, BLOCK_K, H)
        left_ptrs = left_base \
            + (offs_i[:, None, None] * stride_left_i) \
            + (k_start + offs_k)[None, :, None] * stride_left_k \
            + tl.arange(0, H)[None, None, :] * stride_left_h
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None, None] & (k_start + offs_k)[None, :, None] < N, other=0.0)

        # load a tile of right: shape (BLOCK_N, BLOCK_K, H)
        right_ptrs = right_base \
            + (offs_j[:, None, None] * stride_right_j) \
            + (k_start + offs_k)[None, :, None] * stride_right_k \
            + tl.arange(0, H)[None, None, :] * stride_right_h
        right_tile = tl.load(right_ptrs, mask=mask_j[:, None, None] & (k_start + offs_k)[None, :, None] < N, other=0.0)

        # ---- core contraction --------------------------------------------------
        # left_tile[i, k, h] * right_tile[j, k, h] summed over k
        # we vectorise over the hidden dimension H
        acc += tl.sum(left_tile[:, :, None, :] * right_tile[None, :, :, :], axis=2)   # (M,N,H)

    # ---- write back -----------------------------------------------------------
    out_ptrs = out_base \
        + (offs_i[:, None, None] * stride_out_i) \
        + (offs_j[None, :, None] * stride_out_j) \
        + tl.arange(0, H)[None, None, :] * stride_out_h

    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None])


# ------------------------------------------------------------
# Helper: LayerNorm without torch.nn (faster in FP32)
# ------------------------------------------------------------
def fused_layernorm(x, weight, bias, eps=1e-5):
    # x: (..., dim) float32
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    norm = (x - mean) / torch.sqrt(var + eps)
    return norm * weight + bias


# ------------------------------------------------------------
# Entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D]  (float32)
        - mask         : torch.Tensor [B, N, N]   (float32, 0/1)
        - weights      : dict of torch.Parameter tensors
        - config       : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor : output of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim = D)
    # ------------------------------------------------------------------
    x = fused_layernorm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)   left & right : (B,N,N,H)
    # ------------------------------------------------------------------
    # shape: (..., D) @ (D, H)ᵀ  -> (..., H)
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if present) – mask is broadcasted on hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                # (B,N,N,1)
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel: per‑channel matmul left @ rightᵀ  -> out (B,N,N,H)
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # strides (in elements) for each tensor
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tile sizes – tuned for Hopper (SM90) 128‑thread warps
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B,                                 # batch dimension
    )

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output layer‑norm + gating + final linear projection back to D
    # ------------------------------------------------------------------
    out = fused_layernorm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
