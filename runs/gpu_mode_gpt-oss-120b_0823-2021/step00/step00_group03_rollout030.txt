"""
TriMul “outgoing” implementation.

The heavy part is the triple‑product
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is an Einstein‑summation
    '... i k d , ... j k d -> ... i j d'.
For a fixed hidden dimension *d* it is a matrix‑multiply
    (N×N) @ (N×N)^T  (the second operand is transposed on the second axis).
The kernel below computes this ternary product in a fused fashion, tiling over
the sequence dimension *N* and the hidden dimension *H*.
All other linear / layer‑norm / gating operations are done with cheap PyTorch
calls (they are memory‑bound and not the bottleneck).  The final output is
returned as float32.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fused triple‑product (the einsum part)
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    # pointers
    left_ptr,          #  [B,N,N,H]   (i,k,d)
    right_ptr,         #  [B,N,N,H]   (j,k,d)
    out_ptr,           #  [B,N,N,H]   (i,j,d)

    # shape / strides
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    # compile‑time tile sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute out[b,i,j,d] += Σ_k left[b,i,k,d] * right[b,j,k,d].
    Tiling:
        i → BLOCK_I, j → BLOCK_J, k → BLOCK_K, d → BLOCK_H
    """
    pid = tl.program_id(0)                     # linear index over (b, i, j)
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)

    b = pid // (num_i * num_j)
    tmp = pid % (num_i * num_j)
    i_tile = tmp // num_j
    j_tile = tmp % num_j

    # ---------- block offsets ----------
    i_offset = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)
    d_offset = tl.arange(0, BLOCK_H)

    # Masks (out‑of‑bounds rows / cols)
    i_mask = i_offset < N
    j_mask = j_offset < N
    d_mask = d_offset < H

    # Pointers to the beginning of the current (i,j) tile
    left_tile_ptr  = left_ptr  + b * stride_left_b \
                               + i_offset[:, None, None] * stride_left_i \
                               + 0          * stride_left_k \
                               + d_offset[None, None, :] * stride_left_h
    right_tile_ptr = right_ptr + b * stride_right_b \
                               + j_offset[:, None, None] * stride_right_j \
                               + 0          * stride_right_k \
                               + d_offset[None, None, :] * stride_right_h
    out_tile_ptr   = out_ptr   + b * stride_out_b \
                               + i_offset[:, None, None] * stride_out_i \
                               + j_offset[None, :, None] * stride_out_j \
                               + d_offset[None, None, :] * stride_out_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # load left: shape (I, K, H)
        left_ptrs = left_tile_ptr + k_offset[None, :, None] * stride_left_k
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :] & d_mask[None, None, :],
                       other=0.0).to(tl.float32)

        # load right: shape (J, K, H)
        right_ptrs = right_tile_ptr + k_offset[None, :, None] * stride_right_k
        right = tl.load(right_ptrs, mask=j_mask[:, None] & k_mask[None, :] & d_mask[None, None, :],
                        other=0.0).to(tl.float32)

        # we need right transposed on the K axis: (J, K, H) → (J, K, H) (already aligned)
        # multiply and reduce over K
        # broadcasting: left[I,K,H] * right[J,K,H] → (I,J,K,H) → sum over K
        # We perform a dot for each d slice using tl.dot
        for dh in range(0, BLOCK_H, 1):
            # slice one hidden dimension
            l = left[:, :, dh]          # (I, K)
            r = right[:, :, dh]         # (J, K)
            # compute (I, K) @ (J, K)^T => (I, J)
            acc[:, :, dh] += tl.dot(l, r, trans_b=True)   # (I,J)

    # write back
    tl.store(out_tile_ptr, acc.to(tl.float16), mask=i_mask[:, None, None] &
                                                j_mask[None, :, None] &
                                                d_mask[None, None, :])


# ----------------------------------------------------------------------
# Helper for launching the kernel
# ----------------------------------------------------------------------
def trimul_einsum_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B,N,N,H]  (float32)
    returns out    [B,N,N,H]  (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)

    # strides in elements (not bytes)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_H = 16   # hidden dim is a multiple of 16 in all configs

    grid = (
        triton.cdiv(B * N * N, BLOCK_I * BLOCK_J),   # one program per (b,i,j) tile
    )
    trimul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=H,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_h=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_j=stride_right[1],
        stride_right_k=stride_right[2],
        stride_right_h=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_h=stride_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,   # enough for fp16 dot
    )
    return out


# ----------------------------------------------------------------------
# Full TriMul forward (torch for everything else, triton for the einsum)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Args
    ----
    data : (input, mask, weights, config)
        input : [B, N, N, dim]   (float32, on CUDA)
        mask  : [B, N, N]        (float32 or bool)
        weights: dict of all linear / LN parameters
        config : {"dim": int, "hidden_dim": int}
    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ---------- LayerNorm (dim) ----------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # torch.nn.functional.layer_norm works on the last dimension only
    x = nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B,N,N,dim]

    # ---------- Linear projections (no bias) ----------
    # shape: (..., dim) @ (dim, hidden_dim)^T
    left_proj_w  = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]

    # fuse the linear with a view to keep memory contiguous
    x_flat = x.reshape(-1, dim)                       # (B*N*N, dim)
    left = torch.nn.functional.linear(x_flat, left_proj_w).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, right_proj_w).view(B, N, N, hidden_dim)

    # ---------- Apply mask ----------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                    # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ---------- Gates (sigmoid) ----------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x_flat, left_gate_w).view(B, N, N, hidden_dim).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, right_gate_w).view(B, N, N, hidden_dim).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, out_gate_w).view(B, N, N, hidden_dim).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Core einsum via Triton ----------
    # Use fp16 inside the kernel for higher throughput, cast back to fp32
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = trimul_einsum_triton(left_fp16, right_fp16)   # [B,N,N,hidden_dim] fp32 (kernel casts back)
    out = out_fp16.to(torch.float32)

    # ---------- Output norm & linear ----------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate

    to_out_w = weights["to_out.weight"]        # (dim, hidden_dim)
    out_flat = out.reshape(-1, hidden_dim)
    out = torch.nn.functional.linear(out_flat, to_out_w).view(B, N, N, dim)

    return out
