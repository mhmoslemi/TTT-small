"""
TriMul "outgoing" implementation (forward only).

The heavy part is the N‑by‑N pairwise multiplication over the
intermediate dimension `k`:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape [B, N, N, H] (H = hidden_dim).
The kernel computes this reduction block‑wise over i and j while
iterating over k.  All LayerNorm / Linear / gating operations are
performed with PyTorch (they are cheap compared with the O(N³) work).
The kernel works in mixed precision (fp16 for the main tensors,
fp32 accumulation) and returns fp32 output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: pairwise reduction over the `k` dimension
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,            # const float16 *
    right_ptr,           # const float16 *
    out_ptr,             # float32 *
    B, N, H,             # sizes (int32)
    # strides for left  (B,N,K,H)
    stride_lb, stride_li, stride_lk, stride_lh,
    # strides for right (B,N,K,H) – note that the second N is the *j* axis
    stride_rb, stride_rj, stride_rk, stride_rh,
    # strides for out   (B,N,N,H)
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]"""
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Compute block coordinates (b, i0, j0)
    # --------------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    total_blocks = B * num_i * num_j

    b = pid // (num_i * num_j)
    rem = pid % (num_i * num_j)
    i0 = (rem // num_j) * BLOCK_I
    j0 = (rem % num_j) * BLOCK_J

    # --------------------------------------------------------------
    # Offsets for the i, j and k axes
    # --------------------------------------------------------------
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, H), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over k
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < N

        # left[b,i,k,:]   shape (BLOCK_I, H)
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + offs_i[:, None] * stride_li
            + cur_k[None, :] * stride_lk
            + tl.arange(0, H)[None, None, :] * stride_lh
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right[b,j,k,:]  shape (BLOCK_J, H)
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + offs_j[:, None] * stride_rj
            + cur_k[None, :] * stride_rk
            + tl.arange(0, H)[None, None, :] * stride_rh
        )
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # broadcast to (BLOCK_I, BLOCK_J, H) and accumulate
        # left[:,None,:]  -> (BLOCK_I, 1, H)
        # right[None,:,:] -> (1, BLOCK_J, H)
        left_exp  = tl.broadcast_to(left[:, None, :],  (BLOCK_I, BLOCK_J, H))
        right_exp = tl.broadcast_to(right[None, :, :], (BLOCK_I, BLOCK_J, H))
        acc += left_exp * right_exp

    # --------------------------------------------------------------
    # Store result
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + offs_i[:, None, None] * stride_oi
        + offs_j[None, :, None] * stride_oj
        + tl.arange(0, H)[None, None, :] * stride_oh
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None])


# ----------------------------------------------------------------------
# Helper: LayerNorm that works on a 4‑D tensor (B,N,N,C)
# ----------------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    # x: (B,N,N,C)   weight,bias: (C,)
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim]      (float32)
            mask         : torch.Tensor [B, N, N]           (bool/float)
            weights      : dict of torch.Tensors (all float32)
            config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias) – use fp16 for speed
    # ------------------------------------------------------------------
    x16 = x.to(torch.float16)

    left = F.linear(x16, weights["left_proj.weight"]).to(torch.float16)   # (B,N,N,hidden)
    right = F.linear(x16, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣  Apply mask if present
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣  Gating (sigmoid) – keep in fp32 for numerical stability
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))  # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate.to(torch.float16)
    right = right * right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣  Core TriMul contraction via Triton
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=x.device)

    # strides (in elements) for the 4‑D tensors
    left_stride = (
        left.stride(0),  # B
        left.stride(1),  # i
        left.stride(2),  # k
        left.stride(3),  # h
    )
    right_stride = (
        right.stride(0),  # B
        right.stride(1),  # j  (second N)
        right.stride(2),  # k
        right.stride(3),  # h
    )
    out_stride = (
        out.stride(0),   # B
        out.stride(1),   # i
        out.stride(2),   # j
        out.stride(3),   # h
    )

    # Block sizes – tuned for Hopper (sm_90)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    total_blocks = B * triton.cdiv(N, BLOCK_I) * triton.cdiv(N, BLOCK_J)
    grid = (total_blocks,)

    trimul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_lb=left_stride[0],
        stride_li=left_stride[1],
        stride_lk=left_stride[2],
        stride_lh=left_stride[3],
        stride_rb=right_stride[0],
        stride_rj=right_stride[1],
        stride_rk=right_stride[2],
        stride_rh=right_stride[3],
        stride_ob=out_stride[0],
        stride_oi=out_stride[1],
        stride_oj=out_stride[2],
        stride_oh=out_stride[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # good default for H200
    )

    # ------------------------------------------------------------------
    # 6️⃣  to_out_norm (LayerNorm on hidden dim) → out_gate → final linear
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.unsqueeze(-1)               # (B,N,N,hidden)

    out = F.linear(out, weights["to_out.weight"])    # back to dim
    return out.to(torch.float32)
