"""
TriMul “outgoing” implementation.

The expensive part of the forward pass is the
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a reduction over the sequence dimension k for every
(i,j) pair and every hidden channel d.  This kernel computes the
reduction in CUDA with Triton, blocking over the k‑dimension and the
hidden dimension to keep the register pressure low while fully
utilising the Hopper tensor cores (fp16→fp32 accumulation).

All surrounding operations (LayerNorms, linear projections,
gates, final LayerNorm and linear) are performed in PyTorch so that
the host‑side code stays simple and the kernel focuses on the
N³‑like component.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: reduction over k for a tile (i,j) and a slice of d
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,               #  *[B, N, N, H]   left projected & gated
    right_ptr,              #  *[B, N, N, H]   right projected & gated
    out_ptr,                #  *[B, N, N, H]   result before to_out_norm
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,   # strides for left
    stride_rb, stride_rj, stride_rk, stride_rd,   # strides for right
    stride_ob, stride_oi, stride_oj, stride_od,   # strides for out
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Compute a block of the output tensor.
    Each program instance is responsible for one (b, i, j) tile and a
    contiguous slice of the hidden dimension d of size BLOCK_D.
    The reduction over k is performed in a loop with a tile size BLOCK_K.
    """

    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # i index tile
    pid_j = tl.program_id(2)          # j index tile
    pid_d = tl.program_id(3)          # d slice tile

    # ----- compute global offsets -----
    i_offset = pid_i * 1               # one i per program (BLOCK_I = 1)
    j_offset = pid_j * 1               # one j per program (BLOCK_J = 1)
    d_offset = pid_d * BLOCK_D

    # masks for out‑of‑bounds
    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_d = (d_offset + tl.arange(0, BLOCK_D)) < H

    # pointers to the first element of this tile
    left_tile_ptr  = left_ptr  + pid_b * stride_lb \
                                   + i_offset * stride_li \
                                   + 0 * stride_lk \
                                   + d_offset * stride_ld
    right_tile_ptr = right_ptr + pid_b * stride_rb \
                                   + j_offset * stride_rj \
                                   + 0 * stride_rk \
                                   + d_offset * stride_rd
    out_tile_ptr   = out_ptr   + pid_b * stride_ob \
                                   + i_offset * stride_oi \
                                   + j_offset * stride_oj \
                                   + d_offset * stride_od

    # accumulator in fp32
    acc = tl.zeros([BLOCK_D], dtype=tl.float32)

    # ----- k‑reduction loop -----
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = tl.minimum(N - k_offset, BLOCK_K)          # actual tile size

        # load a (cur_k, BLOCK_D) block from left and right
        left = tl.load(
            left_tile_ptr
            + k_offset * stride_lk
            + tl.arange(0, cur_k)[:, None] * stride_lk
            + tl.arange(0, BLOCK_D)[None, :] * stride_ld,
            mask=(tl.arange(0, cur_k)[:, None] < cur_k) & mask_d[None, :],
            other=0.0,
        )
        right = tl.load(
            right_tile_ptr
            + k_offset * stride_rk
            + tl.arange(0, cur_k)[:, None] * stride_rk
            + tl.arange(0, BLOCK_D)[None, :] * stride_rd,
            mask=(tl.arange(0, cur_k)[:, None] < cur_k) & mask_d[None, :],
            other=0.0,
        )

        # element‑wise product and reduction over k
        prod = left * right                     # fp16 * fp16 -> fp16
        acc += tl.sum(prod, axis=0).to(tl.float32)

    # ----- write result -----
    tl.store(
        out_tile_ptr + tl.arange(0, BLOCK_D) * stride_od,
        acc,
        mask=mask_i & mask_j & mask_d,
    )


# ----------------------------------------------------------------------
# Host‑side helper to launch the kernel
# ----------------------------------------------------------------------
def launch_trimul_einsum(left: torch.Tensor,
                         right: torch.Tensor,
                         out: torch.Tensor,
                         BLOCK_K: int = 64,
                         BLOCK_D: int = 32):
    """
    left, right, out : [B, N, N, H]  (fp16)
    """
    B, N, _, H = left.shape
    # strides in elements (not bytes)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    grid = (
        B,
        triton.cdiv(N, 1),          # BLOCK_I = 1
        triton.cdiv(N, 1),          # BLOCK_J = 1
        triton.cdiv(H, BLOCK_D),    # d‑tiles
    )
    trimul_einsum_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Main entry point – must be named `custom_kernel`
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    The heavy triple‑product reduction is executed in the Triton kernel
    `trimul_einsum_kernel`.  All other linear / LayerNorm / gating
    operations stay in PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) -> hidden_dim
    # ------------------------------------------------------------------
    # shape: [B, N, N, hidden_dim]
    left = F.linear(
        x,
        weights["left_proj.weight"],
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"],
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiply
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"], bias=None))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"], bias=None))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"], bias=None))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to fp16 for the reduction kernel (better throughput)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output buffer in fp16
    out_fp16 = torch.empty_like(left_fp16, device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 6. Triton kernel: compute Σ_k left * right
    # ------------------------------------------------------------------
    launch_trimul_einsum(left_fp16, right_fp16, out_fp16,
                         BLOCK_K=64, BLOCK_D=32)

    # back to fp32 for the rest of the pipeline
    out = out_fp16.to(dtype)

    # ------------------------------------------------------------------
    # 7. to_out_norm (LayerNorm over hidden_dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 8. Apply out_gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"], bias=None)

    return out
