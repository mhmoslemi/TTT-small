"""
TriMul (outgoing) forward pass with a fused Triton kernel for the
core N‑N‑H hidden‑dimensional multiplication:

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

All LayerNorms and Linear projections are performed in PyTorch,
while the O(N³·H) multiplication is executed on GPU via Triton.
The final output is cast back to float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused multiplication over the “k” dimension.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # [B, N, N, H] left tensor
    right_ptr,              # [B, N, N, H] right tensor
    out_ptr,                # [B, N, N, H] output tensor
    B, N, H,                # dimensions
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_M: tl.constexpr,  # tile size for i dimension
    BLOCK_N: tl.constexpr,  # tile size for j dimension
    BLOCK_K: tl.constexpr,  # tile size for k reduction
    BLOCK_H: tl.constexpr,  # tile size for hidden dim
):
    """Compute out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]"""
    pid_b = tl.program_id(0)               # batch index
    pid_i = tl.program_id(1)               # i‑tile index
    pid_j = tl.program_id(2)               # j‑tile index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, BLOCK_H)

    # Masks for out‑of‑bounds tiles
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # ------------------------------------------------------------------
    # Pointers at the beginning of the tile (batch fixed)
    # ------------------------------------------------------------------
    left_base  = left_ptr  + pid_b * stride_lb
    right_base = right_ptr + pid_b * stride_rb
    out_base   = out_ptr   + pid_b * stride_ob

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduction over k
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load left[b, i, k, h]
        left_ptrs = (left_base
                     + offs_i[:, None, None] * stride_li
                     + cur_k[None, :, None] * stride_lk
                     + offs_h[None, None, :] * stride_ld)
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0)

        # Load right[b, j, k, h]
        right_ptrs = (right_base
                      + offs_j[:, None, None] * stride_rj
                      + cur_k[None, :, None] * stride_rk
                      + offs_h[None, None, :] * stride_rd)
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0)

        # Compute dot product over the k‑axis for each (i,j,h) pair.
        # We use an explicit multiplication‑accumulation because tl.dot only
        # reduces over the last dimension.
        acc += tl.dot(left, right)   # shape (BLOCK_M, BLOCK_N, BLOCK_H)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_ptrs = (out_base
                + offs_i[:, None, None] * stride_oi
                + offs_j[None, :, None] * stride_oj
                + offs_h[None, None, :] * stride_od)
    tl.store(out_ptrs,
             acc.to(tl.float16),   # store as fp16 to save bandwidth
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

# ----------------------------------------------------------------------
# Host‑side helper that launches the Triton kernel
# ----------------------------------------------------------------------
def tri_mul_fused(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : torch tensors of shape [B, N, N, H] (float32)
    returns      : torch tensor of same shape (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)

    # strides in elements (not bytes)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # Tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64
    BLOCK_H = 32   # hidden_dim is a multiple of 32 in all configs

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul operator.
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   torch.float32 (or fp16)
            - mask         : [B, N, N]        torch.float32/bool
            - weights      : dict of torch tensors (see problem statement)
            - config       : dict with keys "dim" and "hidden_dim"
    Returns:
        out : torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dim (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim], float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) to hidden_dim
    # ------------------------------------------------------------------
    # shape transform: (..., dim) @ (hidden_dim, dim).T
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul via Triton (compute in fp16 for speed, accumulate fp32)
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (kernel internally uses fp16 loads)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # fused multiplication, result is float32
    out = tri_mul_fused(left_h, right_h)          # [B,N,N,hidden_dim] float32

    # ------------------------------------------------------------------
    # 6) to_out_norm + gating + final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to dim

    return out.to(torch.float32)
