"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1.  Layer‑norm the input (dim = C).
2.  Linear projections → left / right (C → H) and three sigmoid gates
    (C → H).  The gates and an optional mask are applied in PyTorch – they
    are cheap compared with the N³ work.
3.  The core operation is
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   i.e. a batched matrix‑multiply for each hidden dimension *d*.
   We reshape the tensor as (B·H, N, N) and launch a single Triton kernel
   that computes many (i,j) blocks while looping over *k* tiles.
   The kernel accumulates in fp32 and writes fp16 (later cast back to fp32).
4.  Apply a second layer‑norm, the output gate and a final linear projection
    (H → C) – also done in PyTorch.

The heavy N³ contraction is therefore executed fully on the GPU with a
custom Triton kernel, while the surrounding lightweight ops stay in
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------------------- #
# Triton kernel: batched mat‑mul for the “einsum” part of TriMul
# --------------------------------------------------------------------------- #
@triton.jit
def _trilu_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                 # pointers
    B, N, H,                             # problem size
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_k, stride_b_j, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes, for every (b, h):
        C[b,h,i,j] = Σ_k A[b,h,i,k] * B[b,h,k,j]
    where A,B have shape (B, N, N, H) and are interpreted as (B·H, N, N).
    """
    pid = tl.program_id(0)

    # total number of (batch, head) pairs
    total_bh = B * H
    # compute which (b, h) this pid works on
    bh_idx = pid // (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    i_block = (pid // tl.cdiv(N, BLOCK_N)) % tl.cdiv(N, BLOCK_M)
    j_block = pid % tl.cdiv(N, BLOCK_N)

    b_idx = bh_idx // H
    h_idx = bh_idx % H

    # block offsets
    i_offset = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_k = k_offset < N

    # pointers to the start of the (b,h) slice
    a_base = a_ptr + b_idx * stride_a_batch + h_idx * stride_a_h
    b_base = b_ptr + b_idx * stride_b_batch + h_idx * stride_b_h
    c_base = c_ptr + b_idx * stride_c_batch + h_idx * stride_c_h

    # compute strided pointers for the tile
    a_ptrs = a_base + i_offset[:, None] * stride_a_i + k_offset[None, :] * stride_a_k
    b_ptrs = b_base + k_offset[:, None] * stride_b_k + j_offset[None, :] * stride_b_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = start_k * BLOCK_K + k_offset
        cur_mask_k = cur_k < N

        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & cur_mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=cur_mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)               # (M,K)·(K,N) → (M,N)

        # advance pointers to the next k‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # write out
    c_ptrs = c_base + i_offset[:, None] * stride_c_i + j_offset[None, :] * stride_c_j
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

# --------------------------------------------------------------------------- #
# Host side helper that launches the kernel
# --------------------------------------------------------------------------- #
def _trilu_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, N, N, H)  fp32
    returns        (B, N, N, H)  fp32
    """
    B, N, _, H = left.shape
    # output allocation
    out = torch.empty_like(left, dtype=torch.float16, device=left.device)

    # strides (in elements, not bytes)
    stride_a = left.stride()
    stride_b = right.stride()
    stride_c = out.stride()

    # block sizes (tuned for Hopper)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(B * H, 1)
        * triton.cdiv(N, BLOCK_M)
        * triton.cdiv(N, BLOCK_N),
    )
    _trilu_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_a[0],
        stride_a[1],
        stride_a[2],
        stride_a[3],
        stride_b[0],
        stride_b[2],
        stride_b[1],
        stride_b[3],
        stride_c[0],
        stride_c[1],
        stride_c[2],
        stride_c[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # cast back to fp32 for the remaining layers
    return out.to(torch.float32)

# --------------------------------------------------------------------------- #
# The public entry point required by the grader
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward‑only TriMul “outgoing” operator.

    data = (input_tensor, mask, weights, config)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1. Layer‑norm on the input (C → C)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"].to(dtype))      # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # ------------------------------------------------------------------- #
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask = mask.unsqueeze(-1)               # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid) – also cheap, keep in PyTorch
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(dtype))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(dtype))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(dtype))
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Core N³ contraction via Triton
    # ------------------------------------------------------------------- #
    out = _trilu_matmul(left, right)               # (B,N,N,H) fp32

    # ------------------------------------------------------------------- #
    # 6. Second layer‑norm, output gate and final projection (H → C)
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
