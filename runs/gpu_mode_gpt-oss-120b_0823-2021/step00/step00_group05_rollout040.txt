"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a per‑channel matrix multiplication
    out_d = left_d  @  (right_d).T
where left_d/right_d have shape (N,N) for each hidden dimension d.
The kernel below launches one program for each (batch, hidden_dim) pair
and computes this GEMM with Triton.  All linear projections, LayerNorms,
gates and the final linear projection are kept in PyTorch (they are cheap
relative to the N³ reduction).

The kernel works for arbitrary batch size (≤2), sequence length N (≤1024)
and hidden dimension H (up to 384).  The final output is returned as
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton GEMM kernel: computes out = A @ B.T  (A,B shape = [N,N])
# One program instance handles a single (batch, hidden_dim) pair.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, out_ptr,               # pointers
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d,   # A strides
    stride_b_batch, stride_b_j, stride_b_k, stride_b_d,   # B strides
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d,   # O strides
    N,                                   # sequence length
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)                       # idx over (batch * hidden)
    batch = pid // tl.constexpr(tl.load(tl.constexpr(0)))  # dummy to force compile
    # The actual batch and hidden indices are computed on the host and passed
    # as offsets in a_ptr/b_ptr/out_ptr (see launch code below).

    # ------------------------------------------------------------------
    # 2‑D block indices within the (N,N) matrix
    # ------------------------------------------------------------------
    pid_m = tl.program_id(1)                     # block row
    pid_n = tl.program_id(2)                     # block col

    # Offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks to avoid OOB accesses
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block [M,K]
        a = tl.load(
            a_ptr
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
            + 0,                     # batch & hidden dim already folded into ptr
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B block [N,K] (note we need B.T, so we load [K,N] and later use transposed indices)
        b = tl.load(
            b_ptr
            + offs_n[:, None] * stride_b_j
            + offs_k[None, :] * stride_b_k,
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Compute dot product over K
        acc += tl.dot(a, b)

    # Store result
    out = acc.to(tl.float16)   # store as fp16 to reduce traffic, convert later
    tl.store(
        out_ptr
        + offs_m[:, None] * stride_o_i
        + offs_n[None, :] * stride_o_j,
        out,
        mask=mask_m[:, None] & mask_n[None, :],
    )

# ----------------------------------------------------------------------
# Helper to launch the kernel for all (batch, hidden) pairs
# ----------------------------------------------------------------------
def _launch_trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: [B, N, N, H]  (float32)
    returns out:   [B, N, N, H]  (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Block sizes – chosen to fit H200 SM and to give good occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute strides (contiguous layout)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d = left.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_d = right.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # Grid: (batch*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    total_prog = B * H
    grid = (
        total_prog,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Launch: we fold batch and hidden index into the base pointer:
    #   a_ptr = left_ptr + (b*stride_a_batch + h*stride_a_d)
    # Same for b_ptr and out_ptr.
    a_ptr = left
    b_ptr = right
    out_ptr = out

    _trimul_gemm_kernel[grid](
        a_ptr,
        b_ptr,
        out_ptr,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_a_d,
        stride_b_batch,
        stride_b_j,
        stride_b_k,
        stride_b_d,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        stride_o_d,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Convert back to fp32 (the kernel stored fp16)
    return out.to(torch.float32)


# ----------------------------------------------------------------------
# Main entry point required by the task
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            input  – Tensor[B,N,N,dim]  (float32)
            mask   – Tensor[B,N,N]      (float32 or bool)
            weights – dict of weight tensors
            config  – dict with keys dim, hidden_dim

    Returns:
        Tensor[B,N,N,dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ---------- LayerNorm (dim) ----------
    # manual LayerNorm to avoid extra kernels
    eps = 1e-5
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # reshape for broadcasting: [1,1,1,dim]
    x = input_tensor
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b

    # ---------- Linear projections ----------
    # shape: (..., dim) @ (dim, hidden_dim).T  => (..., hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)            # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Core einsum via Triton ----------
    out = _launch_trimul_gemm(left, right)   # [B,N,N,hidden_dim]

    # ---------- Output LayerNorm (hidden_dim) ----------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    # LayerNorm over hidden_dim
    mean = out.mean(-1, keepdim=True)
    var = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean) / torch.sqrt(var + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ---------- Apply out gate ----------
    out = out * out_gate

    # ---------- Final linear to original dim ----------
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
