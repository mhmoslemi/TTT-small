"""
TriMul “outgoing” kernel (AlphaFold3)

The heavy part of TriMul is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix multiplication per hidden dimension
(`d`).  The implementation below

* applies LayerNorm, Linear + gating in PyTorch (these are cheap),
* launches a custom Triton kernel that computes the above sum for
  all (b,i,j,d) with block‑wise tiling (BLOCK_M × BLOCK_N × BLOCK_K).
* finally applies the output LayerNorm and Linear in PyTorch.

The kernel works for arbitrary batch size (1‑2), sequence length up to 1024
and hidden dimensions that are a multiple of 32 (the configurations used in
the tests).  All tensors are expected to be on CUDA and the final result is
float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ---------------------------------------------------------------------------
# Triton kernel: batched matmul over the “k” dimension, keeping the hidden dim.
# ---------------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    out_ptr,                     # 4‑D output  (B,N,N,HD)
    left_ptr,                    # 4‑D left   (B,N,N,HD)
    right_ptr,                   # 4‑D right  (B,N,N,HD)
    B, N, HD,                    # problem sizes
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    stride_l_b,  stride_l_i,  stride_l_k,  stride_l_d,
    stride_r_b,  stride_r_j,  stride_r_k,  stride_r_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""

    pid = tl.program_id(0)

    # --------------------------------------------------------------
    #  Compute the 3‑D grid: (batch, i‑block, j‑block)
    # --------------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    b = pid // (num_i * num_j)
    tmp = pid % (num_i * num_j)
    i_block = tmp // num_j
    j_block = tmp %  num_j

    # --------------------------------------------------------------
    #  Offsets for the current tile
    # --------------------------------------------------------------
    i_base = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = j_block * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds for i/j
    i_mask = i_base < N
    j_mask = j_base < N

    # pointers to the output tile
    out_ptrs = out_ptr + (
        b * stride_out_b +
        i_base[:, None] * stride_out_i +
        j_base[None, :] * stride_out_j
    )  # (BM, BN)

    # initialise accumulator (float32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N, HD), dtype=tl.float32)

    # --------------------------------------------------------------
    #  Loop over k in BLOCK_K steps
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_base < N

        # Load left[b, i, k, d]   -> (BM, BK, HD)
        left_ptrs = left_ptr + (
            b * stride_l_b +
            i_base[:, None] * stride_l_i +
            k_base[None, :] * stride_l_k
        )
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        # shape (BM, BK, HD)

        # Load right[b, j, k, d]  -> (BN, BK, HD)
        right_ptrs = right_ptr + (
            b * stride_r_b +
            j_base[:, None] * stride_r_j +
            k_base[None, :] * stride_r_k
        )
        right = tl.load(right_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)
        # shape (BN, BK, HD)

        # transpose right to (BK, BN, HD) so we can do dot over k
        right = tl.trans(right, (1, 0, 2))

        # Compute dot product over the K dimension for every hidden d
        # acc[i,j,d] += Σ_k left[i,k,d] * right[j,k,d]
        # Use tl.dot which does reduction over the second axis of left
        # and first axis of right.
        # We have to loop over d because tl.dot works on 2‑D matrices.
        for d in range(0, HD, 1):
            # left[:, :, d] : (BM, BK)
            # right[:, :, d] : (BK, BN)
            acc[:, :, d] += tl.dot(
                left[:, :, d],           # (BM, BK)
                right[:, :, d]           # (BK, BN)
            )

    # --------------------------------------------------------------
    #  Store the result (convert back to fp16 for memory bandwidth)
    # --------------------------------------------------------------
    # Cast to the same dtype as the input (we assume fp16 storage)
    out = acc.to(tl.float16)

    tl.store(out_ptrs, out, mask=i_mask[:, None] & j_mask[None, :])


# ---------------------------------------------------------------------------
# Helper to launch the kernel
# ---------------------------------------------------------------------------
def trimul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : (B, N, N, HD)  (float16)
    returns     : (B, N, N, HD)  (float16)
    """
    B, N, _, HD = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes)
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()
    stride_l_b,  stride_l_i,  stride_l_k,  stride_l_d  = left.stride()
    stride_r_b,  stride_r_j,  stride_r_k,  stride_r_d  = right.stride()

    # tile sizes – tuned for Hopper (sm_90)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (B * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),)

    trimul_einsum_kernel[grid](
        out,
        left,
        right,
        B, N, HD,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        stride_l_b,  stride_l_i,  stride_l_k,  stride_l_d,
        stride_r_b,  stride_r_j,  stride_r_k,  stride_r_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # Hopper can handle 4‑warp blocks efficiently
    )
    return out


# ---------------------------------------------------------------------------
# Full custom kernel entry point
# ---------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    TriMul “outgoing” forward pass with a fused Triton kernel for the
    N³ reduction.  All other operations (LayerNorm, Linear, gating) are
    performed in PyTorch for simplicity and numerical stability.

    Args:
        data: (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]   (float32)
            - mask  : torch.Tensor [B, N, N]        (float32 or bool)
            - weights: dict with the model parameters (float32)
            - config : {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """

    # -------------------------------------------------------------------
    # unpack
    # -------------------------------------------------------------------
    x, mask, w, cfg = data
    dim, hidden = cfg["dim"], cfg["hidden_dim"]          # dim = C, hidden = C_z

    # -------------------------------------------------------------------
    # 1) LayerNorm over the feature dimension C
    # -------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=1e-5,
    )  # (B,N,N,dim)   float32

    # -------------------------------------------------------------------
    # 2) Linear projections (no bias) - we operate in fp16 for speed
    # -------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left = F.linear(x_fp16, w["left_proj.weight"].to(torch.float16))   # (B,N,N,hidden)
    right = F.linear(x_fp16, w["right_proj.weight"].to(torch.float16))

    # -------------------------------------------------------------------
    # 3) Masking (if required)
    # -------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(torch.float16)        # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # -------------------------------------------------------------------
    # 4) Gating
    # -------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_fp16, w["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x_fp16, w["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x_fp16, w["out_gate.weight"].to(torch.float16))
    )

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------
    # 5) Core N³ reduction – Triton kernel
    # -------------------------------------------------------------------
    # The kernel expects fp16 input and returns fp16 output
    out = trimul_einsum(left, right)   # (B,N,N,hidden) fp16

    # -------------------------------------------------------------------
    # 6) Output LayerNorm + gating + final linear (back to dim)
    # -------------------------------------------------------------------
    out = out.to(torch.float32)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.to(torch.float32)          # broadcast over hidden dim

    out = F.linear(out, w["to_out.weight"])        # (B,N,N,dim)

    return out
