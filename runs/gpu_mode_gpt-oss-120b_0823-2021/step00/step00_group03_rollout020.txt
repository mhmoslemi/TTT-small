"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the 4‑D Einstein‑sum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each (b,d) this is exactly a matrix‑multiply
    L_{b,d} (N×N)  @  R_{b,d}ᵀ (N×N)  →  O_{b,d} (N×N)

The kernel below performs a batched GEMM over the (b,d) dimension using
Triton.  All other operations (LayerNorm, linear projections, gates,
masking and the final linear + LayerNorm) are kept in PyTorch – they are
already highly‑optimized and do not dominate the runtime.

The kernel works with FP16 intermediate arithmetic for speed, but the
final result is cast back to FP32 as required by the specification.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply L @ Rᵀ over the (batch, hidden) axis
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr,                     # *float16   (B, H, N, N)
    right_ptr,                    # *float16   (B, H, N, N)
    out_ptr,                      # *float32   (B, H, N, N)
    B, N, H,
    stride_l_bh, stride_l_n, stride_l_k,
    stride_r_bh, stride_r_n, stride_r_k,
    stride_o_bh, stride_o_n, stride_o_k,
    BLOCK_M: tl.constexpr,        # rows of the output tile
    BLOCK_N: tl.constexpr,        # cols of the output tile
    BLOCK_K: tl.constexpr,        # reduction tile
):
    """
    Compute a tile of O[b,h,i,j] = Σ_k L[b,h,i,k] * R[b,h,j,k]
    (note the transposition on the second operand).
    """
    pid_bh = tl.program_id(0)               # combined batch‑hidden id
    pid_m = tl.program_id(1)                # row tile id
    pid_n = tl.program_id(2)                # col tile id

    # decode batch and hidden indices
    b = pid_bh // H
    h = pid_bh %  H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointer base for this (b,h) slice
    left_base  = left_ptr  + b * stride_l_bh + h * stride_l_n
    right_base = right_ptr + b * stride_r_bh + h * stride_r_n
    out_base   = out_ptr   + b * stride_o_bh + h * stride_o_n

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension K in tiles
    for k_tile in range(0, N, BLOCK_K):
        cur_k = tl.min(BLOCK_K, N - k_tile)

        # k‑axis offsets
        offs_k = k_tile + tl.arange(0, cur_k)

        # load L[i,k]  -> (M, K)
        l_ptrs = left_base + (offs_m[:, None] * stride_l_n
                               + offs_k[None, :] * stride_l_k)
        L = tl.load(l_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < N), other=0.0)

        # load R[j,k]  -> (N, K)   we need its transpose (K, N)
        r_ptrs = right_base + (offs_n[:, None] * stride_r_n
                                + offs_k[None, :] * stride_r_k)
        R = tl.load(r_ptrs, mask=mask_n[:, None] & (offs_k[None, :] < N), other=0.0)

        # Rᵀ : (K, N)
        R_T = tl.trans(R, (1, 0))

        # dot: (M, K) @ (K, N) → (M, N)
        acc += tl.dot(L, R_T, out_dtype=tl.float32)

    # write back
    out_ptrs = out_base + (offs_m[:, None] * stride_o_n
                           + offs_n[None, :] * stride_o_k)
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: launch the Triton kernel
# ----------------------------------------------------------------------
def triton_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, N, N, H)   (float16)
    returns       : (B, N, N, H)   (float32)
    """
    B, N, _, H = left.shape
    # stride convention: (B,N,N,H) → (stride_b, stride_i, stride_k, stride_h)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()

    # output will be (B,N,N,H) float32 – keep same memory layout as input
    out = torch.empty_like(left, dtype=torch.float32)

    stride_o_b, stride_o_i, stride_o_k, stride_o_h = out.stride()

    # combine batch and hidden dimensions into a single program id axis
    total_bh = B * H

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        total_bh,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_l_bh=stride_l_b,
        stride_l_n=stride_l_i,
        stride_l_k=stride_l_k,
        stride_r_bh=stride_r_b,
        stride_r_n=stride_r_i,
        stride_r_k=stride_r_k,
        stride_o_bh=stride_o_b,
        stride_o_n=stride_o_i,
        stride_o_k=stride_o_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (mix of Torch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass using a Triton kernel for the
    O(b,i,j,d) = Σ_k left[b,i,k,d] * right[b,j,k,d] reduction.

    Args
    ----
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor (B, N, N, dim)   float32
        mask         : torch.Tensor (B, N, N)        bool / float32
        weights      : dict of model parameters (see skeleton)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape (B, N, N, dim)  (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # compute mean / var per (B,N,N) – we use torch because it is already fast
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]        # (hidden_dim, dim)

    # shape (B,N,N,hidden_dim) – use torch.matmul for the last two dims
    left = torch.nn.functional.linear(x, left_proj_w)   # FP32
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                     # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) and gating of the projected values
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the heavy reduction and launch Triton kernel
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Triton kernel returns (B,N,N,hidden_dim) in FP32
    out_fp32 = triton_tri_mul(left_fp16, right_fp16)   # shape (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 6. Final LayerNorm + gating + linear projection back to dim
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    out = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    return out.to(torch.float32)
