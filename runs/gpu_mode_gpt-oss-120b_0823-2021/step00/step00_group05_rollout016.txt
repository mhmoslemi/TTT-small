"""
TriMul “outgoing” implementation.

The heavy N³ work is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We keep all cheap operations (LayerNorm, Linear, sigmoids, masking) in
torch and off‑load only the contraction to a Triton kernel.
Each Triton program instance computes a whole vector of size `H`
(hidden_dim) for a single (b,i,j) triple, looping over the `k` dimension
in blocks (`BLOCK_K`).  This yields a very low launch overhead (one
program for every output pair) while keeping the inner‑most multiply‑
accumulate in registers, which is optimal on Hopper (SM‑90).
The final result is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -----------------------------------------------------------
# Triton kernel: compute the N³ contraction for one (b,i,j) pair
# -----------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # *float16   [B,N,N,H]
    right_ptr,         # *float16   [B,N,N,H]
    out_ptr,           # *float32   [B,N,N,H]
    B, N, H,           # scalar dimensions
    stride_lb, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rk, stride_rh,   # right strides
    stride_ob, stride_ok, stride_oh,   # out strides
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Compute the (b,i,j) coordinates handled by this program instance
    # ------------------------------------------------------------------
    pid = tl.program_id(0)
    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i = ij // N
    j = ij % N

    # Offsets for the hidden dimension
    d_offs = tl.arange(0, BLOCK_H)

    # ------------------------------------------------------------------
    # Accumulator for the output vector (float32)
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k in blocks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Pointers for the (b,i,k, d) and (b,j,k, d) slices
        left_base = left_ptr + b * stride_lb + i * stride_lk + cur_k[:, None] * stride_lk + d_offs[None, :] * stride_lh
        right_base = right_ptr + b * stride_rb + j * stride_rk + cur_k[:, None] * stride_rk + d_offs[None, :] * stride_rh

        # Load (BLOCK_K, BLOCK_H) tiles, mask out-of‑range k
        left_tile = tl.load(left_base, mask=mask_k[:, None], other=0.0).to(tl.float32)   # fp16 → fp32
        right_tile = tl.load(right_base, mask=mask_k[:, None], other=0.0).to(tl.float32)

        # Reduce over k dimension: Σ_k left[b,i,k,:] * right[b,j,k,:]
        acc += tl.sum(left_tile * right_tile, axis=0)

    # ------------------------------------------------------------------
    # Write the result (float32)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + b * stride_ob + i * stride_ok + j * stride_ok + d_offs * stride_oh
    tl.store(out_ptrs, acc, mask=d_offs < H)


# -----------------------------------------------------------
# Python entry point
# -----------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    TriMul forward (outgoing) – torch for cheap ops, Triton for the N³ contraction.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to apply a linear weight (no bias) in FP32 → FP16 for speed
    # ------------------------------------------------------------------
    def linear(x, w):
        # x: [..., dim]   w: [dim, hidden_dim]
        return torch.matmul(x, w.T)

    # ------------------------------------------------------------------
    # LayerNorm (torch, using supplied weight/bias)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left = linear(x, weights["left_proj.weight"]).to(torch.float16)
    right = linear(x, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # Apply mask if present
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # Gating scalars (sigmoid) – keep in FP32 for stability
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"])).to(torch.float32)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for Triton kernel
    # ------------------------------------------------------------------
    left_contig = left.contiguous()
    right_contig = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes)
    stride_lb, stride_lk, stride_lh = left_contig.stride()  # B,N,N,H
    stride_rb, stride_rk, stride_rh = right_contig.stride()
    stride_ob, stride_ok, stride_oh = out.stride()

    # Kernel launch parameters
    BLOCK_K = 64                         # k‑tile size – fits L2 and registers
    BLOCK_H = hidden_dim                # we compute the whole hidden vector at once

    total_programs = B * N * N
    grid = (total_programs,)

    trimul_outgoing_kernel[grid](
        left_contig,
        right_contig,
        out,
        B, N, hidden_dim,
        stride_lb, stride_lk, stride_lh,
        stride_rb, stride_rk, stride_rh,
        stride_ob, stride_ok, stride_oh,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm, gating, final linear (all in torch)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.unsqueeze(-1)       # broadcast gate over hidden dim

    out = torch.matmul(out, weights["to_out.weight"].T)  # [B,N,N,dim]
    return out.to(torch.float32)
