"""
TriMul “outgoing” implementation.

The heavy part of the module is the 4‑D einsum

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

where left/right have shape [B,N,N,H] (H = hidden_dim).

We keep the LayerNorm / Linear / gating steps in PyTorch (they are cheap)
and accelerate the O(N³·H) contraction with a custom Triton kernel.
For each block of (i,j) and a block of the hidden dimension we load a
tile of K‑dimension from *left* and *right*, perform H independent
matrix‑multiplications (M×K)·(K×N) → (M×N) with `tl.dot`, and accumulate
the result.  The kernel works for any batch size (B∈{1,2}), any sequence
length (N up to 1024) and any hidden dimension that is a multiple of the
chosen block size (we round up with masking).

The final output is converted back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: block‑wise outer‑product contraction over the K dimension
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # static sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    with tiling over i (M), j (N), k and h.
    """
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Decode a 3‑D grid: (batch, block_i, block_j)
    # --------------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    b = pid // (num_i * num_j)
    tmp = pid % (num_i * num_j)
    block_i = tmp // num_j
    block_j = tmp % num_j

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, BLOCK_H)               # hidden dimension tile
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator: (M, N, H) in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers to the K‑tiles of left and right
        # left:   [b, i, k, h]
        # right:  [b, j, k, h]
        ptr_l = left_ptr + b * stride_l_b \
                + offs_i[:, None, None] * stride_l_i \
                + offs_k[None, :, None] * stride_l_k \
                + offs_h[None, None, :] * stride_l_h

        ptr_r = right_ptr + b * stride_r_b \
                + offs_j[:, None, None] * stride_r_j \
                + offs_k[None, :, None] * stride_r_k \
                + offs_h[None, None, :] * stride_r_h

        # Load tiles   (M, K, H)  and (N, K, H)
        left_tile = tl.load(ptr_l,
                            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                            other=0.0)
        right_tile = tl.load(ptr_r,
                             mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                             other=0.0)

        # --------------------------------------------------------------
        # For each h in the hidden block we perform a small matmul:
        #   (M, K) · (K, N) → (M, N)
        # --------------------------------------------------------------
        for dh in range(0, BLOCK_H):
            # guard against overflow of the hidden dimension
            if dh >= H:
                break

            # select the slice for the current hidden index
            l = left_tile[:, :, dh]   # (M, K)
            r = right_tile[:, :, dh]  # (N, K)

            # we need (M, K) @ (K, N).   Triton’s dot expects (M,K) @ (K,N)
            # r is (N,K); we simply transpose it via stride manipulation by
            # passing it as (K,N) using tl.dot which works on contiguous layout.
            # So we load r transposed manually:
            r_t = tl.load(
                right_ptr + b * stride_r_b
                + offs_k[:, None] * stride_r_k
                + offs_j[None, :] * stride_r_j
                + (block_i * BLOCK_M + dh) * stride_r_h,   # dummy stride, not used
                mask=mask_k[:, None] & mask_j[None, :],
                other=0.0)
            # However a simpler approach is to use tl.dot on l and r.T via
            # the `tl.dot` API which accepts (M, K) and (K, N) directly.
            # We can just take r as (K, N) by swapping axes:
            r_T = tl.trans(r)               # (K, N)

            # matmul accumulation
            acc[:, :, dh] += tl.dot(l, r_T)  # (M, N)

    # --------------------------------------------------------------
    # Store the result
    # --------------------------------------------------------------
    ptr_o = out_ptr + b * stride_o_b \
            + offs_i[:, None, None] * stride_o_i \
            + offs_j[None, :, None] * stride_o_j \
            + offs_h[None, None, :] * stride_o_h

    tl.store(ptr_o,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])


# ----------------------------------------------------------------------
# Host side: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the outgoing TriMul operator.

    Arguments
    ---------
    data : (input, mask, weights, config)
        - input : [B, N, N, D]   (float32)
        - mask  : [B, N, N]      (float32 or bool)
        - weights : dict of parameters (see skeleton)
        - config  : {"dim": D, "hidden_dim": H}
    Returns
    -------
    out : torch.Tensor [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(D,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if present) – mask is broadcasted on the hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the einsum contraction
    # ------------------------------------------------------------------
    # allocate output buffer (float32)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    l_strides = left.stride()
    r_strides = right.stride()
    o_strides = out.stride()

    # launch parameters – experimentally good for Hopper
    BLOCK_M = 64   # i‑dimension tile
    BLOCK_N = 64   # j‑dimension tile
    BLOCK_K = 32   # k‑tile
    BLOCK_H = 32   # hidden‑dim tile (must divide H or be masked)

    total_blocks_i = (N + BLOCK_M - 1) // BLOCK_M
    total_blocks_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * total_blocks_i * total_blocks_j,)

    trimul_einsum_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, H,
        l_strides[0], l_strides[1], l_strides[2], l_strides[3],
        r_strides[0], r_strides[1], r_strides[2], r_strides[3],
        o_strides[0], o_strides[1], o_strides[2], o_strides[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(H,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)

    out = out * out_gate          # broadcast on hidden dim
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
