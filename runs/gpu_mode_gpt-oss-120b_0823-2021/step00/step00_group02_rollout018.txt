"""
TriMul “outgoing” implementation.

The heavy‑lifting part of the module is the three‑dimensional einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix‑multiply for each hidden channel *d*:

    out[b,:,:,d] = left[b,:,:,d] @ right[b,:,:,d].transpose(-2,-1)

The kernel below computes this product with a tiled GEMM written in Triton.
All other operations (LayerNorms, Linear projections and sigmoids) are
performed with regular PyTorch kernels – they are cheap compared with the
O(N³·D) work of the einsum and keeping them in PyTorch lets the Triton kernel
focus on a single, well‑structured computation.
The final result is returned as float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# Triton kernel:  out = left @ rightᵀ   (per batch & per hidden dim)
# --------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr,               # *float32   [B, N, N, D]
    right_ptr,              # *float32   [B, N, N, D]
    out_ptr,                # *float32   [B, N, N, D]

    B, N, D,                # problem sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,

    BLOCK_M: tl.constexpr,  # tile size along i
    BLOCK_N: tl.constexpr,  # tile size along j
    BLOCK_K: tl.constexpr,  # tile size along k
):
    """
    One program instance computes a BLOCK_M×BLOCK_N tile of the output matrix
    for a single (batch, hidden_dim) pair.
    """
    # --------------------------------------------------------------
    # program identifiers
    # --------------------------------------------------------------
    pid = tl.program_id(0)                 # linear id over (batch, d)
    pid_m = tl.program_id(1)               # i‑tile id
    pid_n = tl.program_id(2)               # j‑tile id

    b = pid // D                           # batch index
    d = pid % D                            # hidden‑dim index

    # --------------------------------------------------------------
    # offsets for the i and j dimensions
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # main K‑loop
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, d]  -> shape (M, K)
        left = tl.load(
            left_ptr
            + b * stride_left_b
            + d * stride_left_d
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right[b, j, k, d]  we need the k‑dimension as the inner one,
        # so we load it transposed: (K, N)
        right = tl.load(
            right_ptr
            + b * stride_right_b
            + d * stride_right_d
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # GEMM: (M, K) @ (K, N) -> (M, N)
        acc += tl.dot(left, right)

    # --------------------------------------------------------------
    # write back the result
    # --------------------------------------------------------------
    out = acc.to(tl.float32)  # keep fp32; later PyTorch will cast if needed
    tl.store(
        out_ptr
        + b * stride_out_b
        + d * stride_out_d
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j,
        out,
        mask=mask_i[:, None] & mask_j[None, :],
    )

# Host‑side entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : (input, mask, weights, config)
        input  – torch.Tensor of shape [B, N, N, dim] (float32)
        mask   – torch.Tensor of shape [B, N, N]   (float32 or bool)
        weights – dict with all linear / layer‑norm parameters
        config  – dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]        # (hidden_dim, dim)

    # reshape for a single matmul per channel
    x_flat = x.reshape(B * N * N, dim)                 # (B*N*N, dim)

    left = torch.nn.functional.linear(x_flat, left_proj_w)   # (B*N*N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, right_proj_w)

    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                 # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x_flat, left_gate_w).view(B, N, N, hidden_dim).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, right_gate_w).view(B, N, N, hidden_dim).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, out_gate_w).view(B, N, N, hidden_dim).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton GEMM: out = Σ_k left[...,i,k,d] * right[...,j,k,d]
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements, not bytes) for the 4‑D tensors
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32                        # small enough to fit in registers

    grid = (
        B * hidden_dim,                 # one program per (batch, hidden_dim)
        triton.cdiv(N, BLOCK_M),        # i‑tiles
        triton.cdiv(N, BLOCK_N),        # j‑tiles
    )

    trimul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_d,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_d,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm, out‑gate, final linear)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate
    # final projection back to `dim`
    out_flat = out.view(B * N * N, hidden_dim)
    out = torch.nn.functional.linear(out_flat, to_out_w).view(B, N, N, dim)

    return out
