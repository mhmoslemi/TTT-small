"""
TriMul “outgoing” kernel (AlphaFold3).

We compute the heavy O(N³·C) part
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
by a fused batched matrix‑multiply in Triton.
All LayerNorm / Linear / gating work is kept in PyTorch
(they are cheap compared to the N³ loop).  The kernel
receives the masked & gated tensors `left` and `right`
in FP16, performs the matmul in FP32 and writes the
result back in FP32.  The final Tensor is reshaped to
[B, N, N, C] and passed through the remaining PyTorch
operations.

The launch grid is 3‑D:
    dim0 : batch * hidden_dim   (one matmul per (b,d) pair)
    dim1 : output‑row blocks    (i axis)
    dim2 : output‑col blocks    (j axis)

Block sizes are tuned for Hopper (SM‑90): 64×64 tiles
with a K‑tile of 32.  The kernel works for any
seq_len ≤ 1024 and hidden_dim ≤ 768 as required by the
test matrix.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the TriMul einsum
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    L_ptr,                     # left   [B*D, N, N]
    R_ptr,                     # right  [B*D, N, N]
    O_ptr,                     # out    [B*D, N, N]
    BDN,                       # total number of (batch, hidden) combos
    N,                         # sequence length
    stride_lbn, stride_lnk, stride_ld,   # strides for left   (bn, nk, d)
    stride_rbn, stride_rnk, stride_rd,   # strides for right  (bn, nk, d)
    stride_obn, stride_onn, stride_od,   # strides for out    (bn, nn, d)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute, for each (b,d) slice:
        O[b,d,i,j] = Σ_k L[b,d,i,k] * R[b,d,j,k]
    where L,R are stored as (B*D, N, N) with the last dimension being K.
    """
    pid_bd = tl.program_id(0)               # index over (batch, hidden)
    pid_m  = tl.program_id(1)               # row block index
    pid_n  = tl.program_id(2)               # col block index

    # base offsets for this (b,d) slice
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the beginning of the current (b,d) tile
    L_base = L_ptr + pid_bd * stride_ld
    R_base = R_ptr + pid_bd * stride_rd
    O_base = O_ptr + pid_bd * stride_od

    # accumulate in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load L[i,k]   shape (M,K)
        L_ptrs = L_base + (offs_m[:, None] * stride_lbn
                           + offs_k[None, :] * stride_lnk)
        L = tl.load(L_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load R[j,k]   shape (N,K)  (note we need R[j,k] not R[k,j])
        R_ptrs = R_base + (offs_n[:, None] * stride_rbn
                           + offs_k[None, :] * stride_rnk)
        R = tl.load(R_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # accumulate dot product over K
        acc += tl.dot(L, R)   # (M,K)·(N,K) -> (M,N) because dot does Σ_k L * R

    # write result
    O_ptrs = O_base + (offs_m[:, None] * stride_obn
                       + offs_n[None, :] * stride_onn)
    tl.store(O_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, D]  (FP16)
    returns out   : [B, N, N, D]  (FP32)
    """
    B, N, _, D = left.shape
    # reshape to (B*D, N, N) – we will treat each (b,d) as an independent matmul
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * D, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * D, N, N)

    out_ = torch.empty_like(left_, dtype=torch.float32)

    # strides in element units (not bytes) for the 3‑D tensors
    stride_lbn, stride_lnk, stride_ld = left_.stride()
    stride_rbn, stride_rnk, stride_rd = right_.stride()
    stride_obn, stride_onn, stride_od = out_.stride()

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(B * D, 1),            # dim0: one program per (b,d)
        triton.cdiv(N, BLOCK_M),          # dim1: row tiles
        triton.cdiv(N, BLOCK_N),          # dim2: col tiles
    )
    _trimul_matmul_kernel[grid](
        left_.data_ptr(),
        right_.data_ptr(),
        out_.data_ptr(),
        B * D,
        N,
        stride_lbn, stride_lnk, stride_ld,
        stride_rbn, stride_rnk, stride_rd,
        stride_obn, stride_onn, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # tuned for Hopper
    )

    # reshape back to (B,N,N,D)
    out = out_.view(B, D, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Full custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    Only the N³ einsum is executed in Triton; all other ops stay in PyTorch.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ---------- LayerNorm (dim) ----------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias          # [B,N,N,dim]

    # ---------- Linear projections (dim -> hidden_dim) ----------
    # shape: (dim, hidden_dim) per weight matrix – we use torch.nn.functional.linear
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"], bias=None
    )  # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"], bias=None
    )

    # ---------- Apply mask (if present) ----------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ---------- Gating ----------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"], bias=None
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"], bias=None
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"], bias=None
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Core TriMul (N³) ----------
    # Cast to FP16 for the kernel (good performance, enough precision)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out = trimul_matmul(left_fp16, right_fp16)   # [B,N,N,hidden_dim] float32

    # ---------- Post‑processing ----------
    # to_out_norm (LayerNorm over hidden_dim)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean = out.mean(-1, keepdim=True)
    var = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean) / torch.sqrt(var + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    # gating and final linear
    out = out * out_gate
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"], bias=None
    )  # back to dim

    return out  # float32 tensor of shape [B, N, N, dim]
