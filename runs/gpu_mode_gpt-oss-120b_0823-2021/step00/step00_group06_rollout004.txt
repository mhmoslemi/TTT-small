"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (B,N,N,D).
2. Linear projections → left / right (B,N,N,H) and three sigmoid gates.
3. Apply masks and gates.
4. Core operation: for every batch b and hidden head h
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   This is a batched matrix‑multiply of
       A = left  (shape B,N,N,H)
       B = rightᵀ (shape B,N,N,H)      (transpose the last two dims)
   i.e. out = A @ Bᵀ  performed per (b,h).
   The heavy N³ work is fused in a single Triton kernel that
   tiles the M,N,K dimensions (M=i, N=j, K=k) using three‑level
   tiling (BLOCK_M×BLOCK_N×BLOCK_K).  The kernel works in fp16 for
   speed and converts the accumulator to fp32 for the final result.
5. Apply a second LayerNorm, multiplicative out‑gate and the final
   linear projection back to D (float32).

Only the N³ inner product (step 4) is written in Triton; everything
else stays in PyTorch and uses the supplied weight tensors.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel : batched matmul for the TriMul core
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # problem size
    BATCH, N, H,
    # strides (in elements, not bytes)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    # tile sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C[b,i,j,h] = Σ_k A[b,i,k,h] * B[b,j,k,h]  (B is already transposed)."""
    pid = tl.program_id(0)                     # linear program id over (batch*head)
    num_bh = BATCH * H
    bh = pid                                   # (b * H + h)
    b = bh // H
    h = bh % H

    # tile offsets
    offs_i = tl.arange(0, BLOCK_M)
    offs_j = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this (b,h) pair
    a_base = a_ptr + b * stride_a_batch + h * stride_a_h
    b_base = b_ptr + b * stride_b_batch + h * stride_b_h
    c_base = c_ptr + b * stride_c_batch + h * stride_c_h

    # loop over tiles of the output matrix
    for tile_i in range(0, N, BLOCK_M):
        for tile_j in range(0, N, BLOCK_N):
            # accumulator in fp32
            acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

            # loop over K dimension
            for tile_k in range(0, N, BLOCK_K):
                # compute actual offsets with bounds checking
                i = tile_i + offs_i
                j = tile_j + offs_j
                k = tile_k + offs_k

                a_mask = (i < N) & (k < N)
                b_mask = (j < N) & (k < N)

                a_ptrs = a_base + i[:, None] * stride_a_i + k[None, :] * stride_a_k
                b_ptrs = b_base + j[:, None] * stride_b_j + k[None, :] * stride_b_k

                a = tl.load(a_ptrs, mask=a_mask[:, None], other=0.0).to(tl.float16)
                b = tl.load(b_ptrs, mask=b_mask[:, None], other=0.0).to(tl.float16)

                # dot‑product over K
                acc += tl.dot(a, b).to(tl.float32)

            # store the result tile
            c_i = tile_i + offs_i
            c_j = tile_j + offs_j
            c_mask = (c_i[:, None] < N) & (c_j[None, :] < N)
            c_ptrs = c_base + c_i[:, None] * stride_c_i + c_j[None, :] * stride_c_j
            tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

# ----------------------------------------------------------------------
# Host helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left , right : (B, N, N, H)  float16
    returns out   : (B, N, N, H)  float32
    """
    B, N, _, H = left.shape
    device = left.device
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides in element counts
    stride_a = left.stride()
    stride_b = right.stride()
    stride_c = out.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (B * H,)

    trimul_matmul_kernel[grid](
        left,                     # a_ptr
        right,                    # b_ptr (already transposed)
        out,
        B, N, H,
        stride_a[0], stride_a[1], stride_a[2], stride_a[3],
        stride_b[0], stride_b[2], stride_b[1], stride_b[3],   # note swapped j/k stride for transposed view
        stride_c[0], stride_c[1], stride_c[2], stride_c[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out

# ----------------------------------------------------------------------
# Main entry point required by the judge
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            input : Tensor[B, N, N, D] (float32)
            mask  : Tensor[B, N, N]      (float32 or bool)
            weights: dict of parameters (see problem statement)
            config : dict with keys "dim", "hidden_dim"

    Returns:
        Tensor[B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm (first)
    # ------------------------------------------------------------------
    # LayerNorm parameters
    ln_w = weights["norm.weight"].to(input_tensor.dtype)
    ln_b = weights["norm.bias"].to(input_tensor.dtype)

    # reshape to (D,) and apply per‑element
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.rsqrt(var + 1e-5)
    x_norm = x_norm * ln_w + ln_b                     # [B,N,N,D]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – keep in float16 for speed
    # ------------------------------------------------------------------
    # shape helpers
    def linear_no_bias(x, w):
        # x : [..., D] , w : [hidden, D] (as stored in PyTorch weight = out_features x in_features)
        return torch.einsum('...d,hd->...h', x, w)

    left = linear_no_bias(x_norm, weights["left_proj.weight"]).to(torch.float16)
    right = linear_no_bias(x_norm, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Mask handling (if mask is provided)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(linear_no_bias(x_norm, weights["left_gate.weight"])).to(left.dtype)
    right_gate = torch.sigmoid(linear_no_bias(x_norm, weights["right_gate.weight"])).to(right.dtype)
    out_gate = torch.sigmoid(linear_no_bias(x_norm, weights["out_gate.weight"])).to(left.dtype)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul: compute Σ_k left[...,k,h] * right[...,k,h]
    #    Right needs to be transposed on its last two dims.
    # ------------------------------------------------------------------
    # Transpose the (N,N) axes of right: (B,N,N,H) -> (B,N,N,H) with swapped dim 1 & 2
    right_t = right.permute(0, 2, 1, 3).contiguous()   # now [B,N,N,H] where dim2 is original k

    # Call the Triton kernel (output in fp32)
    out = trimul_matmul(left, right_t)                # shape [B,N,N,H] fp32

    # ------------------------------------------------------------------
    # 6. Final LayerNorm, out‑gate and projection back to D
    # ------------------------------------------------------------------
    # to_out_norm
    to_out_ln_w = weights["to_out_norm.weight"].to(out.dtype)
    to_out_ln_b = weights["to_out_norm.bias"].to(out.dtype)

    mean2 = out.mean(dim=-1, keepdim=True)
    var2 = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean2) / torch.rsqrt(var2 + 1e-5)
    out_norm = out_norm * to_out_ln_w + to_out_ln_b

    out_norm = out_norm * out_gate                           # broadcast over hidden dim

    # final linear projection back to D (no bias)
    to_out_w = weights["to_out.weight"]                     # shape [D, H]
    # use einsum for clarity; result cast to float32
    final = torch.einsum('...h,dh->...d', out_norm, to_out_w).to(torch.float32)

    return final
