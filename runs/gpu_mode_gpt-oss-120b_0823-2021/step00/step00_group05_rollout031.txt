"""
TriMul “outgoing” implementation.

The expensive part of the module is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication of shape
    (B, D, N, N)  ←  (B, D, N, N)  @  (B, D, N, N).transpose(-2,-1)

The kernel below computes this product in fp16 (the model works in fp16/bf16)
while accumulating in fp32 for numeric stability.  All surrounding
operations (LayerNorms, Linear projections, sigmoid gates) are kept in
regular PyTorch – they are cheap compared to the N³ work of the einsum.
The kernel is launched with a 3‑D grid:
    (M‑tiles, N‑tiles, B*hidden_dim)
so each program instance handles one tile of the matrix product for a
specific (batch, hidden) pair.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batch * hidden_dim matmul (left @ right.T)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    # pointers
    left_ptr,          # [B, H, N, N]   (i,k)
    right_ptr,         # [B, H, N, N]   (j,k)  we read it as (k,j) -> transposed
    out_ptr,           # [B, H, N, N]   (i,j)

    # sizes
    N,                 # sequence length
    H,                 # hidden dimension
    # strides (in elements, not bytes)
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_j, stride_right_k,
    stride_out_bh, stride_out_i, stride_out_j,

    # tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)                     # tile over i (rows)
    pid_n = tl.program_id(1)                     # tile over j (cols)
    pid_z = tl.program_id(2)                     # batch * hidden index

    # decode batch and hidden index
    b = pid_z // H
    h = pid_z % H

    # compute offsets for this tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers for the current (b,h)
    left_base = left_ptr + b * stride_left_bh + h * stride_left_i
    right_base = right_ptr + b * stride_right_bh + h * stride_right_j

    # allocate accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute actual k range for this iteration
        k_lo = k * BLOCK_K
        cur_k = k_lo + offs_k
        mask_k = cur_k < N

        # load left[i,k]  (i is rows, k is cols)
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + (cur_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # load right[j,k] but we need it as (k,j) i.e. transpose on the fly
        # right is stored as (j,k) -> we read with swapped indices
        right_ptrs = right_base + (cur_k[:, None] * stride_right_k) + offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float16)

        # matmul (M,K) x (K,N) -> (M,N)
        acc += tl.dot(left, right)

        # advance pointers (implicit via loop index)
    # ------------------------------------------------------------------
    # write result
    # ------------------------------------------------------------------
    out_base = out_ptr + b * stride_out_bh + h * stride_out_i
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    out = acc.to(tl.float16)
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator with a fused Triton kernel
    for the N³ einsum. All other ops (LayerNorm, Linear, sigmoid gates) are
    performed with PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------------
    # Helper functions for linear / layernorm using supplied weights
    # --------------------------------------------------------------
    def linear(x, w):
        # w shape: (out_features, in_features)
        return F.linear(x, w)

    def layernorm(x, weight, bias, normalized_shape):
        return F.layer_norm(x, normalized_shape, weight, bias, eps=1e-5)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = layernorm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        normalized_shape=(dim,),
    )  # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2. Projections (no bias)
    # --------------------------------------------------------------
    # reshape to (B*N*N, dim) for a single matmul call
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)
    left = linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden)
    right = linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden)

    # --------------------------------------------------------------
    # 3. Apply mask (if any)
    # --------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)            # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(linear(x_flat, weights["left_gate.weight"])).view(B, N, N, hidden)
    right_gate = torch.sigmoid(linear(x_flat, weights["right_gate.weight"])).view(B, N, N, hidden)
    out_gate = torch.sigmoid(linear(x_flat, weights["out_gate.weight"])).view(B, N, N, hidden)

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton matmul: out = left @ right.transpose(-2,-1)
    #    Shapes: left/right  -> (B, hidden, N, N)
    # --------------------------------------------------------------
    # transpose to (B, hidden, N, N) for convenient stride handling
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)   i,k
    right_t = right.permute(0, 3, 2, 1).contiguous() # (B,H,N,N)   j,k  (already transposed)

    out = torch.empty_like(left_t)                  # (B,H,N,N)

    # stride extraction (in elements)
    stride_left_bh, stride_left_i, stride_left_k, _ = left_t.stride()
    stride_right_bh, stride_right_j, stride_right_k, _ = right_t.stride()
    stride_out_bh, stride_out_i, stride_out_j, _ = out.stride()

    # tile sizes – chosen to fit H200 shared memory & occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden,
    )
    tri_mul_kernel[grid](
        left_t,
        right_t,
        out,
        N,
        hidden,
        stride_left_bh, stride_left_i, stride_left_k,
        stride_right_bh, stride_right_j, stride_right_k,
        stride_out_bh, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # back to original layout (B,N,N,hidden)
    out = out.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear
    # --------------------------------------------------------------
    out = layernorm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        normalized_shape=(hidden,),
    )
    out = out * out_gate

    out_flat = out.view(-1, hidden)
    out = linear(out_flat, weights["to_out.weight"]).view(B, N, N, dim)

    return out.to(dtype)
