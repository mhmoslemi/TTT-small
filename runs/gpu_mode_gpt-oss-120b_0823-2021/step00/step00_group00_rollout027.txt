"""
tri_mul_outgoing.py

Implementation of the *outgoing* TriMul operator (AlphaFold‑3) using a mixed
PyTorch / Triton approach.

The heavy N³ part

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

is realised as a custom Triton kernel that performs a batched,
head‑wise matrix‑multiply (left @ rightᵀ) for every hidden channel *d*.
All LayerNorms, linear projections and gating are kept in PyTorch – they are
memory‑bound and benefit from the highly‑optimised ATen kernels.  The Triton
kernel works in float32 internally and writes the result back in float16;
the surrounding Python code casts back to float32 before the final
normalisation / linear projection.

The kernel is written for Hopper (SM‑90) but works on any recent GPU.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# ----------------------------------------------------------------------
# Triton kernel: batched head‑wise matmul
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,                # [B,N,N,H]   (float32)
    right_ptr,               # [B,N,N,H]   (float32)
    out_ptr,                 # [B,N,N,H]   (float32)
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # ------------------------------------------------------------------
    BLOCK_M: tl.constexpr,   # tile size in the i‑dimension
    BLOCK_N: tl.constexpr,   # tile size in the j‑dimension
    BLOCK_K: tl.constexpr,   # tile size in the reduction (k) dimension
    BLOCK_H: tl.constexpr,   # number of hidden channels processed together
):
    """Compute:
           out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

       The kernel is launched with a 3‑D grid:
           dim0 -> (batch * ceil(H/BLOCK_H))
           dim1 -> ceil(N/BLOCK_M)
           dim2 -> ceil(N/BLOCK_N)
    """
    pid0 = tl.program_id(0)          # combined batch/head id
    pid_m = tl.program_id(1)          # i‑tile id
    pid_n = tl.program_id(2)          # j‑tile id

    # --------------------------------------------------------------
    # Decode batch and head block from pid0
    # --------------------------------------------------------------
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    b = pid0 // num_h_blocks
    h_block = (pid0 % num_h_blocks) * BLOCK_H               # first hidden index in this block

    # --------------------------------------------------------------
    # Offsets for the i‑ and j‑tiles
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # offsets for the hidden dimension inside this block
    offs_h = h_block + tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main reduction loop over k
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load left[b,i,k,h]   shape (M, K, H)
        # ------------------------------------------------------------------
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            offs_m[:, None, None] * stride_left_i + \
            offs_k[None, :, None] * stride_left_k + \
            offs_h[None, None, :] * stride_left_h
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                       other=0.0)

        # ------------------------------------------------------------------
        # Load right[b,j,k,h]  shape (N, K, H)
        # ------------------------------------------------------------------
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            offs_n[:, None, None] * stride_right_j + \
            offs_k[None, :, None] * stride_right_k + \
            offs_h[None, None, :] * stride_right_h
        right = tl.load(right_ptrs,
                        mask=mask_n[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                        other=0.0)

        # --------------------------------------------------------------
        # For each hidden channel we need a dot‑product over K:
        #   (M,K) . (N,K).T -> (M,N)
        # Triton only provides tl.dot for 2‑D matrices, so we iterate over
        # the hidden dimension inside the kernel (BLOCK_H is small, typically 1‑4).
        # --------------------------------------------------------------
        for hh in range(0, BLOCK_H):
            l = left[:, :, hh]          # (M,K)
            r = right[:, :, hh]         # (N,K)
            # tl.dot expects (M,K) and (K,N); we transpose r on‑the‑fly.
            acc[:, :, hh] += tl.dot(l, r, trans_b=True)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        offs_m[:, None, None] * stride_out_i + \
        offs_n[None, :, None] * stride_out_j + \
        offs_h[None, None, :] * stride_out_h

    tl.store(out_ptrs,
             acc.to(tl.float16),          # keep memory traffic low
             mask=mask_m[:, None, None] & mask_n[None, :, None] & mask_h[None, None, :])


# ----------------------------------------------------------------------
# Helper: launch the Triton kernel
# ----------------------------------------------------------------------
def _run_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B,N,N,H]   (float32)
    returns                out = Σₖ left[...,k,:] * right[...,k,:]   (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # stride bookkeeping (contiguous tensors -> stride = product of later dims)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – tuned for Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 1          # we process one hidden channel per program; a small loop inside the kernel

    grid = (
        triton.cdiv(B * triton.cdiv(H, BLOCK_H), 1),   # dim0 : batch * h‑blocks
        triton.cdiv(N, BLOCK_M),                      # dim1 : i‑tiles
        triton.cdiv(N, BLOCK_N),                      # dim2 : j‑tiles
    )
    _trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   (float32)
            - mask         : [B, N, N]        (float32 or bool)
            - weights      : dict of pre‑loaded weight tensors
            - config       : dict containing "dim" and "hidden_dim"
    Returns:
        output tensor : [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣  LayerNorm on the input (dim‑wise)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    eps = 1e-5
    # manual layer‑norm to keep everything on‑device without extra kernels
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_weight + norm_bias          # shape [B,N,N,dim]

    # --------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # --------------------------------------------------------------
    # weight shapes: [hidden, dim]
    left_proj_weight  = weights["left_proj.weight"]          # (hidden, dim)
    right_proj_weight = weights["right_proj.weight"]
    left  = torch.einsum('...d,hd->...h', x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.einsum('...d,hd->...h', x, right_proj_weight)

    # --------------------------------------------------------------
    # 3️⃣  Apply mask (if present)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)    # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣  Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(torch.einsum('...d,hd->...h', x, left_gate_weight))
    right_gate = torch.sigmoid(torch.einsum('...d,hd->...h', x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.einsum('...d,hd->...h', x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣  N³ reduction – Triton kernel
    # --------------------------------------------------------------
    # left/right are now [B,N,N,hidden]; the kernel expects that layout.
    # Cast to float32 (they already are) and call the kernel.
    out = _run_triton(left, right)          # [B,N,N,hidden]  (float32)

    # --------------------------------------------------------------
    # 6️⃣  Output normalisation + gating + final linear
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]          # (dim, hidden)

    # LayerNorm over hidden dimension
    eps2 = 1e-5
    mean_h = out.mean(dim=-1, keepdim=True)
    var_h  = ((out - mean_h) ** 2).mean(dim=-1, keepdim=True)
    out = (out - mean_h) * torch.rsqrt(var_h + eps2)
    out = out * to_out_norm_weight + to_out_norm_bias        # [B,N,N,hidden]

    # Apply final gate (broadcast over hidden)
    out = out * out_gate

    # Final linear back to dim
    out = torch.einsum('...h,dh->...d', out, to_out_weight)   # [B,N,N,dim]

    return out
