"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

*  LayerNorm, linear projections and gating are performed with plain
   torch.nn functional calls – these are cheap compared with the O(N³)
   reduction.
*  The only heavy part is the tensor contraction

        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

   which is a batched matrix‑multiply of shape (N, N) × (N, N)ᵀ for every
   (batch, hidden) pair.  This is implemented in Triton as a fused
   mat‑mul‑transpose kernel.  Each kernel instance computes a
   BLOCK_M×BLOCK_N tile of the output matrix C = A·Bᵀ for a single
   (batch, hidden) slice.

The kernel works for any batch size (≤2), any sequence length N (up to
1024 in the tests) and any hidden dimension H (typically 128).  All
computations are done in float16 for throughput; the final result is
converted back to float32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A and B are N×N, 16‑bit, tile‑wise)
# ----------------------------------------------------------------------
@triton.jit
def trilu_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH_H,                 # number of (batch, hidden) slices
    N,                       # sequence length
    stride_ah, stride_ak,    # A strides  (batch_h, i, k)
    stride_bh, stride_bk,    # B strides  (batch_h, j, k)
    stride_ch, stride_ci,    # C strides  (batch_h, i, j)
    BLOCK_M: tl.constexpr,   # tile rows
    BLOCK_N: tl.constexpr,   # tile cols
    BLOCK_K: tl.constexpr,   # reduction tile
):
    # each program works on one (batch,hidden) slice
    pid = tl.program_id(0)
    batch_h = pid  # 0 … BATCH_H-1

    # tile offsets
    pid_m = tl.program_id(1)   # row tile
    pid_n = tl.program_id(2)   # col tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the start of our slice
    a_base = a_ptr + batch_h * stride_ah
    b_base = b_ptr + batch_h * stride_bh
    c_base = c_ptr + batch_h * stride_ch

    # accumulate in fp32 for accuracy, store as fp16
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A[i, k]   (shape BLOCK_M×BLOCK_K)
        a_ptrs = a_base + (offs_m[:, None] * stride_ah) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B[j, k]   (shape BLOCK_N×BLOCK_K) – note we need Bᵀ, so we load
        # B[j, k] and will dot with A[i, k] directly.
        b_ptrs = b_base + (offs_n[:, None] * stride_bh) + (offs_k[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # inner‑product across k dimension
        acc += tl.dot(a, b)       # (M×K)·(N×K)ᵀ → (M×N)

    # write back
    c_ptrs = c_base + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper that launches the kernel for one (batch, hidden) slice
# ----------------------------------------------------------------------
def trilu_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]  (float16)
    returns      : [B, N, N, H]  (float16)
    """
    B, N, _, H = left.shape
    # contiguous view as (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left = left.view(B * H, N, N)
    right = right.view(B * H, N, N)

    # output tensor
    out = torch.empty_like(left)

    # strides (in elements) for Triton
    stride_ah, stride_ak = left.stride()[:2]   # (batch_h, i, k)
    stride_bh, stride_bk = right.stride()[:2]
    stride_ch, stride_ci = out.stride()[:2]   # (batch_h, i, j)

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        B * H,                                 # program id 0 : each slice
        triton.cdiv(N, BLOCK_M),               # program id 1 : row tiles
        triton.cdiv(N, BLOCK_N),               # program id 2 : col tiles
    )

    trilu_matmul_transpose_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        BATCH_H=B * H,
        N=N,
        stride_ah=stride_ah,
        stride_ak=stride_ak,
        stride_bh=stride_bh,
        stride_bk=stride_bk,
        stride_ch=stride_ch,
        stride_ci=stride_ci,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to original layout
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    All heavy reductions (the N³ contraction) are performed in Triton.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (float32 -> float16 for speed)
    # ------------------------------------------------------------------
    # weight & bias are fp32 in the checkpoint; we keep them fp32 here.
    norm_weight = weights["norm.weight"].to(input_tensor.dtype)
    norm_bias   = weights["norm.bias"].to(input_tensor.dtype)

    # fused layer‑norm (no built‑in Triton version – torch is fine)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight,
                     bias=norm_bias, eps=1e-5)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – we fuse two‑step matmul + reshape
    # ------------------------------------------------------------------
    # shape: [..., dim] @ dim×hidden_dimᵀ  →  [..., hidden_dim]
    def linear_no_bias(t, w):
        # t: (B, N, N, dim)   w: (hidden_dim, dim)  – note transposed weight
        return torch.einsum('...d,hd->...h', t, w)

    left = linear_no_bias(x, weights["left_proj.weight"]).to(torch.float16)
    right = linear_no_bias(x, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any) – mask shape (B, N, N) → (...,1)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – keep in fp16 for the element‑wise ops
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(linear_no_bias(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(linear_no_bias(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate   = torch.sigmoid(linear_no_bias(x, weights["out_gate.weight"])).to(torch.float16)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction:   out = Σ_k left[...,i,k,h] * right[...,j,k,h]
    # ------------------------------------------------------------------
    out = trilu_matmul_transpose(left, right)      # fp16, shape [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm + gating + final linear (back to dim)
    # ------------------------------------------------------------------
    # to_out_norm (hidden_dim) – still fp16 but we need fp32 for final proj
    to_out_norm_w = weights["to_out_norm.weight"].to(out.dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(out.dtype)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w,
                       bias=to_out_norm_b, eps=1e-5)

    out = out * out_gate      # gating in fp16

    # final linear back to dim (no bias)
    to_out_w = weights["to_out.weight"]          # (dim, hidden_dim)
    out = torch.einsum('...h,dh->...d', out, to_out_w)   # fp16 → fp16

    # return as float32 as required by the reference
    return out.to(torch.float32)
