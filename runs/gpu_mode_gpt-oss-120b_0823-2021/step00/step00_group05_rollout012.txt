"""
TriMul “outgoing” forward pass

We keep the cheap point‑wise ops (layer‑norm, linear projections and gates)
in PyTorch and move the heavy N³ einsum
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
into a fused Triton kernel.  The einsum is equivalent to a batch of
matrix‑multiplications  (left @ rightᵀ)  for every hidden dimension *d*.
The kernel processes a combined (batch × hidden_dim) dimension so that the
inner GEMM works on 2‑D tiles (BLOCK_M × BLOCK_N) with a BLOCK_K reduction
loop.  All computations are done in fp32; inputs/outputs are expected to be
cuda tensors (float16/float32 are both supported).  The final output is cast
to float32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel:  out = left @ rightᵀ   (per hidden dimension)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix sizes (all = seq_len)
    stride_ah, stride_ak,               # left: batch*hid stride, i‑stride, k‑stride
    stride_bh, stride_bj, stride_bk,    # right: batch*hid stride, j‑stride, k‑stride
    stride_ch, stride_ck,               # out: batch*hid stride, i‑stride, j‑stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """compute C = A @ Bᵀ  for a single (batch, hidden) slice."""
    bh   = tl.program_id(0)          # combined batch*hidden index
    pid_m = tl.program_id(1)         # tile row
    pid_n = tl.program_id(2)         # tile col

    # ------------------- tile offsets -------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds (required for non‑multiple of BLOCK sizes)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------- pointers -------------------
    # A has shape (M, K)   : left[b,i,k,hd]
    a_ptrs = a_ptr + bh * stride_ah + offs_m[:, None] * stride_ah + offs_k[None, :] * stride_ak

    # Bᵀ (transpose on‑the‑fly) has shape (K, N) where original B is (j, k)
    # original strides: j‑stride = stride_bj, k‑stride = stride_bk
    b_ptrs = b_ptr + bh * stride_bh + offs_k[:, None] * stride_bj + offs_n[None, :] * stride_bk

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- reduction over K -------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[None, :] & (offs_k[:, None] < K), other=0.0)
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) • (BLOCK_K, BLOCK_N)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bj

    # ------------------- write result -------------------
    c_ptrs = c_ptr + bh * stride_ch + offs_m[:, None] * stride_ch + offs_n[None, :] * stride_ck
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ----------
    data : tuple
        (input, mask, weights, config)
        - input  : [B, N, N, C]   (float16/float32 on cuda)
        - mask   : [B, N, N]      (bool or float)
        - weights: dict of pretrained parameters
        - config : dict with keys ``dim`` and ``hidden_dim``
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm ----------
    # use torch.nn.functional.layer_norm which is fast on GPU
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (C,), weight=norm_w, bias=norm_b, eps=eps)

    # ---------- Linear projections ----------
    # left_proj & right_proj (bias=False)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton GEMM: left @ rightᵀ  ->  out  (shape [B, N, N, hidden])
    # ------------------------------------------------------------------
    # reshape to combine batch and hidden dim for the kernel
    left_flat = left.reshape(B * hidden, N, N)          # (BH, M, K)
    right_flat = right.reshape(B * hidden, N, N)        # (BH, J, K)
    out_flat = torch.empty_like(left_flat)              # (BH, M, N)

    # strides (in elements) for the original contiguous layout
    stride_BH = N * N * hidden          # advance one (batch,hidden) slice
    stride_i = N * hidden               # i‑row stride inside a slice
    stride_k = hidden                   # k‑col stride inside a slice
    # right uses the same i‑stride for its j dimension
    stride_j = N * hidden               # j‑row stride inside a slice

    # output strides
    stride_out_i = N * hidden
    stride_out_j = hidden

    # tile sizes – tuned for Hopper (sm90); 128 is a good default
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden,                     # bh dimension
        triton.cdiv(N, BLOCK_M),        # row tiles
        triton.cdiv(N, BLOCK_N),        # col tiles
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                                   # M,N,K
        stride_BH, stride_i,                       # left strides
        stride_BH, stride_j, stride_k,             # right strides (bh, j, k)
        stride_BH, stride_i, stride_j,             # out strides (bh, i, j)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    out = out_flat.view(B, N, N, hidden)            # back to [B,N,N,hidden]

    # ---------- Output norm and final linear ----------
    out = F.layer_norm(out, (hidden,), weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"], eps=eps)
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # ensure float32 output
    return out.to(torch.float32)
