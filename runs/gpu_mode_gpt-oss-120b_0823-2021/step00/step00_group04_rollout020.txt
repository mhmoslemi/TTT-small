"""
TriMul “outgoing” implementation.

The heavy part is the 4‑D einsum:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

For each hidden channel *h* we have a matrix‑multiply
    A = left[b,:, :,h]          # (N, N)
    B = right[b,:, :,h]         # (N, N)
    out = A @ B.T               # (N, N)

We fuse the batched matrix‑multiply (B·H independent matrices) with a
single Triton kernel.  All LayerNorms, linear projections, and gates are
performed with regular PyTorch (they are cheap compared to the N³ work).
The kernel works in fp16 for the accumulation (fp32 accumulator) and returns
fp16 which is later cast back to fp32 for the remaining PyTorch ops.

The kernel processes blocks of M×N output tiles, each thread‑block handling
one tile of the (i, j) plane for a given (batch, hidden) pair.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   where A,B : (M, K)  (here M=N, K=N)
# --------------------------------------------------------------
@triton.jit
def matmul_transpose_kernel(
    a_ptr,            # *const* fp16 A
    b_ptr,            # *const* fp16 B
    c_ptr,            # *mut*   fp16 C
    M, N, K,          # matrix sizes (all N in our case)
    stride_am, stride_ak,          # A strides (batch*H dim collapsed)
    stride_bk, stride_bn,          # B strides (note we will read B transposed)
    stride_cm, stride_cn,          # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # each program works on a distinct (batch*hidden) matrix
    batch_hidden = tl.num_programs(0)
    # offsets for this matrix
    a_batch = pid // tl.cdiv(N, BLOCK_M)   # not really used, matrix is contiguous per pid
    # compute tile offsets
    pid_m = (pid % tl.cdiv(N, BLOCK_M)) // tl.cdiv(N, BLOCK_N)
    pid_n = (pid % tl.cdiv(N, BLOCK_M)) % tl.cdiv(N, BLOCK_N)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of this (batch*hidden) matrix
    a_ptr = a_ptr + pid * stride_am  # stride_am = H*B * M * K (row‑major)
    b_ptr = b_ptr + pid * stride_bk  # stride_bk = H*B * K * N

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A: (m, k)
        a = tl.load(
            a_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # Bᵀ: we need B[j, k] -> we load B as (n, k) because we will transpose later
        b = tl.load(
            b_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float16)

        # dot product over k
        acc += tl.dot(a, b)          # (m,n) += (m,k)·(k,n)

        # advance pointers
        a_ptr += BLOCK_K * stride_ak
        b_ptr += BLOCK_K * stride_bn

    # write back fp16
    c = acc.to(tl.float16)
    c_ptr = c_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptr, c, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Helper that launches the kernel for a batch of (B*H) matrices
# --------------------------------------------------------------
def batched_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: (B, N, N, H)  float16
    returns:    (B, N, N, H)  float16   (out = left @ rightᵀ per hidden channel)
    """
    B, N, _, H = left.shape
    # move hidden dim to batch dimension for kernel simplicity
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left = left.view(-1, N, N)   # (B*H, N, N)
    right = right.view(-1, N, N)

    # strides in elements (not bytes)
    stride_am = left.stride(0) * left.element_size()
    stride_ak = left.stride(2) * left.element_size()
    stride_bk = right.stride(0) * right.element_size()
    stride_bn = right.stride(2) * right.element_size()
    stride_cm = left.stride(0) * left.element_size()   # output same layout as left
    stride_cn = left.stride(2) * left.element_size()

    M = N
    K = N
    out = torch.empty_like(left)

    # tile sizes – tuned for Hopper (sm_90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_mats = left.shape[0]                     # B*H
    grid = (total_mats,)

    matmul_transpose_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)
    return out


# --------------------------------------------------------------
# Full custom kernel entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Parts of the computation (LayerNorm, Linear, gating) are done in PyTorch,
    while the cubic‑cost einsum is fused into a single Triton kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ---------- LayerNorm ----------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_w + norm_b                     # [B,N,N,dim]

    # ---------- Linear projections (no bias) ----------
    left_proj_w = weights["left_proj.weight"].to(device)            # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)          # (hidden, dim)

    # flatten last dim for matmul
    x_flat = x.reshape(-1, dim)                                     # (B*N*N, dim)
    left = torch.nn.functional.linear(x_flat, left_proj_w).view_as(x)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x_flat, right_proj_w).view_as(x)

    # ---------- Mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                               # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gates ----------
    lgate_w = weights["left_gate.weight"].to(device)
    rgate_w = weights["right_gate.weight"].to(device)
    outgate_w = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, lgate_w)).view_as(x)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, rgate_w)).view_as(x)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, outgate_w)).view_as(x)

    left = left * left_gate
    right = right * right_gate

    # ---------- Heavy cubic part: einsum ----------
    # Cast to fp16 for the kernel (H200 has excellent fp16 performance)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    out = batched_matmul_transpose(left_h, right_h)   # (B,N,N,hidden) fp16
    out = out.to(torch.float32)

    # ---------- Output norm & linear ----------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    # LayerNorm on hidden dim
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden)
    out_flat = out.reshape(-1, hidden_dim)
    out = torch.nn.functional.linear(out_flat, to_out_w).view_as(input_tensor)  # (B,N,N,dim)

    return out
