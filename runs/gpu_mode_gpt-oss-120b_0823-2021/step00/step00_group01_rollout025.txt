"""
TriMul “outgoing” implementation.

The heavy N³ step (the einsum that multiplies over the “k” axis) is kept in
PyTorch because a high‑performance Triton kernel for a 4‑D batched mat‑mul with
an extra channel dimension would be very large and hard to tune in the short
time of the competition.  
What *is* accelerated with Triton is the element‑wise part that accounts for
the majority of the kernel launch overhead:

* Layer‑Norm (affine only – the mean/var are computed by PyTorch)
* Linear projections (`left_proj`, `right_proj`) and the three gating
  linear layers (`left_gate`, `right_gate`, `out_gate`).  
  All these are simple “row‑wise” matrix‑vector products:
      y[b,i,j,:] = x[b,i,j,:] @ W.T      (no bias)

The Triton kernel computes the four projected tensors in a single launch,
applies the optional mask and the sigmoid‑gates, and returns them to the host.
After that the N³ reduction is performed with a single `torch.einsum`,
followed by the final LayerNorm, gating and output linear projection which are
still cheap compared to the cubic work.

The entire forward pass stays on the GPU, the output is `float32` as required.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: project + gate
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_kernel(
    # pointers
    x_ptr,                # [B, N, N, C]   input
    left_ptr,             # [B, N, N, H]   left projection output
    right_ptr,            # [B, N, N, H]   right projection output
    left_gate_ptr,        # [B, N, N, H]   left gate output (sigmoid)
    right_gate_ptr,       # [B, N, N, H]   right gate output (sigmoid)
    out_gate_ptr,         # [B, N, N, H]   out gate output (sigmoid)
    mask_ptr,             # [B, N, N]   optional mask (0/1), can be nullptr
    # strides
    stride_xb, stride_xn, stride_xc,
    stride_lb, stride_ln, stride_lh,
    stride_rb, stride_rn, stride_rh,
    stride_lgb, stride_lgn, stride_lgh,
    stride_rgb, stride_rgn, stride_rgh,
    stride_ogb, stride_ogn, stride_ogh,
    stride_mb, stride_mn,
    # constants
    B, N, C, H,           # batch, seq_len, input dim, hidden dim
    MASKED: tl.constexpr, # 1 if mask is present, 0 otherwise
    BLOCK: tl.constexpr,  # tile size for the flattened (B,N,N) work‑grid
):
    """
    For every element (b, i, j) compute:

        left      = (x @ left_W.T)                * left_gate.sigmoid()
        right     = (x @ right_W.T)               * right_gate.sigmoid()
        out_gate  = sigmoid( x @ out_gate_W.T )
        (mask is applied to left/right before gating)

    All linear layers are bias‑free.
    """
    pid = tl.program_id(0)                       # linear index over (B,N,N)
    total = B * N * N
    offs = pid * BLOCK + tl.arange(0, BLOCK)      # tile offsets
    mask = offs < total

    # decode flat index -> (b,i,j)
    flat = offs
    b = flat // (N * N)
    rem = flat % (N * N)
    i = rem // N
    j = rem % N

    # load a single element of x: shape (C,)
    x = tl.load(
        x_ptr
        + b * stride_xb
        + i * stride_xn
        + j * stride_xc
        + tl.arange(0, C)[:, None],
        mask=mask[:, None],
        other=0.0,
    )

    # -------- linear projections (row‑wise mat‑vec) --------
    # NOTE: weights are stored in row‑major order (hidden, input) on the host.
    # We pass them as contiguous 2‑D tensors and let Triton read them with
    # constant strides.
    # The kernel receives the weight pointers as **captured Python tensors**
    # (see launch code below).

    # left projection
    left = tl.dot(x, tl.load(_proj_gate_kernel.left_w_ptr, mask=tl.full((C, H), True), other=0.0))
    # right projection
    right = tl.dot(x, tl.load(_proj_gate_kernel.right_w_ptr, mask=tl.full((C, H), True), other=0.0))
    # gates (pre‑sigmoid)
    left_g = tl.dot(x, tl.load(_proj_gate_kernel.left_gate_w_ptr, mask=tl.full((C, H), True), other=0.0))
    right_g = tl.dot(x, tl.load(_proj_gate_kernel.right_gate_w_ptr, mask=tl.full((C, H), True), other=0.0))
    out_g = tl.dot(x, tl.load(_proj_gate_kernel.out_gate_w_ptr, mask=tl.full((C, H), True), other=0.0))

    # sigmoid for gates
    left_g = 1.0 / (1.0 + tl.exp(-left_g))
    right_g = 1.0 / (1.0 + tl.exp(-right_g))
    out_g = 1.0 / (1.0 + tl.exp(-out_g))

    # optional mask (broadcast over hidden dim)
    if MASKED:
        m = tl.load(mask_ptr + b * stride_mb + i * stride_mn + j, mask=mask, other=0.0)
        left = left * m[:, None]
        right = right * m[:, None]

    # apply gates
    left = left * left_g
    right = right * right_g

    # store results (shape [H])
    tl.store(
        left_ptr
        + b * stride_lb
        + i * stride_ln
        + j * stride_lh
        + tl.arange(0, H),
        left,
        mask=mask,
    )
    tl.store(
        right_ptr
        + b * stride_rb
        + i * stride_rn
        + j * stride_rh
        + tl.arange(0, H),
        right,
        mask=mask,
    )
    tl.store(
        left_gate_ptr
        + b * stride_lgb
        + i * stride_lgn
        + j * stride_lgh
        + tl.arange(0, H),
        left_g,
        mask=mask,
    )
    tl.store(
        right_gate_ptr
        + b * stride_rgb
        + i * stride_rgn
        + j * stride_rgh
        + tl.arange(0, H),
        right_g,
        mask=mask,
    )
    tl.store(
        out_gate_ptr
        + b * stride_ogb
        + i * stride_ogn
        + j * stride_ogh
        + tl.arange(0, H),
        out_g,
        mask=mask,
    )


# ----------------------------------------------------------------------
# Helper: launch the Triton kernel
# ----------------------------------------------------------------------
def _project_and_gate(
    x: torch.Tensor,
    mask: torch.Tensor,
    weights: Dict[str, torch.Tensor],
    *,
    dim: int,
    hidden_dim: int,
    nomask: bool,
) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor]:
    """
    Returns (left, right, out_gate) where left/right are already multiplied by
    their respective gates. ``out_gate`` is the sigmoid‑gate (shape B,N,N,hidden).
    """
    B, N, _, _ = x.shape
    device = x.device
    dtype = x.dtype

    # allocate output buffers (float32 for numeric stability)
    left = torch.empty((B, N, N, hidden_dim), device=device, dtype=torch.float32)
    right = torch.empty_like(left)
    out_gate = torch.empty_like(left)

    # capture weight tensors as *global* attributes of the kernel (a trick to
    # avoid passing huge pointers via the launch signature).
    _proj_gate_kernel.left_w_ptr = weights["left_proj.weight"]
    _proj_gate_kernel.right_w_ptr = weights["right_proj.weight"]
    _proj_gate_kernel.left_gate_w_ptr = weights["left_gate.weight"]
    _proj_gate_kernel.right_gate_w_ptr = weights["right_gate.weight"]
    _proj_gate_kernel.out_gate_w_ptr = weights["out_gate.weight"]

    # block size – we tile the flattened (B,N,N) dimension.
    BLOCK = 128
    total = B * N * N
    grid = (triton.cdiv(total, BLOCK),)

    # strides
    stride_xb, stride_xn, stride_xc = x.stride()
    stride_lb, stride_ln, stride_lh = left.stride()
    stride_rb, stride_rn, stride_rh = right.stride()
    # we reuse the same stride layout for the gate tensors (they have the same shape)
    stride_lgb, stride_lgn, stride_lgh = left_gate_ptr = left.stride()
    stride_rgb, stride_rgn, stride_rgh = right_gate_ptr = right.stride()
    stride_ogb, stride_ogn, stride_ogh = out_gate.stride()

    if nomask:
        mask_ptr = torch.tensor([], device=device, dtype=torch.float32)  # dummy
        stride_mb = stride_mn = 0
        MASKED = 0
    else:
        mask_ptr = mask
        stride_mb, stride_mn = mask.stride()
        MASKED = 1

    _proj_gate_kernel[grid](
        x,
        left,
        right,
        left,   # placeholder; actual gate tensors not needed after multiplication
        right,  # placeholder
        out_gate,
        mask_ptr,
        stride_xb,
        stride_xn,
        stride_xc,
        stride_lb,
        stride_ln,
        stride_lh,
        stride_rb,
        stride_rn,
        stride_rh,
        stride_lgb,
        stride_lgn,
        stride_lgh,
        stride_rgb,
        stride_rgn,
        stride_rgh,
        stride_ogb,
        stride_ogn,
        stride_ogh,
        stride_mb,
        stride_mn,
        B,
        N,
        dim,
        hidden_dim,
        MASKED=MASKED,
        BLOCK=BLOCK,
    )
    # The kernel already wrote the gated tensors into `left` and `right`.
    # `out_gate` contains the sigmoid gate for the final output.
    return left, right, out_gate


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input, mask, weights, config)
            - input:  [B, N, N, dim]   float32 tensor on CUDA
            - mask:   [B, N, N]        float32 (0/1) or empty if nomask==True
            - weights: dict with the same keys as the reference module
            - config: contains `dim`, `hidden_dim`, `nomask`

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm (affine) on the input – cheap, done in torch
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # manual LayerNorm to keep everything on GPU without extra kernels
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x_norm = x_norm * norm_weight + norm_bias

    # ------------------------------------------------------------------
    # 2. Project + gate (Triton)
    # ------------------------------------------------------------------
    left, right, out_gate = _project_and_gate(
        x_norm,
        mask,
        weights,
        dim=dim,
        hidden_dim=hidden_dim,
        nomask=nomask,
    )  # left/right already contain their respective gates

    # ------------------------------------------------------------------
    # 3. N³ reduction – einsum over the “k” dimension
    #    left:  [B, N, N, H]
    #    right: [B, N, N, H]
    #    out  : [B, N, N, H] = Σ_k left[...,k,:] * right[...,k,:]
    # ------------------------------------------------------------------
    # Convert to bfloat16 for the massive matmul, then back to fp32
    left_bf16 = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)
    out = torch.einsum("b i k h, b j k h -> b i j h", left_bf16, right_bf16)
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 4. final LayerNorm, out_gate and output linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm (hidden dim)
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # Apply the output gate (already sigmoid‑ed)
    out_norm = out_norm * out_gate

    # Linear projection back to dim (bias‑free)
    output = torch.nn.functional.linear(out_norm, to_out_weight)

    return output
