"""
TriMul “outgoing” implementation.

The heavy part of TriMul is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where
    left  = (x @ W_left)   * mask * sigmoid(x @ W_left_gate)
    right = (x @ W_right)  * mask * sigmoid(x @ W_right_gate)

Both left and right have shape [B, N, N, H] (H = hidden_dim).  The
contraction is an N×N matrix‑multiply for every hidden channel d.
We implement this reduction in Triton (a fused GEMM over the k‑axis) and
perform the surrounding linear / layer‑norm / gating work in PyTorch.
The kernel operates on a tile of (i, j, d) and streams the k‑dimension
in blocks, accumulating in FP32.  The final output is cast back to
float32 (the kernel already works in FP32, but we keep the signature
compatible with mixed‑precision inputs).

The overall algorithm
    1. Layer‑norm on the input (torch, because the statistics are cheap).
    2. Linear projections + gating (torch, using the provided weight tensors).
    3. Triton kernel that computes the N‑by‑N “outer‑product” for each
       hidden channel.
    4. Output layer‑norm + out‑gate + final linear (torch).

The Triton kernel is the only non‑trivial piece – everything else is a
few torch ops, which keeps the implementation simple while still giving
a large speed‑up over the naïve `einsum` version.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# ----------------------------------------------------------------------
# Triton kernel: batched outer‑product over the k‑axis
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # const float*   [B, N, N, H]
    right_ptr,              # const float*   [B, N, N, H]
    out_ptr,                # float*         [B, N, N, H]
    B, N, H,
    stride_left_bs, stride_left_in, stride_left_jn, stride_left_hd,
    stride_right_bs, stride_right_in, stride_right_jn, stride_right_hd,
    stride_out_bs, stride_out_in, stride_out_jn, stride_out_hd,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] """
    pid = tl.program_id(0)          # linear index over (b, i_tile, j_tile, h_tile)
    num_b = B
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    num_h = tl.cdiv(H, BLOCK_H)

    # decode pid
    tmp = pid
    h_tile = tmp % num_h;        tmp = tmp // num_h
    j_tile = tmp % num_j;        tmp = tmp // num_j
    i_tile = tmp % num_i;        b = tmp // num_i

    # tile offsets
    i_offset = i_tile * BLOCK_I
    j_offset = j_tile * BLOCK_J
    h_offset = h_tile * BLOCK_H

    # ranges of the tile
    i_range = i_offset + tl.arange(0, BLOCK_I)
    j_range = j_offset + tl.arange(0, BLOCK_J)
    h_range = h_offset + tl.arange(0, BLOCK_H)

    i_mask = i_range < N
    j_mask = j_range < N
    h_mask = h_range < H

    # allocate accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over k in chunks
    for k_base in range(0, N, BLOCK_K):
        k_range = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # load left[b,i,k,h]
        left_ptrs = left_ptr + \
            b * stride_left_bs + \
            i_range[:, None, None] * stride_left_in + \
            k_range[None, :, None] * stride_left_jn + \
            h_range[None, None, :] * stride_left_hd
        left = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :], other=0.0)

        # load right[b,j,k,h]
        right_ptrs = right_ptr + \
            b * stride_right_bs + \
            j_range[:, None, None] * stride_right_in + \
            k_range[None, :, None] * stride_right_jn + \
            h_range[None, None, :] * stride_right_hd
        right = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :], other=0.0)

        # accumulate product over k (broadcasted reduction)
        # left: (I, K, H) , right: (J, K, H) -> we need sum_k left[i,k,h] * right[j,k,h]
        # We compute a batched dot for each h via tl.dot
        for h in range(0, BLOCK_H):
            # slice h dimension
            l = tl.view(left[:, :, h], (BLOCK_I, BLOCK_K))
            r = tl.view(right[:, :, h], (BLOCK_J, BLOCK_K))
            # tl.dot expects (M, K) @ (K, N) -> (M, N)
            # we need (I, K) dot (J, K).transpose second operand.
            acc[:, :, h] += tl.dot(l, r, trans_b=True)   # (I,J)

    # store result
    out_ptrs = out_ptr + \
        b * stride_out_bs + \
        i_range[:, None, None] * stride_out_in + \
        j_range[None, :, None] * stride_out_jn + \
        h_range[None, None, :] * stride_out_hd
    tl.store(out_ptrs,
             acc.to(tl.float32),
             mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :])


# ----------------------------------------------------------------------
# Helper: launch the Triton kernel
# ----------------------------------------------------------------------
def trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right:  [B, N, N, H] float32
    returns out of same shape.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in element units
    stride_left_bs, stride_left_in, stride_left_jn, stride_left_hd = left.stride()
    stride_right_bs, stride_right_in, stride_right_jn, stride_right_hd = right.stride()
    stride_out_bs, stride_out_in, stride_out_jn, stride_out_hd = out.stride()

    # tile sizes – chosen to fit H200 shared memory and give good occupancy
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32
    BLOCK_K = 64   # stream k in 64‑element chunks

    grid = (
        triton.cdiv(B, 1) *
        triton.cdiv(N, BLOCK_I) *
        triton.cdiv(N, BLOCK_J) *
        triton.cdiv(H, BLOCK_H),
    )  # one‑dim grid; pid encodes (b,i_tile,j_tile,h_tile)

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_bs, stride_left_in, stride_left_jn, stride_left_hd,
        stride_right_bs, stride_right_in, stride_right_jn, stride_right_hd,
        stride_out_bs, stride_out_in, stride_out_jn, stride_out_hd,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full custom kernel (forward only)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]  (float32 or bfloat16)
            - mask:          torch.Tensor [B, N, N]   (bool / float)
            - weights:       dict of model weight tensors
            - config:        dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim = C)
    # ------------------------------------------------------------------
    # Manual LayerNorm to avoid extra Python overhead
    # x_norm = (x - mean) / sqrt(var + eps) * weight + bias
    eps = 1e-5
    weight_ln = weights["norm.weight"].to(device)
    bias_ln = weights["norm.bias"].to(device)

    # compute mean/var over last dimension
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * weight_ln + bias_ln

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) + mask
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden_dim]
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # apply mask if present (mask may be bool or float)
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Triton kernel for the N×N contraction over hidden_dim
    # ------------------------------------------------------------------
    # left/right are now [B,N,N,H] float32
    out = trimul_outgoing_triton(left, right)   # result in float32

    # ------------------------------------------------------------------
    # 5. Output LayerNorm + out‑gate + final linear
    # ------------------------------------------------------------------
    # to_out_norm over hidden_dim
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - out_mean) / torch.sqrt(out_var + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    out = out * out_gate

    # final projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
