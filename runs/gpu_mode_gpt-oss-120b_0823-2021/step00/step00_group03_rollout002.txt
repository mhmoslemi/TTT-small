"""
 Triton‐accelerated “outgoing” TriMul (AlphaFold3) forward pass.

 The heavy part of the module is the 4‑D Einstein sum

        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

 where left/right have shape (B,N,N,H).  For each channel *d* this is a
 matrix‑multiply of the (i,k) slice of *left* with the (j,k) slice of *right*.
 The kernel tiles the i‑ and j‑ dimensions (BLOCK_M × BLOCK_N) and the
 reduction dimension *k* (BLOCK_K).  Inputs are cast to fp16 for
 bandwidth, the accumulation is performed in fp32 and the final output is
 written as fp32.  All LayerNorms, Linear projections and gating are kept
 in PyTorch – they are cheap compared with the N³ reduction – and only the
 einsum‑style reduction is off‑loaded to Triton.

 The kernel works for any batch size ≤ 2, any sequence length N that fits
 into the tile grid and any hidden dimension H (the reference config
 uses H = 128).  It respects an optional mask that is applied before the
 reduction.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: computes the N³ reduction for one batch element.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,               # ptr to left  [B,N,N,H]  (fp16)
    right_ptr,              # ptr to right [B,N,N,H]  (fp16)
    out_ptr,                # ptr to output [B,N,N,H] (fp32)

    B, N, H,                # problem size
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_M: tl.constexpr,  # tile size for i   (rows)
    BLOCK_N: tl.constexpr,  # tile size for j   (cols)
    BLOCK_K: tl.constexpr,  # tile size for reduction over k
):
    ############################################################
    # program IDs:                         (grid_x, grid_y, grid_z)
    #   grid_x : i‑tile index
    #   grid_y : j‑tile index
    #   grid_z : batch index
    ############################################################
    pid_i = tl.program_id(0)          # tile row
    pid_j = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # batch

    # ----- compute global offsets for the i‑ and j‑ dimensions -----
    i_base = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_base < N
    mask_j = j_base < N

    # ----- accumulator in fp32 (shape: BLOCK_M × BLOCK_N × H) -----
    acc = tl.zeros((BLOCK_M, BLOCK_N, H), dtype=tl.float32)

    # ----- iterate over k in tiles --------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_base = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_base < N

        # load left:   shape (BLOCK_M, BLOCK_K, H)
        left_offset = (
            pid_b * stride_left_b
            + i_base[:, None, None] * stride_left_i
            + k_base[None, :, None] * stride_left_k
            + tl.arange(0, H)[None, None, :] * stride_left_h
        )
        left = tl.load(left_ptr + left_offset,
                       mask=mask_i[:, None] & mask_k[None, :] ,
                       other=0.0).to(tl.float32)          # fp16 → fp32

        # load right:  shape (BLOCK_N, BLOCK_K, H)
        # note that right is indexed as (batch, j, k, h)
        right_offset = (
            pid_b * stride_right_b
            + j_base[:, None, None] * stride_right_j
            + k_base[None, :, None] * stride_right_k
            + tl.arange(0, H)[None, None, :] * stride_right_h
        )
        right = tl.load(right_ptr + right_offset,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)         # fp16 → fp32

        # ---- dot‑product over k for every channel h -------------
        # left : (M, K, H)   → (M, K, 1, H)
        # right: (N, K, H)   → (1, N, K, H)
        # after broadcasting we multiply and sum over K
        # we use tl.dot for each channel separately (vector‑dot)
        for h in range(0, H):
            # (M, K) dot (N, K).T   → (M, N)
            acc[:, :, h] += tl.dot(left[:, :, h], right[:, :, h].transpose())

    # ----- write result -------------------------------------------------
    out_offset = (
        pid_b * stride_out_b
        + i_base[:, None, None] * stride_out_i
        + j_base[None, :, None] * stride_out_j
        + tl.arange(0, H)[None, None, :] * stride_out_h
    )
    tl.store(out_ptr + out_offset,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None])


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator with a Triton kernel
    for the N³ reduction.

    Parameters
    ----------
    data : tuple
        (input, mask, weights, config) where
        * input  : Tensor[B, N, N, dim]   (float32)
        * mask   : Tensor[B, N, N]        (bool / uint8)
        * weights: dict of model parameters (float32)
        * config : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]

    device = inp.device

    # ------------------------------------------------------------------
    # Helper: linear + layernorm + gate (all in PyTorch)
    # ------------------------------------------------------------------
    def linear(x, w):
        # x: [..., dim] , w: [out, dim]
        # bias is omitted (all layers are bias‑free except LayerNorm)
        return torch.nn.functional.linear(x, w)

    # LayerNorms (bias & weight are provided)
    def layer_norm(x, w, b):
        eps = 1e-5
        mean = x.mean(-1, keepdim=True)
        var = ((x - mean) ** 2).mean(-1, keepdim=True)
        return (x - mean) * torch.rsqrt(var + eps) * w + b

    # ----- normalisation of the input ----------------------------------
    x = layer_norm(inp,
                   weights["norm.weight"],
                   weights["norm.bias"])

    # ----- projections --------------------------------------------------
    left = linear(x, weights["left_proj.weight"])
    right = linear(x, weights["right_proj.weight"])

    # ----- mask (if present) -------------------------------------------
    if mask is not None:
        mask_u = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_u
        right = right * mask_u

    # ----- gating -------------------------------------------------------
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (bandwidth) – keep a copy in fp32 for
    # final operations.
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp32 = torch.empty((B, N, N, hidden_dim), dtype=torch.float32,
                           device=device)

    # ------------------------------------------------------------------
    # launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32                     # empirically good on Hopper

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B,
    )

    # compute strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_fp16.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_fp32.stride()

    _trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # post‑processing: to_out_norm -> gate -> linear back to dim
    # ------------------------------------------------------------------
    out = layer_norm(out_fp32,
                     weights["to_out_norm.weight"],
                     weights["to_out_norm.bias"])
    out = out * out_gate
    out = linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
