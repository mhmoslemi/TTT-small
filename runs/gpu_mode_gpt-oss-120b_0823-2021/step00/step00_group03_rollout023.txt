"""
TriMul "outgoing" kernel (AlphaFold3)

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched “matrix‑multiply‑over‑k” for every hidden dimension *d*.
The kernel below computes this operation on GPU with Triton, using
a classic GEMM tiling (BLOCK_M × BLOCK_N tiles over the (i,j) plane and
a BLOCK_K reduction over the summed dimension).  All other
operations (LayerNorm, Linear projections, gates, final LayerNorm + Linear)
are kept in PyTorch – this already gives a huge speed‑up because the
N³ work is off‑loaded to a fused Triton kernel.

The kernel works for arbitrary batch size, sequence length N and hidden
dimension H (called ``hidden_dim``).  It accepts the tensors in
*float16* (or *bfloat16*) for the matmul and returns *float32* to keep the
final precision identical to the reference implementation.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# ----------------------------------------------------------------------
# Triton kernel: batched matmul over the summed dimension (the einsum)
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # *[B, N, N, H]*
    right_ptr,              # *[B, N, N, H]*
    out_ptr,                # *[B, N, N, H]*
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,  # tile size over i
    BLOCK_N: tl.constexpr,  # tile size over j
    BLOCK_K: tl.constexpr,  # tile size over k
    FP16_ACC: tl.constexpr, # use fp16 accumulation?
):
    # --------------------------------------------------------------
    # 1) program ids – we tile over (batch, i, j)
    # --------------------------------------------------------------
    pid = tl.program_id(0)
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    b = pid // (num_i * num_j)                     # batch index
    pid_ij = pid % (num_i * num_j)
    i_tile = pid_ij // num_j                       # i‑tile index
    j_tile = pid_ij % num_j                         # j‑tile index

    # --------------------------------------------------------------
    # 2) compute start offsets for this tile
    # --------------------------------------------------------------
    i_offset = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds rows / cols
    mask_i = i_offset < N
    mask_j = j_offset < N

    # pointers to the beginning of the tile (no k yet)
    left_tile_ptr = left_ptr + \
        b * stride_left_b + \
        i_offset[:, None] * stride_left_i + \
        tl.arange(0, BLOCK_K)[None, :] * stride_left_k + \
        0 * stride_left_h               # h will be added later

    right_tile_ptr = right_ptr + \
        b * stride_right_b + \
        j_offset[:, None] * stride_right_j + \
        tl.arange(0, BLOCK_K)[None, :] * stride_right_k + \
        0 * stride_right_h

    # --------------------------------------------------------------
    # 3) accumulate over k
    # --------------------------------------------------------------
    # accumulator per hidden dimension -> shape (BLOCK_M, BLOCK_N, H)
    acc = tl.zeros((BLOCK_M, BLOCK_N, H), dtype=tl.float16 if FP16_ACC else tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # offsets for this K‑tile
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left:   [i, k, h]   -> broadcast k dim to j tile
        left = tl.load(
            left_tile_ptr + \
            k_offset[None, :] * stride_left_k + \
            tl.arange(0, H)[None, None, :] * stride_left_h,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # right:  [j, k, h]   -> broadcast k dim to i tile
        right = tl.load(
            right_tile_ptr + \
            k_offset[None, :] * stride_right_k + \
            tl.arange(0, H)[None, None, :] * stride_right_h,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0
        )

        # left shape (BLOCK_M, BLOCK_K, H)
        # right shape (BLOCK_N, BLOCK_K, H)
        # we need sum_k left[...,h] * right[...,h]
        # perform dot for each hidden dimension separately
        for h in range(H):
            a = left[:, :, h]   # (M, K)
            b = right[:, :, h]  # (N, K)
            # dot over K -> (M,N)
            acc[:, :, h] += tl.dot(a, b, allow_tf32=False)

        # advance pointers
        left_tile_ptr += BLOCK_K * stride_left_k
        right_tile_ptr += BLOCK_K * stride_right_k

    # --------------------------------------------------------------
    # 4) write back results
    # --------------------------------------------------------------
    out_tile_ptr = out_ptr + \
        b * stride_out_b + \
        i_offset[:, None] * stride_out_i + \
        j_offset[None, :] * stride_out_j + \
        tl.arange(0, H)[None, None, :] * stride_out_h

    tl.store(out_tile_ptr, acc.to(tl.float32), mask=mask_i[:, None, None] & mask_j[None, :, None])


# ----------------------------------------------------------------------
# Helper: launch the kernel
# ----------------------------------------------------------------------
def trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, H] float16 (or bfloat16)
    returns out  : [B, N, N, H] float32
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tunable block sizes – good defaults for H200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    FP16_ACC = True   # accumulate in fp16 then cast to fp32 (still accurate)

    # total number of tiles = B * ceil(N/BM) * ceil(N/BN)
    grid = (
        triton.cdiv(B * N * N, BLOCK_M * BLOCK_N),  # one program per (b,i,j) tile
    )
    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        FP16_ACC=FP16_ACC,
    )
    return out


# ----------------------------------------------------------------------
# Full custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]  float32
            - mask         : [B, N, N]       bool/float (0/1)
            - weights      : dict of Parameter tensors
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm on last dim ----------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias

    # ---------- Linear projections (no bias) ----------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask ----------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(x)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ---------- Gates ----------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Convert to FP16 / BF16 for the heavy matmul ----------
    left_h = left.to(torch.bfloat16)
    right_h = right.to(torch.bfloat16)

    # ---------- Triton kernel for the einsum ----------
    out = trimul_outgoing_triton(left_h, right_h)   # float32 result

    # ---------- Final LayerNorm + gating + linear ----------
    # to_out_norm
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate

    # final projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
