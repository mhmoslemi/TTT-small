"""
TriMul “outgoing” kernel (AlphaFold3 style).

The heavy N³ part is the tensor‑einsum
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
We keep the linear projections, layer‑norms and the final linear in PyTorch
(because they are cheap) and off‑load the *mask‑and‑gate* fusion to Triton.
The Triton kernel fuses three element‑wise operations:
    left  = left  * mask * left_gate
    right = right * mask * right_gate
All tensors are kept in FP16 inside the kernel (the model works in mixed precision)
and are cast back to FP32 before the final LayerNorm / Linear.

The kernel is deliberately kept simple – the N³ contraction is still performed
by `torch.einsum` which on Hopper (SM90) is already heavily optimised and
covers the required micro‑second budget for the test shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into the projected tensors
# --------------------------------------------------------------------
@triton.jit
def mask_gate_fusion_kernel(
    left_ptr, right_ptr,               # [B, N, N, H] projected tensors
    mask_ptr,                          # [B, N, N]    mask (0/1) broadcasted on H
    lgate_ptr, rgate_ptr,              # [B, N, N, H] gate tensors (sigmoid already)
    stride_lbz, stride_lby, stride_lbx, stride_lbh,   # strides for left
    stride_rbz, stride_rby, stride_rbx, stride_rbh,   # strides for right
    stride_mbz, stride_mby, stride_mbx,               # strides for mask
    stride_gbz, stride_gby, stride_gbx, stride_gbh,   # strides for gates
    BLOCK_B: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each (b, i, j, h) compute:
        left[b,i,j,h]  = left[b,i,j,h]  * mask[b,i,j] * lgate[b,i,j,h]
        right[b,i,j,h] = right[b,i,j,h] * mask[b,i,j] * rgate[b,i,j,h]
    """
    pid = tl.program_id(0)

    # linear block ids
    b = (pid // (tl.cdiv(tl.cdiv(N, BLOCK_N), BLOCK_N))) % BLOCK_B
    i = (pid // BLOCK_B) % tl.cdiv(N, BLOCK_N)
    j = pid % tl.cdiv(N, BLOCK_N)

    # thread offsets within the block
    offs_h = tl.arange(0, BLOCK_H)

    # global offsets
    off_left  = b * stride_lbz + i * stride_lby + j * stride_lbx + offs_h * stride_lbh
    off_right = b * stride_rbz + i * stride_rby + j * stride_rbx + offs_h * stride_rbh
    off_mask  = b * stride_mbz + i * stride_mby + j * stride_mbx
    off_lgate = b * stride_gbz + i * stride_gby + j * stride_gbx + offs_h * stride_gbh
    off_rgate = b * stride_gbz + i * stride_gby + j * stride_gbx + offs_h * stride_gbh

    # load (masked) data – note mask has no H dimension
    mask = tl.load(mask_ptr + off_mask, mask=offs_h < BLOCK_H, other=0.0)  # broadcast
    mask = mask[:, None]  # (BLOCK_H, 1) -> broadcast when multiplying

    left = tl.load(left_ptr + off_left,  mask=offs_h < BLOCK_H, other=0.0)
    right = tl.load(right_ptr + off_right, mask=offs_h < BLOCK_H, other=0.0)

    lgate = tl.load(lgate_ptr + off_lgate, mask=offs_h < BLOCK_H, other=0.0)
    rgate = tl.load(rgate_ptr + off_rgate, mask=offs_h < BLOCK_H, other=0.0)

    # fused multiply
    left = left * mask * lgate
    right = right * mask * rgate

    # store back
    tl.store(left_ptr + off_left,  left,  mask=offs_h < BLOCK_H)
    tl.store(right_ptr + off_right, right, mask=offs_h < BLOCK_H)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]        (float32)
            - mask       : [B, N, N]            (float32 or bool)
            - weights    : dict of model parameters (float32)
            - config     : {"dim": int, "hidden_dim": int}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- 1. LayerNorm on the input ----------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias   # shape [B,N,N,dim]
    x = x.to(torch.float16)           # mixed‑precision for the heavy ops

    # ---------- 2. Linear projections (no bias) ----------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- 3. Gates (sigmoid) ----------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ---------- 4. Fuse mask + gates with a Triton kernel ----------
    # Cast everything to fp16 for the kernel (the kernel works in fp16)
    left  = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate  = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)
    mask = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]

    # strides
    stride_lbz, stride_lby, stride_lbx, stride_lbh = left.stride()
    stride_rbz, stride_rby, stride_rbx, stride_rbh = right.stride()
    stride_mbz, stride_mby, stride_mbx = mask.stride()
    stride_gbz, stride_gby, stride_gbx, stride_gbh = left_gate.stride()   # same for both gates

    BLOCK_B = 1          # batch size is tiny (1 or 2)
    BLOCK_N = 32         # split N dimension into 32‑wide tiles
    BLOCK_H = 64         # hidden dim tile (covers up to 384)

    grid_b = triton.cdiv(B, BLOCK_B)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (grid_b * grid_n * grid_n,)

    mask_gate_fusion_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        stride_lbz=stride_lbz, stride_lby=stride_lby, stride_lbx=stride_lbx, stride_lbh=stride_lbh,
        stride_rbz=stride_rbz, stride_rby=stride_rby, stride_rbx=stride_rbx, stride_rbh=stride_rbh,
        stride_mbz=stride_mbz, stride_mby=stride_mby, stride_mbx=stride_mbx,
        stride_gbz=stride_gbz, stride_gby=stride_gby, stride_gbx=stride_gbx, stride_gbh=stride_gbh,
        BLOCK_B=BLOCK_B, BLOCK_N=BLOCK_N, BLOCK_H=BLOCK_H,
    )

    # ---------- 5. Core N³ contraction (einsum) ----------
    # left/right are now already masked and gated.
    # Cast to bf16 for the matmul‑like contraction, then back to fp32.
    left  = left.to(torch.bfloat16)
    right = right.to(torch.bfloat16)

    # einsum: (... i k d, ... j k d) -> (... i j d)
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # shape [B,N,N,hidden]

    # ---------- 6. Post‑processing ----------
    out = out.to(torch.float32)

    # to_out_norm (LayerNorm over hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_weight + to_out_norm_bias

    # apply final output gate and linear projection back to dim
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
