"""
TriMul “outgoing” implementation.
---------------------------------
We keep the cheap LayerNorm / Linear / gating ops in PyTorch (they are
memory‑bound) and accelerate the cubic “einsum”:
    out[b, i, j, d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
for every batch b and hidden channel d.

The operation is exactly a batched matrix‑multiply for each hidden
dimension:
    out_d = left_d @ right_d.transpose(-2,-1)

We fuse this computation in a single Triton kernel.  The kernel processes
a (BLOCK_M × BLOCK_N) tile of the output matrix for a fixed (batch,
hidden) pair while looping over K‑tiles.  All intermediate
accumulators are fp32; the final result is written as fp32.  The rest of
the TriMul pipeline (LayerNorm, Linear, gating, final LayerNorm+Linear)
remains in PyTorch.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel: batch‑wise, hidden‑wise matrix multiply
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # [B, N, N, H]   left tensor (already gated & masked)
    right_ptr,         # [B, N, N, H]   right tensor (already gated & masked)
    out_ptr,           # [B, N, N, H]   output
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    batch = tl.program_id(1)          # 0 .. B-1
    hidden = tl.program_id(2)         # 0 .. H-1

    # tile coordinates
    pid_m = pid // tl.cdiv(N, BLOCK_N)   # row tile index
    pid_n = pid %  tl.cdiv(N, BLOCK_N)   # col tile index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for boundary
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers for the current batch / hidden slice
    left_batch_ptr  = left_ptr  + batch * stride_left_b  + hidden * stride_left_h
    right_batch_ptr = right_ptr + batch * stride_right_b + hidden * stride_right_h
    out_batch_ptr   = out_ptr   + batch * stride_out_b   + hidden * stride_out_h

    # load left tile (i,k) and right tile (j,k) – note right needs transpose
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left:  (i,k)
        left_ptrs = (left_batch_ptr
                     + offs_m[:, None] * stride_left_i
                     + offs_k[None, :] * stride_left_k)
        left_tile = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right: (j,k) → we need (j,k) then multiply with left (i,k)
        # later we will do dot(left[i,k], right[j,k]) = sum_k left[i,k] * right[j,k]
        right_ptrs = (right_batch_ptr
                      + offs_n[:, None] * stride_right_j
                      + offs_k[None, :] * stride_right_k)
        right_tile = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # accumulate
        acc += tl.dot(left_tile, right_tile, trans_b=False)  # (M,K)·(K,N) → (M,N)

        # advance pointers
        # (handled automatically by indexing with offs_k each iteration)

    # write back
    out_ptrs = (out_batch_ptr
                + offs_m[:, None] * stride_out_i
                + offs_n[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------
# Helper to launch the kernel
# --------------------------------------------------------------
def launch_trimap_kernel(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]  (float32, already masked & gated)
    returns out : [B, N, N, H]   (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes) – torch uses contiguous layout (B,N,N,H)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (grid_m * grid_n, B, H)   # (tile id, batch, hidden)

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                # H200 works well with 4 warps
    )
    return out

# --------------------------------------------------------------
# Full custom kernel entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the “outgoing” TriMul.

    data = (input_tensor, mask, weights, config)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]            # input channel dimension
    hidden_dim = config["hidden_dim"]

    # ---------- 1. LayerNorm on last dimension ----------
    # Manual LayerNorm to avoid extra Python overhead
    eps = 1e-5
    # (B,N,N,dim)
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + eps
    )
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # ---------- 2. Linear projections ----------
    # shape after proj: (B,N,N,hidden_dim)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])      # bias=False
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- 3. Apply mask (if provided) ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ---------- 4. Gating ----------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- 5. Heavy cubic einsum (Triton) ----------
    # left/right are (B,N,N,hidden_dim) float32
    out = launch_trimap_kernel(left, right)   # still (B,N,N,hidden_dim)

    # ---------- 6. Post‑norm + gating + final linear ----------
    # to_out_norm (LayerNorm on hidden_dim)
    eps2 = 1e-5
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + eps2
    )
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    out = out * out_gate                     # broadcast over hidden dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
