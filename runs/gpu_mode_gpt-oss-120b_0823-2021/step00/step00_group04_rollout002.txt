"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the ternary einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We perform all cheap operations (layer‑norm, linear projections, masks and
gates) with PyTorch and fuse the N³ reduction into a single Triton kernel.
The kernel processes one batch at a time and tiles the (i,j) matrix with
BLOCK_I × BLOCK_J thread‑blocks while iterating over the reduction dimension
k in tiles of size BLOCK_K.  For each hidden‑dimension slice we use
`tl.dot` (i.e. a GEMM of shape (BLOCK_I, BLOCK_K) × (BLOCK_J, BLOCK_K)ᵀ)
and accumulate into a float32 accumulator.  The final result is written
as float32; intermediate loads are in fp16 for higher arithmetic throughput.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# ----------------------------------------------------------------------
# Triton kernel ---------------------------------------------------------
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,          # const * left [N, N, H]   (i,k,d)
    right_ptr,         # const * right[N, N, H]   (j,k,d)
    out_ptr,           # * out [N, N, H]          (i,j,d)
    N, H,
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_j, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    # -----------------
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute  out[i, j, h] = Σ_k left[i, k, h] * right[j, k, h]
    for a single batch element.
    """
    pid = tl.program_id(0)

    # tile coordinates
    i_tile = (pid // tl.cdiv(N, BLOCK_J)) * BLOCK_I
    j_tile = (pid % tl.cdiv(N, BLOCK_J)) * BLOCK_J

    # thread offsets within a tile
    i_off = tl.arange(0, BLOCK_I)
    j_off = tl.arange(0, BLOCK_J)
    h_off = tl.arange(0, BLOCK_H)

    # mask for out‑of‑bounds tiles
    i_valid = i_tile + i_off < N
    j_valid = j_tile + j_off < N
    h_valid = h_off < H

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over k
    # ------------------------------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        # range of k for this iteration
        k_off = tl.arange(0, BLOCK_K)
        k_valid = k_base + k_off < N

        # ---- load left[i, k, h] ------------------------------------------------
        left_ptrs = left_ptr + \
            (i_tile + i_off[:, None, None]) * stride_left_i + \
            (k_base + k_off[None, :, None]) * stride_left_k + \
            (h_off[None, None, :] ) * stride_left_h
        left = tl.load(left_ptrs,
                       mask=i_valid[:, None, None] & k_valid[None, :, None] & h_valid[None, None, :],
                       other=0.0).to(tl.float16)

        # ---- load right[j, k, h] ------------------------------------------------
        right_ptrs = right_ptr + \
            (j_tile + j_off[:, None, None]) * stride_right_j + \
            (k_base + k_off[None, :, None]) * stride_right_k + \
            (h_off[None, None, :] ) * stride_right_h
        right = tl.load(right_ptrs,
                        mask=j_valid[:, None, None] & k_valid[None, :, None] & h_valid[None, None, :],
                        other=0.0).to(tl.float16)

        # ---- compute dot over k for each hidden slice ---------------------------
        # left : (I, K, H)   right : (J, K, H)
        # we need Σ_k left[i,k,h] * right[j,k,h]   →   (I,J,H)
        # Perform a broadcasting multiply then sum over K.
        prod = tl.cast(left, tl.float32) * tl.cast(right, tl.float32)   # (I,K,H) * (J,K,H) -> broadcast to (I,J,K,H)
        # sum over K dimension
        acc += tl.sum(prod, axis=2)   # sum over K → shape (I,J,H)

    # ------------------------------------------------------------------
    # write result
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        (i_tile + i_off[:, None, None]) * stride_out_i + \
        (j_tile + j_off[None, :, None]) * stride_out_j + \
        (h_off[None, None, :] ) * stride_out_h

    tl.store(out_ptrs,
             acc,
             mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :])


# ----------------------------------------------------------------------
# Helper functions ------------------------------------------------------
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """Manual LayerNorm (no torch.nn to keep everything on GPU)."""
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    inv = torch.rsqrt(var + eps)
    return (x - mean) * inv * weight + bias


# ----------------------------------------------------------------------
# Entry point -----------------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, D]   torch.float32 on CUDA
        - mask         : [B, N, N]       torch.float32 (0/1) on CUDA
        - weights      : dict of torch tensors (all float32 on CUDA)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor : [B, N, N, D] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension
    # ------------------------------------------------------------------
    x = layer_norm(input_tensor,
                   weights["norm.weight"],
                   weights["norm.bias"])

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – we use torch.nn.functional for speed
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])        # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the ternary reduction
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # kernel launch parameters
    BLOCK_I, BLOCK_J, BLOCK_K, BLOCK_H = 32, 32, 32, 32
    grid_i = triton.cdiv(N, BLOCK_I)
    grid_j = triton.cdiv(N, BLOCK_J)
    grid = (grid_i * grid_j,)

    # strides for the three tensors (batch dimension is handled in a Python loop)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_j = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    left_ptr = left
    right_ptr = right
    out_ptr = out

    for b in range(B):
        trimul_einsum_kernel[grid](
            left_ptr + b * left.stride(0),
            right_ptr + b * right.stride(0),
            out_ptr + b * out.stride(0),
            N, hidden_dim,
            stride_left_i, stride_left_k, stride_left_h,
            stride_right_j, stride_right_k, stride_right_h,
            stride_out_i, stride_out_j, stride_out_h,
            BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
        )

    # ------------------------------------------------------------------
    # 6. Final LayerNorm, gating and output projection (still in PyTorch)
    # ------------------------------------------------------------------
    out = layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
