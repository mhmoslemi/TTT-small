"""
tri_mul_out.py
----------------
Custom Triton+PyTorch implementation of the *outgoing* TriMul operator
described in the AlphaFold‑3 paper.

Algorithm
~~~~~~~~~
1.  Layer‑norm + three linear projections (left/right projections and three
    gates) are performed with regular PyTorch – they are cheap compared to the
    N³ “pair‑wise” reduction.
2.  The heavy part is

        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

    (i.e. an einsum `'... i k d, ... j k d -> ... i j d'`).

    A Triton kernel computes this reduction.  For each block of (i,j) the
    kernel loads tiles of the k‑dimension, performs an element‑wise product,
    and accumulates the result in FP32.  The final tensor is written back in
    float‑32, after which the remaining LayerNorm‑/Linear‑post‑processing is
    done in PyTorch.

The kernel works for any batch size (B ≤ 2), sequence length N (128‑1024) and
hidden dimension H (normally 128).  The implementation is memory‑bound
efficient on Hopper (SM‑90) GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes the N³ reduction
# ----------------------------------------------------------------------
@triton.jit
def trimul_reduce_kernel(
    left_ptr,                     #   const * FP32   [B, N, N, H]
    right_ptr,                    #   const * FP32   [B, N, N, H]
    out_ptr,                      #   *   FP32   [B, N, N, H]
    B, N, H,                      #   int32
    stride_lb, stride_lk, stride_ld,   # strides for left  (B,N,K,H)
    stride_rb, stride_rk, stride_rd,   # strides for right (B,N,K,H)
    stride_ob, stride_oi, stride_od,   # strides for out   (B,N,N,H)
    BLOCK_M: tl.constexpr,        # tile size for i dimension
    BLOCK_N: tl.constexpr,        # tile size for j dimension
    BLOCK_K: tl.constexpr,        # tile size for k dimension
):
    pid_b = tl.program_id(2)                # batch index
    pid_i = tl.program_id(0)                # block‑i
    pid_j = tl.program_id(1)                # block‑j

    # --------------------------------------------------------------
    # Block offsets
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = tl.arange(0, H)                # hidden dimension (small, kept in register)

    # masks for boundary conditions
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # ------------------------------------------------------------------
    # Pointers base for this (b,i,j) tile
    # ------------------------------------------------------------------
    left_base = left_ptr + pid_b * stride_lb \
                + offs_i[:, None, None] * stride_lk \
                + 0 * stride_ld                # k will be added in loop
    right_base = right_ptr + pid_b * stride_rb \
                + offs_j[None, :, None] * stride_rk \
                + 0 * stride_rd                # k will be added in loop

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, H), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over k dimension in blocks
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for blk in range(0, num_k_blocks):
        offs_k = blk * BLOCK_K + tl.arange(0, BLOCK_K)

        mask_k = offs_k < N

        # load left[b, i, k, d]   -> shape (M, K, H)
        left_ptrs = left_base + offs_k[None, :, None] * stride_ld
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :] & mask_d[None, None, :], other=0.0)

        # load right[b, j, k, d]  -> shape (N, K, H)
        right_ptrs = right_base + offs_k[None, :, None] * stride_rd
        right = tl.load(right_ptrs, mask=mask_j[None, :] & mask_k[:, None] & mask_d[None, None, :], other=0.0)

        # left: (M, K, H)   right: (N, K, H)
        # we need product over K, keep H.
        # Transpose right to (K, N, H) for broadcasting
        right_t = tl.transposed(right, (1, 0, 2))   # now (K, N, H)

        # Multiply & accumulate: broadcasting M×K×H * K×N×H -> M×N×H
        # tl.dot works on 2‑D, so we do element‑wise multiply & sum manually.
        # Because H is small we can keep it in registers.
        # Expand dims for broadcasting:
        left_exp  = left[:, :, None, :]            # (M, K, 1, H)
        right_exp = right_t[None, :, :, :]        # (1, K, N, H)

        prod = left_exp * right_exp               # (M, K, N, H)
        acc += tl.sum(prod, axis=1)               # sum over K -> (M, N, H)

    # --------------------------------------------------------------
    # Write result
    # --------------------------------------------------------------
    out_ptrs = out_ptr + pid_b * stride_ob \
               + offs_i[:, None, None] * stride_oi \
               + offs_j[None, :, None] * stride_od \
               + offs_d[None, None, :] * 1
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

# ----------------------------------------------------------------------
# Host side helper
# ----------------------------------------------------------------------
def launch_trimul_reduce(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]  (float32)
    returns      : [B, N, N, H]  (float32)
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes)
    stride_lb, stride_lk, stride_ld = left.stride(0), left.stride(1), left.stride(3)
    stride_rb, stride_rk, stride_rd = right.stride(0), right.stride(1), right.stride(3)
    stride_ob, stride_oi, stride_od = out.stride(0), out.stride(1), out.stride(2)

    # Tile sizes (tuned for Hopper)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B,
    )
    trimul_reduce_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_lb, stride_lk, stride_ld,
        stride_rb, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out

# ----------------------------------------------------------------------
# Full custom kernel (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]    float32
        - mask_tensor  : [B, N, N]        (bool / 0‑1 float)
        - weights_dict : mapping of all weight tensors
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    output : torch.Tensor  [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (no bias handling – bias is supplied)
    # ------------------------------------------------------------------
    eps = 1e-5
    w = weights["norm.weight"]
    b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x_norm * w + b

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy N³ reduction with Triton
    # ------------------------------------------------------------------
    out = launch_trimul_reduce(left, right)          # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm + gating + final linear)
    # ------------------------------------------------------------------
    w_out = weights["to_out_norm.weight"]
    b_out = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out_norm * w_out + b_out

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
