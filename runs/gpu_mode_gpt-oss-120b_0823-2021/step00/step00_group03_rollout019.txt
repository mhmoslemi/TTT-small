"""
TriMul “outgoing” kernel (AlphaFold3 style)

We compute
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where
    left  = (x @ W_left )  * mask * sigmoid(x @ W_left_gate )
    right = (x @ W_right)  * mask * sigmoid(x @ W_right_gate)

The heavy part is the 4‑D “tri‑mul” contraction – a batched
matrix‑multiply with an extra feature dimension *d*.
The kernel below performs this contraction in Triton using
a classic block‑wise GEMM loop (blocks M×N×K with K‑reduction).
All other operations (layer‑norms, linear projections, gates,
final norm + linear) stay in PyTorch for simplicity.

The kernel works for arbitrary batch size *B*, sequence length *N*,
and hidden dimension *H* (called `hidden_dim` in the config).  It
produces FP32 output (intermediate accumulation is FP32, the loads
use the input dtype which may be FP16/BF16).
"""
import torch
import triton
import triton.language as tl
from torch import nn

# ----------------------------------------------------------------------
# Triton kernel:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # const float*  left  [B,N,N,H]
    right_ptr,         # const float*  right [B,N,N,H]
    out_ptr,           # float*        out   [B,N,N,H]
    B, N, H,           # int32 scalars
    stride_lb, stride_li, stride_lk, stride_lz,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rz,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oz,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_Z: tl.constexpr,  # block over hidden dim
):
    pid = tl.program_id(0)

    # --------- grid decomposition ----------
    # grid is (B, ceil(N/BM), ceil(N/BN), ceil(H/BZ))
    B_grid = tl.cdiv(B, 1)                # we keep B inside pid
    M_grid = tl.cdiv(N, BLOCK_M)
    N_grid = tl.cdiv(N, BLOCK_N)
    Z_grid = tl.cdiv(H, BLOCK_Z)

    total = B_grid * M_grid * N_grid * Z_grid
    # guard (should never happen)
    if pid >= total:
        return

    # decode pid -> (b, m, n, z)
    tmp = pid
    z = tmp % Z_grid;   tmp = tmp // Z_grid
    n = tmp % N_grid;   tmp = tmp // N_grid
    m = tmp % M_grid;   b = tmp // M_grid

    # --------- pointers to tile ----------
    offs_m = m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_z = z * BLOCK_Z + tl.arange(0, BLOCK_Z)

    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_z = offs_z < H

    # left:  [b, i, k, d]  -> (M, K, Z)
    left_tile_ptr = left_ptr + \
        b * stride_lb + \
        offs_m[:, None, None] * stride_li + \
        tl.arange(0, BLOCK_K)[None, :, None] * stride_lk + \
        offs_z[None, None, :] * stride_lz

    # right: [b, j, k, d] -> (N, K, Z)
    right_tile_ptr = right_ptr + \
        b * stride_rb + \
        offs_n[None, :, None] * stride_rj + \
        tl.arange(0, BLOCK_K)[None, :, None] * stride_rk + \
        offs_z[None, None, :] * stride_rz

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_Z), dtype=tl.float32)

    # --------- K‑reduction loop ----------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = cur_k < N

        # load a block of left and right (broadcast over Z)
        left = tl.load(left_tile_ptr,
                       mask=mask_m[:, None, None] & k_mask[None, :, None] & mask_z[None, None, :],
                       other=0.0)
        right = tl.load(right_tile_ptr,
                        mask=mask_n[None, :, None] & k_mask[None, :, None] & mask_z[None, None, :],
                        other=0.0)

        # matmul over K dimension (dot is over last two axes)
        # we need a dot for each Z slice → use tl.dot for each Z lane
        # trick: flatten M*K and N*K then dot, then reshape
        left_flat = left.reshape(BLOCK_M * BLOCK_K, BLOCK_Z)
        right_flat = right.reshape(BLOCK_N * BLOCK_K, BLOCK_Z)
        # accumulate per‑Z dot products
        # tl.dot works on 2‑D tensors, we compute each Z separately via broadcasting
        for zz in range(0, BLOCK_Z):
            l = left_flat[:, zz]
            r = right_flat[:, zz]
            acc[:, :, zz] += tl.dot(l, r, out_dtype=tl.float32).reshape(BLOCK_M, BLOCK_N)

        # advance pointers
        left_tile_ptr += BLOCK_K * stride_lk
        right_tile_ptr += BLOCK_K * stride_rk

    # --------- write back ----------
    out_tile_ptr = out_ptr + \
        b * stride_ob + \
        offs_m[:, None, None] * stride_oi + \
        offs_n[None, :, None] * stride_oj + \
        offs_z[None, None, :] * stride_oz

    tl.store(out_tile_ptr,
             acc.to(tl.float32),
             mask=mask_m[:, None, None] & mask_n[None, :, None] & mask_z[None, None, :])


# ----------------------------------------------------------------------
# Helper functions (layer‑norm & linear) – kept in PyTorch for simplicity
# ----------------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_hat = (x - mean) * torch.rsqrt(var + eps)
    return x_hat * weight + bias

def linear(x, weight):
    # x: [..., dim_in]   weight: [dim_out, dim_in]
    return torch.nn.functional.linear(x, weight)

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  (float32/float16/bfloat16)
        - mask         : [B, N, N]       (bool or 0/1 float)
        - weights      : dict of weight tensors (see docstring)
        - config       : dict with keys “dim”, “hidden_dim”
    Returns:
        out : [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- layer‑norm ----------
    x = layer_norm(input_tensor,
                   weights["norm.weight"],
                   weights["norm.bias"])

    # ---------- linear projections ----------
    # shape: [B,N,N,hidden_dim]
    left = linear(x, weights["left_proj.weight"])
    right = linear(x, weights["right_proj.weight"])

    # ---------- mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)      # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- gates ----------
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Triton tri‑mul ----------
    # ensure contiguous and same dtype (we keep original dtype for load)
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left, dtype=torch.float32)   # allocate float32 output

    # strides (in elements)
    stride_lb, stride_li, stride_lk, stride_lz = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rz = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oz = out.stride()

    # block sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_Z = 32   # hidden_dim is multiple of 32 in all test configs

    grid = (
        triton.cdiv(B, 1) *
        triton.cdiv(N, BLOCK_M) *
        triton.cdiv(N, BLOCK_N) *
        triton.cdiv(hidden_dim, BLOCK_Z),
    )
    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_lz=stride_lz,
        stride_rb=stride_rb,
        stride_rj=stride_rj,
        stride_rk=stride_rk,
        stride_rz=stride_rz,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_oz=stride_oz,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_Z=BLOCK_Z,
        num_warps=4,
    )

    # ---------- final norm, gate and linear ----------
    out = layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = linear(out, weights["to_out.weight"]).to(torch.float32)

    return out
