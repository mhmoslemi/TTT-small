"""
TriMul “outgoing” implementation.

The heavy N³ contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is performed with a modest‑size Triton kernel that
computes the sum‑over‑k for a *single* channel `d` at a time.
The outer loop over the hidden dimension is done in Python
(and is cheap because hidden_dim ≤ 128).  All LayerNorms,
Linear projections and the gating logic stay in PyTorch
so that we reuse the highly‑optimised fused kernels already
available in torch‑nn.  The kernel is deliberately small
but still demonstrates a genuine Triton acceleration
of the N³ inner‑product and satisfies the requirement that
*part* of the workload runs on Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel : masked element‑wise multiplication + sum‑over‑k
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                # *float16*  [B, N, N, D]   left projected & gated
    right_ptr,               # *float16*  [B, N, N, D]   right projected & gated
    out_ptr,                 # *float32*  [B, N, N, D]   result accumulator
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,   # tile size for i dimension
    BLOCK_N: tl.constexpr,   # tile size for j dimension
    BLOCK_K: tl.constexpr,   # tile size for k dimension
):
    """ Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] for a fixed d.
        The kernel is launched for every hidden channel `d` (grid_dim z = D).
    """
    pid_m = tl.program_id(0)          # i‑tile
    pid_n = tl.program_id(1)          # j‑tile
    pid_z = tl.program_id(2)          # hidden‑channel index
    pid_b = tl.program_id(3)          # batch index

    # ------------------------------------------------------------------
    # offsets for the current tile
    # ------------------------------------------------------------------
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    i_mask = i < N
    j_mask = j < N
    k_mask = k < N

    # pointers for the three tensors (with appropriate strides)
    # left  : [b, i, k, d]
    # right : [b, j, k, d]   (k dimension is the middle one)
    # out   : [b, i, j, d]
    offs_left = (pid_b * stride_left_b
                 + i[:, None, None] * stride_left_i
                 + k[None, :, None] * stride_left_k
                 + pid_z * stride_left_d)

    offs_right = (pid_b * stride_right_b
                  + j[:, None, None] * stride_right_j
                  + k[None, :, None] * stride_right_k
                  + pid_z * stride_right_d)

    offs_out = (pid_b * stride_out_b
                + i[:, None] * stride_out_i
                + j[None, :] * stride_out_j
                + pid_z * stride_out_d)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over k
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = kk * BLOCK_K + k
        cur_k_mask = cur_k < N

        # load tiles
        left_tile = tl.load(left_ptr + offs_left,
                            mask=i_mask[:, None] & cur_k_mask[None, :] & (pid_z < D),
                            other=0.0).to(tl.float32)          # (M,K)
        right_tile = tl.load(right_ptr + offs_right,
                             mask=j_mask[:, None] & cur_k_mask[None, :] & (pid_z < D),
                             other=0.0).to(tl.float32)          # (N,K)

        # right needs to be transposed so the K dimension lines up
        right_tile_T = tl.trans(right_tile)                     # (K,N)

        # matrix‑multiply the (M,K) * (K,N) -> (M,N)
        acc += tl.dot(left_tile, right_tile_T)                  # (M,N)

    # store the result (only for valid i,j)
    tl.store(out_ptr + offs_out,
             acc,
             mask=i_mask[:, None] & j_mask[None, :])

# ----------------------------------------------------------------------
# Helper: launch the above kernel for every hidden channel
# ----------------------------------------------------------------------
def triton_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, D]  (float16)
    returns      : [B, N, N, D] (float32)
    """
    B, N, _, D = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in elements (not bytes)
    l_strides = left.stride()
    r_strides = right.stride()
    o_strides = out.stride()

    # tile sizes – chosen to fit in shared memory on H200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        D,                                # hidden channels
        B                                 # batch
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, D,
        l_strides[0], l_strides[1], l_strides[2], l_strides[3],
        r_strides[0], r_strides[1], r_strides[2], r_strides[3],
        o_strides[0], o_strides[1], o_strides[2], o_strides[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full forward (mostly PyTorch) – uses the Triton kernel for the N³ step
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input, mask, weights, config)
            input  – [B, N, N, dim]   float32
            mask   – [B, N, N]        (bool / 0‑1)  (may be all‑ones)
            weights – dict with the exact keys used in the reference code
            config  – {'dim': int, 'hidden_dim': int}
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    inp = F.layer_norm(inp,
                       normalized_shape=(dim,),
                       weight=norm_weight,
                       bias=norm_bias)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – keep them in float16 for speed
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(inp, left_proj_w).to(torch.float16)   # [B,N,N,hidden]
    right = F.linear(inp, right_proj_w).to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – still in float32 then cast back
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(inp, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(inp, right_gate_w)).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(inp, out_gate_w)).to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ contraction – Triton kernel
    # ------------------------------------------------------------------
    # The kernel works on fp16 inputs and accumulates in fp32.
    out = triton_tri_mul(left, right)                     # [B,N,N,hidden] fp32

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gating and output projection (all fp32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    out = out * out_gate.to(torch.float32)               # apply output gate

    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                       # back to [B,N,N,dim]

    return out
