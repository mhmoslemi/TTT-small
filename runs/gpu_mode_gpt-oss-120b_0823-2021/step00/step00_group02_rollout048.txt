"""
TriMul (outgoing) – fast Triton implementation of the core N³ “einsum”
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

The surrounding LayerNorm / Linear / gating ops are performed in PyTorch
(because they are cheap compared to the N³ reduction).  Only the
triple‑sum is off‑loaded to a Triton kernel that:
  * processes one (b,i,j) tile per program instance,
  * reduces over the sequence dimension k in blocks,
  * vector‑wise multiplies/accumulates the hidden dimension (hidden_dim).

The kernel works for any batch size (≤2), any sequence length (≤1024) and
any hidden dimension that is a multiple of 32 (the default BLOCK_HD).  The
final output is cast back to float32.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# ----------------------------------------------------------------------
# Triton kernel: batched “dot‑over‑k” for the hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,                     # const float*  [B,N,N,HD]
    right_ptr,                    # const float*  [B,N,N,HD]
    out_ptr,                      #       float*  [B,N,N,HD]
    B, N, HD,                     # shape scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_K: tl.constexpr,        # reduction block over k
    BLOCK_HD: tl.constexpr,       # vector size in hidden dim
):
    """
    One program instance computes the full hidden‑dim vector for a single
    (b,i,j) triple.  The reduction over k is tiled with BLOCK_K.
    """
    pid = tl.program_id(0)                     # flattened (b,i,j)
    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i = ij // N
    j = ij % N

    # pointer bases for this (b,i,j)
    left_base  = left_ptr  + b * stride_left_b  + i * stride_left_i
    right_base = right_ptr + b * stride_right_b + j * stride_right_i
    out_base   = out_ptr   + b * stride_out_b   + i * stride_out_i + j * stride_out_j

    # accumulator in float32
    acc = tl.zeros([BLOCK_HD], dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # offsets within the current k‑block
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # load left[b,i,k,:]  -> (BLOCK_K, BLOCK_HD)
        left_ptrs = left_base + k_off[:, None] * stride_left_k + tl.arange(0, BLOCK_HD)[None, :] * stride_left_d
        left = tl.load(left_ptrs, mask=mask_k[:, None] & (tl.arange(0, BLOCK_HD) < HD), other=0.0)

        # load right[b,j,k,:] -> (BLOCK_K, BLOCK_HD)
        right_ptrs = right_base + k_off[:, None] * stride_right_k + tl.arange(0, BLOCK_HD)[None, :] * stride_right_d
        right = tl.load(right_ptrs, mask=mask_k[:, None] & (tl.arange(0, BLOCK_HD) < HD), other=0.0)

        # element‑wise multiply and sum over k
        # tl.dot works on 1‑D vectors, so we first reshape to (BLOCK_K, BLOCK_HD) and then
        # perform a reduction manually.
        #   Σₖ leftₖ·rightₖ   ==   tl.dot(left[:,d], right[:,d])  for each d
        #   we achieve this with tl.dot on the flattened view.
        acc += tl.sum(left * right, axis=0)      # shape (BLOCK_HD,)

    # store the result
    out_ptrs = out_base + tl.arange(0, BLOCK_HD) * stride_out_d
    tl.store(out_ptrs, acc.to(tl.float32), mask=tl.arange(0, BLOCK_HD) < HD)


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim]
            mask         : torch.Tensor [B, N, N]   (may be all‑ones)
            weights      : dict of parameter tensors
            config       : dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -------------------- 1. LayerNorm and linear projections (PyTorch) --------------------
    # Build a lightweight module to reuse nn.functional utilities
    # (no autograd needed – we operate in inference mode)
    # ----------------------------------------------------------------------
    # Helper for linear projection without bias
    def linear(x, w):
        # x: [..., dim]  , w: [out, dim]
        return torch.nn.functional.linear(x, w)

    # LayerNorm (no bias in AlphaFold implementation, but bias may be present)
    def layer_norm(x, weight, bias, eps=1e-5):
        mean = x.mean(-1, keepdim=True)
        var = ((x - mean) * (x - mean)).mean(-1, keepdim=True)
        return (x - mean) * torch.rsqrt(var + eps) * weight + bias

    # ----- Norm on input -------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"].to(torch.float32),
        weights["norm.bias"].to(torch.float32),
    ).to(torch.float32)

    # ----- Projections ----------------------------------------------------
    left  = linear(x, weights["left_proj.weight"].to(torch.float32))
    right = linear(x, weights["right_proj.weight"].to(torch.float32))

    # ----- Apply mask (if present) ----------------------------------------
    if mask is not None:
        # shape to [B,N,N,1] and broadcast
        mask_ = mask.unsqueeze(-1).to(torch.float32)
        left  = left  * mask_
        right = right * mask_

    # ----- Gating ---------------------------------------------------------
    left_gate  = torch.sigmoid(linear(x, weights["left_gate.weight"].to(torch.float32)))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"].to(torch.float32)))
    out_gate   = torch.sigmoid(linear(x, weights["out_gate.weight"].to(torch.float32)))

    left  = left  * left_gate
    right = right * right_gate

    # -------------------- 2. Core N³ reduction with Triton --------------------
    # Convert to contiguous FP16 for higher throughput; accumulation stays FP32.
    left  = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    out   = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Triton launch configuration
    BLOCK_K = 64                # reasonable reduction tile for N up to 1024
    BLOCK_HD = 32               # hidden_dim is a multiple of 32 in all configs
    total_programs = B * N * N
    grid = (total_programs,)

    trimul_einsum_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_i, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_K=BLOCK_K,
        BLOCK_HD=BLOCK_HD,
    )

    out = out.to(torch.float32)

    # -------------------- 3. Post‑processing (PyTorch) --------------------
    # to_out_norm
    out = layer_norm(
        out,
        weights["to_out_norm.weight"].to(torch.float32),
        weights["to_out_norm.bias"].to(torch.float32),
    )

    out = out * out_gate
    out = linear(out, weights["to_out.weight"].to(torch.float32))

    return out
