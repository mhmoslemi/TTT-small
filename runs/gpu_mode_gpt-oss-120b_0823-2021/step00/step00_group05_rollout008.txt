"""
Triton‑accelerated TriMul (outgoing) forward pass.

The heavy part of the module is the 4‑dimensional einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched GEMM over the “k” dimension for every hidden channel *d*.
We implement this contraction with a custom Triton kernel that tiles the (i,j)
plane (BLOCK_M×BLOCK_N) and reduces over k in blocks (BLOCK_K).  The hidden
dimension (hidden_dim) is small (≤128) so we load it as a whole vector and
accumulate in float32.  All other operations (LayerNorm, Linear, gating,
final Linear) are left in PyTorch for simplicity.

The kernel returns a float32 tensor of shape [B, N, N, hidden_dim] which is then
post‑processed exactly like the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# -------------------------------------------------------------
# Triton kernel for the einsum reduction
# -------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,               # *[B, N, N, H]  left tensor
    right_ptr,              # *[B, N, N, H]  right tensor
    out_ptr,                # *[B, N, N, H]  output
    B, N, H,                # sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,  # tile size for i dimension
    BLOCK_N: tl.constexpr,  # tile size for j dimension
    BLOCK_K: tl.constexpr   # tile size for reduction dimension k
):
    pid = tl.program_id(0)                 # combine batch and i tile
    pid_b = pid // tl.cdiv(N, BLOCK_M)     # batch index
    pid_i = (pid % tl.cdiv(N, BLOCK_M))    # i‑tile index

    pid_j = tl.program_id(1)               # j‑tile index

    # -----------------------------------------------------------------
    # Offsets for the current tile
    # -----------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, H)               # full hidden dim (vectorized)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # -----------------------------------------------------------------
    # Pointers to the (i, k, h) and (j, k, h) slices for this tile
    # -----------------------------------------------------------------
    # left : [B, i, k, h]
    # right: [B, j, k, h]  (note the second index is j after transpose)
    # We will iterate over k in blocks.
    acc = tl.zeros((BLOCK_M, BLOCK_N, H), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------------------------------------------------------------
        # Load left[b, i, k, h]  -> shape (BLOCK_M, BLOCK_K, H)
        # -------------------------------------------------------------
        left_ptrs = left_ptr + (
            pid_b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_h[None, None, :] * stride_left_h
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0)

        # -------------------------------------------------------------
        # Load right[b, j, k, h]  -> shape (BLOCK_N, BLOCK_K, H)
        # -------------------------------------------------------------
        right_ptrs = right_ptr + (
            pid_b * stride_right_b
            + offs_j[:, None, None] * stride_right_i   # right_i is actually the j‑dim after the einsum
            + offs_k[None, :, None] * stride_right_k
            + offs_h[None, None, :] * stride_right_h
        )
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0)

        # -------------------------------------------------------------
        # Accumulate: for each hidden dim vector do a dot over k
        #   acc[i, j, h] += Σ_k left[i, k, h] * right[j, k, h]
        # -------------------------------------------------------------
        # tl.dot works on 2‑D matrices, we collapse the hidden dim as the
        # inner dimension using broadcasting.
        for h_idx in range(0, H):
            l_vec = left[:, :, h_idx]      # (BLOCK_M, BLOCK_K)
            r_vec = right[:, :, h_idx]     # (BLOCK_N, BLOCK_K)
            acc[:, :, h_idx] += tl.dot(l_vec, r_vec, trans_b=True)  # (BLOCK_M, BLOCK_N)

    # -----------------------------------------------------------------
    # Store the result back
    # -----------------------------------------------------------------
    out_ptrs = out_ptr + (
        pid_b * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_h[None, None, :] * stride_out_h
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None])


# -------------------------------------------------------------
# Host‑side helper that launches the kernel
# -------------------------------------------------------------
def trimul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, H]  (float32)
    returns out   : [B, N, N, H]  (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes) – Triton expects element strides
    (sb, si, sk, sh) = left.stride()
    (rb, ri, rk, rh) = right.stride()
    (ob, oi, oj, oh) = out.stride()

    # Tunable block sizes – chosen for Hopper (SM90) shared‑memory capacity
    BLOCK_M = 64   # i‑tile
    BLOCK_N = 64   # j‑tile
    BLOCK_K = 32   # k‑reduction

    grid = (
        triton.cdiv(N, BLOCK_M) * B,   # program_id(0) combines batch and i‑tile
        triton.cdiv(N, BLOCK_N),       # program_id(1) is j‑tile
    )

    trimul_einsum_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        sb, si, sk, sh,
        rb, ri, rk, rh,
        ob, oi, oj, oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# -------------------------------------------------------------
# Full custom_kernel implementation
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the outgoing TriMul module.
    data = (input, mask, weights, config)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]               # input channel dimension
    hidden_dim = config["hidden_dim"] # projection dimension

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(input_tensor.dtype),
        bias=weights["norm.bias"].to(input_tensor.dtype),
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias) -> [B,N,N,hidden_dim]
    # -----------------------------------------------------------------
    # Linear = x @ W.T   where W has shape (hidden_dim, dim)
    # torch.nn.functional.linear expects weight of shape (out_features, in_features)
    left = F.linear(
        x,
        weights["left_proj.weight"].to(input_tensor.dtype),  # (hidden_dim, dim)
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(input_tensor.dtype),
        bias=None,
    )

    # -----------------------------------------------------------------
    # 3) Apply mask if present
    # -----------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise multiplication
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(
            x,
            weights["left_gate.weight"].to(input_tensor.dtype),
            bias=None,
        )
    )
    right_gate = torch.sigmoid(
        F.linear(
            x,
            weights["right_gate.weight"].to(input_tensor.dtype),
            bias=None,
        )
    )
    out_gate = torch.sigmoid(
        F.linear(
            x,
            weights["out_gate.weight"].to(input_tensor.dtype),
            bias=None,
        )
    )
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core einsum – Triton kernel
    # -----------------------------------------------------------------
    # Cast to float32 for accumulation (kernel expects float32)
    left = left.float()
    right = right.float()
    out = trimul_einsum(left, right)   # [B,N,N,hidden_dim] float32

    # -----------------------------------------------------------------
    # 6) Final LayerNorm, gating and projection back to dim
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(out.dtype),
        bias=weights["to_out_norm.bias"].to(out.dtype),
        eps=1e-5,
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(
        out,
        weights["to_out.weight"].to(out.dtype),  # (dim, hidden_dim)
        bias=None,
    )
    # Ensure final dtype is float32 as required
    return out.float()
