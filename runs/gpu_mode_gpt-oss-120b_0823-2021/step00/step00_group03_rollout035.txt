"""
TriMul “outgoing” implementation.
The heavy N³ work (the einsum over the third axis) is performed by a
custom Triton kernel that computes

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

for all batches B, sequence length N and hidden dimension H.
All other operations (LayerNorms, linear projections, gating and the final
linear) are kept in PyTorch – they are cheap compared with the N³ term.
The kernel works block‑wise over (i, j, d) and iterates over k in chunks.
It uses FP16 for the accumulation to keep memory traffic low and finally
writes FP32 results.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the N³ “einsum” part
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,          # *[B, N, N, H]  (i, k, d)
    right_ptr,         # *[B, N, N, H]  (j, k, d)
    out_ptr,           # *[B, N, N, H]  (i, j, d)
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # 1) map program id to a (batch, i, j) tile
    # ------------------------------------------------------------------
    total_tiles_i = tl.cdiv(N, BLOCK_I)
    total_tiles_j = tl.cdiv(N, BLOCK_J)
    b = pid // (total_tiles_i * total_tiles_j)
    tile_ij = pid % (total_tiles_i * total_tiles_j)
    i_tile = tile_ij // total_tiles_j
    j_tile = tile_ij % total_tiles_j

    # range of i / j handled by this program
    i_offset = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)

    # mask out-of-range i/j
    mask_i = i_offset < N
    mask_j = j_offset < N

    # ------------------------------------------------------------------
    # 2) allocate accumulator for (i, j, d)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) iterate over k
    # ------------------------------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        cur_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load left: shape (i, k, d)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            i_offset[:, None, None] * stride_left_i + \
            cur_k[None, :, None] * stride_left_k + \
            tl.arange(0, BLOCK_H)[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0).to(tl.float32)

        # Load right: shape (j, k, d)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            j_offset[:, None, None] * stride_right_j + \
            cur_k[None, :, None] * stride_right_k + \
            tl.arange(0, BLOCK_H)[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0).to(tl.float32)

        # multiply‑and‑accumulate over k dimension
        # left: (i, k, d)   right: (j, k, d)
        # we need Σ_k left[i,k,d] * right[j,k,d]
        # broadcast i over j and vice‑versa
        # expand dims to (i,1,k,d) and (1,j,k,d) then mul and sum over k
        left_exp = left[:, None, :, :]          # (i,1,k,d)
        right_exp = right[None, :, :, :]        # (1,j,k,d)
        prod = left_exp * right_exp             # (i,j,k,d)
        acc += tl.sum(prod, axis=2)             # sum over k -> (i,j,d)

    # ------------------------------------------------------------------
    # 4) write back
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        i_offset[:, None, None] * stride_out_i + \
        j_offset[None, :, None] * stride_out_j + \
        tl.arange(0, BLOCK_H)[None, None, :] * stride_out_h

    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None])


# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, D]  (float32)
            - mask         : [B, N, N]       (float32 or bool)
            - weights      : dict of model parameters
            - config       : dict with keys "dim", "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim = D)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) -> [B, N, N, H]
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask if present
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton kernel for the N³ einsum
    # ------------------------------------------------------------------
    out = torch.empty_like(left)  # shape [B,N,N,H], will be overwritten

    # strides (in elements) for each tensor
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # block sizes – tuned for Hopper (SM90) with 128‑thread warps
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32   # hidden dim is a multiple of 32 in all test configs
    BLOCK_K = 64   # k‑chunk size

    total_tiles_i = triton.cdiv(N, BLOCK_I)
    total_tiles_j = triton.cdiv(N, BLOCK_J)
    grid = (B * total_tiles_i * total_tiles_j,)

    trimul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (hidden dim) -> [B,N,N,H]
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # gating
    out = out * out_gate

    # final linear back to D
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
