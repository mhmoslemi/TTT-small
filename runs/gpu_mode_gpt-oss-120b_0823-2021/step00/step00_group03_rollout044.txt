"""
TriMul “outgoing” kernel (AlphaFold3 style) – fast Triton implementation
======================================================================

The heavy part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape (B, N, N, H) (H = hidden_dim).
For every batch `b` and every hidden channel `d` this is a matrix‑multiply
(L ∈ ℝ^{N×N}) · (Rᵀ ∈ ℝ^{N×N}) → (O ∈ ℝ^{N×N}).

The surrounding operations (LayerNorms, linear projections, sigmoid gates
and the final output projection) are cheap compared to the N³ accumulation,
so they stay in PyTorch.  Only the N³ reduction is off‑loaded to a
Triton kernel that

* works on FP16 internally (input is casted, accumulation in FP32)
* tiles the (i, j) plane with BLOCK_M × BLOCK_N threads
* loops over K‑tiles of size BLOCK_K
* multiplies a tile of ``left`` (i‑dim × k‑dim) with a tile of ``right``
  transposed (j‑dim × k‑dim) using ``tl.dot``

The kernel is launched with a 3‑D grid:
    (batch * hidden, ceil(N/BLOCK_M), ceil(N/BLOCK_N))

All strides are passed explicitly so the kernel works for any contiguous
layout that PyTorch produces (the default contiguous layout of a 4‑D tensor).

The final output of the whole module is converted back to FP32.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: tiled matrix‑multiply for each (batch, hidden) pair
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,   # [B, N, N, H]   (i, k, d) layout
    right_ptr,  # [B, N, N, H]   (j, k, d) layout, we will read transposed on K
    out_ptr,    # [B, N, N, H]   (i, j, d)

    # sizes
    B, N, H,

    # strides (in elements)
    stride_blh, stride_bl_i, stride_bl_k,   # left: (b, i, k, h)
    stride_brh, stride_br_j, stride_br_k,   # right: (b, j, k, h)
    stride_oh, stride_oi, stride_oj,        # out  : (b, i, j, h)

    # tile sizes – compile‑time constants
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_z = tl.program_id(0)                     # combined batch*hidden axis
    pid_m = tl.program_id(1)                     # i‑tile
    pid_n = tl.program_id(2)                     # j‑tile

    # decode batch and hidden channel from pid_z
    batch_id = pid_z // H
    hidden_id = pid_z % H

    # offsets within the N×N matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for tiles that go out of bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # offsets for the output tile (i, j)
    out_ptrs = out_ptr + \
        batch_id * stride_oh + \
        hidden_id * stride_oh + \
        offs_m[:, None] * stride_oi + \
        offs_n[None, :] * stride_oj

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in BLOCK_K steps
    for k_pos in range(0, tl.cdiv(N, BLOCK_K)):
        # K‑range for this iteration
        k_start = k_pos * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile: shape (M, K) -----
        left_ptrs = left_ptr + \
            batch_id * stride_blh + \
            hidden_id * stride_blh + \
            offs_m[:, None] * stride_bl_i + \
            offs_k[None, :] * stride_bl_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float16)

        # ----- load right tile (transposed): shape (N, K) -----
        # right layout is (b, j, k, h); we need Rᵀ so we load (j, k) as (k, j)
        # but we can simply read the same layout and treat k as inner dim.
        right_ptrs = right_ptr + \
            batch_id * stride_brh + \
            hidden_id * stride_brh + \
            offs_n[:, None] * stride_br_j + \
            offs_k[None, :] * stride_br_k
        right = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)
        right = right.to(tl.float16)

        # ----- accumulate dot product over K -----
        acc += tl.dot(left, right)   # (M,K)·(K,N) → (M,N)  (fp32 accumulation)

    # write result (convert back to fp16 then fp32 store)
    out = acc.to(tl.float16)
    tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Helper to launch the kernel
# --------------------------------------------------------------
def trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : torch.Tensor of shape [B, N, N, H] (float16)
    returns out : torch.Tensor of same shape (float16, later cast to fp32)
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, N2, H = left.shape
    assert N == N2

    out = torch.empty_like(left, dtype=torch.float16)

    # strides in number of elements (not bytes)
    # layout: (B, N, N, H) -> (b, i, k, h)
    stride_blh = left.stride(0)
    stride_bl_i = left.stride(1)
    stride_bl_k = left.stride(2)
    stride_bl_h = left.stride(3)

    # right layout: (b, j, k, h)
    stride_brh = right.stride(0)
    stride_br_j = right.stride(1)
    stride_br_k = right.stride(2)
    stride_br_h = right.stride(3)

    # out layout: (b, i, j, h)
    stride_oh = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_oh_h = out.stride(3)

    # tile sizes – tuned for Hopper (sm90) shared‑memory and registers
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        B * H,                                 # pid_z
        triton.cdiv(N, BLOCK_M),               # pid_m
        triton.cdiv(N, BLOCK_N),               # pid_n
    )

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_blh=stride_blh,
        stride_bl_i=stride_bl_i,
        stride_bl_k=stride_bl_k,
        stride_brh=stride_brh,
        stride_br_j=stride_br_j,
        stride_br_k=stride_br_k,
        stride_oh=stride_oh,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # enough for the chosen tile size
    )
    return out


# --------------------------------------------------------------
# Main entry point – combines PyTorch pre/post‑processing with the kernel
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data : tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            - mask         : torch.Tensor [B, N, N]        (bool/float)
            - weights      : dict of parameter tensors
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ----------------------------------------------------------
    # 1. LayerNorm on the last dimension
    # ----------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = (input_tensor - torch.mean(input_tensor, dim=-1, keepdim=True)) / torch.sqrt(
        torch.var(input_tensor, dim=-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias

    # ----------------------------------------------------------
    # 2. Linear projections (no bias) – keep in FP32 for weight
    # ----------------------------------------------------------
    # cast to FP16 for the heavy multiply later
    x_fp16 = x.to(torch.float16)

    left = torch.nn.functional.linear(x_fp16, weights["left_proj.weight"]).to(torch.float16)
    right = torch.nn.functional.linear(x_fp16, weights["right_proj.weight"]).to(torch.float16)

    # ----------------------------------------------------------
    # 3. Apply mask (if any)
    # ----------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype=torch.float16)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ----------------------------------------------------------
    # 4. Gate values (sigmoid) – keep in FP16
    # ----------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_fp16, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 5. Core triangular multiplication (Triton)
    # ----------------------------------------------------------
    # left, right : [B, N, N, H]  fp16
    out = trimul_outgoing_triton(left, right)   # fp16

    # ----------------------------------------------------------
    # 6. Post‑processing (LayerNorm, gate, final linear)
    # ----------------------------------------------------------
    out = out.to(torch.float32)

    # to_out_norm (LayerNorm on hidden_dim)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = (out - torch.mean(out, dim=-1, keepdim=True)) / torch.sqrt(
        torch.var(out, dim=-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate.to(torch.float32)

    # final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
