"""
TriMul (outgoing) – fast Triton implementation of the heavy N³ reduction.

The kernel computes
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]

All layer‑norms, linear projections and gating are kept in PyTorch (they are
memory‑bound and inexpensive compared with the N³ reduction).  Only the
reduction itself is executed on the GPU with a custom Triton kernel that
tiles the reduction over the sequence dimension (`k`) and accumulates in
float32.  The final tensor is brought back to FP32, normalized, gated and
projected in PyTorch – the required output type of the operator.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------------
# Triton kernel
# --------------------------------------------------------------------
@triton.jit
def _trimul_out_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,                         # tensor dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_K: tl.constexpr,           # tile size along the reduction axis
):
    """Compute the outgoing TriMul reduction.

    Grid: (B, N, N)  →  each program instance produces one (i,j) element
    for a given batch `b`.  The hidden dimension `H` is a compile‑time constant
    (hidden_dim from the model) so we can keep the whole reduction vector in
    registers.
    """
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # first sequence index
    pid_j = tl.program_id(2)          # second sequence index

    # ----------------------------------------------------------------
    # offsets for the hidden dimension (treated as a static vector)
    # ----------------------------------------------------------------
    offs_h = tl.arange(0, H)          # [0 … H‑1]

    acc = tl.zeros([H], dtype=tl.float32)   # accumulator for this (b,i,j)

    # ----------------------------------------------------------------
    # loop over k in tiles of BLOCK_K
    # ----------------------------------------------------------------
    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = start_k * BLOCK_K + tl.arange(0, BLOCK_K)   # [k0, k0+…]
        mask_k = cur_k < N

        # left[b, i, k, h]
        left_off = (
            pid_b * stride_left_b
            + pid_i * stride_left_i
            + cur_k[:, None] * stride_left_k
            + offs_h[None, :] * stride_left_h
        )
        left = tl.load(left_ptr + left_off, mask=mask_k[:, None], other=0.0)

        # right[b, j, k, h]
        right_off = (
            pid_b * stride_right_b
            + pid_j * stride_right_j
            + cur_k[:, None] * stride_right_k
            + offs_h[None, :] * stride_right_h
        )
        right = tl.load(right_ptr + right_off, mask=mask_k[:, None], other=0.0)

        # element‑wise product and reduction over the k‑tile
        acc += tl.sum(left * right, axis=0)

    # ----------------------------------------------------------------
    # write result
    # ----------------------------------------------------------------
    out_off = (
        pid_b * stride_out_b
        + pid_i * stride_out_i
        + pid_j * stride_out_j
        + offs_h * stride_out_h
    )
    tl.store(out_ptr + out_off, acc)


# --------------------------------------------------------------------
# Helper – a tiny LayerNorm that works on the last dimension only
# --------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias


# --------------------------------------------------------------------
# Entry point
# --------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : (input, mask, weights, config)
        input  – [B, N, N, dim]   (float32, already on CUDA)
        mask   – [B, N, N]        (float32 or bool, already on CUDA)
        weights – dict of model parameters
        config  – dict containing "dim" and "hidden_dim"

    Returns
    -------
    out – [B, N, N, dim] (float32, on the same device as the inputs)
    """
    # ----------------------------------------------------------------
    # unpack
    # ----------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]

    # ----------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (dim‑wise)
    # ----------------------------------------------------------------
    x = _layer_norm(inp,
                    weight=weights["norm.weight"],
                    bias=weights["norm.bias"]).to(torch.float32)

    # ----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – keep them in FP32
    # ----------------------------------------------------------------
    # shape: [B, N, N, hidden]
    left = torch.nn.functional.linear(
        x, weight=weights["left_proj.weight"]
    )
    right = torch.nn.functional.linear(
        x, weight=weights["right_proj.weight"]
    )

    # ----------------------------------------------------------------
    # 3️⃣ Apply mask (if present).  `mask` is broadcastable to [...,1]
    # ----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – also dim‑wise
    # ----------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weight=weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weight=weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weight=weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # 5️⃣ The heavy N³ reduction – Triton kernel
    # ----------------------------------------------------------------
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=inp.device)

    # strides in elements (not bytes) – required for Triton
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # launch parameters
    BLOCK_K = 64                         # tile size along reduction axis; fits in shared memory

    grid = (B, N, N)                     # one program instance per (b,i,j)
    _trimul_out_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_h=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_j=stride_right[1],
        stride_right_k=stride_right[2],
        stride_right_h=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_h=stride_out[3],
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gating and projection (back to `dim`)
    # ----------------------------------------------------------------
    out = _layer_norm(out,
                      weight=weights["to_out_norm.weight"],
                      bias=weights["to_out_norm.bias"])

    out = out * out_gate                 # broadcast over hidden dim
    out = torch.nn.functional.linear(
        out, weight=weights["to_out.weight"]
    )                                     # -> [B,N,N,dim]

    return out.to(torch.float32)
