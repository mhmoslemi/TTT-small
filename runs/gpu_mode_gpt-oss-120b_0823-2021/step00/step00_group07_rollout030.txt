"""
TriMul “outgoing” implementation.

The heavy part of the TriMul forward pass is the ternary einsum

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

For a fixed batch index *b* and channel *d* this is exactly a matrix‑multiply
(L @ Rᵀ) where L,R ∈ ℝ^{N×N}.  The kernel below fuses the
(N×N) × (N×N) multiply for **all** (b,d) pairs in a single launch.
All other operations (LayerNorm, Linear projections, sigmoids, masking,
output‑layer Linear+LayerNorm) remain in native PyTorch – they are
memory‑bound and cheap compared to the O(N³) work of the einsum.

The kernel works in fp16 (or bf16 if the hardware prefers it) and writes
the result in fp32 to match the reference implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul (L @ Rᵀ) for many (batch, hidden) pairs.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    # pointers
    left_ptr,          # (B*H, N, N)   fp16
    right_ptr,         # (B*H, N, N)   fp16
    out_ptr,           # (B*H, N, N)   fp32
    # strides
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_j, stride_right_k,
    stride_out_bh, stride_out_i, stride_out_j,
    # sizes
    N,                 # sequence length
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,
):
    """
    Compute out = left @ rightᵀ   (float16 → float32)

    left  : (B*H, N, N)          layout (bh, i, k)
    right : (B*H, N, N)          layout (bh, j, k)
    out   : (B*H, N, N)          layout (bh, i, j)
    """
    pid = tl.program_id(axis=0)          # index of (bh) pair
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # --------- choose a 2‑D tile (i,j) ---------------------------------
    pid_m = (pid // GROUP_SIZE_M) % num_pid_m
    pid_n = (pid // GROUP_SIZE_M) // num_pid_m
    pid_k = pid % GROUP_SIZE_M

    # compute the start offsets of the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask to avoid OOB
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers for this (bh) slice
    left_base  = left_ptr  + pid * stride_left_bh
    right_base = right_ptr + pid * stride_right_bh
    out_base   = out_ptr   + pid * stride_out_bh

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------- loop over k ---------------------------------------------
    for k in range(0, N, BLOCK_K):
        # k‑range for this iteration
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load a tile of left  (i,k)
        left = tl.load(
            left_base
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load a tile of right (j,k) – note we need the *same* k axis,
        # but we will treat it as transposed later.
        right = tl.load(
            right_base
            + offs_j[:, None] * stride_right_j
            + offs_k[None, :] * stride_right_k,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )
        # convert to fp32 for the dot product
        left_f32  = left.to(tl.float32)
        right_f32 = right.to(tl.float32)

        # accumulate: Σₖ left[i,k] * right[j,k]
        acc += tl.dot(left_f32, right_f32, out_dtype=tl.float32)

    # write result
    out = acc.to(tl.float32)
    tl.store(
        out_base
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j,
        out,
        mask=mask_i[:, None] & mask_j[None, :],
    )

# ----------------------------------------------------------------------
# Host helper to launch the kernel
# ----------------------------------------------------------------------
def _launch_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : torch.float16 tensors of shape (B, N, N, H)
    returns     : torch.float32 tensor of shape (B, N, N, H)
    """
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for the kernel
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out = torch.empty_like(left_, dtype=torch.float32)

    # strides in elements (not bytes)
    stride_l_bh, stride_l_i, stride_l_k = left_.stride()
    stride_r_bh, stride_r_j, stride_r_k = right_.stride()
    stride_o_bh, stride_o_i, stride_o_j = out.stride()

    # tuning parameters – these work well on Hopper (sm_90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    GROUP_SIZE_M = 8

    grid = (
        triton.cdiv(B * H, GROUP_SIZE_M),  # number of (bh) tiles (flattened)
    )
    _tri_mul_kernel[grid](
        left_.data_ptr(),
        right_.data_ptr(),
        out.data_ptr(),
        stride_left_bh=stride_l_bh,
        stride_left_i=stride_l_i,
        stride_left_k=stride_l_k,
        stride_right_bh=stride_r_bh,
        stride_right_j=stride_r_j,
        stride_right_k=stride_r_k,
        stride_out_bh=stride_o_bh,
        stride_out_i=stride_o_i,
        stride_out_j=stride_o_j,
        N=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
        num_warps=4,
    )
    # reshape back to (B, N, N, H)
    return out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()


# ----------------------------------------------------------------------
# Full custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        mask         : torch.Tensor [B, N, N]        (bool or 0/1)
        weights      : dict name→torch.Tensor (float32)
        config       : dict with keys 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_w + norm_b

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: (dim, hidden_dim)
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = torch.einsum("...c,ch->...h", x, left_proj_w)   # [B,N,N,hidden]
    right = torch.einsum("...c,ch->...h", x, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Masking (if mask provided)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(torch.einsum("...c,ch->...h", x, left_gate_w))
    right_gate = torch.sigmoid(torch.einsum("...c,ch->...h", x, right_gate_w))
    out_gate   = torch.sigmoid(torch.einsum("...c,ch->...h", x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core ternary einsum => fused matmul (fp16 for speed)
    # ------------------------------------------------------------------
    # Cast to fp16 before the heavy kernel
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # out shape (B,N,N,hidden) in fp32
    out = _launch_tri_mul(left_f16, right_f16)   # fp32 result

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm + gating + final Linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # LayerNorm over hidden_dim
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    # Apply output gate
    out = out * out_gate

    # Final linear back to dim
    out = torch.einsum("...h,hd->...d", out, to_out_w)

    return out.to(torch.float32)
