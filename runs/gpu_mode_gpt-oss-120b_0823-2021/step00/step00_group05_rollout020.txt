"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part of TriMul is the ternary einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]          (1)

where `left` and `right` are obtained from linear projections of the input
tensor (shape [B, N, N, H]) after masking and gating.
Equation (1) is equivalent to a batched matrix‑multiply **per hidden
dimension**:

    out[b, :, :, d] = left[b, :, :, d] @ right[b, :, :, d].T    (2)

We implement (2) with a Triton kernel that processes a 4‑D block
( BLOCK_M × BLOCK_N × BLOCK_D ) and iterates over the reduction
dimension `K = N` in tiles of `BLOCK_K`.  All arithmetic is performed in
FP16 for the loads (the model weights are stored in FP16) while the
accumulator is FP32 to retain numerical quality.  The final result is
converted back to FP32, layer‑normed and linearly projected with regular
PyTorch code.

Only the core multiplication‑accumulation is in Triton; the surrounding
LayerNorm, linear layers, masking, gating and the final linear projection
remain in PyTorch for simplicity and to keep the implementation concise.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM over the hidden dimension (outgoing)
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,          #   [B, N, N, H]   (FP16)
    right_ptr,         #   [B, N, N, H]   (FP16)
    out_ptr,           #   [B, N, N, H]   (FP32)
    # geometry
    B, N, H,
    # strides (in elements, not bytes)
    stride_bl, stride_bi, stride_bk, stride_bd,   # left  (B,N,N,H)
    stride_rl, stride_ri, stride_rk, stride_rd,   # right (B,N,N,H)
    stride_ol, stride_oi, stride_ok, stride_od,   # out   (B,N,N,H)
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute out = left @ rightᵀ per hidden slice."""
    pid_b = tl.program_id(0)                     # batch index
    pid_m = tl.program_id(1)                     # row tile (i)
    pid_n = tl.program_id(2)                     # col tile (j)
    pid_d = tl.program_id(3)                     # hidden tile

    # ------------------------------------------------------------------
    # Offsets for this tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    # mask out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_d = offs_d < H

    # ------------------------------------------------------------------
    # Pointers to the tile corners
    # ------------------------------------------------------------------
    left_tile_ptr  = left_ptr \
        + pid_b * stride_bl \
        + offs_m[:, None, None] * stride_bi \
        + tl.arange(0, BLOCK_K)[None, :, None] * stride_bk \
        + offs_d[None, None, :] * stride_bd

    right_tile_ptr = right_ptr \
        + pid_b * stride_rl \
        + offs_n[:, None, None] * stride_ri \
        + tl.arange(0, BLOCK_K)[None, :, None] * stride_rk \
        + offs_d[None, None, :] * stride_rd

    # Output accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduction over K (the second “N” dimension)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        cur_k = tl.minimum(BLOCK_K, N - k_base)

        # Load left:   [M, K, D]
        left = tl.load(
            left_tile_ptr
            + k_base * stride_bk,
            mask=mask_m[:, None] & (tl.arange(0, cur_k)[None, :] < cur_k)[:, None] & mask_d[None, None, :],
            other=0.0,
        )
        # Load right:  [N, K, D]
        right = tl.load(
            right_tile_ptr
            + k_base * stride_rk,
            mask=mask_n[:, None] & (tl.arange(0, cur_k)[None, :] < cur_k)[:, None] & mask_d[None, None, :],
            other=0.0,
        )

        # Convert to FP16 (they are stored as FP16)
        left = left.to(tl.float16)
        right = right.to(tl.float16)

        # Element‑wise multiplication and sum over K
        #   (M, K, D) * (N, K, D)  ->  (M, N, D) by broadcasting K
        #   we perform a dot for each hidden slice
        for d in range(0, BLOCK_D):
            if not mask_d[d]:
                continue
            # (M, K) dot (N, K)ᵀ  -> (M, N)
            acc[:, :, d] += tl.dot(left[:, :, d], right[:, :, d].t())

    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    out_tile_ptr = out_ptr \
        + pid_b * stride_ol \
        + offs_m[:, None, None] * stride_oi \
        + offs_n[None, :, None] * stride_ok \
        + offs_d[None, None, :] * stride_od

    tl.store(
        out_tile_ptr,
        acc,
        mask=mask_m[:, None, None] & mask_n[None, :, None] & mask_d[None, None, :],
    )


# --------------------------------------------------------------
# Helper functions (LayerNorm + Linear) – stay in PyTorch
# --------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    """Manual LayerNorm over last dimension (FP32)."""
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias


def linear(x, weight):
    """Linear without bias: y = x @ weight.T  (weight shape [out, in])"""
    return torch.nn.functional.linear(x, weight)

# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]   (float32)
            mask         : [B, N, N]        (float32 or bool)
            weights      : dict of tensors (all float32)
            config       : {'dim': int, 'hidden_dim': int}
    Returns:
        output tensor: [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm (dim) – stay in PyTorch (fast enough)
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float16)        # cast to FP16 for the linear projections

    # ------------------------------------------------------------------
    # 2. Linear proj + gating (all FP16, no bias)
    # ------------------------------------------------------------------
    left  = linear(x, weights["left_proj.weight"]).to(torch.float16)          # [B,N,N,H]
    right = linear(x, weights["right_proj.weight"]).to(torch.float16)

    # mask handling
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    left_gate  = torch.sigmoid(linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate   = torch.sigmoid(linear(x, weights["out_gate.weight"])).to(torch.float16)

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core Triton kernel: out = Σ_k left[...,i,k,d] * right[...,j,k,d]
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides in elements (not bytes)
    stride_bl, stride_bi, stride_bk, stride_bd = left.stride()
    stride_rl, stride_ri, stride_rk, stride_rd = right.stride()
    stride_ol, stride_oi, stride_ok, stride_od = out.stride()

    # Tile sizes – tuned for Hopper (SM90).  Adjust if needed.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 32

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(hidden_dim, BLOCK_D),
    )

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_bl=stride_bl,
        stride_bi=stride_bi,
        stride_bk=stride_bk,
        stride_bd=stride_bd,
        stride_rl=stride_rl,
        stride_ri=stride_ri,
        stride_rk=stride_rk,
        stride_rd=stride_rd,
        stride_ol=stride_ol,
        stride_oi=stride_oi,
        stride_ok=stride_ok,
        stride_od=stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 4. Post‑processing (still in PyTorch)
    # ------------------------------------------------------------------
    # to_out_norm (LayerNorm over hidden dim)
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                       # gating
    out = linear(out, weights["to_out.weight"])  # final projection → [B,N,N,dim]

    return out.to(torch.float32)
