"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the einsum

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

For a fixed hidden dimension `h` this is a matrix multiplication
`left_h @ right_h.T`.  We fuse this reduction with a Triton kernel that
computes a batched matmul over all `B * hidden_dim` matrices.
All LayerNorm / Linear / gating operations stay in PyTorch (they are
memory‑bound and already highly optimized).  The kernel works in fp16
and the final result is cast back to fp32 as required by the
reference model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: batched matmul for the TriMul einsum
# -------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr,               # *B* N * N * H  (row‑major)
    right_ptr,              # *B* N * N * H  (row‑major)
    out_ptr,                # *B* N * N * H  (row‑major)
    B, N, H,
    stride_lb, stride_lk, stride_lh,   # left strides: (batch, i, k, h)
    stride_rb, stride_rj, stride_rk, stride_rh,  # right strides: (batch, j, k, h)
    stride_ob, stride_oi, stride_oj, stride_oh,  # out strides:   (batch, i, j, h)
    BLOCK_M: tl.constexpr,  # tile size for i dimension
    BLOCK_N: tl.constexpr,  # tile size for j dimension
    BLOCK_K: tl.constexpr,  # tile size for k reduction
):
    pid = tl.program_id(0)                     # 0 … B*H-1
    b = pid // H                               # batch index
    h = pid % H                                # hidden dim index

    # ------------------------------------------------------------------
    # Offsets for the tile we are computing
    # ------------------------------------------------------------------
    offs_i = tl.arange(0, BLOCK_M)
    offs_j = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out‑of‑bounds
    i_mask = (offs_i < N)
    j_mask = (offs_j < N)

    # ------------------------------------------------------------------
    # Pointers for the current tile
    # ------------------------------------------------------------------
    # left[b, i, k, h]
    left_base = left_ptr + b * stride_lb + h * stride_lh
    left_tile_ptrs = left_base + (offs_i[:, None] * stride_lk + offs_k[None, :] * 1)

    # right[b, j, k, h]   (note the transposition: we need right[j, k])
    right_base = right_ptr + b * stride_rb + h * stride_rh
    right_tile_ptrs = right_base + (offs_j[None, :] * stride_rj + offs_k[:, None] * 1)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k dimension (reduction)
    # ------------------------------------------------------------------
    for k_offset in range(0, tl.cdiv(N, BLOCK_K)):
        # load a tile of left and right, apply masks
        left = tl.load(left_tile_ptrs,
                       mask=(i_mask[:, None] & (k_offset * BLOCK_K + offs_k < N))[..., None],
                       other=0.0).to(tl.float32)

        right = tl.load(right_tile_ptrs,
                        mask=((k_offset * BLOCK_K + offs_k < N)[:, None] & j_mask[None, :])[..., None],
                        other=0.0).to(tl.float32)

        # dot‑product acc += left @ right
        acc += tl.dot(left, right)

        # advance pointers to the next k‑block
        left_tile_ptrs += BLOCK_K * stride_lk
        right_tile_ptrs += BLOCK_K * stride_rk

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_base = out_ptr + b * stride_ob + h * stride_oh
    out_tile_ptrs = out_base + (offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj)

    tl.store(out_tile_ptrs, acc.to(tl.float16), mask=(i_mask[:, None] & j_mask[None, :]))


# -------------------------------------------------------------
# Host‑side entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, dim]  (float32)
            - mask        : torch.Tensor[B, N, N]      (float32 or bool)
            - weights     : dict of torch.Tensor containing all model weights
            - config      : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    import torch.nn as nn

    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ---------- 1. LayerNorm ----------
    # norm: (dim,) weight & bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias

    # ---------- 2. Linear projections ----------
    # linear without bias
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])        # [B,N,N,hidden]

    # ---------- 3. Apply mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)           # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- 4. Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- 5. Core einsum via Triton ----------
    # Cast to fp16 for the kernel (H200 has strong fp16 performance)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)  # same shape [B,N,N,hidden]

    # strides (in elements) for the tensors
    stride_lb, stride_lk, stride_lh = left_fp16.stride()[-4:]  # batch, i, k, h
    stride_rb, stride_rj, stride_rk, stride_rh = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out_fp16.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B * hidden,)

    _trimul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden,
        stride_lb, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    out = out_fp16.to(torch.float32)   # back to fp32

    # ---------- 6. Output norm & gate ----------
    # to_out_norm (LayerNorm over hidden)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate

    # ---------- 7. Final linear ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
