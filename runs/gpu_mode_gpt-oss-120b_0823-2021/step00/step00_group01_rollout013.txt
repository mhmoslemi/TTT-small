"""
TriMul “outgoing” implementation.

The heavy part of the module is the ternary einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is O(N³·C).  We keep the cheap LayerNorm/Linear/gate work in
PyTorch and fuse the triple‑product into one Triton kernel.
The kernel iterates over the k‑dimension in a simple for‑loop (the
sequence length is at most 1024, so the loop is cheap) and processes
the hidden dimension `hidden_dim` in blocks (`BLOCK_D`).  The kernel is
launched with a 2‑D grid:
    (batch * N * N) × ceil(hidden_dim / BLOCK_D)

All tensors are expected to be on the CUDA device already.  The final
output is cast back to float32 to match the reference implementation.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------------
# Triton kernel for the core ternary multiplication
# --------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,               # const float*  [B, N, N, H]
    right_ptr,              # const float*  [B, N, N, H]
    out_ptr,                # float*        [B, N, N, H]
    stride_lb0, stride_lb1, stride_lb2, stride_lb3,   # left strides
    stride_rb0, stride_rb1, stride_rb2, stride_rb3,   # right strides
    stride_ob0, stride_ob1, stride_ob2, stride_ob3,   # out strides
    N,                      # sequence length
    H,                      # hidden_dim
    BLOCK_D: tl.constexpr,  # block size for hidden dimension
):
    # ----------------------------------------------------------------
    # Decode the 2‑D program ids
    # pid0 : (b,i,j) flattened
    # pid1 : block index for hidden dimension
    # ----------------------------------------------------------------
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    # ----- decode (b,i,j) -----------------------------------------
    b = pid0 // (N * N)
    tmp = pid0 % (N * N)
    i = tmp // N
    j = tmp % N

    # ----- hidden‑dim block ----------------------------------------
    d_start = pid1 * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)
    mask_d = offs_d < H

    # ----------------------------------------------------------------
    # Accumulate Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ----------------------------------------------------------------
    acc = tl.zeros([BLOCK_D], dtype=tl.float32)

    for k in range(0, N):
        # pointers to left[b,i,k,:] and right[b,j,k,:]
        left_offset = (
            b * stride_lb0
            + i * stride_lb1
            + k * stride_lb2
            + offs_d * stride_lb3
        )
        right_offset = (
            b * stride_rb0
            + j * stride_rb1
            + k * stride_rb2
            + offs_d * stride_rb3
        )
        l = tl.load(left_ptr + left_offset, mask=mask_d, other=0.0)
        r = tl.load(right_ptr + right_offset, mask=mask_d, other=0.0)
        acc += l * r

    # ----- write result --------------------------------------------
    out_offset = (
        b * stride_ob0
        + i * stride_ob1
        + j * stride_ob2
        + offs_d * stride_ob3
    )
    tl.store(out_ptr + out_offset, acc, mask=mask_d)


# --------------------------------------------------------------------
# Helper: LayerNorm without bias (weights & bias are passed explicitly)
# --------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# --------------------------------------------------------------------
# Entry point called by the grader
# --------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    Only the O(N³·C) multiplication is performed in Triton.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ----------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # ----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ----------------------------------------------------------------
    # shape: [B,N,N,hidden_dim]
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------
    # 3. Apply mask (if present) – mask is broadcast on the hidden dim
    # ----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                 # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ----------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(
        x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(
        x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(
        x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # 5. Triton kernel for the ternary product
    # ----------------------------------------------------------------
    # allocate output
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # strides (in elements, not bytes)
    stride_lb = left.stride()   # (B, N, N, H)
    stride_rb = right.stride()
    stride_ob = out.stride()

    # block size for hidden dimension – 64 works for all hidden sizes
    BLOCK_D = 64

    grid = (
        triton.cdiv(B * N * N, 1),          # pid0 dimension
        triton.cdiv(hidden_dim, BLOCK_D),   # pid1 dimension
    )

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        stride_lb0=stride_lb[0],
        stride_lb1=stride_lb[1],
        stride_lb2=stride_lb[2],
        stride_lb3=stride_lb[3],
        stride_rb0=stride_rb[0],
        stride_rb1=stride_rb[1],
        stride_rb2=stride_rb[2],
        stride_rb3=stride_rb[3],
        stride_ob0=stride_ob[0],
        stride_ob1=stride_ob[1],
        stride_ob2=stride_ob[2],
        stride_ob3=stride_ob[3],
        N=N,
        H=hidden_dim,
        BLOCK_D=BLOCK_D,
    )

    # ----------------------------------------------------------------
    # 6. Post‑processing in PyTorch
    # ----------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(dtype)
