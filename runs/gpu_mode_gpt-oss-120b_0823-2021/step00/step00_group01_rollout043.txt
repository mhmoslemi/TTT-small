"""
TriMul “outgoing” implementation.

Algorithm
---------
1. Layer‑Norm + three Linear projections (left/right) and three gating
   Linear → sigmoid (all done in PyTorch – they are cheap compared to the
   N³ multiplication).
2. Apply mask and gates to the projected tensors → tensors `left` and `right`
   of shape [B, N, N, H]  (H = hidden_dim).
3. Triton kernel computes, for every (b,i,j) pair, the inner product over the
   third axis `k` for every hidden channel d:

        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

   This is the expensive O(B·N³·H) step.  The kernel loops over `k` in blocks,
   accumulates a per‑channel sum and finally writes the result.
4. Remaining Layer‑Norm, output‑gate and output Linear are performed in PyTorch.
   Final tensor is returned as float32.

The kernel is deliberately simple: each program processes one (b,i,j) triple.
`BLOCK_K` (tiles over the summed dimension) and `BLOCK_D` (tiles over hidden)
are compile‑time constants; they are chosen to fit the H200 shared‑memory
budget while keeping the arithmetic intensity high.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: computes the N³ “einsum” part
# ----------------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr,          # const * fp32   [B, N, N, H]
    right_ptr,         # const * fp32   [B, N, N, H]
    out_ptr,           #       * fp32   [B, N, N, H]

    B, N, H,

    # strides for left tensor
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right tensor
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for output tensor
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_K: tl.constexpr,   # tile size along summed dimension k
    BLOCK_D: tl.constexpr,   # tile size along hidden dimension d
):
    pid = tl.program_id(0)                     # 0 ≤ pid < B*N*N
    total = B * N * N

    # ----- decode (b,i,j) from pid -----
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # ----- hidden‑dim tile offsets -----
    offs_d = tl.arange(0, BLOCK_D)
    d_mask = offs_d < H                         # mask for the last partial tile

    # accumulator for the dot‑product over k, one value per hidden channel
    acc = tl.zeros([BLOCK_D], dtype=tl.float32)

    # ----- loop over k in blocks -----
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_start = kb * BLOCK_K + tl.arange(0, BLOCK_K)   # [BLOCK_K]
        k_mask = k_start < N

        # pointers for left[b,i,k,d]
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i * stride_left_i
            + k_start[:, None] * stride_left_k
            + offs_d[None, :] * stride_left_h
        )
        # pointers for right[b,j,k,d]
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + j * stride_right_j
            + k_start[:, None] * stride_right_k
            + offs_d[None, :] * stride_right_h
        )

        # load with masking
        left = tl.load(left_ptrs,
                       mask=k_mask[:, None] & d_mask[None, :],
                       other=0.0)
        right = tl.load(right_ptrs,
                        mask=k_mask[:, None] & d_mask[None, :],
                        other=0.0)

        # element‑wise product and reduction over k
        prod = left * right                     # shape (BLOCK_K, BLOCK_D)
        acc += tl.sum(prod, axis=0)             # sum across k‑dimension

    # ----- write output -----
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i * stride_out_i
        + j * stride_out_j
        + offs_d * stride_out_h
    )
    tl.store(out_ptrs, acc, mask=d_mask)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : (input, mask, weights, config)
        input  – torch.Tensor of shape [B, N, N, dim] (float32 or float16)
        mask   – torch.Tensor of shape [B, N, N] (bool / uint8 / float)
        weights – dict with all linear / layer‑norm parameters (fp32)
        config  – dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    dtype = torch.float32

    # ------------------------------------------------------------------
    # 1. LayerNorm and linear projections (PyTorch)
    # ------------------------------------------------------------------
    # layer norm
    norm_w = weights["norm.weight"].to(dtype)
    norm_b = weights["norm.bias"].to(dtype)
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_w + norm_b

    # linear projections (no bias)
    left_proj_w = weights["left_proj.weight"].to(dtype)      # [hidden, dim]
    right_proj_w = weights["right_proj.weight"].to(dtype)    # [hidden, dim]

    # reshape to (..., dim) → (..., hidden)
    left = torch.einsum('...d,hd->...h', x, left_proj_w)       # [B,N,N,hidden]
    right = torch.einsum('...d,hd->...h', x, right_proj_w)

    # ------------------------------------------------------------------
    # 2. Mask (if present) and gating
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)                 # [B,N,N,1]
        left = left * mask
        right = right * mask

    # gates
    left_gate_w = weights["left_gate.weight"].to(dtype)
    right_gate_w = weights["right_gate.weight"].to(dtype)
    out_gate_w = weights["out_gate.weight"].to(dtype)

    left_gate = torch.sigmoid(torch.einsum('...d,hd->...h', x, left_gate_w))
    right_gate = torch.sigmoid(torch.einsum('...d,hd->...h', x, right_gate_w))
    out_gate = torch.sigmoid(torch.einsum('...d,hd->...h', x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Triton kernel: compute the N³ inner product over k
    # ------------------------------------------------------------------
    # allocate output buffer
    out = torch.empty((B, N, N, hidden), dtype=dtype, device=device)

    # strides (in elements, not bytes) required by the kernel
    left_stride = left.stride()
    right_stride = right.stride()
    out_stride = out.stride()

    # launch parameters
    BLOCK_K = 64                      # tile size along summed axis
    BLOCK_D = 128 if hidden >= 128 else hidden   # hidden dim fits in one tile
    grid = (B * N * N,)

    _trimul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_left_b=left_stride[0],
        stride_left_i=left_stride[1],
        stride_left_k=left_stride[2],
        stride_left_h=left_stride[3],
        stride_right_b=right_stride[0],
        stride_right_j=right_stride[1],
        stride_right_k=right_stride[2],
        stride_right_h=right_stride[3],
        stride_out_b=out_stride[0],
        stride_out_i=out_stride[1],
        stride_out_j=out_stride[2],
        stride_out_h=out_stride[3],
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 4. Post‑processing (PyTorch)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)

    # layer‑norm over hidden dimension
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    # output gate
    out = out * out_gate

    # final linear back to original dim
    to_out_w = weights["to_out.weight"].to(dtype)          # [dim, hidden]
    out = torch.einsum('...h,dh->...d', out, to_out_w)

    return out.to(torch.float32)   # guarantee fp32 output
