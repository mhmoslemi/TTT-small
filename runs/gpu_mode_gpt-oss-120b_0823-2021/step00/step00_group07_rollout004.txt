"""
TriMul “outgoing” implementation.

The heavy N³ part is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We fuse this reduction into a single Triton kernel.  All linear
projections, layer‑norms and final linear are kept in PyTorch (they are
memory‑bound) while the cubic accumulation runs on the GPU with
explicit tiling over (i, j, k).  The kernel computes a tile
(M_BLOCK × N_BLOCK) of the output for a fixed batch and accumulates over
the K dimension in blocks of K_BLOCK.  The partial results are summed
with `tl.sum` (no dot needed) and written back as FP32.  Mixed precision
is used inside the kernel (FP16 loads, FP32 accumulation) and the final
output is cast back to FP32 as required by the model.

The host side builds the tiled grid as
    grid = (ceil(N/M_BLOCK), ceil(N/N_BLOCK), B)
and launches the kernel once.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel for the cubic reduction
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # *[B, N, N, H]   left projection (FP16)
    right_ptr,         # *[B, N, N, H]   right projection (FP16)
    out_ptr,           # *[B, N, N, H]   output (FP32)
    B, N, H,           # static sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    M_BLOCK: tl.constexpr,
    N_BLOCK: tl.constexpr,
    K_BLOCK: tl.constexpr,
):
    # program ids for tiling
    pid_i = tl.program_id(0)          # tile over output rows (i)
    pid_j = tl.program_id(1)          # tile over output cols (j)
    pid_b = tl.program_id(2)          # batch dimension

    # start offsets for this tile
    i_base = pid_i * M_BLOCK
    j_base = pid_j * N_BLOCK

    # range of indices inside the tile
    offs_i = i_base + tl.arange(0, M_BLOCK)
    offs_j = j_base + tl.arange(0, N_BLOCK)

    # mask out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # allocate accumulator (M_BLOCK, N_BLOCK, H) in fp32
    acc = tl.zeros((M_BLOCK, N_BLOCK, H), dtype=tl.float32)

    # loop over k dimension
    num_k_blocks = tl.cdiv(N, K_BLOCK)
    for kb in range(0, num_k_blocks):
        k_base = kb * K_BLOCK
        offs_k = k_base + tl.arange(0, K_BLOCK)
        mask_k = offs_k < N

        # ------- load left[b, i, k, :] ---------------------------------
        # shape (M_BLOCK, K_BLOCK, H)
        left_ptrs = left_ptr + \
            pid_b * stride_left_b + \
            offs_i[:, None, None] * stride_left_i + \
            offs_k[None, :, None] * stride_left_k + \
            tl.arange(0, H)[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None],
                       other=0.0).to(tl.float16)

        # ------- load right[b, j, k, :] ---------------------------------
        # shape (N_BLOCK, K_BLOCK, H)
        right_ptrs = right_ptr + \
            pid_b * stride_right_b + \
            offs_j[:, None, None] * stride_right_i + \
            offs_k[None, :, None] * stride_right_k + \
            tl.arange(0, H)[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None],
                        other=0.0).to(tl.float16)

        # multiply element‑wise and reduce over k
        # left: (M,K,H) , right: (N,K,H) -> want sum_k left[:,k,:] * right[:,k,:]
        # expand dims to broadcast over i/j
        prod = left[:, None, :, :] * right[None, :, :, :]   # (M,N,K,H)

        # sum over K axis
        acc += tl.sum(prod, axis=2)   # (M,N,H)

    # store result
    out_ptrs = out_ptr + \
        pid_b * stride_out_b + \
        offs_i[:, None, None] * stride_out_i + \
        offs_j[None, :, None] * stride_out_j + \
        tl.arange(0, H)[None, None, :] * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None])

# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module with the cubic reduction
    performed by a Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, D = input_tensor.shape          # D == dim
    hidden_dim = config["hidden_dim"]        # H

    # ---------- LayerNorm on the input ----------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(D,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ---------- Linear projections (no bias) ----------
    # We'll work in FP16 inside the kernel for bandwidth.
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    # shape: [B,N,N,H]
    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)

    # ---------- Apply mask (if present) ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)   # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gating ----------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ---------- Triton reduction ----------
    # Cast to FP16 for the kernel (already fp16 from linear)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements, not bytes) for each tensor
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    M_BLOCK = 32
    N_BLOCK = 32
    K_BLOCK = 32

    grid = (triton.cdiv(N, M_BLOCK),
            triton.cdiv(N, N_BLOCK),
            B)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        M_BLOCK=M_BLOCK,
        N_BLOCK=N_BLOCK,
        K_BLOCK=K_BLOCK,
        num_warps=4,  # enough for FP16 compute on H200
    )

    # ---------- Output norm, gating, final linear ----------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    out = out * out_gate   # broadcasting over (B,N,N,hidden)

    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    return out.to(torch.float32)
