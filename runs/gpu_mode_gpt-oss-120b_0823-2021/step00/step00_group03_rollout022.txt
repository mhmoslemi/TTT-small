"""
TriMul “outgoing” kernel (AlphaFold3) – fast Triton implementation.

The heavy part of the forward pass is the three‑way einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication per hidden dimension d.
We rewrite it as

    out = (left) @ (right.permute(0,2,1,3))

and launch a Triton GEMM that processes a block of rows (i) and a block
of columns (j) for a fixed batch index b and hidden dimension d.
The kernel works in fp16 for speed and accumulates in fp32; the final
result is cast back to fp32 to match the reference implementation.

All surrounding operations (LayerNorm, Linear projections, gating,
masking, final LayerNorm + Linear) remain in PyTorch – they are cheap
relative to the O(N³) matmul and keep the code simple.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton GEMM kernel: C[b,d,i,j] = A[b,d,i,k] @ B[b,d,j,k]   (note B is
# stored as (..., j, k, d) → we pass a transposed view so that the inner
# dimension is contiguous).
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_head, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_head, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_head, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr, dtype: tl.constexpr
):
    """Compute C = A @ B  (A: [B,H,N,N], B: [B,H,N,N] transposed on the last two dims)."""
    pid = tl.program_id(0)                     # linear index over (b, h, m_tile, n_tile)
    num_h = H
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    # decompose pid
    n_tile = pid % num_n
    m_tile = (pid // num_n) % num_m
    h = (pid // (num_m * num_n)) % num_h
    b = pid // (num_m * num_n * num_h)

    # pointers to the start of the tile
    a_block_ptr = a_ptr + \
        b * stride_a_batch + h * stride_a_head + \
        (m_tile * BLOCK_M) * stride_a_row

    b_block_ptr = b_ptr + \
        b * stride_b_batch + h * stride_b_head + \
        (n_tile * BLOCK_N) * stride_b_row   # note: B is already permuted (j,k)

    # offsets
    offs_m = tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    a_offs = a_block_ptr + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
    b_offs = b_block_ptr + offs_n[:, None] * stride_b_row + offs_k[None, :] * stride_b_col

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_offs, mask=offs_m[:, None] < N & offs_k[None, :] < N,
                    other=0.0, dtype=dtype)
        b = tl.load(b_offs, mask=offs_n[:, None] < N & offs_k[None, :] < N,
                    other=0.0, dtype=dtype)

        # convert to fp32 for accumulation
        a_fp32 = a.to(tl.float32)
        b_fp32 = b.to(tl.float32)
        acc += tl.dot(a_fp32, b_fp32)

        # advance pointers
        a_offs += BLOCK_K * stride_a_col
        b_offs += BLOCK_K * stride_b_col

    # write back
    c = acc.to(dtype)   # back to fp16 for storage
    c_ptrs = c_ptr + \
        b * stride_c_batch + h * stride_c_head + \
        (m_tile * BLOCK_M) * stride_c_row + (n_tile * BLOCK_N) * stride_c_col
    mask_m = offs_m[:, None] < N
    mask_n = offs_n[None, :] < N
    tl.store(c_ptrs + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col,
             c, mask=mask_m & mask_n)


# ----------------------------------------------------------------------
# Helper that launches the kernel for a given tensor pair (left, right_t)
# ----------------------------------------------------------------------
def trimul_gemm(left: torch.Tensor, right_t: torch.Tensor, block_m=64, block_n=64, block_k=64):
    """
    left :  [B, N, N, H]   fp16
    right_t : [B, N, N, H]  (right already transposed on the last two dims) fp16
    returns: [B, N, N, H]  fp16
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides for contiguous NHWC layout
    stride_a_batch, stride_a_row, stride_a_col, stride_a_head = left.stride()
    stride_b_batch, stride_b_row, stride_b_col, stride_b_head = right_t.stride()
    stride_c_batch, stride_c_row, stride_c_col, stride_c_head = out.stride()

    # convert to 1‑D pointers
    a_ptr = left.data_ptr()
    b_ptr = right_t.data_ptr()
    c_ptr = out.data_ptr()

    # grid size = B * H * ceil(N/BLOCK_M) * ceil(N/BLOCK_N)
    num_tiles_m = (N + block_m - 1) // block_m
    num_tiles_n = (N + block_n - 1) // block_n
    total = B * H * num_tiles_m * num_tiles_n
    grid = (total,)

    trimul_gemm_kernel[grid](
        a_ptr, b_ptr, c_ptr,
        B, N, H,
        stride_a_batch, stride_a_head, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_head, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_head, stride_c_row, stride_c_col,
        BLOCK_M=block_m, BLOCK_N=block_n, BLOCK_K=block_k,
        num_stages=3,
        dtype=tl.float16,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            mask         : torch.Tensor [B, N, N]        (float32 or bool)
            weights_dict : dict of parameter tensors
            config_dict  : {'dim': int, 'hidden_dim': int, ...}
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # torch.nn.functional.layer_norm expects normalized_shape = last dim
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)  -->  fp16 for speed
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3) Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(torch.float16)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) – keep in fp16, then back‑to‑fp16 multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core trigram multiplication: out = Σ_k left[...,i,k,d] * right[...,j,k,d]
    #    Implemented as GEMM with right transposed on (j,k)
    # ------------------------------------------------------------------
    # right_t has shape [B, N, N, H] with axes (j,k) → we need (k,j)
    right_t = right.permute(0, 2, 1, 3).contiguous()   # now (B, N, N, H) with k as dim 1
    # Triton kernel works on [B, N, N, H] layout (row‑major)
    out_fp16 = trimul_gemm(left, right_t, block_m=64, block_n=64, block_k=64)

    # ------------------------------------------------------------------
    # 6) to_out_norm (LayerNorm over hidden_dim) + out_gate + final linear
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)   # back to fp32 for the final layers

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate.to(torch.float32)        # broadcast over (B,N,N,hidden)

    out = F.linear(out, weights["to_out.weight"].to(torch.float32))   # -> dim

    return out
