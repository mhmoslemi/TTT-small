"""
TriMul “outgoing” kernel (AlphaFold3)

We keep the LayerNorm / linear / gating logic in PyTorch (they are cheap)
and accelerate the expensive N³ einsum:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden channel *d* the operation is a batched matrix‑multiply:
    out_d = left_d @ right_dᵀ                     (B×N×N) @ (B×N×N)ᵀ

The Triton kernel tiles the k‑dimension (shared reduction axis) and
computes a block of the output matrix for each (b, d) pair.
Result is accumulated in fp32 for numerical stability, while the
loads are performed in fp16/bf16 (the reference uses bfloat16).  The final
tensor is returned in float32 as required by the spec.
"""
import torch
import triton
import triton.language as tl
from torch import nn

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for a single hidden channel (d)
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,                     # const * left  [B, N, N, H]
    right_ptr,                    # const * right [B, N, N, H]
    out_ptr,                      #       * out   [B, N, N, H]
    B, N, H,                      # problem sizes
    stride_lbn, stride_lbk, stride_lbd,   # left strides (batch, N, hidden)
    stride_rbn, stride_rbk, stride_rbd,   # right strides
    stride_obn, stride_obk, stride_obd,   # out strides
    BLOCK_M: tl.constexpr,        # tile rows  (i)
    BLOCK_N: tl.constexpr,        # tile cols  (j)
    BLOCK_K: tl.constexpr,        # tile reduction (k)
    USE_BF16: tl.constexpr,       # 1: load as bf16, else fp16
):
    pid = tl.program_id(0)                # linear id over (b, d)
    b = pid // H
    d = pid %  H

    # Offsets for the output tile (i,j)
    offs_i = tl.arange(0, BLOCK_M)
    offs_j = tl.arange(0, BLOCK_N)

    # Global row / col indices
    i = tl.program_id(1) * BLOCK_M + offs_i   # i in [0, N)
    j = tl.program_id(2) * BLOCK_N + offs_j   # j in [0, N)

    # Masks for out‑of‑bounds
    mask_i = i < N
    mask_j = j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # Pointers to the start of the (b, :, :, d) slices
    left_base  = left_ptr  + b * stride_lbn + d * stride_lbd
    right_base = right_ptr + b * stride_rbn + d * stride_rbd
    out_base   = out_ptr   + b * stride_obn + d * stride_obd

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left[i, k] and right[j, k] (note transposition on right)
        left_ptrs  = left_base  + i[:, None] * stride_lbk + offs_k[None, :] * stride_lbk
        right_ptrs = right_base + j[:, None] * stride_rbk + offs_k[None, :] * stride_rbk

        if USE_BF16:
            left_tile  = tl.load(left_ptrs,  mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
            right_tile = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        else:
            left_tile  = tl.load(left_ptrs,  mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
            right_tile = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right is taken transposed: we need right[j,k] not right[k,j]; the
        # pointer arithmetic above already reads the correct layout because
        # the reduction axis is the last of the two dimensions we multiply.
        # Perform dot‑product over k
        acc += tl.dot(left_tile, right_tile, allow_tf32=False)

    # Write result
    out_ptrs = out_base + i[:, None] * stride_obk + j[None, :] * stride_obk
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_ij)


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_outgoing_triton(x: torch.Tensor,
                           mask: torch.Tensor,
                           weights: dict,
                           config: dict) -> torch.Tensor:
    """
    x    : [B, N, N, dim]   (float32)
    mask : [B, N, N]        (float32 or bool)
    returns: [B, N, N, dim] (float32)
    """
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]
    device = x.device

    # ------------------------------------------------------------------
    # 1. PyTorch part ---------------------------------------------------
    # ------------------------------------------------------------------
    # LayerNorm
    x = (x - x.mean(-1, keepdim=True)) / torch.sqrt(x.var(-1, unbiased=False, keepdim=True) + 1e-5)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # Linear projections (no bias)
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask (if present)
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # Cast to bfloat16 for the heavy N³ part (matches reference)
    left  = left.to(torch.bfloat16)
    right = right.to(torch.bfloat16)

    # ------------------------------------------------------------------
    # 2. Triton kernel -------------------------------------------------
    # ------------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    stride_lbn, stride_lbk, stride_lbd = left.stride()
    stride_rbn, stride_rbk, stride_rbd = right.stride()
    stride_obn, stride_obk, stride_obd = out.stride()

    # Tiling parameters – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * hidden,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_lbn=stride_lbn,
        stride_lbk=stride_lbk,
        stride_lbd=stride_lbd,
        stride_rbn=stride_rbn,
        stride_rbk=stride_rbk,
        stride_rbd=stride_rbd,
        stride_obn=stride_obn,
        stride_obk=stride_obk,
        stride_obd=stride_obd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        USE_BF16=1,                     # we loaded BF16 above
    )

    # ------------------------------------------------------------------
    # 3. Post‑processing (LayerNorm, gating, final linear) ------------
    # ------------------------------------------------------------------
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(out.var(-1, unbiased=False, keepdim=True) + 1e-5)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    return trimul_outgoing_triton(input_tensor, mask, weights, config)
