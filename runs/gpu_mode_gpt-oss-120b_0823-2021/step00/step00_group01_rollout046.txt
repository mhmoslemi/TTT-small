"""
TriMul “outgoing” implementation.

The heavy part of TriMul is the tensor‑product
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed (b,d) this is exactly a matrix multiplication
    L (N×N)  @  Rᵀ (N×N)   ->   O (N×N)

We fuse this N×N mat‑mul for every batch element and every hidden
dimension into a single Triton kernel.  All other ops (LayerNorm,
Linear, sigmoid gating, masking) are kept in PyTorch – they are cheap
compared to the O(N³) work of the einsum and keep the code simple.
The kernel works with float16 inputs (the model uses bfloat16/float16)
and writes float32 accumulators that are finally cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batch‑wise mat‑mul for the TriMul einsum
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
        a_ptr, b_ptr, c_ptr,        # pointers
        B, N, D,                    # problem sizes
        stride_ab, stride_ak, stride_ah,   # A: (b,i,k,d) strides
        stride_bb, stride_bk, stride_bh,   # B: (b,j,k,d) strides
        stride_cb, stride_ci, stride_ch,   # C: (b,i,j,d) strides
        BLOCK_M: tl.constexpr,      # tile size in i‑dimension
        BLOCK_N: tl.constexpr,      # tile size in j‑dimension
        BLOCK_K: tl.constexpr,      # tile size in reduction k
):
    """
    Compute for a single (batch, hidden) pair:
        C[i, j] = Σ_k A[i, k] * B[j, k]   (note B is accessed with swapped axes)
    """
    pid = tl.program_id(0)                # 0 … B*D-1
    batch_id = pid // D
    d_id     = pid % D

    # pointers to the start of the (batch,d) slice
    a_base = a_ptr + batch_id * stride_ab + d_id * stride_ah
    b_base = b_ptr + batch_id * stride_bb + d_id * stride_bh
    c_base = c_ptr + batch_id * stride_cb + d_id * stride_ch

    # grid indices
    pid_m = tl.program_id(1)              # block row
    pid_n = tl.program_id(2)              # block col

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A: [M, K]   (i, k)
        a_ptrs = a_base + (offs_m[:, None] * stride_ak
                           + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # B: [N, K]   we need B[j, k] → address uses stride_bb for j and stride_bk for k
        b_ptrs = b_base + (offs_n[:, None] * stride_bb
                           + offs_k[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # dot‑product over k
        acc += tl.dot(a, b)

        # advance pointers (not needed because we recompute a_ptrs each iteration)
    # store result
    c_ptrs = c_base + (offs_m[:, None] * stride_ci
                       + offs_n[None, :] * stride_ci)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def trimul_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, D]  (float16 or bfloat16)
    returns      : [B, N, N, D]  (float32)
    """
    assert left.shape == right.shape
    B, N, _, D = left.shape
    device = left.device

    # output buffer (float32)
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides in elements (not bytes) for each tensor
    stride_ab, stride_ak, stride_ah = left.stride()   # (b,i,k,d)
    stride_bb, stride_bk, stride_bh = right.stride()
    stride_cb, stride_ci, stride_ch = out.stride()

    # block sizes – tuned for Hopper SM90 (shared‑memory friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # launch grid: one (batch,head) program per (b,d)
    grid = (B * D, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    trimul_matmul_kernel[grid](
        left, right, out,
        B, N, D,
        stride_ab, stride_ak, stride_ah,
        stride_bb, stride_bk, stride_bh,
        stride_cb, stride_ci, stride_ch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,   # enough parallelism for H200
    )
    return out  # already fp32


# ----------------------------------------------------------------------
# Custom kernel entry point (the function required by the grader)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    data = (input, mask, weights, config)
        input : [B, N, N, dim]    (float16 / bfloat16)
        mask  : [B, N, N]         (0/1, may be omitted)
        weights: dict with all linear / norm parameters
        config : dict with "dim" and "hidden_dim"
    returns: tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dim (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – we keep them in torch for simplicity
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]      # [hidden, dim]
    right_proj_w = weights["right_proj.weight"]     # [hidden, dim]

    # torch.nn.functional.linear does: out = x @ w.T
    left = F.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Apply mask if present (mask may be all‑ones)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul einsum → fused matmul in Triton
    #    left/right are (B,N,N,hidden); we cast to fp16 for the kernel.
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Triton kernel gives fp32 result
    out = trimul_matmul(left_fp16, right_fp16)   # [B,N,N,hidden] float32

    # ------------------------------------------------------------------
    # 6. Final LayerNorm, gating, linear projection back to dim
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    out = out * out_gate  # out_gate is still fp32 (broadcast over hidden)

    out = F.linear(out, to_out_w)   # back to [B,N,N,dim]
    return out  # float32 as required
