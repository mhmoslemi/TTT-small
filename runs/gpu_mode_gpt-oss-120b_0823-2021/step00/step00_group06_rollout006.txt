"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part of TriMul is the 4‑D “einsum”

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

where `left` and `right` have shape (B, N, N, H) and H = hidden_dim.
For every head `h` this is exactly a batched matrix‑multiply
A_h @ B_h.T  (A_h = left[..., :, :, h] , B_h = right[..., :, :, h]).
The kernel below implements a fused batched GEMM for all heads and batches.
All LayerNorms and Linear projections are kept in PyTorch – they are cheap
compared to the O(N³·H) contraction.  The kernel works in FP16 internally
and writes the result back in FP32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    # pointers
    a_ptr,               # left  [B, N, N, H]   (contiguous)
    b_ptr,               # right [B, N, N, H]   (contiguous)
    c_ptr,               # out   [B, N, N, H]   (contiguous)
    # problem size
    B, N, H,
    # strides (in elements, not bytes)
    stride_ah, stride_ai, stride_ak, stride_ahd,   # a: batch, i, k, head
    stride_bh, stride_bj, stride_bk, stride_bhd,   # b: batch, j, k, head
    stride_ch, stride_cm, stride_cn, stride_chd,   # c: batch, i, j, head
    # compile‑time block sizes
    BLOCK_M: tl.constexpr,  # i‑tile
    BLOCK_N: tl.constexpr,  # j‑tile
    BLOCK_K: tl.constexpr,  # k‑tile
):
    """
    Computes, for one (batch, head) pair:
        C[i, j] = Σ_k A[i, k] * B[j, k]   (note the transposition on B)
    The kernel is launched with a 1‑D grid of size B*H.
    """

    pid = tl.program_id(0)                     # 0 … B*H‑1
    total = B * H
    if pid >= total:
        return

    # decode batch and head from pid
    b = pid // H                               # batch index
    h = pid % H                                 # head  index

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    # block coordinates in the i‑j matrix
    block_i = tl.program_id(1)                 # tile row
    block_j = tl.program_id(2)                 # tile col

    i_base = block_i * BLOCK_M
    j_base = block_j * BLOCK_N

    # ------------------------------------------------------------------
    # Accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the k‑dimension
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k_start, BLOCK_K)

        # offsets within the current tile
        offs_i = i_base + tl.arange(0, BLOCK_M)
        offs_j = j_base + tl.arange(0, BLOCK_N)
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # masks for out‑of‑bounds loads
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load A tile: shape (BLOCK_M, cur_k)
        # ------------------------------------------------------------------
        a_ptrs = (
            a_ptr
            + b * stride_ah
            + h * stride_ahd
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # ------------------------------------------------------------------
        # Load B tile: shape (BLOCK_N, cur_k)
        # Note: we need B[j, k] (no transposition in memory) → we will
        # multiply a[i,k] * b[j,k] and then sum over k.
        # ------------------------------------------------------------------
        b_ptrs = (
            b_ptr
            + b * stride_bh
            + h * stride_bhd
            + offs_j[:, None] * stride_bj
            + offs_k[None, :] * stride_bk
        )
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # ------------------------------------------------------------------
        # Compute outer‑product over k and accumulate
        # ------------------------------------------------------------------
        # (BLOCK_M, cur_k) @ (BLOCK_N, cur_k).T  -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b, allow_tf32=False)   # dot does the Σ_k internally

    # ------------------------------------------------------------------
    # Write back the result
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + b * stride_ch
        + h * stride_chd
        + offs_i[:, None] * stride_cm
        + offs_j[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side helper to launch the kernel
# ----------------------------------------------------------------------
def launch_trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: float16 tensors of shape [B, N, N, H]
    returns: float32 tensor of shape [B, N, N, H]
    """
    assert left.is_cuda and right.is_cuda
    B, N, N2, H = left.shape
    assert N == N2
    out = torch.empty_like(left, dtype=torch.float32)

    # strides in elements (torch strides are in element count)
    s_ah, s_ai, s_ak, s_ahd = left.stride()
    s_bh, s_bj, s_bk, s_bhd = right.stride()
    s_ch, s_cm, s_cn, s_chd = out.stride()

    # Block sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(B * H, 1),          # program id 0 = (batch, head) pair
        triton.cdiv(N, BLOCK_M),       # program id 1 = i‑tile
        triton.cdiv(N, BLOCK_N),       # program id 2 = j‑tile
    )

    trimul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_ah,
        s_ai,
        s_ak,
        s_ahd,
        s_bh,
        s_bj,
        s_bk,
        s_bhd,
        s_ch,
        s_cm,
        s_cn,
        s_chd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# Full TriMul forward (mostly PyTorch, core contraction in Triton)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Args
    ----
    data : (input_tensor, mask, weights, config)
        input_tensor : [B, N, N, dim]   (float32)
        mask         : [B, N, N]        (bool / float)
        weights      : dict of parameters (all float32)
        config       : {'dim': int, 'hidden_dim': int}
    Returns
    -------
    output : [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # LayerNorm and Linear projections (PyTorch – cheap)
    # --------------------------------------------------------------
    # norm
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)                     # work in fp16 for the rest

    # linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # mask handling (mask may be None)
    if mask is not None:
        mask = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"])).to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Core contraction – Triton kernel
    # --------------------------------------------------------------
    # left/right are [B,N,N,hidden_dim] in fp16
    out = launch_trimul_gemm(left, right)   # returns fp32

    # --------------------------------------------------------------
    # Post‑processing (LayerNorm, gating, final linear)
    # --------------------------------------------------------------
    # to_out_norm (hidden_dim)  – fp32
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)

    # final linear back to dim (no bias)
    out = F.linear(out, weights["to_out.weight"])

    return out
