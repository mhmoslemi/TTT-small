"""
TriMul “outgoing” implementation.

The heavy part of the module is the tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched “matrix‑multiply‑over‑k” with a separate hidden
dimension *d*.  The kernel below computes this contraction by iterating
over the *k* dimension in blocks, loading a (BLOCK_K, BLOCK_D) tile from
`left` and `right`, multiplying element‑wise, and reducing the k‑axis.
All other operations (layer‑norms, linear projections, gates) are
performed with PyTorch which is already highly optimized.

The kernel is launched with one program‑id per (batch, i, j) tuple,
so the output for a given (b,i,j,*) is independent and can be
computed in parallel.  The final result is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------
# Triton kernel: compute the triple‑mul contraction
# -------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,                # const float*  [B, N, N, H]
    right_ptr,               # const float*  [B, N, N, H]
    out_ptr,                 #       float*  [B, N, N, H]
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_ri, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_K: tl.constexpr,   # tile size over k
    BLOCK_H: tl.constexpr,   # tile size over hidden dim
):
    pid = tl.program_id(0)

    # ---------------------------------------------------------
    # decompose pid into (b,i,j)
    # ---------------------------------------------------------
    total_ij = N * N
    b = pid // total_ij
    rem = pid % total_ij
    i = rem // N
    j = rem % N

    # offsets for the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)

    # accumulator for the hidden dimension
    acc = tl.zeros((BLOCK_H,), dtype=tl.float32)

    # ---------------------------------------------------------
    # loop over k in blocks
    # ---------------------------------------------------------
    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_start * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # pointers to the (k, h) tiles
        left_tile_ptr = (
            left_ptr
            + b * stride_lb
            + i * stride_li
            + k_offset[:, None] * stride_lk
            + offs_h[None, :] * stride_lh
        )
        right_tile_ptr = (
            right_ptr
            + b * stride_rb
            + j * stride_ri
            + k_offset[:, None] * stride_rk
            + offs_h[None, :] * stride_rh
        )

        # load tiles (BLOCK_K, BLOCK_H)
        left_tile = tl.load(left_tile_ptr, mask=mask_k[:, None], other=0.0)
        right_tile = tl.load(right_tile_ptr, mask=mask_k[:, None], other=0.0)

        # element‑wise product and reduction over k
        prod = left_tile * right_tile
        acc += tl.sum(prod, axis=0)          # shape (BLOCK_H,)

    # ---------------------------------------------------------
    # write back, applying mask for the hidden dimension
    # ---------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + i * stride_oi
        + j * stride_oj
        + offs_h * stride_oh
    )
    mask_h = offs_h < H
    tl.store(out_ptrs, acc, mask=mask_h)


# -------------------------------------------------------------
# Host‑side entry point
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm on the last dimension ----------
    eps = 1e-5
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
        eps=eps,
    )  # [B,N,N,dim]

    # ---------- Linear projections (bias‑free) ----------
    left = F.linear(
        x,
        weights["left_proj.weight"].to(dtype),  # shape [hidden_dim, dim]
        bias=None,
    )
    right = F.linear(
        x,
        weights["right_proj.weight"].to(dtype),
        bias=None,
    )  # both [B,N,N,hidden_dim]

    # ---------- Apply mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gates ----------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(dtype), bias=None)
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(dtype), bias=None)
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(dtype), bias=None)
    )  # [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ---------- Triton contraction ----------
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # strides (in elements) for the tensors
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    BLOCK_K = 128   # works well for N up to 1024 on Hopper
    BLOCK_H = 32    # hidden dim is a multiple of 32 in all configs

    total_programs = B * N * N
    grid = (triton.cdiv(total_programs, 1),)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_lh=stride_lh,
        stride_rb=stride_rb,
        stride_ri=stride_ri,
        stride_rk=stride_rk,
        stride_rh=stride_rh,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_oh=stride_oh,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ---------- Output post‑processing ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype), bias=None)

    return out
