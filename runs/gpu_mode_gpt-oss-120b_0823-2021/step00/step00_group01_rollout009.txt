"""
TriMul “outgoing” implementation.

The expensive part of the forward pass is the
```
out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
```
which is a batched matrix‑multiply of the hidden dimension
`d` (called hidden_dim).  By reshaping the tensors to
[batch*hidden_dim, N, N] we can compute

    out = left @ right.transpose(-2,-1)

with a single Triton kernel that performs a tiled GEMM for each
(batch*hidden_dim) matrix.  All other operations (layer‑norms,
linear projections, sigmoid gates and the final linear) are kept in
PyTorch because they are cheap relative to the N³ work.

The kernel works for any batch size (1 or 2), any sequence length up
to 1024 and any hidden dimension that is a multiple of the tile size.
The final result is returned as `float32`.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B shape: [M, N]; C shape: [M, N])
# One kernel instance per (batch*hidden) matrix.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix dimensions (all int32)
    stride_am, stride_ak,               # strides for A (batch, row, col)
    stride_bm, stride_bk,               # strides for B (batch, row, col)
    stride_cm, stride_cn,               # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a single batch (already folded batch*head).
    All matrices are stored in row‑major order.
    """
    pid = tl.program_id(0)

    # ---------------------------------------------------------------
    # 1) block index decomposition (same pattern as in the triton docs)
    # ---------------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # “grouped” launch to improve L2 cache reuse
    group_id = pid // (GROUP_SIZE_M * num_blocks_n)
    first_pid_in_group = group_id * GROUP_SIZE_M * num_blocks_n
    pid_in_group = pid - first_pid_in_group

    num_pid_m = tl.minimum(GROUP_SIZE_M, num_blocks_m - group_id * GROUP_SIZE_M)
    pid_m = pid_in_group // num_blocks_n
    pid_n = pid_in_group %  num_blocks_n

    # global offsets for the block
    offs_m = (group_id * GROUP_SIZE_M + pid_m) * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---------------------------------------------------------------
    # 2) pointers to the start of the tile
    # ---------------------------------------------------------------
    a_ptrs = a_ptr + (offs_m[:, None] * stride_am) + (tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_ptr + (offs_n[:, None] * stride_bm) + (tl.arange(0, BLOCK_K)[None, :] * stride_bk)

    # ---------------------------------------------------------------
    # 3) accumulation
    # ---------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None], other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[:, None], other=0.0)   # B is NOT transposed yet
        # note: we need Bᵀ, so we dot a row with b row, which is equivalent
        # to a @ bᵀ because b is stored row‑major.  Using tl.dot on the
        # inner dimension does exactly that.
        acc += tl.dot(a, b, trans_b=True)   # trans_b flips the second operand
        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ---------------------------------------------------------------
    # 4) write back
    # ---------------------------------------------------------------
    c_ptrs = c_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side helper to launch the kernel
# ----------------------------------------------------------------------
def batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a, b : torch tensors of shape [B*H, N, N]   (float16)
    returns c of same shape, computed as a @ bᵀ   (float16)
    """
    assert a.is_cuda and b.is_cuda
    B_H, N, K = a.shape
    assert b.shape == (B_H, N, K)

    # output tensor
    c = torch.empty_like(a, dtype=torch.float16)

    # Tile sizes – chosen to fit H200 SMs and L2.  64×64 is a good default.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    GROUP_SIZE_M = 8

    grid = lambda meta: (
        triton.cdiv(N, meta['BLOCK_M']) *
        triton.cdiv(N, meta['BLOCK_N']) *
        B_H,   # one program per (batch*head) matrix
    )

    batched_matmul_kernel[grid](
        a,
        b,
        c,
        M=N,
        N=N,
        K=K,
        stride_am=a.stride(0),
        stride_ak=a.stride(2),      # because a is [B*H, N, K] (row‑major)
        stride_bm=b.stride(0),
        stride_bk=b.stride(2),
        stride_cm=c.stride(0),
        stride_cn=c.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
    )
    return c


# ----------------------------------------------------------------------
# The public entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.
    Arguments
    ----------
    data : tuple (input, mask, weights, config)
        input  : [B, N, N, dim]   torch.float32 on CUDA
        mask   : [B, N, N]       (may be all‑ones)
        weights: dict of tensors (already on CUDA)
        config : dict with keys "dim", "hidden_dim"
    Returns
    -------
    output : torch.Tensor  [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) Layer‑norm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )               # [B,N,N,dim]  fp32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → hidden_dim
    # ------------------------------------------------------------------
    # shape helpers for torch.nn.functional.linear: (…, in_features) @ (out_features, in_features).T
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"]) # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3) Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul:   out = Σ_k left[...,k,:] * right[...,k,:]  (einsum)
    #    We move this to Triton.
    # ------------------------------------------------------------------
    # Cast to fp16 for the GEMM (speed / memory)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # reshape to (B*hidden, N, N) for the kernel
    left_mat = left_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # GEMM: left @ rightᵀ   →   shape (B*hidden, N, N)
    out_mat_fp16 = batched_matmul(left_mat, right_mat)

    # reshape back to [B, N, N, hidden]
    out = out_mat_fp16.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Final LayerNorm, gating and linear back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to dim, no bias

    return out
