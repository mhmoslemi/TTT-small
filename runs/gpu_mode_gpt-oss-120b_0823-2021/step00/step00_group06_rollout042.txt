"""
TriMul “outgoing” implementation.
*   LayerNorm, Linear and gating are performed in PyTorch (they are cheap).
*   The expensive N‑by‑N‑by‑H einsum
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
  is computed with a custom Triton kernel that treats the (b,i) axis as the
  M‑dimension, j as N and the hidden dimension H is tiled (BLOCK_D).
  The kernel performs a blocked matrix‑multiply for each hidden‑slice and
  accumulates in FP32, finally storing FP16 (the later PyTorch stages cast back
  to FP32).
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ------------------------------------------------------------
# Triton kernel for the 4‑D einsum (the N³ part)
# ------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,               # [B,N,N,H]  (b,i,k,d)
    right_ptr,              # [B,N,N,H]  (b,j,k,d)
    out_ptr,                # [B,N,N,H]  (b,i,j,d)
    B, N, H,
    stride_left_bs, stride_left_i, stride_left_k, stride_left_h,
    stride_right_bs, stride_right_j, stride_right_k, stride_right_h,
    stride_out_bs, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,   # tiles over (b,i)   => M = B*N
    BLOCK_N: tl.constexpr,   # tiles over j
    BLOCK_K: tl.constexpr,   # tiles over k
    BLOCK_D: tl.constexpr,   # tiles over hidden dim
):
    pid = tl.program_id(0)
    num_m = tl.cdiv(B * N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_n                      # which (b,i) tile
    pid_n = pid % num_n                       # which j tile

    # ----- global offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # (M,)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # (N,)
    offs_d = tl.arange(0, BLOCK_D)                     # (D,)

    # mask for out‑of‑range
    mask_m = offs_m < B * N
    mask_n = offs_n < N

    # recover batch index b and i from the flattened M index
    b_idx = offs_m // N
    i_idx = offs_m % N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ----- main K‑loop ----------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left : shape (M, K, D) --------------------------------
        left_ptrs = (
            left_ptr
            + b_idx[:, None, None] * stride_left_bs
            + i_idx[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_d[None, None, :] * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_m[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )                      # (M, K, D)

        # ----- load right : shape (N, K, D) -------------------------------
        right_ptrs = (
            right_ptr
            + b_idx[:, None, None] * stride_right_bs          # same batch
            + offs_n[None, :, None] * stride_right_j          # j dimension
            + offs_k[None, :, None] * stride_right_k
            + offs_d[None, None, :] * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_n[None, :, None] & mask_k[None, :, None],
            other=0.0,
        )                      # (N, K, D)

        # ----- compute per‑slice matrix multiplication --------------------
        # Loop over hidden slices (D is tiny, 128 → 4‑8 iterations)
        for d in range(0, BLOCK_D):
            # (M,K) * (N,K)^T  =>  (M,N)
            l_slice = left[:, :, d]          # (M,K)
            r_slice = right[:, :, d]         # (N,K)
            # Triton dot expects (M,K) @ (K,N) ; we need (N,K) → transpose
            acc[:, :, d] += tl.dot(l_slice, r_slice, trans_b=True)

    # ----- write result ----------------------------------------------------
    out_ptrs = (
        out_ptr
        + b_idx[:, None, None] * stride_out_bs
        + i_idx[:, None, None] * stride_out_i
        + offs_n[None, :, None] * stride_out_j
        + offs_d[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),    # store as FP16, later cast back to FP32 in PyTorch
        mask=mask_m[:, None, None] & mask_n[None, :, None],
    )


# ------------------------------------------------------------
# Host side helper
# ------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """Manual LayerNorm over the last dimension."""
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.
    Arguments
    ---------
    data : (input, mask, weights, config)
        input  – Tensor[B, N, N, dim]  (float32, on CUDA)
        mask   – Tensor[B, N, N]       (float32 or bool, on CUDA)
        weights – dict of pre‑loaded model parameters
        config  – dict containing "dim" and "hidden_dim"
    Returns
    -------
    Tensor[B, N, N, dim]  (float32)
    """
    # ------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    B, N, _, _ = inp.shape

    # ------------------------------------------------------------
    # 1. LayerNorm + Linear projections (PyTorch – cheap)
    # ------------------------------------------------------------
    x = _layer_norm(inp, weights["norm.weight"], weights["norm.bias"])

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # mask handling
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                       # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # gating (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------
    # 2. Expensive einsum via Triton
    # ------------------------------------------------------------
    # Prepare contiguous tensors in fp16 for the kernel
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out_h = torch.empty_like(left_h)          # same shape [B,N,N,hidden]

    # strides (in elements, not bytes) for the 4‑D tensors
    stride_left_bs, stride_left_i, stride_left_k, stride_left_h = left_h.stride()
    stride_right_bs, stride_right_j, stride_right_k, stride_right_h = right_h.stride()
    stride_out_bs, stride_out_i, stride_out_j, stride_out_h = out_h.stride()

    # launch configuration
    BLOCK_M = 64   # tile over (B,N) → up to 64*B per block
    BLOCK_N = 64   # tile over j
    BLOCK_K = 32   # tile over k
    BLOCK_D = 32   # tile over hidden dimension (hidden≤384, fits)

    grid_m = triton.cdiv(B * N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (grid_m * grid_n,)

    trimul_einsum_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        hidden,
        stride_left_bs,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_bs,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_bs,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # Convert back to FP32 for the remaining ops
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------
    # 3. Final LayerNorm, gating and output linear (PyTorch)
    # ------------------------------------------------------------
    out = _layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
