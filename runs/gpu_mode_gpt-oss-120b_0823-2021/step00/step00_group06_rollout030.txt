"""
TriMul “outgoing” implementation.

The heavy O(N³·H) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed by a custom Triton kernel.  All surrounding
LayerNorm / Linear / sigmoid work is kept in PyTorch so that we
only need to launch one kernel – this gives a large speed‑up on
H200 while keeping the code simple and numerically stable.

The kernel tiles the (i,j) output plane (BLOCK_M × BLOCK_N) and
reduces over the k dimension in blocks of size BLOCK_K.
Each thread computes a vector of length BLOCK_H (the hidden
dimension) and accumulates in float32.  Input/weight tensors are
stored as fp16 to exploit tensor‑core throughput; the final
output is cast back to fp32 before the remaining PyTorch
post‑processing.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------
# Triton kernel:  out = Σ_k left[...,k,:] * right[...,k,:]
# ---------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr,           # *[B, N, N, H]  (i,k,h)
    right_ptr,          # *[B, N, N, H]  (j,k,h)
    out_ptr,            # *[B, N, N, H]  (i,j,h)
    B, N, H,
    stride_l_bn, stride_l_ik, stride_l_h,   # left strides
    stride_r_bn, stride_r_jk, stride_r_h,   # right strides
    stride_o_bn, stride_o_ij, stride_o_h,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch dimension
    pid_i = tl.program_id(0)          # output i‑block
    pid_j = tl.program_id(1)          # output j‑block

    # ------------------------------------------------------------------
    # Offsets for the current block
    # ------------------------------------------------------------------
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds in i/j
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Create a 2‑D grid of (i,j) coordinates for this block
    i_idx = tl.broadcast_to(i_offset[:, None], (BLOCK_M, BLOCK_N))
    j_idx = tl.broadcast_to(j_offset[None, :], (BLOCK_M, BLOCK_N))

    # ------------------------------------------------------------------
    # Accumulator: one vector of length BLOCK_H for each (i,j)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k in tiles
    # ------------------------------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        k_offset = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ------------------------------------------------------------------
        # Load left[b,i,k,:]  shape (BLOCK_M, BLOCK_K, BLOCK_H)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + pid_b * stride_l_bn
            + i_idx[:, :, None] * stride_l_ik
            + k_offset[None, :, None] * stride_l_h
            + tl.arange(0, BLOCK_H)[None, None, :] * 1
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0).to(tl.float32)

        # ------------------------------------------------------------------
        # Load right[b,j,k,:] shape (BLOCK_N, BLOCK_K, BLOCK_H)
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + pid_b * stride_r_bn
            + j_idx[None, :, None] * stride_r_jk
            + k_offset[:, None, None] * stride_r_h
            + tl.arange(0, BLOCK_H)[None, None, :] * 1
        )
        right = tl.load(right_ptrs, mask=mask_j[None, :, None] & mask_k[:, None, None], other=0.0).to(tl.float32)

        # ------------------------------------------------------------------
        # Multiply‑accumulate over k (broadcast over H)
        #   left: (M,K,H)   right: (N,K,H) -> need (M,N,H)
        # ------------------------------------------------------------------
        # transpose right to (K,N,H) then broadcast K dimension
        # compute elementwise product and sum over K
        acc += tl.sum(left[:, :, None, :] * right[None, :, :, :], axis=1)  # sum over K

    # ------------------------------------------------------------------
    # Write result
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_o_bn
        + i_idx[:, :, None] * stride_o_ij
        + j_idx[:, :, None] * stride_o_h
        + tl.arange(0, BLOCK_H)[None, None, :] * 1
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, :, None] & mask_j[:, :, None])


# ---------------------------------------------------------------
# Helper: simple LayerNorm that works on the last dimension
# ---------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    return F.layer_norm(x, x.shape[-1:], weight, bias, eps)


# ---------------------------------------------------------------
# Main entry point
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, D]   float32
            - mask          : [B, N, N]    (bool / 0‑1 float)
            - weights       : dict of torch tensors
            - config        : dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]        # hidden dimension

    # ------------------------------------------------------------------
    # 1) LayerNorm on input (dim = D)
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – cast to fp16 for speed
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) – stay in fp32 for stability, then back to fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"])).to(torch.float32)  # used later

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton kernel for the O(N³·H) contraction
    # ------------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float16)   # [B,N,N,H]

    # strides (contiguous layout)
    stride_l_bn, stride_l_ik, stride_l_h = left.stride()
    stride_r_bn, stride_r_jk, stride_r_h = right.stride()
    stride_o_bn, stride_o_ij, stride_o_h = out.stride()

    BLOCK_M = 64   # tile size for i
    BLOCK_N = 64   # tile size for j
    BLOCK_K = 32   # tile size for reduction over k
    BLOCK_H = 32   # tile size for hidden dimension (must divide H, we pad via mask)

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B,
    )
    _trimul_einsum_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_bn, stride_l_ik, stride_l_h,
        stride_r_bn, stride_r_jk, stride_r_h,
        stride_o_bn, stride_o_ij, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
    )
    # Cast back to fp32 for the remaining ops
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm, gating and final linear
    # ------------------------------------------------------------------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.unsqueeze(-1)          # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])

    return out
