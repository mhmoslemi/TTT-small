"""
TriMul “outgoing” implementation.
-------------------------------------------------
The heavy part of the module is the ternary einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `d` is the hidden dimension (often 128‑384).  We implement this
reduction in Triton – one program processes a (BLOCK_M, BLOCK_N)
tile of the (i,j) plane for a single batch.  For every hidden channel
we perform a small GEMM `left_d (M×K) @ right_d.T (K×N)` and store the
result.  All surrounding operations (LayerNorm, linear projections,
masking, gating and the final linear) stay in native PyTorch – they are
memory‑bound and cheap compared to the O(N³) reduction.

The kernel works for any `N` (up to 1024 in the tests) and any hidden
size that fits into shared memory (we tile the hidden dimension in
chunks of `BLOCK_H`).  The final output is cast back to `float32`.
"""

import torch
import triton
import triton.language as tl
from torch import nn, einsum


# ---------------------------------------------------------------------------
# Triton kernel:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# ---------------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,                     # batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Triton implementation of the core ternary multiplication.
    Each program instance computes a (BLOCK_M, BLOCK_N) tile of the
    (i,j) plane for one batch element.  Hidden dimension is tiled with
    BLOCK_H (e.g. 32) to stay inside shared memory.
    """
    pid = tl.program_id(0)
    # linearisation: pid = b * (num_i * num_j) + i_tile * num_j + j_tile
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    b_id   = pid // (num_i * num_j)
    pid_xy = pid % (num_i * num_j)
    i_tile = pid_xy // num_j
    j_tile = pid_xy % num_j

    # offsets for the i / j axes
    offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # allocate accumulator for the current tile (M, N, H)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # loop over k tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left slice [M, K, H]   (broadcast over j)
        left = tl.load(
            left_ptr
            + b_id * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_left_h,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )  # (M, K, H)

        # load right slice [N, K, H]  (broadcast over i)
        right = tl.load(
            right_ptr
            + b_id * stride_right_b
            + offs_j[:, None, None] * stride_right_j
            + offs_k[None, :, None] * stride_right_k
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_right_h,
            mask=mask_j[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )  # (N, K, H)

        # compute contribution for this k-tile:
        #   Σ_k left[i,k,h] * right[j,k,h]
        # we do an explicit reduction over the K dimension
        # (M, K, H) * (N, K, H) → (M, N, H)
        # using broadcasting + sum
        acc += tl.sum(left[:, None, :, :] * right[None, :, :, :], axis=2)

    # write back
    out = acc.to(tl.float32)
    tl.store(
        out_ptr
        + b_id * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + tl.arange(0, BLOCK_H)[None, None, :] * stride_out_h,
        out,
        mask=mask_i[:, None, None] & mask_j[None, :, None],
    )


# ---------------------------------------------------------------------------
# Helper: fused LayerNorm (no bias‑gradient needed)
# ---------------------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]   weight/bias: (C,)
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias


# ---------------------------------------------------------------------------
# Entry point ---------------------------------------------------------------
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # -----------------------------------------------------------------------
    # 1. Pre‑processing in PyTorch (LayerNorm + Linear + gating)
    # -----------------------------------------------------------------------
    x = layer_norm(input_tensor, weights["norm.weight"], weights["norm.bias"])
    # linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # apply mask if present
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 2. Core ternary multiplication – Triton
    # -----------------------------------------------------------------------
    # shapes: (B,N,N,hidden)
    left = left.view(B, N, N, hidden).contiguous()
    right = right.view(B, N, N, hidden).contiguous()
    out = torch.empty_like(left)  # allocate output (B,N,N,hidden)

    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 32                     # hidden dimension is tiled

    grid = (
        triton.cdiv(B, 1)                     # batch dimension (we keep it inside program id)
        * triton.cdiv(N, BLOCK_M)
        * triton.cdiv(N, BLOCK_N),
    )
    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # -----------------------------------------------------------------------
    # 3. Post‑processing (norm, gating, final linear)
    # -----------------------------------------------------------------------
    out = layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
