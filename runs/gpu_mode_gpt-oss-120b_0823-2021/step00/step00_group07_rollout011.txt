"""
TriMul (outgoing) – Triton‑accelerated core.

The heavy part of the module is the 4‑D “einsum”

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right are the linearly projected and gated tensors
(shape [B, N, N, D]).  The kernel tiles the reduction dimension *k*
and computes a block of (i, j, d) per program instance, accumulating the
partial products in fp32 and finally writing fp32 results.
All surrounding operations (LayerNorm, Linear, sigmoid gates, final
Linear+LayerNorm) are expressed in PyTorch – they are cheap compared
with the O(B·N³·D) reduction and therefore do not need to be fused.
The kernel works for any batch size (1‑2), any sequence length (≤1024)
and any hidden dimension D (typically 128, 384 or 768).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel: blockwise sum‑over‑k of element‑wise products
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # strides (in elements, not bytes!)
    stride_bL, stride_iL, stride_kL, stride_dL,   # left  [B, i, k, d]
    stride_bR, stride_jR, stride_kR, stride_dR,   # right [B, j, k, d]
    stride_bO, stride_iO, stride_jO, stride_dO,   # out   [B, i, j, d]
    # sizes
    B, N, D,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_D: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Each program computes a tile of shape (BLOCK_I, BLOCK_J, BLOCK_D)
    of the output tensor for a fixed batch index.
    The reduction over k is tiled with BLOCK_K.
    """
    pid = tl.program_id(0)                     # linear program id
    # decode program id into (b, i_tile, j_tile)
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    b = pid // (num_i * num_j)
    tmp = pid % (num_i * num_j)
    i_tile = tmp // num_j
    j_tile = tmp % num_j

    # coordinate offsets for the output tile
    offs_i = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_d = tl.arange(0, BLOCK_D)

    # masks for out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # broadcast masks to the 3‑D tile
    mask_ij = mask_i[:, None] & mask_j[None, :]          # (BI, BJ)
    mask_ij_d = mask_ij[:, :, None] & mask_d[None, None, :]  # (BI, BJ, BD)

    # pointers to the beginning of the tile in the output tensor
    out_ptr_tile = out_ptr + \
        b * stride_bO + \
        offs_i[:, None, None] * stride_iO + \
        offs_j[None, :, None] * stride_jO + \
        offs_d[None, None, :] * stride_dO

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ----------------------------------------------------------
    # reduction over k
    # ----------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left : [BI, BK, BD]
        left_ptrs = left_ptr + \
            b * stride_bL + \
            offs_i[:, None] * stride_iL + \
            cur_k[None, :] * stride_kL + \
            offs_d[None, None, :] * stride_dL
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] &
                                          mask_d[None, None, :], other=0.0).to(tl.float32)

        # right : [BJ, BK, BD]   (note the j dimension)
        right_ptrs = right_ptr + \
            b * stride_bR + \
            offs_j[:, None] * stride_jR + \
            cur_k[None, :] * stride_kR + \
            offs_d[None, None, :] * stride_dR
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] &
                                           mask_d[None, None, :], other=0.0).to(tl.float32)

        # multiply and reduce over k
        # left: (BI, BK, BD)   -> (BI, 1, BK, BD)
        # right: (BJ, BK, BD)  -> (1, BJ, BK, BD)
        # after broadcasting we sum over k (axis=2)
        prod = left[:, None, :, :] * right[None, :, :, :]   # (BI, BJ, BK, BD)
        acc += tl.sum(prod, axis=2)                         # (BI, BJ, BD)

    # write result (only where in‑bounds)
    tl.store(out_ptr_tile, acc, mask=mask_ij_d)


# --------------------------------------------------------------
# Host‑side wrapper
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward of TriMul (outgoing) as described in the
    AlphaFold3 paper.

    Args:
        data: (input, mask, weights, config)
            - input  : Tensor[B, N, N, dim] (float32)
            - mask   : Tensor[B, N, N]      (float32 or bool)
            - weights: dict with all linear / layer‑norm parameters
            - config : dict with "dim" and "hidden_dim"

    Returns:
        Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # called D in the kernel

    # ----------------------------------------------------------
    # Helper: linear w/o bias (weights are stored as (out, in))
    # ----------------------------------------------------------
    def linear(x, w):
        # x: [..., in], w: (out, in)
        return F.linear(x, w)

    # ----------------------------------------------------------
    # 1) LayerNorm on the last dimension (dim)
    # ----------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=1e-5)

    # ----------------------------------------------------------
    # 2) Projections (no bias) and gates
    # ----------------------------------------------------------
    left = linear(x, weights["left_proj.weight"])      # [B,N,N,hidden]
    right = linear(x, weights["right_proj.weight"])

    # mask handling – mask is [B,N,N]; we expand to [...,1] and cast to float
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 3) Core Triton reduction (outgoing einsum)
    # ----------------------------------------------------------
    # allocate output tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides in elements (not bytes)
    s_bL, s_iL, s_kL, s_dL = left.stride()
    s_bR, s_jR, s_kR, s_dR = right.stride()
    s_bO, s_iO, s_jO, s_dO = out.stride()

    # tuning parameters – chosen to fit H200 SM90 and typical N
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_D = 32        # hidden_dim is a multiple of 32 in all test configs
    BLOCK_K = 32

    # total number of programs = B * ceil(N/BLOCK_I) * ceil(N/BLOCK_J)
    grid_i = triton.cdiv(N, BLOCK_I)
    grid_j = triton.cdiv(N, BLOCK_J)
    total_blocks = B * grid_i * grid_j
    grid = (total_blocks,)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        stride_bL=s_bL, stride_iL=s_iL, stride_kL=s_kL, stride_dL=s_dL,
        stride_bR=s_bR, stride_jR=s_jR, stride_kR=s_kR, stride_dR=s_dR,
        stride_bO=s_bO, stride_iO=s_iO, stride_jO=s_jO, stride_dO=s_dO,
        B=B, N=N, D=hidden_dim,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_D=BLOCK_D, BLOCK_K=BLOCK_K,
        num_warps=4,   # empirical good value on Hopper
    )

    # ----------------------------------------------------------
    # 4) Final LayerNorm + gating + linear back to dim
    # ----------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)
    out = out * out_gate
    out = linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
