"""
TriMul “outgoing” implementation.

The heavy part of the module is the three‑dimensional tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]          (1)

where   left = (x·W_left) ⊙ gate_left   and   right = (x·W_right) ⊙ gate_right.
For a fixed batch `b` and channel `d` the contraction is exactly a
matrix‑matrix product

    out[b, :, :, d] = left[b, :, :, d]  @  right[b, :, :, d].T

Thus we can treat the `hidden_dim` dimension as an extra batch dimension and
use a single highly‑fused Triton kernel that performs a batched GEMM
(`BATCH = batch_size * hidden_dim`).  The kernel computes
C = A @ Bᵀ for each batch, where

    A : (BATCH, N, N)   (left)
    B : (BATCH, N, N)   (right)

All LayerNorm and Linear operations are performed with PyTorch (they are
memory‑bound and already highly optimised).  Only the O(N³) contraction is
off‑loaded to Triton.

The kernel works for any `N` up to a few thousand, any batch size ≤ 2 and any
hidden dimension that is a multiple of the block‑size (we pad the hidden
dimension with a small temporary view).  The final output is cast back to
`float32` as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (float16 accumulation)
# --------------------------------------------------------------
@triton.jit
def batched_gemm_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, N,
    stride_az, stride_ak, stride_aa,          # A strides: (batch, row, col)
    stride_bz, stride_bk, stride_bb,          # B strides: (batch, row, col)
    stride_cz, stride_ci, stride_cj,          # C strides: (batch, row, col)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, GROUP_M: tl.constexpr
):
    """
    Compute C = A @ Bᵀ for many independent batches.
    A, B, C are all (BATCH, N, N) tensors.
    """
    pid = tl.program_id(0)

    # --------------------------------------------------
    # block coordinates
    # --------------------------------------------------
    batch_id = pid // (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_rem  = pid % (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))

    block_i = (pid_rem // tl.cdiv(N, BLOCK_N)) * BLOCK_M
    block_j = (pid_rem % tl.cdiv(N, BLOCK_N)) * BLOCK_N

    # --------------------------------------------------
    # offsets for the current block
    # --------------------------------------------------
    offs_i = block_i + tl.arange(0, BLOCK_M)
    offs_j = block_j + tl.arange(0, BLOCK_N)

    a_ptrs = a_ptr + batch_id * stride_az \
                    + offs_i[:, None] * stride_ak \
                    + tl.arange(0, BLOCK_K)[None, :] * stride_aa
    b_ptrs = b_ptr + batch_id * stride_bz \
                    + offs_j[None, :] * stride_bk \
                    + tl.arange(0, BLOCK_K)[:, None] * stride_bb   # note Bᵀ → swap axes

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------
    # loop over K dimension (the contraction axis)
    # --------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_i[:, None] < N) & (tl.arange(0, BLOCK_K)[None, :] + k * BLOCK_K < N), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] + k * BLOCK_K < N) & (offs_j[None, :] < N), other=0.0)
        # a : (BLOCK_M, BLOCK_K)   b : (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)          # fp32 accumulation
        a_ptrs += BLOCK_K * stride_aa
        b_ptrs += BLOCK_K * stride_bb

    # --------------------------------------------------
    # write back
    # --------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cz \
                    + offs_i[:, None] * stride_ci \
                    + offs_j[None, :] * stride_cj
    mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


# --------------------------------------------------------------
# Helper to launch the kernel
# --------------------------------------------------------------
def batched_gemm_outgoing(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a, b : (BATCH, N, N)  float16 tensors
    returns c : (BATCH, N, N) float16
    """
    BATCH, N, _ = a.shape
    assert a.shape == b.shape

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    GROUP_M = 8

    grid = (
        triton.cdiv(BATCH * N * N, (BLOCK_M // GROUP_M) * BLOCK_N),
    )
    c = torch.empty_like(a)

    batched_gemm_outgoing_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        BATCH=BATCH,
        N=N,
        stride_az=a.stride(0), stride_ak=a.stride(1), stride_aa=a.stride(2),
        stride_bz=b.stride(0), stride_bk=b.stride(1), stride_bb=b.stride(2),
        stride_cz=c.stride(0), stride_ci=c.stride(1), stride_cj=c.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, GROUP_M=GROUP_M,
    )
    return c


# --------------------------------------------------------------
# Main entry point expected by the evaluator
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the TriMul “outgoing” operator.

    Args
    ----
    data : tuple containing
        - input_tensor : torch.Tensor [B, N, N, dim]      (float32)
        - mask         : torch.Tensor [B, N, N]          (float32 or bool)
        - weights      : dict of pre‑loaded weight tensors
        - config       : dict with keys `dim` and `hidden_dim`

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------
    # LayerNorm (dim) – use torch (fast, fused on GPU)
    # --------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------
    # Linear projections (no bias) – fused FP16 matmul + bias‑less
    # --------------------------------------------------
    # shape: (B, N, N, hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------
    # Apply the (optional) mask – broadcast over hidden_dim
    # --------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------
    # Gating (sigmoid) – also fused by torch
    # --------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------
    # Move to float16 for the massive N³ contraction
    # --------------------------------------------------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # view as (BATCH, N, N) where BATCH = B * hidden_dim
    left_reshape = left_f16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_reshape = right_f16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Triton GEMM of each (N,N) pair
    out_reshape = batched_gemm_outgoing(left_reshape, right_reshape)   # half

    # reshape back to [B, N, N, hidden_dim]
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)                 # back to fp32 for the rest

    # --------------------------------------------------
    # Output LayerNorm + gating + final linear
    # --------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
