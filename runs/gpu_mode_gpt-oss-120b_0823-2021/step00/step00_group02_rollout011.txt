"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part of the forward pass is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is exactly a batched matrix‑multiply for every hidden‑dim slice d:

    out = left @ right.transpose(-2,-1)          # shape (B,N,N,hd)

The implementation below performs all cheap element‑wise work (LayerNorm,
linear projections, gating, masking) with PyTorch and hands the 3‑D matmul to a
custom Triton kernel.  The kernel works on FP16 (or BF16) for speed and
produces FP32 output as required.

The kernel is written for the Hopper H200 (sm_90) and uses 64×64 tiles with
a K‑tile of 32.  It is fully fused: loads, computes, and stores without any
intermediate materialisation on the host.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton batched matmul kernel (out = A @ B^T)  A,B : (B,N,N,HD)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_outgoing_kernel(
    # Pointers
    a_ptr, b_ptr, c_ptr,
    # Shapes
    B, N, HD,
    # Strides (in elements, not bytes)
    stride_ab, stride_ak, stride_ai, stride_ah,   # A: (B,N,K,HD)
    stride_bb, stride_bk, stride_bj, stride_bh,   # B: (B,N,K,HD)
    stride_cb, stride_ci, stride_cj, stride_ch,   # C: (B,N,N,HD)
    # Tile sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B^T for each (b, h) slice.
    A, B: (B, N, K, HD)  with K == N
    C: (B, N, N, HD)
    """
    pid = tl.program_id(0)                         # 1‑D grid over (b, h, tile_m, tile_n)
    total_tiles_m = tl.cdiv(N, BLOCK_M)
    total_tiles_n = tl.cdiv(N, BLOCK_N)
    # decompose pid
    pid_h = pid % HD
    pid_bn = (pid // HD) % B
    pid_tile = pid // (HD * B)
    pid_m = pid_tile // total_tiles_n
    pid_n = pid_tile % total_tiles_n

    # Offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # Pointers to the start of the tile (cast to ptr)
    a_base = a_ptr + pid_bn * stride_ab + pid_h * stride_ah
    b_base = b_ptr + pid_bn * stride_bb + pid_h * stride_bh
    c_base = c_ptr + pid_bn * stride_cb + pid_h * stride_ch

    # Load A tile: (BLOCK_M, BLOCK_K)
    a_ptrs = a_base + (offs_m[:, None] * stride_ai
                      + offs_k[None, :] * stride_ak)
    a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

    # Load B tile and transpose it on‑the‑fly: (BLOCK_N, BLOCK_K)
    b_ptrs = b_base + (offs_n[:, None] * stride_bj
                      + offs_k[None, :] * stride_bk)
    b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

    # Accumulator in fp32 for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = k_offset + offs_k
        kmask = cur_k < N

        a_ptrs = a_base + (offs_m[:, None] * stride_ai
                          + cur_k[None, :] * stride_ak)
        b_ptrs = b_base + (offs_n[:, None] * stride_bj
                          + cur_k[None, :] * stride_bk)

        a_tile = tl.load(a_ptrs, mask=mask_m[:, None] & kmask[None, :], other=0.0)
        b_tile = tl.load(b_ptrs, mask=mask_n[:, None] & kmask[None, :], other=0.0)

        # dot product over K
        acc += tl.dot(a_tile, b_tile)               # (M,N) += (M,K)·(N,K)

    # Write back
    c_ptrs = c_base + (offs_m[:, None] * stride_ci
                       + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: LayerNorm without bias (bias handled separately)
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """Fused LayerNorm (weight, bias are 1‑D of size C)."""
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the TriMul “outgoing” operator.

    Arguments
    ---------
    data : (input, mask, weights, config)
        input  – (B, N, N, dim)  float32 tensor on CUDA
        mask   – (B, N, N)       float32 or bool, broadcastable
        weights – dict with the following keys (all float32 tensors on CUDA):
            norm.weight, norm.bias,
            left_proj.weight, right_proj.weight,
            left_gate.weight, right_gate.weight, out_gate.weight,
            to_out_norm.weight, to_out_norm.bias,
            to_out.weight
        config – dict with entries:
            dim (int), hidden_dim (int)

    Returns
    -------
    output – (B, N, N, dim) float32 tensor on CUDA
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, w, cfg = data
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = layer_norm(inp, w["norm.weight"], w["norm.bias"])

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → (B,N,N,hidden)
    # ------------------------------------------------------------------
    # Use torch.nn.functional.linear for speed (weight shape: hidden x dim)
    left = F.linear(x, w["left_proj.weight"])
    right = F.linear(x, w["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask if present (mask shape (B,N,N,1))
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, w["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, w["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Main contraction: out = left @ right^T   (per hidden slice)
    #    Use FP16/BF16 for compute, cast back to FP32
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (H200 has strong fp16 performance)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)  # (B,N,N,hidden)

    # strides in element units
    stride_al = left_fp16.stride(0)  # B
    stride_ak = left_fp16.stride(1)  # N (i)
    stride_ai = left_fp16.stride(2)  # N (k)
    stride_ah = left_fp16.stride(3)  # hidden

    stride_bl = right_fp16.stride(0)
    stride_bk = right_fp16.stride(1)  # N (j)
    stride_bj = right_fp16.stride(2)  # N (k)
    stride_bh = right_fp16.stride(3)

    stride_cl = out_fp16.stride(0)
    stride_ci = out_fp16.stride(1)  # i
    stride_cj = out_fp16.stride(2)  # j
    stride_ch = out_fp16.stride(3)

    # Tile sizes – tuned for Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute total number of program instances
    total_tiles_m = triton.cdiv(N, BLOCK_M)
    total_tiles_n = triton.cdiv(N, BLOCK_N)
    total_programs = B * hidden * total_tiles_m * total_tiles_n

    grid = (total_programs,)

    batched_matmul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden,
        stride_al, stride_ai, stride_ak, stride_ah,
        stride_bl, stride_bj, stride_bk, stride_bh,
        stride_cl, stride_ci, stride_cj, stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                # good default for fp16 on Hopper
    )

    # Cast back to fp32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    out = layer_norm(out, w["to_out_norm.weight"], w["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, w["to_out.weight"])

    return out
