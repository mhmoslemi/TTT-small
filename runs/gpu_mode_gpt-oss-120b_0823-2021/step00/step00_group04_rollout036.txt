"""
TriMul (outgoing) – Triton accelerated core.
We compute

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are the projected and gated tensors.
The reduction over `k` is a batched matrix‑multiply for every hidden
dimension `d`.  The kernel fuses the K‑reduction and works on FP32
(accumulating in FP32, inputs may be FP16/BF16).  All LayerNorms,
Linear projections and gating are done in PyTorch – they are cheap
compared to the O(B·N²·D) reduction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: one matrix‑multiply per (batch, hidden_dim) pair
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,               # *[B, N, N, D]   left projected tensor
    right_ptr,              # *[B, N, N, D]   right projected tensor
    out_ptr,                # *[B, N, N, D]   output tensor
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,  # rows of the output tile
    BLOCK_N: tl.constexpr,  # cols of the output tile
    BLOCK_K: tl.constexpr,  # reduction tile size
):
    """
    Each program instance computes a BLOCK_M × BLOCK_N tile of the
    matrix product for a single (batch, hidden) pair.
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Decode program id → (batch, hidden, tile_i, tile_j)
    # ------------------------------------------------------------------
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)
    blocks_per_pair = num_blocks_i * num_blocks_j               # tiles per (b,d)

    pair_id   = pid // blocks_per_pair                           # index over B*D
    tile_id   = pid %  blocks_per_pair

    b = pair_id // D                                            # batch index
    d = pair_id %  D                                            # hidden index

    tile_i = tile_id // num_blocks_j
    tile_j = tile_id %  num_blocks_j

    # --------------------------------------------------------------
    # Offsets for the tiles
    # --------------------------------------------------------------
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------- load left[b, i, k, d] --------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + d * stride_left_d
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # ------------------- load right[b, j, k, d] -------------------
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_j[:, None] * stride_right_j
            + offs_k[None, :] * stride_right_k
            + d * stride_right_d
        )
        # note: we need right transposed on the K axis, so we load as (j,k)
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # ------------------- dot product over K -----------------------
        # left : (M, K) , right : (N, K) → we want Σ_k left[i,k]*right[j,k]
        acc += tl.dot(left, right, trans_b=False)   # (M,N)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + d * stride_out_d
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Host‑side helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, N, N, D]  (float16 / bfloat16 / float32)
    returns out of same shape, dtype float32 (converted after kernel)
    """
    B, N, _, D = left.shape
    out = torch.empty_like(left, dtype=torch.float16, device=left.device)

    # strides in elements (not bytes) for the 4‑D layout
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Tuning parameters (good for Hopper/H200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # total number of program instances
    num_blocks_i = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * D * num_blocks_i * num_blocks_j,)

    trimul_kernel[grid](
        left,
        right,
        out,
        B, N, D,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_j, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # convert back to FP32 for the remaining PyTorch ops
    return out.to(torch.float32)


# ----------------------------------------------------------------------
# Full custom kernel entry point (the function required by the grader)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator using Triton for the
    expensive O(N²·D) reduction.

    Arguments
    ---------
    data : (input, mask, weights, config)
        input   – tensor [B, N, N, dim] (float16/float32)
        mask    – tensor [B, N, N] (bool or 0/1)
        weights – dict of pre‑trained parameters
        config  – dict with keys "dim" and "hidden_dim"

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # PyTorch LayerNorm – cheap compared to the N³ work
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) – produce [B,N,N,hidden_dim]
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]           # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    # using torch.nn.functional.linear which handles broadcasting
    left = F.linear(x, left_proj_w)        # -> [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3. Apply mask (if provided)
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core TriMul reduction – Triton kernel
    #    Cast to FP16/BF16 for speed; accumulation in FP32.
    # --------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)
    out = trimul_triton(left, right)          # shape [B,N,N,hidden_dim], dtype float32

    # --------------------------------------------------------------
    # 6. Final LayerNorm, gated output and projection back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)
    out = out * out_gate                       # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # -> [B,N,N,dim]

    return out.to(torch.float32)
