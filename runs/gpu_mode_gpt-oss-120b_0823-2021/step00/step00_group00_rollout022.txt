"""
TriMul “outgoing” kernel (AlphaFold3).

The forward pass consists of
  1. Layer‑norm + four linear projections (pure‑torch, cheap).
  2. Masking and gated multiplication (torch, also cheap).
  3. A triple‑wise product:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]          (1)
     for every batch b, residue pair (i,j) and hidden channel d.
     This is a batched matrix‑multiply per hidden channel:
        out[...,d] = left[...,d] @ right[...,d].transpose(-2,-1)

Step‑3 dominates the runtime (N³·H).  The kernel below computes (1) using
Triton: each program instance works on a TILE (BLOCK_M×BLOCK_N) of the
(i,j) plane and processes BLOCK_D hidden channels at a time.  The reduction
over the sequence dimension `k` is tiled with BLOCK_K and accumulated in
FP32.  All tensors are stored in FP16 (the reference model uses bfloat16 for the
einsum) and the final result is cast back to FP32 before the output
normalisation.

The kernel is launched from `custom_kernel`.  Only the heavy reduction is
off‑loaded to Triton; everything else remains in PyTorch for simplicity and
numeric‑stability.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict

# -----------------------------------------------------------------------
# Triton kernel ---------------------------------------------------------
# -----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,          # [B, N, N, H]  (i,k,d)
    right_ptr,         # [B, N, N, H]  (j,k,d)
    out_ptr,           # [B, N, N, H]  (i,j,d)
    # strides (in elements, not bytes)
    stride_lb, stride_lk, stride_ld,   # left  strides for (b,i,k,d)
    stride_rb, stride_rj, stride_rd,   # right strides for (b,j,k,d)
    stride_ob, stride_oi, stride_od,   # out   strides for (b,i,j,d)
    # sizes
    B, N, H,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""

    pid = tl.program_id(axis=0)                     # 0‑D grid
    # each program works on a (i0,j0) tile and a slice of d
    total_tiles_i = tl.cdiv(N, BLOCK_M)
    total_tiles_j = tl.cdiv(N, BLOCK_N)
    total_tiles_d = tl.cdiv(H, BLOCK_D)

    # decode linear pid
    pid_i = (pid // (total_tiles_j * total_tiles_d)) % total_tiles_i
    pid_j = (pid // total_tiles_d) % total_tiles_j
    pid_d = pid % total_tiles_d

    # starting offsets of this tile
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d_off = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    # mask for out‑of‑bounds (needed for the last partial tile)
    mask_i = i_off < N
    mask_j = j_off < N
    mask_d = d_off < H

    # ----- accumulate over k -------------------------------------------------
    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over k dimension in tiles of BLOCK_K
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(0, num_k_tiles):
        k_off = tk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # ----- load left[b,i,k,d] -------------------------------------------
        # shape (B, BLOCK_M, BLOCK_K, BLOCK_D) but we broadcast over batch
        # via program_id axis 1 (we will launch one program per batch)
        # For simplicity we fold batch into the first dimension of the grid.
        # Here we assume pid runs over (B * total_tiles_i * total_tiles_j * total_tiles_d)
        # and we recover the batch id:
        pid_batch = pid // (total_tiles_i * total_tiles_j * total_tiles_d)
        b = pid_batch

        # compute flat pointers
        left_offset = (
            b * stride_lb +
            i_off[:, None, None] * stride_lk +   # i dimension
            k_off[None, :, None] * stride_ld +   # k dimension
            d_off[None, None, :] * 1
        )
        right_offset = (
            b * stride_rb +
            j_off[:, None, None] * stride_rd +   # j dimension (right uses j as first spatial)
            k_off[None, :, None] * stride_rd +   # k dimension (same stride as d in packed layout)
            d_off[None, None, :] * 1
        )
        # load tiles, cast to fp16
        left_tile = tl.load(left_ptr + left_offset,
                            mask=mask_i[:, None, None] & mask_k[None, :] & mask_d[None, None, :],
                            other=0.0).to(tl.float16)

        right_tile = tl.load(right_ptr + right_offset,
                             mask=mask_j[:, None, None] & mask_k[None, :] & mask_d[None, None, :],
                             other=0.0).to(tl.float16)

        # ----- compute dot over k ---------------------------------------------
        # left_tile: (M, K, D), right_tile: (N, K, D)
        # we want Σ_k left[i,k,d] * right[j,k,d]
        # we can use tl.dot on the K dimension after flattening (M*D, K) and (K, N*D)
        # but it is simpler to loop over K in chunks:
        for kk in range(0, BLOCK_K):
            mask_kk = mask_k[kk]
            if not tl.any(mask_kk):
                continue
            # extract vector (M, D) and (N, D) for this k
            l_vec = left_tile[:, kk, :]   # (M, D)
            r_vec = right_tile[:, kk, :]  # (N, D)
            # compute outer product for each d :
            # acc[i,j,d] += l_vec[i,d] * r_vec[j,d]
            # broadcasting via tl.dot on (M,1) x (1,N) per d is expensive;
            # we use elementwise multiplication with broadcasting:
            # expand dimensions:
            l_exp = l_vec[:, None, :]      # (M,1,D)
            r_exp = r_vec[None, :, :]      # (1,N,D)
            acc += (l_exp * r_exp).to(tl.float32)   # (M,N,D)

    # ----- store result -------------------------------------------------------
    out_offset = (
        b * stride_ob +
        i_off[:, None, None] * stride_oi +   # i
        j_off[None, :, None] * stride_od +   # j
        d_off[None, None, :] * 1
    )
    tl.store(out_ptr + out_offset,
             acc.to(tl.float16),
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])


# -----------------------------------------------------------------------
# Helper ----------------------------------------------------------------
# -----------------------------------------------------------------------
def launch_trimul_kernel(left: torch.Tensor,
                         right: torch.Tensor,
                         out: torch.Tensor,
                         BLOCK_M=64, BLOCK_N=64, BLOCK_D=32, BLOCK_K=64):
    """
    left/right/out : [B,N,N,H]  (float16)
    The kernel is launched with a 1‑D grid where each program handles a
    (BLOCK_M, BLOCK_N, BLOCK_D) tile.
    """
    B, N, _, H = left.shape
    stride_lb, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rd = right.stride()
    stride_ob, stride_oi, stride_od = out.stride()

    total_tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    total_tiles_j = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles_d = (H + BLOCK_D - 1) // BLOCK_D
    grid = (B * total_tiles_i * total_tiles_j * total_tiles_d,)

    trimul_outgoing_kernel[grid](
        left, right, out,
        stride_lb, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rd,
        stride_ob, stride_oi, stride_od,
        B, N, H,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
        BLOCK_D=BLOCK_D, BLOCK_K=BLOCK_K,
        num_warps=4,  # good for fp16 matmul on Hopper
    )


# -----------------------------------------------------------------------
# Main entry point -------------------------------------------------------
# -----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Arguments
    ----------
    data:
        (input, mask, weights, config)
        - input:  [B, N, N, dim]      float32
        - mask :  [B, N, N]           (bool or float)
        - weights: dict of parameters (see problem statement)
        - config : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    out : [B, N, N, dim]  float32
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- 1. LayerNorm (dim) ----------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    input_norm = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.rsqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    input_norm = input_norm * norm_weight + norm_bias   # [B,N,N,dim]

    # ---------- 2. Linear projections (FP16) ----------
    # All linear layers are weight‑only, no bias.
    # We use torch.nn.functional.linear for speed.
    left = torch.nn.functional.linear(
        input_norm, weights["left_proj.weight"]
    )  # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(
        input_norm, weights["right_proj.weight"]
    )
    # mask handling
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- 3. Gates (sigmoid) ----------
    left_gate  = torch.sigmoid(
        torch.nn.functional.linear(input_norm, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(input_norm, weights["right_gate.weight"])
    )
    out_gate   = torch.sigmoid(
        torch.nn.functional.linear(input_norm, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # ---------- 4. Triton reduction (the N³ part) ----------
    # Cast to fp16 for the kernel (the reference uses bfloat16; fp16 is faster on H200)
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out_h   = torch.empty_like(left_h)   # [B,N,N,hidden_dim]

    launch_trimul_kernel(left_h, right_h, out_h,
                         BLOCK_M=64, BLOCK_N=64, BLOCK_D=32, BLOCK_K=64)

    # ---------- 5. Output norm, gating and final linear ----------
    out = out_h.to(torch.float32)                     # back to FP32
    # LayerNorm on hidden_dim
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = (out - out.mean(-1, keepdim=True)) / torch.rsqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_weight + to_out_norm_bias

    out = out * out_gate                               # gating

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
