"""
TriMul “outgoing” implementation.

The heavy part of TriMul is the triple‐wise product

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication for every (batch, hidden_dim) pair.
The kernel below implements this matmul in Triton with a 2‑D block‑grid over
(i,j) and a 1‑D grid over (batch, hidden_dim).  All surrounding
LayerNorm/Linear/Gating ops are performed with PyTorch (they are cheap
relative to the N³ accumulation).  The final tensor is returned in
float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel:  out = left @ rightᵀ   (sum over k)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    left_ptr,               # *const float*
    right_ptr,              # *const float*
    out_ptr,                # *float*
    B, N, D,                # int32 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,  # tile size in i
    BLOCK_N: tl.constexpr,  # tile size in j
    BLOCK_K: tl.constexpr,  # tile size in k
):
    # program ids
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_bd = tl.program_id(2)         # combined batch‑hidden_dim id
    b = pid_bd // D                   # batch index
    d = pid_bd % D                    # hidden‑dim index

    # block start offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # range of rows / cols for this program
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers to the start of the current tile
    left_tile_ptr = left_ptr + \
        b * stride_left_b + d * stride_left_d + \
        offs_i[:, None] * stride_left_i + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    right_tile_ptr = right_ptr + \
        b * stride_right_b + d * stride_right_d + \
        offs_j[:, None] * stride_right_j + tl.arange(0, BLOCK_K)[None, :] * stride_right_k

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load tiles
        left = tl.load(left_tile_ptr, mask=mask_i[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < N), other=0.0)
        right = tl.load(right_tile_ptr, mask=mask_j[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < N), other=0.0)

        # dot product over the reduction axis
        acc += tl.dot(left, right, trans_b=True)   # left: (M,K), right: (N,K) → (M,N)

        # advance pointers
        left_tile_ptr += BLOCK_K * stride_left_k
        right_tile_ptr += BLOCK_K * stride_right_k

    # write back
    out_tile_ptr = out_ptr + \
        b * stride_out_b + d * stride_out_d + \
        offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_tile_ptr, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : (input, mask, weights, config)
        input   – Tensor[B, N, N, dim]  (float32, on CUDA)
        mask    – Tensor[B, N, N]       (float32/bool, on CUDA)
        weights – dict of model parameters (float32 tensors)
        config  – dict containing "dim" and "hidden_dim"

    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) -> [B,N,N,hidden_dim]
    # --------------------------------------------------------------
    # left_proj
    w_left = weights["left_proj.weight"]          # (hidden_dim, dim)
    left = torch.einsum("b i j d, h d -> b i j h", x, w_left)   # (B,N,N,hidden_dim)
    # right_proj
    w_right = weights["right_proj.weight"]
    right = torch.einsum("b i j d, h d -> b i j h", x, w_right)

    # --------------------------------------------------------------
    # 3. Apply mask (if any)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and apply
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # used later

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton kernel: out = Σ_k left[...,k,:] * right[...,k,:]ᵀ
    # --------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # contiguous tensors & strides (in elements, not bytes)
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    stride_left = left.stride()     # (b,i,j,d)
    stride_right = right.stride()
    stride_out = out.stride()

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    _trimul_matmul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=hidden_dim,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_d=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_j=stride_right[1],
        stride_right_k=stride_right[2],
        stride_right_d=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_d=stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6. to_out_norm + gating + final linear
    # --------------------------------------------------------------
    # LayerNorm on hidden dimension
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # Apply output gate (broadcast over i,j)
    out = out * out_gate

    # Final linear projection back to dim
    w_out = weights["to_out.weight"]          # (dim, hidden_dim)
    out = torch.einsum("b i j h, d h -> b i j d", out, w_out)   # (B,N,N,dim)

    return out
