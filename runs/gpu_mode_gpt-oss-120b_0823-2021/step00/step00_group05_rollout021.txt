"""
TriMul “outgoing” implementation.

The heavy N³ work is the einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We keep the cheap LayerNorm / Linear / gating logic in PyTorch and
off‑load the triple‑product to a Triton kernel.  The kernel processes a
tile (BLOCK_I × BLOCK_J) of the (i,j) plane at a time and accumulates
over k in blocks (BLOCK_K).  The channel dimension d is split into
sub‑tiles (BLOCK_H) so that a single accumulation can be kept in registers.
The final result is written back in float32.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------------------- #
# Triton kernel
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # static sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Computes:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for a single batch index ``b``.
    """

    # --------------------------------------------------------------- #
    # program id encodes (batch, tile_i, tile_j)
    # --------------------------------------------------------------- #
    pid = tl.program_id(0)
    tiles_i = tl.cdiv(N, BLOCK_I)
    tiles_j = tl.cdiv(N, BLOCK_J)

    b   = pid // (tiles_i * tiles_j)                     # batch index
    tmp = pid % (tiles_i * tiles_j)
    ti  = tmp // tiles_j                                 # tile i
    tj  = tmp % tiles_j                                   # tile j

    # offsets for the tile
    i_offset = ti * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = tj * BLOCK_J + tl.arange(0, BLOCK_J)

    # mask for out‑of‑bounds rows / cols
    i_mask = i_offset < N
    j_mask = j_offset < N

    # accumulator: (BLOCK_I, BLOCK_J, BLOCK_H)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # -----------------------------------------------------------------
    # loop over k in blocks
    # -----------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        k_offset = k_start + tl.arange(0, BLOCK_K)
        k_mask   = k_offset < N

        # -------------------------------------------------------------
        # left[b,i,k,h]   shape (BLOCK_I, BLOCK_K, BLOCK_H)
        # right[b,j,k,h]  shape (BLOCK_J, BLOCK_K, BLOCK_H)
        # -------------------------------------------------------------
        left = tl.load(
            left_ptr
            + b * stride_left_b
            + i_offset[:, None, None] * stride_left_i
            + k_offset[None, :, None] * stride_left_k
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_left_h,
            mask=(i_mask[:, None, None] & k_mask[None, :, None] &
                  (tl.arange(0, BLOCK_H)[None, None, :] < H)),
            other=0.0,
        )  # (BI, BK, BH)

        right = tl.load(
            right_ptr
            + b * stride_right_b
            + j_offset[:, None, None] * stride_right_j
            + k_offset[None, :, None] * stride_right_k
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_right_h,
            mask=(j_mask[:, None, None] & k_mask[None, :, None] &
                  (tl.arange(0, BLOCK_H)[None, None, :] < H)),
            other=0.0,
        )  # (BJ, BK, BH)

        # element‑wise product, then sum over k
        prod = left * right  # (BI, BK, BH) broadcast with (BJ, BK, BH) → (BI, BJ, BK, BH)
        # broadcasting trick: expand left over j and right over i
        prod = tl.reshape(prod, (BLOCK_I, 1, BLOCK_K, BLOCK_H)) * \
               tl.reshape(right, (1, BLOCK_J, BLOCK_K, BLOCK_H))
        # sum over k dimension
        acc += tl.sum(prod, axis=2)  # (BI, BJ, BH)

    # -----------------------------------------------------------------
    # write result
    # -----------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i_offset[:, None, None] * stride_out_i
        + j_offset[None, :, None] * stride_out_j
        + tl.arange(0, BLOCK_H)[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=i_mask[:, None, None] & j_mask[None, :, None] &
             (tl.arange(0, BLOCK_H)[None, None, :] < H),
    )


# --------------------------------------------------------------------------- #
# Host‑side helper that launches the kernel for each batch
# --------------------------------------------------------------------------- #
def launch_trimul_kernel(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]  (float32)
    returns out   : [B, N, N, H] (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 32

    # strides (in elements, not bytes)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    grid = (B * triton.cdiv(N, BLOCK_I) * triton.cdiv(N, BLOCK_J),)

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_h=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_j=stride_right[1],
        stride_right_k=stride_right[2],
        stride_right_h=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_h=stride_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    return out


# --------------------------------------------------------------------------- #
# Full custom kernel – mixes PyTorch (layernorm / linear / gating) with Triton
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    data = (input_tensor, mask, weights, config)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]

    # --------------------------------------------------------------- #
    # 1. LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------- #
    x = input_tensor
    # LayerNorm: (x - mean) / sqrt(var + eps) * w + b
    eps = 1e-5
    w = weights["norm.weight"].to(x.dtype)
    b = weights["norm.bias"].to(x.dtype)

    # compute mean / var over the last dim (dim)
    x_flat = x.view(-1, dim)
    mean = x_flat.mean(-1, keepdim=True)
    var = x_flat.var(-1, unbiased=False, keepdim=True)
    x = (x - mean.view_as(x)) * torch.rsqrt(var + eps) * w + b

    # --------------------------------------------------------------- #
    # 2. Linear projections (no bias) and gates
    # --------------------------------------------------------------- #
    left = torch.nn.functional.linear(x, weights["left_proj.weight"].to(x.dtype))
    right = torch.nn.functional.linear(x, weights["right_proj.weight"].to(x.dtype))

    # mask handling
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)            # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # gates (sigmoid)
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"].to(x.dtype)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"].to(x.dtype)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"].to(x.dtype)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # 3. Core N³ multiplication – Triton
    # --------------------------------------------------------------- #
    out = launch_trimul_kernel(left, right)          # [B,N,N,H]

    # --------------------------------------------------------------- #
    # 4. Output norm + gating + final linear
    # --------------------------------------------------------------- #
    # LayerNorm on hidden dimension
    w_out = weights["to_out_norm.weight"].to(out.dtype)
    b_out = weights["to_out_norm.bias"].to(out.dtype)
    out_flat = out.view(-1, hidden_dim)
    mean = out_flat.mean(-1, keepdim=True)
    var = out_flat.var(-1, unbiased=False, keepdim=True)
    out = (out - mean.view_as(out)) * torch.rsqrt(var + eps) * w_out + b_out

    out = out * out_gate

    # final linear back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(out.dtype))

    return out.to(torch.float32)
