"""
TriMul “outgoing” kernel (AlphaFold‑3)

The only costly operation is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For every hidden channel *d* this is a batched matrix‑multiply
    (left[b,:, :,d])  @  (right[b,:, :,d]).T
with shape (N,N)×(N,N) → (N,N).

The kernel below performs a tiled mat‑mul for a **block of hidden
channels** (C_BLOCK) at a time, using fp16 inputs and fp32 accumulation.
All LayerNorms, Linear projections, and gating are kept in PyTorch (they
are cheap compared with the N³ work).  Only the einsum is off‑loaded to
Triton, giving a large speed‑up on Hopper GPUs.

The host‑side `custom_kernel` prepares the tensors, calls the Triton
kernel and finishes the remaining PyTorch post‑processing.
"""
import torch
import torch.nn as nn
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:  tiled batched mat‑mul for a block of hidden channels
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,          # [B, N, N, C]  (i,k,c)
    right_ptr,         # [B, N, N, C]  (j,k,c)
    out_ptr,           # [B, N, N, C]  (i,j,c)

    # shape
    B, N, C,

    # strides (in elements)
    stride_bli, stride_blk, stride_blc,   # left  (batch, i, k, c)
    stride_brj, stride_brk, stride_brc,   # right (batch, j, k, c)
    stride_oui, stride_ouj, stride_ouc,   # out   (batch, i, j, c)

    # tile sizes
    BLOCK_M: tl.constexpr,   # i dimension
    BLOCK_N: tl.constexpr,   # j dimension
    BLOCK_K: tl.constexpr,   # k dimension
    C_BLOCK:  tl.constexpr,  # hidden‑channel block

    # other constants
    FP16: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # decode pid -> (batch, c_block, block_i, block_j)
    # --------------------------------------------------------------
    total_c_blocks = (C + C_BLOCK - 1) // C_BLOCK
    total_i_blocks = tl.cdiv(N, BLOCK_M)
    total_j_blocks = tl.cdiv(N, BLOCK_N)

    pid_b   = pid // (total_c_blocks * total_i_blocks * total_j_blocks)
    pid_rem = pid % (total_c_blocks * total_i_blocks * total_j_blocks)

    pid_c   = pid_rem // (total_i_blocks * total_j_blocks)
    pid_rem = pid_rem % (total_i_blocks * total_j_blocks)

    pid_i   = pid_rem // total_j_blocks
    pid_j   = pid_rem % total_j_blocks

    # --------------------------------------------------------------
    # compute start offsets
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # hidden channel offsets within this block
    offs_c = pid_c * C_BLOCK + tl.arange(0, C_BLOCK)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_c = offs_c < C

    # pointers to the first element of the tile
    left_tile_ptr  = left_ptr \
        + pid_b * stride_bli * N * N \
        + offs_i[:, None, None] * stride_bli \
        + offs_k[None, :, None] * stride_blk \
        + offs_c[None, None, :] * stride_blc

    right_tile_ptr = right_ptr \
        + pid_b * stride_brj * N * N \
        + offs_j[:, None, None] * stride_brj \
        + offs_k[None, :, None] * stride_brk \
        + offs_c[None, None, :] * stride_brc

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N, C_BLOCK), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # actual K offsets for this iteration
        cur_k = k * BLOCK_K + offs_k
        k_mask = cur_k < N

        # load tiles
        left = tl.load(
            left_tile_ptr + cur_k[None, :, None] * stride_blk,
            mask=mask_i[:, None, None] & k_mask[None, :, None] & mask_c[None, None, :],
            other=0.0
        )
        right = tl.load(
            right_tile_ptr + cur_k[None, :, None] * stride_brk,
            mask=mask_j[:, None, None] & k_mask[None, :, None] & mask_c[None, None, :],
            other=0.0
        )
        # left : (M, K, Cb)   right : (N, K, Cb)
        # need (M,K,C) * (N,K,C) -> sum over K, keep C
        # we perform dot per channel
        for cb in range(C_BLOCK):
            acc[:, :, cb] += tl.dot(left[:, :, cb].to(tl.float32),
                                   right[:, :, cb].to(tl.float32).T,
                                   allow_tf32=False)

    # --------------------------------------------------------------
    # write result
    # --------------------------------------------------------------
    out_tile_ptr = out_ptr \
        + pid_b * stride_oui * N * N \
        + offs_i[:, None, None] * stride_oui \
        + offs_j[None, :, None] * stride_ouj \
        + offs_c[None, None, :] * stride_ouc

    tl.store(
        out_tile_ptr,
        acc.to(tl.float16),
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_c[None, None, :]
    )


# ----------------------------------------------------------------------
# Host helper – launches the Triton kernel
# ----------------------------------------------------------------------
def triton_einsum_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left , right : [B, N, N, C] (dtype=float16)
    returns out   : [B, N, N, C] (dtype=float16)
    """
    B, N, _, C = left.shape
    out = torch.empty_like(left)

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    C_BLOCK = 32        # hidden dimension is multiple of 32 in all configs

    # strides (in elements)
    stride_bli, stride_blk, stride_blc = left.stride()
    stride_brj, stride_brk, stride_brc = right.stride()
    stride_oui, stride_ouj, stride_ouc = out.stride()

    total_programs = B * ((C + C_BLOCK - 1) // C_BLOCK) * \
                     ((N + BLOCK_M - 1) // BLOCK_M) * ((N + BLOCK_N - 1) // BLOCK_N)

    grid = (total_programs,)

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, C,
        stride_bli, stride_blk, stride_blc,
        stride_brj, stride_brk, stride_brc,
        stride_oui, stride_ouj, stride_ouc,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        C_BLOCK=C_BLOCK,
        FP16=tl.constexpr,   # just a dummy constexpr to keep the signature stable
    )
    return out


# ----------------------------------------------------------------------
# Full custom_kernel – mixes PyTorch + Triton
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor with shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1) LayerNorm (dim)
    # --------------------------------------------------------------
    eps = 1e-5
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b   # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    # shape: (...,dim) @ (dim,hidden) -> (...,hidden)
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Apply mask (if present)
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4) Gating (sigmoid) – also no bias
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Core einsum – move to Triton (use fp16 for speed)
    # --------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp16 = triton_einsum_outgoing(left_fp16, right_fp16)   # [B,N,N,hidden] fp16

    out = out_fp16.to(torch.float32)

    # --------------------------------------------------------------
    # 6) Post‑processing (LayerNorm + gating + final linear)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    # layer norm on hidden dim
    mean_h = out.mean(-1, keepdim=True)
    var_h = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_h) * torch.rsqrt(var_h + eps)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate

    # final linear back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
