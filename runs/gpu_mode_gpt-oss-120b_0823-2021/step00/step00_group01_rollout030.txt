"""
TriMul “outgoing” kernel (AlphaFold3)

The heavy part of TriMul is the three‑tensor einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where *left* and *right* are the projected and gated pairwise
representations of shape (B,N,N, D_hidden).  For every hidden dimension *d*
the operation is a matrix multiplication of the (i,k) slice with the
transpose of the (j,k) slice.  The kernel below computes this reduction
in a fused Triton kernel, performing a blocked GEMM for each *d* while
keeping the reduction over *k* on‑device.  All surrounding LayerNorm,
Linear and gating work is done in PyTorch (fast enough) and the final
output is cast back to float32.

The kernel is written for Hopper (SM‑90) but works on any recent
CUDA device.  Block sizes were chosen to give high occupancy while
respecting H200’s large shared memory.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: blocked reduction over k, vectorised over hidden dim D
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # * left  [B,N,N,D]
    right_ptr,         # * right [B,N,N,D]
    out_ptr,           # * out   [B,N,N,D]
    B, N, D,
    stride_lbz, stride_lby, stride_lbk, stride_lbd,   # left strides
    stride_rbz, stride_rby, stride_rbk, stride_rbd,   # right strides
    stride_obz, stride_oby, stride_obj, stride_obd,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""

    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # i‑block
    pid_j = tl.program_id(2)          # j‑block

    # ------------------------------------------------------------------
    #  thread‑local offsets
    # ------------------------------------------------------------------
    i_range = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_range = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d_range = tl.arange(0, BLOCK_D)

    mask_i = i_range < N
    mask_j = j_range < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  loop over k dimension in tiles
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_range = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # ------------------------------------------------------------------
        #  Load left[b,i,k,d]  and  right[b,j,k,d]
        # ------------------------------------------------------------------
        # left pointer: (b,i,k,d)
        left_ptrs = (
            left_ptr
            + pid_b * stride_lbz
            + i_range[:, None, None] * stride_lby
            + k_range[None, :, None] * stride_lbk
            + d_range[None, None, :] * stride_lbd
        )
        # right pointer: (b,j,k,d)
        right_ptrs = (
            right_ptr
            + pid_b * stride_rbz
            + j_range[:, None, None] * stride_rby
            + k_range[None, :, None] * stride_rbk
            + d_range[None, None, :] * stride_rbd
        )

        left = tl.load(
            left_ptrs,
            mask=(mask_i[:, None, None] & mask_k[None, :, None]),
            other=0.0,
        )
        right = tl.load(
            right_ptrs,
            mask=(mask_j[:, None, None] & mask_k[None, :, None]),
            other=0.0,
        )
        # element‑wise product, sum over k (axis=1)
        acc += tl.sum(left * right, axis=1)

    # ------------------------------------------------------------------
    #  Store result
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_obz
        + i_range[:, None, None] * stride_oby
        + j_range[None, :, None] * stride_obj
        + d_range[None, None, :] * stride_obd
    )
    mask_store = mask_i[:, None, None] & mask_j[None, :, None]
    tl.store(out_ptrs, acc, mask=mask_store)


# ----------------------------------------------------------------------
# Helper: apply LayerNorm without creating a module (faster)
# ----------------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    """x: [..., C]  weight/bias: (C,)"""
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    norm = (x - mean) * torch.rsqrt(var + eps)
    return norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    data = (input, mask, weights, config)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on input (dim)
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"].to(device),
        weights["norm.bias"].to(device),
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # use torch.nn.functional.linear for speed
    left = F.linear(x, weights["left_proj.weight"].to(device))      # (B,N,N,hd)
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(x)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device)))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel: compute Σ_k left[...,i,k,d] * right[...,j,k,d]
    # ------------------------------------------------------------------
    # pad hidden dimension to a multiple of BLOCK_D for the kernel
    BLOCK_M, BLOCK_N, BLOCK_K, BLOCK_D = 32, 32, 32, 16
    pad_d = (BLOCK_D - hidden_dim % BLOCK_D) % BLOCK_D
    if pad_d:
        left = F.pad(left, (0, pad_d))
        right = F.pad(right, (0, pad_d))
    D_eff = hidden_dim + pad_d

    out = torch.empty((bs, N, N, D_eff), dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    l_strides = left.stride()
    r_strides = right.stride()
    o_strides = out.stride()

    grid = (
        bs,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=N,
        D=D_eff,
        stride_lbz=l_strides[0],
        stride_lby=l_strides[1],
        stride_lbk=l_strides[2],
        stride_lbd=l_strides[3],
        stride_rbz=r_strides[0],
        stride_rby=r_strides[1],
        stride_rbk=r_strides[2],
        stride_rbd=r_strides[3],
        stride_obz=o_strides[0],
        stride_oby=o_strides[1],
        stride_obj=o_strides[2],
        stride_obd=o_strides[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    # truncate possible padding
    out = out[..., :hidden_dim]

    # ------------------------------------------------------------------
    # 6. to_out_norm (hidden_dim) and final linear
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"].to(device),
        weights["to_out_norm.bias"].to(device),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(device))

    return out
