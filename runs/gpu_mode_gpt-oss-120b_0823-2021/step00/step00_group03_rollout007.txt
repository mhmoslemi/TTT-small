"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part is the three‑dimensional contraction

    out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]          (1)

where left/right ∈ ℝ^{B×N×N×H}.  The surrounding LayerNorm,
Linear and gating operations are cheap and stay in PyTorch.
The contraction (1) is implemented in Triton with a tiled
i‑j block and a reduction over k.  Mixed‑precision (FP16) is used
inside the kernel and the result is accumulated in FP32, matching
the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the contraction (1)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # problem size
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Grid: one program per (b,i,j) tile.
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # decode pid → (b, i_tile, j_tile)
    # ------------------------------------------------------------------
    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i_tile = ij // N
    j_tile = ij % N

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_i = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over k
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(0, num_k_tiles):
        k_start = tk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # load left[b, i, k, :]   shape (BLOCK_I, BLOCK_K, H)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + offs_i[:, None, None] * stride_li
            + offs_k[None, :, None] * stride_lk
            + tl.arange(0, H)[None, None, :] * stride_lh
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        ).to(tl.float32)   # cast to fp32 for accumulation

        # ------------------------------------------------------------------
        # load right[b, j, k, :]  shape (BLOCK_J, BLOCK_K, H)
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + offs_j[:, None, None] * stride_rj
            + offs_k[None, :, None] * stride_rk
            + tl.arange(0, H)[None, None, :] * stride_rh
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None],
            other=0.0,
        ).to(tl.float32)

        # ------------------------------------------------------------------
        # multiply and reduce over k
        #   left  : (I, K, H)
        #   right : (J, K, H)
        #   we broadcast to (I, J, K, H) and sum on K
        # ------------------------------------------------------------------
        prod = left[:, None, :, :] * right[None, :, :, :]          # (I,J,K,H)
        acc += tl.sum(prod, axis=2)                               # (I,J,H)

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + offs_i[:, None, None] * stride_oi
        + offs_j[None, :, None] * stride_oj
        + tl.arange(0, H)[None, None, :] * stride_oh
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None],
    )


# ----------------------------------------------------------------------
# Host wrapper (the required `custom_kernel` entry point)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Arguments
    ---------
    data : Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
        - input   : [B, N, N, D]   (float32)
        - mask    : [B, N, N]      (bool / float, 1.0 where kept)
        - weights : dict with the same keys as in the reference module
        - config  : {"dim": D, "hidden_dim": H, "nomask": bool}
    Returns
    -------
    torch.Tensor : [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) + optional mask
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    if not nomask:
        # mask shape [B,N,N] → [B,N,N,1] for broadcasting
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Cast to fp16 for the heavy contraction (helps bandwidth)
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # Prepare output buffer (fp16, will be cast back later)
    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=left.device)

    # ------------------------------------------------------------------
    # 5. Triton launch
    # ------------------------------------------------------------------
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    grid = (B * N * N,)   # one program per (b,i_tile,j_tile) block

    # compute strides for each tensor (they are contiguous in memory)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out_fp16.stride()

    tri_mul_kernel[grid](
        left,
        right,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6. Restore fp32, LayerNorm, gating and final linear
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
