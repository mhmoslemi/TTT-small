"""
TriMul “outgoing” implementation.

The heavy part of the TriMul forward pass is the
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix‑multiply for every hidden channel *d*.
We implement this reduction in Triton (one kernel) and keep the
remaining LayerNorm / gating / linear projections in PyTorch.

All tensors are expected to be on the same CUDA device.
The final output is returned in float32.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel : batched matmul per hidden channel
# ----------------------------------------------------------------------
@triton.jit
def triton_trimul_kernel(
    left_ptr,                     # *B *N *N *H*   (float16)
    right_ptr,                    # *B *N *N *H*   (float16)
    out_ptr,                      # *B *N *N *H*   (float32)
    B, N, H,
    stride_lbn, stride_lbk, stride_lbh,   # left strides (b,n,k,h)
    stride_rbn, stride_rbk, stride_rbh,   # right strides
    stride_obn, stride_obk, stride_obh,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] """
    pid = tl.program_id(axis=0)               # linear index for (b,h) pair
    num_bh = B * H
    bh = pid                                   # 0 … B*H-1
    b = bh // H
    h = bh % H

    # tile coordinates in the i‑j plane
    pid_m = tl.program_id(axis=1)              # i‑tile
    pid_n = tl.program_id(axis=2)              # j‑tile

    # start offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds i/j
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the first element of the tile (i,j) for this batch & channel
    left_tile_ptr  = left_ptr \
        + b * stride_lbn \
        + h * stride_lbh \
        + offs_m[:, None] * stride_lbk   # (i,k)
    right_tile_ptr = right_ptr \
        + b * stride_rbn \
        + h * stride_rbh \
        + offs_n[None, :] * stride_rbk   # (j,k)

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k‑range for this iteration
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left[i,k]  -> (BLOCK_M, BLOCK_K)
        l = tl.load(left_tile_ptr + offs_k[None, :] * stride_lbh,
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right[j,k] -> (BLOCK_N, BLOCK_K)
        r = tl.load(right_tile_ptr + offs_k[:, None] * stride_rbh,
                    mask=mask_n[None, :] & mask_k[:, None], other=0.0).to(tl.float32)

        # element‑wise multiply and sum over k
        # we need Σ_k l[i,k] * r[j,k]  (note r is indexed by j,k)
        # we can use tl.dot on the k‑dimension: dot(l[i,:], r[j,:])
        acc += tl.dot(l, r.T)   # (BLOCK_M, BLOCK_N)

        # advance pointers for next k‑block
        left_tile_ptr  += BLOCK_K * stride_lbh
        right_tile_ptr += BLOCK_K * stride_rbh

    # store result (float32)
    out_tile_ptr = out_ptr \
        + b * stride_obn \
        + h * stride_obh \
        + offs_m[:, None] * stride_obk \
        + offs_n[None, :] * stride_obk

    tl.store(out_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: LayerNorm without bias (we use PyTorch implementation)
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: [..., C]  float32
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    std = torch.rsqrt(var + eps)
    return (x - mean) * std * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Only the O(N³·H) reduction is performed in Triton; everything else stays in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device)
    )
    # ensure contiguous float32 for the linear layers
    x = x.to(torch.float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gating
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"].to(device), bias=None)          # [B,N,N, H]
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"].to(device), bias=None)

    # mask handling (optional)
    if mask is not None:
        mask = mask.unsqueeze(-1)            # [B,N,N,1]
        left = left * mask
        right = right * mask

    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"].to(device), bias=None).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"].to(device), bias=None).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"].to(device), bias=None).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Triton kernel: compute Σ_k left[...,i,k,d] * right[...,j,k,d]
    # ------------------------------------------------------------------
    # Cast to bfloat16 for better throughput, accumulation is float32
    left_t = left.to(torch.bfloat16)
    right_t = right.to(torch.bfloat16)

    out = torch.empty((bs, seq_len, seq_len, hidden_dim), dtype=torch.float32,
                      device=device)

    # stride extraction (torch tensors are contiguous in the last dim)
    stride_lbn, stride_lbk, stride_lbh = left_t.stride()
    stride_rbn, stride_rbk, stride_rbh = right_t.stride()
    stride_obn, stride_obk, stride_obh = out.stride()

    # launch configuration
    BLOCK_M = 64   # tiles in i dimension
    BLOCK_N = 64   # tiles in j dimension
    BLOCK_K = 32   # reduction tile size

    grid = (bs * hidden_dim,
            triton.cdiv(seq_len, BLOCK_M),
            triton.cdiv(seq_len, BLOCK_N))

    triton_trimul_kernel[grid](
        left_t,
        right_t,
        out,
        bs, seq_len, hidden_dim,
        stride_lbn, stride_lbk, stride_lbh,
        stride_rbn, stride_rbk, stride_rbh,
        stride_obn, stride_obk, stride_obh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                     # H200 has many warps, 4 is a good default
    )

    # ------------------------------------------------------------------
    # 4. Output LayerNorm, gating and final linear
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device)
    )
    out = out * out_gate
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(device), bias=None)

    return out.to(torch.float32)
