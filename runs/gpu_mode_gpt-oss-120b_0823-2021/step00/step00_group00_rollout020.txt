"""
TriMul “outgoing” kernel (AlphaFold‑3)

Algorithm
---------
1.  Layer‑norm + the four linear projections + the three sigmoids are
    performed in PyTorch (they are cheap compared to the O(N³) step).
2.  The core operation is

        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]          (d = hidden_dim)

    which is a reduction over the sequence dimension *k* of an
    element‑wise product.  We implement this reduction with a fused
    Triton kernel that processes tiles of *(i,j)* and streams the *k*
    dimension block‑wise, accumulating the result in FP32.
3.  The kernel produces a tensor of shape (B, N, N, hidden_dim) in
    float32.  It is then passed through the final LayerNorm, gating
    and output‑projection (again in PyTorch) to obtain the final
    (B, N, N, dim) result.

The implementation works for any batch size ≤2, any sequence length
up to 1024 and hidden dimensions that are multiples of 32 (the hidden
dimension in the tests is 128).  Mixed‑precision (FP16 storage) is
used inside the kernel to reduce memory traffic while the accumulation
remains in FP32 for numerical stability.
"""

import torch
import triton
import triton.language as tl
from torch import nn
from typing import Tuple, Dict, Any

# ---------------------------------------------------------------------------
# Triton kernel: reduction over k of element‑wise product
# ---------------------------------------------------------------------------
@triton.jit
def trimul_out_kernel(
    left_ptr,          # const float*  [B, N, N, H]   (i,k,d)
    right_ptr,         # const float*  [B, N, N, H]   (j,k,d)
    out_ptr,           # float*        [B, N, N, H]   (i,j,d)

    B, N, H,
    stride_lbn, stride_lbi, stride_lbk, stride_lbh,   # left strides
    stride_rbn, stride_rbj, stride_rbk, stride_rbh,   # right strides
    stride_obn, stride_obi, stride_obj, stride_obh,   # out  strides

    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    """Compute out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:].

    Tile sizes (BLOCK_*) are compile‑time constants.
    """
    pid = tl.program_id(0)          # linear index over (i_tile, j_tile)
    pid_b = tl.program_id(1)        # batch dimension

    # Number of tiles in each axis
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)

    i_tile = pid % num_i
    j_tile = pid // num_i

    # Starting coordinates of the tile
    i_offset = i_tile * BLOCK_I
    j_offset = j_tile * BLOCK_J

    # Offsets (1‑D) inside the tile
    offs_i = i_offset + tl.arange(0, BLOCK_I)
    offs_j = j_offset + tl.arange(0, BLOCK_J)

    # Masks for out‑of‑bounds rows / columns
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Hidden‑dim offsets (stored as FP16 to save bandwidth)
    offs_h = tl.arange(0, BLOCK_H)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # -----------------------------------------------------------------------
    # Loop over k dimension in blocks
    # -----------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_offset = k_iter * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---------- left[b,i,k,h] ----------
        # shape (BLOCK_I, BLOCK_K, BLOCK_H)
        left_ptrs = (
            left_ptr
            + pid_b * stride_lbn
            + offs_i[:, None, None] * stride_lbi
            + offs_k[None, :, None] * stride_lbk
            + offs_h[None, None, :] * stride_lbh
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
            dtype=tl.float16,
        )
        left = left.to(tl.float32)

        # ---------- right[b,j,k,h] ----------
        # shape (BLOCK_J, BLOCK_K, BLOCK_H)
        right_ptrs = (
            right_ptr
            + pid_b * stride_rbn
            + offs_j[:, None, None] * stride_rbj
            + offs_k[None, :, None] * stride_rbk
            + offs_h[None, None, :] * stride_rbh
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None],
            other=0.0,
            dtype=tl.float16,
        )
        right = right.to(tl.float32)

        # ---------- multiply + reduction over k ----------
        # left  : (I, K, H)
        # right : (J, K, H) -> broadcast to (I, J, K, H)
        # product : (I, J, K, H)   -> sum over K
        prod = left[:, None, :, :] * right[None, :, :, :]   # (I, J, K, H)
        acc += tl.sum(prod, axis=2)                         # (I, J, H)

    # -----------------------------------------------------------------------
    # Write result
    # -----------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_obn
        + offs_i[:, None, None] * stride_obi
        + offs_j[None, :, None] * stride_obj
        + offs_h[None, None, :] * stride_obh
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None],
    )


# ---------------------------------------------------------------------------
# Helper: LayerNorm in Triton (optional – we keep the PyTorch version)
# ---------------------------------------------------------------------------
def layer_norm_triton(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """Simple per‑feature LayerNorm implemented with torch for clarity."""
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_hat = (x - mean) / torch.sqrt(var + eps)
    return x_hat * weight + bias


# ---------------------------------------------------------------------------
# Entry point
# ---------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : (input, mask, weights, config)
        input   – Tensor[B, N, N, dim] (float32, on CUDA)
        mask    – Tensor[B, N, N]      (float32/bool, on CUDA)
        weights – dict of model parameters (already on CUDA)
        config  – dict with keys:
                  * dim          – input dimensionality
                  * hidden_dim   – hidden dimensionality (H)
                  * eps          – LayerNorm epsilon (optional)

    Returns
    -------
    Tensor[B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # -----------------------------------------------------------------------
    # 1.  LayerNorm + linear projections (PyTorch)
    # -----------------------------------------------------------------------
    # norm
    x = layer_norm_triton(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps,
    )
    # linear projections (bias‑free)
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------------
    # 2.  Masking
    # -----------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # -----------------------------------------------------------------------
    # 3.  Gating (sigmoid)
    # -----------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 4.  Core O(N³) reduction – Triton kernel
    # -----------------------------------------------------------------------
    # Cast to FP16 for bandwidth‑limited kernel, keep a FP32 copy for the
    # final accumulation (the kernel does the promotion internally).
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    stride_lbn, stride_lbi, stride_lbk, stride_lbh = left_fp16.stride()
    stride_rbn, stride_rbj, stride_rbk, stride_rbh = right_fp16.stride()
    stride_obn, stride_obi, stride_obj, stride_obh = out.stride()

    # Tile sizes – chosen to fit into shared memory on Hopper
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_H = 32                     # hidden_dim is a multiple of 32 (128)

    grid_i = triton.cdiv(N, BLOCK_I)
    grid_j = triton.cdiv(N, BLOCK_J)
    grid = (grid_i * grid_j, B)      # (i*j tiles, batch)

    trimul_out_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B, N, hidden_dim,
        stride_lbn, stride_lbi, stride_lbk, stride_lbh,
        stride_rbn, stride_rbj, stride_rbk, stride_rbh,
        stride_obn, stride_obi, stride_obj, stride_obh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # -----------------------------------------------------------------------
    # 5.  Final LayerNorm, gating and projection (PyTorch)
    # -----------------------------------------------------------------------
    out = layer_norm_triton(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"], eps)
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.float()
