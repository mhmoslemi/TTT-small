"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batch of N×N matrix products for each hidden channel d
(left @ rightᵀ).  A single Triton kernel computes this 3‑D product
for a block of rows (i) × columns (j) × channels (d) while
tiling over the reduction dimension k.  All other operations
(LayerNorm, Linear projections, sigmoid gates, final LayerNorm
and output Linear) are inexpensive and are performed with
regular PyTorch primitives using the supplied weight tensors.

The kernel works in half‑precision (fp16) for speed and converts the
result back to fp32 before the final output Linear layer.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel : batched matmul for each hidden channel (d)
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,            # [B, N, N, H]  fp16
    right_ptr,           # [B, N, N, H]  fp16
    out_ptr,             # [B, N, N, H]  fp16 (temporary)
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)                      # 1‑D grid over (b, i, j, h) blocks
    # decode grid indices
    block_h = pid % tl.cdiv(H, BLOCK_H)
    pid //= tl.cdiv(H, BLOCK_H)
    block_j = pid % tl.cdiv(N, BLOCK_N)
    pid //= tl.cdiv(N, BLOCK_N)
    block_i = pid % tl.cdiv(N, BLOCK_M)
    b       = pid // tl.cdiv(N, BLOCK_M)        # batch index

    # pointers to the start of the block
    i_off = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    h_off = block_h * BLOCK_H + tl.arange(0, BLOCK_H)

    # mask for out‑of‑bounds (needed for the last partial blocks)
    i_mask = i_off < N
    j_mask = j_off < N
    h_mask = h_off < H

    # accumulators in fp32 for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # ---- load left :  (i,k,h)  ----
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        left_ptrs = left_ptr + \
            b * stride_left_b + \
            i_off[:, None, None] * stride_left_i + \
            k_off[None, :, None] * stride_left_k + \
            h_off[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=(i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :]), other=0.0)

        # ---- load right (transposed) : (j,k,h) -> we need right[j,k,h] but the dot wants (k,j)
        # loading as (k,j,h) and later dot will treat k as reduction dim
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            j_off[:, None, None] * stride_right_j + \
            k_off[None, :, None] * stride_right_k + \
            h_off[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=(j_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :]), other=0.0)

        # dot over k : (i,k)·(j,k)ᵀ  => (i,j) for each h
        # tl.dot works on 2‑D matrices, so we loop over h inside the block
        for hh in range(0, BLOCK_H):
            # shape (BLOCK_M, BLOCK_K) * (BLOCK_N, BLOCK_K)ᵀ
            acc[:, :, hh] += tl.dot(left[:, :, hh].to(tl.float32),
                                   right[:, :, hh].to(tl.float32), trans_b=True)
    # store result
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        i_off[:, None, None] * stride_out_i + \
        j_off[None, :, None] * stride_out_j + \
        h_off[None, None, :] * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=(i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :]))

# --------------------------------------------------------------
# Host helper that launches the kernel
# --------------------------------------------------------------
def trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right :  [B, N, N, H]  (fp16)
    returns      :  [B, N, N, H]  (fp16)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in number of elements (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – tuned for Hopper SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 32   # hidden dim is multiple of 32 (128,384)

    grid = (
        triton.cdiv(B, 1) *
        triton.cdiv(N, BLOCK_M) *
        triton.cdiv(N, BLOCK_N) *
        triton.cdiv(H, BLOCK_H),
    )
    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
    )
    return out

# --------------------------------------------------------------
# Main entry point required by the grader
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : (input, mask, weights, config)
        input   – [B, N, N, dim]   float32 tensor
        mask    – [B, N, N]        (may be all‑ones)
        weights – dict of weight tensors (see problem statement)
        config  – dict with keys "dim" and "hidden_dim"

    Returns
    -------
    output : [B, N, N, dim] float32
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    # manual LayerNorm to avoid extra memory copies
    eps = 1e-5
    x = input_tensor
    x_flat = x.view(-1, dim)
    mean = x_flat.mean(dim=-1, keepdim=True)
    var = x_flat.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x_flat - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]
    x_norm = x_norm.view_as(x)                # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)  ->  left/right  [B,N,N,hidden]
    # ------------------------------------------------------------------
    # weight shape: [hidden, dim]
    x_proj = x_norm.to(torch.float16)          # fp16 for the heavy path
    left = F.linear(x_proj, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_proj, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3️⃣  Apply mask (if present) – mask is broadcasted on hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(torch.float16)    # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣  Gating (sigmoid) – all in fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_proj, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x_proj, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x_proj, weights["out_gate.weight"].to(torch.float16))
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Core 3‑D multiplication via Triton
    # ------------------------------------------------------------------
    # left/right are [B,N,N,H] fp16
    out = trimul_outgoing_triton(left, right)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣  Final LayerNorm (hidden) + gating + output Linear
    # ------------------------------------------------------------------
    out = out.to(torch.float32)                 # back to fp32 for LayerNorm
    B, N, _, H = out.shape
    out_flat = out.view(-1, H)
    mean = out_flat.mean(dim=-1, keepdim=True)
    var = out_flat.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_flat - mean) / torch.sqrt(var + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]
    out_norm = out_norm.view(B, N, N, H)

    out_norm = out_norm * out_gate.to(torch.float32)

    # final Linear (hidden -> dim), no bias
    out_final = F.linear(out_norm, weights["to_out.weight"])

    return out_final
