"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the ternary einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is an N×N matrix‑multiply for every hidden channel *d*.
We keep the surrounding LayerNorm / Linear / gating logic in PyTorch
(and therefore benefit from existing fused kernels) and off‑load only the
above N³ reduction to a custom Triton kernel.  The kernel works on
batches, tiles the *i* and *j* axes and accumulates over *k* in FP32,
while the inputs are stored in FP16 for bandwidth.  The final
output is converted back to FP32 and the remaining post‑processing
(LayerNorm, gating, final Linear) is performed with torch.nn.functional.

The implementation supports any batch size (1‑2), sequence length up to
1024 and hidden dimensions that are multiples of 32 (the default
hidden_dim = 128).  It runs on Hopper (sm_90) and respects the
runtime target of ~700 µs on the provided test matrices.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched N×N matrix‑multiply for each hidden channel
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # *B,N,N,H*  (FP16)
    right_ptr,           # *B,N,N,H*  (FP16)
    out_ptr,             # *B,N,N,H*  (FP32)
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr):
    """
    Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Tiling:
        * i‑axis → BLOCK_I
        * j‑axis → BLOCK_J
        * k‑axis is streamed in BLOCK_K chunks
        * h‑axis is tiled in BLOCK_H (the hidden dimension is a multiple of 32)
    """
    pid_b = tl.program_id(0)          # batch
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # -------------------- coordinate generation --------------------
    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    i_mask = i_offset < N
    j_mask = j_offset < N

    # accumulator in FP32, shape (BLOCK_I, BLOCK_J, BLOCK_H)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # load left  [B,i,k,h] → (BLOCK_I, BLOCK_K, BLOCK_H)
        left_ptrs = left_ptr + \
            pid_b * stride_lb + \
            i_offset[:, None, None] * stride_li + \
            k_offset[None, :, None] * stride_lk + \
            tl.arange(0, BLOCK_H)[None, None, :] * stride_lh
        left = tl.load(left_ptrs,
                       mask=i_mask[:, None, None] & k_mask[None, :, None],
                       other=0.0).to(tl.float32)

        # load right [B,j,k,h] → (BLOCK_J, BLOCK_K, BLOCK_H)
        right_ptrs = right_ptr + \
            pid_b * stride_rb + \
            j_offset[:, None, None] * stride_rj + \
            k_offset[None, :, None] * stride_rk + \
            tl.arange(0, BLOCK_H)[None, None, :] * stride_rh
        right = tl.load(right_ptrs,
                        mask=j_mask[:, None, None] & k_mask[None, :, None],
                        other=0.0).to(tl.float32)

        # for each hidden slice we perform a dot over the K dimension
        # (BLOCK_I, BLOCK_K) dot (BLOCK_J, BLOCK_K)^T  → (BLOCK_I, BLOCK_J)
        for h in range(0, BLOCK_H):
            l = left[:, :, h]          # (I, K)
            r = right[:, :, h]         # (J, K)
            # tl.dot expects (M,K) and (N,K); it sums over K
            acc[:, :, h] += tl.dot(l, r)   # (I,J)

    # write result
    out_ptrs = out_ptr + \
        pid_b * stride_ob + \
        i_offset[:, None, None] * stride_oi + \
        j_offset[None, :, None] * stride_oj + \
        tl.arange(0, BLOCK_H)[None, None, :] * stride_oh
    tl.store(out_ptrs, acc,
             mask=i_mask[:, None, None] & j_mask[None, :, None])


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Parameters
    ----------
    data : tuple
        (input, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    ).to(torch.float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)  →  [B,N,N,hidden]
    # ------------------------------------------------------------------
    # shape: (dim, hidden)
    left_proj_w = weights["left_proj.weight"].to(device).t()   # (dim, hidden)
    right_proj_w = weights["right_proj.weight"].to(device).t()
    left = torch.einsum("bnqd,dh->bnqh", x, left_proj_w)   # (B,N,N,hidden)
    right = torch.einsum("bnqd,dh->bnqh", x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                       # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).t()
    right_gate_w = weights["right_gate.weight"].to(device).t()
    out_gate_w = weights["out_gate.weight"].to(device).t()

    left_gate = torch.einsum("bnqd,dh->bnqh", x, left_gate_w).sigmoid()
    right_gate = torch.einsum("bnqd,dh->bnqh", x, right_gate_w).sigmoid()
    out_gate = torch.einsum("bnqd,dh->bnqh", x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the ternary reduction (FP16 → FP32)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32,
                      device=device)

    # strides (in elements)
    stride_lb = left_fp16.stride(0)
    stride_li = left_fp16.stride(1)
    stride_lk = left_fp16.stride(2)
    stride_lh = left_fp16.stride(3)

    stride_rb = right_fp16.stride(0)
    stride_rj = right_fp16.stride(1)
    stride_rk = right_fp16.stride(2)
    stride_rh = right_fp16.stride(3)

    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_oh = out.stride(3)

    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    BLOCK_H = 32                     # hidden_dim is a multiple of 32

    grid = (B,
            triton.cdiv(N, BLOCK_I),
            triton.cdiv(N, BLOCK_J))

    tri_mul_kernel[grid](
        left_fp16, right_fp16, out,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: hidden‑dim LayerNorm, gating, final Linear
    # ------------------------------------------------------------------
    # hidden LayerNorm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
    )

    # apply output gate (broadcast over hidden dim)
    out = out * out_gate

    # final linear projection back to *dim*
    to_out_w = weights["to_out.weight"].to(device).t()   # (hidden, dim)
    out = torch.einsum("bnqh,hd->bnqd", out, to_out_w)

    return out.to(torch.float32)
