"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where
    left  = Linear(x) * mask * sigmoid(Linear_gate(x))
    right = Linear(x) * mask * sigmoid(Linear_gate(x))

`left` and `right` have shape (B, N, N, H)  (H = hidden_dim).  
The kernel below computes the above reduction directly on GPU, keeping
the channel dimension (`H`) inside the register file (vectorized loads)
so that we only have to loop over the long sequence dimension `N`.
All other operations (LayerNorm, linear layers, gating, final
normalisation / projection) are performed in PyTorch – they are
memory‑bound and cheap compared with the O(B·N³·H) work of the einsum.

The kernel is launched with a 2‑D grid (i‑block × j‑block). Each program
instance computes a tile of shape (BLOCK_I, BLOCK_J, BLOCK_H) of the
output.  The inner reduction over `k` is performed in a tiled fashion
with `BLOCK_K` elements per iteration.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel that performs the einsum reduction
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    # pointers
    left_ptr,          # const float*  [B, N, N, H]
    right_ptr,         # const float*  [B, N, N, H]
    out_ptr,           #       float*  [B, N, N, H]
    # strides
    stride_bl, stride_bi, stride_bk, stride_bd,   # left  strides
    stride_rl, stride_rj, stride_rk, stride_rd,   # right strides
    stride_ol, stride_oi, stride_oj, stride_od,   # out   strides
    # sizes
    B, N, H,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_i = tl.program_id(0)          # tile index along i
    pid_j = tl.program_id(1)          # tile index along j
    pid_b = tl.program_id(2)          # batch index

    # tile start coordinates
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # range of indices inside the tile
    i_off = i_base + tl.arange(0, BLOCK_I)
    j_off = j_base + tl.arange(0, BLOCK_J)

    # mask out‑of‑bounds (i,j may exceed N)
    i_mask = i_off < N
    j_mask = j_off < N

    # ------------------------------------------------------------------
    # allocate accumulator for the tile (BLOCK_I, BLOCK_J, BLOCK_H)
    # using float32 for numeric stability
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over k in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        # actual k range for this iteration
        cur_k = k_start + tl.arange(0, BLOCK_K)
        k_mask = cur_k < N

        # --------------------------------------------------------------
        # Load left[ b, i, k, : ]  -> shape (BLOCK_I, BLOCK_K, BLOCK_H)
        # --------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + pid_b * stride_bl
            + i_off[:, None, None] * stride_bi
            + cur_k[None, :, None] * stride_bk
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_bd
        )
        left_tile = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None], other=0.0)

        # --------------------------------------------------------------
        # Load right[ b, j, k, : ]  -> shape (BLOCK_J, BLOCK_K, BLOCK_H)
        # --------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + pid_b * stride_rl
            + j_off[:, None, None] * stride_rj
            + cur_k[None, :, None] * stride_rk
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_rd
        )
        right_tile = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_mask[None, :, None], other=0.0)

        # --------------------------------------------------------------
        # Multiply element‑wise over the hidden dimension and reduce over k
        # left_tile : (I, K, H)
        # right_tile: (J, K, H)
        # We need Σ_k left[i,k,h] * right[j,k,h] for every (i,j,h)
        # --------------------------------------------------------------
        # broadcast left over j and right over i
        left_b = left_tile[:, None, :, :]          # (I,1,K,H)
        right_b = right_tile[None, :, :, :]        # (1,J,K,H)
        prod = left_b * right_b                    # (I,J,K,H)

        # sum over K
        acc += tl.sum(prod, axis=2)                # (I,J,H)

    # ------------------------------------------------------------------
    # store the result
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_ol
        + i_off[:, None, None] * stride_oi
        + j_off[None, :, None] * stride_oj
        + tl.arange(0, BLOCK_H)[None, None, :] * stride_od
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float32),
        mask=i_mask[:, None, None] & j_mask[None, :, None],
    )


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, D]   (float32)
        - mask         : torch.Tensor  [B, N, N]      (float32 or bool)
        - weights      : dict of torch.Tensor, containing all Linear /
                         LayerNorm parameters described in the prompt.
        - config       : dict with keys `dim` (D) and `hidden_dim` (H).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to apply a Linear layer given weight (no bias) in FP32
    # ------------------------------------------------------------------
    def linear(x, w):
        # x: (..., D)  w: (D, H)
        return torch.nn.functional.linear(x, w)

    # ------------------------------------------------------------------
    # LayerNorm (no bias handling – bias already supplied in weight dict)
    # ------------------------------------------------------------------
    def layer_norm(x, weight, bias):
        return torch.nn.functional.layer_norm(
            x, normalized_shape=(D,), weight=weight, bias=bias, eps=1e-5
        )

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Projections
    # ------------------------------------------------------------------
    left = linear(x, weights["left_proj.weight"])          # (B,N,N,H)
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_ = mask.unsqueeze(-1).to(left.dtype)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))   # (B,N,N,H)
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core einsum – Triton kernel
    # ------------------------------------------------------------------
    # allocate output buffer (float32)
    out = torch.empty((B, N, N, H), device=input_tensor.device, dtype=torch.float32)

    # strides (in elements, not bytes) for each tensor
    stride_bl, stride_bi, stride_bk, stride_bd = left.stride()
    stride_rl, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ol, stride_oi, stride_oj, stride_od = out.stride()

    # tile sizes – tuned for Hopper (SM90) with H up to 384
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_H = min(64, H)          # we process up to 64 channels per program

    grid = (
        triton.cdiv(N, BLOCK_I),   # i‑tiles
        triton.cdiv(N, BLOCK_J),   # j‑tiles
        B,                         # batch dimension
    )

    # launch
    trimul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        stride_bl=tl.constant(stride_bl), stride_bi=tl.constant(stride_bi),
        stride_bk=tl.constant(stride_bk), stride_bd=tl.constant(stride_bd),
        stride_rl=tl.constant(stride_rl), stride_rj=tl.constant(stride_rj),
        stride_rk=tl.constant(stride_rk), stride_rd=tl.constant(stride_rd),
        stride_ol=tl.constant(stride_ol), stride_oi=tl.constant(stride_oi),
        stride_oj=tl.constant(stride_oj), stride_od=tl.constant(stride_od),
        B=B, N=N, H=H,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Output normalisation, gating and final projection back to D
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])   # (B,N,N,D)

    return out.to(torch.float32)
