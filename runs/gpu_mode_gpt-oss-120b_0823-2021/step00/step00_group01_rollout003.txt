"""
TriMul “outgoing” kernel (AlphaFold‑3 style).

The expensive part of the module is the batched tensor‑product

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right have shape (B, N, N, H) (H = hidden_dim).
We fuse this N‑reducing contraction into a single Triton kernel.
All surrounding ops (LayerNorm, Linear, sigmoid, masking) are kept in PyTorch
so that we can reuse the supplied weight tensors directly.

The kernel tiles the (i, j) plane with BLOCK_I × BLOCK_J threads
and the hidden dimension with BLOCK_D.  The reduction over k is performed
in blocks of size BLOCK_K and accumulated in FP32.  The final result is
written in FP32 (the model works in float32, but the kernel can read
in FP16/BF16 for higher bandwidth).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the N‑reducing outer‑product
# ----------------------------------------------------------------------
@triton.jit
def trimul_outer_kernel(
    # pointers
    left_ptr,          # left  [B, N, N, H]
    right_ptr,         # right [B, N, N, H]
    out_ptr,           # out   [B, N, N, H]
    # shapes / strides
    B, N, H,           # int scalars
    stride_lbz, stride_lbz2, stride_lbk, stride_lbd,   # left  strides (b, i, k, d)
    stride_rbz, stride_rbk, stride_rbj, stride_rbd,   # right strides (b, j, k, d)
    stride_obz, stride_obi, stride_obj, stride_obd,   # out   strides (b, i, j, d)
    # tile sizes (compile‑time constants)
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # ------------------------------------------------------------------
    # compute base offsets for the current tile
    # ------------------------------------------------------------------
    i_off = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    d_off = tl.arange(0, BLOCK_D)

    mask_i = i_off < N
    mask_j = j_off < N
    mask_d = d_off < H

    # pointers to the start of the tile (batch dimension fixed)
    left_tile_ptr  = left_ptr  + pid_b * stride_lbz
    right_tile_ptr = right_ptr + pid_b * stride_rbz
    out_tile_ptr   = out_ptr   + pid_b * stride_obz

    # ------------------------------------------------------------------
    # accumulator in fp32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over k
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # ------------------------------------------------------------------
        # load left[b, i, k, d]  -> shape (BLOCK_I, BLOCK_K, BLOCK_D)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_tile_ptr
            + i_off[:, None, None] * stride_lbz2
            + cur_k[None, :, None] * stride_lbk
            + d_off[None, None, :] * stride_lbd
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)
        # cast to fp32 for accumulation
        left = left.to(tl.float32)

        # ------------------------------------------------------------------
        # load right[b, j, k, d] -> shape (BLOCK_J, BLOCK_K, BLOCK_D)
        # ------------------------------------------------------------------
        right_ptrs = (
            right_tile_ptr
            + j_off[:, None, None] * stride_rbj
            + cur_k[None, :, None] * stride_rbk
            + d_off[None, None, :] * stride_rbd
        )
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)
        right = right.to(tl.float32)

        # ------------------------------------------------------------------
        # multiply‑accumulate over k and d
        #   left: (i, k, d)   right: (j, k, d)
        #   we need Σ_k left[i,k,d] * right[j,k,d] for each d
        # ------------------------------------------------------------------
        # expand dimensions to broadcast over i/j
        left_exp  = left[:, :, None, :]   # (i, k, 1, d)
        right_exp = right[None, :, :, :]  # (1, j, k, d)
        # product over k (and sum)
        prod = left_exp * right_exp       # (i, j, k, d)
        acc += tl.sum(prod, axis=1)       # sum over k -> (i, j, d)

    # ------------------------------------------------------------------
    # write result
    # ------------------------------------------------------------------
    out_ptrs = (
        out_tile_ptr
        + i_off[:, None, None] * stride_obi
        + j_off[None, :, None] * stride_obj
        + d_off[None, None, :] * stride_obd
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float32),
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


# ----------------------------------------------------------------------
# Host‑side helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_outer_product(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : torch.Tensor of shape (B, N, N, H)  (dtype fp16/bf16/float32)
    returns      : torch.Tensor of shape (B, N, N, H)  (dtype float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in number of elements (not bytes)
    stride_lbz, stride_lbz2, stride_lbk, stride_lbd = left.stride()
    stride_rbz, stride_rbj, stride_rbk, stride_rbd = right.stride()
    stride_obz, stride_obi, stride_obj, stride_obd = out.stride()

    # tile sizes – tuned for Hopper (sm_90)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_D = 32
    BLOCK_K = 32

    grid = (
        B,
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
    )
    trimul_outer_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lbz,
        stride_lbz2,
        stride_lbk,
        stride_lbd,
        stride_rbz,
        stride_rbk,
        stride_rbj,
        stride_rbd,
        stride_obz,
        stride_obi,
        stride_obj,
        stride_obd,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_D=BLOCK_D,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# Full TriMul forward (no backward) – mixes PyTorch and Triton
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim]   torch.float32 (on CUDA)
            - mask_tensor  : [B, N, N]        torch.float32 (0/1) or None
            - weights_dict : mapping name → torch.Tensor (FP32)
            - config_dict  : {"dim": int, "hidden_dim": int}
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. LayerNorm on input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask if present (mask shape (B,N,N,1))
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N‑reducing outer product – Triton kernel
    # ------------------------------------------------------------------
    # cast to fp16 for higher bandwidth (kernel accumulates in fp32)
    left_t = left.to(torch.float16)
    right_t = right.to(torch.float16)
    out = trimul_outer_product(left_t, right_t)   # (B,N,N,hidden_dim), fp32

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
