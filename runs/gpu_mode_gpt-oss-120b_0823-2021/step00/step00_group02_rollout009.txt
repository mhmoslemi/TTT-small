"""
TriMul “outgoing” kernel (AlphaFold‑3).

The heavy part is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We keep all linear / LayerNorm / gating logic in PyTorch (they are cheap)
and implement the 3‑D “big” contraction in Triton.
The kernel is tiled over (i,j) and over the hidden dimension d.
For each tile we accumulate a mat‑mul over the k‑dimension.
The final tensor is post‑processed (norm, gating, linear) in PyTorch
so the output type is float32 as required.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: tiled mat‑mul for each hidden‑dim slice
# --------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    # pointers
    left_ptr,    # [B, N, N, H]  (i, k) ordering
    right_ptr,   # [B, N, N, H]  (j, k) ordering
    out_ptr,     # [B, N, N, H]  result
    # sizes
    B, N, H,
    # strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    """
    pid_b = tl.program_id(0)          # batch
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile
    pid_d = tl.program_id(3)          # d‑tile

    # ------------------------------------------------------------------
    # 1) Tile coordinates
    # ------------------------------------------------------------------
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d_offset = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_d = d_offset < H

    # ------------------------------------------------------------------
    # 2) Accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) Loop over k (the contracted dimension)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # --------------------------------------------------------------
        # Load left[b, i, k, d]  – shape (M, K, D)
        # --------------------------------------------------------------
        left = tl.load(
            left_ptr
            + pid_b * stride_left_b
            + i_offset[:, None, None] * stride_left_i
            + k_offset[None, :, None] * stride_left_k
            + d_offset[None, None, :] * stride_left_h,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # --------------------------------------------------------------
        # Load right[b, j, k, d] – shape (N, K, D)
        # --------------------------------------------------------------
        right = tl.load(
            right_ptr
            + pid_b * stride_right_b
            + j_offset[:, None, None] * stride_right_j
            + k_offset[None, :, None] * stride_right_k
            + d_offset[None, None, :] * stride_right_h,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # --------------------------------------------------------------
        # Compute Σ_k left[...,d] * right[...,d] for every d in the tile.
        # We unroll over the hidden‑dim tile because tl.dot works on 2‑D
        # matrices only.
        # --------------------------------------------------------------
        for dd in range(0, BLOCK_D):
            # shape: (M, K) • (N, K).T  -> (M, N)
            acc[:, :, dd] += tl.dot(
                left[:, :, dd],          # (M, K)
                right[:, :, dd].T,       # (K, N)
            )

    # ------------------------------------------------------------------
    # 4) Write back
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_out_b
        + i_offset[:, None, None] * stride_out_i
        + j_offset[None, :, None] * stride_out_j
        + d_offset[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )

# Host‑side wrapper
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    The heavy contraction is executed on GPU via Triton.
    All other linear / norm / gate operations stay in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1) LayerNorm on the input (dim)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias

    # --------------------------------------------------------------
    # 2) Linear projections (no bias) + masking
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"]
    )  # [B,N,N,hidden]
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"]
    )
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 3) Gating (sigmoid) and application
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"]
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"]
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4) Triton contraction: out_hidden[b,i,j,d]
    # --------------------------------------------------------------
    out_hidden = torch.empty(
        (B, N, N, hidden_dim), device=device, dtype=dtype
    )

    # strides for the three tensors (batch‑first contiguous)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out_hidden.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_D = 32  # hidden_dim is a multiple of 32 in all test configs

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(hidden_dim, BLOCK_D),
    )

    trimul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_hidden,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=s_left[0],
        stride_left_i=s_left[1],
        stride_left_k=s_left[2],
        stride_left_h=s_left[3],
        stride_right_b=s_right[0],
        stride_right_j=s_right[1],
        stride_right_k=s_right[2],
        stride_right_h=s_right[3],
        stride_out_b=s_out[0],
        stride_out_i=s_out[1],
        stride_out_j=s_out[2],
        stride_out_h=s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 5) Post‑processing (LayerNorm, gating, final linear) – stay in PyTorch
    # --------------------------------------------------------------
    # to_out_norm
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = (out_hidden - out_hidden.mean(-1, keepdim=True)) / torch.sqrt(
        out_hidden.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    # out_gate (already computed earlier, same shape as x)
    out = out * out_gate

    # final linear back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(dtype)
