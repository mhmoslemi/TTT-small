"""
TriMul “outgoing” kernel (AlphaFold‑3)

The forward pass consists of
1. Layer‑norm over the last channel.
2. Two linear projections (left / right) and three gating‑linear layers.
3. Optional element‑wise masking of the projected tensors.
4. A tri‑multiplicative update:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   (implemented with a fused Triton kernel that computes the
    per‑channel matrix‑multiplication left_h @ right_h.T).
5. Output normalization, gating and final linear projection.

Only the N‑by‑N‑by‑H matrix‑multiplication (step 4) is performed in
Triton; all other simple linear / norm / sigmoid operations are kept
in PyTorch for readability and to avoid excessive kernel launch
overheads.  The kernel works for any batch size (B = 1 or 2), any
sequence length N (≤ 1024) and any channel dimensions C (the model
dimension) and H (the hidden dimension, typically 128).  The final
output is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ------------------------------------------------------------
# Triton kernel: per‑channel matrix multiplication
# ------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    left_ptr,                 # const *float32
    right_ptr,                # const *float32
    out_ptr,                  # *float32
    B, N, H,                  # int meta
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,    # tile size for i dimension
    BLOCK_J: tl.constexpr,    # tile size for j dimension
    BLOCK_K: tl.constexpr,    # tile size for k reduction
    BLOCK_H: tl.constexpr,    # tile size for channel dimension
):
    """
    Compute, for a single batch element,
        out[i, j, h] = Σ_k left[i, k, h] * right[j, k, h]

    The computation is blocked over i, j and h; the reduction over k
    is performed in a loop.  All tensors are stored in row‑major
    layout (the default for torch.contiguous()).
    """

    pid = tl.program_id(0)                     # linear index over (i,j,h) blocks
    # decode block coordinates
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    num_h = tl.cdiv(H, BLOCK_H)

    i_block = (pid // (num_j * num_h)) * BLOCK_I
    rem = pid % (num_j * num_h)
    j_block = (rem // num_h) * BLOCK_J
    h_block = (rem % num_h) * BLOCK_H

    # offsets within the block
    offs_i = i_block + tl.arange(0, BLOCK_I)
    offs_j = j_block + tl.arange(0, BLOCK_J)
    offs_h = h_block + tl.arange(0, BLOCK_H)

    # mask out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulate result
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # reduction over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- load left : (I, K, H) ----------
        # left layout: (b, i, k, h)
        left_ptrs = left_ptr + \
            offs_i[:, None, None] * stride_left_i + \
            offs_k[None, :, None] * stride_left_k + \
            offs_h[None, None, :] * stride_left_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                       other=0.0)

        # -------- load right : (J, K, H) ----------
        # right layout: (b, j, k, h)
        right_ptrs = right_ptr + \
            offs_j[:, None, None] * stride_right_j + \
            offs_k[None, :, None] * stride_right_k + \
            offs_h[None, None, :] * stride_right_h
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                        other=0.0)

        # element‑wise multiply and reduce over k
        # left: (I,K,H), right: (J,K,H) → we need Σ_k left[i,k,h] * right[j,k,h]
        # we broadcast over i and j, then sum k
        prod = left[:, None, :, :] * right[None, :, :, :]          # (I, J, K, H)
        acc += tl.sum(prod, axis=2)                               # sum over K → (I, J, H)

    # store the result
    out_ptrs = out_ptr + \
        offs_i[:, None, None] * stride_out_i + \
        offs_j[None, :, None] * stride_out_j + \
        offs_h[None, None, :] * stride_out_h
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])


# ------------------------------------------------------------
# Helper: launch the Triton kernel for a single batch
# ------------------------------------------------------------
def _trimul_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left , right : [B, N, N, H]  (float32, contiguous)
    returns out   : [B, N, N, H]
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in element units (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # kernel launch parameters
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64          # reduction tile
    BLOCK_H = 32

    # total number of program instances = ceil(N/BLOCK_I)*ceil(N/BLOCK_J)*ceil(H/BLOCK_H)*B
    total_blocks = triton.cdiv(N, BLOCK_I) * triton.cdiv(N, BLOCK_J) * triton.cdiv(H, BLOCK_H) * B
    grid = (total_blocks,)

    _trimul_matmul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, C]   float32 (already on CUDA)
        - mask         : [B, N, N]      (bool / float, already on CUDA)
        - weights      : dict of leaf tensors
        - config       : dict with keys
              "dim"        -> C
              "hidden_dim" -> H

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]

    # ----------------------------------------------------
    # 1. LayerNorm over the last channel (C)
    # ----------------------------------------------------
    eps = 1e-5
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # mean & var over last dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                     # [B,N,N,C]

    # ----------------------------------------------------
    # 2. Linear projections (no bias)
    # ----------------------------------------------------
    left_proj_w = weights["left_proj.weight"]        # (H, C)
    right_proj_w = weights["right_proj.weight"]      # (H, C)

    left = F.linear(x, left_proj_w)                  # [B,N,N,H]
    right = F.linear(x, right_proj_w)                # [B,N,N,H]

    # ----------------------------------------------------
    # 3. Apply mask (if present)
    # ----------------------------------------------------
    # mask is [B,N,N]; we broadcast to H
    if mask is not None and mask.numel() > 0:
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ----------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise gating
    # ----------------------------------------------------
    left_gate_w = weights["left_gate.weight"]        # (H, C)
    right_gate_w = weights["right_gate.weight"]      # (H, C)
    out_gate_w = weights["out_gate.weight"]          # (H, C)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------
    # 5. Tri‑multiplicative update (core O(N³) part) – Triton
    # ----------------------------------------------------
    # left/right are [B,N,N,H]; Triton kernel expects contiguous memory
    left = left.contiguous()
    right = right.contiguous()
    out = _trimul_matmul(left, right)               # [B,N,N,H]

    # ----------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # ----------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    # layernorm over hidden dimension H
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate

    to_out_w = weights["to_out.weight"]              # (C, H)
    out = F.linear(out, to_out_w)                    # back to [B,N,N,C]

    return out.to(torch.float32)
