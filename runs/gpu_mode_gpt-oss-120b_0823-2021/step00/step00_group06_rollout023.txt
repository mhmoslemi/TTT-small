"""
TriMul “outgoing” kernel (AlphaFold‑3)

We implement the heavy N³ part:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden dimension `d` this is exactly a matrix multiplication
    L_d (N×N) @ R_d.T (N×N)  →  (N×N)

The kernel therefore performs a batched GEMM over the
    (batch, hidden) axis.  All LayerNorm / Linear / sigmoid / mask
operations are kept in PyTorch – they are cheap compared to the N³
multiply‑accumulate.  The kernel accumulates in float32 and writes
float32, while the inner loads use the input dtype (fp16/bf16) for
maximum throughput.

The kernel is launched with a 4‑D grid:
    pid_b   – batch index
    pid_h   – hidden‑dimension index
    pid_m   – output row block (i)
    pid_n   – output column block (j)

Block sizes were chosen for Hopper (SM‑90): 128×128 tiles with
k‑tiles of 32.  These fit in shared memory and give good occupancy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: out = left @ rightᵀ   (per batch & hidden dim)
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left  [B, N, N, H]
    stride_lb, stride_li, stride_lk, stride_ld,
    # strides for right [B, N, N, H]
    stride_rb, stride_rj, stride_rk, stride_rd,
    # strides for out   [B, N, N, H]
    stride_ob, stride_oi, stride_oj, stride_od,
    # tile sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_b = tl.program_id(0)           # batch
    pid_h = tl.program_id(1)           # hidden dimension
    pid_m = tl.program_id(2)           # i‑block
    pid_n = tl.program_id(3)           # j‑block

    # offsets within the N×N matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # left[b, i, k, h]   shape (BLOCK_M, BLOCK_K)
        # right[b, j, k, h]  shape (BLOCK_N, BLOCK_K)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + pid_b * stride_lb
            + pid_h * stride_ld
            + offs_m[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        right_ptrs = (
            right_ptr
            + pid_b * stride_rb
            + pid_h * stride_rd
            + offs_n[:, None] * stride_rj
            + offs_k[None, :] * stride_rk
        )

        left = tl.load(
            left_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        right = tl.load(
            right_ptrs,
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right is (j,k); we need its transpose (k,j) for the dot product
        acc += tl.dot(left, tl.trans(right))

    # write the result
    out_ptrs = (
        out_ptr
        + pid_b * stride_ob
        + pid_h * stride_od
        + offs_m[:, None] * stride_oi
        + offs_n[None, :] * stride_oj
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------
# Host‑side wrapper
# --------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass with a Triton kernel for the N³ einsum.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input    (dim -> dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape after proj: [B, N, N, hidden]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                      # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel: out = left @ rightᵀ   (per hidden dimension)
    # ------------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B,                                 # batch axis
        hidden,                            # hidden‑dim axis
        triton.cdiv(N, BLOCK_M),           # i‑blocks
        triton.cdiv(N, BLOCK_N),           # j‑blocks
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # good default for fp32 accumulation
    )

    # ------------------------------------------------------------------
    # 6. Final LayerNorm + output gate + linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # ensure float32 output
    return out.to(torch.float32)
