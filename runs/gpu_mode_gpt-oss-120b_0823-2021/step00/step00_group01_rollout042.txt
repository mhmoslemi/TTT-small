"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The expensive part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For every hidden dimension *d* this is exactly a matrix multiplication
between the (i,k) slice of ``left`` and the transpose of the (j,k) slice of
``right``:

    out[...,d] = left[...,d] @ right[...,d].T

The kernel below computes this batched matrix‑multiply for a block of
hidden dimensions (``BLOCK_D``) at a time.  All surrounding LayerNorm,
linear projections and gates are performed in PyTorch (they are cheap
compared with the N³ work).  The kernel works for arbitrary batch size,
sequence length and hidden dimension (the hidden dimension may not be a
multiple of the block size – the code masks the out‑of‑bounds elements).

The final output is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------------------- #
# Triton kernel:  batched GEMM for a slice of the hidden dimension
# --------------------------------------------------------------------------- #
@triton.jit
def _trimul_gemm_kernel(
    # pointers
    left_ptr,        # [B, N, N, D]  – left projection
    right_ptr,       # [B, N, N, D]  – right projection
    out_ptr,         # [B, N, N, D]  – result
    # shape / strides
    B, N, D,
    stride_bl, stride_il, stride_kl, stride_dl,   # left  (B,N,K,D)
    stride_br, stride_jr, stride_kr, stride_dr,   # right (B,N,K,D)
    stride_bo, stride_io, stride_jo, stride_do,   # out   (B,N,N,D)
    # compile‑time block sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Batched GEMM for a slice of the hidden dimension.

    Computes, for each batch ``b`` and hidden block ``d_blk``:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    where ``right`` is implicitly transposed on the ``k`` axis.
    """
    pid = tl.program_id(0)

    # --------------------------------------------------------------- #
    # Decode program id → (batch, i‑block, j‑block)
    # --------------------------------------------------------------- #
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    pid_b   = pid // (num_i * num_j)
    pid_ij  = pid %  (num_i * num_j)
    pid_i   = pid_ij // num_j
    pid_j   = pid_ij %  num_j

    # --------------------------------------------------------------- #
    # Offsets for this block
    # --------------------------------------------------------------- #
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ----------------------------------------------------------------- #
    # Loop over hidden‑dimension blocks
    # ----------------------------------------------------------------- #
    for d_blk in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_blk * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # accumulator in fp32 (left/right may be fp16 / bf16)
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

        # ----------------------------------------------------------- #
        # Main K‑loop (blocked)
        # ----------------------------------------------------------- #
        for k_start in range(0, tl.cdiv(N, BLOCK_K)):
            cur_k = k_start * BLOCK_K + offs_k
            mask_k = cur_k < N

            # -------------------- load left ------------------------ #
            # left[b, i, k, d]
            left = tl.load(
                left_ptr
                + pid_b * stride_bl
                + offs_i[:, None, None] * stride_il
                + cur_k[None, :, None] * stride_kl
                + offs_d[None, None, :] * stride_dl,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                other=0.0,
            )

            # -------------------- load right (transposed) ---------- #
            # we need right[b, j, k, d] but we will treat the K‑axis as the *column*
            # of a GEMM, i.e. we load right[b, j, k, d] and later perform a dot
            # over the K‑axis.
            right = tl.load(
                right_ptr
                + pid_b * stride_br
                + offs_j[:, None, None] * stride_jr
                + cur_k[None, :, None] * stride_kr
                + offs_d[None, None, :] * stride_dr,
                mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                other=0.0,
            )

            # -------------------- multiply‑accumulate -------------- #
            # left: (I,K,D), right: (J,K,D) → need dot over K
            # tl.dot works on 2‑D tensors, so we do an element‑wise multiply
            # then sum over K.
            prod = left * right                    # (I, K, D)
            acc += tl.sum(prod, axis=1)            # (I, D)  → broadcast over J later

            # The above gives a (I, D) result that is the same for every J.
            # To obtain the full (I, J, D) tensor we need to broadcast the
            # summed values across the J dimension.
            # This broadcasting is cheap (just a view), so we expand:
            acc = acc[:, None, :] + tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

        # ----------------------------------------------------------- #
        # Store the block
        # ----------------------------------------------------------- #
        out = acc.to(tl.float32)
        tl.store(
            out_ptr
            + pid_b * stride_bo
            + offs_i[:, None, None] * stride_io
            + offs_j[None, :, None] * stride_jo
            + offs_d[None, None, :] * stride_do,
            out,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
        )


# --------------------------------------------------------------------------- #
# Helper: launch wrapper
# --------------------------------------------------------------------------- #
def _launch_trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    `left`/`right` : [B, N, N, D] (float16 / bfloat16 / float32)
    Returns          : [B, N, N, D] (float32)
    """
    B, N, _, D = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in elements (not bytes)
    stride_bl, stride_il, stride_kl, stride_dl = left.stride()
    stride_br, stride_jr, stride_kr, stride_dr = right.stride()
    stride_bo, stride_io, stride_jo, stride_do = out.stride()

    # Block sizes – tuned for Hopper SM90 (32‑wide warp, 128 KB shared memory)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64
    BLOCK_D = 32   # hidden‑dim block, fits nicely in registers

    grid = (
        triton.cdiv(B, 1)
        * triton.cdiv(N, BLOCK_I)
        * triton.cdiv(N, BLOCK_J),
    )
    _trimul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        D,
        stride_bl,
        stride_il,
        stride_kl,
        stride_dl,
        stride_br,
        stride_jr,
        stride_kr,
        stride_dr,
        stride_bo,
        stride_io,
        stride_jo,
        stride_do,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    return out


# --------------------------------------------------------------------------- #
# Main entry point – the full TriMul forward pass
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask        : [B, N, N]       (bool / float)
        - weights     : dict of pre‑loaded weight tensors
        - config      : dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # --------------------------------------------------------------- #
    # 1. LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias

    # --------------------------------------------------------------- #
    # 2. Linear projections (no bias) – we keep them in fp16 for speed
    # --------------------------------------------------------------- #
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_w).to(torch.float16)
    right = F.linear(x, right_proj_w).to(torch.float16)

    # --------------------------------------------------------------- #
    # 3. Apply mask (if present)
    # --------------------------------------------------------------- #
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)        # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------- #
    # 4. Gating
    # --------------------------------------------------------------- #
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w)).to(left.dtype)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)).to(right.dtype)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w)).to(left.dtype)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # 5. Core TriMul – heavy N³ work – done in Triton
    # --------------------------------------------------------------- #
    #   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = _launch_trimul_gemm(left, right)   # fp32 result

    # --------------------------------------------------------------- #
    # 6. Final LayerNorm, gating and projection back to dim
    # --------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # LayerNorm over hidden_dim
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(out.var(-1, unbiased=False, keepdim=True) + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate

    # Linear projection back to original dim
    out = F.linear(out, to_out_w)

    return out
