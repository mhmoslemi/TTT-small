"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the ternary product

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix‑multiply for every hidden dimension d

    out_d = left_d @ right_d.transpose(-2,-1)          # (B,N,N) @ (B,N,N)^T

The kernel below performs this batched GEMM in FP32 (inputs are FP16/BF16 for
speed) using a 2‑D tiling over the (i,j) output axes and a 1‑D tiling over the
reduction axis k.  All surrounding LayerNorm / Linear / gating logic is kept
in PyTorch because they are cheap compared with the O(B·N³·D) product.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM over the hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # dimensions
    B, N, D,
    # strides (in elements, not bytes)
    stride_lb, stride_li, stride_lk, stride_ld,   # left  [b,i,k,d]
    stride_rb, stride_rj, stride_rk, stride_rd,   # right [b,j,k,d]
    stride_ob, stride_oi, stride_oj, stride_od,   # out   [b,i,j,d]
    # tiling constants
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    # program id encodes (block_i, block_j, block_d, batch)
    total_i = tl.cdiv(N, BLOCK_M)
    total_j = tl.cdiv(N, BLOCK_N)
    total_d = tl.cdiv(D, BLOCK_D)

    b = pid // (total_i * total_j * total_d)
    tmp = pid % (total_i * total_j * total_d)
    d_block = tmp // (total_i * total_j)
    tmp2 = tmp % (total_i * total_j)
    i_block = tmp2 // total_j
    j_block = tmp2 % total_j

    # offsets within the tile
    i_offset = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    d_offset = d_block * BLOCK_D + tl.arange(0, BLOCK_D)

    # mask for out‑of‑bounds
    i_mask = i_offset < N
    j_mask = j_offset < N
    d_mask = d_offset < D
    ij_mask = i_mask[:, None] & j_mask[None, :]   # (M,N)

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over reduction k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # load left:  [M,K,D]
        left_ptrs = (left_ptr
            + b * stride_lb
            + i_offset[:, None, None] * stride_li
            + k_offset[None, :, None] * stride_lk
            + d_offset[None, None, :] * stride_ld)
        left = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :], other=0.0)

        # load right: [N,K,D] -> we need right[j,k,d]
        right_ptrs = (right_ptr
            + b * stride_rb
            + j_offset[:, None, None] * stride_rj
            + k_offset[None, :, None] * stride_rk
            + d_offset[None, None, :] * stride_rd)
        right = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :], other=0.0)

        # multiply‑accumulate over k
        # left (M,K,D) * right (N,K,D) -> broadcast k dim
        # we compute sum_k left[...,k,d] * right[...,k,d]
        acc += tl.dot(left, right, transpose_y=True)   # (M,N,D)

    # store result
    out_ptrs = (out_ptr
        + b * stride_ob
        + i_offset[:, None, None] * stride_oi
        + j_offset[None, :, None] * stride_oj
        + d_offset[None, None, :] * stride_od)
    tl.store(out_ptrs, acc.to(tl.float32), mask=ij_mask[:, :, None] & d_mask[None, None, :])


# ----------------------------------------------------------------------
# Helper: launch the kernel
# ----------------------------------------------------------------------
def launch_trimul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B,N,N,D]  (fp16 or bf16)
    returns out:   [B,N,N,D]  (fp32)
    """
    B, N, _, D = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in elements
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # tile sizes – tuned for Hopper (sm90).  Small D tiling helps register reuse.
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8

    total_i = triton.cdiv(N, BLOCK_M)
    total_j = triton.cdiv(N, BLOCK_N)
    total_d = triton.cdiv(D, BLOCK_D)
    total_programs = B * total_i * total_j * total_d

    grid = (total_programs,)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, D=D,
        stride_lb=stride_lb, stride_li=stride_li, stride_lk=stride_lk, stride_ld=stride_ld,
        stride_rb=stride_rb, stride_rj=stride_rj, stride_rk=stride_rk, stride_rd=stride_rd,
        stride_ob=stride_ob, stride_oi=stride_oi, stride_oj=stride_oj, stride_od=stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
        num_warps=4,  # enough for fp16/bf16
    )
    return out


# ----------------------------------------------------------------------
# Full TriMul forward (outgoing) – mix of torch & triton
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Executes the outgoing TriMul operation.
    data = (input_tensor, mask, weights, config)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5)
    x = x * norm_weight + norm_bias

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – fp16 for speed
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]        # (hidden_dim, dim)

    # Cast to fp16/bf16 (GPU friendly)
    x_fp = x.to(torch.float16)

    left = torch.einsum('b i j d, h d -> b i j h', x_fp, left_proj_w)   # [B,N,N,hidden]
    right = torch.einsum('b i j d, h d -> b i j h', x_fp, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – still in fp16, then back‑cast to fp32 for matmul
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.einsum('b i j d, h d -> b i j h', x_fp, left_gate_w).sigmoid()
    right_gate = torch.einsum('b i j d, h d -> b i j h', x_fp, right_gate_w).sigmoid()
    out_gate = torch.einsum('b i j d, h d -> b i j h', x_fp, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core ternary product – Triton kernel
    # ------------------------------------------------------------------
    # left/right are [B,N,N,hidden] fp16
    out = launch_trimul_outgoing(left, right)   # fp32 result

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # LayerNorm over hidden dimension
    mean = out.mean(-1, keepdim=True)
    var = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean) / torch.sqrt(var + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate                     # gating (broadcasted)
    out = torch.einsum('b i j h, d h -> b i j d', out, to_out_w)   # final projection

    return out.to(torch.float32)
