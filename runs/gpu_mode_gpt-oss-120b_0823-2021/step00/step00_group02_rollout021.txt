"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the 4‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be rewritten as a batched matrix multiplication per output
feature *d*:

    left_d   : (B, N, N)          – left[..., :, :, d]
    right_dᵀ : (B, N, N)          – right[..., :, :, d].transpose(-2,-1)
    out_d    = left_d @ right_dᵀ   → (B, N, N)

We fuse this contraction in a Triton kernel.  All other operations
(LayerNorm, linear projections, sigmoid gates and the final
LayerNorm/Linear) are performed with regular PyTorch kernels because
they are cheap compared to the O(B·N³·C) work of the contraction.

The kernel works in mixed precision:
* left/right are loaded as fp16
* the accumulation is fp32
* the result is written back as fp16 (later cast to fp32)

The kernel is launched with three dimensions:
    dim0 : batch * hidden_dim  (flattened B and C_hidden)
    dim1 : ceil(N / BLOCK_M)   – tiles over the i‑dimension
    dim2 : ceil(N / BLOCK_N)   – tiles over the j‑dimension
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the 4‑D einsum
# ----------------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    L_ptr,                     # *fp16* left   (B*H, N, N)
    R_ptr,                     # *fp16* right  (B*H, N, N) – already transposed on last two dims
    O_ptr,                     # *fp16* output (B*H, N, N)
    N,                         # seq_len
    stride_l_batch, stride_l_m, stride_l_n,
    stride_r_batch, stride_r_m, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr,    # tile size in i
    BLOCK_N: tl.constexpr,    # tile size in j
    BLOCK_K: tl.constexpr,    # tile size in k (usually 32‑64)
):
    # program ids
    batch_id = tl.program_id(0)               # flattened batch*hidden_dim
    pid_i   = tl.program_id(1)                # tile over i
    pid_j   = tl.program_id(2)                # tile over j

    # compute the start offsets of the tile
    off_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    off_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds
    mask_i = off_i < N
    mask_j = off_j < N

    # pointers for the current tile
    L_tile_ptr = L_ptr + batch_id * stride_l_batch \
                           + off_i[:, None] * stride_l_m \
                           + tl.arange(0, BLOCK_K)[None, :] * stride_l_n
    R_tile_ptr = R_ptr + batch_id * stride_r_batch \
                           + off_j[None, :] * stride_r_n \
                           + tl.arange(0, BLOCK_K)[:, None] * stride_r_m   # note transposed layout

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load a BLOCK_M × BLOCK_K tile of left and a BLOCK_K × BLOCK_N tile of rightᵀ
        L = tl.load(L_tile_ptr, mask=mask_i[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < N), other=0.0).to(tl.float32)
        R = tl.load(R_tile_ptr, mask=mask_j[None, :] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < N), other=0.0).to(tl.float32)

        # dot‑product over the k dimension
        acc += tl.dot(L, R)

        # advance pointers to the next k‑block
        L_tile_ptr += BLOCK_K * stride_l_n
        R_tile_ptr += BLOCK_K * stride_r_m

    # write back in fp16
    O = acc.to(tl.float16)
    O_ptr_tile = O_ptr + batch_id * stride_o_batch \
                           + off_i[:, None] * stride_o_m \
                           + off_j[None, :] * stride_o_n
    tl.store(O_ptr_tile, O, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper: layer‑norm without bias (bias is added later)
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the TriMul “outgoing” module.

    Arguments
    ----------
    data : (input, mask, weights, config)
        input  – torch.Tensor, shape [B, N, N, dim]
        mask   – torch.Tensor, shape [B, N, N] (bool or 0/1)
        weights – dict of parameters (see problem statement)
        config   – dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: (B, N, N, hidden_dim)
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask if it exists
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Compute the contraction with Triton
    # ------------------------------------------------------------------
    # Convert to fp16 for the kernel (accumulation is fp32 inside)
    left_f16  = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()

    # We need right transposed on the last two axes: (..., j, k, d) -> (..., k, j, d)
    right_f16_t = right_f16.permute(0, 1, 3, 2).contiguous()   # [B, N, hidden_dim, N]

    # Flatten batch and hidden_dim together -> shape (B*hidden_dim, N, N)
    left_flat  = left_f16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right_f16_t.permute(0, 2, 1, 3).reshape(B * hidden_dim, N, N)
    out_flat   = torch.empty_like(left_flat)   # fp16 output

    # strides (in number of elements)
    stride_l_batch, stride_l_m, stride_l_n = left_flat.stride()
    stride_r_batch, stride_r_m, stride_r_n = right_flat.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_flat.stride()

    # Triton launch parameters
    BLOCK_M = 64          # tile over i
    BLOCK_N = 64          # tile over j
    BLOCK_K = 64          # tile over k (must be divisible by warp size)

    grid = (
        B * hidden_dim,                 # batch * hidden_dim
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _trimul_einsum_kernel[grid](
        left_flat, right_flat, out_flat,
        N,
        stride_l_batch, stride_l_m, stride_l_n,
        stride_r_batch, stride_r_m, stride_r_n,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,            # adequate for fp16 compute on H200
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Final LayerNorm + gating + linear projection to dim
    # ------------------------------------------------------------------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
