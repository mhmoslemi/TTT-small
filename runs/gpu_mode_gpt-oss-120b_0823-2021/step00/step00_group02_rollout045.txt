"""
TriMul “outgoing” forward pass (AlphaFold‑3).

The heavy N³ part is the einsum  
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We fuse this contraction in a Triton kernel.
All LayerNorms and Linear projections are kept in PyTorch (they are cheap
compared to the N³ work) while the triple‑product is performed on‑device.
The kernel computes a batched matrix‑multiply for each hidden dimension
block, i.e. for a fixed d‑block it evaluates

    (left_slice)_{i,k}  · (right_slice)_{j,k}ᵀ   →   (i,j)   (dot‑product over k)

and accumulates the result over the K‑tiles.
The final tensor is brought back to FP32.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,                 # *[B, N, N, H]  (already masked & gated)
    right_ptr,                # *[B, N, N, H]
    out_ptr,                  # *[B, N, N, H]  (FP32 result)
    B, N, H,                  # static sizes
    stride_lb, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rk, stride_rd,   # right strides
    stride_ob, stride_ok, stride_od,   # out strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    The kernel works on a tile (BLOCK_I, BLOCK_J, BLOCK_D).
    """
    pid = tl.program_id(0)                 # linear id over (B, i_tile, j_tile, d_tile)
    total_i = tl.cdiv(N, BLOCK_I)
    total_j = tl.cdiv(N, BLOCK_J)
    total_d = tl.cdiv(H, BLOCK_D)

    # decode pid
    d_tile = pid % total_d
    tmp   = pid // total_d
    j_tile = tmp % total_j
    i_tile = tmp // total_j

    # offsets for the current tile
    i_offset = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)
    d_offset = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_d = d_offset < H

    # allocate accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # loop over K dimension
    for k_start in range(0, N, BLOCK_K):
        k_offset = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left tile  shape (I, K, D)
        left_ptrs = left_ptr + \
            i_offset[:, None, None] * stride_lb + \
            k_offset[None, :, None] * stride_lk + \
            d_offset[None, None, :] * stride_ld
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

        # right tile shape (J, K, D)
        right_ptrs = right_ptr + \
            j_offset[:, None, None] * stride_rb + \
            k_offset[None, :, None] * stride_rk + \
            d_offset[None, None, :] * stride_rd
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

        # For each d‑block we perform a matrix‑multiply I×K dot K×Jᵀ
        # Loop over the hidden‑dim block (small, e.g. 32 or 64)
        for d in range(0, BLOCK_D):
            # slice (I,K) and (J,K)
            l = left[:, :, d]          # (I, K)
            r = right[:, :, d]         # (J, K)

            # compute matmul I×K @ K×Jᵀ  → (I,J)
            prod = tl.dot(l, r, trans_b=True)   # (I,J) in FP32
            acc[:, :, d] += prod
    # --------------------------------------------------------------
    # Store result
    # --------------------------------------------------------------
    out_ptrs = out_ptr + \
        i_offset[:, None, None] * stride_ob + \
        j_offset[None, :, None] * stride_ok + \
        d_offset[None, None, :] * stride_od
    tl.store(out_ptrs, acc.to(tl.float32),
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])


# --------------------------------------------------------------
# Host‑side wrapper
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Only the N³ contraction is executed in Triton.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----------------- 1. PyTorch part: LayerNorm & Linear -----------------
    # LayerNorm
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / \
        torch.sqrt(input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5)
    x = x * norm_w + norm_b

    # Linear projections (no bias)
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left = torch.einsum('b i j d, h d -> b i j h', x, left_proj_w)   # [B,N,N,H]
    right = torch.einsum('b i j d, h d -> b i j h', x, right_proj_w)

    # ----------------- 2. Apply mask -----------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)      # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ----------------- 3. Gates -----------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.einsum('b i j d, h d -> b i j h', x, left_gate_w))
    right_gate = torch.sigmoid(torch.einsum('b i j d, h d -> b i j h', x, right_gate_w))
    out_gate = torch.sigmoid(torch.einsum('b i j d, h d -> b i j h', x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ----------------- 4. Triton N³ contraction -----------------
    # Prepare pointers and strides
    left_ptr = left.contiguous().to(device)
    right_ptr = right.contiguous().to(device)
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides (in element units)
    stride_lb, stride_lk, stride_ld = left_ptr.stride()
    stride_rb, stride_rk, stride_rd = right_ptr.stride()
    stride_ob, stride_ok, stride_od = out.stride()

    # Tile sizes (tuned for Hopper)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_D = 32   # hidden_dim is 128 → 4 tiles

    total_i = triton.cdiv(N, BLOCK_I)
    total_j = triton.cdiv(N, BLOCK_J)
    total_d = triton.cdiv(hidden_dim, BLOCK_D)
    grid = (B * total_i * total_j * total_d, )

    trimul_outgoing_kernel[grid](
        left_ptr, right_ptr, out,
        B, N, hidden_dim,
        stride_lb, stride_lk, stride_ld,
        stride_rb, stride_rk, stride_rd,
        stride_ob, stride_ok, stride_od,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
        num_warps=4,          # good for matmul‑like work
    )

    # ----------------- 5. Post‑processing (LayerNorm, gating, final linear) -----------------
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # to_out_norm (LayerNorm over hidden_dim)
    out = (out - out.mean(-1, keepdim=True)) / \
          torch.sqrt(out.var(-1, unbiased=False, keepdim=True) + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate
    out = torch.einsum('b i j h, d h -> b i j d', out, to_out_w)   # final linear, back to dim
    return out
