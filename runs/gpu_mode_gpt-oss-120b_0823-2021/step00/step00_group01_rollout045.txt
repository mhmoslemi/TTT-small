"""
TriMul “outgoing” implementation (AlphaFold‑3)

The heavy N³ part is the pairwise multiplication

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

which is exactly a batched matrix‑multiply of the form

    out₍b,d₎ = left₍b,d₎  @  right₍b,d₎ᵀ

where left₍b,d₎ and right₍b,d₎ are (N×N) matrices.
The kernel below performs this batched GEMM in FP16 (input) /
FP32 (accumulator) and writes the result back in FP32.
All other operations (LayerNorm, Linear, gating, final LayerNorm + Linear)
are done with regular PyTorch for simplicity – they are cheap compared
to the N³ multiplication.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply C = A @ Bᵀ   (A,B : fp16, C : fp32)
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_t_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,                     # BATCH = B*H , M=N , N=N , K=N
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr
):
    """C = A @ Bᵀ   (A,B FP16, C FP32)"""
    pid = tl.program_id(0)
    #--- 2‑D grid, each program works on a BLOCK_M × BLOCK_N tile of one batch
    batch_id = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_in_batch = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))

    # group tiles to improve L2 reuse (standard Triton pattern)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    group_id = pid_in_batch // GROUP_M
    first_pid_m = group_id * GROUP_M % num_pid_m
    pid_m = first_pid_m + (pid_in_batch % GROUP_M) // num_pid_n
    pid_n = (pid_in_batch % GROUP_M) % num_pid_n

    #--- tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    #--- mask out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers for this batch
    a_ptrs = A_ptr + batch_id * stride_a_batch \
                       + offs_m[:, None] * stride_a_m \
                       + offs_k[None, :] * stride_a_k
    b_ptrs = B_ptr + batch_id * stride_b_batch \
                       + offs_n[:, None] * stride_b_n \
                       + offs_k[None, :] * stride_b_k

    # accumulate in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        # dot‑product over K‑dimension
        acc += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # write result
    c_ptrs = C_ptr + batch_id * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper to launch the kernel
# ----------------------------------------------------------------------
def batch_matmul_t(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    A,B : (B*H, N, N)   fp16
    returns C : (B*H, N, N)   fp32   = A @ Bᵀ
    """
    assert A.is_contiguous() and B.is_contiguous()
    BATCH, M, K = A.shape
    _, N, _ = B.shape
    assert K == N, "inner dimensions must match"

    C = torch.empty((BATCH, M, N), dtype=torch.float32, device=A.device)

    # tile sizes – chosen for Hopper SM‑90 (max 128‑thread warps)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    GROUP_M = 8   # groups of tiles along M for better L2 reuse

    grid = (triton.cdiv(BATCH * M * N, BLOCK_M * BLOCK_N),)

    batch_matmul_t_kernel[grid](
        A,
        B,
        C,
        BATCH,
        M,
        N,
        K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_M=GROUP_M,
        num_warps=4,
    )
    return C


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask         : [B, N, N]        (float32 or bool)
        - weights      : dict of parameter tensors
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]          # hidden dim (c_z)

    # --------------------------------------------------------------
    # 1. LayerNorm on the last dim (no bias in the reference code)
    # --------------------------------------------------------------
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    # manual LN to keep everything on GPU and avoid extra Python overhead
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x_norm = x_norm * w_norm + b_norm                # [B,N,N,D]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) – we keep fp16 for the heavy part
    # --------------------------------------------------------------
    # shape: (D, H)
    left_proj_w = weights["left_proj.weight"]          # [H, D] because torch Linear stores (out,in)
    right_proj_w = weights["right_proj.weight"]

    # we cast to fp16 to reduce memory traffic for the N³ kernel
    x_fp16 = x_norm.to(torch.float16)

    left = torch.einsum("bnid,hd->bnih", x_fp16, left_proj_w)   # [B,N,N,H]
    right = torch.einsum("bnid,hd->bnih", x_fp16, right_proj_w)# [B,N,N,H]

    # --------------------------------------------------------------
    # 3. Apply mask (if present) – mask broadcast over H
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)        # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gates (sigmoid) – applied in FP32 for stability
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.einsum("bnid,hd->bnih", x_norm, left_gate_w))
    right_gate = torch.sigmoid(torch.einsum("bnid,hd->bnih", x_norm, right_gate_w))
    out_gate = torch.sigmoid(torch.einsum("bnid,hd->bnih", x_norm, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core N³ multiplication:   out = left @ rightᵀ per hidden channel
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) and run the Triton kernel
    left_resh = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_resh = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # fp16

    out_resh = batch_matmul_t(left_resh, right_resh)                    # fp32

    # reshape back to [B,N,N,H]
    out = out_resh.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # --------------------------------------------------------------
    # 6. Final LayerNorm + gating + linear back to dim
    # --------------------------------------------------------------
    w_out_norm = weights["to_out_norm.weight"]
    b_out_norm = weights["to_out_norm.bias"]
    w_out = weights["to_out.weight"]          # [D, H]

    # LayerNorm over hidden dim H
    mean_h = out.mean(dim=-1, keepdim=True)
    var_h = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_h) / torch.sqrt(var_h + 1e-5)
    out_norm = out_norm * w_out_norm + b_out_norm

    # gating (already computed as out_gate, shape [B,N,N,H])
    out_norm = out_norm * out_gate

    # final linear (fp16 -> fp32)
    out_fp16 = out_norm.to(torch.float16)
    final = torch.einsum("bnih,dh->bnid", out_fp16, w_out)   # [B,N,N,D]
    return final.to(torch.float32)
