"""
TriMul “outgoing” kernel (AlphaFold‑3)

We compute
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are the masked, gated projections of the
input tensor.  The heavy N³ reduction is performed in a custom Triton
kernel that tiles the three reduction axes (i, j, k) and the feature
dimension d.  All LayerNorms, linear layers and gating are kept in
torch for simplicity; the kernel works on FP16 tensors and accumulates
in FP32 for numerical stability.  The final output is returned in
float32 as required by the reference implementation.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# ---------------------------------------------------------------
# Triton kernel that performs the 3‑D einsum reduction
# ---------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,           # FP16 left  [B, N, N, D]
    right_ptr,          # FP16 right [B, N, N, D]
    out_ptr,            # FP32 out   [B, N, N, D]

    B, N, D,

    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,

    # Compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_D: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Tile layout:
        (i, j) plane is split into (BLOCK_I, BLOCK_J) blocks.
        The reduction over k is tiled with BLOCK_K.
        Features d are processed in BLOCK_D chunks.
    """
    pid = tl.program_id(0)

    # ---- map 1‑D pid to (b, i_block, j_block) -----------------
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    b = pid // (num_i * num_j)
    tmp = pid % (num_i * num_j)
    i_block = tmp // num_j
    j_block = tmp % num_j

    i_offset = i_block * BLOCK_I
    j_offset = j_block * BLOCK_J

    # ---- allocate accumulator (FP32) ---------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ---- loop over k -------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        k_offset = k_start

        # mask for out‑of‑bounds loads
        i_mask = i_offset + tl.arange(0, BLOCK_I) < N
        j_mask = j_offset + tl.arange(0, BLOCK_J) < N
        k_mask = k_offset + tl.arange(0, BLOCK_K) < N
        d_mask = tl.arange(0, BLOCK_D) < D

        # ----- load left tile: shape (I, K, D) -----------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + (i_offset[:, None, None]) * stride_left_i
            + (k_offset + tl.arange(0, BLOCK_K))[None, :, None] * stride_left_k
            + tl.arange(0, BLOCK_D)[None, None, :] * stride_left_d
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
            other=0.0,
        ).to(tl.float32)          # (I, K, D)

        # ----- load right tile: shape (J, K, D) ----------------------
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + (j_offset[:, None, None]) * stride_right_j
            + (k_offset + tl.arange(0, BLOCK_K))[None, :, None] * stride_right_k
            + tl.arange(0, BLOCK_D)[None, None, :] * stride_right_d
        )
        right = tl.load(
            right_ptrs,
            mask=j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
            other=0.0,
        ).to(tl.float32)          # (J, K, D)

        # ----- compute dot products for each feature d ---------------
        # for each d we need:  out[i,j,d] += Σ_k left[i,k,d] * right[j,k,d]
        # this is a matrix multiplication with the k dimension reduced.
        for d in range(0, BLOCK_D):
            # shape (I, K) and (J, K)
            a = left[:, :, d]
            b = right[:, :, d]
            # tl.dot expects (M, K) and (N, K) with trans_b flag to treat second
            # operand as (K, N).  Setting trans_b=True yields Σ_k a[i,k] * b[j,k].
            acc[:, :, d] += tl.dot(a, b, trans_b=True)

    # ---- store the result ----------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + (i_offset[:, None, None]) * stride_out_i
        + (j_offset + tl.arange(0, BLOCK_J))[None, :, None] * stride_out_j
        + tl.arange(0, BLOCK_D)[None, None, :] * stride_out_d
    )
    tl.store(
        out_ptrs,
        acc,
        mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :],
    )


# ---------------------------------------------------------------
# Host‑side entry point
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   torch.float32
        - mask         : [B, N, N]        torch.float32 (0/1) or None
        - weights      : dict of torch tensors (see docstring above)
        - config       : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm + linear projections + gating (pure torch)
    # ------------------------------------------------------------------
    # layer‑norm on the last dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias

    # linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # apply optional mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # gates (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 2️⃣  Triton kernel for the 3‑D einsum reduction
    # ------------------------------------------------------------------
    # work in FP16 inside the kernel, accumulate in FP32
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp32 = torch.empty((B, N, N, dim), dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left_fp16.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out_fp32.stride()

    # tile sizes – chosen to fit into shared memory on Hopper
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_D = 32
    BLOCK_K = 64

    # grid: one block per (b, i_tile, j_tile)
    grid_i = triton.cdiv(N, BLOCK_I)
    grid_j = triton.cdiv(N, BLOCK_J)
    grid = (B * grid_i * grid_j,)

    trimul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B,
        N,
        dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_d,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_d,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_d,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_D=BLOCK_D,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 3️⃣  Final LayerNorm, gating and projection (torch)
    # ------------------------------------------------------------------
    # to_out_norm
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = (out_fp32 - out_fp32.mean(-1, keepdim=True)) / torch.sqrt(
        out_fp32.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    # multiply by output gate
    out = out * out_gate

    # final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
