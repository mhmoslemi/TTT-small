"""
TriMul forward (outgoing) kernel.

The expensive part is the 3‑D tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]        (d = hidden_dim)
It is computed with a fused Triton kernel that tiles the (i,j) plane and
accumulates over k.  All surrounding linear / layer‑norm / gating
operations are performed with PyTorch (they are cheap compared to the
N³ contraction).  The kernel works for arbitrary batch size (1‑2), sequence
length N (≤1024) and hidden dimension (≤384).  The final result is cast back
to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def trilayer_mul_kernel(
    left_ptr,                # *[B, N, N, H] left tensor
    right_ptr,               # *[B, N, N, H] right tensor
    out_ptr,                 # *[B, N, N, H] output tensor
    B, N, H,                 # static sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,   # tile size over i
    BLOCK_J: tl.constexpr,   # tile size over j
    BLOCK_K: tl.constexpr,   # tile size over k
):
    pid = tl.program_id(axis=0)                     # linear id over (b, i_tile, j_tile)
    # decompose program id
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    b = pid // (num_i * num_j)
    rem = pid % (num_i * num_j)
    i_tile = rem // num_j
    j_tile = rem % num_j

    # offsets within the tile
    offs_i = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)

    # masks for boundary
    mask_i = offs_i < N
    mask_j = offs_j < N

    # allocate accumulator (float32 for precision)
    acc = tl.zeros((BLOCK_I, BLOCK_J, H), dtype=tl.float32)

    # iterate over k dimension
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.arange(0, BLOCK_K) + k_start
        mask_k = cur_k < N

        # ----- load left[b, i, k, h] -----
        # shape: (BLOCK_I, BLOCK_K, H)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            offs_i[:, None, None] * stride_left_i + \
            cur_k[None, :, None] * stride_left_k + \
            tl.arange(0, H)[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0)

        # ----- load right[b, j, k, h] -----
        # shape: (BLOCK_J, BLOCK_K, H)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            offs_j[:, None, None] * stride_right_j + \
            cur_k[None, :, None] * stride_right_k + \
            tl.arange(0, H)[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0)

        # now compute outer product over k for each h
        # left: (I, K, H)  -> (I, K, 1, H)
        # right: (J, K, H) -> (1, J, K, H)
        # we want sum_k left[i,k,h] * right[j,k,h]  for each h
        # using broadcasting and dot over K
        # loop over h to keep it simple (H ≤ 384)
        for h in range(H):
            l = left[:, :, h]          # (I, K)
            r = right[:, :, h]         # (J, K)
            # compute (I, J) = l @ r.T   (K dimension)
            acc[:, :, h] += tl.dot(l, r, allow_tf32=False)

    # ----- store result -----
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        offs_i[:, None, None] * stride_out_i + \
        offs_j[None, :, None] * stride_out_j + \
        tl.arange(0, H)[None, None, :] * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None, None] & mask_j[None, :, None])


# ----------------------------------------------------------------------
# Helper: launch Triton kernel
# ----------------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]   (float16 or float32)
    returns      : [B, N, N, H]   (float32)
    """
    B, N, _, H = left.shape
    # ensure contiguous and half precision for faster compute
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes (tuned for Hopper)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    # grid over (B, i_tile, j_tile)
    grid_i = triton.cdiv(N, BLOCK_I)
    grid_j = triton.cdiv(N, BLOCK_J)
    grid = (B * grid_i * grid_j,)

    tri_layer_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# Full custom_kernel implementation
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]   torch.float32 on CUDA
            mask         : [B, N, N]        torch.float32 (0/1) on CUDA
            weights      : dict of parameter tensors
            config       : dict containing "dim" and "hidden_dim"

    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm on input (dim) -----
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----- Linear projections (no bias) -----
    # shape after proj: [B, N, N, hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Apply mask if present -----
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ----- Gates (sigmoid) -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----- Core N³ contraction via Triton -----
    # Result is [B, N, N, hidden_dim] in float32
    out = tri_mul_triton(left, right)   # already float32

    # ----- Post‑processing -----
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to dim

    return out
