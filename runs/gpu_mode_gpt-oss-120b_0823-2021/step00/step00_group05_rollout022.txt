"""
TriMul "outgoing" kernel (AlphaFold3).

The heavy part of the forward pass is the einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right have shape (B, N, N, H) (H = hidden_dim).  
We implement this reduction in Triton. All other ops
(LayerNorm, Linear, sigmoid gates) are performed with PyTorch
because they are cheap compared with the N³ work.

The Triton kernel iterates over the reduction dimension `k` in
tiles (BLOCK_K). For each output element (b,i,j) it accumulates a
vector of length `H` in FP32 registers and finally writes the result.
The kernel is launched with a 3‑D grid (batch, i, j) and works for
any batch size ≤2, any sequence length up to 1024 and hidden_dim up
to 384 (the compile‑time constant `BLOCK_H` is set to the exact
hidden_dim value for each launch).

The final output is converted back to FP32 after the optional
to_out_norm, out_gate and final linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel for the einsum reduction
# -------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides of left  (B,N,K,H)
    stride_lb, stride_li, stride_lk, stride_ld,
    # strides of right (B,N,K,H)
    stride_rb, stride_ri, stride_rk, stride_rd,
    # strides of out   (B,N,N,H)
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_K: tl.constexpr,  # tile size along reduction dim
    BLOCK_H: tl.constexpr,  # hidden dimension (compile‑time)
):
    pid_b = tl.program_id(0)          # batch axis
    pid_i = tl.program_id(1)          # i axis
    pid_j = tl.program_id(2)          # j axis

    # current i, j indices
    i = pid_i
    j = pid_j

    # offsets for the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_H,), dtype=tl.float32)

    # loop over k in tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_start < N                           # mask for out‑of‑bounds k

        # left[b,i,k,:]   shape (BLOCK_K, BLOCK_H)
        left_off = (
            pid_b * stride_lb +
            i * stride_li +
            k_start * stride_lk +
            offs_h * stride_ld
        )
        left = tl.load(left_ptr + left_off, mask=k_mask[:, None], other=0.0)

        # right[b,j,k,:]  shape (BLOCK_K, BLOCK_H)
        right_off = (
            pid_b * stride_rb +
            j * stride_ri +
            k_start * stride_rk +
            offs_h * stride_rd
        )
        right = tl.load(right_ptr + right_off, mask=k_mask[:, None], other=0.0)

        # element‑wise product and sum over k (axis 0)
        prod = left * right
        acc += tl.sum(prod, axis=0)

    # write the result: out[b,i,j,:]
    out_off = (
        pid_b * stride_ob +
        i * stride_oi +
        j * stride_oj +
        offs_h * stride_od
    )
    tl.store(out_ptr + out_off, acc.to(tl.float32))


# -------------------------------------------------------------
# Host‑side entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a fused Triton kernel
    for the N³ reduction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32 or bool)
            - weights: dict of model parameters (all float32)
            - config: dict containing "dim", "hidden_dim", and optionally "eps"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # -----------------------------------------------------------------
    # Helper: LayerNorm without bias (bias handled manually)
    # -----------------------------------------------------------------
    def layer_norm(x, weight, bias):
        # x: [..., D]
        return F.layer_norm(x, (x.shape[-1],), weight, bias, eps)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    # shape after proj: [B, N, N, H]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Apply mask if present (mask already on device)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4) Gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Triton kernel: out = Σ_k left[...,i,k,:] * right[...,j,k,:]
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # strides (contiguous tensors)
    left_s = left.stride()
    right_s = right.stride()
    out_s = out.stride()

    BLOCK_K = 32                       # tile size for reduction over k
    BLOCK_H = hidden_dim               # compile‑time hidden dimension

    grid = (B, N, N)                    # (batch, i, j)

    _trimul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_lb=left_s[0],
        stride_li=left_s[1],
        stride_lk=left_s[2],
        stride_ld=left_s[3],
        stride_rb=right_s[0],
        stride_ri=right_s[1],
        stride_rk=right_s[2],
        stride_rd=right_s[3],
        stride_ob=out_s[0],
        stride_oi=out_s[1],
        stride_oj=out_s[2],
        stride_od=out_s[3],
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # -----------------------------------------------------------------
    # 6) to_out_norm (LayerNorm on hidden dim)
    # -----------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )

    # -----------------------------------------------------------------
    # 7) Apply out_gate and final linear projection back to dim
    # -----------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(dtype)
