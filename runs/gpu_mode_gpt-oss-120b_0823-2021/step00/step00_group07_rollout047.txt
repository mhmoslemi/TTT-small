"""
TriMul “outgoing” kernel (AlphaFold‑3)

The forward pass consists of:
1. Layer‑norm over the last dimension (dim).
2. Linear projections → left/right tensors of shape [B,N,N,HD].
3. Optional mask and per‑position gating (sigmoid).
4. Core “tri‑mul” operation:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply for every hidden channel d.
   We implement this reduction in a fused Triton kernel (the only part
   written in Triton).  The remaining LayerNorms, gates and final
   Linear/LayerNorm are performed in PyTorch for simplicity.

The kernel processes one (batch, hidden‑channel) pair at a time,
computing a BLOCK_M × BLOCK_N tile of the output matrix.
It uses FP16 for the intermediate multiply‑accumulate and writes
the result back in FP32 (the model works in float32).
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel that computes the tri‑mul reduction for a single
# (batch, hidden_dim) slice.
# --------------------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr,                # [B, N, N, HD]  (i,k) slice
    right_ptr,               # [B, N, N, HD]  (j,k) slice
    out_ptr,                 # [B, N, N, HD]  result
    # sizes
    B, N, HD,
    # strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # block sizes (compile‑time)
    BLOCK_M: tl.constexpr,   # tile size for i dimension
    BLOCK_N: tl.constexpr,   # tile size for j dimension
    BLOCK_K: tl.constexpr,   # tile size for reduction k dimension
):
    # ------------------------------------------------------------------
    # each program works on one (batch, hidden_channel) pair
    # ------------------------------------------------------------------
    pid = tl.program_id(0)                     # index over B*HD
    batch_id = pid // HD
    d_id     = pid %  HD

    # tile offsets in i/j
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # pointers to the beginning of this (b,d) slice
    # ------------------------------------------------------------------
    left_base  = left_ptr \
               + batch_id * stride_left_b \
               + d_id     * stride_left_d
    right_base = right_ptr \
               + batch_id * stride_right_b \
               + d_id     * stride_right_d
    out_base   = out_ptr \
               + batch_id * stride_out_b \
               + d_id     * stride_out_d

    # accumulation buffer in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over k
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: shape [BLOCK_M, BLOCK_K]
        left_ptrs = left_base \
                  + offs_i[:, None] * stride_left_i \
                  + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right: we need (j,k) → transpose to (k,j) for dot product
        # right original layout: (batch, j, k, d)
        right_ptrs = right_base \
                   + offs_j[:, None] * stride_right_j \
                   + offs_k[None, :] * stride_right_k
        # after load we have shape [BLOCK_N, BLOCK_K]; transpose to [BLOCK_K, BLOCK_N]
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        right = tl.trans(right)   # now [BLOCK_K, BLOCK_N]

        # matrix multiply‑accumulate
        acc += tl.dot(left, right)   # (M,K)·(K,N) → (M,N)

    # ------------------------------------------------------------------
    # write result
    # ------------------------------------------------------------------
    out_ptrs = out_base \
             + offs_i[:, None] * stride_out_i \
             + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Helper to launch the kernel
# --------------------------------------------------------------
def launch_trimul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B,N,N,HD]  (float16 or float32)
    returns out   : [B,N,N,HD]  (float32)
    """
    B, N, _, HD = left.shape
    device = left.device

    # ensure fp16 inside kernel for speed
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides (in elements) for the 4‑D tensors
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b,   stride_out_i,   stride_out_j,   stride_out_d   = out.stride()

    # block sizes – tuned for Hopper (SM90) shared‑memory & register pressure
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * HD,                        # program_id(0)
        triton.cdiv(N, BLOCK_M),       # program_id(1)
        triton.cdiv(N, BLOCK_N),       # program_id(2)
    )

    trimul_kernel[grid](
        left_ptr       = left,
        right_ptr      = right,
        out_ptr        = out,
        B=B, N=N, HD=HD,
        stride_left_b=stride_left_b, stride_left_i=stride_left_i,
        stride_left_k=stride_left_k, stride_left_d=stride_left_d,
        stride_right_b=stride_right_b, stride_right_j=stride_right_j,
        stride_right_k=stride_right_k, stride_right_d=stride_right_d,
        stride_out_b=stride_out_b, stride_out_i=stride_out_i,
        stride_out_j=stride_out_j, stride_out_d=stride_out_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,   # good default for H200
    )
    return out


# --------------------------------------------------------------
# Full custom kernel (entry point)
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            input_tensor : [B, N, N, dim]   torch.float32 on CUDA
            mask         : [B, N, N]        torch.float32 (0/1) or bool
            weights_dict : model parameters (see problem statement)
            config_dict  : {"dim": int, "hidden_dim": int, ...}

    Returns:
        output tensor : [B, N, N, dim]   torch.float32
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. LayerNorm over dim
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # PyTorch LayerNorm (fast enough, runs on CUDA)
    norm = nn.LayerNorm(dim, elementwise_affine=True).to(input_tensor.device)
    norm.weight.data.copy_(norm_weight)
    norm.bias.data.copy_(norm_bias)
    x = norm(input_tensor)                     # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = nn.Linear(dim, hidden_dim, bias=False).to(x.device)
    right_proj = nn.Linear(dim, hidden_dim, bias=False).to(x.device)
    left_proj.weight.data.copy_(weights["left_proj.weight"])
    right_proj.weight.data.copy_(weights["right_proj.weight"])

    left = left_proj(x)        # [B,N,N,HD]
    right = right_proj(x)      # [B,N,N,HD]

    # ------------------------------------------------------------------
    # 3. Mask (if present) – broadcast over hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                 # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – also no bias
    # ------------------------------------------------------------------
    left_gate = nn.Linear(dim, hidden_dim, bias=False).to(x.device)
    right_gate = nn.Linear(dim, hidden_dim, bias=False).to(x.device)
    out_gate = nn.Linear(dim, hidden_dim, bias=False).to(x.device)

    left_gate.weight.data.copy_(weights["left_gate.weight"])
    right_gate.weight.data.copy_(weights["right_gate.weight"])
    out_gate.weight.data.copy_(weights["out_gate.weight"])

    left_gate_val = left_gate(x).sigmoid()
    right_gate_val = right_gate(x).sigmoid()
    out_gate_val = out_gate(x).sigmoid()

    left = left * left_gate_val
    right = right * right_gate_val

    # ------------------------------------------------------------------
    # 5. Core tri‑mul (fused Triton kernel)
    # ------------------------------------------------------------------
    out = launch_trimul(left, right)   # [B,N,N,HD]  FP32

    # ------------------------------------------------------------------
    # 6. Final LayerNorm + gating + Linear
    # ------------------------------------------------------------------
    to_out_norm = nn.LayerNorm(hidden_dim, elementwise_affine=True).to(x.device)
    to_out_norm.weight.data.copy_(weights["to_out_norm.weight"])
    to_out_norm.bias.data.copy_(weights["to_out_norm.bias"])

    to_out = nn.Linear(hidden_dim, dim, bias=False).to(x.device)
    to_out.weight.data.copy_(weights["to_out.weight"])

    out = to_out_norm(out)          # [B,N,N,HD]
    out = out * out_gate_val        # broadcast over hidden dim
    out = to_out(out)               # [B,N,N,dim]

    return out
