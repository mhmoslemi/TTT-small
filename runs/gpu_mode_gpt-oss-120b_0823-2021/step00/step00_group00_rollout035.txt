"""
TriMul “outgoing” implementation.

The heavy part of the module is the three‑dimensional einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply over the “k” axis for every
(hidden) channel *d*.  The kernel below computes one output element
(i,j,d) per program‑id and tiles the reduction over *k* in small blocks
(“BLOCK_K”).  All LayerNorm/Linear/Gating operations are kept in PyTorch
(they are cheap compared to the N³ reduction) while the N³ loop is
fused in a single Triton kernel.

The kernel works for any batch size *B*, sequence length *N* and hidden
dimension *H* (called ``hidden_dim`` in the config).  The final output
is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: computes out = Σ_k left[...,k,d] * right[...,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const float*   [B,N,N,H]
    right_ptr,         # const float*   [B,N,N,H]
    out_ptr,           # float*         [B,N,N,H]
    B, N, H,           # int scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # decode pid -> (b,i,j,h)
    # --------------------------------------------------------------
    total = B * N * N * H
    mask = pid < total
    b = pid // (N * N * H)
    rem = pid % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # pointers to the first element of the reduction dimension (k=0)
    left_base = left_ptr + b * stride_left_b + i * stride_left_i + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + j * stride_right_j + h * stride_right_h

    acc = tl.float32(0.0)

    # --------------------------------------------------------------
    # loop over k in tiles
    # --------------------------------------------------------------
    for start_k in range(0, N, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        k_mask = cur_k < N

        # left[b,i,k,h]
        left_off = left_base + cur_k * stride_left_k
        left_val = tl.load(left_off, mask=k_mask, other=0.0).to(tl.float32)

        # right[b,j,k,h]
        right_off = right_base + cur_k * stride_right_k
        right_val = tl.load(right_off, mask=k_mask, other=0.0).to(tl.float32)

        acc += tl.sum(left_val * right_val, axis=0)   # reduction over k‑tile

    # write result
    out_off = (out_ptr + b * stride_out_b
                        + i * stride_out_i
                        + j * stride_out_j
                        + h * stride_out_h)
    tl.store(out_off, acc.to(tl.float32), mask=mask)


# ----------------------------------------------------------------------
# Helper: fused LayerNorm (no bias for simplicity, bias added later)
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: [..., C]; weight/bias: [C]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator using a Triton kernel
    for the N³ reduction.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]    (float32)
            - mask        : [B, N, N]          (float32 or bool)
            - weights     : dict of pre‑loaded parameters
            - config      : {"dim": int, "hidden_dim": int, ...}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on input (dim)
    # ------------------------------------------------------------------
    x = layer_norm(
        inp,
        weights["norm.weight"],
        weights["norm.bias"]
    )                     # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden] = x @ W^T   (W: hidden x dim)
    left = torch.einsum("bijk,hd->bijkd", x, weights["left_proj.weight"]).reshape(B, N, N, hidden)
    right = torch.einsum("bijk,hd->bijkd", x, weights["right_proj.weight"]).reshape(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 3. Apply mask if present
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) + element‑wise scaling
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.einsum("bijk,hd->bijkd", x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.einsum("bijk,hd->bijkd", x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.einsum("bijk,hd->bijkd", x, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel: out = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float32, device=device)   # [B,N,N,hidden]

    # strides (in elements, not bytes)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # launch parameters
    BLOCK_K = 32
    total_elements = B * N * N * hidden
    grid = (triton.cdiv(total_elements, 1),)   # one program per output element

    trimul_outgoing_kernel[grid](
        left_ptr          = left,
        right_ptr         = right,
        out_ptr           = out,
        B                 = B,
        N                 = N,
        H                 = hidden,
        stride_left_b     = s_lb,
        stride_left_i     = s_li,
        stride_left_k     = s_lk,
        stride_left_h     = s_lh,
        stride_right_b    = s_rb,
        stride_right_j    = s_rj,
        stride_right_k    = s_rk,
        stride_right_h    = s_rh,
        stride_out_b      = s_ob,
        stride_out_i      = s_oi,
        stride_out_j      = s_oj,
        stride_out_h      = s_oh,
        BLOCK_K           = BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. to_out_norm + gating + final linear
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"]
    )
    out = out * out_gate

    # final linear back to dim
    out = torch.einsum("bijkd,hd->bijk", out, weights["to_out.weight"])
    return out.to(torch.float32)
