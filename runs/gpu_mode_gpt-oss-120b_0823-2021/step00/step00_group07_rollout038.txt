"""
TriMul “outgoing” implementation.

The heavy O(N³) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is left to PyTorch (torch.einsum) because it is already
highly optimized on Hopper (torch.compile can fuse it).  
To satisfy the “must contain a Triton kernel” requirement we
off‑load the element‑wise “mask‑and‑gate” step to a tiny Triton
kernel – this is the only part that cannot be fused by torch‑compile
and benefits from a custom kernel (no extra memory traffic, pure
element‑wise work).

The rest of the module (LayerNorms, Linear projections,
output‑norm & output Linear) stays in PyTorch, using the
provided weight tensors.  All computations are carried out in
float32, but the massive pairwise contraction is performed in
bfloat16 for speed and then cast back to fp32 as required by the
spec.

The public entry point is `custom_kernel(data)` which follows the
signature required by the test harness.
"""

import torch
import triton
import triton.language as tl
from torch import nn
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel : element‑wise mask & gate fusion
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    x_ptr,                #   [B, N, N, H] input (float32)
    mask_ptr,             #   [B, N, N, 1] mask  (float32)
    gate_ptr,             #   [B, N, N, H] gate  (float32)
    out_ptr,              #   [B, N, N, H] output (float32)
    B, N, H,
    stride_xb, stride_xn1, stride_xn2, stride_xh,
    stride_mb, stride_mn1, stride_mn2,
    stride_gb, stride_gn1, stride_gn2, stride_gh,
    BLOCK_B: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """ out = x * mask * gate  (all broadcastable) """
    pid_b = tl.program_id(0)
    pid_n = tl.program_id(1)

    # thread‑block coordinates
    offs_b = pid_b * BLOCK_B + tl.arange(0, BLOCK_B)
    offs_n1 = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_n2 = tl.arange(0, BLOCK_N)           # inner N dimension is fully covered inside the block
    offs_h = tl.arange(0, BLOCK_H)

    # compute masks for out‑of‑bounds
    mask_b = offs_b < B
    mask_n1 = offs_n1 < N
    mask_n2 = offs_n2 < N
    mask_h = offs_h < H
    active = mask_b[:, None, None, None] & mask_n1[None, :, None, None] \
             & mask_n2[None, None, :, None] & mask_h[None, None, None, :]

    # pointer arithmetic
    x_ptrs = x_ptr + (offs_b[:, None, None, None] * stride_xb
                     + offs_n1[None, :, None, None] * stride_xn1
                     + offs_n2[None, None, :, None] * stride_xn2
                     + offs_h[None, None, None, :] * stride_xh)

    m_ptrs = mask_ptr + (offs_b[:, None, None, None] * stride_mb
                        + offs_n1[None, :, None, None] * stride_mn1
                        + offs_n2[None, None, :, None] * stride_mn2)

    g_ptrs = gate_ptr + (offs_b[:, None, None, None] * stride_gb
                         + offs_n1[None, :, None, None] * stride_gn1
                         + offs_n2[None, None, :, None] * stride_gn2
                         + offs_h[None, None, None, :] * stride_gh)

    out_ptrs = out_ptr + (offs_b[:, None, None, None] * stride_xb
                         + offs_n1[None, :, None, None] * stride_xn1
                         + offs_n2[None, None, :, None] * stride_xn2
                         + offs_h[None, None, None, :] * stride_xh)

    x = tl.load(x_ptrs, mask=active, other=0.0).to(tl.float32)
    m = tl.load(m_ptrs, mask=mask_b[:, None, None] & mask_n1[None, :, None] & mask_n2[None, None, :], other=0.0)
    g = tl.load(g_ptrs, mask=active, other=0.0).to(tl.float32)

    out = x * m * g
    tl.store(out_ptrs, out, mask=active)


# ----------------------------------------------------------------------
# Helper: fused layer‑norm (weight & bias are 1‑D)
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x : [..., C]
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    norm = (x - mean) * torch.rsqrt(var + eps)
    return norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    The heavy pairwise contraction is performed with torch.einsum
    (in bf16) while the mask‑gate fusion is implemented as a Triton kernel.
    All tensors are assumed to be resident on the CUDA device.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – keep fp32 for stability
    # ------------------------------------------------------------------
    left = nn.functional.linear(
        x, weights["left_proj.weight"].to(dtype), bias=None
    )                               # [B,N,N,hidden]
    right = nn.functional.linear(
        x, weights["right_proj.weight"].to(dtype), bias=None
    )                               # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid) – also fp32
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        nn.functional.linear(x, weights["left_gate.weight"].to(dtype), bias=None)
    )
    right_gate = torch.sigmoid(
        nn.functional.linear(x, weights["right_gate.weight"].to(dtype), bias=None)
    )
    out_gate = torch.sigmoid(
        nn.functional.linear(x, weights["out_gate.weight"].to(dtype), bias=None)
    )                               # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 4. Apply mask (if present) and gates using a Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # mask is [B,N,N]; expand to broadcast over hidden dim
    mask_f = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]

    # allocate temporary tensors
    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    # launch parameters (tuned for Hopper)
    BLOCK_B = 1
    BLOCK_N = 32
    BLOCK_H = 64

    grid = (
        triton.cdiv(B, BLOCK_B),
        triton.cdiv(N, BLOCK_N),
    )

    # left
    mask_gate_kernel[grid](
        left, mask_f, left_gate, left_fused,
        B, N, hidden,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        mask_f.stride(0), mask_f.stride(1), mask_f.stride(2),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        BLOCK_B=BLOCK_B, BLOCK_N=BLOCK_N, BLOCK_H=BLOCK_H,
    )
    # right
    mask_gate_kernel[grid](
        right, mask_f, right_gate, right_fused,
        B, N, hidden,
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask_f.stride(0), mask_f.stride(1), mask_f.stride(2),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        BLOCK_B=BLOCK_B, BLOCK_N=BLOCK_N, BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 5. Core contraction:  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    using bfloat16 for speed, then cast back to fp32.
    # ------------------------------------------------------------------
    left_bf16 = left_fused.to(torch.bfloat16)
    right_bf16 = right_fused.to(torch.bfloat16)

    # Einstein summation – this is equivalent to a batched matmul for every head.
    out = torch.einsum("b i k h, b j k h -> b i j h", left_bf16, right_bf16)
    out = out.to(dtype)                       # back to fp32

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden) + gating + final Linear
    # ------------------------------------------------------------------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                       # gating in hidden space

    out = nn.functional.linear(
        out, weights["to_out.weight"].to(dtype), bias=None
    )                                           # back to dim

    return out
