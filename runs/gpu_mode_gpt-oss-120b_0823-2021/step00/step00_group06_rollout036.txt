"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the ternary multiplication

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

for every batch `b`, sequence position `i,j` and hidden channel `h`.
For a fixed channel `h` this is exactly a matrix‑multiply:

    out_h = left_h @ right_hᵀ          (left_h, right_h : [B, N, N])

The kernel below performs a batched mat‑mul on the `(B*hidden_dim)` “matrices”
simultaneously, using a classic Triton tiled algorithm.  All other
operations (LayerNorm, Linear projections, gates, final LayerNorm/Linear)
are kept in PyTorch – they are cheap compared with the O(B·N³·H) work of the
einsum and benefit from the highly‑optimized torch kernels.

The kernel works in FP16 for the inner product (the model uses float32
weights, but the intermediate matrix‑multiply can be done in FP16 without
loss of accuracy) and stores the result back in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply for (B*H) matrices of size N×N
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # matrix dimensions
    M, N, K,
    # strides (in elements)
    stride_am, stride_ak,          # a: [batch_h, i, k]
    stride_bk, stride_bn,          # b: [batch_h, j, k]   (k is the reduction dim)
    stride_cm, stride_cn,          # c: [batch_h, i, j]
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch of matrices.
    A : [B*H, M, K]   (left projection)
    B : [B*H, N, K]   (right projection, we need its transpose)
    C : [B*H, M, N]   (output)
    """
    pid = tl.program_id(0)
    num_batch = tl.cdiv(M * N, GROUP_M * BLOCK_M * BLOCK_N)
    batch_id = pid // num_batch
    block_index = pid % num_batch

    # 2‑D grid within each batch
    block_i = (block_index // tl.cdiv(N, BLOCK_N)) * BLOCK_M
    block_j = (block_index % tl.cdiv(N, BLOCK_N)) * BLOCK_N

    # pointers for this batch
    a_batch_ptr = a_ptr + batch_id * stride_am * M
    b_batch_ptr = b_ptr + batch_id * stride_bn * N
    c_batch_ptr = c_ptr + batch_id * stride_cm * M

    # range of rows / cols handled by this program
    offs_m = block_i + tl.arange(0, BLOCK_M)
    offs_n = block_j + tl.arange(0, BLOCK_N)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A block  (M x K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am
                                + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B block  (N x K) – we need Bᵀ, so we load as (K x N) and later dot
        b_ptrs = b_batch_ptr + (offs_n[:, None] * stride_bn
                                + offs_k[None, :] * stride_bk)
        # note: we load Bᵀ by swapping axes in the address calculation
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # dot‑product over K
        acc += tl.dot(a, b)   # result shape (BLOCK_M, BLOCK_N)

        # advance pointers
        a_batch_ptr += BLOCK_K * stride_ak
        b_batch_ptr += BLOCK_K * stride_bk

    # write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm
                            + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: fused layer‑norm (weight, bias) that works on 4‑D tensors
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: [..., C]
    orig_shape = x.shape
    x_flat = x.view(-1, x.shape[-1])
    mean = x_flat.mean(dim=-1, keepdim=True)
    var = x_flat.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x_flat - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * weight + bias
    return x_norm.view(orig_shape)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, C]  (float32)
        - mask         : [B, N, N]    (float32 or bool)
        - weights      : dict of model parameters
        - config       : dict with keys "dim", "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, C = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (C -> C)
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B,N,N,C]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) : dim -> hidden
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden]
    left = F.linear(x, weights["left_proj.weight"])          # weight shape [hidden, C]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if present).  The mask is broadcast over the hidden dim.
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)            # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) and gating of left/right
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul: out = Σ_k left[...,i,k,h] * right[...,j,k,h]
    #    Implemented as batched mat‑mul on (B*hidden) matrices of size N×N.
    # ------------------------------------------------------------------
    # reshape to [B*hidden, N, N] for the kernel
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # [BH, N, N]
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N) # [BH, N, N]

    # allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)   # [BH, N, N]

    # strides (in elements) for the kernel
    stride_am = left_flat.stride(1)          # stride between rows (i)
    stride_ak = left_flat.stride(2)          # stride between k
    stride_bk = right_flat.stride(1)         # note: we treat right as transposed, so we stride over j
    stride_bn = right_flat.stride(2)         # stride between k
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # tile sizes – tuned for Hopper (sm_90).  64×64 tiles give good occupancy.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    GROUP_M = 8   # grouping for better L2 cache reuse

    total_programs = B * hidden * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N)
    grid = (total_programs,)

    batched_matmul_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_M=GROUP_M,
    )

    # reshape back to [B,N,N,hidden]
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden -> hidden) and final linear projection
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate          # gating after norm
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,C]

    return out.to(torch.float32)
