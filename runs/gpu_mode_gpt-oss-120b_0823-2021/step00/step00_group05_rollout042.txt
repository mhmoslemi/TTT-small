"""
Trimer Multiplicative Update – outgoing version

The heavy part of the module is the 4‑D “einsum”

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiply of the (i,k) slice with the
transpose of the (j,k) slice for every hidden dimension *d*.
The kernel below tiles the i‑ and j‑axes (BLOCK_M × BLOCK_N) and
accumulates over k in tiles of size BLOCK_K.  All arithmetic is
performed in fp32 and the result is stored back as fp16 (the
subsequent LayerNorm + linear layers convert back to fp32).

Only the einsum is executed in Triton; the remaining linear
projections, LayerNorms and sigmoids are cheap and stay in PyTorch.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------
# Triton kernel: out = Σ_k left[...,i,k,d] * right[...,j,k,d]
# ---------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,        # pointers
    B, N, H,                             # dimensions
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_bn = tl.program_id(0)          # combined batch & hidden
    pid_i  = tl.program_id(1)          # block row   (i)
    pid_j  = tl.program_id(2)          # block col   (j)

    # decode batch and hidden dim from the first grid dimension
    b = pid_bn // H
    h = pid_bn % H

    # offsets for the i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- load left[b,i,k,d] -------
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + h * stride_ld
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # (BLOCK_M, BLOCK_K) fp32

        # ------- load right[b,j,k,d] as (k,j) -------
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + h * stride_rd
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_rj
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )  # (BLOCK_K, BLOCK_N) fp32

        # matrix multiply over the k‑dimension
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # write result
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + h * stride_od
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ---------------------------------------------------------------
# Host‑side entry point
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]       (float32 or bool)
        - weights      : dict of torch.Tensor parameters
        - config       : dict with keys "dim", "hidden_dim"
    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ---------- LayerNorm on input ----------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                       # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gating ----------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ---------- Triton einsum ----------
    # allocate output buffer (fp16 for reduced memory traffic)
    out = torch.empty((B, N, N, hidden), dtype=torch.float16, device=device)

    # strides for the three tensors (in elements, not bytes)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # tile sizes – tuned for Hopper (sm_90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(B * hidden, 1),               # we will fold batch*hidden into pid 0
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_ld=stride_ld,
        stride_rb=stride_rb,
        stride_rj=stride_rj,
        stride_rk=stride_rk,
        stride_rd=stride_rd,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_od=stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    out = out.to(torch.float32)   # back to fp32 for the remaining ops

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(out, (hidden,), weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
