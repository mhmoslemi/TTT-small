"""
TriMul “outgoing” implementation.

We keep the cheap element‑wise parts (LayerNorm, Linear, sigmoid gates) in PyTorch
and only accelerate the cubic‑cost tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

by a fused batched GEMM per hidden dimension `d`.  After the contraction we
apply a second LayerNorm, a gate and a final linear projection – also in
PyTorch – to obtain the final 4‑D result in float32.

The kernel works for any batch size ≤2, any sequence length N that is a
multiple of the block size (we pad with a mask), and any hidden dimension
`hidden_dim`.  All tensors are expected on the CUDA device.
"""

import torch
import triton
import triton.language as tl
from torch import nn

# --------------------------------------------------------------
# Triton kernel: batched GEMM over the “k” dimension for each
# (batch, hidden) pair.  The kernel computes
#   C[b, d, i, j] = Σ_k A[b, d, i, k] * B[b, d, k, j]
# where A = left   (shape B,N,N,HD)   and
#       B = right_t (right transposed on dims 1↔2, same shape)
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_gemm_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # tensor sizes
    B, N, D,
    # strides (in elements) for the 4‑D tensors laid out as
    # (batch, i, k, d) for left and (batch, k, j, d) for right_t
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_k, stride_right_j, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # block size constexprs
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr):
    """
    A very standard tiled GEMM with an extra outer dimension that groups
    (batch, hidden) pairs into a single 1‑D program id.
    """

    # ----------------------------------------------------------
    # 1) identify (batch, hidden) pair and 2‑D tile (i,j)
    # ----------------------------------------------------------
    pid = tl.program_id(0)
    # (batch, hidden) is the outermost dimension of the launch grid
    num_pairs = B * D
    pair_id = pid // (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_in_pair = pid % (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))

    b = pair_id // D
    d = pair_id % D

    # 2‑D tile indices (i_tile, j_tile) using a simple row‑major mapping
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    i_tile = pid_in_pair // num_tiles_n
    j_tile = pid_in_pair % num_tiles_n

    # ----------------------------------------------------------
    # 2) compute offsets for the blocks we will load/store
    # ----------------------------------------------------------
    offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds (when N is not a multiple of the block size)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointer base for this (b,d) pair
    left_base  = left_ptr  + b * stride_left_b  + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_b   + d * stride_out_d

    # ------------------------------------------------------------------
    # 3) accumulate over k
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left:   (i,k)
        a_ptrs = left_base + \
                 offs_i[:, None] * stride_left_i + \
                 offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right_t: (k,j)  (right was originally (j,k) before transpose)
        b_ptrs = right_base + \
                 offs_k[:, None] * stride_right_k + \
                 offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot‑product over the k‑dimension
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # 4) write back
    # ------------------------------------------------------------------
    out_ptrs = out_base + \
               offs_i[:, None] * stride_out_i + \
               offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Host‑side helper that launches the kernel
# --------------------------------------------------------------
def trimul_outgoing_gemm(left: torch.Tensor, right_t: torch.Tensor) -> torch.Tensor:
    """
    left, right_t : [B, N, N, D]  (float32 or float16)
    returns out    : [B, N, N, D]  (float32)
    """
    assert left.is_contiguous() and right_t.is_contiguous()
    B, N, _, D = left.shape
    out = torch.empty_like(left, dtype=torch.float16)

    # strides in number of elements (torch strides are in bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_k, stride_right_j, stride_right_d = right_t.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # block sizes – tuned for Hopper (sm_90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # launch grid: one program per (batch, hidden) pair * tile grid
    num_tiles_m = triton.cdiv(N, BLOCK_M)
    num_tiles_n = triton.cdiv(N, BLOCK_N)
    total_tiles = num_tiles_m * num_tiles_n
    grid = (B * D * total_tiles,)

    trimul_outgoing_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right_t,
        out_ptr=out,
        B=B,
        N=N,
        D=D,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_d=stride_left_d,
        stride_right_b=stride_right_b,
        stride_right_k=stride_right_k,
        stride_right_j=stride_right_j,
        stride_right_d=stride_right_d,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_d=stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=8,          # not used – placeholder for future tiling
    )

    return out.to(torch.float32)


# --------------------------------------------------------------
# Main entry point required by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    data = (input_tensor, mask, weights, config)

    Returns
    -------
    out : torch.Tensor
        Shape [bs, seq_len, seq_len, dim], dtype float32, on the same device.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------- 1. LayerNorm on the input ----------
    # Manual LayerNorm to avoid extra Python overhead; we use the provided
    # weight / bias tensors.
    eps = 1e-5
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)

    mean = input_tensor.mean(-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x_norm * norm_w + norm_b          # [B,N,N,dim]

    # ---------- 2. Linear projections (no bias) ----------
    # shape: (..., dim) @ (dim, hidden_dim)
    def linear_no_bias(x, w):
        # x: [B,N,N,dim]  w: [dim, hidden_dim]
        return torch.matmul(x, w.t())

    left = linear_no_bias(x, weights["left_proj.weight"].to(device))
    right = linear_no_bias(x, weights["right_proj.weight"].to(device))

    # ---------- 3. Apply mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- 4. Gating ----------
    left_gate = torch.sigmoid(
        linear_no_bias(x, weights["left_gate.weight"].to(device))
    )
    right_gate = torch.sigmoid(
        linear_no_bias(x, weights["right_gate.weight"].to(device))
    )
    out_gate = torch.sigmoid(
        linear_no_bias(x, weights["out_gate.weight"].to(device))
    )
    left = left * left_gate
    right = right * right_gate

    # ---------- 5. Core contraction (outgoing) ----------
    # right needs to be transposed on the second and third axes
    right_t = right.transpose(1, 2).contiguous()
    # Cast to fp16 for the heavy GEMM, then cast back to fp32
    left_h = left.to(torch.float16)
    right_h = right_t.to(torch.float16)
    out = trimul_outgoing_gemm(left_h, right_h)   # [B,N,N,hidden_dim] float32

    # ---------- 6. Second LayerNorm ----------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    mean2 = out.mean(-1, keepdim=True)
    var2 = ((out - mean2) ** 2).mean(-1, keepdim=True)
    out = (out - mean2) * torch.rsqrt(var2 + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ---------- 7. Apply output gate ----------
    out = out * out_gate

    # ---------- 8. Final linear ----------
    out = torch.matmul(out, weights["to_out.weight"].t().to(device))

    return out.to(torch.float32)
