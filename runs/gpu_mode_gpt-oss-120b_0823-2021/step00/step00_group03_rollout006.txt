"""
TriMul “outgoing” implementation.

The heavy part of the module is the pairwise multiplication
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]           (1)

For every channel *d* this is a matrix multiplication
            out_d = left_d @ right_d.T
where left_d, right_d are (N,N) matrices.  The kernel below
computes (1) for all batches, all channels and all i‑j blocks.
All LayerNorms, linear projections and sigmoid gates are kept in
plain PyTorch (they are cheap compared to the O(N³) work).  The
kernel works with fp16 / bf16 inputs and accumulates in fp32,
producing a fp32 output that matches the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel:   out = Σ_k left[...,i,k,d] * right[...,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # * left  [B,N,N,D]
    right_ptr,              # * right [B,N,N,D]
    out_ptr,                # * out   [B,N,N,D]

    B, N, D,                # static sizes
    stride_lb, stride_lk, stride_ld,               # left strides
    stride_rb, stride_rk, stride_rd,               # right strides
    stride_ob, stride_oi, stride_oj, stride_od,    # out strides

    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Grid: (ceil(N/BLOCK_I), ceil(N/BLOCK_J), B)
    """
    pid_i = tl.program_id(0)          # i‑block
    pid_j = tl.program_id(1)          # j‑block
    pid_b = tl.program_id(2)          # batch

    # coordinate ranges for this program
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    d = tl.arange(0, BLOCK_D)

    mask_i = i < N
    mask_j = j < N

    # accumulator: (BLOCK_I, BLOCK_J, BLOCK_D) in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k_off in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ----- load a tile of left : (i,k,d) -------------------------
        left_tile = tl.load(
            left_ptr
            + pid_b * stride_lb
            + i[:, None, None] * stride_lb
            + k[None, :, None] * stride_lk
            + d[None, None, :] * stride_ld,
            mask=(mask_i[:, None, None] & mask_k[None, :, None]),
            other=0.0,
        )  # fp16 / bf16

        # ----- load a tile of right : (j,k,d) ------------------------
        # we need right transposed on the k‑dimension for the dot product
        right_tile = tl.load(
            right_ptr
            + pid_b * stride_rb
            + j[:, None, None] * stride_oj   # note: stride_oj is the stride of the j index
            + k[None, :, None] * stride_rk
            + d[None, None, :] * stride_rd,
            mask=(mask_j[:, None, None] & mask_k[None, :, None]),
            other=0.0,
        )  # fp16 / bf16

        # now compute for every channel d:
        #   acc[:,:,d] += left_tile[:,:,d] @ right_tile[:,:,d].T
        # we implement this as a series of tl.dot calls.
        for di in range(0, BLOCK_D):
            left_mat = left_tile[:, :, di]          # (BLOCK_I, BLOCK_K)
            right_mat = right_tile[:, :, di]        # (BLOCK_J, BLOCK_K)

            # dot expects (M,K) @ (K,N).  right_mat is (BLOCK_J, BLOCK_K);
            # we therefore transpose it on‑the‑fly by swapping strides.
            # triton dot can take a transposed operand via the `trans` kw.
            acc[:, :, di] += tl.dot(left_mat, right_mat, trans_b=True)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_ptr_base = (
        pid_b * stride_ob
        + i[:, None, None] * stride_oi
        + j[None, :, None] * stride_oj
        + d[None, None, :] * stride_od
    )
    tl.store(
        out_ptr + out_ptr_base,
        acc.to(tl.float32),
        mask=(mask_i[:, None, None] & mask_j[None, :, None]),
    )


# ----------------------------------------------------------------------
# Host‑side function
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    Only the O(N³) multiplication (einsum) is executed in Triton.
    All other linear / norm / gate operations are performed with PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    hidden = config["hidden_dim"]

    # ---------- LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias

    # ---------- Linear projections ----------
    # Shapes: (C, hidden)
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = torch.einsum("...c,cd->...d", x, left_proj_w)          # (B,N,N,hidden)
    right = torch.einsum("...c,cd->...d", x, right_proj_w)

    # ---------- Mask ----------
    if mask is not None:
        mask_ext = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_ext
        right = right * mask_ext

    # ---------- Gates ----------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.einsum("...c,cd->...d", x, left_gate_w))
    right_gate = torch.sigmoid(torch.einsum("...c,cd->...d", x, right_gate_w))
    out_gate = torch.sigmoid(torch.einsum("...c,cd->...d", x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------- Triton multiplication ------------------------
    # Cast to bf16/fp16 for the kernel (faster on H200)
    left_k = left.to(torch.bfloat16)
    right_k = right.to(torch.bfloat16)

    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides (in elements) for the tensors
    stride_lb = left_k.stride(0)
    stride_lk = left_k.stride(2)
    stride_ld = left_k.stride(3)

    stride_rb = right_k.stride(0)
    stride_rk = right_k.stride(2)
    stride_rd = right_k.stride(3)

    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_od = out.stride(3)

    # Tuning parameters – good trade‑off on Hopper
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    BLOCK_D = 32

    grid = (
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        B,
    )
    trimul_outgoing_kernel[grid](
        left_k,
        right_k,
        out,
        B,
        N,
        hidden,
        stride_lb,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ---------- Final norm, gating and projection ----------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # LayerNorm on hidden dim
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate          # gating
    out = torch.einsum("...d,df->...f", out, to_out_w)   # final linear
    return out.to(torch.float32)
