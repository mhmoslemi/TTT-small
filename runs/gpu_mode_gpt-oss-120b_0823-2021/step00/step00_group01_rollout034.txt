"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a Triton kernel for the
N³ pairwise multiplication.

Algorithm
---------
1. Layer‑norm over the last dimension (done in PyTorch – cheap).
2. Linear projections + sigmoid gates (still in PyTorch, they are 2‑D mat‑vec
   ops and are bandwidth bound, not compute bound).
3. **Core**:   out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
   This is a batched “outer‑product‑sum” over the sequence dimension *k*.
   We implement it as a tiled kernel:
   * each program processes a tile (i0…i0+BM‑1, j0…j0+BN‑1)
   * the hidden dimension *h* is split into chunks of size *BH* (≤32)
   * for every k‑tile we load a (BM, Kt, BH) block of *left* and a
     (BN, Kt, BH) block of *right*, multiply, and reduce over the K‑tile.
   * accumulation is performed in FP32 and the final tile is stored back.

The kernel works for any batch size B (≤2 in the tests) and any
sequence length N that is a multiple of the tile size (the kernel
handles the remainder with a mask).  Hidden dimension *H* is assumed to be
a multiple of BH (we pad the last tile with zeros if needed).

The final output is passed through a LayerNorm + linear projection (again
in PyTorch) and returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: tiled outer‑product‑sum over the sequence dimension
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # sizes
    B, N, H,
    # strides (flattened 1‑D tensor view)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # tile sizes (compile‑time constants)
    BM: tl.constexpr, BN: tl.constexpr, BK: tl.constexpr, BH: tl.constexpr,
):
    pid = tl.program_id(axis=0)               # linear id over (i_tile, j_tile, batch)
    # decompose pid
    total_tiles_i = tl.cdiv(N, BM)
    total_tiles_j = tl.cdiv(N, BN)
    b = pid // (total_tiles_i * total_tiles_j)
    rem = pid % (total_tiles_i * total_tiles_j)
    i_tile = rem // total_tiles_j
    j_tile = rem % total_tiles_j

    # start indices of the tile
    i0 = i_tile * BM
    j0 = j_tile * BN

    # offsets within the tile
    offs_i = i0 + tl.arange(0, BM)
    offs_j = j0 + tl.arange(0, BN)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ----------------------------------------------------------------------------
    # Accumulator: (BM, BN, BH)  – we compute the hidden dim in chunks of BH
    # ----------------------------------------------------------------------------
    acc = tl.zeros((BM, BN, BH), dtype=tl.float32)

    # iterate over k‑tiles
    num_k_tiles = tl.cdiv(N, BK)
    for k_tile in range(num_k_tiles):
        k0 = k_tile * BK
        offs_k = k0 + tl.arange(0, BK)
        mask_k = offs_k < N

        # --------------------------------------------------------------------
        # Load left  : shape (BM, BK, BH)
        # --------------------------------------------------------------------
        left_ptrs = left_ptr \
            + b * stride_left_b \
            + offs_i[:, None, None] * stride_left_i \
            + offs_k[None, :, None] * stride_left_k \
            + tl.arange(0, BH)[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0)

        # --------------------------------------------------------------------
        # Load right : shape (BN, BK, BH)
        # --------------------------------------------------------------------
        right_ptrs = right_ptr \
            + b * stride_right_b \
            + offs_j[:, None, None] * stride_right_j \
            + offs_k[None, :, None] * stride_right_k \
            + tl.arange(0, BH)[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0)

        # --------------------------------------------------------------------
        # Multiply & reduce over k
        #   acc[i,j,h] += Σₖ left[i,k,h] * right[j,k,h]
        # --------------------------------------------------------------------
        # expand dimensions to broadcast the k‑axis
        left_exp = left[:, :, None, :]          # (BM, BK, 1, BH)
        right_exp = right[None, :, :, :]        # (1, BN, BK, BH)
        prod = left_exp * right_exp             # (BM, BN, BK, BH)
        acc += tl.sum(prod, axis=2)             # sum over k‑tile → (BM, BN, BH)

    # ------------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------------
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + offs_i[:, None, None] * stride_out_i \
        + offs_j[None, :, None] * stride_out_j \
        + tl.arange(0, BH)[None, None, :] * stride_out_h
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None])

# ----------------------------------------------------------------------
# Helper to launch the Triton kernel
# ----------------------------------------------------------------------
def launch_triton_trimul(
    left: torch.Tensor,          # [B,N,N,H]  FP32
    right: torch.Tensor,         # [B,N,N,H]  FP32
) -> torch.Tensor:
    """Run the tiled outer‑product‑sum kernel and return the result tensor."""
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes) – required by Triton
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – tuned for Hopper (SM90)  (BM*BN*BH fits into 64 KB shared memory)
    BM, BN, BK, BH = 64, 64, 32, 32

    total_tiles_i = (N + BM - 1) // BM
    total_tiles_j = (N + BN - 1) // BN
    total_programs = B * total_tiles_i * total_tiles_j
    grid = (total_programs,)

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BM=BM, BN=BN, BK=BK, BH=BH,
        num_warps=8,             # enough warps for the compute intensity
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch for the cheap parts, Triton for the heavy part)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args
    ----
    data : (input, mask, weights, config)
        input  – [B, N, N, dim]  (float32)
        mask   – [B, N, N]      (float32 or bool)
        weights – dict with all linear / layer‑norm parameters
        config  – contains "dim" and "hidden_dim"

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (in‑place to avoid extra alloc)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_w + norm_b

    # ------------------------------------------------------------------
    # 2. Linear projections + sigmoid gates (plain PyTorch – cheap)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # mask handling
    if mask is not None:
        mask = mask.unsqueeze(-1)                # [B,N,N,1]
        left = left * mask
        right = right * mask

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core N³ multiplication – Triton kernel
    # ------------------------------------------------------------------
    # Cast to float32 (the kernel works in FP32)
    left = left.to(torch.float32)
    right = right.to(torch.float32)

    out = launch_triton_trimul(left, right)    # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4. Final LayerNorm + output gate + linear projection (PyTorch)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to original dim

    return out.to(torch.float32)
