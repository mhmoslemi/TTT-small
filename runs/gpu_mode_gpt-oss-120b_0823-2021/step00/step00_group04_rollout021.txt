"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed batch `b` and hidden channel `d` this is exactly a matrix‑matrix
product

    L = left[b,:,:,d]          # (N,N)
    R = right[b,:,:,d]         # (N,N)
    out[b,:,:,d] = L @ R.T      # (N,N)

We therefore launch a Triton kernel that computes many independent mat‑muls
simultaneously: one mat‑mul for every (batch, hidden) pair.  The kernel
operates on fp16 inputs, accumulates in fp32, and writes fp32 results.
All LayerNorm / Linear / gating operations are kept in PyTorch – they are
cheap compared to the N³ reduction and benefit from the highly‑optimised
PyTorch kernels.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel – block‑wise matmul for a single hidden channel
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # problem sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,   # left strides
    stride_right_b, stride_right_j, stride_right_k, stride_right_h, # right strides
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,       # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HIDDEN: tl.constexpr,                  # hidden dim (constant per launch)
):
    """
    One program instance works on a single (batch, hidden) pair.
    It computes a BLOCK_M × BLOCK_N tile of the N×N output matrix.
    """
    pid = tl.program_id(0)                     # combined batch*hidden index
    b = pid // HIDDEN
    h = pid % HIDDEN

    # tile coordinates
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the start of the tile in each tensor
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        offs_m[:, None] * stride_out_i + \
        offs_n[None, :] * stride_out_j + \
        h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in tiles
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k_start, BLOCK_K)
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # load left [i,k] and right [j,k] (note: right is NOT transposed,
        # we will dot with left which yields L @ R.T)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            offs_m[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k + \
            h * stride_left_h
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            offs_n[:, None] * stride_right_j + \
            offs_k[None, :] * stride_right_k + \
            h * stride_right_h

        left = tl.load(left_ptrs, mask=mask_m[:, None] & (offs_k < N), other=0.0).to(tl.float16)
        right = tl.load(right_ptrs, mask=mask_n[:, None] & (offs_k < N), other=0.0).to(tl.float16)

        # dot over k (the inner dimension)
        acc += tl.dot(left, right, trans_b=True)   # (M,K)·(N,K)ᵀ → (M,N)

    # store result
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper – layer‑norm without bias (bias handled separately)
# ----------------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]; weight/bias: [C]
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    norm = (x - mean) * torch.rsqrt(var + eps)
    return norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   (float32 on CUDA)
            - mask         : [B, N, N] (bool/float, already on CUDA)
            - weights      : dict of parameters (float32)
            - config       : dict with keys "dim", "hidden_dim"

    Returns:
        out : torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    ##################################################################
    # 1. LayerNorm + linear projections (pure PyTorch)
    ##################################################################
    # LayerNorm over the last dimension (dim)
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Linear projections (no bias)
    # shape after proj: [B, N, N, hidden]
    left = F.linear(x, weights["left_proj.weight"])          # (dim, hidden) transposed inside F.linear
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if mask is not None:
        mask3 = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask3
        right = right * mask3

    # Gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    ##################################################################
    # 2. Triton kernel for the einsum (the N³ part)
    ##################################################################
    # Cast to fp16 for the kernel (precision is recovered by fp32 accumulation)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=input_tensor.device)

    # stride information (in elements, not bytes)
    s_lb, s_li, s_lk, s_lh = left_fp16.stride()
    s_rb, s_rj, s_rk, s_rh = right_fp16.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Triton launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                     # program_id(0) – batch * hidden
        triton.cdiv(N, BLOCK_M),       # program_id(1) – i‑tile
        triton.cdiv(N, BLOCK_N),       # program_id(2) – j‑tile
    )
    _trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B, N, hidden,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN=hidden,
    )

    ##################################################################
    # 3. Final post‑processing (PyTorch)
    ##################################################################
    # to_out_norm (LayerNorm over hidden)
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # multiply by out gate (broadcast over hidden)
    out = out * out_gate
    # final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
