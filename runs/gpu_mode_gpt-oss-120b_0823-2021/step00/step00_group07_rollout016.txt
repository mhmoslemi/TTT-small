"""
TriMul “outgoing” kernel (AlphaFold‑3).

The heavy part of the forward pass is the three‑tensor “einsum”

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

with shapes   left/right : (B,N,N,H)   (H = hidden_dim).
We keep the surrounding LayerNorm / Linear / gating logic in PyTorch
(because they are cheap) and fuse the N³ multiplication‑accumulation
into a single Triton kernel.

The kernel processes a tile of (i,j) for a single batch element.
For each output element (i,j) we keep an accumulator per hidden
dimension (H) in FP32, load a BLOCK_K slice of `k`,
multiply element‑wise across H and reduce with a dot‑product.
The kernel works for any N up to 1024, any hidden_dim that fits in
registers (the tests use ≤ 128) and both masked / un‑masked inputs.

The final output is cast back to FP32 (the kernel already works in FP32)
and the usual “to_out” linear + LayerNorm are applied in PyTorch.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel : N³ fused multiply‑add
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    # pointers
    left_ptr,          #  *[B, N, N, H]
    right_ptr,         #  *[B, N, N, H]
    out_ptr,           #  *[B, N, N, H]
    mask_ptr,          #  *[B, N, N]   (optional, can be NULL)
    # strides
    stride_lb, stride_lk, stride_li, stride_ld,   # left  B,N,K,H strides
    stride_rb, stride_rk, stride_rj, stride_rd,   # right B,N,J,H strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out   B,N,J,H strides
    stride_mb, stride_mi, stride_mj,              # mask  B,N,J strides
    # sizes
    B, N, H,
    # compile‑time constants
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, MASKED: tl.constexpr,
):
    """
    Compute a tile (BLOCK_I × BLOCK_J) of the output for a single batch.
    """
    pid = tl.program_id(0)                     #  pid = b * num_i * num_j + i_tile * num_j + j_tile
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)

    b = pid // (num_i * num_j)
    tmp = pid % (num_i * num_j)
    i_tile = tmp // num_j
    j_tile = tmp % num_j

    # block start indices
    i_start = i_tile * BLOCK_I
    j_start = j_tile * BLOCK_J

    # offsets within the tile
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)

    # mask for out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ----- accumulator (FP32) -----
    acc = tl.zeros((BLOCK_I, BLOCK_J, H), dtype=tl.float32)

    # ----- loop over K -----
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # load left : shape (BLOCK_I, BLOCK_K, H)
        left_ptrs = left_ptr + \
            b * stride_lb + \
            (offs_i[:, None, None] * stride_li) + \
            (cur_k[None, :, None] * stride_lk) + \
            (tl.arange(0, H)[None, None, :] * stride_ld)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :] , other=0.0)

        # load right : shape (BLOCK_J, BLOCK_K, H)
        right_ptrs = right_ptr + \
            b * stride_rb + \
            (tl.arange(0, BLOCK_J)[None, :, None] + j_start) * stride_rj + \
            (cur_k[:, None] * stride_rk) + \
            (tl.arange(0, H)[None, None, :] * stride_rd)
        right = tl.load(right_ptrs,
                         mask=mask_j[None, :] & mask_k[:, None],
                         other=0.0)

        # optional mask applied to left/right before multiplication
        if MASKED:
            # mask shape (B,N,N) – broadcast over H
            mask_ptrs = mask_ptr + \
                b * stride_mb + \
                (offs_i[:, None] * stride_mi) + \
                (cur_k[None, :] * stride_mj)
            lmask = tl.load(mask_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
            # left already masked by element‑wise multiplication
            left = left * lmask[..., None]

            mask_ptrs = mask_ptr + \
                b * stride_mb + \
                (offs_j[:, None] * stride_mi) + \
                (cur_k[None, :] * stride_mj)
            rmask = tl.load(mask_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
            right = right * rmask[..., None]

        # ---- compute dot‑product over K for each hidden dimension ----
        # we unroll over H (≤128) – still fast because it stays in registers
        for h in range(0, H):
            # (BLOCK_I, BLOCK_K) * (BLOCK_J, BLOCK_K)^T -> (BLOCK_I, BLOCK_J)
            # right is stored as (BLOCK_J, BLOCK_K, H); we need the K‑dim as dot dim
            l = left[:, :, h]          # (BI, BK)
            r = right[:, :, h]         # (BJ, BK)
            # transpose r to (BK, BJ) for dot
            r_t = tl.trans(r, (0, 1))   # (BK, BJ)
            acc[:, :, h] += tl.dot(l, r_t)   # (BI, BJ)

    # ----- write back -----
    out_ptrs = out_ptr + \
        b * stride_ob + \
        (offs_i[:, None, None] * stride_oi) + \
        (offs_j[None, :, None] * stride_oj) + \
        (tl.arange(0, H)[None, None, :] * stride_od)

    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None])


# ----------------------------------------------------------------------
# Helper – launch wrapper
# ----------------------------------------------------------------------
def launch_trimul_einsum(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor | None,
    *,
    BLOCK_I=64,
    BLOCK_J=64,
    BLOCK_K=32,
) -> torch.Tensor:
    """
    left / right : [B,N,N,H]  (float32)
    mask         : [B,N,N]   (float32) or None
    Returns out  : [B,N,N,H]
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides for each tensor (contiguous layout)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()
    if mask is None:
        mask_ptr = torch.tensor([0], device=left.device, dtype=left.dtype)  # dummy
        stride_mb = stride_mi = stride_mj = 0
        MASKED = 0
    else:
        stride_mb, stride_mi, stride_mj = mask.stride()
        mask_ptr = mask
        MASKED = 1

    # launch grid: one program per (b, i_tile, j_tile)
    num_i = (N + BLOCK_I - 1) // BLOCK_I
    num_j = (N + BLOCK_J - 1) // BLOCK_J
    grid = (B * num_i * num_j,)

    trimul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        mask_ptr=mask_ptr,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_ld=stride_ld,
        stride_rb=stride_rb,
        stride_rj=stride_rj,
        stride_rk=stride_rk,
        stride_rd=stride_rd,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_od=stride_od,
        stride_mb=stride_mb,
        stride_mi=stride_mi,
        stride_mj=stride_mj,
        B=B,
        N=N,
        H=H,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        MASKED=MASKED,
    )
    return out


# ----------------------------------------------------------------------
# Full custom kernel (the entry point required by the grader)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward‑only implementation of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = torch.float32

    # --------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias   = weights["norm.bias"].to(dtype)
    eps = 1e-5
    # fused layernorm: (x - μ) / sqrt(var + eps) * w + b
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias    # shape [B,N,N,dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) – keep in FP32 for stability
    # --------------------------------------------------------------
    def linear_no_bias(x, w):
        # x : [..., dim]  ; w : [hidden_dim, dim]
        return torch.nn.functional.linear(x, w)          # bias=0

    left_proj_weight = weights["left_proj.weight"].to(dtype)     # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"].to(dtype)

    left = linear_no_bias(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = linear_no_bias(x_norm, right_proj_weight)

    # --------------------------------------------------------------
    # 3. Apply optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(dtype)      # [B,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) – also FP32
    # --------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"].to(dtype)
    right_gate_weight = weights["right_gate.weight"].to(dtype)
    out_gate_weight = weights["out_gate.weight"].to(dtype)

    left_gate = torch.sigmoid(linear_no_bias(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(linear_no_bias(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(linear_no_bias(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core cubic multiplication – Triton kernel
    # --------------------------------------------------------------
    # Shapes: [B,N,N,hidden_dim]
    out = launch_trimul_einsum(left, right, mask if mask is not None else None,
                               BLOCK_I=64, BLOCK_J=64, BLOCK_K=32)

    # --------------------------------------------------------------
    # 6. Post‑processing: LayerNorm → gating → final linear
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(dtype)

    # layernorm on hidden_dim
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # gating
    out_norm = out_norm * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"].to(dtype)   # [dim, hidden_dim]
    out_final = torch.nn.functional.linear(out_norm, to_out_weight)  # bias=0

    return out_final.to(dtype)
