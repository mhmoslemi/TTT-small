"""
TriMul "outgoing" implementation.

The heavy part of the forward pass is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed channel *d* this is exactly a matrix multiplication
    L_d (seq_len×seq_len)  @  R_d.T (seq_len×seq_len)
The kernel below performs a batched GEMM over the channel dimension
using Triton (fp16 accumulation, fp32 output).  All other ops
(LayerNorm, Linear, gating, final LayerNorm / Linear) are kept in
torch for simplicity.

The kernel is launched with a grid of size
    (B * hidden_dim, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
so each program instance works on one (batch, channel) pair and
produces a BLOCK_M×BLOCK_N tile of the output matrix.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# -------------------------------------------------------------
# Triton kernel: batched matmul over the “k” dimension (seq_len)
# -------------------------------------------------------------
@triton.jit
def triton_trimul_kernel(
    # pointers
    left_ptr,          # [B, N, N, D]  (row‑major)
    right_ptr,         # [B, N, N, D]
    out_ptr,           # [B, N, N, D]
    # sizes
    B, N, D,
    # strides (in elements, not bytes)
    stride_lb, stride_lk, stride_ld,   # left  : batch, row(i), channel
    stride_rb, stride_rk, stride_rd,   # right : batch, row(j), channel
    stride_ob, stride_ok, stride_od,   # out   : batch, row(i), channel
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,
):
    pid = tl.program_id(0)                # linear program id over (batch*D)
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # -------------------------------------------------
    # map pid -> (batch, channel, tile_m, tile_n)
    # -------------------------------------------------
    # each group of GROUP_SIZE_M tiles processes same batch+channel
    pid_m = pid // (B * D * num_pid_n)          # which tile row
    pid_n = pid % num_pid_n                     # which tile col
    batch_chan = (pid // num_pid_n) % (B * D)   # combined batch & channel
    batch_id = batch_chan // D
    chan_id  = batch_chan % D

    # compute start offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the first element of the tile
    left_tile_ptr  = left_ptr  + batch_id * stride_lb \
                                  + chan_id   * stride_ld \
                                  + offs_m[:, None] * stride_lk
    right_tile_ptr = right_ptr + batch_id * stride_rb \
                                  + chan_id   * stride_rd \
                                  + offs_n[None, :] * stride_rk

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load a BLOCK_M×BLOCK_K panel of left and BLOCK_N×BLOCK_K of right
        left_panel = tl.load(left_tile_ptr + k_offset[None, :] * stride_lk,
                             mask=mask_m[:, None] & mask_k[None, :],
                             other=0.0).to(tl.float16)
        right_panel = tl.load(right_tile_ptr + k_offset[:, None] * stride_rk,
                              mask=mask_k[:, None] & mask_n[None, :],
                              other=0.0).to(tl.float16)

        # dot‑product across k
        acc += tl.dot(left_panel, right_panel)

        # advance pointers
        left_tile_ptr  += BLOCK_K * stride_lk
        right_tile_ptr += BLOCK_K * stride_rk

    # write result back (still fp32)
    out_tile_ptr = out_ptr + batch_id * stride_ob \
                               + chan_id   * stride_od \
                               + offs_m[:, None] * stride_ok \
                               + offs_n[None, :] * stride_od
    tl.store(out_tile_ptr, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])

# -------------------------------------------------------------
# Helper that launches the kernel
# -------------------------------------------------------------
def triton_trimul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : [B, N, N, D]  (float16 or float32)
    returns    : same shape, float32
    """
    assert left.shape == right.shape
    B, N, _, D = left.shape
    # ensure fp16 for the kernel (better throughput) – cast if needed
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out = torch.empty_like(left_fp16, dtype=torch.float32, device=left.device)

    # strides in elements
    stride_lb, stride_lk, stride_ld = left_fp16.stride()
    stride_rb, stride_rk, stride_rd = right_fp16.stride()
    stride_ob, stride_ok, stride_od = out.stride()

    # tile sizes – tuned for Hopper SM90 (shared‑memory 64KB)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    GROUP_SIZE_M = 8

    grid = (
        triton.cdiv(N, BLOCK_M) *
        triton.cdiv(N, BLOCK_N) *
        B * D,
    )
    # launch
    triton_trimul_kernel[grid](
        left_fp16, right_fp16, out,
        B, N, D,
        stride_lb, stride_lk, stride_ld,
        stride_rb, stride_rk, stride_rd,
        stride_ob, stride_ok, stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
        num_warps=4,   # enough for fp16 matmul
    )
    return out

# -------------------------------------------------------------
# Full custom kernel entry point
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Performs the forward pass of the “outgoing” TriMul operator.
    Only the O(N³) pairwise multiplication is executed on Triton.
    All other linear / norm / gating ops are done with PyTorch.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm ----------
    # x shape: [B, N, N, dim]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # B,N,N,dim  (float32)

    # ---------- Linear projections ----------
    # shape -> [B, N, N, hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Triton matmul (einsum) ----------
    # result shape [B, N, N, hidden_dim]
    out = triton_trimul(left, right)          # fp32 output

    # ---------- Final LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ---------- Output gating and linear ----------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
