"""
TriMul “outgoing” kernel (AlphaFold3) – Triton‑accelerated reduction.

The heavy part of the forward pass is the Einstein sum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where
    left  = (x @ W_left)   * σ(x @ W_left_gate)   * mask
    right = (x @ W_right)  * σ(x @ W_right_gate)  * mask

The reduction over the sequence dimension *k* is O(N³) and dominates the
runtime.  We keep all linear projections, LayerNorms and gating in PyTorch
(because they are cheap and already highly‑optimised) and off‑load only the
3‑D reduction to a custom Triton kernel.  The kernel works for arbitrary
batch size B, sequence length N and hidden size H (the hidden dimension of
the TriMul block).  It processes a tile of (M×N) output positions and a tile
of K reduction elements, accumulating in float32 and finally storing the
result in float32 (the surrounding code casts to the model dtype as needed).

The launch grid is
    grid = (B * ceil(N/BM) * ceil(N/BN),)
with constexpr tile sizes BM, BN, BK chosen for the H200 (BM=BN=64, BK=32
works well for the tested shapes).  The kernel uses only basic Triton
operations (load, store, element‑wise mul, tl.sum) and therefore runs
efficiently on Hopper SM90.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: reduction over the third sequence dimension (k)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_reduce_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, H,                                     # static sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    pid = tl.program_id(0)

    # ----- decompose program id into (batch, i‑tile, j‑tile) -------------
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_i * num_j)
    tile_idx = pid % (num_i * num_j)
    tile_i = tile_idx // num_j
    tile_j = tile_idx % num_j

    # ----- tile offsets -------------------------------------------------
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, H)                # full hidden dim (no tiling)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, H), dtype=tl.float32)

    # ----- loop over k --------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,i,k,h]
        left = tl.load(
            left_ptr
            + batch * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_h[None, None, :] * stride_left_h,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )
        # right[b,j,k,h]
        right = tl.load(
            right_ptr
            + batch * stride_right_b
            + offs_j[:, None, None] * stride_right_i   # note: reuse i‑stride for j‑dim
            + offs_k[None, :, None] * stride_right_k
            + offs_h[None, None, :] * stride_right_h,
            mask=mask_j[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )
        # element‑wise product, then sum over k (axis=1)
        prod = left * right                     # (M, K, H) * (N, K, H) → broadcast to (M, N, K, H)
        acc += tl.sum(prod, axis=1)             # sum over reduction axis K

    # ----- store result -------------------------------------------------
    out = acc.to(tl.float32)
    tl.store(
        out_ptr
        + batch * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_h[None, None, :] * stride_out_h,
        out,
        mask=mask_i[:, None, None] & mask_j[None, :, None],
    )
# ----------------------------------------------------------------------


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]   (float32 or bfloat16 on device)
        - mask_tensor  : [B, N, N]        (0/1 mask, same device)
        - weights_dict : mapping of all PyTorch weights (LayerNorm, Linear, etc.)
        - config_dict  : contains "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim) – use torch for simplicity
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – torch.nn.functional.linear
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]        # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]       # (hidden, dim)

    left = F.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_w) # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3. Apply mask (if present) – mask may be all‑ones when nomask=True
    # ------------------------------------------------------------------
    mask_ = mask.unsqueeze(-1)                 # [B,N,N,1]
    left = left * mask_
    right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – also in torch
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # [B,N,N,hidden]
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # [B,N,N,hidden]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton reduction: Σ_k left[...,i,k,d] * right[...,j,k,d]
    # ------------------------------------------------------------------
    # Cast to float16 for higher bandwidth; accumulation stays in fp32
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides (in elements) for each tensor
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # constexpr tile sizes – tuned for Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(B * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N), 1),
    )

    _trimul_reduce_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Final LayerNorm, out‑gate, and output linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate  # out_gate is still fp16; cast implicitly
    out = F.linear(out, to_out_w)  # result back to [B,N,N,dim]

    return out.to(torch.float32)
