"""
TriMul “outgoing” implementation.
The heavy O(N³·C) part is the einsum
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

We rewrite it as a batched matrix‑multiply per channel:
    left   : (B, N, N, D)  → (B·D, N, N)   (i,k)
    right  : (B, N, N, D)  → (B·D, N, N)   (j,k)
    rightᵀ : (B·D, N, N)   → (B·D, N, N)   (k,j)
    out    : (B·D, N, N) = left @ rightᵀ

The kernel is a fused FP16 mat‑mul ( BLOCK_M×BLOCK_N×BLOCK_K tiles )
followed by an FP32 accumulation and final cast to FP32.
All surrounding LayerNorm / Linear / gating work are done with
torch.nn.functional for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: batched matmul (C = A @ B)
# -------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,         # a: (batch, M, K)
    stride_bk, stride_bn,         # b: (batch, K, N)
    stride_cm, stride_cn,         # c: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,
):
    pid = tl.program_id(0)
    batch_pid = tl.program_id(1)

    # --- 2D block coordinates -------------------------------------------------
    block_i = pid // tl.cdiv(N, BLOCK_N)
    block_j = pid % tl.cdiv(N, BLOCK_N)

    # offset inside the current block
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this batch
    a_batch_ptr = a_ptr + batch_pid * stride_am * M
    b_batch_ptr = b_ptr + batch_pid * stride_bn * N

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am
                                + offs_k[None, :] * stride_ak)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk
                                + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptr + batch_pid * stride_cm * M \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a : (B*D, M, K)   FP16
    b : (B*D, K, N)   FP16
    returns c : (B*D, M, N) FP16 (cast to FP32 by caller)
    """
    assert a.is_cuda and b.is_cuda
    B_D, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    GROUP_SIZE_M = 8

    grid = (triton.cdiv(M, BLOCK_M) * triton.cdiv(N, BLOCK_N), B_D)
    c = torch.empty((B_D, M, N), device=a.device, dtype=torch.float16)

    batched_matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        a.stride(0), a.stride(2),       # stride_am, stride_ak
        b.stride(1), b.stride(2),       # stride_bk, stride_bn
        c.stride(0), c.stride(2),       # stride_cm, stride_cn
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
    )
    return c


# -------------------------------------------------------------
# Custom kernel entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass using a fused Triton mat‑mul.
    Arguments:
        data = (input_tensor, mask, weights, config)
        input_tensor : [B, N, N, dim]   float32
        mask         : [B, N, N]        (optional, bool/float)
        weights      : dict of torch tensors (fp32)
        config       : dict with keys "dim", "hidden_dim"
    Returns:
        out : [B, N, N, dim]  float32
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm ----------
    # x = (x - mean) / sqrt(var + eps) * w + b
    eps = 1e-5
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * w_norm + b_norm

    # ---------- Linear projections (fp16) ----------
    # Linear without bias: out = x @ w.T
    def linear_no_bias(t, w):
        # t : (..., dim)   fp32
        # w : (out_dim, dim)   fp32
        return F.linear(t, w)        # fp32, will cast later

    left = linear_no_bias(x, weights["left_proj.weight"]).to(torch.float16)
    right = linear_no_bias(x, weights["right_proj.weight"]).to(torch.float16)

    # ---------- Apply mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gating ----------
    left_gate = torch.sigmoid(linear_no_bias(x, weights["left_gate.weight"])).to(left.dtype)
    right_gate = torch.sigmoid(linear_no_bias(x, weights["right_gate.weight"])).to(right.dtype)
    out_gate = torch.sigmoid(linear_no_bias(x, weights["out_gate.weight"])).to(left.dtype)

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare for batched matmul ----------
    # reshape to (B*hidden_dim, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (i,k)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N) # (j,k)
    right_T = right_flat.transpose(1, 2)                               # (k,j)

    # ---------- Compute out = left @ right_T ----------
    out_flat = batched_matmul(left_flat, right_T)   # fp16 result
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,hidden_dim]
    out = out.to(torch.float32)

    # ---------- to_out_norm ----------
    w_out_norm = weights["to_out_norm.weight"]
    b_out_norm = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * w_out_norm + b_out_norm

    # ---------- Apply out_gate ----------
    out = out * out_gate

    # ---------- Final linear ----------
    out = F.linear(out, weights["to_out.weight"]).to(torch.float32)

    return out
