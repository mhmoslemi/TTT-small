"""
TriMul “outgoing” kernel (AlphaFold3)

The heavy part of TriMul is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be rewritten per‑channel as a matrix multiplication:
    out[d] = left[d] @ right[d].T   (left/right have shape B×N×N)

We implement this reduction in Triton.  All other operations
(LayerNorm, Linear projections, gates, final LayerNorm/Linear) are kept in
regular PyTorch for simplicity.  The Triton kernel is launched with a
1‑D grid where each thread computes a single (b,i,j,d) element by looping over
the reduction dimension *k* in tiles (BLOCK_K).  The kernel works for any
batch size (1‑2), sequence length N up to 1024 and hidden dimension H (≤384).
The final result is cast back to float32.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: one thread = one output element (b,i,j,d)
# ----------------------------------------------------------------------
@triton.jit
def trimul_reduce_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # static sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ---------- decode linear pid into (b,i,j,d) ----------
    total = B * N * N * H
    pid = tl.where(pid < total, pid, 0)        # guard out‑of‑range

    b = pid // (N * N * H)
    rem = pid % (N * N * H)
    i = rem // (N * H)
    rem = rem % (N * H)
    j = rem // H
    d = rem % H

    # ---------- pointers to the first element of the reduction ----------
    left_base = left_ptr + b * stride_left_b \
                        + i * stride_left_i \
                        + d * stride_left_h
    right_base = right_ptr + b * stride_right_b \
                          + j * stride_right_j \
                          + d * stride_right_h

    # ---------- accumulation ----------
    acc = tl.float32(0.0)

    num_k = tl.cdiv(N, BLOCK_K)
    for k_block in range(0, num_k):
        k_offset = k_block * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load a tile of size BLOCK_K from left[i,k,:] and right[j,k,:]
        left_val = tl.load(
            left_base + k_offset * stride_left_k,
            mask=mask_k,
            other=0.0
        )
        right_val = tl.load(
            right_base + k_offset * stride_right_k,
            mask=mask_k,
            other=0.0
        )
        # multiply‑accumulate (both are float16 or float32 – we cast to fp32)
        acc += tl.dot(left_val.to(tl.float32), right_val.to(tl.float32))

    # ---------- write result ----------
    out_ptr += b * stride_out_b + i * stride_out_i + j * stride_out_j + d * stride_out_h
    tl.store(out_ptr, acc)


# ----------------------------------------------------------------------
# Helper: launch the Triton kernel
# ----------------------------------------------------------------------
def trimul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]  (float16 or float32)
    returns      : [B, N, N, H]  (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in elements (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    BLOCK_K = 32  # good trade‑off for N up to 1024

    total_threads = B * N * N * H
    grid = (triton.cdiv(total_threads, 1),)   # 1‑D grid

    trimul_reduce_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # enough to hide latency
    )
    return out


# ----------------------------------------------------------------------
# Full TriMul forward (mostly PyTorch, core reduction in Triton)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Args
    ----
    data : (input, mask, weights, config)
        input  : [B, N, N, dim]          (float32, on CUDA)
        mask   : [B, N, N]               (bool/float, on CUDA)
        weights: dict of parameters (all float32, on CUDA)
        config : {"dim": int, "hidden_dim": int, ...}

    Returns
    -------
    output : [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm on the last dimension
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # manual layernorm for speed / to avoid extra modules
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    eps = 1e-5
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_weight + norm_bias   # shape [B,N,N,dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) – keep in FP16 to save bandwidth
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left  = torch.nn.functional.linear(x, left_proj_w).to(torch.float16)
    right = torch.nn.functional.linear(x, right_proj_w).to(torch.float16)

    # --------------------------------------------------------------
    # 3. Apply mask (if any)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)  # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4. Gates (sigmoid) – also FP16
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid().to(torch.float16)
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid().to(torch.float16)
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid().to(torch.float16)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core reduction: Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    (implemented in Triton)
    # --------------------------------------------------------------
    # left/right are [B,N,N,hidden_dim] (FP16)
    out = trimul_triton(left, right)          # float32 result, shape [B,N,N,H]

    # --------------------------------------------------------------
    # 6. Final LayerNorm + out_gate + projection back to dim
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # LayerNorm on hidden_dim
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out = (out - mean_o) * torch.rsqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # apply out gate (broadcast over hidden_dim)
    out = out * out_gate

    # final linear projection back to dim (no bias)
    out = torch.nn.functional.linear(out, to_out_w)   # float32

    return out
