"""
TriMul “outgoing” forward pass (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm the input (float32).
2. Linear projections + sigmoid gates (bias‑free linear layers).
3. Apply the mask (if present) and the gates → tensors
      L, R ∈ [B, N, N, H]   (H = hidden_dim)
4. Core operation:
      out[b,i,j,:] = Σₖ L[b,i,k,:] * R[b,j,k,:]
   This is exactly a batched matrix‑multiply for every hidden channel:
      out[:,h] = L[:,h] @ R[:,h].T        (L,R : [B, N, N])
   We implement this multiplication with a custom Triton kernel that
   processes many (B·H) “matmul batches” simultaneously.
5. Layer‑norm over the hidden dimension, sigmoid out‑gate, final linear
   projection back to dim.  All post‑gating work is kept in PyTorch for
   simplicity – the heavy N³ work is done in Triton.

The kernel is a standard block‑wise GEMM (no persistent workspace) tuned
for Hopper (SM‑90).  BLOCK sizes were chosen to give high occupancy while
respecting H200’s large shared‑memory bandwidth.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ---------------------------------------------------------------------------
# Triton GEMM kernel: C = A @ Bᵀ   (A,B : [batch, M, K] , Bᵀ : [batch, K, N])
# ---------------------------------------------------------------------------
@triton.jit
def triton_gemm_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (all int32)
    stride_am, stride_ak,              # strides for A  (batch,M,K)
    stride_bk, stride_bj,              # strides for B  (batch,K,N)
    stride_cm, stride_cn,              # strides for C  (batch,M,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,             # tiling grouping for better L2 reuse
):
    pid = tl.program_id(0)            # linear id for (batch) dimension
    # -----------------------------------------------------------------------
    # Compute the block coordinates (block_m, block_n) for this program
    # -----------------------------------------------------------------------
    num_block_m = tl.cdiv(M, BLOCK_M)
    num_block_n = tl.cdiv(N, BLOCK_N)

    # Grouped ordering of blocks to increase cache reuse (see Triton docs)
    pid_z = pid // (num_block_m * num_block_n)
    pid_in = pid % (num_block_m * num_block_n)
    block_m = (pid_in // num_block_n) * GROUP_M + (pid_in % GROUP_M)
    block_n = (pid_in % num_block_n)

    # Guard against out‑of‑bounds groups
    block_m = tl.where(block_m < num_block_m, block_m, 0)

    # -----------------------------------------------------------------------
    # Offsets for the current block
    # -----------------------------------------------------------------------
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for the edges of the tensors
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for the current batch (pid_z)
    a_ptrs = a_ptr + pid_z * stride_am * M + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_z * stride_bk * K + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj

    # -----------------------------------------------------------------------
    # Accumulator
    # -----------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)          # dot does the inner‑product over K‑tile
        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # -----------------------------------------------------------------------
    # Write back
    # -----------------------------------------------------------------------
    c_ptrs = c_ptr + pid_z * stride_cm * M + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# ---------------------------------------------------------------------------
# Helper to launch the kernel
# ---------------------------------------------------------------------------
def triton_matmul_outgoing(L: torch.Tensor, R: torch.Tensor) -> torch.Tensor:
    """
    Compute   out = L @ Rᵀ   for each hidden channel independently.
    L,R : [B, N, N, H]   (float32)
    Returns out : [B, N, N, H]  (float32)
    """
    B, N, _, H = L.shape
    # reshape to (batch = B*H, M=N, K=N)
    L = L.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    R = R.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    L = L.view(B * H, N, N)
    R = R.view(B * H, N, N)

    # allocate output
    out = torch.empty_like(L)

    # strides (in elements) for Triton
    stride_am = L.stride(0)
    stride_ak = L.stride(2)
    stride_bk = R.stride(0)
    stride_bj = R.stride(2)
    stride_cm = out.stride(0)
    stride_cn = out.stride(2)

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    GROUP_M = 8

    total_batches = B * H
    grid = (total_batches,)

    triton_gemm_outgoing_kernel[grid](
        L.data_ptr(),
        R.data_ptr(),
        out.data_ptr(),
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bj,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_M=GROUP_M,
    )
    # reshape back to original layout
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]
    return out.float()   # keep float32 for later layers

# ---------------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ---------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ---------
    data : tuple
        (input, mask, weights, config)
        - input : Tensor[B, N, N, dim]          (float32 or bfloat16)
        - mask  : Tensor[B, N, N] (may be all‑ones if nomask)
        - weights : dict of leaf tensors (see problem statement)
        - config  : dict with keys "dim", "hidden_dim"
    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------------
    # 1️⃣  Layer‑norm over the last dimension
    # -----------------------------------------------------------------------
    eps = 1e-5
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # fused layer‑norm (manual, to stay on‑device)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b

    # -----------------------------------------------------------------------
    # 2️⃣  Linear projections (bias‑free)  + 3 gates
    # -----------------------------------------------------------------------
    # Linear layers: weight shape [hidden_dim, dim]
    def linear_no_bias(x, w):
        # x : [..., dim] , w : [hidden_dim, dim]
        return torch.einsum('...d,hd->...h', x, w)

    left  = linear_no_bias(x_norm, weights["left_proj.weight"])
    right = linear_no_bias(x_norm, weights["right_proj.weight"])

    # mask handling
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # gates (sigmoid)
    left_gate  = torch.sigmoid(linear_no_bias(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_no_bias(x_norm, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear_no_bias(x_norm, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 3️⃣  Core N³ operation – Triton‑accelerated batched matmul
    # -----------------------------------------------------------------------
    # left, right : [B, N, N, H] (float32)
    out = triton_matmul_outgoing(left, right)      # [B, N, N, H]

    # -----------------------------------------------------------------------
    # 4️⃣  Output layer‑norm, gating and final linear
    # -----------------------------------------------------------------------
    # to_out_norm: LayerNorm over hidden_dim
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    eps2 = 1e-5
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps2)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    out_norm = out_norm * out_gate

    # final projection back to dim (bias‑free)
    out_final = torch.einsum('...h,dh->...d', out_norm, weights["to_out.weight"])
    return out_final.float()
