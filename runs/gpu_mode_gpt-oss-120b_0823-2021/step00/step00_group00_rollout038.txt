"""
TriMul “outgoing” implementation.

The heavy part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is an O(N³·C) contraction.  We keep the cheap LayerNorm / Linear
operations in PyTorch and off‑load only this contraction to a Triton
kernel.  The kernel works on blocks of (i,j) and reduces over k while
vectorising the hidden dimension `d`.  Mixed‑precision (fp16 for the
input tiles, fp32 accumulation) yields the same numeric result as the
reference implementation after a final cast to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel that computes the einsum over the k‑dimension.
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,           # *float16   [B, N, N, H]
    right_ptr,          # *float16   [B, N, N, H]
    out_ptr,            # *float32   [B, N, N, H]
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    # program id layout: pid = b * (grid_i * grid_j) + i_block * grid_j + j_block
    grid_i = tl.cdiv(N, BLOCK_I)
    grid_j = tl.cdiv(N, BLOCK_J)

    b = pid // (grid_i * grid_j)
    tmp = pid % (grid_i * grid_j)
    i_block = tmp // grid_j
    j_block = tmp % grid_j

    # block coordinates
    i_start = i_block * BLOCK_I + tl.arange(0, BLOCK_I)
    j_start = j_block * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i_start < N
    j_mask = j_start < N

    # initialise accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # loop over k
    for k_start in range(0, N, BLOCK_K):
        k_end = tl.minimum(k_start + BLOCK_K, N)
        cur_k = tl.arange(0, BLOCK_K)
        k_mask = cur_k < (k_end - k_start)

        # ----- load left : [B,i,k,d] -----
        left_off = (
            b * stride_lb
            + i_start[:, None, None] * stride_li
            + (k_start + cur_k)[None, :, None] * stride_lk
            + tl.arange(0, BLOCK_D)[None, None, :] * stride_ld
        )
        left = tl.load(left_ptr + left_off, mask=i_mask[:, None, None] & k_mask[None, :, None], other=0.0)
        # left shape (BLOCK_I, BLOCK_K, BLOCK_D)

        # ----- load right : [B,j,k,d] -----
        right_off = (
            b * stride_rb
            + j_start[:, None, None] * stride_rj
            + (k_start + cur_k)[None, :, None] * stride_rk
            + tl.arange(0, BLOCK_D)[None, None, :] * stride_rd
        )
        right = tl.load(right_ptr + right_off, mask=j_mask[:, None, None] & k_mask[None, :, None], other=0.0)
        # right shape (BLOCK_J, BLOCK_K, BLOCK_D)

        # bring to same k dimension order for broadcasting
        left = left.to(tl.float32)
        right = right.to(tl.float32)

        # multiply and reduce over k
        # left: (i,k,d) , right: (j,k,d) -> we need sum_k left[i,k,d]*right[j,k,d]
        # expand dims for broadcasting
        left_exp = left[:, :, None, :]          # (i,k,1,d)
        right_exp = right[None, :, :, :]        # (1,j,k,d)
        prod = left_exp * right_exp              # (i,k,j,d)
        # sum over k (axis=1)
        acc += tl.sum(prod, axis=1)               # (i,j,d)

    # ----- write out -----
    out_off = (
        b * stride_ob
        + i_start[:, None, None] * stride_oi
        + j_start[None, :, None] * stride_oj
        + tl.arange(0, BLOCK_D)[None, None, :] * stride_od
    )
    tl.store(out_ptr + out_off,
             acc,
             mask=i_mask[:, None, None] & j_mask[None, :, None])

# --------------------------------------------------------------
# Host‑side helper to launch the kernel
# --------------------------------------------------------------
def trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : [B,N,N,H]  fp16
    returns out : [B,N,N,H]  fp32
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)

    # strides in element units
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # block sizes – tuned for Hopper (sm90)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_D = 32   # hidden dimension is a multiple of 32 in all configs

    grid_i = triton.cdiv(N, BLOCK_I)
    grid_j = triton.cdiv(N, BLOCK_J)
    grid = (B * grid_i * grid_j,)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=H,
        stride_lb=stride_lb, stride_li=stride_li, stride_lk=stride_lk, stride_ld=stride_ld,
        stride_rb=stride_rb, stride_rj=stride_rj, stride_rk=stride_rk, stride_rd=stride_rd,
        stride_ob=stride_ob, stride_oi=stride_oi, stride_oj=stride_oj, stride_od=stride_od,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
    )
    return out

# --------------------------------------------------------------
# Main entry point required by the evaluator
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module with the heavy
    contraction accelerated by a Triton kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    # manual LayerNorm to avoid extra Python overhead
    eps = 1e-5
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    # input: [B,N,N,dim]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * w_norm + b_norm        # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16 for speed)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"]        # [hidden_dim, dim]
    left = F.linear(x_norm, left_proj_w)               # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                       # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) and gating of left/right
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton‑accelerated contraction over k
    # ------------------------------------------------------------------
    # Cast to fp16 before kernel (reduces memory traffic)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out = trimul_outgoing_triton(left_fp16, right_fp16)   # [B,N,N,hidden_dim] fp32

    # ------------------------------------------------------------------
    # 6. to_out_norm + gating + final linear
    # ------------------------------------------------------------------
    w_out_norm = weights["to_out_norm.weight"]
    b_out_norm = weights["to_out_norm.bias"]
    w_to_out = weights["to_out.weight"]                 # [dim, hidden_dim]

    # LayerNorm on hidden dimension
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * w_out_norm + b_out_norm

    # apply output gate
    out_norm = out_norm * out_gate

    # final linear projection back to dim
    output = F.linear(out_norm, w_to_out)                # [B,N,N,dim]

    return output
