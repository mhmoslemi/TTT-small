"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The forward pass consists of:
1. LayerNorm on the input (dim‑wise) – kept in PyTorch (fast, highly
   optimized and needs global mean/var).
2. Linear projections + element‑wise gating – also done in PyTorch.
3. Core “triangle multiplicative update”:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched, per‑channel matrix‑multiply over the
   sequence dimension k.  We implement the reduction with Triton,
   processing a tile of (i,j) at a time and accumulating over k.
   The kernel works in fp16 for the loads (the model weights are
   stored in fp32 but we cast to fp16 for speed) and accumulates
   in fp32 to retain numeric quality.  After the kernel we
   apply the final LayerNorm, output‑gate and output linear
   (again in PyTorch) and cast the result back to float32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# -------------------------------------------------------------
# Triton kernel – computes one channel (d) of the triangle mul.
# -------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # const float16*   [B, N, N, D]   (b,i,k,d)
    right_ptr,         # const float16*   [B, N, N, D]   (b,j,k,d)
    out_ptr,           #       float32*   [B, N, N, D]   (b,i,j,d)
    B, N, D,
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Grid: (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B)

    Each program processes a tile (i0:i0+BLOCK_M, j0:j0+BLOCK_N) for a
    single batch element `b`.  The reduction over k is tiled with
    BLOCK_K.
    """
    pid_i = tl.program_id(0)          # tile row   (i)
    pid_j = tl.program_id(1)          # tile col   (j)
    pid_b = tl.program_id(2)          # batch index

    # tile start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # offsets within the tile
    i_offset = i0 + tl.arange(0, BLOCK_M)
    j_offset = j0 + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds i/j
    mask_i = i_offset < N
    mask_j = j_offset < N

    # pointers to the start of the tile (b,i0,j0,0)
    out_ptrs = out_ptr + \
        pid_b * stride_ob + \
        i_offset[:, None] * stride_oi + \
        j_offset[None, :] * stride_oj

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the reduction dimension k
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        k_offset = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left:   (b, i, k, d)
        left_tile = tl.load(
            left_ptr + pid_b * stride_lb
            + i_offset[:, None] * stride_li
            + k_offset[None, :] * stride_lk
            + tl.arange(0, 1) * stride_ld,          # d=0, we will broadcast later
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        ).to(tl.float32)    # (BLOCK_M, BLOCK_K)

        # right:  (b, j, k, d)
        right_tile = tl.load(
            right_ptr + pid_b * stride_rb
            + j_offset[:, None] * stride_rj
            + k_offset[None, :] * stride_rk
            + tl.arange(0, 1) * stride_rd,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0
        ).to(tl.float32)    # (BLOCK_N, BLOCK_K)

        # transpose right to (BLOCK_K, BLOCK_N) for dot
        right_tile = tl.trans(right_tile, (1, 0))

        # dot over k
        acc += tl.dot(left_tile, right_tile)   # (BLOCK_M, BLOCK_N)

    # store the result (only where i,j are in‑bounds)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# -------------------------------------------------------------
# Host‑side helper that launches the kernel for all channels d
# -------------------------------------------------------------
def triangle_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, D]  (float16)
    returns      : [B, N, N, D]  (float32)
    """
    B, N, _, D = left.shape
    out = torch.empty((B, N, N, D), dtype=torch.float32, device=left.device)

    # strides in elements (not bytes)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # Tile sizes – tuned for Hopper SM90 (large shared‑mem, 128‑wide warp)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B,
    )

    trimul_kernel[grid](
        left,
        right,
        out,
        B, N, D,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                     # good default for H200
    )
    return out

# -------------------------------------------------------------
# Full custom kernel – mix PyTorch + Triton
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module.

    data = (input, mask, weights, config)
        input : [B, N, N, dim]   float32
        mask  : [B, N, N]        (bool / 0‑1)   – may be all‑ones
        weights : dict of model parameters (float32)
        config  : {"dim": int, "hidden_dim": int}
    Returns:
        output : [B, N, N, dim]  float32
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ---------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ---------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # manual LayerNorm to avoid an extra torch.nn.Module
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    eps  = 1e-5
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias          # [B,N,N,dim]

    # ---------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – cast to fp16 for speed
    # ---------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]          # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"]
    left  = torch.einsum("...d,hd->...h", x, left_proj_w)   # [B,N,N,hidden]
    right = torch.einsum("...d,hd->...h", x, right_proj_w)

    # ---------------------------------------------------------
    # 3️⃣ Apply mask (if any) – mask shape [B,N,N,1]
    # ---------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(left)   # broadcast
        left  = left * mask_
        right = right * mask_

    # ---------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – also cast to fp16
    # ---------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(torch.einsum("...d,hd->...h", x, left_gate_w))
    right_gate = torch.sigmoid(torch.einsum("...d,hd->...h", x, right_gate_w))
    out_gate   = torch.sigmoid(torch.einsum("...d,hd->...h", x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 5️⃣ Core triangle multiplication – Triton
    # ---------------------------------------------------------
    # Cast to fp16 for the kernel (weights are already fp32)
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # Kernel returns fp32 accumulator
    out = triangle_mul_triton(left_f16, right_f16)   # [B,N,N,hidden]

    # ---------------------------------------------------------
    # 6️⃣ Final LayerNorm + out‑gate + output projection
    # ---------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # LayerNorm on hidden dimension
    mean_h = out.mean(dim=-1, keepdim=True)
    var_h  = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_h) / torch.sqrt(var_h + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    out_norm = out_norm * out_gate          # gating

    # Linear projection back to dim
    output = torch.einsum("...h,dh->...d", out_norm, to_out_w)   # [B,N,N,dim]

    return output.to(torch.float32)
