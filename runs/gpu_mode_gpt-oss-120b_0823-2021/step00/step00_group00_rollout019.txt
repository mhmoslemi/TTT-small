"""
TriMul “outgoing” implementation.

The expensive part of TriMul is the contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]          (d = hidden)

which is a batched matrix‑multiply per hidden channel:
    out_d = left_d @ right_d.T      where left_d,right_d ∈ ℝ^{N×K}

We keep the cheap LayerNorm / Linear / gating ops in PyTorch and accelerate
the N×K × N×K^T multiplication with a fused Triton kernel.  The kernel
computes a block of the (i,j) output matrix for a single batch element and a
single hidden dimension; the host loops over the hidden dimension (≤384) and
launches the kernel for each slice.  Mixed‑precision (FP16) is used inside the
kernel, while the final output is accumulated in FP32 to match the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,          # * const uint8   left  [B,N,N,H]  (FP16)
    right_ptr,         # * const uint8   right [B,N,N,H]  (FP16)
    out_ptr,           # *       uint8   out   [B,N,N,H]  (FP32)
    B, N, H,           # sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute a block of the (i,j) matrix for a single batch element and a
    single hidden channel d:

        out[b,i0:i0+BM, j0:j0+BN, d] = left[b,i0:i0+BM, :, d] @
                                      right[b,j0:j0+BN, :, d].T
    """
    pid = tl.program_id(0)

    # ----- decode batch, i‑block and j‑block from pid -----
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    b = pid // (num_i * num_j)
    rem = pid % (num_i * num_j)
    i_block = rem // num_j
    j_block = rem % num_j

    # ----- offsets within the tensors -----
    i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K (= hidden_dim)
    for kk in range(0, tl.cdiv(H, BLOCK_K)):
        k_off = kk * BLOCK_K + k
        mask_k = k_off < H

        # left[b,i,k,d]   (M×K)
        left = tl.load(
            left_ptr
            + b * stride_left_b
            + i[:, None] * stride_left_i
            + k_off[None, :] * stride_left_k
            + 0 * stride_left_d,                # d is handled by host loop
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right[b,j,k,d]  (N×K)
        right = tl.load(
            right_ptr
            + b * stride_right_b
            + j[:, None] * stride_right_j
            + k_off[None, :] * stride_right_k
            + 0 * stride_right_d,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )
        # sum_k left * right
        acc += tl.dot(left, right, trans_b=False)   # (M,K)·(N,K) → (M,N)

    # store the block (still FP32)
    out = acc
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i[:, None] * stride_out_i
        + j[None, :] * stride_out_j
        + 0 * stride_out_d
    )
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32, on CUDA)
        - mask         : [B, N, N]        (float32/bool, on CUDA)
        - weights      : dict of parameters (float32 tensors)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1) LayerNorm and Linear / gating – keep in torch (fast enough)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # optional mask (shape [B,N,N,1])
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 2) Cast to FP16 for the heavy contraction
    # --------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # output buffer for the contraction (still FP32)
    out = torch.empty((B, N, N, hidden), device=device, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 3) Launch Triton kernel for each hidden slice
    # ------------------------------------------------------------------
    # strides (in elements, not bytes) are needed by Triton
    s_left = left_h.stride()
    s_right = right_h.stride()
    s_out = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32          # hidden is at most 384 → 32 works well

    total_blocks = B * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N)

    # one launch per hidden channel (loop on host, cheap because hidden ≤ 384)
    for d in range(hidden):
        # compute pointers that point to the beginning of the d‑th channel
        left_ptr = left_h.data_ptr() + d * s_left[3] * left_h.element_size()
        right_ptr = right_h.data_ptr() + d * s_right[3] * right_h.element_size()
        out_ptr = out.data_ptr() + d * s_out[3] * out.element_size()

        _trimul_kernel[
            (total_blocks,)
        ](
            left_ptr,
            right_ptr,
            out_ptr,
            B,
            N,
            hidden,
            s_left[0],
            s_left[1],
            s_left[2],
            s_left[3],
            s_right[0],
            s_right[1],
            s_right[2],
            s_right[3],
            s_out[0],
            s_out[1],
            s_out[2],
            s_out[3],
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

    # --------------------------------------------------------------
    # 4) Post‑processing (LayerNorm, out‑gate, final linear)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
