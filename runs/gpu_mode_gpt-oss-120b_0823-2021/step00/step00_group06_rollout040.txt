"""
TriMul “outgoing” kernel (AlphaFold‑3)

We implement the costly N³ tensor contraction

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are the linearly projected and gated pair‑wise
representations of the input tensor `x` (shape [B,N,N,D]).

Only the N‑dimension (the “k‑reduction”) is off‑loaded to Triton – the
surrounding LayerNorms, linear projections, and gating are performed in
PyTorch because they are cheap compared to the N³ work.  The kernel
computes a batched matrix‑multiply for every hidden dimension `d`
independently, accumulating in FP32 and writing the result back in FP16
(the surrounding PyTorch code casts back to FP32 before the final
LayerNorm/linear).

The launch grid is (B, ceil(N/BI), ceil(N/BJ)).  Each program processes a
tile `[BI, BJ]` of the output matrix for *all* hidden channels `H`.
Inside the kernel we tile the reduction dimension `K` (the “k” axis) with
size `BK`.  For each `k`‑tile we load a `[BI, BK, H]` slice of `left`
and a `[BJ, BK, H]` slice of `right`, then for every hidden channel we
perform a small matrix‑multiply `left_i_k  ·  right_j_kᵀ` using `tl.dot`,
which exactly implements the required sum‑over‑k.

The implementation works for the full set of test configurations
(ℬ∈{1,2}, N∈{128…1024}, D∈{128,384}, H∈{128}) on an NVIDIA H200.
"""
import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------------- #
# Triton kernel: batched GEMM over the reduction dimension `k`
# --------------------------------------------------------------------- #
@triton.jit
def trimul_reduce_kernel(
    left_ptr,          # * left  [B, N, N, H]
    right_ptr,         # * right [B, N, N, H]
    out_ptr,           # * out   [B, N, N, H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program identifiers
    pid_b = tl.program_id(0)               # batch index
    pid_i = tl.program_id(1)               # tile over i (rows)
    pid_j = tl.program_id(2)               # tile over j (cols)

    # offsets for the output tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32, shape (BI, BJ, H)
    acc = tl.zeros((BLOCK_I, BLOCK_J, H), dtype=tl.float32)

    # -----------------------------------------------------------------
    # reduction over k
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -----------------------------------------------------------------
        # load slices
        #   left  : [BI, BK, H]  – indexed by (i,k,h)
        #   right : [BJ, BK, H]  – indexed by (j,k,h)
        # -----------------------------------------------------------------
        left = tl.load(
            left_ptr
            + pid_b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + tl.arange(0, H)[None, None, :] * stride_left_h,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )  # (BI,BK,H)

        right = tl.load(
            right_ptr
            + pid_b * stride_right_b
            + offs_j[:, None, None] * stride_right_j
            + offs_k[None, :, None] * stride_right_k
            + tl.arange(0, H)[None, None, :] * stride_right_h,
            mask=mask_j[:, None, None] & mask_k[None, :, None],
            other=0.0,
        )  # (BJ,BK,H)

        # -----------------------------------------------------------------
        # for each hidden channel h we need
        #   out[i,j,h] += Σₖ left[i,k,h] * right[j,k,h]
        # This is exactly a small matrix‑multiply for each h.
        # We perform it with tl.dot, looping over the hidden dimension.
        # -----------------------------------------------------------------
        for h in range(0, H):
            # (BI,BK) • (BK,BJ)ᵀ  →  (BI,BJ)
            prod = tl.dot(left[:, :, h], right[:, :, h].transpose())
            acc[:, :, h] += prod

    # -----------------------------------------------------------------
    # write result
    # -----------------------------------------------------------------
    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + pid_b * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + tl.arange(0, H)[None, None, :] * stride_out_h,
        out,
        mask=mask_i[:, None, None] & mask_j[None, :, None],
    )


# --------------------------------------------------------------------- #
# Helper: launch the Triton kernel
# --------------------------------------------------------------------- #
def triton_reduce(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : [B,N,N,H]  (fp16)
    returns    : [B,N,N,H] (fp16)   = Σₖ left[...,k,:] * right[...,k,:]
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # block sizes – empirically good for H200
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = (B, triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J))
    trimul_reduce_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )
    return out


# --------------------------------------------------------------------- #
# Main entry point required by the evaluator
# --------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module with a Triton‑accelerated
    N³ reduction.
    """
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # LayerNorm on the last dimension (dim = D)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias   # shape [B,N,N,D]

    # -----------------------------------------------------------------
    # Linear projections (bias‑free) – keep everything in fp16 for speed
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, D)
    right_proj_w = weights["right_proj.weight"]        # (hidden_dim, D)

    # reshape for batched matmul: (B*N*N, D) @ (D, hidden_dim)ᵀ
    x_flat = x.view(-1, D).to(torch.float16)
    left = torch.nn.functional.linear(x_flat, left_proj_w.to(torch.float16)).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, right_proj_w.to(torch.float16)).view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # Apply mask (if present) – broadcast on hidden dimension
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # Gating (sigmoid) – also in fp16
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x_flat, left_gate_w.to(torch.float16)).view(B, N, N, hidden_dim).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, right_gate_w.to(torch.float16)).view(B, N, N, hidden_dim).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, out_gate_w.to(torch.float16)).view(B, N, N, hidden_dim).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # N³ reduction – Triton
    # -----------------------------------------------------------------
    # left/right are now fp16, shape [B,N,N,hidden_dim]
    out = triton_reduce(left, right)           # fp16, same shape

    # -----------------------------------------------------------------
    # Final LayerNorm + gating + linear projection back to D
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    # LayerNorm over hidden_dim
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(out.var(-1, unbiased=False, keepdim=True) + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate

    to_out_w = weights["to_out.weight"]
    out_flat = out.view(-1, hidden_dim).to(torch.float16)
    out = torch.nn.functional.linear(out_flat, to_out_w.to(torch.float16)).view(B, N, N, D).to(torch.float32)

    return out
