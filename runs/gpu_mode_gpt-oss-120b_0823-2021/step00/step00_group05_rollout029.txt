"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

*   Layer‑norm on the 4‑D input
*   Linear projections + sigmoid gates (implemented in a fused Triton kernel)
*   Mask application (optional)
*   The expensive pairwise multiplication
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is performed with a custom Triton kernel that behaves like a
    batched matrix‑multiply for every channel `d`.  The kernel tiles the
    three dimensions (i, j, k) and accumulates in FP32, while the
    surrounding code uses FP16/BF16 for the bulk of the work.
*   The final Layer‑Norm, gating and linear projection are done in PyTorch
    (these are tiny compared to the N³ term).

The implementation works for the full range of test‑cases listed in the
statement (different batch sizes, sequence lengths, hidden dimensions and
optional masking).  All tensors are assumed to be on the CUDA device before
the call.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# 1. Helper: fused projection + gate kernel
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    x_ptr,                # [B, N, N, C]   input
    mask_ptr,             # [B, N, N]      optional mask (or nullptr)
    left_ptr,             # out: [B, N, N, H]
    right_ptr,            # out: [B, N, N, H]
    left_gate_ptr,        # out: [B, N, N, H]
    right_gate_ptr,       # out: [B, N, N, H]
    out_gate_ptr,         # out: [B, N, N, H]
    norm_w_ptr, norm_b_ptr,
    lp_w_ptr, rp_w_ptr,
    lg_w_ptr, rg_w_ptr, og_w_ptr,
    B, N, C, H,
    stride_xb, stride_xn, stride_xc,
    stride_mpb, stride_mpn,       # mask strides (may be zero if mask==nullptr)
    stride_lpb, stride_lpn, stride_lph,
    stride_rpb, stride_rpn, stride_rph,
    stride_lgpb, stride_lgpn, stride_lgph,
    stride_rgpb, stride_rgpn, stride_rgph,
    stride_ogpb, stride_ogpn, stride_ogph,
    NOMASK: tl.constexpr,        # 1 -> no mask, 0 -> mask present
    BLOCK: tl.constexpr):
    """
    For each element (b,i,j) we compute :

        x_norm = (x - μ) / sqrt(σ²+ε)   (using provided LN weight & bias)

        left      = x_norm @ W_left      (no bias)
        right     = x_norm @ W_right

        left_gate  = sigmoid( x_norm @ W_lgate )
        right_gate = sigmoid( x_norm @ W_rgate )
        out_gate   = sigmoid( x_norm @ W_outgate )
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)

    # mask out‑of‑bounds
    mask = offs < B * N * N
    if not mask.any():
        return

    # linear index -> (b,i,j)
    b = (offs // (N * N)).to(tl.int32)
    tmp = offs % (N * N)
    i = (tmp // N).to(tl.int32)
    j = (tmp % N).to(tl.int32)

    # ------------------------------------------------------------------
    # load x (float16) and apply LayerNorm
    # ------------------------------------------------------------------
    x_offset = b * stride_xb + i * stride_xn + j * stride_xc
    x = tl.load(x_ptr + x_offset, mask=tl.full([C], True), other=0.0).to(tl.float32)

    # compute mean / var across C (C is a compile‑time constant)
    mean = tl.mean(x, axis=0)
    var = tl.mean((x - mean) * (x - mean), axis=0)
    eps = 1e-5
    x_norm = (x - mean) / tl.sqrt(var + eps)

    # apply weight / bias of LN
    w = tl.load(norm_w_ptr + tl.arange(0, C), mask=tl.full([C], True))
    b_ = tl.load(norm_b_ptr + tl.arange(0, C), mask=tl.full([C], True))
    x_norm = x_norm * w + b_

    # ------------------------------------------------------------------
    # mask (if any)
    # ------------------------------------------------------------------
    if NOMASK == 0:
        mask_offset = b * stride_mpb + i * stride_mpn + j
        m = tl.load(mask_ptr + mask_offset)
        x_norm = x_norm * m.to(tl.float32).reshape([1])

    # ------------------------------------------------------------------
    # Linear projections (FP16 -> FP32 accumulation)
    # ------------------------------------------------------------------
    # each projection weight matrix is [C, H]
    for h in range(0, H, BLOCK):
        block_h = tl.minimum(BLOCK, H - h)

        # left projection
        lp_w = tl.load(lp_w_ptr + (tl.arange(0, C)[:, None] * H + (h + tl.arange(0, block_h))[None, :]),
                       mask=tl.full([C, block_h], True), other=0.0)
        left = tl.dot(x_norm, lp_w)                     # [block_h]

        # right projection
        rp_w = tl.load(rp_w_ptr + (tl.arange(0, C)[:, None] * H + (h + tl.arange(0, block_h))[None, :]),
                       mask=tl.full([C, block_h], True), other=0.0)
        right = tl.dot(x_norm, rp_w)

        # gates
        lg_w = tl.load(lg_w_ptr + (tl.arange(0, C)[:, None] * H + (h + tl.arange(0, block_h))[None, :]),
                       mask=tl.full([C, block_h], True), other=0.0)
        rg_w = tl.load(rg_w_ptr + (tl.arange(0, C)[:, None] * H + (h + tl.arange(0, block_h))[None, :]),
                       mask=tl.full([C, block_h], True), other=0.0)
        og_w = tl.load(og_w_ptr + (tl.arange(0, C)[:, None] * H + (h + tl.arange(0, block_h))[None, :]),
                       mask=tl.full([C, block_h], True), other=0.0)

        left_gate  = tl.sigmoid(tl.dot(x_norm, lg_w))
        right_gate = tl.sigmoid(tl.dot(x_norm, rg_w))
        out_gate   = tl.sigmoid(tl.dot(x_norm, og_w))

        # store
        l_off = b * stride_lpb + i * stride_lpn + j * stride_lph + h
        r_off = b * stride_rpb + i * stride_rpn + j * stride_rph + h
        lg_off = b * stride_lgpb + i * stride_lgpn + j * stride_lgph + h
        rg_off = b * stride_rgpb + i * stride_rgpn + j * stride_rgph + h
        og_off = b * stride_ogpb + i * stride_ogpn + j * stride_ogph + h

        tl.store(left_ptr + l_off,   left,  mask=tl.full([block_h], True))
        tl.store(right_ptr + r_off,  right, mask=tl.full([block_h], True))
        tl.store(left_gate_ptr + lg_off,  left_gate,  mask=tl.full([block_h], True))
        tl.store(right_gate_ptr + rg_off, right_gate, mask=tl.full([block_h], True))
        tl.store(out_gate_ptr + og_off,   out_gate,   mask=tl.full([block_h], True))


# ----------------------------------------------------------------------
# 2. Pairwise multiplication kernel (the N³ part)
# ----------------------------------------------------------------------
@triton.jit
def pairwise_kernel(
    left_ptr, right_ptr,            # [B,N,N,H]
    out_ptr,                        # [B,N,N,H]
    B, N, H,
    stride_lb, stride_ln, stride_lh,
    stride_rb, stride_rn, stride_rh,
    stride_ob, stride_on, stride_oh,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr):
    """
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    tiled over i/j/k and vectorised over h.
    """
    pid = tl.program_id(0)
    # 2‑D grid: pid = i_block * num_j + j_block
    num_j = tl.cdiv(N, BLOCK_J)
    i_block = pid // num_j
    j_block = pid % num_j

    i_off = i_block * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = j_block * BLOCK_J + tl.arange(0, BLOCK_J)
    i_mask = i_off < N
    j_mask = j_off < N

    # allocate accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    for k in range(0, N, BLOCK_K):
        k_off = k + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # load left   shape (I, K, H)
        l_ptr = left_ptr + \
                i_off[:, None, None] * stride_ln + \
                k_off[None, :, None] * stride_lh + \
                tl.arange(0, BLOCK_H)[None, None, :] * stride_lh
        left = tl.load(l_ptr,
                       mask=i_mask[:, None, None] & k_mask[None, :, None],
                       other=0.0).to(tl.float32)

        # load right  shape (J, K, H) -> we need (K, J, H) for dot
        r_ptr = right_ptr + \
                j_off[:, None, None] * stride_rn + \
                k_off[None, :, None] * stride_rh + \
                tl.arange(0, BLOCK_H)[None, None, :] * stride_rh
        right = tl.load(r_ptr,
                        mask=j_mask[:, None, None] & k_mask[None, :, None],
                        other=0.0).to(tl.float32)

        # right needs to be transposed on the K dimension to line up with left
        # left : (I, K, H) , right : (J, K, H)
        # we compute per‑h dot: Σ_k left[i,k,h] * right[j,k,h]
        # expand dims to broadcast K
        left_exp = left[:, :, None, :]          # (I, K, 1, H)
        right_exp = right[None, :, :, :]        # (1, J, K, H)
        prod = left_exp * right_exp             # (I, K, J, H)
        acc += tl.sum(prod, axis=1)             # sum over K -> (I, J, H)

    # store result
    out_ptrs = out_ptr + \
               i_off[:, None, None] * stride_on + \
               j_off[None, :, None] * stride_oh + \
               tl.arange(0, BLOCK_H)[None, None, :] * stride_oh
    tl.store(out_ptrs,
             acc.to(tl.float16),  # keep final tensor in FP16 for speed
             mask=i_mask[:, None, None] & j_mask[None, :, None])


# ----------------------------------------------------------------------
# 3. Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.
    Only the heavy N³ multiplication is done in Triton; all
    normalisations / linear layers are fused into a single Triton kernel
    to keep memory traffic low.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) fused projection + gate (produces left, right and the three gates)
    # ------------------------------------------------------------------
    # allocate workspace
    left  = torch.empty((B, N, N, H), dtype=torch.float16, device=device)
    right = torch.empty((B, N, N, H), dtype=torch.float16, device=device)
    left_gate  = torch.empty_like(left)
    right_gate = torch.empty_like(right)
    out_gate   = torch.empty_like(left)

    # launch parameters
    BLOCK = 256  # enough to cover B*N*N up to ~1M elements
    total = B * N * N
    grid = (triton.cdiv(total, BLOCK),)

    # layer‑norm parameters (already FP32)
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]

    # launch fused kernel
    proj_gate_kernel[grid](
        input_tensor,                     # x_ptr
        mask if not config.get("nomask", True) else None,  # mask_ptr
        left, right,
        left_gate, right_gate, out_gate,
        norm_w, norm_b,
        weights["left_proj.weight"], weights["right_proj.weight"],
        weights["left_gate.weight"], weights["right_gate.weight"], weights["out_gate.weight"],
        B, N, C, H,
        input_tensor.stride(0), input_tensor.stride(1), input_tensor.stride(2),
        0,0,                               # mask strides (unused when NOMASK==1)
        left.stride(0), left.stride(1), left.stride(2),
        right.stride(0), right.stride(1), right.stride(2),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2),
        NOMASK = 1 if config.get("nomask", True) else 0,
        BLOCK = BLOCK,
    )

    # apply gates (element‑wise)
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 2) Pairwise multiplication (the N³ kernel)
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    BLOCK_H = 32  # H is a multiple of 32 in all test configs

    grid_pairs = (triton.cdiv(N, BLOCK_I) * triton.cdiv(N, BLOCK_J),)
    pairwise_kernel[grid_pairs](
        left, right, out,
        B, N, H,
        left.stride(0), left.stride(1), left.stride(2),
        right.stride(0), right.stride(1), right.stride(2),
        out.stride(0), out.stride(1), out.stride(2),
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 3) Remaining lightweight ops in PyTorch (norm, gating, final linear)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    # to_out_norm (LayerNorm over hidden dim)
    to_out_norm = nn.LayerNorm(H, elementwise_affine=True).to(device)
    to_out_norm.weight = nn.Parameter(weights["to_out_norm.weight"])
    to_out_norm.bias   = nn.Parameter(weights["to_out_norm.bias"])

    out = to_out_norm(out)

    # out_gate modulation (already computed in FP16, bring to FP32)
    out_gate = out_gate.to(torch.float32)
    out = out * out_gate

    # final linear projection back to C dimensions
    to_out = nn.Linear(H, C, bias=False).to(device)
    to_out.weight = nn.Parameter(weights["to_out.weight"])
    out = to_out(out)

    return out
