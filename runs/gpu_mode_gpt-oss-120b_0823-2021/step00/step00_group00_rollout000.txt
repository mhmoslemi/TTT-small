"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape (B,N,N,H) (H = hidden_dim).
We implement this reduction with a Triton kernel that tiles the
(i,j) output plane and accumulates over the k‑dimension block‑wise.
All other linear / layer‑norm / gating ops are kept in PyTorch
(easily‑vectorised and cheap compared with the N³ reduction).

The kernel works for any batch size (≤2), sequence length N (up to 1024)
and hidden dimension H (multiple of 32).  The final result is cast back
to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel:  sum‑over‑k of element‑wise products, keeping the last dim
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # * left  [B, N, N, H]
    right_ptr,         # * right [B, N, N, H]
    out_ptr,           # * out   [B, N, N, H]
    B, N, H,
    stride_lb, stride_lk, stride_ld,   # left strides  (batch, i, k, d)
    stride_rb, stride_rj, stride_rd,   # right strides (batch, j, k, d)
    stride_ob, stride_oi, stride_oj, stride_od,  # out strides
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(0)                # batch index
    pid_i = tl.program_id(1)                # tile over i (rows)
    pid_j = tl.program_id(2)                # tile over j (cols)

    # --------------------------------------------------------------
    # Offsets for the output tile (i,j,d)
    # --------------------------------------------------------------
    i_base = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_base = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    d_base = tl.arange(0, BLOCK_D)

    mask_i = i_base < N
    mask_j = j_base < N
    mask_d = d_base < H

    # broadcast masks to 3‑D
    mask_ij = mask_i[:, None] & mask_j[None, :]
    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over k in blocks
    # --------------------------------------------------------------
    for k_offset in range(0, N, BLOCK_K):
        cur_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # ---------- load left tile: (i, k, d) ----------
        left_ptrs = (left_ptr
                     + pid_b * stride_lb
                     + i_base[:, None, None] * stride_lk
                     + cur_k[None, :, None] * stride_ld
                     + d_base[None, None, :] * 1)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                       other=0.0).to(tl.float32)

        # ---------- load right tile: (j, k, d) ----------
        right_ptrs = (right_ptr
                      + pid_b * stride_rb
                      + j_base[:, None, None] * stride_rj
                      + cur_k[None, :, None] * stride_rd
                      + d_base[None, None, :] * 1)
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                        other=0.0).to(tl.float32)

        # element‑wise product and sum over k
        # left: (i,k,d)   right: (j,k,d)  -> we need (i,j,d) = Σ_k left[i,k,d] * right[j,k,d]
        # we broadcast left over j and right over i, then sum over k
        prod = tl.broadcast_to(left[:, None, :, :], (BLOCK_I, BLOCK_J, BLOCK_K, BLOCK_D)) * \
               tl.broadcast_to(right[None, :, :, :], (BLOCK_I, BLOCK_J, BLOCK_K, BLOCK_D))
        acc += tl.sum(prod, axis=2)   # sum over k

    # --------------------------------------------------------------
    # Store result
    # --------------------------------------------------------------
    out_ptrs = (out_ptr
                + pid_b * stride_ob
                + i_base[:, None, None] * stride_oi
                + j_base[None, :, None] * stride_oj
                + d_base[None, None, :] * stride_od)
    tl.store(out_ptrs, acc.to(tl.float32),
             mask=mask_ij[:, :, None] & mask_d[None, None, :])


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: (input, mask, weights, config)
            - input:  [B, N, N, dim]   (float32 or bfloat16)
            - mask :  [B, N, N]        (bool / 0‑1)
            - weights: dict with all linear / ln parameters
            - config : {"dim":..., "hidden_dim":...}
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections ----------
    # shape: [B,N,N,hidden]
    left = F.linear(x, weights["left_proj.weight"])          # bias=False
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask (if provided) ----------
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel for the N³ reduction
    # ------------------------------------------------------------------
    # Cast to float16 for faster matmul‑like work; accumulator stays fp32.
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # kernel launch dimensions
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_D = 32                     # hidden_dim is a multiple of 32 in our tests

    grid = (
        B,
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
    )

    # strides (in elements) for each tensor
    stride_lb = left.stride(0)
    stride_lk = left.stride(1)   # i dimension
    stride_ld = left.stride(2)   # k dimension
    stride_lh = left.stride(3)   # hidden dim

    stride_rb = right.stride(0)
    stride_rj = right.stride(1)  # j dimension
    stride_rk = right.stride(2)  # k dimension
    stride_rh = right.stride(3)

    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_oh = out.stride(3)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_lb=stride_lb,
        stride_lk=stride_lk,
        stride_ld=stride_ld,
        stride_rb=stride_rb,
        stride_rj=stride_rj,
        stride_rd=stride_rk,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_od=stride_oh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # Post‑processing (LayerNorm, gating, final linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out.to(torch.float32)
```

The implementation keeps the memory‑intensive N³ reduction inside a single Triton kernel while all cheap Linear/LayerNorm/Gate operations stay in PyTorch. The kernel tiles the (i, j) output plane, loops over k in blocks, and performs an element‑wise multiplication followed by a sum‑reduce over k, accumulating in FP32 for precision. The final tensor is returned in `float32` as required.
