"""
TriMul “outgoing” kernel.

The forward pass consists of:
1. Layer‑norm on the (B,N,N,C) input.
2. Three linear projections (left, right, three gates) – done in PyTorch.
3. Masking and element‑wise gating.
4. A fused “tri‑multiplicative” update
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is an Einstein‑summation over the shared k dimension.
   This O(N³·C) operation dominates the runtime, so it is implemented in
   Triton.  The kernel works per‑batch (B ≤ 2) and processes blocks of
   (i,j) while looping over k in tiles.  The hidden dimension (C) is
   vectorised with a small block (BLOCK_H) and accumulated in FP32.
5. A final LayerNorm, gating and linear projection – back in PyTorch.
The kernel returns FP32 results; the surrounding code casts to the
required dtype and applies the remaining linear layers.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: tri‑multiplicative update (the einsum part)
# --------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    N, H,                                  # problem size
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_j, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute   out[i,j,h] = Σ_k left[i,k,h] * right[j,k,h]

    left  : (N, N, H)   indexed as (i, k, h)
    right : (N, N, H)   indexed as (j, k, h)
    out   : (N, N, H)   indexed as (i, j, h)

    The kernel is launched with a 2‑D grid over (i, j) blocks.
    """
    pid_i = tl.program_id(0)      # i‑block id
    pid_j = tl.program_id(1)      # j‑block id

    i_off = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i_off < N
    j_mask = j_off < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over k in tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k):
        k_off = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # ----- load left[i, k, h] -------------------------------------------------
        # shape (BLOCK_I, BLOCK_K, BLOCK_H)
        left = tl.load(
            left_ptr
            + i_off[:, None, None] * stride_left_i
            + k_off[None, :, None] * stride_left_k
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_left_h,
            mask=(i_mask[:, None, None] & k_mask[None, :, None] &
                  (tl.arange(0, BLOCK_H)[None, None, :] < H)),
            other=0.0,
        )

        # ----- load right[j, k, h] ------------------------------------------------
        # shape (BLOCK_J, BLOCK_K, BLOCK_H)
        right = tl.load(
            right_ptr
            + j_off[:, None, None] * stride_right_j
            + k_off[None, :, None] * stride_right_k
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_right_h,
            mask=(j_mask[:, None, None] & k_mask[None, :, None] &
                  (tl.arange(0, BLOCK_H)[None, None, :] < H)),
            other=0.0,
        )

        # ----- compute dot over k for every hidden slice ---------------------------
        # We cannot call tl.dot on a 3‑D tensor, so we loop over the hidden block.
        for h in range(0, BLOCK_H):
            # (BLOCK_I, BLOCK_K)  x  (BLOCK_J, BLOCK_K)^T  -> (BLOCK_I, BLOCK_J)
            a = left[:, :, h]          # (I, K)
            b = right[:, :, h]         # (J, K)
            acc[:, :, h] += tl.dot(a, b, trans_b=True)  # sum over K

    # ----- write results ---------------------------------------------------------
    out = acc.to(tl.float32)   # keep FP32 (final cast is done in PyTorch)

    tl.store(
        out_ptr
        + i_off[:, None, None] * stride_out_i
        + j_off[None, :, None] * stride_out_j
        + tl.arange(0, BLOCK_H)[None, None, :] * stride_out_h,
        out,
        mask=(i_mask[:, None, None] & j_mask[None, :, None] &
              (tl.arange(0, BLOCK_H)[None, None, :] < H)),
    )


# --------------------------------------------------------------
# Host‑side helper that launches the kernel for one batch
# --------------------------------------------------------------
def trimul_einsum_one_batch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : (N, N, H)   FP32
    returns      : (N, N, H)   FP32
    """
    assert left.is_contiguous() and right.is_contiguous()
    N, _, H = left.shape

    out = torch.empty_like(left)

    # strides in element units (not bytes)
    stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tuning parameters – good for Hopper (sm_90)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    BLOCK_H = 32   # hidden dim (≤ 384) fits into a few blocks

    grid = (
        triton.cdiv(N, BLOCK_I),   # i‑blocks
        triton.cdiv(N, BLOCK_J),   # j‑blocks
    )
    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        N=N,
        H=H,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out


# --------------------------------------------------------------
# Full Triton‑accelerated TriMul forward (no gradients required)
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : [B, N, N, C]   FP32 (already on CUDA)
        mask         : [B, N, N]      (bool / float, 1.0 means keep)
        weights      : dict of all linear / ln parameters (FP32)
        config       : dict containing "dim" and "hidden_dim"
    Returns
    -------
    output : torch.Tensor  [B, N, N, C]  (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, C = input_tensor.shape
    hidden = config["hidden_dim"]

    # ---------- LayerNorm -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # manual LayerNorm (no torch.nn to stay on GPU)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x_norm = x_norm * norm_weight.view(1, 1, 1, -1) + norm_bias.view(1, 1, 1, -1)

    # ---------- Linear projections (no bias) -------------------------------
    # shape: (C, hidden)
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # project
    left  = torch.einsum('b i j c, c h -> b i j h', x_norm, left_proj_w)   # (B,N,N,hidden)
    right = torch.einsum('b i j c, c h -> b i j h', x_norm, right_proj_w)

    # ---------- Mask (if present) -----------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # (B,N,N,1)
        left  = left * mask_
        right = right * mask_

    # ---------- Gates -------------------------------------------------------
    left_gate  = torch.einsum('b i j c, c h -> b i j h', x_norm, left_gate_w).sigmoid()
    right_gate = torch.einsum('b i j c, c h -> b i j h', x_norm, right_gate_w).sigmoid()
    out_gate   = torch.einsum('b i j c, c h -> b i j h', x_norm, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ---------- Tri‑multiplicative update (Triton) -------------------------
    # Process each batch independently (B ≤ 2)
    out_list = []
    for b in range(B):
        out_b = trimul_einsum_one_batch(left[b], right[b])   # (N,N,hidden)
        out_list.append(out_b)
    out = torch.stack(out_list, dim=0)   # (B,N,N,hidden)

    # ---------- Final LayerNorm + gating + linear -------------------------
    # to_out_norm (LayerNorm over hidden)
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out_norm = out_norm * to_out_norm_w.view(1,1,1,-1) + to_out_norm_b.view(1,1,1,-1)

    out_norm = out_norm * out_gate          # gating

    # final linear back to C
    output = torch.einsum('b i j h, h c -> b i j c', out_norm, to_out_w)

    return output.float()
