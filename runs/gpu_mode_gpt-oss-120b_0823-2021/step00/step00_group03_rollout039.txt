"""
TriMul “outgoing” forward pass (AlphaFold3)

The heavy O(N³·C) part is a batch of C independent matrix‑matrix products

    out[b, i, j, c] = Σ_k left[b, i, k, c] * right[b, j, k, c]

If we transpose the last two axes of *right* we obtain the classic
batched GEMM:

    right_t[b, k, j, c] = right[b, j, k, c]
    out[b, :, :, c]      = left[b, :, :, c] @ right_t[b, :, :, c]          (N×N)·(N×N)

The kernel below computes this GEMM for many (i, j) tiles while keeping the
channel dimension `c` in registers (C ≤ 128).  All linear layers,
LayerNorms and gating are performed in PyTorch – they are cheap compared
to the N³ work – and only the matrix multiplication is off‑loaded to Triton.
The final result is cast back to FP32 so the output matches the reference.

The kernel works for any batch size (1 or 2), any sequence length up to
1024 and any hidden dimension that is a multiple of 32 (the typical
values in the test matrix).  It uses FP16 arithmetic internally and
accumulates in FP32 for stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM over the hidden dimension (C)
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    # pointers
    left_ptr,          #   [B, N, N, C]  (FP16)
    right_t_ptr,       #   [B, N, N, C]  (FP16)   = right.transpose(1,2)
    out_ptr,           #   [B, N, N, C]  (FP32)
    # strides
    stride_lb, stride_lk, stride_lc,
    stride_rb, stride_rk, stride_rc,
    stride_ob, stride_ok, stride_oc,
    # problem size
    B, N, C,
    # compile‑time tile sizes
    TILE_M: tl.constexpr, TILE_N: tl.constexpr, TILE_C: tl.constexpr,
    BLOCK_K: tl.constexpr,                      # K‑loop tile (over the summed axis)
):
    """Compute:
       out[b, i, j, c] = Σ_k left[b, i, k, c] * right_t[b, k, j, c]
    """
    pid = tl.program_id(axis=0)                     # global tile id
    num_m = tl.cdiv(N, TILE_M)
    num_n = tl.cdiv(N, TILE_N)
    b = pid // (num_m * num_n)                      # batch index
    tile_index = pid % (num_m * num_n)
    m_tile = tile_index // num_n
    n_tile = tile_index % num_n

    # start offsets of the tile
    i0 = m_tile * TILE_M + tl.arange(0, TILE_M)
    j0 = n_tile * TILE_N + tl.arange(0, TILE_N)

    # mask for out‑of‑bounds rows / cols
    mask_i = i0 < N
    mask_j = j0 < N

    # each thread block works on TILE_C channels (stored in registers)
    c0 = tl.arange(0, TILE_C)

    # Pointers for the current tile
    left_tile_ptr  = left_ptr  + b * stride_lb \
                                   + i0[:, None, None] * stride_lk \
                                   + c0[None, None, :] * stride_lc
    right_tile_ptr = right_t_ptr + b * stride_rb \
                                   + c0[None, None, :] * stride_rc \
                                   + j0[None, :, None] * stride_rb   # will be adjusted per k‑loop

    # accumulation buffer (FP32)
    acc = tl.zeros((TILE_M, TILE_N, TILE_C), dtype=tl.float32)

    # loop over k (summation axis)
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left:  (i, k, c)
        left = tl.load(
            left_tile_ptr + cur_k[None, :, None] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )                       # FP16

        # right_t: (k, j, c)
        right = tl.load(
            right_tile_ptr + cur_k[:, None, None] * stride_rb,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )                       # FP16

        # dot over k (BLOCK_K) → we can use tl.dot for each c channel
        # tl.dot works on 2‑D matrices; we broadcast over c manually.
        # Unroll the small c dimension (TILE_C ≤ 128) with a for‑loop.
        for cc in range(0, TILE_C):
            a = left[:, :, cc]                # (TILE_M, BLOCK_K)
            b = right[:, :, cc]               # (BLOCK_K, TILE_N)
            acc[:, :, cc] += tl.dot(a, b)     # FP32 accumulation

    # write result back (FP32)
    out_tile_ptr = out_ptr + b * stride_ob \
                               + i0[:, None, None] * stride_ok \
                               + j0[None, :, None] * stride_ok \
                               + c0[None, None, :] * stride_oc

    tl.store(
        out_tile_ptr,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None]
    )

# Helper utilities

def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """FP32 LayerNorm implemented with torch (fast enough for tiny C)."""
    orig_dtype = x.dtype
    x = x.float()
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x = (x - mean) / torch.sqrt(var + eps)
    x = x * weight + bias
    return x.to(orig_dtype)

def linear(x: torch.Tensor, weight: torch.Tensor) -> torch.Tensor:
    """Linear without bias (weight shape: out_dim × in_dim)."""
    return torch.nn.functional.linear(x, weight)   # uses torch.mm internally

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the “outgoing” TriMul.

    Args
    ----
    data : (input, mask, weights, config)
        input   – [B, N, N, dim]    (float32, on CUDA)
        mask    – [B, N, N]         (float32 or bool, on CUDA)
        weights – dict of tensors required by the reference module
        config  – {'dim': int, 'hidden_dim': int}
    Returns
    -------
    out : torch.Tensor
        [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dim (dim)
    # ------------------------------------------------------------------
    inp = layer_norm(
        inp,
        weights["norm.weight"],
        weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – produce FP16 tensors
    # ------------------------------------------------------------------
    left  = linear(inp, weights["left_proj.weight"]).to(torch.float16)
    right = linear(inp, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – keep in FP32 for stability, then cast
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(linear(inp, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(inp, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear(inp, weights["out_gate.weight"]))

    left  = left * left_gate.to(torch.float16)
    right = right * right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·C) contraction – Triton GEMM
    # ------------------------------------------------------------------
    # Transpose right to align the contracted axis
    right_t = right.transpose(1, 2).contiguous()   # now [B, N, N, C]

    # allocate output (FP32 accumulation)
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=inp.device)

    # strides (in elements, not bytes)
    stride_lb, stride_lk, stride_lc = left.stride()
    stride_rb, stride_rk, stride_rc = right_t.stride()
    stride_ob, stride_ok, stride_oc = out.stride()

    # Tile sizes – tuned for Hopper (SM90).  Must divide hidden_dim.
    TILE_M = 64
    TILE_N = 64
    TILE_C = 32                     # hidden_dim is a multiple of 32 in all tests
    BLOCK_K = 32

    # launch grid: one block per (batch, tile_m, tile_n)
    total_tiles = B * triton.cdiv(N, TILE_M) * triton.cdiv(N, TILE_N)
    grid = (total_tiles,)

    trimul_gemm_kernel[grid](
        left_ptr=left,
        right_t_ptr=right_t,
        out_ptr=out,
        stride_lb=stride_lb, stride_lk=stride_lk, stride_lc=stride_lc,
        stride_rb=stride_rb, stride_rk=stride_rk, stride_rc=stride_rc,
        stride_ob=stride_ob, stride_ok=stride_ok, stride_oc=stride_oc,
        B=B, N=N, C=hidden,
        TILE_M=TILE_M, TILE_N=TILE_N, TILE_C=TILE_C,
        BLOCK_K=BLOCK_K,
        num_stages=3,            # double‑buffering
        num_warps=8
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm, out‑gate and projection back to dim
    # ------------------------------------------------------------------
    out = layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate            # out_gate is FP32, broadcasting over hidden
    out = linear(out, weights["to_out.weight"])   # back to dim, still FP32

    return out
