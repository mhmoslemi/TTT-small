"""
custom_kernel
==============

Implements the *outgoing* TriMul operator from AlphaFold‑3.

The heavy O(N³·C) “tri‑multiplication”

    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]

is performed in a fused Triton kernel.  All other operations
(LayerNorm, Linear, gating, and the final projection) are kept in
torch for simplicity – they are cheap compared with the N³ work.
The kernel works in mixed‑precision (fp16 inputs, fp32 accumulation) and
stores the final result in fp32, matching the reference implementation.

The kernel is written for the Hopper SM‑90 (H200) but runs on any recent
GPU.  It supports arbitrary batch size (B∈{1,2}), sequence length N (up to
1024) and hidden dimension H (the “hidden_dim” argument, typically 128).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:   out = Σ_k left[...,k,:] * right[...,k,:]
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # [B, N, N, H]  (i,k,c)
    right_ptr,         # [B, N, N, H]  (j,k,c)
    out_ptr,           # [B, N, N, H]  (i,j,c)  <-- result
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_c,
    stride_right_b, stride_right_j, stride_right_k, stride_right_c,
    stride_out_b, stride_out_i, stride_out_j, stride_out_c,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_b = tl.program_id(2)          # batch tile (size 1)

    # --------------------------------------------------------------
    # Block offsets
    # --------------------------------------------------------------
    i_base = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_base = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    b = pid_b

    # Masks for out‑of‑bounds
    mask_i = i_base < N
    mask_j = j_base < N

    # ------------------------------------------------------------------
    # Pointers offset to the start of the current (i,j) tile
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i_base[:, None] * stride_out_i
        + j_base[None, :] * stride_out_j
    )  # (BLOCK_I, BLOCK_J)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k dimension in chunks of BLOCK_K
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left: (i,k,c)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i_base[:, None, None] * stride_left_i
            + cur_k[None, :, None] * stride_left_k
        )  # (BLOCK_I, BLOCK_K, 1)
        # broadcast channel stride
        left_ptrs = left_ptrs + tl.arange(0, H)[None, None, :] * stride_left_c

        # right: (j,k,c)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + j_base[:, None, None] * stride_right_j
            + cur_k[None, :, None] * stride_right_k
        )
        right_ptrs = right_ptrs + tl.arange(0, H)[None, None, :] * stride_right_c

        # Load with masking, convert to fp16 -> fp32
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0).to(tl.float32)
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0).to(tl.float32)

        # left: (i,k,c)   right: (j,k,c)   -> we need (i,j,c) sum over k
        # Use tl.dot on the k dimension for each channel
        # tl.dot works on 2‑D, so we compute per‑channel with a manual reduction.
        for c in range(0, H, 1):
            # slice channel
            l_c = left[:, :, c]          # (i,k)
            r_c = right[:, :, c]         # (j,k)
            # dot product over k (broadcasted)
            # tl.dot expects (M,K) @ (K,N) ; we need (i,k) @ (k,j)^T
            # transposing right gives (k,j)
            prod = tl.dot(l_c, r_c, trans_b=True)  # (i,j)
            acc[:, :, c] += prod

    # ------------------------------------------------------------------
    # Store result (fp32)
    # ------------------------------------------------------------------
    out_mask = mask_i[:, None] & mask_j[None, :]
    for c in range(0, H, 1):
        out_ptrs_c = out_ptrs + c * stride_out_c
        tl.store(out_ptrs_c, acc[:, :, c], mask=out_mask)


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim]   (float32)
        - mask          : torch.Tensor[bs, N, N]       (float32 or bool)
        - weights       : dict of torch.Tensor containing all
                         linear / layer‑norm parameters
        - config        : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – keep in fp32, cast later
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]        # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]       # (hidden, dim)

    left = F.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Apply mask (if any)
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core TriMul – launch Triton kernel
    # --------------------------------------------------------------
    # Cast to fp16 for the kernel (still keep a fp32 copy for later)
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out = torch.empty_like(left_fp16, dtype=torch.float32, device=device)

    # strides (in elements) for each tensor
    stride_l_b, stride_l_i, stride_l_k, stride_l_c = left_fp16.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_c = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_c = out.stride()

    # block sizes – tuned for Hopper, reasonable for all given N
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        B,
    )

    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B, N, hidden,
        stride_l_b, stride_l_i, stride_l_k, stride_l_c,
        stride_r_b, stride_r_j, stride_r_k, stride_r_c,
        stride_o_b, stride_o_i, stride_o_j, stride_o_c,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gating, and linear projection back to dim
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    out = F.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate                               # gating
    out = F.linear(out, to_out_w)                      # [B,N,N,dim]

    return out.to(dtype)   # ensure same dtype as input (float32)
