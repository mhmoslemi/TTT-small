"""
TriMul “outgoing” implementation.

The expensive part of TriMul is the three‑dimensional
einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix‑multiply for each hidden
dimension *d*:

    out[b,:,:,d] = left[b,:,:,d] @ right[b,:,:,d].transpose(-2,-1)

The kernel below computes this batched mat‑mul for a block of
output rows *i* and columns *j* and a block of hidden channels *d*.
All other operations (layer‑norms, linear projections, gating,
masking) are done in PyTorch – they are cheap compared to the
N³ work and keeping them in PyTorch simplifies handling of bias /
running‑mean statistics.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul over the hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    # pointers
    left_ptr,          # [B, N, N, H]   left  tensor
    right_ptr,         # [B, N, N, H]   right tensor
    out_ptr,           # [B, N, N, H]   output tensor
    # strides (in elements, not bytes)
    stride_blh, stride_lnh, stride_lkh, stride_lh,   # left  strides
    stride_brh, stride_rnh, stride_rkh, stride_rh,   # right strides
    stride_obh, stride_ohn, stride_ojn, stride_oh,   # out   strides
    # problem size
    B, N, H,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute a block of the batched mat‑mul:
        out[b, i:i+BM, j:j+BN, h:h+BH] +=
            Σ_k left[b, i:i+BM, k, h:h+BH] *
            right[b, j:j+BN, k, h:h+BH].transpose(-2,-1)
    """
    pid_b = tl.program_id(0)          # batch dimension
    pid_m = tl.program_id(1)          # output‑row block
    pid_n = tl.program_id(2)          # output‑col block
    pid_h = tl.program_id(3)          # hidden‑channel block

    # ---------------------------------------------
    # offsets for this program instance
    # ---------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    # masks for out‑of‑bounds accesses
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_h = offs_h < H

    # Pointers to the start of the block (batch dimension is fixed)
    left_base  = left_ptr  + pid_b * stride_blh
    right_base = right_ptr + pid_b * stride_brh
    out_base   = out_ptr   + pid_b * stride_obh

    # -----------------------------------------------------------------
    # accumulate into FP32 accumulator (input may be FP16/BF16)
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # loop over K dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # K‑range for this iteration
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -----------------------------------------------------------------
        # load left block  [M, K, H]   (no transpose)
        # -----------------------------------------------------------------
        left_ptrs = (
            left_base
            + offs_m[:, None, None] * stride_lnh
            + offs_k[None, :, None] * stride_lkh
            + offs_h[None, None, :] * stride_lh
        )
        left = tl.load(left_ptrs, mask=mask_m[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0)

        # -----------------------------------------------------------------
        # load right block [N, K, H]  and transpose K & N on‑the‑fly
        # right[b, j, k, h] -> we need right_T[b, k, j, h]
        # -----------------------------------------------------------------
        right_ptrs = (
            right_base
            + offs_n[:, None, None] * stride_rnh
            + offs_k[None, :, None] * stride_rkh
            + offs_h[None, None, :] * stride_rh
        )
        # after load we have shape [N, K, H]; we need [K, N, H] for dot
        right = tl.load(right_ptrs, mask=mask_n[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0)
        right = tl.trans(right, (1, 0, 2))                     # now [K, N, H]

        # -----------------------------------------------------------------
        # compute dot over K dimension (broadcasted over H)
        # acc[i,j,h] += Σ_k left[i,k,h] * right[k,j,h]
        # -----------------------------------------------------------------
        # left:  [M, K, H]   -> (M, K, 1, H)
        # right: [K, N, H]   -> (1, K, N, H)
        # after broadcasting we get (M, K, N, H) and sum over K
        # Triton provides tl.dot which works on the last two axes, so we
        # reshape to collapse H into the dot‑product lanes.
        left_ = left.reshape((BLOCK_M, BLOCK_K, 1, BLOCK_H))
        right_ = right.reshape((1, BLOCK_K, BLOCK_N, BLOCK_H))
        # perform element‑wise multiply + sum over K
        acc += tl.sum(left_ * right_, axis=1)

    # -----------------------------------------------------------------
    # write back
    # -----------------------------------------------------------------
    out_ptrs = (
        out_base
        + offs_m[:, None, None] * stride_ohn
        + offs_n[None, :, None] * stride_ojn
        + offs_h[None, None, :] * stride_oh
    )
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_m[:, None, None] & mask_n[None, :, None] & mask_h[None, None, :])


# ----------------------------------------------------------------------
# Helper: fused layer‑norm (torch implementation – cheap)
# ----------------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]   weight/bias: [C]
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    The heavy N³ computation is carried out by a Triton kernel
    (trimul_matmul_kernel). All linear projections, gating,
    masking and layer‑norms are performed with PyTorch (they are
    cheap compared to the N³ work).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"].to(dtype),
        weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden_dim]
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"].to(dtype)
    )
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"].to(dtype)
    )

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if present) – mask already broadcasted to hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"].to(dtype)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"].to(dtype)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"].to(dtype)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ operation – batched mat‑mul over hidden dim
    # ------------------------------------------------------------------
    # allocate output tensor (float32 accumulator, final cast later)
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides (in number of elements)
    stride_blh, stride_lnh, stride_lkh, stride_lh = left.stride()
    stride_brh, stride_rnh, stride_rkh, stride_rh = right.stride()
    stride_obh, stride_ohn, stride_ojn, stride_oh = out.stride()

    # block sizes – tuned for Hopper SM90 (shared‑memory ≈ 64KB)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 32   # hidden dimension block (hidden_dim is multiple of 32)

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(hidden_dim, BLOCK_H),
    )
    trimul_matmul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        stride_blh, stride_lnh, stride_lkh, stride_lh,
        stride_brh, stride_rnh, stride_rkh, stride_rh,
        stride_obh, stride_ohn, stride_ojn, stride_oh,
        B, N, hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,               # H200 works well with 4 warps per block
    )

    # ------------------------------------------------------------------
    # 6️⃣ to_out_norm (hidden_dim) + gating + final linear
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"].to(dtype),
        weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate

    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(dtype)
    )

    return out.to(torch.float32)
