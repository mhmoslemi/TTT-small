"""
TriMul “outgoing” forward pass (AlphaFold‑3 style).

* LayerNorm, Linear, sigmoid‑gates – done in regular torch (they are cheap
  compared with the O(N³·H) contraction).
* The core contraction  
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]                (1)  
  is a batched matrix‑multiply for every hidden dimension *d*:
        out_d = left_d @ right_d.T
  and is implemented in a custom Triton kernel.  The kernel processes a tile
  of the (i,j) output matrix for a **single** hidden‑dimension slice; the host
  loops over the hidden‑dimension blocks.

The kernel works with float16 accumulation (the inputs are cast to fp16 to
exploit Tensor‑Core throughput on Hopper) and writes the result back in
float32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------------- #
# Triton kernel: one‑dimensional block of the hidden dimension
# ----------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    # pointers
    left_ptr,          # left  [B, N, N, H]
    right_ptr,         # right [B, N, N, H]
    out_ptr,           # out   [B, N, N, H]
    # sizes
    B, N, H,
    # strides (torch uses row‑major)
    stride_blh, stride_blk, stride_blhk,   # left  : (B,N,N,H)  -> (B,N,N,H) strides
    stride_brh, stride_brk, stride_brkh,   # right : (B,N,N,H)  -> (B,N,N,H) strides
    stride_oh, stride_ik, stride_ijk,       # out   : (B,N,N,H)  -> (B,N,N,H) strides
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute a tile of out[b, i, j, h] for a single hidden‑dimension block.
    The computation performed is:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where h runs over a contiguous slice of size BLOCK_H.
    """
    pid = tl.program_id(0)                      # linear index over (B, tile_i, tile_j, tile_h)
    num_h_tiles = tl.cdiv(H, BLOCK_H)

    # decompose linear pid
    pid_h = pid % num_h_tiles
    pid_rest = pid // num_h_tiles
    pid_nj = pid_rest % (tl.cdiv(N, BLOCK_N) * tl.cdiv(N, BLOCK_M))
    pid_b = pid_rest // (tl.cdiv(N, BLOCK_N) * tl.cdiv(N, BLOCK_M))

    tile_i = pid_nj // tl.cdiv(N, BLOCK_N)
    tile_j = pid_nj % tl.cdiv(N, BLOCK_N)

    # offsets for the tile
    i_offset = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    h_offset = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    # bounds
    i_mask = i_offset < N
    j_mask = j_offset < N
    h_mask = h_offset < H

    # pointers for the current batch
    left_batch_ptr  = left_ptr  + pid_b * stride_blh
    right_batch_ptr = right_ptr + pid_b * stride_brh
    out_batch_ptr   = out_ptr   + pid_b * stride_oh

    # initialise accumulator in fp32 (higher precision for sum)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # load left:  [i, k, h]
        left_ptrs = (
            left_batch_ptr
            + i_offset[:, None, None] * stride_blk
            + k_offset[None, :, None] * stride_blhk
            + h_offset[None, None, :] * stride_blh
        )
        left = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :] & h_mask[None, None, :], other=0.0)
        left = left.to(tl.float16)        # cast to fp16 for dot

        # load right (need transpose on the last two dims): right[j, k, h]
        right_ptrs = (
            right_batch_ptr
            + j_offset[:, None, None] * stride_brk
            + k_offset[None, :, None] * stride_brkh
            + h_offset[None, None, :] * stride_brh
        )
        right = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_mask[None, :] & h_mask[None, None, :], other=0.0)
        right = right.to(tl.float16)

        # accumulate: Σ_k left[i,k,h] * right[j,k,h]
        #   we can use tl.dot on the k dimension for each (i,j) pair and each h.
        #   Because tl.dot works on 2‑D matrices, we flatten the h dimension.
        left_flat  = left.reshape(BLOCK_M, BLOCK_K * BLOCK_H)
        right_flat = right.reshape(BLOCK_N, BLOCK_K * BLOCK_H)

        # dot over K*H (still small enough for Tensor‑Core)
        acc += tl.dot(left_flat, right_flat, out_dtype=tl.float32).reshape(BLOCK_M, BLOCK_N, BLOCK_H)

    # store result
    out_ptrs = (
        out_batch_ptr
        + i_offset[:, None, None] * stride_ik
        + j_offset[None, :, None] * stride_ijk
        + h_offset[None, None, :] * stride_oh
    )
    tl.store(out_ptrs, acc, mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :])


# ----------------------------------------------------------------------------- #
# Helper: fused LayerNorm (no bias handling because bias is passed separately)
# ----------------------------------------------------------------------------- #
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    norm = (x - mean) * torch.rsqrt(var + eps)
    return norm * weight + bias


# ----------------------------------------------------------------------------- #
# Main entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32, on CUDA)
        - mask         : [B, N, N]        (float32 or bool, on CUDA)
        - weights      : dict of torch.Parameters as described in the prompt
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # --------------------------------------------------------------------- #
    # 1. LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------------- #
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float32)

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias) -> shape [B,N,N,hidden_dim]
    # --------------------------------------------------------------------- #
    #   left_proj = x @ W_left^T , right_proj = x @ W_right^T
    #   we implement the matmul via torch.einsum which is cheap compared
    #   to the O(N³·H) contraction.
    left = torch.einsum('bijk,dh->bijd', x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = torch.einsum('bijk,dh->bijd', x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    # 3. Apply mask (if present)
    # --------------------------------------------------------------------- #
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------------- #
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(torch.einsum('bijk,dh->bijd', x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.einsum('bijk,dh->bijd', x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.einsum('bijk,dh->bijd', x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Core contraction – Triton kernel
    # --------------------------------------------------------------------- #
    # Cast to fp16 for the kernel (Tensor‑Core friendly)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output buffer
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides in elements (torch uses contiguous layout)
    stride_blh, stride_blk, stride_blhk = left_fp16.stride()
    stride_brh, stride_brk, stride_brkh = right_fp16.stride()
    stride_oh, stride_ik, stride_ijk = out.stride()

    # Tile sizes – chosen to fit in H200 shared memory and to hit Tensor‑Core
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 32   # hidden‑dim block; 32 fits nicely with fp16 matmul units

    # Number of program instances (grid)
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid_h = triton.cdiv(hidden_dim, BLOCK_H)
    total_programs = B * grid_m * grid_n * grid_h
    grid = (total_programs,)

    # launch
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B, N, hidden_dim,
        stride_blh, stride_blk, stride_blhk,
        stride_brh, stride_brk, stride_brkh,
        stride_oh, stride_ik, stride_ijk,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,               # empirical good number on Hopper
    )

    # --------------------------------------------------------------------- #
    # 6. Post‑processing: LayerNorm + gating + final linear
    # --------------------------------------------------------------------- #
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.einsum('bijd,dh->bijk', out, weights["to_out.weight"])
    return out.to(torch.float32)
