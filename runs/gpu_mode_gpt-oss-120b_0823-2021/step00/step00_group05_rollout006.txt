"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where
    left  = (x @ W_left)   * mask * sigmoid(x @ W_left_gate)
    right = (x @ W_right)  * mask * sigmoid(x @ W_right_gate)

For a fixed batch `b` and hidden channel `h` this is exactly a matrix‑multiply
    (N×N) @ (N×N)^T  → (N×N)

Consequently we can fuse the whole reduction in a single Triton kernel.
The kernel computes a tiled GEMM for every (batch, hidden) pair, using
fp16 accumulation (the model works in fp16/bf16) and writes the result back
in fp32.  All surrounding LayerNorms, linear projections and gating are
performed with regular PyTorch – they are cheap compared with the N³ work.

The public entry‑point `custom_kernel` follows the required signature and
returns a float32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM across (batch, hidden) dim
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_out_kernel(
    # pointers
    left_ptr,            # *[B, N, N, H]  (i,k)
    right_ptr,           # *[B, N, N, H]  (j,k)
    out_ptr,             # *[B, N, N, H]  (i,j)
    # sizes
    B, N, H,
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    # fast‑math mode
    FP16_ACC: tl.constexpr,
):
    # program ids
    pid_batch = tl.program_id(0)                # 0 … B*H-1
    pid_m = tl.program_id(1)                    # tiles along i (M)
    pid_n = tl.program_id(2)                    # tiles along j (N)

    # decode batch and hidden index
    batch_idx = pid_batch // H
    h_idx     = pid_batch % H

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the start of the tile in left and right (both have k as inner dim)
    left_tile_ptr  = left_ptr \
        + batch_idx * stride_left_b \
        + h_idx    * stride_left_h \
        + offs_m[:, None] * stride_left_i \
        + tl.arange(0, BLOCK_K)[None, :] * stride_left_k

    right_tile_ptr = right_ptr \
        + batch_idx * stride_right_b \
        + h_idx    * stride_right_h \
        + offs_n[None, :] * stride_right_j \
        + tl.arange(0, BLOCK_K)[:, None] * stride_right_k

    # accumulate in fp32 or fp16 (if FP16_ACC True)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        kp = k * BLOCK_K
        cur_k = tl.arange(0, BLOCK_K) + kp
        mask_k = cur_k < N

        # load left[i,k] and right[j,k] (note the transpose on right)
        left = tl.load(left_tile_ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)
        # right is stored as (j,k); we need (k,j) → load then transpose
        right = tl.load(right_tile_ptr,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)
        right = tl.trans(right)   # now shape (BLOCK_K, BLOCK_N)

        # dot‑product
        if FP16_ACC:
            # cast to fp16 for dot, accumulate in fp32
            left = left.to(tl.float16)
            right = right.to(tl.float16)
        acc += tl.dot(left, right)

        # advance pointers
        left_tile_ptr  += BLOCK_K * stride_left_k
        right_tile_ptr += BLOCK_K * stride_right_k

    # final store (output expected in fp32)
    out_tile_ptr = out_ptr \
        + batch_idx * stride_out_b \
        + h_idx    * stride_out_h \
        + offs_m[:, None] * stride_out_i \
        + offs_n[None, :] * stride_out_j

    tl.store(out_tile_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper: layer‑norm without bias/weight (the bias/weight are applied manually)
# ----------------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   (float32)
            - mask         : [B, N, N]        (float32 or bool)
            - weights      : dict of tensors (already on CUDA)
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        out : torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – keep in fp32, cast later
    # ------------------------------------------------------------------
    # shape: (dim, hidden)
    left_proj_w  = weights["left_proj.weight"]          # [hidden, dim]
    right_proj_w = weights["right_proj.weight"]

    # torch.einsum for dense matmul across last dim
    left = torch.einsum('bnid,dh->bnih', x, left_proj_w.t())   # [B,N,N,hidden]
    right = torch.einsum('bnid,dh->bnih', x, right_proj_w.t())

    # ------------------------------------------------------------------
    # 3) Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)            # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(torch.einsum('bnid,dh->bnih', x, left_gate_w.t()))
    right_gate = torch.sigmoid(torch.einsum('bnid,dh->bnih', x, right_gate_w.t()))
    out_gate   = torch.sigmoid(torch.einsum('bnid,dh->bnih', x, out_gate_w.t()))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton kernel: out = Σ_k left[...,i,k,h] * right[...,j,k,h]
    # ------------------------------------------------------------------
    # Cast to fp16 for the compute‑intensive part
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out = torch.empty_like(left_fp16)   # [B,N,N,hidden] fp16 buffer

    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_fp16.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – empirically good for Hopper (sm90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # grid: (B*hidden, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        B * hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_out_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B, N, hidden,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        FP16_ACC=True,               # accumulate in fp32 internally
        num_warps=4,
    )

    # back to fp32 for the remaining ops
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) to_out_norm + gating + final linear
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # final projection (hidden → dim)
    to_out_w = weights["to_out.weight"]          # [dim, hidden]
    out = torch.einsum('bnih,dh->bnid', out, to_out_w.t())   # [B,N,N,dim]

    return out
