"""
TriMul “outgoing” implementation.

The heavy part of the module is the ternary einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batch of hidden‑dim (d) matrix‑multiplications:

    out_d[b] = left_d[b] @ right_d[b].T          (left_d : N×N, right_d : N×N)

The kernel below computes a block of (i,j) for a block of hidden
dimensions d.  All LayerNorm / Linear / gating operations are done in
PyTorch (they are cheap compared with the N³ work).  The kernel
accumulates the result in FP32 and writes the final tensor in FP32
as required by the specification.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ------------------------------------------------------------------
# Triton kernel: computes Σ_k left[...,k,:] * right[...,k,:]
# ------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # * const float16 *
    right_ptr,         # * const float16 *
    out_ptr,           # * float32 *
    B, N, D,           # runtime constants
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_D: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    One program instance works on a tile (BLOCK_M x BLOCK_N) of the
    (i,j) matrix for a tile of hidden dimensions (BLOCK_D).
    It iterates over k in chunks of BLOCK_K and accumulates the
    product left[b,i,k,d] * right[b,j,k,d].
    """

    pid_b = tl.program_id(0)          # batch index
    pid_m = tl.program_id(1)          # i‑tile index
    pid_n = tl.program_id(2)          # j‑tile index

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers with full 4‑D indexing
        left_ptrs = (
            left_ptr
            + pid_b * stride_lb
            + offs_i[:, None, None] * stride_li
            + offs_k[None, :, None] * stride_lk
            + offs_d[None, None, :] * stride_ld
        )
        right_ptrs = (
            right_ptr
            + pid_b * stride_rb
            + offs_j[:, None, None] * stride_rj
            + offs_k[None, :, None] * stride_rk
            + offs_d[None, None, :] * stride_rd
        )

        # Load a (M x K x D) and (N x K x D) tile – fp16
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # Multiply and reduce over k
        # left : (M, K, D)   right : (N, K, D)
        # we want   Σ_k left[m,k,d] * right[n,k,d]   → (M,N,D)
        # This is implemented as an explicit reduction over the K‑axis.
        acc += tl.sum(left * right, axis=1)   # (M,N,D)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_ob
        + offs_i[:, None, None] * stride_oi
        + offs_j[None, :, None] * stride_oj
        + offs_d[None, None, :] * stride_od
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )

# ------------------------------------------------------------------
# Helper : LayerNorm (no bias fusion – use PyTorch)
# ------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_hat = (x - mean) * torch.rsqrt(var + eps)
    return x_hat * weight + bias

# ------------------------------------------------------------------
# Entry point
# ------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Only the O(N³) reduction is executed on the GPU via Triton.
    All other linear / gating layers are performed with PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1.  LayerNorm on input (dim)
    # --------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2.  Linear projections (no bias) – cast to fp16 for the kernel
    # --------------------------------------------------------------
    #   left = x @ left_proj.T   (shape B,N,N,hidden_dim)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"]).to(torch.float16)
    right = torch.nn.functional.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # --------------------------------------------------------------
    # 3.  Apply mask if present
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4.  Gating (sigmoid) – stays in fp32, then cast to fp16
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    ).to(torch.float16)
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    ).to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5.  Triton kernel – compute Σ_k left * right  (output in fp32)
    # --------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # kernel launch configuration
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_D = 32                     # hidden dim is a multiple of 32 in all tests
    BLOCK_K = 32

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # strides (in element units) – convert to int64 for Triton
    s_lb, s_li, s_lk, s_ld = left.stride()
    s_rb, s_rj, s_rk, s_rd = right.stride()
    s_ob, s_oi, s_oj, s_od = out.stride()

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=hidden_dim,
        stride_lb=s_lb,
        stride_li=s_li,
        stride_lk=s_lk,
        stride_ld=s_ld,
        stride_rb=s_rb,
        stride_rj=s_rj,
        stride_rk=s_rk,
        stride_rd=s_rd,
        stride_ob=s_ob,
        stride_oi=s_oi,
        stride_oj=s_oj,
        stride_od=s_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_D=BLOCK_D,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6.  Output LayerNorm + gating + final linear (all fp32)
    # --------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )
    out = out * out_gate.unsqueeze(1).unsqueeze(2)      # broadcast over i,j
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.float()
