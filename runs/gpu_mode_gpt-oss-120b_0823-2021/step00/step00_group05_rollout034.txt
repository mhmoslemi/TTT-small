"""
TriMul "outgoing" implementation (forward only).

Algorithm
---------
1. Layer‑norm + linear + gating are done in PyTorch (they are cheap compared
   to the O(N³) pairwise product).
2. The core operation is
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication for each hidden channel `d`.
3. A Triton kernel computes this triple product.  For a tile (BM×BN) of the
   output we accumulate over K‑tiles, loading a (BM×BK) slice of `left`
   and a (BN×BK) slice of `right` (the latter is accessed transposed so that
   the dot‑product is over the K‑dimension).  The computation is performed
   for a small chunk of the hidden dimension (CH) at a time; this keeps the
   per‑thread register pressure low and allows the use of `tl.dot` which
   maps to Tensor‑Core MMA on Hopper.

The kernel returns float32 results, while intermediate loads are performed
in float16 (or bfloat16) to increase bandwidth.  All other operations remain
in float32, matching the reference implementation.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel that computes the triple product for a chunk of the hidden dim.
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr,        # [B, N, N, CH]
    right_ptr,       # [B, N, N, CH] (will be read transposed on the last two axes)
    out_ptr,         # [B, N, N, CH]
    # sizes
    B, N, CH,
    # strides (in elements, not bytes)
    stride_blh, stride_lhn, stride_lnk, stride_lch,   # left  (B,N,K,CH)
    stride_brh, stride_rhn, stride_rnk, stride_rch,   # right (B,N,K,CH) – we will read as (B,K,N,CH)
    stride_oh, stride_ohn, stride_onk, stride_och,    # out   (B,N,N,CH)
    # tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch dimension
    pid_m = tl.program_id(1)          # output i‑tile
    pid_n = tl.program_id(2)          # output j‑tile
    pid_c = tl.program_id(3)          # hidden‑channel chunk

    # --------------------------------------------------------------
    # compute global offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_c = pid_c * BLOCK_C + tl.arange(0, BLOCK_C)

    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_c = offs_c < CH

    # pointers to the beginning of the current batch
    left_batch_ptr  = left_ptr   + pid_b * stride_blh
    right_batch_ptr = right_ptr  + pid_b * stride_brh
    out_batch_ptr   = out_ptr    + pid_b * stride_oh

    # --------------------------------------------------------------
    # accumulate over K
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_C), dtype=tl.float32)

    # loop over k tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # ---------- load left: shape (BM, BK, BC) ----------
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left[b, i, k, c]
        left_ptrs = (
            left_batch_ptr
            + offs_m[:, None] * stride_lhn
            + cur_k[None, :] * stride_lnk
            + offs_c[None, None, :] * stride_lch
        )
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :] & mask_c[None, None, :],
                       other=0.0).to(tl.float16)

        # ---------- load right (transposed): shape (BN, BK, BC) ----------
        # we need right[b, j, k, c] -> we access as right[b, k, j, c]
        right_ptrs = (
            right_batch_ptr
            + cur_k[:, None] * stride_rhn      # k dimension becomes first
            + offs_n[None, :] * stride_rnk      # j becomes second (original N)
            + offs_c[None, None, :] * stride_rch
        )
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :] & mask_c[None, None, :],
                        other=0.0).to(tl.float16)

        # ---------- dot over K ----------
        # left: (BM, BK, BC)   right: (BN, BK, BC)
        # we compute for each channel independently via tl.dot
        for c in range(0, BLOCK_C):
            # extract the c‑th channel slice
            l = left[:, :, c]          # (BM, BK)
            r = right[:, :, c]          # (BN, BK)
            # tl.dot expects (M,K) @ (K,N) -> (M,N)
            # right needs to be transposed on K dimension
            acc[:, :, c] += tl.dot(l, r, allow_tf32=False)

    # --------------------------------------------------------------
    # write result
    # --------------------------------------------------------------
    out_ptrs = (
        out_batch_ptr
        + offs_m[:, None] * stride_ohn
        + offs_n[None, :] * stride_onk
        + offs_c[None, None, :] * stride_och
    )
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :] & mask_c[None, None, :])


# ----------------------------------------------------------------------
# Helper functions (layer‑norm + linear + gating) – stay in PyTorch.
# ----------------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    # x: (..., C)
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]   (float32)
            - mask:         [B, N, N]        (float32 or bool)
            - weights: dict of model parameters (all float32)
            - config:  {"dim": int, "hidden_dim": int}
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = _layer_norm(input_tensor,
                    weight=weights["norm.weight"],
                    bias=weights["norm.bias"])

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free) and gating
    # ------------------------------------------------------------------
    # Linear: view as (B*N*N, dim) matmul weight^T
    x_flat = x.reshape(-1, dim)

    left = torch.nn.functional.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden)
    right = torch.nn.functional.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden)

    left_gate = torch.nn.functional.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden).sigmoid()

    # apply mask if present
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                         # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # gating
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3️⃣ Triton kernel for the triple product
    # ------------------------------------------------------------------
    # allocate output
    out = torch.empty_like(left)   # shape [B,N,N,hidden], float32

    # strides (in elements)
    stride_blh, stride_lhn, stride_lnk, stride_lch = left.stride()
    stride_brh, stride_rhn, stride_rnk, stride_rch = right.stride()
    stride_oh, stride_ohn, stride_onk, stride_och = out.stride()

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_C = 32   # channel chunk, must divide hidden (hidden <= 384)

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(hidden, BLOCK_C),
    )

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, CH=hidden,
        stride_blh=stride_blh,
        stride_lhn=stride_lhn,
        stride_lnk=stride_lnk,
        stride_lch=stride_lch,
        stride_brh=stride_brh,
        stride_rhn=stride_rhn,
        stride_rnk=stride_rnk,
        stride_rch=stride_rch,
        stride_oh=stride_oh,
        stride_ohn=stride_ohn,
        stride_onk=stride_onk,
        stride_och=stride_och,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K, BLOCK_C=BLOCK_C,
        num_warps=4,  # enough for the compute intensity
    )

    # ------------------------------------------------------------------
    # 4️⃣ Final LayerNorm, gating and linear projection back to dim
    # ------------------------------------------------------------------
    out = _layer_norm(out,
                      weight=weights["to_out_norm.weight"],
                      bias=weights["to_out_norm.bias"])

    out = out * out_gate

    # final linear back to dim
    out_flat = out.reshape(-1, hidden)
    out = torch.nn.functional.linear(out_flat, weights["to_out.weight"]).view(B, N, N, dim)

    return out
