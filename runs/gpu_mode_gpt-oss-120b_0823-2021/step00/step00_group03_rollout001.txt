"""
TriMul “outgoing” implementation.

The heavy O(N³) part is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed head dimension d this is exactly a matrix‑multiply:
    out_d = left_d @ right_d.T   (shape B×N×N)

We keep the cheap LayerNorm / Linear / gating logic in PyTorch and
off‑load the per‑head matrix multiplication to a fused Triton kernel.
The kernel works on FP16 (mixed‑precision) and writes the result back as
FP32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B shape: B × N × N)
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N,
    stride_Ab, stride_Ak, stride_Al,   # A: (batch, i, k)
    stride_Bb, stride_Bk, stride_Bl,   # B: (batch, j, k)
    stride_Cb, stride_Ci, stride_Cj,   # C: (batch, i, j)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    FP16: tl.constexpr,
):
    pid = tl.program_id(0)
    # program id maps to (batch, block_i, block_j)
    b = pid // (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_rest = pid % (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_i = pid_rest // tl.cdiv(N, BLOCK_N)
    block_j = pid_rest % tl.cdiv(N, BLOCK_N)

    # offsets inside the block
    i_offset = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_i = i_offset < N
    mask_j = j_offset < N

    # pointers to the start of the (b, i, k) and (b, j, k) tiles
    a_ptrs = a_ptr + b * stride_Ab + i_offset[:, None] * stride_Ai + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = b_ptr + b * stride_Bb + j_offset[:, None] * stride_Bj + tl.arange(0, BLOCK_K)[None, :] * stride_Bk

    # accumulation buffer
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < N), other=0.0)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < N), other=0.0)
        # convert to FP16 if requested (improves bandwidth)
        if FP16:
            a = a.to(tl.float16)
            b = b.to(tl.float16)
        acc += tl.dot(a, b)          # dot over K (BLOCK_K)
        # advance pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # write result
    c_ptrs = c_ptr + b * stride_Cb + i_offset[:, None] * stride_Ci + j_offset[None, :] * stride_Cj
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Helper to launch the kernel for one head (d) slice
# ----------------------------------------------------------------------
def batch_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a, b : torch.Tensor of shape [B, N, N]  (float16)
    returns c : torch.Tensor of shape [B, N, N] (float32)
    """
    B, N, _ = a.shape
    c = torch.empty_like(a, dtype=torch.float32, device=a.device)

    # strides in elements (not bytes)
    stride_Ab, stride_Ai, stride_Ak = a.stride()
    stride_Bb, stride_Bj, stride_Bk = b.stride()
    stride_Cb, stride_Ci, stride_Cj = c.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),
    )
    batch_matmul_kernel[grid](
        a, b, c,
        B, N,
        stride_Ab, stride_Ak, stride_Ai,
        stride_Bb, stride_Bk, stride_Bj,
        stride_Cb, stride_Ci, stride_Cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        FP16=True,
    )
    return c

# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : [B, N, N, dim]   (float32, on CUDA)
        mask        : [B, N, N]        (float32 or bool, on CUDA)
        weights     : dict of torch.Tensor containing all linear / LN params
        config      : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm -------------------------------------------------
    # x = LN(x)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----- Linear projections (no bias) -------------------------------
    # shape after proj: [B, N, N, hidden_dim]
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Apply mask (if present) -----------------------------------
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ----- Gating -----------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ----- Core O(N^3) einsum via Triton -------------------------------
    # We'll compute per‑head (dim -> hidden_dim) matmul.
    # Convert to fp16 for the kernel (saves bandwidth)
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output tensor (float32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # loop over hidden dimension in blocks for better occupancy
    BLOCK_D = 32
    for d_start in range(0, hidden_dim, BLOCK_D):
        d_end = min(d_start + BLOCK_D, hidden_dim)
        # slice the head dimension
        left_slice  = left_fp16[..., d_start:d_end]   # [B,N,N,blk]
        right_slice = right_fp16[..., d_start:d_end]  # [B,N,N,blk]

        # reshape to (blk, B, N, N) -> compute each block independently
        # we iterate over the block dimension in Python because the kernel
        # handles one head at a time (simpler and still fast).
        for h in range(d_end - d_start):
            a = left_slice[..., h]   # [B,N,N]
            b = right_slice[..., h]  # [B,N,N]
            c = batch_matmul(a, b)   # [B,N,N] (float32)
            out[..., d_start + h] = c

    # ----- Output processing (LN, gating, final linear) ---------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
