"""
TriMul (outgoing) – fast Triton implementation of the heavy
pair‑wise contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right are the projected, gated and masked tensors.
All cheap operations (LayerNorm, Linear, gating, final Linear) are
performed in PyTorch.  The N³ contraction is executed by a fused
batched‑matmul kernel written in Triton.  The kernel tiles the
(k)‑dimension to keep shared‑memory usage low and uses FP16
accumulators for speed, converting the final result back to FP32.
The output tensor remains FP32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched matmul over the “k” dimension
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               #   [B, N, N, H]  (i,k)
    right_ptr,              #   [B, N, N, H]  (j,k)
    out_ptr,                #   [B, N, N, H]  (i,j)
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,  # tile size for i
    BLOCK_N: tl.constexpr,  # tile size for j
    BLOCK_K: tl.constexpr,  # tile size for k
):
    pid = tl.program_id(0)                     # each program handles one (b, h)
    b = pid // H
    h = pid % H

    # offsets for the tile of the output matrix
    offs_i = tl.arange(0, BLOCK_M)
    offs_j = tl.arange(0, BLOCK_N)

    # start indices (i0, j0) for this tile
    i0 = tl.program_id(1) * BLOCK_M
    j0 = tl.program_id(2) * BLOCK_N

    # mask out-of-bounds
    mask_i = i0 + offs_i < N
    mask_j = j0 + offs_j < N

    # compute base pointers for this (b,h) slice
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator in FP32 (we will load in FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k0 in range(0, N, BLOCK_K):
        cur_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # load left:  [M, K]
        left_ptrs = left_base \
            + (i0 + offs_i[:, None]) * stride_left_i \
            + (cur_k[None, :]) * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # load right:  [N, K]   (note: right is indexed (j,k) – we need the same K order)
        right_ptrs = right_base \
            + (j0 + offs_j[None, :]) * stride_right_j \
            + (cur_k[:, None]) * stride_right_k
        right = tl.load(right_ptrs, mask=mask_j[None, :] & mask_k[:, None], other=0.0).to(tl.float16)

        # dot‑product over the K dimension
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # write back
    out_ptrs = out_base \
        + (i0 + offs_i[:, None]) * stride_out_i \
        + (j0 + offs_j[None, :]) * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Helper to launch the kernel
# --------------------------------------------------------------
def launch_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]   (FP32)
    returns out : [B, N, N, H]    (FP32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in element units (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H,                     # program_id(0) – batch * heads
            triton.cdiv(N, BLOCK_M),  # program_id(1) – i‑tiles
            triton.cdiv(N, BLOCK_N))  # program_id(2) – j‑tiles

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                # good default on H200
    )
    return out


# --------------------------------------------------------------
# Full custom kernel entry‑point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32/float16)
        - mask         : torch.Tensor [B, N, N]      (bool or 0/1)
        - weights      : dict of parameter tensors
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # cheap ops in PyTorch (LayerNorm, Linear, gating, masking)
    # ------------------------------------------------------------------
    # LayerNorm over the last dimension
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # mask handling
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # gating – sigmoid after linear
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # heavy N³ contraction – Triton batched matmul
    # ------------------------------------------------------------------
    # Cast to FP16 for the inner kernel (H200 has excellent FP16 throughput)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # compute out in FP16 inside the kernel, then cast back
    out_fp16 = launch_tri_mul(left_fp16, right_fp16)

    out = out_fp16.to(torch.float32)          # back to FP32 for the rest

    # ------------------------------------------------------------------
    # post‑processing (LayerNorm, gating, final linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # gating
    out = F.linear(out, weights["to_out.weight"])

    return out
