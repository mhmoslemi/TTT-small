"""
tri_mul_outgoing_kernel.py

Implementation of the “outgoing” TriMul operator (AlphaFold‑3) in Triton.
Only the O(N³·C) reduction (the einsum) is performed on the GPU;
pre‑/post‑processing (LayerNorms, linear projections, gating) stay in
torch for simplicity and to keep the code short.

The kernel computes

    out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]

for a batch of size B, sequence length N and hidden dimension C (= hidden_dim).
The kernel works block‑wise over the (i,j) plane and accumulates over k.
It processes C in chunks (BLOCK_C) so that a single dot‑product can be used
(`tl.dot`) which is highly optimized on Hopper (SM‑90).
The final result is summed into a float32 accumulator and written back
as float32 (the rest of the model uses fp32).

The host function `custom_kernel` assembles the full TriMul forward pass,
loads the provided weights into the corresponding torch modules and
launches the Triton kernel for the expensive reduction.
"""
import torch
import triton
import triton.language as tl
from torch import nn

# ----------------------------------------------------------------------
# Triton kernel for the einsum reduction
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,          # [B, N, N, C]  (already projected, gated, masked)
    right_ptr,         # [B, N, N, C]  (already projected, gated, masked)
    out_ptr,           # [B, N, N, C]  (output accumulator)
    # strides
    stride_lb, stride_lk, stride_lc,   # left  : B,N,N,C
    stride_rb, stride_rk, stride_rc,   # right : B,N,N,C
    stride_ob, stride_ok, stride_oc,   # out   : B,N,N,C
    # sizes
    B, N, C,
    # compile‑time constants
    BLOCK_M: tl.constexpr,   # tile size over i
    BLOCK_N: tl.constexpr,   # tile size over j
    BLOCK_K: tl.constexpr,   # tile size over k (reduction)
    BLOCK_C: tl.constexpr,   # tile size over channel dimension
):
    """Compute out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]."""

    pid = tl.program_id(0)
    # 2‑D grid over (i, j)
    pid_i = pid // tl.cdiv(N, BLOCK_N)
    pid_j = pid % tl.cdiv(N, BLOCK_N)

    # start indices for this tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # range of i/j indices handled by this program instance
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # broadcast batch dimension
    offs_b = tl.arange(0, B)[:, None, None]

    # ------------------------------------------------------------------
    # Accumulator over k and C
    # ------------------------------------------------------------------
    acc = tl.zeros((B, BLOCK_M, BLOCK_N, BLOCK_C), dtype=tl.float32)

    # loop over k in tiles of BLOCK_K
    for k_start in range(0, N, BLOCK_K):
        # k‑range for this tile
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # --------------------------------------------------------------
        # Load left  : [B, i, k, c]
        # --------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + offs_b * stride_lb
            + offs_i[:, None, None] * stride_lk
            + offs_k[None, :, None] * stride_lc
        )  # shape (B, BLOCK_M, BLOCK_K, 1)
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & (tl.arange(0, BLOCK_C) < C),
            other=0.0,
        )
        # left shape: (B, BLOCK_M, BLOCK_K, BLOCK_C) after broadcast

        # --------------------------------------------------------------
        # Load right : [B, j, k, c]  (note the order of k and j)
        # --------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + offs_b * stride_rb
            + offs_j[None, :, None] * stride_rk
            + offs_k[None, None, :] * stride_rc
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[None, :, None] & mask_k[None, None, :] & (tl.arange(0, BLOCK_C) < C),
            other=0.0,
        )
        # right shape: (B, BLOCK_N, BLOCK_K, BLOCK_C)

        # --------------------------------------------------------------
        # Compute contribution for this k‑tile
        # We need Σₖ left[...,k,c] * right[...,k,c] per (i,j,c).
        # This can be expressed as a dot product over the K dimension for each
        # channel chunk.
        # --------------------------------------------------------------
        # reshape to (B, BLOCK_M, BLOCK_K, BLOCK_C) and (B, BLOCK_N, BLOCK_K, BLOCK_C)
        # Loop over channel sub‑tiles
        for c_start in range(0, C, BLOCK_C):
            c_end = tl.minimum(c_start + BLOCK_C, C)
            c_slice = tl.arange(0, BLOCK_C)[: c_end - c_start]

            # slice channel dimension
            left_c = left[..., c_slice]   # (B, BLOCK_M, BLOCK_K, c_tile)
            right_c = right[..., c_slice] # (B, BLOCK_N, BLOCK_K, c_tile)

            # move channel dim outward so that we can dot over K
            # left_c: (B, BLOCK_M, BLOCK_K, c_tile) -> (B, c_tile, BLOCK_M, BLOCK_K)
            # right_c: (B, BLOCK_N, BLOCK_K, c_tile) -> (B, c_tile, BLOCK_N, BLOCK_K)
            left_c = tl.trans(left_c, (0, 3, 1, 2))
            right_c = tl.trans(right_c, (0, 3, 1, 2))

            # iterate over channel tiles
            for ci in range(c_end - c_start):
                # view as 2‑D matrices for dot
                l = left_c[:, ci, :, :]   # (B, BLOCK_M, BLOCK_K)
                r = right_c[:, ci, :, :]  # (B, BLOCK_N, BLOCK_K)

                # dot over K => (B, BLOCK_M, BLOCK_N)
                # tl.dot works on 2‑D matrices; we expand batch via loop
                for b in range(B):
                    acc[b, :, :, ci] += tl.dot(l[b], r[b])

    # ------------------------------------------------------------------
    # Write back accumulator
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + offs_b * stride_ob
        + offs_i[:, None, None] * stride_ok
        + offs_j[None, :, None] * stride_oc
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & (tl.arange(0, BLOCK_C) < C),
    )


# ----------------------------------------------------------------------
# Host‑side helper that launches the kernel
# ----------------------------------------------------------------------
def _launch_trimul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, C]   (float32)
    Returns      : out  [B, N, N, C] (float32)
    """
    B, N, _, C = left.shape
    out = torch.empty_like(left)

    # stride information (in elements)
    stride_lb, stride_lk, stride_lc = left.stride()
    stride_rb, stride_rk, stride_rc = right.stride()
    stride_ob, stride_ok, stride_oc = out.stride()

    # Tuning parameters – chosen to give good occupancy on H200
    BLOCK_M = 32   # tiles over i
    BLOCK_N = 32   # tiles over j
    BLOCK_K = 64   # reduction tile over k
    BLOCK_C = 16   # channel tile (must divide hidden_dim for best perf)

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (grid_m * grid_n,)

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        stride_lb,
        stride_lk,
        stride_lc,
        stride_rb,
        stride_rk,
        stride_rc,
        stride_ob,
        stride_ok,
        stride_oc,
        B,
        N,
        C,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_C=BLOCK_C,
        num_warps=4,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Full TriMul forward pass (mostly torch, kernel for the heavy reduction)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim]  torch.float32 on cuda
            - mask_tensor  : [B, N, N]       torch.float32 or bool on cuda
            - weights_dict : mapping str → torch.Tensor (already on cuda)
            - config_dict  : {"dim": int, "hidden_dim": int}
    Returns:
        output tensor [B, N, N, dim] (float32, cuda)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dim (dim)
    # ------------------------------------------------------------------
    norm = nn.LayerNorm(dim, elementwise_affine=True).cuda()
    norm.weight.data = weights["norm.weight"]
    norm.bias.data   = weights["norm.bias"]
    x = norm(input_tensor)                      # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) + mask
    # ------------------------------------------------------------------
    # Linear layers without bias can be expressed as a matrix‑multiply on the
    # last dimension.
    left_proj = nn.Linear(dim, hidden_dim, bias=False).cuda()
    right_proj = nn.Linear(dim, hidden_dim, bias=False).cuda()
    left_proj.weight.data = weights["left_proj.weight"]
    right_proj.weight.data = weights["right_proj.weight"]

    left = left_proj(x)   # [B,N,N,hidden_dim]
    right = right_proj(x) # [B,N,N,hidden_dim]

    # apply mask (broadcast on hidden_dim)
    mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
    left = left * mask_
    right = right * mask_

    # ------------------------------------------------------------------
    # 3) Gating (sigmoid) – also linear without bias
    # ------------------------------------------------------------------
    left_gate_lin = nn.Linear(dim, hidden_dim, bias=False).cuda()
    right_gate_lin = nn.Linear(dim, hidden_dim, bias=False).cuda()
    out_gate_lin = nn.Linear(dim, hidden_dim, bias=False).cuda()
    left_gate_lin.weight.data = weights["left_gate.weight"]
    right_gate_lin.weight.data = weights["right_gate.weight"]
    out_gate_lin.weight.data = weights["out_gate.weight"]

    left_gate = left_gate_lin(x).sigmoid()
    right_gate = right_gate_lin(x).sigmoid()
    out_gate = out_gate_lin(x).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Core reduction (einsum) using Triton
    # ------------------------------------------------------------------
    out = _launch_trimul_outgoing(left, right)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 5) Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    to_out_norm = nn.LayerNorm(hidden_dim, elementwise_affine=True).cuda()
    to_out_norm.weight.data = weights["to_out_norm.weight"]
    to_out_norm.bias.data   = weights["to_out_norm.bias"]
    out = to_out_norm(out)                      # [B,N,N,hidden_dim]

    out = out * out_gate                        # gating

    to_out = nn.Linear(hidden_dim, dim, bias=False).cuda()
    to_out.weight.data = weights["to_out.weight"]
    out = to_out(out)                           # [B,N,N,dim]

    return out.to(torch.float32)
