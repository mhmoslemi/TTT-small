"""
TriMul “outgoing” kernel (AlphaFold‑3).  
The expensive part is the 4‑D tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which we rewrite as a batched matrix multiplication per hidden dimension *d*:

    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T

The kernel computes this mat‑mul for all batches and hidden dimensions in a
single launch.  All other operations (LayerNorm, Linear, gating, final
normalisation) are performed with regular PyTorch because they are cheap
relative to the O(B·D·N³) contraction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,                      # const float*  (B,N,N,D)
    right_ptr,                     # const float*  (B,N,N,D)
    out_ptr,                       #       float*  (B,N,N,D)
    B, N, D,                       # int32 sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr,         # tile rows   of output (i)
    BLOCK_N: tl.constexpr,         # tile cols   of output (j)
    BLOCK_K: tl.constexpr,         # reduction tile over k
):
    """
    Each program instance processes a tile (BLOCK_M x BLOCK_N) of the output
    matrix for a single (batch, hidden_dim) pair.
    Grid dimensions:
        dim0 : B * D                (combined batch & hidden dim)
        dim1 : ceil(N / BLOCK_M)    (row tiles)
        dim2 : ceil(N / BLOCK_N)    (col tiles)
    """
    # ------------------------------------------------------------------
    # decode program ids
    # ------------------------------------------------------------------
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    b = pid // D                     # batch index
    d = pid % D                      # hidden dimension index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # main reduction loop over k
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k_blocks):
        k_base = k_iter * BLOCK_K
        cur_k = k_base + offs_k
        mask_k = cur_k < N

        # -------- left: shape (i,k) -----------------------------------
        # ptr = left_ptr + b*stride_l_b + d*stride_l_d
        #                     + i*stride_l_i + k*stride_l_k
        a_ptrs = (
            left_ptr
            + b * stride_l_b
            + d * stride_l_d
            + offs_m[:, None] * stride_l_i
            + cur_k[None, :] * stride_l_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # -------- right (transposed): shape (k,j) --------------------
        # we need right[b, j, k, d] -> treat j as first dim, k as second.
        # Original strides: (b, i, k, d).  After transpose we use:
        #   stride for k -> stride_r_i   (original i dimension)
        #   stride for j -> stride_r_k   (original k dimension)
        b_ptrs = (
            right_ptr
            + b * stride_r_b
            + d * stride_r_d
            + cur_k[:, None] * stride_r_i   # k dimension (was i)
            + offs_n[None, :] * stride_r_k   # j dimension (was k)
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # dot‑product over k
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # write back FP16 (save bandwidth)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + d * stride_o_d
        + offs_m[:, None] * stride_o_i
        + offs_n[None, :] * stride_o_j
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator with a fused Triton kernel
    for the O(N³) contraction.

    Arguments
    ---------
    data : (input, mask, weights, config)
        - input : [B, N, N, dim]  torch.float32 (on CUDA)
        - mask  : [B, N, N]       torch.float32 (0/1)   (may be all‑ones)
        - weights: dictionary with the exact keys used in the reference model
        - config : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # Helper: linear w/o bias
    # --------------------------------------------------------------
    def linear(inp, w):
        # inp: [..., dim] , w: [out_dim, dim]
        return torch.nn.functional.linear(inp, w)

    # --------------------------------------------------------------
    # 1) LayerNorm over the last dim
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2) Projections + mask
    # --------------------------------------------------------------
    left = linear(x, weights["left_proj.weight"])
    right = linear(x, weights["right_proj.weight"])

    # mask broadcasted over hidden dim
    mask4 = mask.unsqueeze(-1)          # [B,N,N,1]
    left = left * mask4
    right = right * mask4

    # --------------------------------------------------------------
    # 3) Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4) Triton kernel: batched matmul per hidden dim
    # --------------------------------------------------------------
    # allocate output buffer (FP16 for bandwidth)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=x.device)

    # strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 64      # rows  of output tile
    BLOCK_N = 64      # cols  of output tile
    BLOCK_K = 32      # reduction tile

    grid = (
        triton.cdiv(B * hidden_dim, 1),                     # dim0 : batch*hidden
        triton.cdiv(N, BLOCK_M),                           # dim1 : rows
        triton.cdiv(N, BLOCK_N),                           # dim2 : cols
    )

    _trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=hidden_dim,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_d=stride_l_d,
        stride_r_b=stride_r_b,
        stride_r_i=stride_r_i,
        stride_r_k=stride_r_k,
        stride_r_d=stride_r_d,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_d=stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to float32 for the remaining ops
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # 5) Output LayerNorm, gating and final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = linear(out, weights["to_out.weight"])
    return out
