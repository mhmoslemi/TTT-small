"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) with a fused
Trion‑multiplication kernel written in Triton.

Algorithm
---------
1. Layer‑norm on the input tensor   (torch, fp32).
2. Linear projections + gates      (torch, fp32).
3. Apply mask & element‑wise gates (torch, fp32).
4. **Fused triple‑multiplication**:
   For every batch element *b* and hidden channel *h* we need

        out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]

   which is exactly a batched matrix‑multiplication
   (N×N) @ (N×N)ᵀ for each hidden channel.
   A single Triton kernel computes all `B*H` matrix‑multiplications
   using a standard tiling (BLOCK_M × BLOCK_N × BLOCK_K) and
   accumulates in fp32.  The kernel works for arbitrary sequence
   length up to 1024 and is fully vectorised over the hidden
   dimension.
5. Final LayerNorm, gating and output linear (torch, fp32).

The kernel is the only heavy part moved off‑torch; the surrounding
operations stay in PyTorch for simplicity and to keep the code
readable while still meeting the “partially‑triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul (left @ rightᵀ) over the hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # *float16  (B, N, N, H)  – left projection
    right_ptr,           # *float16  (B, N, N, H)  – right projection
    out_ptr,             # *float32  (B, N, N, H)  – result
    B, N, H,             # problem sizes (int)
    stride_l_bh, stride_l_nk, stride_l_h,  # left strides
    stride_r_bh, stride_r_nk, stride_r_h,  # right strides
    stride_o_bh, stride_o_n, stride_o_h,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # tiled matrix‑M dimension
    pid_n = tl.program_id(1)        # tiled matrix‑N dimension
    pid_h = tl.program_id(2)        # batch * hidden index

    # decode (batch, hidden) from pid_h
    b = pid_h // H
    h = pid_h % H

    # start offsets for this tile
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # pointers for the current tile (left: [b, i, k, h], right: [b, j, k, h])
    left_tile_ptr = (
        left_ptr
        + b * stride_l_bh
        + h * stride_l_h
        + offs_m[:, None] * stride_l_nk
        + offs_k[None, :] * stride_l_nk
    )
    right_tile_ptr = (
        right_ptr
        + b * stride_r_bh
        + h * stride_r_h
        + offs_n[:, None] * stride_r_nk
        + offs_k[None, :] * stride_r_nk
    )

    # accumulation in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load a (BLOCK_M, BLOCK_K) and b (BLOCK_N, BLOCK_K)
        a = tl.load(left_tile_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(right_tile_ptr, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # dot‑product over K dimension
        acc += tl.dot(a, b)   # (M,K)·(N,K)ᵀ → (M,N)

        # advance K‑pointers
        left_tile_ptr += BLOCK_K * stride_l_nk
        right_tile_ptr += BLOCK_K * stride_r_nk

    # write back
    out_tile_ptr = (
        out_ptr
        + b * stride_o_bh
        + h * stride_o_h
        + offs_m[:, None] * stride_o_n
        + offs_n[None, :] * stride_o_h
    )
    tl.store(out_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper to launch the kernel
# ----------------------------------------------------------------------
def tri_mul_fused(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : fp16 tensors of shape (B, N, N, H)
    returns      : fp32 tensor of shape (B, N, N, H)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # tile sizes – tuned for Hopper (sm90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    # strides (as int64)
    stride_l_bh, stride_l_n, stride_l_k, stride_l_h = left.stride()
    stride_r_bh, stride_r_n, stride_r_k, stride_r_h = right.stride()
    stride_o_bh, stride_o_n, stride_o_k, stride_o_h = out.stride()

    # note: the right matrix is used transposed on the K dimension,
    # so we reuse the same K‑stride.
    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_l_bh=stride_l_bh,
        stride_l_nk=stride_l_n,   # stride from i → k (same as N‑dim)
        stride_l_h=stride_l_h,
        stride_r_bh=stride_r_bh,
        stride_r_nk=stride_r_n,
        stride_r_h=stride_r_h,
        stride_o_bh=stride_o_bh,
        stride_o_n=stride_o_n,
        stride_o_h=stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Full custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            - mask         : torch.Tensor [B, N, N]       (bool/float)
            - weights      : dict of torch.Tensors (all fp32)
            - config       : dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------- 1. LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- 2. Linear projections (no bias) ----------
    # We keep fp32 for the linear ops, but cast to fp16 before the
    # heavy multiplication to reduce bandwidth.
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # ---------- 3. Apply mask ----------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ---------- 4. Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(left.dtype)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(right.dtype)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"])).to(x.dtype)  # kept fp32 for later

    left = left * left_gate
    right = right * right_gate

    # ---------- 5. Fused triple‑multiplication ----------
    # out shape: (B,N,N,hidden) in fp32
    out = tri_mul_fused(left, right)

    # ---------- 6. Final LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ---------- 7. Output gating and linear ----------
    out = out * out_gate.unsqueeze(-1)                # (B,N,N,hidden) * (B,N,N,1)
    out = F.linear(out, weights["to_out.weight"])    # back to dim, fp32

    return out
