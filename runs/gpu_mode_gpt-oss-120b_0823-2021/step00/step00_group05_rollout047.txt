"""
TriMul “outgoing” implementation.
The heavy part – the N‑by‑N pairwise reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed in a custom Triton kernel.  All linear
projections, layer‑norms and gating are kept in PyTorch
(for flexibility) while the reduction is fused,
vectorised over the hidden dimension (d) and tiled over
the i‑j plane.  The final result is cast back to float32.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: tiled reduction over the sequence dimension `k`
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,            # *[B, N, N, H]   left projection (i‑k‑d)
    right_ptr,           # *[B, N, N, H]   right projection (j‑k‑d)
    out_ptr,             # *[B, N, N, H]   output
    B, N, H,             # problem sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    # ------------------------------------------------------------------
    # program ids – we launch a 3‑D grid: (i‑tile, j‑tile, batch)
    # ------------------------------------------------------------------
    pid_i = tl.program_id(0)          # tile over i (rows)
    pid_j = tl.program_id(1)          # tile over j (cols)
    pid_b = tl.program_id(2)          # batch index

    # compute start indices of this tile
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # offsets inside the tile
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, BLOCK_H)

    # mask for out‑of‑bounds in i/j dimensions
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # ------------------------------------------------------------------
    # initialise accumulator for the reduction over k
    # shape: (BLOCK_M, BLOCK_N, BLOCK_H)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over k dimension in blocks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load left  : shape (M, K, H)   – indexed by (i, k, d)
        # Load right : shape (N, K, H)   – indexed by (j, k, d)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + pid_b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_h[None, None, :] * stride_left_h
        )
        right_ptrs = (
            right_ptr
            + pid_b * stride_right_b
            + offs_j[:, None, None] * stride_right_j
            + offs_k[None, :, None] * stride_right_k
            + offs_h[None, None, :] * stride_right_h
        )

        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0)
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0)

        # element‑wise product and sum over k
        # we first multiply, then reduce the K dimension
        prod = left * right               # (M, K, H)
        acc += tl.sum(prod, axis=1)       # sum over K → (M, H) broadcast to (M,N,H) later

        # advance k‑pointers implicitly by loop index
    # ------------------------------------------------------------------
    # write back – we need to broadcast the reduced result over the j dimension
    # acc currently has shape (BLOCK_M, BLOCK_H); expand to (M, N, H) using mask_ij
    # ------------------------------------------------------------------
    # expand acc to (M, N, H)
    acc_exp = acc[:, None, :, :]          # (M,1,H) → broadcast
    out_ptrs = (
        out_ptr
        + pid_b * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_h[None, None, :] * stride_out_h
    )
    tl.store(out_ptrs, acc_exp.to(tl.float32), mask=mask_ij[:, :, None])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. PyTorch part – layernorms, linear projections & gating
    # ------------------------------------------------------------------
    # helper to apply a linear weight (no bias)
    def linear(x, w):
        # x: [..., dim]   w: [out, dim]
        return torch.nn.functional.linear(x, w)

    # LayerNorm (affine)
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_w + norm_b

    # Projections
    left = linear(x, weights["left_proj.weight"])
    right = linear(x, weights["right_proj.weight"])

    # Apply mask if present (mask shape [B,N,N] → unsqueeze dim)
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gating (sigmoid)
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 2. Triton kernel – the N^2 * hidden reduction
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # Strides (in elements, not bytes)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # Block sizes – tuned for Hopper (SM90)
    BLOCK_M = 64   # tile over i
    BLOCK_N = 64   # tile over j
    BLOCK_K = 64   # tile over k (reduction axis)
    BLOCK_H = 32   # tile over hidden dimension

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B,
    )

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_h=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_j=stride_right[1],
        stride_right_k=stride_right[2],
        stride_right_h=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_h=stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,  # enough for the compute bound work
    )

    # ------------------------------------------------------------------
    # 3. Remaining PyTorch ops – output norm, gating and final linear
    # ------------------------------------------------------------------
    # to_out_norm
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    # out gating
    out = out * out_gate

    # final linear back to `dim`
    out = linear(out, weights["to_out.weight"])

    return out
