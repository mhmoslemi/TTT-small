"""
TriMul “outgoing’’ implementation.

The heavy part of the forward pass is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batch of hidden‑dim (d) matrix‑multiplications
    (left[..., :, d]) @ (right[..., :, d]).T .

The kernel below performs this contraction completely on‑device.
All surrounding operations (LayerNorm, Linear, gating, final LayerNorm + Linear)
are kept in PyTorch – they are tiny compared to the O(B·N³·C) work of the
contraction and the kernels stay simple and easy to reason about.

The kernel works in FP16 for the inner accumulation and writes the result in
FP32 (the final model expects float32).  It tiles the three axes with
configurable block sizes; the default values were chosen for Hopper (SM‑90)
but they can be overridden by the host if needed.
"""

import torch
import triton
import triton.language as tl
from torch import nn

# --------------------------------------------------------------
# Triton kernel:  hidden‑dim d is processed in small tiles (BLOCK_D)
# --------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # const float16 *
    right_ptr,         # const float16 *
    out_ptr,           # float32 *
    B, N, H,           # int32  batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)               # 1‑D grid over (i, j, d) tiles
    # flatten i‑j‑d space:
    total_tiles = (tl.cdiv(N, BLOCK_M) *
                   tl.cdiv(N, BLOCK_N) *
                   tl.cdiv(H, BLOCK_D))
    # map linear pid to three coordinates
    pid_m = pid // (tl.cdiv(N, BLOCK_N) * tl.cdiv(H, BLOCK_D))
    pid_rest = pid % (tl.cdiv(N, BLOCK_N) * tl.cdiv(H, BLOCK_D))
    pid_n = pid_rest // tl.cdiv(H, BLOCK_D)
    pid_d = pid_rest % tl.cdiv(H, BLOCK_D)

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # initialise accumulator in fp32 (more stable than fp16 accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over k dimension in BLOCK_K tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for the current tile
        left_tile = left_ptr + \
            offs_i[:, None, None] * stride_left_i + \
            offs_k[None, :, None] * stride_left_k + \
            offs_d[None, None, :] * stride_left_h

        right_tile = right_ptr + \
            offs_j[:, None, None] * stride_right_j + \
            offs_k[None, :, None] * stride_right_k + \
            offs_d[None, None, :] * stride_right_h

        # broadcast batch dimension later
        # load a tile for every batch (vectorized over batch axis)
        for b in range(B):
            l = tl.load(left_tile + b * stride_left_b,
                        mask=mask_i[:, None] & mask_k[None, :] & mask_d[None, None],
                        other=0.0).to(tl.float32)          # (M,K,D)
            r = tl.load(right_tile + b * stride_right_b,
                        mask=mask_j[:, None] & mask_k[None, :] & mask_d[None, None],
                        other=0.0).to(tl.float32)          # (N,K,D)

            # compute (M,K)·(N,K)^T for each d – use dot for each d slice
            # we can vectorise over D by reshaping
            # l: (M,K,D) -> (M,K*D)
            # r: (N,K,D) -> (N,K*D)
            l_flat = tl.reshape(l, (BLOCK_M, BLOCK_K * BLOCK_D))
            r_flat = tl.reshape(r, (BLOCK_N, BLOCK_K * BLOCK_D))

            # dot produces (M,N)   but we need per‑d accumulation, so we compute manually:
            # accumulate over k: Σ_k l[i,k,d] * r[j,k,d]
            # use broadcasting: (M,1,K,D) * (1,N,K,D) -> (M,N,K,D) -> sum over K
            l_exp = l[:, None, :, :]               # (M,1,K,D)
            r_exp = r[None, :, :, :]               # (1,N,K,D)
            prod = l_exp * r_exp                   # (M,N,K,D)
            acc += tl.sum(prod, axis=2)            # sum over K -> (M,N,D)

    # write results
    out_tile = out_ptr + \
        offs_i[:, None, None] * stride_out_i + \
        offs_j[None, :, None] * stride_out_j + \
        offs_d[None, None, :] * stride_out_h
    for b in range(B):
        tl.store(out_tile + b * stride_out_b,
                 acc,
                 mask=mask_i[:, None] & mask_j[None, :] & mask_d[None, None])

# --------------------------------------------------------------
# Helper wrappers (PyTorch → Triton launch)
# --------------------------------------------------------------
def launch_trimul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right:  [B, N, N, H]  (float16)
    returns:      [B, N, N, H]  (float32)
    """
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape

    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in elements (not bytes) – used inside the kernel
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # block sizes – reasonable defaults for Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 32

    total_tiles = (triton.cdiv(N, BLOCK_M) *
                   triton.cdiv(N, BLOCK_N) *
                   triton.cdiv(H, BLOCK_D))

    grid = (total_tiles,)

    trimul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    return out


# --------------------------------------------------------------
# Full custom kernel entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing’’ TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B,N,N,dim]        (float32)
            - mask  : torch.Tensor [B,N,N]           (bool / float)
            - weights: dict of the model parameters
            - config : dict with keys 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor [B,N,N,dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (float32 → keep it in fp32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # manual layer‑norm to avoid extra Module overhead
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    eps = 1e-6
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias                     # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – we keep them in fp16 for the kernel
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]        # (hidden, dim)

    x_fp16 = x.to(torch.float16)

    left = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Apply mask (if any) – mask is broadcasted on the hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – stay in fp32 for the sigmoid non‑linearity
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w  = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate.to(torch.float16)
    right = right * right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton kernel
    # ------------------------------------------------------------------
    # left/right are now [B,N,N,hidden] in fp16
    out = launch_trimul(left, right)    # fp32 result, shape [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm + gating + final linear)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]               # (dim, hidden)

    # hidden‑dim LayerNorm (float32)
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # gating
    out = out * out_gate

    # final linear (no bias)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]

    return out
