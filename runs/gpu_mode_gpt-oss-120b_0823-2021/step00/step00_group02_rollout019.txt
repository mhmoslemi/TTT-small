"""
TriMul “outgoing” forward pass (AlphaFold‑3 style).

Algorithm
---------
1.  Layer‑norm the [B,N,N,D] input.
2.  Linear projections → left/right tensors of shape [B,N,N,H].
3.  Optional mask applied to the projections.
4.  Gated “left/right” (sigmoid gates) → element‑wise multiplication.
5.  Core operation:  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
    This is a batched matrix‑multiply  left @ rightᵀ  for every head * batch.
    The heavy part (step 5) is executed by a custom Triton kernel.
6.  Layer‑norm + out‑gate + final linear projection back to D.

The kernel works on the flattened dimension (B*H, N, N) and computes
C = A @ Bᵀ   (A,B,C are fp32 accumulators, stored as fp16 for bandwidth).
All other ops stay in PyTorch (they are cheap compared with the N³ work).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B,C : (batch, M, K) , (batch, N, K) )
# ----------------------------------------------------------------------
@triton.jit
def matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,  # pointers
    batch, M, N, K,       # problem size
    stride_am, stride_ak,   # A strides (batch, M, K)
    stride_bn, stride_bk,   # B strides (batch, N, K)   (note: B is NOT transposed in memory)
    stride_cm, stride_cn,   # C strides (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # each program works on one (batch*head) slice
    batch_idx = pid // tl.cdiv(M, BLOCK_M) // tl.cdiv(N, BLOCK_N)
    pid_m = (pid // tl.cdiv(N, BLOCK_N)) % tl.cdiv(M, BLOCK_M)
    pid_n = pid % tl.cdiv(N, BLOCK_N)

    # offsets for the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the tile
    a_ptrs = a_ptr + batch_idx * stride_am * M + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch_idx * stride_bn * N + offs_n[:, None] * stride_bn + tl.arange(0, BLOCK_K)[None, :] * stride_bk

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        a = tl.load(a_ptrs + k_offset * stride_ak, mask=mask_m[:, None] & (k_offset + tl.arange(0, BLOCK_K) < K), other=0.0)
        # B needs to be read transposed: we want B[j,k] -> we load B[j, k] and later dot with A[i,k]
        b = tl.load(b_ptrs + k_offset * stride_bk, mask=mask_n[:, None] & (k_offset + tl.arange(0, BLOCK_K) < K), other=0.0)

        # a : (M, K_tile)   b : (N, K_tile)
        # we need sum_k a[i,k] * b[j,k]  → dot product of rows of a and rows of b
        acc += tl.dot(a, b)   # tl.dot does row‑wise dot product

    # write result
    c_ptrs = c_ptr + batch_idx * stride_cm * M + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side helper to launch the kernel
# ----------------------------------------------------------------------
def batch_gemm_transpose(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a, b : torch.float16 tensors of shape (B*H, N, N)
    returns c = a @ bᵀ  with shape (B*H, N, N)   (float16)
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, N, _ = a.shape
    K = N
    c = torch.empty_like(a)

    # launch config
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (B * grid_m * grid_n,)

    matmul_transpose_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        batch=B,
        M=N,
        N=N,
        K=K,
        stride_am=a.stride(0),
        stride_ak=a.stride(2),
        stride_bn=b.stride(0),
        stride_bk=b.stride(2),
        stride_cm=c.stride(0),
        stride_cn=c.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, D]   (float32)
        - mask         : torch.Tensor  [B, N, N]      (float32 or bool)
        - weights      : dict of leaf tensors (all float32)
        - config       : dict with keys ``dim`` (D) and ``hidden_dim`` (H)

    Returns
    -------
    torch.Tensor  [B, N, N, D]  (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Layer‑norm on D dimension  (no bias handling needed – we have bias)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(D,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: (B,N,N,D) @ (D,H) -> (B,N,N,H)
    left = torch.einsum("bnnd,dh->bnnh", x, weights["left_proj.weight"])
    right = torch.einsum("bnnd,dh->bnnh", x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (mask shape [B,N,N] → broadcast)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.einsum("bnnd,dh->bnnh", x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.einsum("bnnd,dh->bnnh", x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.einsum("bnnd,dh->bnnh", x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul: out = left @ rightᵀ   (batched GEMM over the “k” axis)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) and compute in fp16 for bandwidth
    left_h = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N).to(torch.float16)
    right_h = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N).to(torch.float16)

    out_h = batch_gemm_transpose(left_h, right_h)               # (B*H,N,N) fp16
    out = out_h.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6) Final projection back to D
    # ------------------------------------------------------------------
    # layer‑norm on hidden dim
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(H,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    out = out * out_gate  # apply out‑gate (broadcasted over H)

    # final linear (H → D)
    out = torch.einsum("bnnh,hd->bnnd", out, weights["to_out.weight"])

    return out
