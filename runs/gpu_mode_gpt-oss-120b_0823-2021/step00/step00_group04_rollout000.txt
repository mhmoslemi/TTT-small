"""
TriMul “outgoing” implementation.

We keep the cheap point‑wise work (LayerNorm, Linear, gating, masking) in
torch and accelerate the N³ contraction:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed feature d this is a classic matrix multiply

    out_d = left_d @ right_d.T          (M=N, K=N, N=N)

We therefore launch a batched GEMM kernel where the batch dimension is
`B * hidden_dim`.  The kernel is a simplified version of the classic
triton matmul, working in fp16 internally and writing fp32 results.
The rest of the pipeline (norms, linear layers, gates) is performed with
torch (which is already highly optimized on the target Hopper GPUs).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# GEMM kernel: C = A @ Bᵀ  (A,B are square N×N, batch = B*H)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,                     # logical sizes
    stride_ah, stride_ai, stride_ak,    # A strides (batch, i, k)
    stride_bh, stride_bi, stride_bk,    # B strides (batch, i, k) – we read Bᵀ
    stride_ch, stride_ci, stride_cj,    # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute a batched GEMM with the second matrix transposed."""
    pid = tl.program_id(0)
    pid_batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = (pid // tl.cdiv(N, BLOCK_N)) % tl.cdiv(M, BLOCK_M)
    pid_n = pid % tl.cdiv(N, BLOCK_N)

    # ---- compute offsets ------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    a_offset = (pid_batch * stride_ah
                + offs_m[:, None] * stride_ai
                + offs_k[None, :] * stride_ak)
    b_offset = (pid_batch * stride_bh
                + offs_n[:, None] * stride_bi           # note: we read Bᵀ → swap i<->k
                + offs_k[None, :] * stride_bk)

    a_ptrs = a_ptr + a_offset
    b_ptrs = b_ptr + b_offset

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K ----------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_n < N)[:, None] & (offs_k < K)[None, :], other=0.0)
        # a: (M,K) , b: (N,K) → dot over K
        acc += tl.dot(a, b)          # result in fp32

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ---- write back ------------------------------------------------------
    c_offset = (pid_batch * stride_ch
                + offs_m[:, None] * stride_ci
                + offs_n[None, :] * stride_cj)
    c_ptrs = c_ptr + c_offset
    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask)


def batched_gemm_transpose(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    A, B : (batch*H, N, N)  fp16 tensors, contiguous.
    Returns C : (batch*H, N, N) fp32 tensor.
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, N, K = A.shape
    assert batch == B.shape[0] and N == K == B.shape[1] == B.shape[2]

    C = torch.empty_like(A, dtype=torch.float32)

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (grid_m * grid_n * batch,)

    batched_gemm_transpose_kernel[grid](
        A,
        B,
        C,
        batch,
        N,
        N,
        N,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(2), B.stride(1),   # note swapped stride_i/k for transpose
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return C


# ----------------------------------------------------------------------
# Top‑level entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]  (float32/float16)
        - mask         : torch.Tensor  [B, N, N]      (bool or 0/1)
        - weights      : dict of torch.Tensors (all float32)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dim
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden_dim]
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if provided)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                       # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ contraction – Triton kernel
    # ------------------------------------------------------------------
    # Work in fp16 for the GEMM, keep a copy of the fp32 output later.
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # move the feature dim (hidden_dim) into the batch dimension
    B, N, _, H = left_fp16.shape
    A = left_fp16.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, N)
    B_mat = right_fp16.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # GEMM: C = A @ Bᵀ   -> result shape (B*H, N, N) fp32
    C = batched_gemm_transpose(A, B_mat)                     # fp32

    # reshape back to [B,N,N,hidden_dim]
    out = C.reshape(B, H, N, N).permute(0, 2, 3, 1)          # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Output projection (LayerNorm + Linear + gate)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])            # back to dim

    # final dtype = float32 as required
    return out.to(torch.float32)
