#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
TriMul “outgoing” implementation.
---------------------------------
*   Normalisation and linear projections are performed with PyTorch (they are
    cheap compared to the O(N³) core).
*   The core operation
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
  is a batched matrix‑multiply over the sequence dimension *k* for every
  hidden channel *h*.  This is implemented as a single Triton kernel that
  fuses the three‑dimensional loop (i, j, k) into a tiled GEMM‑like kernel.
*   The kernel works for any batch size B∈{1,2}, any sequence length N,
    and any hidden size H (called `hidden_dim` in the config).  All strides are
    derived from the input tensors so the kernel can be launched with a simple
    2‑D grid (i‑tile × j‑tile) and internally loops over the reduction axis *k*.
*   The kernel uses float16 arithmetic internally (the model weights are
    stored in FP32, but the multiplicative part is cast to FP16 for higher
    throughput on Hopper).  The final result is cast back to FP32 before the
    downstream LayerNorm / linear layers.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# -----------------------------------------------------------------------
# Triton kernel  ---------------------------------------------------------
# -----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,          #  [B, N, N, H]   (float16)
    right_ptr,         #  [B, N, N, H]   (float16)
    out_ptr,           #  [B, N, N, H]   (float16)
    # shapes
    B, N, H,
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr
):
    """
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    The kernel tiles the i‑ and j‑ dimensions (M, N) and loops over the k‑axis.
    """
    pid = tl.program_id(axis=0)  # linear program id
    # -------------------------------------------------------------------
    # 2‑D grid: pid -> (block_i, block_j)
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # group blocks to improve L2 cache reuse (standard GEMM trick)
    block_i = (pid // GROUP_SIZE_M) * BLOCK_M
    block_j = (pid % GROUP_SIZE_M) * BLOCK_N
    block_i = block_i % (num_blocks_m * BLOCK_M)
    block_j = block_j % (num_blocks_n * BLOCK_N)

    # offsets inside the tile
    offs_i = block_i + tl.arange(0, BLOCK_M)
    offs_j = block_j + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # -------------------------------------------------------------------
    # We treat the hidden dimension as a batched dimension: each (b, h)
    # gets its own independent GEMM.
    # Loop over hidden dimension in the outermost Python launch.
    for h_idx in range(0, H):
        # Pointers for this hidden slice
        left_base  = left_ptr  + h_idx * stride_left_h
        right_base = right_ptr + h_idx * stride_right_h
        out_base   = out_ptr   + h_idx * stride_out_h

        # Initialise accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over reduction axis k
        for k in range(0, tl.cdiv(N, BLOCK_K)):
            # Compute actual offset for this k‑tile
            cur_k = k * BLOCK_K + offs_k
            cur_mask_k = cur_k < N

            # Load A: left[b, i, k, h]
            a_ptrs = left_base + \
                     offs_i[:, None] * stride_left_i + \
                     cur_k[None, :] * stride_left_k
            a = tl.load(a_ptrs, mask=mask_i[:, None] & cur_mask_k[None, :],
                        other=0.0).to(tl.float32)

            # Load B: right[b, j, k, h]   (note the j dimension)
            b_ptrs = right_base + \
                     offs_j[:, None] * stride_right_j + \
                     cur_k[None, :] * stride_right_k
            b = tl.load(b_ptrs, mask=mask_j[:, None] & cur_mask_k[None, :],
                        other=0.0).to(tl.float32)

            # Accumulate
            acc += tl.dot(a, b)

        # ----------------------------------------------------------------
        # Store result
        out_ptrs = out_base + \
                   offs_i[:, None] * stride_out_i + \
                   offs_j[None, :] * stride_out_j
        tl.store(out_ptrs, acc.to(tl.float16),
                 mask=mask_i[:, None] & mask_j[None, :])


# -----------------------------------------------------------------------
# Helper to launch the kernel
# -----------------------------------------------------------------------
def launch_trimul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    `left` / `right` :  [B, N, N, H]   (float16)
    Returns `out` :      [B, N, N, H]   (float16)
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # compute strides (in elements!)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # tile sizes – chosen to fit in shared memory on Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    GROUP_SIZE_M = 8  # typical value for good L2 reuse

    grid = lambda META: (triton.cdiv(N, META['BLOCK_M']) *
                         triton.cdiv(N, META['BLOCK_N']) //
                         META['GROUP_SIZE_M'] + 1, )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
    )
    return out


# -----------------------------------------------------------------------
# Main entry point -------------------------------------------------------
# -----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.
    The heavy O(N³) multiplication is computed in a fused Triton kernel,
    while all linear layers / layer‑norms are performed with PyTorch
    (they are memory‑bound and inexpensive relative to the core).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]          # input / output feature dimension
    hidden_dim = config["hidden_dim"]

    # -------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # -------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias   = weights["norm.bias"].to(dtype)
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias

    # -------------------------------------------------------------------
    # 2. Linear projections (no bias) – we keep them in FP32 then cast
    # -------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(dtype)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(dtype)

    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)

    # -------------------------------------------------------------------
    # 3. Mask handling (optional)
    # -------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(dtype)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -------------------------------------------------------------------
    # 4. Gates (sigmoid) – all in FP32
    # -------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(dtype)
    right_gate_w = weights["right_gate.weight"].to(dtype)
    out_gate_w   = weights["out_gate.weight"].to(dtype)

    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------
    # 5. Core O(N³) multiplication – convert to FP16, launch Triton kernel
    # -------------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # out_fp16 : [B,N,N,hidden]
    out_fp16 = launch_trimul_outgoing(left_fp16, right_fp16)

    # back to FP32 for the remaining ops
    out = out_fp16.to(dtype)

    # -------------------------------------------------------------------
    # 6. Output LayerNorm (hidden) + gating + final linear
    # -------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)

    # LayerNorm over hidden dimension
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    # Apply out_gate (broadcast on hidden dim)
    out = out * out_gate

    # Final linear projection back to dim
    to_out_w = weights["to_out.weight"].to(dtype)   # (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_w)

    return out
