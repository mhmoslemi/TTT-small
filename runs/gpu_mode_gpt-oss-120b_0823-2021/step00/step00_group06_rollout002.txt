"""
TriMul “outgoing” kernel (AlphaFold3)

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where
    left  = Linear(x) * gate_left  * mask
    right = Linear(x) * gate_right * mask

We compute all cheap ops (LayerNorm, Linear, sigmoids, masking, final
normalisation / projection) with PyTorch and off‑load the cubic reduction
to a highly‑fused Triton kernel.  The kernel treats the hidden dimension *d*
as an extra batch axis, i.e. each block of D channels is processed by an
independent program ID.  For a given batch *b* we launch a 3‑D grid:

    (ceil(N/BM), ceil(N/BN), ceil(H/BD))

Inside the kernel we perform a classic blocked matrix‑multiply over the
*K* dimension, accumulating in FP32 and finally storing FP32 results.
All inputs/outputs are kept in FP32 (the surrounding PyTorch code may use
mixed‑precision, but the final tensor is returned as float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel : blocked matmul (M×K)·(K×N) for each hidden‑dim block
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # shapes
    N, H,
    # strides (left: [i,k,d], right: [j,k,d], out: [i,j,d])
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_j, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    # block sizes (constexpr)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    # ------------------------------------------------------------------
    # program IDs
    # ------------------------------------------------------------------
    pid_i = tl.program_id(0)  # i‑block
    pid_j = tl.program_id(1)  # j‑block
    pid_d = tl.program_id(2)  # hidden‑dim block

    # ------------------------------------------------------------------
    # offsets for this block
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    d_base = pid_d * BLOCK_D

    # masks for out‑of‑bounds
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    offs_d = d_base + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # ------------------------------------------------------------------
    # accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over k dimension
    # ------------------------------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        # ----------------------------------------------------------------
        # Load a tile of left: shape (M, K, D)
        # ----------------------------------------------------------------
        k_range = tl.arange(0, BLOCK_K)
        offs_k = k_base + k_range
        mask_k = offs_k < N

        # left[i,k,d]
        left_ptrs = (
            left_ptr
            + (offs_i[:, None, None] * stride_left_i)
            + (offs_k[None, :, None] * stride_left_k)
            + (offs_d[None, None, :] * stride_left_d)
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

        # ----------------------------------------------------------------
        # Load a tile of right: shape (N, K, D)
        # ----------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + (offs_j[:, None, None] * stride_right_j)
            + (offs_k[None, :, None] * stride_right_k)
            + (offs_d[None, None, :] * stride_right_d)
        )
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

        # ----------------------------------------------------------------
        # Compute outer‑product over the K dimension for each d channel:
        #   acc[i,j,d] += Σ_k left[i,k,d] * right[j,k,d]
        # ----------------------------------------------------------------
        # We broadcast the K dimension and perform a dot‑product manually.
        # Using tl.dot on 1‑D vectors inside the loop is costly; instead we
        # rely on the fact that both tensors are contiguous in K and perform
        # a vector‑wise multiply‑reduce.
        # The reduction is done with tl.sum over the K axis.
        prod = left * right      # (M, K, D) * (N, K, D) -> broadcasting creates (M, N, K, D)
        # bring K dimension to innermost position for tl.sum
        prod = tl.transposed(prod, (0, 2, 1, 3))   # (M, K, N, D)
        # sum over K
        acc += tl.sum(prod, axis=1)               # (M, N, D)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + (offs_i[:, None, None] * stride_out_i)
        + (offs_j[None, :, None] * stride_out_j)
        + (offs_d[None, None, :] * stride_out_d)
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.
    Arguments:
        data = (input, mask, weights, config)
            input : [B, N, N, C]   (float32)
            mask  : [B, N, N]      (float32 or bool)
            weights: dict of learned parameters
            config: {"dim": C, "hidden_dim": H}
    Returns:
        out : [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]
    dtype = torch.float32

    # --------------------------------------------------------------
    # Helper to apply linear weight (no bias) + optional mask
    # --------------------------------------------------------------
    def linear(x, w):
        # x: [..., C]   w: [H, C]  (as in nn.Linear(C->H))
        return F.linear(x, w)

    # --------------------------------------------------------------
    # LayerNorm (C)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(dtype)
    norm_b = weights["norm.bias"].to(dtype)
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_w + norm_b

    # --------------------------------------------------------------
    # Projections
    # --------------------------------------------------------------
    left = linear(x, weights["left_proj.weight"].to(dtype))
    right = linear(x, weights["right_proj.weight"].to(dtype))

    # mask handling
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"].to(dtype)))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Triton kernel for the cubic reduction
    # --------------------------------------------------------------
    # allocate output in FP32
    out = torch.empty((B, N, N, H), dtype=dtype, device=device)

    # block sizes (tuned for Hopper)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64
    BLOCK_D = 16   # hidden‑dim block

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(H, BLOCK_D),
    )

    for b in range(B):
        # compute strides for this batch
        left_ptr = left[b].contiguous()
        right_ptr = right[b].contiguous()
        out_ptr = out[b].contiguous()

        trimul_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            N,
            H,
            # strides
            left_ptr.stride(0), left_ptr.stride(1), left_ptr.stride(2),
            right_ptr.stride(0), right_ptr.stride(1), right_ptr.stride(2),
            out_ptr.stride(0), out_ptr.stride(1), out_ptr.stride(2),
            # compile‑time constants
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # --------------------------------------------------------------
    # Final normalisation + projection (still in PyTorch)
    # --------------------------------------------------------------
    # to_out_norm : LayerNorm over hidden dimension
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate

    # final linear projection back to C
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
