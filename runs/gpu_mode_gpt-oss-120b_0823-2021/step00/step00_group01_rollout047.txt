"""
TriMul “outgoing” implementation.
The heavy part (the N‑dimensional einsum) is executed with a custom
Triton kernel that performs a batched, per‑channel matrix multiplication

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

All other operations (LayerNorm, Linear projections, gating and the final
LayerNorm/Linear) stay in PyTorch – they are cheap compared to the O(N³)
computation and keep the code simple.  The kernel works in fp16 for the
multiplications, accumulates in fp32 and finally writes float‑32 results.
It supports arbitrary batch size (B≤2), sequence length N (up to 1024) and
hidden dimension H (any multiple of the block size, typically 128).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched per‑channel matrix multiplication
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # *const fp16  [B, N, N, H]
    right_ptr,         # *const fp16  [B, N, N, H]
    out_ptr,           # *mut   fp32  [B, N, N, H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """ Compute a block of the output matrix for one (batch, head) pair. """

    # program id encodes (batch, head)
    pid = tl.program_id(0)
    batch_id = pid // H
    head_id   = pid %  H

    # block offsets in the i (row) and j (col) dimensions
    pid_m = tl.program_id(1)   # i‑block
    pid_n = tl.program_id(2)   # j‑block

    # start indices for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the start of the current (b, h) slice
    left_base  = left_ptr \
               + batch_id * stride_left_b \
               + head_id   * stride_left_h
    right_base = right_ptr \
               + batch_id * stride_right_b \
               + head_id   * stride_right_h
    out_base   = out_ptr \
               + batch_id * stride_out_b \
               + head_id   * stride_out_h

    # create 2‑D pointers for the current block
    left_tile  = left_base \
               + offs_m[:, None] * stride_left_i \
               + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    right_tile = right_base \
               + offs_n[:, None] * stride_right_j \
               + tl.arange(0, BLOCK_K)[None, :] * stride_right_k

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = tl.arange(0, BLOCK_K) + k_offset
        cur_k = tl.where(cur_k < N, cur_k, 0)               # avoid OOB read
        # load tiles
        left = tl.load(left_tile + cur_k * stride_left_k,
                       mask=mask_m[:, None] & (cur_k[None, :] < N),
                       other=0.0).to(tl.float32)
        right = tl.load(right_tile + cur_k * stride_right_k,
                        mask=mask_n[:, None] & (cur_k[None, :] < N),
                        other=0.0).to(tl.float32)
        # inner product
        acc += tl.dot(left, right)

    # write result (fp32)
    out_ptrs = out_base \
             + offs_m[:, None] * stride_out_i \
             + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Host‑side helper that launches the kernel
# ----------------------------------------------------------------------
def launch_trimul_kernel(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : torch.float16 tensors of shape (B,N,N,H)
    returns      : torch.float32 tensor of shape (B,N,N,H)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides (in elements) for each dimension
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    BLOCK_M = 64   # rows per block
    BLOCK_N = 64   # cols per block
    BLOCK_K = 64   # reduction block size

    # grid: (batch*head, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        triton.cdiv(B * H, 1),                      # we will split (b,h) manually inside kernel
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # note: we encode (b,h) in pid using the first dimension; each kernel instance
    # processes exactly one head for one batch.
    trimul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul “outgoing” operator.
    Arguments
    ---------
    data : (input, mask, weights, config)
        - input : Tensor[B, N, N, dim] (float32)
        - mask  : Tensor[B, N, N]      (float32 or bool)
        - weights: dict of layer‑norm and linear parameters (float32)
        - config : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – we keep them in fp16 for speed
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]  # (hidden_dim, dim)

    # shape: (B,N,N,hidden_dim)
    left  = F.linear(x, left_proj_w).to(torch.float16)
    right = F.linear(x, right_proj_w).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # (B,N,N,1)
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – keep as fp32 then cast back
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left  * left_gate.to(left.dtype)
    right = right * right_gate.to(right.dtype)

    # ------------------------------------------------------------------
    # 5. Core N‑dimensional einsum → Triton kernel
    # ------------------------------------------------------------------
    # left/right are (B,N,N,hidden_dim) in fp16
    out = launch_trimul_kernel(left, right)   # (B,N,N,hidden_dim) fp32

    # ------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]   # (dim, hidden_dim)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate          # broadcasting over last dim
    out = F.linear(out, to_out_w) # back to (B,N,N,dim)
    return out.to(torch.float32)
