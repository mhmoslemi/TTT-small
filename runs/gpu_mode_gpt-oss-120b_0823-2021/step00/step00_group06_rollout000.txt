"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the three‑dimensional einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be viewed as a batched matrix multiplication for every hidden
dimension *d* :

    left  : (B, N, K)   with K = N
    right : (B, N, K)   (the K‑axis is summed, the second axis is the output
                         row index, therefore we need right transposed).

We keep all inexpensive operations (layer‑norm, linear projections,
gating, final layer‑norm + linear) in PyTorch and fuse the O(N³) part with
a custom Triton kernel.  The kernel performs a FP32 GEMM for each
(batch, hidden_dim) pair, accumulating into fp32 and finally writing the
result back.  The kernel is launched with a three‑dimensional grid:

    dim0 = B * hidden_dim          (combined batch‑hidden index)
    dim1 = ceil(N / BLOCK_M)       (output‑row tiles)
    dim2 = ceil(N / BLOCK_N)       (output‑col tiles)

The kernel works for any batch (1 or 2), any sequence length N up to 1024,
and any hidden dimension (128 in the reference, but the code is generic).
Mixed‑precision is allowed inside the kernel – we load as fp16 to utilize
tensor‑cores and accumulate in fp32 for numerical stability.  The final
output is cast back to float32 as required by the reference implementation.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# ---------------------------------------------------------------------
# Triton kernel: batched GEMM for the three‑dimensional einsum
# ---------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr,          # * (B, N, K, D)   left projection after gating & mask
    right_ptr,         # * (B, N, K, D)   right projection after gating & mask
    out_ptr,           # * (B, N, N, D)   result tensor
    B, N, D,           # static problem size
    stride_lb, stride_lk, stride_ld,   # left strides: (batch, i, k, d)
    stride_rb, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute, for a given (batch, hidden) pair, the matrix product
        out[i, j] = Σ_k left[i, k] * right[j, k]      (right is used transposed)
    """
    # program identifiers
    pid_bn_hd = tl.program_id(0)               # combined batch*hidden index
    pid_m = tl.program_id(1)                   # tile over output rows (i)
    pid_n = tl.program_id(2)                   # tile over output cols (j)

    # decode batch and hidden dimension from the combined index
    b = pid_bn_hd // D                         # batch id
    d = pid_bn_hd % D                          # hidden dim id

    # offsets within the i/j dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers to the beginning of the tile for this (b,d)
    left_base  = left_ptr  + b * stride_lb + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k‑range for this iteration
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left[i, k]  -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (offs_i[:, None] * stride_lk
                                 + offs_k[None, :] * stride_lk
                                 + 0)          # d stride already added
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float16)

        # load right[j, k] (note: we need right transposed, so we load as (j,k))
        right_ptrs = right_base + (offs_j[:, None] * stride_rk
                                   + offs_k[None, :] * stride_rk
                                   + 0)
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float16)

        # fp16 dot, accumulate in fp32
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # write result
    out_ptrs = out_ptr + b * stride_ob + d * stride_od \
               + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ---------------------------------------------------------------------
# Helper: LayerNorm without bias (bias handled separately)
# ---------------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    """Simple fused LayerNorm (no extra sync needed)."""
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    y = (x - mean) * torch.rsqrt(var + eps)
    if weight is not None:
        y = y * weight
    if bias is not None:
        y = y + bias
    return y


# ---------------------------------------------------------------------
# Main entry point
# ---------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]   float32
            - mask:         [B, N, N]        float32 (0/1) or None
            - weights: dict of torch tensors (see problem statement)
            - config:  dict with keys "dim" and "hidden_dim"

    Returns:
        out: [B, N, N, dim]   float32
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input (dim)
    # -----------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias) + mask
    # -----------------------------------------------------------------
    # shape: (dim, hidden_dim)
    left_proj_w = weights["left_proj.weight"]          # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"]
    # Linear without bias: y = x @ w.T
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hd]
    right = torch.nn.functional.linear(x, right_proj_w)

    if mask is not None:
        mask_ = mask.unsqueeze(-1)            # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 3) Gating (sigmoid) and apply
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4) Triton GEMM for the einsum over k
    # -----------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=x.device)

    # strides (in elements)
    stride_lb = left.stride(0)          # batch
    stride_lk = left.stride(2)          # k dimension
    stride_ld = left.stride(3)          # hidden dim

    stride_rb = right.stride(0)
    stride_rk = right.stride(2)
    stride_rd = right.stride(3)

    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_od = out.stride(3)

    # Tile sizes – tuned for Hopper (sm_90).  128 works well for N up to 1024.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(B * hidden_dim, 1),          # dim0: each (b,d) gets its own program
        triton.cdiv(N, BLOCK_M),                # dim1: rows
        triton.cdiv(N, BLOCK_N),                # dim2: cols
    )

    trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=hidden_dim,
        stride_lb=stride_lb,
        stride_lk=stride_lk,
        stride_ld=stride_ld,
        stride_rb=stride_rb,
        stride_rk=stride_rk,
        stride_rd=stride_rd,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_od=stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 5) to_out_norm + gating + final linear
    # -----------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
