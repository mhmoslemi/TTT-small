"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication for every hidden dimension `d`.
We off‑load this N³ reduction to a Triton kernel (one kernel call per
(batch, hidden_dim) pair).  All LayerNorms, Linear projections and gates are
performed with PyTorch (they are cheap compared with the N³ work), but the
core einsum is fully fused in Triton.

The kernel computes a classic GEMM:
    out = left @ right.transpose(-2,-1)
where left/right are 3‑D tensors of shape (N, N) for a fixed
(batch, hidden_dim) slice.
The kernel uses fp16 arithmetic for speed and casts the final
result back to fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for a single hidden dimension slice
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                     # *const float16
    right_ptr,                    # *const float16
    out_ptr,                      # *float16
    N,                            # int32
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # batch * hidden_dim flattened
    # decode batch and hidden dimension from pid
    batch_id = pid // tl.constexpr(N_HIDDEN)   # N_HIDDEN set at launch
    hidden_id = pid %  tl.constexpr(N_HIDDEN)

    # pointers to the start of the (batch, hidden) slice
    left = left_ptr + batch_id * stride_left_b + hidden_id * stride_left_d
    right = right_ptr + batch_id * stride_right_b + hidden_id * stride_right_d
    out = out_ptr + batch_id * stride_out_b + hidden_id * stride_out_d

    # 2‑D program IDs for the output tile
    pid_m = tl.program_id(1)   # row block index (i)
    pid_n = tl.program_id(2)   # col block index (j)

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Load a block of left: shape (BLOCK_M, BLOCK_K)
    left_ptrs = left + (offs_m[:, None] * stride_left_i
                        + offs_k[None, :] * stride_left_k)
    a = tl.load(left_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < N), other=0.0)

    # Load a block of right (transposed): we need right[j,k] -> we load right^T
    # right original layout (i,k) ; we want (k,j). So we load with swapped axes.
    right_ptrs = right + (offs_k[:, None] * stride_right_i
                          + offs_n[None, :] * stride_right_k)
    b = tl.load(right_ptrs, mask=(offs_k[:, None] < N) & mask_n[None, :], other=0.0)

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # a: (BLOCK_M, BLOCK_K) ; b: (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs = left_ptrs + k * BLOCK_K * stride_left_k
        b_ptrs = right_ptrs + k * BLOCK_K * stride_right_i
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] + k*BLOCK_K < N), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] + k*BLOCK_K < N) & mask_n[None, :], other=0.0)

    # write back
    out_ptrs = out + (offs_m[:, None] * stride_out_i
                      + offs_n[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim]   (float32)
            mask        : torch.Tensor [B, N, N]       (float32/bool)
            weights     : dict of parameter tensors
            config      : dict with keys dim, hidden_dim, nomask (bool)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- LayerNorm ----------
    # x = (x - mean) / sqrt(var + eps) * w + b   (per‑channel)
    eps = 1e-5
    w = weights["norm.weight"]
    b = weights["norm.bias"]
    # compute mean/var over last dimension (dim)
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * w + b

    # ---------- Linear projections ----------
    # use torch.nn.functional.linear for speed
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Mask ----------
    if not nomask:
        mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Core TriMul (Triton) ----------
    # Cast to fp16 for the kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)   # same shape [B,N,N,hidden]

    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left_fp16.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_d = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out_fp16.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # we need N_HIDDEN as a compile‑time constant for the kernel.
    # Triton does not allow dynamic constexpr, so we expose it via a
    # global variable trick: set a module‑level const and read it inside
    # the kernel via tl.constexpr.  Here we use a closure hack:
    global N_HIDDEN
    N_HIDDEN = hidden_dim

    grid = (
        triton.cdiv(B * hidden_dim, 1),   # pid (flattened batch*hidden)
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_i, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # back to fp32
    out = out_fp16.to(torch.float32)

    # ---------- Output processing ----------
    # LayerNorm on hidden dim
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # -> [B,N,N,dim]

    return out
