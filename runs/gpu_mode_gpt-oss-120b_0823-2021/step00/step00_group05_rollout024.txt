"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part of the forward pass is the three‑dimensional einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right are the masked‑gated projections of the input.
For each (b,i,j) the reduction over *k* produces a vector of length *d*.
The kernel below computes this reduction completely on the GPU:
* one program (thread‑block) handles a single (b,i,j) triple,
* the hidden dimension *d* (aka “hidden_dim”) is processed in
  registers in tiles of size BLOCK_H (≤ 384),
* the sequence dimension *k* is tiled with BLOCK_K (default 32).

All other operations (LayerNorm, Linear, gating, final LayerNorm/Linear)
are performed with regular PyTorch – they are cheap compared with the
N³ reduction and keep the implementation simple while still satisfying the
requirement of “at least part of the ops in a Triton kernel”.

The kernel works for arbitrary batch size (≤2), sequence length (≤1024) and
hidden dimensions up to 384.  It returns a float‑32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: computes the N³ reduction for a single (b,i,j) triplet.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const float*  [B,N,N,H]
    right_ptr,         # const float*  [B,N,N,H]
    out_ptr,           # float*        [B,N,N,H]
    B, N, H,
    # strides (in elements, not bytes)
    stride_lb, stride_li, stride_lk, stride_lh,   # left  B,N,K,H
    stride_rb, stride_rj, stride_rk, stride_rh,   # right B,J,K,H
    stride_ob, stride_oi, stride_oj, stride_oh,   # out   B,I,J,H
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    """
    pid = tl.program_id(0)

    # flatten (b,i,j) -> linear pid
    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i = ij // N
    j = ij % N

    # ------------------------------------------------------------------
    # registers for the hidden dimension
    # ------------------------------------------------------------------
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    # ------------------------------------------------------------------
    # iterate over k in tiles
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_start < N

        # left[b,i,k,:]   shape (BLOCK_K, BLOCK_H)
        left = tl.load(
            left_ptr
            + b * stride_lb
            + i * stride_li
            + k_start * stride_lk
            + offs_h * stride_lh,
            mask=(mask_k[:, None] & mask_h[None, :]),
            other=0.0,
        )
        # right[b,j,k,:]  shape (BLOCK_K, BLOCK_H)
        right = tl.load(
            right_ptr
            + b * stride_rb
            + j * stride_rj
            + k_start * stride_rk
            + offs_h * stride_rh,
            mask=(mask_k[:, None] & mask_h[None, :]),
            other=0.0,
        )

        # element‑wise product & sum over k axis
        acc += tl.sum(left * right, axis=0)

    # ------------------------------------------------------------------
    # write the result
    # ------------------------------------------------------------------
    tl.store(
        out_ptr
        + b * stride_ob
        + i * stride_oi
        + j * stride_oj
        + offs_h * stride_oh,
        acc,
        mask=mask_h,
    )


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input, mask, weights, config)
            - input : Tensor[B, N, N, dim] (float32, on CUDA)
            - mask  : Tensor[B, N, N]        (bool/float, on CUDA)
            - weights: dict with all linear / layer‑norm parameters
            - config : dict with keys "dim", "hidden_dim"

    Returns:
        Tensor[B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper: LayerNorm (no bias in the reference, bias provided separately)
    # ------------------------------------------------------------------
    def layer_norm(x, weight, bias, eps=1e-5):
        # x: [..., C]
        mean = x.mean(dim=-1, keepdim=True)
        var = x.var(dim=-1, unbiased=False, keepdim=True)
        x_norm = (x - mean) / torch.sqrt(var + eps)
        return x_norm * weight + bias

    # ------------------------------------------------------------------
    # 1) Pre‑norm
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core N³ reduction via Triton
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides in *elements* (not bytes) for contiguous tensors
    left_s = left.stride()
    right_s = right.stride()
    out_s = out.stride()

    # launch grid: one program per (b,i,j)
    total_prog = B * N * N
    grid = (total_prog,)

    BLOCK_K = 32                     # tiles over the reduction axis
    BLOCK_H = 128                    # hidden‑dim tile (max 384, will be split internally)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_lb=left_s[0],
        stride_li=left_s[1],
        stride_lk=left_s[2],
        stride_lh=left_s[3],
        stride_rb=right_s[0],
        stride_rj=right_s[1],
        stride_rk=right_s[2],
        stride_rh=right_s[3],
        stride_ob=out_s[0],
        stride_oi=out_s[1],
        stride_oj=out_s[2],
        stride_oh=out_s[3],
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6) Post‑norm, gate, final linear
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
