"""
TriMul “outgoing” implementation.

The heavy part of the TriMul forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]          (1)

where `left` and `right` are already multiplied by their
corresponding gates and masked.  Equation (1) is a reduction over the
sequence dimension *k* with an element‑wise product across the hidden
dimension *d*.

The kernel below evaluates (1) in pure Triton:

*   The batch dimension is folded into the program id, so a single
    kernel launch covers all batches.
*   Tiling is performed over the two output axes `i` and `j`
    (BLOCK_M × BLOCK_N) and over the reduction axis `k`
    (BLOCK_K).  The hidden dimension `d` is also tiled (BLOCK_H) and
    accumulated in FP32 for numeric stability.
*   The final output is stored in FP32 – later PyTorch layers (LayerNorm
    and the final Linear) operate in FP32 as required by the reference
    implementation.

The rest of the TriMul module (LayerNorms, Linear projections, gates,
and the final output projection) is kept in PyTorch because they are
memory‑bound and already highly optimized; only the cubic‑cost core is
off‑loaded to Triton.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -------------------------------------------------
# Triton kernel for the einsum reduction (1)
# -------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # shapes
    B, N, H,
    # strides
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    with tiling over i, j, k and the hidden dimension h.
    """

    pid = tl.program_id(0)

    # -------------------------------------------------
    # Decompose pid -> batch index + tile indices
    # -------------------------------------------------
    grid_m = tl.cdiv(N, BLOCK_M)
    grid_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = grid_m * grid_n

    b = pid // tiles_per_batch                     # batch index
    tile_id = pid % tiles_per_batch                # tile inside the batch
    pid_m = tile_id // grid_n                      # tile row (i)
    pid_n = tile_id % grid_n                       # tile col (j)

    # -------------------------------------------------
    # Offsets for the current tile
    # -------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, BLOCK_H)

    # Masks for the tail of i / j / h dimensions
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # -----------------------------------------------------------------
    # Accumulator (FP32) for the tile (M, N, H)
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over k dimension in blocks of size BLOCK_K
    # -----------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left[b,i,k,:]  -> shape (M, K, H)
        left = tl.load(
            left_ptr
            + b * stride_lb
            + offs_i[:, None, None] * stride_li
            + offs_k[None, :, None] * stride_lk
            + offs_h[None, None, :] * stride_lh,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        # Load right[b,j,k,:] -> shape (N, K, H)
        right = tl.load(
            right_ptr
            + b * stride_rb
            + offs_j[:, None, None] * stride_rj
            + offs_k[None, :, None] * stride_rk
            + offs_h[None, None, :] * stride_rh,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # Element‑wise multiply over the reduction axis k and accumulate.
        # left has shape (M, K, H), right has shape (N, K, H)
        # We need a broadcasting product over (M, N, K, H) and sum over K.
        # Triton can broadcast the K dimension by expanding left/right:
        acc += tl.sum(left[:, None, :, :] * right[None, :, :, :], axis=2)

    # -----------------------------------------------------------------
    # Store the result
    # -----------------------------------------------------------------
    out_ptr_base = (
        out_ptr
        + b * stride_ob
        + offs_i[:, None, None] * stride_oi
        + offs_j[None, :, None] * stride_oj
        + offs_h[None, None, :] * stride_oh
    )
    store_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptr_base, acc, mask=store_mask)


# -------------------------------------------------
# Helper: launch the kernel for a given tensor
# -------------------------------------------------
def launch_trimul_kernel(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]   (float32)
    returns out : [B, N, N, H]   (float32)
    """
    B, N, _, H = left.shape

    out = torch.empty_like(left)

    # strides in elements (not bytes) – Triton works with element strides
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Tile sizes – tuned for Hopper, can be adjusted if needed
    BLOCK_M = 64   # i‑axis
    BLOCK_N = 64   # j‑axis
    BLOCK_K = 64   # reduction axis
    BLOCK_H = 32   # hidden axis (must divide H for best perf)

    # Grid: one program per (batch * tile_i * tile_j)
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_tiles = B * grid_m * grid_n
    grid = (total_tiles,)

    trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    return out


# -------------------------------------------------
# Main entry point required by the evaluation harness
# -------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Arguments
    ----------
    data : Tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : [B, N, N, dim]   (float32 or bf16)
        - mask_tensor  : [B, N, N]       (float32, 0/1)
        - weights_dict : mapping from string to weight tensors
        - config_dict  : contains "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # -----------------------------
    # 1. LayerNorm on the input
    # -----------------------------
    # LayerNorm parameters
    eps = 1e-5
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    # compute mean / var per last dim (dim)
    x = input_tensor
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + eps)
    x = x * w_norm + b_norm

    # -----------------------------
    # 2. Linear projections (no bias)
    # -----------------------------
    # proj: [dim, hidden]
    left_proj_w = weights["left_proj.weight"]          # (hidden, dim) in PyTorch Linear
    right_proj_w = weights["right_proj.weight"]
    left = torch.einsum("bijk,dj->bijd", x, left_proj_w.t())   # shape [B,N,N,hidden]
    right = torch.einsum("bijk,dj->bijd", x, right_proj_w.t())

    # -----------------------------
    # 3. Apply mask (if present)
    # -----------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------
    # 4. Gating (sigmoid)
    # -----------------------------
    def linear_gate(tensor, weight):
        # weight shape (hidden, dim)
        return torch.einsum("bijk,dj->bijd", tensor, weight.t())

    left_gate = torch.sigmoid(linear_gate(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_gate(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear_gate(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------
    # 5. Core TriMul reduction – Triton kernel
    # -----------------------------
    # Move to float32 for the kernel (the reference does bfloat16 -> fp32,
    # we keep fp32 throughout for simplicity)
    left = left.to(torch.float32)
    right = right.to(torch.float32)

    out = launch_trimul_kernel(left, right)          # [B,N,N,hidden]

    # -----------------------------
    # 6. Final LayerNorm + gating + linear out
    # -----------------------------
    # to_out_norm is LayerNorm over hidden dim
    w_outnorm = weights["to_out_norm.weight"]
    b_outnorm = weights["to_out_norm.bias"]
    eps_out = 1e-5
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps_out)
    out = out * w_outnorm + b_outnorm

    out = out * out_gate

    # final linear projection back to dim
    to_out_w = weights["to_out.weight"]          # (dim, hidden)
    out = torch.einsum("bijk,dk->bijd", out, to_out_w.t())
    return out.to(torch.float32)
