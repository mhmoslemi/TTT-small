"""
TriMul “outgoing” kernel (AlphaFold‑3).

The heavy part of TriMul is the three‑tensor einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right are already masked and gated.
We implement this contraction with a custom Triton kernel that
    * iterates over the reduction axis k in blocks,
    * accumulates a vector of length `hidden_dim` (d) for every (b,i,j) pair,
    * stores the result back to a dense tensor.

All other operations (LayerNorm, linear projections, gates, final
normalisation and projection) are performed with highly‑optimised
torch.nn.functional calls.  The kernel is launched with a 1‑D grid of
size B·N·N (one program per output (i,j) pair) and uses
BLOCK_I/J/H/K tunable at launch time (default values work for the
tested shapes on an NVIDIA H200).
The final output is returned as float‑32.
"""
import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the 3‑tensor contraction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # --------------------------------------------------------------
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for a single (b,i,j) program; H‑dimension is vectorised.
    """
    pid = tl.program_id(0)                     # one program per (b,i,j)

    # decode pid -> (b,i,j)
    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i = ij // N
    j = ij % N

    # offsets for the current i / j
    offs_i = i * stride_left_i
    offs_j = j * stride_right_j

    # accumulator for the hidden dimension (vector length BLOCK_H)
    acc = tl.zeros((BLOCK_H,), dtype=tl.float32)

    # loop over k in BLOCK_K chunks
    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = start_k * BLOCK_K + tl.arange(0, BLOCK_K)          # [0, BLOCK_K)
        mask_k = cur_k < N

        # ------------------------------------------------------------------
        # Load left[b,i,k,:]  -> shape (BLOCK_K, BLOCK_H)
        # ------------------------------------------------------------------
        # pointer arithmetic:  b*stride_left_b + i*stride_left_i + k*stride_left_k + h*stride_left_h
        left_ptrs = (left_ptr
                     + b * stride_left_b
                     + offs_i
                     + cur_k[:, None] * stride_left_k
                     + tl.arange(0, BLOCK_H)[None, :] * stride_left_h)
        left = tl.load(left_ptrs, mask=mask_k[:, None] & (tl.arange(0, BLOCK_H)[None, :] < H), other=0.0)

        # ------------------------------------------------------------------
        # Load right[b,j,k,:] -> shape (BLOCK_K, BLOCK_H)
        # ------------------------------------------------------------------
        right_ptrs = (right_ptr
                      + b * stride_right_b
                      + offs_j
                      + cur_k[:, None] * stride_right_k
                      + tl.arange(0, BLOCK_H)[None, :] * stride_right_h)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & (tl.arange(0, BLOCK_H)[None, :] < H), other=0.0)

        # multiply‑accumulate over k
        acc += tl.sum(left * right, axis=0)        # result shape (BLOCK_H,)

    # ----------------------------------------------------------------------
    # Store the accumulated vector to out[b,i,j,:]
    # ----------------------------------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_out_b
                + i * stride_out_i
                + j * stride_out_j
                + tl.arange(0, BLOCK_H) * stride_out_h)
    tl.store(out_ptrs, acc, mask=tl.arange(0, BLOCK_H) < H)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a Triton‑accelerated
    contraction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]  (float32, on CUDA)
            - mask         : [B, N, N]       (float32 or bool, on CUDA)
            - weights      : dict of torch tensors (model parameters)
            - config       : dict with keys "dim", "hidden_dim"

    Returns:
        output tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last channel (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton contraction
    # ------------------------------------------------------------------
    # allocate output tensor in FP32
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements, not bytes) for the three tensors
    stride_left = left.stride()      # (B, N, N, H)
    stride_right = right.stride()
    stride_out = out.stride()

    # launch parameters
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_H = 32                         # hidden_dim ≤ 384, 32 works well

    grid = (B * N * N, )                 # 1‑D grid, one program per (b,i,j)

    _trimul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_h=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_j=stride_right[1],   # note: right uses dim 1 as j
        stride_right_k=stride_right[2],
        stride_right_h=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_h=stride_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (hidden_dim) + gating + final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
