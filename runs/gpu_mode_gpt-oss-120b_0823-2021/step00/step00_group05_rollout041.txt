"""
TriMul “outgoing” implementation.

*   LayerNorm, Linear projections and gating are performed with PyTorch
    (they are cheap compared to the N³ core).
*   The only heavy operation – the einsum  
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] – is executed in Triton.
*   The kernel computes one output element (b,i,j,d) per thread and
    accumulates over k in blocks (BLOCK_K = 32).  The kernel works for
    any batch size ≤ 2, any sequence length N and any hidden dimension
    H (must be a multiple of 32 for best performance, otherwise masking
    handles the tail).
*   All tensors are expected on the same CUDA device.  The final output
    is cast back to float32.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: computes the N³ einsum for a single (b,i,j,d)
# --------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr,          # const float*  left  [B,N,N,H]
    right_ptr,         # const float*  right [B,N,N,H]
    out_ptr,           #       float*  out   [B,N,N,H]

    B, N, H,

    # strides of left  (b,i,k,d)
    stride_lb, stride_li, stride_lk, stride_ld,
    # strides of right (b,j,k,d)
    stride_rb, stride_rj, stride_rk, stride_rd,
    # strides of out   (b,i,j,d)
    stride_ob, stride_oi, stride_oj, stride_od,

    BLOCK_K: tl.constexpr,               # k‑loop tile size
):
    # ----------------------------------------------------------
    # linear index → (b, i, j, d)
    # ----------------------------------------------------------
    pid = tl.program_id(0)
    total = B * N * N * H
    if pid >= total:
        return

    # decompose pid
    b = pid // (N * N * H)
    rem = pid - b * (N * N * H)
    d = rem // (N * N)
    rem2 = rem - d * (N * N)
    i = rem2 // N
    j = rem2 - i * N

    # accumulator in fp32
    acc = tl.float32(0.0)

    # ----------------------------------------------------------
    # loop over k in tiles
    # ----------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tile_k in range(0, num_k_tiles):
        # k indices inside this tile
        k_offset = tile_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # pointers for the current tile
        left_off  = (b * stride_lb
                     + i * stride_li
                     + k_offset * stride_lk
                     + d * stride_ld)
        right_off = (b * stride_rb
                     + j * stride_rj
                     + k_offset * stride_rk
                     + d * stride_rd)

        left_tile  = tl.load(left_ptr + left_off,  mask=mask_k, other=0.0).to(tl.float32)
        right_tile = tl.load(right_ptr + right_off, mask=mask_k, other=0.0).to(tl.float32)

        # element‑wise product + reduction over k
        acc += tl.dot(left_tile, right_tile)

    # ----------------------------------------------------------
    # write result
    # ----------------------------------------------------------
    out_off = (b * stride_ob
               + i * stride_oi
               + j * stride_oj
               + d * stride_od)
    tl.store(out_ptr + out_off, acc)


# --------------------------------------------------------------
# Host‑side wrapper
# --------------------------------------------------------------
def _launch_triton_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : torch.Tensor of shape [B,N,N,H] (dtype=torch.float32)
    returns out  : torch.Tensor of shape [B,N,N,H] (dtype=torch.float32)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape

    out = torch.empty_like(left)

    # strides as int64 (bytes / element) → divide by element size (4) for pointer arithmetic
    ls = [s // left.element_size() for s in left.stride()]
    rs = [s // right.element_size() for s in right.stride()]
    os = [s // out.element_size() for s in out.stride()]

    # launch configuration
    total_elements = B * N * N * H
    BLOCK_K = 32                                 # fits well in registers / shared memory
    grid = (triton.cdiv(total_elements, 1),)      # 1‑D grid, each thread handles 1 element

    _trimul_einsum_kernel[grid](
        left_ptr   = left,
        right_ptr  = right,
        out_ptr    = out,
        B = B, N = N, H = H,
        stride_lb = ls[0], stride_li = ls[1], stride_lk = ls[2], stride_ld = ls[3],
        stride_rb = rs[0], stride_rj = rs[1], stride_rk = rs[2], stride_rd = rs[3],
        stride_ob = os[0], stride_oi = os[1], stride_oj = os[2], stride_od = os[3],
        BLOCK_K = BLOCK_K,
    )
    return out


# --------------------------------------------------------------
# Main entry point expected by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
              input_tensor – [B, N, N, dim]  (float32)
              mask         – [B, N, N]       (float32 or bool)
              weights      – dict of model weights
              config       – dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------
    # Helper: LayerNorm implemented with explicit parameters
    # ------------------------------------------------------
    def layer_norm(x, weight, bias, eps=1e-5):
        # x: [..., C]
        mean = x.mean(dim=-1, keepdim=True)
        var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
        x_norm = (x - mean) / torch.sqrt(var + eps)
        return x_norm * weight + bias

    # ------------------------------------------------------
    # Load all weight tensors (they are already on `device`)
    # ------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------
    # 1. Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------
    x = layer_norm(input_tensor, norm_w, norm_b)      # [B,N,N,dim]

    # ------------------------------------------------------
    # 2. Linear projections (no bias) – use torch.einsum for speed
    # ------------------------------------------------------
    # left = x @ left_proj_w.T   shape [B,N,N,hidden_dim]
    left = torch.einsum("bnijc,ch->bnijh", x, left_proj_w)   # c=dim, h=hidden_dim
    right = torch.einsum("bnijc,ch->bnijh", x, right_proj_w)

    # ------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise gating
    # ------------------------------------------------------
    left_gate = torch.sigmoid(torch.einsum("bnijc,ch->bnijh", x, left_gate_w))
    right_gate = torch.sigmoid(torch.einsum("bnijc,ch->bnijh", x, right_gate_w))
    out_gate = torch.sigmoid(torch.einsum("bnijc,ch->bnijh", x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------
    # 5. Core N³ multiplication – Triton kernel
    # ------------------------------------------------------
    # left/right are [B,N,N,hidden_dim] float32
    out = _launch_triton_einsum(left, right)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------
    # 6. Output transformation
    # ------------------------------------------------------
    out = layer_norm(out, to_out_norm_w, to_out_norm_b)   # [B,N,N,hidden_dim]
    out = out * out_gate                                 # gating after norm
    # final linear back to `dim`
    out = torch.einsum("bnijh,ch->bnijc", out, to_out_w)  # [B,N,N,dim]

    return out.to(torch.float32)
