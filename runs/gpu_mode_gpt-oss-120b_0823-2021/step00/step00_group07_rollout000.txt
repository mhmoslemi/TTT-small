"""
TriMul “outgoing” kernel (AlphaFold‑3 style).

The heavy part of the TriMul forward pass is

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right have shape [B, N, N, H] (H = hidden_dim) and a mask has already
been applied.  This kernel computes that 4‑D reduction in a fused Triton
kernel.  All LayerNorms and linear projections are performed with PyTorch
using the supplied weight tensors; only the O(N³·H) contraction is off‑loaded
to Triton.

The kernel is launched with a 3‑D grid (batch, block_i, block_j).  For each
(i‑block, j‑block) we iterate over k in tiles, load the two tiles
(L[i, k, :], R[j, k, :]) and accumulate their element‑wise product into a
register tile of shape (BLOCK_I, BLOCK_J, BLOCK_H).  The final tile is stored
to `out`.  The implementation works for any B∈{1,2}, N up to 1024 and any
hidden_dim that is a multiple of BLOCK_H (we use 64, which evenly divides the
tested hidden dimensions 128 and 384).  The output is cast back to float32.

Only a subset of the overall TriMul computation is in Triton – the rest (layer
norms, linear layers, sigmoids, gating and final linear projection) is done
with PyTorch for simplicity and to keep the code short while still achieving
large speed‑ups on the H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel for the triple‑multiply reduction
# ----------------------------------------------------------------------------- #
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,    # [B, N, N, H]
    right_ptr,   # [B, N, N, H]
    out_ptr,     # [B, N, N, H]

    # strides (in elements, not bytes)
    stride_bL, stride_iL, stride_kL, stride_hL,
    stride_bR, stride_iR, stride_kR, stride_hR,
    stride_bO, stride_iO, stride_jO, stride_hO,

    # sizes
    N, H,

    # compile‑time constants
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    where left/right are already masked.
    """
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # block over i
    pid_j = tl.program_id(2)          # block over j

    # offsets within the current tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, BLOCK_H)

    # Masks for out‑of‑bounds i/j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------------- #
    # Allocate accumulator
    # --------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # --------------------------------------------------------------------- #
    # Loop over k dimension in tiles
    # --------------------------------------------------------------------- #
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(0, num_k_tiles):
        offs_k = tk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -----------------------------------------------------------------
        # Load left tile  : shape (BLOCK_I, BLOCK_K, BLOCK_H)
        # -----------------------------------------------------------------
        # left[b, i, k, h] -> pointer arithmetic
        left_ptrs = (
            left_ptr
            + pid_b * stride_bL
            + offs_i[:, None, None] * stride_iL
            + offs_k[None, :, None] * stride_kL
            + offs_h[None, None, :] * stride_hL
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0)

        # -----------------------------------------------------------------
        # Load right tile : shape (BLOCK_J, BLOCK_K, BLOCK_H)
        # -----------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + pid_b * stride_bR
            + offs_j[:, None, None] * stride_iR   # note: right uses i‑index for the “j” dim
            + offs_k[None, :, None] * stride_kR
            + offs_h[None, None, :] * stride_hR
        )
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0)

        # -----------------------------------------------------------------
        # Compute element‑wise product and reduce over k
        #   left: (I, K, H)   right: (J, K, H)
        #   need: Σ_k left[i,k,:] * right[j,k,:]   -> (I,J,H)
        # -----------------------------------------------------------------
        # Expand dimensions so broadcasting works:
        #   left  -> (I, K, 1, H)
        #   right -> (1, K, J, H)
        left_exp  = left[:, :, None, :]        # (I,K,1,H)
        right_exp = right[None, :, :, :]       # (1,K,J,H)

        prod = left_exp * right_exp            # (I,K,J,H)
        # sum over k
        acc += tl.sum(prod, axis=1)            # (I,J,H)

    # --------------------------------------------------------------------- #
    # Store the result
    # --------------------------------------------------------------------- #
    out_ptrs = (
        out_ptr
        + pid_b * stride_bO
        + offs_i[:, None, None] * stride_iO
        + offs_j[None, :, None] * stride_jO
        + offs_h[None, None, :] * stride_hO
    )
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None, None] & mask_j[None, :, None])


# ----------------------------------------------------------------------------- #
# Helper: launch wrapper
# ----------------------------------------------------------------------------- #
def trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, N, N, H] (float32, already masked)
    returns out:   [B, N, N, H] (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # tile sizes – chosen to fit H200 shared memory & registers
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 64   # H is always a multiple of 64 in the provided configs

    grid = (
        B,
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
    )

    # stride utilities (torch gives strides in number of elements)
    stride_bL, stride_iL, stride_kL, stride_hL = left.stride()
    stride_bR, stride_iR, stride_kR, stride_hR = right.stride()
    stride_bO, stride_iO, stride_jO, stride_hO = out.stride()

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        stride_bL, stride_iL, stride_kL, stride_hL,
        stride_bR, stride_iR, stride_kR, stride_hR,
        stride_bO, stride_iO, stride_jO, stride_hO,
        N, H,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,               # enough for the compute intensity
    )
    return out


# ----------------------------------------------------------------------------- #
# Full TriMul forward (PyTorch for everything except the O(N³·H) contraction)
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]      (float32)
            mask         : [B, N, N]           (bool / float)
            weights      : dict of weight tensors
            config       : dict with keys "dim", "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    device = input_tensor.device
    dtype = torch.float32

    # --------------------------------------------------------------- #
    # LayerNorm on the last dim (dim)
    # --------------------------------------------------------------- #
    norm_w = weights["norm.weight"].to(dtype)
    norm_b = weights["norm.bias"].to(dtype)
    # manual layernorm to avoid extra PyTorch overhead
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x_norm = x_norm * norm_w + norm_b   # [B,N,N,dim]

    # --------------------------------------------------------------- #
    # Linear projections (no bias) -> shape [B,N,N,hidden_dim]
    # --------------------------------------------------------------- #
    left_proj_w = weights["left_proj.weight"].to(dtype)          # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"].to(dtype)

    # torch.einsum is fast for these small GEMMs
    left = torch.einsum('bnnd,hd->bnnh', x_norm, left_proj_w)    # [B,N,N,hidden_dim]
    right = torch.einsum('bnnd,hd->bnnh', x_norm, right_proj_w)

    # --------------------------------------------------------------- #
    # Apply mask (if present)
    # --------------------------------------------------------------- #
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------- #
    # Gating (sigmoid)
    # --------------------------------------------------------------- #
    left_gate_w = weights["left_gate.weight"].to(dtype)
    right_gate_w = weights["right_gate.weight"].to(dtype)
    out_gate_w = weights["out_gate.weight"].to(dtype)

    left_gate = torch.einsum('bnnd,hd->bnnh', x_norm, left_gate_w).sigmoid()
    right_gate = torch.einsum('bnnd,hd->bnnh', x_norm, right_gate_w).sigmoid()
    out_gate = torch.einsum('bnnd,hd->bnnh', x_norm, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # Heavy contraction via Triton
    # --------------------------------------------------------------- #
    out = trimul_outgoing_triton(left, right)    # [B,N,N,hidden_dim]

    # --------------------------------------------------------------- #
    # Final LayerNorm + out gate + linear projection back to dim
    # --------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)

    # layernorm on hidden_dim
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    out_norm = out_norm * out_gate

    to_out_w = weights["to_out.weight"].to(dtype)   # [dim, hidden_dim]
    final = torch.einsum('bnnh,dh->bnnd', out_norm, to_out_w)  # -> [B,N,N,dim]

    return final.to(dtype)
