"""
TriMul “outgoing” implementation.

The heavy part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batch of H (= hidden_dim) independent matrix multiplications:
for each channel d we need (N×N) @ (N×N)ᵀ.
The kernel below performs a fused batched GEMM on
    A : (B*H, N, K)   with K = N
    B : (B*H, N, K)   (interpreted transposed, i.e. we use B_{k,j})
and writes the result C : (B*H, N, N).

All layer‑norms, linear projections and gating are kept in PyTorch
(they are cheap compared with the O(N³) product).
Only the einsum is executed on the GPU via Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ---------------------------------------------------------------------------
# Triton kernel: batched GEMM with the second operand interpreted transposed.
# ---------------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    batch_h, m, n, k,                  # logical sizes
    stride_ah, stride_ak,              # A strides: (batch_h, m, k)
    stride_bh, stride_bk, stride_bn,  # B strides: (batch_h, k, n)   (k is the fast dim)
    stride_ch, stride_cm, stride_cn,  # C strides: (batch_h, m, n)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[bh, i, j] = Σ_k A[bh, i, k] * B[bh, j, k]   (B is logically transposed).
    """
    pid_bh = tl.program_id(0)                     # which (batch * hidden) matrix
    pid_m  = tl.program_id(1)                     # block row inside the matrix
    pid_n  = tl.program_id(2)                     # block column inside the matrix

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < m
    mask_n = offs_n < n

    # pointers for the current batch‑hidden slice
    a_base = a_ptr + pid_bh * stride_ah
    b_base = b_ptr + pid_bh * stride_bh
    c_base = c_ptr + pid_bh * stride_ch

    # load A tile   shape (BLOCK_M, BLOCK_K)
    a_ptrs = a_base + (offs_m[:, None] * stride_ak) + (offs_k[None, :] * stride_ak)
    a = tl.load(a_ptrs, mask=mask_m[:, None], other=0.0).to(tl.float16)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k_idx in range(0, tl.cdiv(k, BLOCK_K)):
        # compute global k offset
        cur_k = k_idx * BLOCK_K
        cur_k = tl.minimum(cur_k + offs_k, k - 1)

        # B is logically transposed: we need B[bh, k, j]
        b_ptrs = b_base + (cur_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=cur_k[:, None] < k, other=0.0).to(tl.float16)

        # dot product over K dimension
        acc += tl.dot(a, b)          # (M,K)·(K,N) → (M,N)

        # advance A pointer for next K‑tile
        a_ptrs = a_ptrs + BLOCK_K * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None], other=0.0).to(tl.float16)

    # write back
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ---------------------------------------------------------------------------
# Helper to launch the kernel
# ---------------------------------------------------------------------------
def tri_mul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]   (float16 or float32)
    returns      : [B, N, N, H]   (same dtype as input, but we compute in fp16)
    """
    B, N, _, H = left.shape
    # flatten batch and channel dimensions
    left = left.reshape(B * H, N, N).contiguous()
    right = right.reshape(B * H, N, N).contiguous()

    # output buffer
    out = torch.empty_like(left)

    # strides (in elements) for the flattened tensors
    stride_ah = left.stride(0)   # batch*H stride
    stride_ak = left.stride(1)   # row stride (N)
    stride_al = left.stride(2)   # col stride (1)

    stride_bh = right.stride(0)
    stride_bk = right.stride(1)
    stride_bn = right.stride(2)

    stride_ch = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    # choose tile sizes (tuned for Hopper, fp16)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_gemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B * H,
        N,
        N,
        N,
        stride_ah,
        stride_al,
        stride_bh,
        stride_bk,
        stride_bn,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to original 4‑D layout
    return out.reshape(B, N, N, H)


# ---------------------------------------------------------------------------
# Main entry point
# ---------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module, using a fused Triton kernel for
    the O(N³) einsum. All other ops (LayerNorm, Linear, Sigmoid, masking) are
    performed with PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = torch.float32

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections ----------
    # Linear: y = x @ Wᵀ   (torch.nn.functional.linear does this)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)            # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Core einsum via Triton ----------
    # Cast to fp16 for the kernel (still keep a copy in fp32 for later layers)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out = tri_mul_gemm(left_fp16, right_fp16)          # [B,N,N,H] fp16
    out = out.to(dtype)                               # back to fp32

    # ---------- Output projection ----------
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
