"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication for every hidden
dimension *d*:

    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T

We implement this inner product with a Triton kernel.  The kernel is launched
with three program‑ids:

* pid_m – block of the *i* axis (size BLOCK_M)
* pid_n – block of the *j* axis (size BLOCK_N)
* pid_h – the hidden dimension *d* (size 1 → one kernel per hidden channel)

For each (b,i,j,d) tile the kernel iterates over *k* in BLOCK_K steps,
loads the left and right tiles (shape BLOCK_M×BLOCK_K and BLOCK_N×BLOCK_K)
and accumulates the dot‑product using `tl.dot`.  All computations are done
in FP16 for speed, the final result is stored in FP32 to match the PyTorch
reference.

The surrounding LayerNorm, Linear projections and gating are kept in PyTorch
(because they are cheap compared with the O(N³) term) and only the einsum
is off‑loaded to Triton.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: per‑hidden‑channel batched matrix multiplication
# -------------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # *   [B, N, N, H]  (i,k,d)
    right_ptr,              # *   [B, N, N, H]  (j,k,d)
    out_ptr,                # *   [B, N, N, H]  (i,j,d)
    B, N, H,
    stride_lbn, stride_lbk, stride_lbh,   # left  strides (b, i, k, d)
    stride_rbn, stride_rbk, stride_rbh,   # right strides (b, j, k, d)
    stride_obn, stride_obh,               # out   strides (b, i, j, d)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch dimension
    pid_m = tl.program_id(1)          # i block
    pid_n = tl.program_id(2)          # j block
    pid_h = tl.program_id(3)          # hidden channel (d)

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers for this (b, d) slice
    left_base  = left_ptr  + pid_b * stride_lbn + pid_h * stride_lbh
    right_base = right_ptr + pid_b * stride_rbn + pid_h * stride_rbh
    out_base   = out_ptr   + pid_b * stride_obn + pid_h * stride_obh

    # initialise accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < N

        # left: [M, K]
        l_ptr = left_base + (offs_m[:, None] * stride_lbk) + (cur_k[None, :] * stride_lbk)
        l = tl.load(l_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # right: [N, K] (need the same K ordering)
        r_ptr = right_base + (offs_n[:, None] * stride_rbk) + (cur_k[None, :] * stride_rbk)
        r = tl.load(r_ptr, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # accumulate dot‑product over K
        # tl.dot works on vectors; we use broadcasting on the K‑axis
        acc += tl.dot(l, r, trans_b=False)   # (M,N) += (M,K)·(N,K)ᵀ
        # advance pointers (not needed because we recompute each iteration)

    # write result
    out_ptrs = out_base + (offs_m[:, None] * stride_obh) + (offs_n[None, :] * stride_obh)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------------------
# Helper to launch the kernel
# -------------------------------------------------------------------------
def trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]   (float16 or float32)
    returns      : [B, N, N, H]   (float32)
    """
    B, N, _, H = left.shape
    assert right.shape == (B, N, N, H)

    # ensure fp16 for the compute (cast if necessary)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides (in elements) for the contiguous layout
    stride_lbn, stride_lik, stride_lkj, stride_ldh = left.stride()
    stride_rbn, stride_rjk, stride_rkj, stride_rdh = right.stride()
    stride_obn, stride_oij, stride_oij2, stride_odh = out.stride()

    # Triton block sizes – tuned for Hopper, can be adjusted
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        H,                     # one program-id per hidden channel
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lbn,
        stride_lik,
        stride_lkj,
        stride_ldh,
        stride_rbn,
        stride_rjk,
        stride_rkj,
        stride_rdh,
        stride_obn,
        stride_oij,
        stride_odh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# -------------------------------------------------------------------------
# Full custom kernel entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
        input_tensor : [B, N, N, dim]   (float32)
        mask         : [B, N, N]        (float32 or bool)
        weights      : dict of weight tensors
        config       : dict with keys "dim" and "hidden_dim"
    Returns:
        output tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm ----------
    norm_w = weights["norm.weight"].to(input_tensor.dtype)
    norm_b = weights["norm.bias"].to(input_tensor.dtype)
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_w + norm_b                # [B,N,N,dim]

    # ---------- Linear projections ----------
    # x shape: [B,N,N,dim]
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)            # (B*N*N, dim)

    left_proj_w = weights["left_proj.weight"].to(x.dtype)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(x.dtype)

    left = torch.nn.functional.linear(x_flat, left_proj_w)   # (B*N*N, hidden)
    right = torch.nn.functional.linear(x_flat, right_proj_w)

    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # ---------- Mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gates ----------
    left_gate_w = weights["left_gate.weight"].to(x.dtype)
    right_gate_w = weights["right_gate.weight"].to(x.dtype)
    out_gate_w = weights["out_gate.weight"].to(x.dtype)

    left_gate = torch.nn.functional.linear(x_flat, left_gate_w).view(B, N, N, hidden_dim).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, right_gate_w).view(B, N, N, hidden_dim).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, out_gate_w).view(B, N, N, hidden_dim).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Core TriMul einsum (Triton) ----------
    out = trimul_outgoing_triton(left, right)      # [B,N,N,hidden]   fp32

    # ---------- Output transformation ----------
    to_out_norm_w = weights["to_out_norm.weight"].to(out.dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(out.dtype)

    # LayerNorm over hidden dim
    mean = out.mean(-1, keepdim=True)
    var = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean) / torch.sqrt(var + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(out.dtype)   # (dim, hidden)
    out = torch.nn.functional.linear(out.view(-1, hidden_dim), to_out_w).view(B, N, N, dim)

    return out
