"""
TriMul "outgoing" implementation.

We perform the expensive part
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
with a Triton kernel.  The surrounding LayerNorm / gating / linear
layers stay in PyTorch (they are cheap compared to the N³ reduction).
The kernel works for any batch size ≤2, sequence length up to 1024 and
hidden dimension that is a multiple of 32 (the hidden dimensions used in the
tests are 128 and 384).  The final output is cast back to float32.
"""
import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: N³ reduction over the “k” dimension
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # sizes
    B, N, H,
    # strides (flattened, last dimension changes fastest)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_i = tl.program_id(0)   # tile over i
    pid_j = tl.program_id(1)   # tile over j
    pid_b = tl.program_id(2)   # batch (≤2)

    # start indices of the tile
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    b = pid_b

    # offsets within the tile
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, BLOCK_H)

    # masks for out‑of‑bounds (when N is not a multiple of the block size)
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over k in BLOCK_K chunks
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # ------- load left[b,i,k,:] -------
        # shape (BLOCK_I, BLOCK_K, BLOCK_H)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + cur_k[None, :, None] * stride_left_k
            + offs_h[None, None, :] * stride_left_h
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0)

        # ------- load right[b,j,k,:] -------
        # shape (BLOCK_J, BLOCK_K, BLOCK_H)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_j[:, None, None] * stride_right_j
            + cur_k[None, :, None] * stride_right_k
            + offs_h[None, None, :] * stride_right_h
        )
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0)

        # accumulate: for each k we need left[i,k,:] * right[j,k,:] (element‑wise)
        # left  -> (BLOCK_I, BLOCK_K, BLOCK_H)
        # right -> (BLOCK_J, BLOCK_K, BLOCK_H)
        # we broadcast over k and sum
        #   left[:,k,:] (i,h)  *  right[:,k,:] (j,h)  -> (i,j,h)
        # This can be done with a simple loop over k (BLOCK_K ≤ 32)
        for kk in range(BLOCK_K):
            km = mask_k[kk]
            if km:
                l = left[:, kk, :]            # (BLOCK_I, BLOCK_H)
                r = right[:, kk, :]           # (BLOCK_J, BLOCK_H)
                # broadcasting: (i,1,h) * (1,j,h) -> (i,j,h)
                acc += tl.broadcast_to(l[:, None, :], (BLOCK_I, BLOCK_J, BLOCK_H)) \
                     * tl.broadcast_to(r[None, :, :], (BLOCK_I, BLOCK_J, BLOCK_H))

    # ------- store -------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_h[None, None, :] * stride_out_h
    )
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])


# ----------------------------------------------------------------------
# Host‑side helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]  (float32)
    returns      : [B, N, N, H]  (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes) – triton expects element‑wise strides
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # tile size choices – tuned for Hopper (SM90)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 128   # hidden_dim is a multiple of 128 in the tests

    grid = (
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        B,
    )
    trimul_kernel[grid](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
        num_warps=4,  # enough for the compute intensity
    )
    return out


# ----------------------------------------------------------------------
# Full TriMul forward (mostly PyTorch, kernel for the N³ part)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    data = (input, mask, weights, config)

    Implements the “outgoing” TriMul forward pass.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm on the input ----------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = (input_tensor - (input_tensor * norm_w + norm_b).mean(dim=-1, keepdim=True)) / torch.sqrt(
        ((input_tensor * norm_w + norm_b) ** 2).mean(dim=-1, keepdim=True) + 1e-5
    )
    # Simpler: use torch.nn.functional.layer_norm (fast enough)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )

    # ---------- Linear projections ----------
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"]
    )          # [B,N,N,hidden]
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"]
    )          # [B,N,N,hidden]

    # ---------- Apply mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gating ----------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # ---------- N³ reduction via Triton ----------
    # Cast to float32 for the kernel (the kernel only supports fp32)
    left = left.to(torch.float32)
    right = right.to(torch.float32)
    out = trimul_triton(left, right)          # [B,N,N,hidden]

    # ---------- Output norm, gating and final linear ----------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out.to(torch.float32)
