"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* this is a standard matrix‑multiply:
    out_d = left_d @ right_d.T          (shape N×N)

The kernel below computes many (i,j) tiles for a block of hidden
dimensions at once, accumulating in float32.  Everything else
(LayerNorm, Linear, gating, final LayerNorm/Linear) is done with
regular torch kernels – they are cheap compared with the N³ work.

The kernel is launched with a 3‑D grid:
    (grid_i, grid_j, grid_bd) where
        grid_i = ceil(N / BLOCK_M)
        grid_j = ceil(N / BLOCK_N)
        grid_bd = B * ceil(H / BLOCK_H)
Each program instance therefore knows its batch index,
its hidden‑block index, and the (i,j) tile it must compute.
The inner‑most loop iterates over K in BLOCK_K steps and uses
tl.dot to multiply a (BLOCK_M, BLOCK_K) tile of *left* with a
(BLOCK_K, BLOCK_H) tile of *right* (the K‑dimension is shared).
All arithmetic is carried out in float32, the result is stored as
float16 to save bandwidth and finally cast back to float32 on the
host.
"""
import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# --------------------------------------------------------------
# Triton kernel: batched‑per‑hidden‑dim GEMM (left @ right.T)
# --------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,               # [B, N, N, H]  left tensor
    right_ptr,              # [B, N, N, H]  right tensor
    out_ptr,                # [B, N, N, H]  output
    B, N, H,
    stride_lb, stride_lk, stride_ld,   # left strides  (b, i, k, d)
    stride_rb, stride_rj, stride_rd,   # right strides (b, j, k, d)
    stride_ob, stride_oi, stride_od,   # out strides   (b, i, j, d)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(axis=0)          # combine batch * hidden block
    pid_i = tl.program_id(axis=1)        # tile over i dimension
    pid_j = tl.program_id(axis=2)        # tile over j dimension

    # ------------------------------------------------------------------
    # decode program ids
    # ------------------------------------------------------------------
    # batch index and hidden‑block index
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    b = pid // num_h_blocks
    h_block = pid % num_h_blocks
    # tile origins
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    d0 = h_block * BLOCK_H

    # block ranges
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_d = d0 + tl.arange(0, BLOCK_H)

    # masks for out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # ------------------------------------------------------------------
    # pointers to the start of the tiles for this program
    # ------------------------------------------------------------------
    # left:   [b, i, k, d]
    left_tile_ptr = left_ptr + b * stride_lb \
                               + offs_i[:, None, None] * stride_lk \
                               + 0 * stride_ld \
                               + offs_d[None, None, :] * stride_ld

    # right:  [b, j, k, d]  (we need right.T -> we keep k as the shared dim)
    right_tile_ptr = right_ptr + b * stride_rb \
                                 + offs_j[:, None, None] * stride_rj \
                                 + 0 * stride_rd \
                                 + offs_d[None, None, :] * stride_rd

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over k dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute actual k offset
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile   shape (M, K, H)
        l_ptr = left_tile_ptr + offs_k[None, :, None] * stride_lk
        left = tl.load(l_ptr, mask=mask_i[:, None] & mask_k[None, :] & mask_d[None, None, :],
                       other=0.0).to(tl.float32)

        # load right tile  shape (N, K, H)  (note we will dot with left over K)
        r_ptr = right_tile_ptr + offs_k[None, :, None] * stride_rd
        right = tl.load(r_ptr, mask=mask_j[:, None] & mask_k[None, :] & mask_d[None, None, :],
                        other=0.0).to(tl.float32)

        # dot over K -> (M,N,H)
        # tl.dot works on 2‑D matrices; we vectorise over H by looping manually.
        # Unroll a tiny loop (BLOCK_H is modest, 128/256) – use tensor dot.
        # Compute (M,K) @ (N,K)^T for each hidden channel independently.
        # We can use broadcasting: left[..., :] @ right[..., :].T
        # but Triton only has tl.dot for 2‑D, so we compute per channel.
        for dh in range(0, BLOCK_H):
            # guard hidden dim
            if not mask_d[dh]:
                continue
            l = left[:, :, dh]          # (M, K)
            r = right[:, :, dh]          # (N, K)
            # r needs to be transposed: (K, N)
            acc[:, :, dh] += tl.dot(l, r, trans_b=True)

    # ------------------------------------------------------------------
    # write result
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + b * stride_ob \
                        + offs_i[:, None, None] * stride_oi \
                        + offs_j[None, :, None] * stride_od \
                        + offs_d[None, None, :] * stride_od
    tl.store(out_ptrs,
             acc.to(tl.float16),          # store as fp16 to reduce traffic
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])


# --------------------------------------------------------------
# Helper: fused LayerNorm (weight & bias are 1‑D)
# --------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    # x: (..., C)
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: float32 tensor [B, N, N, dim]
            - mask:          float32 or bool tensor [B, N, N] (may be all‑ones)
            - weights: dict of all model parameters (already on CUDA)
            - config:  dict with keys "dim", "hidden_dim"

    Returns:
        out: float32 tensor [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------
    x = layer_norm(input_tensor,
                   weights["norm.weight"],
                   weights["norm.bias"])

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    # (B,N,N,dim) @ (dim, hidden) -> (B,N,N,hidden)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Apply mask if required (mask is broadcastable)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) and multiplication
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton kernel for the einsum (heavy N³ work)
    # --------------------------------------------------------------
    # allocate output (float32, will be cast from fp16 later)
    out = torch.empty_like(left, dtype=torch.float16, device=device)  # temporary fp16

    # strides (in elements)
    stride_lb = left.stride(0)   # B
    stride_lk = left.stride(1)   # i
    stride_ld = left.stride(2)   # k (shared)
    stride_lh = left.stride(3)   # hidden dimension

    stride_rb = right.stride(0)  # B
    stride_rj = right.stride(1)  # j
    stride_rk = right.stride(2)  # k (shared)
    stride_rh = right.stride(3)  # hidden

    stride_ob = out.stride(0)    # B
    stride_oi = out.stride(1)    # i
    stride_oj = out.stride(2)    # j
    stride_oh = out.stride(3)    # hidden

    # block sizes – tuned for Hopper SM90 (shared‑memory friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 32

    grid = (
        triton.cdiv(B * triton.cdiv(hidden, BLOCK_H), 1),   # pid over (batch, hidden‑block)
        triton.cdiv(N, BLOCK_M),                           # i‑tiles
        triton.cdiv(N, BLOCK_N),                           # j‑tiles
    )

    trimul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=hidden,
        stride_lb=left.stride(0), stride_lk=left.stride(1), stride_ld=left.stride(2),
        stride_rb=right.stride(0), stride_rj=right.stride(1), stride_rd=right.stride(2),
        stride_ob=out.stride(0), stride_oi=out.stride(1), stride_od=out.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
        # Note: Triton infers the type of the strides from the tensors; we only
        # need to pass them as integer arguments.
    )

    # Cast back to float32 for the remaining ops
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # 6. Final LayerNorm + gating + linear projection back to dim
    # --------------------------------------------------------------
    out = layer_norm(out,
                     weights["to_out_norm.weight"],
                     weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
