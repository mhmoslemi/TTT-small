"""
TriMul “outgoing” kernel (forward only)

We compute the heavy 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where
    left  = (x·W_left )   ⊙ left_gate   ⊙ mask
    right = (x·W_right)   ⊙ right_gate  ⊙ mask

Only the N³ reduction is off‑loaded to Triton.  All LayerNorms,
Linear projections and gating are cheap and stay in PyTorch.
The kernel works for any batch size ≤ 2, any sequence length up to 1024,
and any hidden dimension that is a multiple of 32 (the default
BLOCK_D).  The final output is recovered in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: computes the N³ sum‑product.
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # const float*  [B, N, N, H]
    right_ptr,         # const float*  [B, N, N, H]
    out_ptr,           #       float*  [B, N, N, H]
    B, N, H,           # dimensions (int32)
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)               # 0‑dim grid: tiles over (i,j)
    batch = tl.program_id(1)             # 1‑dim grid: batch index

    # ---- tile coordinates ------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_I)
    i_tile = (pid % num_i) * BLOCK_I
    j_tile = (pid // num_i) * BLOCK_J

    # ---- offsets ---------------------------------------------------------
    offs_i = i_tile + tl.arange(0, BLOCK_I)
    offs_j = j_tile + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)
    offs_d = tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N
    mask_d = offs_d < H

    # pointers for the current batch
    left_base  = left_ptr  + batch * stride_lb
    right_base = right_ptr + batch * stride_rb
    out_base   = out_ptr   + batch * stride_ob

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ---- main K‑reduction loop -------------------------------------------
    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_start * BLOCK_K + offs_k
        cur_k_mask = cur_k < N

        # ---- load left[b, i, cur_k, d]  -> (I, K, D) --------------------
        left_ptrs = (left_base
                     + offs_i[:, None, None] * stride_li
                     + cur_k[None, :, None] * stride_lk
                     + offs_d[None, None, :] * stride_lh)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & cur_k_mask[None, :, None] & mask_d[None, None, :],
                       other=0.0)

        # ---- load right[b, j, cur_k, d] -> (J, K, D) --------------------
        right_ptrs = (right_base
                      + offs_j[:, None, None] * stride_rj
                      + cur_k[None, :, None] * stride_rk
                      + offs_d[None, None, :] * stride_rh)
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] & cur_k_mask[None, :, None] & mask_d[None, None, :],
                        other=0.0)

        # ---- accumulate: Σ_k left[i,k,d] * right[j,k,d] ------------------
        # we broadcast over the K dimension manually
        # left  : (I, K, D) -> (I, 1, K, D)
        # right : (J, K, D) -> (1, J, K, D)
        # product -> (I, J, K, D) -> sum over K
        prod = left[:, None, :, :] * right[None, :, :, :]
        acc += tl.sum(prod, axis=2)   # sum over K

    # ---- write result ----------------------------------------------------
    out_ptrs = (out_base
                + offs_i[:, None, None] * stride_oi
                + offs_j[None, :, None] * stride_oj
                + offs_d[None, None, :] * stride_oh)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul “outgoing” operator.

    Arguments
    ---------
    data : (input, mask, weights, config)
        input  – [B, N, N, C]  (float32/float16)
        mask   – [B, N, N]    (float32, 0/1)
        weights – mapping with all linear / layer‑norm parameters
        config  – dict with keys ``dim`` (C) and ``hidden_dim`` (H)

    Returns
    -------
    torch.Tensor  – [B, N, N, C]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (C‑dim)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    # manual LayerNorm to avoid extra Python overhead
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    eps = 1e-5
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                         # [B,N,N,C]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) – use F.linear for speed
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device)      # [H, C]
    right_proj_w = weights["right_proj.weight"].to(device)    # [H, C]

    left = F.linear(x, left_proj_w)     # [B,N,N,H]
    right = F.linear(x, right_proj_w)   # [B,N,N,H]

    # --------------------------------------------------------------
    # 3. Mask (optional) – broadcast on hidden dim
    # --------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gating
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device)   # [H, C]
    right_gate_w = weights["right_gate.weight"].to(device) # [H, C]
    out_gate_w = weights["out_gate.weight"].to(device)     # [H, C]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton kernel for the N³ reduction
    # --------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides (contiguous layout)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # tile sizes – chosen to fit H200 shared memory & occupancy
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 16
    BLOCK_D = 32                     # H must be multiple of 32 (true for configs)

    grid_i = triton.cdiv(N, BLOCK_I)
    grid_j = triton.cdiv(N, BLOCK_J)
    grid = (grid_i * grid_j, B)      # (tiles over i,j) × (batch)

    trimul_kernel[grid](
        left_ptr   = left,
        right_ptr  = right,
        out_ptr    = out,
        B = B, N = N, H = H,
        stride_lb = stride_lb, stride_li = stride_li, stride_lk = stride_lk, stride_lh = stride_lh,
        stride_rb = stride_rb, stride_rj = stride_rj, stride_rk = stride_rk, stride_rh = stride_rh,
        stride_ob = stride_ob, stride_oi = stride_oi, stride_oj = stride_oj, stride_oh = stride_oh,
        BLOCK_I = BLOCK_I, BLOCK_J = BLOCK_J, BLOCK_K = BLOCK_K, BLOCK_D = BLOCK_D,
    )

    # --------------------------------------------------------------
    # 6. Output LayerNorm + gate + final linear
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    to_out_w = weights["to_out.weight"].to(device)          # [C, H]

    # LayerNorm on hidden dim H
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    eps = 1e-5
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # Apply the output gate (same shape as x)
    out_norm = out_norm * out_gate

    # Final projection back to C
    output = F.linear(out_norm, to_out_w)   # [B,N,N,C]

    return output
