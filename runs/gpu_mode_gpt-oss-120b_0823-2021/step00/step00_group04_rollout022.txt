"""
TriMul “outgoing” implementation with a fused Triton kernel for the
core N‑by‑N tensor contraction:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

All other linear / layer‑norm / gating operations are performed with
PyTorch (they are cheap compared with the O(N³) contraction).  The
kernel works on fp16 data for higher throughput and writes the result
in fp16; the surrounding code casts back to fp32 as required by the
module API.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel : fused block‑wise matrix‑multiply over the “k” dimension
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,          # const float16 *
    right_ptr,         # const float16 *
    out_ptr,           # float16 *
    # strides (in elements, not bytes)
    stride_bl, stride_il, stride_kl, stride_dl,   # left  [b,i,k,d]
    stride_br, stride_jr, stride_kr, stride_dr,   # right [b,j,k,d] (k is second axis)
    stride_bo, stride_io, stride_jo, stride_do,   # out   [b,i,j,d]
    # sizes
    B, N, H,
    # block sizes (compile‑time)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute a block of the output tensor.

    Each program instance is identified by (batch_id, pid_m, pid_n) where
    pid_m iterates over i‑blocks and pid_n over j‑blocks.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for partial tiles (when N is not a multiple of BLOCK_*)
    mask_i = i_offset < N
    mask_j = j_offset < N

    # ------------------------------------------------------------------
    # Pointers to the beginning of the current tile for left/right
    # ------------------------------------------------------------------
    # left:  (b, i, k, d)
    left_tile_ptr = left_ptr + batch_id * stride_bl \
                     + i_offset[:, None] * stride_il \
                     + tl.arange(0, BLOCK_K)[None, :] * stride_kl \
                     + 0 * stride_dl          # d‑dimension handled later by broadcasting

    # right: (b, j, k, d)  -- we need the k‑axis aligned with left,
    # therefore we load it as (k, j) and later treat it as transposed.
    right_tile_ptr = right_ptr + batch_id * stride_br \
                     + j_offset[None, :] * stride_jr \
                     + tl.arange(0, BLOCK_K)[:, None] * stride_kr \
                     + 0 * stride_dr

    # ------------------------------------------------------------------
    # Accumulator for the hidden dimension (d) -> we compute one d at a time
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, H), dtype=tl.float32)

    # Loop over K in tiles
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k_start, BLOCK_K)

        # Load left tile  (M, K, H)
        left = tl.load(
            left_tile_ptr + k_start * stride_kl,
            mask=(mask_i[:, None] & (tl.arange(0, cur_k)[None, :] < cur_k))[..., None],
            other=0.0,
        ).to(tl.float32)   # (M, K, H)

        # Load right tile (K, N, H) – note the transposed layout
        right = tl.load(
            right_tile_ptr + k_start * stride_kr,
            mask=((tl.arange(0, cur_k)[:, None] < cur_k) & mask_j[None, :])[..., None],
            other=0.0,
        ).to(tl.float32)   # (K, N, H)

        # ------------------- inner product over K -----------------------
        # left:  (M, K, H)   right: (K, N, H)
        # broadcasting over H -> compute per‑channel dot product
        for dh in range(0, H, 1):
            # slice a single channel to keep register pressure low
            lch = left[..., dh]    # (M, K)
            rch = right[..., dh]   # (K, N)
            acc[..., dh] += tl.dot(lch, rch)   # (M, N)
        # ---------------------------------------------------------------

    # ------------------------------------------------------------------
    # Write back – cast to fp16 as required by the surrounding code
    # ------------------------------------------------------------------
    out = acc.to(tl.float16)

    out_ptrs = out_ptr + batch_id * stride_bo \
               + i_offset[:, None] * stride_io \
               + j_offset[None, :] * stride_jo \
               + tl.arange(0, H)[None, None, :] * stride_do

    tl.store(out_ptrs, out, mask=mask_i[:, None, None] & mask_j[None, :, None])


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operation.
    Parts of the computation (layernorm, linear projections, gating) are
    performed with PyTorch; the O(N³) tensor contraction is executed by a
    custom Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Helper: LayerNorm without torch.nn (to use supplied weight/bias)
    # ------------------------------------------------------------------
    def layer_norm(x, weight, bias):
        mean = x.mean(-1, keepdim=True)
        var = ((x - mean) ** 2).mean(-1, keepdim=True)
        x_norm = (x - mean) * torch.rsqrt(var + eps)
        return x_norm * weight + bias

    # ------------------------------------------------------------------
    # Linear + gating
    # ------------------------------------------------------------------
    # 1) Input normalization
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )   # shape [B,N,N,dim]

    # 2) Projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # 3) Mask handling (mask shape [B,N,N])
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # 4) Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel for the contraction
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # strides (in elements)
    stride_bl, stride_il, stride_kl, stride_dl = left_fp16.stride()
    stride_br, stride_jr, stride_kr, stride_dr = right_fp16.stride()
    stride_bo, stride_io, stride_jo, stride_do = out_fp16.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        stride_bl, stride_il, stride_kl, stride_dl,
        stride_br, stride_jr, stride_kr, stride_dr,
        stride_bo, stride_io, stride_jo, stride_do,
        B, N, hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑processing (still in fp32 as required by the module)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # to_out_norm
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out
