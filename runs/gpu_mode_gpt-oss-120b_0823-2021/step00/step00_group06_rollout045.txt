"""
TriMul “outgoing” forward pass.

The heavy O(N³·D) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a custom Triton kernel that tiles the (i,j) pair
and accumulates over k.  All remaining linear / layer‑norm / gate
operations are carried out with native PyTorch (they are cheap compared
to the N³ contraction).  The kernel works for any batch size (1‑2),
sequence length up to 1024 and hidden dimension that is a multiple of
32 (the default block size).  The final result is returned in float32
as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: computes the N³ contraction for one batch element.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,                     # *const uint8   (B,N,N,D) left tensor
    right_ptr,                    # *const uint8   (B,N,N,D) right tensor
    out_ptr,                      # *mut   uint8   (B,N,N,D) output
    B, N, D,                      # static sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,        # tile size for i dimension
    BLOCK_N: tl.constexpr,        # tile size for j dimension
    BLOCK_K: tl.constexpr,        # tile size for k loop
    BLOCK_D: tl.constexpr,        # tile size for hidden dimension
):
    """
    Each program instance computes a (BLOCK_M, BLOCK_N, BLOCK_D) tile of the
    output for a fixed batch index.  The contraction over k is performed in a
    blocked loop while keeping the hidden dimension inside the register file.
    """
    pid = tl.program_id(0)                     # linear id over (B,N,N)
    total_tiles = B * tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N)

    # decode pid → (b, i0, j0)
    b = pid // (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    rem = pid % (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    i_tile = rem // tl.cdiv(N, BLOCK_N)
    j_tile = rem % tl.cdiv(N, BLOCK_N)

    i_offset = i_tile * BLOCK_M
    j_offset = j_tile * BLOCK_N

    # offsets inside the tile
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)
    offs_d = tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # pointer to the beginning of the tile (batch‑wise)
    left_tile_ptr  = left_ptr  + b * stride_left_b
    right_tile_ptr = right_ptr + b * stride_right_b
    out_tile_ptr   = out_ptr   + b * stride_out_b

    # accumulate in fp32, later cast to fp16 for storage
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over k
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,i,k,d]  → shape (M,K,D)
        left_ptrs = (
            left_tile_ptr
            + offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_d[None, None, :] * stride_left_d
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

        # right[b,j,k,d] → shape (N,K,D)
        right_ptrs = (
            right_tile_ptr
            + offs_j[:, None, None] * stride_right_j
            + offs_k[None, :, None] * stride_right_k
            + offs_d[None, None, :] * stride_right_d
        )
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

        # outer product over the k‑dimension for each hidden channel d
        # left : (M,K,D) , right : (N,K,D)
        # we sum over K → (M,N,D)
        acc += tl.sum(left * right, axis=1)   # sum over the K axis

    # store result in fp16 (the model later casts to fp32)
    out_fp16 = acc.to(tl.float16)

    out_ptrs = (
        out_tile_ptr
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_d[None, None, :] * stride_out_d
    )
    tl.store(out_ptrs, out_fp16,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])


# ----------------------------------------------------------------------
# Host side helper that launches the kernel
# ----------------------------------------------------------------------
def launch_trimul_kernel(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : (B,N,N,D) float32
    returns out  : (B,N,N,D) float32
    """
    B, N, _, D = left.shape
    out = torch.empty_like(left)

    # strides in element units (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # block sizes – chosen to fit into shared memory on Hopper
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64
    BLOCK_D = 32   # D is always a multiple of 32 in the provided configs

    grid = (B * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),)

    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, D,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_j, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
    )
    return out


# ----------------------------------------------------------------------
# Full TriMul forward (outgoing version)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Args
    ----
    data : (input, mask, weights, config)
        input   – (B, N, N, dim)   float32 on CUDA
        mask    – (B, N, N)        float32/bool on CUDA
        weights – dict of model parameters (already on CUDA)
        config  – {'dim': int, 'hidden_dim': int}
    Returns
    -------
    Tensor of shape (B, N, N, dim)  (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    # LayerNorm = (x - mean)/sqrt(var+eps) * weight + bias
    eps = 1e-5
    x = input_tensor
    # compute mean / var over the channel dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x = (x - mean) * torch.rsqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # ------------------------------------------------------------------
    # 2. Project to hidden dimension (no bias)
    # ------------------------------------------------------------------
    # weight shape : (hidden, dim) → we need (dim, hidden) for linear
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. N³ contraction – Triton
    # ------------------------------------------------------------------
    out = launch_trimul_kernel(left, right)          # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    # LayerNorm on hidden dimension
    mean_h = out.mean(dim=-1, keepdim=True)
    var_h = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_h) * torch.rsqrt(var_h + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])    # back to dim
    return out
