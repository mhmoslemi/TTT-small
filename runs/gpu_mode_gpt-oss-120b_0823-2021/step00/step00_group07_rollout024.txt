"""
TriMul “outgoing” implementation.

*   LayerNorm + linear projections + element‑wise gating are done in PyTorch
    (they are cheap compared to the O(N³) part).
*   The heavy tensor contraction

        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    is executed by a fused Triton kernel that performs a batched GEMM
    for every hidden channel `d`.  The kernel tiles the output matrix
    (i‑j plane) with BLOCK_M × BLOCK_N tiles and reduces over the
    reduction dimension `k` in BLOCK_K steps.

The final output is brought back to float32, normalized and linearly
projected to the original model dimension.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel for the core contraction `einsum('... i k d, ... j k d -> ... i j d')`
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          #   [B, N, N, H]   left  (i,k) plane
    right_ptr,         #   [B, N, N, H]   right (j,k) plane
    out_ptr,           #   [B, N, N, H]   output
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # tile row   (i)
    pid_j = tl.program_id(1)          # tile col   (j)
    pid_z = tl.program_id(2)          # combined batch & hidden dim

    # decode batch and hidden index
    b = pid_z // H
    h = pid_z %  H

    # start offsets for this tile
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds
    i_mask = i_offset < N
    j_mask = j_offset < N

    # pointer arithmetic helpers
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # allocate accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # offsets for the current k‑tile
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # load left[i, k]  shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base \
            + i_offset[:, None] * stride_left_i \
            + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # load right[j, k] shape (BLOCK_N, BLOCK_K) -> we need transpose to (BLOCK_K, BLOCK_N)
        right_ptrs = right_base \
            + j_offset[:, None] * stride_right_j \
            + k_offset[None, :] * stride_right_k
        right = tl.load(right_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)
        # right is (BLOCK_N, BLOCK_K); transpose for dot
        right_t = tl.trans(right)

        # accumulate dot product over k
        acc += tl.dot(left, right_t)   # (BLOCK_M, BLOCK_N)

    # write back
    out_ptrs = out_base \
        + i_offset[:, None] * stride_out_i \
        + j_offset[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Args
    ----
    data : (input, mask, weights, config)
        - input   : torch.Tensor[B, N, N, dim]   (float32)
        - mask    : torch.Tensor[B, N, N]        (float32/bool)
        - weights : dict of parameter tensors (see code)
        - config  : {'dim': int, 'hidden_dim': int}

    Returns
    -------
    torch.Tensor[B, N, N, dim]   (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # helpers to load linear weights (no bias)
    # ------------------------------------------------------------------
    def linear(x, w):
        # x: [..., dim] , w: [out_dim, dim]
        return torch.nn.functional.linear(x, w)

    # ------------------------------------------------------------------
    # 1. LayerNorm over `dim`
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_w + norm_b   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Projections + gating (pure PyTorch, cheap)
    # ------------------------------------------------------------------
    left = linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = linear(x, weights["right_proj.weight"])

    # apply mask if present
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    left_gate  = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core contraction with Triton
    # ------------------------------------------------------------------
    # cast to float16 for compute, keep a copy for later fp32 conversion
    left_half  = left.to(torch.float16)
    right_half = right.to(torch.float16)
    out_half = torch.empty_like(left_half)   # same shape [B,N,N,hidden]

    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_half.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right_half.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_half.stride()

    # tile sizes – tuned for Hopper (SM90) shared‑memory ~64KB
    BLOCK_M = 64   # rows of i
    BLOCK_N = 64   # cols of j
    BLOCK_K = 32   # reduction tile

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden,
    )

    trimul_outgoing_kernel[grid](
        left_half,
        right_half,
        out_half,
        B, N, hidden,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # enough parallelism
    )

    out = out_half.to(torch.float32)   # back to fp32

    # ------------------------------------------------------------------
    # 4. Final LayerNorm + gating + projection
    # ------------------------------------------------------------------
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]
    out = out * out_gate
    out = linear(out, weights["to_out.weight"])
    return out
```
