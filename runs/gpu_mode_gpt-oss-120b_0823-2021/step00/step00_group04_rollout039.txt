"""
TriMul “outgoing” implementation.

The heavy part is the pair‑wise multiplication summed over the third sequence
dimension:

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

For every batch element *b* and hidden channel *d* this is exactly a matrix
multiply

    out_d = left_d @ right_d.T           (left_d, right_d : N×N)

The kernel below performs a batched FP16 GEMM for all *d* in parallel.
All LayerNorm / Linear / gating work is done in PyTorch (they are cheap
compared to the N³ reduction).  The final result is cast back to float‑32
so the surrounding model sees the same datatype as the reference
implementation.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: batched GEMM over the hidden dimension (D = hidden_dim)
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # strides
    stride_bl, stride_il, stride_kl, stride_dl,   # left  [B,N,N,D]
    stride_br, stride_jr, stride_kr, stride_dr,   # right [B,N,N,D]
    stride_bo, stride_io, stride_jo, stride_do,   # out   [B,N,N,D]
    # sizes
    B, N, D,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr
):
    """
    Compute out[d] = left[d] @ right[d].T  for every batch element.
    Tile sizes are chosen to fill the Hopper SM.
    """
    pid = tl.program_id(axis=0)        # = batch * D  (flattened)
    pid_m = tl.program_id(axis=1)      # tile‑row over N
    pid_n = tl.program_id(axis=2)      # tile‑col over N

    # -----------------------------------------------------------------
    # decode batch and hidden dimension from pid
    # -----------------------------------------------------------------
    b = pid // D
    d = pid % D

    # block offsets in the output matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds (N may not be multiple of tile)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the start of the current tile
    left_tile_ptr  = left_ptr  + b*stride_bl + d*stride_dl \
                     + offs_m[:, None] * stride_il + tl.arange(0, BLOCK_K)[None, :] * stride_kl
    # right is transposed -> we read (j,k) as (k,j)
    right_tile_ptr = right_ptr + b*stride_br + d*stride_dr \
                     + tl.arange(0, BLOCK_K)[:, None] * stride_kr + offs_n[None, :] * stride_jr

    # accumulator in FP32 (more stable for the final cast)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K dimension
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load a tile of left (M×K) and right (K×N)
        left_tile  = tl.load(left_tile_ptr,
                             mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < N),
                             other=0.0).to(tl.float16)
        right_tile = tl.load(right_tile_ptr,
                             mask=(k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < N) & mask_n[None, :],
                             other=0.0).to(tl.float16)

        # dot‑product over K (inner dimension)
        acc += tl.dot(left_tile, right_tile).to(tl.float32)

        # advance pointers to next K‑block
        left_tile_ptr  += BLOCK_K * stride_kl
        right_tile_ptr += BLOCK_K * stride_kr

    # -----------------------------------------------------------------
    # Store the result (cast back to FP16 inside the kernel then to FP32
    # at the Python level)
    # -----------------------------------------------------------------
    out_tile_ptr = out_ptr + b*stride_bo + d*stride_do \
                   + offs_m[:, None] * stride_io + offs_n[None, :] * stride_jo
    tl.store(out_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

# -------------------------------------------------------------------------
# Helper: LayerNorm without bias (bias handled separately)
# -------------------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    norm = (x - mean) * torch.rsqrt(var + eps)
    return norm * weight + bias

# -------------------------------------------------------------------------
# Entry point called by the grader
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns: Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1) LayerNorm on the input (dim)
    # --------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2) Linear projections (no bias) – we keep them in FP16 for speed
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left  = torch.nn.functional.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 3) Apply mask (if present)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)      # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4) Gates (sigmoid) – still in FP16, then back‑to‑FP16 multiplication
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(
        torch.nn.functional.linear(x_fp16, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_fp16, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate   = torch.sigmoid(
        torch.nn.functional.linear(x_fp16, weights["out_gate.weight"].to(torch.float16))
    )

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Triton GEMM: out = left @ right.T  (per batch & per hidden channel)
    # --------------------------------------------------------------
    # Allocate output (FP16, will be cast to FP32 later)
    out = torch.empty((B, N, N, hidden_dim), device=device, dtype=torch.float16)

    # Strides (in elements)
    stride_bl, stride_il, stride_kl, stride_dl = left.stride()
    stride_br, stride_jr, stride_kr, stride_dr = right.stride()
    stride_bo, stride_io, stride_jo, stride_do = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32          # works well on H200
    GROUP_SIZE_M = 8      # number of tiles processed per program‑id (optional)

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        stride_bl=stride_bl,
        stride_il=stride_il,
        stride_kl=stride_kl,
        stride_dl=stride_dl,
        stride_br=stride_br,
        stride_jr=stride_jr,
        stride_kr=stride_kr,
        stride_dr=stride_dr,
        stride_bo=stride_bo,
        stride_io=stride_io,
        stride_jo=stride_jo,
        stride_do=stride_do,
        B=B,
        N=N,
        D=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
        num_warps=4,
        num_stages=2,
    )

    # Cast back to FP32 for the remaining layers
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # 6) to_out_norm + out_gate + final linear
    # --------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate

    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float32)
    )
    return out
