"""
TriMul “outgoing” kernel (AlphaFold3).

The implementation follows the reference PyTorch code but moves the
computationally‑heavy N³ einsum into a custom Triton kernel.
The flow is:

1. Layer‑norm on the input (torch, because it is cheap and needs mean/var).
2. Linear projections and gating (torch, bias‑free mat‑muls).
3. Apply the mask and gates (torch).
4. Compute   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batch of N×N Gram‑matrix products for each hidden dimension.
   This is performed by a fused Triton GEMM:
        out_{(b,d)} = left_{(b,d)} @ right_{(b,d)}.T
5. Final Layer‑norm, out‑gate and output linear (torch).  The final tensor
   is returned in float32.

Only the N³ part is executed in Triton; everything else stays in
torch‑autograd‑free code (no gradients are required).
"""

import torch
import triton
import triton.language as tl
from torch import nn

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for each (batch, hidden_dim) pair
# ----------------------------------------------------------------------
@triton.jit
def triton_gram_kernel(
    a_ptr,                     # left  [B*HD, N, N] (row‑major)
    b_ptr,                     # right [B*HD, N, N] (row‑major)
    c_ptr,                     # out   [B*HD, N, N] (row‑major)
    N,                         # sequence length
    stride_ah, stride_ak,      # strides for a (batch, row, col)
    stride_bh, stride_bk,      # strides for b
    stride_ch, stride_ck,      # strides for c
    BLOCK_M: tl.constexpr,     # tile rows of A / C
    BLOCK_N: tl.constexpr,     # tile cols of B / C
    BLOCK_K: tl.constexpr,     # tile reduction dim
):
    # program ids
    pid_h = tl.program_id(0)          # index over (batch * hidden_dim)
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # offsets within the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute linear offsets
    a_ptrs = a_ptr + pid_h * stride_ah + offs_m[:, None] * stride_ak + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_h * stride_bh + offs_n[None, :] * stride_bk + offs_k[:, None] * stride_bk

    # mask out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # main reduction loop
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load tiles (convert to fp16 for bandwidth, compute in fp32)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)
        # dot‑product accumulate
        acc += tl.dot(a, b)  # (M,K)·(K,N) -> (M,N)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back result in fp32 then cast to fp16 for storage (we keep fp32 later)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_h * stride_ch + offs_m[:, None] * stride_ck + offs_n[None, :] * stride_ck
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper: launch the Triton kernel for a given tensor pair
# ----------------------------------------------------------------------
def gram_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, HD]  (float16 or float32)
    returns      : [B, N, N, HD]  (float32)
    """
    B, N, _, HD = left.shape
    # collapse batch and hidden dim for a single 3‑D GEMM batch
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * HD, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * HD, N, N)

    # output buffer
    out = torch.empty_like(left_, dtype=torch.float16)

    # tile sizes – tuned for Hopper GPUs (shared memory ~ 64KB per SM)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * HD,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # strides (row‑major)
    stride_ah = left_.stride(0)
    stride_ak = left_.stride(1)
    stride_bh = right_.stride(0)
    stride_bk = right_.stride(1)
    stride_ch = out.stride(0)
    stride_ck = out.stride(1)

    triton_gram_kernel[grid](
        left_.data_ptr(),
        right_.data_ptr(),
        out.data_ptr(),
        N,
        stride_ah, stride_ak,
        stride_bh, stride_bk,
        stride_ch, stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # reshape back to original layout and cast to fp32
    out = out.view(B, HD, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)
    return out

# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a fused Triton
    kernel for the N³ inner product.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask        : torch.Tensor [B, N, N] (bool/float)
            weights     : dict of torch.Tensor (all float32)
            config      : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (no bias in the reference code except
    #    bias for norm & to_out_norm – they are provided in weights)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # manual layer‑norm to avoid extra module overhead
    # x shape: [B,N,N,dim]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    eps = 1e-5
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias   # broadcast over B,N,N

    # --------------------------------------------------------------
    # 2. Linear projections (bias‑free) – use torch.einsum for speed
    # --------------------------------------------------------------
    # left_proj: (dim, hidden_dim)
    left_proj_w = weights["left_proj.weight"]        # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"]      # [hidden_dim, dim]

    # shape after projection: [B,N,N,hidden_dim]
    left = torch.einsum('bnid,hd->bnih', x_norm, left_proj_w)   # d->h
    right = torch.einsum('bnid,hd->bnih', x_norm, right_proj_w)

    # --------------------------------------------------------------
    # 3. Mask (if present) – mask already broadcasted to 4‑D
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]   # [hidden_dim, dim]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(torch.einsum('bnid,hd->bnih', x_norm, left_gate_w))
    right_gate = torch.sigmoid(torch.einsum('bnid,hd->bnih', x_norm, right_gate_w))
    out_gate   = torch.sigmoid(torch.einsum('bnid,hd->bnih', x_norm, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. N³ inner product via Triton (produces [B,N,N,hidden_dim])
    # --------------------------------------------------------------
    # Cast to fp16 for the kernel (H200 has good fp16 throughput)
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out = gram_triton(left_fp16, right_fp16)   # fp32 result

    # --------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate and final linear
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]          # [dim, hidden_dim]

    # LayerNorm over hidden_dim
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    out_norm = out_norm * out_gate

    # final projection back to dim
    output = torch.einsum('bnih,dh->bnid', out_norm, to_out_w)

    return output.to(torch.float32)
