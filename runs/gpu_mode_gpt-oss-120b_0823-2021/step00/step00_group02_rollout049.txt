"""
TriMul "outgoing" forward pass (AlphaFold3) with a fused Triton kernel.

The expensive part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is exactly a batched matrix multiplication for every
( batch , channel ) pair:

    out_{b,d} =   left_{b,d}   @   right_{b,d}.T          (N×N) @ (N×N)ᵀ

The kernel below computes this product in FP16 and accumulates
in FP32, then the result is cast back to FP32 for the remaining
LayerNorm / linear layers that stay in PyTorch.  All other
operations (LayerNorm, Linear projections, gating) are cheap
and are performed with native PyTorch / torch.nn.functional
calls.

The kernel works for any batch size B∈{1,2}, sequence length N
(up to 1024 in the test suite) and hidden dimension H (the
output channel count of the projections).  The launch grid is

    (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B*H)

so each (block_row, block_col, batch*channel) computes a tile of
the output matrix.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel: batched matmul L @ Rᵀ  (L,R : [B,N,N,H] in FP16)
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # *fp16  (B,N,N,H)
    right_ptr,         # *fp16  (B,N,N,H)
    out_ptr,           # *fp32  (B,N,N,H)
    B, N, H,           # int32 scalars
    stride_lbn, stride_lbk, stride_lbh,   # strides for left  (B,N,N,H)
    stride_rbn, stride_rbk, stride_rbh,   # strides for right (B,N,N,H)
    stride_obn, stride_obk, stride_obh,   # strides for out   (B,N,N,H)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_z = tl.program_id(2)          # combined batch*head index

    batch_id = pid_z // H
    head_id  = pid_z %  H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # Pointers to the start of the tile for left/right
    left_tile_ptr  = left_ptr  + batch_id * stride_lbn \
                                   + head_id  * stride_lbh \
                                   + offs_m[:,None] * stride_lbk \
                                   + offs_k[None,:] * stride_lbn
    right_tile_ptr = right_ptr + batch_id * stride_rbn \
                                   + head_id  * stride_rbh \
                                   + offs_n[:,None] * stride_rbk \
                                   + offs_k[None,:] * stride_rbn   # note: same k‑dim

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load a tile of L and Rᵀ (R transposed in memory)
        l = tl.load(left_tile_ptr, mask=mask_m[:,None] & mask_k[None,:], other=0.0).to(tl.float32)
        r = tl.load(right_tile_ptr, mask=mask_n[:,None] & mask_k[None,:], other=0.0).to(tl.float32)

        # compute dot-product for this K‑tile
        acc += tl.dot(l, r)

        # advance pointers
        left_tile_ptr  += BLOCK_K * stride_lbn
        right_tile_ptr += BLOCK_K * stride_rbn

    # write result (FP32) back
    out_tile_ptr = out_ptr + batch_id * stride_obn \
                               + head_id  * stride_obh \
                               + offs_m[:,None] * stride_obbk() \
                               + offs_n[None,:] * stride_obbn()
    # Stride helpers (cannot be constexpr, so compute inline)
    # stride_obbk = stride_obk   # stride over K dimension (same as N)
    # stride_obbn = stride_obn   # stride over N dimension
    tl.store(out_tile_ptr, acc, mask=mask_m[:,None] & mask_n[None,:])

# --------------------------------------------------------------
# Helper to compute stride values for a contiguous tensor
# --------------------------------------------------------------
def _contig_strides(tensor: torch.Tensor) -> Tuple[int, int, int]:
    """Return (stride_batch, stride_row, stride_col) assuming layout
    (B, N, N, H).  Strides are in elements (not bytes)."""
    s = tensor.stride()
    # (B, N, N, H)
    stride_b = s[0]
    stride_n = s[1]
    stride_k = s[2]
    stride_h = s[3]
    return stride_b, stride_n, stride_h

# --------------------------------------------------------------
# Main entry point required by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    Parts that are cheap are done in PyTorch; the N³ multiplication
    is executed by a fused Triton kernel.

    Arguments
    ----------
    data : (input, mask, weights, config)
        input  – [B, N, N, dim]  (float32)
        mask   – [B, N, N]       (float32 or bool)
        weights – dict of pre‑trained parameters
        config – dict with keys "dim" and "hidden_dim"

    Returns
    -------
    out – [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm + Linear projections (PyTorch, FP32 -> FP16 for kernel)
    # ------------------------------------------------------------------
    # norm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # apply mask (mask may be all‑ones)
    if mask is not None:
        mask_ = mask.unsqueeze(-1)            # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # cast to fp16 for the kernel (still keep a copy in fp32 for later)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # Allocate output buffer (FP32) – shape [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Triton launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * hidden_dim,                   # batch*head
    )

    # stride helpers (in elements)
    stride_lbn, stride_lnk, stride_lhk = _contig_strides(left_fp16)   # (B,N,N,H)
    stride_rbn, stride_rnk, stride_rhk = _contig_strides(right_fp16)
    stride_obn, stride_onn, stride_ohk = _contig_strides(out)

    # launch kernel
    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        stride_lbn, stride_lnk, stride_lhk,
        stride_rbn, stride_rnk, stride_rhk,
        stride_obn, stride_onn, stride_ohk,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # post‑processing (LayerNorm + gating + final linear)
    # ------------------------------------------------------------------
    # out is [B,N,N,hidden_dim] (FP32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
