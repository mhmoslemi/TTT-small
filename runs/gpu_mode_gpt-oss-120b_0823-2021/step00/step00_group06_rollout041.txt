"""
TriMul “outgoing” implementation (forward only).

We perform the cheap LayerNorm / Linear / gating work in PyTorch and hand‑off the
core *pairwise* reduction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

to a custom Triton kernel.  The kernel tiles the (i,j) plane and reduces over the
sequence dimension `k` in blocks.  Hidden dimension `d` is processed in
chunks (default 32) that fit in registers; each chunk is reduced with a
`tl.dot` (vector‑dot) and accumulated in fp32 before being written back.
The final result is cast to float16 for the reduction and back to float32 at
the end, matching the reference implementation.
"""

import torch
import triton
import triton.language as tl
from torch import nn, einsum

# ----------------------------------------------------------------------
# Triton kernel: tri_mul_reduce
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_reduce(
    # pointers
    left_ptr,      #   [B, N, N, D]  (float16)
    right_ptr,     #   [B, N, N, D]  (float16)
    out_ptr,       #   [B, N, N, D]  (float32)
    # strides (in elements, not bytes)
    stride_lbz, stride_lby, stride_lbx, stride_lbd,   # left  B,N,N,D
    stride_rbz, stride_rby, stride_rbx, stride_rbd,   # right B,N,N,D
    stride_obz, stride_oby, stride_obx, stride_obd,   # out   B,N,N,D
    # sizes
    B, N, D,
    # compile‑time constants
    BLOCK_M: tl.constexpr,   # tile size for i dimension
    BLOCK_N: tl.constexpr,   # tile size for j dimension
    BLOCK_K: tl.constexpr,   # tile size for k reduction
    CHUNK_D: tl.constexpr,   # number of hidden channels processed together
):
    """Compute: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].

    The kernel works on a 2‑D tile of (i,j) and reduces over k in blocks.
    Hidden dimension d is processed in chunks of CHUNK_D (e.g. 32).
    """
    pid = tl.program_id(axis=0)               # linear index over batches × i‑tiles
    num_i_tiles = tl.cdiv(N, BLOCK_M)
    b = pid // num_i_tiles                     # batch index
    i_tile = pid % num_i_tiles                 # i‑tile index

    j_tile = tl.program_id(axis=1)            # j‑tile index

    # starting offsets for this tile
    i0 = i_tile * BLOCK_M
    j0 = j_tile * BLOCK_N

    # create range vectors
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_d = tl.arange(0, CHUNK_D)

    # masks for boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers offset to the start of the tile (for a given d‑chunk)
    # we will advance k inside the loop, so pointers are only offset in (b,i/j,d)
    left_tile_ptr  = left_ptr  + b * stride_lbz \
                                 + offs_i[:, None, None] * stride_lbx \
                                 + 0 * stride_lby \
                                 + offs_d[None, None, :] * stride_lbd
    right_tile_ptr = right_ptr + b * stride_rbz \
                                 + 0 * stride_rbx \
                                 + offs_j[None, :, None] * stride_rby \
                                 + offs_d[None, None, :] * stride_rbd

    # accumulator for the output tile (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N, CHUNK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over k
    # ------------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        cur_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # load left[b, i, k, d]   -> shape (M, K, chunk)
        left = tl.load(
            left_tile_ptr
            + cur_k[None, :, None] * stride_lby,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )   # fp16

        # load right[b, j, k, d]  -> shape (N, K, chunk)
        right = tl.load(
            right_tile_ptr
            + cur_k[None, :, None] * stride_rby,
            mask=mask_j[None, :] & mask_k[None, :],
            other=0.0
        )   # fp16

        # transpose right to (K, N, chunk) for dot product
        right = tl.trans(right, (1, 0, 2))

        # compute dot product over the reduction axis k for every (i,j) pair
        # using tl.dot on vectors of length BLOCK_K
        # we loop over the chunk dimension in a vectorized way
        for d_off in range(0, CHUNK_D):
            # slice the d‑th channel
            l_vec = left[:, :, d_off]   # (M, K)
            r_vec = right[:, :, d_off]  # (N, K)

            # compute outer product reduction: sum_k l_vec[i,k] * r_vec[j,k]
            # This can be expressed as a matrix multiply of (M,K) @ (K,N)^T
            # but we implement it manually with tl.dot because tl.dot only accepts
            # 1‑D vectors.  We therefore unroll over K in blocks using dot.
            # Here we simply rely on tl.dot for each i,j pair:
            for ii in range(0, BLOCK_M):
                for jj in range(0, BLOCK_N):
                    if mask_i[ii] and mask_j[jj]:
                        acc[ii, jj, d_off] += tl.dot(l_vec[ii, :], r_vec[jj, :], dtype=tl.float32)

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    out_tile_ptr = out_ptr + b * stride_obz \
                               + offs_i[:, None, None] * stride_obx \
                               + offs_j[None, :, None] * stride_oby \
                               + offs_d[None, None, :] * stride_obd

    tl.store(
        out_tile_ptr,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None]
    )


# ----------------------------------------------------------------------
# Helper: fused layer‑norm (float32) – we keep it in PyTorch for simplicity
# ----------------------------------------------------------------------
def fused_layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C] float32
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor   [B, N, N, C]   (float32)
            - mask         : torch.Tensor   [B, N, N]      (float32 or bool)
            - weights      : dict of torch.Tensor containing all model parameters
            - config       : dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (float32)
    # ------------------------------------------------------------------
    x = fused_layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2) Linear projections & gating (keep in float32, later cast to fp16)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask handling (optional)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # Cast to half for the reduction (saves bandwidth)
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Triton reduction: core tri‑mul
    # ------------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides (in elements)
    stride_lbz, stride_lby, stride_lbx, stride_lbd = left_h.stride()
    stride_rbz, stride_rby, stride_rbx, stride_rbd = right_h.stride()
    stride_obz, stride_oby, stride_obx, stride_obd = out.stride()

    # launch configuration
    BLOCK_M = 64          # i‑tile
    BLOCK_N = 64          # j‑tile
    BLOCK_K = 32          # k‑reduction tile
    CHUNK_D = 32          # process 32 channels at a time (must divide hidden_dim)

    assert hidden_dim % CHUNK_D == 0, "hidden_dim must be a multiple of CHUNK_D"

    grid = (
        triton.cdiv(B * N, BLOCK_M),   # program_id axis 0
        triton.cdiv(N, BLOCK_N),       # program_id axis 1
    )

    for d_start in range(0, hidden_dim, CHUNK_D):
        # slice the channel chunk
        left_chunk  = left_h[:, :, :, d_start:d_start + CHUNK_D]
        right_chunk = right_h[:, :, :, d_start:d_start + CHUNK_D]
        out_chunk   = out[:, :, :, d_start:d_start + CHUNK_D]

        tri_mul_reduce[grid](
            left_chunk, right_chunk, out_chunk,
            stride_lbz, stride_lby, stride_lbx, stride_lbd,
            stride_rbz, stride_rby, stride_rbx, stride_rbd,
            stride_obz, stride_oby, stride_obx, stride_obd,
            B, N, CHUNK_D,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, CHUNK_D=CHUNK_D,
        )

    # ------------------------------------------------------------------
    # 4) Output projection, final layer‑norm and gating (still in PyTorch)
    # ------------------------------------------------------------------
    out = fused_layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
