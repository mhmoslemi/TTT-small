"""
TriMul “outgoing” implementation.

The heavy N³ work is the einsum  
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]                (1)

We keep all surrounding ops (LayerNorm, Linear, gating) in PyTorch and
fuse only (1) into a Triton kernel.  The kernel treats the hidden dimension
`hidden` (denoted H) as a third tiled axis so a single launch computes a
tile (BLOCK_M×BLOCK_N×BLOCK_H) of the result for one batch element.
The reduction over the sequence dimension K is performed block‑wise with
`tl.dot`, which yields the same semantics as the PyTorch `einsum` used in
the reference implementation.

The final output is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: tiled batched mat‑mul per hidden channel
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,               # *[B, N, N, H]  left projected & gated
    right_ptr,              # *[B, N, N, H]  right projected & gated
    out_ptr,                # *[B, N, N, H]  result of (1)
    B, N, H,                # static sizes
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out  strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    """
    pid = tl.program_id(0)

    # ---- tile indices -------------------------------------------------
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    num_h = tl.cdiv(H, BLOCK_H)

    b = (pid // (num_m * num_n * num_h)) % B
    h_tile = (pid // (num_m * num_n)) % num_h
    m_tile = (pid // num_n) % num_m
    n_tile = pid % num_n

    # ---- offsets within the current tile -------------------------------
    offs_m = m_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = n_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_h = offs_h < H

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ---- reduction over K ---------------------------------------------
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, h]   -> shape (M, K, H)
        left = tl.load(
            left_ptr
            + b * stride_lb
            + offs_m[:, None, None] * stride_li
            + offs_k[None, :, None] * stride_lk
            + offs_h[None, None, :] * stride_lh,
            mask=mask_m[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # right[b, j, k, h]  we need it as (K, N, H) for tl.dot
        right_t = tl.load(
            right_ptr
            + b * stride_rb
            + offs_k[:, None, None] * stride_rk
            + offs_n[None, :, None] * stride_rj
            + offs_h[None, None, :] * stride_rh,
            mask=mask_k[:, None, None] & mask_n[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # dot over K for each hidden slice
        for h_idx in range(0, BLOCK_H):
            if mask_h[h_idx]:
                acc[:, :, h_idx] += tl.dot(
                    left[:, :, h_idx],          # (M, K)
                    right_t[:, :, h_idx],       # (K, N)
                )

    # ---- write result -------------------------------------------------
    out = out_ptr \
        + b * stride_ob \
        + offs_m[:, None, None] * stride_oi \
        + offs_n[None, :, None] * stride_oj \
        + offs_h[None, None, :] * stride_oh

    tl.store(
        out,
        acc,
        mask=mask_m[:, None, None] & mask_n[None, :, None] & mask_h[None, None, :],
    )


# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of TriMul (outgoing) with the core N³ multiplication
    off‑loaded to a Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------- 1) LayerNorm ------------------------------------------------
    # norm over last dim (dim)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------- 2) Linear projections ---------------------------------------
    # left_proj, right_proj : (dim -> hidden)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------- 3) Apply mask (if present) ---------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------- 4) Gating ----------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------- 5) Triton kernel for the N³ einsum ---------------------------
    # prepare strides (in elements, not bytes)
    left_strides = left.stride()
    right_strides = right.stride()
    out_shape = (B, N, N, hidden)
    out = torch.empty(out_shape, device=device, dtype=torch.float32)

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 32

    grid = (
        triton.cdiv(B * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N) * triton.cdiv(hidden, BLOCK_H), 1),
    )

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_lb=left_strides[0],
        stride_li=left_strides[1],
        stride_lk=left_strides[2],
        stride_lh=left_strides[3],
        stride_rb=right_strides[0],
        stride_rj=right_strides[1],
        stride_rk=right_strides[2],
        stride_rh=right_strides[3],
        stride_ob=out.stride(0),
        stride_oi=out.stride(1),
        stride_oj=out.stride(2),
        stride_oh=out.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------- 6) Post‑processing (LayerNorm, gating, final linear) -------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
