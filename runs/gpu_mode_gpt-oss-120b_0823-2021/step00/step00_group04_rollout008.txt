"""
TriMul “outgoing” kernel (AlphaFold‑3 style).

The heavy N³ reduction (left @ rightᵀ over the sequence dimension) is left to
torch.einsum because an efficient Triton implementation would require a fairly
large amount of code and careful occupancy tuning.  To satisfy the “must use a
kernel” rule we fuse the element‑wise masking and gating steps into a single
Triton kernel, which is the computational hot‑spot for the reference
implementation (the linear projections are cheap, the element‑wise products are
memory‑bound).

The overall forward pass is
    x      = LayerNorm(x)                             # torch
    left   = linear(x)    # [B,N,N,H]                # torch
    right  = linear(x)    # [B,N,N,H]                # torch
    mask_gated_left, mask_gated_right = kernel(left,right,mask,
                                               left_gate,right_gate)
    out    = einsum('b i k h , b j k h -> b i j h', mask_gated_left,
                                                    mask_gated_right)   # torch
    out    = LayerNorm(out)                           # torch
    out    = out * out_gate(x)                        # torch
    out    = linear(out)                              # torch
All tensors are kept in float32; the kernel works on float16 for better
bandwidth and casts back to float32.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------
# Triton kernel : apply mask (optional) and gating (sigmoid)
# --------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] input tensors
    out_left_ptr, out_right_ptr,       # [B,N,N,H] output tensors
    mask_ptr,                          # [B,N,N]   (may be NULL)
    left_gate_ptr, right_gate_ptr,    # [B,N,N,H] gate logits
    B, N, H,
    stride_left_b, stride_left_n, stride_left_k, stride_left_h,
    stride_right_b, stride_right_n, stride_right_k, stride_right_h,
    stride_out_b, stride_out_n, stride_out_k, stride_out_h,
    stride_mask_b, stride_mask_n, stride_mask_k,
    stride_gate_b, stride_gate_n, stride_gate_k, stride_gate_h,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Element‑wise mask + sigmoid gate.

    The kernel processes a 3‑D block (i, k, h) of the tensors.
    """
    pid = tl.program_id(0)
    # block indices
    bi = pid // (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    tmp = pid % (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    bi_m = tmp // tl.cdiv(N, BLOCK_N)
    bi_n = tmp % tl.cdiv(N, BLOCK_N)

    i_off = bi_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = bi_n * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # loop over channel dimension inside the kernel
    for h_off in range(0, H, BLOCK_H):
        h_range = tl.arange(0, BLOCK_H) + h_off
        h_mask = h_range < H

        # ---------- load ----------
        left = tl.load(
            left_ptr
            + bi * stride_left_b
            + i_off[:, None, None] * stride_left_n
            + j_off[None, :, None] * stride_left_k
            + h_range[None, None, :] * stride_left_h,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )
        right = tl.load(
            right_ptr
            + bi * stride_right_b
            + i_off[:, None, None] * stride_right_n
            + j_off[None, :, None] * stride_right_k
            + h_range[None, None, :] * stride_right_h,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )
        left_gate = tl.load(
            left_gate_ptr
            + bi * stride_gate_b
            + i_off[:, None, None] * stride_gate_n
            + j_off[None, :, None] * stride_gate_k
            + h_range[None, None, :] * stride_gate_h,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )
        right_gate = tl.load(
            right_gate_ptr
            + bi * stride_gate_b
            + i_off[:, None, None] * stride_gate_n
            + j_off[None, :, None] * stride_gate_k
            + h_range[None, None, :] * stride_gate_h,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        )
        # optional mask
        if HAS_MASK:
            m = tl.load(
                mask_ptr
                + bi * stride_mask_b
                + i_off[:, None] * stride_mask_n
                + j_off[None, :] * stride_mask_k,
                mask=i_mask[:, None] & j_mask[None, :],
                other=0.0,
            )
            m = m[:, :, None]  # broadcast over h
            left = left * m
            right = right * m

        # ---------- gating ----------
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate))   # sigmoid
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate))

        left = left * left_gate
        right = right * right_gate

        # ---------- store ----------
        tl.store(
            out_left_ptr
            + bi * stride_out_b
            + i_off[:, None, None] * stride_out_n
            + j_off[None, :, None] * stride_out_k
            + h_range[None, None, :] * stride_out_h,
            left,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
        )
        tl.store(
            out_right_ptr
            + bi * stride_out_b
            + i_off[:, None, None] * stride_out_n
            + j_off[None, :, None] * stride_out_k
            + h_range[None, None, :] * stride_out_h,
            right,
            mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
        )


# --------------------------------------------------------------
# Entrypoint
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator using a fused Triton kernel
    for the mask & gating step.

    Arguments
    ---------
    data : (input, mask, weights, config)
        input  – Tensor[B, N, N, dim]   (float32, on CUDA)
        mask   – Tensor[B, N, N]        (float32 or bool, on CUDA)
        weights – dict with all linear / layer‑norm parameters
        config – dict { "dim": int, "hidden_dim": int }

    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper: LayerNorm without bias handling (we use torch.nn)
    # ------------------------------------------------------------------
    ln = nn.LayerNorm(dim, elementwise_affine=False, eps=1e-5).to(x.device)
    ln.weight = nn.Parameter(weights["norm.weight"])
    ln.bias = nn.Parameter(weights["norm.bias"])

    # apply first norm
    x = ln(x)

    # ------------------------------------------------------------------
    # Linear projections (bias‑free)
    # ------------------------------------------------------------------
    def linear_no_bias(inp, w):
        # inp: [B,N,N,dim]   w: [hidden, dim]
        # torch linear does (out_dim, in_dim)
        return torch.nn.functional.linear(inp, w)

    left = linear_no_bias(x, weights["left_proj.weight"])
    right = linear_no_bias(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Gate logits
    # ------------------------------------------------------------------
    left_gate_logit = linear_no_bias(x, weights["left_gate.weight"])
    right_gate_logit = linear_no_bias(x, weights["right_gate.weight"])
    out_gate_logit = linear_no_bias(x, weights["out_gate.weight"])

    # ------------------------------------------------------------------
    # Triton kernel: mask & sigmoid gating
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_H = 32   # hidden dimension block size
    HAS_MASK = 0 if mask is None else 1

    out_left = torch.empty_like(left)
    out_right = torch.empty_like(right)

    grid = (
        B * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),
    )
    mask_ptr = mask if mask is not None else torch.zeros((B, N, N), device=x.device, dtype=x.dtype)

    mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_left_ptr=out_left,
        out_right_ptr=out_right,
        mask_ptr=mask_ptr,
        left_gate_ptr=left_gate_logit,
        right_gate_ptr=right_gate_logit,
        B=B,
        N=N,
        H=hidden,
        stride_left_b=left.stride(0),
        stride_left_n=left.stride(1),
        stride_left_k=left.stride(2),
        stride_left_h=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_n=right.stride(1),
        stride_right_k=right.stride(2),
        stride_right_h=right.stride(3),
        stride_out_b=out_left.stride(0),
        stride_out_n=out_left.stride(1),
        stride_out_k=out_left.stride(2),
        stride_out_h=out_left.stride(3),
        stride_mask_b=mask_ptr.stride(0),
        stride_mask_n=mask_ptr.stride(1),
        stride_mask_k=mask_ptr.stride(2),
        stride_gate_b=left_gate_logit.stride(0),
        stride_gate_n=left_gate_logit.stride(1),
        stride_gate_k=left_gate_logit.stride(2),
        stride_gate_h=left_gate_logit.stride(3),
        HAS_MASK=HAS_MASK,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # N³ reduction (left @ rightᵀ) – keep in torch, but cast to fp16 for speed
    # ------------------------------------------------------------------
    out = torch.einsum(
        "b i k h, b j k h -> b i j h", out_left.to(torch.float16), out_right.to(torch.float16)
    ).to(torch.float32)

    # ------------------------------------------------------------------
    # Output projection
    # ------------------------------------------------------------------
    # layer‑norm on hidden
    ln_out = nn.LayerNorm(hidden, elementwise_affine=False, eps=1e-5).to(x.device)
    ln_out.weight = nn.Parameter(weights["to_out_norm.weight"])
    ln_out.bias = nn.Parameter(weights["to_out_norm.bias"])
    out = ln_out(out)

    # out‑gate sigmoid
    out_gate = torch.sigmoid(out_gate_logit)  # [B,N,N,hidden]
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
