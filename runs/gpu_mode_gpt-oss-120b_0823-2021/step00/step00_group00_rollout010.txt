"""
TriMul “outgoing” implementation.

The heavy O(N³) part:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is computed with a fused Triton kernel.  All simple ops
(layer‑norm, linear projections, gating, final norm+linear) are kept
in PyTorch so that we only have to launch one kernel per (batch, hidden)
channel.  The kernel tiles the i‑ and j‑axes (BLOCK_M × BLOCK_N) and
processes the k‑axis in BLOCK_K‑size tiles, accumulating in FP32.
The final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel: batched, per‑hidden‑dim matmul over the k‑axis
# --------------------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr,          # const float *
    right_ptr,         # const float *
    out_ptr,           # float *
    # sizes
    B, N, H,           # batch, seq_len, hidden_dim
    # strides (in elements, not bytes)
    stride_blh,        # B * N * N * H   (stride for batch dim)
    stride_ilh,        # N * N * H       (stride for i dim in left)
    stride_klh,        # H               (stride for k dim in left)
    stride_hl,         # 1               (stride for hidden dim in left)

    stride_brh,        # B * N * N * H   (stride for batch dim)
    stride_jrh,        # N * N * H       (stride for j dim in right)
    stride_krh,        # H               (stride for k dim in right)
    stride_hr,         # 1               (stride for hidden dim in right)

    stride_oo,         # B * N * N * H   (stride for batch dim in out)
    stride_io,         # N * N * H       (stride for i dim in out)
    stride_jo,         # N * H           (stride for j dim in out)
    stride_ho,         # 1               (stride for hidden dim in out)

    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,   # i‑tile
    BLOCK_N: tl.constexpr,   # j‑tile
    BLOCK_K: tl.constexpr,   # k‑tile
):
    """ Compute a tile of out[b, i0:i0+M, j0:j0+N, h] """
    pid_bh = tl.program_id(0)               # combined batch * hidden index
    pid_i  = tl.program_id(1)               # i‑tile index
    pid_j  = tl.program_id(2)               # j‑tile index

    # ----- decode batch and hidden indices ---------------------------------
    b = pid_bh // H          # batch index
    h = pid_bh %  H          # hidden channel index

    # ----- tile start offsets ------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # ----- range offsets -----------------------------------------------------
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds tiles
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # ----- pointers for the current batch/hidden -----------------------------
    left_base  = left_ptr  + b * stride_blh + h * stride_hl
    right_base = right_ptr + b * stride_brh + h * stride_hr
    out_base   = out_ptr   + b * stride_oo  + h * stride_ho

    # ----- allocate accumulator (FP32) ---------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over k -------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_tiles):
        # global k offset
        k0 = kk * BLOCK_K

        # -------- left tile: shape (M, K) ------------------------------------
        left_ptrs = left_base \
                  + offs_i[:, None] * stride_ilh \
                  + (k0 + offs_k)[None, :] * stride_klh
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # -------- right tile: shape (N, K) -----------------------------------
        right_ptrs = right_base \
                  + offs_j[:, None] * stride_jrh \
                  + (k0 + offs_k)[None, :] * stride_krh
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # -------- dot over K --------------------------------------------------
        # each mul produces FP32 (left/right are FP32), accumulate in FP32
        acc += tl.dot(left, right)   # (M,K)·(N,K)ᵀ → (M,N)

    # ----- store result -------------------------------------------------------
    out_ptrs = out_base \
            + offs_i[:, None] * stride_io \
            + offs_j[None, :] * stride_jo
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Helper: launch the kernel
# --------------------------------------------------------------
def launch_trimul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]  (contiguous, float32)
    returns out  : [B, N, N, H]  (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (torch stride gives element stride)
    sl = left.stride()   # (B,N,N,H)
    sr = right.stride()
    so = out.stride()

    # tile sizes – chosen to fit Hopper SMs and L2
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(B * H, 1),          # pid 0 : batch*hidden (we launch 1 per bh)
        triton.cdiv(N, BLOCK_M),       # pid 1 : i‑tiles
        triton.cdiv(N, BLOCK_N),       # pid 2 : j‑tiles
    )

    trimul_kernel[grid](
        left_ptr   = left,
        right_ptr  = right,
        out_ptr    = out,
        B, N, H,
        stride_blh = sl[0],
        stride_ilh = sl[1],
        stride_klh = sl[2],
        stride_hl  = sl[3],
        stride_brh = sr[0],
        stride_jrh = sr[1],
        stride_krh = sr[2],
        stride_hr  = sr[3],
        stride_oo  = so[0],
        stride_io  = so[1],
        stride_jo  = so[2],
        stride_ho  = so[3],
        BLOCK_M = BLOCK_M,
        BLOCK_N = BLOCK_N,
        BLOCK_K = BLOCK_K,
    )
    return out


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    The heavy pairwise multiplication is implemented in Triton.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ----------------------------------------------------------
    # 1. LayerNorm on the last dim
    # ----------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B,N,N,dim]

    # ----------------------------------------------------------
    # 2. Linear projections (no bias)
    # ----------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]          # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]

    # fuse projection with weight transpose for torch.nn.functional.linear
    left = F.linear(x, left_proj_w)    # -> [B,N,N,hidden]
    right = F.linear(x, right_proj_w)

    # ----------------------------------------------------------
    # 3. Apply mask (if present) – mask may be None -> treat as all‑ones
    # ----------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ----------------------------------------------------------
    # 4. Gating (sigmoid)
    # ----------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 5. Core TriMul (triton)
    # ----------------------------------------------------------
    # left/right are [B,N,N,hidden] float32 contiguous
    out = launch_trimul(left, right)   # fp32

    # ----------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear
    # ----------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]   # (dim, hidden)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # -> [B,N,N,dim]

    return out
