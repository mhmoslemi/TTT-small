"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The heavy part of TriMul is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batch of matrix‑multiplications per hidden
dimension `d`.  The kernel below computes this operation exactly in
float‑32 while the surrounding LayerNorm / Linear / gating work is done
with PyTorch (float‑32 → float‑16 where beneficial).

The kernel works on:
* B  ∈ {1,2}
* N  up to 1024
* hidden_dim (hd) up to 384

A three‑dimensional grid (batch, i‑tile, j‑tile) is used; the hidden
dimension is tiled inside the kernel.  All arithmetic is accumulated in
float‑32 and the final output is written as float‑32.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: batched mat‑mul per hidden dimension
# -------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # strides (in elements, not bytes)
    stride_bl, stride_il, stride_kl, stride_dl,   # left  [B, N, K, D]
    stride_br, stride_jr, stride_kr, stride_dr,   # right [B, N, K, D]
    stride_bo, stride_io, stride_jo, stride_do,   # out   [B, N, N, D]
    # problem size
    B, N, K, D,
    # tile sizes (constexpr)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # tile over i (rows)
    pid_j = tl.program_id(2)          # tile over j (cols)

    # offsets for the i / j tiles
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑range indices
    mask_i = offs_i < N
    mask_j = offs_j < N

    # allocate accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # iterate over K tiles
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- load left block ----
        # shape: (BLOCK_M, BLOCK_K, BLOCK_D)
        left_ptrs = (
            left_ptr
            + pid_b * stride_bl
            + offs_i[:, None, None] * stride_il
            + offs_k[None, :, None] * stride_kl
            + tl.arange(0, BLOCK_D)[None, None, :] * stride_dl
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0)

        # ---- load right block (need transpose on K dimension) ----
        # shape: (BLOCK_N, BLOCK_K, BLOCK_D) → we will use it as (BLOCK_K, BLOCK_N, BLOCK_D)
        right_ptrs = (
            right_ptr
            + pid_b * stride_br
            + offs_j[:, None, None] * stride_jr
            + offs_k[None, :, None] * stride_kr
            + tl.arange(0, BLOCK_D)[None, None, :] * stride_dr
        )
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0)

        # right is (BLOCK_N, BLOCK_K, D); we need (BLOCK_K, BLOCK_N, D)
        right = tl.trans(right, (1, 0, 2))

        # ---- compute dot for each hidden dim slice ----
        # We iterate over the hidden‑dim tile manually to keep the dot operation simple
        for d in range(0, BLOCK_D):
            # left[:, :, d] : (M, K)    right[:, :, d] : (K, N)
            acc[:, :, d] += tl.dot(left[:, :, d], right[:, :, d])

    # ---- write result ----
    out_ptrs = (
        out_ptr
        + pid_b * stride_bo
        + offs_i[:, None, None] * stride_io
        + offs_j[None, :, None] * stride_jo
        + tl.arange(0, BLOCK_D)[None, None, :] * stride_do
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None])


# -------------------------------------------------------------
# Host‑side helper
# -------------------------------------------------------------
def _launch_trimul_kernel(left, right):
    """
    left, right : torch.Tensor of shape [B, N, N, D]   (float16/float32)
    returns out : torch.Tensor of shape [B, N, N, D]   (float32)
    """
    B, N, _, D = left.shape
    K = N                               # the summed dimension equals seq_len

    # Choose reasonable tile sizes for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 32                # hidden dim tile; D <= 384 → at most 12 tiles

    # pad hidden dim to a multiple of BLOCK_D for simpler indexing
    D_pad = triton.cdiv(D, BLOCK_D) * BLOCK_D

    # allocate output
    out = torch.empty((B, N, N, D), dtype=torch.float32, device=left.device)

    # launch grid: (batch, i‑tiles, j‑tiles)
    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # compute strides (in elements)
    stride_bl, stride_il, stride_kl, stride_dl = left.stride()
    stride_br, stride_jr, stride_kr, stride_dr = right.stride()
    stride_bo, stride_io, stride_jo, stride_do = out.stride()

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        stride_bl, stride_il, stride_kl, stride_dl,
        stride_br, stride_jr, stride_kr, stride_dr,
        stride_bo, stride_io, stride_jo, stride_do,
        B, N, K, D,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
    )
    return out

# Full custom kernel (PyTorch + Triton)
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]   torch.float32
            mask         : [B, N, N]        torch.float32 (0/1) or None
            weights      : dict of torch tensors (already on CUDA)
            config       : dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ---------- LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # torch.nn.functional.layer_norm works on the last dimension only,
    # we need to flatten the leading three dims.
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ---------- Linear projections (no bias) ----------
    # shape: [B,N,N,hidden_dim]
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- Mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gates (sigmoid) ----------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to half for the heavy einsum ----------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # ---------- Triton kernel for the einsum ----------
    out_h = _launch_trimul_kernel(left_h, right_h)          # float32 result

    # ---------- Final LayerNorm, gating and output linear ----------
    out = torch.nn.functional.layer_norm(
        out_h,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
