"""
TriMul “outgoing” implementation.

The heavy part is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For every hidden dimension `d` this is exactly a batched matrix‑multiply:
    L_d  = left[..., d]          # (B, N, N)
    R_d  = right[..., d]         # (B, N, N)
    out_d = L_d @ R_d.transpose(1, 2)   # (B, N, N)

We fuse this batched GEMM with Triton.  All other ops (LayerNorm, Linear,
gates, final LayerNorm + Linear) stay in PyTorch – they are cheap compared
to the O(B·N³·hidden) work of the einsum.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul (B*D, N, N)  where each “batch” is one
# (original batch, hidden_dim) pair.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,      # pointers
    M, N, K,                  # matrix sizes (M=N=N_seq, K=N_seq)
    stride_am, stride_ak,     # a strides: (batch*D, M, K)
    stride_bk, stride_bn,     # b strides: (batch*D, K, N)
    stride_cm, stride_cn,     # c strides: (batch*D, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)                       # combined (batch*D) index
    # block start offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)

    # mask for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the current batch
    a_batch_ptr = a_ptr + pid * stride_am
    b_batch_ptr = b_ptr + pid * stride_bk

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # loads
        a = tl.load(
            a_batch_ptr
            + (offs_m[:, None] * stride_am
               + k_offset[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_batch_ptr
            + (k_offset[:, None] * stride_bk
               + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot‑product update
        acc += tl.dot(a, b)

        # advance pointers (not strictly needed because we recompute each iter)
        # kept for clarity
        # a_batch_ptr += BLOCK_K * stride_ak
        # b_batch_ptr += BLOCK_K * stride_bk

    # write back
    c = acc.to(tl.float16)          # use fp16 store for bandwidth
    tl.store(
        c_ptr
        + (offs_m[:, None] * stride_cm
           + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: (B, N, N, D)   fp32
    returns out: (B, N, N, D)   fp32
    """
    B, N, _, D = left.shape
    # combine batch and hidden dim -> treat each (b,d) as a separate batch
    batch = B * D

    # reshape to (batch, N, N)
    left_ = left.permute(0, 3, 1, 2).reshape(batch, N, N).contiguous()
    # right needs to be transposed on the last two axes before matmul
    right_ = right.permute(0, 3, 2, 1).reshape(batch, N, N).contiguous()

    out = torch.empty_like(left_)

    # launch parameters (tuned for Hopper)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(batch, 1),)   # one program per combined batch
    batched_matmul_kernel[grid](
        left_.ptr,
        right_.ptr,
        out.ptr,
        N, N, N,                         # M, N, K (all equal to seq_len)
        left_.stride(0), left_.stride(2),  # stride_am, stride_ak
        right_.stride(2), right_.stride(1),# stride_bk, stride_bn
        out.stride(0), out.stride(2),      # stride_cm, stride_cn
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back
    out = out.view(B, D, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape (bs, seq_len, seq_len, dim) in float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm ----------
    # Manual LayerNorm to keep everything on the same device & dtype
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * norm_weight + norm_bias

    # ---------- Linear projections (no bias) ----------
    def linear_no_bias(x, w):
        # x: (..., dim), w: (hidden_dim, dim)
        return torch.nn.functional.linear(x, w)

    left = linear_no_bias(x, weights["left_proj.weight"])
    right = linear_no_bias(x, weights["right_proj.weight"])

    # ---------- Apply mask ----------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gates ----------
    left_gate = torch.sigmoid(linear_no_bias(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_no_bias(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear_no_bias(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Core TriMul (Triton) ----------
    # left/right : (B,N,N,hidden_dim)
    # we need shape (B,N,N,hidden_dim) -> same hidden dim as hidden_dim
    # our kernel expects hidden_dim as last dim
    out = triton_batched_matmul(left, right)          # (B,N,N,hidden_dim)

    # ---------- Post‑processing ----------
    # to_out_norm (LayerNorm over hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * to_out_norm_weight + to_out_norm_bias

    out = out * out_gate

    # final linear back to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
