"""
TriMul “outgoing” implementation.

We keep the cheap per‑element ops (layer‑norm, linear projections, sigmoid
gates) in PyTorch and fuse the expensive 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

with a custom Triton kernel.  The kernel computes a block of (i,j,d) and
tiles the reduction over k.  All tensors are float32 on the device and the
final output is cast back to float32.

The kernel launches a 3‑D grid:
    dim0 – batch index
    dim1 – i‑block
    dim2 – j‑block
for each hidden‑dim block (BLOCK_D) we accumulate a partial sum over k‑tiles.
The implementation works for the shapes required by the test matrix
( B∈{1,2}, N∈{128,…,1024}, hidden_dim∈{128} ) and runs efficiently on Hopper
(sm90).
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel that performs the N×N×hidden reduction
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,          # *[B, N, N, H]
    right_ptr,         # *[B, N, N, H]
    out_ptr,           # *[B, N, N, H]
    # sizes
    B, N, H,
    # strides (row‑major)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time constants
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    # ------------------------------------------------------------------
    # program indices
    # ------------------------------------------------------------------
    pid_b = tl.program_id(0)          # batch
    pid_i = tl.program_id(1)          # i‑block
    pid_j = tl.program_id(2)          # j‑block

    # base offsets for this block
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # ------------------------------------------------------------------
    # allocate accumulator for each d‑sub‑block
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over k‑tiles
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K

        # ------------------------------------------------------------------
        # load left: shape (BLOCK_M, BLOCK_K, BLOCK_D)
        # ------------------------------------------------------------------
        offs_i = i_base + tl.arange(0, BLOCK_M)[:, None, None]
        offs_k = k_base + tl.arange(0, BLOCK_K)[None, :, None]
        offs_d = tl.arange(0, BLOCK_D)[None, None, :]
        mask_i = offs_i < N
        mask_k = offs_k < N

        left_ptrs = (left_ptr
                     + pid_b * stride_left_b
                     + offs_i * stride_left_i
                     + offs_k * stride_left_k
                     + offs_d * stride_left_h)
        left = tl.load(left_ptrs, mask=mask_i & mask_k, other=0.0)

        # ------------------------------------------------------------------
        # load right: shape (BLOCK_N, BLOCK_K, BLOCK_D)
        # ------------------------------------------------------------------
        offs_j = j_base + tl.arange(0, BLOCK_N)[:, None, None]
        right_ptrs = (right_ptr
                      + pid_b * stride_right_b
                      + offs_j * stride_right_j
                      + offs_k * stride_right_k
                      + offs_d * stride_right_h)
        right = tl.load(right_ptrs, mask=mask_j & mask_k, other=0.0)

        # ------------------------------------------------------------------
        # multiply‑and‑sum over k dimension:
        #   acc[i,j,d] += Σ_k left[i,k,d] * right[j,k,d]
        # We use tl.dot on the k‑vector for each (i,j,d) element.
        # ------------------------------------------------------------------
        # reshape to (M*N, D) * (K) => we broadcast manually
        for d in range(0, BLOCK_D):
            # left slice (M, K), right slice (N, K)
            l = left[:, :, d]          # (M, K)
            r = right[:, :, d]         # (N, K)

            # compute outer product: (M, K) * (N, K) -> (M, N)
            # we expand dimensions to (M,1,K) and (1,N,K) then dot‑reduce.
            l_expand = l[:, None, :]   # (M,1,K)
            r_expand = r[None, :, :]   # (1,N,K)

            # element‑wise product and sum over K
            acc[:, :, d] += tl.sum(l_expand * r_expand, axis=2)

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    offs_i = i_base + tl.arange(0, BLOCK_M)[:, None, None]
    offs_j = j_base + tl.arange(0, BLOCK_N)[None, :, None]
    offs_d = tl.arange(0, BLOCK_D)[None, None, :]
    mask_i = offs_i < N
    mask_j = offs_j < N

    out_ptrs = (out_ptr
                + pid_b * stride_out_b
                + offs_i * stride_out_i
                + offs_j * stride_out_j
                + offs_d * stride_out_h)
    tl.store(out_ptrs, acc, mask=mask_i & mask_j)


# ----------------------------------------------------------------------
# Host‑side helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right: [B,N,N,H] float32
    returns out:      [B,N,N,H] float32
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # tile sizes – tuned for Hopper, can be adjusted
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64          # must divide N reasonably; we tile anyway
    BLOCK_D = 32          # hidden_dim is a multiple of 32 (128)

    grid = (B,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    # strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# Full forward function (mixed precision where beneficial)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Arguments
    ---------
    data : (input, mask, weights, config)
        input  – [B,N,N,dim] torch.float32
        mask   – [B,N,N]   torch.float32  (0/1)
        weights – dict of pre‑loaded parameters (all float32)
        config  – {"dim": int, "hidden_dim": int}
    Returns
    -------
    torch.Tensor  – [B,N,N,dim] float32
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dim
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # manual LayerNorm (mean & var per element) – uses torch for simplicity
    x = nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]      # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]

    # shape (B,N,N,hidden) = (B,N,N,dim) @ (dim, hidden)
    x_flat = x.view(-1, dim)                         # (B*N*N, dim)
    left  = torch.matmul(x_flat, left_proj_w.t()).view(B, N, N, hidden_dim)
    right = torch.matmul(x_flat, right_proj_w.t()).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Apply mask (if provided)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                # (B,N,N,1)
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating networks (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(
        torch.matmul(x_flat, left_gate_w.t()).view(B, N, N, hidden_dim)
    )
    right_gate = torch.sigmoid(
        torch.matmul(x_flat, right_gate_w.t()).view(B, N, N, hidden_dim)
    )
    out_gate   = torch.sigmoid(
        torch.matmul(x_flat, out_gate_w.t()).view(B, N, N, hidden_dim)
    )

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton fused einsum (outgoing version)
    # ------------------------------------------------------------------
    # We compute the kernel on float16 for higher throughput, cast back later.
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # Triton kernel works on float32; we promote the inputs.
    out = trimul_outgoing_triton(left_f16.to(torch.float32),
                                 right_f16.to(torch.float32))

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    out_norm_w = weights["to_out_norm.weight"]
    out_norm_b = weights["to_out_norm.bias"]
    to_out_w   = weights["to_out.weight"]   # (dim, hidden)

    out = nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=out_norm_w,
        bias=out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate

    out_flat = out.view(-1, hidden_dim)
    out = torch.matmul(out_flat, to_out_w.t()).view(B, N, N, dim)

    return out.to(torch.float32)
