"""
TriMul “outgoing” kernel (AlphaFold‑3 style).

The heavy part is the 3‑D tensor contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

We keep the lightweight LayerNorm / linear / gating work in PyTorch and
off‑load the contraction to a fused Triton kernel.  The kernel works on
blocks of the (i,j) plane while iterating over the k‑dimension in tiles.
All arithmetic is performed in float32 (the input tensors are cast to
fp16 for memory bandwidth, then accumulated in fp32 and finally written
back as fp32).  The kernel supports arbitrary batch size (≤2) and
sequence length up to 1024, and hidden‑dim up to 384.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched “einsum” over the k dimension
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,          # [B, N, N, H]   left projection (fp16)
    right_ptr,         # [B, N, N, H]   right projection (fp16)
    out_ptr,           # [B, N, N, H]   output (fp32)
    # sizes & strides
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)          # linear id over (b, i_tile, j_tile)
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    b = pid // (num_i * num_j)           # batch index
    tile_ij = pid % (num_i * num_j)
    i_tile = tile_ij // num_j
    j_tile = tile_ij %  num_j

    # starting coordinates of the tile
    i0 = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j0 = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_i = i0 < N
    mask_j = j0 < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, H), dtype=tl.float32)

    # iterate over k‑tiles
    for k_start in range(0, N, BLOCK_K):
        k0 = k_start + tl.arange(0, BLOCK_K)
        mask_k = k0 < N

        # ---- load left[b, i, k, :]  ----
        # shape (BLOCK_M, BLOCK_K, H)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            i0[:, None, None] * stride_left_i + \
            k0[None, :, None] * stride_left_k + \
            tl.arange(0, H)[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0).to(tl.float32)

        # ---- load right[b, j, k, :]  ----
        # shape (BLOCK_N, BLOCK_K, H)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            j0[:, None, None] * stride_right_j + \
            k0[None, :, None] * stride_right_k + \
            tl.arange(0, H)[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0).to(tl.float32)

        # ---- compute product over k (broadcast over H) ----
        # left: (M, K, H)   right: (N, K, H)
        # we need Σₖ left[m,k,h] * right[n,k,h]
        # Perform a dot for each h separately via tl.dot
        for h in range(0, H):
            l = left[:, :, h]   # (M, K)
            r = right[:, :, h]  # (N, K)
            # tl.dot expects (M,K) @ (K,N) -> (M,N)
            acc[:, :, h] += tl.dot(l, r, triton_language=tl)

    # ---- write result ----
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        i0[:, None, None] * stride_out_i + \
        j0[None, :, None] * stride_out_j + \
        tl.arange(0, H)[None, None, :] * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None])

# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:         torch.Tensor [B, N, N] (bool or float)
            - weights: dict of model parameters (torch.Tensor)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------ 1. PyTorch preprocessing ------------------------
    # LayerNorm
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_w + norm_b

    # Linear projections (no bias) – we keep them in fp32 first then cast to fp16 for the kernel
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"]
    )  # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"]
    )

    # Apply mask (if any)
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)       # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # Gating
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for the contraction (memory bound)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------ 2. Triton contraction ------------------------
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32, device=device)

    # Strides (in elements, not bytes) for each tensor
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_fp16.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_fp32.stride()

    # Block sizes – tuned for Hopper (SM90)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64   # fits well in shared memory

    total_tiles = B * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N)
    grid = (total_tiles,)

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------ 3. Post‑processing (LayerNorm, gate, linear) ------------------------
    # to_out_norm
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    out = (out_fp32 - out_fp32.mean(-1, keepdim=True)) / torch.sqrt(
        out_fp32.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    # Multiply by out_gate (broadcast over hidden dim)
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
