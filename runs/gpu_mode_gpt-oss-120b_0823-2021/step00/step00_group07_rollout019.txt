"""
TriMul “outgoing” kernel (AlphaFold3)

The expensive part of the forward pass is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* this is a matrix multiplication
    L_d (N×N)  ×  R_d^T (N×N)  →  out_d (N×N)

We therefore reshape the tensors to (B*H, N, N) and launch a batched
GEMM Triton kernel that computes C = A @ Bᵀ for every (B*H) slice.
All LayerNorm, linear projections and gating are kept in PyTorch
(they are memory‑bound and already heavily optimized).  The kernel
operates in fp16 for speed and the result is cast back to fp32
before the final LayerNorm / linear projection.

The kernel works for any batch size (1‑2), sequence length up to 1024
and hidden dimensions that are multiples of the block size (default 64).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ Bᵀ)   A,B: fp16, C: fp16
# --------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes
    stride_am, stride_ak,         # A stride (batch, row, col)
    stride_bm, stride_bn,         # B stride (batch, row, col) – note B is already transposed in memory
    stride_cm, stride_cn,         # C stride (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr,        # number of row tiles processed per program
):
    pid_batch = tl.program_id(0)
    pid = tl.program_id(1)
    # ----------- 2‑D tiled grid (row‑major) --------------------
    num_m_blocks = tl.cdiv(M, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)

    # map linear pid → (m_tile, n_tile) respecting GROUP_M for better L2 reuse
    group_id = pid // (GROUP_M * num_n_blocks)
    first_pid_in_group = group_id * GROUP_M * num_n_blocks
    local_pid = pid - first_pid_in_group

    pid_m = first_pid_in_group // num_n_blocks + local_pid // num_n_blocks
    pid_n = local_pid % num_n_blocks

    # offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the current batch
    a_batch_ptr = a_ptr + pid_batch * stride_am
    b_batch_ptr = b_ptr + pid_batch * stride_bm
    c_batch_ptr = c_ptr + pid_batch * stride_cm

    # accumulate in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am
                                + (k * BLOCK_K + offs_k)[None, :] * stride_ak)
        b_ptrs = b_batch_ptr + (offs_n[None, :] * stride_bn
                                + (k * BLOCK_K + offs_k)[:, None] * stride_bn)  # B is stored transposed

        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + offs_k < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_n[None, :] & (k * BLOCK_K + offs_k < K), other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm
                            + offs_n[None, :] * stride_cn)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Helper to launch the kernel
# --------------------------------------------------------------
def batched_gemm_fp16(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    A,B shape : (batch, M, K) , (batch, N, K)   (B is *not* transposed)
    Returns C shape (batch, M, N) = A @ Bᵀ   computed in fp16.
    """
    assert A.is_contiguous() and B.is_contiguous()
    B_t = B  # we will treat B as already transposed in memory (same layout as Bᵀ)

    batch, M, K = A.shape
    _, N, _ = B.shape

    # strides in element units
    stride_am = A.stride(0)
    stride_ak = A.stride(2)
    stride_bm = B_t.stride(0)
    stride_bn = B_t.stride(1)  # because we load B as (N, K) but want col‑major view
    stride_cm = A.stride(0)    # output has same batch stride
    stride_cn = N * 1          # contiguous last dim

    # allocate output
    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    GROUP_M = 8

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M) * triton.cdiv(N, BLOCK_N),  # total number of tiles per batch
    )
    batched_gemm_kernel[grid](
        A, B_t, C,
        M, N, K,
        stride_am, stride_ak,
        stride_bm, stride_bn,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_M=GROUP_M,
    )
    return C


# --------------------------------------------------------------
# Main entry point required by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module.
    Only the O(N³) multiplication is Triton‑accelerated.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ---------- LayerNorm ----------
    eps = 1e-5
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + eps
    )
    x = x * w_norm + b_norm

    # ---------- Linear projections ----------
    # shape: (B,N,N,hidden)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare for batched GEMM ----------
    # bring hidden dimension to the batch axis: (B*H, N, N)
    B, N, _, H = left.shape
    left_ = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_ = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()

    # compute C = left @ rightᵀ   in fp16
    left_fp16 = left_.to(torch.float16)
    right_fp16 = right_.to(torch.float16)
    out_fp16 = batched_gemm_fp16(left_fp16, right_fp16)   # (B*H, N, N)

    # reshape back to (B,N,N,hidden)
    out = out_fp16.to(torch.float32).reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Output LayerNorm & linear ----------
    w_out_norm = weights["to_out_norm.weight"]
    b_out_norm = weights["to_out_norm.bias"]
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + eps
    )
    out = out * w_out_norm + b_out_norm

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out  # float32 tensor of shape (B, N, N, dim)
