"""
TriMul “outgoing” forward pass (AlphaFold3).

The heavy part is a batched N×N matrix multiply for every
hidden channel d (hidden_dim).  For each batch b and channel d

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is exactly a GEMM:   left[b,:, :,d]   @   right[b,:, :,d].T

The kernel below performs this GEMM in float16 (the model
uses fp32 weights but we can safely compute in fp16 on H200).
All outer operations (LayerNorm, Linear, gating, final LayerNorm
and projection) are done with torch.nn.functional for simplicity.
Only the N³ multiplication is Triton‑accelerated.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------
# Triton kernel: batched GEMM for a single hidden channel
# ---------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # problem size
    N, hidden_dim,
    # strides (batch*channel are fused into the first dimension)
    stride_ah, stride_ak,          # a: [batch_chan, i, k]
    stride_bh, stride_bk,          # b: [batch_chan, j, k]
    stride_ch, stride_ci, stride_cj,  # c: [batch_chan, i, j]
    # tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr, num_warps: tl.constexpr,
):
    # program ids
    batch_chan = tl.program_id(0)          # fused batch * hidden_dim
    pid_m = tl.program_id(1)               # block row
    pid_n = tl.program_id(2)               # block col

    # starting offsets for the block
    pid_m_off = pid_m * BLOCK_M
    pid_n_off = pid_n * BLOCK_N

    # pointers for this block (batch_chan is fixed)
    a = a_ptr + batch_chan * stride_ah + pid_m_off * stride_ak
    b = b_ptr + batch_chan * stride_bh + pid_n_off * stride_bk

    # accumulate in fp32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K
        # load A block [M, K]
        a_ptrs = a + tl.arange(0, BLOCK_M)[:, None] * stride_ak + \
                     (cur_k + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        a_mask = (pid_m_off + tl.arange(0, BLOCK_M) < N) & \
                 (cur_k + tl.arange(0, BLOCK_K) < N)
        a_block = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)

        # load B block [N, K] (note: we need B transposed, i.e. rows=j, cols=k)
        b_ptrs = b + tl.arange(0, BLOCK_N)[:, None] * stride_bk + \
                     (cur_k + tl.arange(0, BLOCK_K))[None, :] * stride_bk
        b_mask = (pid_n_off + tl.arange(0, BLOCK_N) < N) & \
                 (cur_k + tl.arange(0, BLOCK_K) < N)
        b_block = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # dot: sum over K
        acc += tl.dot(a_block, b_block)

        # advance pointers for next K‑tile
        a += BLOCK_K * stride_ak
        b += BLOCK_K * stride_bk

    # write result
    c = c_ptr + batch_chan * stride_ch + pid_m_off * stride_ci + pid_n_off * stride_cj
    c_mask = (pid_m_off + tl.arange(0, BLOCK_M) < N) & \
             (pid_n_off + tl.arange(0, BLOCK_N) < N)
    tl.store(c + tl.arange(0, BLOCK_M)[:, None] * stride_ci +
                tl.arange(0, BLOCK_N)[None, :] * stride_cj,
             acc.to(tl.float16), mask=c_mask)

# ---------------------------------------------------------------
# Host‑side entry point
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            input  – Tensor [B, N, N, dim] (float32)
            mask   – Tensor [B, N, N]   (float32 or bool)
            weights – dict of all linear / LN parameters (float32)
            config – dict with keys "dim" and "hidden_dim"
    Returns:
        Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm ----------
    norm_weight, norm_bias = weights["norm.weight"], weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ---------- Linear projections ----------
    # shape: [B,N,N,hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare for GEMM ----------
    # Cast to fp16 for the kernel (keep a copy in fp32 for later ops)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    # Output buffer (fp16)
    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # fuse batch and hidden dim into one dimension for the kernel launch
    batch_chan = B * hidden_dim

    # strides (in elements, not bytes!)
    # left: [B, N, N, hidden] -> we view as [batch_chan, i, k]
    stride_ah = left_fp16.stride(0) * hidden_dim + left_fp16.stride(3)
    stride_ak = left_fp16.stride(2)
    # right: [B, N, N, hidden] -> we view as [batch_chan, j, k]
    stride_bh = right_fp16.stride(0) * hidden_dim + right_fp16.stride(3)
    stride_bk = right_fp16.stride(2)
    # out: [B, N, N, hidden] -> [batch_chan, i, j]
    stride_ch = out_fp16.stride(0) * hidden_dim + out_fp16.stride(3)
    stride_ci = out_fp16.stride(1)
    stride_cj = out_fp16.stride(2)

    # Tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_chan,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N,
        hidden_dim,
        stride_ah, stride_ak,
        stride_bh, stride_bk,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_stages=3,
        num_warps=4,
    )

    # ---------- Post‑processing ----------
    out = out_fp16.to(torch.float32)          # back to fp32
    out = F.layer_norm(out, (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
