"""
TriMul “outgoing” kernel (forward only)

The heavy part of the TriMul forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication of shape
    (B, hidden, N, N) × (B, hidden, N, N)ᵀ → (B, hidden, N, N)

The implementation below

* performs the cheap LayerNorm / Linear / gating work in PyTorch
* launches a custom Triton kernel that computes the
  (B, hidden, N, N) matmul on the GPU (using fp16 accumulation and
  fp32 final result) – this satisfies the “must contain a Triton kernel”
  requirement while still being fast on the H200.
* finally runs the output LayerNorm / Linear in PyTorch.

The kernel is written for the H200 (SM‑90) but works on any recent
CUDA‑capable GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the (B, H, N, N) tensors.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr,                     # left  tensor pointer   (B,H,N,N)  row‑major
    b_ptr,                     # right tensor pointer   (B,H,N,N)  row‑major
    c_ptr,                     # output pointer         (B,H,N,N)
    B, H, N,                   # static sizes
    stride_ah, stride_ak,      # strides for a: (batch, hidden, row, col)
    stride_ai, stride_aj,
    stride_bh, stride_bk,      # strides for b
    stride_bi, stride_bj,
    stride_ch, stride_ck,      # strides for c
    stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,    # tile size for the i‑dimension
    BLOCK_N: tl.constexpr,    # tile size for the j‑dimension
    BLOCK_K: tl.constexpr,    # tile size for the reduction k
):
    pid = tl.program_id(0)                # 1‑D grid
    # decode linear pid → (b, h, i_tile, j_tile)
    num_h = H
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    total_tiles = B * num_h * num_i * num_j
    pid = tl.minimum(pid, total_tiles - 1)

    b = pid // (num_h * num_i * num_j)
    tmp = pid % (num_h * num_i * num_j)
    h = tmp // (num_i * num_j)
    tmp = tmp % (num_i * num_j)
    i_tile = tmp // num_j
    j_tile = tmp % num_j

    # offsets for the current tile
    i_offset = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds
    i_mask = i_offset < N
    j_mask = j_offset < N

    # pointers to the beginning of the tile
    a_tile_ptr = a_ptr + b * stride_ah + h * stride_ak \
               + i_offset[:, None] * stride_ai \
               + tl.arange(0, BLOCK_K)[None, :] * stride_aj
    b_tile_ptr = b_ptr + b * stride_bh + h * stride_bk \
               + j_offset[:, None] * stride_bi \
               + tl.arange(0, BLOCK_K)[None, :] * stride_bj

    # accumulator in fp32 (more stable than fp16‑only)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------- reduction over k ----------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # k‑range for this iteration
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # load a block: shape (M, K)
        a = tl.load(
            a_tile_ptr + k_offset[None, :] * stride_aj,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        )
        # load b block: shape (N, K) – note we need the *same* k dimension
        b = tl.load(
            b_tile_ptr + k_offset[None, :] * stride_bj,
            mask=j_mask[:, None] & k_mask[None, :],
            other=0.0,
        )

        # dot‑product over k (inner product)
        # a and b are fp16 -> convert to fp32 for the dot
        acc += tl.dot(a, b, out_dtype=tl.float32)

        # advance pointers to next k‑block
        a_tile_ptr += BLOCK_K * stride_aj
        b_tile_ptr += BLOCK_K * stride_bj

    # write result
    c_tile_ptr = c_ptr + b * stride_ch + h * stride_ck \
               + i_offset[:, None] * stride_ci \
               + j_offset[None, :] * stride_cj
    tl.store(c_tile_ptr, acc.to(tl.float32), mask=i_mask[:, None] & j_mask[None, :])

# ----------------------------------------------------------------------
# Helper: fused LayerNorm (no bias handling needed for speed)
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: [..., C]
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    normalized = (x - mean) / torch.sqrt(var + eps)
    return normalized * weight + bias

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : (input, mask, weights, config)
        input   – torch.Tensor [B, N, N, dim] (float32)
        mask    – torch.Tensor [B, N, N]   (float32 or bool)
        weights – dict with all linear / layer‑norm parameters
        config  – dict with "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )                               # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → hidden
    # ------------------------------------------------------------------
    # shape: (..., dim) @ (hidden, dim)ᵀ  ==  (..., hidden)
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"]   # shape (hidden, dim)
    )
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"]
    )                               # both [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 3) Apply mask if present
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate               # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 5) Heavy part: batched matmul over the k dimension.
    #    Re‑shape to (B, hidden, N, N) and launch the Triton kernel.
    # ------------------------------------------------------------------
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # allocate output in fp32 (kernel writes fp32)
    out_t = torch.empty_like(left_t, dtype=torch.float32)

    # strides (in elements) for each tensor
    stride_ah, stride_ak, stride_ai, stride_aj = left_t.stride()
    stride_bh, stride_bk, stride_bi, stride_bj = right_t.stride()
    stride_ch, stride_ck, stride_ci, stride_cj = out_t.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    total_tiles = B * hidden * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N)
    grid = (total_tiles,)

    batched_matmul_kernel[grid](
        left_t,
        right_t,
        out_t,
        B, hidden, N,
        stride_ah, stride_ak,
        stride_ai, stride_aj,
        stride_bh, stride_bk,
        stride_bi, stride_bj,
        stride_ch, stride_ck,
        stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                 # tune for Hopper
    )

    # back to original layout [B, N, N, hidden]
    out = out_t.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate                    # out_gate broadcasted over hidden dim
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"]
    )                                        # → [B, N, N, dim]

    return out.to(torch.float32)
