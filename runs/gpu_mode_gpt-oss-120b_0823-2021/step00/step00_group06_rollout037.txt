"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The heavy part of the forward pass is the 4‑D einsum

    out[b, i, j, d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape (B, N, N, H) (H = hidden_dim).
For each batch‑and‑head pair (b,d) this is exactly a matrix multiplication
      (N×N)  ×  (N×N)ᵀ  → (N×N)

We fuse this with a custom Triton kernel that performs a batched GEMM
over the `(B*H)` independent “heads”.  All other operations
(layer‑norms, linear projections, gating and the final linear layer)
are cheap enough to keep in PyTorch.

The kernel uses the classic three‑level tiling (BLOCK_M, BLOCK_N,
BLOCK_K) and works on the H200 (SM‑90) in fp16 for the multiply‑accumulate,
producing fp32 output that is finally reshaped back to `[B,N,N,H]`.
The rest of the module (norms, gates, final linear) is carried out in
torch.nn.functional for clarity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------
# Triton kernel: batched matrix multiplication (N×N)·(N×N)ᵀ
# ---------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # dimensions
    N, HEADS,
    # strides of the input tensors (contiguous NHWC layout)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # tiling sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute
        out[head, i, j] = Σ_k left[head, i, k] * right[head, j, k]
    where `head = b * H + h` indexes the (batch, hidden) pair.
    """
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_h = tl.program_id(2)          # head (batch*hidden)

    # ------------------- offsets within the tile -------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # compute base pointer for this head
    # head = b*H + h  →  we treat it as a single leading dimension
    head_offset = pid_h * (
        stride_left_b * 0   # zero because we will add the per‑head offset below
    )
    # stride of the “head” dimension (i.e. batch*hidden stride)
    stride_head = stride_left_b // HEADS  # equals N*N

    # pointer arithmetic: add head offset + i/k strides
    left_ptr   = left_ptr   + pid_h * stride_head
    right_ptr  = right_ptr  + pid_h * stride_head
    out_ptr    = out_ptr    + pid_h * stride_head

    # ------------------- load tiles -------------------
    # left:  (BLOCK_M, BLOCK_K)
    a_ptrs = left_ptr + (offs_i[:, None] * stride_left_i
                         + offs_k[None, :] * stride_left_k)
    a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

    # right: (BLOCK_N, BLOCK_K) – we need the transpose on the K dimension
    b_ptrs = right_ptr + (offs_j[:, None] * stride_right_j
                          + offs_k[None, :] * stride_right_k)
    b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- reduction over K -------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load current K‑tile
        cur_a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        cur_b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # fp16 dot, accumulate in fp32
        acc += tl.dot(cur_a, cur_b).to(tl.float32)

        # advance pointers
        a_ptrs   += BLOCK_K * stride_left_k
        b_ptrs   += BLOCK_K * stride_right_k
        mask_k   = (k * BLOCK_K + offs_k) < N

    # ------------------- store result -------------------
    out_ptrs = out_ptr + (offs_i[:, None] * stride_out_i
                         + offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ---------------------------------------------------------
# Host‑side wrapper
# ---------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator using a fused Triton kernel.
    Input:
        data = (input_tensor, mask, weights, config)
    Returns:
        out tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper: linear w/o bias (weight shape: out_features x in_features)
    # ------------------------------------------------------------------
    def linear(x, w):
        # x: [..., dim]; w: [out_dim, dim]
        return F.linear(x, w)

    # ------------------------------------------------------------------
    # LayerNorm (no epsilon tweaking – use PyTorch defaults)
    # ------------------------------------------------------------------
    def layer_norm(x, w, b):
        return F.layer_norm(x, (x.shape[-1],), weight=w, bias=b)

    # ------------------------------------------------------------------
    # Load weights (they are already on the correct device)
    # ------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm over the last dim (dim)
    # ------------------------------------------------------------------
    x = layer_norm(input_tensor, norm_w, norm_b)          # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections to hidden_dim
    # ------------------------------------------------------------------
    left  = linear(x, left_proj_w)   # [B,N,N,hidden]
    right = linear(x, right_proj_w)  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3️⃣  Apply mask (if provided)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣  Gating (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(linear(x, left_gate_w))
    right_gate = torch.sigmoid(linear(x, right_gate_w))
    out_gate   = torch.sigmoid(linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Triton batched GEMM  (left @ rightᵀ) → shape [B,N,N,hidden]
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (weights already in fp32)
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)
    out_f16 = torch.empty_like(left_f16)   # allocate output

    # strides for contiguous layout [B, N, N, H] (NHWC)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_f16.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right_f16.stride()
    stride_out_b,   stride_out_i,   stride_out_j,   stride_out_h = out_f16.stride()

    # kernel launch parameters – tuned for Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_heads = B * hidden_dim                     # treat (b,h) as one dimension
    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        total_heads                # head tiles
    )

    tri_mul_kernel[grid](
        left_f16, right_f16, out_f16,
        N, total_heads,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # Hopper prefers 4‑warp kernels
    )

    # Convert back to fp32 for the remaining ops
    out = out_f16.to(torch.float32)               # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6️⃣  Post‑projection LayerNorm + gating + final Linear
    # ------------------------------------------------------------------
    out = layer_norm(out, to_out_norm_w, to_out_norm_b)   # LN over hidden
    out = out * out_gate                                 # element‑wise gate
    out = linear(out, to_out_w)                          # back to `dim`

    return out
