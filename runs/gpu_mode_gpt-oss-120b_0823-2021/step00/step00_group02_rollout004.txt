"""
TriMul “outgoing” forward pass (AlphaFold3).

The heavy part is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix‑multiply for every hidden channel *d*.
The kernel below tiles the (i,j) output plane (BLOCK_M×BLOCK_N) and the
reduction dimension *k* (BLOCK_K).  The hidden dimension *d* is kept inside the
registers (BLOCK_D = hidden_dim, typically 128) and accumulated in FP32.
All other operations (LayerNorm, linear projections, gating, masking) are
performed with native PyTorch – they are cheap compared with the N³ reduction.
The kernel works for any batch size ≤2 and any sequence length that fits into
GPU memory (the test suite uses up to 1024).  The final output is cast back to
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -------------------------------------------------------------------------
# Triton kernel : tiled 4‑D einsum
# -------------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,          # const float16*   [B,N,N,D]  (i,k)
    right_ptr,         # const float16*   [B,N,N,D]  (j,k)
    out_ptr,           #       float32*   [B,N,N,D]  (i,j)
    B, N, D,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)                     # one program per (b,i,j) block
    total_blocks = B * N * N
    if pid >= total_blocks:
        return

    # -----------------------------------------------------------------
    # Decode pid → (b,i,j)
    # -----------------------------------------------------------------
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # -----------------------------------------------------------------
    # Tile offsets
    # -----------------------------------------------------------------
    offs_i = i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_d = tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Main k‑loop
    # -----------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_tile * BLOCK_K + offs_k
        mask_k = cur_k < N

        # left[b,i,k,d]  -> shape (M,K,D)
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + offs_i[:, None, None] * stride_li
            + cur_k[None, :, None] * stride_lk
            + offs_d[None, None, :] * stride_ld
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & (offs_d[None, None, :] < D), other=0.0)

        # right[b,j,k,d] -> shape (N,K,D)
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + offs_j[None, :, None] * stride_rj
            + cur_k[:, None, None] * stride_rk
            + offs_d[None, None, :] * stride_rd
        )
        right = tl.load(right_ptrs, mask=mask_j[None, :, None] & mask_k[:, None, None] & (offs_d[None, None, :] < D), other=0.0)

        # accumulate: (M,K,D) * (N,K,D) → sum over K
        # broadcasting: left[..., d] * right[..., d] → (M,N,D)
        acc += tl.dot(left, right, out_dtype=tl.float32)  # dot does reduction over K

    # -----------------------------------------------------------------
    # Write back
    # -----------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + offs_i[:, None, None] * stride_oi
        + offs_j[None, :, None] * stride_oj
        + offs_d[None, None, :] * stride_od
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & (offs_d[None, None, :] < D),
    )


# -------------------------------------------------------------------------
# Host entry point
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    expensive N³ einsum.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # LayerNorm (input)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    # shape: (B,N,N,hidden)
    left = F.linear(x, weights["left_proj.weight"].to(device))          # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # -----------------------------------------------------------------
    # Apply mask (if present)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                       # (B,N,N,1)
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # Gating (sigmoid) and apply
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device)))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device)))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Cast to float16 for compute (helps throughput on H200)
    # -----------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # -----------------------------------------------------------------
    # Allocate output buffer (float32 as required)
    # -----------------------------------------------------------------
    out = torch.empty_like(input_tensor, dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # Strides (in elements) for the three tensors
    # -----------------------------------------------------------------
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # -----------------------------------------------------------------
    # Launch parameters
    # -----------------------------------------------------------------
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = hidden                       # hidden_dim is small (≤128)

    total_blocks = bs * N * N
    grid = (total_blocks,)

    trimul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=N,
        D=hidden,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_ld=stride_ld,
        stride_rb=stride_rb,
        stride_rj=stride_rj,
        stride_rk=stride_rk,
        stride_rd=stride_rd,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_od=stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # -----------------------------------------------------------------
    # Post‑processing: to_out_norm, gating, final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
    )
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"].to(device))
    return out
