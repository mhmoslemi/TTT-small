"""
TriMul “outgoing” kernel (AlphaFold‑3)

The expensive part of TriMul is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed head *d* this is a regular matrix multiplication
    L_d   : (B,N,N)   where L_d[b,i,k] = left[b,i,k,d]
    R_d^T : (B,N,N)   where R_d^T[b,k,j] = right[b,j,k,d]

so   out[:,:, :,d] = L_d @ R_d^T.

We implement a tiled batched‑matmul kernel in Triton that computes a
block of (M×N) for a *subset* of the head dimension.  The surrounding
Python code:

*   LayerNorm, linear projections, and gating are done with regular
    PyTorch (they are cheap compared to the N³ work).
*   The projection results are reshaped to (B,N,N,H) and passed to the
    Triton kernel together with the mask.
*   The kernel works in mixed precision (FP16 accumulator, FP32 output)
    and writes the result in float32.
*   Finally we apply the output LayerNorm, output‑gate and linear layer
    in PyTorch.

The kernel is written for SM‑90 (H200) – block sizes were chosen to keep
the shared‑memory footprint low while exposing enough parallelism.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for a slice of the head dimension
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # tensor sizes
    B, N, H,
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time constants
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,  # number of head channels processed per program
):
    """Compute a block of the output for a subset of the head dimension."""
    pid = tl.program_id(axis=0)                     # unique program id
    # linearise the three axes (batch, head block, block row, block col)
    total_blocks = B * tl.cdiv(H, BLOCK_H) * tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N)
    pid = tl.program_id(0)
    # decode
    tmp = pid
    block_col = tmp % tl.cdiv(N, BLOCK_N); tmp //= tl.cdiv(N, BLOCK_N)
    block_row = tmp % tl.cdiv(N, BLOCK_M); tmp //= tl.cdiv(N, BLOCK_M)
    head_block = tmp % tl.cdiv(H, BLOCK_H);    tmp //= tl.cdiv(H, BLOCK_H)
    batch = tmp

    # offsets for the current block
    offs_i = block_row * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_col * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = head_block * BLOCK_H + tl.arange(0, BLOCK_H)

    # mask out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # pointers to the start of the block (including batch & head offsets)
    left_tile_ptr = left_ptr  + batch * stride_left_b \
                               + offs_i[:, None, None] * stride_left_i \
                               + tl.arange(0, BLOCK_M)[None, :, None] * 0 \
                               + offs_h[None, None, :] * stride_left_h
    # right is transposed on k dimension, we load it as (j,k) and later use k as reduction axis
    right_tile_ptr = right_ptr + batch * stride_right_b \
                                 + offs_j[None, :, None] * stride_right_j \
                                 + tl.arange(0, BLOCK_N)[None, None, :] * 0 \
                                 + offs_h[None, None, :] * stride_right_h

    # accumulator in fp32 (good accuracy) – we accumulate over k (the reduction axis)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_M)):
        # compute actual k offset for this tile
        offs_k = k * BLOCK_M + tl.arange(0, BLOCK_M)
        mask_k = offs_k < N

        # left:  (B, i, k, h)   -> we need L[b,i,k,h]
        L = tl.load(
            left_ptr
            + batch * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_h[None, None, :] * stride_left_h,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # right: (B, j, k, h)  we need R[b,j,k,h]; note we will use the same k index
        R = tl.load(
            right_ptr
            + batch * stride_right_b
            + offs_j[None, :, None] * stride_right_j
            + offs_k[None, None, :] * stride_right_k
            + offs_h[None, None, :] * stride_right_h,
            mask=mask_j[None, :, None] & mask_k[None, None, :] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # outer product over k -> (i,j,h) accumulation
        # broadcasting: L (i,k,h) , R (j,k,h) -> we need sum_k L[i,k,h] * R[j,k,h]
        # we compute via tl.dot over the k dimension for each head slice
        # to keep it simple we use explicit multiplication+sum
        acc += tl.sum(L[:, :, None] * R[None, :, :], axis=1)

    # write result
    out_ptrs = out_ptr \
        + batch * stride_out_b \
        + offs_i[:, None, None] * stride_out_i \
        + offs_j[None, :, None] * stride_out_j \
        + offs_h[None, None, :] * stride_out_h

    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])


# ----------------------------------------------------------------------
# Host‑side wrapper that launches the kernel
# ----------------------------------------------------------------------
def trimul_matmul_tiled(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : torch.Tensor of shape (B, N, N, H)  (dtype=float16 or float32)
    returns      : torch.Tensor of shape (B, N, N, H)  (dtype=float32)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    device = left.device

    # ensure fp16 for compute (helps occupancy); accumulate in fp32 inside kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides in elements (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # ------------------------------------------------------------------
    # launch parameters – tuned for Hopper
    # ------------------------------------------------------------------
    BLOCK_M = 64   # rows
    BLOCK_N = 64   # cols
    BLOCK_H = 32   # head channels per program

    total_blocks = B * ( (H + BLOCK_H - 1)//BLOCK_H ) * ( (N + BLOCK_M - 1)//BLOCK_M ) * ( (N + BLOCK_N - 1)//BLOCK_N )
    grid = (total_blocks,)

    trimul_matmul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# Full custom kernel (entry point)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward for the “outgoing” TriMul module.

    Args
    ----
    data : (input, mask, weights, config)
        input  – tensor [B, N, N, dim] (float32)
        mask   – tensor [B, N, N] (float32 or bool)
        weights – dict with all linear / layer‑norm parameters
        config  – {"dim": int, "hidden_dim": int}
    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1) LayerNorm on the last dim
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # LayerNorm = (x - mean) / sqrt(var+eps) * w + b
    eps = 1e-5
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_w + norm_b   # shape [B,N,N,dim]

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)   ->  [B,N,N,hidden_dim]
    # --------------------------------------------------------------
    def linear_no_bias(x, w):
        # w: (out, in)
        return torch.nn.functional.linear(x, w)   # bias = None

    left = linear_no_bias(x, weights["left_proj.weight"])
    right = linear_no_bias(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Apply mask (if present) – mask is broadcasted on hidden dim
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(linear_no_bias(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_no_bias(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear_no_bias(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Core N³ operation – Triton kernel
    # --------------------------------------------------------------
    out = trimul_matmul_tiled(left, right)   # shape [B,N,N,hidden_dim], dtype=float32

    # --------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    # LayerNorm on hidden dimension
    mean_h = out.mean(dim=-1, keepdim=True)
    var_h = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_h) * torch.rsqrt(var_h + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # apply output gate (broadcasted)
    out = out * out_gate

    # final linear back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out  # float32, shape [B,N,N,dim]
