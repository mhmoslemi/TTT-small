"""
TriMul “outgoing” kernel (AlphaFold3)

The heavy part is the 3‑dimensional “tri‑mul”:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We fuse this N³ reduction into a single Triton kernel.
All LayerNorm / Linear steps are kept in PyTorch (they are cheap compared
to the N³ work).  The kernel processes tiles of (i, j, d) and iterates over
k in blocks, accumulating in fp32 and finally writing fp32 results.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,          # [B, N, N, H]   (i, k, d)
    right_ptr,         # [B, N, N, H]   (j, k, d)
    out_ptr,           # [B, N, N, H]   (i, j, d)

    # sizes
    B, N, H,          # batch, seq_len, hidden_dim
    stride_lb_i, stride_lb_k, stride_lb_d,   # left strides (i,k,d)
    stride_rb_j, stride_rb_k, stride_rb_d,   # right strides (j,k,d)
    stride_ob_i, stride_ob_j, stride_ob_d,   # out strides (i,j,d)

    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,   # tile size for i
    BLOCK_N: tl.constexpr,   # tile size for j
    BLOCK_D: tl.constexpr,   # tile size for d
    BLOCK_K: tl.constexpr,   # tile size for k
):
    pid = tl.program_id(0)                # linear id over (b, i_tile, j_tile)
    num_b = B
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    b = pid // (num_i * num_j)
    tmp = pid % (num_i * num_j)
    i_tile = tmp // num_j
    j_tile = tmp % num_j

    # tile offsets
    offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = tl.arange(0, BLOCK_D)

    # masks for out‑of‑bounds tiles
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # allocate accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # iterate over k
    for k_base in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k_base, BLOCK_K)

        # load left  [B, i, k, d]
        left_offset = (
            b * stride_lb_i * N * H
            + offs_i[:, None, None] * stride_lb_i
            + (k_base + tl.arange(0, cur_k))[None, :, None] * stride_lb_k
            + offs_d[None, None, :] * stride_lb_d
        )
        left = tl.load(left_ptr + left_offset,
                       mask=(mask_i[:, None, None] & (tl.arange(0, cur_k) < cur_k)[:, None] & mask_d[None, None, :]),
                       other=0.0).to(tl.float32)

        # load right [B, j, k, d]
        right_offset = (
            b * stride_rb_j * N * H
            + offs_j[:, None, None] * stride_rb_j
            + (k_base + tl.arange(0, cur_k))[None, :, None] * stride_rb_k
            + offs_d[None, None, :] * stride_rb_d
        )
        right = tl.load(right_ptr + right_offset,
                        mask=(mask_j[:, None, None] & (tl.arange(0, cur_k) < cur_k)[:, None] & mask_d[None, None, :]),
                        other=0.0).to(tl.float32)

        # multiply and accumulate over k dimension
        # left: (M, K, D)   right: (N, K, D)  -> we need sum_k left[m,k,d] * right[n,k,d]
        # use broadcasting: (M,1,K,D) * (1,N,K,D) then sum over K
        # but we can vectorize: for each d we do dot over k
        for d in range(0, BLOCK_D):
            l = left[:, :, d]          # (M, K)
            r = right[:, :, d]         # (N, K)
            # tl.dot expects (M,K) * (K,N) -> (M,N)
            acc[:, :, d] += tl.dot(l, r.T)

    # write result
    out_offset = (
        b * stride_ob_i * N * H
        + offs_i[:, None, None] * stride_ob_i
        + offs_j[None, :, None] * stride_ob_j
        + offs_d[None, None, :] * stride_ob_d
    )
    tl.store(out_ptr + out_offset,
             acc.to(tl.float32),
             mask=(mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]))


# ----------------------------------------------------------------------
# Host helper (LayerNorm + Linear in PyTorch)
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Args:
        data: (input, mask, weights, config)
            input  – [B, N, N, dim]   (float32)
            mask   – [B, N, N]        (float32 or bool)
            weights – dict with all linear / norm parameters
            config – dict containing dim, hidden_dim

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the input (dim)
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    ).to(torch.float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) + mask
    # ------------------------------------------------------------------
    # shape: [B, N, N, hidden_dim]
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 3) Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Triton kernel: tri‑mul over k
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements) for the three tensors
    stride_lb_i, stride_lb_k, stride_lb_d = left.stride(1), left.stride(2), left.stride(3)
    stride_rb_j, stride_rb_k, stride_rb_d = right.stride(1), right.stride(2), right.stride(3)
    stride_ob_i, stride_ob_j, stride_ob_d = out.stride(1), out.stride(2), out.stride(3)

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_D = 32
    BLOCK_K = 64

    # grid: one program per (batch, i_tile, j_tile)
    grid = (triton.cdiv(B * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N), 1),)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_lb_i=stride_lb_i,
        stride_lb_k=stride_lb_k,
        stride_lb_d=stride_lb_d,
        stride_rb_j=stride_rb_j,
        stride_rb_k=stride_rb_k,
        stride_rb_d=stride_rb_d,
        stride_ob_i=stride_ob_i,
        stride_ob_j=stride_ob_j,
        stride_ob_d=stride_ob_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_D=BLOCK_D,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # enough to hide latency on H200
    )

    # ------------------------------------------------------------------
    # 5) to_out_norm + gating + final linear
    # ------------------------------------------------------------------
    out = _layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
