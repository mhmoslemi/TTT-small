"""
TriMul “outgoing” implementation.

The expensive step is the N‑by‑N pairwise dot‑product over the
sequence dimension:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We fuse the three element‑wise stages (mask, left‑gate, right‑gate) in a
single Triton kernel `apply_mask_and_gate`.  The remaining linear
projections, the LayerNorms and the large N³ reduction are performed
with regular PyTorch (the reduction is a batched matmul that PyTorch /
torch.cuda already executes at near‑optimal performance on the H200).

The kernel works for any batch size, sequence length and hidden size
that fit in GPU memory.  All intermediate tensors are kept in FP16
while the final output is cast back to FP32 as required by the
reference model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel – mask + gate fusion
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_and_gate_kernel(
    x_ptr,                # [B, N, N, C]      input (FP16)
    mask_ptr,             # [B, N, N]         mask  (FP16, 0/1)
    left_proj_ptr,        # [C, H]            left_proj weight (FP16)
    right_proj_ptr,       # [C, H]            right_proj weight (FP16)
    left_gate_ptr,        # [C, H]            left gate weight (FP16)
    right_gate_ptr,       # [C, H]            right gate weight (FP16)
    out_gate_ptr,         # [C, H]            out gate weight (FP16)
    left_out_ptr,         # [B, N, N, H]      left after proj & gate (FP16)
    right_out_ptr,        # [B, N, N, H]      right after proj & gate (FP16)
    out_gate_out_ptr,     # [B, N, N, H]      out‑gate (FP16)
    B, N, C, H,           # static sizes
    BLOCK_B: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    """
    For every element (b,i,j) we compute
        left  = (x @ left_proj)   * mask * sigmoid(x @ left_gate)
        right = (x @ right_proj)  * mask * sigmoid(x @ right_gate)
        out_g = sigmoid(x @ out_gate)
    All operations are fused to minimise global memory traffic.
    """

    pid = tl.program_id(0)
    # 1‑D launch: each program works on a (b, i, j) tile
    b_idx = pid // (tl.cdiv(N, BLOCK_N) * tl.cdiv(N, BLOCK_N))
    rem   = pid % (tl.cdiv(N, BLOCK_N) * tl.cdiv(N, BLOCK_N))
    i_idx = rem // tl.cdiv(N, BLOCK_N)
    j_idx = rem % tl.cdiv(N, BLOCK_N)

    # coordinate offsets inside the tile
    offs_b = b_idx * BLOCK_B + tl.arange(0, BLOCK_B)
    offs_i = i_idx * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_j = j_idx * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_c = tl.arange(0, BLOCK_C)

    # predicates – handle tail
    mask_b = offs_b < B
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_c = offs_c < C

    # ------------------------------------------------------------------
    # Load a single (b,i,j) vector of size C
    # ------------------------------------------------------------------
    # Shape we load: [B, N, N, C]
    x_ptrs = (
        x_ptr
        + (offs_b[:, None, None, None] * N * N * C)
        + (offs_i[None, :, None, None] * N * C)
        + (offs_j[None, None, :, None] * C)
        + offs_c[None, None, None, :]
    )
    x = tl.load(x_ptrs, mask=mask_b[:, None, None, None] &
                         mask_i[None, :, None, None] &
                         mask_j[None, None, :, None] &
                         mask_c[None, None, None, :], other=0.0)

    # ------------------------------------------------------------------
    # Load mask (float16 0/1)
    # ------------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + (offs_b[:, None, None] * N * N)
        + (offs_i[None, :, None] * N)
        + offs_j[None, None, :]
    )
    m = tl.load(mask_ptrs, mask=mask_b[:, None, None] &
                           mask_i[None, :, None] &
                           mask_j[None, None, :], other=0.0)
    # expand mask to channel dim
    m = m[..., None]  # [B,N,N,1]

    # ------------------------------------------------------------------
    # Linear projections (C -> H)
    # ------------------------------------------------------------------
    # weight layout: [C, H]  (row‑major)
    # Compute x @ W   =>   Σ_c x[...,c] * W[c,h]
    # We do a small matmul inside the kernel (C is at most 384)
    # ------------------------------------------------------------------
    # reshape for dot: (B,N,N,1,C) dot (C,H) -> (B,N,N,1,H)
    x_reshape = x.reshape(-1, C)                     # [(B*N*N), C]
    # Load weight once per program (tiny)
    w_left  = tl.load(left_proj_ptr + offs_c[:, None] * H, mask=mask_c[:, None], other=0.0)
    w_right = tl.load(right_proj_ptr + offs_c[:, None] * H, mask=mask_c[:, None], other=0.0)
    w_lgate = tl.load(left_gate_ptr + offs_c[:, None] * H, mask=mask_c[:, None], other=0.0)
    w_rgate = tl.load(right_gate_ptr + offs_c[:, None] * H, mask=mask_c[:, None], other=0.0)
    w_ogate = tl.load(out_gate_ptr  + offs_c[:, None] * H, mask=mask_c[:, None], other=0.0)

    # Compute projections (using dot for each output channel)
    # Because C is small we unroll manually with a reduction over C.
    # acc_* shape: [B,N,N,H]
    acc_left = tl.zeros((BLOCK_B, BLOCK_N, BLOCK_N, H), dtype=tl.float16)
    acc_right= tl.zeros((BLOCK_B, BLOCK_N, BLOCK_N, H), dtype=tl.float16)
    acc_lgate= tl.zeros((BLOCK_B, BLOCK_N, BLOCK_N, H), dtype=tl.float16)
    acc_rgate= tl.zeros((BLOCK_B, BLOCK_N, BLOCK_N, H), dtype=tl.float16)
    acc_ogate= tl.zeros((BLOCK_B, BLOCK_N, BLOCK_N, H), dtype=tl.float16)

    # loop over channel dimension C
    for c in range(0, C, BLOCK_C):
        cur_c = c + offs_c
        cm = cur_c < C
        x_c = tl.load(x_ptr + (cur_c[None, None, None, :] * 1), mask=cm, other=0.0)  # [B,N,N,blockC]

        # broadcast weight tiles
        w_l  = tl.load(left_proj_ptr + cur_c[:, None] * H, mask=cm[:, None], other=0.0)
        w_r  = tl.load(right_proj_ptr + cur_c[:, None] * H, mask=cm[:, None], other=0.0)
        w_lg = tl.load(left_gate_ptr + cur_c[:, None] * H, mask=cm[:, None], other=0.0)
        w_rg = tl.load(right_gate_ptr + cur_c[:, None] * H, mask=cm[:, None], other=0.0)
        w_og = tl.load(out_gate_ptr  + cur_c[:, None] * H, mask=cm[:, None], other=0.0)

        # perform dot‑product over the current channel block
        # x_c shape: [B,N,N,blkC] ; w shape: [blkC,H]
        # use tl.dot on the last dim
        acc_left  += tl.dot(x_c, w_l)   # -> [B,N,N,H]
        acc_right += tl.dot(x_c, w_r)
        acc_lgate += tl.dot(x_c, w_lg)
        acc_rgate += tl.dot(x_c, w_rg)
        acc_ogate += tl.dot(x_c, w_og)

    # ------------------------------------------------------------------
    # Apply mask and sigmoid gates
    # ------------------------------------------------------------------
    left  = acc_left * m * tl.sigmoid(acc_lgate)
    right = acc_right * m * tl.sigmoid(acc_rgate)
    out_g = tl.sigmoid(acc_ogate)

    # ------------------------------------------------------------------
    # Store results
    # ------------------------------------------------------------------
    left_ptrs   = (left_out_ptr   + (offs_b[:, None, None, None] * N * N * H)
                                 + (offs_i[None, :, None, None] * N * H)
                                 + (offs_j[None, None, :, None] * H)
                                 + tl.arange(0, H)[None, None, None, :])
    right_ptrs  = (right_out_ptr  + (offs_b[:, None, None, None] * N * N * H)
                                 + (offs_i[None, :, None, None] * N * H)
                                 + (offs_j[None, None, :, None] * H)
                                 + tl.arange(0, H)[None, None, None, :])
    outg_ptrs   = (out_gate_out_ptr + (offs_b[:, None, None, None] * N * N * H)
                                 + (offs_i[None, :, None, None] * N * H)
                                 + (offs_j[None, None, :, None] * H)
                                 + tl.arange(0, H)[None, None, None, :])

    tl.store(left_ptrs,  left,  mask=mask_b[:, None, None, None] &
                                 mask_i[None, :, None, None] &
                                 mask_j[None, None, :, None] &
                                 (tl.arange(0, H) < H)[None, None, None, :])
    tl.store(right_ptrs, right, mask=mask_b[:, None, None, None] &
                                 mask_i[None, :, None, None] &
                                 mask_j[None, None, :, None] &
                                 (tl.arange(0, H) < H)[None, None, None, :])
    tl.store(outg_ptrs,  out_g, mask=mask_b[:, None, None, None] &
                                 mask_i[None, :, None, None] &
                                 mask_j[None, None, :, None] &
                                 (tl.arange(0, H) < H)[None, None, None, :])

# ----------------------------------------------------------------------
# Host side helper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the TriMul “outgoing” operator.
    Only the element‑wise mask / projection / gating stage is executed
    on the GPU via Triton; the N³ reduction is performed with a
    batched matmul (torch.einsum) which already runs at near‑optimal speed
    on Hopper GPUs.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dim (dim) – PyTorch (fast)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias   = weights["norm.bias"].to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), norm_weight, norm_bias)

    # cast to FP16 for the kernel (H200 has excellent FP16 throughput)
    x_fp16   = x.to(torch.float16)
    mask_fp16= mask.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Allocate buffers for the three fused tensors
    # ------------------------------------------------------------------
    left  = torch.empty((bs, N, N, hidden), dtype=torch.float16, device=device)
    right = torch.empty((bs, N, N, hidden), dtype=torch.float16, device=device)
    out_gate = torch.empty((bs, N, N, hidden), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 3. Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_B = 1          # batch size is tiny (≤2)
    BLOCK_N = 64         # tile size in sequence dimension
    BLOCK_C = 32         # channel tile for the inner reduction (C ≤ 384)

    grid = (
        triton.cdiv(bs, BLOCK_B) *
        triton.cdiv(N, BLOCK_N) *
        triton.cdiv(N, BLOCK_N),
    )
    apply_mask_and_gate_kernel[grid](
        x_fp16,
        mask_fp16,
        weights["left_proj.weight"].to(torch.float16).contiguous(),
        weights["right_proj.weight"].to(torch.float16).contiguous(),
        weights["left_gate.weight"].to(torch.float16).contiguous(),
        weights["right_gate.weight"].to(torch.float16).contiguous(),
        weights["out_gate.weight"].to(torch.float16).contiguous(),
        left,
        right,
        out_gate,
        bs, N, dim, hidden,
        BLOCK_B=BLOCK_B,
        BLOCK_N=BLOCK_N,
        BLOCK_C=BLOCK_C,
    )

    # ------------------------------------------------------------------
    # 4. N³ reduction: batched matmul over the K dimension for every d
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H) -> (B, N, N, H) ; we can use einsum
    out = torch.einsum('bikd,bjkd->bijd', left, right)          # FP16 result

    # ------------------------------------------------------------------
    # 5. to_out_norm, out_gate multiplication and final linear projection
    # ------------------------------------------------------------------
    out = out.to(torch.float32)   # convert back to FP32 for LayerNorm

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32)
    out = F.layer_norm(out, (hidden,), to_out_norm_w, to_out_norm_b)

    out_gate_fp32 = out_gate.to(torch.float32)
    out = out * out_gate_fp32

    # final linear projection (hidden -> dim)
    to_out_w = weights["to_out.weight"].to(torch.float32)   # shape (hidden, dim)
    out = torch.matmul(out, to_out_w)                      # [..., dim]

    return out
