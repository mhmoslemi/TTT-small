"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the tensor contraction

    out[b,i,j,d] = Σ_k  left[b,i,k,d] * right[b,j,k,d]

where
    left  = (x·W_left )  * sigmoid(x·W_left_gate )
    right = (x·W_right)  * sigmoid(x·W_right_gate)

We keep the cheap element‑wise work (layer‑norms, linear layers, gates and the
final linear projection) in PyTorch and move the N³ reduction to a custom
Triton kernel.  The hidden dimension (`hidden_dim`) is processed in
small chunks (`CHUNK`) so that the kernel fits into registers while still
exploiting the massive parallelism of the H200.

The kernel works on a 3‑D block (i‑block × j‑block) for a single batch and a
single hidden‑chunk.  For each (i,j) tile it accumulates over the k dimension
in a blocked‑gemm fashion:

    acc[i_tile, j_tile, h] += dot( left[i_tile, k_tile, h],
                                 right[j_tile, k_tile, h]ᵀ )

`tl.dot` is used for every hidden channel (`h`) inside a compile‑time
`tl.static_range` loop, which is unrolled by the compiler.  After the
reduction the result is stored back to the output tensor.

The final output is brought back to float32, layer‑normed, gated and projected
to the original `dim`.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# --------------------------------------------------------------
# Triton kernel: one hidden‑chunk of size BLOCK_H (compile‑time)
# --------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # [B, N, K, H]  (K == N)
    right_ptr,         # [B, N, K, H]
    out_ptr,           # [B, N, N, H]
    B, N, H,           # runtime sizes
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Block‑wise N³ reduction for one hidden‑chunk."""
    pid = tl.program_id(0)
    # decode linear pid -> (b, m, n)
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    b = pid // (num_m * num_n)
    tmp = pid % (num_m * num_n)
    m = tmp // num_n
    n = tmp % num_n

    # tile coordinates
    i = m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = n * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_i = i < N
    mask_j = j < N

    # accumulator: (BLOCK_M, BLOCK_N, BLOCK_H)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # iterate over K dimension in blocks
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = kk * BLOCK_K + k
        mask_k = k_off < N

        # ---- load left[b, i, k, h] ---------------------------------
        left_ptrs = (left_ptr
                     + b * stride_lb
                     + i[:, None, None] * stride_li
                     + k_off[None, :, None] * stride_lk
                     + tl.arange(0, BLOCK_H)[None, None, :] * stride_lh)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None],
                       other=0.0)

        # ---- load right[b, j, k, h] --------------------------------
        right_ptrs = (right_ptr
                      + b * stride_rb
                      + j[:, None, None] * stride_rj
                      + k_off[None, :, None] * stride_rk
                      + tl.arange(0, BLOCK_H)[None, None, :] * stride_rh)
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] & mask_k[None, :, None],
                        other=0.0)

        # ---- multiply‑accumulate over k for each hidden channel -----
        # left : (M, K, H)   right : (N, K, H)
        # we need Σ_k left[i,k,h] * right[j,k,h]  →  dot(A, Bᵀ)
        for h in tl.static_range(0, BLOCK_H):
            a = left[:, :, h]          # (M, K)
            b_mat = right[:, :, h]     # (N, K)
            # tl.dot expects (M,K) × (K,N) → (M,N); set trans_b=True
            acc[:, :, h] += tl.dot(a, b_mat, trans_b=True)

    # ---- store ----------------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_ob
                + i[:, None, None] * stride_oi
                + j[None, :, None] * stride_oj
                + tl.arange(0, BLOCK_H)[None, None, :] * stride_oh)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None])


# --------------------------------------------------------------
# Host entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]        (float32)
        - mask         : torch.Tensor [B, N, N] (may be all‑ones)
        - weights_dict : mapping of all weight tensors (float32)
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # cheap element‑wise sub‑graph (stay in PyTorch)
    # ------------------------------------------------------------------
    eps = 1e-5
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # apply mask if present
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton reduction: out = Σ_k left[...,k,:] * right[...,k,:]
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden), device=x.device, dtype=torch.float32)

    # kernel launch parameters
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    CHUNK = 32                     # hidden dim processed per kernel launch

    for h_start in range(0, hidden, CHUNK):
        h_end = min(h_start + CHUNK, hidden)
        cur_H = h_end - h_start

        # contiguous slices for the current hidden chunk
        left_chunk = left[..., h_start:h_end].contiguous()
        right_chunk = right[..., h_start:h_end].contiguous()
        out_chunk = out[..., h_start:h_end].contiguous()

        # strides as plain integers (required by Triton)
        s_lb, s_li, s_lk, s_lh = left_chunk.stride()
        s_rb, s_rj, s_rk, s_rh = right_chunk.stride()
        s_ob, s_oi, s_oj, s_oh = out_chunk.stride()

        grid = (triton.cdiv(N, BLOCK_M) *
                triton.cdiv(N, BLOCK_N) *
                B,)

        trimul_kernel[grid](
            left_chunk,
            right_chunk,
            out_chunk,
            B, N, cur_H,
            s_lb, s_li, s_lk, s_lh,
            s_rb, s_rj, s_rk, s_rh,
            s_ob, s_oi, s_oj, s_oh,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_H=cur_H,
        )

    # ------------------------------------------------------------------
    # post‑processing (still in PyTorch)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
