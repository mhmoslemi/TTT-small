"""
TriMul “outgoing” kernel (AlphaFold3 style)

The heavy part of the forward pass is the 4‑D einsum
    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
where `left`/`right` have shape [B, N, N, H] (H = hidden_dim).

For a fixed channel *d* this is a batched matrix‑multiply:
    out_d = left_d @ right_d.transpose(-2, -1)          # shape [B, N, N]

The kernel below performs this multiplication for all batches *B* and all
hidden channels *H* in a single launch.  The rest of the TriMul block
(LayerNorms, linear projections, gating, final LayerNorm + linear) is done
in regular PyTorch to keep the implementation simple while still moving the
N³ work to Triton.

The kernel works with fp16 inputs (the model weights are fp16/float32) and
accumulates in fp32 for numerical stability.  The final output is cast back
to fp32 as required by the specification.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul for many channels (H) at once
# ----------------------------------------------------------------------
@triton.jit
def triton_trimul_kernel(
    # Pointers
    left_ptr,          # left  [B, N, N, H]
    right_ptr,         # right [B, N, N, H]
    out_ptr,           # out   [B, N, N, H]

    # Strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,

    B, N, H,                         # static sizes
    BLOCK_M: tl.constexpr,           # tile rows   (i)
    BLOCK_N: tl.constexpr,           # tile cols   (j)
    BLOCK_K: tl.constexpr,           # reduction tile (k)
    GROUP_SIZE_M: tl.constexpr,      # for better cache reuse
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1)  compute 2‑D grid of (i, j) tiles for each (batch, hidden) combo
    # --------------------------------------------------------------
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # each program works on a (batch, hidden) pair
    pid_batch = pid // (B * H)
    pid_rest  = pid %  (B * H)
    batch_id = pid_batch
    h_id     = pid_rest

    # 2‑D tile coordinates inside the (i, j) matrix
    pid_m = (pid % (num_pid_m * num_pid_n)) // num_pid_n
    pid_n = (pid % (num_pid_m * num_pid_n)) % num_pid_n

    # compute start offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds tiles
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers for the current (batch, hidden) slice
    left_base  = left_ptr  + batch_id * stride_l_b + h_id * stride_l_h
    right_base = right_ptr + batch_id * stride_r_b + h_id * stride_r_h
    out_base   = out_ptr   + batch_id * stride_o_b + h_id * stride_o_h

    # --------------------------------------------------------------
    # 3)   Accumulate over k
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for start_k in range(0, N, BLOCK_K):
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A: left[..., i, k, h]
        a_ptrs = left_base + \
                 (offs_i[:, None] * stride_l_i) + \
                 (offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # load B: right[..., j, k, h]   note the k‑dimension is the same
        b_ptrs = right_base + \
                 (offs_j[:, None] * stride_r_i) + \
                 (offs_k[None, :] * stride_r_k)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        b = b.to(tl.float32)

        # multiply‑accumulate
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # 4)   Write results
    # --------------------------------------------------------------
    out_ptrs = out_base + \
               (offs_i[:, None] * stride_o_i) + \
               (offs_j[None, :] * stride_o_j)

    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper to launch the kernel
# ----------------------------------------------------------------------
def batched_matmul_trimul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, H]  (fp16 or fp32)
    returns       : [B, N, N, H]  (fp32)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape

    # ensure contiguous and fp16 for kernel (fp32 will be cast)
    left  = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    out   = torch.empty_like(left, dtype=torch.float16, device=left.device)

    # strides in elements
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tile sizes – chosen for Hopper (SM90) – 128‑128 tiles work well
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # small enough to fit in shared memory
    GROUP_SIZE_M = 8

    total_programs = B * H                # one program per (batch, hidden)
    # 2‑D grid for (i,j) tiles is folded into program id via GROUP_SIZE_M
    # Triton expects a 1‑D grid, so we flatten everything:
    grid = (triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N) * total_programs,)

    triton_trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_h=stride_l_h,
        stride_r_b=stride_r_b,
        stride_r_i=stride_r_i,
        stride_r_k=stride_r_k,
        stride_r_h=stride_r_h,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_h=stride_o_h,
        B=B,
        N=N,
        H=H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
        num_warps=4,
    )
    # Cast back to fp32 for the rest of the pipeline
    return out.to(torch.float32)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    Only the N³ accumulation (einsum) is done in Triton; everything else
    stays in PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = torch.float32

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dim
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(dtype)
    norm_b = weights["norm.bias"].to(dtype)
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_w + norm_b

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(dtype)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(dtype)

    # x: [B,N,N,dim] -> reshape for linear matmul
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)

    left = torch.nn.functional.linear(x_flat, left_proj_w).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, right_proj_w).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3) Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)        # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(dtype)
    right_gate_w = weights["right_gate.weight"].to(dtype)
    out_gate_w   = weights["out_gate.weight"].to(dtype)

    left_gate  = torch.nn.functional.linear(x_flat, left_gate_w).view(B, N, N, hidden_dim).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, right_gate_w).view(B, N, N, hidden_dim).sigmoid()
    out_gate   = torch.nn.functional.linear(x_flat, out_gate_w).view(B, N, N, hidden_dim).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core N³ multiplication – Triton kernel
    # ------------------------------------------------------------------
    out = batched_matmul_trimul(left, right)      # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + gate + final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)
    to_out_w      = weights["to_out.weight"].to(dtype)

    # LayerNorm over hidden_dim
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate

    # Final linear (hidden_dim -> dim)
    out_flat = out.reshape(-1, hidden_dim)
    out = torch.nn.functional.linear(out_flat, to_out_w).view(B, N, N, dim)

    return out.to(torch.float32)
