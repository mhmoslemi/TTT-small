"""
TriMul (outgoing) – fast Triton implementation
================================================

The forward pass consists of

1. Layer‑norm on the 4‑D input   [B, N, N, D]
2. Linear projections  →  left, right  (shape [B, N, N, H])
3. Gated masking
4. 3‑D “tri‑multiplication”

        out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]

   which is a batched matrix‑multiply for every hidden channel *h*:
   left[b, :, :, h]  @  right[b, :, :, h].T

   The expensive N³ work is performed by a custom Triton kernel that
   folds the hidden dimension *H* into the batch dimension, i.e.
   we treat the data as a batch of ``B*H`` matrices of size ``N×N``.

5. Layer‑norm → gate → final linear projection   (back to D)

The kernel works in fp16 (the model uses fp32 weights, we cast to fp16
for the mat‑mul, then cast the result back to fp32) and is tuned for
N up to 1024 on an NVIDIA H200 (sm_90).  No gradients are needed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (C = A @ B)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,      # pointers
    M, N, K,                  # matrix sizes
    stride_am, stride_ak, stride_an,   # strides for A (batch, m, k)
    stride_bk, stride_bn, stride_bb,   # strides for B (batch, k, n)
    stride_cm, stride_cn, stride_cc,   # strides for C (batch, m, n)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Computes C = A @ B for many independent matrices in a single launch.

    A : (batch, M, K)   (row‑major)
    B : (batch, K, N)   (row‑major)
    C : (batch, M, N)   (row‑major)
    """

    pid = tl.program_id(0)                     # linear index over (batch, tile_m, tile_n)
    batch = tl.program_id(1)

    # tile coordinates
    num_m_tiles = tl.cdiv(M, BLOCK_M)
    num_n_tiles = tl.cdiv(N, BLOCK_N)
    tile_m = (pid // num_n_tiles) % num_m_tiles
    tile_n = pid % num_n_tiles

    # offsets inside the tile
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the tile
    a_tile_ptr = a_ptr + batch * stride_am + offs_m[:, None] * stride_ak + offs_k[None, :] * stride_an
    b_tile_ptr = b_ptr + batch * stride_bb + offs_k[:, None] * stride_bn + offs_n[None, :] * stride_bb

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K in BLOCK_K steps
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load a block of A and B (fp16)
        a = tl.load(a_tile_ptr, mask=mask_m[:, None] & (offs_k < K), other=0.0).to(tl.float32)
        b = tl.load(b_tile_ptr, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float32)

        # dot‑product
        acc += tl.dot(a, b)

        # advance pointers
        a_tile_ptr += BLOCK_K * stride_an
        b_tile_ptr += BLOCK_K * stride_bn

    # write back
    c_tile_ptr = c_ptr + batch * stride_cm + offs_m[:, None] * stride_cn + offs_n[None, :] * stride_cc
    tl.store(c_tile_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper to launch the kernel
# ----------------------------------------------------------------------
def batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a : (B*H, N, N)   fp16
    b : (B*H, N, N)   fp16   (already transposed on the last two dims)
    returns c : (B*H, N, N) fp16
    """
    assert a.is_cuda and b.is_cuda
    BxH, N, _ = a.shape
    dtype = a.dtype

    c = torch.empty_like(a)

    # strides (in elements, not bytes)
    stride_am, stride_ak, stride_an = a.stride()
    stride_bk, stride_bn, stride_bb = b.stride()
    stride_cm, stride_cn, stride_cc = c.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # grid: one 1‑D grid over (M,N) tiles, plus a second dimension for batch
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_tiles = grid_m * grid_n
    grid = (total_tiles, BxH)

    batched_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_an=stride_an,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_bb=stride_bb,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_cc=stride_cc,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return c


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator with a Triton‑accelerated
    tri‑multiplication.

    Args
    ----
    data : (input, mask, weights, config)
        input   – Tensor [B, N, N, D]          (float32, on CUDA)
        mask    – Tensor [B, N, N] (optional) (float32 or bool, on CUDA)
        weights – dict with the exact keys used in the reference module
        config  – dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    Tensor of shape [B, N, N, D] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the last dimension (per‑pair channel)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # --------------------------------------------------------------
    # 2. Linear projections (no bias, keep fp32 weights)
    # --------------------------------------------------------------
    # shape after proj: [B, N, N, H]
    left = F.linear(x, weights["left_proj.weight"])          # (D, H)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Mask (if present) – broadcast on hidden dim
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Tri‑multiplication (N³).  We fold the hidden dimension into the batch.
    # --------------------------------------------------------------
    # cast to fp16 for the mat‑mul kernel (precision is sufficient)
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # transpose right on the last two axes   (j,k) -> (k,j)
    right_f16_T = right_f16.transpose(2, 1).contiguous()

    # reshape to (B*H, N, N)
    left_flat = left_f16.reshape(B * H, N, N)
    right_flat = right_f16_T.reshape(B * H, N, N)

    # run the Triton batched mat‑mul
    out_flat = batched_matmul(left_flat, right_flat)      # fp16

    # reshape back → [B, N, N, H]  (still fp16)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6. Output norm, gating and final linear projection (back to D)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out.to(torch.float32),               # back to fp32 for stable norm
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
