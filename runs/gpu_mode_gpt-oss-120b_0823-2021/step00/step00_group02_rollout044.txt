"""
TriMul “outgoing” kernel (AlphaFold3) – Triton + PyTorch hybrid.

The heavy N³ work is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We keep the lightweight LayerNorm / Linear / sigmoid work in PyTorch and
implement the above reduction in a custom Triton kernel.
The kernel computes a batched block‑matrix‑multiply for every hidden
dimension d (the hidden_dim is usually 128).  The final result is converted
back to float32.

Only the core multiply‑and‑reduce is in Triton – this satisfies the
requirement “at least part of the operations in a kernel” while keeping
the implementation simple and correct for all test shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # *[B, N, N, H]   left tensor (float16)
    right_ptr,         # *[B, N, N, H]   right tensor (float16)
    out_ptr,           # *[B, N, N, H]   output (float32)
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    Grid dimensions: (num_i, num_j, B)
    Each program works on a BLOCK_M × BLOCK_N tile of (i,j) for a given batch b.
    The hidden dimension H is processed in chunks of BLOCK_H.
    """
    pid_i = tl.program_id(0)            # tile over i
    pid_j = tl.program_id(1)            # tile over j
    pid_b = tl.program_id(2)            # batch

    # start indices of the tile
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N
    b_offset = pid_b

    # offsets within the hidden dimension (processed in a loop)
    out = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # global k index
        k_offset = k * BLOCK_K

        # ------------------------------------------------------------------- #
        # Load left  : shape (BLOCK_M, BLOCK_K, BLOCK_H)
        # Load right : shape (BLOCK_N, BLOCK_K, BLOCK_H)
        # ------------------------------------------------------------------- #
        offs_i = i_offset + tl.arange(0, BLOCK_M)
        offs_j = j_offset + tl.arange(0, BLOCK_N)
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        offs_h = tl.arange(0, BLOCK_H)

        # mask for out‑of‑bounds in i/j/k/h
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N
        mask_h = offs_h < H

        # left[i,k,d]
        left_ptrs = left_ptr \
            + b_offset * stride_left_b \
            + offs_i[:, None, None] * stride_left_i \
            + offs_k[None, :, None] * stride_left_k \
            + offs_h[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0)

        # right[j,k,d]
        right_ptrs = right_ptr \
            + b_offset * stride_right_b \
            + offs_j[:, None, None] * stride_right_i \
            + offs_k[None, :, None] * stride_right_k \
            + offs_h[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0)

        # ------------------------------------------------------------------- #
        # Multiply‑accumulate over k
        # out[i,j,d] += Σ_k left[i,k,d] * right[j,k,d]
        # ------------------------------------------------------------------- #
        # (BLOCK_M, BLOCK_K, BLOCK_H) * (BLOCK_N, BLOCK_K, BLOCK_H)  → (BLOCK_M,BLOCK_N,BLOCK_H)
        # We perform a dot product over the K dimension for each hidden slice.
        for dh in range(0, BLOCK_H):
            # slice at hidden index dh
            l = left[:, :, dh]          # (M, K)
            r = right[:, :, dh]         # (N, K)
            # accumulate dot product across K
            out[:, :, dh] += tl.dot(l, r)   # (M, N)

    # ----------------------------------------------------------------------- #
    # Store the result (float32)
    # ----------------------------------------------------------------------- #
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    out_ptrs = out_ptr \
        + b_offset * stride_out_b \
        + offs_i[:, None, None] * stride_out_i \
        + offs_j[None, :, None] * stride_out_j \
        + offs_h[None, None, :] * stride_out_h

    tl.store(out_ptrs, out, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])


# --------------------------------------------------------------------------- #
# Helper for launching the kernel
# --------------------------------------------------------------------------- #
def launch_trimul(out, left, right, block_m=32, block_n=32, block_k=64, block_h=8):
    """
    left/right : [B, N, N, H] float16
    out        : [B, N, N, H] float32 (pre‑allocated)
    """
    B, N, _, H = left.shape

    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    grid = (
        triton.cdiv(N, block_m),   # i tiles
        triton.cdiv(N, block_n),   # j tiles
        B,                         # batch
    )
    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        BLOCK_H=block_h,
        num_warps=4,
    )
    return out


# --------------------------------------------------------------------------- #
# Full custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor : shape [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------- #
    # 1. LayerNorm on the input (float32)
    # --------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B,N,N,dim]  float32

    # --------------------------------------------------------------- #
    # 2. Linear projections (no bias) – keep in float16 for speed
    # --------------------------------------------------------------- #
    # left_proj(x) = x @ W_left.T
    left_proj_weight = weights["left_proj.weight"]          # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]        # [hidden_dim, dim]

    # reshape to (B*N*N, dim) for a single matmul
    x_flat = x.view(-1, dim).to(torch.float16)               # fp16

    left = F.linear(x_flat, left_proj_weight.to(torch.float16))   # (B*N*N, hidden_dim)
    right = F.linear(x_flat, right_proj_weight.to(torch.float16))

    left = left.view(bs, N, N, hidden_dim)   # [B,N,N,hidden_dim] fp16
    right = right.view(bs, N, N, hidden_dim)

    # --------------------------------------------------------------- #
    # 3. Mask (optional)
    # --------------------------------------------------------------- #
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------- #
    # 4. Gating (sigmoid) – still fp16
    # --------------------------------------------------------------- #
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x_flat, left_gate_w.to(torch.float16))).view(bs, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w.to(torch.float16))).view(bs, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w.to(torch.float16))).view(bs, N, N, hidden_dim)

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # 5. Core TriMul reduction (Triton)
    # --------------------------------------------------------------- #
    out = torch.empty_like(left, dtype=torch.float32)  # [B,N,N,hidden_dim]
    launch_trimul(out, left, right,
                  block_m=32, block_n=32, block_k=64, block_h=8)

    # --------------------------------------------------------------- #
    # 6. Post‑processing: to_out_norm -> out_gate -> to_out
    # --------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # final linear projection back to dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out.view(-1, hidden_dim), to_out_w.to(torch.float32))
    out = out.view(bs, N, N, dim)

    return out
