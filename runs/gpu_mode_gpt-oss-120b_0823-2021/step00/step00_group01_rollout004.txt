"""
TriMul “outgoing” implementation.

The heavy part of the module is the ternary contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]          (1)

where `left` and `right` have shape (B,N,N,H) with H = hidden_dim.
Equation (1) is an N×N matrix‑multiply for every feature dimension d.
We fuse this contraction in a single Triton kernel:

* The kernel is launched with a 3‑D grid (batch, i‑tile, j‑tile).
* Each program computes a BLOCK_M × BLOCK_N tile of the output.
* The reduction over k is tiled with BLOCK_K and accumulated in fp32.
* Inside the k‑tile we load a (BLOCK_M, BLOCK_K, H) block of `left`
  and a (BLOCK_N, BLOCK_K, H) block of `right`.  For every kk in the
  k‑tile we compute an outer product of the two vectors of length H
  with `tl.dot(..., trans_b=True)` which yields a (BLOCK_M, BLOCK_N)
  contribution that is added to the accumulator.
* The kernel writes the result in fp16 and the surrounding Python code
  finishes the forward pass (LayerNorms, linear projections, gates)
  using regular PyTorch for simplicity.

The final output is returned in float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel for the N³ contraction (outgoing direction)
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,                # *const float*  [B,N,N,H]
    right_ptr,               # *const float*  [B,N,N,H]
    out_ptr,                 # *float*        [B,N,N,H]
    B, N, H,
    stride_lb, stride_lk, stride_lh,   # left strides  (b,i,k,d)
    stride_rb, stride_rk, stride_rh,   # right strides (b,j,k,d)
    stride_ob, stride_oi, stride_oj,   # out strides   (b,i,j,d)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # tile over i
    pid_j = tl.program_id(2)          # tile over j

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in tiles
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kb * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for the current k‑tile
        # left:   (i,k,d)   right: (j,k,d)
        left_tile_ptr = left_ptr + (
            pid_b * stride_lb
            + offs_i[:, None] * stride_lk
            + offs_k[None, :] * stride_lh
        )
        right_tile_ptr = right_ptr + (
            pid_b * stride_rb
            + offs_j[:, None] * stride_rk
            + offs_k[None, :] * stride_rh
        )

        # inner reduction over the k‑tile
        for kk in range(0, BLOCK_K):
            k_valid = (k_base + kk) < N
            if not k_valid:
                continue

            # (BLOCK_M, H) and (BLOCK_N, H)
            left_vec  = tl.load(left_tile_ptr[:, kk, :],
                                mask=mask_i[:, None] & k_valid,
                                other=0.0)
            right_vec = tl.load(right_tile_ptr[:, kk, :],
                                mask=mask_j[:, None] & k_valid,
                                other=0.0)

            # outer product over the feature dimension H.
            # tl.dot with trans_b=True computes left_vec @ right_vec.T  -> (M,N)
            acc += tl.dot(left_vec, right_vec, trans_b=True)

    # write back the tile
    out_tile_ptr = out_ptr + (
        pid_b * stride_ob
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
    )
    tl.store(out_tile_ptr, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper to launch the Triton kernel
# ----------------------------------------------------------------------
def trimul_outgoing_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : (B,N,N,H)  float32
    returns out  : (B,N,N,H)  float32
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float16, device=left.device)

    # strides in elements (not bytes)
    stride_lb, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_lb, stride_lk, stride_lh,
        stride_rb, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                 # good default for H200
    )

    return out.to(torch.float32)


# ----------------------------------------------------------------------
# The required entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    data = (input, mask, weights, config)
        input : (B, N, N, dim)      float32
        mask  : (B, N, N)           float32 (0/1) – may be all‑ones
        weights: dict of all linear / norm parameters (float32)
        config : {"dim": int, "hidden_dim": int}
    Returns
        output : (B, N, N, dim)    float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    inp = F.layer_norm(inp, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]    # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]
    left  = F.linear(inp, left_proj_w)   # (B,N,N,hidden)
    right = F.linear(inp, right_proj_w)

    # ------------------------------------------------------------------
    # Apply mask (if any).  mask is broadcasted on the feature dim.
    # ------------------------------------------------------------------
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1)            # (B,N,N,1)
        left  = left * mask_unsq
        right = right * mask_unsq

    # ------------------------------------------------------------------
    # Gates
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(inp, left_gate_w))
    right_gate = torch.sigmoid(F.linear(inp, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(inp, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton contraction (the expensive N³ part)
    # ------------------------------------------------------------------
    out = trimul_outgoing_torch(left, right)   # (B,N,N,hidden)  float32

    # ------------------------------------------------------------------
    # to_out_norm + gating + final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    return out
