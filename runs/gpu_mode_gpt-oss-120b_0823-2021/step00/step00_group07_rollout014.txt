"""
TriMul “outgoing” kernel (AlphaFold‑3).

The heavy part of the forward pass is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is an N³ reduction.  The surrounding LayerNorm / Linear /
sigmoid / gating steps are cheap and are kept in PyTorch.
The kernel below treats each (batch, hidden‑dim) pair as an independent
matrix‑multiply of shape (N,N) × (N,N)⁽ᵀ⁾ for every hidden channel *d*.
The computation is tiled in three dimensions (M‑tile, N‑tile, K‑tile)
and accumulates in FP32.  The kernel is launched with a 3‑D grid:

    pid0 : batch * hidden_dim + d          (combined outer dimension)
    pid1 : tile over the i‑axis (rows)
    pid2 : tile over the j‑axis (cols)

The result is written back as FP32, matching the reference implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel that performs the core N³ reduction.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    out_ptr,               # [B, N, N, D]  output
    left_ptr,              # [B, N, N, D]  left  (already masked & gated)
    right_ptr,             # [B, N, N, D]  right (already masked & gated)
    B, N, D,
    # strides of left tensor
    stride_lb, stride_li, stride_lk, stride_ld,
    # strides of right tensor
    stride_rb, stride_rj, stride_rk, stride_rd,
    # strides of out tensor
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute for a fixed (batch, d) a tiled matrix‑multiply:
        out[i, j] = Σ_k left[i, k] * right[j, k]   (all in the same channel d)
    """
    pid0 = tl.program_id(0)                # combined batch*D index
    pid1 = tl.program_id(1)                # tile over i (rows)
    pid2 = tl.program_id(2)                # tile over j (cols)

    b = pid0 // D
    d = pid0 % D

    # Offsets for the i‑ and j‑tiles
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to the first tile of left/right for this (b,d)
    left_base = left_ptr + b * stride_lb + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd
    out_base = out_ptr + b * stride_ob + d * stride_od

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k‑tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left[i,k]   shape (M,K)
        left_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right[j,k]  shape (N,K) – note the j‑axis is the first dim
        right_ptrs = right_base + offs_j[:, None] * stride_rj + offs_k[None, :] * stride_rk
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Inner product over K
        acc += tl.dot(left, right)          # (M,K)·(N,K)ᵀ → (M,N)

    # Write back
    out_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper to launch the kernel
# ----------------------------------------------------------------------
def trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : float32 tensors of shape [B, N, N, D]
    returns      : float32 tensor of same shape
    """
    B, N, _, D = left.shape
    out = torch.empty_like(left)

    # stride extraction (torch tensors are contiguous in the last dimension)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # tile sizes – tuned for Hopper (SM90) 16‑bit compute units
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32   # a multiple of 32 works well for WMMA‑friendly loads

    grid = (
        B * D,                                     # pid0
        triton.cdiv(N, BLOCK_M),                  # pid1
        triton.cdiv(N, BLOCK_N),                  # pid2
    )

    trimul_outgoing_kernel[grid](
        out,
        left,
        right,
        B, N, D,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # Hopper handles 4‑warp blocks nicely
    )
    return out


# ----------------------------------------------------------------------
# Full forward implementation that mixes PyTorch + Triton
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : [B, N, N, dim]  torch.float32  (on CUDA)
            mask_tensor  : [B, N, N]       torch.float32  (1.0 / 0.0)
            weights_dict : mapping of weight names to tensors
            config_dict  : {"dim": dim, "hidden_dim": hidden_dim, "nomask": bool}
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) for left, right and three gates
    # --------------------------------------------------------------
    # shape: [B,N,N,hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # apply mask if present (mask shape [B,N,N] -> broadcast)
    if not nomask:
        mask = mask_tensor.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 3. Core N³ reduction via Triton
    # --------------------------------------------------------------
    # Keep tensors in float32 for the kernel (the kernel expects fp32)
    # If desired, they could be cast to fp16 for higher throughput,
    # but the final spec requires float32 output.
    out = trimul_outgoing_triton(left, right)   # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 4. Output LayerNorm + gating + final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                       # broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])   # back to dim

    return out
