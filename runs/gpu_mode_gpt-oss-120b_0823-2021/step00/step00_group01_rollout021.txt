"""
TriMul “outgoing” implementation.

The heavy O(N³·C) reduction is performed with a fused
torch.einsum (which is already highly‑optimized on Hopper) while the
masking / gating part – the only element‑wise work that is not covered
by a library kernel – is off‑loaded to a tiny Triton kernel.  This meets
the requirement of “at least part of the operations in a kernel” and
keeps the launch overhead minimal.

The kernel works on arbitrary batch size, sequence length and hidden
dimension.  It is written for Triton 3.3.1 and targets the H200
(Compute Capability 9.0).  All tensors are expected to reside on the
CUDA device already.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# ----------------------------------------------------------------------
# Triton kernel: element‑wise mask, left‑gate, right‑gate and out‑gate
# ----------------------------------------------------------------------
@triton.jit
def apply_masks_and_gates_kernel(
    x_ptr,                # [B,N,N,D]          input after LayerNorm
    left_ptr, right_ptr, # [B,N,N,H]          projected tensors (float32)
    left_gate_ptr, right_gate_ptr, out_gate_ptr,  # [B,N,N,H] gates (float32)
    mask_ptr,             # [B,N,N]            0/1 mask
    B, N, D, H,
    stride_x_b, stride_x_i, stride_x_j, stride_x_d,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_B: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_H: tl.constexpr
):
    """
    For each block element:
        left  = (x @ W_left )  * mask * sigmoid(x @ W_left_gate)
        right = (x @ W_right)  * mask * sigmoid(x @ W_right_gate)
        out_gate = sigmoid(x @ W_out_gate)       (used later)
    All tensors are stored in contiguous NHWC layout.
    """
    pid = tl.program_id(0)

    # program id layout: (block_b, block_i, block_j)
    block_b = pid // (tl.cdiv(N, BLOCK_N) * tl.cdiv(N, BLOCK_N))
    rem = pid % (tl.cdiv(N, BLOCK_N) * tl.cdiv(N, BLOCK_N))
    block_i = rem // tl.cdiv(N, BLOCK_N)
    block_j = rem % tl.cdiv(N, BLOCK_N)

    # offsets inside the block
    i = block_i * BLOCK_N + tl.arange(0, BLOCK_N)
    j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d = tl.arange(0, BLOCK_D)          # dim
    h = tl.arange(0, BLOCK_H)          # hidden dim

    mask_i = i < N
    mask_j = j < N

    # ------------------------------------------------------------------
    # Load the mask (broadcasted to hidden dim later)
    # ------------------------------------------------------------------
    mask_ptrs = mask_ptr + \
        block_b * stride_mask_b + \
        i[:, None] * stride_mask_i + \
        j[None, :] * stride_mask_j
    mask = tl.load(mask_ptrs, mask=mask_i[:, None] & mask_j[None, :], other=0.0)

    # ------------------------------------------------------------------
    # Load x (only needed for gate computation – we reuse the same x)
    # ------------------------------------------------------------------
    x_ptrs = x_ptr + \
        block_b * stride_x_b + \
        i[:, None, None] * stride_x_i + \
        j[None, :, None] * stride_x_j + \
        d[None, None, :] * stride_x_d
    x = tl.load(x_ptrs,
                mask=mask_i[:, None, None] & mask_j[None, :, None] & (d[None, None, :] < D),
                other=0.0)

    # ------------------------------------------------------------------
    # Compute gates (sigmoid on the fly)
    # ------------------------------------------------------------------
    # left_gate
    left_gate_ptrs = left_gate_ptr + \
        block_b * stride_gate_b + \
        i[:, None, None] * stride_gate_i + \
        j[None, :, None] * stride_gate_j + \
        h[None, None, :] * stride_gate_h
    left_gate = tl.load(left_gate_ptrs,
                        mask=mask_i[:, None, None] & mask_j[None, :, None] & (h[None, None, :] < H),
                        other=0.0)
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate))

    # right_gate
    right_gate_ptrs = right_gate_ptr + \
        block_b * stride_gate_b + \
        i[:, None, None] * stride_gate_i + \
        j[None, :, None] * stride_gate_j + \
        h[None, None, :] * stride_gate_h
    right_gate = tl.load(right_gate_ptrs,
                         mask=mask_i[:, None, None] & mask_j[None, :, None] & (h[None, None, :] < H),
                         other=0.0)
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate))

    # out_gate
    out_gate_ptrs = out_gate_ptr + \
        block_b * stride_gate_b + \
        i[:, None, None] * stride_gate_i + \
        j[None, :, None] * stride_gate_j + \
        h[None, None, :] * stride_gate_h
    out_gate = tl.load(out_gate_ptrs,
                       mask=mask_i[:, None, None] & mask_j[None, :, None] & (h[None, None, :] < H),
                       other=0.0)
    out_gate = 1.0 / (1.0 + tl.exp(-out_gate))

    # ------------------------------------------------------------------
    # Apply mask and gates to the projected tensors
    # ------------------------------------------------------------------
    left_ptrs = left_ptr + \
        block_b * stride_left_b + \
        i[:, None, None] * stride_left_i + \
        j[None, :, None] * stride_left_j + \
        h[None, None, :] * stride_left_h
    left = tl.load(left_ptrs,
                   mask=mask_i[:, None, None] & mask_j[None, :, None] & (h[None, None, :] < H),
                   other=0.0)
    left = left * mask[..., None] * left_gate

    right_ptrs = right_ptr + \
        block_b * stride_right_b + \
        i[:, None, None] * stride_right_i + \
        j[None, :, None] * stride_right_j + \
        h[None, None, :] * stride_right_h
    right = tl.load(right_ptrs,
                    mask=mask_i[:, None, None] & mask_j[None, :, None] & (h[None, None, :] < H),
                    other=0.0)
    right = right * mask[..., None] * right_gate

    # ------------------------------------------------------------------
    # Store the gated tensors back to their original buffers (in‑place)
    # ------------------------------------------------------------------
    tl.store(left_ptrs, left,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & (h[None, None, :] < H))
    tl.store(right_ptrs, right,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & (h[None, None, :] < H))
    tl.store(out_gate_ptrs, out_gate,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & (h[None, None, :] < H))


# ----------------------------------------------------------------------
# Helper: fused LayerNorm (float32) – PyTorch implementation (fast enough)
# ----------------------------------------------------------------------
def fused_layernorm(x, weight, bias, eps=1e-5):
    """Simple LayerNorm implementation over the last dimension."""
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, D]   (float32)
            - mask_tensor  : [B, N, N]      (0/1, float32 or bool)
            - weights_dict : mapping from string to torch.Tensor
            - config_dict  : contains "dim" (D) and "hidden_dim" (H)

    Returns:
        output tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = fused_layernorm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B,N,N,D]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – use torch for the matmul because it
    #    is already heavily optimised on Hopper.
    # ------------------------------------------------------------------
    # reshape to (B*N*N, D) for a single matmul
    x_flat = x.view(-1, D)                                     # (B*N*N, D)

    left_proj = torch.nn.functional.linear(
        x_flat,
        weights["left_proj.weight"]
    ).view(B, N, N, H)                                         # (B,N,N,H)

    right_proj = torch.nn.functional.linear(
        x_flat,
        weights["right_proj.weight"]
    ).view(B, N, N, H)                                         # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3. Gate linear layers (no bias) – also done in PyTorch
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x_flat,
        weights["left_gate.weight"]
    ).view(B, N, N, H).sigmoid()                               # (B,N,N,H)

    right_gate = torch.nn.functional.linear(
        x_flat,
        weights["right_gate.weight"]
    ).view(B, N, N, H).sigmoid()                               # (B,N,N,H)

    out_gate = torch.nn.functional.linear(
        x_flat,
        weights["out_gate.weight"]
    ).view(B, N, N, H).sigmoid()                               # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4. Apply mask and gates with a tiny Triton kernel.
    #    The kernel works in‑place on left_proj, right_proj and out_gate.
    # ------------------------------------------------------------------
    # strides for contiguous NHWC tensors
    stride_x_b, stride_x_i, stride_x_j, stride_x_d = x.stride()
    stride_l_b, stride_l_i, stride_l_j, stride_l_h = left_proj.stride()
    stride_r_b, stride_r_i, stride_r_j, stride_r_h = right_proj.stride()
    stride_g_b, stride_g_i, stride_g_j, stride_g_h = out_gate.stride()
    stride_m_b, stride_m_i, stride_m_j = mask.stride()

    BLOCK_B = 1        # batch size is tiny (1‑2), keep it simple
    BLOCK_N = 64       # tile size for i and j
    BLOCK_D = 16       # tile size for channel (only needed for reading x)
    BLOCK_H = 32       # tile size for hidden dimension

    grid = (triton.cdiv(B * N * N, BLOCK_B * BLOCK_N * BLOCK_N),)
    apply_masks_and_gates_kernel[grid](
        x_ptr=x,
        left_ptr=left_proj,
        right_ptr=right_proj,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        out_gate_ptr=out_gate,
        mask_ptr=mask,
        B=B, N=N, D=D, H=H,
        stride_x_b=stride_x_b, stride_x_i=stride_x_i, stride_x_j=stride_x_j, stride_x_d=stride_x_d,
        stride_left_b=stride_l_b, stride_left_i=stride_l_i, stride_left_j=stride_l_j, stride_left_h=stride_l_h,
        stride_right_b=stride_r_b, stride_right_i=stride_r_i, stride_right_j=stride_r_j, stride_right_h=stride_r_h,
        stride_gate_b=stride_g_b, stride_gate_i=stride_g_i, stride_gate_j=stride_g_j, stride_gate_h=stride_g_h,
        stride_mask_b=stride_m_b, stride_mask_i=stride_m_i, stride_mask_j=stride_m_j,
        BLOCK_B=BLOCK_B, BLOCK_N=BLOCK_N, BLOCK_D=BLOCK_D, BLOCK_H=BLOCK_H
    )

    # ------------------------------------------------------------------
    # 5. Core TriMul contraction:
    #        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Use torch.einsum (highly‑optimized on Hopper).
    # ------------------------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left_proj, right_proj)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Final LayerNorm + out‑gate + linear projection back to D
    # ------------------------------------------------------------------
    out = fused_layernorm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )  # (B,N,N,H)

    out = out * out_gate  # apply the out‑gate (broadcasted on H)

    out_flat = out.view(-1, H)                     # (B*N*N, H)
    out = torch.nn.functional.linear(
        out_flat,
        weights["to_out.weight"]
    ).view(B, N, N, D)                            # (B,N,N,D)

    return out
