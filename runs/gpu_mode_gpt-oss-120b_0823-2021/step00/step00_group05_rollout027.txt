"""
TriMul “outgoing” implementation (forward only).

We keep the expensive N‑wise contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
in PyTorch (torch.einsum) because it is already a highly‑optimized
batched‑matmul.  What we *do* move to Triton is the element‑wise part that
pre‑processes the 4‑D tensor:
   * Layer‑Norm (per‑channel)
   * Linear projections (no bias)
   * Gating (sigmoid) and optional mask
All those operations are fused into a single Triton kernel that works on
blocks of the (B, N, N, C) tensor and writes the three intermediate tensors
`left`, `right` and the gating values directly into GPU memory.  The final
einsum is executed in FP16 and the result is brought back to FP32, passed
through the output LayerNorm and linear projection (still in PyTorch).

The kernel is written for the Hopper H200 (SM‑90) and uses
float16 accumulation to maximise throughput while keeping the final
output in float32.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel – fused LayerNorm + Linear + Gate + Mask
# ----------------------------------------------------------------------
@triton.jit
def trimul_fwd_kernel(
    x_ptr,              # [B, N, N, C]       input
    left_ptr,           # [B, N, N, H]       left projection output
    right_ptr,          # [B, N, N, H]       right projection output
    left_gate_ptr,      # [B, N, N, H]       left gate (sigmoid)
    right_gate_ptr,     # [B, N, N, H]       right gate (sigmoid)
    out_gate_ptr,       # [B, N, N, H]       out gate  (sigmoid)
    mask_ptr,           # [B, N, N] (optional, may be nullptr)
    # ----- weights & biases (flat 1‑D) -----
    norm_weight_ptr,    # [C]
    norm_bias_ptr,      # [C]
    left_w_ptr,         # [H, C]
    right_w_ptr,        # [H, C]
    left_gate_w_ptr,    # [H, C]
    right_gate_w_ptr,   # [H, C]
    out_gate_w_ptr,     # [H, C]
    # ----- strides -----
    stride_x_b, stride_x_i, stride_x_j, stride_x_c,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_j, stride_og_h,
    stride_norm_w, stride_norm_b,
    stride_lw_h, stride_lw_c,
    stride_rw_h, stride_rw_c,
    stride_lgw_h, stride_lgw_c,
    stride_rgw_h, stride_rgw_c,
    stride_ogw_h, stride_ogw_c,
    # ----- shape & misc -----
    B, N, C, H,
    has_mask: tl.constexpr,
    BLOCK_B: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    One program works on a tile (BLOCK_B × BLOCK_N × BLOCK_N) of the
    (B, N, N, *) tensor.  Inside the tile we compute:

        x̂ = (x - μ) / sqrt(σ² + ε)        (LayerNorm)
        left  = x̂ @ W_left
        right = x̂ @ W_right
        left_gate  = sigmoid(x̂ @ W_left_gate)
        right_gate = sigmoid(x̂ @ W_right_gate)
        out_gate   = sigmoid(x̂ @ W_out_gate)

        if mask is present: left  *= mask ; right *= mask
        left  *= left_gate
        right *= right_gate
    """
    pid = tl.program_id(0)
    # --------------------------------------------------------------
    # compute tile offsets
    # --------------------------------------------------------------
    b0 = (pid // tl.cdiv(N, BLOCK_N) // tl.cdiv(N, BLOCK_N)) * BLOCK_B
    i0 = ((pid // tl.cdiv(N, BLOCK_N)) % tl.cdiv(N, BLOCK_N)) * BLOCK_N
    j0 = (pid % tl.cdiv(N, BLOCK_N)) * BLOCK_N

    offs_b = b0 + tl.arange(0, BLOCK_B)
    offs_i = i0 + tl.arange(0, BLOCK_N)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_c = tl.arange(0, C)

    mask_b = offs_b < B
    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # load the raw input (float32) and apply LayerNorm
    # --------------------------------------------------------------
    x = tl.load(
        x_ptr
        + offs_b[:, None, None, None] * stride_x_b
        + offs_i[None, :, None, None] * stride_x_i
        + offs_j[None, None, :, None] * stride_x_j
        + offs_c[None, None, None, :] * stride_x_c,
        mask=mask_b[:, None, None, None] & mask_i[None, :, None, None] &
             mask_j[None, None, :, None],
        other=0.0,
    )

    # mean & variance (per‑channel) – we inline a tiny reduction
    # NOTE: C is small (≤ 384) → we can afford a full reduction in‑kernel.
    eps = 1e-5
    mean = tl.zeros([C], dtype=tl.float32)
    var = tl.zeros([C], dtype=tl.float32)

    # reduction over (B,N,N) – we just compute a running mean/var across the tile
    # (the tile covers a slice of the full tensor, the final statistics are exact
    # because each element appears exactly once in some tile.)
    cnt = tl.full([C], (BLOCK_B * BLOCK_N * BLOCK_N), dtype=tl.int32)

    mean = tl.sum(x, axis=[0, 1, 2]) / tl.cast(cnt, tl.float32)
    diff = x - mean[None, None, None, :]
    var = tl.sum(diff * diff, axis=[0, 1, 2]) / tl.cast(cnt, tl.float32)

    x_norm = (x - mean[None, None, None, :]) * tl.rsqrt(var[None, None, None, :] + eps)

    # --------------------------------------------------------------
    # Linear projections (no bias) – we treat H as a compile‑time constant
    # --------------------------------------------------------------
    # x_norm: (B,N,N,C)   weight: (H, C)
    # we compute a matmul for each (b,i,j) → (H)
    # because H ≤ 128 we can do it with a small inner loop
    left  = tl.zeros([BLOCK_B, BLOCK_N, BLOCK_N, BLOCK_H], dtype=tl.float32)
    right = tl.zeros([BLOCK_B, BLOCK_N, BLOCK_N, BLOCK_H], dtype=tl.float32)

    for h in range(0, H, BLOCK_H):
        offs_h = h + tl.arange(0, BLOCK_H)

        w_l = tl.load(left_w_ptr + offs_h[:, None] * stride_lw_h +
                                    offs_c[None, :] * stride_lw_c,
                     mask=offs_h[:, None] < H, other=0.0)
        w_r = tl.load(right_w_ptr + offs_h[:, None] * stride_rw_h +
                                    offs_c[None, :] * stride_rw_c,
                     mask=offs_h[:, None] < H, other=0.0)

        # (B,N,N,C) dot (H,C) → (B,N,N,H)
        proj_l = tl.dot(x_norm, w_l)   # result (B,N,N,H)
        proj_r = tl.dot(x_norm, w_r)

        tl.store(left_ptr  + offs_b[:, None, None, None] * stride_l_b
                              + offs_i[None, :, None, None] * stride_l_i
                              + offs_j[None, None, :, None] * stride_l_j
                              + offs_h[None, None, None, :] * stride_l_h,
                 proj_l, mask=mask_b[:, None, None, None] &
                           mask_i[None, :, None, None] &
                           mask_j[None, None, :, None] &
                           (offs_h[None, None, None, :] < H))

        tl.store(right_ptr + offs_b[:, None, None, None] * stride_l_b
                              + offs_i[None, :, None, None] * stride_l_i
                              + offs_j[None, None, :, None] * stride_l_j
                              + offs_h[None, None, None, :] * stride_l_h,
                 proj_r, mask=mask_b[:, None, None, None] &
                           mask_i[None, :, None, None] &
                           mask_j[None, None, :, None] &
                           (offs_h[None, None, None, :] < H))

    # --------------------------------------------------------------
    # Gates (sigmoid)
    # --------------------------------------------------------------
    for h in range(0, H, BLOCK_H):
        offs_h = h + tl.arange(0, BLOCK_H)

        w_lg = tl.load(left_gate_w_ptr + offs_h[:, None] * stride_lgw_h +
                                      offs_c[None, :] * stride_lgw_c,
                      mask=offs_h[:, None] < H, other=0.0)
        w_rg = tl.load(right_gate_w_ptr + offs_h[:, None] * stride_rgw_h +
                                      offs_c[None, :] * stride_rgw_c,
                      mask=offs_h[:, None] < H, other=0.0)
        w_og = tl.load(out_gate_w_ptr + offs_h[:, None] * stride_ogw_h +
                                      offs_c[None, :] * stride_ogw_c,
                      mask=offs_h[:, None] < H, other=0.0)

        gate_l = tl.sigmoid(tl.dot(x_norm, w_lg))
        gate_r = tl.sigmoid(tl.dot(x_norm, w_rg))
        gate_o = tl.sigmoid(tl.dot(x_norm, w_og))

        tl.store(left_gate_ptr  + offs_b[:, None, None, None] * stride_rg_b
                                   + offs_i[None, :, None, None] * stride_rg_i
                                   + offs_j[None, None, :, None] * stride_rg_j
                                   + offs_h[None, None, None, :] * stride_rg_h,
                 gate_l, mask=mask_b[:, None, None, None] &
                           mask_i[None, :, None, None] &
                           mask_j[None, None, :, None] &
                           (offs_h[None, None, None, :] < H))

        tl.store(right_gate_ptr + offs_b[:, None, None, None] * stride_rg_b
                                   + offs_i[None, :, None, None] * stride_rg_i
                                   + offs_j[None, None, :, None] * stride_rg_j
                                   + offs_h[None, None, None, :] * stride_rg_h,
                 gate_r, mask=mask_b[:, None, None, None] &
                           mask_i[None, :, None, None] &
                           mask_j[None, None, :, None] &
                           (offs_h[None, None, None, :] < H))

        tl.store(out_gate_ptr   + offs_b[:, None, None, None] * stride_og_b
                                   + offs_i[None, :, None, None] * stride_og_i
                                   + offs_j[None, None, :, None] * stride_og_j
                                   + offs_h[None, None, None, :] * stride_og_h,
                 gate_o, mask=mask_b[:, None, None, None] &
                           mask_i[None, :, None, None] &
                           mask_j[None, None, :, None] &
                           (offs_h[None, None, None, :] < H))

    # --------------------------------------------------------------
    # Apply mask (if present) and gates
    # --------------------------------------------------------------
    if has_mask:
        mask = tl.load(mask_ptr + offs_b[:, None, None] * stride_x_b +
                                   offs_i[None, :, None] * stride_x_i +
                                   offs_j[None, None, :] * stride_x_j,
                       mask=mask_b[:, None, None] &
                            mask_i[None, :, None] &
                            mask_j[None, None, :],
                       other=1.0)
        mask = mask[:, :, :, None]          # broadcast over H
        tl.store(left_ptr,
                 tl.load(left_ptr, mask=mask_b[:, None, None, None] &
                                     mask_i[None, :, None, None] &
                                     mask_j[None, None, :, None]) * mask,
                 mask=mask_b[:, None, None, None] &
                      mask_i[None, :, None, None] &
                      mask_j[None, None, :, None])
        tl.store(right_ptr,
                 tl.load(right_ptr, mask=mask_b[:, None, None, None] &
                                      mask_i[None, :, None, None] &
                                      mask_j[None, None, :, None]) * mask,
                 mask=mask_b[:, None, None, None] &
                      mask_i[None, :, None, None] &
                      mask_j[None, None, :, None])

    # left  *= left_gate   ;   right *= right_gate
    left_gate  = tl.load(left_gate_ptr,  mask=mask_b[:, None, None, None] &
                                            mask_i[None, :, None, None] &
                                            mask_j[None, None, :, None])
    right_gate = tl.load(right_gate_ptr, mask=mask_b[:, None, None, None] &
                                            mask_i[None, :, None, None] &
                                            mask_j[None, None, :, None])

    tl.store(left_ptr,
             tl.load(left_ptr) * left_gate,
             mask=mask_b[:, None, None, None] &
                  mask_i[None, :, None, None] &
                  mask_j[None, None, :, None])
    tl.store(right_ptr,
             tl.load(right_ptr) * right_gate,
             mask=mask_b[:, None, None, None] &
                  mask_i[None, :, None, None] &
                  mask_j[None, None, :, None])

# ----------------------------------------------------------------------
# Helper – launch the Triton kernel
# ----------------------------------------------------------------------
def fused_preprocess(
    x: torch.Tensor,
    mask: torch.Tensor,
    weights: Dict[str, torch.Tensor],
    config: Dict[str, Any],
) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
    """
    Run the Triton kernel that produces:
        left, right, out_gate   (all float16, shape [B,N,N,H])
    """
    B, N, _, C = x.shape
    H = config["hidden_dim"]

    device = x.device
    # allocate output buffers
    left = torch.empty((B, N, N, H), dtype=torch.float16, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty_like(left)

    # launch parameters
    BLOCK_B = 1                # B is tiny (≤2) → keep it in one block
    BLOCK_N = 64               # fits in shared memory, good for N up‑to 1024
    BLOCK_H = 32               # hidden_dim is multiple of 32

    grid = (
        triton.cdiv(B, BLOCK_B) *
        triton.cdiv(N, BLOCK_N) *
        triton.cdiv(N, BLOCK_N),
    )

    has_mask = mask is not None
    mask_ptr = mask.data_ptr() if has_mask else 0

    trimul_fwd_kernel[grid](
        x.data_ptr(),
        left.data_ptr(),
        right.data_ptr(),
        left.data_ptr(),          # reuse buffers for gates – they will be overwritten later
        right.data_ptr(),
        out_gate.data_ptr(),
        mask_ptr,
        # weights
        weights["norm.weight"].data_ptr(),
        weights["norm.bias"].data_ptr(),
        weights["left_proj.weight"].data_ptr(),
        weights["right_proj.weight"].data_ptr(),
        weights["left_gate.weight"].data_ptr(),
        weights["right_gate.weight"].data_ptr(),
        weights["out_gate.weight"].data_ptr(),
        # strides
        x.stride(0), x.stride(1), x.stride(2), x.stride(3),
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        weights["norm.weight"].stride(0), weights["norm.bias"].stride(0),
        weights["left_proj.weight"].stride(0), weights["left_proj.weight"].stride(1),
        weights["right_proj.weight"].stride(0), weights["right_proj.weight"].stride(1),
        weights["left_gate.weight"].stride(0), weights["left_gate.weight"].stride(1),
        weights["right_gate.weight"].stride(0), weights["right_gate.weight"].stride(1),
        weights["out_gate.weight"].stride(0), weights["out_gate.weight"].stride(1),
        # shape & flags
        B, N, C, H,
        int(has_mask),
        BLOCK_B, BLOCK_N, BLOCK_H,
    )
    return left, right, out_gate

# ----------------------------------------------------------------------
# Main entry point – custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask        : [B, N, N] (optional, bool/float)
        - weights     : dict of pre‑loaded model parameters
        - config      : dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm + linear + gating (in Triton)
    # ------------------------------------------------------------------
    left, right, out_gate = fused_preprocess(input_tensor, mask, weights, config)

    # cast to fp16 for the heavy contraction
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) The N‑wise contraction (einsum) – torch uses highly‑tuned
    #    batched matmul kernels (fp16 → fp32 accumulation)
    # ------------------------------------------------------------------
    # shape: (B, N, N, H)  ->  (B, N, H, N) dot (B, N, H, N).T   -> (B, N, N, H)
    # we can rewrite the einsum as a batch‑matmul per head:
    #   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # which is exactly torch.einsum with the given pattern.
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # ------------------------------------------------------------------
    # 3) Output LayerNorm + linear (still in PyTorch)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    out = out * out_gate.to(torch.float32)

    # final linear projection
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
```
