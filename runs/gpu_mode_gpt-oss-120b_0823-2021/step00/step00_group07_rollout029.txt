"""
TriMul “outgoing” kernel (AlphaFold3) -------------------------------------------------
We implement the expensive 4‑D tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape [B,N,N,D] (D = hidden_dim).
The outer LayerNorm / Linear / gating logic stays in PyTorch – only the
N³ contraction is fused into a Triton kernel.  The kernel tiles the (i,j)
plane with BLOCK_M×BLOCK_N threads, iterates over k in BLOCK_K steps and
accumulates a D‑dimensional vector for each output element.  The final
output is cast back to float32, LayerNorm‑ed and projected with the supplied
weights.

The implementation works for any batch size (B∈{1,2}), any sequence length
N (≤1024) and hidden dimension D (≤384).  It tolerates an optional mask
(which is simply broadcasted into the left/right tensors before the kernel
launch).  All tensors are expected to be on the CUDA device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# --------------------------------------------------------------------------- #
# Triton kernel: computes out = Σ_k left[...,k,:] * right[...,k,:]
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # shapes
    B, N, D,
    # strides (in elements, not bytes)
    stride_bl_i_kd, stride_bl_i_k, stride_bl_i, stride_bl_b,
    stride_br_j_kd, stride_br_j_k, stride_br_j, stride_br_b,
    stride_o_b_i_jd, stride_o_i_jd, stride_o_jd, stride_o_b,
    # compile‑time constants
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, D_BLOCK: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # tile over i
    pid_j = tl.program_id(2)          # tile over j

    # ------- offsets for the output tile (i,j) ------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # broadcast i and j offsets to form a (BLOCK_M, BLOCK_N) grid
    offs_i = tl.where(mask_i, offs_i, 0)
    offs_j = tl.where(mask_j, offs_j, 0)

    # ----------- allocate accumulator for D dimension -----------------------
    # D may be larger than the register limit, so we tile it in D_BLOCK steps
    acc = tl.zeros((BLOCK_M, BLOCK_N, D_BLOCK), dtype=tl.float32)

    # ----------- loop over k -----------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        # k‑tile offsets
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load left : shape (B, i, k, D) --------------------------------
        # left_ptr + b*stride_bl_b + i*stride_bl_i + k*stride_bl_i_k + d
        left_ptrs = (
            left_ptr
            + pid_b * stride_bl_b
            + offs_i[:, None, None] * stride_bl_i
            + offs_k[None, :, None] * stride_bl_i_k
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0)

        # ---- load right : shape (B, j, k, D) --------------------------------
        # note: we need right[j, k, d] but the contraction uses right[b, j, k, d]
        right_ptrs = (
            right_ptr
            + pid_b * stride_br_b
            + offs_j[:, None, None] * stride_br_j
            + offs_k[None, :, None] * stride_br_j_k
        )
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0)

        # both left and right have shape (BLOCK_M/BLOCK_N, BLOCK_K, D)
        # compute element‑wise product and sum over k
        # we process D in chunks of D_BLOCK
        for d0 in range(0, D, D_BLOCK):
            d_slice = tl.arange(0, D_BLOCK) + d0
            # load d‑slice
            left_d = tl.load(
                left_ptr
                + pid_b * stride_bl_b
                + offs_i[:, None, None] * stride_bl_i
                + offs_k[None, :, None] * stride_bl_i_k
                + d_slice[None, None, :] * stride_bl_i_kd,
                mask=mask_i[:, None, None] & mask_k[None, :, None],
                other=0.0,
            )
            right_d = tl.load(
                right_ptr
                + pid_b * stride_br_b
                + offs_j[:, None, None] * stride_br_j
                + offs_k[None, :, None] * stride_br_j_k
                + d_slice[None, None, :] * stride_br_j_kd,
                mask=mask_j[:, None, None] & mask_k[None, :, None],
                other=0.0,
            )
            # multiply and accumulate over k
            # left_d, right_d : (BLOCK_M, BLOCK_K, D_BLOCK)
            acc += tl.sum(left_d * right_d, axis=1)   # sum over k

    # ---------- write back ---------------------------------------------------
    # out pointer offsets
    out_ptrs = (
        out_ptr
        + pid_b * stride_o_b
        + offs_i[:, None, None] * stride_o_i_jd
        + offs_j[None, :, None] * stride_o_jd
    )
    for d0 in range(0, D, D_BLOCK):
        d_slice = tl.arange(0, D_BLOCK) + d0
        out_ptrs_d = out_ptrs + d_slice[None, None, :] * stride_o_b_i_jd
        tl.store(out_ptrs_d, acc[:, :, :], mask=mask_i[:, None, None] & mask_j[None, :, None])

# --------------------------------------------------------------------------- #
# Host‑side helper that launches the kernel
# --------------------------------------------------------------------------- #
def trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, N, N, D] (float16 or float32)
    returns    : [B, N, N, D] (float32)
    """
    assert left.shape == right.shape
    B, N, _, D = left.shape
    device = left.device

    # use float16 inside the kernel for bandwidth, cast later
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # strides in elements (not bytes)
    stride_bl_b, stride_bl_i, stride_bl_k, stride_bl_d = left.stride()
    stride_br_b, stride_br_j, stride_br_k, stride_br_d = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    D_BLOCK = 32                      # fits in registers (D ≤ 384)

    grid = (B,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, D=D,
        stride_bl_b=stride_bl_b,
        stride_bl_i=stride_bl_i,
        stride_bl_i_k=stride_bl_k,
        stride_bl_i_kd=stride_bl_d,
        stride_br_b=stride_br_b,
        stride_br_j=stride_br_j,
        stride_br_j_k=stride_br_k,
        stride_br_j_kd=stride_br_d,
        stride_o_b=stride_o_b,
        stride_o_i_jd=stride_o_i,
        stride_o_jd=stride_o_j,
        stride_o_b_i_jd=stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        D_BLOCK=D_BLOCK,
    )
    return out

# --------------------------------------------------------------------------- #
# Full custom_kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask         : [B, N, N]        (bool / float)
        - weights      : dict of weight/bias tensors
        - config       : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor : [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    device = input_tensor.device

    # ---------- LayerNorm on the last dim ------------------------------------
    # fused manually (mean / var per element) using torch for simplicity
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B,N,N,dim]

    # ---------- Linear projections -------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]        # (hidden_dim, dim)

    # linear without bias: x @ W.T
    left = torch.einsum("bnqd,hd->bnqh", x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.einsum("bnqd,hd->bnqh", x, right_proj_w)

    # ---------- Mask ----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)            # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gating -------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.einsum("bnqd,hd->bnqh", x, left_gate_w))
    right_gate = torch.sigmoid(torch.einsum("bnqd,hd->bnqh", x, right_gate_w))
    out_gate = torch.sigmoid(torch.einsum("bnqd,hd->bnqh", x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ---------- Triton contraction -------------------------------------------
    out = trimul_outgoing_triton(left, right)   # float32 result, shape [B,N,N,hidden_dim]

    # ---------- Output norm + gate + final linear ----------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    out = F.layer_norm(out, normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)
    out = out * out_gate
    out = torch.matmul(out, to_out_w.t())   # [B,N,N,dim]

    return out
