"""
TriMul “outgoing” implementation.

The heavy part is the tensor‑wise multiplication
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix‑multiply for every hidden dimension *d*.
We keep the surrounding LayerNorm / Linear / gating logic in PyTorch
(because they are cheap) and fuse the N³‑loop in a single Triton kernel.
The kernel works on fp16 inputs, accumulates in fp32 and writes fp16
(which is later cast back to fp32 for the final linear layer).
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched matmul per hidden dimension (the einsum in the paper)
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,                # *[B, N, N, H] left tensor   (fp16)
    right_ptr,               # *[B, N, N, H] right tensor  (fp16)
    out_ptr,                 # *[B, N, N, H] output tensor (fp16)
    B, N, H,                 # static sizes
    stride_lbn, stride_lbk, stride_lbh,   # left strides (B,N,N,H)
    stride_rbn, stride_rbk, stride_rbh,   # right strides
    stride_obn, stride_obk, stride_obh,   # out strides
    BLOCK_M: tl.constexpr,   # tile size in i dimension
    BLOCK_N: tl.constexpr,   # tile size in j dimension
    BLOCK_K: tl.constexpr,   # tile size in reduction k
):
    """ Compute out = left @ right.transpose(-2,-1) per batch & hidden dim """
    pid = tl.program_id(0)                      # linear id over (B * H)
    b = pid // H                                 # batch index
    h = pid % H                                  # hidden dimension index

    # offsets for the output tile
    offs_m = tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)
    # start offsets for this tile
    i = tl.program_id(1) * BLOCK_M + offs_m
    j = tl.program_id(2) * BLOCK_N + offs_n

    # mask out‑of‑bounds
    mask_i = i < N
    mask_j = j < N

    # pointers to the start of the current batch/hidden slice
    left_base  = left_ptr  + b * stride_lbn + h * stride_lbh
    right_base = right_ptr + b * stride_rbn + h * stride_rbh
    out_base   = out_ptr   + b * stride_obn + h * stride_obh

    # initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        # offsets for the reduction dimension
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = k_start + offs_k < N

        # left[i,k]   shape (M, K)
        left_ptrs = left_base + i[:, None] * stride_lbk + (k_start + offs_k)[None, :] * stride_lbk
        left_vals = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right[j,k]  we need right transposed => right[k,j]
        # right has layout (B,N,N,H) where dim2 is the *k* axis, dim1 is *j*
        # after transpose we read (k,j)
        right_ptrs = right_base + (k_start + offs_k)[:, None] * stride_rbk + j[None, :] * stride_rbk
        right_vals = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot‑product for the tile
        acc += tl.dot(left_vals, right_vals)

    # write result back as fp16
    out_ptrs = out_base + i[:, None] * stride_obk + j[None, :] * stride_obk
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper: LayerNorm (no torch.nn to avoid extra kernel launches)
# ----------------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]  float32
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    data = (input_tensor, mask, weights, config)
        input_tensor : [B, N, N, C]   float32
        mask         : [B, N, N]      (bool / 0‑1)   – may be None
        weights      : dict of torch tensors (float32)
        config       : {"dim": C, "hidden_dim": H}
    returns
        out : [B, N, N, C] float32
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm + Linear projections + gates (all in pytorch, fp32)
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                     # [B,N,N,C] fp32

    # linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # apply mask if present
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)           # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 2. Triton kernel for the heavy N^3 multiplication
    # ------------------------------------------------------------------
    # cast to fp16 for the kernel (more bandwidth, still accurate)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)   # shape [B,N,N,H]

    # strides (in elements) for the 4‑D tensors
    # layout is contiguous in C order: (B,N,N,H)
    stride_lbn, stride_lbk, stride_lbh = left_fp16.stride()
    stride_rbn, stride_rbk, stride_rbh = right_fp16.stride()
    stride_obn, stride_obk, stride_obh = out_fp16.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(B * H, 1),                   # program id 0: batch * hidden
        triton.cdiv(N, BLOCK_M),                # program id 1: i‑tiles
        triton.cdiv(N, BLOCK_N),                # program id 2: j‑tiles
    )

    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, H,
        stride_lbn, stride_lbk, stride_lbh,
        stride_rbn, stride_rbk, stride_rbh,
        stride_obn, stride_obk, stride_obh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # back to fp32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 3. Final LayerNorm, gating and output projection (pure torch)
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
