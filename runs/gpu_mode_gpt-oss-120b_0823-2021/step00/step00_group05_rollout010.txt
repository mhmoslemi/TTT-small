"""
TriMul “outgoing” implementation.

We keep the LayerNorms and linear projections in PyTorch (they are cheap
compared to the N³ “einsum” part).  The core operation

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is a batched matrix‑multiply for each head d:

    left  → (B, D, I, K)   (permute 0,3,1,2)
    right → (B, D, K, J)   (permute 0,3,2,1)

For every (b,d) we need a matrix product I×K × K×J → I×J.
The kernel below computes many such products in a single launch.
It works in fp16 for the inner multiply‑add and writes the result in fp16;
the host converts the tensor back to fp32 and applies the final
LayerNorm / linear projection.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# ----------------------------------------------------------------------
# Triton kernel: batched 2‑D matrix multiply (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    a_ptr,            # *Pointer* to left matrix   (B*D, I, K)
    b_ptr,            # *Pointer* to right matrix  (B*D, K, J)
    c_ptr,            # *Pointer* to output matrix (B*D, I, J)
    M, N, K,          # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (batch, i, k)
    stride_bz, stride_bk, stride_bn,   # strides for b (batch, k, j)
    stride_cz, stride_cm, stride_cn,   # strides for c (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_m = tl.program_id(0)          # tile along M
    pid_n = tl.program_id(1)          # tile along N
    pid_z = tl.program_id(2)          # batch × head index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current tile
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # dot‑product accumulation (fp32)
        acc += tl.dot(a, b)

    # write result back in fp16 (final conversion to fp32 is done on host)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def triton_batch_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)   fp16
    right : (B, H, K, J)   fp16
    returns: (B, H, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # output allocation
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten batch*head dimension for the kernel
    batch_dim = B * H
    a = left.view(batch_dim, I, K)
    b = right.view(batch_dim, K, J)
    c = out.view(batch_dim, I, J)

    # strides (in elements, not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # block sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),
        triton.cdiv(J, BLOCK_N),
        batch_dim,
    )
    batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # good default for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward with mixed‑precision Triton acceleration
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments:
        data : (input, mask, weights, config)
            input  – torch.Tensor [B, N, N, dim] (float32)
            mask   – torch.Tensor [B, N, N]      (float32 or bool)
            weights– dict of leaf weight tensors
            config – dict with keys “dim” and “hidden_dim”
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ---------- LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias

    # ---------- Linear projections (float32 → fp16) ----------
    # weight shapes: (hidden_dim, dim)
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # fp16 tensors for the heavy part
    x_fp16 = x.to(torch.float16)

    left = torch.nn.functional.linear(x_fp16, left_proj_w.to(torch.float16))
    right = torch.nn.functional.linear(x_fp16, right_proj_w.to(torch.float16))

    # ---------- Apply mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(torch.float16)        # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gating ----------
    left_gate = torch.nn.functional.linear(x_fp16, left_gate_w.to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w.to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_w.to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Triton batched matmul ----------
    # reshape to (B, hidden_dim, N, N) and permute for matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, J)

    out = triton_batch_matmul(left_perm, right_perm)   # (B, H, I, J) fp16

    # bring back to (B, N, N, dim)
    out = out.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,dim)

    # ---------- Output LayerNorm ----------
    # manual LN in fp32 (small tensor)
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    # ---------- Final gating and projection ----------
    out = out * out_gate.to(torch.float32)
    out = torch.nn.functional.linear(out, to_out_w.to(torch.float32))

    return out
