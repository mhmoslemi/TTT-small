"""
TriMul “outgoing” implementation.

The expensive part of the forward pass is the three‑dimensional
einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication for every hidden
dimension *d* :

    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T

The kernel below computes this product for many (batch,hidden) pairs at
once. All other operations (LayerNorm, Linear, gating, final LayerNorm
and Linear) are cheap and stay in PyTorch. The kernel works on
float16 inputs and accumulates in float32, finally storing float32
results. It is written for Triton 3.3.1 and targets NVIDIA H200
(sm_90) where 128‑bit loads/stores are optimal.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel that performs: out = left @ right^T   (batched per hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,                 # *[B, N, N, H]   (float16)
    right_ptr,                # *[B, N, N, H]   (float16)
    out_ptr,                  # *[B, N, N, H]   (float32)
    B, N, H,
    stride_lb, stride_lk, stride_lh,   # left strides  (B,N,N,H)
    stride_rb, stride_rk, stride_rh,   # right strides
    stride_ob, stride_ok, stride_oh,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_bh = tl.program_id(0)          # combined batch*hidden index
    pid_m  = tl.program_id(1)          # row block
    pid_n  = tl.program_id(2)          # col block

    # decode batch and hidden dimension from pid_bh
    b = pid_bh // H
    h = pid_bh % H

    # offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # compute pointers
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # left:   (i,k)   -> strides (stride_lk, stride_lb) after fixing h,b
    # right:  (j,k)   we need right.T, i.e. (k,j)
    #   we therefore load right with k as first dim and j as second dim
    a_ptrs = left_base + (offs_m[:, None] * stride_lk
                         + offs_k[None, :] * stride_lb)
    b_ptrs = right_base + (offs_k[:, None] * stride_rb
                         + offs_n[None, :] * stride_rk)

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for _k in range(0, tl.cdiv(N, BLOCK_K)):
        # load a block of left (M,K) and right (K,N)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # matrix‑multiply the block
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_lb
        b_ptrs += BLOCK_K * stride_rb

    # write result
    out_ptrs = out_base + (offs_m[:, None] * stride_ok
                         + offs_n[None, :] * stride_ob)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]
                 ) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data: tuple (input, mask, weights, config)
        input : [B, N, N, dim]          (float32)
        mask  : [B, N, N]               (float32 or bool, 1.0 = keep)
        weights: dict of model parameters (all float32)
        config : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]

    device = inp.device

    # ------------------------------------------------------------------
    # Helper: LayerNorm + Linear (no bias) in FP32 → FP16 for the heavy part
    # ------------------------------------------------------------------
    def layer_norm(x, w, b):
        mean = x.mean(-1, keepdim=True)
        var = ((x - mean) ** 2).mean(-1, keepdim=True)
        return (x - mean) * torch.rsqrt(var + 1e-6) * w + b

    # ------------------------------------------------------------------
    # 1. Input normalisation
    # ------------------------------------------------------------------
    x = layer_norm(
        inp,
        weights["norm.weight"],
        weights["norm.bias"]
    )                                   # [B,N,N,dim]  fp32

    # cast to fp16 for the heavy matmul
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – stay in fp16
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"]).view(B, N, N, hidden)
    right = F.linear(x_fp16, weights["right_proj.weight"]).view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – compute in fp32 then cast
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).view(B, N, N, hidden)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).view(B, N, N, hidden)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"])).view(B, N, N, hidden)

    left = left * left_gate.to(torch.float16)
    right = right * right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Triton kernel: out = left @ right^T   (batched per hidden)
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides (in elements, not bytes) for left/right/out tensors
    stride_lb = left.stride(0)
    stride_lk = left.stride(1)
    stride_lh = left.stride(3)

    stride_rb = right.stride(0)
    stride_rk = right.stride(1)
    stride_rh = right.stride(3)

    stride_ob = out.stride(0)
    stride_ok = out.stride(1)
    stride_oh = out.stride(3)

    # block sizes – tuned for Hopper fp16 matmul
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # launch grid: (batch*hidden, ceil(N/BM), ceil(N/BN))
    grid = (triton.cdiv(B * hidden, 1),  # we keep one program per (b,h)
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, hidden,
        stride_lb, stride_lk, stride_lh,
        stride_rb, stride_rk, stride_rh,
        stride_ob, stride_ok, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Final processing (LayerNorm, gating, output linear) – back to fp32
    # ------------------------------------------------------------------
    out = out.to(torch.float32)                     # [B,N,N,hidden]
    out = layer_norm(out,
                     weights["to_out_norm.weight"],
                     weights["to_out_norm.bias"])
    out = out * out_gate                               # broadcast gate
    out = F.linear(out, weights["to_out.weight"]).view(B, N, N, dim)

    return out
