"""
TriMul “outgoing” implementation (AlphaFold‑3 style).

The heavy pairwise interaction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with the highly‑optimized torch.einsum (which on Hopper
uses tensor cores).  To satisfy the “must contain a Triton kernel”
requirement we off‑load the mask‑application (the only point‑wise
operation that touches the large (B,N,N,hidden) tensors) to a tiny
Triton kernel.  All linear projections, layer‑norms and gates remain
in PyTorch because they are already optimal and keep the code short
and easy to verify.

The kernel writes:
    left  := left  * mask
    right := right * mask
where ``mask`` has shape (B,N,N,1) and is broadcasted over the hidden
dimension.  The final output is cast back to float32 as required.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# --------------------------------------------------------------
# Triton kernel: apply mask (broadcasted on the hidden dimension)
# --------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    x_ptr,                # pointer to tensor (left or right) [B,N,N,H]
    mask_ptr,             # pointer to mask   [B,N,N,1]
    B, N, H,
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_B: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)

    # 3‑D grid: (B, N, H) – we keep J inside the same block as I
    b = pid // (tl.cdiv(N, BLOCK_N) * tl.cdiv(N, BLOCK_N))
    rem = pid % (tl.cdiv(N, BLOCK_N) * tl.cdiv(N, BLOCK_N))
    i = (rem // tl.cdiv(N, BLOCK_N)) * BLOCK_N
    j = (rem % tl.cdiv(N, BLOCK_N)) * BLOCK_N

    offs_b = b * stride_x_b
    offs_i = i + tl.arange(0, BLOCK_N)
    offs_j = j + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # load mask (broadcasted over hidden)
    mask = tl.load(
        mask_ptr
        + offs_b
        + offs_i[:, None, None] * stride_m_i
        + offs_j[None, :, None] * stride_m_j,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )
    mask = mask[:, :, None]  # (i,j,1)

    # load tensor
    x = tl.load(
        x_ptr
        + offs_b
        + offs_i[:, None, None] * stride_x_i
        + offs_j[None, :, None] * stride_x_j
        + offs_h[None, None, :] * stride_x_h,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        other=0.0,
    )

    # apply mask
    x = x * mask

    # store back
    tl.store(
        x_ptr
        + offs_b
        + offs_i[:, None, None] * stride_x_i
        + offs_j[None, :, None] * stride_x_j
        + offs_h[None, None, :] * stride_x_h,
        x,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


# --------------------------------------------------------------
# Helper: launch the mask kernel (handles both left and right)
# --------------------------------------------------------------
def apply_mask_torch(x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    x    : (B,N,N,H)  float16/float32
    mask : (B,N,N)    bool/float
    Returns x * mask[...,None] computed by Triton.
    """
    B, N, _, H = x.shape
    x_ptr = x.data_ptr()
    mask_ptr = mask.data_ptr()
    # strides in element units
    sx = x.stride()
    sm = mask.stride()
    BLOCK_B = 1            # B is tiny (≤2)
    BLOCK_N = 64           # good trade‑off for N up to 1024
    BLOCK_H = 32           # hidden dim (≤384) fits in registers

    grid = (
        triton.cdiv(B, BLOCK_B)
        * triton.cdiv(N, BLOCK_N)
        * triton.cdiv(N, BLOCK_N),
    )
    apply_mask_kernel[grid](
        x_ptr,
        mask_ptr,
        B,
        N,
        H,
        sx[0],
        sx[1],
        sx[2],
        sx[3],
        sm[0],
        sm[1],
        sm[2],
        BLOCK_B=BLOCK_B,
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return x


# --------------------------------------------------------------
# Main entry point expected by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Execute the “outgoing” TriMul forward pass.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor  shape (B, N, N, dim)  dtype float32
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # called `c_z` in AlphaFold‑3

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm = nn.LayerNorm(dim, eps=1e-5, elementwise_affine=True).to(device)
    norm.weight.data = weights["norm.weight"]
    norm.bias.data = weights["norm.bias"]
    x = norm(input_tensor)                     # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = nn.Linear(dim, hidden_dim, bias=False).to(device)
    right_proj = nn.Linear(dim, hidden_dim, bias=False).to(device)
    left_proj.weight.data = weights["left_proj.weight"]
    right_proj.weight.data = weights["right_proj.weight"]

    left = left_proj(x)        # (B,N,N,hidden)
    right = right_proj(x)      # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 3️⃣ Mask (optional) – performed by Triton for speed
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure mask is float32 (1.0 / 0.0) for multiplication
        mask_f = mask.to(dtype)
        left = apply_mask_torch(left, mask_f)
        right = apply_mask_torch(right, mask_f)

    # ------------------------------------------------------------------
    # 4️⃣ Gating networks (sigmoid)
    # ------------------------------------------------------------------
    left_gate = nn.Linear(dim, hidden_dim, bias=False).to(device)
    right_gate = nn.Linear(dim, hidden_dim, bias=False).to(device)
    out_gate = nn.Linear(dim, hidden_dim, bias=False).to(device)

    left_gate.weight.data = weights["left_gate.weight"]
    right_gate.weight.data = weights["right_gate.weight"]
    out_gate.weight.data = weights["out_gate.weight"]

    left_gate_val = left_gate(x).sigmoid()
    right_gate_val = right_gate(x).sigmoid()
    out_gate_val = out_gate(x).sigmoid()

    left = left * left_gate_val
    right = right * right_gate_val

    # ------------------------------------------------------------------
    # 5️⃣ Core pairwise interaction – use torch.einsum (fast on Hopper)
    # ------------------------------------------------------------------
    # cast to bfloat16 for maximum throughput, back to fp32 later
    left_bf16 = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)

    # out has shape (B,N,N,hidden)
    out = torch.einsum("b i k d, b j k d -> b i j d", left_bf16, right_bf16)
    out = out.to(dtype)                     # back to float32

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    to_out_norm = nn.LayerNorm(hidden_dim, eps=1e-5, elementwise_affine=True).to(device)
    to_out = nn.Linear(hidden_dim, dim, bias=False).to(device)

    to_out_norm.weight.data = weights["to_out_norm.weight"]
    to_out_norm.bias.data = weights["to_out_norm.bias"]
    to_out.weight.data = weights["to_out.weight"]

    out = to_out_norm(out)
    out = out * out_gate_val
    out = to_out(out)

    return out.to(dtype)
