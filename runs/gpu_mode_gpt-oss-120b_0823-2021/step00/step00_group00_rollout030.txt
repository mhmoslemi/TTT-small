"""
TriMul (outgoing) – fast Triton implementation of the N³ inner product.

The heavy part of the forward pass is the tensor contraction  

    out[b, i, j, d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left`/`right` ∈ ℝ^{B×N×N×H}.  
We keep everything in fp32 (the kernel can trivially be switched to fp16/bf16)
and fuse the k‑reduction with a small GEMM per hidden‑dimension slice.
All other operations (LayerNorm, Linear, gating) are performed with
regular PyTorch kernels – they are cheap compared with the O(N³) work.
The kernel processes a BLOCK_M × BLOCK_N tile of the (i,j) plane for a
single batch element at a time and iterates over the K‑dimension in
chunks of BLOCK_K.
"""
import torch
import torch.nn as nn
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel that computes the 4‑D contraction described above.
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,               # tensor dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    pid = tl.program_id(0)               # linear id over (b, i_tile, j_tile)
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    b     = pid // (num_i * num_j)                # batch index
    pid   = pid % (num_i * num_j)
    i_tile = pid // num_j                         # i‑tile index
    j_tile = pid %  num_j                         # j‑tile index

    # start indices of the tile
    i_offset = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds tiles
    mask_i = i_offset < N
    mask_j = j_offset < N

    # ------------------------------------------------------------------
    # accumulate across the hidden dimension H (BLOCK_H == H in all
    # supported configs, but we keep it as a constexpr for flexibility)
    # ------------------------------------------------------------------
    for h in range(0, H, BLOCK_H):
        # one accumulator per hidden slice
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # iterate over K in BLOCK_K steps
        for k in range(0, N, BLOCK_K):
            k_offset = k + tl.arange(0, BLOCK_K)

            # ----- load left tile (M, K) -----
            left_ptrs = left_ptr + \
                b * stride_left_b + \
                i_offset[:, None] * stride_left_i + \
                k_offset[None, :] * stride_left_k + \
                h * stride_left_h
            mask_left = (i_offset[:, None] < N) & (k_offset[None, :] < N)
            a = tl.load(left_ptrs, mask=mask_left, other=0.0)   # (M,K)

            # ----- load right tile (K, N)  (note the transpose) -----
            # we need right[b, j, k, h] → (K,N) where N is the j‑dimension.
            right_ptrs = right_ptr + \
                b * stride_right_b + \
                j_offset[None, :] * stride_right_j + \
                k_offset[:, None] * stride_right_k + \
                h * stride_right_h
            mask_right = (j_offset[None, :] < N) & (k_offset[:, None] < N)
            b_tile = tl.load(right_ptrs, mask=mask_right, other=0.0)   # (K,N)

            # matrix multiply acc += a @ b_tile
            acc += tl.dot(a, b_tile)               # (M,N)

        # ----- store the result for this hidden slice -----
        out_ptrs = out_ptr + \
            b * stride_out_b + \
            i_offset[:, None] * stride_out_i + \
            j_offset[None, :] * stride_out_j + \
            h * stride_out_h
        mask_out = mask_i[:, None] & mask_j[None, :]
        tl.store(out_ptrs, acc.to(tl.float32), mask=mask_out)

# ----------------------------------------------------------------------
# Helper LayerNorm that works on the last dimension only (no bias needed)
# ----------------------------------------------------------------------
def layernorm(x, weight, bias, eps=1e-5):
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias

# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Fast Triton‑based forward pass of the outgoing TriMul module.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣  LayerNorm on the input (dim)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = layernorm(input_tensor, norm_w, norm_b)            # (B,N,N,dim)

    # --------------------------------------------------------------
    # 2️⃣  Linear projections (no bias) – we keep them in fp32
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"]
    )   # (B,N,N,hidden)
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"]
    )   # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 3️⃣  Apply mask (if present) – mask shape [B,N,N] → [...,1]
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                         # (B,N,N,1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4️⃣  Gating (sigmoid) and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣  Core contraction – Triton kernel
    # --------------------------------------------------------------
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # launch configuration
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = hidden   # hidden is small (≤384) – we handle it whole at once

    total_tiles = B * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N)
    grid = (total_tiles,)

    trimul_einsum_kernel[grid](
        left,
        right,
        out,
        B, N, hidden,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣  Final LayerNorm, gating and linear projection back to dim
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = layernorm(out, to_out_norm_w, to_out_norm_b)          # (B,N,N,hidden)
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # (B,N,N,dim)

    return out
