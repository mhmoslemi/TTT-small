"""
TriMul “outgoing” implementation.

The heavy part of TriMul is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a 3‑D batched “inner‑product” over the sequence dimension *k*.
We keep the preprocessing (LayerNorm, Linear, gating) in PyTorch – they are
memory‑bound and already highly optimised – and we fuse the N² × hidden
inner‑products in a custom Triton kernel.

The kernel iterates over *k* in tiles (BLOCK_K) and accumulates a
( BLOCK_M × BLOCK_N × hidden_dim ) tensor in registers, using
`tl.sum(left * right, axis=0)` to perform the reduction over *k*.
After the reduction we store the result back to the output tensor.
All tensors are assumed to be on CUDA and the final output is cast to
float32 (the kernel works in float16 for speed, but the host code converts
to float32 before the final LayerNorm/Linear).

The host‑side `custom_kernel` builds the required tensors, launches the
kernel and then finishes the remaining LayerNorm‑Linear pipeline in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from torch import nn


# ----------------------------------------------------------------------
# Triton kernel – core N² × hidden inner‑product
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          #   *[B, N, N, H]   left projected & gated
    right_ptr,         #   *[B, N, N, H]   right projected & gated
    out_ptr,           #   *[B, N, N, H]   output (still hidden dim)
    B, N, H,
    stride_lb, stride_lk, stride_ld,   # strides for left   (batch, i, k, d)
    stride_rb, stride_rj, stride_rk, stride_rd,   # strides for right (batch, j, k, d)
    stride_ob, stride_oi, stride_oj, stride_od,   # strides for out   (batch, i, j, d)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch index
    pid_m = tl.program_id(1)          # tile over i
    pid_n = tl.program_id(2)          # tile over j

    # offsets within the N×N plane
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds i/j
    mask_m = offs_m < N
    mask_n = offs_n < N

    # offsets for the hidden dimension (process whole hidden dim at once)
    offs_d = tl.arange(0, H)

    # initialise accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, H), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- load left[b, i, k, d] ----------
        # shape (BLOCK_M, BLOCK_K, H)
        left_ptrs = left_ptr + (
            pid_b * stride_lb
            + offs_m[:, None, None] * stride_lk
            + offs_k[None, :, None] * stride_ld
            + offs_d[None, None, :] * 1
        )
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # -------- load right[b, j, k, d] ----------
        # shape (BLOCK_N, BLOCK_K, H)
        right_ptrs = right_ptr + (
            pid_b * stride_rb
            + offs_n[:, None, None] * stride_rj
            + offs_k[None, :, None] * stride_rd
            + offs_d[None, None, :] * 1
        )
        right = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # left: (M,K,H)   right: (N,K,H)
        # compute element‑wise product and sum over K -> (M,N,H)
        # broadcasting over K dimension
        prod = left[:, :, None, :] * right[None, :, :, :]   # (M,K,N,H)
        acc += tl.sum(prod, axis=1)                        # sum over K -> (M,N,H)

    # store result
    out_ptrs = out_ptr + (
        pid_b * stride_ob
        + offs_m[:, None, None] * stride_oi
        + offs_n[None, :, None] * stride_oj
        + offs_d[None, None, :] * stride_od
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None, None] & mask_n[None, :, None])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim]   (float32)
        - mask         : torch.Tensor[bs, N, N]        (float32 or bool)
        - weights      : dict of torch tensors (model parameters)
        - config       : dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ---------- 1. LayerNorm ----------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_w + norm_b          # [bs,N,N,dim]

    # ---------- 2. Linear projections ----------
    # Linear without bias: y = x @ W.T   (dim -> hidden)
    left_proj_w = weights["left_proj.weight"]          # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]        # (hidden, dim)
    left = torch.einsum("bnid,hd->bnih", x, left_proj_w)   # [bs,N,N,hidden]
    right = torch.einsum("bnid,hd->bnih", x, right_proj_w)

    # ---------- 3. Apply mask (if present) ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [bs,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- 4. Gating ----------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.einsum("bnid,hd->bnih", x, left_gate_w).sigmoid()
    right_gate = torch.einsum("bnid,hd->bnih", x, right_gate_w).sigmoid()
    out_gate = torch.einsum("bnid,hd->bnih", x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- 5. Triton fused inner‑product ----------
    # Cast to half for the kernel (H200 is tensor‑core friendly)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    out_h = torch.empty_like(left_h)   # same shape [bs,N,N,hidden]

    # strides (in elements) for the 4‑D tensors
    stride_lb = left_h.stride(0)    # batch
    stride_lk = left_h.stride(1)    # i
    stride_lk = left_h.stride(2)    # k (reuse name)
    stride_ld = left_h.stride(3)    # d

    stride_rb = right_h.stride(0)   # batch
    stride_rj = right_h.stride(1)   # j
    stride_rk = right_h.stride(2)   # k
    stride_rd = right_h.stride(3)   # d

    stride_ob = out_h.stride(0)
    stride_oi = out_h.stride(1)
    stride_oj = out_h.stride(2)
    stride_od = out_h.stride(3)

    # block sizes – tuned for H200, hidden dim ≤ 384
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 32

    grid = (
        bs,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        bs,
        N,
        hidden,
        stride_lb, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # back to float32 for the remaining ops
    out = out_h.to(torch.float32)

    # ---------- 6. to_out_norm ----------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean = out.mean(-1, keepdim=True)
    var = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean) / torch.sqrt(var + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    # ---------- 7. out gate ----------
    out = out * out_gate

    # ---------- 8. final linear ----------
    to_out_w = weights["to_out.weight"]   # (dim, hidden)
    out = torch.einsum("bnih,dh->bnid", out, to_out_w)

    return out
