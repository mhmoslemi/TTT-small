"""
TriMul “outgoing” implementation.

The heavy part is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is reduced over the “k” axis (sequence length).  A custom Triton
kernel computes this reduction for every (b,i,j) while keeping the hidden
dimension `d` (== hidden_dim) in registers.  All other operations
(LayerNorm, Linear, gating, final LayerNorm/Linear) are done with
torch‑ops – they are cheap compared with the O(N³·C) contraction.

The kernel launches one program for each output pair (b,i,j).  Inside the
kernel the `k` dimension is tiled with `BLOCK_K` (default 32) and the
hidden dimension is kept as a compile‑time constant `BLOCK_H`
(default 128).  The accumulation is performed in float32 for accuracy
and the result is written back as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ------------------------------------------------------------------
# Triton kernel
# ------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const float16 *
    right_ptr,         # const float16 *
    out_ptr,           # float32 *
    B, N, H,           # int32 scalars
    # strides of left  [b,i,k,d]
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides of right [b,j,k,d]
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides of out   [b,i,j,d]
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    for a single (b,i,j) program.
    """
    pid = tl.program_id(0)                # one program per (b,i,j)
    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i = ij // N
    j = ij % N

    # ------------------------------------------------------------------
    # offsets for the hidden dimension (stored contiguously)
    # ------------------------------------------------------------------
    offs_h = tl.arange(0, BLOCK_H)                 # [BLOCK_H]

    # accumulator in float32
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over k in tiles
    # ------------------------------------------------------------------
    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_start * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)   # [BLOCK_K]
        mask_k = offs_k < N

        # pointers for the current tile
        left_base = (
            left_ptr
            + b * stride_left_b
            + i * stride_left_i
            + offs_k * stride_left_k
            + offs_h * stride_left_h
        )
        right_base = (
            right_ptr
            + b * stride_right_b
            + j * stride_right_j
            + offs_k * stride_right_k
            + offs_h * stride_right_h
        )

        # load [BLOCK_K, BLOCK_H]
        left = tl.load(left_base, mask=mask_k[:, None], other=0.0)   # fp16
        right = tl.load(right_base, mask=mask_k[:, None], other=0.0) # fp16

        # multiply and reduce over k
        prod = left * right                     # fp16 * fp16 -> fp16
        acc += tl.sum(prod, axis=0)             # sum over k, result fp32

    # ------------------------------------------------------------------
    # write result
    # ------------------------------------------------------------------
    out_base = (
        out_ptr
        + b * stride_out_b
        + i * stride_out_i
        + j * stride_out_j
        + offs_h * stride_out_h
    )
    tl.store(out_base, acc)                     # store fp32


# ------------------------------------------------------------------
# Host‑side entry point
# ------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Args:
        data: (input, mask, weights, config)
            - input:  [B, N, N, dim]  float32
            - mask :  [B, N, N]       (bool / 0‑1)  (may be all‑ones)
            - weights: dict with all linear / norm parameters
            - config : {"dim": dim, "hidden_dim": hidden_dim}
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to apply LayerNorm manually (torch implementation is fast)
    # ------------------------------------------------------------------
    def layer_norm(x, w, b):
        return F.layer_norm(x, (x.shape[-1],), weight=w, bias=b, eps=1e-5)

    # ------------------------------------------------------------------
    # 1) Input norm
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to fp16 for the heavy contraction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Prepare output buffer
    out_fp32 = torch.empty((B, N, N, hidden_dim), dtype=torch.float32,
                           device=device)

    # Strides (in elements) for the three tensors
    stride_left = left_fp16.stride()   # (B,N,N,hidden) -> (N*N*H, N*H, H, 1)
    stride_right = right_fp16.stride()
    stride_out = out_fp32.stride()

    # launch configuration
    BLOCK_K = 32                     # tile over the summed axis
    BLOCK_H = hidden_dim            # hidden_dim is a compile‑time constant
    grid = (triton.cdiv(B * N * N, 1),)   # one program per (b,i,j)

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B, N, hidden_dim,
        # left strides
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        # right strides
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        # out strides
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6) Final LayerNorm, gating and output linear (all in float32)
    # ------------------------------------------------------------------
    out = layer_norm(out_fp32,
                     weights["to_out_norm.weight"],
                     weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
