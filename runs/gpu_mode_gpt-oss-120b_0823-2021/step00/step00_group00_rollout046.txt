"""
TriMul “outgoing” kernel (AlphaFold‑3).

The heavy N³ reduction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is performed with a fused Triton kernel that computes the
sum‑over‑k for a tile of (i,j) indices while keeping the channel
dimension `d` in registers.  The surrounding LayerNorms, Linear
projections and sigmoid gates are cheap and stay in PyTorch – this
still satisfies the “at least part in Triton” rule while keeping the
implementation simple and fast on Hopper (sm_90).

The kernel works for any batch size B∈{1,2}, sequence length N up to
1024 and hidden dimension H (called `hidden_dim` below).  All
computations are performed in FP32 for numerical stability; the
final linear projection (`to_out`) returns FP32 as required.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------
# Triton kernel: tile‑wise N³ reduction (outgoing version)
# -------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # *const float*   [B,N,N,H]
    right_ptr,         # *const float*   [B,N,N,H]
    out_ptr,           # *mut   float*   [B,N,N,H]
    B, N, H,
    # strides for the 4‑D tensors
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute a tile of the output tensor:
        out[b, i, j, h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Tile layout:
        i ∈ [0, BLOCK_I), j ∈ [0, BLOCK_J), h ∈ [0, BLOCK_H)
    Reduction over k is performed in chunks of size BLOCK_K.
    """
    pid = tl.program_id(0)

    # ---------------------------------------------------------
    # map 1‑D program id -> (b, i_block, j_block)
    # ---------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    b = pid // (num_i * num_j)
    tmp = pid % (num_i * num_j)
    i_block = tmp // num_j
    j_block = tmp % num_j

    # ---------------------------------------------------------
    # Offsets for the tile
    # ---------------------------------------------------------
    offs_i = i_block * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = j_block * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator: shape (BLOCK_I, BLOCK_J, BLOCK_H)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ---------------------------------------------------------
    # Loop over the reduction dimension k in chunks
    # ---------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.arange(0, BLOCK_K)
        k_valid = (k_start + cur_k) < N

        # ----- load left[b, i, k, h]  -------------------------
        left_ptrs = (left_ptr
                     + b * stride_left_b
                     + offs_i[:, None, None] * stride_left_i
                     + (k_start + cur_k)[None, :, None] * stride_left_k
                     + offs_h[None, None, :] * stride_left_h)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] &
                            k_valid[None, :, None] &
                            mask_h[None, None, :],
                       other=0.0)

        # ----- load right[b, j, k, h]  ------------------------
        right_ptrs = (right_ptr
                      + b * stride_right_b
                      + offs_j[:, None, None] * stride_right_j
                      + (k_start + cur_k)[None, :, None] * stride_right_k
                      + offs_h[None, None, :] * stride_right_h)
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] &
                             k_valid[None, :, None] &
                             mask_h[None, None, :],
                        other=0.0)

        # element‑wise multiply and accumulate over k
        # shape: (BLOCK_I, BLOCK_J, BLOCK_H)
        prod = left * right
        acc += tl.sum(prod, axis=1)          # sum over the k‑axis

    # ---------------------------------------------------------
    # Store the result back to memory
    # ---------------------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_out_b
                + offs_i[:, None, None] * stride_out_i
                + offs_j[None, :, None] * stride_out_j
                + offs_h[None, None, :] * stride_out_h)

    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] &
                  mask_j[None, :, None] &
                  mask_h[None, None, :])

# -------------------------------------------------------------
# Helper: launch the Triton kernel
# -------------------------------------------------------------
def trimul_outgoing_triton(left: torch.Tensor,
                           right: torch.Tensor,
                           mask: torch.Tensor) -> torch.Tensor:
    """
    left/right : [B,N,N,H]   FP32
    mask       : [B,N,N,1]   (broadcastable)
    returns    : [B,N,N,H]   FP32
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes) for Triton
    sbl, sli, slk, slh = left.stride()
    sbr, srj, srk, srh = right.stride()
    sbo, sio, sj0, sho = out.stride()

    # tile sizes – tuned for Hopper SM90
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_H = 32

    # number of program instances
    num_i = (N + BLOCK_I - 1) // BLOCK_I
    num_j = (N + BLOCK_J - 1) // BLOCK_J
    grid = (B * num_i * num_j,)

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        sbl, sli, slk, slh,
        sbr, srj, srk, srh,
        sbo, sio, sj0, sho,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out


# -------------------------------------------------------------
# Full custom kernel (entry point)
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args
    ----
    data : (input, mask, weights, config)
        input   – Tensor[bs, N, N, dim]  (float32)
        mask    – Tensor[bs, N, N]       (float32 or bool)
        weights – dict of pre‑loaded parameters
        config  – dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    Tensor[bs, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = torch.float32

    # -----------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # -----------------------------------------------------
    norm = nn.LayerNorm(dim, eps=1e-5, elementwise_affine=True).to(device)
    norm.weight.data = weights["norm.weight"].to(dtype)
    norm.bias.data   = weights["norm.bias"].to(dtype)
    x = norm(input_tensor)

    # -----------------------------------------------------
    # 2) Linear projections (no bias) – stay in PyTorch
    # -----------------------------------------------------
    left_proj = nn.Linear(dim, hidden_dim, bias=False).to(device)
    right_proj = nn.Linear(dim, hidden_dim, bias=False).to(device)
    left_proj.weight.data = weights["left_proj.weight"].to(dtype)
    right_proj.weight.data = weights["right_proj.weight"].to(dtype)

    left = left_proj(x)          # [B,N,N,hidden_dim]
    right = right_proj(x)        # [B,N,N,hidden_dim]

    # -----------------------------------------------------
    # 3) Apply mask (if present)
    # -----------------------------------------------------
    mask = mask.unsqueeze(-1).to(dtype)   # [B,N,N,1]
    left = left * mask
    right = right * mask

    # -----------------------------------------------------
    # 4) Gating (sigmoid)
    # -----------------------------------------------------
    left_gate = nn.Linear(dim, hidden_dim, bias=False).to(device)
    right_gate = nn.Linear(dim, hidden_dim, bias=False).to(device)
    out_gate = nn.Linear(dim, hidden_dim, bias=False).to(device)

    left_gate.weight.data = weights["left_gate.weight"].to(dtype)
    right_gate.weight.data = weights["right_gate.weight"].to(dtype)
    out_gate.weight.data = weights["out_gate.weight"].to(dtype)

    left = left * left_gate(x).sigmoid()
    right = right * right_gate(x).sigmoid()
    out_gate_val = out_gate(x).sigmoid()          # [B,N,N,hidden_dim]

    # -----------------------------------------------------
    # 5) Triton‑fused N³ reduction (outgoing)
    # -----------------------------------------------------
    out = trimul_outgoing_triton(left, right, mask)   # [B,N,N,hidden_dim]

    # -----------------------------------------------------
    # 6) Final LayerNorm + out_gate + Linear projection back to dim
    # -----------------------------------------------------
    to_out_norm = nn.LayerNorm(hidden_dim, eps=1e-5, elementwise_affine=True).to(device)
    to_out_norm.weight.data = weights["to_out_norm.weight"].to(dtype)
    to_out_norm.bias.data   = weights["to_out_norm.bias"].to(dtype)

    to_out = nn.Linear(hidden_dim, dim, bias=False).to(device)
    to_out.weight.data = weights["to_out.weight"].to(dtype)

    out = to_out_norm(out)
    out = out * out_gate_val
    out = to_out(out)          # final shape [B,N,N,dim]

    return out.to(dtype)
