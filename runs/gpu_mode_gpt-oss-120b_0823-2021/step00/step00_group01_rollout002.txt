"""
TriMul “outgoing” implementation.

The heavy part of the operator is the three‑tensor contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which is equivalent to a batched matrix multiplication
`out_h = left_h @ right_h.transpose(-1,-2)` for every hidden channel *h*.
The kernel below computes this contraction for a block of
(i, j, h) while streaming over the k‑dimension.  All LayerNorms,
Linear projections and gating are performed in PyTorch (float32);
only the O(N³·H) contraction is moved to Triton.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul over the k‑dimension for a block of h
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,               #   [B, N, N, H]  (i, k, h)
    right_ptr,              #   [B, N, N, H]  (j, k, h)
    out_ptr,                #   [B, N, N, H]  (i, j, h)
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,   # strides of left
    stride_rb, stride_rj, stride_rk, stride_rh,   # strides of right
    stride_ob, stride_oi, stride_oj, stride_oh,   # strides of out
    BLOCK_M: tl.constexpr,   # block size over i
    BLOCK_N: tl.constexpr,   # block size over j
    BLOCK_H: tl.constexpr,   # block size over h
    BLOCK_K: tl.constexpr,   # block size over k
):
    # program ids
    pid_bh = tl.program_id(0)                 # combines batch and h‑block
    pid_m  = tl.program_id(1)                 # i‑block
    pid_n  = tl.program_id(2)                 # j‑block

    # decompose pid_bh -> batch and h‑block
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    b = pid_bh // num_h_blocks
    hb = pid_bh % num_h_blocks

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = hb * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_h = offs_h < H

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # stream over k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left:  (M, K, H)
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + offs_m[:, None, None] * stride_li
            + offs_k[None, :, None] * stride_lk
            + offs_h[None, None, :] * stride_lh
        )
        left = tl.load(
            left_ptrs,
            mask=mask_m[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # right: (N, K, H)
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + offs_n[:, None, None] * stride_rj
            + offs_k[None, :, None] * stride_rk
            + offs_h[None, None, :] * stride_rh
        )
        right = tl.load(
            right_ptrs,
            mask=mask_n[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # contract over k for each h‑slice
        for dh in range(0, BLOCK_H):
            # shape (M, K) and (N, K)
            l = left[:, :, dh]
            r = right[:, :, dh]
            # r needs to be transposed: (K, N)
            acc[:, :, dh] += tl.dot(l, r, trans_b=True)

    # write back
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + offs_m[:, None, None] * stride_oi
        + offs_n[None, :, None] * stride_oj
        + offs_h[None, None, :] * stride_oh
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_m[:, None, None] & mask_n[None, :, None] & mask_h[None, None, :],
    )


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Execute the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, D]  (float32)
        - mask         : [B, N, N]      (float32 or bool, will be broadcast)
        - weights      : dict of pre‑trained parameters
        - config       : {"dim": D, "hidden_dim": H, ...}
    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. PyTorch part – LayerNorm, linear projections, gating
    # --------------------------------------------------------------
    # LayerNorm on the last dim
    norm = nn.LayerNorm(D, elementwise_affine=False).to(input_tensor.device)
    norm.weight = nn.Parameter(weights["norm.weight"])
    norm.bias   = nn.Parameter(weights["norm.bias"])

    x = norm(input_tensor)

    # Linear projections (bias‑free)
    left_proj = nn.Linear(D, H, bias=False).to(input_tensor.device)
    right_proj = nn.Linear(D, H, bias=False).to(input_tensor.device)
    left_proj.weight = nn.Parameter(weights["left_proj.weight"])
    right_proj.weight = nn.Parameter(weights["right_proj.weight"])

    left = left_proj(x)
    right = right_proj(x)

    # apply mask if present
    if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(left)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # gating
    left_gate = nn.Linear(D, H, bias=False).to(input_tensor.device)
    right_gate = nn.Linear(D, H, bias=False).to(input_tensor.device)
    out_gate = nn.Linear(D, H, bias=False).to(input_tensor.device)

    left_gate.weight = nn.Parameter(weights["left_gate.weight"])
    right_gate.weight = nn.Parameter(weights["right_gate.weight"])
    out_gate.weight = nn.Parameter(weights["out_gate.weight"])

    left = left * left_gate(x).sigmoid()
    right = right * right_gate(x).sigmoid()
    out_gate_val = out_gate(x).sigmoid()               # [B,N,N,H]

    # --------------------------------------------------------------
    # 2. Triton kernel – the N³·H contraction
    # --------------------------------------------------------------
    # output buffer
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=input_tensor.device)

    # strides (in elements) for the 4‑D tensors
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # block sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_H = 32               # fits hidden dim up to 384 (12 blocks)
    BLOCK_K = 32

    grid = (
        B * triton.cdiv(H, BLOCK_H),   # batch × h‑blocks
        triton.cdiv(N, BLOCK_M),      # i‑blocks
        triton.cdiv(N, BLOCK_N),      # j‑blocks
    )

    trimul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 3. Final LayerNorm + out‑gate + linear
    # --------------------------------------------------------------
    to_out_norm = nn.LayerNorm(H, elementwise_affine=False).to(input_tensor.device)
    to_out_norm.weight = nn.Parameter(weights["to_out_norm.weight"])
    to_out_norm.bias   = nn.Parameter(weights["to_out_norm.bias"])

    to_out = nn.Linear(H, D, bias=False).to(input_tensor.device)
    to_out.weight = nn.Parameter(weights["to_out.weight"])

    out = to_out_norm(out)                # [B,N,N,H] -> normalized
    out = out * out_gate_val               # gating
    out = to_out(out)                      # back to D

    return out
