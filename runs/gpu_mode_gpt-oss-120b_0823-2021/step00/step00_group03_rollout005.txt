"""
TriMul “outgoing” implementation.

The expensive part of the forward pass is the contraction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

For every batch * b * and hidden channel * d * this is a dense
matrix‑matrix multiply of two (N×N) matrices (the second one is taken
with the k‑dimension as the column).  The kernel below performs a
batched fp16 mat‑mul over the (b, d) dimensions and writes the result
in fp32.  All layer‑norms and linear layers are kept in PyTorch – they
are cheap compared to the O(N³) contraction and keep the code short.
The final output is returned in float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel : batched mat‑mul over the k‑dimension
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,               # const float16 *
    right_ptr,              # const float16 *
    out_ptr,                #       float32 *
    B, N, H,                # int32 scalars
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_M: tl.constexpr,  # tile size in i
    BLOCK_N: tl.constexpr,  # tile size in j
    BLOCK_K: tl.constexpr,  # tile size in k
):
    """
    Compute, for every (b, d) pair,  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    using a tiled mat‑mul.  The kernel is launched with a 1‑D grid that
    enumerates (b, d, tile_i, tile_j).
    """
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # map 1‑D program id to (b, d, tile_i, tile_j)
    # --------------------------------------------------------------
    total_i = tl.cdiv(N, BLOCK_M)
    total_j = tl.cdiv(N, BLOCK_N)
    blocks_per_pair = total_i * total_j               # tiles for a (b,d) pair
    pairs = B * H                                      # total (b,d) pairs
    pair_id = pid // blocks_per_pair
    tile_id = pid % blocks_per_pair

    b = pair_id // H
    d = pair_id % H

    tile_i = tile_id // total_j
    tile_j = tile_id % total_j

    # --------------------------------------------------------------
    # offsets for the current tile
    # --------------------------------------------------------------
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # pointers for the A (left) and B (right) tiles
    # left[b, i, k, d]
    a_ptrs = left_ptr + \
        b * stride_lb + \
        offs_i[:, None] * stride_li + \
        offs_k[None, :] * stride_lk + \
        d * stride_ld

    # right[b, j, k, d]   (note that k is the *column* dimension)
    b_ptrs = right_ptr + \
        b * stride_rb + \
        offs_j[:, None] * stride_rj + \
        offs_k[None, :] * stride_rk + \
        d * stride_rd

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # main k‑loop
    # --------------------------------------------------------------
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (offs_k[None, :] < N), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & (offs_k[None, :] < N), other=0.0).to(tl.float32)
        acc += tl.dot(a, b)                     # (M,K)·(K,N) → (M,N)

        # advance pointers
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    # --------------------------------------------------------------
    # write result
    # --------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_ob + \
        offs_i[:, None] * stride_oi + \
        offs_j[None, :] * stride_oj + \
        d * stride_od

    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    Only the O(N³) contraction is performed in Triton; everything else runs
    in native PyTorch (layer‑norm, linear layers, sigmoids, etc.).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Layer‑norm on the last dim (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)  ->  [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(dtype))          # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # ------------------------------------------------------------------
    # 3) Apply mask if present
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton contraction  (fp16 inside kernel, fp32 output)
    # ------------------------------------------------------------------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # strides (contiguous layout B,N,N,H)
    stride_lb, stride_li, stride_lk, stride_ld = left_f16.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_f16.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # tile sizes – tuned for Hopper (sm_90)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64

    total_i = (N + BLOCK_M - 1) // BLOCK_M
    total_j = (N + BLOCK_N - 1) // BLOCK_N
    blocks_per_pair = total_i * total_j
    grid = (B * hidden_dim * blocks_per_pair,)

    trimul_kernel[grid](
        left_f16,
        right_f16,
        out,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing (layer‑norm, out‑gate, final linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
