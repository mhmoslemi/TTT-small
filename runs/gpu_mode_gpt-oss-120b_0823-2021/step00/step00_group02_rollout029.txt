"""
TriMul “outgoing” kernel (AlphaFold3).

The heavy part of TriMul is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each batch‐and‑head (b,d) this is a matrix multiplication:
    L_{b,d} : (N,N)   (i,k)
    R_{b,d} : (N,N)   (j,k)   → we need R_{b,d}^T
    out_{b,d} = L_{b,d} @ R_{b,d}^T                (i,j)

We implement this N×N matrix multiply for many (B·H) independent
matrices with a fused Triton kernel.  All other ops
(LayerNorm, Linear, sigmoid‑gates, final LayerNorm/Linear) are kept in
PyTorch – they are inexpensive compared with the O(N³) work of the
einsum.

The kernel works in FP16 for the bulk of the work (the input tensors are
cast to fp16) and accumulates in FP32 for numerical stability.  The final
output tensor is cast back to fp32 as required by the reference model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------
# Triton kernel: batched matmul A @ B^T  (A,B shape: [batch, N, N])
# --------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,                              # matrix size
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # 2‑D grid layout (m‑blocks × n‑blocks)
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_n
    pid_n = pid % num_n

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointer to the batch we are processing (batch id == program id)
    a_batch_ptr = a_ptr + pid * stride_a_batch
    b_batch_ptr = b_ptr + pid * stride_b_batch
    c_batch_ptr = c_ptr + pid * stride_c_batch

    # initialise accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k‑dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_row
                               + offs_k[None, :] * stride_a_col)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float16)

        # load B block, but we need B^T → we load (BLOCK_N, BLOCK_K) transposed
        b_ptrs = b_batch_ptr + (offs_n[:, None] * stride_b_row
                               + offs_k[None, :] * stride_b_col)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)
        b = b.to(tl.float16)

        # dot‑product: result accumulates in FP32
        acc += tl.dot(a, b, out_dtype=tl.float32)

        # advance k‑pointers (implicit via loop index)

    # write back C
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_row
                            + offs_n[None, :] * stride_c_col)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------
# Helper: LayerNorm without bias (bias is added afterwards)
# --------------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias


# --------------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, C]   torch.float32
            - mask_tensor  : [B, N, N]      torch.float32 (0/1)
            - weights_dict : mapping of all Linear/LayerNorm params
            - config_dict  : {"dim":C, "hidden_dim":H, ...}

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]

    # ----------------------------------------------------------------
    # 1) LayerNorm on input (C‑dim)
    # ----------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # ensure fp32 for the linear layers (they expect fp32 weights)
    x = x.to(torch.float32)

    # ----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ----------------------------------------------------------------
    # Shape after projection: [B,N,N,H]
    left = F.linear(x, weights["left_proj.weight"])          # (C,H)
    right = F.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------
    # 3) Apply mask (if any)
    # ----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ----------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # 5) Core einsum via Triton: out = left @ right^T per head
    # ----------------------------------------------------------------
    # Cast to fp16 for the kernel (Hopper has fast fp16 compute)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # reshape to [B*H, N, N] so each (batch,head) pair is independent
    left_mat = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # strides
    stride_a_batch, stride_a_row, stride_a_col = left_mat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_mat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_mat.stride()

    # launch parameters (tuned for Hopper)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    total_batches = B * H
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (grid_m * grid_n,)

    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B, N, N, H] (still fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)          # back to fp32 for following layers

    # ----------------------------------------------------------------
    # 6) Post‑processing LayerNorm + gating + final Linear
    # ----------------------------------------------------------------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
