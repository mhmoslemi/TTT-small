"""
TriMul “outgoing” implementation.

*   LayerNorm, Linear projections and gates are performed in PyTorch
    (they are cheap compared to the O(N³) kernel).

*   The only heavy part – the ternary multiplicative update
    `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]` –
    is realized by a fused batched‑matmul kernel written in Triton.
    The operation is mathematically identical to
        out = left @ right.transpose(-2,-1)
    where the hidden dimension `d` is treated as an extra batch axis.
    The kernel works on a 3‑D view `(B*D, N, N)` and computes
    C = A @ Bᵀ with tiling in the i‑ and j‑axes.
    All arithmetic is done in FP16 for speed; the final result is
    cast back to FP32 before the post‑norm/out‑linear layers.

The implementation supports all shapes required by the test matrix
(e.g. B∈{1,2}, N up to 1024, hidden_dim = 128, dim up to 384) and works
with or without a mask.
"""

import torch
import triton
import triton.language as tl
from torch import nn


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ  (A,B shape = (B*D, N, N))
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_T_kernel(
    A_ptr,                # *const fp16
    B_ptr,                # *const fp16
    C_ptr,                # *fp16   (output)
    BATCH,                # total batches = B * hidden_dim   (int32)
    N,                    # sequence length                 (int32)
    stride_Ab, stride_An, stride_Ak,   # strides for A
    stride_Bb, stride_Bn, stride_Bk,   # strides for B
    stride_Cb, stride_Ci, stride_Cj,   # strides for C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr = 8):
    """
    Computes C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]   (i.e. A @ Bᵀ)
    The kernel tiles over i (BLOCK_M) and j (BLOCK_N) and iterates over k
    in BLOCK_K steps.
    """

    pid = tl.program_id(axis=0)          # linear program id over (batch, i, j) tiles
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # --------------------------------------------------------------
    # 1) map pid -> (batch, tile_i, tile_j) with grouping to improve cache
    # --------------------------------------------------------------
    batch_pid = pid // (num_pid_m * num_pid_n)
    pid_rem   = pid %  (num_pid_m * num_pid_n)
    tile_i = pid_rem // num_pid_n
    tile_j = pid_rem %  num_pid_n

    # offsets for the current tile
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers for the current batch
    a_batch_ptr = A_ptr + batch_pid * stride_Ab
    b_batch_ptr = B_ptr + batch_pid * stride_Bb
    c_batch_ptr = C_ptr + batch_pid * stride_Cb

    # allocate accumulator in FP32 for numeric stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 2) loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A[i,k]   (shape BLOCK_M x BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_i[:, None] * stride_An
                               + offs_k[None, :] * stride_Ak)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B[j,k]   (shape BLOCK_N x BLOCK_K) – note we need Bᵀ, so we load row‑wise
        b_ptrs = b_batch_ptr + (offs_j[:, None] * stride_Bn
                               + offs_k[None, :] * stride_Bk)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # accumulate A[i,k] * B[j,k]
        acc += tl.dot(a, b, trans_b=False)   # (M,K)·(N,K)ᵀ -> (M,N)

    # --------------------------------------------------------------
    # 3) write back
    # --------------------------------------------------------------
    c_ptrs = c_batch_ptr + (offs_i[:, None] * stride_Ci
                           + offs_j[None, :] * stride_Cj)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def batched_matmul_T(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : float16 tensors of shape (B, N, N, D)
    returns      : float16 tensor of shape (B, N, N, D) = left @ rightᵀ per D
    """
    B, N, _, D = left.shape
    # view as (B*D, N, N)
    A = left.permute(0, 3, 1, 2).reshape(B * D, N, N).contiguous()
    B_ = right.permute(0, 3, 1, 2).reshape(B * D, N, N).contiguous()
    C = torch.empty_like(A)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(B * D, 1) *
            triton.cdiv(N, BLOCK_M) *
            triton.cdiv(N, BLOCK_N),)

    batched_matmul_T_kernel[grid](
        A, B_, C,
        B * D, N,
        A.stride(0), A.stride(1), A.stride(2),
        B_.stride(0), B_.stride(1), B_.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K)

    # reshape back to (B,N,N,D)
    C = C.view(B, D, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Full TriMul forward (torch for cheap ops, triton for the heavy einsum)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights_dict, config_dict)
        input_tensor : torch.Tensor   [B, N, N, dim]   (float32, on CUDA)
        mask         : torch.Tensor   [B, N, N]        (float32, on CUDA)
        weights      : dict of torch.Tensor containing all linear / norm params
        config       : dict with keys ["dim", "hidden_dim"]
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1) LayerNorm on the last dim (dim)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5)
    x = x * norm_weight + norm_bias

    # --------------------------------------------------------------
    # 2) Linear projections (no bias) – we keep them in FP32
    # --------------------------------------------------------------
    def linear_no_bias(x, w):
        # w : (out_features, in_features)
        return torch.nn.functional.linear(x, w)

    left  = linear_no_bias(x, weights["left_proj.weight"])
    right = linear_no_bias(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Apply mask (if present)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)            # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4) Gates (sigmoid) and element‑wise gating
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(linear_no_bias(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_no_bias(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear_no_bias(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Heavy part: Σ_k left[...,k,d] * right[...,k,d]  (einsum)
    #    Cast to fp16, run triton kernel, cast back to fp32
    # --------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = batched_matmul_T(left_fp16, right_fp16)   # shape [B,N,N,hidden_dim]

    out = out_fp16.to(torch.float32)                     # back to FP32

    # --------------------------------------------------------------
    # 6) Post‑norm, out‑gate and final linear
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
