"""
TriMul “outgoing” implementation.

The expensive part is the N‑wise “pairwise‑multiplication‑sum’’:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We fuse this N³ operation in a Triton kernel.  The rest (LayerNorm,
Linear projections, sigmoids and the final Linear + LayerNorm) are kept in
PyTorch because they are cheap compared with the N³ work.

The kernel works for any batch size ≤2, any sequence length N, and any hidden
dimension H (we assume H is a multiple of BLOCK_D).  It accumulates in FP32
and finally stores FP32 results.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel:  out = Σ_k left[...,i,k,d] * right[...,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          #  [B, N, N, H]  (i,k,d)
    right_ptr,         #  [B, N, N, H]  (j,k,d)
    out_ptr,           #  [B, N, N, H]  (i,j,d)
    B, N, H,
    stride_lbn, stride_lbk, stride_lbd,   # left strides: batch, i, k, d
    stride_rbn, stride_rbk, stride_rbd,   # right strides: batch, j, k, d
    stride_obn, stride_obi, stride_obj, stride_obd,  # out strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # ------------------------------------------------------------------
    # Offsets for the block we are computing
    # ------------------------------------------------------------------
    i_off = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    k_off = tl.arange(0, BLOCK_K)

    d_off = tl.arange(0, BLOCK_D)

    # Masks for out‑of‑bounds
    i_mask = i_off < N
    j_mask = j_off < N
    d_mask = d_off < H

    # ------------------------------------------------------------------
    # Pointers for the current batch
    # ------------------------------------------------------------------
    left_batch_ptr  = left_ptr  + pid_b * stride_lbn
    right_batch_ptr = right_ptr + pid_b * stride_rbn
    out_batch_ptr   = out_ptr   + pid_b * stride_obn

    # ------------------------------------------------------------------
    # Accumulator (float32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k_tile in range(0, N, BLOCK_K):
        cur_k = k_tile + k_off
        k_mask = cur_k < N

        # ------------------------------------------------------------------
        # Load left[i, k, d]  -> shape (I, K, D)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_batch_ptr
            + i_off[:, None, None] * stride_lbk   # i dimension
            + cur_k[None, :, None] * stride_lbd   # k dimension
            + d_off[None, None, :] * stride_lbd   # d dimension (stride same as k‑dim because contiguous in d)
        )
        # Because the tensor is [B,N,N,H] (i,k,d) we need stride for k and d separately:
        # stride_lbk : stride between k entries
        # stride_lbd : stride between d entries
        # In a contiguous layout stride_lbk = H * stride_lbd, but we keep both explicit.
        left_tile = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :], other=0.0)

        # ------------------------------------------------------------------
        # Load right[j, k, d] -> shape (J, K, D)
        # ------------------------------------------------------------------
        right_ptrs = (
            right_batch_ptr
            + j_off[:, None, None] * stride_rbk   # j dimension
            + cur_k[None, :, None] * stride_rbd   # k dimension
            + d_off[None, None, :] * stride_rbd
        )
        right_tile = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :], other=0.0)

        # ------------------------------------------------------------------
        # Compute dot for each d slice:
        #   acc[i,j,d] += Σ_k left[i,k,d] * right[j,k,d]
        # ------------------------------------------------------------------
        # left_tile : (I, K, D)   -> we iterate over D in blocks
        # right_tile: (J, K, D)
        for d_start in range(0, BLOCK_D, 1):          # unrolled over D (BLOCK_D is small)
            l = left_tile[:, :, d_start]   # (I, K)
            r = right_tile[:, :, d_start]  # (J, K)
            # tl.dot expects (M,K) @ (K,N) -> (M,N)
            acc[:, :, d_start] += tl.dot(l, r)   # (I,J)

    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    out_ptrs = (
        out_batch_ptr
        + i_off[:, None, None] * stride_obi   # i
        + j_off[None, :, None] * stride_obj   # j
        + d_off[None, None, :] * stride_obd   # d
    )
    tl.store(out_ptrs,
             acc.to(tl.float32),
             mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :])

# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:          torch.Tensor [B, N, N]        (float32 or bool)
            - weights: dict of torch.Tensor containing all linear / norm weights
            - config:  dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask if present (mask shape [B,N,N], expand dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).type_as(left)   # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton kernel to compute the N³ pairwise multiplication‑sum.
    #    Output shape: [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes) for the contiguous tensors
    stride_lbn, stride_lbi, stride_lbk, stride_lbd = left.stride()
    stride_rbn, stride_rbj, stride_rbk, stride_rbd = right.stride()
    stride_obn, stride_obi, stride_obj, stride_obd = out.stride()

    # Choose block sizes (tuned for Hopper, H200)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    BLOCK_D = 32                     # hidden_dim is a multiple of 32 in all tests

    grid = (
        B,
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
    )

    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, hidden,
        stride_lbn, stride_lbi, stride_lbk, stride_lbd,
        stride_rbn, stride_rbj, stride_rbk, stride_rbd,
        stride_obn, stride_obi, stride_obj, stride_obd,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6) Final LayerNorm + gating + Linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
