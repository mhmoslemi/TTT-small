"""
TriMul “outgoing” kernel (AlphaFold‑3)

We keep the inexpensive LayerNorm / Linear / sigmoid work in PyTorch and
off‑load the only N³‑like operation to Triton:

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

For a fixed hidden dimension `h` this is exactly a matrix multiplication

    out_h = left_h @ right_h.T                (shape N×N)

Thus we can treat the (batch, hidden) axes as a fused outer batch and run
a batched GEMM with a single Triton kernel.  The kernel is a classic
persistent‑workspace matmul (see the example in the statement) but
extended to 3‑D batch indexing.  All tensors are FP16 inside the kernel
and cast back to FP32 for the final linear projection.

The host function `custom_kernel` builds the needed weight tensors,
applies the LayerNorm / Linear / sigmoid gates, launches the Triton
kernel and finishes the forward pass with the output‑norm and output
linear (both in PyTorch).  The implementation works for any
`batch ∈ {1,2}`, `seq_len ∈ {128,…,1024}` and arbitrary `dim`,
`hidden_dim` (the reference tests use 128/384‑dim inputs and 128 hidden).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------
# Triton batched GEMM kernel (left @ right.T) for FP16 inputs
# -------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    # pointers
    left_ptr,          # [B*H, N, N]   fp16
    right_ptr,         # [B*H, N, N]   fp16 (will be read transposed)
    out_ptr,           # [B*H, N, N]   fp16
    # sizes
    N,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch,  stride_out_m,  stride_out_n,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,
):
    """Batched GEMM: out = left @ right.T  (fp16)."""

    pid = tl.program_id(0)        # row tile
    pid_n = tl.program_id(1)      # col tile
    pid_b = tl.program_id(2)      # fused batch (batch * hidden)

    # ---------------------------------------------------------
    # 2‑D tiling with 1‑D grouping for better L2 reuse
    # ---------------------------------------------------------
    num_pid_m = tl.cdiv(N, BLOCK_M)
    group_id = (pid * GROUP_SIZE_M + pid_n) // num_pid_m
    pid_m = group_id * GROUP_SIZE_M + (pid % GROUP_SIZE_M)
    pid_n = (pid * GROUP_SIZE_M + pid_n) % num_pid_m

    # compute offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointer arithmetic for the current batch
    left_batch_offset  = pid_b * stride_left_batch
    right_batch_offset = pid_b * stride_right_batch
    out_batch_offset   = pid_b * stride_out_batch

    # base pointers for the tile
    a_ptrs = left_ptr + left_batch_offset + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
    b_ptrs = right_ptr + right_batch_offset + offs_n[None, :] * stride_right_n + offs_k[:, None] * stride_right_k

    # accumulate in fp32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < N), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_n[None, :] & (offs_k[:, None] < N), other=0.0).to(tl.float32)
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # cast back to fp16 and store
    c = acc.to(tl.float16)
    c_ptrs = out_ptr + out_batch_offset + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------
# Helper: fused LayerNorm (weight & bias) in FP32 → FP16
# -------------------------------------------------------------
def layer_norm_fp32(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """Standard LayerNorm (no bias‑grad) – returns FP16."""
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    out = x_norm * weight + bias
    return out.to(torch.float16)


# -------------------------------------------------------------
# Main entry point expected by the grader
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, D]   float32
        - mask         : [B, N, N]      (0/1)  float32 or bool
        - weights      : dict of learned parameters
        - config       : dict with keys `dim`, `hidden_dim`

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    hidden = config["hidden_dim"]

    # -------------------------------------------------
    # 1.  LayerNorm + Linear projections (all in PyTorch)
    # -------------------------------------------------
    # LayerNorm on the last dimension
    x = layer_norm_fp32(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )   # -> BF16 (actually fp16)

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"]).to(torch.float16)
    right = torch.nn.functional.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # Apply mask if present
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"])).to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 2.  Triton batched GEMM: out_h = left_h @ right_h.T
    # -------------------------------------------------
    # reshape to (B*hidden, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)   # [B*H, N, N]
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)  # same layout

    out_flat = torch.empty_like(left_flat)   # fp16 output

    # strides (in elements)
    stride_l_batch = left_flat.stride(0)
    stride_l_m = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    # right will be read transposed, so swap m↔n in strides
    stride_r_batch = right_flat.stride(0)
    stride_r_n = right_flat.stride(1)   # corresponds to original m after transpose
    stride_r_k = right_flat.stride(2)

    stride_o_batch = out_flat.stride(0)
    stride_o_m = out_flat.stride(1)
    stride_o_n = out_flat.stride(2)

    # Tile sizes (tuned for Hopper)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    GROUP_SIZE_M = 8

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden,
    )
    tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_n,
        stride_r_k,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
    )

    # reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # -------------------------------------------------
    # 3.  Post‑processing (LayerNorm, gating, final linear)
    # -------------------------------------------------
    # to_out_norm on hidden dimension
    out = layer_norm_fp32(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )   # fp16

    out = out * out_gate   # broadcast over hidden dim

    # final linear (hidden -> dim)
    out = torch.nn.functional.linear(out, weights["to_out.weight"]).to(torch.float32)

    return out
