"""
TriMul “outgoing” implementation.

The expensive part of the module is the Einstein‐summation

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden dimension `d` this is exactly a batched matrix‑multiply:
    out_d = left_d @ right_d.T   (left_d : B×N×N , right_d : B×N×N)

We keep the LayerNorm/Linear/gating logic in PyTorch (they are cheap) and
off‑load the N³ operation to a single Triton kernel that performs a
batched GEMM for *all* hidden dimensions in one launch.  The kernel works
in FP16 for the inner multiply‑accumulate and converts the result back to
FP32, matching the reference implementation.

The kernel grid is three‑dimensional:
    dim0 – batch index
    dim1 – hidden‑dimension block
    dim2 – tiling over the (i,j) matrix (M×N)

Block sizes are chosen for Hopper (sm_90) – 64×64 tiles with a K‑tile of
64.  The kernel is fully fused: loads, masking (optional), dot‑product and
store.  The final FP32 post‑processing (LayerNorm, gating, output‑projection)
is performed with native PyTorch ops.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------------
# Triton kernel: batched matmul for all hidden channels at once
# --------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,          # B×N×N×H  (FP16)
    right_ptr,         # B×N×N×H  (FP16)
    out_ptr,           # B×N×N×H  (FP32)
    # shapes
    B, N, H,
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # mask pointer (optional, 0 = no mask)
    mask_ptr,
    # compile‑time constants
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch
    pid_h = tl.program_id(1)          # hidden‑dim block
    pid_mn = tl.program_id(2)         # tile over (i,j)

    # compute tile offsets
    num_m_tiles = tl.cdiv(N, BLOCK_M)
    i_tile = pid_mn // num_m_tiles
    j_tile = pid_mn % num_m_tiles

    offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows/cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ----------------------------------------------------------------
    # accumulate over k
    # ----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # k‑loop
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for the current k‑tile
        left_tile_ptr = left_ptr + \
            pid_b * stride_left_b + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k + \
            pid_h * BLOCK_K * stride_left_h   # each hidden block is contiguous

        right_tile_ptr = right_ptr + \
            pid_b * stride_right_b + \
            offs_j[:, None] * stride_right_i + \
            offs_k[None, :] * stride_right_k + \
            pid_h * BLOCK_K * stride_right_h

        # load (FP16)
        left = tl.load(left_tile_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        right = tl.load(right_tile_ptr, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # if a mask over the pairwise matrix is provided we apply it now
        if HAS_MASK:
            # mask shape: B×N×N (broadcasted over H)
            mask_tile_ptr = mask_ptr + \
                pid_b * N * N + \
                offs_i[:, None] * N + \
                offs_j[None, :]
            m = tl.load(mask_tile_ptr, mask=mask_i[:, None] & mask_j[None, :], other=0.0)
            # mask is 0/1 – we simply zero‑out the contribution of masked positions
            left = left * m[:, None]

        # dot‑product over k
        acc += tl.dot(left, right)   # (M,N)

    # write result (FP32)
    out_tile_ptr = out_ptr + \
        pid_b * stride_out_b + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j + \
        pid_h * BLOCK_K * stride_out_h

    tl.store(out_tile_ptr, acc, mask=mask_i[:, None] & mask_j[None, :])

# --------------------------------------------------------------------
# Helper: LayerNorm (FP32) without torch.nn.LayerNorm to avoid extra kernels
# --------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x : [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias

# --------------------------------------------------------------------
# Entry point
# --------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]  (float32)
            - mask:        [B, N, N]        (float32 or bool)
            - weights:     dict of all linear / layer‑norm parameters
            - config:      {'dim': int, 'hidden_dim': int}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # --------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    ).to(torch.float16)         # cast to fp16 for the cheap linear layers

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) + gating
    # --------------------------------------------------------------
    # shape: [B,N,N,hidden]  (fp16)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"]).to(torch.float16)
    right = torch.nn.functional.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # apply mask (if present)
    has_mask = mask is not None and mask.numel() > 0
    if has_mask:
        mask_ = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # gates (sigmoid, fp16)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"])).to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 3. Triton kernel for the N³ reduction (fp16→fp32)
    # --------------------------------------------------------------
    # Prepare output buffer
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides (in elements)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # constants
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B,                                    # batch dimension
        triton.cdiv(hidden, BLOCK_K),         # hidden‑dim blocks
        triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N)   # tiles over (i,j)
    )

    # optional mask pointer (0 if not used)
    mask_ptr = mask.data_ptr() if has_mask else 0

    trimul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        mask_ptr,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HAS_MASK=int(has_mask),
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 4. Post‑processing (LayerNorm, gating, final projection)
    # --------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate   # out_gate is still fp16; promote
    out = torch.nn.functional.linear(out, weights["to_out.weight"]).to(torch.float32)

    return out
