"""
TriMul “outgoing” implementation.

The heavy N³ part (the einsum `left[i,k,d] * right[j,k,d] → out[i,j,d]`)
is expressed as a batched matrix‑multiply for every hidden dimension `d`:

    out[b, d, i, j] = Σ_k left[b, i, k, d] * right[b, j, k, d]

We flatten the batch and hidden dimensions (B*H) and launch a single
Triton kernel that behaves like a regular GEMM with shape
    (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)

All the surrounding LayerNorms, Linear projections, gates and the final
LayerNorm/Linear are cheap and stay in PyTorch.  The kernel works in
float16 (the model runs in FP16/BF16) and returns float32 results, which
matches the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM (no bias, no activation)
# --------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr,               # left   [B*H, N, K]   (K == N)
    B_ptr,               # right  [B*H, N, K]   (will be read as K, N)
    C_ptr,               # out    [B*H, N, N]
    stride_ah, stride_ak, stride_ab,   # strides for A
    stride_bk, stride_bn, stride_bb,   # strides for B (note: B is read as KxN)
    stride_ch, stride_cn, stride_cc,   # strides for C
    BATCH_H,            # B*H flattened dimension
    N, K,               # matrix sizes (K == N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid = tl.program_id(0)
    batch_head = pid // (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_rem   = pid % (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = pid_rem // tl.cdiv(N, BLOCK_N)          # block row
    pid_n = pid_rem % tl.cdiv(N, BLOCK_N)          # block col

    # offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers for this batch‑head
    a_ptrs = A_ptr + batch_head * stride_ah + offs_m[:, None] * stride_ak + offs_k[None, :] * stride_ab
    b_ptrs = B_ptr + batch_head * stride_bk + offs_k[:, None] * stride_bn + offs_n[None, :] * stride_bb
    c_ptrs = C_ptr + batch_head * stride_ch + offs_m[:, None] * stride_cn + offs_n[None, :] * stride_cc

    # accumulation in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # a,b are fp16 → promote to fp32 inside dot
        acc += tl.dot(a, b)   # (M,K)·(K,N) → (M,N)
        # advance pointers
        a_ptrs += BLOCK_K * stride_ab
        b_ptrs += BLOCK_K * stride_bn

    # write back
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------
# Helper to launch the kernel
# --------------------------------------------------------------
def batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]  (float16 or bfloat16)
    returns out : [B, N, N, H]   (float32)
    """
    B, N, _, H = left.shape
    # flatten batch & hidden as one leading dimension
    LH = B * H
    left_ = left.permute(0, 3, 1, 2).contiguous().view(LH, N, N)   # [B*H, N, K]
    right_ = right.permute(0, 3, 1, 2).contiguous().view(LH, N, N) # [B*H, N, K]

    out = torch.empty_like(left_, dtype=torch.float16)

    # strides (in elements)
    stride_ah = left_.stride(0)
    stride_ak = left_.stride(1)
    stride_ab = left_.stride(2)

    stride_bk = right_.stride(0)
    stride_bn = right_.stride(1)
    stride_bb = right_.stride(2)

    stride_ch = out.stride(0)
    stride_cn = out.stride(1)
    stride_cc = out.stride(2)

    # launch config
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_m = triton.cdiv(N, BLOCK_M)
    num_n = triton.cdiv(N, BLOCK_N)
    total_blocks = LH * num_m * num_n

    grid = (total_blocks,)

    _batched_gemm_kernel[grid](
        left_.data_ptr(),
        right_.data_ptr(),
        out.data_ptr(),
        stride_ah, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_ch, stride_cn, stride_cc,
        LH,
        N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # reshape back to original layout and cast to float32
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().float()
    return out

# --------------------------------------------------------------
# Full custom kernel entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   torch.float32
            - mask         : [B, N, N]        (bool or 0/1) – may be omitted
            - weights      : dict of model parameters
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ---------- LayerNorm ----------
    # LayerNorm weight & bias are 1‑D (dim)
    eps = 1e-5
    # manual LN to keep everything on device without extra Python overhead
    mean = input_tensor.mean(-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(-1, keepdim=True)
    inv_std = torch.rsqrt(var + eps)
    x = (input_tensor - mean) * inv_std
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # ---------- Linear projections (FP16 for speed) ----------
    x_fp16 = x.to(torch.float16)

    left = F.linear(x_fp16, weights["left_proj.weight"]).view_as(x_fp16[..., :hidden_dim])
    right = F.linear(x_fp16, weights["right_proj.weight"]).view_as(x_fp16[..., :hidden_dim])

    # ---------- Mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Core N³ einsum (Triton) ----------
    # left/right are [B,N,N,hidden_dim] in fp16
    out = batched_gemm(left, right)          # -> float32, shape [B,N,N,hidden_dim]

    # ---------- Output projection ----------
    out = F.layer_norm(out, (hidden_dim,), weights["to_out_norm.weight"],
                       weights["to_out_norm.bias"], eps=eps)
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
