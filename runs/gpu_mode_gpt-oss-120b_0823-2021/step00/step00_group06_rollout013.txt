"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The heavy part of the forward pass is the tensor contraction

    out[b, i, j, d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* this is exactly a matrix‑multiply

    L_d (N×N)  @  R_d.T (N×N)  →  out_d (N×N)

We therefore launch a Triton kernel that computes many (i,j) tiles
simultaneously while iterating over *k* in blocks.  The kernel works in
float32 internally and writes the result in float32; the surrounding
PyTorch code handles the LayerNorms, linear projections and gates.
The implementation supports arbitrary batch size (B∈{1,2}), sequence
length N up to 1024 and hidden dimension H (named *hidden_dim* in the
configuration).  The kernel is launched with a 3‑D grid:

    grid = (B,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

Inside the kernel the program id ``pid`` encodes the hidden‑dim block
(``d_offset``) – we iterate over the hidden dimension in the outermost
Python loop to keep the kernel signature simple and compatible with the
current Triton version.

The kernel is deliberately written to be memory‑bound‑friendly:
* loads are masked with the ``k < N`` condition,
* accumulators are kept in float32 (the input may be bf16/float16),
* stores are performed only after the full reduction over *k*.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel – computes one hidden‑dim slice:
#   out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]   (all 2‑D matrices)
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    left_ptr,          # const * left  (B,N,N)  – packed as [b,i,k]
    right_ptr,         # const * right (B,N,N)  – packed as [b,j,k]
    out_ptr,           #       * out   (B,N,N)  – packed as [b,i,j]
    B, N,               # static sizes
    stride_left_bi, stride_left_ik,
    stride_right_bj, stride_right_jk,
    stride_out_bi, stride_out_ij,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a single hidden‑dimension block of the TriMul contraction.
    Each program works on a tile (i0:i0+BLOCK_M, j0:j0+BLOCK_N) for a
    particular batch ``b``.  The reduction over *k* is tiled by BLOCK_K.
    """

    # --------------------------------------------------------------
    # 1) identify tile coordinates
    # --------------------------------------------------------------
    pid_b = tl.program_id(0)               # batch index
    pid_m = tl.program_id(1)               # tile over rows   (i)
    pid_n = tl.program_id(2)               # tile over cols   (j)

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # --------------------------------------------------------------
    # 2) offsets for the three tensors
    # --------------------------------------------------------------
    # left  : [b, i, k]
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_k = tl.arange(0, BLOCK_K)
    mask_i = offs_i < N
    mask_k = offs_k < N

    left_ptrs = left_ptr + \
        pid_b * stride_left_bi + \
        offs_i[:, None] * stride_left_ik + \
        offs_k[None, :] * stride_left_ik

    # right : [b, j, k]
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_j = offs_j < N

    right_ptrs = right_ptr + \
        pid_b * stride_right_bj + \
        offs_j[:, None] * stride_right_jk + \
        offs_k[None, :] * stride_right_jk

    # --------------------------------------------------------------
    # 3) accumulator (float32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 4) loop over k dimension
    # --------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + offs_k
        cur_mask_k = cur_k < N

        # load a tile of left  (M×K) and right (N×K)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & cur_mask_k[None, :],
                       other=0.0).to(tl.float32)
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & cur_mask_k[None, :],
                        other=0.0).to(tl.float32)

        # inner product over k
        acc += tl.dot(left, right, trans_b=True)   # (M×K)·(K×N) → (M×N)

        # advance pointers
        left_ptrs += BLOCK_K * stride_left_ik
        right_ptrs += BLOCK_K * stride_right_jk

    # --------------------------------------------------------------
    # 5) write back
    # --------------------------------------------------------------
    out_ptrs = out_ptr + \
        pid_b * stride_out_bi + \
        offs_i[:, None] * stride_out_ij + \
        offs_j[None, :] * stride_out_ij

    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


# ----------------------------------------------------------------------
# Helper: apply LayerNorm in FP32 (torch implementation – cheap)
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_hat = (x - mean) / torch.sqrt(var + eps)
    return x_hat * weight + bias


# ----------------------------------------------------------------------
# Main entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator using a Triton kernel for the
    N³ contraction.  All linear projections, gating, and LayerNorms are performed
    with PyTorch; only the heavy matrix‑multiply (over the sequence dimension)
    is accelerated by Triton.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    ).to(torch.float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – produce [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    #   shape: (..., dim) @ (dim, hidden_dim).T  →  (..., hidden_dim)
    def linear_no_bias(t, w):
        # t: [B,N,N,dim]  w: [hidden_dim, dim]
        # we cast to fp16 for the matmul (speed) and back to fp32 later
        return torch.nn.functional.linear(t, w).to(torch.float32)

    left = linear_no_bias(x, weights["left_proj.weight"])
    right = linear_no_bias(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask (optional) – shape broadcasted to hidden_dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) – also no bias
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton contraction: out = Σ_k left[...,k,d] * right[...,k,d]
    #    We compute it slice‑by‑slice over the hidden dimension.
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # constants for the kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # stride information (all tensors are contiguous in C order)
    stride_left_bi, stride_left_ik, stride_left_kd = left.stride()
    stride_right_bj, stride_right_jk, stride_right_kd = right.stride()
    stride_out_bi, stride_out_ij, stride_out_jd = out.stride()

    # The hidden dimension is not part of the kernel; we loop over it in Python.
    # This keeps the kernel signature simple and avoids the 4‑D grid limitation.
    for d in range(hidden_dim):
        # pointers for the d‑th slice
        left_ptr = left[..., d].contiguous()
        right_ptr = right[..., d].contiguous()
        out_ptr = out[..., d]

        # launch
        grid = (B,
                triton.cdiv(N, BLOCK_M),
                triton.cdiv(N, BLOCK_N))
        trimul_matmul_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            B, N,
            stride_left_bi, stride_left_ik,
            stride_right_bj, stride_right_jk,
            stride_out_bi, stride_out_ij,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

    # ------------------------------------------------------------------
    # 6) Output post‑processing: LayerNorm, gating and final linear
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out.to(torch.float32)
