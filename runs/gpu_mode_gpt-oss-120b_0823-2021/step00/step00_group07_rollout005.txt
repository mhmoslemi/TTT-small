"""
TriMul “outgoing” implementation.

The expensive part of the module is the pair‑wise multiplication

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

for every batch `b`, sequence index `i`, `j` and hidden channel `h`.
For a fixed `(b,h)` this is exactly a matrix multiplication

    L = left[b,:, :, h]          # (N, N)
    R = right[b,:, :, h]         # (N, N)
    out[b,:, :, h] = L @ R.T

The kernel below performs a batched mat‑mul over the combined
`batch*head` dimension.  All other operations (layer‑norms,
linear projections, gating, final linear) are kept in PyTorch – they are
cheap compared with the O(N³) work and allow us to reuse the existing
weight tensors without extra bookkeeping.

The kernel is written for Triton 3.3.1 and targets NVIDIA Hopper
(H200/H100).  Tile sizes (BLOCK_M/N/K) are chosen to give good
occupancy while staying inside the shared‑memory budget of the
architecture.  Mixed‑precision (fp16) is used inside the kernel; the
result is cast back to fp32 before the final linear projection.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul L @ Rᵀ  (L,R : (B*H, N, N))
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr,                # left  : (B*H, N, N)   fp16
    b_ptr,                # right : (B*H, N, N)   fp16
    c_ptr,                # out   : (B*H, N, N)   fp32 (accumulated as fp32)
    M, N, K,              # all equal to seq_len
    stride_am, stride_ak, stride_ah,   # a: batch*head, row, col
    stride_bm, stride_bk, stride_bh,   # b: batch*head, row, col
    stride_cm, stride_cn, stride_ch,   # c: batch*head, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)               # tile row
    pid_n = tl.program_id(1)               # tile col
    pid_bh = tl.program_id(2)              # combined batch*head index

    # tile start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the current batch*head
    a_batch_ptr = a_ptr + pid_bh * stride_ah
    b_batch_ptr = b_ptr + pid_bh * stride_bh
    c_batch_ptr = c_ptr + pid_bh * stride_ch

    # initialize accumulator in fp32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # K‑tile offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load a tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am
                                + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)               # fp16 → fp32

        # load b tile, but we need Rᵀ, so we load transposed tile:
        #   we want b[j,k] where j is column index in output, k is reduction.
        #   Hence we load (BLOCK_K, BLOCK_N) and later treat it as transposed.
        b_ptrs = b_batch_ptr + (offs_n[:, None] * stride_bm   # note: row‑major of Rᵀ
                                + offs_k[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)
        b = b.to(tl.float32)

        # accumulate:   Σ_k a[i,k] * b[j,k]   (note b already corresponds to Rᵀ)
        acc += tl.dot(a, b)               # (BLOCK_M, BLOCK_N)

    # write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm
                            + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: LayerNorm (no running stats, just affine)
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: (..., C)  float32
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : (input, mask, weights, config)
        input   – Tensor[bs, N, N, dim] (float32)
        mask    – Tensor[bs, N, N]     (float32 or bool)
        weights – dict of all linear / layer‑norm parameters
        config  – dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    Tensor[bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]           # called hidden_dim in the paper

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)  –  fp16 for speed
    # ------------------------------------------------------------------
    # shapes: (dim, hidden)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    # reshape for a single matmul per head
    x_flat = x.view(B * N * N, dim)                     # (B*N*N, dim)
    left = torch.nn.functional.linear(x_flat, left_proj_w).view(B, N, N, hidden)
    right = torch.nn.functional.linear(x_flat, right_proj_w).view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 3) Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x.view(B * N * N, dim), weights["left_gate.weight"]
    ).view(B, N, N, hidden).sigmoid()
    right_gate = torch.nn.functional.linear(
        x.view(B * N * N, dim), weights["right_gate.weight"]
    ).view(B, N, N, hidden).sigmoid()
    out_gate = torch.nn.functional.linear(
        x.view(B * N * N, dim), weights["out_gate.weight"]
    ).view(B, N, N, hidden).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul: batched matmul over the K dimension (N)
    #    We treat (batch, head) as a single leading dimension.
    # ------------------------------------------------------------------
    # cast to fp16 for the kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    # output will be accumulated in float32
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)

    # strides (in elements, not bytes)
    stride_ah = left_fp16.stride(0)   # batch*head stride
    stride_am = left_fp16.stride(1)   # row (i) stride
    stride_ak = left_fp16.stride(2)   # col (k) stride

    stride_bh = right_fp16.stride(0)
    stride_bm = right_fp16.stride(1)
    stride_bk = right_fp16.stride(2)

    stride_ch = out_fp32.stride(0)
    stride_cm = out_fp32.stride(1)
    stride_cn = out_fp32.stride(2)

    # combine batch and hidden into a single dimension for the kernel
    BH = B * hidden
    a_ptr = left_fp16.view(BH, N, N).contiguous()
    b_ptr = right_fp16.view(BH, N, N).contiguous()
    c_ptr = out_fp32.view(BH, N, N).contiguous()

    # tile sizes – empirically good for Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        BH,                               # batch*head
    )
    tri_mul_kernel[grid](
        a_ptr,
        b_ptr,
        c_ptr,
        N, N, N,                         # M, N, K  (all = seq_len)
        stride_am, stride_ak, stride_ah,
        stride_bm, stride_bk, stride_bh,
        stride_cm, stride_cn, stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B,N,N,hidden)
    out = out_fp32.view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6) to_out_norm (LayerNorm over hidden) and final linear
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # final linear: hidden -> dim (no bias)
    to_out_w = weights["to_out.weight"]            # (hidden, dim)
    out = torch.nn.functional.linear(out.view(B * N * N, hidden), to_out_w)
    out = out.view(B, N, N, dim)

    return out
