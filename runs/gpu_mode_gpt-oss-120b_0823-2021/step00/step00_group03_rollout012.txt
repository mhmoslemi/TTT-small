"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

We compute
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

for a 4‑D tensor x of shape [B, N, N, C].
The forward pass consists of:
1️⃣ Layer‑norm on the last dimension (C)                         – PyTorch
2️⃣ Linear projections + sigmoid gates (no bias)                  – PyTorch
3️⃣ Masking (optional)                                            – PyTorch
4️⃣ The *core* N³ contraction (the einsum above)                 – Triton
5️⃣ Post‑norm + gate + final linear                               – PyTorch

Only step‑4 is accelerated with Triton; everything else stays in
regular torch code so we keep the implementation simple and
numerically exact.  The Triton kernel treats each hidden dimension
`d` as an independent GEMM:
    left_d (B,N,N)  @  right_dᵀ (B,N,N)   →  out_d (B,N,N)

We launch a 2‑D grid:
    program_id(0) → block of output rows (i)
    program_id(1) → (batch * hidden_dim) index, i.e. a distinct GEMM
The kernel is written for FP16 compute (the inputs are cast to
fp16 for speed) and the result is accumulated in fp32, then stored
back as fp32 to satisfy the required output dtype.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -------------------------------------------------------------
# Triton kernel: one hidden‑dim slice of the N×N GEMM per batch
# -------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr,          # B * H * N * N  (contiguous)
    right_ptr,         # B * H * N * N  (contiguous)
    out_ptr,           # B * H * N * N  (contiguous)
    B, N, H,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,  # j‑dimension is first in the GEMM
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    FP16: tl.constexpr,          # compute in fp16?
):
    # -------------------------------------------------
    # program IDs
    # -------------------------------------------------
    pid_m = tl.program_id(0)                     # row block of the output (i)
    pid_bn = tl.program_id(1)                    # combined (batch, hidden) index
    # de‑compose pid_bn
    b = pid_bn // H
    h = pid_bn % H

    # -------------------------------------------------
    # offsets
    # -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds rows
    mask_m = offs_m < N

    # pointers to the start of the current GEMM
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # -----------------------------------------------------------------
    # accumulator (fp32 for numerical safety)
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # loop over K dimension (the “summation” axis of the einsum)
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # K‑range for this iteration
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------------------------------------------------
        # load a tile from left  : (i,k)
        # -------------------------------------------------
        left_ptrs = left_base \
            + offs_m[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        a = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        if FP16:
            a = a.to(tl.float16)

        # -------------------------------------------------
        # load a tile from right : (j,k)   ---- note we need rightᵀ
        # -------------------------------------------------
        right_ptrs = right_base \
            + offs_n[:, None] * stride_right_j \
            + offs_k[None, :] * stride_right_k
        b_tile = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)
        if FP16:
            b_tile = b_tile.to(tl.float16)

        # -------------------------------------------------
        # dot‑product accumulation
        # -------------------------------------------------
        # tl.dot works on (M,K)×(K,N) → (M,N)
        acc += tl.dot(a, b_tile)

        # advance pointers (not needed because we recompute offsets each iter)

    # -----------------------------------------------------------------
    # write‑back
    # -----------------------------------------------------------------
    out_ptrs = out_base \
        + offs_m[:, None] * stride_out_i \
        + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])

# -------------------------------------------------------------
# Host wrapper that launches the kernel
# -------------------------------------------------------------
def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]   (float32)
    returns      : [B, N, N, H]   (float32)
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)

    # strides in elements (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),            # rows
        B * H,                              # batch * hidden_dim
    )
    trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=H,
        stride_left_b=stride_left_b,
        stride_left_h=stride_left_h,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_right_b=stride_right_b,
        stride_right_h=stride_right_h,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_out_b=stride_out_b,
        stride_out_h=stride_out_h,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        FP16=1,          # compute in fp16 for speed, accumulate in fp32
    )
    return out

# -------------------------------------------------------------
# Full TriMul forward using the Triton kernel for the heavy part
# -------------------------------------------------------------
class TritonTriMul(nn.Module):
    def __init__(self, dim: int, hidden_dim: int):
        super().__init__()
        self.dim = dim
        self.hidden_dim = hidden_dim

        self.norm = nn.LayerNorm(dim, eps=1e-5, elementwise_affine=True)
        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim, eps=1e-5, elementwise_affine=True)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x    : [B, N, N, C]
        mask : [B, N, N]   (may be all‑ones if nomask)
        """
        B, N, _, C = x.shape

        # 1. LayerNorm on the channel dimension
        x = self.norm(x)

        # 2. Linear projections (no bias)
        left = self.left_proj(x)   # [B,N,N,H]
        right = self.right_proj(x)

        # 3. Optional mask (broadcast over hidden dim)
        if mask is not None:
            mask = mask.unsqueeze(-1)           # [B,N,N,1]
            left = left * mask
            right = right * mask

        # 4. Gating
        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        # 5. Core N³ contraction – Triton
        # cast to fp16 for the kernel then back to fp32
        left_fp16 = left.to(torch.float16).contiguous()
        right_fp16 = right.to(torch.float16).contiguous()
        out = trimul_gemm(left_fp16, right_fp16)   # fp32 result

        # 6. Post‑processing
        out = self.to_out_norm(out)
        out = out * out_gate
        out = self.to_out(out)

        return out

# -------------------------------------------------------------
# Entrypoint required by the evaluation harness
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

    The function builds a TritonTriMul instance, loads the provided
    weights (all tensors are already on CUDA) and runs the forward
    pass.  The final result is returned as float32.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    model = TritonTriMul(dim, hidden_dim).to(input_tensor.device)

    # ---- load weights -------------------------------------------------
    model.norm.weight.data.copy_(weights["norm.weight"])
    model.norm.bias.data.copy_(weights["norm.bias"])

    model.left_proj.weight.data.copy_(weights["left_proj.weight"])
    model.right_proj.weight.data.copy_(weights["right_proj.weight"])

    model.left_gate.weight.data.copy_(weights["left_gate.weight"])
    model.right_gate.weight.data.copy_(weights["right_gate.weight"])
    model.out_gate.weight.data.copy_(weights["out_gate.weight"])

    model.to_out_norm.weight.data.copy_(weights["to_out_norm.weight"])
    model.to_out_norm.bias.data.copy_(weights["to_out_norm.bias"])
    model.to_out.weight.data.copy_(weights["to_out.weight"])

    # ---- compute ------------------------------------------------------
    output = model(input_tensor, mask)

    return output
