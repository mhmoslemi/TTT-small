"""
TriMul “outgoing” kernel (AlphaFold3).

We compute the core three‑tensor multiplication
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where
    left  = (x @ W_left)   * mask * sigmoid(x @ W_left_gate)
    right = (x @ W_right)  * mask * sigmoid(x @ W_right_gate)

The heavy N³ work is performed in a fused Triton kernel that treats the
channel dimension `d` as a third matrix‑multiply batch.  The kernel is a
straightforward tiled GEMM over the `k` dimension, accumulating a block of
(i, j, d).  All surrounding LayerNorm and Linear work is kept in PyTorch
(because they are cheap compared to the N³ inner product) while the
multiplication‑sum is fully on‑device.

The kernel works for arbitrary batch size (1‑2), sequence length N (≤1024)
and hidden dimension H (≤384).  It uses float16 accumulation for speed and
writes the final result in float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr,    # const float16 *
    right_ptr,   # const float16 *
    out_ptr,     # float32 *
    # tensor sizes
    B, N, H,
    # strides (in elements, not bytes)
    stride_lb, stride_li, stride_lk, stride_ld,   # left  [b, i, k, d]
    stride_rb, stride_rj, stride_rk, stride_rd,   # right [b, j, k, d]
    stride_ob, stride_oi, stride_oj, stride_od,   # out   [b, i, j, d]
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,  # i‑tile
    BLOCK_N: tl.constexpr,  # j‑tile
    BLOCK_D: tl.constexpr,  # d‑tile
    BLOCK_K: tl.constexpr,  # k‑tile
):
    """ Compute a tiled (i,j,d) block for a single batch element. """
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # ------------------------------------------------------------------
    # Offsets for the current block
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = tl.arange(0, BLOCK_D)

    # Mask out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # Pointers to the start of the block (b‑dim fixed)
    left_base = left_ptr + pid_b * stride_lb
    right_base = right_ptr + pid_b * stride_rb
    out_base = out_ptr + pid_b * stride_ob

    # ------------------------------------------------------------------
    # Accumulator: use float32 for numerical stability
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Compute k offsets for this tile
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left  [M, K, D]  -> (i, k, d)
        left_ptrs = (left_base
                     + offs_i[:, None, None] * stride_li
                     + offs_k[None, :, None] * stride_lk
                     + offs_d[None, None, :] * stride_ld)
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                       other=0.0).to(tl.float32)

        # Load right [N, K, D]  -> (j, k, d)
        right_ptrs = (right_base
                      + offs_j[:, None, None] * stride_rj
                      + offs_k[None, :, None] * stride_rk
                      + offs_d[None, None, :] * stride_rd)
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                        other=0.0).to(tl.float32)

        # Multiply‑accumulate over k
        # shape: (M,N,D) += (M,K,D) * (N,K,D)  summed over K
        # We do a manual dot: for each d we have a matrix‑multiply
        for dk in range(0, BLOCK_D):
            # slice d‑th channel
            l = left[:, :, dk]   # (M,K)
            r = right[:, :, dk]  # (N,K)
            acc[:, :, dk] += tl.dot(l, r, trans_b=False)   # (M,N)

    # ------------------------------------------------------------------
    # Write result
    # ------------------------------------------------------------------
    out_ptrs = (out_base
                + offs_i[:, None, None] * stride_oi
                + offs_j[None, :, None] * stride_oj
                + offs_d[None, None, :] * stride_od)
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : [B, N, N, C]   float32
            mask_tensor  : [B, N, N]      float32 or bool
            weights_dict : mapping of weight names to tensors
            config_dict  : {"dim": C, "hidden_dim": H}
    Returns:
        output tensor of shape [B, N, N, C]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper: LayerNorm without bias (bias handled separately)
    # ------------------------------------------------------------------
    def layer_norm(x, weight, bias, eps=1e-5):
        # x: [..., C]
        mean = x.mean(dim=-1, keepdim=True)
        var = x.var(dim=-1, unbiased=False, keepdim=True)
        x_norm = (x - mean) * torch.rsqrt(var + eps)
        return x_norm * weight + bias

    # ------------------------------------------------------------------
    # Load weights (they are provided as float32)
    # ------------------------------------------------------------------
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    w_left = weights["left_proj.weight"]          # (H, C)
    w_right = weights["right_proj.weight"]        # (H, C)
    w_left_gate = weights["left_gate.weight"]     # (H, C)
    w_right_gate = weights["right_gate.weight"]   # (H, C)
    w_out_gate = weights["out_gate.weight"]       # (H, C)
    w_to_out_norm = weights["to_out_norm.weight"]
    b_to_out_norm = weights["to_out_norm.bias"]
    w_to_out = weights["to_out.weight"]           # (C, H)

    # ------------------------------------------------------------------
    # Normalisation + linear projections (keep in fp32, cast to fp16 for kernel)
    # ------------------------------------------------------------------
    x = layer_norm(input_tensor, w_norm, b_norm)                      # [B,N,N,C] fp32

    # Linear projections – shape (B,N,N,H)
    left = F.linear(x, w_left)      # (B,N,N,H)
    right = F.linear(x, w_right)

    # Apply mask if present
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(F.linear(x, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x, w_out_gate))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel: compute Σ_k left[...,i,k,d] * right[...,j,k,d]
    # ------------------------------------------------------------------
    # Cast to fp16 for faster arithmetic inside the kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp32 = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Strides in elements (not bytes)
    stride_lb, stride_li, stride_lk, stride_ld = left_fp16.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_fp32.stride()

    # Tile sizes – tuned for Hopper SM90 (shared‑mem ~ 64KB)
    BLOCK_M = 32   # i
    BLOCK_N = 32   # j
    BLOCK_D = 32   # channel block
    BLOCK_K = 32   # k

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_D=BLOCK_D,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑processing: to_out_norm -> gating -> to_out
    # ------------------------------------------------------------------
    out = layer_norm(out_fp32, w_to_out_norm, b_to_out_norm)
    out = out * out_gate
    out = F.linear(out, w_to_out)          # back to [B,N,N,C]

    return out.to(torch.float32)
