"""
TriMul (outgoing) – fast Triton implementation of the core N‑wise update.

The module consists of:
1️⃣ Layer‑norm and four linear projections (left/right + gates) – done in PyTorch.
2️⃣ Masking & gating – PyTorch (very cheap).
3️⃣ The expensive N³ “einsum”:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is exactly a batched matrix‑multiply of shape
        (B, N, N, D) → (B·D, N, N)  with   out = left @ rightᵀ .
   This step is executed in a custom Triton kernel.  The kernel works for any
   batch size B∈{1,2}, sequence length N up to 1024 and hidden dimension D (the
   “hidden_dim” in the paper).  It uses 16‑bit arithmetic for the inner product
   and accumulates in 32‑bit to retain FP32 accuracy.  The final result is cast
   back to float32 before the remaining PyTorch post‑processing.

The kernel is launched with a 2‑D grid:
   dim0 – (B·D) tiled over the “batch‑hidden” dimension,
   dim1 – row‑tiles of the output matrix.
All strides are passed explicitly so the kernel works with the original
contiguous layout `[B, N, N, D]`.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   with A,B shape (M,N) and C (M,N)
# Here M = N (seq_len), N = N (seq_len) and we tile over the combined
# batch*hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def triton_outgoing_matmul_kernel(
    # pointers   (flattened as 1‑D)
    a_ptr, b_ptr, c_ptr,
    # problem size
    N, BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    # strides for the original 4‑D tensor [B, N, N, D]
    stride_a_bn, stride_a_nk, stride_a_kd,   # A = left
    stride_b_bn, stride_b_nk, stride_b_kd,   # B = right
    stride_c_bn, stride_c_nj, stride_c_jd,   # C = out
):
    # program ids
    pid_bhd = tl.program_id(0)          # index over (batch * hidden)
    pid_row = tl.program_id(1)          # tile row (i)

    # Resolve batch and hidden index from the flattened pid
    hidden = tl.program_id(2)           # we expose hidden as a third axis for clarity
    batch = pid_bhd // hidden
    hidden_idx = pid_bhd % hidden

    # Offsets for the tile we are computing
    offs_m = pid_row * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)                     # column offset inside the tile

    # Masks for boundary conditions
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the beginning of the tile for this (batch, hidden) slice
    a_base = a_ptr + batch * stride_a_bn + hidden_idx * stride_a_kd
    b_base = b_ptr + batch * stride_b_bn + hidden_idx * stride_b_kd
    c_base = c_ptr + batch * stride_c_bn + hidden_idx * stride_c_jd

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # k‑range of this iteration
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A[i,k] and B[j,k] (note: B is transposed → we load B[:,k] as rows)
        a = tl.load(
            a_base + offs_m[:, None] * stride_a_nk + offs_k[None, :] * stride_a_kd,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_base + offs_n[:, None] * stride_b_nk + offs_k[None, :] * stride_b_kd,
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Cast to fp16 for the dot product, accumulate in fp32
        a = a.to(tl.float16)
        b = b.to(tl.float16)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) • (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # Write back result
    c = acc.to(tl.float32)
    c_ptrs = c_base + offs_m[:, None] * stride_c_nj + offs_n[None, :] * stride_c_jd
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel for the whole tensor
# ----------------------------------------------------------------------
def outgoing_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, D]   (float32)
    returns      : [B, N, N, D]   (float32)
    """
    B, N, _, D = left.shape
    assert right.shape == (B, N, N, D)

    # Output buffer
    out = torch.empty_like(left)

    # Choose tile sizes (tuned for H200 – 64 works well for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides in the original layout (contiguous)
    stride_a_bn, stride_a_nk, stride_a_kd = left.stride()
    stride_b_bn, stride_b_nk, stride_b_kd = right.stride()
    stride_c_bn, stride_c_nj, stride_c_jd = out.stride()

    # Flatten batch*hidden dimension for the first grid axis
    total_bhd = B * D

    grid = (
        total_bhd,                         # pid_bhd
        triton.cdiv(N, BLOCK_M),          # pid_row (tiles over i)
        D,                                # hidden index – we expose it as a third axis
    )

    # Launch
    triton_outgoing_matmul_kernel[grid](
        left, right, out,
        N,
        BLOCK_M, BLOCK_N, BLOCK_K,
        stride_a_bn, stride_a_nk, stride_a_kd,
        stride_b_bn, stride_b_nk, stride_b_kd,
        stride_c_bn, stride_c_nj, stride_c_jd,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full TriMul (outgoing) implementation – combines PyTorch + Triton
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Args
    ----
    data : (input, mask, weights, config)
        input  – Tensor[B, N, N, dim] (float32, already on CUDA)
        mask   – Tensor[B, N, N] (float32/bool, already on CUDA)
        weights – dict with all linear / layer‑norm parameters
        config – dict with keys "dim" and "hidden_dim"

    Returns
    -------
    Tensor[B, N, N, dim]   (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- 1️⃣ LayerNorm ----------
    # Manual LN to avoid extra kernel launches; uses torch ops (fast enough)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * norm_weight + norm_bias

    # ---------- 2️⃣ Linear projections ----------
    # shape: [B,N,N,hidden_dim]
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- 3️⃣ Mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- 4️⃣ Gating ----------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- 5️⃣ Core N³ operation (Triton) ----------
    # left/right are [B,N,N,hidden_dim]; we need them as (B,N,N,hidden_dim)
    # Triton kernel expects [B,N,N,hidden_dim] contiguous – already the case.
    out = outgoing_matmul(left, right)           # -> [B,N,N,hidden_dim] (float32)

    # ---------- 6️⃣ Final LayerNorm, gating and projection ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # LN over hidden_dim
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * to_out_norm_weight + to_out_norm_bias

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
