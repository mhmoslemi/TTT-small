"""
TriMul “outgoing” implementation.

We keep the cheap point‑wise ops (layernorm, linear projections, gates, final
norm/out linear) in PyTorch and only jump into Triton for the cubic
einsum:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

All tensors are already on the CUDA device.  The kernel works on blocks of
(i,j) and a small tile of the hidden dimension `d`.  It accumulates in FP32
and writes the result back in FP32.  Mixed‑precision (FP16 load) is used for
the inner product to improve bandwidth while keeping the final output in
float32.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict


# ------------------------------------------------------------------
# Triton kernel that computes the 4‑D “tri‑mul” contraction.
# ------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const * left  [B,N,N,H]
    right_ptr,         # const * right [B,N,N,H]
    out_ptr,           #       * out   [B,N,N,H]
    B, N, H,           # problem sizes
    stride_blh, stride_blk, stride_blh_d,   # left strides: (b,i,k,d)
    stride_brh, stride_brk, stride_brh_d,   # right strides: (b,j,k,d)
    stride_oh,  stride_ok,  stride_od,      # out strides:   (b,i,j,d)
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_D: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)                     # batch index
    pid_i = tl.program_id(0)                     # i‑block
    pid_j = tl.program_id(1)                     # j‑block

    # block coordinates
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # tile offsets inside the block
    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)
    offs_d = tl.arange(0, BLOCK_D)

    # create masks for out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # pointers for the current batch
    left_batch_ptr  = left_ptr  + pid_b * stride_blh
    right_batch_ptr = right_ptr + pid_b * stride_brh
    out_batch_ptr   = out_ptr   + pid_b * stride_oh

    # initialise accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # iterate over k in tiles
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # --------------------------------------------------------------
        # load left[b,i,k,d]  -> shape (BLOCK_I, BLOCK_K, BLOCK_D)
        # load right[b,j,k,d] -> shape (BLOCK_J, BLOCK_K, BLOCK_D)
        # --------------------------------------------------------------
        left_ptrs = (
            left_batch_ptr
            + (offs_i[:, None, None] * stride_blk
               + cur_k[None, :, None] * stride_blh_d
               + offs_d[None, None, :] )
        )
        right_ptrs = (
            right_batch_ptr
            + (offs_j[:, None, None] * stride_brk
               + cur_k[None, :, None] * stride_brh_d
               + offs_d[None, None, :] )
        )

        left_tile  = tl.load(left_ptrs,
                             mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                             other=0.0).to(tl.float32)
        right_tile = tl.load(right_ptrs,
                             mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                             other=0.0).to(tl.float32)

        # Perform the reduction over k:
        #   Σ_k left[i,k,d] * right[j,k,d]
        # We compute a batched dot for each d slice.
        # (BLOCK_I, BLOCK_K) dot (BLOCK_J, BLOCK_K)^T -> (BLOCK_I, BLOCK_J)
        # expand d dimension manually.
        # Use tl.dot which works on 2‑D matrices; we loop over d.
        for d in range(0, BLOCK_D):
            li = left_tile[:, :, d]          # (I,K)
            rj = right_tile[:, :, d]         # (J,K)
            # dot over K -> (I,J)
            acc[:, :, d] += tl.dot(li, rj, trans_b=True)

    # --------------------------------------------------------------
    # write result
    # --------------------------------------------------------------
    out_ptrs = (
        out_batch_ptr
        + (offs_i[:, None, None] * stride_ok
           + offs_j[None, :, None] * stride_od
           + offs_d[None, None, :] )
    )
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])


# ------------------------------------------------------------------
# Host‑side entry point
# ------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : (input, mask, weights, config)
        * input  – tensor [B, N, N, dim]   (float32)
        * mask   – tensor [B, N, N]        (float32 or bool)
        * weights – dict with all linear / norm parameters
        * config – dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    out : torch.Tensor  [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]

    ##################################################################
    # 1️⃣  Point‑wise ops in PyTorch (LayerNorm, Linear, gating)
    ##################################################################
    # helper to apply a linear weight (no bias)
    def linear(x, w):
        # x: [..., dim]   w: [out, dim]
        return torch.nn.functional.linear(x, w)

    # layernorm
    norm_weight = weights["norm.weight"].to(inp.dtype)
    norm_bias   = weights["norm.bias"].to(inp.dtype)
    x = torch.nn.functional.layer_norm(inp,
                                       normalized_shape=(dim,),
                                       weight=norm_weight,
                                       bias=norm_bias,
                                       eps=1e-5)

    # projections
    left  = linear(x, weights["left_proj.weight"].to(inp.dtype))
    right = linear(x, weights["right_proj.weight"].to(inp.dtype))

    # mask handling (optional)
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # gates (sigmoid)
    left_gate  = torch.sigmoid(linear(x, weights["left_gate.weight"].to(inp.dtype)))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"].to(inp.dtype)))
    out_gate   = torch.sigmoid(linear(x, weights["out_gate.weight"].to(inp.dtype)))

    left  = left  * left_gate
    right = right * right_gate

    ##################################################################
    # 2️⃣  Triton kernel for the cubic contraction
    ##################################################################
    # stride layout for left/right/out tensors (contiguous, last dim = hidden)
    stride_blh, stride_blk, stride_blh_d = left.stride()
    stride_brh, stride_brk, stride_brh_d = right.stride()
    stride_oh,  stride_ok,  stride_od   = (B * N * N * hidden,
                                          N * N * hidden,
                                          N * hidden)  # we will recompute below

    # Because tensors are contiguous the strides are simple:
    stride_blh, stride_blk, stride_blh_d = left.stride()
    stride_brh, stride_brk, stride_brh_d = right.stride()
    stride_oh,  stride_ok,  stride_od   = out = torch.empty_like(left)  # placeholder
    stride_oh, stride_ok, stride_od = out.stride()  # after allocation

    # allocate output buffer (float32)
    out = torch.empty((B, N, N, hidden),
                      dtype=torch.float32,
                      device=inp.device)

    # launch parameters (tuned for Hopper)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_D = 32            # hidden is a multiple of 32 (128/384)
    BLOCK_K = 64            # tile size along the summed dimension

    grid = (triton.cdiv(N, BLOCK_I),
            triton.cdiv(N, BLOCK_J),
            B)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_blh=stride_blh,
        stride_blk=stride_blk,
        stride_blh_d=stride_blh_d,
        stride_brh=stride_brh,
        stride_brk=stride_brk,
        stride_brh_d=stride_brh_d,
        stride_oh=stride_oh,
        stride_ok=stride_ok,
        stride_od=stride_od,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_D=BLOCK_D,
        BLOCK_K=BLOCK_K,
    )

    ##################################################################
    # 3️⃣  Remaining point‑wise ops (norm, out gating, final linear)
    ##################################################################
    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"].to(inp.dtype)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(inp.dtype)
    out = torch.nn.functional.layer_norm(out,
                                         normalized_shape=(hidden,),
                                         weight=to_out_norm_weight,
                                         bias=to_out_norm_bias,
                                         eps=1e-5)

    out = out * out_gate

    # final linear projection back to dim
    out = linear(out, weights["to_out.weight"].to(inp.dtype))

    return out.to(torch.float32)
