"""
TriMul “outgoing” kernel (AlphaFold‑3 style).

The heavy part is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed (b,d) this is a matrix multiplication
    L = left[b,:,:,d]          (N×N)
    R = right[b,:,:,d]         (N×N)
    out[b,:,:,d] = L @ R.T      (N×N)

The implementation below
* does all LayerNorm / Linear / gating work in PyTorch (they are cheap);
* folds the triple‑loop N³ into a batched GEMM that runs on Triton;
* keeps the final LayerNorm / Linear in PyTorch;
* returns a float‑32 tensor of shape (B, N, N, dim).

The Triton kernel is a classic block‑wise GEMM (Block‑M×Block‑N tiles,
accumulating over K in blocks of size BLOCK_K).  It is launched over the
grid (B * D, ceil(N/BLOCK_M), ceil(N/BLOCK_N)).  The kernel works with
FP16 inside the inner product for higher throughput and converts the
result back to FP32 before storing.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton GEMM kernel:  out = left @ right.T  (both have shape [B, N, N, D])
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, D,                                     # problem sizes
    stride_left_bn, stride_left_bc, stride_left_bk,  # left strides (B,N,N,D) → (b,n,k,d)
    stride_right_bn, stride_right_bc, stride_right_bk,  # right strides (b,n,k,d)
    stride_out_bn, stride_out_bc, stride_out_bd,       # out strides (b,n,j,d)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,   # tile sizes
):
    pid = tl.program_id(0)                     # linear program id over (b,d) pairs
    pid_bd = pid
    total_bd = B * D
    b = pid_bd // D
    d = pid_bd % D

    # tile offsets for the output matrix (i,j)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the start of the tile in each operand
    left_tile_ptr = left_ptr + \
        b * stride_left_bn + d * stride_left_bc + \
        offs_m[:, None] * stride_left_bk + tl.arange(0, BLOCK_K)[None, :] * 0   # k offset added in loop
    right_tile_ptr = right_ptr + \
        b * stride_right_bn + d * stride_right_bc + \
        offs_n[None, :] * stride_right_bc + tl.arange(0, BLOCK_K)[:, None] * 0   # k offset added in loop

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = tl.arange(0, BLOCK_K) + k_offset
        mask_k = cur_k < N

        # Load a BLOCK_M × BLOCK_K tile from left  (L[i,k])
        l_ptr = left_ptr + \
            b * stride_left_bn + d * stride_left_bc + \
            offs_m[:, None] * stride_left_bk + cur_k[None, :] * stride_left_bc
        L = tl.load(l_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load a BLOCK_N × BLOCK_K tile from right (R[j,k]) – note the transpose
        r_ptr = right_ptr + \
            b * stride_right_bn + d * stride_right_bc + \
            offs_n[None, :] * stride_right_bc + cur_k[:, None] * stride_right_bc
        R = tl.load(r_ptr, mask=mask_n[None, :] & mask_k[:, None], other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(L, R, out_dtype=tl.float32)

    # Write back
    out_ptr_tile = out_ptr + \
        b * stride_out_bn + d * stride_out_bd + \
        offs_m[:, None] * stride_out_bc + offs_n[None, :] * stride_out_bd
    tl.store(out_ptr_tile, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out = left @ right.transpose(-2,-1) for tensors of shape
    (B, N, N, D).  Returns a tensor of the same shape, dtype float32.
    """
    B, N, _, D = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # stride ordering (B,N,N,D) → (b, i, k, d)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # Tile sizes – tuned for Hopper (sm90) – 128×128 tiles, 32‑wide K
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B * D,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=D,
        stride_left_bn=stride_left[0],
        stride_left_bc=stride_left[3],
        stride_left_bk=stride_left[2],
        stride_right_bn=stride_right[0],
        stride_right_bc=stride_right[3],
        stride_right_bk=stride_right[2],
        stride_out_bn=stride_out[0],
        stride_out_bc=stride_out[1],
        stride_out_bd=stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full TriMul forward (outgoing) – main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.
    Arguments
    ----------
    data : tuple
        (input, mask, weights, config)
        - input : Tensor[B, N, N, dim]   (float32)
        - mask  : Tensor[B, N, N]        (float32 / bool)
        - weights: dict of all linear / layer‑norm parameters
        - config : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension
    # ------------------------------------------------------------------
    norm = nn.LayerNorm(dim, elementwise_affine=True).to(device)
    norm.weight.data = weights["norm.weight"]
    norm.bias.data   = weights["norm.bias"]
    x = norm(input_tensor)                                     # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – fused into a single matmul per proj
    # ------------------------------------------------------------------
    left_proj = nn.Linear(dim, hidden_dim, bias=False).to(device)
    right_proj = nn.Linear(dim, hidden_dim, bias=False).to(device)
    left_proj.weight.data = weights["left_proj.weight"]
    right_proj.weight.data = weights["right_proj.weight"]

    left = left_proj(x)   # (B,N,N,hidden_dim)
    right = right_proj(x) # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – also linear layers without bias
    # ------------------------------------------------------------------
    left_gate = nn.Linear(dim, hidden_dim, bias=False).to(device)
    right_gate = nn.Linear(dim, hidden_dim, bias=False).to(device)
    out_gate = nn.Linear(dim, hidden_dim, bias=False).to(device)

    left_gate.weight.data = weights["left_gate.weight"]
    right_gate.weight.data = weights["right_gate.weight"]
    out_gate.weight.data = weights["out_gate.weight"]

    left_gate_val = left_gate(x).sigmoid()
    right_gate_val = right_gate(x).sigmoid()
    out_gate_val = out_gate(x).sigmoid()

    left = left * left_gate_val
    right = right * right_gate_val

    # ------------------------------------------------------------------
    # 5. Core TriMul:   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # left/right are (B,N,N,hidden_dim).  hidden_dim == D here.
    out = trimul_gemm(left, right)        # (B,N,N,hidden_dim), FP32

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    to_out_norm = nn.LayerNorm(hidden_dim, elementwise_affine=True).to(device)
    to_out = nn.Linear(hidden_dim, dim, bias=False).to(device)

    to_out_norm.weight.data = weights["to_out_norm.weight"]
    to_out_norm.bias.data   = weights["to_out_norm.bias"]
    to_out.weight.data      = weights["to_out.weight"]

    out = to_out_norm(out)                # (B,N,N,hidden_dim)
    out = out * out_gate_val              # broadcast gating
    out = to_out(out)                     # (B,N,N,dim)

    return out
