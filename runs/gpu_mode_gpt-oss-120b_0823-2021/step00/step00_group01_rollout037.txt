"""
TriMul “outgoing” implementation.

We keep the bulk of the computation in PyTorch (LayerNorms, Linear
projections and the final einsum) because they are already heavily
optimized.  The only operation we move to Triton is the
mask‑application + gating step, which is memory‑bound and benefits
from a fused kernel.  The kernel multiplies the projected tensors with
the optional mask and the learned sigmoid gates in a single pass,
writing the gated tensors back to contiguous buffers that are then
fed to the original einsum.

The overall algorithm
---------------------
1. Layer‑norm the input (torch).
2. Linear projections to `hidden_dim` (torch, fp32 → fp16 for the
   subsequent GEMM‑like einsum).
3. Load the mask (or a tensor of ones when `nomask=True`) and compute
   the three sigmoid gates.
4. Triton kernel:
      left   = left_proj   * mask * left_gate
      right  = right_proj  * mask * right_gate
   The kernel works on a 3‑D tile (batch, i, j) and a vector of
   channels (hidden_dim).  All operations are fused and use FP16
   arithmetic; the result is stored as FP16.
5. Perform the core “tri‑multiplicative” step with a single
   `einsum('... i k d, ... j k d -> ... i j d', left, right)`.
6. Layer‑norm, out‑gate, final linear (torch, fp32 output).

The kernel is written for Hopper (sm_90) but works on any recent
CUDA GPU.  Block sizes were chosen to give high occupancy on the H200
while keeping shared‑memory usage low.
"""

import torch
import triton
import triton.language as tl
from torch import nn, einsum
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel: fused mask & gating
# --------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr,          # [B, N, N, H] projected tensors (fp16)
    mask_ptr,                     # [B, N, N]   (fp16, 1.0 if no mask)
    left_gate_ptr, right_gate_ptr,# [B, N, N, H] (fp16, sigmoid output)
    out_left_ptr, out_right_ptr,  # destination buffers (fp16)
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_B: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Each program processes a tile of shape
        (batch = pid_b, i ∈ [block_i, block_i+BLOCK_N),
         k ∈ [block_k, block_k+BLOCK_M), h ∈ [0, BLOCK_H)

    It writes the gated tensors back to contiguous output buffers.
    """
    pid = tl.program_id(0)
    num_blocks_b = tl.cdiv(B, BLOCK_B)
    pid_b = pid // (tl.cdiv(N, BLOCK_N) * tl.cdiv(N, BLOCK_M))
    rem = pid % (tl.cdiv(N, BLOCK_N) * tl.cdiv(N, BLOCK_M))
    pid_i = rem // tl.cdiv(N, BLOCK_M)   # tile over i dimension
    pid_k = rem % tl.cdiv(N, BLOCK_M)    # tile over k dimension

    # Offsets
    offs_b = pid_b * BLOCK_B + tl.arange(0, BLOCK_B)
    offs_i = pid_i * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = pid_k * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, BLOCK_H)

    mask_b = pid_b * BLOCK_B                        # mask has no H dim
    mask_i = pid_i * BLOCK_N
    mask_k = pid_k * BLOCK_M

    # Guard against out‑of‑bounds
    mask_i_valid = offs_i < N
    mask_k_valid = offs_k < N
    mask_h_valid = offs_h < H
    mask_b_valid = offs_b < B

    # -----------------------
    # Load mask (broadcast to H)
    # -----------------------
    mask = tl.load(
        mask_ptr
        + offs_b[:, None, None] * stride_mask_b
        + mask_i[None, :, None] * stride_mask_i
        + mask_k[None, None, :] * stride_mask_k,
        mask=mask_b_valid[:, None, None] & mask_i_valid[None, :, None] & mask_k_valid[None, None, :],
        other=0.0,
    )
    mask = mask[:, :, None]  # [B,N,N,1] -> broadcast over H

    # -----------------------
    # Load projections
    # -----------------------
    left = tl.load(
        left_ptr
        + offs_b[:, None, None, None] * stride_left_b
        + offs_i[:, None, None, None] * stride_left_i
        + offs_k[None, :, None, None] * stride_left_k
        + offs_h[None, None, None, :] * stride_left_h,
        mask=mask_b_valid[:, None, None, None]
        & mask_i_valid[:, None, None, None]
        & mask_k_valid[None, :, None, None]
        & mask_h_valid[None, None, None, :],
        other=0.0,
    )
    right = tl.load(
        right_ptr
        + offs_b[:, None, None, None] * stride_right_b
        + offs_i[:, None, None, None] * stride_right_i   # note: reuse i‑offset for right‑j tile
        + offs_k[None, :, None, None] * stride_right_k
        + offs_h[None, None, None, :] * stride_right_h,
        mask=mask_b_valid[:, None, None, None]
        & mask_i_valid[:, None, None, None]
        & mask_k_valid[None, :, None, None]
        & mask_h_valid[None, None, None, :],
        other=0.0,
    )

    # -----------------------
    # Load gates (already sigmoid‑ed)
    # -----------------------
    left_gate = tl.load(
        left_gate_ptr
        + offs_b[:, None, None, None] * stride_gate_b
        + offs_i[:, None, None, None] * stride_gate_i
        + offs_k[None, :, None, None] * stride_gate_k
        + offs_h[None, None, None, :] * stride_gate_h,
        mask=mask_b_valid[:, None, None, None]
        & mask_i_valid[:, None, None, None]
        & mask_k_valid[None, :, None, None]
        & mask_h_valid[None, None, None, :],
        other=0.0,
    )
    right_gate = tl.load(
        right_gate_ptr
        + offs_b[:, None, None, None] * stride_gate_b
        + offs_i[:, None, None, None] * stride_gate_i
        + offs_k[None, :, None, None] * stride_gate_k
        + offs_h[None, None, None, :] * stride_gate_h,
        mask=mask_b_valid[:, None, None, None]
        & mask_i_valid[:, None, None, None]
        & mask_k_valid[None, :, None, None]
        & mask_h_valid[None, None, None, :],
        other=0.0,
    )

    # -----------------------
    # Fuse: (proj * mask) * gate
    # -----------------------
    gated_left = left * mask * left_gate
    gated_right = right * mask * right_gate

    # -----------------------
    # Store results
    # -----------------------
    tl.store(
        out_left_ptr
        + offs_b[:, None, None, None] * stride_out_b
        + offs_i[:, None, None, None] * stride_out_i
        + offs_k[None, :, None, None] * stride_out_k
        + offs_h[None, None, None, :] * stride_out_h,
        gated_left,
        mask=mask_b_valid[:, None, None, None]
        & mask_i_valid[:, None, None, None]
        & mask_k_valid[None, :, None, None]
        & mask_h_valid[None, None, None, :],
    )
    tl.store(
        out_right_ptr
        + offs_b[:, None, None, None] * stride_out_b
        + offs_i[:, None, None, None] * stride_out_i
        + offs_k[None, :, None, None] * stride_out_k
        + offs_h[None, None, None, :] * stride_out_h,
        gated_right,
        mask=mask_b_valid[:, None, None, None]
        & mask_i_valid[:, None, None, None]
        & mask_k_valid[None, :, None, None]
        & mask_h_valid[None, None, None, :],
    )


# --------------------------------------------------------------
# Host‑side helper that launches the kernel
# --------------------------------------------------------------
def apply_mask_and_gates(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    `left/right`      : [B, N, N, H]  fp16
    `mask`            : [B, N, N]     fp16 (1.0 if no mask)
    `left_gate/right_gate` : [B, N, N, H] fp16 (sigmoid)

    Returns gated left/right tensors (fp16) suitable for the einsum.
    """
    B, N, _, H = left.shape
    # allocate contiguous output buffers
    out_left = torch.empty_like(left)
    out_right = torch.empty_like(right)

    # strides (in elements)
    sl = left.stride()
    sr = right.stride()
    sm = mask.stride()
    sg = left_gate.stride()  # same for right_gate

    # launch configuration
    BLOCK_B = 1
    BLOCK_N = 32   # tile over i dimension
    BLOCK_M = 32   # tile over k dimension
    BLOCK_H = 32   # channel tile
    grid = (
        triton.cdiv(B, BLOCK_B)
        * triton.cdiv(N, BLOCK_N)
        * triton.cdiv(N, BLOCK_M),
    )
    mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        out_left_ptr=out_left,
        out_right_ptr=out_right,
        B=B,
        N=N,
        H=H,
        stride_left_b=sl[0],
        stride_left_i=sl[1],
        stride_left_k=sl[2],
        stride_left_h=sl[3],
        stride_right_b=sr[0],
        stride_right_i=sr[1],
        stride_right_k=sr[2],
        stride_right_h=sr[3],
        stride_mask_b=sm[0],
        stride_mask_i=sm[1],
        stride_mask_k=sm[2],
        stride_gate_b=sg[0],
        stride_gate_i=sg[1],
        stride_gate_k=sg[2],
        stride_gate_h=sg[3],
        stride_out_b=out_left.stride(0),
        stride_out_i=out_left.stride(1),
        stride_out_k=out_left.stride(2),
        stride_out_h=out_left.stride(3),
        BLOCK_B=BLOCK_B,
        BLOCK_N=BLOCK_N,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out_left, out_right


# --------------------------------------------------------------
# Main entry point required by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args
    ----
    data : (input, mask, weights, config)
        input   – Tensor[B, N, N, dim]   (float32)
        mask    – Tensor[B, N, N]        (bool or float, optional)
        weights – dict containing all LayerNorm / Linear parameters.
        config  – dict with keys:
                    "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm ----------
    # using torch.nn.functional for speed & simplicity
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias

    # ---------- Linear projections ----------
    # we keep everything in fp32 until after the projection,
    # then cast to fp16 for the fused kernel / einsum
    left_proj_w = weights["left_proj.weight"]          # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"]        # [hidden_dim, dim]

    # torch.nn.functional.linear expects (batch, *, in_features)
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ---------- Gates ----------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ---------- Prepare mask ----------
    if nomask:
        mask_f = torch.ones_like(mask, dtype=torch.float16, device=mask.device)
    else:
        mask_f = mask.to(torch.float16)

    # Cast everything to fp16 for the kernel / einsum
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    # ---------- Triton fused mask & gating ----------
    left_gated, right_gated = apply_mask_and_gates(
        left, right, mask_f, left_gate, right_gate
    )

    # ---------- Core TriMul (einsum) ----------
    # Use bfloat16 for the massive N³ reduction; final result is cast back.
    out = einsum(
        "... i k d, ... j k d -> ... i j d",
        left_gated.to(torch.bfloat16),
        right_gated.to(torch.bfloat16),
    )
    out = out.to(torch.float32)

    # ---------- Output projection ----------
    # to_out_norm (LayerNorm over hidden_dim)
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate  # broadcast over hidden_dim

    # final linear back to dim
    out = torch.nn.functional.linear(out, to_out_w)

    return out
