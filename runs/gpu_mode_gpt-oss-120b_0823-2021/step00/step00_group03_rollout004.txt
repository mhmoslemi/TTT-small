"""
TriMul “outgoing” forward pass.

The heavy part is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply per hidden channel:
    out_d = left_d @ right_d.T      (left_d/right_d : [B,N,N])

We keep LayerNorms / Linear / gating in PyTorch (they are cheap) and
implement the batched matmul with a single Triton kernel that works on
FP16 for speed and casts the result back to FP32.
The kernel is written for arbitrary batch size * hidden dimension
(bh = B * hidden_dim) and any sequence length N.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ---------------------------------------------------------------
# Triton kernel: batched (B*H) matmul   C = A @ B.T
# ---------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    bh, N,                # bh = B*hidden_dim
    stride_ah, stride_ak,   # A strides: (bh,N,N) -> (bh,N,N)
    stride_bk, stride_bn,   # B strides (bh,N,N)
    stride_ch, stride_cm, stride_cn,  # C strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    FP16: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_h = tl.program_id(2)          # batch*hidden index

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # Pointers for this (h) instance
    a_base = a_ptr + pid_h * stride_ah
    b_base = b_ptr + pid_h * stride_bk
    c_base = c_ptr + pid_h * stride_ch

    # Load tiles of A and B
    a_ptrs = a_base + (offs_m[:, None] * stride_ak + offs_k[None, :] * stride_ah)  # (M,K)
    b_ptrs = b_base + (offs_n[None, :] * stride_bn + offs_k[:, None] * stride_bk)  # (K,N)   (note: we will use B.T)

    a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
    b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # we already loaded a & b for the first block; subsequent blocks are loaded inside the loop
        acc += tl.dot(a, b)                     # (M,K)·(K,N) → (M,N)
        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

    # write back
    c_ptrs = c_base + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16) if FP16 else acc, mask=mask_m[:, None] & mask_n[None, :])

# ---------------------------------------------------------------
# Helper to launch the kernel
# ---------------------------------------------------------------
def batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a, b : [B, hidden, N, N]   (float16)
    returns c : [B, hidden, N, N] (float32)
    """
    B, H, N, _ = a.shape
    bh = B * H
    a = a.contiguous()
    b = b.contiguous()
    c = torch.empty_like(a, dtype=torch.float32, device=a.device)

    # strides
    stride_ah = a.stride(0)          # bh stride
    stride_ak = a.stride(2)          # N stride (row)
    stride_bk = b.stride(0)
    stride_bn = b.stride(2)
    stride_ch = c.stride(0)
    stride_cm = c.stride(2)
    stride_cn = c.stride(3)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        bh,
    )
    batched_matmul_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        bh=bh,
        N=N,
        stride_ah=stride_ah,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_ch=stride_ch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        FP16=True,
    )
    return c


# ---------------------------------------------------------------
# Custom kernel entry point
# ---------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul using Triton for the
    N^3 einsum part.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm ----------
    # x shape: [B, N, N, dim]
    x = input_tensor
    # LayerNorm over last dim using supplied weights/bias
    eps = 1e-5
    weight = weights["norm.weight"].to(x.dtype)
    bias = weights["norm.bias"].to(x.dtype)
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps) * weight + bias          # [B,N,N,dim]

    # ---------- Linear projections ----------
    # reshape to (..., dim) → (..., hidden_dim)
    def linear(inp, w):
        # w: [hidden_dim, dim]  (no bias)
        return torch.nn.functional.linear(inp, w)

    left  = linear(x_norm, weights["left_proj.weight"].to(x.dtype))      # [B,N,N,hd]
    right = linear(x_norm, weights["right_proj.weight"].to(x.dtype))

    # ---------- Mask ----------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                     # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ---------- Gates ----------
    left_gate  = torch.sigmoid(linear(x_norm, weights["left_gate.weight"].to(x.dtype)))
    right_gate = torch.sigmoid(linear(x_norm, weights["right_gate.weight"].to(x.dtype)))
    out_gate   = torch.sigmoid(linear(x_norm, weights["out_gate.weight"].to(x.dtype)))

    left  = left * left_gate
    right = right * right_gate

    # ---------- Core einsum via Triton ----------
    # reshape to [B, hidden, N, N] and cast to fp16 for the kernel
    B, N, _, _ = left.shape
    left_  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # [B,hd,N,N]
    right_ = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    # batched matmul per hidden channel: out_h = left_h @ right_h.T
    out_ = batched_matmul_fp16(left_, right_)                        # [B,hd,N,N] float32

    # bring back to original layout [B,N,N,hd]
    out = out_.permute(0, 2, 3, 1).contiguous()                      # [B,N,N,hd]

    # ---------- Final projection ----------
    # LayerNorm over hidden dim
    out = (out - out.mean(-1, keepdim=True)) * torch.rsqrt(out.var(-1, unbiased=False, keepdim=True) + eps)
    out = out * weights["to_out_norm.weight"].to(out.dtype) + weights["to_out_norm.bias"].to(out.dtype)

    out = out * out_gate                                            # gating

    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(out.dtype))  # back to dim

    return out.float()
