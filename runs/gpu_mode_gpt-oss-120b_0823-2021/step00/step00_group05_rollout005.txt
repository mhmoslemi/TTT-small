"""
TriMul “outgoing” implementation.

The heavy part of the forward pass is the tensor contraction

    out[b,i,j,d] = Σ_k  left[b,i,k,d] * right[b,j,k,d]

where   left  = (x @ W_left)  * mask * sigmoid(x @ W_left_gate)
        right = (x @ W_right) * mask * sigmoid(x @ W_right_gate)

Both left and right have shape   [B, N, N, H]  (H = hidden_dim).

The kernel below computes the above contraction.  Each program instance
handles a single (b,i,j) triple and reduces over the K‑dimension in
tiles of size BLOCK_K.  The hidden dimension is processed as a small
vector (BLOCK_D ≤ 64) that fits in a register; the inner‐most reduction
over K is performed with an explicit Python‑style loop (fast because
BLOCK_K is modest).  The result is written in FP16 and later cast back
to FP32 in the host code.

All LayerNorm and Linear layers (including the final LayerNorm and
projection) are performed with regular torch operations – they are
memory‑bound and cheap compared with the O(N³·H) contraction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# -----------------------------------------------------------------------
# Triton kernel: computes   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# -----------------------------------------------------------------------
@triton.jit
def trimul_out_kernel(
    left_ptr,      # *float16  [B,N,N,H]
    right_ptr,     # *float16  [B,N,N,H]
    out_ptr,       # *float16  [B,N,N,H]

    B, N, H,                     # tensor dimensions
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides

    BLOCK_K: tl.constexpr,       # tile size along K
    BLOCK_D: tl.constexpr,       # vector size along hidden dim
):
    pid = tl.program_id(0)

    # ---------------------------------------------------------------
    # decode 1‑D program id into (b,i,j)
    # ---------------------------------------------------------------
    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i = ij // N
    j = ij % N

    # ----------------------------------------------------------------
    # offsets for the hidden dimension (vectorized)
    # ----------------------------------------------------------------
    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < H

    # accumulator for the hidden dimension, FP32 for precision
    acc = tl.zeros([BLOCK_D], dtype=tl.float32)

    # ----------------------------------------------------------------
    # loop over K in tiles
    # ----------------------------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        cur_k = k_base + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        mask_k = cur_k < N                               # valid K mask

        # ----------------------------------------------------------------
        # pointers for the current tile (left: [B,i,k,d], right: [B,j,k,d])
        # ----------------------------------------------------------------
        left_tile_ptr  = (left_ptr
                          + b * stride_lb
                          + i * stride_li
                          + cur_k[:, None] * stride_lk
                          + offs_d[None, :] * stride_ld)

        right_tile_ptr = (right_ptr
                          + b * stride_rb
                          + j * stride_rj
                          + cur_k[:, None] * stride_rk
                          + offs_d[None, :] * stride_rd)

        # ----------------------------------------------------------------
        # inner reduction over K (explicit loop, BLOCK_K is small)
        # ----------------------------------------------------------------
        for kk in range(0, BLOCK_K):
            k_valid = mask_k[kk] & mask_d            # broadcasted mask
            l = tl.load(left_tile_ptr + kk * stride_lk,
                        mask=k_valid, other=0.0).to(tl.float32)
            r = tl.load(right_tile_ptr + kk * stride_rk,
                        mask=k_valid, other=0.0).to(tl.float32)
            acc += l * r

    # ----------------------------------------------------------------
    # write result
    # ----------------------------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_ob
                + i * stride_oi
                + j * stride_oj
                + offs_d * stride_od)

    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_d)


# -----------------------------------------------------------------------
# Host entry point
# -----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.
    Arguments
    ---------
    data : (input, mask, weights, config)
        input  – [B, N, N, dim]   (float32)
        mask   – [B, N, N]        (float32, 0/1)
        weights – dict of learned parameters (float32)
        config – dict with keys  dim, hidden_dim
    Returns
    -------
    output – [B, N, N, dim] (float32)
    """
    # -------------------------------------------------------------------
    # unpack
    # -------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # -------------------------------------------------------------------
    # helper to apply a linear layer without bias (weight shape [out,in])
    # -------------------------------------------------------------------
    def linear(x, w):
        # x: [..., dim]   w: [out, dim]
        return torch.nn.functional.linear(x, w)

    # -------------------------------------------------------------------
    # LayerNorm (no bias handling needed because bias is passed explicitly)
    # -------------------------------------------------------------------
    def layer_norm(x, weight, bias):
        # x: [..., dim]
        mean = x.mean(-1, keepdim=True)
        var = x.var(-1, unbiased=False, keepdim=True)
        return (x - mean) * torch.rsqrt(var + 1e-5) * weight + bias

    # -------------------------------------------------------------------
    # 1. Input LayerNorm
    # -------------------------------------------------------------------
    x = layer_norm(input_tensor,
                   weights["norm.weight"],
                   weights["norm.bias"])

    # -------------------------------------------------------------------
    # 2. Linear projections + mask
    # -------------------------------------------------------------------
    left = linear(x, weights["left_proj.weight"])
    right = linear(x, weights["right_proj.weight"])

    if mask is not None:
        mask_ = mask.unsqueeze(-1)                      # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # -------------------------------------------------------------------
    # 3. Gating
    # -------------------------------------------------------------------
    left_gate = torch.sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------
    # 4. Cast to fp16 for the heavy contraction
    # -------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # output buffer
    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16,
                           device=device)

    # ---------------------------------------------------------------
    # launch parameters
    # ---------------------------------------------------------------
    BLOCK_K = 32                      # tile along K
    BLOCK_D = 32 if hidden_dim > 32 else hidden_dim   # vector width
    total_programs = B * N * N
    grid = (total_programs,)

    # strides (triton expects int64)
    stride_lb, stride_li, stride_lk, stride_ld = left_fp16.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_fp16.stride()

    trimul_out_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,   # enough for fp16 compute
    )

    # -------------------------------------------------------------------
    # 5. Final LayerNorm + out gating + projection back to dim
    # -------------------------------------------------------------------
    out = out_fp16.to(torch.float32)
    out = layer_norm(out,
                     weights["to_out_norm.weight"],
                     weights["to_out_norm.bias"])
    out = out * out_gate
    out = linear(out, weights["to_out.weight"])

    return out
