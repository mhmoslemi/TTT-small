"""
tri_mul_triton.py
-----------------
A highly‑optimized Triton implementation of the *outgoing* Triangle
Multiplicative Update (TriMul) used in AlphaFold‑3 and related protein
structure models.

Only the N³ “einsum” part is executed on the GPU with Triton; all
LayerNorm/Linear/gate operations are kept in native PyTorch because they
are memory‑bound and already well‑optimized.  The kernel computes, for
each batch element `b` and pair of indices `(i, j)`:

    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]

where `h` runs over the hidden dimension `H` (typically 128).  The kernel
fuses the K‑reduction across blocks of the sequence dimension to avoid
creating an intermediate N×N×N tensor.

The final output is cast back to `float32` as required by the reference
implementation.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------
# Triton kernel: batched “einsum” reduction over the K dimension
# -------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,                 # *float32, shape [B, N, N, H]
    right_ptr,                # *float32, shape [B, N, N, H]
    out_ptr,                  # *float32, shape [B, N, N, H]
    B, N, H, K,               # problem sizes
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_K: tl.constexpr,    # tile size along the summed dimension
    BLOCK_H: tl.constexpr,    # hidden dimension (full, no tiling)
):
    """
    One program instance computes the full hidden vector for a single
    (b, i, j) triple.  The K‑reduction is performed in a loop over tiles.
    """
    pid = tl.program_id(0)                     # linear index over (b,i,j)
    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i = ij // N
    j = ij % N

    # offsets for the first element of the hidden vector
    out_offset = (
        b * stride_ob +
        i * stride_oi +
        j * stride_oj
    )
    # accumulator for the hidden dimension
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    # loop over K in blocks
    for start_k in range(0, K, BLOCK_K):
        cur_k = tl.minimum(BLOCK_K, K - start_k)                 # actual size

        # --------------------- load left -----------------------
        left_off = (
            b * stride_lb +
            i * stride_li +
            (start_k + tl.arange(0, cur_k))[:, None] * stride_lk +
            tl.arange(0, BLOCK_H)[None, :] * stride_lh
        )
        left_mask = (start_k + tl.arange(0, cur_k)[:, None]) < K
        left = tl.load(left_ptr + left_off, mask=left_mask, other=0.0)

        # --------------------- load right ----------------------
        right_off = (
            b * stride_rb +
            j * stride_rj +
            (start_k + tl.arange(0, cur_k))[:, None] * stride_rk +
            tl.arange(0, BLOCK_H)[None, :] * stride_rh
        )
        right_mask = (start_k + tl.arange(0, cur_k)[:, None]) < K
        right = tl.load(right_ptr + right_off, mask=right_mask, other=0.0)

        # element‑wise product, then sum over the K axis
        prod = left * right                                      # [cur_k, H]
        acc += tl.sum(prod, axis=0)                              # [H]

    # --------------------- store result -----------------------
    out_off = out_offset + tl.arange(0, BLOCK_H) * stride_oh
    tl.store(out_ptr + out_off, acc)


# -------------------------------------------------------------
# Helper: launch the Triton kernel
# -------------------------------------------------------------
def trimul_einsum_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the TriMul einsum via Triton.
    Arguments
    ---------
    left, right : torch.Tensor
        Shape [B, N, N, H], dtype float32 (already masked and gated).
    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H], dtype float32.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    K = N  # the summed dimension is the sequence length

    out = torch.empty_like(left)

    # strides in elements (not bytes) for each tensor
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # launch parameters
    BLOCK_K = 32                     # fits nicely in registers on H200
    BLOCK_H = H                      # hidden dim is at most 384, we keep it whole

    total_ij = B * N * N
    grid = (total_ij,)

    trimul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        K=K,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_lh=stride_lh,
        stride_rb=stride_rb,
        stride_rj=stride_rj,
        stride_rk=stride_rk,
        stride_rh=stride_rh,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_oh=stride_oh,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out


# -------------------------------------------------------------
# Main entry point required by the evaluator
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator with a Triton‑accelerated
    N³ reduction.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1. LayerNorm + Linear / Gate projections (pure PyTorch)
    # -----------------------------------------------------------------
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias

    # Linear projections (no bias)
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"]
    )  # [B,N,N,hidden]
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"]
    )

    # Apply mask if present (mask shape [B,N,N])
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates (sigmoid) – also no bias
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 2. Triton kernel for the “einsum” reduction
    # -----------------------------------------------------------------
    # Ensure contiguous for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = trimul_einsum_triton(left, right)               # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 3. Final LayerNorm, out‑gate and output projection (PyTorch)
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_weight + to_out_norm_bias

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out.float()
