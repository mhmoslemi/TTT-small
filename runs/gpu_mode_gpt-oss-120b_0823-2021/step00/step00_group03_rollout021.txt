"""
TriMul “outgoing” implementation.
--------------------------------
* LayerNorm + Linear + sigmoid gates are executed with PyTorch (they are
  memory‑bandwidth bound and already highly optimized).
* The expensive N³‑like contraction

        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

  is performed by a custom Triton kernel that tiles the reduction
  dimension `k` and the channel dimension `d` (hidden_dim).  The kernel
  computes a full (B,N,N,hidden_dim) tensor in FP16 and returns FP32
  after the final LayerNorm/Linear.

The kernel works for any batch size ≤2, any sequence length up to 1024,
and any hidden dimension that is a multiple of the block size (the
kernel also handles non‑multiples via masking).  All tensors are
expected to reside on GPU.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel that computes the einsum contraction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr,                 # *const float16
    right_ptr,                # *const float16
    out_ptr,                  # *float16
    B, N, H,                  # shape parameters
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_H: tl.constexpr,    # channel block size (e.g. 64)
    BLOCK_K: tl.constexpr,    # reduction tile size (e.g. 32)
):
    """Compute out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]"""
    pid = tl.program_id(0)                     # 1‑D grid over (b,i,j)
    total_ij = B * N * N
    pid_ij = pid % total_ij

    b = pid_ij // (N * N)
    tmp = pid_ij % (N * N)
    i = tmp // N
    j = tmp % N

    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # accumulator in FP32 for numeric stability
    acc = tl.zeros((BLOCK_H,), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k‑tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_tiles):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for the current k‑tile
        left_off = (
            b * stride_left_b +
            i * stride_left_i +
            offs_k[:, None] * stride_left_k +
            offs_h[None, :] * stride_left_h
        )
        right_off = (
            b * stride_right_b +
            j * stride_right_j +
            offs_k[:, None] * stride_right_k +
            offs_h[None, :] * stride_right_h
        )

        # load (BLOCK_K, BLOCK_H) tiles; mask out‑of‑bounds entries
        left_tile  = tl.load(left_ptr + left_off,
                             mask=mask_k[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)
        right_tile = tl.load(right_ptr + right_off,
                             mask=mask_k[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)

        # element‑wise product + reduction over k
        prod = left_tile * right_tile                # (BK, BH)
        acc += tl.sum(prod, axis=0)                  # (BH,)

    # ------------------------------------------------------------------
    # Write result
    # ------------------------------------------------------------------
    out_off = (
        b * stride_out_b +
        i * stride_out_i +
        j * stride_out_j +
        offs_h * stride_out_h
    )
    tl.store(out_ptr + out_off, acc.to(tl.float16), mask=mask_h)


# ----------------------------------------------------------------------
# Helper functions (LayerNorm & Linear without bias for speed)
# ----------------------------------------------------------------------
def _apply_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor) -> torch.Tensor:
    # x: [..., C]  (float32)
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    std = torch.rsqrt(var + 1e-5)
    return (x - mean) * std * weight + bias


def _apply_linear(x: torch.Tensor, w: torch.Tensor) -> torch.Tensor:
    # w: (out, in) weight matrix, bias already folded into x if needed
    return torch.nn.functional.linear(x, w)   # bias=False


# ----------------------------------------------------------------------
# Main entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module with a Triton‑accelerated
    contraction.  All tensors are assumed to be on CUDA.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]                 # input & output channel dim
    hidden_dim = config["hidden_dim"]   # inner channel dim (H)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    ##################################################################
    # 1. LayerNorm on the input (float32)
    ##################################################################
    x = _apply_layernorm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"]
    )

    ##################################################################
    # 2. Linear projections (no bias) – keep as float32
    ##################################################################
    left  = _apply_linear(x, weights["left_proj.weight"])
    right = _apply_linear(x, weights["right_proj.weight"])

    # mask: [B,N,N] → [B,N,N,1] for broadcasting
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # fp32 mask
        left  = left * mask_
        right = right * mask_

    ##################################################################
    # 3. Gating (sigmoid) – all in fp32
    ##################################################################
    left_gate  = torch.sigmoid(_apply_linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(_apply_linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(_apply_linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    ##################################################################
    # 4. Triton contraction (compute out = Σ_k left[...,k,:] * right[...,k,:])
    ##################################################################
    # Cast to FP16 for the heavy reduction (still accurate enough)
    left_f16  = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()
    out_f16   = torch.empty((B, N, N, hidden_dim),
                            dtype=torch.float16, device=device)

    # strides (in elements, not bytes) for contiguous tensors
    s_b, s_i, s_k, s_h = left_f16.stride()   # (B,N,N,H) layout
    # right uses the same layout
    rs_b, rs_i, rs_k, rs_h = right_f16.stride()
    os_b, os_i, os_k, os_h = out_f16.stride()

    # tuning parameters – chosen to fit H200 SM90 shared memory comfortably
    BLOCK_H = 64          # hidden_dim tile (must divide 128 or be masked)
    BLOCK_K = 32          # reduction tile over sequence dimension

    total_programs = B * N * N
    grid = (total_programs,)

    _trimul_einsum_kernel[grid](
        left_f16,
        right_f16,
        out_f16,
        B, N, hidden_dim,
        s_b, s_i, s_k, s_h,
        rs_b, rs_i, rs_k, rs_h,
        os_b, os_i, os_k, os_h,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
    )

    # back to FP32 for the remaining ops
    out = out_f16.to(torch.float32)

    ##################################################################
    # 5. Output LayerNorm + gating + final linear
    ##################################################################
    out = _apply_layernorm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = _apply_linear(out, weights["to_out.weight"])

    return out
