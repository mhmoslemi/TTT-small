"""
TriMul “outgoing” forward pass with the heavy N³ tensor contraction off‑loaded to
a Triton kernel.

The contraction is:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each head dimension *d* this is exactly a batched matrix‑multiply
    out[b,:,:,d] = left[b,:,:,d] @ right[b,:,:,d].T
We implement this with a 3‑D launch (batch, block_i, block_j) and a
K‑loop tiled by BLOCK_K.  All other operations (layer‑norms, linear
projections, sigmoid gates and the final linear) are kept in PyTorch – they
are cheap compared to the N³ contraction.

The kernel works in fp16/bf16 and writes float32 results, matching the
reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched GEMM for each head dimension (hidden_dim)
# --------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    # pointers
    left_ptr,          # [B, N, N, H]
    right_ptr,         # [B, N, N, H]
    out_ptr,           # [B, N, N, H]
    # sizes
    B, N, H,
    # strides (in elements)
    stride_blh, stride_lih, stride_lkh, stride_ldh,   # left: b,l,i,k,d
    stride_brh, stride_rjh, stride_rkh, stride_rdh,   # right: b,r,j,k,d
    stride_boh, stride_oih, stride_ojh, stride_odh,   # out:   b,o,i,j,d
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,    # hidden dim tile
    EPS: tl.constexpr = 1e-6
):
    # program ids -------------------------------------------------
    batch_id = tl.program_id(0)
    pid_i    = tl.program_id(1)          # block on output i dimension
    pid_j    = tl.program_id(2)          # block on output j dimension

    # block start offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = tl.arange(0, BLOCK_D)               # hidden dim within tile

    # mask for out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # pointers offset to the start of the current tile
    left_tile_ptr  = left_ptr + batch_id * stride_blh \
                     + offs_i[:, None, None] * stride_lih \
                     + 0 * stride_lkh \
                     + offs_d[None, None, :] * stride_ldh

    right_tile_ptr = right_ptr + batch_id * stride_brh \
                     + offs_j[None, :, None] * stride_rjh \
                     + 0 * stride_rkh \
                     + offs_d[None, None, :] * stride_rdh

    out_tile_ptr   = out_ptr + batch_id * stride_boh \
                     + offs_i[:, None, None] * stride_oih \
                     + offs_j[None, :, None] * stride_ojh \
                     + offs_d[None, None, :] * stride_odh

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # -----------------------------------------------------------------
    # K‑loop: sum over the third axis (k)
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # offsets for this k‑tile
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # load left: shape [M, K, D]
        l_ptrs = left_tile_ptr \
                 + cur_k[None, :, None] * stride_lkh
        left = tl.load(l_ptrs, mask=mask_i[:, None] & mask_k[None, :] & mask_d[None, None, :], other=0.0)

        # load right: shape [N, K, D] (note we need right[..., j, k, d])
        r_ptrs = right_tile_ptr \
                 + cur_k[None, :, None] * stride_rkh
        right = tl.load(r_ptrs, mask=mask_j[None, :] & mask_k[:, None] & mask_d[None, None, :], other=0.0)

        # dot over k dimension for each d (using tl.dot)
        # tl.dot expects 2‑D tensors, so we iterate over d inside the tile
        for d in range(0, BLOCK_D):
            # shapes: left [:, :, d]  (M x K), right [:, :, d] (N x K)
            l = left[:, :, d]          # (M, K)
            r = right[:, :, d]         # (N, K)

            # accumulate MxN matrix product
            acc[:, :, d] += tl.dot(l, r, trans_b=True)   # (M,N)

        # advance k‑pointers for next tile
        left_tile_ptr  += BLOCK_K * stride_lkh
        right_tile_ptr += BLOCK_K * stride_rkh

    # -----------------------------------------------------------------
    # Store result (convert to float16/bfloat16 if desired)
    # -----------------------------------------------------------------
    out = acc.to(tl.float16)          # keep half precision in memory
    tl.store(out_tile_ptr, out,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])


# --------------------------------------------------------------
# Host‑side helper to launch the kernel
# --------------------------------------------------------------
def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, H]  (fp16 / bf16)
    returns out : [B, N, N, H]  (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in elements (not bytes)
    stride_blh, stride_lih, stride_lkh, stride_ldh = left.stride()
    stride_brh, stride_rjh, stride_rkh, stride_rdh = right.stride()
    stride_boh, stride_oih, stride_ojh, stride_odh = out.stride()

    # tile sizes – tuned for Hopper SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 32   # hidden dim chunk (hidden_dim is ≤ 384)

    grid = (B,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    trimul_gemm_kernel[grid](
        left, right, out,
        B, N, H,
        stride_blh, stride_lih, stride_lkh, stride_ldh,
        stride_brh, stride_rjh, stride_rkh, stride_rdh,
        stride_boh, stride_oih, stride_ojh, stride_odh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=8,  # good for 32×32 tiles on Hopper
    )
    return out


# --------------------------------------------------------------
# Full custom kernel entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            input  – torch.Tensor [B, N, N, dim] (float32)
            mask   – torch.Tensor [B, N, N]      (bool / float)
            weights – dict of model parameters (float32)
            config – dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm on dim ----------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    # use fp16 inside the heavy kernel
    x_h = x.to(torch.float16)

    # ---------- Linear projections ----------
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x_h, left_proj_w)                 # [B,N,N,hidden]
    right = F.linear(x_h, right_proj_w)

    # ---------- Apply mask ----------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x_h.dtype)      # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ---------- Gates ----------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ---------- N³ contraction via Triton ----------
    # left/right are [B,N,N,H] in fp16 – feed to kernel
    out = trimul_gemm(left, right)          # float32 result

    # ---------- Output LayerNorm + gate + linear ----------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden dim

    to_out_w = weights["to_out.weight"]      # (dim, hidden_dim)
    out = F.linear(out, to_out_w)            # final shape [B,N,N,dim]

    return out.to(torch.float32)
