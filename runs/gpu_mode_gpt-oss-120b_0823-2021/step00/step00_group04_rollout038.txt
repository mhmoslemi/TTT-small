"""
TriMul “outgoing” kernel (AlphaFold‑3)

The expensive part of the TriMul forward pass is the 4‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication for every head‑dimension *d*.
The kernel below computes this contraction by tiling the (i,j) output plane
with BLOCK_M × BLOCK_N tiles and iterating over the reduction dimension *k*
with tiles of size BLOCK_K.  All arithmetic is performed in fp32 and the
final result is written back in fp32.  The surrounding LayerNorm, Linear and
gate operations are kept in PyTorch (they are cheap compared to the O(N³)
contraction).

The implementation works for any batch size B∈{1,2}, sequence length N up to
1024 and hidden dimension H (named ``hidden_dim`` in the config).  It also
handles the optional mask – the mask is applied **before** the kernel is
launched, so the kernel always sees a dense tensor.
"""

import torch
import triton
import triton.language as tl
from torch import nn

# ----------------------------------------------------------------------
# Triton kernel: batched matmul over the reduction axis k
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr,           # const float*   [B, N, N, H]
    right_ptr,          # const float*   [B, N, N, H]
    out_ptr,            #       float*   [B, N, N, H]

    # sizes
    B, N, H,            # batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    # tile sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)               # linear id over (batch, head, grid_i, grid_j)

    # decode program id
    num_heads = tl.cdiv(H, BLOCK_N)      # we tile the hidden dimension with BLOCK_N (same as N‑tile)
    num_grid_i = tl.cdiv(N, BLOCK_M)
    num_grid_j = tl.cdiv(N, BLOCK_N)

    # layout of the launch grid: (b, h, i_tile, j_tile)
    b = pid // (num_heads * num_grid_i * num_grid_j)
    rem = pid % (num_heads * num_grid_i * num_grid_j)
    h = rem // (num_grid_i * num_grid_j)
    rem = rem % (num_grid_i * num_grid_j)
    i_tile = rem // num_grid_j
    j_tile = rem % num_grid_j

    # starting offsets for this block
    i_offset = i_tile * BLOCK_M
    j_offset = j_tile * BLOCK_N
    h_offset = h * BLOCK_N                 # we treat hidden dim like the N‑tile

    # masks for out‑of‑bounds rows/cols/head
    mask_i = i_offset + tl.arange(0, BLOCK_M) < N
    mask_j = j_offset + tl.arange(0, BLOCK_N) < N
    mask_h = h_offset + tl.arange(0, BLOCK_N) < H

    # pointers to the first element of the tile
    left_tile_ptr  = left_ptr  + b * stride_left_b \
                                + i_offset * stride_left_i \
                                + 0 * stride_left_k \
                                + h_offset * stride_left_h
    right_tile_ptr = right_ptr + b * stride_right_b \
                                + j_offset * stride_right_j \
                                + 0 * stride_right_k \
                                + h_offset * stride_right_h
    out_tile_ptr   = out_ptr   + b * stride_out_b \
                                + i_offset * stride_out_i \
                                + j_offset * stride_out_j \
                                + h_offset * stride_out_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction axis k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K

        # load a tile of left: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_tile_ptr
            + k_offset * stride_left_k
            + tl.arange(0, BLOCK_M)[:, None] * stride_left_i
            + tl.arange(0, BLOCK_K)[None, :] * stride_left_k,
            mask=mask_i[:, None] & (k_offset + tl.arange(0, BLOCK_K) < N),
            other=0.0,
        )
        # load a tile of right: shape (BLOCK_N, BLOCK_K)  (note the k dimension is the *inner* one)
        right = tl.load(
            right_tile_ptr
            + k_offset * stride_right_k
            + tl.arange(0, BLOCK_N)[:, None] * stride_right_j
            + tl.arange(0, BLOCK_K)[None, :] * stride_right_k,
            mask=mask_j[:, None] & (k_offset + tl.arange(0, BLOCK_K) < N),
            other=0.0,
        )
        # right needs to be transposed to align k with left's second dim
        right_T = tl.trans(right)          # (BLOCK_K, BLOCK_N)

        # dot‑product over k
        acc += tl.dot(left, right_T)       # (BLOCK_M, BLOCK_N)

    # write result
    tl.store(
        out_tile_ptr
        + tl.arange(0, BLOCK_M)[:, None] * stride_out_i
        + tl.arange(0, BLOCK_N)[None, :] * stride_out_j,
        acc,
        mask=mask_i[:, None] & mask_j[None, :] & mask_h[None, :],
    )


# ----------------------------------------------------------------------
# Helper – launch wrapper
# ----------------------------------------------------------------------
def launch_trimul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    Arguments
    ---------
    left, right : torch.Tensor
        Shape [B, N, N, H], dtype torch.float32, on CUDA

    Returns
    -------
    out : torch.Tensor
        Same shape & dtype as inputs
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape

    out = torch.empty_like(left)          # [B,N,N,H]

    # strides (in elements, not bytes)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # tile sizes – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # number of program instances
    num_heads = (H + BLOCK_N - 1) // BLOCK_N
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    total_progs = B * num_heads * grid_i * grid_j

    grid = (total_progs,)

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=s_lb,
        stride_left_i=s_li,
        stride_left_k=s_lk,
        stride_left_h=s_lh,
        stride_right_b=s_rb,
        stride_right_j=s_rj,
        stride_right_k=s_rk,
        stride_right_h=s_rh,
        stride_out_b=s_ob,
        stride_out_i=s_oi,
        stride_out_j=s_oj,
        stride_out_h=s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # good default for fp32 on H200
    )
    return out


# ----------------------------------------------------------------------
# Full TriMul forward (mostly PyTorch, kernel for the O(N³) part)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul “outgoing” operator.

    Input tuple:
        input_tensor : torch.Tensor   [B, N, N, dim]      (float32)
        mask         : torch.Tensor   [B, N, N]           (float32 or bool)
        weights      : dict of torch.Tensor (model parameters)
        config       : dict with keys "dim", "hidden_dim"

    Returns
        torch.Tensor  [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # manual layer‑norm (no mean/var sync needed because dim is small)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * norm_weight + norm_bias          # shape [B,N,N,dim]

    # ---------- Linear projections ----------
    # Linear: y = x @ W.T   (bias = 0)
    left_proj_weight = weights["left_proj.weight"]          # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]        # [hidden_dim, dim]

    left = torch.einsum('b i j d, h d -> b i j h', x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.einsum('b i j d, h d -> b i j h', x, right_proj_weight)

    # ---------- Apply mask (if present) ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)                # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gates ----------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(torch.einsum('b i j d, h d -> b i j h', x, left_gate_weight))
    right_gate = torch.sigmoid(torch.einsum('b i j d, h d -> b i j h', x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.einsum('b i j d, h d -> b i j h', x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ---------- Contraction (the heavy O(N³) part) ----------
    # left : [B,N,N,hidden]   right : same
    out = launch_trimul(left, right)        # [B,N,N,hidden]

    # ---------- Output projection ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]          # [dim, hidden]

    # LayerNorm over hidden dimension
    mean_h = out.mean(dim=-1, keepdim=True)
    var_h  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_h) / torch.sqrt(var_h + 1e-5)
    out = out * to_out_norm_weight + to_out_norm_bias

    out = out * out_gate                                     # gate after norm

    # final linear
    out = torch.einsum('b i j h, d h -> b i j d', out, to_out_weight)

    return out.to(torch.float32)
