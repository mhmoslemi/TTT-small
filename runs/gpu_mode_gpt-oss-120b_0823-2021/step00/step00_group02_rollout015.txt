"""
TriMul “outgoing” forward pass (AlphaFold3 style).

The heavy part is the 3‑D reduction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply for every hidden dimension `d`.
We keep the surrounding LayerNorm / Linear / gating logic in PyTorch
and fuse the N³ reduction into a single Triton kernel.  The kernel
loads tiles of `left` and `right` (shape B×N×N×D) and accumulates the
sum over `k` using `tl.sum`.  The final result is cast back to float32
so the rest of the network can run in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel for the core 3‑D reduction
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,                     # pointers
    B, N, D,                                          # tensor sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # ---------------------------------------------------------------
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute out = Σ_k left[...,k,:] * right[...,k,:]"""
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # tile on i (seq_len)
    pid_j = tl.program_id(2)          # tile on j (seq_len)

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_d = tl.arange(0, BLOCK_D)

    # mask for out‑of‑bounds on i/j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialise accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over k dimension in blocks
    # ------------------------------------------------------------------
    for cur_k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = cur_k * BLOCK_K
        cur_offs_k = k_base + offs_k
        mask_k = cur_offs_k < N

        # load left[b,i,k,d]  -> (M, K, D)
        left = tl.load(
            left_ptr
            + pid_b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + cur_offs_k[None, :, None] * stride_left_k
            + offs_d[None, None, :] * stride_left_d,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & (offs_d[None, None, :] < D),
            other=0.0,
        )
        # load right[b,j,k,d] -> (N, K, D)
        right = tl.load(
            right_ptr
            + pid_b * stride_right_b
            + offs_j[:, None, None] * stride_right_j
            + cur_offs_k[None, :, None] * stride_right_k
            + offs_d[None, None, :] * stride_right_d,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & (offs_d[None, None, :] < D),
            other=0.0,
        )

        # element‑wise product and reduction over k
        # left: (M,K,D)   right: (N,K,D)   -> broadcast to (M,N,K,D)
        # sum over K -> (M,N,D)
        prod = left[:, None, :, :] * right[None, :, :, :]   # (M,N,K,D)
        acc += tl.sum(prod, axis=2)                        # sum over K

    # ------------------------------------------------------------------
    # write back, cast to float32 (the rest of the model expects FP32)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_d[None, None, :] * stride_out_d
    )
    mask_out = mask_i[:, None, None] & mask_j[None, :, None] & (offs_d[None, None, :] < D)
    tl.store(out_ptrs, acc, mask=mask_out)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    TriMul forward pass with the N³ reduction fused in Triton.

    Args
    ----
    data : (input, mask, weights, config)
        input  – Tensor[B, N, N, dim] (float32)
        mask   – Tensor[B, N, N]      (float32 or bool)
        weights – dict of model parameters
        config  – dict containing "dim" and "hidden_dim"

    Returns
    -------
    Tensor[B, N, N, dim] – the module output in float32
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1) LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # torch.layer_norm works per‑element across the last dim
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hd]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Apply mask (if present)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(left)            # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise modulation
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Triton kernel for the reduction over k
    # --------------------------------------------------------------
    # Choose tile sizes (tuned for Hopper, can be tuned further)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # hidden_dim may not be a multiple of BLOCK_D; we tile D inside the kernel
    BLOCK_D = 32

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    out = torch.empty_like(left)  # shape [B,N,N,hidden_dim], dtype=float32

    # strides (in elements) for the 4‑D tensors
    stride_left = left.stride()   # (b, i, k, d)
    stride_right = right.stride()
    stride_out = out.stride()

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=hidden_dim,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_d=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_j=stride_right[1],   # note: right tensor index order is (b, j, k, d)
        stride_right_k=stride_right[2],
        stride_right_d=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_d=stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6) Final LayerNorm, gating and linear projection back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
