"""
TriMul “outgoing” kernel (forward‑only)

The heavy part of TriMul is the tensor contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication for every channel *d* :

    out[b,:,:,d] = left[b,:,:,d] @ right[b,:,:,d].T

The implementation below

1.   performs LayerNorm, linear projections and sigmoid‑gates in PyTorch
2.   launches a Triton kernel that evaluates the above contraction.
     The kernel works on a 4‑D tile (BLOCK_I × BLOCK_J × BLOCK_D) and
     iterates over the K‑dimension in BLOCK_K‑sized chunks, accumulating
     with `tl.dot`.  All arithmetic is done in FP16 for speed, the final
     result is cast back to FP32.
3.   finishes the forward pass (output‑norm, out‑gate, final linear) in
     PyTorch.

The kernel is written for Hopper (SM‑90) but works on any recent GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batch‑matmul across the channel dimension
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               #  *[B, N, N, H]   (i,k,d)
    right_ptr,              #  *[B, N, N, H]   (j,k,d)
    out_ptr,                #  *[B, N, N, H]   (i,j,d)
    B, N, H,
    stride_lbn, stride_lbk, stride_lbd,   # left strides
    stride_rbn, stride_rbk, stride_rbd,   # right strides
    stride_obn, stride_obi, stride_obj, stride_obd,  # out strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""

    pid = tl.program_id(0)                 # flattened id over (b, i_tile, j_tile, d_tile)
    total_tiles_i = tl.cdiv(N, BLOCK_I)
    total_tiles_j = tl.cdiv(N, BLOCK_J)
    total_tiles_d = tl.cdiv(H, BLOCK_D)

    b      = pid // (total_tiles_i * total_tiles_j * total_tiles_d)
    rem    = pid %  (total_tiles_i * total_tiles_j * total_tiles_d)
    i_tile = rem // (total_tiles_j * total_tiles_d)
    rem    = rem %  (total_tiles_j * total_tiles_d)
    j_tile = rem // total_tiles_d
    d_tile = rem % total_tiles_d

    # offsets for this tile
    i_offset = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)
    d_offset = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)

    # masks to avoid out‑of‑bounds work
    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_d = d_offset < H

    # pointers to the beginning of the tile (batch dimension folded in)
    left_base  = left_ptr  + b * stride_lbn
    right_base = right_ptr + b * stride_rbn
    out_base   = out_ptr   + b * stride_obn

    # accumulators: (BLOCK_I, BLOCK_J, BLOCK_D) -> we keep a separate
    # accumulator per d because tl.dot does not support a third dimension.
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # loop over K dimension
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k_start, BLOCK_K)
        k_offset = k_start + tl.arange(0, BLOCK_K)

        # mask for K
        mask_k = k_offset < N

        # ----- load left tile -----
        # shape: (BLOCK_I, cur_k, BLOCK_D)
        left_ptrs = (
            left_base
            + i_offset[:, None, None] * stride_lbn
            + k_offset[None, :, None] * stride_lbk
            + d_offset[None, None, :] * stride_lbd
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                       other=0.0).to(tl.float16)

        # ----- load right tile -----
        # shape: (BLOCK_J, cur_k, BLOCK_D)
        right_ptrs = (
            right_base
            + j_offset[:, None, None] * stride_rbn
            + k_offset[None, :, None] * stride_rbk
            + d_offset[None, None, :] * stride_rbd
        )
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                        other=0.0).to(tl.float16)

        # ----- compute dot products for each d -----
        # we iterate over d inside the tile
        for d in range(0, BLOCK_D):
            if not mask_d[d]:
                continue
            # (BLOCK_I, cur_k) dot (cur_k, BLOCK_J) -> (BLOCK_I, BLOCK_J)
            a = left[:, :, d]      # (I, K)
            b = right[:, :, d]     # (J, K)
            # transpose b to (K, J) for dot
            b_t = tl.trans(b)      # (K, J)
            acc[:, :, d] += tl.dot(a, b_t).to(tl.float32)

    # ----- store result -----
    out_ptrs = (
        out_base
        + i_offset[:, None, None] * stride_obn
        + j_offset[None, :, None] * stride_obi
        + d_offset[None, None, :] * stride_obd
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])


# ----------------------------------------------------------------------
# Helper to launch the kernel
# ----------------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]  (float32)
    returns out   : [B, N, N, H] (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides (in elements)
    stride_lbn, stride_lbk, stride_lbd = left.stride()
    stride_rbn, stride_rbk, stride_rbd = right.stride()
    stride_obn, stride_obi, stride_obd = out.stride()

    # tile sizes – tuned for Hopper (can be adjusted)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_D = 16

    total_tiles_i = triton.cdiv(N, BLOCK_I)
    total_tiles_j = triton.cdiv(N, BLOCK_J)
    total_tiles_d = triton.cdiv(H, BLOCK_D)
    total_programs = B * total_tiles_i * total_tiles_j * total_tiles_d
    grid = (total_programs,)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_lbn, stride_lbk, stride_lbd,
        stride_rbn, stride_rbk, stride_rbd,
        stride_obn, stride_obi, stride_obj, stride_obd,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module.
    The heavy contraction is evaluated on the GPU with a custom Triton kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    eps = 1e-5
    weight = weights["norm.weight"]
    bias   = weights["norm.bias"]
    # manual layer‑norm to avoid extra memory (float32 -> fp16 inside the kernel)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * weight + bias                     # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – keep FP16 for speed
    # ------------------------------------------------------------------
    def linear_no_bias(t, w):
        # w: [out, in] – we need to perform a matmul on the last dim
        return torch.nn.functional.linear(t, w)

    left  = linear_no_bias(x, weights["left_proj.weight"]).to(torch.float16)
    right = linear_no_bias(x, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Apply mask (if any) – mask is broadcast on the hidden dimension
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – stay in FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(linear_no_bias(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(linear_no_bias(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate   = torch.sigmoid(linear_no_bias(x, weights["out_gate.weight"])).to(torch.float16)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: Σ_k left * right   (Triton)
    # ------------------------------------------------------------------
    # left/right are [B,N,N,hidden] in fp16
    out = tri_mul_triton(left, right)          # still fp16

    # ------------------------------------------------------------------
    # 6. Output norm, out‑gate and final linear (back to FP32)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    # to_out_norm (LayerNorm over hidden_dim)
    out_weight = weights["to_out_norm.weight"]
    out_bias   = weights["to_out_norm.bias"]
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var  = ((out - out_mean) ** 2).mean(dim=-1, keepdim=True)
    out = (out - out_mean) * torch.rsqrt(out_var + eps)
    out = out * out_weight + out_bias

    out = out * out_gate.to(torch.float32)

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
