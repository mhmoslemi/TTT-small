"""
TriMul “outgoing” kernel (AlphaFold‑3 style).

The heavy part of the module is the triple‑wise multiplication

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication
for each hidden dimension `d` :

    out_d[b] = left_d[b] @ right_d[b].T            # (N,N) @ (N,N) → (N,N)

The kernel below computes this product for all `d` in a single launch,
tiling the `i` and `j` axes and streaming over `k`.  The surrounding
LayerNorms, linear projections and gates are performed in PyTorch – they
are cheap compared with the O(N³·H) work of the einsum.

The kernel works with FP16 inputs (the model weights are FP32, we cast to
FP16 for the mat‑mul) and accumulates in FP32 for numerical stability.
The final output is cast back to FP32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched mat‑mul for every hidden dimension `d`
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          #  *[B, N, N, H]   left tensor
    right_ptr,         #  *[B, N, N, H]   right tensor
    out_ptr,           #  *[B, N, N, H]   result
    B, N, H,           # problem sizes
    stride_lb, stride_lk, stride_ld,   # left strides (b, i, k, d)
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides (b, j, k, d)
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,   # how many M blocks per program group
):
    """
    Compute   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    using a tiled GEMM where the inner product is over `k` and the
    outer dimensions `i` and `j` are blocked.
    """
    pid = tl.program_id(0)

    # ----------- 2‑D grid over (i,j)  ---------------------------
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    # group the program IDs to improve L2 reuse of the K‑tiles
    group_id = pid // (GROUP_SIZE_M * num_n)
    first_pid_m = group_id * GROUP_SIZE_M
    group_size = tl.minimum(num_m - first_pid_m, GROUP_SIZE_M)

    pid_m = first_pid_m + (pid % group_size)
    pid_n = (pid // group_size) % num_n

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑range rows / columns
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------- iterate over batches --------------------------------
    for b in range(B):
        # pointers to the beginning of the current batch
        left_batch_ptr  = left_ptr  + b * stride_lb
        right_batch_ptr = right_ptr + b * stride_rb
        out_batch_ptr   = out_ptr   + b * stride_ob

        # ---------- accumulate over k --------------------------------
        acc = tl.zeros((BLOCK_M, BLOCK_N, H), dtype=tl.float32)

        for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
            # ----------- load a tile of left (i,k)  -----------------
            k_start = k_tile * BLOCK_K
            offs_k = k_start + tl.arange(0, BLOCK_K)

            left_ptrs = left_batch_ptr \
                        + (offs_m[:, None, None] * stride_lk) \
                        + (offs_k[None, :, None] * stride_ld) \
                        + (tl.arange(0, H)[None, None, :] * stride_ld)

            left_mask = (offs_m[:, None] < N) & (offs_k[None, :] < N)
            left_tile = tl.load(left_ptrs, mask=left_mask[..., None], other=0.0).to(tl.float16)

            # ----------- load a tile of right (j,k)  ----------------
            right_ptrs = right_batch_ptr \
                         + (offs_n[:, None, None] * stride_rj) \
                         + (offs_k[None, :, None] * stride_rd) \
                         + (tl.arange(0, H)[None, None, :] * stride_rd)

            right_mask = (offs_n[:, None] < N) & (offs_k[None, :] < N)
            right_tile = tl.load(right_ptrs, mask=right_mask[..., None], other=0.0).to(tl.float16)

            # right is (j,k,d); we need right[j,k,d] with k as the inner dim.
            # The GEMM we want is: (M,K,H) ⋅ (N,K,H)ᵀ → (M,N,H)
            # Transpose the K dimension of right by swapping axes 0↔1.
            right_tile = tl.trans(right_tile, (1, 0, 2))

            # ----------- compute dot product over K -------------------
            # tl.dot works on 2‑D matrices, so we loop over the hidden dim.
            for d in range(0, H, 1):
                a = left_tile[:, :, d]   # (M, K)
                b = right_tile[:, :, d]  # (N, K)
                acc[:, :, d] += tl.dot(a, b)   # (M,N)

        # ----------- write back the result ---------------------------
        out_ptrs = out_batch_ptr \
                   + (offs_m[:, None, None] * stride_oi) \
                   + (offs_n[None, :, None] * stride_oj) \
                   + (tl.arange(0, H)[None, None, :] * stride_od)

        out_mask = mask_m[:, None] & mask_n[None, :]  # (M,N)
        tl.store(out_ptrs, acc.to(tl.float32), mask=out_mask[..., None])


# --------------------------------------------------------------
# Host‑side helper
# --------------------------------------------------------------
def launch_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    `left` / `right`:  [B,N,N,H]  (FP16 or FP32)
    Returns tensor of same shape & dtype (float32).
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    device = left.device

    # cast to fp16 for the kernel (the model keeps fp32 weights)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out = torch.empty_like(left_fp16, dtype=torch.float32, device=device)

    # stride extraction (in elements, not bytes)
    stride_lb, stride_lk, stride_ld = left_fp16.stride()
    stride_rb, stride_rj, stride_rd = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # tuning parameters – work well on Hopper H200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    GROUP_SIZE_M = 8

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_programs = B * grid_m * grid_n

    tri_mul_kernel[(total_programs,)](
        left_fp16,
        right_fp16,
        out,
        B, N, H,
        stride_lb, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
    )
    return out


# --------------------------------------------------------------
# Full custom kernel (PyTorch + Triton)
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ---- LayerNorm -------------------------------------------------
    # using functional LayerNorm to avoid extra Parameter objects
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---- Linear projections (FP32 → FP16 for the heavy kernel) ------
    # x: [B,N,N,dim]
    # Linear: y = x @ W.T   (bias=False)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---- Apply mask (if present) ----------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---- Gates ------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---- Heavy O(N³·H) part -----------------------------------------
    # launch the Triton kernel
    out = launch_tri_mul(left, right)          # float32, shape [B,N,N,hidden_dim]

    # ---- Output LayerNorm + gating + final linear -----------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # ensure final dtype is float32
    return out.float()
