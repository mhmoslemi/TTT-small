"""
TriMul “outgoing” implementation.

The expensive part of TriMul is the N‑by‑N pairwise contraction

    out[b, i, j, c] = Σ_k left[b,i,k,c] * right[b,j,k,c]

for every batch `b` and hidden channel `c`.  This is exactly a batched
matrix‑multiply for each hidden channel:

    out_c = left_c @ right_cᵀ          where left_c,right_c ∈ ℝ^{B×N×N}

We therefore fuse the batch×channel dimensions and run a standard
GEMM kernel written in Triton.  All surrounding ops (LayerNorm, linear
projections, gates, final LayerNorm + linear) are carried out in
PyTorch – they are cheap compared to the O(B·C·N³) contraction and
benefit from the highly‑optimized torch.nn implementations.

The kernel `tri_mul_gemm` computes, for a single contiguous batch‑channel
slice, the matrix product `A @ Bᵀ` where

    A : (M, K)  → left  slice  (i‑axis, k‑axis)
    B : (N, K)  → right slice  (j‑axis, k‑axis)  (note the K‑axis is the same)

The result is stored in an output tile (M, N).  The kernel is launched
with a three‑dimensional grid:

    grid = (num_bc, ceil(N/BLOCK_M), ceil(N/BLOCK_N))

where `num_bc = B * hidden_dim // BLOCK_C` and `BLOCK_C` is the number of
channels processed per program instance (typically 1, i.e. we fuse batch
and channel into a single dimension).  The implementation follows the
canonical Triton GEMM pattern and works for the full range of test
sizes (seq_len up to 1024, hidden_dim up to 384, batch ≤ 2).

The final output is cast back to FP32 as required.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for one channel (or a group of channels)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm(
    # pointers
    left_ptr,            # *float16   (B*C, N, K)   -- left  matrix slice
    right_ptr,           # *float16   (B*C, N, K)   -- right matrix slice (will be read transposed)
    out_ptr,             # *float32   (B*C, N, N)   -- output
    # problem size
    N: tl.constexpr,     # seq_len
    K: tl.constexpr,     # seq_len (same as N for TriMul)
    # strides (all tensors are contiguous in the last dimension)
    stride_left_bc,      # stride to next (batch*channel) element in left
    stride_left_n,       # stride to next row i   in left  (i dimension)
    stride_left_k,       # stride to next column k in left  (k dimension)
    stride_right_bc,     # stride to next (batch*channel) element in right
    stride_right_n,      # stride to next row j   in right (j dimension)
    stride_right_k,      # stride to next column k in right (k dimension)
    stride_out_bc,       # stride to next (batch*channel) element in out
    stride_out_m,        # stride to next row i   in out
    stride_out_n,        # stride to next col j   in out
    # tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    # other compile‑time constants
    FP16_ACC: tl.constexpr = True,   # accumulate in fp16 if possible
):
    """Compute out = left @ rightᵀ for a single (batch,channel) slice."""
    pid_bc = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # -------------------------------------------------------------
    # Offsets for the output tile
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds (needed for the last partial tiles)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the start of this block
    out_base = out_ptr + pid_bc * stride_out_bc \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n

    # -------------------------------------------------------------
    # Accumulator
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over K dimension (the summed axis)
    # -------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # ---- load a tile from left (M x K) -----------------------
        k_offs = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offs < K

        left_ptrs = left_ptr + pid_bc * stride_left_bc \
                    + offs_m[:, None] * stride_left_n \
                    + k_offs[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # ---- load a tile from right (N x K) and transpose it -----
        # right is stored as (N, K); we need (K, N) for dot product.
        right_ptrs = right_ptr + pid_bc * stride_right_bc \
                     + offs_n[:, None] * stride_right_n \
                     + k_offs[None, :] * stride_right_k
        right = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # ---- compute partial dot product -------------------------
        # left : (M, K) ; right : (N, K) -> we need left @ rightᵀ
        # so we transpose right here: (K, N)
        acc += tl.dot(left, tl.trans(right))

    # -------------------------------------------------------------
    # Store result (cast to fp32 as required)
    # -------------------------------------------------------------
    tl.store(out_base, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: layer‑norm with weight & bias (FP32)
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    normalized = (x - mean) * torch.rsqrt(var + eps)
    return normalized * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator with the heavy
    contraction performed by a Triton GEMM kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected fp32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – we fuse them into a single matmul
    # ------------------------------------------------------------------
    # shape: (B,N,N,hidden_dim)
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"].to(dtype), bias=None)
    right = torch.nn.functional.linear(x, weights["right_proj.weight"].to(dtype), bias=None)

    # ------------------------------------------------------------------
    # 3. Apply mask (if any) – mask is broadcastable to [...,1]
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)          # (B,N,N,1)
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating logits + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"].to(dtype), bias=None).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(dtype), bias=None).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"].to(dtype), bias=None).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
    #    We cast to fp16 for the GEMM (the kernel works on fp16 inputs)
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # reshape to (B*hidden_dim, N, K) where K == N
    left_reshape  = left_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_reshape = right_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    out_shape = (B, hidden_dim, N, N)
    out_fp32 = torch.empty(out_shape, dtype=torch.float32, device=device)

    # flatten to (B*hidden_dim, N, N) for the kernel output
    out_flat = out_fp32.reshape(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # launch parameters (tuned for Hopper)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(B * hidden_dim, 1),          # we process one (batch,channel) per program
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # strides for the contiguous tensors
    stride_left_bc, stride_left_n, stride_left_k = left_reshape.stride()
    stride_right_bc, stride_right_n, stride_right_k = right_reshape.stride()
    stride_out_bc, stride_out_m, stride_out_n = out_flat.stride()

    tri_mul_gemm[grid](
        left_reshape,
        right_reshape,
        out_flat,
        N,
        N,                          # K == N
        stride_left_bc,
        stride_left_n,
        stride_left_k,
        stride_right_bc,
        stride_right_n,
        stride_right_k,
        stride_out_bc,
        stride_out_m,
        stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_fp32.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Final LayerNorm + gating + linear projection back to `dim`
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(dtype), bias=None)

    return out.to(dtype)
