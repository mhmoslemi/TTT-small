"""
TriMul “outgoing” implementation.

The heavy part of TriMul is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are already masked and gated.
The kernel below tiles the (i,j) plane and accumulates the sum over k
in blocks of size BLOCK_K.  For every (i,j) we keep a small vector of
the hidden dimension (BLOCK_D = 32) in registers and use `tl.dot`
to perform the inner‑product over the k‑block.  The kernel is launched
with a grid of (batch * H_blocks, ceil(N/BLOCK_I), ceil(N/BLOCK_J)).
All other operations (LayerNorm, linear projections, sigmoid gates and the
final LayerNorm+Linear) are carried out with native PyTorch which is
already highly‑optimized on Hopper.
"""

import torch
import torch.nn as nn
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # *[B, N, N, D] left tensor
    right_ptr,         # *[B, N, N, D] right tensor
    out_ptr,           # *[B, N, N, D] output
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    """
    Compute   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    Tiling:
        i → BLOCK_I
        j → BLOCK_J
        d → BLOCK_D   (vector of hidden dim kept in registers)
        k → BLOCK_K   (reduced sequentially)
    """
    pid = tl.program_id(0)               # combined batch * H_block
    pid_i = tl.program_id(1)              # i‑tile
    pid_j = tl.program_id(2)              # j‑tile

    # ----- decode batch and hidden‑dim block ----------
    H_BLOCKS = tl.cdiv(D, BLOCK_D)
    batch = pid // H_BLOCKS
    d_block = pid % H_BLOCKS

    # ----- offsets for the tiled block -----------------
    i_off = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    d_off = d_block * BLOCK_D + tl.arange(0, BLOCK_D)

    # mask out‑of‑bounds
    i_mask = i_off < N
    j_mask = j_off < N
    d_mask = d_off < D

    # ----- pointers to the first elements ------------
    left_base = left_ptr   + batch * stride_left_b \
                         + i_off[:, None, None] * stride_left_i \
                         + tl.arange(0, BLOCK_K)[None, :, None] * stride_left_k \
                         + d_off[None, None, :] * stride_left_d

    right_base = right_ptr + batch * stride_right_b \
                         + j_off[:, None, None] * stride_right_j \
                         + tl.arange(0, BLOCK_K)[None, :, None] * stride_right_k \
                         + d_off[None, None, :] * stride_right_d

    # accumulator: [BLOCK_I, BLOCK_J, BLOCK_D] in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ----- reduction over k ---------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute actual k‑range
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # load left[b,i,k,d]   → shape (BLOCK_I, BLOCK_K, BLOCK_D)
        left_ptrs = left_base \
                    + k_off[None, :, None] * stride_left_k
        left = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
                       other=0.0)

        # load right[b,j,k,d]  → shape (BLOCK_J, BLOCK_K, BLOCK_D)
        right_ptrs = right_base \
                     + k_off[None, :, None] * stride_right_k
        right = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
                        other=0.0)

        # we need Σ_k left[i,k,d] * right[j,k,d]
        # perform a dot over the K dimension for each (i,j,d)
        # tl.dot works on 1‑D vectors, so we reshape manually:
        #   left  -> (BLOCK_I, BLOCK_K)   per d
        #   right -> (BLOCK_J, BLOCK_K)   per d
        for d in range(0, BLOCK_D):
            # vectors of length BLOCK_K
            l_vec = left[:, :, d]   # (BLOCK_I, BLOCK_K)
            r_vec = right[:, :, d]  # (BLOCK_J, BLOCK_K)

            # compute outer product of dot products:
            #   tmp[i,j] = dot(l_vec[i,:], r_vec[j,:])
            # we achieve this with a matmul using tl.dot
            # (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J)ᵀ
            # tl.dot expects (M,K) @ (K,N)
            tmp = tl.dot(l_vec, r_vec, trans_b=True)   # (BLOCK_I, BLOCK_J)
            acc[:, :, d] += tmp
    # ----- write result --------------------------------
    out_ptrs = out_ptr + batch * stride_out_b \
                       + i_off[:, None, None] * stride_out_i \
                       + j_off[None, :, None] * stride_out_j \
                       + d_off[None, None, :] * stride_out_d

    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None, None] &
                                            j_mask[None, :, None] &
                                            d_mask[None, None, :])

# --------------------------------------------------------------
# Host‑side helper
# --------------------------------------------------------------
def launch_trimap_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, D]   (float16)
    returns out   : [B, N, N, D]   (float16, later cast to fp32)
    """
    B, N, _, D = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes)
    s_lb, s_li, s_lk, s_ld = left.stride()
    s_rb, s_rj, s_rk, s_rd = right.stride()
    s_ob, s_oi, s_oj, s_od = out.stride()

    # tile sizes – tuned for Hopper
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_D = 32               # D is a multiple of 32 in all tests

    grid = (
        triton.cdiv(B * triton.cdiv(D, BLOCK_D), 1),      # program_id(0): batch * d‑block
        triton.cdiv(N, BLOCK_I),                         # program_id(1): i‑tile
        triton.cdiv(N, BLOCK_J),                         # program_id(2): j‑tile
    )

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, D=D,
        stride_left_b=s_lb, stride_left_i=s_li, stride_left_k=s_lk, stride_left_d=s_ld,
        stride_right_b=s_rb, stride_right_j=s_rj, stride_right_k=s_rk, stride_right_d=s_rd,
        stride_out_b=s_ob, stride_out_i=s_oi, stride_out_j=s_oj, stride_out_d=s_od,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
        num_warps=8
    )
    return out


# --------------------------------------------------------------
# Full custom kernel (front‑end)
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input, mask, weights, config)
        - input : [B, N, N, dim]  (float32, on CUDA)
        - mask  : [B, N, N]       (float32 or bool)
        - weights: dict of pre‑allocated weight tensors (float32)
        - config : dict with keys
            * dim            – model dimension
            * hidden_dim    – inner hidden dimension

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # Torch's LayerNorm implementation is already very fast on H200
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – keep in float16 for speed
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]          # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"]
    left  = torch.nn.functional.linear(x, left_proj_w).to(torch.float16)
    right = torch.nn.functional.linear(x, right_proj_w).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Apply mask   (mask shape [B,N,N] → broadcast on hidden)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – all in float16 then cast back to fp16
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w)).to(torch.float16)
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w)).to(torch.float16)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction – Triton kernel
    # ------------------------------------------------------------------
    # kernel expects fp16
    out = launch_trimap_outgoing(left, right)          # [B,N,N,hidden_dim] fp16

    # ------------------------------------------------------------------
    # 6. Post‑projection (LayerNorm + Linear) – back to fp32
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    # LayerNorm on hidden dimension
    mean = out.mean(-1, keepdim=True)
    var  = out.var(-1, unbiased=False, keepdim=True)
    out  = (out - mean) / torch.sqrt(var + 1e-5)
    out  = out * to_out_norm_w + to_out_norm_b

    # gate
    out = out * out_gate.to(torch.float32)

    # final linear
    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]

    return out
