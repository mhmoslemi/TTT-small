"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The expensive part of TriMul is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For a fixed hidden dimension *d* this is a matrix multiplication
    (N×N) × (N×N)^T  → (N×N)
so we can fuse the whole reduction into a single batched GEMM.
All cheap operations (layer‑norm, linear, sigmoid, masking) are done in
PyTorch; the GEMM is executed by a custom Triton kernel that works on
float16 inputs and accumulates in float32 for numerical stability.
The final result is converted back to float32 and passed through the
output normalisation / linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel: C = A @ B^T   (A: [B*H, M, K], B: [B*H, N, K])
# --------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_am, stride_ak, stride_ab,    # strides for A (batch*H, M, K)
    stride_bk, stride_bn, stride_bb,    # strides for B (batch*H, K, N) after transpose
    stride_cm, stride_cn,               # strides for C (batch*H, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)                     # index of (batch*H) block
    batch_head = pid * (M // BLOCK_M) * (N // BLOCK_N)   # not used, just to keep pid

    # offsets for the block of C we are computing
    offs_m = tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)

    # pointer offsets for A and B
    a_batch_ptr = a_ptr + pid * stride_ab
    b_batch_ptr = b_ptr + pid * stride_bb

    # accumulate in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load a tile of A (M x K)
        a_offs_m = offs_m[:, None] * stride_am
        a_offs_k = (k * BLOCK_K + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        a_ptrs = a_batch_ptr + a_offs_m + a_offs_k
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (k * BLOCK_K + tl.arange(0, BLOCK_K)[None, :] < K),
                    other=0.0).to(tl.float32)

        # load a tile of B^T (N x K) → we read B as (K x N) and treat it as transposed
        b_offs_k = (k * BLOCK_K + tl.arange(0, BLOCK_K))[:, None] * stride_bk
        b_offs_n = offs_n[None, :] * stride_bn
        b_ptrs = b_batch_ptr + b_offs_k + b_offs_n
        b = tl.load(b_ptrs, mask=(k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0).to(tl.float32)

        # dot‑product over the K dimension
        acc += tl.dot(a, b)

    # write back C
    c_offs_m = offs_m[:, None] * stride_cm
    c_offs_n = offs_n[None, :] * stride_cn
    c_ptrs = a_batch_ptr + c_offs_m + c_offs_n   # same base pointer as A (same batch*H stride)
    tl.store(c_ptrs, acc, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# --------------------------------------------------------------
# Host‑side helper to launch the kernel
# --------------------------------------------------------------
def batched_gemm(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a : [B*H, M, K]   (float16)
    b : [B*H, N, K]   (float16)   – note the K dimension is the last one,
                                   we will treat it as transposed inside the kernel.
    returns c : [B*H, M, N] (float16)
    """
    assert a.is_cuda and b.is_cuda
    BHK, M, K = a.shape
    assert b.shape == (BHK, N := a.shape[1], K)  # N == M in TriMul, but keep generic
    N = a.shape[1]

    # output allocation
    c = torch.empty((BHK, M, N), device=a.device, dtype=a.dtype)

    # strides (in elements, not bytes)
    stride_am, stride_ak, stride_ab = a.stride()
    stride_bk, stride_bn, stride_bb = b.stride()
    stride_cm, stride_cn, _ = c.stride()

    # block sizes – tuned for Hopper SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (BHK,)

    batched_gemm_kernel[grid](
        a, b, c,
        M, N, K,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return c


# --------------------------------------------------------------
# Full TriMul forward (only forward pass)
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the “outgoing” TriMul operator.
    Arguments:
        data = (input, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)  →  [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if present) – mask shape [B,N,N] → broadcast
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core TriMul: out = Σ_k left[...,i,k,d] * right[...,j,k,d]
    #    => batched GEMM over the hidden dimension
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*hidden_dim, N, N) and cast to fp16 for speed
    left_fp16 = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N).to(torch.float16)
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N).to(torch.float16)

    # GEMM with transposed right (right^T)
    out_fp16 = batched_gemm(left_fp16, right_fp16)   # shape [B*H, N, N]

    # reshape back to [B, N, N, hidden_dim]
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # -> [B,N,N,dim]

    return out
