"""
TriMul “outgoing” kernel (AlphaFold3)

The forward pass consists of

1. Layer‑norm on the 4‑D input  (B,N,N,D)
2. Two linear projections → (B,N,N,HD)   (left, right)
3. Optional mask multiplication
4. Gated‑multiplication (sigmoid gates)
5. A batched “tri‑mul” :   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is exactly a batch of HD independent matrix‑multiplies
   out_d = left_d @ right_dᵀ   (shape B×N×N)
   This is the only part accelerated with Triton.
6. Layer‑norm, output‑gate and final linear back to D.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel – batched matmul for the “tri‑mul” operation
# -------------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,                      # const float*  [B,N,N,HD]
    right_ptr,                     # const float*  [B,N,N,HD]
    out_ptr,                       #       float*  [B,N,N,HD]
    B, N, HD,
    stride_l_bn, stride_l_bk, stride_l_kd,   # left strides   (bn,bk,kd)
    stride_r_bn, stride_r_bk, stride_r_kd,   # right strides  (bn,bk,kd)
    stride_o_bn, stride_o_bk, stride_o_kd,   # out strides    (bn,bk,kd)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program_id(0) encodes (batch, hidden_dim)
    pid = tl.program_id(0)
    b = pid // HD          # batch index
    d = pid %  HD          # hidden‑dim index

    # program_id(1) – M (i) tiles, program_id(2) – N (j) tiles
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # -------------------- compute tile offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the start of the current (b,d) slice
    left_base  = left_ptr  + b * stride_l_bn + d * stride_l_kd
    right_base = right_ptr + b * stride_r_bn + d * stride_r_kd
    out_base   = out_ptr   + b * stride_o_bn + d * stride_o_kd

    # Accumulator in fp32 (left/right are fp16 for speed, can be fp32 as well)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- loop over K dimension --------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute offsets inside the K‑tile
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left:  (..., i, k, d)
        l_ptrs = left_base + (offs_m[:, None] * stride_l_bk
                              + offs_k[None, :] * stride_l_kd)
        # right: (..., j, k, d)  – we need the transpose over (k,j)
        r_ptrs = right_base + (offs_n[:, None] * stride_r_kd
                               + offs_k[None, :] * stride_r_bk)

        # load tiles (with masks)
        l_tile = tl.load(l_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        r_tile = tl.load(r_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # multiply‑accumulate
        acc += tl.dot(l_tile, r_tile)   # dot does (M×K)·(K×N) → (M×N)

    # -------------------- write result --------------------
    o_ptrs = out_base + (offs_m[:, None] * stride_o_bk
                         + offs_n[None, :] * stride_o_kd)
    tl.store(o_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------------------
# Helper: launch the kernel
# -------------------------------------------------------------------------
def trimul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : torch.Tensor of shape (B,N,N,HD), dtype=torch.float16
    returns      : torch.Tensor of shape (B,N,N,HD), dtype=torch.float16
    """
    assert left.shape == right.shape
    B, N, _, HD = left.shape

    out = torch.empty_like(left)

    # strides in elements
    stride_l_bn, stride_l_bk, stride_l_kd = left.stride()
    stride_r_bn, stride_r_bk, stride_r_kd = right.stride()
    stride_o_bn, stride_o_bk, stride_o_kd = out.stride()

    # block sizes – tuned for Hopper (sm_90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * HD,                     # one program per (batch, hidden_dim)
        triton.cdiv(N, BLOCK_M),   # tiles over i
        triton.cdiv(N, BLOCK_N),   # tiles over j
    )

    trimul_kernel[grid](
        left,
        right,
        out,
        B, N, HD,
        stride_l_bn, stride_l_bk, stride_l_kd,
        stride_r_bn, stride_r_bk, stride_r_kd,
        stride_o_bn, stride_o_bk, stride_o_kd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# -------------------------------------------------------------------------
# Full custom kernel – most ops stay in torch, the heavy N^3 reduction is
# performed by the Triton kernel above.
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul.

    Args:
        data: tuple (input, mask, weights, config)
            input  – torch.Tensor[B, N, N, D]  (float32, CUDA)
            mask   – torch.Tensor[B, N, N]    (float32 or bool, CUDA)
            weights – dict of torch.Tensors with the model parameters
            config – dict with keys: dim, hidden_dim

    Returns:
        torch.Tensor[B, N, N, D] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    HD = config["hidden_dim"]
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1) LayerNorm on the last dim (D)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (D,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    # shape: (B,N,N,D) @ (D,HD) → (B,N,N,HD)
    left_proj_w = weights["left_proj.weight"]          # (HD, D)
    right_proj_w = weights["right_proj.weight"]        # (HD, D)

    # use torch.einsum for the projection – cheap compared to N^3 part
    left = torch.einsum("bnid,hd->bnihd", x, left_proj_w)  # (B,N,N,HD)
    right = torch.einsum("bnid,hd->bnihd", x, right_proj_w)

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast on HD)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)           # (B,N,N,1)
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.einsum("bnid,hd->bnihd", x, left_gate_w).sigmoid()
    right_gate = torch.einsum("bnid,hd->bnihd", x, right_gate_w).sigmoid()
    out_gate = torch.einsum("bnid,hd->bnihd", x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Tri‑mul – heavy N³ reduction, done in fp16 for speed
    # -----------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Triton kernel computes: out_d = left_d @ right_dᵀ   (B,N,N,HD)
    out_fp16 = trimul_triton(left_fp16, right_fp16)

    out = out_fp16.to(torch.float32)   # back to fp32 for the rest

    # -----------------------------------------------------------------
    # 6) Final layer‑norm, output gate and linear projection back to D
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]          # (D, HD)

    out = F.layer_norm(out, (HD,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = torch.einsum("bnidh,dh->bnid", out, to_out_w)   # (B,N,N,D)

    return out
