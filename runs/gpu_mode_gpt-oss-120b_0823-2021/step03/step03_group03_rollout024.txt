"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

We implement the forward pass of the outgoing TriMul operator.
The heavy cubic contraction  

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is performed with a tiny Triton kernel that treats each (batch, head)
pair as an independent GEMM (N×N)·(N×N) → (N×N).  All other
operations (layer‑norm, linear projections, sigmoids, final layer‑norm
and projection) are kept in PyTorch because they are cheap compared
to the O(N³) work and because PyTorch already fuses them efficiently
with Torch‑Compile.

The kernel works in fp16 for the two inputs, accumulates in fp32 and
writes fp16.  After the contraction we reshape back to the original
layout, apply the second LayerNorm, the output‑gate and the final
linear projection – all in fp32.  The implementation supports
optional masking and works for the full range of test configurations
(N ∈ {128,…,1024}, B∈{1,2}, dim∈{128,384}, hidden_dim=128).
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched GEMM (I×K)·(K×J) → (I×J) for each (batch, head)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,            # left tensor  (Z, I, K)   fp16
    b_ptr,            # right tensor (Z, K, J)   fp16
    c_ptr,            # output      (Z, I, J)   fp16
    M, N, K,          # extents
    stride_az, stride_am, stride_ak,   # a strides (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    One program per (z, tile_i, tile_j).
    Accumulates in fp32, writes fp16.
    """
    pid_z = tl.program_id(2)       # combined batch*head index
    pid_i = tl.program_id(0)       # tile along M (i dimension)
    pid_j = tl.program_id(1)       # tile along N (j dimension)

    # offset of the first element of the tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (M x K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_i[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile: (K x N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # fp16 dot -> fp32 accumulation
        acc += tl.dot(a, b)

    # write result (convert to fp16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_i[:, None] * stride_cm
        + offs_j[None, :] * stride_cn,
        c,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous
    Returns (B, H, I, J) fp16.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # Collapse batch*head into a single dimension for the kernel
    a = left.view(B * H, I, K)
    b = right.view(B * H, K, J)
    c = out.view(B * H, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes tuned for Hopper (sm_90) – good occupancy and shared‑mem usage
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),          # tiles along I (i)
        triton.cdiv(J, BLOCK_N),          # tiles along J (j)
        B * H,                            # one program per (batch,head)
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # 4 warps give high throughput for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Entry point – forward of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor of shape [B, N, N, dim]  (float32)
        mask         : torch.Tensor of shape [B, N, N] or None
        weights      : dict of leaf weight tensors
        config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32, manual – avoids extra kernel launch)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    # mean & var over the channel dimension
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gates (cast to FP16 for speed)
    # --------------------------------------------------------------
    def _fp16(t): return t.to(torch.float16)

    # weight matrices have shape (out_features, in_features)
    left_proj_w   = _fp16(weights["left_proj.weight"])
    right_proj_w  = _fp16(weights["right_proj.weight"])
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)            # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional pairwise mask (broadcast on hidden_dim)
    # --------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction with Triton (batched GEMM)
    # --------------------------------------------------------------
    # The contraction is over the *sequence* dimension k.
    # left  shape: (B, N, N, H)  → (B, H, N, N)  where H = hidden_dim
    # right shape: (B, N, N, H)  → (B, H, N, N) and we need (K,J) = (N,N)
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    # right must be indexed as (..., j, k, d) → (B, H, K, J)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # batched GEMM in fp16 with our Triton kernel
    out_fp16 = _triton_batched_gemm(left_t, right_t)  # (B, H, I, J)

    # reshape back to (B, N, N, hidden_dim) and cast to fp32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection (still FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)         # broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])  # (B,N,N,dim)

    return out
