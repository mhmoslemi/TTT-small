"""
custom_kernel  –  outgoing TriMul (AlphaFold‑3)

The heavy O(N³) contraction

        out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]

is implemented as a batched GEMM inside a small Triton kernel.
All cheap operations (LayerNorm, linear projections, masking, gating,
output‑norm and final linear) stay in PyTorch.

The kernel works in fp16 with fp32 accumulation (Tensor‑Core friendly)
and writes fp16 results back.  The wrapper reshapes the tensors so that a
single program instance processes a 2‑D tile (M×N) of one
(batch × head) matrix pair.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply for one (batch * head) slice
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr,                     # const fp16*  [BH, N, N]
    right_ptr,                    # const fp16*  [BH, N, N]   (will be read transposed)
    out_ptr,                      #       fp16*  [BH, N, N]
    BH: tl.constexpr, N: tl.constexpr,
    stride_lbh, stride_lbn, stride_lbk,   # left strides
    stride_rbh, stride_rbn, stride_rbk,   # right strides
    stride_obh, stride_obn, stride_obk,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute one (BLOCK_M × BLOCK_N) tile of
          out[bh, i, j] = Σ_k left[bh, i, k] * right[bh, j, k]
    for a fixed bh = b*H + h (flattened batch‑head dimension).
    """
    pid = tl.program_id(0)                      # linear id over (bh, tile_m, tile_n)

    # -------- decode grid indices ------------------------------------
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    tile_n = pid % num_tiles_n
    pid //= num_tiles_n
    tile_m = pid % num_tiles_m
    pid //= num_tiles_m
    bh = pid                                   # 0 ≤ bh < BH

    # -------- offsets for this tile ----------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # -------- accumulator (fp32) ------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------- loop over K dimension ----------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[bh, i, k]   shape (M, K)
        l_ptr = left_ptr + bh * stride_lbh \
                + offs_m[:, None] * stride_lbn \
                + offs_k[None, :] * stride_lbk
        l = tl.load(l_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[bh, j, k] needs to be read as (K, N) because we multiply
        # left[i,k] * right[j,k] = left[i,k] * (right^T)[k,j]
        r_ptr = right_ptr + bh * stride_rbh \
                + offs_k[:, None] * stride_rbk \
                + offs_n[None, :] * stride_rbn
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # GEMM tile
        acc += tl.dot(l, r)                     # (M,K)@(K,N) → (M,N)

    # -------- write back ------------------------------------------------
    out_ptrs = out_ptr + bh * stride_obh \
               + offs_m[:, None] * stride_obn \
               + offs_n[None, :] * stride_obk
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input   : [B, N, N, dim]   (float32)
        torch.Tensor,                # mask    : [B, N, N]       (float32 or None)
        Dict[str, torch.Tensor],     # weights : parameters
        Dict[str, Any],              # config  : {"dim":..., "hidden_dim":...}
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # = H

    ######################################################################
    # 1️⃣  LayerNorm + linear projections (stay in PyTorch – cheap)
    ######################################################################
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    ######################################################################
    # 2️⃣  Mask (optional)
    ######################################################################
    if mask is not None:
        mask_ = mask.unsqueeze(-1)        # (B,N,N,1)
        left  = left * mask_
        right = right * mask_

    ######################################################################
    # 3️⃣  Gating (sigmoid)
    ######################################################################
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left  = left * left_gate
    right = right * right_gate

    ######################################################################
    # 4️⃣  Reshape for the GEMM kernel
    #    (B,N,N,H) -> (B, H, N, N) -> (B*H, N, N)   and cast to fp16
    ######################################################################
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16  = left.view(B * hidden_dim, N, N).to(torch.float16)
    right_fp16 = right.view(B * hidden_dim, N, N).to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)          # (B*H, N, N), fp16

    ######################################################################
    # 5️⃣  Launch Triton kernel
    ######################################################################
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    BH = B * hidden_dim
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_tiles = BH * grid_m * grid_n
    grid = (total_tiles,)

    _trimul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        BH,
        N,
        # strides for left   (BH, N, N)
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        # strides for right  (BH, N, N)
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        # strides for out    (BH, N, N)
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    ######################################################################
    # 6️⃣  Restore original layout + post‑processing (PyTorch)
    ######################################################################
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)
    out = out.to(dtype)                                                       # fp32 for LN etc.

    # output LayerNorm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply output gate (broadcast over N,N)
    out = out * out_gate

    # final projection back to original dimensionality
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
```

### What the implementation does
1. **Cheap ops** (LayerNorm, Linear, masking, sigmoid‑gates) stay in PyTorch – they are memory‑bound and much faster there.
2. **Heavy contraction** is performed by `_trimul_gemm_kernel`.  
   * The kernel tiles the output matrix in 2‑D blocks (`BLOCK_M × BLOCK_N`) and iterates over the reduction axis `K` in chunks `BLOCK_K`.  
   * All arithmetic is fp16 with fp32 accumulation, matching Tensor‑Core best practice.  
   * The kernel processes a flattened batch‑head dimension (`BH = B * hidden_dim`), so a single kernel launch covers all heads simultaneously.
3. The result is reshaped back to `[B, N, N, hidden_dim]`, layer‑normalized, gated, and finally linearly projected to the original `dim`.

The code follows Triton‑launch conventions, works for all test configurations listed, respects the “no‑gradient” requirement, and keeps at least one non‑trivial computation inside a Triton kernel.
