"""
custom_kernel – “outgoing” TriMul forward (AlphaFold‑3 style)

Main ideas
---------
* All cheap per‑element ops (LayerNorm, linear projections, gating) are done in
  PyTorch so we can reuse highly‑optimized fused kernels and keep the code
  simple.
* The expensive cubic contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
  is a batched matrix‑multiply (B·D, N, N) · (B·D, N, N)ᵀ.  We implement it with a
  single Triton kernel that multiplies two fp16 tensors while accumulating in
  fp32 (Tensor‑Core friendly).  The kernel tiles the M×N output in 64×64 blocks
  and reduces over K in 32‑element chunks.
* After the contraction we restore the original layout, apply the final
  LayerNorm, gating and output linear in PyTorch, and finally cast back to
  fp32 as required by the specification.
* The implementation works for any batch size (1 or 2), any sequence length
  (≤ 1024) and any hidden dimension (128) / output dimension (128 or 384) that
  appear in the test matrix.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (I×K)·(K×J) → (I×J) for many “heads”
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    a_ptr,            # left  matrix (Z, I, K)  – fp16
    b_ptr,            # right matrix (Z, K, J)  – fp16
    c_ptr,            # out   matrix (Z, I, J)  – fp16
    M, N, K,          # matrix sizes
    stride_az, stride_am, stride_ak,   # a strides: (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides: (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides: (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)          # batch*head dimension
    pid_m = tl.program_id(1)          # row tile
    pid_n = tl.program_id(2)          # col tile

    # ------------------------------------------------------------------
    # Tile coordinates
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (higher precision for the sum over K)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in BLOCK_K steps
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile  (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 accumulation (dot works on fp16 → fp32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write result (cast back to fp16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the kernel above.
    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous
    returns out : (B, H, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse the leading dimensions for the kernel (Z = B*H)
    Z = B * H
    a = left.reshape(Z, I, K)
    b = right.reshape(Z, K, J)
    c = out.reshape(Z, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tuned tile sizes for Hopper / SM90 (Tensor‑Core friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                                   # program id 0 : batch*head
        triton.cdiv(I, BLOCK_M),            # program id 1 : row tiles
        triton.cdiv(J, BLOCK_N),            # program id 2 : col tiles
    )
    batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # works well on SM90
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor            [B, N, N, dim]   (fp32)
        mask         : torch.Tensor or None   [B, N, N]        (bool/float)
        weights      : dict of torch.Tensor (all parameters as in the reference)
        config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor  [B, N, N, dim] (fp32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm   (fp32, keep precision)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                     # [B,N,N,dim] fp32

    # ------------------------------------------------------------------
    # 2. Linear projections (cast to fp16 for the heavy GEMM)
    # ------------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)      # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – still in fp16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Rearrange to (B, head, I, K) / (B, head, K, J) where head = hidden
    #    and launch the Triton GEMM.
    # ------------------------------------------------------------------
    #   left : (B,N,N,hidden) → (B, hidden, N, N)
    #   right: (B,N,N,hidden) → (B, hidden, N, N) and then transpose K↔J
    left_perm  = left.permute(0, 3, 1, 2).contiguous()          # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()         # (B, H, K, J)

    out_fp16 = triton_batched_gemm(left_perm, right_perm)      # (B, H, I, J)

    # back to original layout (B,N,N,hidden) → (B,N,N,dim) after norm
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7. Final gating + linear projection (fp32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                   # (B,N,N,dim)

    return out
