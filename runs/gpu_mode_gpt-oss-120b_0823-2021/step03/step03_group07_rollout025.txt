"""
TriMul “outgoing” forward pass (AlphaFold‑3).

The dominant O(N³) operation is the contracted sum

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

where `left`/`right` are the linearly‑projected, gated tensors of shape
[B, H, N, N] (H = hidden_dim).  The implementation below

* performs cheap work (LayerNorm, linear projections, gating, final
  LayerNorm/linear) with regular PyTorch – these are memory‑bound and
  far cheaper than the cubic contraction.
* launches a single Triton kernel that evaluates the contraction as a
  batched GEMM using a classic three‑dimensional tiling (BLOCK_M ×
  BLOCK_N × BLOCK_K).  The kernel works on fp16 data with fp32
  accumulation and stores the result back as fp16.
* finally reshapes the result back to the original layout and finishes the
  forward pass in PyTorch.

The kernel is deliberately simple and runs efficiently on Hopper (SM90)
by keeping shared‑memory usage modest (BLOCK_M/N/K ≤ 64) and letting the
hardware‑accelerated `tl.dot` exploit Tensor‑cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # const fp16*  [B, H, N, N]
    right_ptr,         # const fp16*  [B, H, N, N] (will be read transposed)
    out_ptr,           #       fp16*  [B, H, N, N]
    B, N, H,           # int metadata
    # strides of the three tensors (row‑major)
    stride_lb, stride_lh, stride_ln, stride_lk,   # left  (B,H,N,N)
    stride_rb, stride_rh, stride_rn, stride_rk,   # right (B,H,N,N)
    stride_ob, stride_oh, stride_on, stride_ok,   # out   (B,H,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a (BLOCK_M × BLOCK_N) tile of the 4‑D result

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    One program instance is responsible for a single (b, h, tile_i, tile_j)
    tile.
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # decode pid  -> (b, h, tile_i, tile_j)
    # ------------------------------------------------------------------
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)

    tile_j = pid % num_tiles_j
    pid //= num_tiles_j
    tile_i = pid % num_tiles_i
    pid //= num_tiles_i
    h = pid % H
    b = pid // H

    # ------------------------------------------------------------------
    # tile coordinates
    # ------------------------------------------------------------------
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (to keep precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K‑dimension in blocks
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, h, i, k]   -> shape (M, K)
        l_ptr = (left_ptr +
                 b * stride_lb +
                 h * stride_lh +
                 offs_i[:, None] * stride_ln +
                 offs_k[None, :] * stride_lk)
        l = tl.load(l_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[b, h, j, k] must be read transposed so that its shape is (K, N)
        #   right[b, h, j, k]  (b, h, row=j, col=k)
        #   we need (k, j) => swap last two dims while loading
        r_ptr = (right_ptr +
                 b * stride_rb +
                 h * stride_rh +
                 offs_k[:, None] * stride_rk +   # k dimension becomes row
                 offs_j[None, :] * stride_rn)    # j dimension becomes column
        r_t = tl.load(r_ptr,
                      mask=mask_k[:, None] & mask_j[None, :],
                      other=0.0).to(tl.float32)

        # compute dot product (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(l, r_t)

    # ------------------------------------------------------------------
    # write back the tile
    # ------------------------------------------------------------------
    out_ptrs = (out_ptr +
                b * stride_ob +
                h * stride_oh +
                offs_i[:, None] * stride_on +
                offs_j[None, :] * stride_ok)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host‑side wrapper implementing the full forward pass
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input   [B, N, N, dim]
        torch.Tensor,                     # mask    [B, N, N]  (or None)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any]                    # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device
    # Use float32 for the cheap ops – enough precision and matches the
    # reference implementation.
    f32 = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]          # H

    # ------------------------------------------------------------------
    # 1. Cheap element‑wise work with PyTorch
    # ------------------------------------------------------------------
    # LayerNorm over the feature dimension
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(f32),
        bias=weights["norm.bias"].to(f32),
    )

    # Linear projections (bias=False)
    left = F.linear(x, weights["left_proj.weight"].to(f32))    # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(f32))

    # optional masking
    if mask is not None:
        mask_view = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_view
        right = right * mask_view

    # gating (sigmoid after linear)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(f32)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(f32)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(f32)))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 2. Rearrange to [B, H, N, N] and cast to fp16 for the kernel
    # ------------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output buffer in fp16
    out_fp16 = torch.empty_like(left_fp16)   # (B,H,N,N), fp16

    # ------------------------------------------------------------------
    # 3. Launch the Triton kernel (blocked GEMM)
    # ------------------------------------------------------------------
    # Tile sizes – chosen to keep shared memory < 64KB on Hopper and to
    # allow Tensor‑core usage (multiple of 8).
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Number of tiles in each dimension
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)

    # total number of program instances = B * H * grid_m * grid_n
    grid = (B * hidden_dim * grid_m * grid_n,)

    _trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        # strides for left
        left_fp16.stride(0),   # B
        left_fp16.stride(1),   # H
        left_fp16.stride(2),   # N (i)
        left_fp16.stride(3),   # N (k)
        # strides for right
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        # strides for out
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 4. Post‑process back to the original layout
    # ------------------------------------------------------------------
    # (B,H,N,N) -> (B,N,N,H)
    out = out_fp16.to(f32).permute(0, 2, 3, 1).contiguous()

    # final LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(f32),
        bias=weights["to_out_norm.bias"].to(f32),
    )

    # apply output gate (broadcast over pairwise axes)
    out = out * out_gate

    # linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"].to(f32))

    return out
```

**Explanation of design choices**

* **Kernel vs. PyTorch split** – The O(N³) contraction dominates runtime, so it is the only part moved to Triton. All LayerNorms, linear projections and sigmoids stay in PyTorch because they are memory‑bound and cheap relative to the cubic work.
* **Tiling** – BLOCK_M/N/K = 64/64/32 are multiples of 8, enabling Tensor‑core `tl.dot` while keeping shared memory per thread‑block ≈ (64·64 + 64·32 + 64·32)·2 bytes ≈ 28 KB, comfortably within the Hopper limit.
* **Transposed read of `right`** – By loading `right` with the last two dimensions swapped we obtain a (K, N) matrix, allowing a single `tl.dot` to compute the inner product for each tile.
* **FP16 compute / FP32 accumulation** – Matches best‑practice for speed on Hopper (Tensor‑cores) while preserving numerical quality for the final float32 output.
* **Stride handling** – Strides are passed explicitly from the host so the kernel works for any contiguous layout (the tensors are made contiguous before launch).

The function conforms to the required signature, runs entirely on the device, and produces the same result as the reference PyTorch implementation (up to floating‑point rounding).
