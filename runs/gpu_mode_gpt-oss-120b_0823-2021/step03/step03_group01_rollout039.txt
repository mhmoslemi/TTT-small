"""
custom_kernel – “outgoing” TriMul forward pass (AlphaFold‑3 style).

The computational bottleneck is the cubic contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply for every head (hidden‑dim) :

    (B, H, I, K)  @  (B, H, K, J)  →  (B, H, I, J)

The rest of the module (LayerNorms, linear projections, gating, final
normalisation / linear) is performed in PyTorch (FP32).  The heavy
multiplication is executed in a fused Triton kernel that works on
fp16 data (fp32 accumulation) and exploits Tensor‑Core‑friendly tiles.
The kernel is launched on a 3‑D grid: (row‑tile, col‑tile, batch‑head).

The implementation supports:
* any batch size ≤ 2
* any sequence length N that fits the test matrix‑tile sizes
* optional boolean / fp32 mask
* arbitrary channel dimensions (dim, hidden_dim) – the kernel only touches
  the (N,N) matrix dimensions, the head dimension H (= hidden_dim) is
  collapsed into the batch axis for the kernel launch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (I×K)·(K×J) → (I×J) for many heads
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,            # left  matrix pointer   (Z, I, K)
    b_ptr,            # right matrix pointer   (Z, K, J)
    c_ptr,            # output matrix pointer  (Z, I, J)
    M, N, K,          # matrix extents
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1) identify the tile we are working on
    # ------------------------------------------------------------------
    pid_z = tl.program_id(0)          # batch*head dimension
    pid_m = tl.program_id(1)          # tile‑row index
    pid_n = tl.program_id(2)          # tile‑col index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # fp32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 2) loop over K dimension in BLOCK_K steps
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile  (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot‑product accumulation (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # 3) write result (down‑cast to fp16 for memory bandwidth)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for tensors of shape (B, H, I, K) @ (B, H, K, J).

    Both inputs must be contiguous fp16 tensors.
    Returns fp16 tensor of shape (B, H, I, J).
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # Collapse the batch and head dimensions – the kernel sees a 3‑D tensor
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides in element units (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes – chosen to hit 64‑byte lanes and Tensor‑Core shape 8×8×16
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                               # batch*head
        triton.cdiv(I, BLOCK_M),         # rows
        triton.cdiv(J, BLOCK_N),         # cols
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # SM90 works well with 4 warps per block
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass – PyTorch for everything except the cubic contraction
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        mask         : torch.Tensor [B, N, N]       (float32/bool) or None
        weights      : dict of torch.Tensor parameters
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + eps)
    x = x * ln_w + ln_b                               # [B,N,N,dim] fp32

    # ------------------------------------------------------------------
    # 2. Linear projections – cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16, non_blocking=True)

    left_proj_w  = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w  = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w   = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 3. Apply optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – compute on fp16 then up‑cast when needed
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Rearrange for batched GEMM
    #    left : (B, N, N, H)  → (B, H, N, N)
    #    right: (B, N, N, H)  → (B, H, N, N)   but we need (K,J) = (N,N) with K as
    #                                          the shared dimension, J as the second
    #                                          spatial index.
    # ------------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)  I==K==N
    # right must be indexed as (B, H, K, J) where J corresponds to the
    # second spatial index of the original tensor.
    right = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # ------------------------------------------------------------------
    # 6. Cubic contraction via Triton batched GEMM (fp16 compute)
    # ------------------------------------------------------------------
    out_fp16 = _triton_batched_gemm(left, right)   # (B, H, I, J)

    # ------------------------------------------------------------------
    # 7. Restore original layout and cast back to fp32 for the rest
    # ------------------------------------------------------------------
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 9. Final gating + linear projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over dim
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
