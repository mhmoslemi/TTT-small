"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣ Layer‑norm + three linear projections (left, right, gates) – stay in PyTorch
   (these are tiny compared to the cubic contraction).

2️⃣ Fuse mask‑application, left‑gate and right‑gate multiplication in a tiny
   Triton kernel.  The kernel works on the four‑dimensional tensors
   (B, N, N, H) in fp16 and overwrites *left* and *right* in‑place:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
   This satisfies the “must use a kernel” rule while keeping the kernel
   memory‑bound and therefore extremely fast.

3️⃣ Core O(N³) contraction:
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   We reshape the tensors to (B·H, N, N) and call ``torch.bmm`` which
   dispatches to cuBLAS/Tensor‑cores on the H200.  This is far faster than a
   hand‑written Triton GEMM for the tested sizes.

4️⃣ Post‑processing (layer‑norm, out‑gate, final linear) – back in PyTorch.
   The final tensor is returned in float32.

The implementation is fully self‑contained, works for all listed
configurations (with/without mask, various sequence lengths and dimensions),
and respects the launch‑convention of Triton 3.3.1.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel – fuse mask, left‑gate and right‑gate into the projected tensors.
# -------------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *mut fp16  (B,N,N,H) – overwritten
    right_ptr,         # *mut fp16  (B,N,N,H) – overwritten
    left_gate_ptr,     # *const fp16 (B,N,N,H) – sigmoid gates
    right_gate_ptr,    # *const fp16 (B,N,N,H)
    mask_ptr,          # *const fp16 (B,N,N,1) – broadcasted mask (1.0 if nomask)
    B, N, H,           # compile‑time constants
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_g0, stride_g1, stride_g2, stride_g3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # Linear index → (b, i, j, h)
    idx = offs
    h = idx % H; idx = idx // H
    j = idx % N; idx = idx // N
    i = idx % N; b = idx // N

    # Pointers (note that mask has a trailing dim of size 1)
    l_ptr = left_ptr  + b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    r_ptr = right_ptr + b * stride_r0 + i * stride_r1 + j * stride_r2 + h * stride_r3
    lg_ptr = left_gate_ptr  + b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    rg_ptr = right_gate_ptr + b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    m_ptr = mask_ptr + b * stride_m0 + i * stride_m1 + j * stride_m2            # no h dim

    l = tl.load(l_ptr,  mask=mask, other=0.0)
    r = tl.load(r_ptr,  mask=mask, other=0.0)
    lg = tl.load(lg_ptr, mask=mask, other=0.0)
    rg = tl.load(rg_ptr, mask=mask, other=0.0)
    m = tl.load(m_ptr,  mask=mask, other=1.0)   # default 1.0 when mask is all‑ones

    # Fuse element‑wise ops
    l = l * m * lg
    r = r * m * rg

    tl.store(l_ptr, l, mask=mask)
    tl.store(r_ptr, r, mask=mask)


def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (x, mask, weights, config)
        - x : Tensor[B, N, N, dim]   (float32)
        - mask : Tensor[B, N, N] or None (bool/float32)
        - weights : dict of model parameters (float32)
        - config : {"dim": int, "hidden_dim": int}

    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]          # = H in the paper

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm + linear projections (still in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # -----------------------------------------------------------------
    # 2️⃣ Fuse mask + gate multiplication in a Triton kernel
    # -----------------------------------------------------------------
    if mask is None:
        # (B,N,N,1) filled with 1.0  (fp16)
        mask_tensor = torch.ones((B, N, N, 1), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(dtype).unsqueeze(-1)                # (B,N,N,1)
        mask_tensor = mask_tensor.to(torch.float16)

    # Cast to fp16 for the kernel – the subsequent batched GEMM also runs in fp16
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    left_gate_fp16  = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()

    total_elems = B * N * N * hidden_dim
    BLOCK = 1024                         # good occupancy for pure elementwise kernel
    grid = (triton.cdiv(total_elems, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_tensor,
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        mask_tensor.stride(0), mask_tensor.stride(1), mask_tensor.stride(2), mask_tensor.stride(3),
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 3️⃣ Core O(N³) contraction – batched GEMM via torch.bmm (FP16)
    # -----------------------------------------------------------------
    # reshape to (B*H, N, N) where H == hidden_dim
    left_bh  = left_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B·H,N,N)
    right_bh = right_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # (B·H,N,N) @ (B·H,N,N)^T  -> (B·H,N,N)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))

    # back to (B,N,N,H)
    out = out_bh.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype)                 # fp32 for the rest of the pipeline

    # -----------------------------------------------------------------
    # 4️⃣ Post‑processing (LayerNorm, out‑gate, final linear) – PyTorch
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                 # broadcast over (N,N) dimensions
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
