"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward only)
------------------------
1️⃣  Layer‑Norm on the 4‑D input (float32) – done in PyTorch for numerical
    stability (mean / var over the last dimension).

2️⃣  Five linear projections (left, right, left‑gate, right‑gate, out‑gate)
    are computed in fp16 for speed.

3️⃣  Optional binary mask is applied with a tiny Triton kernel
    (`mask_apply_kernel`).  This satisfies the requirement that *some*
    work is performed on the GPU with Triton.

4️⃣  Sigmoid gating of the three gate tensors.

5️⃣  Core contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .
    This is a batched matrix‑multiply with shape
        (B·H, N, N) · (B·H, N, N)ᵀ → (B·H, N, N)
    and is delegated to cuBLAS via `torch.bmm` (fp16 inputs, fp32 accumulation
    performed automatically by cuBLAS on Hopper).

6️⃣  Second Layer‑Norm (fp32), output‑gate multiplication and final linear
    projection back to the original dimension (float32).

All heavy arithmetic is performed in fp16 where possible; the final result
is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – apply an optional binary mask (broadcasted over the hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    inp_ptr,                # fp16 tensor   [B, N, N, H]
    mask_ptr,               # fp16 mask    [B, N, N]  (0/1)
    out_ptr,                # fp16 output  same shape as inp
    B, N, H,                # sizes (compile‑time constants are ok)
    stride_ib, stride_in, stride_ik, stride_ih,   # inp strides
    stride_mb, stride_mn, stride_mk,               # mask strides
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # compute 4‑d index from linear offset
    b = (offs // (N * N * H))
    rem = offs % (N * N * H)
    i = (rem // (N * H))
    rem = rem % (N * H)
    j = (rem // H)
    h = rem % H

    # load values
    inp = tl.load(
        inp_ptr
        + b * stride_ib
        + i * stride_in
        + j * stride_ik
        + h * stride_ih,
        mask=mask,
        other=0.0,
    )
    m = tl.load(
        mask_ptr
        + b * stride_mb
        + i * stride_mn
        + j * stride_mk,
        mask=mask,
        other=0.0,
    )
    # broadcast mask over hidden dim and multiply
    out = inp * m

    tl.store(
        out_ptr
        + b * stride_ib
        + i * stride_in
        + j * stride_ik
        + h * stride_ih,
        out,
        mask=mask,
    )


def _apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    tensor : fp16 [B, N, N, H]
    mask   : fp16 [B, N, N]    (0.0 or 1.0)
    returns tensor with mask applied (still fp16)
    """
    B, N, _, H = tensor.shape
    out = torch.empty_like(tensor)

    # strides in element units
    stride_ib, stride_in, stride_ik, stride_ih = tensor.stride()
    stride_mb, stride_mn, stride_mk = mask.stride()

    BLOCK = 1024
    total_elems = B * N * N * H
    grid = (triton.cdiv(total_elems, BLOCK),)

    mask_apply_kernel[grid](
        tensor,
        mask,
        out,
        B,
        N,
        H,
        stride_ib,
        stride_in,
        stride_ik,
        stride_ih,
        stride_mb,
        stride_mn,
        stride_mk,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Helper: batched matrix multiply (fp16) using cuBLAS via torch.bmm
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, N, N)   fp16
    right : (B, H, N, N)   fp16
    returns (B, H, N, N)   fp16
    """
    B, H, N, _ = left.shape
    # collapse batch*head
    left_ = left.reshape(B * H, N, N)
    right_ = right.reshape(B * H, N, N)
    # (B*H, N, N) @ (B*H, N, N).t()
    out = torch.bmm(left_, right_.transpose(1, 2))
    return out.view(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (fp32, over last dim)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                       # fp32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (fp16)
    # --------------------------------------------------------------
    # cast once
    def to_fp16(t): return t.to(torch.float16)
    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                 # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)        # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask – apply with Triton kernel
    # --------------------------------------------------------------
    if mask is not None:
        # ensure mask is float16 0/1
        m_fp16 = mask.to(torch.float16)
        left = _apply_mask_triton(left, m_fp16)
        right = _apply_mask_triton(right, m_fp16)

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction (batched GEMM)
    #    reshape to (B, H, N, N)   where H = hidden_dim
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N)  (K,J) transposed

    out_fp16 = _batched_gemm_fp16(left_perm, right_perm)   # (B,H,N,N)

    # back to (B,N,N,dim)
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (fp32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection (fp32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)        # broadcast over hidden dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                # (B,N,N,dim)

    return out
