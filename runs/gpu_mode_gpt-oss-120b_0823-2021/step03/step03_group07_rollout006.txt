"""
TriMul “outgoing” forward pass (AlphaFold‑3) – fast implementation.

* All cheap ops (LayerNorm, Linear, final LayerNorm/Linear) are done in PyTorch.
* The only GPU‑side work that really benefits from a custom kernel is the
  element‑wise masking & gating of the projected tensors.  This is written as a
  tiny Triton kernel that works on a flattened view and therefore has virtually
  no launch overhead.
* The dominant O(N³) contraction
        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
  is performed with a single batched GEMM (`torch.bmm`) after reshaping the
  tensors to (B*H, N, N).  cuBLAS/Tensor‑cores on the Hopper H200 handle this
  extremely fast, so we do not need a full‑blown Triton GEMM.
* The final output is returned in float32 as required by the reference module.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – element‑wise mask & gate
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    src_ptr,           # fp16* input tensor (flattened)
    mask_ptr,          # fp16* mask  (flattened)   – can be nullptr
    gate_ptr,          # fp16* gate  (flattened)
    dst_ptr,           # fp16* output (flattened)
    N_ELEM: tl.constexpr,   # total number of elements
    BLOCK: tl.constexpr,
):
    """out = src * mask * gate  (mask omitted => 1)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM

    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # mask may be None → we treat it as all‑ones
    if tl.constexpr(mask_ptr is not None):
        m = tl.load(mask_ptr + offs, mask=mask, other=1.0)
    else:
        m = tl.full((BLOCK,), 1.0, dtype=tl.float16)

    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    out = src * m * g
    tl.store(dst_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input   [B, N, N, dim]
        torch.Tensor,                     # mask    [B, N, N]   (or None)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any]                    # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32            # final precision

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]    # = H

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm + linear projections (PyTorch – cheap, memory bound)
    # ------------------------------------------------------------------
    #   norm → left_proj / right_proj → left_gate / right_gate / out_gate
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # Linear projections (bias=False)
    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # Gates – sigmoid after linear
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 2️⃣ Apply mask and gates with a tiny Triton kernel (fp16)
    # ------------------------------------------------------------------
    #   we keep the tensors in fp16 for the subsequent GEMM
    left  = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    left_gate  = left_gate.to(torch.float16).contiguous()
    right_gate = right_gate.to(torch.float16).contiguous()
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).contiguous()
    else:
        mask_fp16 = None

    # helper to launch the kernel on a flattened view
    def _apply(src, gate, mask_tensor):
        out = torch.empty_like(src)
        N_ELEM = src.numel()
        BLOCK = 128
        grid = triton.cdiv(N_ELEM, BLOCK)
        # Pass nullptr when mask is None – Triton sees a Python `None`
        _mask_gate_kernel[grid](
            src_ptr=src,
            mask_ptr=mask_tensor,
            gate_ptr=gate,
            dst_ptr=out,
            N_ELEM=N_ELEM,
            BLOCK=BLOCK,
        )
        return out

    left  = _apply(left,  left_gate,  mask_fp16)
    right = _apply(right, right_gate, mask_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Batched GEMM (cuBLAS) over the hidden dimension
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) and use bmm; right must be transposed on the last two axes
    left  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B*H, N, N)
    right = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)  # (B*H, N, N)

    # right is used as right^[T] in the einsum → we need (N, N) where the
    # second axis corresponds to the summed‑over `k`.  Transposing last two dims
    # of each batch achieves the same effect as the original einsum.
    out = torch.bmm(left, right.transpose(1, 2))                     # (B*H, N, N)

    # reshape back to (B, H, N, N) then to (B, N, N, H)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Final LayerNorm, output gate and linear back to `dim`
    # ------------------------------------------------------------------
    out = out.to(dtype)   # to float32 for the remaining ops

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # out_gate is (B,N,N,H) – broadcast over the last dim after the norm
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,dim)

    return out
