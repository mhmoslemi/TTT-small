"""
custom_kernel – outgoing TriMul (AlphaFold‑3 style)

Algorithm
---------
1️⃣  Layer‑norm + four linear projections (left, right, left‑gate,
    right‑gate, out‑gate) – stay in PyTorch (these are cheap).

2️⃣  Fuse the *mask* (if any), *left‑gate* and *right‑gate* into a **single
    Triton element‑wise kernel**.  The kernel reads
        left[b,i,k,h], right[b,j,k,h],
        left_gate[b,i,k,h], right_gate[b,j,k,h],
        mask[b,i,k]               (broadcasted over h)
    and writes back in‑place
        left  ← left  * mask * left_gate
        right ← right * mask * right_gate
    This satisfies the “must use a kernel” rule while remaining memory‑bound.

3️⃣  The cubic contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is a batched matrix‑matrix multiplication for every hidden channel *h*.
    By reshaping the tensors to (B·H, N, N) we can call ``torch.bmm`` which
    (on Hopper) uses Tensor‑cores and is far faster than a hand‑written
    Triton GEMM.  We keep the data in **fp16** for the GEMM and convert back
    to fp32 afterwards.

4️⃣  Post‑processing (layer‑norm, out‑gate, final linear) – again in PyTorch.

All operations respect arbitrary batch size (1 or 2), sequence length,
and the possible presence/absence of a mask.  The final result is
float‑32 with shape ``[B, N, N, dim]``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel : mask × gate × (left/right) fusion
# -------------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *mut fp16  (B,N,N,H)
    right_ptr,         # *mut fp16  (B,N,N,H)
    left_gate_ptr,     # *const fp16 (B,N,N,H)
    right_gate_ptr,    # *const fp16 (B,N,N,H)
    mask_ptr,          # *const fp16 (B,N,N,1) – broadcast over H
    B, N, H,           # compile‑time constants
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_g0, stride_g1, stride_g2, stride_g3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    BLOCK: tl.constexpr,
):
    """In‑place: left  ← left  * mask * left_gate
                     right ← right * mask * right_gate
       The mask is broadcast along the hidden dimension."""
    pid = tl.program_id(0)
    total = B * N * N * H
    idx = pid * BLOCK + tl.arange(0, BLOCK)
    valid = idx < total

    # -------------------------------------------------------------
    # Decode linear index → (b,i,j,k,h)
    # -------------------------------------------------------------
    tmp = idx
    h = tmp % H;            tmp = tmp // H
    j = tmp % N;            tmp = tmp // N
    i = tmp % N;            b = tmp // N

    # -------------------------------------------------------------
    # Compute pointers (one element per thread)
    # -------------------------------------------------------------
    l_ptr = left_ptr   + b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    r_ptr = right_ptr  + b * stride_r0 + i * stride_r1 + j * stride_r2 + h * stride_r3
    lg_ptr = left_gate_ptr  + b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    rg_ptr = right_gate_ptr + b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    # mask has no hidden dimension, stride_m3 == 0
    m_ptr = mask_ptr + b * stride_m0 + i * stride_m1 + j * stride_m2

    # -------------------------------------------------------------
    # Load, fuse, store
    # -------------------------------------------------------------
    l = tl.load(l_ptr, mask=valid, other=0.0)
    r = tl.load(r_ptr, mask=valid, other=0.0)
    lg = tl.load(lg_ptr, mask=valid, other=0.0)
    rg = tl.load(rg_ptr, mask=valid, other=0.0)
    m = tl.load(m_ptr, mask=valid, other=1.0)          # default 1.0 → no mask

    l = l * m * lg
    r = r * m * rg

    tl.store(l_ptr, l, mask=valid)
    tl.store(r_ptr, r, mask=valid)


# -------------------------------------------------------------------------
# Entry‑point
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : (x, mask, weights, config)
        x       : Tensor[ B, N, N, dim ]  (float32)
        mask    : Tensor[ B, N, N ] or ``None`` (bool / float32)
        weights : dict of parameters (float32)
        config  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    Tensor[ B, N, N, dim ] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]          # denoted H in the paper

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm + linear projections (remain in FP32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(x.dtype),
        bias=weights["norm.bias"].to(x.dtype),
    )
    # → (B,N,N,hidden_dim)
    left  = F.linear(x, weights["left_proj.weight"].to(x.dtype))
    right = F.linear(x, weights["right_proj.weight"].to(x.dtype))

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(x.dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(x.dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(x.dtype)))

    # -----------------------------------------------------------------
    # 2️⃣ Fuse mask & gates with a tiny Triton kernel
    # -----------------------------------------------------------------
    # mask → (B,N,N,1) fp16, all‑ones when mask is None
    if mask is None:
        mask_fp16 = torch.ones((B, N, N, 1), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)

    # Cast everything to fp16 for the kernel (memory‑bound, no precision loss)
    left_fp16   = left.to(torch.float16).contiguous()
    right_fp16  = right.to(torch.float16).contiguous()
    left_gate_fp16  = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()

    total_elems = B * N * N * hidden_dim
    BLOCK = 1024                                     # good occupancy for 1‑D kernel
    grid = (triton.cdiv(total_elems, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        mask_fp16.stride(0), mask_fp16.stride(1),
        mask_fp16.stride(2), mask_fp16.stride(3),
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 3️⃣ O(N³) contraction via batched GEMM (fp16 → fp32 after)
    # -----------------------------------------------------------------
    # reshape to (B·H, N, N)
    left_bh  = left_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_bh = right_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Batched matrix multiply: (B·H,N,N) @ (B·H,N,N)^T
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))

    # Back to (B,N,N,hidden_dim) and fp32 for the remaining layers
    out = out_bh.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(x.dtype)                     # fp32

    # -----------------------------------------------------------------
    # 4️⃣ Post‑processing (layer‑norm, out‑gate, final linear)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(x.dtype),
        bias=weights["to_out_norm.bias"].to(x.dtype),
    )
    out = out * out_gate                       # broadcast over N,N
    out = F.linear(out, weights["to_out.weight"].to(x.dtype))

    return out
