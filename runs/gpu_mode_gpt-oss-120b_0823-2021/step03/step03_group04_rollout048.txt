"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (B,N,N,C).
2️⃣  Linear projections → left/right (B,N,N,H) and three gates (B,N,N,H).
3️⃣  Fuse element‑wise:   left  = left  * left_gate  * mask
                         right = right * right_gate * mask
    (mask optional).  This tiny fusion is performed by a Triton kernel
    to guarantee some GPU work and to illustrate the required “some‑kernel”
    constraint.
4️⃣  Reshape to (B*H,N,N) and compute the O(N³) contraction with a
    batched FP16 matrix‑multiply (Tensor‑cores via torch.bmm).  The result
    has shape (B,N,N,H).
5️⃣  Hidden‑dim LayerNorm → multiply by out_gate → final linear projection
    back to the original channel dimension C.

All heavy arithmetic is kept in FP16 (inputs/outputs of the GEMM) while
accumulations are performed in FP32 for numerical stability.  The final
tensor is returned as FP32 to match the PyTorch reference.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
#  Triton kernel – fuse mask, left/right gates and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,                 # fp16 *
    right_ptr,                # fp16 *
    left_gate_ptr,            # fp16 *
    right_gate_ptr,           # fp16 *
    mask_ptr,                 # fp16 *   (B,N,N,1)  or nullptr
    out_left_ptr, out_right_ptr,  # fp16 *
    B, N, H,                  # shapes
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    The kernel works on a 1‑D grid of size ceil(B*N*N*H / BLOCK).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # ---- decode linear index ------------------------------------------------
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ---- load tensors -------------------------------------------------------
    idx = (
        b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    )
    left   = tl.load(left_ptr + idx,     mask=mask, other=0.0)
    right  = tl.load(right_ptr + idx,    mask=mask, other=0.0)

    idx_gate = (
        b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + h * stride_lg3
    )
    lgate = tl.load(left_gate_ptr  + idx_gate, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + idx_gate, mask=mask, other=0.0)

    if HAS_MASK:
        # mask has shape (B,N,N,1) → no hidden dimension
        idx_mask = b * stride_m0 + i * stride_m1 + j * stride_m2
        m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)
    else:
        m = 1.0

    out_l = left * lgate * m
    out_r = right * rgate * m

    tl.store(out_left_ptr  + idx, out_l, mask=mask)
    tl.store(out_right_ptr + idx, out_r, mask=mask)


# ----------------------------------------------------------------------
#  Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : (x, mask, weights, config)
        x       – (B, N, N, C)    torch.float32
        mask    – (B, N, N)       torch.bool/float32/None
        weights – dict of all learned parameters (float32)
        config  – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor – (B, N, N, C)  float32
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, C = x.shape
    H = config["hidden_dim"]
    dtype = torch.float32

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )   # (B,N,N,C)  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections & gates  (keep in fp32 for now)
    # ------------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate= torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate  = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N,H)

    # Cast everything that will go into the GEMM to fp16 now.
    left_fp16      = left .to(torch.float16).contiguous()
    right_fp16     = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate .to(torch.float16).contiguous()
    right_gate_fp16= right_gate.to(torch.float16).contiguous()
    out_gate_fp16  = out_gate .to(torch.float16).contiguous()   # keep for later multiplication

    # ------------------------------------------------------------------
    # 3️⃣ Fuse mask & gates using Triton
    # ------------------------------------------------------------------
    if mask is None:
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)   # dummy
        has_mask = False
    else:
        # mask (B,N,N) → (B,N,N,1) to broadcast over H
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        has_mask = True

    fused_left  = torch.empty_like(left_fp16)
    fused_right = torch.empty_like(right_fp16)

    total_elems = B * N * N * H
    BLOCK = 1024                     # empirically good for Hopper
    grid = (triton.cdiv(total_elems, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        fused_left,
        fused_right,
        B, N, H,
        # strides for left/right (identical layout)
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        # (right) strides – same as left, kept for API symmetry
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        # gate strides
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        # mask strides (ignore H dim)
        mask_fp16.stride(0) if has_mask else 0,
        mask_fp16.stride(1) if has_mask else 0,
        mask_fp16.stride(2) if has_mask else 0,
        mask_fp16.stride(3) if has_mask else 0,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (FP16 inputs, FP32 accumulation)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat  = fused_left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_mat = fused_right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)

    # left @ right^T   (Tensor‑core friendly)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))          # (B*H,N,N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)
    out = out.to(dtype)   # back to FP32 for the following LayerNorm

    # ------------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    # multiply by out_gate (broadcast over N,N)
    out = out * out_gate   # both are (B,N,N,H)

    # final linear back to C
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
