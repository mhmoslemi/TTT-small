"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)  
Fast forward implementation that

1️⃣  performs the two LayerNorms in pure PyTorch (float32) – they are cheap.  
2️⃣  casts the normalized tensor to fp16 and runs all linear projections with
    torch.nn.functional.linear (the weight tensors are stored as fp16).  
3️⃣  **fuses the optional binary mask application** for the left/right
    projected tensors in a tiny Triton kernel (one‑load / one‑store per element).  
    This is the only required Triton component – it removes a memory‑bound
    PyTorch *mul* that would otherwise be executed twice.  
4️⃣  uses cuBLAS (torch.bmm) for the dominant O(N³) contraction.  The tensors
    are reshaped to (B·H, N, N) and multiplied in fp16 with fp32 accumulation,
    which on Hopper gives tensor‑core throughput.  
5️⃣  finishes the remaining element‑wise ops (LayerNorm, gating, final linear)
    in PyTorch.  

All intermediate tensors stay on the GPU; the final result is float32
[bs, seq_len, seq_len, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – mask fusion (optional)
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    src_ptr,                # left or right projection, fp16
    mask_ptr,               # binary mask, fp16, broadcasted over hidden dim
    dst_ptr,                # output, fp16
    N, H,                   # N = seq_len, H = hidden_dim
    stride_src_n, stride_src_h, stride_src_c,   # src strides (n, h, c)
    stride_mask_n, stride_mask_h,               # mask strides (n, h) – mask has shape (B,N,N,1)
    stride_dst_n, stride_dst_h, stride_dst_c,   # dst strides (n, h, c)
    BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each element (n, h, c) copies src * mask to dst.
    src shape : (B, N, N, H)  (flattened as n,h,c)
    mask shape: (B, N, N, 1)  (broadcast over H)
    """
    pid_n = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask = tl.load(mask_ptr + offs_n[:, None] * stride_mask_n
                              + offs_h[None, :] * stride_mask_h,
                   mask=offs_n[:, None] < N,
                   other=1.0)          # if no mask we launch with mask==1

    src = tl.load(src_ptr + offs_n[:, None] * stride_src_n
                            + offs_h[None, :] * stride_src_h,
                  mask=offs_n[:, None] < N,
                  other=0.0)

    dst = src * mask

    tl.store(dst_ptr + offs_n[:, None] * stride_dst_n
                     + offs_h[None, :] * stride_dst_h,
             dst,
             mask=offs_n[:, None] < N)


def _apply_mask(src: torch.Tensor, mask: torch.Tensor | None) -> torch.Tensor:
    """
    src : (B, N, N, H)   fp16
    mask: (B, N, N)      bool/float16  or None
    returns src * mask (fp16) – if mask is None the input is returned unchanged.
    """
    if mask is None:
        return src

    B, N, _, H = src.shape
    src = src.contiguous()
    mask = mask.to(torch.float16).unsqueeze(-1).contiguous()          # (B,N,N,1)

    out = torch.empty_like(src)

    # strides in element units
    stride_src_n, stride_src_h, stride_src_c = src.stride()
    stride_mask_n, stride_mask_h, _ = mask.stride()
    stride_out_n, stride_out_h, stride_out_c = out.stride()

    BLOCK_N = 64
    BLOCK_H = 64

    grid = (
        triton.cdiv(N, BLOCK_N),   # n‑tiles
        triton.cdiv(H, BLOCK_H),   # h‑tiles
        B,                         # batch dimension (program_id(2) == batch)
    )

    _apply_mask_kernel[grid](
        src_ptr=src,
        mask_ptr=mask,
        dst_ptr=out,
        N=N,
        H=H,
        stride_src_n=stride_src_n,
        stride_src_h=stride_src_h,
        stride_src_c=stride_src_c,
        stride_mask_n=stride_mask_n,
        stride_mask_h=stride_mask_h,
        stride_dst_n=stride_out_n,
        stride_dst_h=stride_out_h,
        stride_dst_c=stride_out_c,
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors (see reference implementation)
        config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]
    device = x.device

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (float32, manual)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                               # still fp32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (fp16)
    # --------------------------------------------------------------
    def _fp16(t): return t.to(torch.float16)
    left_w   = _fp16(weights["left_proj.weight"])
    right_w  = _fp16(weights["right_proj.weight"])
    left_gate_w  = _fp16(weights["left_gate.weight"])
    right_gate_w = _fp16(weights["right_gate.weight"])
    out_gate_w   = _fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_w)          # (B,N,N,hidden)
    right = F.linear(x_fp16, right_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask – fused Triton kernel
    # --------------------------------------------------------------
    left  = _apply_mask(left,  mask)
    right = _apply_mask(right, mask)

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – stay in fp16
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy contraction with cuBLAS (batched GEMM)
    #    left  : (B, N, N, H) → (B*H, N, N)
    #    right : (B, N, N, H) → (B*H, N, N) then transpose K,J
    # --------------------------------------------------------------
    left_reshape  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_reshape = right.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N)

    BH, _, _ = left_reshape.shape[:2]
    left_mat  = left_reshape.view(B * hidden, N, N)          # (B*H, I, K)
    right_mat = right_reshape.view(B * hidden, N, N)         # (B*H, K, J)

    # torch.bmm uses fp16×fp16→fp32 accumulation on Hopper
    out_mat = torch.bmm(left_mat, right_mat)                # (B*H, N, N)

    out_fp16 = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_fp16.to(torch.float32)                        # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (float32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear (float32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over hidden dim
    to_out_w = weights["to_out.weight"]             # (dim, hidden)
    out = F.linear(out, to_out_w)                   # (B,N,N,dim)

    return out
