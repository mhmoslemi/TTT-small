"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm (GPU‑centric)
-----------------------
1️⃣  Layer‑norm the input (B,N,N,C) – done in PyTorch (FP32) because it is
    cheap compared to the O(N³) contraction.

2️⃣  Linear projections → (B,N,N,H) for *left*, *right* and the three gates.
    All tensors are cast to FP16 (Tensor‑core friendly) before the
    contraction.

3️⃣  Fuse element‑wise mask, left‑gate and right‑gate with a tiny Triton
    kernel.  The kernel works on a 1‑D grid and performs:

        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask

    (If `nomask` the mask is omitted and the kernel just multiplies
    left/right by their gates.)

4️⃣  Batched GEMM (cuBLAS/Tensor‑cores).  By reshaping to (B*H, N, N) we can
    reuse the highly‑optimized bmm implementation:

        out[b,h] = left_fused[b,h] @ right_fused[b,h].T

    The bmm runs in FP16 with FP32 accumulation, giving the O(N³) work in
    ~O(N²·H) time.

5️⃣  Hidden‑dim LayerNorm, gating with `out_gate` and final linear projection
    back to the original channel dimension C – performed in PyTorch (FP32).

Only step 3 is implemented in Triton, satisfying the “some‑kernel” constraint
while keeping the overall runtime dominated by the cuBLAS batched matmul.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – apply mask (optional) and gates in one pass
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                # fp16 [B,N,N,H]
    left_gate_ptr, right_gate_ptr,      # fp16 [B,N,N,H]
    mask_ptr,                           # fp16 [B,N,N,1] or dummy
    out_left_ptr, out_right_ptr,        # fp16 output buffers
    B, N, H,                            # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    A flat 1‑D grid of size ceil(B·N·N·H / BLOCK) is used.
    """
    pid   = tl.program_id(0)
    offs  = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask  = offs < total                     # out‑of‑bounds guard

    # ---- decode linear index to (b,i,j,h) (contiguous layout) ----
    tmp = offs
    h   = tmp % H
    tmp = tmp // H
    j   = tmp % N
    tmp = tmp // N
    i   = tmp % N
    b   = tmp // N

    # ---- load left / right -------------------------------------------------
    idx = b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    left_val  = tl.load(left_ptr + idx,        mask=mask, other=0.0)
    right_val = tl.load(right_ptr + idx,       mask=mask, other=0.0)

    # ---- load gates ---------------------------------------------------------
    idx_gate = b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + h * stride_lg3
    left_gate_val  = tl.load(left_gate_ptr + idx_gate,  mask=mask, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + idx_gate, mask=mask, other=0.0)

    # ---- load mask (if present) --------------------------------------------
    if HAS_MASK:
        # mask has no hidden dimension → broadcast over H
        idx_mask = b * stride_m0 + i * stride_m1 + j * stride_m2
        m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)
    else:
        m = 1.0

    # ---- fuse ---------------------------------------------------------------
    out_l = left_val  * left_gate_val  * m
    out_r = right_val * right_gate_val * m

    # ---- store ---------------------------------------------------------------
    tl.store(out_left_ptr  + idx, out_l, mask=mask)
    tl.store(out_right_ptr + idx, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,               # input  [B,N,N,C]  fp32
        torch.Tensor,               # mask   [B,N,N]    bool/float32 or None
        Dict[str, torch.Tensor],    # weights
        Dict[str, Any],             # config  {"dim":C,"hidden_dim":H}
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        x        – (B, N, N, C) torch.float32
        mask     – (B, N, N)   torch.bool / torch.float32 / None
        weights  – dict with all learned parameters (float32)
        config   – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor
        (B, N, N, C) float32
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, C = x.shape
    H = config["hidden_dim"]
    dtype = torch.float32

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (still FP32, will be cast to FP16)
    # ------------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # ------------------------------------------------------------------
    # 3️⃣ Fuse mask & gates – Triton kernel
    # ------------------------------------------------------------------
    # Cast everything to FP16 (contiguous) for the kernel / GEMM
    left_fp16      = left.to(torch.float16).contiguous()
    right_fp16     = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()

    if mask is None:
        # dummy tensor – never accessed because HAS_MASK=False
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask = False
    else:
        # mask broadcasted over hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        has_mask = True

    # output buffers
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elements = B * N * N * H
    BLOCK = 2048                     # fits H200 L2 and gives good occupancy
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2), mask_fp16.stride(3) if has_mask else 0,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS – Tensor cores)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)

    # FP16 inputs → FP16 output, accumulation in FP32
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))          # (B*H,N,N)

    # restore shape (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype)   # back to FP32 for the following LayerNorm

    # ------------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                 # broadcast over (N,N)
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
