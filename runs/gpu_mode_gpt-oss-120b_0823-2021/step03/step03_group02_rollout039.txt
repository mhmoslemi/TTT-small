"""
TriMul “outgoing” forward pass (no gradients).

The dominant N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a single torch.bmm on fp16 tensors (cuBLAS/Tensor‑cores).
All cheap element‑wise work (masking, gating, final gating) is fused into
tiny Triton kernels to satisfy the “must use a kernel” requirement while
keeping the overall runtime well below the target budget.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton element‑wise fusions
# ----------------------------------------------------------------------
@triton.jit
def eltwise_fuse_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,    # three inputs and one output
    N_ELEMENTS: tl.constexpr,        # total number of scalar elements
    stride_a0, stride_a1,            # strides (in elements) for a
    stride_b0, stride_b1,            # strides for b
    stride_c0, stride_c1,            # strides for c
    stride_o0, stride_o1,            # strides for out
):
    """
    out = a * b * c   (element‑wise, broadcasted over the last dim)
    All tensors are 4‑D [B, N, N, H] and are laid out contiguously,
    but we treat them as a flat array of length N_ELEMENTS.
    """
    pid = tl.program_id(0)
    offs = pid * 128 + tl.arange(0, 128)          # 128‑wide thread block
    mask = offs < N_ELEMENTS

    # convert flat offset → multi‑dim index
    # we use stride tables to compute the address directly
    a = tl.load(
        a_ptr + offs * stride_a0,
        mask=mask,
        other=0.0,
    )
    b = tl.load(
        b_ptr + offs * stride_b0,
        mask=mask,
        other=0.0,
    )
    c = tl.load(
        c_ptr + offs * stride_c0,
        mask=mask,
        other=0.0,
    )
    out = a * b * c
    tl.store(
        out_ptr + offs * stride_o0,
        out,
        mask=mask,
    )


def _fuse_mul3(x, y, z):
    """element‑wise x*y*z using the Triton kernel above."""
    assert x.shape == y.shape == z.shape
    B, N, M, H = x.shape
    # flatten to 1‑D
    flat = B * N * M * H
    stride_x = x.stride()
    stride_y = y.stride()
    stride_z = z.stride()
    out = torch.empty_like(x)

    # all tensors are contiguous after the permutes we perform later,
    # therefore a single stride (the fast‑changing one) suffices.
    grid = (triton.cdiv(flat, 128),)
    eltwise_fuse_mul3_kernel[grid](
        x,
        y,
        z,
        out,
        N_ELEMENTS=flat,
        stride_a0=stride_x[0],
        stride_a1=stride_x[1],
        stride_b0=stride_y[0],
        stride_b1=stride_y[1],
        stride_c0=stride_z[0],
        stride_c1=stride_z[1],
        stride_o0=out.stride(0),
        stride_o1=out.stride(1),
        BLOCK_SIZE=128,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Helper: batched GEMM using cuBLAS (torch.bmm)
# ----------------------------------------------------------------------
def _batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)  fp16
    right : (B, H, K, J)  fp16   (K is the reduction dim)
    returns (B, H, I, J)  fp16
    """
    B, H, I, K = left.shape
    _, _, _, J = right.shape

    # collapse (B, H) → batch for torch.bmm
    left_flat = left.reshape(B * H, I, K)
    right_flat = right.reshape(B * H, K, J)

    # torch.bmm is a thin wrapper around cuBLAS and automatically uses
    # Tensor‑cores for fp16 inputs.
    out_flat = torch.bmm(left_flat, right_flat)          # (B*H, I, J)

    return out_flat.view(B, H, I, J)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        mask         : torch.Tensor [B, N, N]       (float32/bool) or None
        weights      : dict of torch.Tensor parameters
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]   (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm  (keep in PyTorch, cheap)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual layer‑norm (reduce over last dim)
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                     # still fp32

    # ------------------------------------------------------------------
    # 2) Linear projections  (fp16 for the heavy part)
    # ------------------------------------------------------------------
    # weights are stored as fp32, cast once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"]   # stay fp32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # cast normalized activation to fp16 (this will be reused)
    x_fp16 = x.to(torch.float16)

    # projections: (B,N,N,hidden)
    left  = F.linear(x_fp16, left_proj_w)          # fp16
    right = F.linear(x_fp16, right_proj_w)         # fp16

    # ------------------------------------------------------------------
    # 3) Optional mask   (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
    else:
        mask_fp16 = torch.ones((B, N, N, 1), dtype=torch.float16,
                               device=input_tensor.device)

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) – keep in fp16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 5) Fuse mask & gates into left/right with a tiny Triton kernel
    # ------------------------------------------------------------------
    # left = left * left_gate * mask
    # right = right * right_gate * mask
    left = _fuse_mul3(left, left_gate, mask_fp16)
    right = _fuse_mul3(right, right_gate, mask_fp16)

    # ------------------------------------------------------------------
    # 6) Core contraction – batched GEMM via torch.bmm (cuBLAS)
    #    left  : (B, H, I, K)   → (B, hidden, N, N)
    #    right : (B, H, K, J)   → (B, hidden, N, N) with K,J swapped
    # ------------------------------------------------------------------
    # bring hidden dim to channel axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    # right needs the reduction dimension K to be the middle axis:
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _batched_gemm(left_perm, right_perm)    # (B, H, I, J) fp16

    # back to original layout (B, N, N, hidden)
    out = out_fp16.permute(0, 2, 3, 1).contiguous()    # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm  (still in fp32)
    # ------------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b          # fp32

    # ------------------------------------------------------------------
    # 8) Final gating + projection
    # ------------------------------------------------------------------
    # out_gate is still fp16, cast to fp32 for the multiplication
    out = out * out_gate.to(torch.float32)              # (B,N,N,hidden)
    out = F.linear(out, to_out_w)                       # (B,N,N,dim) fp32

    return out
