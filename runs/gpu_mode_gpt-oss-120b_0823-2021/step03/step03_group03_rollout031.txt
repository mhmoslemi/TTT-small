"""
TriMul “outgoing” forward – fast implementation for H200.

Algorithm
---------
1️⃣  Layer‑norm the input (FP32) with a tiny Triton kernel.
2️⃣  Cast to FP16 and apply the four linear layers (two projections + three
    gates).  All linear ops are done with torch.nn.functional.linear which
    internally uses cuBLAS / Tensor‑cores.
3️⃣  Optional pair‑wise mask (broadcasted) and gating (sigmoid) – still FP16.
4️⃣  Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply over the *k* dimension.  We reshape the
   tensors to (B·H, N, N) and call torch.bmm – cuBLAS on H200 is far faster
   than any hand‑written Triton N³ GEMM.
5️⃣  Convert back to FP32, second Layer‑norm (again Triton), multiply by the
   output gate and a final linear projection to the original dimension.

Only the two Layer‑Norms are written in Triton; everything else stays in
PyTorch so we keep the implementation simple, correct and fast.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton Layer‑Norm kernel (FP32)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    src_ptr,                # input  (rows, dim)
    dst_ptr,                # output (rows, dim)
    weight_ptr,             # gamma  (dim,)
    bias_ptr,               # beta   (dim,)
    rows, dim,              # extents
    stride_src_row, stride_src_col,
    stride_dst_row, stride_dst_col,
    stride_w,
    stride_b,
    eps: tl.constexpr,
    BLOCK_D: tl.constexpr,          # tile size on the feature dimension
):
    row_id = tl.program_id(0)                 # which row we work on
    offs_d = tl.arange(0, BLOCK_D)            # feature offsets

    # mask for the last (partial) block
    mask_d = offs_d < dim

    # ------------------------------------------------------------------
    # Load a tile of the row
    # ------------------------------------------------------------------
    src = tl.load(
        src_ptr + row_id * stride_src_row + offs_d * stride_src_col,
        mask=mask_d,
        other=0.0,
    )

    # ------------------------------------------------------------------
    # Reduce to compute mean and variance (two‑pass reduction)
    # ------------------------------------------------------------------
    # 1) mean
    sum_val = tl.sum(src, axis=0)
    n = dim
    mean = sum_val / n

    # 2) variance
    diff = src - mean
    var = tl.sum(diff * diff, axis=0) / n
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Normalize, apply affine parameters
    # ------------------------------------------------------------------
    norm = (src - mean) * inv_std

    weight = tl.load(weight_ptr + offs_d, mask=mask_d, other=1.0)
    bias   = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

    out = norm * weight + bias

    # ------------------------------------------------------------------
    # Store
    # ------------------------------------------------------------------
    tl.store(
        dst_ptr + row_id * stride_dst_row + offs_d * stride_dst_col,
        out,
        mask=mask_d,
    )


def _triton_layernorm(x: torch.Tensor, weight: torch.Tensor,
                      bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """
    Layer‑norm over the last dimension of a FP32 tensor.
    x : (B, N, N, C)  – contiguous
    weight, bias : (C,)
    Returns a new tensor of same shape and dtype.
    """
    assert x.is_contiguous()
    B, N, _, C = x.shape
    rows = B * N * N
    dim = C

    out = torch.empty_like(x)

    # strides for a (rows, dim) view
    x_view = x.view(rows, dim)
    out_view = out.view(rows, dim)

    stride_src_row, stride_src_col = x_view.stride()
    stride_dst_row, stride_dst_col = out_view.stride()
    stride_w = weight.stride(0)
    stride_b = bias.stride(0)

    BLOCK_D = 128                                    # good trade‑off on H200
    grid = (triton.cdiv(rows, 1),)                  # one program per row

    _layernorm_kernel[grid](
        src_ptr=x_view,
        dst_ptr=out_view,
        weight_ptr=weight,
        bias_ptr=bias,
        rows=rows,
        dim=dim,
        stride_src_row=stride_src_row,
        stride_src_col=stride_src_col,
        stride_dst_row=stride_dst_row,
        stride_dst_col=stride_dst_col,
        stride_w=stride_w,
        stride_b=stride_b,
        eps=eps,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors (see reference implementation)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (Triton, FP32)
    # --------------------------------------------------------------
    inp_norm = _triton_layernorm(
        inp,
        weights["norm.weight"],
        weights["norm.bias"],
    )                       # (B,N,N,dim)  FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gates – cast to FP16 for speed
    # --------------------------------------------------------------
    # weight tensors are stored as FP32 – cast once
    def _fp16(t): return t.to(torch.float16)

    left_proj_w   = _fp16(weights["left_proj.weight"])    # (hidden_dim, dim)
    right_proj_w  = _fp16(weights["right_proj.weight"])
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])

    x_fp16 = inp_norm.to(torch.float16)                 # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted) – still FP16
    # --------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)       # (B,N,N,1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – FP16
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM via torch.bmm (cuBLAS/Tensor‑cores)
    #    left : (B, N, N, H) → (B·H, N, N)
    #    right: (B, N, N, H) → (B·H, N, N) and need (K,J) = (N,N) transposed
    # --------------------------------------------------------------
    # bring the hidden dimension to the second axis
    left_t  = left.permute(0, 3, 1, 2).contiguous()    # (B, H, I, K) == (B,H,N,N)
    right_t = right.permute(0, 3, 2, 1).contiguous()   # (B, H, K, J)

    B_, H_, I_, K_ = left_t.shape
    _, _, K2_, J_ = right_t.shape
    assert K_ == K2_ and I_ == J_ == N

    # collapse batch & heads → a single bmm batch
    left_bmm  = left_t.view(B_ * H_, I_, K_)            # (B*H, N, N)
    right_bmm = right_t.view(B_ * H_, K_, J_)           # (B*H, N, N)

    # batched matrix‑multiply: (B*H, N, N) @ (B*H, N, N)^T  → (B*H, N, N)
    out_fp16 = torch.bmm(left_bmm, right_bmm)          # FP16 accumulation, fast

    # reshape back
    out_fp16 = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # (B, N, N, hidden_dim) → FP32 for the following layers
    out = out_fp16.to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (Triton, FP32)
    # --------------------------------------------------------------
    out = _triton_layernorm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 7️⃣ Output gating + final projection (still FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)               # broadcast over dim
    out = F.linear(out, weights["to_out.weight"])       # (B,N,N,dim)

    return out
