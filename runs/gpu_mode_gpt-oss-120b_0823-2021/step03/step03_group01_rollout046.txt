"""
custom_kernel – “outgoing” TriMul forward pass (AlphaFold‑3 style).

The computational bottleneck is the cubic contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply over the “pair‑wise” dimension N.
We implement this contraction with a small Triton kernel that performs a
batched GEMM on the tensors (B, H, N, N) → (B, H, N, N) where H = hidden_dim
(the channel dimension after the linear projections).  All the surrounding
operations – LayerNorm, linear projections, gating, final LayerNorm and the
output linear – are carried out in regular PyTorch (fp32 → fp16 where
beneficial).  Mixed‑precision (fp16 compute, fp32 accumulation) is used inside
the kernel, and the final result is converted back to fp32 so that the rest of
the model sees the same datatype as the reference implementation.

The kernel is deliberately simple and tuned for Hopper (SM‑90):
* BLOCK_M = BLOCK_N = 64 tiles a 64×64 sub‑matrix per program‑instance.
* BLOCK_K = 32 gives good Tensor‑Core utilisation.
* 4 warps per block (the default on SM‑90) and fp32 accumulation keep
  numerical error well below the reference tolerance.

The overall algorithm matches the reference PyTorch code exactly, including
the optional mask handling and the three gating sigmoids.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (I×K)·(K×J) → (I×J) for each head H
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    a_ptr,                # left  tensor pointer   (Z, I, K)
    b_ptr,                # right tensor pointer   (Z, K, J)
    c_ptr,                # output tensor pointer  (Z, I, J)
    M, N, K,              # matrix shapes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)          # batch * head index
    pid_m = tl.program_id(1)          # tile‑row index
    pid_n = tl.program_id(2)          # tile‑col index

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (higher precision for the sum over K)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load tiles
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 accumulation (a, b are fp16)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write result (store as fp16, then cast back to fp32 outside)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    Parameters
    ----------
    left : Tensor (B, H, I, K)  – dtype float16, contiguous
    right: Tensor (B, H, K, J)  – dtype float16, contiguous

    Returns
    -------
    out : Tensor (B, H, I, J)   – dtype float16, contiguous
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse leading dims so the kernel sees (Z, *, *)
    Z = B * H
    a = left.reshape(Z, I, K)
    b = right.reshape(Z, K, J)
    c = out.reshape(Z, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling (tuned for Hopper / Tensor‑Core)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                                 # program_id(0) = batch*head
        triton.cdiv(I, BLOCK_M),           # program_id(1) = tile‑row
        triton.cdiv(J, BLOCK_N),           # program_id(2) = tile‑col
    )

    batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # SM‑90 works well with 4 warps per block
    )

    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor  [B, N, N, dim]   (float32)
        mask         : torch.Tensor  [B, N, N]       (bool/float32) or None
        weights      : dict of parameter tensors (float32)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 – numerical stability)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                                 # [B,N,N,dim] fp32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections – cast to FP16 for the heavy work
    # --------------------------------------------------------------
    def to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w  = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w  = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w   = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)    # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask – broadcast over hidden dimension
    # --------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – still in FP16
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Contract over the pair‑wise dimension N with Triton GEMM
    # --------------------------------------------------------------
    # Rearrange to (B, H, I, K) and (B, H, K, J) where H = hidden_dim,
    # I = J = N, K = N.
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = triton_batched_gemm(left_t, right_t)   # (B, H, I, J)

    # Back to the original layout and convert to FP32 for the remaining ops
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,dim)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm
    # --------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = out * to_out_norm_w + to_out_norm_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)            # broadcast over dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                     # (B,N,N,dim)

    return out
