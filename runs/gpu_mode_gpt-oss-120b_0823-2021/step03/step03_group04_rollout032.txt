"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm (GPU‑accelerated parts)
---------------------------------
1️⃣ Layer‑norm the [B,N,N,C] input (torch‑level, fp32).
2️⃣ Linear projections → tensors of shape (B,N,N, H) for left/right and three
   gating branches (still fp32).
3️⃣ Fuse element‑wise mask, left‑gate and right‑gate using a tiny Triton kernel.
   The kernel works on a 1‑D grid over the total element count B·N·N·H and
   writes two output tensors (left_fused, right_fused) in fp16.
4️⃣ Batched GEMM (cuBLAS / Tensor‑cores) on the fused tensors:
       left_fused  @ right_fusedᵀ  →  (B·H, N, N)
   The contraction implements the Einstein sum
   `einsum('... i k d, ... j k d -> ... i j d')` of the reference model.
5️⃣ Hidden‑dim LayerNorm, out‑gate multiplication and final linear projection
   back to the original channel dimension C (torch‑level, fp32).

Only step 3 is performed in Triton – this satisfies the “some‑kernel” constraint
while keeping the dominant O(N³) work on cuBLAS where it runs fastest on the
H200 (Tensor‑core friendly fp16).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – fuse mask and gates (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    # input tensors (fp16)
    left_ptr, right_ptr,               # (B,N,N,H)
    left_gate_ptr, right_gate_ptr,     # (B,N,N,H)
    mask_ptr,                          # (B,N,N,1)  or nullptr
    # output tensors (fp16)
    out_left_ptr, out_right_ptr,       # (B,N,N,H)
    # shape / stride metadata
    B, N, H,
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_g0, stride_g1, stride_g2, stride_g3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    # compile‑time flags
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    The kernel is launched with a 1‑D grid of size
        ceil(B*N*N*H / BLOCK).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # ----- decode linear index → (b,i,j,h) ---------------------------
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ----- load left, right, gates ------------------------------------
    idx = (
        b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    )
    left  = tl.load(left_ptr  + idx, mask=mask, other=0.0)
    right = tl.load(right_ptr + idx, mask=mask, other=0.0)

    idx_gate = (
        b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    )
    lgate = tl.load(left_gate_ptr  + idx_gate, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + idx_gate, mask=mask, other=0.0)

    # ----- optional mask ----------------------------------------------
    if HAS_MASK:
        # mask has no H dimension → broadcast
        idx_mask = b * stride_m0 + i * stride_m1 + j * stride_m2
        m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)
    else:
        m = 1.0

    # ----- write fused results -----------------------------------------
    out_l = left * lgate * m
    out_r = right * rgate * m
    tl.store(out_left_ptr  + idx, out_l, mask=mask)
    tl.store(out_right_ptr + idx, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : (x, mask, weights, config)
        x       – (B, N, N, C)      torch.float32
        mask    – (B, N, N) or None (torch.bool/float32)
        weights – dict with LayerNorm / Linear parameters (float32)
        config  – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor – (B, N, N, C)  float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, C = x.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (fp32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (fp32 → fp16 later)
    # ------------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))      # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # ------------------------------------------------------------------
    # 3️⃣ Fuse mask + gates in fp16 via Triton
    # ------------------------------------------------------------------
    #   keep everything contiguous & in fp16 for the upcoming GEMM
    left_fp16      = left.to(torch.float16).contiguous()
    right_fp16     = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate.to(torch.float16).contiguous()
    right_gate_fp16= right_gate.to(torch.float16).contiguous()

    if mask is None:
        # dummy tensor – strides are irrelevant because HAS_MASK=False
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask = False
    else:
        # mask broadcast over hidden dim H
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        has_mask = True

    # allocate destination buffers
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elements = B * N * N * H
    BLOCK = 1024                      # fits well in H200 shared‑memory budget
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        mask_fp16.stride(0) if has_mask else 0,
        mask_fp16.stride(1) if has_mask else 0,
        mask_fp16.stride(2) if has_mask else 0,
        mask_fp16.stride(3) if has_mask else 0,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (fp16 inputs → fp32 accumulation)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)   (order: B,H,N,N)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)

    # out = left @ rightᵀ   (cuBLAS/Tensor‑cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))           # (B*H,N,N)

    # back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype)          # fp32 for next LayerNorm

    # ------------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, out‑gate & final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                # broadcast over (N,N)
    out = F.linear(out, weights["to_out.weight"].to(dtype))  # (B,N,N,C)

    return out
