"""
TriMul “outgoing” forward (no gradients).

The O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a single torch.bmm on the reshaped tensors – this lets
cuBLAS/Tensor‑cores on the H200 do the heavy work.  The only operation that is
off‑loaded to Triton is the element‑wise combination of the (optional) mask
with the left/right tensors and their gates, which is a memory‑bound
broadcast that benefits from a fused kernel.

All other cheap ops (LayerNorms, linear projections, final gate &
linear) stay in PyTorch (FP16 where possible, FP32 for the final output).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:  out = a * mask * gate   (all fp16, broadcast on last dim)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    a_ptr,                 # left or right tensor      (Z, H, N, N) fp16
    mask_ptr,              # mask tensor (broadcast)  (Z, 1, N, N) fp16 or 0‑dim
    gate_ptr,              # gate tensor               (Z, H, N, N) fp16
    out_ptr,               # output tensor             (Z, H, N, N) fp16
    Z, H, N,               # logical dimensions
    stride_az, stride_ah, stride_an, stride_am,   # strides for a
    stride_mz, stride_mn, stride_mm,              # strides for mask (z, n, n)
    stride_gz, stride_gh, stride_gn, stride_gm,   # strides for gate
    stride_oz, stride_oh, stride_on, stride_om,   # strides for out
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # linear thread index
    total = Z * H * N * N
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # element offsets
    mask = offs < total

    # 1‑D index → 4‑D coordinates
    #  idx = ((z * H + h) * N + i) * N + j
    idx = offs
    j = idx % N
    i = (idx // N) % N
    h = (idx // (N * N)) % H
    z = idx // (H * N * N)

    # compute pointers
    a = a_ptr + z * stride_az + h * stride_ah + i * stride_an + j * stride_am
    g = gate_ptr + z * stride_gz + h * stride_gh + i * stride_gn + j * stride_gm
    m = mask_ptr + z * stride_mz + i * stride_mn + j * stride_mm   # mask has no head dim

    out = out_ptr + z * stride_oz + h * stride_oh + i * stride_on + j * stride_om

    a_val = tl.load(a, mask=mask, other=0.0)
    g_val = tl.load(g, mask=mask, other=0.0)
    m_val = tl.load(m, mask=mask, other=0.0)

    tl.store(out, a_val * m_val * g_val, mask=mask)


def apply_mask_and_gate(x, mask, gate):
    """
    Fuse ``x = x * mask * gate`` with a Triton kernel.
    Shapes:
        x   : (B, H, N, N)   fp16
        mask: (B, N, N)      fp16  (or None)
        gate: (B, H, N, N)   fp16
    Returns a new tensor of shape (B, H, N, N) fp16.
    """
    if mask is None:
        # simple element‑wise multiplication with gate
        return x * gate

    B, H, N, _ = x.shape
    Z = B                     # we keep batch dimension separate from head dim

    # make mask broadcastable to (B, H, N, N)
    mask_b = mask.unsqueeze(1).contiguous()          # (B,1,N,N)

    out = torch.empty_like(x)

    # strides in elements (not bytes)
    stride_x = x.stride()
    stride_m = mask_b.stride()
    stride_g = gate.stride()
    stride_o = out.stride()

    BLOCK = 128
    total_elems = B * H * N * N
    grid = (triton.cdiv(total_elems, BLOCK),)

    mask_gate_fuse_kernel[grid](
        a_ptr=x,
        mask_ptr=mask_b,
        gate_ptr=gate,
        out_ptr=out,
        Z=Z,
        H=H,
        N=N,
        stride_az=stride_x[0],
        stride_ah=stride_x[1],
        stride_an=stride_x[2],
        stride_am=stride_x[3],
        stride_mz=stride_m[0],
        stride_mn=stride_m[2],
        stride_mm=stride_m[3],
        stride_gz=stride_g[0],
        stride_gh=stride_g[1],
        stride_gn=stride_g[2],
        stride_gm=stride_g[3],
        stride_oz=stride_o[0],
        stride_oh=stride_o[1],
        stride_on=stride_o[2],
        stride_om=stride_o[3],
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim] (float32, CUDA)
        mask         : torch.Tensor [B, N, N] (float32/bool) or None
        weights      : dict of torch.Tensor parameters
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep in FP32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b               # still FP32

    # ------------------------------------------------------------------
    # 2) Linear projections (FP16) – no bias in the reference model
    # ------------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16, non_blocking=True)

    left_proj_w   = to_fp16(weights["left_proj.weight"])   # (hidden, dim)
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Gates (FP16)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    # bring hidden dim to 2nd axis for the upcoming batched GEMM
    # (B, H, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 2, 1).contiguous()   # note transpose on last two dims
    left_gate  = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous()

    # ------------------------------------------------------------------
    # 4) Apply optional mask and gates with a fused Triton kernel
    # ------------------------------------------------------------------
    # mask is (B,N,N) -> broadcast to (B,1,N,N)
    mask_fp16 = mask.to(torch.float16) if mask is not None else None

    left  = apply_mask_and_gate(left,  mask_fp16, left_gate)   # (B,H,N,N)
    right = apply_mask_and_gate(right, mask_fp16, right_gate)

    # ------------------------------------------------------------------
    # 5) Core contraction – batched GEMM via torch.bmm (leverages cuBLAS)
    #    left  : (B, H, N, N) -> (B*H, N, N)
    #    right : (B, H, N, N) -> (B*H, N, N)
    # ------------------------------------------------------------------
    BH = B * hidden_dim
    left_flat  = left.reshape(BH, N, N)
    right_flat = right.reshape(BH, N, N)

    # matmul over K dimension (the second N)
    out_flat = torch.bmm(left_flat, right_flat)          # (BH, N, N) fp16

    out_fp16 = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # now shape (B, N, N, hidden) – still fp16

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)

    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # ------------------------------------------------------------------
    # 7) Final gating (reuse out_gate computed earlier) and linear projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # (B,N,N,hidden)
    out = F.linear(out, weights["to_out.weight"])  # (B,N,N,dim)

    return out
