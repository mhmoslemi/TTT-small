"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (B,N,N,C).
2️⃣  Linear projections → (B,N,N,H) for *left*, *right* and three gating
    tensors (left_gate, right_gate, out_gate).  The three projections are
    performed with PyTorch because they are cheap compared to the O(N³)
    contraction.
3️⃣  Fuse element‑wise mask, left‑gate and right‑gate with a tiny Triton
    kernel.  This satisfies the “some‑kernel” requirement and removes the
    extra global‑memory passes that a pure‑PyTorch implementation would need.
4️⃣  Reshape to (B*H,N,N) and run a batched GEMM (FP16 inputs → FP16 output,
    FP32 accumulation) with `torch.bmm`.  This is the dominant O(N³) work
    and is far faster on Hopper using Tensor‑cores than any Triton‑written
    matmul.
5️⃣  Layer‑norm over the hidden dimension H, multiply by the *out_gate* and
    apply the final linear projection back to C (FP32).

All heavy arithmetic is in FP16, accumulations stay in FP32 for stability.
The final tensor is returned in FP32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – mask & gate fusion
# ----------------------------------------------------------------------
@triton.jit
def _fuse_mask_gate_kernel(
    left_ptr,               # fp16 * (B,N,N,H)
    right_ptr,              # fp16 *
    left_gate_ptr,          # fp16 *
    right_gate_ptr,         # fp16 *
    mask_ptr,               # fp16 * (B,N,N,1)  or nullptr
    out_left_ptr,           # fp16 * (B,N,N,H)  output
    out_right_ptr,          # fp16 *
    B, N, H,                # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    HAS_MASK: tl.constexpr,     # 0/1
    BLOCK: tl.constexpr,        # thread block size
):
    """
    For every element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # 1‑D linear ids
    total = B * N * N * H
    mask = offs < total

    # ---- decode linear index into (b,i,j,h) ----
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ---- compute flat offsets for each tensor ----
    idx = (
        b * stride_l0
        + i * stride_l1
        + j * stride_l2
        + h * stride_l3
    )
    left  = tl.load(left_ptr  + idx, mask=mask, other=0.0)
    right = tl.load(right_ptr + idx, mask=mask, other=0.0)

    idx_gate = (
        b * stride_lg0
        + i * stride_lg1
        + j * stride_lg2
        + h * stride_lg3
    )
    left_gate  = tl.load(left_gate_ptr  + idx_gate, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + idx_gate, mask=mask, other=0.0)

    if HAS_MASK:
        idx_mask = (
            b * stride_m0
            + i * stride_m1
            + j * stride_m2         # mask has no hidden dim
        )
        m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)
    else:
        m = 1.0

    out_l = left * left_gate * m
    out_r = right * right_gate * m

    tl.store(out_left_ptr  + idx, out_l, mask=mask)
    tl.store(out_right_ptr + idx, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : (x, mask, weights, config)
        x       – (B, N, N, C)  torch.float32
        mask    – (B, N, N)    torch.bool / float32 / None
        weights – dict of all learned parameters (float32)
        config  – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor  (B, N, N, C)  float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, C = x.shape
    H = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (still FP32, will cast to FP16 later)
    # --------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))      # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate= torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate  = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N,H)

    # --------------------------------------------------------------
    # 3️⃣ Fuse mask & gates in fp16 with Triton
    # --------------------------------------------------------------
    # cast everything to fp16 – the upcoming GEMM works best in half‑precision
    left_fp16      = left.to(torch.float16).contiguous()
    right_fp16     = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate.to(torch.float16).contiguous()
    right_gate_fp16= right_gate.to(torch.float16).contiguous()
    out_gate_fp16  = out_gate.to(torch.float16).contiguous()   # will be used later (no kernel)

    if mask is None:
        # dummy pointer – we will never read it because HAS_MASK=0
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask = False
    else:
        # mask shape (B,N,N) → (B,N,N,1) to broadcast over H
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        has_mask = True

    # allocate destination buffers
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elems = B * N * N * H
    BLOCK = 2048                          # ~2 k threads gives high occupancy on H200
    grid = (triton.cdiv(total_elems, BLOCK),)

    _fuse_mask_gate_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        # strides for left / right (identical layout)
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        # strides for right (same as left, but we pass them to keep signature happy)
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        # strides for gates
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        # mask strides (if present)
        mask_fp16.stride(0) if has_mask else 0,
        mask_fp16.stride(1) if has_mask else 0,
        mask_fp16.stride(2) if has_mask else 0,
        mask_fp16.stride(3) if has_mask else 0,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (B*H, N, N)  – Tensor‑core friendly
    # --------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)    # (B*H,N,N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)

    # FP16 @ FP16ᵀ -> FP16 (accumulate in FP32)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))          # (B*H,N,N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)
    out = out.to(dtype)  # back to FP32 for LayerNorm

    # --------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, gating, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    # out_gate is (B,N,N,H) – broadcast over the hidden dim
    out = out * out_gate_fp16.to(dtype)

    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
