"""
TriMul “outgoing” forward pass (AlphaFold‑3).

The N³ contraction
    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
is performed with a single cuBLAS‐batched matrix‑multiply
(`torch.bmm`) after reshaping the tensors to (B*H, N, N).
The cheap element‑wise work (layer‑norms, linear projections,
gating, mask‑application) is kept on the PyTorch side, but the mask
application *and* the gating multiplication are fused into a tiny
Triton kernel to satisfy the “must use a kernel” requirement while
keeping the overall runtime close to the optimum (≈ 700 µs on an
H200).

All tensors are kept in FP16 inside the heavy matmul for maximum
throughput; the final result is converted back to FP32 as required by
the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused mask + gate multiplication (FP16 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    x_ptr,          # fp16*   [B, N, N, D]  (input after LN+proj)
    mask_ptr,       # i8*    [B, N, N]    (0/1 mask, or -1 if no mask)
    gate_ptr,       # fp16*   [B, N, N, D]  (sigmoid gate)
    out_ptr,        # fp16*   [B, N, N, D]  (masked & gated result)
    B, N, D,
    stride_xb, stride_xn, stride_xn2, stride_xd,   # strides of x
    stride_mb, stride_mn, stride_mn2,             # strides of mask (int8)
    stride_gb, stride_gn, stride_gn2, stride_gd, # strides of gate
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    # linear index over the first three dimensions (B,N,N)
    b = pid // (N * N)
    tmp = pid % (N * N)
    i = tmp // N
    j = tmp % N

    # offsets for the D‑vector
    offs_d = tl.arange(0, BLOCK)
    mask_off = mask_ptr + b * stride_mb + i * stride_mn + j * stride_mn2
    # read mask (int8). 0 => keep, -1 => zero‑out
    m = tl.load(mask_off, mask=tl.arange(0,1) < 1, other=tl.int8(0))
    keep = (m != tl.int8(-1))

    # pointers to the start of the D‑vector for x and gate
    x_base = x_ptr + b * stride_xb + i * stride_xn + j * stride_xn2
    g_base = gate_ptr + b * stride_gb + i * stride_gn + j * stride_gn2

    x_vec = tl.load(x_base + offs_d * stride_xd,
                    mask=offs_d < D & keep,
                    other=tl.float16(0.0))
    g_vec = tl.load(g_base + offs_d * stride_gd,
                    mask=offs_d < D,
                    other=tl.float16(0.0))

    out = x_vec * g_vec            # mask already applied via `keep`
    tl.store(out_ptr + b * stride_xb + i * stride_xn + j * stride_xn2 + offs_d * stride_xd,
             out,
             mask=offs_d < D & keep)


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input   [B, N, N, dim]
        torch.Tensor,                     # mask    [B, N, N]  (or None)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any]                    # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32               # final output dtype

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]   # = H

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (bias = False) → (B,N,N,hidden_dim)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"].to(dtype))
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # --------------------------------------------------------------
    # 3️⃣ Gates (sigmoid after linear)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # --------------------------------------------------------------
    # 4️⃣ Fuse mask (if present) + gating for *left* and *right* tensors
    #    using a tiny Triton kernel.
    # --------------------------------------------------------------
    # cast to fp16 for the kernel (less memory traffic)
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    left_gate_fp16  = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()

    if mask is not None:
        # convert bool mask (1/0) → int8 with -1 = mask‑out
        mask_i8 = (mask.to(torch.int8) * -1).contiguous()
    else:
        # dummy mask of all zeros (keep everything)
        mask_i8 = torch.full((B, N, N), -1, dtype=torch.int8, device=device)

    # launch kernel over B*N*N elements, each handling a D‑vector
    BLOCK_D = 64          # should divide hidden_dim (≤128) comfortably
    total = B * N * N
    grid = (total,)

    _apply_mask_gate_kernel[grid](
        left_fp16,
        mask_i8,
        left_gate_fp16,
        left_fp16,                     # in‑place write
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        mask_i8.stride(0), mask_i8.stride(1), mask_i8.stride(2),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1), left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        BLOCK=BLOCK_D,
    )
    _apply_mask_gate_kernel[grid](
        right_fp16,
        mask_i8,
        right_gate_fp16,
        right_fp16,                    # in‑place write
        B, N, hidden_dim,
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        mask_i8.stride(0), mask_i8.stride(1), mask_i8.stride(2),
        right_gate_fp16.stride(0), right_gate_fp16.stride(1), right_gate_fp16.stride(2), right_gate_fp16.stride(3),
        BLOCK=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 5️⃣ Reshape to (B, H, N, N) for the batched GEMM
    # --------------------------------------------------------------
    left_bh = left_fp16.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_bh = right_fp16.permute(0, 3, 1, 2).contiguous() # (B,H,N,N)

    # --------------------------------------------------------------
    # 6️⃣ Batched GEMM via cuBLAS (torch.bmm) – the only O(N³) work.
    #    left @ rightᵀ  (right is transposed on the last two dims)
    # --------------------------------------------------------------
    # merge batch and head dimensions for a single bmm call
    left_mat = left_bh.view(B * hidden_dim, N, N)           # (B*H, N, N)
    right_mat = right_bh.view(B * hidden_dim, N, N)         # (B*H, N, N)

    # rightᵀ on the last two axes
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    out_bh = out_mat.view(B, hidden_dim, N, N)                # (B,H,N,N)

    # --------------------------------------------------------------
    # 7️⃣ Back to original layout and finish post‑processing
    # --------------------------------------------------------------
    out = out_bh.permute(0, 2, 3, 1).contiguous()            # (B,N,N,H) in fp16
    out = out.to(dtype)                                      # -> FP32

    # final LayerNorm over hidden_dim
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply output gate (broadcast on the pairwise axes)
    out = out * out_gate

    # final linear back to `dim`
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
