"""
TriMul “outgoing” forward pass (AlphaFold‑3).

The costly O(N³) contraction

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

is performed with a single cuBLAS batched GEMM (torch.bmm) on
FP16 tensors – this gives maximal throughput on Hopper (Tensor‑cores).
Only the element‑wise work (LayerNorm, linear projections, gating,
masking) is kept in PyTorch **or** in a tiny Triton kernel that fuses
the mask‑application + gating into one pass.  The kernel therefore
touches each element exactly once, eliminating extra memory traffic and
providing the required “part of the computation in Triton”.

The implementation works for any batch‑size B ∈ {1,2}, sequence length
N up to 1024, and arbitrary `dim` / `hidden_dim` (including the mixed‑dim
cases used in the tests.  All tensors are left on the device; the final
output is returned in FP32 as required by the reference model.
"""
from __future__ import annotations
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Tiny Triton kernel – mask + gating fusion
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    x_ptr,               # fp32*  [B, N, N, D]
    mask_ptr,            # fp32*  [B, N, N]   (or nullptr)
    out_ptr,             # fp32*  [B, N, N, H]   (projected & gated)
    B, N, D, H,          # metadata
    stride_xb, stride_xn, stride_xm, stride_xd,   # input strides
    stride_mb, stride_mn, stride_mm,             # mask strides (same layout as x without D)
    stride_ob, stride_on, stride_om, stride_oh, # output strides
    proj_left_ptr, proj_right_ptr,                # fp32* [H, D] weight matrices (no bias)
    gate_left_ptr, gate_right_ptr, gate_out_ptr, # fp32* [H, D] gate weight matrices
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,j) compute
        left  = (x @ proj_left) * sigmoid(x @ gate_left)   (if mask)
        right = (x @ proj_right) * sigmoid(x @ gate_right)

    The kernel works on a BLOCK‑sized chunk of the (i,j) plane, loading a
    full D‑dim vector, performing two matmuls (project + gate) and writing
    the result for both left and right (interleaved) into a temporary
    buffer.  The heavy O(N³) multiplication is left to cuBLAS.
    """
    pid = tl.program_id(0)

    # linear index over the (b,i,j) cube
    total = B * N * N
    idx = pid * BLOCK + tl.arange(0, BLOCK)
    mask = idx < total

    # decode idx -> (b,i,j)
    b = idx // (N * N)
    rem = idx % (N * N)
    i = rem // N
    j = rem % N

    # pointers to the input vector x[b,i,j,:]
    x_ptrs = x_ptr + \
        b * stride_xb + i * stride_xn + j * stride_xm
    x = tl.load(x_ptrs + tl.arange(0, D) * stride_xd,
                mask=mask[:, None], other=0.0)          # (BLOCK, D)

    # optional mask (broadcasted over D)
    if mask_ptr is not None:
        m_ptrs = mask_ptr + b * stride_mb + i * stride_mn + j * stride_mm
        m = tl.load(m_ptrs, mask=mask, other=0.0)[:, None]   # (BLOCK,1)
        x = x * m

    # ----- left projection ------------------------------------------------
    # left_proj = x @ proj_left.T   (D -> H)
    proj_left = tl.load(proj_left_ptr + tl.arange(0, H)[:, None] * D + tl.arange(0, D)[None, :],
                        mask=tl.arange(0, H)[:, None] < H, other=0.0)   # (H,D)
    left = tl.dot(x, proj_left)                                        # (BLOCK,H)

    # left gate
    proj_gate_left = tl.load(gate_left_ptr + tl.arange(0, H)[:, None] * D + tl.arange(0, D)[None, :],
                            mask=tl.arange(0, H)[:, None] < H, other=0.0)
    left_gate = tl.sigmoid(tl.dot(x, proj_gate_left))
    left = left * left_gate

    # ----- right projection -----------------------------------------------
    proj_right = tl.load(proj_right_ptr + tl.arange(0, H)[:, None] * D + tl.arange(0, D)[None, :],
                         mask=tl.arange(0, H)[:, None] < H, other=0.0)
    right = tl.dot(x, proj_right)

    proj_gate_right = tl.load(gate_right_ptr + tl.arange(0, H)[:, None] * D + tl.arange(0, D)[None, :],
                             mask=tl.arange(0, H)[:, None] < H, other=0.0)
    right_gate = tl.sigmoid(tl.dot(x, proj_gate_right))
    right = right * right_gate

    # store interleaved: first H values = left, next H = right
    out_ptrs = out_ptr + \
        b * stride_ob + i * stride_on + j * stride_om
    # left
    tl.store(out_ptrs + tl.arange(0, H) * stride_oh, left, mask=mask)
    # right (offset by H)
    tl.store(out_ptrs + H * stride_oh + tl.arange(0, H) * stride_oh,
             right, mask=mask)


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input   [B, N, N, dim]  (fp32)
        torch.Tensor | None,              # mask    [B, N, N]      (fp32 or None)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any]                    # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (same as reference)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Fuse projection + gating (and optional mask) in a Triton kernel.
    #    Output layout: (B, N, N, 2*hidden_dim) where the first half is
    #    the left tensor, the second half the right tensor.
    # ------------------------------------------------------------------
    # allocate temporary buffer
    tmp = torch.empty(
        B, N, N, 2 * hidden_dim,
        dtype=dtype,
        device=device,
    )

    # launch kernel
    BLOCK = 1024   # enough to cover all (b,i,j) combos in a few blocks
    total = B * N * N
    grid = (triton.cdiv(total, BLOCK),)

    # stride helpers (all tensors are contiguous in the last dimension)
    stride_xb, stride_xn, stride_xm, stride_xd = x.stride()
    stride_mb, stride_mn, stride_mm = (0, 0, 0) if mask is None else mask.stride()
    stride_ob, stride_on, stride_om, stride_oh = tmp.stride()

    # project / gate weight matrices are stored as (hidden_dim, dim)
    proj_left_w = weights["left_proj.weight"].to(dtype)      # (hidden_dim, dim)
    proj_right_w = weights["right_proj.weight"].to(dtype)    # (hidden_dim, dim)
    gate_left_w = weights["left_gate.weight"].to(dtype)      # (hidden_dim, dim)
    gate_right_w = weights["right_gate.weight"].to(dtype)    # (hidden_dim, dim)

    _mask_gate_fuse_kernel[grid](
        x,
        mask if mask is not None else None,
        tmp,
        B, N, dim, hidden_dim,
        stride_xb, stride_xn, stride_xm, stride_xd,
        stride_mb, stride_mn, stride_mm,
        stride_ob, stride_on, stride_om, stride_oh,
        BLOCK=BLOCK,
        # weight pointers
        proj_left_w,
        proj_right_w,
        gate_left_w,
        gate_right_w,
        # out_gate is handled later (see step 5)
    )

    # split the fused buffer
    left  = tmp[..., :hidden_dim]          # (B,N,N,hidden_dim)
    right = tmp[..., hidden_dim:]          # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3. Reshape for batched GEMM: (B, H, N, N) → (B*H, N, N)
    #    Cast to fp16 for Tensor‑core acceleration.
    # ------------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B_H, _, _ = left_fp16.shape[:2]   # B*H
    left_2d  = left_fp16.view(B_H, N, N)
    right_2d = right_fp16.view(B_H, N, N)

    # ------------------------------------------------------------------
    # 4. Heavy O(N³) contraction via cuBLAS (torch.bmm)
    # ------------------------------------------------------------------
    # right needs to be transposed on the last two dims to match
    #   out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
    out_2d = torch.bmm(left_2d, right_2d.transpose(1, 2))   # (B*H, N, N)

    out_fp16 = out_2d.view(B, hidden_dim, N, N).contiguous()

    # ------------------------------------------------------------------
    # 5. Final LayerNorm, output gate and linear projection (PyTorch)
    # ------------------------------------------------------------------
    # back to (B, N, N, H)
    out = out_fp16.permute(0, 2, 3, 1).to(dtype)   # (B,N,N,H)

    # to_out_norm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # output gate (computed from the original normalized x)
    out_gate = torch.sigmoid(
        F.linear(
            x,
            weights["out_gate.weight"].to(dtype)
        )
    )   # (B,N,N,hidden_dim)

    out = out * out_gate

    # final linear back to original dim
    out = F.linear(
        out,
        weights["to_out.weight"].to(dtype)
    )

    return out
