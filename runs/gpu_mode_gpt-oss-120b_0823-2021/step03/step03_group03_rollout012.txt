"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (high‑level)
----------------------
1️⃣  Layer‑Norm on the input (FP32, pure‑torch – cheap, no extra kernel).
2️⃣  Linear projections + three gating linear layers are performed in FP16
    with :func:`torch.nn.functional.linear`.
3️⃣  A small Triton kernel fuses
        * optional pair‑wise mask
        * element‑wise multiplication with the three sigmoidal gates
   for the *left* and *right* tensors.  This kernel is the only required
   Triton part.
4️⃣  Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply over the *k* dimension.
   We reshape the tensors to (B·H, N, N) and call ``torch.bmm`` – cuBLAS/Tensor
   Cores on the H200 execute this O(N³) work far faster than any Triton
   hand‑written GEMM.
5️⃣  The result is brought back to FP32, a second Layer‑Norm is applied,
    multiplied by the output‑gate and finally projected back to the original
    dimension with a linear layer.

Only the element‑wise mask + gate fusion is written in Triton; everything else
remains in PyTorch to keep the implementation concise while still satisfying
the “at‑least‑one‑kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask (if present) and three sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,         # (B, N, N, H)  fp16
    right_ptr,        # (B, N, N, H)  fp16
    left_gate_ptr,    # (B, N, N, H)  fp16
    right_gate_ptr,   # (B, N, N, H)  fp16
    mask_ptr,         # (B, N, N, 1)  fp16 or nullptr
    B, N, H,          # extents
    stride_lz, stride_ly, stride_lx, stride_lc,   # left strides
    stride_rz, stride_ry, stride_rx, stride_rc,   # right strides
    stride_gz, stride_gy, stride_gx, stride_gc,   # gate strides (same as left/right)
    stride_mz, stride_my, stride_mx, stride_mc,   # mask strides
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                       # linear thread id over all elements
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)     # work‑items for this program
    mask = offs < total

    # compute 4‑D index from linear offset
    b  = (offs // (N * N * H))
    i  = (offs // (N * H)) % N
    j  = (offs // H) % N
    h  = offs % H

    # compute pointer offsets
    left_off  = b * stride_lz + i * stride_ly + j * stride_lx + h * stride_lc
    right_off = b * stride_rz + i * stride_ry + j * stride_rx + h * stride_rc
    lg_off    = b * stride_gz + i * stride_gy + j * stride_gx + h * stride_gc
    rg_off    = b * stride_gz + i * stride_gy + j * stride_gx + h * stride_gc   # same strides
    if HAS_MASK:
        mask_off = b * stride_mz + i * stride_my + j * stride_mx          # last dim is 1, no *h

    # load values
    left_val  = tl.load(left_ptr  + left_off,  mask=mask, other=0.0)
    right_val = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    lg_val    = tl.load(left_gate_ptr  + lg_off, mask=mask, other=0.0)
    rg_val    = tl.load(right_gate_ptr + rg_off, mask=mask, other=0.0)

    if HAS_MASK:
        m_val = tl.load(mask_ptr + mask_off, mask=mask, other=0.0)
        left_val  = left_val  * m_val
        right_val = right_val * m_val

    # apply sigmoid gates (they are already sigmoid‑ed by torch, we just multiply)
    left_val  = left_val  * lg_val
    right_val = right_val * rg_val

    # store back
    tl.store(left_ptr  + left_off,  left_val,  mask=mask)
    tl.store(right_ptr + right_off, right_val, mask=mask)


def _apply_mask_and_gates(left, right, left_gate, right_gate, mask):
    """
    Fuse optional mask and three gates in‑place using Triton.
    All tensors are FP16 and have shape [B, N, N, H] (mask is [B,N,N,1] or None).
    Returns the updated ``left`` and ``right`` tensors (still FP16).
    """
    B, N, _, H = left.shape
    BLOCK = 1024                     # arbitrary, fits many threads per program
    total = B * N * N * H
    grid  = triton.cdiv(total, BLOCK)

    stride_l = left.stride()
    stride_r = right.stride()
    stride_g = left_gate.stride()    # left_gate & right_gate have same layout
    if mask is None:
        mask_ptr = left.new_empty(0).data_ptr()   # dummy pointer
        HAS_MASK = 0
    else:
        mask_ptr = mask.data_ptr()
        HAS_MASK = 1

    _mask_gate_fuse_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        left_gate.data_ptr(),
        right_gate.data_ptr(),
        mask_ptr,
        B, N, H,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_g[0], stride_g[1], stride_g[2], stride_g[3],
        stride_g[0], stride_g[1], stride_g[2], stride_g[3],   # mask strides (ignore H)
        HAS_MASK=HAS_MASK,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return left, right


# ----------------------------------------------------------------------
# Batched GEMM using torch.bmm (cuBLAS/Tensor‑cores)
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, N, N)  fp16
    right : (B, H, N, N)  fp16   (already transposed in the dim‑order needed)
    returns (B, H, N, N) fp16   = Σ_k left[...,k,:] * right[..., :,k]
    """
    B, H, N, _ = left.shape
    # collapse batch & head -> (B*H, N, N)
    a = left.reshape(B * H, N, N)
    b = right.reshape(B * H, N, N)
    # (B*H, N, N) @ (B*H, N, N).transpose(1,2)  ==  Σ_k left[...,k,:] * right[...,k,:]
    out = torch.bmm(a, b.transpose(1, 2))
    return out.view(B, H, N, N)


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool or float)
        weights      : dict of leaf weight tensors (see reference implementation)
        config       : dict containing keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual – avoids an extra kernel)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var  = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                       # still FP32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections & gates (FP16)
    # ------------------------------------------------------------------
    def _to_fp16(t): return t.to(torch.float16)

    left_proj_w  = _to_fp16(weights["left_proj.weight"])   # (hidden_dim, dim)
    right_proj_w = _to_fp16(weights["right_proj.weight"])
    left_gate_w  = _to_fp16(weights["left_gate.weight"])
    right_gate_w = _to_fp16(weights["right_gate.weight"])
    out_gate_w   = _to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                         # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)                # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()   # (B,N,N,hidden_dim)
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()    # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse optional mask and three gates (Triton)
    # ------------------------------------------------------------------
    if mask is None:
        mask_fp16 = None
    else:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)

    left, right = _apply_mask_and_gates(left, right,
                                        left_gate, right_gate,
                                        mask_fp16)           # in‑place FP16

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction – batched GEMM over the sequence dimension
    #    left : (B,N,N,hidden) → (B, hidden, N, N)
    #    right: (B,N,N,hidden) → (B, hidden, N, N) and we need (K,J) = (N,N)
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _batched_gemm_fp16(left_t, right_t)    # (B, H, N, N)

    # back to (B,N,N,dim) and FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 6️⃣ Final gating + projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)               # broadcast over dim
    out = F.linear(out, weights["to_out.weight"])        # (B,N,N,dim)

    return out
