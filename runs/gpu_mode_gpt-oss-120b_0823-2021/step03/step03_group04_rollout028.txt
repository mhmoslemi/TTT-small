"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm (GPU‑accelerated parts)
---------------------------------
1️⃣  Layer‑norm the input (B,N,N,C)   – PyTorch (FP32)
2️⃣  Linear projections to hidden dim H (left, right) and three gates – PyTorch (FP32)
3️⃣  Element‑wise fusion of mask (optional) and gate sigmoids with a tiny Triton
    kernel.  All tensors are cast to FP16 for the upcoming GEMM.
4️⃣  Batched matrix‑multiply (B*H,N,N) × (B*H,N,N)ᵀ  – torch.bmm which uses
    cuBLAS/Tensor‑cores (FP16→FP16, accumulation in FP32).
5️⃣  Hidden‑dim Layer‑norm, out‑gate multiplication and final linear projection
    back to C – PyTorch (FP32)

Only step 3 is performed in Triton; everything else stays in PyTorch so we
preserve numerical stability and keep the implementation simple while still
meeting the “some‑kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – fuse mask, left‑gate and right‑gate
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    # input tensors (FP16)
    left_ptr, right_ptr,                # (B,N,N,H)
    left_gate_ptr, right_gate_ptr,      # (B,N,N,H)
    mask_ptr,                           # (B,N,N,1) or nullptr
    # output tensors (FP16)
    out_left_ptr, out_right_ptr,        # (B,N,N,H)
    # shape & strides
    B, N, H,
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_m0, stride_m1, stride_m2,   # mask has no hidden dim
    # compile‑time constants
    HAS_MASK: tl.constexpr,
    BLOCK:    tl.constexpr,
):
    """
    For each element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    mask is either 1.0 (no mask) or a scalar broadcast over H.
    The kernel processes a 1‑D grid of size ceil(B*N*N*H / BLOCK).
    """
    pid   = tl.program_id(0)
    offs  = pid * BLOCK + tl.arange(0, BLOCK)               # linear ids
    total = B * N * N * H
    mask  = offs < total

    # ----- decode linear offset to (b,i,j,h) -------------------------
    tmp = offs
    h   = tmp % H
    tmp = tmp // H
    j   = tmp % N
    tmp = tmp // N
    i   = tmp % N
    b   = tmp // N

    # ----- load left / right ----------------------------------------
    idx = (b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3)
    left_val  = tl.load(left_ptr  + idx, mask=mask, other=0.0)
    right_val = tl.load(right_ptr + idx, mask=mask, other=0.0)

    # ----- load gates ------------------------------------------------
    idx_gate = (b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + h * stride_lg3)
    left_gate_val  = tl.load(left_gate_ptr  + idx_gate, mask=mask, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + idx_gate, mask=mask, other=0.0)

    # ----- load mask (if present) -----------------------------------
    if HAS_MASK:
        idx_mask = (b * stride_m0 + i * stride_m1 + j * stride_m2)
        m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)
    else:
        m = 1.0

    # ----- fuse ------------------------------------------------------
    out_l = left_val  * left_gate_val  * m
    out_r = right_val * right_gate_val * m

    # ----- store ------------------------------------------------------
    tl.store(out_left_ptr  + idx, out_l, mask=mask)
    tl.store(out_right_ptr + idx, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host wrapper
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input   (B,N,N,C)  float32
        torch.Tensor | None,        # mask    (B,N,N)    bool / float32 / None
        Dict[str, torch.Tensor],    # weights (all float32)
        Dict[str, Any],             # config  {"dim":C, "hidden_dim":H}
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input, mask, weights, config)
        * input  – (B, N, N, C)      torch.float32
        * mask   – (B, N, N) or None
        * weights – dict with the 9 weight tensors from the reference model
        * config  – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor
        (B, N, N, C) float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype  = torch.float32

    B, N, _, C = x.shape
    H = config["hidden_dim"]          # hidden dimension (c_z)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (FP32 → FP16 later)
    # ------------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))      # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate= torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate  = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse mask & gates – tiny Triton kernel
    # ------------------------------------------------------------------
    # Cast everything to FP16 (Tensor‑core friendly)
    left_fp16      = left.to(torch.float16).contiguous()
    right_fp16     = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate.to(torch.float16).contiguous()
    right_gate_fp16= right_gate.to(torch.float16).contiguous()

    if mask is None:
        # dummy mask tensor – will never be accessed because HAS_MASK=False
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask  = False
    else:
        # mask shape (B,N,N) → (B,N,N,1) so it broadcasts over H
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        has_mask  = True

    # Allocate output buffers
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elements = B * N * N * H
    BLOCK = 1024                     # 1‑K threads gives good occupancy on H200
    grid  = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fusion_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        # strides for left/right (identical layout)
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        # strides for right (same as left, but we still pass them for completeness)
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        # strides for gates (same layout)
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        # mask strides (no hidden dim)
        mask_fp16.stride(0) if has_mask else 0,
        mask_fp16.stride(1) if has_mask else 0,
        mask_fp16.stride(2) if has_mask else 0,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM – (B*H, N, N) @ (B*H, N, N)^T
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)  (contiguous layout required by cuBLAS)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)

    # FP16 GEMM, accumulation in FP32 (torch.bmm automatically does that)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))          # (B*H,N,N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    out = out.to(dtype)  # back to FP32 for the following LayerNorm

    # ------------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                     # broadcast over (N,N)
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
