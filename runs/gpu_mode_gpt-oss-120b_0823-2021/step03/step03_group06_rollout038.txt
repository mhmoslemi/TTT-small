"""
TriMul “outgoing” kernel (AlphaFold‑3).

The expensive contraction

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

is implemented as a batched GEMM for every head `h`.  All cheap
operations (LayerNorm, linear projections, masking, gating and the final
normalisation) stay in PyTorch.  The core O(N³) work is performed
inside a single Triton kernel that tiles the output matrix (M×N) and
loops over the K‑dimension in blocks.  We use FP16 arithmetic with an
FP32 accumulator (Tensor‑Core friendly) and write the result back as
FP16; the surrounding Python code casts back to FP32 for the remaining
ops.

The kernel works for any batch size B∈{1,2}, any sequence length N
(<=1024), and any hidden dimension H (the “head” dimension).  It is
compatible with both masked and un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM for a single (batch * head) slice
# -------------------------------------------------------------------------
@triton.jit
def _trimul_batched_gemm_kernel(
    left_ptr,          # const fp16 *   [BH, N, N]   (row‑major)
    right_ptr,         # const fp16 *   [BH, N, N]   (row‑major, will be read transposed)
    out_ptr,           #       fp16 *   [BH, N, N]
    BH, N,             # problem sizes
    stride_lbh, stride_lbn, stride_lbk,   # strides for left  (BH,N,N)
    stride_rbh, stride_rbn, stride_rbk,   # strides for right (BH,N,N)
    stride_obh, stride_obn, stride_obk,   # strides for out   (BH,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a tile (BLOCK_M × BLOCK_N) of

        out[bh, i, j] = Σ_k left[bh, i, k] * right[bh, j, k]

    i.e. left is multiplied with the *transpose* of right.
    One program instance processes a single (bh, tile_m, tile_n) triplet.
    """
    pid = tl.program_id(0)                     # linear index over (bh, tile_m, tile_n)

    # -----------------------------------------------------------------
    # decode pid → (bh, tile_m, tile_n)
    # -----------------------------------------------------------------
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    tile_n = pid % num_tiles_n
    pid //= num_tiles_n
    tile_m = pid % num_tiles_m
    pid //= num_tiles_m
    bh = pid                                   # 0 ≤ bh < BH

    # -----------------------------------------------------------------
    # offsets for the current tile
    # -----------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32 (Tensor‑Core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # loop over K dimension in BLOCK_K chunks
    # -----------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[bh, i, k]  → shape (M, K)
        l_ptr = left_ptr + bh * stride_lbh \
                + offs_m[:, None] * stride_lbn \
                + offs_k[None, :] * stride_lbk
        l = tl.load(l_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[bh, j, k] needs to be read as (K, N) because we multiply
        # left[i,k] * right[j,k] = left[i,k] * (rightᵀ)[k,j]
        r_ptr = right_ptr + bh * stride_rbh \
                + offs_k[:, None] * stride_rbk \
                + offs_n[None, :] * stride_rbn
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # GEMM: (M,K) @ (K,N) → (M,N)
        acc += tl.dot(l, r)

    # -----------------------------------------------------------------
    # write back FP16 result
    # -----------------------------------------------------------------
    out_ptrs = out_ptr + bh * stride_obh \
               + offs_m[:, None] * stride_obn \
               + offs_n[None, :] * stride_obk
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input   : [B, N, N, dim]   (float32)
        torch.Tensor,                # mask    : [B, N, N]       (bool/float) or None
        Dict[str, torch.Tensor],     # weights : model parameters (float32)
        Dict[str, Any],              # config  : {"dim":..., "hidden_dim":...}
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Returns
    -------
    out : torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]   # also called H in the paper

    # -----------------------------------------------------------------
    # 1️⃣  LayerNorm + Linear projections   (cheap – keep in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )   # (B, N, N, dim)

    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B, N, N, H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 2️⃣  Optional mask
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # (B, N, N, 1)
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 3️⃣  Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4️⃣  Rearrange to (B·H, N, N) and cast to fp16 for the kernel
    # -----------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → reshape to (BH, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left.view(B * hidden_dim, N, N).to(torch.float16)
    right_fp16 = right.view(B * hidden_dim, N, N).to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)          # (BH, N, N), fp16

    # -----------------------------------------------------------------
    # 5️⃣  Launch Triton kernel
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    BH = B * hidden_dim
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_tiles = BH * grid_m * grid_n
    grid = (total_tiles,)

    _trimul_batched_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        BH,
        N,
        # strides for left  (BH, N, N)
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        # strides for right (BH, N, N)
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        # strides for out   (BH, N, N)
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6️⃣  Restore original layout and apply post‑processing
    # -----------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)
    out = out.to(dtype)   # back to FP32 for the remaining ops

    # to_out_norm (LayerNorm over hidden_dim)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply output gate (broadcast over the N×N matrix)
    out = out * out_gate

    # final linear back to original dim
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
