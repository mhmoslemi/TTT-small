"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm
---------
1. Layer‑norm the input (B,N,N,C) on the host (torch.nn.functional.layer_norm).
2. Linear projections (left, right, left_gate, right_gate, out_gate) – still on the host.
3. Fuse three element‑wise ops (mask·gate·proj) into a single Triton kernel.
   The kernel works on the flattened index (b,i,j,h) and writes two fused tensors:
   left_fused = left * left_gate * mask
   right_fused = right * right_gate * mask
   (mask is optional).
   All tensors are cast to fp16 before the kernel to exploit Tensor‑cores later.
4. Batched GEMM using torch.bmm (cuBLAS / Tensor‑core) on the fused tensors:
   left_fused  → (B,H,N,N) → (B*H,N,N)
   right_fused → (B,H,N,N) → (B*H,N,N)
   out_mat = left_mat @ right_mat.transpose(1,2)        # (B*H,N,N)
5. Reshape back to (B,N,N,H) and cast to fp32.
6. Hidden‑dim layer‑norm, gating with out_gate, final linear projection back to C
   – all performed with PyTorch ops.

Only the element‑wise fusion (step 3) is off‑loaded to Triton, satisfying the
“some‑kernel” requirement while keeping the dominant O(N³) work on cuBLAS for
maximum speed on Hopper (H200).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: apply mask (if present) and per‑element gates to left/right
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # fp16* (B,N,N,H)
    right_ptr,              # fp16*
    left_gate_ptr,          # fp16* (B,N,N,H)
    right_gate_ptr,         # fp16*
    mask_ptr,               # fp16* (B,N,N) or nullptr
    out_left_ptr,           # fp16* (B,N,N,H) output
    out_right_ptr,          # fp16* (B,N,N,H) output
    B, N, H,                # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_m0, stride_m1, stride_m2,
    HAS_MASK: tl.constexpr,     # 0 / 1
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    The kernel is launched on a 1‑D grid covering B*N*N*H elements.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    total = B * N * N * H
    mask = offs < total

    # ---- decode linear index -> (b,i,j,h) (contiguous layout) ----
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ---- load left / right projections ----
    idx = (
        b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    )
    l_val = tl.load(left_ptr + idx, mask=mask, other=tl.float16(0.0))
    r_val = tl.load(right_ptr + idx, mask=mask, other=tl.float16(0.0))

    # ---- load gates ----
    idx_gate = (
        b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + h * stride_lg3
    )
    l_gate = tl.load(left_gate_ptr + idx_gate, mask=mask, other=tl.float16(0.0))
    r_gate = tl.load(right_gate_ptr + idx_gate, mask=mask, other=tl.float16(0.0))

    # ---- mask (broadcast over H) ----
    if HAS_MASK:
        idx_mask = b * stride_m0 + i * stride_m1 + j * stride_m2
        m = tl.load(mask_ptr + idx_mask, mask=mask, other=tl.float16(1.0))
    else:
        m = tl.float16(1.0)

    # ---- fused write ----
    tl.store(out_left_ptr + idx,  l_val * l_gate * m, mask=mask)
    tl.store(out_right_ptr + idx, r_val * r_gate * m, mask=mask)


# ----------------------------------------------------------------------
# Host wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : (x, mask, weights, config)
        x        – (B, N, N, C)    torch.float32
        mask     – (B, N, N)      torch.bool / torch.float32 / None
        weights  – dict of learned parameters (float32)
        config   – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor  (B, N, N, C)  float32
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, C = x.shape
    H = config["hidden_dim"]
    dtype = torch.float32

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (still in FP32)
    # --------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate= torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate  = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N,H)

    # --------------------------------------------------------------
    # 3️⃣ Cast to fp16 and fuse mask + gates with Triton
    # --------------------------------------------------------------
    # Make tensors contiguous & fp16 for the kernel / subsequent GEMM
    left_fp16      = left.to(torch.float16).contiguous()
    right_fp16     = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate.to(torch.float16).contiguous()
    right_gate_fp16= right_gate.to(torch.float16).contiguous()

    if mask is None:
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)   # dummy
        has_mask = False
    else:
        # mask shape (B,N,N) → (B,N,N) (no H dim); we keep it fp16
        mask_fp16 = mask.to(torch.float16).contiguous()
        has_mask = True

    # allocate output buffers
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elements = B * N * N * H
    BLOCK = 1024                     # good occupancy for Hopper
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        # mask strides (no H dim)
        mask_fp16.stride(0) if has_mask else 0,
        mask_fp16.stride(1) if has_mask else 0,
        mask_fp16.stride(2) if has_mask else 0,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (fp16 inputs → fp16 output, acc in fp32)
    # --------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N) -> (B*H,N,N)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))          # (B*H,N,N)

    # reshape back to (B,N,N,H) and convert to FP32 for the next norm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(dtype)

    # --------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, gating with out_gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    # out_gate has shape (B,N,N,H) already
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
