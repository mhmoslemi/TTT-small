"""
TriMul “outgoing” forward pass (no gradients).

The O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a batched GEMM with a tiny Triton kernel that
loads a tile of the left and right tensors, accumulates in FP32
and writes the result in FP16.  All other operations (layer‑norms,
linear projections, sigmoid gates) stay in PyTorch – they are
memory‑bound and inexpensive compared with the cubic term.

The kernel works on the layout (B, H, I, K) × (B, H, K, J) → (B, H, I, J)
where H = hidden_dim, I = J = seq_len.
The heavy contraction therefore runs at ~Tensor‑Core speed on the
Hopper H200 while keeping the overall implementation simple.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (I×K)·(K×J) → (I×J)   (fp16 input, fp32 acc, fp16 out)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,            # left  tensor   (Z, I, K)   fp16
    b_ptr,            # right tensor   (Z, K, J)   fp16
    c_ptr,            # output tensor  (Z, I, J)   fp16
    M, N, K,          # matrix dimensions
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM (Z batches) using tiled loads and fp32 accumulation."""
    pid_z = tl.program_id(0)   # batch index (flattened B*H)
    pid_m = tl.program_id(1)   # tile‑row index
    pid_n = tl.program_id(2)   # tile‑col index

    # --------------------------------------------------------------
    # 1) compute tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 2) loop over K dimension in tiles
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile (BLOCK_M × BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile (BLOCK_K × BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP32 accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # 3) write result in FP16
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_batch_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous
    returns (B, H, I, J)   fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten the batch×head dimension – the kernel expects a 3‑D tensor
    Z = B * H
    a = left.reshape(Z, I, K)
    b = right.reshape(Z, K, J)
    c = out.reshape(Z, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling parameters – chosen empirically for Hopper (FP16/Tensor‑Core friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                           # program_id(0) – batch
        triton.cdiv(I, BLOCK_M),    # program_id(1) – row tiles
        triton.cdiv(J, BLOCK_N),    # program_id(2) – col tiles
    )

    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # sufficient for the chosen tile sizes
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor [B, N, N] (float32/bool) or None
        - weights      : dict of torch.Tensor parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep in FP32, cheap)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual LayerNorm to avoid Module overhead
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                     # still FP32

    # ------------------------------------------------------------------
    # 2) Linear projections (cast once to FP16)
    # ------------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16, non_blocking=True)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = weights["to_out_norm.weight"]   # stays FP32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # FP16 copy of the normalized tensor – all heavy ops use FP16
    x_fp16 = x.to(torch.float16)

    # projections → (B, N, N, hidden)
    left  = F.linear(x_fp16, left_proj_w)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) – still in FP16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction – batched GEMM via Triton
    #    reshape to (B, H, I, K) and (B, H, K, J)
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _triton_batch_gemm(left_perm, right_perm)   # (B, H, I, J) fp16

    # back to (B, N, N, hidden)
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm
    # ------------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7) Final gating + linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # same gate as in reference
    out = F.linear(out, to_out_w)                  # (B, N, N, dim)   FP32

    return out
