"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (high‑level)
----------------------
1️⃣  Layer‑norm the 4‑D input (FP32, manual – cheap, no extra kernel).
2️⃣  Four linear layers (left/right projections + three gates) are performed in
    FP16 for speed.  The input is cast once to FP16.
3️⃣  Optional pair‑wise mask is applied to the left/right tensors.
4️⃣  Gating (sigmoid) is applied.
5️⃣  Core contraction  
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]                (1)
    is a batched matrix‑multiply over the sequence dimension **k**.
    We launch a small Triton kernel that treats each (batch, head) pair as an
    independent GEMM (I×K)·(K×J) → (I×J).  The kernel works in FP16, accumulates
    in FP32 and writes FP16.  Tile sizes are tuned for Hopper/SM‑90.
6️⃣  Convert the result back to FP32, apply a second LayerNorm, multiply by the
    output‑gate and finish with a linear projection back to the original dim.

Only the cubic contraction (step 5) is executed on Triton; everything else
remains in PyTorch.  This satisfies the “use at least one kernel” rule while
keeping the implementation simple and fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched GEMM (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,            # left tensor  (B*H, I, K)
    b_ptr,            # right tensor (B*H, K, J)
    c_ptr,            # output       (B*H, I, J)
    M, N, K,          # logical extents
    stride_az, stride_am, stride_ak,   # strides of a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides of b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides of c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the outgoing TriMul contraction.

    Each program processes a tile (BLOCK_M×BLOCK_N) of the output matrix for a
    particular (batch*head) slice.  Accumulation is performed in FP32.
    """
    pid_z = tl.program_id(2)          # index over (batch * head)
    pid_m = tl.program_id(0)          # tile‑row index (i)
    pid_n = tl.program_id(1)          # tile‑col index (j)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M×K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (K×N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16 dot → FP32 accumulation
        acc += tl.dot(a, b)

    # Write back (convert to FP16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication used by TriMul.

    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous
    returns: (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # Collapse (B,H) into one program dimension
    a = left.view(B * H, I, K)
    b = right.view(B * H, K, J)
    c = out.view(B * H, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes chosen for Hopper (SM‑90) – good occupancy & shared‑mem usage
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),          # tiles along I (i‑dimension)
        triton.cdiv(J, BLOCK_N),          # tiles along J (j‑dimension)
        B * H,                            # one program per (batch, head)
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # tuned for fp16 on SM‑90
    )
    return out


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool or float)
        weights      : dict of leaf weight tensors (see reference implementation)
        config       : dict containing keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    # mean / var over last dimension (dim)
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still FP32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections & gates (cast to FP16)
    # ------------------------------------------------------------------
    # Helper to promote weights to FP16 once
    def _to_fp16(t): return t.to(torch.float16)

    left_proj_w   = _to_fp16(weights["left_proj.weight"])    # (hidden_dim, dim)
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])
    to_out_w      = _to_fp16(weights["to_out.weight"])      # (dim, hidden_dim)

    x_fp16 = x.to(torch.float16)  # (B,N,N,dim)

    # Linear layers (bias=False in the reference implementation)
    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction – Triton batched GEMM
    # ------------------------------------------------------------------
    # left : (B,N,N,hidden_dim)  -> (B, hidden_dim, N, N)
    # right: (B,N,N,hidden_dim)  -> (B, hidden_dim, N, N) and we need K↔J swapped
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # Batched GEMM in FP16 via Triton (produces (B, H, N, N) fp16)
    out_fp16 = _triton_batched_gemm(left_t, right_t)

    # Back to (B,N,N,hidden_dim) and to FP32 for the following LayerNorm
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32, manual)
    # ------------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)            # broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])    # (B,N,N,dim)

    return out
```

**Explanation of the design choices**

* **LayerNorms** – Implemented manually in FP32 (no extra kernel) because the
  per‑element cost is negligible compared with the cubic contraction.

* **Linear layers & gating** – Performed with `torch.nn.functional.linear` on
  FP16 tensors; this gives Tensor‑Core speed while keeping the code simple.

* **Mask handling** – The mask is optional; when present it is cast to FP16
  and broadcasted over the hidden dimension before the contraction.

* **Core contraction** – The only heavy O(N³) work.  By reshaping to
  `(B, H, I, K)` and `(B, H, K, J)` we treat each `(batch, head)` pair as an
  independent GEMM.  The Triton kernel tiles the output matrix with
  `BLOCK_M = BLOCK_N = 64` and the reduction dimension with `BLOCK_K = 32`,
  which fits well in shared memory on Hopper and allows 4 warps (default for
  fp16) to achieve near‑Tensor‑Core throughput.

* **Output processing** – After the GEMM we convert back to FP32, apply the
  second LayerNorm, multiply by the output‑gate, and finish with a final FP32
  linear projection – exactly matching the reference PyTorch implementation.

The kernel satisfies the requirement of “using at least one Triton kernel”,
keeps the implementation correct for all listed configurations, and is
optimised for the target H200 hardware.
