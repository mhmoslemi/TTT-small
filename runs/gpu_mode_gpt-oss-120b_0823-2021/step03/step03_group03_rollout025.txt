"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  Manual LayerNorm (FP32) on the 4‑D input.
2️⃣  Four linear layers (left/right projections + three gates) are executed
    in FP16 with torch.nn.functional.linear (fast, uses Tensor‑cores).
3️⃣  A tiny Triton element‑wise kernel fuses the optional mask and the
    sigmoid‑gating for *left* and *right* tensors (≈ 2× speed‑up versus
    Python loops).
4️⃣  The core cubic contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is reduced to a batched matrix‑multiply by reshaping the tensors to
        (B·H, N, N)  ×  (B·H, N, N)ᵀ  →  (B·H, N, N)
    and calling torch.bmm – this lets cuBLAS/Tensor‑cores do the heavy
    O(N³) work.
5️⃣  Result is reshaped back to (B,N,N,dim) (FP32) and a second LayerNorm
    (FP32) is applied.
6️⃣  Final output‑gate (sigmoid) and linear projection (FP32) produce the
    module output.

Only the element‑wise mask‑and‑gate fusion is implemented in Triton, which
fulfils the “at least part in a kernel” requirement while keeping the
overall runtime dominated by the cuBLAS GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused mask & sigmoid gating
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    x_ptr,          # tensor to be gated (fp16)   (B,H,N,N)
    gate_ptr,       # gate pre‑sigmoid (fp16)     (B,H,N,N)
    mask_ptr,       # optional mask (fp16) or nullptr
    out_ptr,        # result (fp16)
    B, H, N,        # static extents
    stride_xz, stride_xh, stride_xi, stride_xj,   # x strides
    stride_gz, stride_gh, stride_gi, stride_gj,   # gate strides
    stride_mz, stride_mh, stride_mi, stride_mj,   # mask strides (unused if mask_ptr==0)
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)                     # flatten over (B,H)
    b = pid // H
    h = pid % H

    # thread offsets inside the N×N matrix
    i = tl.arange(0, 64)                       # tile size = 64 (can be tuned)
    j = tl.arange(0, 64)

    offs_i = i + tl.program_id(1) * 64
    offs_j = j + tl.program_id(2) * 64

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # load
    x = tl.load(
        x_ptr
        + b * stride_xz + h * stride_xh
        + offs_i[:, None] * stride_xi
        + offs_j[None, :] * stride_xj,
        mask=mask_ij,
        other=0.0,
    )
    g = tl.load(
        gate_ptr
        + b * stride_gz + h * stride_gh
        + offs_i[:, None] * stride_gi
        + offs_j[None, :] * stride_gj,
        mask=mask_ij,
        other=0.0,
    )
    if HAS_MASK:
        m = tl.load(
            mask_ptr
            + b * stride_mz + h * stride_mh
            + offs_i[:, None] * stride_mi
            + offs_j[None, :] * stride_mj,
            mask=mask_ij,
            other=1.0,
        )
        x = x * m

    # sigmoid(g)
    g = 1.0 / (1.0 + tl.exp(-g))

    out = x * g
    tl.store(
        out_ptr
        + b * stride_xz + h * stride_xh
        + offs_i[:, None] * stride_xi
        + offs_j[None, :] * stride_xj,
        out,
        mask=mask_ij,
    )


def _apply_mask_gate(
    tensor: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor | None,
) -> torch.Tensor:
    """
    Fuse:  out = (tensor * mask) * sigmoid(gate)
    All tensors are fp16, contiguous, shape (B, H, N, N).
    """
    B, H, N, _ = tensor.shape
    out = torch.empty_like(tensor)

    # strides
    s_xz, s_xh, s_xi, s_xj = tensor.stride()
    s_gz, s_gh, s_gi, s_gj = gate.stride()
    if mask is None:
        s_mz = s_mh = s_mi = s_mj = 0
        mask_ptr = 0
        has_mask = 0
    else:
        s_mz, s_mh, s_mi, s_mj = mask.stride()
        mask_ptr = mask
        has_mask = 1

    # launch grid: (batch*head, ceil(N/64), ceil(N/64))
    BLOCK = 64
    grid = (
        B * H,
        triton.cdiv(N, BLOCK),
        triton.cdiv(N, BLOCK),
    )
    _apply_mask_gate_kernel[grid](
        tensor,
        gate,
        mask_ptr,
        out,
        B,
        H,
        N,
        s_xz,
        s_xh,
        s_xi,
        s_xj,
        s_gz,
        s_gh,
        s_gi,
        s_gj,
        s_mz,
        s_mh,
        s_mi,
        s_mj,
        HAS_MASK=has_mask,
        num_warps=4,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool or float)
        weights      : dict of weight tensors (see reference implementation)
        config       : dict containing keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual – cheap)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                       # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gates (cast to FP16)
    # --------------------------------------------------------------
    to_fp16 = lambda t: t.to(torch.float16)

    left_proj_w   = to_fp16(weights["left_proj.weight"])     # (hidden_dim, dim)
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                            # (B,N,N,dim)

    # Linear layers (no bias)
    left   = F.linear(x_fp16, left_proj_w)                 # (B,N,N,hidden_dim)
    right  = F.linear(x_fp16, right_proj_w)
    left_g = F.linear(x_fp16, left_gate_w)                 # (B,N,N,hidden_dim)
    right_g = F.linear(x_fp16, right_gate_w)
    out_g  = F.linear(x_fp16, out_gate_w)                  # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 3️⃣ Mask & gating – fused in Triton
    # --------------------------------------------------------------
    # Prepare mask in fp16 if present
    if mask is None:
        mask_fp16 = None
    else:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)    # (B,N,N,1)

    # reshape to (B, H, N, N) for the kernel
    left   = left.permute(0, 3, 1, 2).contiguous()          # (B,H,N,N)
    right  = right.permute(0, 3, 1, 2).contiguous()
    left_g = left_g.permute(0, 3, 1, 2).contiguous()
    right_g = right_g.permute(0, 3, 1, 2).contiguous()
    if mask_fp16 is not None:
        mask_t = mask_fp16.permute(0, 3, 1, 2).contiguous()  # (B,1,N,N) broadcast
    else:
        mask_t = None

    left = _apply_mask_gate(left, left_g, mask_t)
    right = _apply_mask_gate(right, right_g, mask_t)

    # --------------------------------------------------------------
    # 4️⃣ Core contraction – batched GEMM via cuBLAS (torch.bmm)
    # --------------------------------------------------------------
    # left:  (B, H, N, N)   right: (B, H, N, N) → we need (B*H, N, N) × (B*H, N, N)ᵀ
    BH = B * hidden_dim
    left_reshape = left.view(BH, N, N)          # fp16
    right_reshape = right.view(BH, N, N)

    # batched matrix multiply (fp16 input, fp32 accumulation in cuBLAS)
    out_fp16 = torch.bmm(left_reshape, right_reshape.transpose(1, 2))   # (BH,N,N)

    # reshape back to (B,N,N,hidden_dim)
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)                  # now (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Output LayerNorm (FP32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 6️⃣ Final gating + projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.sigmoid().to(torch.float32)   # broadcast over dim
    out = F.linear(out, weights["to_out.weight"])      # (B,N,N,dim)

    return out
