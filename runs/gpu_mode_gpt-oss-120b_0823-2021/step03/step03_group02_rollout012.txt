"""
TriMul “outgoing” forward pass (no gradients).

The O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is carried out with a single batched matrix‑multiply (torch.bmm) that
leverages cuBLAS/Tensor‑cores on Hopper.  The only part that really
benefits from a custom Triton kernel is the element‑wise post‑linear
operations (masking + gate multiplication) which are fused into one
kernel to avoid an extra launch per tensor and to keep all heavy
memory traffic in fp16.

All other cheap operations (LayerNorm, linear layers, final
projection) stay in PyTorch.  The final result is returned in fp32 as
required by the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) and gate multiplication
#   left  = left * mask * left_gate
#   right = right * mask * right_gate
# Shapes: (B, N, N, H) as fp16, mask: (B, N, N, 1) fp16 (or all‑ones if None)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    left_ptr,          # fp16  (B,N,N,H)
    right_ptr,         # fp16  (B,N,N,H)
    mask_ptr,          # fp16  (B,N,N,1) – may be NULL (all‑ones)
    left_gate_ptr,     # fp16  (B,N,N,H)
    right_gate_ptr,    # fp16  (B,N,N,H)
    out_left_ptr,      # fp16  (B,N,N,H) – written in‑place
    out_right_ptr,     # fp16  (B,N,N,H) – written in‑place
    B, N, H,
    stride_lbz, stride_lby, stride_lbx, stride_lbh,   # left strides
    stride_rbz, stride_rby, stride_rbx, stride_rbh,   # right strides
    stride_mbz, stride_mby, stride_mbx,               # mask strides
    stride_gbz, stride_gby, stride_gbx, stride_gbh,   # gate strides
    BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_n = tl.program_id(0)          # tile along N dimension
    pid_b = tl.program_id(1)          # batch dimension

    # offsets for the N‑N plane (i and j are both N)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # we treat the two N axes identically: we will load a BLOCK_N × BLOCK_H
    # tile for each row i, applying it to both left and right tensors.
    # The kernel is launched with grid = (ceil(N/BLOCK_N), B)
    mask_is_null = mask_ptr == 0

    # Masks for boundaries
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # loop over the inner H dimension (hidden) in chunks of BLOCK_H
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # ------- load left, right, gates, mask (if any) ----------------
        left = tl.load(
            left_ptr
            + pid_b * stride_lbz
            + offs_n[:, None] * stride_lby
            + offs_n[None, :] * stride_lbx
            + offs_h[None, None] * stride_lbh,
            mask=mask_n[:, None] & mask_n[None, :] & mask_h[None, None],
            other=0.0,
        )
        right = tl.load(
            right_ptr
            + pid_b * stride_rbz
            + offs_n[:, None] * stride_rby
            + offs_n[None, :] * stride_rbx
            + offs_h[None, None] * stride_rbh,
            mask=mask_n[:, None] & mask_n[None, :] & mask_h[None, None],
            other=0.0,
        )
        left_gate = tl.load(
            left_gate_ptr
            + pid_b * stride_gbz
            + offs_n[:, None] * stride_gby
            + offs_n[None, :] * stride_gbx
            + offs_h[None, None] * stride_gbh,
            mask=mask_n[:, None] & mask_n[None, :] & mask_h[None, None],
            other=0.0,
        )
        right_gate = tl.load(
            right_gate_ptr
            + pid_b * stride_gbz
            + offs_n[:, None] * stride_gby
            + offs_n[None, :] * stride_gbx
            + offs_h[None, None] * stride_gbh,
            mask=mask_n[:, None] & mask_n[None, :] & mask_h[None, None],
            other=0.0,
        )
        # mask is broadcasted on the hidden axis (shape B,N,N,1)
        if not mask_is_null:
            m = tl.load(
                mask_ptr
                + pid_b * stride_mbz
                + offs_n[:, None] * stride_mby
                + offs_n[None, :] * stride_mbx,
                mask=mask_n[:, None] & mask_n[None, :],
                other=1.0,
            )
        else:
            m = tl.full((BLOCK_N, BLOCK_N), 1.0, dtype=tl.float16)

        # ------- fuse ----------------------------------------------------
        left  = left * m[:, :, None] * left_gate
        right = right * m[:, :, None] * right_gate

        # ------- store ---------------------------------------------------
        tl.store(
            out_left_ptr
            + pid_b * stride_lbz
            + offs_n[:, None] * stride_lby
            + offs_n[None, :] * stride_lbx
            + offs_h[None, None] * stride_lbh,
            left,
            mask=mask_n[:, None] & mask_n[None, :] & mask_h[None, None],
        )
        tl.store(
            out_right_ptr
            + pid_b * stride_rbz
            + offs_n[:, None] * stride_rby
            + offs_n[None, :] * stride_rbx
            + offs_h[None, None] * stride_rbh,
            right,
            mask=mask_n[:, None] & mask_n[None, :] & mask_h[None, None],
        )


def fuse_mask_and_gate(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor | None,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
) -> tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse: left = left * mask * left_gate
          right = right * mask * right_gate
    All tensors are fp16 and have shape (B, N, N, H).  ``mask`` may be ``None``,
    in which case it is treated as all‑ones.
    Returns the two fused tensors (still fp16).
    """
    B, N, _, H = left.shape
    BLOCK_N = 64
    BLOCK_H = 32

    out_left = torch.empty_like(left)
    out_right = torch.empty_like(right)

    # strides (in elements, not bytes)
    stride_l = left.stride()
    stride_r = right.stride()
    stride_g = left_gate.stride()
    stride_m = (0, 0, 0, 0) if mask is None else mask.stride()

    grid = (triton.cdiv(N, BLOCK_N), B)

    mask_ptr = 0 if mask is None else mask.data_ptr()
    mask_gate_fuse_kernel[
        grid
    ](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask_ptr,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        out_left_ptr=out_left,
        out_right_ptr=out_right,
        B=B,
        N=N,
        H=H,
        stride_lbz=stride_l[0],
        stride_lby=stride_l[1],
        stride_lbx=stride_l[2],
        stride_lbh=stride_l[3],
        stride_rbz=stride_r[0],
        stride_rby=stride_r[1],
        stride_rbx=stride_r[2],
        stride_rbh=stride_r[3],
        stride_mbz=stride_m[0],
        stride_mby=stride_m[1],
        stride_mbx=stride_m[2],
        stride_gbz=stride_g[0],
        stride_gby=stride_g[1],
        stride_gbx=stride_g[2],
        stride_gbh=stride_g[3],
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out_left, out_right


# ----------------------------------------------------------------------
# Full forward (PyTorch for cheap ops, Triton for mask+gate fusion)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensor parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]   (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32 – cheap)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                      # still FP32

    # ------------------------------------------------------------------
    # 2) Linear projections (FP16)
    # ------------------------------------------------------------------
    def linear_fp16(inp, w):
        # w is stored as FP32 in the checkpoint; we cast once
        return F.linear(inp, w.to(torch.float16))

    x_fp16 = x.to(torch.float16)

    left = linear_fp16(x_fp16, weights["left_proj.weight"])   # (B,N,N,H)
    right = linear_fp16(x_fp16, weights["right_proj.weight"])

    # gating tensors (also FP16)
    left_gate = linear_fp16(x_fp16, weights["left_gate.weight"]).sigmoid()
    right_gate = linear_fp16(x_fp16, weights["right_gate.weight"]).sigmoid()
    out_gate = linear_fp16(x_fp16, weights["out_gate.weight"]).sigmoid()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3) Fuse mask & gate with Triton
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
    else:
        mask_fp16 = None

    left, right = fuse_mask_and_gate(left, right, mask_fp16, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 4) Core contraction – batched GEMM via cuBLAS (torch.bmm)
    #    left : (B, H, N, N)  -> (B*H, N, N)
    #    right: (B, H, N, N) -> (B*H, N, N) and we need its transpose on the
    #           last two axes (K dimension)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B, H, I, K = left_perm.shape
    _, _, K2, J = right_perm.shape
    assert I == J == N and K == K2

    left_mat = left_perm.view(B * H, I, K)          # (B*H, N, N)
    right_mat = right_perm.view(B * H, K, J)        # (B*H, N, N)

    # torch.bmm uses Tensor‑cores on Hopper for fp16 inputs
    out_mat = torch.bmm(left_mat, right_mat)       # (B*H, N, N)  fp16

    out = out_mat.view(B, H, I, J).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    out = out.to(torch.float32)                     # back to FP32 for LN

    # ------------------------------------------------------------------
    # 5) Output LayerNorm + final gating + projection
    # ------------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    out = out * out_gate.to(torch.float32)          # same gate as in reference
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
