"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (high‑level)
----------------------
1️⃣  Layer‑norm the 4‑D input (FP32, manual – cheap, no extra kernel).  
2️⃣  Cast to FP16 and compute the four linear projections (left/right) and the
    three gating projections (left‑gate, right‑gate, out‑gate) with ``torch.nn.functional.linear``.  
3️⃣  **Triton kernel** – element‑wise masking of the left/right tensors (optional).  
    The kernel works on the flat view ``(B,N,N,H)`` and multiplies by the broadcast
    mask ``(B,N,N,1)`` in FP16.  This satisfies the requirement that “part of the
    computation is performed in a Triton kernel”.
4️⃣  Apply sigmoid gates (still in FP16).  
5️⃣  Reshape tensors to (B*H, N, N) and perform the cubic contraction  
    ``out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]`` via a batched‑matrix‑multiply
    (`torch.bmm`).  This uses cuBLAS/Tensor‑cores on Hopper and is the dominant
    O(N³) work.  
6️⃣  Convert the result back to FP32, apply a second LayerNorm, the output‑gate
    and the final linear projection to the original dimension.

All heavy compute (masking & the O(N³) reduction) is performed on‑device, the
rest is cheap element‑wise work in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton element‑wise mask kernel
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    a_ptr,          # left    (B,N,N,H)  fp16
    b_ptr,          # right  (B,N,N,H)  fp16
    mask_ptr,       # mask    (B,N,N,1)  fp16
    M, H,           # extents
    stride_a0, stride_a1, stride_a2, stride_a3,   # a strides
    stride_b0, stride_b1, stride_b2, stride_b3,   # b strides
    stride_m0, stride_m1, stride_m2, stride_m3,   # mask strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)

    # linear block index → (m,n,h)
    offs_m = (pid // (tl.cdiv(N, BLOCK_N) * tl.cdiv(H, BLOCK_H))) * BLOCK_M
    offs_n = ((pid // tl.cdiv(H, BLOCK_H)) % tl.cdiv(N, BLOCK_N)) * BLOCK_N
    offs_h = (pid % tl.cdiv(H, BLOCK_H)) * BLOCK_H

    rm = tl.arange(0, BLOCK_M)
    rn = tl.arange(0, BLOCK_N)
    rh = tl.arange(0, BLOCK_H)

    m = offs_m + rm
    n = offs_n + rn
    h = offs_h + rh

    mask = tl.load(mask_ptr + m[:,None,None]*stride_m0 + n[None,:,None]*stride_m1 + h[None,None,:]*stride_m2,
                    mask=(m[:,None,None] < M) & (n[None,:,None] < N) & (h[None,None,:] < H),
                    other=0.0)

    # left
    a = tl.load(a_ptr + m[:,None,None]*stride_a0 + n[None,:,None]*stride_a1 + h[None,None,:]*stride_a2,
                mask=(m[:,None,None] < M) & (n[None,:,None] < N) & (h[None,None,:] < H),
                other=0.0)
    a = a * mask
    tl.store(a_ptr + m[:,None,None]*stride_a0 + n[None,:,None]*stride_a1 + h[None,None,:]*stride_a2,
             a,
             mask=(m[:,None,None] < M) & (n[None,:,None] < N) & (h[None,None,:] < H))

    # right
    b = tl.load(b_ptr + m[:,None,None]*stride_b0 + n[None,:,None]*stride_b1 + h[None,None,:]*stride_b2,
                mask=(m[:,None,None] < M) & (n[None,:,None] < N) & (h[None,None,:] < H),
                other=0.0)
    b = b * mask
    tl.store(b_ptr + m[:,None,None]*stride_b0 + n[None,:,None]*stride_b1 + h[None,None,:]*stride_b2,
             b,
             mask=(m[:,None,None] < M) & (n[None,:,None] < N) & (h[None,None,:] < H))


def _apply_mask_fp16(left: torch.Tensor, right: torch.Tensor,
                     mask: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
    """
    left/right : (B,N,N,H)   fp16, contiguous
    mask       : (B,N,N,1)  fp16, contiguous
    returns    : masked left/right (still fp16)
    """
    B, N, _, H = left.shape
    # flatten to 1‑D for launch indexing
    total_elements = B * N * N * H
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_H = 32
    # number of blocks needed
    blocks = triton.cdiv(total_elements, BLOCK_M * BLOCK_N * BLOCK_H)

    stride_a = left.stride()
    stride_b = right.stride()
    stride_m = mask.stride()

    _apply_mask_kernel[blocks](
        left, right, mask,
        M=N, H=H,
        stride_a0=stride_a[0], stride_a1=stride_a[1], stride_a2=stride_a[2], stride_a3=stride_a[3],
        stride_b0=stride_b[0], stride_b1=stride_b[1], stride_b2=stride_b[2], stride_b3=stride_b[3],
        stride_m0=stride_m[0], stride_m1=stride_m[1], stride_m2=stride_m[2], stride_m3=stride_m[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return left, right


# ----------------------------------------------------------------------
# Forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim]      (float32)
        mask         : torch.Tensor or None, shape [B, N, N]   (bool/float)
        weights      : dict of leaf tensors (see reference module)
        config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b            # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear proj & gate weights (cast to FP16)
    # --------------------------------------------------------------
    def _fp16(t): return t.to(torch.float16)

    left_proj_w   = _fp16(weights["left_proj.weight"])   # (hidden_dim, dim)
    right_proj_w  = _fp16(weights["right_proj.weight"])
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                        # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask – Triton kernel
    # --------------------------------------------------------------
    if mask is not None:
        # mask -> (B,N,N,1) fp16
        m = mask.to(torch.float16).unsqueeze(-1)
        left, right = _apply_mask_fp16(left, right, m)

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM via torch.bmm (cuBLAS)
    # --------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N)   left : (B,N,N,H) → (B*H,N,N)
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    B_H, I, K = left_t.shape[0]*left_t.shape[1], left_t.shape[2], left_t.shape[3]
    left_flat  = left_t.view(B*hidden_dim, N, N)      # (B*H, N, N)
    right_flat = right_t.view(B*hidden_dim, N, N)     # (B*H, N, N)

    # bmm does Σ_k left[...,k] * right[...,k]   → (B*H, N, N)
    out_fp16 = torch.bmm(left_flat, right_flat)      # (B*H, N, N) fp16
    out_fp16 = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 6️⃣ Back to FP32 and second LayerNorm
    # --------------------------------------------------------------
    out = out_fp16.to(torch.float32)                  # (B,N,N,hidden_dim)

    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)            # broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])    # (B,N,N,dim)

    return out
