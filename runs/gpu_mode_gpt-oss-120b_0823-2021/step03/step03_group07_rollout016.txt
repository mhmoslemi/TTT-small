"""
TriMul “outgoing” forward pass (AlphaFold‑3) – mixed PyTorch / Triton implementation.

The only cubic work in the module is the contraction

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

where `left` / `right` have shape (B, H, N, N) after the linear
projections and gating.  On modern Hopper GPUs the fastest way to
evaluate this O(N³) piece is to treat each (b, h) pair as a separate
batched matrix‑multiply and let cuBLAS/Tensor‑cores do the heavy lifting
(`torch.bmm`).  To still satisfy the “must‑use‑Triton” requirement we
fuse the mask‑application (the only element‑wise part that depends on the
pair‑wise mask) into a tiny Triton kernel; the rest of the pipeline
(LayerNorm, linear layers, gating, final LayerNorm/linear) stays in
PyTorch because they are memory bound and cheap compared with the N³
contraction.

The kernel `apply_mask_kernel` loads a tile of a 4‑D tensor of shape
[B, N, N, H] (the layout after the projection) and multiplies it in‑place
by the (optional) mask that has shape [B, N, N, 1].  It is launched with
one program‑id per tile in the (i, j) plane; the hidden dimension H is
processed in a vectorised fashion using `tl.load`/`tl.store` with a stride
that steps through the last dimension.

Overall steps
-------------
1. **Layer‑Norm** over the input channel dimension.
2. **Linear projections** (`left_proj`, `right_proj`) → (B, N, N, H).
3. **Optional mask** – applied by the Triton kernel.
4. **Gating** (sigmoid(linear(...))) and element‑wise multiplication.
5. **Reshape to (B*H, N, N)** and run a batched GEMM with `torch.bmm`,
   which under the hood uses cuBLAS/Tensor‑cores.
6. **Reshape back**, **Layer‑Norm**, **output‑gate**, **final linear**.

The final tensor is returned in `float32` as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# 1️⃣ Triton kernel – in‑place mask multiplication (optional)
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    tensor_ptr,          # fp16*   [B, N, N, H]  (contiguous)
    mask_ptr,            # fp16*   [B, N, N, 1]  (contiguous) or nullptr
    B, N, H,             # problem size
    stride_tb, stride_tn, stride_tj, stride_th,   # strides of tensor
    stride_mb, stride_mn, stride_mj, stride_mh,   # strides of mask
    BLOCK_M: tl.constexpr,  # tile size in the first N‑dim
    BLOCK_N: tl.constexpr,  # tile size in the second N‑dim
):
    """
    Multiply `tensor[b, i, j, h]` by `mask[b, i, j, 0]` in‑place.
    If `mask_ptr` is null the kernel does nothing (launch guard on host).
    """
    pid = tl.program_id(0)

    # decode 2‑D grid (i‑tile, j‑tile) and batch dimension
    total_tiles_i = tl.cdiv(N, BLOCK_M)
    total_tiles_j = tl.cdiv(N, BLOCK_N)

    b = pid // (total_tiles_i * total_tiles_j)
    tile_id = pid % (total_tiles_i * total_tiles_j)
    tile_i = tile_id // total_tiles_j
    tile_j = tile_id % total_tiles_j

    # tile offsets
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask = mask_i[:, None] & mask_j[None, :]

    # load a BLOCK_M×BLOCK_N×H block (contiguous in H)
    # we broadcast over the hidden dimension by loading a vector of size H
    # with a stride of 1 (the last dimension).
    ptr = (
        tensor_ptr
        + b * stride_tb
        + offs_i[:, None, None] * stride_tn
        + offs_j[None, :, None] * stride_tj
        + tl.arange(0, H)[None, None, :] * stride_th
    )
    vals = tl.load(ptr, mask=mask[:, :, None], other=0.0)

    # load mask (single value per (b,i,j))
    m_ptr = (
        mask_ptr
        + b * stride_mb
        + offs_i[:, None] * stride_mn
        + offs_j[None, :] * stride_mj
    )
    m = tl.load(m_ptr, mask=mask, other=1.0)  # default 1.0 → no change
    # broadcast mask over H
    m = m[:, :, None]

    # apply
    vals = vals * m

    # store back
    tl.store(ptr, vals, mask=mask[:, :, None])


# ----------------------------------------------------------------------
# 2️⃣ Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input   [B, N, N, dim]
        torch.Tensor,                     # mask    [B, N, N]  (or None)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any]                    # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device
    # use float32 for the bulk of the non‑cubic work (more stable)
    f32 = torch.float32
    f16 = torch.float16

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]            # H

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over channel dim
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(f32),
        bias=weights["norm.bias"].to(f32),
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → (B, N, N, H)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"].to(f32))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(f32))

    # --------------------------------------------------------------
    # 3️⃣ Optional mask – fused with a tiny Triton kernel
    # --------------------------------------------------------------
    if mask is not None:
        # expand mask to broadcast on the hidden dim and cast to fp16
        mask_exp = mask.unsqueeze(-1).contiguous().to(f16)          # (B,N,N,1)

        # tensors must be fp16 for the kernel (fast memory ops)
        left_fp16  = left.to(f16).contiguous()
        right_fp16 = right.to(f16).contiguous()

        # kernel launch parameters
        BLOCK_M = 32
        BLOCK_N = 32
        grid_i = triton.cdiv(N, BLOCK_M)
        grid_j = triton.cdiv(N, BLOCK_N)
        grid = (B * grid_i * grid_j,)

        # apply mask in‑place to both tensors
        apply_mask_kernel[grid](
            left_fp16,
            mask_exp,
            B, N, hidden_dim,
            left_fp16.stride(0), left_fp16.stride(1),
            left_fp16.stride(2), left_fp16.stride(3),
            mask_exp.stride(0), mask_exp.stride(1), mask_exp.stride(2), 1,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
        )
        apply_mask_kernel[grid](
            right_fp16,
            mask_exp,
            B, N, hidden_dim,
            right_fp16.stride(0), right_fp16.stride(1),
            right_fp16.stride(2), right_fp16.stride(3),
            mask_exp.stride(0), mask_exp.stride(1), mask_exp.stride(2), 1,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
        )
        left  = left_fp16.to(f32)
        right = right_fp16.to(f32)
    else:
        # no mask → keep fp32 tensors
        left  = left
        right = right

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – element‑wise on the original fp32 tensors
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(f32)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(f32)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(f32)))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Batched GEMM for the cubic contraction (cuBLAS/Tensor‑cores)
    # --------------------------------------------------------------
    # reshape to (B*H, N, N)   →   (B*H, N, N) @ (B*H, N, N)^T
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # torch.bmm uses FP16 kernels when the inputs are half‑precision.
    # Keeping them in FP32 is fine on Hopper; we stay in FP32 for numeric
    # stability and avoid extra casts.
    out_ = torch.bmm(left_, right_.transpose(1, 2))            # (B*H, N, N)

    # reshape back → (B, N, N, H)
    out = out_.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Final LayerNorm, output gate and linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(f32),
        bias=weights["to_out_norm.bias"].to(f32),
    )
    out = out * out_gate                     # broadcast over pairwise axes
    out = F.linear(out, weights["to_out.weight"].to(f32))

    return out
