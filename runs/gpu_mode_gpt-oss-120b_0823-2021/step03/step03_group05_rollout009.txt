"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣ Layer‑norm + the four linear projections (left, right, left_gate,
   right_gate, out_gate) are cheap and stay in PyTorch.

2️⃣ One‑shot Triton kernel fuses
   * optional mask   (shape [B,N,N,1])
   * left‑gate       (sigmoid)
   * right‑gate      (sigmoid)
   * multiplication of the gates into the projected tensors.
   The kernel works in FP16 (the dominant work‑size) and writes the
   masked‑gated tensors back in‑place.

3️⃣ Core cubic contraction
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   is a batched matrix‑multiply.  By reshaping to (B·H,N,N) we can call
   ``torch.bmm`` (Tensor‑core accelerated on Hopper).  No Triton GEMM is
   required – this is the fastest approach on the target hardware.

4️⃣ Post‑processing (layer‑norm, out‑gate, final linear) are again performed
   in PyTorch.  The final tensor is returned in float‑32.

The only mandatory Triton component is the element‑wise kernel (2).  It
covers both the mask‑present and mask‑absent cases and is fully vectorised.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# 2️⃣ Triton kernel – fuse mask + gate multiplication.
# -------------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *mut fp16   (B,N,N,H)  – written in‑place
    right_ptr,         # *mut fp16   (B,N,N,H)  – written in‑place
    left_gate_ptr,     # *const fp16 (B,N,N,H)  – sigmoid outputs
    right_gate_ptr,    # *const fp16 (B,N,N,H)
    mask_ptr,          # *const fp16 (B,N,N,1)  – 1.0 if no mask
    B, N, H,           # compile‑time constants
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_g0, stride_g1, stride_g2, stride_g3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # ----- decode linear offset to (b,i,j,h) ----------------------------
    tmp = offs
    h = tmp % H; tmp = tmp // H
    j = tmp % N; tmp = tmp // N
    i = tmp % N; b = tmp // N

    # ----- compute element pointers ------------------------------------
    l_ptr = left_ptr   + b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    r_ptr = right_ptr  + b * stride_r0 + i * stride_r1 + j * stride_r2 + h * stride_r3
    lg_ptr = left_gate_ptr  + b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    rg_ptr = right_gate_ptr + b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    m_ptr = mask_ptr   + b * stride_m0 + i * stride_m1 + j * stride_m2          # broadcast over H

    # ----- load ---------------------------------------------------------
    l  = tl.load(l_ptr, mask=mask, other=0.0)
    r  = tl.load(r_ptr, mask=mask, other=0.0)
    lg = tl.load(lg_ptr, mask=mask, other=0.0)
    rg = tl.load(rg_ptr, mask=mask, other=0.0)
    m  = tl.load(m_ptr, mask=mask, other=1.0)   # default 1.0 → no mask

    # ----- fused operation ---------------------------------------------
    # left = left * mask * left_gate
    # right = right * mask * right_gate
    l = l * m * lg
    r = r * m * rg

    # ----- store --------------------------------------------------------
    tl.store(l_ptr, l, mask=mask)
    tl.store(r_ptr, r, mask=mask)


def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        x        : Tensor[B, N, N, dim]          (float32)
        mask     : Tensor[B, N, N] or None       (bool/float32)
        weights  : dict of model parameters (float32)
        config   : {"dim": int, "hidden_dim": int}

    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]          # = H

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm + linear projections (remain in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(x.dtype),
        bias=weights["norm.bias"].to(x.dtype)
    )
    # → (B,N,N,hidden_dim)
    left  = F.linear(x, weights["left_proj.weight"].to(x.dtype))
    right = F.linear(x, weights["right_proj.weight"].to(x.dtype))

    # gates (pre‑sigmoid)
    left_gate  = F.linear(x, weights["left_gate.weight"].to(x.dtype))
    right_gate = F.linear(x, weights["right_gate.weight"].to(x.dtype))
    out_gate   = F.linear(x, weights["out_gate.weight"].to(x.dtype))

    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # -----------------------------------------------------------------
    # 2️⃣ Fuse mask & gate multiplication in‑place via Triton
    # -----------------------------------------------------------------
    # mask → (B,N,N,1) fp16, all‑ones when mask is None
    if mask is None:
        mask_tensor = torch.ones((B, N, N, 1), dtype=torch.float16,
                                 device=device)
    else:
        mask_tensor = mask.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        mask_tensor = mask_tensor.to(torch.float16)

    # Cast to FP16 for the kernel (lossless for the later FP32 steps)
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    lg_fp16    = left_gate.to(torch.float16).contiguous()
    rg_fp16    = right_gate.to(torch.float16).contiguous()

    total_elements = B * N * N * hidden_dim
    BLOCK = 1024                         # element‑wise kernel works well with 1K
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        lg_fp16,
        rg_fp16,
        mask_tensor,
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        lg_fp16.stride(0), lg_fp16.stride(1),
        lg_fp16.stride(2), lg_fp16.stride(3),
        mask_tensor.stride(0), mask_tensor.stride(1),
        mask_tensor.stride(2), mask_tensor.stride(3),
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 3️⃣ Core O(N³) contraction via batched GEMM (Tensor‑core fast)
    # -----------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_bh  = left_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B·H,N,N)
    right_bh = right_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # batched matmul: (B·H,N,N) @ (B·H,N,N)^T → (B·H,N,N)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))

    # back to (B,N,N, H)
    out = out_bh.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(x.dtype)                     # back to float32 for the rest

    # -----------------------------------------------------------------
    # 4️⃣ Post‑processing: layer‑norm, out‑gate, final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(x.dtype),
        bias=weights["to_out_norm.bias"].to(x.dtype)
    )
    out = out * out_gate                       # broadcast over N,N
    out = F.linear(out, weights["to_out.weight"].to(x.dtype))

    return out
