"""
custom_kernel – outgoing TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the input (FP32, pure torch – cheap & no extra kernel).
2️⃣  Cast to FP16 and compute the four linear maps (left_proj, right_proj,
    left_gate, right_gate, out_gate) with torch.nn.functional.linear.
3️⃣  **Fusion kernel (triton)** – for every element in the (B,N,N,hidden_dim)
    tensors we apply:
        left  = left  * left_gate  * mask   (mask is 1 if absent)
        right = right * right_gate * mask
   The kernel works in FP16, reads the four tensors plus an optional mask and
   writes back the two gated tensors.  This replaces three separate element‑wise
   PyTorch kernels and gives a noticeable speed‑up on H200.
4️⃣  Core contraction (the N³ part) is just a batched matrix‑multiply:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   By reshaping we turn each (b,d) pair into a (N×N) matrix multiply and
   call the highly‑optimised cuBLAS kernel via torch.bmm (FP16 → FP32
   accumulation).  This is the fastest approach on Hopper.
5️⃣  Convert the result back to FP32, apply a second LayerNorm, multiply by the
    output‑gate and finish with the final linear projection.
   All remaining ops are performed with torch (FP32) – they are cheap compared
   with the cubic contraction.

The implementation respects the required signature `custom_kernel(data)` and
works for all listed configurations (different seq‑len, batch‑size, dim and
hidden_dim, with or without a mask, and for both normal / Cauchy input
distributions).  The final result has dtype ``torch.float32``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask + gating into left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_and_gate_kernel(
    left_ptr,          # fp16, (B,N,N,H)
    right_ptr,         # fp16, (B,N,N,H)
    left_gate_ptr,     # fp16, (B,N,N,H)
    right_gate_ptr,    # fp16, (B,N,N,H)
    mask_ptr,          # fp16, (B,N,N,1) or nullptr
    B, N, H,           # extents
    stride_lz, stride_ly, stride_lx, stride_lc,   # left strides (z,y,x,c)
    stride_rz, stride_ry, stride_rx, stride_rc,   # right strides
    stride_lgz, stride_lgy, stride_lgx, stride_lgc,  # left_gate strides
    stride_rgz, stride_rgy, stride_rgx, stride_rgc,  # right_gate strides
    stride_mz, stride_my, stride_mx,               # mask strides (z,y,x)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                # linear index over all elements
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # convert linear offset to 4‑D index
    idx = offs
    c = idx % H
    idx = idx // H
    x = idx % N
    idx = idx // N
    y = idx % N
    z = idx // N

    # compute flat pointers
    left  = left_ptr  + z * stride_lz  + y * stride_ly  + x * stride_lx  + c * stride_lc
    right = right_ptr + z * stride_rz  + y * stride_ry  + x * stride_rx  + c * stride_rc
    lg    = left_gate_ptr  + z * stride_lgz + y * stride_lgy + x * stride_lgx + c * stride_lgc
    rg    = right_gate_ptr + z * stride_rgz + y * stride_rgy + x * stride_rgx + c * stride_rgc
    m_ptr = mask_ptr
    if mask_ptr != 0:
        m = m_ptr + z * stride_mz + y * stride_my + x * stride_mx
    else:
        m = None

    # load
    l = tl.load(left,  mask=mask, other=0.0)
    r = tl.load(right, mask=mask, other=0.0)
    lgv = tl.load(lg, mask=mask, other=0.0)
    rgv = tl.load(rg, mask=mask, other=0.0)
    if m is not None:
        mv = tl.load(m, mask=mask, other=1.0)   # mask is 1 where valid
        l = l * mv
        r = r * mv

    # apply gates
    l = l * lgv
    r = r * rgv

    # store back
    tl.store(left,  l, mask=mask)
    tl.store(right, r, mask=mask)


def _apply_mask_and_gate(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor | None,
):
    """
    In‑place fusion of mask and gating.  All tensors are fp16 and contiguous.
    """
    B, N, _, H = left.shape
    assert left.shape == right.shape == left_gate.shape == right_gate.shape
    # strides in the order (z, y, x, c)
    sz_l, sy_l, sx_l, sc_l = left.stride()
    sz_r, sy_r, sx_r, sc_r = right.stride()
    sz_lg, sy_lg, sx_lg, sc_lg = left_gate.stride()
    sz_rg, sy_rg, sx_rg, sc_rg = right_gate.stride()
    if mask is None:
        mask_ptr = 0
        sz_m = sy_m = sx_m = 0
    else:
        mask = mask.to(torch.float16).contiguous()
        sz_m, sy_m, sx_m = mask.stride()
        mask_ptr = mask.data_ptr()

    total = B * N * N * H
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _apply_mask_and_gate_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        left_gate.data_ptr(),
        right_gate.data_ptr(),
        mask_ptr,
        B, N, H,
        sz_l, sy_l, sx_l, sc_l,
        sz_r, sy_r, sx_r, sc_r,
        sz_lg, sy_lg, sx_lg, sc_lg,
        sz_rg, sy_rg, sx_rg, sc_rg,
        sz_m, sy_m, sx_m,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return left, right


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool or float)
        weights      : dict of leaf tensors (see reference implementation)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32, manual)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 and compute linear maps
    # --------------------------------------------------------------
    # weights are stored in FP32 – cast once
    def _fp16(t): return t.to(torch.float16)

    left_proj_w   = _fp16(weights["left_proj.weight"])    # (hidden_dim, dim)
    right_proj_w  = _fp16(weights["right_proj.weight"])
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)          # (B,N,N,dim)

    left      = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right     = F.linear(x_fp16, right_proj_w)
    left_g    = F.linear(x_fp16, left_gate_w)   # (B,N,N,hidden_dim)
    right_g   = F.linear(x_fp16, right_gate_w)
    out_gate  = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 3️⃣ Fuse mask + gating via Triton
    # --------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_g = left_g.contiguous()
    right_g = right_g.contiguous()
    if mask is not None:
        # mask expected shape (B,N,N) – expand to (B,N,N,1)
        mask_fp = mask.to(torch.float16).unsqueeze(-1).contiguous()
    else:
        mask_fp = None

    left, right = _apply_mask_and_gate(left, right, left_g, right_g, mask_fp)

    # --------------------------------------------------------------
    # 4️⃣ Core contraction (batched GEMM) using cuBLAS via torch.bmm
    # --------------------------------------------------------------
    # reshape: (B, N, N, H) -> (B*H, N, N)
    left_b = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B*H, I, K)
    right_b = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)  # (B*H, K, J)

    # torch.bmm works in FP16 → FP32 accumulation on Hopper
    out_b = torch.bmm(left_b, right_b)                               # (B*H, I, J)

    # back to (B, N, N, hidden_dim)
    out = out_b.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Output LayerNorm (FP32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 6️⃣ Final gating + projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
