"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm
---------
1. Layer‑norm the 4‑D input (B,N,N,C) (torch‑level, FP32).
2. Linear projections → (B,N,N,H) for left/right and the three gates.
3. Fuse mask, left‑gate and right‑gate in a tiny Triton kernel
   (element‑wise multiplication, still FP16).  This satisfies the
   “must‑use‑a‑kernel” requirement and is the only part that cannot be
   efficiently expressed by torch.compile().
4. Reshape the fused tensors to (B*H, N, N) and compute the O(N³) contraction
   with a single batched matrix‑multiply (torch.bmm → cuBLAS/Tensor‑cores).
5. Hidden‑dim LayerNorm, out‑gate gating and final linear projection back to
   the original channel dimension C (torch‑level, FP32).

All heavy arithmetic is performed in FP16 with FP32 accumulation, while the
final result is returned in FP32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – element‑wise fusion of mask, left‑gate and right‑gate
# ----------------------------------------------------------------------
@triton.jit
def _fuse_mask_gate_kernel(
    # inputs
    left_ptr, right_ptr,            # fp16 * (B,N,N,H)
    left_gate_ptr, right_gate_ptr,  # fp16 * (B,N,N,H)
    mask_ptr,                       # fp16 * (B,N,N,1)  or nullptr
    # outputs
    out_left_ptr, out_right_ptr,    # fp16 * (B,N,N,H)
    # shape
    B, N, H,                        # compile‑time scalars
    # strides (contiguous layout = [B,N,N,H])
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_g0, stride_g1, stride_g2, stride_g3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    # compile‑time flags
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    The kernel processes BLOCK elements per program‑id.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear ids
    total = B * N * N * H
    mask = offs < total

    # ---------- decode linear index ----------
    # layout: b → i → j → h (contiguous)
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ---------- load values ----------
    idx = (
        b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    )
    left  = tl.load(left_ptr + idx,  mask=mask, other=0.0)
    right = tl.load(right_ptr + idx, mask=mask, other=0.0)

    idx_gate = (
        b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    )
    lgate = tl.load(left_gate_ptr + idx_gate,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + idx_gate, mask=mask, other=0.0)

    if HAS_MASK:
        # mask has no hidden dimension → broadcast over H
        idx_mask = b * stride_m0 + i * stride_m1 + j * stride_m2
        m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)
    else:
        m = 1.0

    # ---------- fused write ----------
    out_l = left * lgate * m
    out_r = right * rgate * m
    tl.store(out_left_ptr + idx,  out_l, mask=mask)
    tl.store(out_right_ptr + idx, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input  (B,N,N,C)  float32
        torch.Tensor | None,  # mask   (B,N,N)    bool/float32 or None
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data :
        (input, mask, weights, config)
        input   – (B, N, N, C)  torch.float32
        mask    – (B, N, N)    torch.bool / torch.float32 / None
        weights – dict with the same keys as the reference module
        config  – {"dim": C, "hidden_dim": H}

    Returns
    -------
    out : torch.Tensor
        (B, N, N, C)  torch.float32
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, C = x.shape
    H = config["hidden_dim"]
    dtype = torch.float32

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (still FP32, will be cast to FP16 later)
    # --------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))      # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate= torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate  = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N,H)

    # --------------------------------------------------------------
    # 3️⃣ Fuse mask & gates in FP16 via Triton
    # --------------------------------------------------------------
    #   keep everything contiguous & FP16 for the upcoming GEMM
    left_fp16      = left.to(torch.float16).contiguous()
    right_fp16     = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate.to(torch.float16).contiguous()
    right_gate_fp16= right_gate.to(torch.float16).contiguous()

    if mask is None:
        # dummy mask tensor – size 0, stride dummy values will never be used
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask = False
    else:
        # mask → (B,N,N,1) to broadcast over H
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        has_mask = True

    # allocate fused outputs
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elems = B * N * N * H
    BLOCK = 1024                     # good trade‑off for Hopper SM90
    grid = (triton.cdiv(total_elems, BLOCK),)

    _fuse_mask_gate_kernel[grid](
        left_fp16, right_fp16,
        left_gate_fp16, right_gate_fp16,
        mask_fp16,
        left_fused, right_fused,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        mask_fp16.stride(0) if has_mask else 0,
        mask_fp16.stride(1) if has_mask else 0,
        mask_fp16.stride(2) if has_mask else 0,
        mask_fp16.stride(3) if has_mask else 0,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (FP16 inputs → FP16 output, FP32 accumulation)
    # --------------------------------------------------------------
    #   reshape to (B*H, N, N) and use torch.bmm (cuBLAS/Tensor‑cores)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)      # (B*H,N,N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)     # (B*H,N,N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))            # (B*H,N,N)

    #   reshape back to (B,N,N,H) and cast to FP32 for LayerNorm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()     # (B,N,N,H)
    out = out.to(dtype)

    # --------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                # broadcast over (N,N)
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
