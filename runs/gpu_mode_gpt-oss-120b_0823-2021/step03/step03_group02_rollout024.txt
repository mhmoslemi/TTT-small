"""
TriMul “outgoing” forward (no gradients).

The O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a single batched matrix‑multiply (torch.bmm) which
leverages cuBLAS/Tensor‑cores on the H200.  All element‑wise work
(masking, gating, layer‑norms, linear projections) is kept in fp16
except for the final LayerNorm / linear that need fp32 accuracy.

A tiny Triton kernel fuses the mask‑and‑gate multiplication for the
two projection tensors – this reduces the number of launches and
removes an extra memory pass compared with a naïve PyTorch
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton: fused mask & gate on the projected tensors (fp16 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr,          # (B,N,N,H) flattened pointer – output buffer
    src_ptr,           # (B,N,N,H) input buffer (already projected)
    mask_ptr,          # (B,N,N,1) mask (broadcasted)
    gate_ptr,          # (B,N,N,H) gate (sigmoid output)
    B, N, H,           # static shapes
    stride_src_b, stride_src_n, stride_src_n2, stride_src_h,
    stride_mask_b, stride_mask_n, stride_mask_n2,
    stride_gate_b, stride_gate_n, stride_gate_n2, stride_gate_h,
    BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    # linear index over (B,N,N) – each program works on a tile of H values
    total = B * N * N
    tile_start = pid * BLOCK_N
    offs = tile_start + tl.arange(0, BLOCK_N)
    mask = offs < total

    # convert linear index to (b,n,i) coordinates
    b = offs // (N * N)
    rem = offs % (N * N)
    i = rem // N
    j = rem % N

    # pointers for the 3‑D index
    src = src_ptr + b * stride_src_b + i * stride_src_n + j * stride_src_n2
    mask = mask_ptr + b * stride_mask_b + i * stride_mask_n + j * stride_mask_n2
    gate = gate_ptr + b * stride_gate_b + i * stride_gate_n + j * stride_gate_n2

    # iterate over hidden dimension H in blocks
    for offset_h in range(0, H, BLOCK_H):
        h_range = offset_h + tl.arange(0, BLOCK_H)
        h_mask = h_range < H

        # load
        src_val = tl.load(src + h_range * stride_src_h,
                          mask=mask[:, None] & h_mask[None, :],
                          other=0.0)
        mask_val = tl.load(mask, mask=mask, other=0.0)   # (B,N,N,1)
        gate_val = tl.load(gate + h_range * stride_gate_h,
                           mask=mask[:, None] & h_mask[None, :],
                           other=0.0)

        # fused multiply
        out_val = src_val * mask_val * gate_val

        # store
        tl.store(proj_ptr + b * stride_src_b + i * stride_src_n + j * stride_src_n2
                 + h_range * stride_src_h,
                 out_val,
                 mask=mask[:, None] & h_mask[None, :])


def _apply_mask_gate(proj, mask, gate):
    """
    proj : (B,N,N,H) fp16 tensor (already projected)
    mask : (B,N,N,1) fp16 tensor or None
    gate : (B,N,N,H) fp16 tensor (sigmoid)
    returns masked & gated tensor, same shape as proj
    """
    if mask is None:
        # no mask -> just element‑wise multiply with gate
        return proj * gate

    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # strides in element units (not bytes)
    s_proj = proj.stride()
    s_mask = mask.stride()
    s_gate = gate.stride()

    BLOCK_N = 128          # tile over (B,N,N) dimension
    BLOCK_H = 32           # tile over hidden dimension

    grid = (triton.cdiv(B * N * N, BLOCK_N),)

    mask_gate_fuse_kernel[grid](
        out,
        proj,
        mask,
        gate,
        B, N, H,
        stride_src_b=s_proj[0], stride_src_n=s_proj[1],
        stride_src_n2=s_proj[2], stride_src_h=s_proj[3],
        stride_mask_b=s_mask[0], stride_mask_n=s_mask[1],
        stride_mask_n2=s_mask[2],
        stride_gate_b=s_gate[0], stride_gate_n=s_gate[1],
        stride_gate_n2=s_gate[2], stride_gate_h=s_gate[3],
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Arguments
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]  (float32, CUDA)
        mask         : torch.Tensor [B, N, N] or None (float32/bool)
        weights      : dict of torch.Tensor parameters
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep in FP32, tiny)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b          # [B,N,N,dim] FP32

    # ------------------------------------------------------------------
    # 2) Linear projections (FP16)
    # ------------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16, non_blocking=True)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = weights["to_out_norm.weight"]   # keep FP32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)          # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)         # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 3) Gating (sigmoid) – still FP16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 4) Fuse mask + gate on the projected tensors via Triton
    # ------------------------------------------------------------------
    mask_fp16 = None
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)

    left  = _apply_mask_gate(left,  mask_fp16, left_gate)
    right = _apply_mask_gate(right, mask_fp16, right_gate)

    # ------------------------------------------------------------------
    # 5) Core contraction – batched GEMM using cuBLAS (torch.bmm)
    #    left : (B, hidden, N, N) -> (B*hidden, N, N)
    #    right: (B, hidden, N, N) -> (B*hidden, N, N) then transpose last two dims
    # ------------------------------------------------------------------
    # bring hidden dim to batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B, H, I, K = left_perm.shape
    _, _, K2, J = right_perm.shape
    assert K == K2

    left_flat  = left_perm.view(B * H, I, K)            # (B*H, N, N)
    right_flat = right_perm.view(B * H, K, J)           # (B*H, N, N)

    # batched matrix multiply in FP16, result FP16
    out_flat = torch.bmm(left_flat, right_flat)        # (B*H, N, N)

    # reshape back to (B, H, N, N) and permute to (B, N, N, H)
    out_fp16 = out_flat.view(B, H, I, J).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7) Final gating + linear projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)   # same out_gate as in reference
    out = F.linear(out, to_out_w)            # (B,N,N,dim) FP32

    return out
