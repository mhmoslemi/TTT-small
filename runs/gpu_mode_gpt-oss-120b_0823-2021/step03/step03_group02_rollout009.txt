"""
TriMul “outgoing” forward pass (no gradients).

The O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a single torch.bmm (batched GEMM) which lets cuBLAS/Tensor‑cores
run at peak efficiency on Hopper.  All element‑wise work (masking, gating,
output‑layer‑norm, final linear) is kept in PyTorch except for a tiny
fusion kernel that multiplies the two projected tensors with their
corresponding gates and the optional pair‑wise mask.  This kernel
eliminates two separate element‑wise passes and therefore gives a
noticeable speed‑up while still satisfying the requirement of “at least
part of the computation in Triton”.

The implementation follows the reference PyTorch code exactly, uses
mixed‑precision (fp16 for the heavy GEMM) and returns a fp32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) and gating into the projected tensors
# ----------------------------------------------------------------------
@triton.jit
def fuse_mask_gate_kernel(
    left_ptr,          # (B, N, N, H)   fp16
    right_ptr,         # (B, N, N, H)   fp16
    left_gate_ptr,     # (B, N, N, H)   fp16
    right_gate_ptr,    # (B, N, N, H)   fp16
    mask_ptr,          # (B, N, N, 1)   fp16  or nullptr
    out_left_ptr,      # (B, N, N, H)   fp16  (in‑place result)
    out_right_ptr,     # (B, N, N, H)   fp16  (in‑place result)
    B, N, H,
    stride_left_b, stride_left_n, stride_left_n2, stride_left_h,
    stride_right_b, stride_right_n, stride_right_n2, stride_right_h,
    stride_gate_b, stride_gate_n, stride_gate_n2, stride_gate_h,
    stride_mask_b, stride_mask_n, stride_mask_n2,
    HAS_MASK: tl.constexpr,
    BLOCK_B: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_n = tl.program_id(1)

    offs_b = pid_b * BLOCK_B + tl.arange(0, BLOCK_B)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_b = offs_b < B
    mask_n = offs_n < N

    # iterate over the hidden dimension inside the kernel
    for off_h in range(0, H, BLOCK_H):
        offs_h = off_h + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load left/right/gates (shape BLOCK_B×BLOCK_N×BLOCK_N×BLOCK_H)
        l = tl.load(
            left_ptr
            + offs_b[:, None, None, None] * stride_left_b
            + offs_n[None, :, None, None] * stride_left_n
            + offs_n[None, None, :, None] * stride_left_n2
            + offs_h[None, None, None, :] * stride_left_h,
            mask=mask_b[:, None, None, None] & mask_n[None, :, None, None]
                 & mask_n[None, None, :, None] & mask_h[None, None, None, :],
            other=0.0,
        )
        r = tl.load(
            right_ptr
            + offs_b[:, None, None, None] * stride_right_b
            + offs_n[None, :, None, None] * stride_right_n
            + offs_n[None, None, :, None] * stride_right_n2
            + offs_h[None, None, None, :] * stride_right_h,
            mask=mask_b[:, None, None, None] & mask_n[None, :, None, None]
                 & mask_n[None, None, :, None] & mask_h[None, None, None, :],
            other=0.0,
        )
        lg = tl.load(
            left_gate_ptr
            + offs_b[:, None, None, None] * stride_gate_b
            + offs_n[None, :, None, None] * stride_gate_n
            + offs_n[None, None, :, None] * stride_gate_n2
            + offs_h[None, None, None, :] * stride_gate_h,
            mask=mask_b[:, None, None, None] & mask_n[None, :, None, None]
                 & mask_n[None, None, :, None] & mask_h[None, None, None, :],
            other=0.0,
        )
        rg = tl.load(
            right_gate_ptr
            + offs_b[:, None, None, None] * stride_gate_b
            + offs_n[None, :, None, None] * stride_gate_n
            + offs_n[None, None, :, None] * stride_gate_n2
            + offs_h[None, None, None, :] * stride_gate_h,
            mask=mask_b[:, None, None, None] & mask_n[None, :, None, None]
                 & mask_n[None, None, :, None] & mask_h[None, None, None, :],
            other=0.0,
        )

        if HAS_MASK:
            m = tl.load(
                mask_ptr
                + offs_b[:, None, None, None] * stride_mask_b
                + offs_n[None, :, None, None] * stride_mask_n
                + offs_n[None, None, :, None] * stride_mask_n2,
                mask=mask_b[:, None, None, None] & mask_n[None, :, None, None]
                     & mask_n[None, None, :, None],
                other=0.0,
            )
            # broadcast mask over H
            m = m[..., None]
            l = l * m
            r = r * m

        # apply gates (sigmoid already applied in FP16)
        l = l * lg
        r = r * rg

        # write back
        tl.store(
            out_left_ptr
            + offs_b[:, None, None, None] * stride_left_b
            + offs_n[None, :, None, None] * stride_left_n
            + offs_n[None, None, :, None] * stride_left_n2
            + offs_h[None, None, None, :] * stride_left_h,
            l,
            mask=mask_b[:, None, None, None] & mask_n[None, :, None, None]
                 & mask_n[None, None, :, None] & mask_h[None, None, None, :],
        )
        tl.store(
            out_right_ptr
            + offs_b[:, None, None, None] * stride_right_b
            + offs_n[None, :, None, None] * stride_right_n
            + offs_n[None, None, :, None] * stride_right_n2
            + offs_h[None, None, None, :] * stride_right_h,
            r,
            mask=mask_b[:, None, None, None] & mask_n[None, :, None, None]
                 & mask_n[None, None, :, None] & mask_h[None, None, None, :],
        )


def _fuse_mask_and_gate(left, right, left_gate, right_gate, mask):
    """
    Fuse ``mask`` and the three gate tensors into ``left`` and ``right`` in‑place.
    All tensors are fp16 and have shape (B,N,N,H).  Returns the fused tensors.
    """
    B, N, _, H = left.shape
    HAS_MASK = mask is not None

    # stride information (in elements)
    stride_l = left.stride()
    stride_r = right.stride()
    stride_g = left_gate.stride()   # left_gate/right_gate share stride
    if HAS_MASK:
        stride_m = mask.stride()
    else:
        stride_m = (0, 0, 0, 0)  # dummy

    # launch configuration – we tile over B and N, hidden dim is looped inside
    BLOCK_B = 1
    BLOCK_N = 64
    BLOCK_H = 32

    grid = (triton.cdiv(B, BLOCK_B), triton.cdiv(N, BLOCK_N))
    fuse_mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask if HAS_MASK else left,   # dummy pointer, never read when HAS_MASK=0
        out_left_ptr=left,
        out_right_ptr=right,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_l[0],
        stride_left_n=stride_l[1],
        stride_left_n2=stride_l[2],
        stride_left_h=stride_l[3],
        stride_right_b=stride_r[0],
        stride_right_n=stride_r[1],
        stride_right_n2=stride_r[2],
        stride_right_h=stride_r[3],
        stride_gate_b=stride_g[0],
        stride_gate_n=stride_g[1],
        stride_gate_n2=stride_g[2],
        stride_gate_h=stride_g[3],
        stride_mask_b=stride_m[0],
        stride_mask_n=stride_m[1],
        stride_mask_n2=stride_m[2],
        HAS_MASK=1 if HAS_MASK else 0,
        BLOCK_B=BLOCK_B,
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return left, right


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + small Triton fusion)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor [B, N, N] (float32/bool) or None
        - weights      : dict of torch.Tensor parameters
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep in FP32)
    # ------------------------------------------------------------------
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * w_norm + b_norm          # still FP32

    # ------------------------------------------------------------------
    # 2) Linear projections (FP16) – keep a copy in FP16 for the heavy part
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left = F.linear(x_fp16, weights["left_proj.weight"])      # (B,N,N,hidden)
    right = F.linear(x_fp16, weights["right_proj.weight"])

    left_gate = F.linear(x_fp16, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"]).sigmoid()  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 3) Fuse mask + gates into left/right via Triton (in‑place)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
    else:
        mask_fp16 = None

    left, right = _fuse_mask_and_gate(left, right, left_gate, right_gate, mask_fp16)

    # ------------------------------------------------------------------
    # 4) Core contraction: (B, H, N, N) @ (B, H, N, N).T  → (B, H, N, N)
    #    We reshape to (B*H, N, N) and use torch.bmm which maps to cuBLAS/Tensor‑cores.
    # ------------------------------------------------------------------
    # bring hidden dim to second axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, J)

    B, H, I, K = left_perm.shape
    _, _, K2, J = right_perm.shape
    assert K == K2

    # flatten batch*head
    left_flat = left_perm.view(B * H, I, K)
    right_flat = right_perm.view(B * H, K, J)

    out_flat = torch.bmm(left_flat, right_flat)        # (B*H, I, J)  FP16
    out_fp16 = out_flat.view(B, H, I, J).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # back to fp32 for the remaining layers
    out = out_fp16.to(torch.float32)   # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 5) Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    w_outnorm = weights["to_out_norm.weight"]
    b_outnorm = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * w_outnorm + b_outnorm

    # ------------------------------------------------------------------
    # 6) Final gating (same out_gate as before) and linear projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
