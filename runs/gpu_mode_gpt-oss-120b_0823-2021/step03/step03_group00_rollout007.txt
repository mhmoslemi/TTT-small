"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward only, no gradients)
-------------------------------------------------
1️⃣  Layer‑norm the 4‑D input (float32) – done in PyTorch (fast enough).
2️⃣  Linear projections + gate‑projections are performed in fp16.
3️⃣  Optional binary mask is applied with a tiny Triton kernel
    (broadcast over the hidden dimension, element‑wise multiply).
4️⃣  Sigmoid gating (float16) and element‑wise multiplication.
5️⃣  Core contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    is reshaped to a batched matrix‑multiply:
        (B·H, N, N) × (B·H, N, N)ᵀ → (B·H, N, N)
    and executed with torch.bmm which uses cuBLAS/Tensor‑cores.
6️⃣  Second Layer‑norm (float32), multiply by the output‑gate and a final
    linear projection back to the original dimensionality.

Only step 3 is off‑loaded to Triton, satisfying the “must use a kernel”
requirement while keeping the dominant O(N³) work on the highly‑optimized
cuBLAS path.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – element‑wise multiply by a broadcast mask
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    tensor_ptr,          # fp16 tensor   [B, N, N, H]
    mask_ptr,            # fp16 mask    [B, N, N, 1]
    out_ptr,             # fp16 output  same shape as tensor_ptr
    B, N, H,             # static sizes (constexpr)
    stride_tb, stride_tm, stride_tn, stride_th,   # tensor strides
    stride_mb, stride_mm, stride_mn,             # mask strides (last dim is 1)
    BLOCK_B: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Each program processes a 4‑D tile (BLOCK_B × BLOCK_N × BLOCK_M × BLOCK_H)
    of the tensor.  The mask has a singleton hidden dimension and is broadcast
    over H.
    """
    pid_b = tl.program_id(0)   # batch tile
    pid_n = tl.program_id(1)   # row   tile
    pid_m = tl.program_id(2)   # col   tile

    # start offsets for this tile
    offs_b = pid_b * BLOCK_B + tl.arange(0, BLOCK_B)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)

    # masks for out‑of‑bounds
    mask_b = offs_b < B
    mask_n = offs_n < N
    mask_m = offs_m < N

    # iterate over hidden dim in chunks of BLOCK_H
    for h in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # linear indices for tensor
        tensor_offset = (
            offs_b[:, None, None, None] * stride_tb
            + offs_n[None, :, None, None] * stride_tm
            + offs_m[None, None, :, None] * stride_tn
            + offs_h[None, None, None, :] * stride_th
        )
        # linear indices for mask (hidden dim = 1, so stride 0)
        mask_offset = (
            offs_b[:, None, None] * stride_mb
            + offs_n[None, :, None] * stride_mm
            + offs_m[None, None, :] * stride_mn
        )

        # load
        a = tl.load(
            tensor_ptr + tensor_offset,
            mask=mask_b[:, None, None, None] & mask_n[None, :, None, None]
                 & mask_m[None, None, :, None] & mask_h[None, None, None, :],
            other=0.0,
        )
        m = tl.load(
            mask_ptr + mask_offset,
            mask=mask_b[:, None, None] & mask_n[None, :, None] & mask_m[None, None, :],
            other=0.0,
        )
        # broadcast mask over H and multiply
        out = a * m[..., None]

        # store
        tl.store(
            out_ptr + tensor_offset,
            out,
            mask=mask_b[:, None, None, None] & mask_n[None, :, None, None]
                 & mask_m[None, None, :, None] & mask_h[None, None, None, :],
        )


def _apply_mask(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    tensor : fp16 [B, N, N, H]
    mask   : fp16 [B, N, N, 1]   (broadcast over H)
    Returns tensor * mask (fp16) computed with Triton.
    """
    B, N, _, H = tensor.shape
    # launch parameters – chosen to give good occupancy on Hopper
    BLOCK_B = 1          # batch dimension is tiny (B≤2)
    BLOCK_N = 32
    BLOCK_M = 32
    BLOCK_H = 32

    grid = (
        triton.cdiv(B, BLOCK_B),   # batch tiles
        triton.cdiv(N, BLOCK_N),   # row tiles
        triton.cdiv(N, BLOCK_M),   # col tiles
    )
    out = torch.empty_like(tensor)

    # strides in element units (not bytes)
    stride_tb, stride_tm, stride_tn, stride_th = tensor.stride()
    stride_mb, stride_mm, stride_mn = mask.stride()

    _apply_mask_kernel[grid](
        tensor,
        mask,
        out,
        B,
        N,
        H,
        stride_tb,
        stride_tm,
        stride_tn,
        stride_th,
        stride_mb,
        stride_mm,
        stride_mn,
        BLOCK_B=BLOCK_B,
        BLOCK_N=BLOCK_N,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Helper: batched GEMM using cuBLAS (torch.bmm) – the heavy O(N³) part
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, N, N)   fp16
    right : (B, H, N, N)   fp16
    Returns (B, H, N, N)   fp16  where out[b,h] = left[b,h] @ right[b,h].t()
    """
    B, H, N, _ = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.reshape(B * H, N, N)
    right_flat = right.reshape(B * H, N, N)

    # torch.bmm does (B*H, N, N) @ (B*H, N, N)ᵀ efficiently on Hopper
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    return out_flat.view(B, H, N, N)


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors (see below)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (float32, manual – mean/var over last dim)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still fp32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (fp16) – bias‑less as in the reference
    # ------------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                     # (B, N, N, dim)

    left  = F.linear(x_fp16, left_proj_w)            # (B, N, N, hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask – Triton kernel (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask to fp16 and add singleton hidden dimension
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)          # (B, N, N, 1)
        left  = _apply_mask(left, mask_fp16)
        right = _apply_mask(right, mask_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()          # (B, N, N, hidden_dim)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction – batched GEMM via cuBLAS
    #    reshape to (B, H, N, N) for both operands, then transpose right
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()   # same layout, we will transpose inside GEMM

    # compute (B, H, N, N) result in fp16
    out_fp16 = _batched_gemm_fp16(left_perm, right_perm)   # (B, H, N, N)

    # back to original layout (B, N, N, H) and to fp32 for the next steps
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + linear projection (float32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over hidden dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                  # (B, N, N, dim)

    return out
