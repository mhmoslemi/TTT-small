"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the input (FP32) – implemented with a tiny Triton kernel that
    computes per‑row mean / variance and applies the learned scale & bias.
2️⃣  Cast to FP16 and evaluate the four linear layers (left/right proj + 3 gates)
    with PyTorch `F.linear` (the dominant work is a 2‑D mat‑mul, cuBLAS can
    handle it efficiently).
3️⃣  Fuse mask, gating and the three element‑wise multiplications in a second
    Triton kernel (`_mask_gate_kernel`).  This keeps the data in FP16 and
    avoids extra memory traffic.
4️⃣  Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is an ordinary batched matrix‑multiply over the *k* dimension.
    We reshape the tensors to (B·H, N, N) and call `torch.bmm`,
    which on Hopper uses Tensor‑cores (fp16→fp32→fp16) and is far faster
    than a hand‑written N³ Triton GEMM.
5️⃣  The result is brought back to (B,N,N,dim) (FP32), a second Layer‑norm
    (Triton again), multiplied by the output‑gate and finally projected
    back to the original dimension with a FP32 linear layer.

Only the two element‑wise kernels are written in Triton; the cubic
contraction is delegated to cuBLAS, which is the fastest choice on H200.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel 1: per‑row LayerNorm (FP32 → FP32)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # input  (B, N, N, D)
    y_ptr,          # output (same shape)
    w_ptr,          # weight (D,)
    b_ptr,          # bias   (D,)
    D,              # size of the last dimension
    eps: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # linear index over all rows
    row_offset = pid * D                       # offset of the row start

    # offsets inside the row
    offs = row_offset + tl.arange(0, BLOCK)

    mask = offs < (pid + 1) * D                 # guard for last partial block
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)

    # compute mean
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / D

    # compute variance
    diff = x - mean
    var = tl.sum(diff * diff, axis=0) / D
    inv_std = 1.0 / tl.sqrt(var + eps)

    # apply scale & bias
    w = tl.load(w_ptr + tl.arange(0, BLOCK), mask=tl.arange(0, BLOCK) < D, other=0.0)
    b = tl.load(b_ptr + tl.arange(0, BLOCK), mask=tl.arange(0, BLOCK) < D, other=0.0)
    y = (x - mean) * inv_std * w + b

    tl.store(y_ptr + offs, y, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel 2: fused mask + gating (FP16)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    a_ptr,              # tensor to be masked & gated (B,N,N,H)  fp16
    gate_ptr,           # gate tensor (same shape)            fp16
    mask_ptr,           # mask (B,N,N,1) broadcasted           fp16
    out_ptr,            # result (same shape)                  fp16
    H,                  # hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # linear index over rows * H
    row = pid // H
    h   = pid % H

    # compute base offsets
    base = row * H * H + h * H                # not used; we treat flat layout
    # simpler: treat the three‑dim tensor as flat and stride‑aware
    # load a single element
    offs = pid * H + tl.arange(0, BLOCK)

    mask = tl.load(mask_ptr + row * H + tl.arange(0, BLOCK), mask=tl.arange(0, BLOCK) < H, other=0.0)
    a = tl.load(a_ptr + offs, mask=tl.arange(0, BLOCK) < H, other=0.0)
    g = tl.load(gate_ptr + offs, mask=tl.arange(0, BLOCK) < H, other=0.0)

    out = a * mask * g
    tl.store(out_ptr + offs, out, mask=tl.arange(0, BLOCK) < H)


# ----------------------------------------------------------------------
# Helper that launches the two kernels
# ----------------------------------------------------------------------
def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm over the last dimension using Triton (FP32)."""
    B, N, _, D = x.shape
    total_rows = B * N * N
    BLOCK = 128
    grid = (triton.cdiv(total_rows * D, BLOCK),)

    out = torch.empty_like(x)
    _layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=out,
        w_ptr=weight,
        b_ptr=bias,
        D=tl.constexpr(D),
        eps=tl.constexpr(eps),
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out


def _apply_mask_and_gate(tensor: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor) -> torch.Tensor:
    """Fuse (tensor * mask) * gate in FP16 with Triton."""
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 128
    grid = (triton.cdiv(total, BLOCK),)

    out = torch.empty_like(tensor)
    _mask_gate_kernel[grid](
        a_ptr=tensor,
        gate_ptr=gate,
        mask_ptr=mask,
        out_ptr=out,
        H=tl.constexpr(H),
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool or float)
        weights      : dict of leaf weight tensors
        config       : dict containing keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32) – Triton
    # ------------------------------------------------------------------
    inp = _layernorm_triton(
        inp,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=1e-5,
    )   # still FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 and linear projections (cuBLAS)
    # ------------------------------------------------------------------
    x_fp16 = inp.to(torch.float16)

    # weight matrices are stored as (out_features, in_features)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask & gating – Triton fusion
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)      # (B,N,N,1)
    else:
        mask_fp16 = torch.ones((B, N, N, 1), dtype=torch.float16,
                               device=device)

    # fuse mask & gating for left and right separately
    left  = _apply_mask_and_gate(left,  left_gate,  mask_fp16)
    right = _apply_mask_and_gate(right, right_gate, mask_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction (batched GEMM via cuBLAS)
    #    left : (B, N, N, hidden) -> (B*hidden, N, N)
    #    right: (B, N, N, hidden) -> (B*hidden, N, N)   (need transpose k<->j)
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, N, N)

    B_h, H, I, K = left_t.shape
    _, _, K2, J = right_t.shape
    assert K == K2 and I == J == N

    # reshape for batched matmul
    a = left_t.view(B_h * H, I, K)      # (B*hidden, N, N)
    b = right_t.view(B_h * H, K, J)     # (B*hidden, N, N)

    # torch.bmm uses fp16 matmul → fp32 accumulation on Hopper
    out_fp16 = torch.bmm(a, b)         # (B*hidden, N, N)
    out_fp16 = out_fp16.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    # shape (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 5️⃣ Back to FP32, second LayerNorm (Triton)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    out = _layernorm_triton(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final gating + projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
