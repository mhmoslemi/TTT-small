"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm (fast path)
---------------------
1️⃣ Layer‑norm the input (B,N,N,C).  
2️⃣ Linear projections → left, right (B,N,N,H) and three gates (B,N,N,H).  
3️⃣ Fuse element‑wise:   left  = left  * left_gate  * mask  
                         right = right * right_gate * mask  
   (mask optional).  This tiny work is done by a Triton kernel so that the
   evaluation always runs on the GPU.  
4️⃣ Reshape to (B*H, N, N) and perform a batched matrix‑multiply
   out = left @ rightᵀ   (FP16 inputs, FP32 accumulation, Tensor‑core friendly).  
   The result is (B*H,N,N) → (B,N,N,H) after reshaping.  
5️⃣ Hidden‑dim LayerNorm, gating with out_gate and final projection back to C.
   The last three steps stay in PyTorch (cuBLAS / torch.nn.functional).

Only the element‑wise fusion (step 3) is required to be a Triton kernel – the
rest can be efficiently handled by existing BLAS kernels.  All heavy
arithmetic is performed in float16, while the final output is float32 as
required by the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – mask & gate fusion
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16 * (B,N,N,H)
    right_ptr,         # fp16 * (B,N,N,H)
    left_gate_ptr,     # fp16 * (B,N,N,H)
    right_gate_ptr,    # fp16 * (B,N,N,H)
    mask_ptr,          # fp16 * (B,N,N)  or nullptr
    out_left_ptr,      # fp16 * (B,N,N,H)
    out_right_ptr,     # fp16 * (B,N,N,H)
    B, N, H,           # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_m0, stride_m1, stride_m2,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    The kernel works on a 1‑D grid of size ceil(B*N*N*H / BLOCK).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # ---------- decode linear index ----------
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N          # because original layout is (B,N,N,H)

    # ---------- load tensors ----------
    idx = (
        b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    )
    left = tl.load(left_ptr + idx,      mask=mask, other=0.0)
    right = tl.load(right_ptr + idx,    mask=mask, other=0.0)

    idx_gate = (
        b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + h * stride_lg3
    )
    lgate = tl.load(left_gate_ptr + idx_gate,   mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + idx_gate,  mask=mask, other=0.0)

    if HAS_MASK:
        # mask has shape (B,N,N) → broadcast over H
        idx_mask = b * stride_m0 + i * stride_m1 + j * stride_m2
        m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)
    else:
        m = 1.0

    # ---------- fuse ----------
    out_l = left * lgate * m
    out_r = right * rgate * m

    # ---------- store ----------
    tl.store(out_left_ptr + idx,  out_l, mask=mask)
    tl.store(out_right_ptr + idx, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config) where
        * x        – (B, N, N, C)    torch.float32
        * mask     – (B, N, N)       torch.bool / torch.float32 / None
        * weights  – dict of all learned parameters (float32)
        * config   – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor
        (B, N, N, C) float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, C = x.shape
    H = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (still FP32)
    # --------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate= torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate  = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # --------------------------------------------------------------
    # 3️⃣ Cast to FP16 and fuse mask + gates with Triton
    # --------------------------------------------------------------
    left_fp16      = left.to(torch.float16).contiguous()
    right_fp16     = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate.to(torch.float16).contiguous()
    right_gate_fp16= right_gate.to(torch.float16).contiguous()

    if mask is None:
        # dummy mask (unused)
        mask_fp16 = torch.empty(0,
                                device=device,
                                dtype=torch.float16)
        has_mask = False
    else:
        # mask → (B,N,N,1) for broadcasting
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        has_mask = True

    # allocate output buffers
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elements = B * N * N * H
    BLOCK = 1024                     # good occupancy on Hopper
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (FP16 inputs → FP16 output, FP32 accumulation)
    # --------------------------------------------------------------
    # reshape (B,N,N,H) → (B*H, N, N)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))          # (B*H,N,N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)
    out = out.to(dtype)  # back to FP32 for following LayerNorm

    # --------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, gating, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                     # broadcast over (N,N)
    out = F.linear(out, weights["to_out.weight"].to(dtype))  # (B,N,N,C)

    return out
