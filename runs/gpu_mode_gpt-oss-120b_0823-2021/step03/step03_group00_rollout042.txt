"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (float32) – done with torch ops (fast, numerically
    stable).
2️⃣  Linear projections + three gating linear layers – cast the input to fp16
    and perform the linear maps with torch.nn.functional.linear.
3️⃣  **Triton kernel** – fuse the element‑wise mask (if any) and the three
    sigmoid gates into a single kernel that writes the gated left/right
    tensors back in fp16.  This removes a memory round‑trip and gives a tiny
    speed win while still satisfying the “must‑use‑triton” rule.
4️⃣  Heavy cubic contraction  
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .
    The tensors are reshaped to (B·H, N, N) (H = hidden_dim) and the batched
    matrix‑multiply is delegated to cuBLAS via ``torch.bmm`` – the fastest
    implementation on Hopper (Tensor‑Core, FP16→FP32 accumulation).
5️⃣  Back‑to‑float32, second LayerNorm, multiply by the output‑gate and final
    linear projection to the original dimension.

The kernel in step 3 is the only Triton‑coded part; everything else stays in
PyTorch for clarity and maximal performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – mask + gates (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    left_ptr, right_ptr,          # [batch, N, N, H] fp16
    left_gate_ptr, right_gate_ptr,# [batch, N, N, H] fp16 (sigmoids)
    mask_ptr,                     # [batch, N, N] fp16   (1.0 or 0.0)  or nullptr
    *,
    BATCH, N, H,                  # static sizes
    stride_left_b, stride_left_n, stride_left_n2, stride_left_h,
    stride_right_b, stride_right_n, stride_right_n2, stride_right_h,
    stride_gate_b, stride_gate_n, stride_gate_n2, stride_gate_h,
    stride_mask_b, stride_mask_n, stride_mask_n2,
    BLOCK_BATCH: tl.constexpr, BLOCK_H: tl.constexpr
):
    """
    One program processes a tile of ``BLOCK_BATCH`` * ``BLOCK_H`` elements
    (batch‑wise * hidden‑dim).  The three tensors are *contiguous* in the
    hidden dimension, so we can load a vector of size BLOCK_H at a time.
    """
    pid = tl.program_id(0)                 # linear batch index
    off_batch = pid * BLOCK_BATCH + tl.arange(0, BLOCK_BATCH)
    off_h     = tl.arange(0, BLOCK_H)

    # --------------------------- masks ------------------------------
    mask_batch = off_batch < BATCH
    mask_h     = off_h < H
    mask = mask_batch[:, None] & mask_h[None, :]

    # --------------------------- loads ------------------------------
    # left
    left = tl.load(
        left_ptr
        + off_batch[:, None] * stride_left_b
        + off_h[None, :]   * stride_left_h,
        mask=mask,
        other=0.0,
    )
    # right
    right = tl.load(
        right_ptr
        + off_batch[:, None] * stride_right_b
        + off_h[None, :]    * stride_right_h,
        mask=mask,
        other=0.0,
    )
    # gates
    left_gate = tl.load(
        left_gate_ptr
        + off_batch[:, None] * stride_gate_b
        + off_h[None, :]    * stride_gate_h,
        mask=mask,
        other=0.0,
    )
    right_gate = tl.load(
        right_gate_ptr
        + off_batch[:, None] * stride_gate_b
        + off_h[None, :]    * stride_gate_h,
        mask=mask,
        other=0.0,
    )
    # optional mask (broadcasted over H)
    if mask_ptr != 0:
        mask_val = tl.load(
            mask_ptr
            + off_batch * stride_mask_b,
            mask=mask_batch,
            other=1.0,
        )
        mask_val = mask_val[:, None]                # (BATCH,1)
    else:
        mask_val = 1.0

    # --------------------------- compute ------------------------------
    left  = left  * left_gate  * mask_val
    right = right * right_gate * mask_val

    # --------------------------- store ---------------------------------
    tl.store(
        left_ptr
        + off_batch[:, None] * stride_left_b
        + off_h[None, :]    * stride_left_h,
        left,
        mask=mask,
    )
    tl.store(
        right_ptr
        + off_batch[:, None] * stride_right_b
        + off_h[None, :]    * stride_right_h,
        right,
        mask=mask,
    )


def _apply_mask_and_gates(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor | None,
) -> tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse mask and three gates with a tiny Triton kernel.

    All tensors are fp16, shape [B, N, N, H] (contiguous).
    Returns the in‑place‑updated left and right tensors.
    """
    B, N, _, H = left.shape
    total = B * N * N                     # flatten batch & two spatial dims
    # view as [total, H] for the kernel
    left_flat  = left.view(total, H).contiguous()
    right_flat = right.view(total, H).contiguous()
    left_gate_flat  = left_gate.view(total, H).contiguous()
    right_gate_flat = right_gate.view(total, H).contiguous()
    mask_flat = None
    if mask is not None:
        mask_flat = mask.view(total).contiguous().to(torch.float16)

    # strides in elements (after view)
    s_l_b, s_l_h = left_flat.stride()
    s_r_b, s_r_h = right_flat.stride()
    s_g_b, s_g_h = left_gate_flat.stride()
    s_m_b = 0
    if mask_flat is not None:
        s_m_b = mask_flat.stride(0)

    # launch parameters
    BLOCK_BATCH = 128          # number of (flattened) rows per program
    BLOCK_H = 64               # hidden dim tile size
    grid = (triton.cdiv(total, BLOCK_BATCH),)

    _mask_gate_kernel[grid](
        left_ptr=left_flat,
        right_ptr=right_flat,
        left_gate_ptr=left_gate_flat,
        right_gate_ptr=right_gate_flat,
        mask_ptr=mask_flat if mask_flat is not None else 0,
        BATCH=total,
        N=N,
        H=H,
        stride_left_b=s_l_b,
        stride_left_n=0,          # unused after flattening
        stride_left_n2=0,
        stride_left_h=s_l_h,
        stride_right_b=s_r_b,
        stride_right_n=0,
        stride_right_n2=0,
        stride_right_h=s_r_h,
        stride_gate_b=s_g_b,
        stride_gate_n=0,
        stride_gate_n2=0,
        stride_gate_h=s_g_h,
        stride_mask_b=s_m_b,
        stride_mask_n=0,
        stride_mask_n2=0,
        BLOCK_BATCH=BLOCK_BATCH,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return left, right


# ----------------------------------------------------------------------
# Batched GEMM via cuBLAS (torch.bmm) – the O(N³) part
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left : (B, H, I, K)   fp16, contiguous
    right: (B, H, K, J)   fp16, contiguous
    returns (B, H, I, J) fp16
    """
    B, H, I, K = left.shape
    _, _, _, J = right.shape
    # collapse batch*head
    left2  = left.reshape(B * H, I, K)
    right2 = right.reshape(B * H, K, J)
    out2   = torch.bmm(left2, right2)               # fp16 matmul → fp32 acc, then fp16 result
    return out2.view(B, H, I, J)


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors (see code)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (float32, torch implementation)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var  = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still fp32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (fp16) – left, right and three gates
    # ------------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16)

    left_proj_w  = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w  = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w   = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                     # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)            # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse mask + gates with Triton (in‑place on left/right)
    # ------------------------------------------------------------------
    left, right = _apply_mask_and_gates(
        left, right, left_gate, right_gate, mask
    )   # both remain fp16

    # ------------------------------------------------------------------
    # 4️⃣ Heavy contraction – batched GEMM (cuBLAS)
    #    reshape to (B, H, I, K) and (B, H, K, J)
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _batched_gemm_fp16(left_perm, right_perm)   # (B, H, I, J)

    # back to (B, N, N, dim) and to float32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 6️⃣ Final gating + projection (float32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over hidden dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                  # (B,N,N,dim)

    return out
