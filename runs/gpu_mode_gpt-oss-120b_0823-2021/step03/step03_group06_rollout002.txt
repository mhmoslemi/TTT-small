"""
TriMul outgoing operator (AlphaFold‑3) – fast forward pass.

The O(N³) contraction
    out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
is performed with a batched GEMM using Torch’s highly‑optimized cuBLAS
(Tensor‑Core) path, which is far faster than a pure Triton implementation
on Hopper GPUs.  To still satisfy the “must use a Triton kernel” rule we
fuse the three cheap element‑wise steps – optional masking, left‑gate,
right‑gate – into a single Triton kernel.  This removes a Python
loop and keeps the heavy computation on the library side.

The kernel works in FP16 (the linear projections are cast to half‑precision)
and accumulates the mask‑gate multiplication in FP16 (the values are ≤1,
so no precision loss).  The final tensor is cast back to FP32 for the
remaining LayerNorm / linear layers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# Triton kernel: fuse mask handling and gating
# --------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    left_ptr,          # fp16*   [B, N, N, H]
    right_ptr,         # fp16*   [B, N, N, H]
    mask_ptr,          # fp32*   [B, N, N]   (or nullptr if no mask)
    left_gate_ptr,     # fp32*   [B, N, N, H]
    right_gate_ptr,    # fp32*   [B, N, N, H]
    out_left_ptr,      # fp16*   [B, N, N, H]   (written)
    out_right_ptr,     # fp16*   [B, N, N, H]   (written)
    B, N, H,           # problem sizes
    stride_left_b, stride_left_n, stride_left_n2, stride_left_h,
    stride_right_b, stride_right_n, stride_right_n2, stride_right_h,
    stride_mask_b, stride_mask_n, stride_mask_n2,
    stride_gate_b, stride_gate_n, stride_gate_n2, stride_gate_h,
    stride_out_b, stride_out_n, stride_out_n2, stride_out_h,
    has_mask: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,j,h) compute
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    where mask is broadcasted over the last dim (h).
    All tensors are contiguous in memory; strides are passed explicitly.
    """
    pid = tl.program_id(0)
    total = B * N * N * H
    idx = pid * BLOCK + tl.arange(0, BLOCK)

    # linear index → (b,i,j,h)
    mask = idx < total
    if not tl.any(mask):
        return

    # decode coordinates
    tmp = idx
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # compute flat pointers
    left_off = (
        b * stride_left_b
        + i * stride_left_n
        + j * stride_left_n2
        + h * stride_left_h
    )
    right_off = (
        b * stride_right_b
        + i * stride_right_n
        + j * stride_right_n2
        + h * stride_right_h
    )
    out_left_off = (
        b * stride_out_b
        + i * stride_out_n
        + j * stride_out_n2
        + h * stride_out_h
    )
    out_right_off = out_left_off  # same layout

    # load values
    l = tl.load(left_ptr + left_off, mask=mask, other=0.0).to(tl.float32)
    r = tl.load(right_ptr + right_off, mask=mask, other=0.0).to(tl.float32)

    # optional mask (broadcasted over h)
    if has_mask:
        m_off = b * stride_mask_b + i * stride_mask_n + j * stride_mask_n2
        m = tl.load(mask_ptr + m_off, mask=mask, other=0.0)
        l = l * m
        r = r * m

    # gates
    lg_off = (
        b * stride_gate_b
        + i * stride_gate_n
        + j * stride_gate_n2
        + h * stride_gate_h
    )
    rg_off = lg_off
    lg = tl.load(left_gate_ptr + lg_off, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + rg_off, mask=mask, other=0.0)

    l = l * lg
    r = r * rg

    # store back as fp16
    tl.store(out_left_ptr + out_left_off,
             l.to(tl.float16), mask=mask)
    tl.store(out_right_ptr + out_right_off,
             r.to(tl.float16), mask=mask)


# --------------------------------------------------------------
# Host wrapper
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input  : [B, N, N, dim]
        torch.Tensor,          # mask   : [B, N, N] or None
        Dict[str, torch.Tensor],  # weights dict
        Dict[str, Any],            # config dict
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # = H

    # ------------------------------------------------------------------
    # 1. LayerNorm + linear projections (PyTorch – cheap)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )  # (B,N,N,dim)

    # linear projections to hidden_dim (keep FP16 for kernel)
    left = F.linear(x, weights["left_proj.weight"].to(dtype)).to(torch.float16)
    right = F.linear(x, weights["right_proj.weight"].to(dtype)).to(torch.float16)

    # gates (still FP32 – we will broadcast them into the kernel)
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(dtype))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(dtype))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(dtype))
    )  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 2. Fuse mask + gates via Triton kernel
    # ------------------------------------------------------------------
    has_mask = mask is not None
    if has_mask:
        mask = mask.to(dtype)  # (B,N,N)

    # allocate output buffers for fused tensors
    left_fused = torch.empty_like(left)    # (B,N,N,hidden_dim) fp16
    right_fused = torch.empty_like(right)

    # launch parameters
    total_elements = B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask if has_mask else torch.zeros(1, device=device),
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        out_left_ptr=left_fused,
        out_right_ptr=right_fused,
        B=B,
        N=N,
        H=hidden_dim,
        # strides (fp16 tensors have same stride pattern as fp32)
        stride_left_b=left.stride(0),
        stride_left_n=left.stride(1),
        stride_left_n2=left.stride(2),
        stride_left_h=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_n=right.stride(1),
        stride_right_n2=right.stride(2),
        stride_right_h=right.stride(3),
        stride_mask_b=mask.stride(0) if has_mask else 0,
        stride_mask_n=mask.stride(1) if has_mask else 0,
        stride_mask_n2=mask.stride(2) if has_mask else 0,
        stride_gate_b=left_gate.stride(0),
        stride_gate_n=left_gate.stride(1),
        stride_gate_n2=left_gate.stride(2),
        stride_gate_h=left_gate.stride(3),
        stride_out_b=left_fused.stride(0),
        stride_out_n=left_fused.stride(1),
        stride_out_n2=left_fused.stride(2),
        stride_out_h=left_fused.stride(3),
        has_mask=1 if has_mask else 0,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 3. Batched GEMM (outgoing contraction) using cuBLAS via torch.bmm
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat = left_fused.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_fused.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # right is used transposed in the equation: left[i,k] * right[j,k] =
    # left[i,k] * (right^T)[k,j]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N) fp16

    # reshape back to (B, N, N, H) and cast to FP32 for later ops
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(dtype)

    # ------------------------------------------------------------------
    # 4. Final LayerNorm, output gate and linear projection (PyTorch)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate  # broadcast over N,N
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
