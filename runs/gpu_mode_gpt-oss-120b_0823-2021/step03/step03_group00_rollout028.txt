"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward only)
------------------------
1️⃣ Layer‑norm the 4‑D input tensor (float32, simple PyTorch reduction).
2️⃣ Five linear projections (left, right, left‑gate, right‑gate, out‑gate) are
   performed in fp16 for speed.
3️⃣ Optional binary mask is applied to the left/right projections with a tiny
   Triton element‑wise kernel (no branching, only a single load/store per
   element).
4️⃣ Sigmoid gating of the three gate tensors.
5️⃣ Core contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .
   This is a batched GEMM of shape (B·H, N, N)·(B·H, N, N)ᵀ → (B·H, N, N) and
   is delegated to cuBLAS via ``torch.bmm`` – the dominant O(N³) work is thus
   executed by the highly‑optimised Tensor‑Core implementation on H200.
6️⃣ Output LayerNorm (float32), multiplication by the output‑gate and final
   linear projection back to the original dimension.
The only Triton code required is the mask‑apply kernel; everything else stays
in PyTorch so the implementation is concise, correct and fast on the target
hardware.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – apply a broadcast mask to a 4‑D tensor
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    src_ptr,               # input  (B, N, N, H) fp16
    mask_ptr,              # mask   (B, N, N, 1) fp16
    dst_ptr,               # output (B, N, N, H) fp16
    B, N, H,               # extents
    stride_bs, stride_bn, stride_bk, stride_bc,   # src strides
    stride_ms, stride_mn, stride_mk, stride_mc,   # mask strides
    stride_ds, stride_dn, stride_dk, stride_dc,   # dst strides
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # linear program id over all elements
    linear_idx = pid * BLOCK + tl.arange(0, BLOCK)

    # compute 4‑D indices
    b = linear_idx // (N * N * H)
    rest = linear_idx % (N * N * H)
    i = rest // (N * H)
    rest = rest % (N * H)
    j = rest // H
    h = rest % H

    mask = tl.load(mask_ptr + b * stride_ms + i * stride_mn + j * stride_mk + h * stride_mc,
                   mask=(b < B) & (i < N) & (j < N) & (h < H), other=0.0)
    src  = tl.load(src_ptr  + b * stride_bs + i * stride_bn + j * stride_bk + h * stride_bc,
                   mask=(b < B) & (i < N) & (j < N) & (h < H), other=0.0)

    tl.store(dst_ptr + b * stride_ds + i * stride_dn + j * stride_dk + h * stride_dc,
             src * mask,
             mask=(b < B) & (i < N) & (j < N) & (h < H))


def _apply_mask_fp16(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    tensor : (B,N,N,H) fp16
    mask   : (B,N,N)   bool/float16  (will be broadcasted over H)
    returns tensor * mask (fp16)
    """
    B, N, _, H = tensor.shape
    # ensure mask has a trailing dimension
    mask = mask.to(tensor.dtype).unsqueeze(-1)          # (B,N,N,1)

    out = torch.empty_like(tensor)

    BLOCK = 1024
    total_elements = B * N * N * H
    grid = (triton.cdiv(total_elements, BLOCK),)

    _apply_mask_kernel[grid](
        tensor,
        mask,
        out,
        B, N, H,
        *tensor.stride(),
        *mask.stride(),
        *out.stride(),
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Helper: batched GEMM for the heavy contraction (cuBLAS)
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, N, N) fp16
    right : (B, H, N, N) fp16   (note: right already transposed on K‑J axis)
    returns (B, H, N, N) fp16
    """
    B, H, N, _ = left.shape
    # collapse batch*head
    left_flat  = left.reshape(B * H, N, N)          # (B*H, N, K)
    right_flat = right.reshape(B * H, N, N)         # (B*H, K, N) – already K,J
    # torch.bmm uses cuBLAS/Tensor‑cores
    out_flat = torch.bmm(left_flat, right_flat)    # (B*H, N, N)
    return out_flat.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool or float)
        weights      : dict of leaf tensors (see code below)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # unpack & basic shapes
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32, manual)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var  = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still fp32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections in fp16
    # ------------------------------------------------------------------
    # cast once
    def _fp16(t): return t.to(torch.float16)
    left_proj_w   = _fp16(weights["left_proj.weight"])
    right_proj_w  = _fp16(weights["right_proj.weight"])
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                     # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)            # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)           # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask – Triton kernel
    # ------------------------------------------------------------------
    if mask is not None:
        left  = _apply_mask_fp16(left,  mask)
        right = _apply_mask_fp16(right, mask)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction – reshape and cuBLAS batched GEMM
    #    left  : (B,N,N,hidden) → (B, hidden, N, N)
    #    right : (B,N,N,hidden) → (B, hidden, N, N) and transpose K↔J axis
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _batched_gemm_fp16(left_perm, right_perm)   # (B, H, N, N)

    # back to (B,N,N,dim) and float32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + projection (float32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                  # (B,N,N,dim)

    return out
