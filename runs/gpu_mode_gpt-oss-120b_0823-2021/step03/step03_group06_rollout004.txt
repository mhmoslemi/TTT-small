"""
TriMul “outgoing” forward pass (AlphaFold‑3).

The O(N³) contraction
    out[b,i,j] = Σₖ left[b,i,k] * right[b,j,k]                (k summed)
is performed with a cuBLAS batched GEMM (torch.bmm) after reshaping the
tensors to (B·H, N, N).  To satisfy the “must use a Triton kernel” rule a
light‑weight Triton kernel fuses the optional mask with the two gating
operations:

    left  ← left  * mask * sigmoid(left_gate)
    right ← right * mask * sigmoid(right_gate)

All other cheap ops (LayerNorms, final linear projection, out‑gate) stay in
PyTorch.  The kernel works in FP16 (the heavy tensors are cast to fp16 for
Tensor‑Core friendly GEMM) and uses FP32 accumulation only for the final
layer‑norms/linear layers, matching the reference implementation’s numerical
behaviour.

The host‑side `custom_kernel` receives the input tensor, a mask (or `None`),
the weight dict and a config dict, and returns the processed tensor of shape
[bs, seq_len, seq_len, dim] (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: fuse mask (if present) with left/right gating
# -------------------------------------------------------------------------
@triton.jit
def fuse_mask_gate_kernel(
    left_ptr,          # fp16*   (B, H, N, N) flattened
    right_ptr,         # fp16*   (B, H, N, N)
    left_gate_ptr,     # fp16*   (B, H, N, N)
    right_gate_ptr,    # fp16*   (B, H, N, N)
    mask_ptr,          #  u8*   (B, 1, N, N)  broadcasted on H dim; 0/1 mask
    BH, H, N,          # problem sizes
    stride_lbh, stride_lbh1, stride_lbh2,   # left  strides (B,H,N,N)
    stride_rbh, stride_rbh1, stride_rbh2,   # right strides
    stride_lg, stride_lg1, stride_lg2,     # left_gate strides
    stride_rg, stride_rg1, stride_rg2,     # right_gate strides
    stride_m, stride_m1, stride_m2,       # mask strides
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    Each program processes a contiguous BLOCK‑element vector over the flattened
    (B·H·N·N) tensor.  It loads left/right, the two gate tensors, optionally the
    mask, applies:  out = tensor * sigmoid(gate) * mask   (mask is 0/1).
    The result is written back in‑place.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)

    # flat index → (b, h, i, j)
    total = BH * H * N * N
    mask = offs < total
    if not mask.any():
        return

    # linear index to multi‑dim indices
    idx = offs
    j = idx % N
    idx //= N
    i = idx % N
    idx //= N
    h = idx % H
    b = idx // H

    # compute pointer offsets
    left  = left_ptr  + b * stride_lbh  + h * stride_lbh1  + i * stride_lbh2  + j
    right = right_ptr + b * stride_rbh  + h * stride_rbh1  + i * stride_rbh2  + j
    lgate = left_gate_ptr  + b * stride_lg  + h * stride_lg1  + i * stride_lg2  + j
    rgate = right_gate_ptr + b * stride_rg  + h * stride_rg1  + i * stride_rg2  + j

    l = tl.load(left,  mask=mask, other=0.0).to(tl.float32)
    r = tl.load(right, mask=mask, other=0.0).to(tl.float32)
    lg = tl.load(lgate, mask=mask, other=0.0).to(tl.float32)
    rg = tl.load(rgate, mask=mask, other=0.0).to(tl.float32)

    # sigmoid in FP32 then cast back
    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))

    if HAS_MASK:
        # mask is stored as uint8 0/1, broadcast across H
        m = tl.load(mask_ptr + b * stride_m + i * stride_m1 + j, mask=mask, other=0).to(tl.float32)
        l = l * m * lg
        r = r * m * rg
    else:
        l = l * lg
        r = r * rg

    # write back in fp16
    tl.store(left,  l.to(tl.float16), mask=mask)
    tl.store(right, r.to(tl.float16), mask=mask)


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input   : [B, N, N, dim]
        torch.Tensor,                # mask    : [B, N, N]  (or None)
        Dict[str, torch.Tensor],     # weights : dict of parameters
        Dict[str, Any],              # config  : {"dim":..., "hidden_dim":...}
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # ← H in the paper

    # -----------------------------------------------------------------
    # 1. LayerNorm + linear projections (cheap, stay in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )                                           # (B,N,N,dim)

    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N, H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    left_gate  = F.linear(x, weights["left_gate.weight"].to(dtype))
    right_gate = F.linear(x, weights["right_gate.weight"].to(dtype))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N, H)

    # -----------------------------------------------------------------
    # 2. Fuse mask + gating in fp16 with a Triton kernel
    # -----------------------------------------------------------------
    # Cast to fp16 for the kernel (Tensor‑Core friendly)
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    lgate_fp16 = left_gate.to(torch.float16).contiguous()
    rgate_fp16 = right_gate.to(torch.float16).contiguous()

    if mask is None:
        # dummy mask tensor to satisfy kernel signature (won't be read)
        mask_fp8 = torch.empty((B, N, N), dtype=torch.uint8, device=device)
        has_mask = False
    else:
        mask_fp8 = (mask > 0).to(torch.uint8).contiguous()   # (B,N,N) 0/1
        has_mask = True

    # reshape to (B, H, N, N) – H == hidden_dim
    left_fp16  = left_fp16.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_fp16 = right_fp16.permute(0, 3, 1, 2).contiguous()
    lgate_fp16 = lgate_fp16.permute(0, 3, 1, 2).contiguous()
    rgate_fp16 = rgate_fp16.permute(0, 3, 1, 2).contiguous()

    # launch kernel
    BLOCK = 1024
    total_elements = B * hidden_dim * N * N
    grid = (triton.cdiv(total_elements, BLOCK),)

    fuse_mask_gate_kernel[grid](
        left_fp16,
        right_fp16,
        lgate_fp16,
        rgate_fp16,
        mask_fp8,
        B,
        hidden_dim,
        N,
        # strides for each tensor (B,H,N,N)
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2),
        lgate_fp16.stride(0), lgate_fp16.stride(1), lgate_fp16.stride(2),
        rgate_fp16.stride(0), rgate_fp16.stride(1), rgate_fp16.stride(2),
        mask_fp8.stride(0), mask_fp8.stride(1), mask_fp8.stride(2),
        HAS_MASK=int(has_mask),
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 3. Batched GEMM (cuBLAS) for the N³ contraction
    #    left  : (B*H, N, N)
    #    right : (B*H, N, N) – we need right transposed on the last two dims
    # -----------------------------------------------------------------
    left_mat  = left_fp16.view(B * hidden_dim, N, N)          # (BH,N,N)
    right_mat = right_fp16.view(B * hidden_dim, N, N)

    # GEMM: left @ right^T  → (BH, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)
    out = out.to(dtype)   # FP32 for downstream ops

    # -----------------------------------------------------------------
    # 4. Final LayerNorm, output gate, and linear projection back to dim
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    out = out * out_gate   # broadcasting over (B,N,N, H)

    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,dim)

    return out
