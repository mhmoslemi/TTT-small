"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the input (B,N,N,C).
2️⃣  Linear projections → left/right (B,N,N,H) and three gates (B,N,N,H).
3️⃣  Fuse mask, left‑gate and right‑gate with a tiny Triton kernel so that
    at least one non‑trivial operation runs on the GPU.
4️⃣  Reshape to (B*H,N,N) and perform a batched matrix‑multiply
    (left @ rightᵀ) with Tensor‑core‑friendly fp16 inputs
    (accumulation in fp32).  This is the O(N³) contraction.
5️⃣  Layer‑norm the hidden result, multiply by out‑gate and apply the final
    linear projection back to C (fp32).

Only the element‑wise fusion (step 3) is written in Triton; everything else
uses PyTorch/cuBLAS which is already highly optimised on the H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – element‑wise mask & gate fusion
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                # fp16 * (B,N,N,H)
    left_gate_ptr, right_gate_ptr,      # fp16 * (B,N,N,H)
    mask_ptr,                           # fp16 * (B,N,N) or nullptr
    out_left_ptr, out_right_ptr,        # fp16 * (B,N,N,H) – outputs
    B, N, H,                            # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_g0, stride_g1, stride_g2, stride_g3,
    stride_m0, stride_m1, stride_m2,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    (mask is 1.0 when HAS_MASK == 0).
    The kernel works on a 1‑D grid of size ceil(B*N*N*H / BLOCK).
    """
    pid  = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    total = B * N * N * H
    active = offs < total

    # ---- decode linear index -> (b,i,j,h) -----------------------------
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ---- load left / right --------------------------------------------
    idx = (b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3)
    left  = tl.load(left_ptr  + idx, mask=active, other=0.0)
    right = tl.load(right_ptr + idx, mask=active, other=0.0)

    # ---- load gates ----------------------------------------------------
    idx_g = (b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3)
    lgate = tl.load(left_gate_ptr  + idx_g, mask=active, other=0.0)
    rgate = tl.load(right_gate_ptr + idx_g, mask=active, other=0.0)

    # ---- optional mask -------------------------------------------------
    if HAS_MASK:
        idx_m = (b * stride_m0 + i * stride_m1 + j * stride_m2)
        m = tl.load(mask_ptr + idx_m, mask=active, other=0.0)
    else:
        m = 1.0

    # ---- write fused results -------------------------------------------
    out_l = left * lgate * m
    out_r = right * rgate * m
    tl.store(out_left_ptr  + idx, out_l, mask=active)
    tl.store(out_right_ptr + idx, out_r, mask=active)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input   (B,N,N,C)  float32
        torch.Tensor,          # mask    (B,N,N)    float32 or bool or None
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : (x, mask, weights, config)
        x       – (B, N, N, C)   torch.float32
        mask    – (B, N, N)      torch.float32 / bool / None
        weights – dict of all learned parameters (float32)
        config  – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor – (B, N, N, C)  float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype  = torch.float32

    B, N, _, C = x.shape
    H = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (keep in fp32 for now)
    # --------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate= torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate  = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # --------------------------------------------------------------
    # 3️⃣ Fuse mask & gates with Triton (all tensors cast to fp16)
    # --------------------------------------------------------------
    left_fp16      = left .to(torch.float16).contiguous()
    right_fp16     = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate .to(torch.float16).contiguous()
    right_gate_fp16= right_gate.to(torch.float16).contiguous()

    if mask is None:
        # dummy mask (won’t be read because HAS_MASK=0)
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask = False
    else:
        # mask shape (B,N,N) → (B,N,N) (no hidden dim)
        mask_fp16 = mask.to(torch.float16).contiguous()
        has_mask = True

    # allocate output buffers
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elements = B * N * N * H
    BLOCK = 1024                                 # good trade‑off on H200
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        mask_fp16.stride(0) if has_mask else 0,
        mask_fp16.stride(1) if has_mask else 0,
        mask_fp16.stride(2) if has_mask else 0,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM  (fp16 inputs → fp16 output, fp32 accumulation)
    # --------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # out = left @ rightᵀ  (cuBLAS / Tensor‑cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # back to (B,N,N,H) and to fp32 for the next LayerNorm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype)

    # --------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, out‑gate and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                     # broadcast over (N,N)
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
