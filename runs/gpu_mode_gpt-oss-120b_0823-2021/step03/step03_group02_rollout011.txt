"""
TriMul “outgoing” forward (no gradients).

The dominant N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a batched GEMM on fp16 tensors inside a small
Triton kernel (fp16‑input, fp32‑accumulation, fp16‑output).
All inexpensive element‑wise work (layer‑norms, linear projections,
gating) stays in PyTorch.  The final tensor is converted back to fp32,
layer‑normed, gated and projected to the original hidden dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (H×I×K) · (H×K×J) → (H×I×J)   (fp16 → fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,            # left  (Z, I, K)   fp16
    b_ptr,            # right (Z, K, J)   fp16
    c_ptr,            # out   (Z, I, J)   fp16
    M, N, K,          # matrix sizes
    stride_az, stride_am, stride_ak,   # a strides (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)          # batch*head dimension
    pid_m = tl.program_id(1)          # tile rows   (I)
    pid_n = tl.program_id(2)          # tile cols   (J)

    # start offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 for numeric stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile (K x N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # accumulate
        acc += tl.dot(a, b)

    # write result back as fp16
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_batch_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.
    Shapes:
        left  : (B, H, I, K)   fp16, contiguous
        right : (B, H, K, J)   fp16, contiguous
    Returns (B, H, I, J) fp16.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse batch*head dimension for the kernel launch
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling parameters – good for Hopper/Tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                                   # program_id(0)  -> batch*head
        triton.cdiv(I, BLOCK_M),            # program_id(1)  -> rows
        triton.cdiv(J, BLOCK_N),            # program_id(2)  -> cols
    )

    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch for cheap ops + Triton for heavy contract)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of “outgoing” TriMul.

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        mask         : torch.Tensor [B, N, N]        (float32/bool) or None
        weights      : dict of torch.Tensor parameters (see code)
        config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]  (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1) Input LayerNorm (keep in fp32)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                         # [B,N,N,dim] fp32

    # --------------------------------------------------------------
    # 2) Linear projections + gating (cast to fp16 once)
    # --------------------------------------------------------------
    def _to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w   = _to_fp16(weights["left_proj.weight"])   # (dim, hidden)
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])

    to_out_norm_w = weights["to_out_norm.weight"]          # stays fp32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]              # (hidden, dim)

    x_fp16 = x.to(torch.float16)

    # projections → (B,N,N,hidden)
    left  = F.linear(x_fp16, left_proj_w)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4) Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Core contraction – batched GEMM via Triton
    #    left : (B,H,N,N) → (B,H,N,N)   (H = hidden_dim)
    #    right: (B,H,N,N) → (B,H,N,N)   (transpose K‑axis)
    # --------------------------------------------------------------
    # bring hidden dim to axis 1 and make reduction axis contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _triton_batch_gemm(left_perm, right_perm)   # (B, H, I, J) fp16

    # back to original layout (B, N, N, hidden)
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6) Output LayerNorm
    # --------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b          # fp32

    # --------------------------------------------------------------
    # 7) Final gating + linear projection back to dim
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)             # apply same out_gate
    out = F.linear(out, to_out_w)                     # (B,N,N,dim)

    return out
