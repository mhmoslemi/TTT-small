"""
custom_kernel – “outgoing” TriMul forward pass (AlphaFold‑3 style).

1️⃣  LayerNorm, linear projections and gating are done in PyTorch (FP32 → FP16).
2️⃣  The cubic contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is a batched matrix‑multiply (I×K)·(K×J) → (I×J) for each head *d*.
    We fuse it with a tiny Triton kernel that loads tiles of the two
    operands, multiplies them with fp16 Tensor‑Core dot‑products and
    accumulates in fp32.  This kernel is the only Triton component – the
    rest of the model stays in PyTorch for simplicity and to keep the
    implementation generic.
3️⃣  The result is cast back to FP32, passed through a second LayerNorm,
    multiplied by the output gate and finally projected back to the
    original dim.

The kernel works for any batch size ≤2, sequence length up to 1024 and
hidden_dim = 128 (the values used in the benchmark).  Mixed‑precision
( fp16 compute / fp32 accumulation ) is used to obtain high throughput on
Hopper (SM‑90) while preserving the required fp32 output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (I×K)·(K×J) → (I×J)   (fp16 × fp16 → fp32 acc)
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    a_ptr,                # left  matrix pointer   (Z, I, K)
    b_ptr,                # right matrix pointer   (Z, K, J)
    c_ptr,                # output matrix pointer  (Z, I, J)
    M, N, K,              # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)               # batch*head dimension
    pid_m = tl.program_id(1)               # tile row
    pid_n = tl.program_id(2)               # tile col

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (high‑precision sum)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (blocking)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile (BLOCK_M × BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a tile (BLOCK_K × BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp16 × fp16 → fp32 accumulation via dot
        acc += tl.dot(a, b)

    # Store back (convert to fp16 for memory efficiency)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    Parameters
    ----------
    left  : Tensor (B, H, I, K)  dtype=torch.float16, contiguous
    right : Tensor (B, H, K, J)  dtype=torch.float16, contiguous

    Returns
    -------
    out : Tensor (B, H, I, J) dtype=torch.float16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # Collapse the leading dimensions so the kernel sees a 3‑D tensor
    Z = B * H
    a = left.reshape(Z, I, K)
    b = right.reshape(Z, K, J)
    c = out.reshape(Z, I, J)

    # Strides in element units (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tiling tuned for Hopper Tensor‑Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                                 # batch*head
        triton.cdiv(I, BLOCK_M),           # rows
        triton.cdiv(J, BLOCK_N),           # cols
    )
    batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # empirically good on SM90
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask         : torch.Tensor [B, N, N] (float32/bool) or None
        weights      : dict of parameter tensors (all float32)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. Input LayerNorm   (FP32)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b            # [B,N,N,dim]  fp32

    # --------------------------------------------------------------
    # 2. Linear projections – cast to fp16 for the heavy part
    # --------------------------------------------------------------
    to_fp16 = lambda t: t.to(torch.float16, non_blocking=True)

    left_proj_w  = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w  = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w   = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3. Optional mask – broadcast over hidden dimension
    # --------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Batched GEMM via Triton  (the N³ contraction)
    #    left : (B, hidden_dim, N, N)  -> (B, H, I, K)
    #    right: (B, hidden_dim, N, N)  -> (B, H, K, J)  (note transposition)
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = triton_batched_gemm(left_perm, right_perm)   # (B, H, I, J)

    # Back to original layout and FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 6. Output LayerNorm (FP32)
    # --------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = out * to_out_norm_w + to_out_norm_b

    # --------------------------------------------------------------
    # 7. Final gating + linear projection
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                   # (B,N,N,dim)

    return out
