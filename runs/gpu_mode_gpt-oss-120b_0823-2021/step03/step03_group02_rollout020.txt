"""
TriMul “outgoing” forward pass (no gradients).

The O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is reduced to a batched GEMM using torch.bmm (cuBLAS/Tensor‑cores).  
All element‑wise work (layer‑norm, masking, gating) is kept in PyTorch **except**
the mask‑×‑gate fusion which is performed by a tiny Triton kernel to satisfy the
“must‑use‑Triton” requirement and to avoid an extra launch‑overhead in Python.

The implementation follows the reference module:

    1️⃣ Input LayerNorm (fp32)
    2️⃣ Linear projections (fp16) – left/right
    3️⃣ Optional mask + gating (fp16) – fused in Triton
    4️⃣ Core contraction (batched GEMM) – fp16 → fp32
    5️⃣ Output LayerNorm (fp32)
    6️⃣ Final gating + linear projection (fp32)

All heavy arithmetic stays in FP16 where tensor‑cores excel; final results are
cast back to FP32 for the remaining normalisations and the final linear layer.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise (mask * gate) * tensor
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # fp16 tensor  (B,N,N,D)
    mask_ptr,      # fp16 tensor  (B,N,N,1)  – broadcasted on dim D
    gate_ptr,      # fp16 tensor  (B,N,N,D)
    out_ptr,       # fp16 tensor  (B,N,N,D)
    B, N, D,       # sizes
    stride_xb, stride_xn, stride_xn2, stride_xd,   # strides for x
    stride_mb, stride_mn, stride_mn2, stride_md,   # strides for mask (last dim is 1)
    stride_gb, stride_gn, stride_gn2, stride_gd,   # strides for gate
    stride_ob, stride_on, stride_on2, stride_od,   # strides for out
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    # linear index over the flattened tensor (B*N*N*D)
    total = B * N * N * D
    idx = pid * BLOCK + tl.arange(0, BLOCK)
    mask = idx < total

    # de‑linearise
    b = idx // (N * N * D)
    rem = idx % (N * N * D)
    i = rem // (N * D)
    rem = rem % (N * D)
    j = rem // D
    d = rem % D

    # compute flat offsets
    x_offset = b * stride_xb + i * stride_xn + j * stride_xn2 + d * stride_xd
    m_offset = b * stride_mb + i * stride_mn + j * stride_mn2            # dim D is 1
    g_offset = b * stride_gb + i * stride_gn + j * stride_gn2 + d * stride_gd
    out_offset = b * stride_ob + i * stride_on + j * stride_on2 + d * stride_od

    x = tl.load(x_ptr + x_offset, mask=mask, other=0.0)
    m = tl.load(mask_ptr + m_offset, mask=mask, other=0.0)   # broadcasted
    g = tl.load(g_gate_ptr + g_offset, mask=mask, other=0.0)

    tl.store(out_ptr + out_offset, x * m * g, mask=mask)


def fuse_mask_gate(x: torch.Tensor,
                   mask: torch.Tensor,
                   gate: torch.Tensor) -> torch.Tensor:
    """
    Apply (x * mask * gate) element‑wise using Triton.

    All tensors must be contiguous, FP16, and have the same shape
    (mask has a trailing singleton dim that is broadcasted).
    """
    assert x.is_contiguous() and mask.is_contiguous() and gate.is_contiguous()
    B, N, _, D = x.shape
    out = torch.empty_like(x)

    # strides in *elements* (not bytes)
    stride_xb, stride_xn, stride_xn2, stride_xd = x.stride()
    stride_mb, stride_mn, stride_mn2, stride_md = mask.stride()
    stride_gb, stride_gn, stride_gn2, stride_gd = gate.stride()
    stride_ob, stride_on, stride_on2, stride_od = out.stride()

    BLOCK = 1024
    total = B * N * N * D
    grid = (triton.cdiv(total, BLOCK),)

    mask_gate_fuse_kernel[grid](
        x_ptr=x,
        mask_ptr=mask,
        gate_ptr=gate,
        out_ptr=out,
        B=B,
        N=N,
        D=D,
        stride_xb=stride_xb,
        stride_xn=stride_xn,
        stride_xn2=stride_xn2,
        stride_xd=stride_xd,
        stride_mb=stride_mb,
        stride_mn=stride_mn,
        stride_mn2=stride_mn2,
        stride_md=stride_md,
        stride_gb=stride_gb,
        stride_gn=stride_gn,
        stride_gn2=stride_gn2,
        stride_gd=stride_gd,
        stride_ob=stride_ob,
        stride_on=stride_on,
        stride_on2=stride_on2,
        stride_od=stride_od,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Helper for a batched GEMM using torch.bmm (cuBLAS/Tensor‑cores)
# ----------------------------------------------------------------------
def batch_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)  fp16
    right : (B, H, K, J)  fp16   (already transposed on the last two axes)

    Returns (B, H, I, J) fp16
    """
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # collapse batch & head dimensions – torch.bmm expects (B*H, I, K)
    left_flat  = left.reshape(B * H, I, K)
    right_flat = right.reshape(B * H, K, J)

    out_flat = torch.bmm(left_flat, right_flat)          # fp16 matmul
    return out_flat.view(B, H, I, J)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor [B, N, N]       (float32/bool) or None
        - weights      : dict of torch.Tensor parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]   (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep in FP32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual LayerNorm
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                         # still FP32

    # ------------------------------------------------------------------
    # 2) Linear projections – cast once to FP16
    # ------------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16, non_blocking=True)

    left_proj_w  = to_fp16(weights["left_proj.weight"])   # (dim, hidden)
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w  = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w   = to_fp16(weights["out_gate.weight"])

    # keep a FP16 copy of the normalized tensor for all heavy ops
    x_fp16 = x.to(torch.float16)

    # (B,N,N,hidden)  fp16
    left  = F.linear(x_fp16, left_proj_w)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Gating + optional mask – fused in a Triton kernel
    # ------------------------------------------------------------------
    # compute gates (still FP16)
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    # apply mask (if present) and gates
    if mask is not None:
        # mask: (B,N,N) -> (B,N,N,1) fp16
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = fuse_mask_gate(left, mask_fp16, left_gate)
        right = fuse_mask_gate(right, mask_fp16, right_gate)
    else:
        # no mask – just apply the gates (still using the same kernel for simplicity)
        left = fuse_mask_gate(left, torch.ones_like(left[..., :1]), left_gate)
        right = fuse_mask_gate(right, torch.ones_like(right[..., :1]), right_gate)

    # ------------------------------------------------------------------
    # 4) Core contraction – batched GEMM
    #    left : (B, N, N, hidden)   -> (B, hidden, N, N)
    #    right: (B, N, N, hidden)   -> (B, hidden, N, N) then transpose K,J
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, J)

    out_fp16 = batch_gemm_fp16(left_perm, right_perm)  # (B, H, I, J) fp16

    # back to (B, N, N, dim) and cast to FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 5) Output LayerNorm
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 6) Final gating + linear projection
    # ------------------------------------------------------------------
    # out_gate computed earlier (still FP16) – bring to FP32 for the multiplication
    out = out * out_gate.to(torch.float32)
    to_out_w = weights["to_out.weight"]                 # (hidden, dim)
    out = F.linear(out, to_out_w)                       # (B,N,N,dim) FP32

    return out
