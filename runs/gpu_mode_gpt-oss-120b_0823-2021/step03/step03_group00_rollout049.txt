"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward only)
------------------------
1️⃣  Layer‑norm the 4‑D tensor (fp32, manual – mean/var over the last dim).
2️⃣  Linear projections + gating are performed in fp16 for speed.
3️⃣  A tiny Triton kernel fuses
        • optional binary mask   (broadcast over the hidden dim)
        • element‑wise multiplication with the three gate tensors
   This eliminates three separate PyTorch kernels and keeps the
   memory traffic inside a single kernel launch.
4️⃣  The expensive cubic contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   is reshaped into a batched matrix‑multiply
        (B·H, N, N) @ (B·H, N, N)ᵀ  → (B·H, N, N)
   and executed with torch.bmm which uses cuBLAS/Tensor‑cores on the
   H200 – far faster than a hand‑written Triton GEMM.
5️⃣  The result is reshaped back, a second Layer‑norm (fp32) is applied,
    multiplied by the output‑gate and finally projected back to the
    original dimension with a fp32 linear layer.

The only Triton‑implemented part is the element‑wise kernel (step 3); the
rest of the computation stays in PyTorch and therefore benefits from
high‑performance library calls.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – mask + gating fusion
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,     # fp16 left projection          (B,N,N,H)
    right_ptr,    # fp16 right projection         (B,N,N,H)
    left_gate_ptr,   # fp16 left gate                (B,N,N,H)
    right_gate_ptr,  # fp16 right gate               (B,N,N,H)
    out_gate_ptr,    # fp16 out gate                 (B,N,N,H)
    mask_ptr,    # fp16 mask (0/1) broadcasted   (B,N,N,1)  – may be nullptr
    B, N, H,    # static sizes
    stride_lbz, stride_lby, stride_lbx, stride_lbh,   # left strides
    stride_rbz, stride_rby, stride_rbx, stride_rbh,   # right strides
    stride_gbz, stride_gby, stride_gbx, stride_gbh,   # gate strides (same layout)
    stride_mbz, stride_mby, stride_mbx,               # mask strides (no H dim)
    BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    One program works on a tile (BLOCK_N × BLOCK_H) of the hidden dimension
    for a fixed (batch, row) pair.
    """
    pid = tl.program_id(0)               # linear index over (B,N) pairs
    batch = pid // N
    row   = pid %  N

    # tile coordinates inside the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    offs_n = tl.arange(0, BLOCK_N)

    # compute absolute indices
    offs_h = offs_h[None, :] + 0          # (1, BLOCK_H)
    offs_n = offs_n[:, None] + 0          # (BLOCK_N, 1)

    # masks for out‑of‑bounds in hidden dimension
    mask_h = offs_h < H
    mask_n = offs_n < N

    # full masks
    m_h = mask_h & mask_n

    # ------- load projections ------------------------------------------------
    # left : (B,N,N,H) → (row, col, hidden)
    left = tl.load(
        left_ptr
        + batch * stride_lbz
        + row   * stride_lby
        + offs_n * stride_lbx
        + offs_h * stride_lbh,
        mask=m_h,
        other=0.0,
    )   # (BLOCK_N, BLOCK_H)

    right = tl.load(
        right_ptr
        + batch * stride_rbz
        + row   * stride_rby
        + offs_n * stride_rbx
        + offs_h * stride_rbh,
        mask=m_h,
        other=0.0,
    )

    # ------- load gates -------------------------------------------------------
    left_gate = tl.load(
        left_gate_ptr
        + batch * stride_gbz
        + row   * stride_gby
        + offs_n * stride_gbx
        + offs_h * stride_gbh,
        mask=m_h,
        other=0.0,
    )
    right_gate = tl.load(
        right_gate_ptr
        + batch * stride_gbz
        + row   * stride_gby
        + offs_n * stride_gbx
        + offs_h * stride_gbh,
        mask=m_h,
        other=0.0,
    )
    out_gate = tl.load(
        out_gate_ptr
        + batch * stride_gbz
        + row   * stride_gby
        + offs_n * stride_gbx
        + offs_h * stride_gbh,
        mask=m_h,
        other=0.0,
    )

    # ------- optional mask ----------------------------------------------------
    if mask_ptr != 0:
        mask = tl.load(
            mask_ptr
            + batch * stride_mbz
            + row   * stride_mby
            + offs_n * stride_mbx,
            mask=mask_n,
            other=0.0,
        )   # (BLOCK_N, 1)
        left  = left * mask
        right = right * mask

    # ------- apply gates -------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # store back the fused tensors (overwrites original buffers)
    tl.store(
        left_ptr
        + batch * stride_lbz
        + row   * stride_lby
        + offs_n * stride_lbx
        + offs_h * stride_lbh,
        left,
        mask=m_h,
    )
    tl.store(
        right_ptr
        + batch * stride_rbz
        + row   * stride_rby
        + offs_n * stride_rbx
        + offs_h * stride_rbh,
        right,
        mask=m_h,
    )
    tl.store(
        out_gate_ptr
        + batch * stride_gbz
        + row   * stride_gby
        + offs_n * stride_gbx
        + offs_h * stride_gbh,
        out_gate,
        mask=m_h,
    )


# ----------------------------------------------------------------------
# Helper – fused mask & gating
# ----------------------------------------------------------------------
def _apply_mask_and_gating(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
    mask: torch.Tensor | None,
) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
    """
    In‑place fusion of:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
        out_gate = out_gate (sigmoid already applied)
    All tensors are fp16, contiguous, shape (B,N,N,H).
    """
    B, N, _, H = left.shape
    # strides in element units
    l_str = left.stride()
    r_str = right.stride()
    g_str = left_gate.stride()            # all gate tensors share same layout

    if mask is None:
        mask_ptr = 0
        m_str = (0, 0, 0)
    else:
        mask = mask.to(torch.float16)
        mask_ptr = mask.data_ptr()
        m_str = mask.stride()             # (B,N,N)

    # launch one program per (batch, row) pair
    total = B * N
    BLOCK_N = 64          # tile over the second pair‑dim
    BLOCK_H = 32          # tile over hidden dimension

    grid = (total,)
    _mask_gate_fusion_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        out_gate_ptr=out_gate,
        mask_ptr=mask_ptr,
        B=B,
        N=N,
        H=H,
        stride_lbz=l_str[0],
        stride_lby=l_str[1],
        stride_lbx=l_str[2],
        stride_lbh=l_str[3],
        stride_rbz=r_str[0],
        stride_rby=r_str[1],
        stride_rbx=r_str[2],
        stride_rbh=r_str[3],
        stride_gbz=g_str[0],
        stride_gby=g_str[1],
        stride_gbx=g_str[2],
        stride_gbh=g_str[3],
        stride_mbz=m_str[0] if mask is not None else 0,
        stride_mby=m_str[1] if mask is not None else 0,
        stride_mbx=m_str[2] if mask is not None else 0,
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return left, right, out_gate


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors (see docstring above)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (fp32, manual)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var  = input_tensor.var(-1, unbiased=False, keepdim=True)
    eps = 1e-5
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * ln_w + ln_b               # still fp32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (fp16) – bias‑free as in the reference
    # ------------------------------------------------------------------
    def _to_fp16(t): return t.to(torch.float16)

    left_proj_w  = _to_fp16(weights["left_proj.weight"])
    right_proj_w = _to_fp16(weights["right_proj.weight"])
    left_gate_w  = _to_fp16(weights["left_gate.weight"])
    right_gate_w = _to_fp16(weights["right_gate.weight"])
    out_gate_w   = _to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)               # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 3️⃣ Fuse mask & gating with a small Triton kernel
    # ------------------------------------------------------------------
    left, right, out_gate = _apply_mask_and_gating(
        left, right, left_gate, right_gate, out_gate, mask
    )   # all remain fp16, shape (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Heavy cubic contraction via cuBLAS batched GEMM
    #    left : (B,N,N,H) → (B*H, N, N)
    #    right: (B,N,N,H) → (B*H, N, N) then transpose K,J
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N)

    BH, _, I, K = left_perm.shape                 # I == K == N
    left_mat  = left_perm.view(BH, I, K)          # (B*H, N, N)
    right_mat = right_perm.view(BH, K, I)         # (B*H, N, N)

    # batched matmul – fp16×fp16→fp16 (cuBLAS uses tensor cores)
    out_mat = torch.bmm(left_mat, right_mat)      # (B*H, N, N)

    # reshape back to (B,N,N,hidden_dim)
    out_fp16 = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (fp32) → gating → final linear (fp32)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * out_ln_w + out_ln_b

    # out_gate still fp16, cast to fp32 for the final element‑wise multiply
    out = out * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)               # (B,N,N,dim), fp32

    return out
