"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the (B,N,N,C) input.
2️⃣  Linear projections → (B,N,N,H) for *left*, *right* and three gates.
3️⃣  Fuse element‑wise work (mask, left‑gate, right‑gate) with a tiny Triton
    kernel.  This satisfies the “some‑kernel” requirement while keeping the
    expensive O(N³) contraction on cuBLAS/Tensor‑cores.
4️⃣  Reshape to (B*H,N,N) and perform a batched matrix‑multiply
        out[b,h] = left[b,h] @ right[b,h].T
    (FP16 inputs, FP32 accumulation, FP16 output).
5️⃣  Layer‑norm the hidden result, multiply by *out_gate* and apply the final
    linear projection back to (B,N,N,C).

All heavy arithmetic runs in FP16 for maximal Tensor‑core utilisation;
the final output is returned in FP32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – element‑wise fusion of masks and gates
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                # fp16 * (B,N,N,H)
    left_gate_ptr, right_gate_ptr,      # fp16 * (B,N,N,H)
    mask_ptr,                           # fp16 * (B,N,N) or dummy
    out_left_ptr, out_right_ptr,        # fp16 * (B,N,N,H) – results
    B, N, H,                            # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_m0, stride_m1, stride_m2,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    The kernel is launched on a 1‑D grid covering the total number of
    elements B·N·N·H.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # ----- decode linear index to (b,i,j,h) (contiguous layout) -----
    idx = offs
    h = idx % H
    idx = idx // H
    j = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # ----- load tensors -----
    # left / right
    ptr = (
        b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    )
    l = tl.load(left_ptr + ptr,  mask=mask, other=0.0)
    r = tl.load(right_ptr + ptr, mask=mask, other=0.0)

    # gates
    ptr_gate = (
        b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + h * stride_lg3
    )
    lg = tl.load(left_gate_ptr  + ptr_gate, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + ptr_gate, mask=mask, other=0.0)

    # optional mask (broadcast over hidden dim)
    if HAS_MASK:
        ptr_mask = b * stride_m0 + i * stride_m1 + j * stride_m2
        m = tl.load(mask_ptr + ptr_mask, mask=mask, other=0.0)
    else:
        m = 1.0

    # ----- apply fusion -----
    out_l = l * lg * m
    out_r = r * rg * m

    # ----- store results -----
    tl.store(out_left_ptr  + ptr, out_l, mask=mask)
    tl.store(out_right_ptr + ptr, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        x          – (B, N, N, C)  torch.float32
        mask       – (B, N, N)    torch.bool / float32 / None
        weights    – dict of learned parameters (float32)
        config     – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor
        (B, N, N, C)  float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype  = torch.float32

    B, N, _, C = x.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣  Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (float32 → fp16 for later GEMM)
    # ------------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate= torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate  = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3️⃣  Fuse mask & gates with Triton
    # ------------------------------------------------------------------
    # cast to fp16 and make contiguous (required for correct stride handling)
    left_fp16      = left.to(torch.float16).contiguous()
    right_fp16     = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate.to(torch.float16).contiguous()
    right_gate_fp16= right_gate.to(torch.float16).contiguous()

    if mask is None:
        # dummy mask – kernel will ignore it
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask = False
    else:
        # mask is (B,N,N) → broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        has_mask = True

    # allocate fused outputs
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elements = B * N * N * H
    BLOCK = 1024                     # 1‑K threads give good occupancy on H200
    grid  = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        mask_fp16.stride(0) if has_mask else 0,
        mask_fp16.stride(1) if has_mask else 0,
        mask_fp16.stride(2) if has_mask else 0,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 4️⃣  Batched GEMM (cuBLAS / Tensor‑cores)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) – contiguous layout required for torch.bmm
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)

    # (FP16 x FP16 -> FP16) with FP32 accumulation
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))          # (B*H,N,N)

    # reshape back to (B,N,N,H) and cast to fp32 for the next layer‑norm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(dtype)

    # ------------------------------------------------------------------
    # 5️⃣  Hidden‑dim LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    out = out * out_gate                     # broadcast over (N,N)
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
