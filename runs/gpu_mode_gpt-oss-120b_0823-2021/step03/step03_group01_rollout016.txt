"""
custom_kernel – “outgoing” TriMul forward (AlphaFold‑3 style).

The expensive 3‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is performed with a tiny Triton batched‑GEMM kernel that multiplies the
(N×N) matrices for each head (hidden dimension) in fp16 while accumulating
in fp32.  All LayerNorms, linear projections and gating are kept in PyTorch
so that the code stays simple, numerically stable and works for all
configuration variants (mask / no‑mask, different hidden sizes, etc.).

The kernel works for an arbitrary batch size B (≤2 in the tests) and an
arbitrary hidden dimension H (called *hidden_dim*).  The input tensor has
shape (B, N, N, dim) and the hidden tensors have shape (B, N, N, hidden_dim).
The kernel receives the tensors in the layout (B, H, I, K) and (B, H, K, J) –
i.e. (batch, head, row, column) – which maps directly to a batched GEMM
with shapes (Z, I, K)·(Z, K, J) → (Z, I, J) where Z = B·H.

The implementation uses fp16 for the matrix data (Tensor‑Core friendly) and
fp32 for the accumulator to preserve numerical quality.  The final result
is cast back to fp32 before the remaining post‑processing steps.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr,          # left  (Z, I, K)
    b_ptr,          # right (Z, K, J)
    c_ptr,          # out   (Z, I, J)
    M, N, K,        # matrix sizes
    stride_az, stride_am, stride_ak,   # strides of a
    stride_bz, stride_bk, stride_bn,   # strides of b
    stride_cz, stride_cm, stride_cn,   # strides of c
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute one tile of the batched GEMM."""

    pid_z = tl.program_id(2)          # batch*head dimension
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile [BLOCK_M, BLOCK_K]
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile [BLOCK_K, BLOCK_N]
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 accumulation
        acc += tl.dot(a, b)

    # write back – cast to fp16 (the rest of the pipeline expects fp16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    Parameters
    ----------
    left  : Tensor (B, H, I, K)  dtype=torch.float16, contiguous
    right : Tensor (B, H, K, J)  dtype=torch.float16, contiguous

    Returns
    -------
    out   : Tensor (B, H, I, J) dtype=torch.float16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # output allocation
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse leading dims for the kernel (Z = B*H)
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides in element units
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes – tuned for Hopper/Tensor‑Core friendly work‑groups
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # rows
        triton.cdiv(J, BLOCK_N),   # cols
        Z,                         # batch*head
    )
    _batched_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # works well on SM90
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass: PyTorch for everything except the N³ contraction
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim] (float32, on CUDA)
        mask         : torch.Tensor [B, N, N] (float32/bool) or None
        weights      : dict of parameter tensors (float32)
        config       : dict with keys
                         - "dim"          : input channel dimension
                         - "hidden_dim"   : hidden dimension for projections

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm  (FP32 for numerical stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # mean / var over the channel dimension
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                       # (B,N,N,dim) fp32

    # ------------------------------------------------------------------
    # 2. Linear projections – cast to fp16 for the heavy GEMM
    # ------------------------------------------------------------------
    def _to_fp16(t): return t.to(torch.float16, non_blocking=True)

    left_proj_w   = _to_fp16(weights["left_proj.weight"])
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim) fp16
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. N³ contraction via the Triton batched GEMM
    #    left  : (B, N, N, H) → (B, H, I, K)   where I = K = N
    #    right : (B, N, N, H) → (B, H, K, J)   where J = N
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _batched_gemm(left_perm, right_perm)    # (B, H, I, J)

    # back to original layout and to fp32 for the rest of the pipeline
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm
    # ------------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7. Final gating + projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                  # (B,N,N,dim) fp32

    return out
