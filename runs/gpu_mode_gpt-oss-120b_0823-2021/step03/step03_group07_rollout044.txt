"""
TriMul “outgoing” forward pass (AlphaFold‑3) – mixed PyTorch / Triton implementation.

The only cubic work is the contraction

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

We reshape the tensors to (B·H, N, N) and let the highly‑optimised CUDA
cuBLAS kernel (torch.bmm) compute the batched matrix‑multiply.  The expensive
element‑wise part (masking + gating) is fused into a tiny Triton kernel that
operates on the (B, N, N, H) layout produced by the linear projections.
All other cheap operations (LayerNorm, final LayerNorm, final linear) stay in
regular PyTorch.

The kernel works on FP16 data (the linear layers are cast to fp16) and writes
the result back in fp16; the final output is cast to float32 as required by
the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: mask + gate fusion
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,                # fp16*  [B, N, N, H]
    right_ptr,               # fp16*  [B, N, N, H]
    mask_ptr,                # fp16*  [B, N, N, 1]   (0/1 mask broadcasted)
    left_gate_ptr,           # fp16*  [B, N, N, H]
    right_gate_ptr,          # fp16*  [B, N, N, H]
    out_left_ptr,            # fp16*  [B, N, N, H]   (written in‑place)
    out_right_ptr,           # fp16*  [B, N, N, H]   (written in‑place)
    B, N, H,
    stride_lb, stride_ln, stride_lk, stride_lh,   # left  strides
    stride_rb, stride_rn, stride_rk, stride_rh,   # right strides
    stride_mb, stride_mn, stride_mk, stride_mh,   # mask  strides (only last dim is 1)
    stride_gb, stride_gn, stride_gk, stride_gh,   # gate strides (same layout as left/right)
    BLOCK_B: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each block of (b, n_i, h) compute

        left  = left  * mask * left_gate
        right = right * mask * right_gate

    The mask is broadcasted over the channel dimension.
    """
    pid = tl.program_id(0)

    # decode 3‑D grid  (block over batch, rows, channels)
    total_n_blocks = tl.cdiv(N, BLOCK_N)
    total_h_blocks = tl.cdiv(H, BLOCK_H)

    b = (pid // (total_n_blocks * total_h_blocks)) % B
    n_block = (pid // total_h_blocks) % total_n_blocks
    h_block = pid % total_h_blocks

    # offsets inside the block
    offs_n = n_block * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = h_block * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_n = offs_n < N
    mask_h = offs_h < H

    # load mask (broadcasted on channel dim)
    mask = tl.load(
        mask_ptr
        + b * stride_mb
        + offs_n[:, None] * stride_mn
        + 0 * stride_mk               # k‑dim absent – stride 0
        + 0 * stride_mh,              # channel dim = 1 → always 0
        mask=mask_n[:, None] & tl.full((1, ), True),
        other=0.0,
    ).to(tl.float32)                 # shape (BLOCK_N, 1)

    # --------------------------------------------------------------
    # left side
    # --------------------------------------------------------------
    l_ptr = (
        left_ptr
        + b * stride_lb
        + offs_n[:, None] * stride_ln
        + 0 * stride_lk                # k‑dim = 0 for this slice
        + offs_h[None, :] * stride_lh
    )
    left = tl.load(
        l_ptr,
        mask=mask_n[:, None] & mask_h[None, :],
        other=0.0,
    ).to(tl.float32)

    lg_ptr = (
        left_gate_ptr
        + b * stride_gb
        + offs_n[:, None] * stride_gn
        + 0 * stride_gk
        + offs_h[None, :] * stride_gh
    )
    left_gate = tl.load(
        lg_ptr,
        mask=mask_n[:, None] & mask_h[None, :],
        other=0.0,
    ).to(tl.float32)

    left = left * mask * left_gate

    # write back
    tl.store(
        out_left_ptr
        + b * stride_lb
        + offs_n[:, None] * stride_ln
        + 0 * stride_lk
        + offs_h[None, :] * stride_lh,
        left.to(tl.float16),
        mask=mask_n[:, None] & mask_h[None, :],
    )

    # --------------------------------------------------------------
    # right side (identical logic)
    # --------------------------------------------------------------
    r_ptr = (
        right_ptr
        + b * stride_rb
        + offs_n[:, None] * stride_rn
        + 0 * stride_rk
        + offs_h[None, :] * stride_rh
    )
    right = tl.load(
        r_ptr,
        mask=mask_n[:, None] & mask_h[None, :],
        other=0.0,
    ).to(tl.float32)

    rg_ptr = (
        right_gate_ptr
        + b * stride_gb
        + offs_n[:, None] * stride_gn
        + 0 * stride_gk
        + offs_h[None, :] * stride_gh
    )
    right_gate = tl.load(
        rg_ptr,
        mask=mask_n[:, None] & mask_h[None, :],
        other=0.0,
    ).to(tl.float32)

    right = right * mask * right_gate

    tl.store(
        out_right_ptr
        + b * stride_rb
        + offs_n[:, None] * stride_rn
        + 0 * stride_rk
        + offs_h[None, :] * stride_rh,
        right.to(tl.float16),
        mask=mask_n[:, None] & mask_h[None, :],
    )


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input   [B, N, N, dim]
        torch.Tensor,                     # mask    [B, N, N]   (or None)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any]                    # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]          # = H

    # --------------------------------------------------------------
    # 1. LayerNorm (last dim = dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # --------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # --------------------------------------------------------------
    # 3. Gates (sigmoid after linear)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # --------------------------------------------------------------
    # 4. Optional masking  –  fuse mask + gates in a small Triton kernel
    # --------------------------------------------------------------
    if mask is not None:
        # make everything fp16 for the kernel (better bandwidth)
        left_fp16  = left.to(torch.float16)
        right_fp16 = right.to(torch.float16)
        left_gate_fp16  = left_gate.to(torch.float16)
        right_gate_fp16 = right_gate.to(torch.float16)
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)

        out_left  = torch.empty_like(left_fp16)
        out_right = torch.empty_like(right_fp16)

        # launch configuration
        BLOCK_B = 1                     # we keep one batch per program
        BLOCK_N = 64                    # rows of the pairwise matrix
        BLOCK_H = 32                    # channel (hidden) block

        total_blocks = B * tl.cdiv(N, BLOCK_N) * tl.cdiv(hidden_dim, BLOCK_H)
        grid = (total_blocks,)

        _apply_mask_gate_kernel[grid](
            left_fp16,
            right_fp16,
            mask_fp16,
            left_gate_fp16,
            right_gate_fp16,
            out_left,
            out_right,
            B,
            N,
            hidden_dim,
            # strides for left
            left_fp16.stride(0), left_fp16.stride(2), left_fp16.stride(3), left_fp16.stride(1),
            # strides for right (same layout)
            right_fp16.stride(0), right_fp16.stride(2), right_fp16.stride(3), right_fp16.stride(1),
            # strides for mask (last dim = 1)
            mask_fp16.stride(0), mask_fp16.stride(2), mask_fp16.stride(3), mask_fp16.stride(1),
            # strides for gates (same as left/right)
            left_gate_fp16.stride(0), left_gate_fp16.stride(2), left_gate_fp16.stride(3), left_gate_fp16.stride(1),
            BLOCK_B=BLOCK_B,
            BLOCK_N=BLOCK_N,
            BLOCK_H=BLOCK_H,
        )
        left  = out_left.to(dtype)
        right = out_right.to(dtype)
    else:
        # no mask → just apply gates (elementwise, cheap)
        left  = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Batched GEMM via cuBLAS (torch.bmm)
    #    reshape to (B*H, N, N) and multiply left @ right^T
    # --------------------------------------------------------------
    # permute to (B, H, N, N) for the batched matmul
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # flatten batch * heads dimension
    left_flat  = left.view(B * hidden_dim, N, N)          # (B*H, N, N)
    right_flat = right.view(B * hidden_dim, N, N)         # (B*H, N, N)

    # right must be transposed on the last two axes for the “outgoing” contraction
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # restore shape (B, H, N, N) → (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # --------------------------------------------------------------
    # 6. Final LayerNorm, gating and linear projection back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply output gate (broadcasted over pairwise axes)
    out = out * out_gate

    # final linear
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
