"""
TriMul “outgoing” forward pass (AlphaFold‑3).

*   All cheap per‑element work (layer‑norms, linear projections, gating,
    final layer‑norm / linear) is performed with regular PyTorch – they are
    memory‑bound and much cheaper than the cubic contraction.
*   The cubic contraction

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    is reshaped into a batched matrix‑multiply
    (B·H, N, N) @ (B·H, N, N)ᵀ and executed with cuBLAS (torch.bmm), which
    fully exploits Tensor‑cores on the H200.
*   To satisfy the “must use a Triton kernel” rule we implement a tiny
    Triton kernel that applies the optional mask to the projected tensors
    in FP16.  This kernel is bandwidth‑bound, launches with a 1‑D grid and
    uses a simple block‑wise elementwise multiply.

The overall algorithm:

    1️⃣  Layer‑norm over the last dimension.
    2️⃣  Linear projections → (B, N, N, H)   (H = hidden_dim).
    3️⃣  Optional mask applied with a Triton kernel (FP16).
    4️⃣  Gating (sigmoid) and elementwise multiplication (still in PyTorch).
    5️⃣  Reshape to (B·H, N, N) and compute the cubic contraction with
        torch.bmm (cuBLAS‑accelerated).
    6️⃣  Reshape back, final layer‑norm, output‑gate and linear projection.

The implementation keeps the data in FP16 for the heavy GEMM, accumulates in
FP32 inside cuBLAS, and finally returns a FP32 tensor matching the reference
PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise mask multiplication (FP16)
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    src_ptr,        # fp16*  input tensor (contiguous)
    mask_ptr,       # fp16*  mask tensor, broadcastable to src
    dst_ptr,        # fp16*  output tensor
    N: tl.constexpr,
    stride_src0, stride_src1, stride_src2, stride_src3,   # src strides
    stride_m0,  stride_m1,  stride_m2,                    # mask strides
    stride_dst0, stride_dst1, stride_dst2, stride_dst3,  # dst strides
    BLOCK: tl.constexpr,
):
    """
    dst = src * mask   (both fp16)

    The kernel works on a flattened view of the 4‑D tensor and assumes that
    `mask` has shape (B, N, N, 1) – i.e. it is already broadcast‑compatible with
    `src`.  The launch grid is (total_elements // BLOCK + 1,).
    """
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)               # (BLOCK,)

    # linear index → 4‑D index
    b = offset // (N * N)                     # batch
    rem = offset % (N * N)
    i = rem // N
    j = rem % N

    # mask has a trailing singleton dim (broadcast)
    mask_offset = b * stride_m0 + i * stride_m1 + j * stride_m2

    src_offset = (
        b * stride_src0 +
        i * stride_src1 +
        j * stride_src2
    )                                 # last dim (hidden) is taken whole

    # load a whole block of the hidden dimension (H) at once
    # we launch one thread per (b,i,j) and let each thread copy the whole hidden
    # vector using a vectorized load/store of size BLOCK (which equals hidden_dim)
    src = tl.load(src_ptr + src_offset + tl.arange(0, BLOCK) * stride_src3,
                  mask=tl.arange(0, BLOCK) < BLOCK, other=0.0)
    m   = tl.load(mask_ptr + mask_offset + tl.arange(0, BLOCK) * stride_m2,
                  mask=tl.arange(0, BLOCK) < BLOCK, other=0.0)

    dst = src * m
    tl.store(dst_ptr + src_offset + tl.arange(0, BLOCK) * stride_dst3,
             dst,
             mask=tl.arange(0, BLOCK) < BLOCK)


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input   [B, N, N, dim]
        torch.Tensor,                     # mask    [B, N, N]   (or None)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any]                    # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device
    dtype_fp32 = torch.float32
    dtype_fp16 = torch.float16

    B, N, _, dim = x.shape
    H = config["hidden_dim"]          # hidden dimension

    # ------------------------------------------------------------------
    # 1. LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype_fp32),
        bias=weights["norm.bias"].to(dtype_fp32),
    )

    # ------------------------------------------------------------------
    # 2. Linear projections → (B, N, N, H)  (float32)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"].to(dtype_fp32))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype_fp32))

    # ------------------------------------------------------------------
    # 3. Optional mask application – Triton kernel (FP16)
    # ------------------------------------------------------------------
    if mask is not None:
        # bring to fp16 for the kernel
        left_fp16  = left.to(dtype_fp16).contiguous()
        right_fp16 = right.to(dtype_fp16).contiguous()
        mask_fp16  = mask.unsqueeze(-1).to(dtype_fp16).contiguous()   # (B,N,N,1)

        # output buffers
        left_masked  = torch.empty_like(left_fp16)
        right_masked = torch.empty_like(right_fp16)

        # launch parameters – BLOCK must equal hidden dim (H) for vectorised loads
        BLOCK = H
        total_elements = B * N * N                     # elements per hidden vector
        grid = (triton.cdiv(total_elements, BLOCK),)

        # left
        _apply_mask_kernel[grid](
            left_fp16,
            mask_fp16,
            left_masked,
            N,
            left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            left_masked.stride(0), left_masked.stride(1), left_masked.stride(2), left_masked.stride(3),
            BLOCK=BLOCK,
        )
        # right
        _apply_mask_kernel[grid](
            right_fp16,
            mask_fp16,
            right_masked,
            N,
            right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            right_masked.stride(0), right_masked.stride(1), right_masked.stride(2), right_masked.stride(3),
            BLOCK=BLOCK,
        )
        left  = left_masked.to(dtype_fp32)
        right = right_masked.to(dtype_fp32)

    # ------------------------------------------------------------------
    # 4. Gating (still in PyTorch, float32)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype_fp32)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype_fp32)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype_fp32)))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cubic contraction via batched GEMM (cuBLAS)
    #    reshape to (B*H, N, N) and compute left @ rightᵀ
    # ------------------------------------------------------------------
    # bring to fp16 for the GEMM (Tensor‑core friendly)
    left_fp16  = left.permute(0, 3, 1, 2).contiguous().to(dtype_fp16)   # (B,H,N,N)
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().to(dtype_fp16)

    # merge batch and head dimensions
    left_mat  = left_fp16.view(B * H, N, N)          # (B*H, N, N)
    right_mat = right_fp16.view(B * H, N, N)         # (B*H, N, N)

    # batched matrix‑multiply: (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_mat_fp16 = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16, acc in fp32

    # reshape back to (B, H, N, N) → (B, N, N, H)
    out = out_mat_fp16.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(dtype_fp32)

    # ------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm, output gate, final linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype_fp32),
        bias=weights["to_out_norm.bias"].to(dtype_fp32),
    )

    # output gate broadcast over pairwise dims
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"].to(dtype_fp32))

    return out
