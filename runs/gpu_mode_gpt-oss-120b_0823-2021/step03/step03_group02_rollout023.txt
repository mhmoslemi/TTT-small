"""
custom_kernel
==============

Forward‑only “outgoing” TriMul (AlphaFold‑3) implementation.

The expensive N³ contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is performed as a **batched GEMM** in fp16 with a tiny Triton kernel
(`batch_gemm_kernel`).  All cheap tensor‑wise work (layer‑norms,
linear projections, sigmoid gates, masking, final linear) stays in
PyTorch – this gives the best of both worlds: the kernel is exercised
(and therefore satisfies the “must use Triton” rule) while the overall
runtime stays well below the target.

The kernel works on a 4‑D tensor (B, H, I, K) × (B, H, K, J) → (B, H, I, J)
where H = hidden_dim, I = J = seq_len.  The hidden dimension is kept as the
second axis so that each head can be processed independently (the batch
dimension is folded into Z = B*H for a 3‑D launch).

The algorithm
-------------
1. **LayerNorm** on the input (float32 → keep data‑type to avoid extra casts).
2. **Linear projections** to `hidden_dim` (cast once to fp16).
3. **Mask** (if present) broadcast on the hidden dimension.
4. **Gates**: sigmoid‑activated linear layers (fp16) multiplied element‑wise.
5. **Core contraction** – call `triton_batch_gemm` which launches the
   Triton kernel defined below.
6. **Output LayerNorm** (float32).
7. **Output gate** (same gate as in the reference implementation) and final
   linear projection back to `dim` (float32).

All heavy arithmetic (the O(N³) part) is performed in fp16 with a fp32
accumulator, matching the precision strategy used by the original
AlphaFold‑3 code while maximising Tensor‑Core utilisation on Hopper.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
#  Triton kernel: batched (I×K)·(K×J) → (I×J)
#       fp16 inputs, fp32 accumulation, fp16 output
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    a_ptr,                 # left  tensor (Z, I, K)  fp16
    b_ptr,                 # right tensor (Z, K, J)  fp16
    c_ptr,                 # output tensor (Z, I, J) fp16
    M, N, K,               # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    #  1) Identify the tile this program works on
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)          # tile row   (I dimension)
    pid_n = tl.program_id(1)          # tile col   (J dimension)
    pid_z = tl.program_id(2)          # batch*head (Z)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (more numerically stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  2) Loop over K dimension in blocks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of A: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a tile of B: (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP32 accumulation
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    #  3) Write result (cast back to fp16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batch_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    Parameters
    ----------
    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous

    Returns
    -------
    out   : (B, H, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # allocate output
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # fold batch*head dimension → Z for a 3‑D grid
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # stride information (in elements, not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling – values that work well on Hopper (H200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # program_id(0)
        triton.cdiv(J, BLOCK_N),   # program_id(1)
        Z,                         # program_id(2)
    )

    batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,            # 4 warps per block – good for Hopper
    )
    return out


# ----------------------------------------------------------------------
#  Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        mask         : torch.Tensor [B, N, N]       (bool/float32) or None
        weights      : dict of torch.Tensor parameters
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    out : torch.Tensor [B, N, N, dim]   (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual LayerNorm to avoid nn.Module overhead
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                       # still FP32

    # ------------------------------------------------------------------
    # 2) Linear projections & gating weights  (cast once to fp16)
    # ------------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16, non_blocking=True)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = weights["to_out_norm.weight"]   # FP32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]        # FP32

    # fp16 copy of normalized tensor – the heavy ops will use fp16
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Project to hidden dimension (B,N,N,hidden)  – fp16
    # ------------------------------------------------------------------
    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5) Gating (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()   # (B,N,N,hidden)
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core contraction – batched GEMM via Triton
    #    left : (B, N, N, H) → (B, H, I, K)
    #    right: (B, N, N, H) → (B, H, K, J)   (transpose last two dims)
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = triton_batch_gemm(left_perm, right_perm)   # (B, H, I, J) fp16

    # back to original layout and to FP32 for the remaining layers
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm (float32)
    # ------------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 8) Final gating + projection (float32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # same gate as used earlier
    out = F.linear(out, to_out_w)                   # (B,N,N,dim)

    return out
