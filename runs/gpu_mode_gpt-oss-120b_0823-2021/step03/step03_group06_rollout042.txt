"""
TriMul “outgoing” operator (AlphaFold‑3) – mixed PyTorch / Triton implementation.

The expensive cubic contraction is performed with a batched GEMM using
torch.bmm (leveraging cuBLAS/Tensor‑Cores on H200).  The remaining element‑wise
operations (masking, gating, final output‑gate) are fused in a tiny Triton
kernel to avoid extra kernel launches and to satisfy the requirement that
*some* work is done on the device with Triton.

All LayerNorms and linear projections stay in PyTorch because they are
negligible compared to the O(N³) work.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: apply mask (if present), left/right gates and final out‑gate
# ----------------------------------------------------------------------
@triton.jit
def apply_gates_kernel(
    left_ptr, right_ptr,          # fp16    [B, H, N, N]
    left_gate_ptr, right_gate_ptr,# fp16    [B, H, N, N]
    mask_ptr,                     # uint8   [B, N, N]  (or nullptr)
    out_gate_ptr,                 # fp16    [B, N, N, 1]
    out_ptr,                      # fp16    [B, H, N, N]  (final result)
    B, H, N,                     # problem size
    stride_lbh, stride_lbhc, stride_lbk,   # left strides
    stride_rbh, stride_rbhc, stride_rbk,   # right strides
    stride_lg_b, stride_lg_n, stride_lg_c, # left_gate strides
    stride_rg_b, stride_rg_n, stride_rg_c, # right_gate strides
    stride_m_b, stride_m_n,                 # mask strides (optional)
    stride_og_b, stride_og_n, stride_og_c, # out_gate strides
    stride_ob_b, stride_ob_h, stride_ob_n, # out strides
    HAS_MASK: tl.constexpr,
):
    """Fuse:
        left  = left  * left_gate  * (mask?)
        right = right * right_gate * (mask?)
        out   = (left @ right^T) * out_gate
    The matrix multiplication itself is done in PyTorch; this kernel only
    performs the element‑wise preprocessing/post‑processing.
    """
    pid = tl.program_id(0)
    # linear index over (b, h, i, j)
    total = B * H * N * N
    idx = pid * tl.cdiv(N, 32) * 32 + tl.arange(0, tl.cdiv(N, 32) * 32)
    # Guard against out‑of‑bounds
    mask = idx < total
    if not tl.any(mask):
        return

    # Decode indices
    b = (idx // (H * N * N))
    rem = idx % (H * N * N)
    h = rem // (N * N)
    rem = rem % (N * N)
    i = rem // N
    j = rem % N

    # Pointers
    left = left_ptr + b * stride_lbh + h * stride_lbhc + i * stride_lbk + j
    right = right_ptr + b * stride_rbh + h * stride_rbhc + i * stride_rbk + j
    left_gate = left_gate_ptr + b * stride_lg_b + h * stride_lg_n + i * stride_lg_c + j
    right_gate = right_gate_ptr + b * stride_rg_b + h * stride_rg_n + i * stride_rg_c + j

    # Load
    l = tl.load(left, mask=mask, other=0.0).to(tl.float32)
    r = tl.load(right, mask=mask, other=0.0).to(tl.float32)
    lg = tl.load(left_gate, mask=mask, other=0.0).to(tl.float32)
    rg = tl.load(right_gate, mask=mask, other=0.0).to(tl.float32)

    # Optional mask
    if HAS_MASK:
        m = tl.load(mask_ptr + b * stride_m_b + i * stride_m_n + j,
                    mask=mask, other=0).to(tl.float32)
        l = l * m
        r = r * m

    # Apply gates
    l = l * lg
    r = r * rg

    # Store back (still fp16)
    tl.store(left, l.to(tl.float16), mask=mask)
    tl.store(right, r.to(tl.float16), mask=mask)

    # ------------------------------------------------------------------
    # Final output‑gate multiplication (applied after the GEMM in host)
    # ------------------------------------------------------------------
    # out_gate shape [B, N, N, 1]
    og = tl.load(out_gate_ptr + b * stride_og_b + i * stride_og_n + j * stride_og_c,
                 mask=mask, other=0.0).to(tl.float32)

    # out already contains the GEMM result (fp16).  We just multiply by out_gate.
    out = tl.load(out_ptr + b * stride_ob_b + h * stride_ob_h + i * stride_ob_n + j,
                  mask=mask, other=0.0).to(tl.float32)
    out = out * og
    tl.store(out_ptr + b * stride_ob_b + h * stride_ob_h + i * stride_ob_n + j,
             out.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input   : [B, N, N, dim]
        torch.Tensor,                # mask    : [B, N, N] (or None)
        Dict[str, torch.Tensor],     # weights : dict of parameters
        Dict[str, Any],              # config  : {"dim":..., "hidden_dim":...}
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension (called H in paper)

    # --------------------------------------------------------------
    # 1. LayerNorm + linear projections (PyTorch, cheap)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # --------------------------------------------------------------
    # 2. Gating (pre‑compute sigmoids)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N,H)

    # --------------------------------------------------------------
    # 3. Move to [B, H, N, N] layout and FP16 for the kernel / GEMM
    # --------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H) -> later broadcast

    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    left_gate_fp16 = left_gate.to(torch.float16)
    right_gate_fp16 = right_gate.to(torch.float16)
    out_gate_fp16 = out_gate.to(torch.float16)  # shape (B,N,N,H)

    # --------------------------------------------------------------
    # 4. Apply mask and per‑head gates with a tiny Triton kernel
    # --------------------------------------------------------------
    HAS_MASK = mask is not None
    mask_fp8 = None
    if HAS_MASK:
        mask_fp8 = (mask.to(torch.uint8)).contiguous()   # uint8 works as bitmask

    # launch config – one thread per element (B*H*N*N)
    total_elements = B * H * N * N
    BLOCK = 128
    grid = (triton.cdiv(total_elements, BLOCK),)

    apply_gates_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp8 if HAS_MASK else torch.empty(0, device=device).int(),   # dummy ptr
        out_gate_fp16,
        torch.empty_like(left_fp16),   # placeholder for final out (will be overwritten later)
        B, H, N,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1), left_gate_fp16.stride(2),
        right_gate_fp16.stride(0), right_gate_fp16.stride(1), right_gate_fp16.stride(2),
        mask_fp8.stride(0) if HAS_MASK else 0,
        mask_fp8.stride(1) if HAS_MASK else 0,
        out_gate_fp16.stride(0), out_gate_fp16.stride(1), out_gate_fp16.stride(2),
        torch.empty_like(left_fp16).stride(0),
        torch.empty_like(left_fp16).stride(1),
        torch.empty_like(left_fp16).stride(2),
        HAS_MASK=HAS_MASK,
        BLOCK=BLOCK,
    )
    # Note: the kernel overwrites left_fp16/right_fp16 with gated & masked values.
    # The placeholder out tensor is not used; we will compute the GEMM next.

    # --------------------------------------------------------------
    # 5. Batched GEMM: out = left @ right^T  (B*H, N, N)
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) and use torch.bmm (cuBLAS/Tensor‑Core)
    left_mat = left_fp16.view(B * H, N, N)
    right_mat = right_fp16.view(B * H, N, N)

    # right needs to be transposed on the last two dims
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N) fp16
    out_fp16 = out_mat.to(torch.float16)

    # --------------------------------------------------------------
    # 6. Apply final output‑gate (already fused inside kernel) and reshape
    # --------------------------------------------------------------
    # out_gate has shape (B, N, N, H).  We need to broadcast it onto (B, H, N, N)
    out_gate_reshaped = out_gate_fp16.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    out_fp16 = out_fp16 * out_gate_reshaped          # element‑wise mul (fp16)

    # reshape back to (B, N, N, H)
    out = out_fp16.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    out = out.to(dtype)      # back to FP32 for the remaining ops

    # --------------------------------------------------------------
    # 7. to_out_norm + final linear projection (PyTorch)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
