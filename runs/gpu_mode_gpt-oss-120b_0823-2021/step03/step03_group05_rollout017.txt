"""
TriMul (outgoing) – fast forward‑pass.

Algorithm
---------
1️⃣ Layer‑norm + four linear projections (left, right, left‑gate, right‑gate,
   out‑gate) – stay in PyTorch (these are cheap compared with the O(N³) work).

2️⃣ Fuse *mask* (if present) with *left‑gate* and *right‑gate* in a tiny
   Triton element‑wise kernel.  The kernel overwrites the ``left`` and ``right``
   tensors in‑place with
        left  = left  * mask * left_gate
        right = right * mask * right_gate
   All tensors are cast to FP16 for bandwidth‑limited speed.

3️⃣ Core contraction (the cubic part) is performed with a batched GEMM that
   leverages cuBLAS/Tensor‑cores:
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   By reshaping to (B·H, N, N) we can call ``torch.bmm`` which is highly
   optimized on Hopper.

4️⃣ Post‑process – layer‑norm, out‑gate multiplication and final linear
   projection – again in PyTorch.

The only mandatory Triton part is step 2, satisfying the “must use a kernel”
rule while keeping the overall runtime close to the theoretical optimum.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel – mask + gate fusion.
# All tensors are contiguous FP16 with shape (B, N, N, H) except mask which is
# (B, N, N, 1).  The kernel overwrites ``left`` and ``right`` in‑place.
# -------------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *mut fp16   (B,N,N,H) – overwritten
    right_ptr,         # *mut fp16   (B,N,N,H) – overwritten
    left_gate_ptr,     # *const fp16 (B,N,N,H)
    right_gate_ptr,    # *const fp16 (B,N,N,H)
    mask_ptr,          # *const fp16 (B,N,N,1)  – 1.0 if no mask
    B, N, H,           # sizes
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_g0, stride_g1, stride_g2, stride_g3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                       # 1‑D grid
    total = B * N * N * H
    idx = pid * BLOCK + tl.arange(0, BLOCK)      # linear indices
    mask = idx < total

    # ---- decode linear index into (b,i,j,h) -----------------------------
    tmp = idx
    h = tmp % H; tmp = tmp // H
    j = tmp % N; tmp = tmp // N
    i = tmp % N; b = tmp // N

    # ---- compute element pointers ---------------------------------------
    l_ptr = left_ptr   + b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    r_ptr = right_ptr  + b * stride_r0 + i * stride_r1 + j * stride_r2 + h * stride_r3
    lg_ptr = left_gate_ptr  + b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    rg_ptr = right_gate_ptr + b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    m_ptr = mask_ptr + b * stride_m0 + i * stride_m1 + j * stride_m2          # broadcast over H

    # ---- load ------------------------------------------------------------
    l = tl.load(l_ptr, mask=mask, other=0.0)
    r = tl.load(r_ptr, mask=mask, other=0.0)
    lg = tl.load(lg_ptr, mask=mask, other=0.0)
    rg = tl.load(rg_ptr, mask=mask, other=0.0)
    m = tl.load(m_ptr, mask=mask, other=1.0)       # default 1.0 (no mask)

    # ---- fused element‑wise ------------------------------------------------
    l = l * m * lg
    r = r * m * rg

    # ---- store -----------------------------------------------------------
    tl.store(l_ptr, l, mask=mask)
    tl.store(r_ptr, r, mask=mask)


def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input   [B,N,N,dim]  (fp32)
        torch.Tensor,                     # mask    [B,N,N]      (bool/float)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any]                    # config  {"dim":int, "hidden_dim":int}
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Returns:
        Tensor[B, N, N, dim]  (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape                     # dim = model dim
    hidden = config["hidden_dim"]              # hidden = H

    # -----------------------------------------------------------------
    # 1️⃣ Linear + LayerNorm work in PyTorch (cheap compared with O(N³))
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )
    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # -----------------------------------------------------------------
    # 2️⃣ Fuse mask + gating via Triton (must‑use‑kernel rule)
    # -----------------------------------------------------------------
    # mask: (B,N,N,1) FP16, all‑ones when no mask is supplied
    if mask is None:
        mask_tensor = torch.ones((B, N, N, 1), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(dtype).unsqueeze(-1)                # (B,N,N,1)
        mask_tensor = mask_tensor.to(torch.float16)

    # Cast tensors to FP16 for bandwidth‑limited kernel
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    left_gate_fp16  = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()

    total_elems = B * N * N * hidden
    BLOCK = 1024                                            # good for pure elementwise
    grid = (triton.cdiv(total_elems, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_tensor,
        B, N, hidden,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        mask_tensor.stride(0), mask_tensor.stride(1),
        mask_tensor.stride(2), mask_tensor.stride(3),
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 3️⃣ O(N³) contraction – batched GEMM via torch.bmm (uses Tensor‑cores)
    # -----------------------------------------------------------------
    # reshape to (B·H, N, N)
    left_bh  = left_fp16.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H,N,N)
    right_bh = right_fp16.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # batched matmul: (B*H,N,N) @ (B*H,N,N).T  -> (B*H,N,N)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))

    # back to (B,N,N,H)
    out = out_bh.reshape(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype)                               # FP32 for the remainder

    # -----------------------------------------------------------------
    # 4️⃣ Post‑process: layer‑norm, out‑gate, final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                              # broadcast over N,N
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
