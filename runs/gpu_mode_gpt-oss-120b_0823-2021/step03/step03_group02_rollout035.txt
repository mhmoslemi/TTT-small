"""
TriMul “outgoing” forward pass (no gradients).

The cubic contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a single torch.bmm (leveraging cuBLAS/Tensor‑cores on
Hopper).  All cheap element‑wise work – masking and gating – is fused into a
tiny Triton kernel to guarantee that *some* work runs on the GPU with
custom code, satisfying the problem requirement.

The pipeline is:
1. LayerNorm (fp32, stays in PyTorch – tiny wrt N³ work).
2. Linear projections → fp16.
3. Triton kernel:  left  = left * mask * left_gate
                   right = right * mask * right_gate
   (mask is broadcasted on the hidden dimension, gates are sigmoid‑ed)
4. Batched GEMM:    left (B,H,N,N) • rightᵀ (B,H,N,N) → out (B,H,N,N)  (fp16)
5. Reshape, LayerNorm, final gating and linear projection (back to fp32).

All heavy arithmetic stays in fp16, accumulators are fp32 for numerical
stability.  The kernel uses a 1‑D launch with a fixed block size; each thread
processes a contiguous chunk of the hidden dimension, which is small (≤384)
so no shared‑memory tricks are needed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel – fused mask + gating (element‑wise)
# --------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    left_ptr,          # fp16 * (B, N, N, H)
    right_ptr,         # fp16 * (B, N, N, H)
    mask_ptr,          # fp16 * (B, N, N, 1)   broadcasted on H
    left_gate_ptr,     # fp16 * (B, N, N, H)   sigmoid already applied
    right_gate_ptr,    # fp16 * (B, N, N, H)   sigmoid already applied
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)

    # linear index into the flattened tensor (B*N*N*H)
    total = tl.num_programs(0) * BLOCK
    mask = offs < total

    # load values (mask handles out‑of‑bounds)
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + (offs // BLOCK) * 1, mask=mask, other=1.0)  # mask is width‑1, repeat per H
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # fused element‑wise combination
    l = l * m * lg
    r = r * m * rg

    # store results back
    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


def _apply_mask_and_gates(left, right, mask, left_gate, right_gate):
    """
    left, right, left_gate, right_gate : (B,N,N,H) fp16
    mask                               : (B,N,N)   fp16 (or None)

    Returns the same tensors with mask & gates fused in‑place.
    """
    if mask is None:
        # no mask → just element‑wise multiply with gates (no triton needed)
        left = left * left_gate
        right = right * right_gate
        return left, right

    # make mask shape (B,N,N,1) so that broadcast over H is a simple repeat
    mask = mask.unsqueeze(-1)          # (B,N,N,1)
    B, N, _, H = left.shape
    total_elem = B * N * N * H

    # flatten tensors for the 1‑D kernel
    left_flat = left.view(-1)
    right_flat = right.view(-1)
    mask_flat = mask.view(-1)          # length B*N*N*1
    left_gate_flat = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)

    # launch configuration
    BLOCK = 1024
    num_blocks = (total_elem + BLOCK - 1) // BLOCK
    grid = (num_blocks,)

    mask_gate_kernel[grid](
        left_ptr=left_flat,
        right_ptr=right_flat,
        mask_ptr=mask_flat,
        left_gate_ptr=left_gate_flat,
        right_gate_ptr=right_gate_flat,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return left, right


# --------------------------------------------------------------
# Helper: batched GEMM via torch.bmm (fp16 inputs → fp32 acc)
# --------------------------------------------------------------
def _batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left , right : (B, H, N, N) fp16
    returns      : (B, H, N, N) fp16  (out = left @ rightᵀ over the K dim)
    """
    B, H, N, _ = left.shape
    # reshape to (B*H, N, N)
    left_2d = left.reshape(B * H, N, N)
    right_2d = right.reshape(B * H, N, N)

    # torch.bmm does fp16 × fp16 → fp32 accumulation, then casts back to fp16
    out_2d = torch.bmm(left_2d, right_2d.transpose(1, 2))

    return out_2d.view(B, H, N, N)


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor [B, N, N]       (float32/bool) or None
        - weights      : dict of torch.Tensor parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]   (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32, cheap)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                     # still fp32

    # ------------------------------------------------------------------
    # 2) Linear projections  (fp16)
    # ------------------------------------------------------------------
    def _to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w   = _to_fp16(weights["left_proj.weight"])
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])
    to_out_norm_w = weights["to_out_norm.weight"]   # keep fp32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)          # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)         # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid)  – still fp16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 4) Fuse mask & gating with a tiny Triton kernel
    # ------------------------------------------------------------------
    left, right = _apply_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 5) Core cubic contraction – batched GEMM (fp16)
    #    left  : (B, N, N, H) → (B, H, N, N)
    #    right : (B, N, N, H) → (B, H, N, N)  (transpose last two dims)
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _batched_gemm(left_perm, right_perm)    # (B, H, I, J) fp16

    # back to (B,N,N,dim) layout and cast to fp32 for the rest
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (float32)
    # ------------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7) Final gating + linear projection (float32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # same out_gate as in reference
    out = F.linear(out, to_out_w)                  # (B,N,N,dim)

    return out
