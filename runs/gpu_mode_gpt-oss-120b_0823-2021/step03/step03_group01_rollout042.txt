"""
custom_kernel – “outgoing” TriMul forward pass (AlphaFold‑3 style).

The dominant O(N³) contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is performed by a fused batched GEMM Triton kernel (fp16 compute, fp32
accumulate).  All other operations (LayerNorm, linear projections, gating and
the final LayerNorm + linear) stay in PyTorch – this gives a very small
launch overhead while still keeping the heavy work on Tensor‑Core‑friendly
code path.

Input
    data = (input, mask, weights, config)
        input : (B, N, N, dim)    float32
        mask  : (B, N, N) or None (float32 / bool)
        weights: dict with the same keys as the reference PyTorch module
        config : {"dim": dim, "hidden_dim": hidden_dim}
Output
    (B, N, N, dim) float32
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (I×K)·(K×J) → (I×J)   (fp16 compute, fp32 acc)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,          # left   : (Z, I, K)
    b_ptr,          # right  : (Z, K, J)
    c_ptr,          # out    : (Z, I, J)
    M, N, K,        # sizes  : I, J, K
    stride_az, stride_am, stride_ak,   # a strides (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # program IDs – tile indices (row, col, batch)
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_z = tl.program_id(2)          # batch (Z = B * hidden_dim)

    # --------------------------------------------------------------
    # offsets within the matrix
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (higher precision for the sum over K)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over K dimension in tiles
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # write back (convert to fp16 to reduce memory traffic)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous
    returns: (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # allocate output
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse leading dims => Z = B*H  (kernel works on 3‑D tensors)
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides in element units (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tile sizes – chosen for Hopper Tensor‑Core friendliness
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # rows
        triton.cdiv(J, BLOCK_N),   # cols
        Z,                         # batch*head
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for SM90
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass – PyTorch + Triton
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor  [B, N, N, dim]   float32
        mask         : torch.Tensor  [B, N, N]       float32/bool or None
        weights      : dict of torch.Tensor parameters (see docstring above)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]  float32
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. Input LayerNorm   (fp32, keep highest precision)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual LayerNorm to stay in pure PyTorch (fast enough)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                                   # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2. Linear projections – cast to fp16 for the heavy GEMM
    # --------------------------------------------------------------
    def to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w  = to_fp16(weights["left_proj.weight"])   # (hidden_dim, dim)
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w  = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w   = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None:
        # mask may be bool or float – convert to fp16 and add channel dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) – still fp16
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden_dim)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton batched GEMM (N³ contraction)
    #    left  : (B, H, I, K)   = (B, hidden_dim, N, N)
    #    right : (B, H, K, J)   = (B, hidden_dim, N, N) with K,J swapped
    # --------------------------------------------------------------
    #   permute to bring the hidden_dim axis before the two spatial axes
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _batched_gemm(left_perm, right_perm)    # (B, H, I, J)

    # Back to original layout and to fp32 for the rest of the pipeline
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,dim)

    # --------------------------------------------------------------
    # 6. Output LayerNorm
    # --------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = out * to_out_norm_w + to_out_norm_b

    # --------------------------------------------------------------
    # 7. Final gating + linear projection back to dim
    # --------------------------------------------------------------
    # out_gate is still fp16 – cast to fp32 for the element‑wise mul
    out = out * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"]                # (dim, dim)
    out = F.linear(out, to_out_w)                      # (B,N,N,dim) fp32

    return out
