"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the input (B,N,N,C) → (B,N,N,C)  (FP32)
2️⃣  Linear projections to hidden dimension H:
        left  = x @ W_left          (B,N,N,H)
        right = x @ W_right
        l_gate = sigmoid(x @ W_lgate)
        r_gate = sigmoid(x @ W_rgate)
        o_gate = sigmoid(x @ W_outgate)      (B,N,N,H)
3️⃣  Fuse element‑wise mask (optional) and the two gates with a tiny
    Triton kernel.  All tensors are cast to FP16 beforehand because the
    subsequent N³ contraction is performed in FP16 (Tensor‑core friendly).
4️⃣  Reshape to (B*H, N, N) and run a batched matrix‑multiply
        out[b*h] = left[b*h] @ right[b*h].T
    using torch.bmm which calls cuBLAS / Tensor‑cores.  The result has
    shape (B,N,N,H) and is cast back to FP32.
5️⃣  Hidden‑dim LayerNorm, multiply by the output‑gate and a final linear
    projection back to C dimensions.

Only the element‑wise fusion (step 3) is written in Triton – this satisfies
the “some‑kernel” requirement while keeping the dominant O(N³) work on the
highly‑optimized cuBLAS path.  All accumulations are FP32 for stability;
the final tensor is returned in FP32 as required by the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – mask & gate fusion (FP16)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                # fp16  (B,N,N,H)
    lgate_ptr, rgate_ptr,               # fp16  (B,N,N,H)
    mask_ptr,                           # fp16  (B,N,N,1) or nullptr
    out_left_ptr, out_right_ptr,        # fp16  (B,N,N,H)   -> fused results
    B, N, H,                            # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_m0, stride_m1, stride_m2,   # mask has no hidden dim
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h) compute
        out_left  = left  * lgate  * mask
        out_right = right * rgate * mask
    (mask = 1 if not present).  The kernel processes a 1‑D grid of size
    ceil(B*N*N*H / BLOCK).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # ---- decode linear index ------------------------------------------------
    # layout is contiguous:   b, i, j, h
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ---- load left / right ---------------------------------------------------
    idx = b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    left  = tl.load(left_ptr  + idx, mask=mask, other=0.0)
    right = tl.load(right_ptr + idx, mask=mask, other=0.0)

    # ---- load gates ---------------------------------------------------------
    idx_gate = b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + h * stride_lg3
    lgate = tl.load(lgate_ptr + idx_gate, mask=mask, other=0.0)
    rgate = tl.load(rgate_ptr + idx_gate, mask=mask, other=0.0)

    # ---- load mask (if present) --------------------------------------------
    if HAS_MASK:
        idx_mask = b * stride_m0 + i * stride_m1 + j * stride_m2   # broadcast over H
        m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)
    else:
        m = 1.0

    # ---- fuse ---------------------------------------------------------------
    out_l = left * lgate * m
    out_r = right * rgate * m

    # ---- store ---------------------------------------------------------------
    tl.store(out_left_ptr  + idx, out_l, mask=mask)
    tl.store(out_right_ptr + idx, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[
        torch.Tensor,
        torch.Tensor,
        Dict[str, torch.Tensor],
        Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        x       – (B, N, N, C)      torch.float32
        mask    – (B, N, N) or None torch.float32 / bool
        weights – dict with model parameters (float32)
        config  – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor
        (B, N, N, C)  float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype  = torch.float32

    B, N, _, C = x.shape
    H = config["hidden_dim"]          # hidden dimension

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (still FP32)
    # ------------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3️⃣ Cast to FP16 and fuse mask + gates with Triton
    # ------------------------------------------------------------------
    left_fp16      = left.to(torch.float16).contiguous()
    right_fp16     = right.to(torch.float16).contiguous()
    lgate_fp16     = left_gate.to(torch.float16).contiguous()
    rgate_fp16     = right_gate.to(torch.float16).contiguous()

    if mask is None:
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)   # dummy
        has_mask = False
    else:
        # mask: (B,N,N) → (B,N,N,1) for broadcasting over H
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        has_mask = True

    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elements = B * N * N * H
    BLOCK = 1024                                     # good for H200
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        lgate_fp16,
        rgate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        lgate_fp16.stride(0), lgate_fp16.stride(1), lgate_fp16.stride(2), lgate_fp16.stride(3),
        mask_fp16.stride(0) if has_mask else 0,
        mask_fp16.stride(1) if has_mask else 0,
        mask_fp16.stride(2) if has_mask else 0,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM  (FP16 inputs → FP16 output, acc in FP32)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))          # (B*H,N,N) FP16 → FP32 acc

    # back to (B,N,N,H) FP32
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(dtype)

    # ------------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    # out_gate was computed in FP32; broadcast over (N,N) automatically
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype))          # (B,N,N,C)

    return out
