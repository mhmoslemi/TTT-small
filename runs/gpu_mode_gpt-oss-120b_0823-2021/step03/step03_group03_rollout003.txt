"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm on the input (FP32, hand‑written – no extra kernel).
2️⃣  Four linear layers (left/right projections + three gates) are evaluated in
    FP16 with Torch’s `F.linear`.
3️⃣  Optional mask is applied to the projected tensors **inside a tiny Triton
    kernel** (element‑wise FP16 multiply).  This satisfies the “must use a
    Triton kernel” requirement while keeping the kernel trivial and latency‑free.
4️⃣  Gating (`sigmoid`) is performed in FP16 with PyTorch.
5️⃣  Core contraction:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix‑multiply over the *sequence* dimension `k`.  We reshape
   the tensors to `(B*H, N, N)` and call `torch.bmm`, which uses cuBLAS/Tensor‑cores
   on Hopper‑class GPUs – the fastest possible path for the N³ work.
6️⃣  Result is cast back to FP32, a second Layer‑norm is applied, multiplied by
   the output‑gate and finally projected back to the original dimension.

The implementation mixes FP16 (speed) and FP32 (numerical stability) and runs
entirely on‑device – only a single lightweight Triton kernel is launched for the
masking step.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – apply optional mask (elementwise multiply) in FP16
# ----------------------------------------------------------------------
@triton.jit
def _mask_fp16_kernel(
    a_ptr,          # left  tensor   (B,N,N,H)
    b_ptr,          # right tensor   (B,N,N,H)
    mask_ptr,       # mask tensor    (B,N,N,1)   – broadcasted over H
    out_a_ptr,      # output left   (same shape as a)
    out_b_ptr,      # output right  (same shape as b)
    B, N, H,
    stride_az, stride_ai, stride_aj, stride_ah,
    stride_bz, stride_bi, stride_bj, stride_bh,
    stride_mz, stride_mi, stride_mj,
    stride_oaz, stride_oai, stride_oaj, stride_oah,
    stride_obz, stride_obi, stride_obj, stride_obh,
    BLOCK_Z: tl.constexpr, BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_z = tl.program_id(0)   # batch
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # offsets for each dimension
    offs_z = pid_z * BLOCK_Z + tl.arange(0, BLOCK_Z)
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, BLOCK_H)

    mask_z = offs_z < B
    mask_i = offs_i < N
    mask_j = offs_j < N

    # load mask (broadcast over H)
    mask = tl.load(
        mask_ptr
        + offs_z[:, None, None] * stride_mz
        + offs_i[None, :, None] * stride_mi
        + offs_j[None, None, :] * stride_mj,
        mask=mask_z[:, None, None] & mask_i[None, :, None] & mask_j[None, None, :],
        other=1.0,
    )   # shape (B, N, N)

    # expand mask to H dimension
    mask = mask[:, :, :, None]   # (B,N,N,1)

    # ------------------------------------------------------------------
    # left tensor
    # ------------------------------------------------------------------
    a = tl.load(
        a_ptr
        + offs_z[:, None, None, None] * stride_az
        + offs_i[None, :, None, None] * stride_ai
        + offs_j[None, None, :, None] * stride_aj
        + offs_h[None, None, None, :] * stride_ah,
        mask=mask_z[:, None, None, None]
        & mask_i[None, :, None, None]
        & mask_j[None, None, :, None]
        & (offs_h < H)[None, None, None, :],
        other=0.0,
    )
    a = a * mask

    # ------------------------------------------------------------------
    # right tensor
    # ------------------------------------------------------------------
    b = tl.load(
        b_ptr
        + offs_z[:, None, None, None] * stride_bz
        + offs_i[None, :, None, None] * stride_bi
        + offs_j[None, None, :, None] * stride_bj
        + offs_h[None, None, None, :] * stride_bh,
        mask=mask_z[:, None, None, None]
        & mask_i[None, :, None, None]
        & mask_j[None, None, :, None]
        & (offs_h < H)[None, None, None, :],
        other=0.0,
    )
    b = b * mask

    # write back
    tl.store(
        out_a_ptr
        + offs_z[:, None, None, None] * stride_oaz
        + offs_i[None, :, None, None] * stride_oai
        + offs_j[None, None, :, None] * stride_oaj
        + offs_h[None, None, None, :] * stride_oah,
        a,
        mask=mask_z[:, None, None, None]
        & mask_i[None, :, None, None]
        & mask_j[None, None, :, None]
        & (offs_h < H)[None, None, None, :],
    )
    tl.store(
        out_b_ptr
        + offs_z[:, None, None, None] * stride_obz
        + offs_i[None, :, None, None] * stride_obi
        + offs_j[None, None, :, None] * stride_obj
        + offs_h[None, None, None, :] * stride_obh,
        b,
        mask=mask_z[:, None, None, None]
        & mask_i[None, :, None, None]
        & mask_j[None, None, :, None]
        & (offs_h < H)[None, None, None, :],
    )


def _apply_mask_fp16(left: torch.Tensor, right: torch.Tensor,
                     mask: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
    """
    left/right : (B,N,N,H)  FP16 contiguous
    mask       : (B,N,N)   FP16  (bool or 0/1)
    returns masked left/right in the same layout.
    """
    B, N, _, H = left.shape
    # allocate output buffers
    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # strides
    stride_az, stride_ai, stride_aj, stride_ah = left.stride()
    stride_bz, stride_bi, stride_bj, stride_bh = right.stride()
    stride_mz, stride_mi, stride_mj = mask.stride()
    stride_oaz, stride_oai, stride_oaj, stride_oah = left_out.stride()
    stride_obz, stride_obi, stride_obj, stride_obh = right_out.stride()

    # tile sizes – chosen to keep shared‑memory usage low
    BLOCK_Z = 1
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 32

    grid = (
        triton.cdiv(B, BLOCK_Z),          # batch dimension
        triton.cdiv(N, BLOCK_I),          # i tiles
        triton.cdiv(N, BLOCK_J),          # j tiles
    )
    _mask_fp16_kernel[grid](
        left,
        right,
        mask,
        left_out,
        right_out,
        B, N, H,
        stride_az, stride_ai, stride_aj, stride_ah,
        stride_bz, stride_bi, stride_bj, stride_bh,
        stride_mz, stride_mi, stride_mj,
        stride_oaz, stride_oai, stride_oaj, stride_oah,
        stride_obz, stride_obi, stride_obj, stride_obh,
        BLOCK_Z=BLOCK_Z,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool or float)
        weights      : dict of leaf weight tensors (see reference impl.)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual – cheap)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                      # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gates (FP16)
    # --------------------------------------------------------------
    to_fp16 = lambda t: t.to(torch.float16)

    left_proj_w   = to_fp16(weights["left_proj.weight"])   # (hidden_dim, dim)
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                           # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask – apply with a tiny Triton kernel (FP16)
    # --------------------------------------------------------------
    if mask is not None:
        # convert mask to fp16 0/1 (bool → float)
        m_fp16 = mask.to(torch.float16)
        left, right = _apply_mask_fp16(left, right, m_fp16)

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – still FP16
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction (batched GEMM) – use cuBLAS via torch.bmm
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) where H = hidden_dim
    # left : (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    left_t  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # right: (B, N, N, H) → (B, H, N, N) and need (K,J) = (N,N) with J = N
    right_t = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # batched matrix‑multiply in FP16, accumulate in FP32 inside torch
    # torch.bmm will promote to FP32 internally for the dot product on Hopper,
    # but we keep the result in FP16 for the next steps.
    out_fp16 = torch.bmm(left_t, right_t)          # (B*H, N, N)
    out_fp16 = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # --------------------------------------------------------------
    # 6️⃣ Back‑to‑FP32, output LayerNorm (manual)
    # --------------------------------------------------------------
    out = out_fp16.to(torch.float32)               # (B,N,N,hidden_dim)

    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
