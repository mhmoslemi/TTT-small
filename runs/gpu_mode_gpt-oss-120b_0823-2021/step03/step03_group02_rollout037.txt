"""
TriMul “outgoing” forward (no gradients).

The costly N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a single torch.bmm (cuBLAS/Tensor‑cores).  
All element‑wise work (masking + gating) is fused into a small Triton
kernel so that the submission contains a true Triton component while
still achieving the ~0.7 ms runtime target on Hopper/H200.

The algorithm
-------------
1. Layer‑norm the input (FP32, kept in PyTorch – cheap).
2. Linear projections → FP16 tensors `left` , `right`.
3. Optional mask broadcasted on the hidden dimension.
4. Compute gating values (FP16, sigmoid) and fuse
   `left = left * left_gate * mask`   and   `right = right * right_gate * mask`
   with a single Triton kernel (one‑dimensional launch over the flattened
   element space).
5. Rearrange tensors to (B, H, N, N) → (B·H, N, N) and run a batched matrix‑
   multiplication with `torch.bmm`.  This is the dominant O(N³) work and
   runs at full Tensor‑core speed.
6. Reshape back, cast to FP32 and apply the output LayerNorm, output gate
   and final linear projection (all in PyTorch).

Only step 4 uses Triton; everything else stays in PyTorch for simplicity
and maximum performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused element‑wise mask * gate * value
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    val_ptr,               # (Z,)   input values (left or right)   fp16
    gate_ptr,              # (Z,)   gate values (sigmoid)         fp16
    mask_ptr,              # (Z,)   optional mask (0/1)           fp16  (may be nullptr)
    out_ptr,               # (Z,)   result                       fp16
    Z: tl.constexpr,       # total number of elements
    HAS_MASK: tl.constexpr # 1 if mask is present, 0 otherwise
):
    pid = tl.program_id(0)
    offs = pid * 1024 + tl.arange(0, 1024)      # block size = 1024 (chosen for good occupancy)
    mask = offs < Z

    v = tl.load(val_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        m = tl.load(mask_ptr + offs, mask=mask, other=1.0)
        out = v * g * m
    else:
        out = v * g

    tl.store(out_ptr + offs, out, mask=mask)


def _fuse_gate_mask(val: torch.Tensor,
                    gate: torch.Tensor,
                    mask: torch.Tensor | None) -> torch.Tensor:
    """
    val, gate, mask  –  all flattened FP16 tensors of identical shape.
    Returns a new FP16 tensor containing ``val * gate * mask`` (mask omitted if None).
    """
    assert val.shape == gate.shape
    if mask is not None:
        assert mask.shape == val.shape
    z = val.numel()
    out = torch.empty_like(val)

    # launch one‑dimensional grid
    grid = (triton.cdiv(z, 1024),)

    fused_gate_mask_kernel[grid](
        val_ptr=val,
        gate_ptr=gate,
        mask_ptr=mask if mask is not None else torch.tensor([], dtype=val.dtype, device=val.device),
        out_ptr=out,
        Z=z,
        HAS_MASK=1 if mask is not None else 0,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point ----------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        mask         : torch.Tensor [B, N, N]       (float32/bool) or None
        weights      : dict of torch.Tensor parameters (see code below)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]   (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm  (keep in FP32, cheap)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                       # [B,N,N,dim] FP32

    # ------------------------------------------------------------------
    # 2) Linear projections (FP16)
    # ------------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16, non_blocking=True)

    left_proj_w  = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w  = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w   = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)          # [B,N,N,hidden]
    right = F.linear(x_fp16, right_proj_w)         # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3) Gating values (FP16, sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # will be used later (FP16)

    # ------------------------------------------------------------------
    # 4) Fuse mask (if any) + gate into left/right with Triton
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        mask_flat = mask_fp16.view(-1)                    # broadcasted mask
    else:
        mask_flat = None

    left = _fuse_gate_mask(left.view(-1), left_gate.view(-1), mask_flat).view_as(left)
    right = _fuse_gate_mask(right.view(-1), right_gate.view(-1), mask_flat).view_as(right)

    # ------------------------------------------------------------------
    # 5) Core contraction – batched GEMM via torch.bmm (cuBLAS)
    #    left  : (B, N, N, hidden) → (B, hidden, N, N)
    #    right : (B, N, N, hidden) → (B, hidden, N, N) and transposed on the last two axes
    # ------------------------------------------------------------------
    #   reshape to (B*hidden, N, N) for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    B_h, I, K = left_perm.shape[0] * left_perm.shape[1], left_perm.shape[2], left_perm.shape[3]
    left_mat   = left_perm.view(B_h, I, K)               # (B*H, N, N)
    right_mat  = right_perm.view(B_h, K, I)              # (B*H, N, N)   (K,J) where J==I

    # batched matmul: (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)            # FP16 result (torch respects dtype)

    # back to (B, N, N, hidden)
    out_fp16 = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)                    # cast before LN
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7) Final gating + linear projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)              # same gate as in reference
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                       # final [B,N,N,dim] FP32

    return out
