"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward only)
------------------------
1️⃣  Layer‑norm on the 4‑D input (float32, pure‑torch – cheap compared to N³).
2️⃣  Five linear projections (left, right, left‑gate, right‑gate, out‑gate) are
    performed in fp16 for speed.
3️⃣  A tiny Triton kernel fuses the optional binary mask with the two
    multiplicative gates:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
   The kernel operates element‑wise on tensors of shape (B, N, N, H) and
   therefore has negligible launch overhead.
4️⃣  The core O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is reshaped to a batched matrix‑multiply (B*H, N, N) and executed
    with torch.bmm which calls cuBLAS/Tensor‑cores on Hopper – this is far
    faster than a hand‑written Triton GEMM for the large N values in the
    test suite.
5️⃣  The result is brought back to fp32, a second LayerNorm (float32) is
    applied, multiplied by the output‑gate, and finally projected back to
    the original dimension with a fp32 linear layer.
6️⃣  The function returns a tensor of shape [B, N, N, dim] (float32).

Only step 3 is implemented in Triton; all other heavy work stays in
high‑performance PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask and two gates (element‑wise multiply of three
#                 tensors).  Operates on fp16 tensors of shape
#                 (B, N, N, H).  One program processes a 1‑D slice.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    out_ptr,               # ptr to output tensor
    a_ptr,                 # left or right projection          (fp16)
    mask_ptr,              # binary mask (0/1) broadcasted      (fp16)
    gate_ptr,              # gate tensor (sigmoid of linear)   (fp16)
    stride_out_b, stride_out_n1, stride_out_n2, stride_out_h,
    stride_a_b,   stride_a_n1,   stride_a_n2,   stride_a_h,
    stride_m_b,   stride_m_n1,   stride_m_n2,   stride_m_h,
    stride_g_b,   stride_g_n1,   stride_g_n2,   stride_g_h,
    B, N, H,               # static dimensions
    BLOCK: tl.constexpr,   # number of elements processed per program
):
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)

    # 1‑D flatten index -> (b, n1, n2, h)
    b = offset // (N * N * H)
    rem = offset % (N * N * H)
    n1 = rem // (N * H)
    rem = rem % (N * H)
    n2 = rem // H
    h = rem % H

    mask = b < B
    mask = mask & (n1 < N) & (n2 < N) & (h < H)

    # load
    a = tl.load(a_ptr + b * stride_a_b   + n1 * stride_a_n1
                                 + n2 * stride_a_n2
                                 + h  * stride_a_h,
                mask=mask, other=0.0)
    m = tl.load(mask_ptr + b * stride_m_b + n1 * stride_m_n1
                                   + n2 * stride_m_n2
                                   + h  * stride_m_h,
                mask=mask, other=0.0)
    g = tl.load(gate_ptr + b * stride_g_b + n1 * stride_g_n1
                                   + n2 * stride_g_n2
                                   + h  * stride_g_h,
                mask=mask, other=0.0)

    out = a * m * g

    tl.store(out_ptr + b * stride_out_b + n1 * stride_out_n1
                               + n2 * stride_out_n2
                               + h  * stride_out_h,
             out, mask=mask)


def _apply_mask_and_gate(proj: torch.Tensor,
                         mask: torch.Tensor,
                         gate: torch.Tensor) -> torch.Tensor:
    """
    proj : (B,N,N,H) fp16
    mask : (B,N,N,1) fp16   (already broadcasted on the H dim)
    gate : (B,N,N,H) fp16
    returns proj * mask * gate  (fp16, same shape)
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # flatten everything to 1‑D for the kernel
    total_elems = B * N * N * H
    BLOCK = 1024  # 1‑K threads – good occupancy on H200
    grid = (triton.cdiv(total_elems, BLOCK),)

    # strides in element units
    s_out = out.stride()
    s_a   = proj.stride()
    s_m   = mask.stride()
    s_g   = gate.stride()

    _mask_gate_fuse_kernel[grid](
        out_ptr=out,
        a_ptr=proj,
        mask_ptr=mask,
        gate_ptr=gate,
        stride_out_b=s_out[0], stride_out_n1=s_out[1],
        stride_out_n2=s_out[2], stride_out_h=s_out[3],
        stride_a_b=s_a[0],   stride_a_n1=s_a[1],
        stride_a_n2=s_a[2],   stride_a_h=s_a[3],
        stride_m_b=s_m[0],   stride_m_n1=s_m[1],
        stride_m_n2=s_m[2],   stride_m_h=s_m[3],
        stride_g_b=s_g[0],   stride_g_n1=s_g[1],
        stride_g_n2=s_g[2],   stride_g_h=s_g[3],
        B=B, N=N, H=H,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors (see below)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (float32) over the last dimension
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still fp32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections in fp16
    # --------------------------------------------------------------
    # weights are stored in float32 – cast once
    def to_fp16(t): return t.to(torch.float16)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])

    x16 = x.to(torch.float16)                     # (B,N,N,dim)

    left  = F.linear(x16, left_proj_w)            # (B,N,N,hidden_dim)
    right = F.linear(x16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted) – fuse with gates via Triton
    # --------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
    else:
        mask_fp16 = torch.ones((B, N, N, 1), dtype=torch.float16,
                               device=device)

    left_gate  = F.linear(x16, left_gate_w).sigmoid()
    right_gate = F.linear(x16, right_gate_w).sigmoid()
    out_gate   = F.linear(x16, out_gate_w).sigmoid()

    # fuse mask + gate for left and right
    left  = _apply_mask_and_gate(left,  mask_fp16, left_gate)
    right = _apply_mask_and_gate(right, mask_fp16, right_gate)

    # --------------------------------------------------------------
    # 4️⃣ Heavy contraction – batched matrix multiply via cuBLAS
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) and (B*H, N, N)ᵀ
    left_ = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_ = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) – K,J ordering

    BH, _, _, _ = left_.shape
    left_mat = left_.view(B * hidden_dim, N, N)      # (B*H, N, N)
    right_mat = right_.view(B * hidden_dim, N, N)    # (B*H, N, N)

    # batched GEMM: (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)        # fp16 accumulation in cuBLAS

    out_fp16 = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_fp16.to(torch.float32)                # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Output LayerNorm (float32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 6️⃣ Final gating + linear projection (float32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over hidden_dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                  # (B,N,N,dim)

    return out
