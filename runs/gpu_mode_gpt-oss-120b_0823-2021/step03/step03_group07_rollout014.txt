"""
TriMul “outgoing” forward pass (AlphaFold‑3).

This implementation follows the reference PyTorch module but off‑loads the
O(N³) contraction

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

to a tiny Triton kernel that performs a blocked GEMM on the
[B, H, N, N] tensors (left and rightᵀ).  All cheap operations
(LayerNorm, linear projections, gates, final LayerNorm + linear) are
performed with regular PyTorch which is memory‑bound and far cheaper
than the cubic work.  The kernel works on fp16 data with fp32
accumulation and writes the result back as fp16; the surrounding host
code then reshapes / casts to the required layout and dtype.

The kernel is deliberately simple:
* one program id per (batch, head, tile‑M, tile‑N) block,
* a K‑loop that loads a tile of `left` (M×K) and a *transposed* tile of
  `right` (K×N) and accumulates with `tl.dot`,
* fp16‑fp32‑fp16 conversion to keep Tensor‑cores happy while preserving
  numeric quality.

The launch parameters (BLOCK_M, BLOCK_N, BLOCK_K) have been tuned for
the H200’s Tensor‑core friendly geometry (multiples of 32) and fit well
within shared‑memory limits.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the cubic contraction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # const fp16*  [B, H, N, N]   (M‑K layout)
    right_ptr,         # const fp16*  [B, H, N, N]   (will be read transposed)
    out_ptr,           #       fp16*  [B, H, N, N]   (M‑N layout)
    B, N, H,           # meta‑data (int32)
    # strides (row‑major) of the three tensors
    stride_lb, stride_lh, stride_ln, stride_lk,   # left
    stride_rb, stride_rh, stride_rn, stride_rk,   # right
    stride_ob, stride_oh, stride_on, stride_ok,   # out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a (BLOCK_M × BLOCK_N) tile of

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    The kernel is launched with one program ID per (b, h, tile_m, tile_n).
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # decode the linear pid into (b, h, tile_m, tile_n)
    # ------------------------------------------------------------------
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    tile_n = pid % num_n
    pid //= num_n
    tile_m = pid % num_m
    pid //= num_m
    h = pid % H
    b = pid // H

    # ------------------------------------------------------------------
    # tile offsets & masks
    # ------------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # fp32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # main K loop
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, h, i, k]  -> shape (M, K)
        l_ptr = (
            left_ptr
            + b * stride_lb
            + h * stride_lh
            + offs_m[:, None] * stride_ln
            + offs_k[None, :] * stride_lk
        )
        l = tl.load(
            l_ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right[b, h, j, k]  (we need its transpose: (K, N))
        #   original indexing: (b, h, j, k)
        #   after transpose we view it as (k, j)
        r_ptr = (
            right_ptr
            + b * stride_rb
            + h * stride_rh
            + offs_k[:, None] * stride_rk   # k‑dim becomes row
            + offs_n[None, :] * stride_rn   # j‑dim becomes col
        )
        r_t = tl.load(
            r_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # GEMM accumulate
        acc += tl.dot(l, r_t)

    # ------------------------------------------------------------------
    # write result back (fp16)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + h * stride_oh
        + offs_m[:, None] * stride_on
        + offs_n[None, :] * stride_ok
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input   [B, N, N, dim]
        torch.Tensor,          # mask    [B, N, N]   (or None)
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32               # final output dtype

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]   # = H

    # --------------------------------------------------------------
    # 1. Cheap ops in PyTorch (LayerNorm, Linear, gating, optional mask)
    # --------------------------------------------------------------
    # LayerNorm over the feature dimension
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # Linear projections (bias=False)
    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # Apply mask if present
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # Gates (sigmoid after linear)
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(dtype))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(dtype))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(dtype))
    )

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 2. Rearrange to [B, H, N, N] and cast to fp16 for the kernel
    # --------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)   # (B,H,N,N), fp16

    # --------------------------------------------------------------
    # 3. Launch Triton kernel (blocked GEMM)
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # grid = total number of (b, h, tile_m, tile_n) blocks
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (B * hidden_dim * grid_m * grid_n,)

    _trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        # strides for left (row‑major)
        left_fp16.stride(0),   # B
        left_fp16.stride(1),   # H
        left_fp16.stride(2),   # N (row i)
        left_fp16.stride(3),   # N (col k)
        # strides for right
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        # strides for out
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 4. Post‑process (to_out Norm, gate, final linear)
    # --------------------------------------------------------------
    # reshape back to (B,N,N,H) and cast to float32 for the remaining ops
    out = out_fp16.to(dtype).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # to_out LayerNorm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply output gate (broadcast on pairwise axes)
    out = out * out_gate

    # final linear back to original `dim`
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
