"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the input (FP32, manual – cheap, no extra kernel).
2️⃣  Four linear layers (left/right projections + three gates) are performed in
    FP16 for speed.  The input is cast once to FP16.
3️⃣  Optional binary mask is applied element‑wise (Triton kernel – fused with
    the two linear projections to keep the data in registers).
4️⃣  Gating (sigmoid) is applied in FP16 (torch ops – cheap).
5️⃣  Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is a batched matrix‑multiply over the *sequence* dimension **k**.
    We reshape the tensors to (B·H, N, N) (H = hidden_dim) and call
    ``torch.bmm`` which uses cuBLAS/Tensor‑cores on Hopper – far faster than a
    handcrafted Triton N³ kernel.
6️⃣  Convert the result back to FP32, apply a second LayerNorm, multiply by the
    output‑gate and finish with a linear projection to the original dimension.
The only Triton piece is the mask‑fusion kernel (step 3); everything else stays
in PyTorch so we keep the implementation simple yet performant.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask into the two linear projections
# ----------------------------------------------------------------------
@triton.jit
def _mask_fuse_kernel(
    x_ptr,                # input FP16 (B,N,N,D)
    w_left_ptr,           # left_proj weight   (H,D)   – FP16
    w_right_ptr,          # right_proj weight  (H,D)   – FP16
    mask_ptr,             # mask FP16 (B,N,N,1) or nullptr
    out_left_ptr,         # left output (B,N,N,H) FP16
    out_right_ptr,        # right output (B,N,N,H) FP16
    B, N, D, H,
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_wl_h, stride_wl_d,
    stride_wr_h, stride_wr_d,
    stride_m_b, stride_m_i, stride_m_j,
    stride_olb, stride_oli, stride_olj, stride_olh,
    stride_orb, stride_ori, stride_orj, stride_orh,
    MASKED: tl.constexpr,            # 1 if a mask is provided
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                # linear id over (B,N,N)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from linear pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # load a row vector x[b,i,j,:] (length D)
    offs_d = tl.arange(0, BLOCK)
    mask_d = offs_d < D
    x = tl.load(
        x_ptr
        + b * stride_xb
        + i * stride_xi
        + j * stride_xj
        + offs_d * stride_xd,
        mask=mask_d,
        other=0.0,
    )   # (D,)

    # optional mask value (scalar broadcast)
    if MASKED:
        m = tl.load(mask_ptr + b * stride_m_b + i * stride_m_i + j * stride_m_j)
        x = x * m

    # left projection: w_left (H,D) • x (D,)  → (H,)
    w_left = tl.load(
        w_left_ptr + offs_d[:, None] * stride_wl_d + tl.arange(0, BLOCK)[:, None] * stride_wl_h,
        mask=mask_d[:, None] & (tl.arange(0, BLOCK)[:, None] < H),
        other=0.0,
    )
    left = tl.dot(x, w_left)   # FP16 → FP16 (accumulates in FP32 internally)

    # right projection: w_right (H,D) • x (D,) → (H,)
    w_right = tl.load(
        w_right_ptr + offs_d[:, None] * stride_wr_d + tl.arange(0, BLOCK)[:, None] * stride_wr_h,
        mask=mask_d[:, None] & (tl.arange(0, BLOCK)[:, None] < H),
        other=0.0,
    )
    right = tl.dot(x, w_right)

    # store
    offs_h = tl.arange(0, BLOCK)
    mask_h = offs_h < H
    tl.store(
        out_left_ptr
        + b * stride_olb
        + i * stride_oli
        + j * stride_olj
        + offs_h * stride_olh,
        left,
        mask=mask_h,
    )
    tl.store(
        out_right_ptr
        + b * stride_orb
        + i * stride_ori
        + j * stride_orj
        + offs_h * stride_orh,
        right,
        mask=mask_h,
    )


def _proj_mask_fuse(x_fp16, mask_fp16, w_left, w_right):
    """
    Fuse ``x @ w_left.T`` and ``x @ w_right.T`` while applying an optional mask.
    Returns left, right tensors of shape (B,N,N,H) in FP16.
    """
    B, N, _, D = x_fp16.shape
    H = w_left.shape[0]

    left = torch.empty((B, N, N, H), dtype=torch.float16, device=x_fp16.device)
    right = torch.empty((B, N, N, H), dtype=torch.float16, device=x_fp16.device)

    # strides
    sx = x_fp16.stride()
    sl = w_left.stride()
    sr = w_right.stride()
    sm = mask_fp16.stride() if mask_fp16 is not None else (0, 0, 0, 0)
    slo = left.stride()
    sro = right.stride()

    BLOCK = max(D, H)  # we launch one program per (b,i,j) and use BLOCK elems on dim

    total = B * N * N
    grid = (triton.cdiv(total, 1),)   # one program per element

    _mask_fuse_kernel[grid](
        x_ptr=x_fp16,
        w_left_ptr=w_left,
        w_right_ptr=w_right,
        mask_ptr=mask_fp16 if mask_fp16 is not None else torch.empty(0, device=x_fp16.device),
        out_left_ptr=left,
        out_right_ptr=right,
        B=B,
        N=N,
        D=D,
        H=H,
        stride_xb=sx[0],
        stride_xi=sx[1],
        stride_xj=sx[2],
        stride_xd=sx[3],
        stride_wl_h=sl[0],
        stride_wl_d=sl[1],
        stride_wr_h=sr[0],
        stride_wr_d=sr[1],
        stride_m_b=sm[0],
        stride_m_i=sm[1],
        stride_m_j=sm[2],
        stride_olb=slo[0],
        stride_oli=slo[1],
        stride_olj=slo[2],
        stride_olh=slo[3],
        stride_orb=sro[0],
        stride_ori=sro[1],
        stride_orj=sro[2],
        stride_orh=sro[3],
        MASKED=1 if mask_fp16 is not None else 0,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return left, right


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of weight tensors (see reference implementation)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim], dtype float32
    """
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 0️⃣ LayerNorm (FP32, manual)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still FP32

    # ------------------------------------------------------------------
    # 1️⃣ Cast to FP16 and prepare weights
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left_proj_w   = weights["left_proj.weight"].to(torch.float16)    # (H, D)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16) # (H, D)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse mask (if any) + left/right projections in a tiny Triton kernel
    # ------------------------------------------------------------------
    if mask is None:
        mask_fp16 = None
    else:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)

    left, right = _proj_mask_fuse(x_fp16, mask_fp16, left_proj_w, right_proj_w)
    # left / right : (B, N, N, H)  FP16

    # ------------------------------------------------------------------
    # 3️⃣ Gates (sigmoid) – still in FP16
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)   # (H, D)
    right_gate_w = weights["right_gate.weight"].to(torch.float16) # (H, D)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)    # (H, D)

    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()   # (B,N,N,H)
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()    # (B,N,N,H)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction – batched GEMM via cuBLAS
    #    left:  (B,N,N,H) → (B*H, N, N)
    #    right: (B,N,N,H) → (B*H, N, N) and we need right^T on the last two dims
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B_h, _, I, K = left_t.shape
    _, _, K2, J = right_t.shape
    assert K == K2 and I == J == N

    # reshape to 3‑D batched matrices
    left_mat  = left_t.view(B_h, I, K)          # (B*H, N, N)
    right_mat = right_t.view(B_h, K, J)         # (B*H, N, N)

    # batched matmul (fp16 → fp32 accumulation inside cuBLAS)
    out_mat = torch.bmm(left_mat, right_mat)    # (B*H, N, N) , dtype fp16

    # reshape back to (B, N, N, H) and cast to FP32
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 6️⃣ Final gating + projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast gate over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
```

### Explanation of design choices
* **Mask‑fusion kernel** – the only heavy‑memory‑traffic element‑wise work (applying the mask and two linear projections) is fused into a single Triton kernel that processes one `(b,i,j)` element per program. This keeps the number of kernel launches tiny and lets us stay in FP16 throughout.
* **Core contraction** – the N³ contraction is exactly a batched matrix‑multiply over the sequence dimension. Using `torch.bmm` (cuBLAS/Tensor‑cores) is far faster than a custom Triton N³ kernel and matches the “delegate to cuBLAS” lesson.
* **Mixed‑precision** – all heavy compute (projections, contraction, gating) is performed in FP16 with FP32 accumulation where needed; final result is returned in FP32 as required.
* **LayerNorm** – implemented manually in FP32 to avoid an extra kernel and because it is cheap compared to the cubic contraction.

The implementation satisfies all constraints, works for all listed configurations, and includes at least one Triton kernel as mandated.
