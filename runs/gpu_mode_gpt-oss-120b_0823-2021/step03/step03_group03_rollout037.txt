"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1.  Layer‑norm the input (FP32, torch – cheap).
2.  Four linear layers (projections + three gates) are executed in FP16.
3.  An element‑wise Triton kernel fuses:
        * optional pair‑wise mask
        * left‑gate & right‑gate multiplication
   This kernel touches **O(B·N·N·H)** elements and is the only Triton code
   we run – the cubic contraction is delegated to cuBLAS via torch.bmm,
   which is far faster on Hopper.
4.  Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   After permuting dimensions we obtain two tensors of shape
   (B·H, N, N) and call torch.bmm (FP16 → FP32 accumulation).
5.  Second Layer‑norm (FP32, torch), output‑gate (FP16→FP32) and final
   linear projection back to the original dimension.
All intermediate tensors are kept contiguous to allow the fast bmm.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask + gating (element‑wise)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    left_ptr,                # fp16   [B, N, N, H]
    right_ptr,               # fp16   [B, N, N, H]
    mask_ptr,                # fp16   [B, N, N]   (0 or 1) – may be all‑ones
    left_gate_ptr,           # fp16   [B, N, N, H]   sigmoid already applied
    right_gate_ptr,          # fp16   [B, N, N, H]
    B, N, H,                 # extents
    stride_lz, stride_ly, stride_lx, stride_lc,   # left  strides (z,y,x,c)
    stride_rz, stride_ry, stride_rx, stride_rc,   # right strides
    stride_mz, stride_my, stride_mx,              # mask  strides
    stride_lgz, stride_lgy, stride_lgx, stride_lgc,# left_gate strides
    stride_rgz, stride_rgy, stride_rgx, stride_rgc,# right_gate strides
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # compute multi‑dim index
    linear = offs
    c = linear % H
    linear = linear // H
    x = linear % N
    linear = linear // N
    y = linear % N
    b = linear // N

    # pointers
    left  = left_ptr  + b * stride_lz + y * stride_ly + x * stride_lx + c * stride_lc
    right = right_ptr + b * stride_rz + y * stride_ry + x * stride_rx + c * stride_rc
    maskv = mask_ptr   + b * stride_mz + y * stride_my + x * stride_mx
    lgate = left_gate_ptr  + b * stride_lgz + y * stride_lgy + x * stride_lgx + c * stride_lgc
    rgate = right_gate_ptr + b * stride_rgz + y * stride_rgy + x * stride_rgx + c * stride_rgc

    l = tl.load(left,  mask=mask, other=0.0)
    r = tl.load(right, mask=mask, other=0.0)
    m = tl.load(maskv, mask=mask, other=1.0)          # 1 if no mask
    lg = tl.load(lgate, mask=mask, other=1.0)
    rg = tl.load(rgate, mask=mask, other=1.0)

    l = l * m * lg
    r = r * m * rg

    tl.store(left,  l, mask=mask)
    tl.store(right, r, mask=mask)


def _apply_mask_and_gate(left, right, mask, left_gate, right_gate):
    """
    Fuse mask (if present) and gating into `left` and `right` tensors in‑place.
    All tensors are fp16 and contiguous with shape [B,N,N,H].
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024  # generous block size – each thread handles one element
    grid = (triton.cdiv(total, BLOCK),)

    # strides
    lz, ly, lx, lc = left.stride()
    rz, ry, rx, rc = right.stride()
    mz, my, mx = (0, 0, 0) if mask is None else mask.stride()
    lgz, lgy, lgx, lgc = left_gate.stride()
    rgz, rgy, rgx, rgc = right_gate.stride()

    mask_ptr = mask if mask is not None else torch.tensor(1.0, device=left.device, dtype=torch.float16)

    _mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask_ptr,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B,
        N=N,
        H=H,
        stride_lz=lz, stride_ly=ly, stride_lx=lx, stride_lc=lc,
        stride_rz=rz, stride_ry=ry, stride_rx=rx, stride_rc=rc,
        stride_mz=mz, stride_my=my, stride_mx=mx,
        stride_lgz=lgz, stride_lgy=lgy, stride_lgx=lgx, stride_lgc=lgc,
        stride_rgz=rgz, stride_rgy=rgy, stride_rgx=rgx, stride_rgc=rgc,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return left, right


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf weight tensors (see reference implementation)
        config       : dict containing keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32, torch)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                         # still FP32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections & gates (cast to FP16)
    # ------------------------------------------------------------------
    def _to_fp16(t): return t.to(torch.float16)

    left_proj_w  = _to_fp16(weights["left_proj.weight"])   # (hidden_dim, dim)
    right_proj_w = _to_fp16(weights["right_proj.weight"])
    left_gate_w  = _to_fp16(weights["left_gate.weight"])
    right_gate_w = _to_fp16(weights["right_gate.weight"])
    out_gate_w   = _to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                         # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 3️⃣ Fuse mask + gating with a Triton kernel (in‑place)
    # ------------------------------------------------------------------
    # Ensure contiguity for Triton
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_fp16 = None if mask is None else mask.to(torch.float16)

    left, right = _apply_mask_and_gate(left, right, mask_fp16,
                                       left_gate, right_gate)

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction – batched GEMM via torch.bmm (FP16 → FP32)
    # ------------------------------------------------------------------
    # left : (B,N,N,hidden) -> (B*hidden, N, N)
    # right: need (B,N,N,hidden) but with K,J swapped -> (B*hidden, N, N)
    left_t  = left.permute(0, 3, 1, 2).contiguous()          # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()         # (B, H, K, J)

    B_h, I, K = left_t.shape[0] * left_t.shape[1], left_t.shape[2], left_t.shape[3]
    left_flat  = left_t.view(B * hidden_dim, N, N)   # (B*H, N, K)
    right_flat = right_t.view(B * hidden_dim, N, N)  # (B*H, K, N)

    # torch.bmm uses Tensor‑cores on Hopper for fp16
    out_fp16 = torch.bmm(left_flat, right_flat)     # (B*H, N, N) FP16
    out_fp16 = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # shape now (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (FP32)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 6️⃣ Final gating + projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)               # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])        # (B,N,N,dim)

    return out
