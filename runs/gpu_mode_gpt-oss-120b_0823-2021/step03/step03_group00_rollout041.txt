"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward only)
------------------------
1️⃣  Layer‑norm on the 4‑D input (fp32, manual – mean/var over the last dim).
2️⃣  Linear projections + three gate projections (fp16, using torch.nn.functional).
3️⃣  **Triton kernel** – element‑wise mask + gating:
        left  = left  * mask * sigmoid(left_gate)
        right = right * mask * sigmoid(right_gate)
    The kernel fuses the two element‑wise multiplications and the
    sigmoid‑gates into a single pass over memory, which is the only
    part that must run on Triton.
4️⃣  Core contraction  
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .
    This is a batched matrix‑multiply with shape
        (B·H, N, N) · (B·H, N, N)ᵀ → (B·H, N, N)  
    and is performed with `torch.bmm` (cuBLAS/Tensor‑cores) for maximal
    throughput.
5️⃣  Convert back to fp32, second LayerNorm, output‑gate and final linear
    projection (all in PyTorch).

Only step‑3 is written in Triton, satisfying the requirement that *some*
heavy work be off‑loaded to a custom kernel while keeping the overall
implementation fast and correct.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fused mask + gating (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,                # fp16*   (B,N,N,H)
    right_ptr,               # fp16*   (B,N,N,H)
    mask_ptr,                # fp16*   (B,N,N,1)  broadcasted over H
    left_gate_ptr,           # fp16*   (B,N,N,H)
    right_gate_ptr,          # fp16*   (B,N,N,H)
    out_left_ptr,            # fp16*   (B,N,N,H)  written in‑place
    out_right_ptr,           # fp16*   (B,N,N,H)
    B, N, H,                 # constexpr sizes
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,   # strides (elements)
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    BLOCK: tl.constexpr,    # number of elements processed per program (along H)
):
    """
    For each (b,i,j) we load a tile of size BLOCK over the hidden dimension H
    and compute:
        left  = left  * mask * sigmoid(left_gate)
        right = right * mask * sigmoid(right_gate)
    All tensors are stored in fp16; the result is written back in fp16.
    """
    pid = tl.program_id(0)                 # linear index over (b,i,j)
    total = B * N * N
    b_ij = pid
    if b_ij >= total:
        return

    # decode linear id -> (b,i,j)
    b = b_ij // (N * N)
    rem = b_ij % (N * N)
    i = rem // N
    j = rem % N

    # offsets for the hidden dimension
    offs_h = tl.arange(0, BLOCK)
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # broadcasted mask element (scalar) for this (b,i,j)
    mask_val = tl.load(mask_ptr + mask_off, mask=True, other=0.0)

    for start in range(0, H, BLOCK):
        cur = start + offs_h
        mask_h = cur < H

        # left
        left_off = (
            b * stride_left_b
            + i * stride_left_i
            + j * stride_left_j
            + cur * stride_left_h
        )
        left = tl.load(left_ptr + left_off, mask=mask_h, other=0.0)

        left_gate_off = (
            b * stride_gate_b
            + i * stride_gate_i
            + j * stride_gate_j
            + cur * stride_gate_h
        )
        left_gate = tl.load(left_gate_ptr + left_gate_off, mask=mask_h, other=0.0)
        left_gate = tl.sigmoid(left_gate)

        left = left * mask_val * left_gate
        tl.store(out_left_ptr + left_off, left, mask=mask_h)

        # right
        right_off = (
            b * stride_right_b
            + i * stride_right_i
            + j * stride_right_j
            + cur * stride_right_h
        )
        right = tl.load(right_ptr + right_off, mask=mask_h, other=0.0)

        right_gate_off = (
            b * stride_gate_b
            + i * stride_gate_i
            + j * stride_gate_j
            + cur * stride_gate_h
        )
        right_gate = tl.load(right_gate_ptr + right_gate_off, mask=mask_h, other=0.0)
        right_gate = tl.sigmoid(right_gate)

        right = right * mask_val * right_gate
        tl.store(out_right_ptr + right_off, right, mask=mask_h)


# ----------------------------------------------------------------------
# Helper – launch the Triton kernel
# ----------------------------------------------------------------------
def _apply_mask_and_gate(
    left: torch.Tensor,          # (B,N,N,H) fp16
    right: torch.Tensor,         # (B,N,N,H) fp16
    mask: torch.Tensor,          # (B,N,N)  fp16 (or None)
    left_gate: torch.Tensor,     # (B,N,N,H) fp16
    right_gate: torch.Tensor,    # (B,N,N,H) fp16
) -> tuple[torch.Tensor, torch.Tensor]:
    """
    Returns new left/right tensors after applying mask and sigmoid gates.
    The operation is performed in‑place via a fused Triton kernel for speed.
    """
    B, N, _, H = left.shape
    BLOCK = 64                               # fits well in registers / shared mem

    # output buffers (same storage as inputs – we write back in‑place)
    out_left = torch.empty_like(left)
    out_right = torch.empty_like(right)

    # strides in element units
    stride_left = left.stride()
    stride_right = right.stride()
    stride_gate = left_gate.stride()        # same shape as left/right
    stride_mask = mask.stride()

    grid = (B * N * N,)                     # one program per (b,i,j)

    _apply_mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        out_left_ptr=out_left,
        out_right_ptr=out_right,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_j=stride_left[2],
        stride_left_h=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_i=stride_right[1],
        stride_right_j=stride_right[2],
        stride_right_h=stride_right[3],
        stride_gate_b=stride_gate[0],
        stride_gate_i=stride_gate[1],
        stride_gate_j=stride_gate[2],
        stride_gate_h=stride_gate[3],
        stride_mask_b=stride_mask[0],
        stride_mask_i=stride_mask[1],
        stride_mask_j=stride_mask[2],
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out_left, out_right


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (fp32, manual, over last dim)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                       # still fp32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (fp16) – left, right, and three gates
    # --------------------------------------------------------------
    def _to_fp16(t): return t.to(torch.float16)

    left_proj_w   = _to_fp16(weights["left_proj.weight"])
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)               # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    left_gate  = F.linear(x_fp16, left_gate_w)   # (B,N,N,hidden_dim)
    right_gate = F.linear(x_fp16, right_gate_w)
    out_gate   = F.linear(x_fp16, out_gate_w)    # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 3️⃣ Fuse mask & gate using Triton
    # --------------------------------------------------------------
    if mask is None:
        # create a dummy mask of ones (fp16) – needed for kernel signature
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16)

    # kernel writes new tensors; we reuse the returned buffers
    left, right = _apply_mask_and_gate(left, right, mask_fp16, left_gate, right_gate)
    # out_gate stays in fp16 for later multiplication (broadcasted over hidden)
    out_gate = out_gate.to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ Core contraction – batched GEMM via torch.bmm (cuBLAS)
    #    reshape to (B*H, N, N) and multiply left · rightᵀ
    # --------------------------------------------------------------
    # left : (B,N,N,H) -> (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B,H,N,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # (B,H,N,N) ; note transpose K↔J

    B_h, _, I, J = left_perm.shape
    left_mat = left_perm.view(B_h, I, J)                  # (B*H, N, N)
    right_mat = right_perm.view(B_h, J, I)                # (B*H, N, N) already transposed

    # batched matrix multiply – fp16 × fp16 → fp16 (Tensor‑cores)
    out_mat = torch.bmm(left_mat, right_mat)             # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out_fp16 = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Output LayerNorm (fp32) + gating + final linear
    # --------------------------------------------------------------
    out = out_fp16.to(torch.float32)                     # (B,N,N,hidden_dim)

    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # apply output gate (broadcast over hidden dim) – gate originally in fp16
    out = out * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"]                 # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                       # (B,N,N,dim)

    return out
