"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  FP32 LayerNorm on the input.
2️⃣  Cast to FP16 and apply four linear layers (left/right projections + three gates).
3️⃣  If a pair‑wise mask is present we fuse the mask‑application and the
    gating (left*mask*sigmoid(left_gate), right*mask*sigmoid(right_gate))
    in a tiny Triton element‑wise kernel – this removes two separate PyTorch
    launches and lets the memory reads be coalesced.
4️⃣  Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    This is a batched matrix‑multiply over the “k” dimension (N).  The
    contraction is performed with a single call to ``torch.bmm`` on tensors of
    shape (B*H, N, N) – cuBLAS/Tensor‑cores on the H200 give the best throughput.
    No Triton work is needed for the cubic part.
5️⃣  Convert back to FP32, second LayerNorm, multiply by the output‑gate
    and a final linear projection to the original dimension.

Only the element‑wise mask + gate fusion is written in Triton; all heavy
linear algebra stays in highly‑optimised cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel – fuse mask, left/right gates
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,            # fp16 left  tensor   (B, N, N, H)
    right_ptr,           # fp16 right tensor   (B, N, N, H)
    mask_ptr,            # fp16 mask  tensor   (B, N, N, 1)  (broadcastable)
    left_gate_ptr,       # fp16 left‑gate tensor (B, N, N, H) after sigmoid
    right_gate_ptr,      # fp16 right‑gate tensor (B, N, N, H) after sigmoid
    out_left_ptr,        # fp16 output left  (same shape as left)
    out_right_ptr,       # fp16 output right (same shape as right)
    B, N, H,
    BLOCK: tl.constexpr,
):
    """In‑place fuse:   out = (tensor * mask) * gate   for left & right."""
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)          # 1‑D flatten index
    total = B * N * N * H

    mask = offset < total

    # decode the 1‑D index into (b, i, j, h)
    b = (offset // (N * N * H))
    rem = offset % (N * N * H)
    i = (rem // (N * H))
    rem = rem % (N * H)
    j = (rem // H)
    h = rem % H

    # compute flat pointers
    idx = ((b * N + i) * N + j) * H + h

    left   = tl.load(left_ptr + idx, mask=mask, other=0.0)
    right  = tl.load(right_ptr + idx, mask=mask, other=0.0)
    m      = tl.load(mask_ptr + idx // H, mask=mask, other=0.0)   # mask has no H dim
    lgate  = tl.load(left_gate_ptr + idx, mask=mask, other=0.0)
    rgate  = tl.load(right_gate_ptr + idx, mask=mask, other=0.0)

    left_out  = left * m * lgate
    right_out = right * m * rgate

    tl.store(out_left_ptr + idx,  left_out,  mask=mask)
    tl.store(out_right_ptr + idx, right_out, mask=mask)


def _apply_mask_and_gate_fp16(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
) -> tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse mask and gating in FP16 via Triton.
    All tensors are contiguous, shape (B, N, N, H) except mask (B, N, N, 1).
    """
    B, N, _, H = left.shape
    out_left  = torch.empty_like(left)
    out_right = torch.empty_like(right)

    # flatten dimensions for the kernel
    total = B * N * N * H
    BLOCK = 1024  # large enough to keep occupancy high

    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        out_left_ptr=out_left,
        out_right_ptr=out_right,
        B=B,
        N=N,
        H=H,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out_left, out_right


# ----------------------------------------------------------------------
# Entry point – forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf weight tensors (see reference implementation)
        config       : dict containing keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual – avoids extra CUDA kernel)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(dim=-1, keepdim=True)
    var = inp.var(dim=-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                               # still FP32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections & gates (FP16)
    # ------------------------------------------------------------------
    def _to_fp16(t): return t.to(torch.float16)

    # weight matrices are stored as (out_dim, in_dim) in PyTorch Linear
    left_proj_w   = _to_fp16(weights["left_proj.weight"])    # (hidden_dim, dim)
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                           # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)                 # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()   # (B,N,N,hidden_dim)
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()    # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask + gating fusion
    # ------------------------------------------------------------------
    if mask is not None:
        # mask -> (B,N,N,1) fp16
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left, right = _apply_mask_and_gate_fp16(left, right, mask_fp16,
                                                left_gate, right_gate)
    else:
        # no mask → just element‑wise gate multiplication (pure PyTorch)
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction (batched GEMM) – use cuBLAS via torch.bmm
    # ------------------------------------------------------------------
    # left : (B,N,N,hidden) -> (B*hidden, N, N)
    # right: (B,N,N,hidden) -> (B*hidden, N, N)  and we need rightᵀ on the
    #        second axis (the contraction sums over the third dimension k)
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)  # (B*H, I, K)
    right_ = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N) # (B*H, K, J)

    # batched matrix multiply – fp16 inputs, fp32 accumulation inside cuBLAS
    out_fp16 = torch.bmm(left_, right_)                                 # (B*H, I, J)

    # reshape back to (B,N,N,hidden)
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)          # now (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (FP32) + final gating + projection
    # ------------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # out_gate was computed in FP16; broadcast it in FP32
    out = out * out_gate.to(torch.float32)

    # final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out
