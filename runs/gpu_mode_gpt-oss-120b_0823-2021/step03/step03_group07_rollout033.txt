"""
TriMul “outgoing” forward pass (AlphaFold‑3).

The heavy part of the module is the three‑dimensional contraction

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

where `left`/`right` are the linearly‑projected, gated tensors of shape
[B, H, N, N] (H = hidden_dim).  This implementation

* performs all cheap per‑element work (LayerNorm, linear projections,
  gating, final LayerNorm & linear) with regular PyTorch – these operations are
  memory bound and negligible compared with the O(N³) contraction.
* launches a single Triton kernel that evaluates the contraction using a
  classic blocked GEMM tiling (block‑M × block‑N × block‑K).  The kernel works
  on FP16 data with FP32 accumulation and stores the result back as FP16.
* finally reshapes the result back to the original layout and finishes the
  forward pass in PyTorch.

The kernel processes one (batch, head) pair per program‑id and computes a
tile of the output matrix `out[b, h]`.  All indexing is done with the strides
of the tensors, so the code works for any contiguous layout produced by the
host side (the host always calls `.contiguous()` before launching).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # const fp16*  [B, H, N, N]
    right_ptr,         # const fp16*  [B, H, N, N]   (read transposed)
    out_ptr,           #       fp16*  [B, H, N, N]
    B, N, H,           # int metadata
    # stride layout (row‑major)
    stride_lb, stride_lh, stride_ln, stride_lk,   # left
    stride_rb, stride_rh, stride_rn, stride_rk,   # right
    stride_ob, stride_oh, stride_on, stride_ok,   # out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a (BLOCK_M × BLOCK_N) tile of the 4‑D result

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    One program‑id corresponds to a unique (b, h, tile_m, tile_n) tuple.
    """
    pid = tl.program_id(0)

    # ---------------------------------------------------------------
    # decode pid  -> (b, h, tile_m, tile_n)
    # ---------------------------------------------------------------
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    tile_n = pid % num_n
    pid //= num_n
    tile_m = pid % num_m
    pid //= num_m
    h = pid % H
    b = pid // H

    # ---------------------------------------------------------------
    # tile coordinates
    # ---------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # main K‑loop (blocked)
    # ---------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, h, i, k]  (M×K)
        l_ptr = (left_ptr +
                 b * stride_lb +
                 h * stride_lh +
                 offs_m[:, None] * stride_ln +
                 offs_k[None, :] * stride_lk)
        l = tl.load(l_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[b, h, j, k]  needs to be read as transposed: (K×N)
        #   original layout: (b, h, j, k)
        #   after transpose we want (k, j)
        r_ptr = (right_ptr +
                 b * stride_rb +
                 h * stride_rh +
                 offs_k[:, None] * stride_rk +          # k‑dim becomes row
                 offs_n[None, :] * stride_rn)           # j‑dim becomes col
        r_t = tl.load(r_ptr,
                      mask=mask_k[:, None] & mask_n[None, :],
                      other=0.0).to(tl.float32)

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(l, r_t)

    # ---------------------------------------------------------------
    # write result back
    # ---------------------------------------------------------------
    out_ptrs = (out_ptr +
                b * stride_ob +
                h * stride_oh +
                offs_m[:, None] * stride_on +
                offs_n[None, :] * stride_ok)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side wrapper that follows the required `custom_kernel` signature
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input   [B, N, N, dim]
        torch.Tensor,                     # mask    [B, N, N]   (or None)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any]                    # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device

    # ------------------------------------------------------------------
    # 1.  Cheap per‑element work in PyTorch (LayerNorm, linear, gating)
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # = H

    # LayerNorm over the last dimension of x  (dtype float32 for stability)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(x.dtype),
        bias=weights["norm.bias"].to(x.dtype),
    )

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"].to(x.dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(x.dtype))

    # optional mask (broadcast over hidden dim)
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # gates (sigmoid after linear)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(x.dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(x.dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(x.dtype)))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 2.  Rearrange to [B, H, N, N] and cast to fp16 for the kernel
    # ------------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # Cast to fp16 – this is the datatype used inside the kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # output buffer
    out_fp16 = torch.empty_like(left_fp16)      # (B,H,N,N), fp16

    # ------------------------------------------------------------------
    # 3.  Launch Triton kernel (blocked GEMM)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # grid: one program for each (b, h, tile_m, tile_n)
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_programs = B * hidden_dim * grid_m * grid_n
    grid = (total_programs,)

    _trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        # strides for left
        left_fp16.stride(0),   # B stride
        left_fp16.stride(1),   # H stride
        left_fp16.stride(2),   # N (row) stride
        left_fp16.stride(3),   # N (col) stride
        # strides for right
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        # strides for out
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 4.  Post‑process back to original shape and finish the forward pass
    # ------------------------------------------------------------------
    # (B,H,N,N) -> (B,N,N,H) and cast to float32 for the rest of the pipeline
    out = out_fp16.to(x.dtype).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # final LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(x.dtype),
        bias=weights["to_out_norm.bias"].to(x.dtype),
    )

    # apply the output gate (broadcast over pairwise axes)
    out = out * out_gate

    # final linear back to dim
    out = F.linear(out, weights["to_out.weight"].to(x.dtype))

    return out
