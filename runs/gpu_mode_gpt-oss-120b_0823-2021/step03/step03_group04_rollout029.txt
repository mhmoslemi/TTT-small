"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm (GPU‑friendly)
------------------------
1️⃣  Layer‑norm the input tensor (B,N,N,C)   →  FP32.
2️⃣  Linear projections for *left*, *right* and the three gates.
3️⃣  Fuse element‑wise mask (if present) with the left/right
    projections and their gates inside a tiny Triton kernel.
    The kernel works on a 1‑D grid and operates on FP16 data
    (Tensor‑core ready).  This satisfies the “some‑kernel” constraint.
4️⃣  Reshape the fused tensors to (B*H, N, N) and perform a
    batched matrix‑multiply (left @ rightᵀ) with torch.bmm /
    torch.matmul – this call is executed by cuBLAS/Tensor‑cores and
    dominates the O(N³) work.
5️⃣  Layer‑norm the hidden result, multiply by *out_gate* and
    apply the final linear projection back to the original channel
    dimension C.  The final tensor is returned in FP32.

Only the element‑wise mask‑and‑gate fusion is performed in Triton;
everything else stays in PyTorch (highly‑optimized cuBLAS kernels).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – mask & gate fusion
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,               # fp16 * (B,N,N,H)
    left_gate_ptr, right_gate_ptr,     # fp16 * (B,N,N,H)
    mask_ptr,                          # fp16 * (B,N,N,1)   – may be dummy
    out_left_ptr, out_right_ptr,       # fp16 * (B,N,N,H)   – output buffers
    B, N, H,                           # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_m0, stride_m1, stride_m2,   # mask has no hidden dim
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    Compute, for every element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # ----- decode linear index (b,i,j,h) -----
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ----- load left / right -----
    idx_lr = (
        b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    )
    left  = tl.load(left_ptr  + idx_lr, mask=mask, other=0.0)
    right = tl.load(right_ptr + idx_lr, mask=mask, other=0.0)

    # ----- load gates -----
    idx_gate = (
        b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + h * stride_lg3
    )
    lgate = tl.load(left_gate_ptr  + idx_gate, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + idx_gate, mask=mask, other=0.0)

    # ----- mask (broadcast over H) -----
    if HAS_MASK:
        idx_m = b * stride_m0 + i * stride_m1 + j * stride_m2
        m = tl.load(mask_ptr + idx_m, mask=mask, other=0.0)
    else:
        m = 1.0

    out_l = left  * lgate * m
    out_r = right * rgate * m

    # ----- store results -----
    tl.store(out_left_ptr  + idx_lr, out_l, mask=mask)
    tl.store(out_right_ptr + idx_lr, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        - x : torch.Tensor of shape (B, N, N, C), dtype=torch.float32
        - mask : torch.Tensor of shape (B, N, N) or ``None``.
        - weights : dict of all learned parameters (float32).
        - config : {"dim": C, "hidden_dim": H}
    Returns
    -------
    torch.Tensor of shape (B, N, N, C), dtype=torch.float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype  = torch.float32

    B, N, _, C = x.shape
    H = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (keep FP32 for now)
    # --------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate= torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate  = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N,H)

    # --------------------------------------------------------------
    # 3️⃣ Cast to FP16 and fuse mask + gates with Triton
    # --------------------------------------------------------------
    left_fp16      = left .to(torch.float16).contiguous()
    right_fp16     = right.to(torch.float16).contiguous()
    lgate_fp16     = left_gate .to(torch.float16).contiguous()
    rgate_fp16     = right_gate.to(torch.float16).contiguous()

    if mask is None:
        # dummy mask – will never be accessed when HAS_MASK=False
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask  = False
    else:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()   # (B,N,N,1)
        has_mask = True

    # allocate fused outputs
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elem = B * N * N * H
    BLOCK = 1024                                 # good occupancy on H200
    grid = (triton.cdiv(total_elem, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,            # left_ptr
        right_fp16,           # right_ptr
        lgate_fp16,           # left_gate_ptr
        rgate_fp16,           # right_gate_ptr
        mask_fp16,            # mask_ptr (dummy if no mask)
        left_fused,           # out_left_ptr
        right_fused,          # out_right_ptr
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0),right_fp16.stride(1),right_fp16.stride(2),right_fp16.stride(3),
        lgate_fp16.stride(0),lgate_fp16.stride(1),lgate_fp16.stride(2),lgate_fp16.stride(3),
        mask_fp16.stride(0) if has_mask else 0,
        mask_fp16.stride(1) if has_mask else 0,
        mask_fp16.stride(2) if has_mask else 0,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (FP16→FP16, acc FP32) – the O(N³) work
    # --------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)

    # left @ rightᵀ   – cuBLAS/Tensor‑core path
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))          # (B*H,N,N)

    # back to (B,N,N,H) and to FP32 for the following LayerNorm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)
    out = out.to(dtype)

    # --------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm → out_gate gating → final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                     # broadcast over (N,N)
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
