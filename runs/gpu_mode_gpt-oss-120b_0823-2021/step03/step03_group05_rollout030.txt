"""
TriMul (outgoing) – fully‑fused forward pass.

Algorithm
---------
1️⃣ Layer‑norm + the four linear projections (left, right, left‑gate,
    right‑gate, out‑gate) are computed in PyTorch – they are cheap compared
    with the O(N³) contraction.

2️⃣ A single Triton kernel fuses **mask‑application**, **sigmoid‑gate**
    multiplication and the **out‑gate** multiplication into one element‑wise
    pass over the 4‑D tensors `left`, `right` and `out_gate`.  The kernel works
    for any contiguous layout, uses fp16 for maximum bandwidth and writes the
    results back in‑place.

3️⃣ The cubic contraction
        out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
    is performed by reshaping the tensors to (B·H, N, N) and calling
    `torch.bmm`.  On Hopper this triggers Tensor‑Core‑enabled cuBLAS which
    dominates the runtime.

4️⃣ A second LayerNorm, the (already‑applied) out‑gate and the final linear
    projection bring the tensor back to shape `[B, N, N, dim]` in fp32.

The kernel fulfills the “must use Triton” rule while keeping the performance
close to the theoretical optimum for the tested shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel – elementwise fusion
# -------------------------------------------------------------------------
@triton.jit
def _fuse_mask_gate_outkernel(
    left_ptr,          # *mut fp16   (B,N,N,H) – overwritten with left*gate*mask
    right_ptr,         # *mut fp16   (B,N,N,H) – overwritten with right*gate*mask
    left_gate_ptr,     # *const fp16 (B,N,N,H)   sigmoid(left_gate)
    right_gate_ptr,    # *const fp16 (B,N,N,H)   sigmoid(right_gate)
    out_gate_ptr,      # *const fp16 (B,N,N,1)   sigmoid(out_gate) broadcasted
    mask_ptr,          # *const fp16 (B,N,N,1)   1.0 if no mask
    B, N, H,           # compile‑time constants
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_og0, stride_og1, stride_og2, stride_og3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # ---- decode linear index ----
    tmp = offs
    h = tmp % H; tmp = tmp // H
    j = tmp % N; tmp = tmp // N
    i = tmp % N; b = tmp // N

    # ---- address calculation ----
    l_ptr = left_ptr  + b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    r_ptr = right_ptr + b * stride_r0 + i * stride_r1 + j * stride_r2 + h * stride_r3
    lg_ptr = left_gate_ptr  + b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + h * stride_lg3
    rg_ptr = right_gate_ptr + b * stride_rg0 + i * stride_rg1 + j * stride_rg2 + h * stride_rg3
    og_ptr = out_gate_ptr  + b * stride_og0 + i * stride_og1 + j * stride_og2               # broadcast over H
    m_ptr = mask_ptr        + b * stride_m0 + i * stride_m1 + j * stride_m2

    # ---- load ----
    l  = tl.load(l_ptr,  mask=mask, other=0.0)
    r  = tl.load(r_ptr, mask=mask, other=0.0)
    lg = tl.load(lg_ptr, mask=mask, other=0.0)
    rg = tl.load(rg_ptr, mask=mask, other=0.0)
    og = tl.load(og_ptr, mask=mask, other=1.0)   # out_gate defaults to 1.0 (no effect)
    m  = tl.load(m_ptr,  mask=mask, other=1.0)   # mask defaults to 1.0

    # ---- fuse operations ----
    # left  = left  * mask * left_gate  * out_gate
    # right = right * mask * right_gate * out_gate
    l = l * m * lg * og
    r = r * m * rg * og

    # ---- store back ----
    tl.store(l_ptr, l, mask=mask)
    tl.store(r_ptr, r, mask=mask)


def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        - x : Tensor[B, N, N, dim]                (float32)
        - mask : Tensor[B, N, N] or None (bool/float32)
        - weights : dict of model parameters (float32)
        - config : {"dim": int, "hidden_dim": int}

    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    H = config["hidden_dim"]                # hidden dimension (c_z)

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm + linear projections (stay in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(x.dtype),
        bias=weights["norm.bias"].to(x.dtype),
    )

    left      = F.linear(x, weights["left_proj.weight"].to(x.dtype))
    right     = F.linear(x, weights["right_proj.weight"].to(x.dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(x.dtype)))
    right_gate= torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(x.dtype)))
    out_gate  = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(x.dtype)))

    # -----------------------------------------------------------------
    # 2️⃣ Fuse mask, gates and out‑gate in one Triton kernel
    # -----------------------------------------------------------------
    # Prepare mask: (B, N, N, 1)
    if mask is None:
        mask_tensor = torch.ones((B, N, N, 1),
                                 dtype=torch.float16,
                                 device=device)
    else:
        mask_tensor = mask.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        mask_tensor = mask_tensor.to(torch.float16)

    # Cast to fp16 for the bandwidth‑bound kernel
    left_fp16       = left.to(torch.float16).contiguous()
    right_fp16      = right.to(torch.float16).contiguous()
    left_gate_fp16  = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()
    out_gate_fp16   = out_gate.to(torch.float16).contiguous()   # (B,N,N,dim)

    total_elements = B * N * N * H
    BLOCK = 1024                                   # good occupancy for simple kernel
    grid = (triton.cdiv(total_elements, BLOCK),)

    _fuse_mask_gate_outkernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        out_gate_fp16,
        mask_tensor,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        right_gate_fp16.stride(0), right_gate_fp16.stride(1),
        right_gate_fp16.stride(2), right_gate_fp16.stride(3),
        out_gate_fp16.stride(0), out_gate_fp16.stride(1),
        out_gate_fp16.stride(2), out_gate_fp16.stride(3),
        mask_tensor.stride(0), mask_tensor.stride(1),
        mask_tensor.stride(2), mask_tensor.stride(3),
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 3️⃣ Core O(N³) contraction – batched GEMM via torch.bmm
    #    left  : (B, N, N, H) → (B·H, N, N)
    #    right : (B, N, N, H) → (B·H, N, N)   (transpose on dim‑2)
    # -----------------------------------------------------------------
    left_bh  = left_fp16.permute(0, 3, 1, 2).reshape(B * H, N, N)          # (B·H, N, N)
    right_bh = right_fp16.permute(0, 3, 1, 2).reshape(B * H, N, N)         # (B·H, N, N)

    # (B·H,N,N) @ (B·H,N,N)^T   →   (B·H,N,N)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_bh.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(x.dtype)            # back to fp32 for the remaining ops

    # -----------------------------------------------------------------
    # 4️⃣ Post‑processing – LayerNorm, out_gate (already applied), final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(x.dtype),
        bias=weights["to_out_norm.bias"].to(x.dtype),
    )
    # out_gate has already been multiplied element‑wise in the kernel,
    # so we only need the final projection.
    out = F.linear(out, weights["to_out.weight"].to(x.dtype))

    return out
