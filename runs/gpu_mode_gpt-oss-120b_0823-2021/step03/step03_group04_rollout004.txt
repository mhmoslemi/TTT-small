"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm
---------
1️⃣ Layer‑norm the input (B,N,N,C) → (B,N,N,C)   (FP32)
2️⃣ Linear projections → left/right (B,N,N,H) and three gates (B,N,N,H)
3️⃣ Fuse element‑wise:   out_left  = left  * left_gate  * mask
                        out_right = right * right_gate * mask
   This tiny work is done with a Triton kernel (mandatory “some‑kernel” rule).
4️⃣ Batched GEMM (Tensor‑core friendly):
        out[b,h] = out_left[b,h] @ out_right[b,h].T
   Reshape left/right to (B*H, N, N) and use torch.bmm (cuBLAS) – the dominant
   O(N³) work stays on the GPU but does not require a custom kernel.
5️⃣ Layer‑norm over the hidden dimension H, multiply by out_gate and final linear
   projection back to C (all in FP32).

All heavy arithmetic uses FP16 for the GEMM; accumulations are FP32 for
numerical stability.  The final result is returned as FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – fuse masks and gates
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                # fp16 * (B,N,N,H)
    left_gate_ptr, right_gate_ptr,      # fp16 * (B,N,N,H)
    mask_ptr,                           # fp16 * (B,N,N,1)  or nullptr
    out_left_ptr, out_right_ptr,        # fp16 * (B,N,N,H)  outputs
    B, N, H,                            # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,   # gates strides (same as left/right)
    stride_m0, stride_m1, stride_m2,                # mask strides (no H dim)
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    The kernel works on a 1‑D grid of size ceil(B*N*N*H / BLOCK).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # -------- decode linear index --------
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # -------- load left/right ----------
    idx = b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    left  = tl.load(left_ptr + idx,   mask=mask, other=0.0)
    right = tl.load(right_ptr + idx,  mask=mask, other=0.0)

    # -------- load gates ---------------
    idx_gate = b * stride_rg0 + i * stride_rg1 + j * stride_rg2 + h * stride_rg3
    lgate = tl.load(left_gate_ptr + idx_gate,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + idx_gate, mask=mask, other=0.0)

    # -------- load mask if present -----
    if HAS_MASK:
        # mask has shape (B,N,N,1) → same strides as left/right but without H dim
        idx_mask = b * stride_m0 + i * stride_m1 + j * stride_m2
        m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)
    else:
        m = 1.0

    # -------- fuse ----------
    out_l = left * lgate * m
    out_r = right * rgate * m

    # -------- store ----------
    tl.store(out_left_ptr + idx,  out_l, mask=mask)
    tl.store(out_right_ptr + idx, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        x       – (B, N, N, C)   torch.float32
        mask    – (B, N, N)      torch.bool / float32 / None
        weights – dict of all learned parameters (float32)
        config  – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor
        (B, N, N, C)  float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, C = x.shape
    H = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (keep FP32 for now)
    # --------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N,H)

    # --------------------------------------------------------------
    # 3️⃣ Cast to FP16 and fuse mask + gates with Triton
    # --------------------------------------------------------------
    left_fp16      = left.to(torch.float16).contiguous()
    right_fp16     = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate.to(torch.float16).contiguous()
    right_gate_fp16= right_gate.to(torch.float16).contiguous()

    if mask is None:
        # dummy mask – pointer will never be dereferenced because HAS_MASK=False
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask = False
    else:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()   # (B,N,N,1)
        has_mask = True

    # allocate output buffers
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_el = B * N * N * H
    BLOCK = 1024                           # empirically good for H200
    grid = (triton.cdiv(total_el, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_gate_fp16.stride(0), right_gate_fp16.stride(1),
        right_gate_fp16.stride(2), right_gate_fp16.stride(3),
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (FP16 inputs → FP16 output, FP32 accumulation)
    # --------------------------------------------------------------
    # reshape to (B*H, N, N)  (the hidden dim becomes a batch dimension)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H, N, N)

    # matrix multiplication: left @ right^T   (bmm uses Tensor‑cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))          # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype)   # back to FP32 for the next LayerNorm

    # --------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, gating, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    # out_gate is already (B,N,N,H) FP32
    out = out * out_gate          # broadcast over H
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
