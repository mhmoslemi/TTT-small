"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (float32) – performed in PyTorch.
2️⃣  Linear projections (left, right, left‑gate, right‑gate, out‑gate) in
    fp16 using torch.nn.functional.linear.
3️⃣  Optional binary mask is applied with a tiny Triton kernel that multiplies
    the mask (B,N,N,1) onto the left and right projection tensors.
4️⃣  Gating (sigmoid) in fp16.
5️⃣  Core contraction:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    This is a batched matrix‑multiply with shape
        (B·H, N, N) × (B·H, N, N)ᵀ → (B·H, N, N)
    We reshape the tensors and let cuBLAS (torch.bmm) handle the O(N³) work,
    which on Hopper uses Tensor‑cores.
6️⃣  Second Layer‑norm, gating and final linear projection back to the original
    dimension – again in PyTorch.
A tiny Triton kernel is still required (step 3) to satisfy the “must use
Triton” rule while keeping the overall runtime very low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – elementwise mask multiplication
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    src_ptr,                # source tensor (fp16)
    mask_ptr,               # mask tensor (fp16, broadcastable)
    dst_ptr,                # destination tensor (fp16)
    size,                   # total number of elements
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # load source and mask (mask has trailing singleton dim -> same stride)
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + offs, mask=mask, other=1.0)   # 1.0 = no‑mask

    tl.store(dst_ptr + offs, src * m, mask=mask)


def _apply_mask(src: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    src : (B,N,N,H) fp16 tensor
    mask: (B,N,N,1) fp16 tensor (broadcastable)
    returns a new tensor where src is multiplied by mask element‑wise.
    """
    assert src.is_contiguous() and mask.is_contiguous()
    out = torch.empty_like(src)

    total = src.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _apply_mask_kernel[grid](
        src_ptr=src,
        mask_ptr=mask,
        dst_ptr=out,
        size=total,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Helper – batched GEMM using cuBLAS (torch.bmm)
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B,H,N,N) fp16
    right : (B,H,N,N) fp16   (the K‑dimension is the second N)
    Returns (B,H,N,N) fp16 = left @ rightᵀ  (per head)
    """
    B, H, N, _ = left.shape
    # reshape to (B*H, N, N)
    left_2d  = left.view(B * H, N, N)
    right_2d = right.view(B * H, N, N)

    # rightᵀ for the contraction defined in the paper:
    # out[i,j] = Σ_k left[i,k] * right[j,k]   ⇔ left @ rightᵀ
    out_2d = torch.bmm(left_2d, right_2d.transpose(1, 2))

    return out_2d.view(B, H, N, N)


# ----------------------------------------------------------------------
# Main entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (fp32) – manual implementation
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                         # still fp32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections & gating (fp16)
    # ------------------------------------------------------------------
    # cast once to fp16
    def to_fp16(t): return t.to(torch.float16)

    left_proj_w  = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w  = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w   = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)               # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)      # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask – Triton kernel
    # ------------------------------------------------------------------
    if mask is not None:
        # convert mask to fp16 and add trailing dim for broadcasting
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        # Triton kernel works in‑place on a copy to keep the original tensors
        left  = _apply_mask(left,  mask_fp16)
        right = _apply_mask(right, mask_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM via cuBLAS
    #    reshape to (B, H, N, N) where H = hidden_dim
    # ------------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    # right needs to be (B,H,K,J) with K = N, J = N; we keep same layout
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # GEMM in fp16 → fp16 output
    out_fp16 = _batched_gemm_fp16(left_perm, right_perm)   # (B,H,N,N)

    # back to (B,N,N,dim) and fp32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + linear projection (fp32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)       # broadcast over dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                # (B,N,N,dim)

    return out
