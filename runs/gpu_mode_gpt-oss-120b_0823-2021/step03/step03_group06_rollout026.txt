"""
TriMul “outgoing” operator (AlphaFold‑3) – fast forward pass.

The dominant O(N³) contraction

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

is performed with a custom Triton batched‑GEMM kernel that
  * works in FP16 with FP32 accumulation (Tensor‑Core friendly)
  * tiles the output (M×N) and iterates over K in BLOCK_K chunks
  * processes all (batch * hidden_dim) “heads’’ in a single launch.

All cheap preprocessing (LayerNorm, linear projections, masking,
gating, final LayerNorm and output linear) remain in PyTorch.
The final tensor is returned in FP32 as required by the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM for one (batch * hidden_dim) slice
# -------------------------------------------------------------------------
@triton.jit
def _trimul_batched_gemm_kernel(
    left_ptr,          # const fp16*  [BH, N, N]
    right_ptr,         # const fp16*  [BH, N, N]   (to be read transposed)
    out_ptr,           #       fp16*  [BH, N, N]
    BH, N,             # problem sizes
    stride_lbh, stride_lbn, stride_lbk,   # left strides
    stride_rbh, stride_rbn, stride_rbk,   # right strides
    stride_obh, stride_obn, stride_obk,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    out[bh, i, j] = Σ_k left[bh, i, k] * right[bh, j, k]
    (right is accessed as right^T so the inner dimension aligns).
    """
    pid = tl.program_id(0)                     # linear id over (bh, tile_m, tile_n)

    # -------------------------------------------------------------
    # Decode a 3‑D grid (bh, tile_m, tile_n) from the 1‑D pid
    # -------------------------------------------------------------
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    tile_n = pid % num_tiles_n
    pid //= num_tiles_n
    tile_m = pid % num_tiles_m
    pid //= num_tiles_m
    bh = pid                                   # 0 ≤ bh < BH

    # -------------------------------------------------------------
    # Offsets for this tile
    # -------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator (FP32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over K dimension in BLOCK_K steps
    # -------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[bh, i, k]   shape (M, K)
        l_ptr = left_ptr + bh * stride_lbh \
                + offs_m[:, None] * stride_lbn \
                + offs_k[None, :] * stride_lbk
        l = tl.load(l_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[bh, j, k] – we need right^T so we load (K, N)
        r_ptr = right_ptr + bh * stride_rbh \
                + offs_k[:, None] * stride_rbk \
                + offs_n[None, :] * stride_rbn
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # GEMM: (M,K) @ (K,N) → (M,N)
        acc += tl.dot(l, r)

    # -------------------------------------------------------------
    # Write back in FP16
    # -------------------------------------------------------------
    out_ptrs = out_ptr + bh * stride_obh \
               + offs_m[:, None] * stride_obn \
               + offs_n[None, :] * stride_obk
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input   : [B, N, N, dim]
        torch.Tensor,                # mask    : [B, N, N]  (or None)
        Dict[str, torch.Tensor],     # weights : dict of parameters
        Dict[str, Any],              # config  : {"dim":..., "hidden_dim":...}
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # = H in the paper
    dtype = torch.float32                     # work in FP32 for the cheap ops

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm + linear projections (cheap, stay in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N, H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 2️⃣ Optional mask (broadcastable)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 3️⃣ Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(dtype))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(dtype))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(dtype))
    )

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ Rearrange for the batched GEMM: (B,N,N,H) → (B, H, N, N) → (BH,N,N)
    # -----------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left.view(B * hidden_dim, N, N).to(torch.float16)
    right_fp16 = right.view(B * hidden_dim, N, N).to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)        # (BH, N, N), FP16

    # -----------------------------------------------------------------
    # 5️⃣ Launch the Triton kernel
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    BH = B * hidden_dim
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_tiles = BH * grid_m * grid_n
    grid = (total_tiles,)

    _trimul_batched_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        BH,
        N,
        # strides for left   (BH, N, N)
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        # strides for right  (BH, N, N)
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        # strides for out    (BH, N, N)
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Back‑to original layout, remaining LayerNorm & output linear
    # -----------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    out = out.to(dtype)                                 # FP32 for the next ops

    # to_out_norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply output gate (broadcast over N,N)
    out = out * out_gate

    # final linear back to dim
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
