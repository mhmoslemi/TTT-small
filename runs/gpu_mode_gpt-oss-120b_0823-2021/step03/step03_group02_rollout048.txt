"""
TriMul “outgoing” forward pass – fast Triton‑accelerated contraction.

The heavy N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is carried out as a batched GEMM with a custom Triton kernel.
All lightweight operations (LayerNorm, linear projections, gates) stay in
PyTorch, using fp16 for the contraction and fp32 for the remaining
normalisation / projection steps. The implementation works for any batch,
sequence length and hidden dimension allowed by the specification and
handles optional masking.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (I×K)·(K×J) → (I×J)  (fp16 inputs, fp32 acc, fp16 out)
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    a_ptr,            # left  tensor   (Z, I, K)   fp16
    b_ptr,            # right tensor   (Z, K, J)   fp16
    c_ptr,            # output tensor  (Z, I, J)   fp16
    M, N, K,          # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM tiled over M (rows), N (cols) and K (reduction dim).
    The kernel processes one tile of shape (BLOCK_M, BLOCK_N) for each
    program_id(0) × program_id(1) pair and iterates over the K dimension.
    Accumulation is performed in fp32 for numerical stability,
    the final result is cast back to fp16.
    """
    pid_m = tl.program_id(0)          # tile index in M dimension
    pid_n = tl.program_id(1)          # tile index in N dimension
    pid_z = tl.program_id(2)          # flattened batch dimension (B * hidden)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for tile_k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = tile_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (BLOCK_M x BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (BLOCK_K x BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 accumulation
        acc += tl.dot(a, b)

    # write result (cast back to fp16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[:, None],
    )


def triton_batch_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches `batch_gemm_kernel` on tensors of shape
        left : (B, H, I, K)   fp16, contiguous
        right: (B, H, K, J)   fp16, contiguous
    Returns a tensor of shape (B, H, I, J) fp16.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten the (B, H) dimensions for the kernel
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides in elements
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tiling – empirically good for Hopper/H200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # program_id(0)
        triton.cdiv(J, BLOCK_N),   # program_id(1)
        Z,                         # program_id(2)
    )
    batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch glue + Triton contraction)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor [B, N, N]       (float32/bool) or None
        - weights      : dict of torch.Tensor parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]   (float32)
    """
    # --------------------------- unpack ---------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------- 1) Input LayerNorm ---------------------------
    # Manual LayerNorm to avoid Module overhead
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b            # still fp32

    # --------------------------- 2) Projections & gating (fp16) ---------------------------
    # Helper to cast weight matrices once
    def to_fp16(t): return t.to(torch.float16, non_blocking=True)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = weights["to_out_norm.weight"]   # fp32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # Cast normalized input once
    x_fp16 = x.to(torch.float16)

    # Linear projections (B,N,N,hidden)
    left  = F.linear(x_fp16, left_proj_w)   # fp16
    right = F.linear(x_fp16, right_proj_w)  # fp16

    # --------------------------- 3) Optional mask ---------------------------
    if mask is not None:
        # mask shape (B,N,N) → (B,N,N,1) and cast to fp16
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------- 4) Gating (sigmoid) ---------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------- 5) Core contraction ---------------------------
    # Rearrange tensors to (B, hidden, I, K) and (B, hidden, K, J)
    # where I = J = N, K = N (the reduction dimension)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # Batched GEMM via Triton
    out_fp16 = triton_batch_gemm(left_perm, right_perm)   # (B, H, I, J)

    # Back to original layout (B, N, N, hidden) and cast to fp32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------- 6) Output LayerNorm ---------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # --------------------------- 7) Final gating + linear ---------------------------
    out = out * out_gate.to(torch.float32)   # apply the same out_gate as in reference
    out = F.linear(out, to_out_w)           # (B,N,N,dim) fp32

    return out
```

**Explanation of design choices**

1. **LayerNorm & linear layers** – kept in PyTorch because they are inexpensive compared to the cubic contraction and avoid the overhead of writing custom reduction kernels. Manual LayerNorm uses FP32 arithmetic for stability.

2. **Mixed‑precision** – the normalized tensor is cast once to `float16`; all heavy matrix‑multiplication work (the `einsum`‑equivalent) is performed in FP16 with an FP32 accumulator inside the Triton kernel, which matches cuBLAS Tensor‑core behaviour and gives a good speed‑accuracy trade‑off. The result is cast back to FP32 before the final LayerNorm and linear projection.

3. **Core contraction** – the `einsum('... i k d, ... j k d -> ... i j d')` is exactly a batched GEMM over the reduction dimension `k`. By permuting to `(B, hidden, I, K)` and `(B, hidden, K, J)` we obtain a standard batched matrix multiply that can be tiled efficiently in Triton. The kernel processes one `(BLOCK_M, BLOCK_N)` output tile per program‑id pair and iterates over `K` in `BLOCK_K`‑sized steps, using `tl.dot` for the inner product and an FP32 accumulator.

4. **Tiling parameters** – `BLOCK_M = BLOCK_N = 64` and `BLOCK_K = 32` give good occupancy on Hopper/H200 (the same values that performed well in the previous attempt). `num_warps=4` matches the compute capability of the target GPU.

5. **Mask handling** – the mask, when present, is simply broadcasted on the hidden dimension and multiplied in FP16 before gating. This keeps the kernel free of any conditional logic.

6. **Correctness** – the implementation follows the reference PyTorch code step‑by‑step, only replacing the cubic `einsum` with the Triton‑based batched GEMM. The final tensor shape and dtype (`float32`) match the specification.

The function `custom_kernel` can be called directly with the tuple `(input, mask, weights, config)` and will return the processed tensor while satisfying all constraints of the task.
