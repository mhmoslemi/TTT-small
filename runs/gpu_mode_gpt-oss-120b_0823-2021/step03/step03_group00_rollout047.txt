"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward‑only)
------------------------
1️⃣ Manual LayerNorm over the last dimension (fp32).
2️⃣ Linear projections + gate‑linear layers executed in fp16.
3️⃣ Optional binary/float mask applied with a tiny Triton kernel
   (the only required Triton component – keeps the launch overhead tiny).
4️⃣ Gating (sigmoid) and element‑wise multiplication (still fp16).
5️⃣ Core O(N³) contraction:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   is turned into a batch of matrix‑multiplications:
        (B·H, N, N) @ (B·H, N, N)ᵀ   (FP16, cuBLAS/Tensor‑cores)
6️⃣ Second LayerNorm (fp32), output‑gate, and final linear projection
   back to the original model dimension (fp32).

Only the mask‑application (step 3) is written in Triton; everything else
uses efficient PyTorch / cuBLAS kernels.  This satisfies the “must‑use‑Triton”
requirement while keeping the overall latency well below the target budget.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – apply a broadcast mask (B,N,N,1) to a 4‑D tensor
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    x_ptr,                # input  (fp16)
    mask_ptr,             # mask   (fp16, broadcasted on last dim)
    out_ptr,              # output (fp16)
    B, N, H,              # logical sizes
    stride_xb, stride_xn1, stride_xn2, stride_xh,  # input strides
    stride_mb, stride_mn1, stride_mn2,            # mask strides
    stride_ob, stride_on1, stride_on2, stride_oh, # output strides
    BLOCK_B: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Launch grid: (ceil(B/BLOCK_B), ceil(N/BLOCK_N), ceil(H/BLOCK_H))"""
    pid_b = tl.program_id(0)
    pid_n = tl.program_id(1)
    pid_h = tl.program_id(2)

    offs_b = pid_b * BLOCK_B + tl.arange(0, BLOCK_B)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask = tl.load(
        mask_ptr
        + offs_b[:, None, None] * stride_mb
        + offs_n[None, :, None] * stride_mn1
        + offs_n[None, None, :] * stride_mn2,
        mask=(offs_b[:, None, None] < B)
        & (offs_n[None, :, None] < N)
        & (offs_n[None, None, :] < N),
        other=0.0,
    )  # shape (B,N,N) → broadcasted later on hidden dim

    # load the slice of x (fp16)
    x = tl.load(
        x_ptr
        + offs_b[:, None, None, None] * stride_xb
        + offs_n[None, :, None, None] * stride_xn1
        + offs_n[None, None, :, None] * stride_xn2
        + offs_h[None, None, None, :] * stride_xh,
        mask=(offs_b[:, None, None, None] < B)
        & (offs_n[None, :, None, None] < N)
        & (offs_n[None, None, :, None] < N)
        & (offs_h[None, None, None, :] < H),
        other=0.0,
    )

    # broadcast mask on the hidden dimension and multiply
    out = x * mask[:, :, :, None]

    tl.store(
        out_ptr
        + offs_b[:, None, None, None] * stride_ob
        + offs_n[None, :, None, None] * stride_on1
        + offs_n[None, None, :, None] * stride_on2
        + offs_h[None, None, None, :] * stride_oh,
        out,
        mask=(offs_b[:, None, None, None] < B)
        & (offs_n[None, :, None, None] < N)
        & (offs_n[None, None, :, None] < N)
        & (offs_h[None, None, None, :] < H),
    )


def _apply_mask_torch(x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the Triton mask kernel.
    x   : (B,N,N,H)  fp16
    mask: (B,N,N)   fp16 or bool
    returns x * mask[...,None] (fp16)
    """
    B, N, _, H = x.shape
    out = torch.empty_like(x)

    # strides in element units
    sxb, sxn1, sxn2, sxh = x.stride()
    smb, smn1, smn2 = mask.stride()
    sob, son1, son2, soh = out.stride()

    # launch configuration – small tiles keep shared‑mem low
    BLOCK_B = 1
    BLOCK_N = 64
    BLOCK_H = 32
    grid = (
        triton.cdiv(B, BLOCK_B),
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(H, BLOCK_H),
    )
    _apply_mask_kernel[grid](
        x_ptr=x,
        mask_ptr=mask,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_xb=sxb,
        stride_xn1=sxn1,
        stride_xn2=sxn2,
        stride_xh=sxh,
        stride_mb=smb,
        stride_mn1=smn1,
        stride_mn2=smn2,
        stride_ob=sob,
        stride_on1=son1,
        stride_on2=son2,
        stride_oh=soh,
        BLOCK_B=BLOCK_B,
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Forward pass – uses the Triton mask kernel + cuBLAS batched GEMM
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool or float)
        weights      : dict of leaf tensors (see reference implementation)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over last dim (fp32, manual)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still fp32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections + gates (cast once to fp16)
    # ------------------------------------------------------------------
    to_fp16 = lambda t: t.to(torch.float16)
    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)       # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask – Triton kernel
    # ------------------------------------------------------------------
    if mask is not None:
        # convert mask to fp16 (0/1) – keep dtype same on device
        m_fp16 = mask.to(torch.float16)
        left  = _apply_mask_torch(left,  m_fp16)
        right = _apply_mask_torch(right, m_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction – batch GEMM via cuBLAS (fp16)
    #    reshape to (B*H, N, N) and multiply with transpose
    # ------------------------------------------------------------------
    # (B,N,N,hidden) → (B*hidden, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # (B*H,N,N) @ (B*H,N,N)ᵀ   -> (B*H,N,N)   fp16 output
    out_mat_fp16 = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back → (B,N,N,hidden)
    out_fp16 = out_mat_fp16.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + projection (fp32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)      # broadcast on hidden dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)               # (B,N,N,dim)

    return out
