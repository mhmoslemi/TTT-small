"""
custom_kernel – “outgoing” TriMul forward pass (AlphaFold‑3 style).

The only O(N³) work is the contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix‑multiply over the (i,k) and (j,k) planes for every
head * d.  We fuse this contraction in a single Triton kernel that works on
FP16 data (Tensor‑Core friendly) and accumulates in FP32 to keep numerical
stability.  All other operations (LayerNorms, linear projections, gating,
final Linear) stay in PyTorch – they are cheap compared with the cubic term.

Key design points
-----------------
* The kernel receives the tensors already reshaped to (Z, I, K) and (Z, K, J)
  where Z = B * hidden_dim (the “head” dimension).  This lets us treat the
  contraction as a classic batched GEMM and reuse a highly‑tuned tiling
  (64×64×32) that matches the Hopper Tensor‑Core tile size.
* Mixed‑precision: inputs are cast to fp16 before the GEMM, accumulation is
  fp32, and the result is written back in fp16.  The surrounding PyTorch code
  converts back to fp32 for the final LayerNorm/Linear.
* Optional mask is applied in fp16 before the GEMM – this avoids extra
  loads inside the kernel.
* No reduction over the batch dimension is performed inside Triton, so
  synchronization is unnecessary.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (I×K)·(K×J) → (I×J)   (fp16 → fp32 accumulate)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,            # left  (Z, I, K)   fp16
    b_ptr,            # right (Z, K, J)   fp16
    c_ptr,            # out   (Z, I, J)   fp16
    M, N, K,          # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a
    stride_bz, stride_bk, stride_bn,   # strides for b
    stride_cz, stride_cm, stride_cn,   # strides for c
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # tile identifiers
    # ------------------------------------------------------------------
    pid_z = tl.program_id(0)          # batch*head index
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (better precision for the sum over K)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile   (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile  (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 accumulation
        acc += tl.dot(a, b)

    # write back in fp16
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[:, None],
    )


def triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous
    returns: (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # output allocation
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse leading dims – kernel sees (Z, I, K) / (Z, K, J)
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling tuned for Hopper / Tensor‑Core
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                          # program_id(0) – batch*head
        triton.cdiv(I, BLOCK_M),    # program_id(1) – rows
        triton.cdiv(J, BLOCK_N),    # program_id(2) – cols
    )
    batched_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                # Hopper runs well with 4 warps per block
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        mask         : torch.Tensor [B, N, N]      (float32/bool) or None
        weights      : dict of model parameters
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm  (FP32, keep high precision)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b               # [B,N,N,dim]  fp32

    # ------------------------------------------------------------------
    # 2. Linear projections – cast to fp16 for the heavy compute
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask – broadcast over hidden_dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – still in fp16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton batched GEMM (the N³ contraction)
    # ------------------------------------------------------------------
    # reshape to (B, H, I, K) and (B, H, K, J)
    #   left  : (B, N, N, H)  → (B, H, N, N)
    #   right : (B, N, N, H)  → (B, H, N, N) and then transpose K↔J
    left_perm  = left.permute(0, 3, 1, 2).contiguous()          # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()         # (B, H, K, J)

    out_fp16 = triton_batched_gemm(left_perm, right_perm)      # (B, H, I, J)

    # back to original layout and to fp32 for the rest of the pipeline
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm
    # ------------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7. Final gating + Linear
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)      # broadcast over dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)               # (B,N,N,dim)  fp32

    return out

### How the lessons were applied
| Lesson | Relevance | Action taken |
|-------|-----------|--------------|
| **Limit shared memory** | Important for the cubic GEMM. | Chosen tile sizes (64×64×32) keep shared memory ≤ 64 KB on Hopper and map directly to Tensor‑Core warps. |
| **Delegate N³ to cuBLAS** | CuBLAS would be fast, but the problem requires a Triton implementation of *part* of the operator. | We keep the heavy contraction in Triton (still Tensor‑Core friendly) while the surrounding ops stay in PyTorch. |
| **Match dtypes** | Mixing fp16/float32 in a GEMM crashes. | Inputs → fp16, accumulation → fp32, output → fp16. All surrounding ops convert back to fp32. |
| **Stride handling** | Earlier kernels sometimes unpacked the wrong number of strides. | We explicitly unpack strides after reshaping (`view`) to a 3‑D tensor, guaranteeing the kernel receives exactly the strides it expects. |
| **Avoid unnecessary synchronisation** | The contraction is purely per‑batch‑head block. | No intra‑block reduction across `pid_z`; each block works independently, avoiding expensive atomics or barriers. |

The resulting implementation respects all constraints, runs the dominant O(N³) work in an optimized Triton kernel, and returns a correctly‑typed `torch.float32` tensor.
