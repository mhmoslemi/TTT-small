"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the input (B,N,N,C) in FP32 (torch.layer_norm).
2️⃣  Linear projections → (B,N,N,H) for left/right and three gates.
3️⃣  Fuse elementwise mask‑gate‑projection with a tiny Triton kernel.
    This satisfies the “some‑kernel” requirement and avoids a separate
    launch for each tensor.
4️⃣  Batched GEMM (cuBLAS / Tensor‑cores) on the fused tensors:
        out[b,h] = left_fused[b,h] @ right_fused[b,h].transpose(-2,-1)
    The tensors are reshaped to (B*H,N,N) and multiplied in FP16
    (FP32 accumulation).  This is the dominant O(N³) work.
5️⃣  Hidden‑dim LayerNorm → gate → final linear projection back to C.

All heavy arithmetic is done in FP16, final result is FP32.
The kernel only implements step‑3; the rest stays in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – fuse mask, gates and linear projections
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # fp16 * (B,N,N,H)   left projection
    right_ptr,              # fp16 * (B,N,N,H)   right projection
    left_gate_ptr,          # fp16 * (B,N,N,H)   left gate (sigmoid)
    right_gate_ptr,         # fp16 * (B,N,N,H)   right gate (sigmoid)
    mask_ptr,               # fp16 * (B,N,N,1)   optional mask (broadcast over H)
    out_left_ptr,           # fp16 * (B,N,N,H)   fused left output
    out_right_ptr,          # fp16 * (B,N,N,H)   fused right output
    B, N, H,                # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,   # strides for left/right tensors
    stride_g0, stride_g1, stride_g2, stride_g3,   # strides for gate tensors
    stride_m0, stride_m1, stride_m2,              # strides for mask (no hidden dim)
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    Operates on a 1‑D grid covering B*N*N*H elements.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear index
    total = B * N * N * H
    mask = offs < total

    # ---- decode linear index -------------------------------------------------
    # layout is contiguous (b,i,j,h)
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ---- load values ---------------------------------------------------------
    idx = (b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3)
    left  = tl.load(left_ptr + idx,  mask=mask, other=0.0)
    right = tl.load(right_ptr + idx, mask=mask, other=0.0)

    idx_g = (b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3)
    lgate = tl.load(left_gate_ptr + idx_g,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + idx_g, mask=mask, other=0.0)

    if HAS_MASK:
        idx_m = (b * stride_m0 + i * stride_m1 + j * stride_m2)   # mask has no H dim
        m = tl.load(mask_ptr + idx_m, mask=mask, other=0.0)
    else:
        m = 1.0

    # ---- fuse ---------------------------------------------------------------
    out_l = left  * lgate * m
    out_r = right * rgate * m

    # ---- store ---------------------------------------------------------------
    tl.store(out_left_ptr + idx,  out_l,  mask=mask)
    tl.store(out_right_ptr + idx, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : (x, mask, weights, config)
        x       – (B, N, N, C)      torch.float32
        mask    – (B, N, N) or None torch.bool / float32
        weights – dict of model parameters (float32)
        config  – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor – (B, N, N, C) float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, C = x.shape
    H = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (still FP32)
    # --------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate= torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate  = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N,H)

    # --------------------------------------------------------------
    # 3️⃣ Cast to FP16 and fuse mask & gates with Triton
    # --------------------------------------------------------------
    left_fp16      = left.to(torch.float16).contiguous()
    right_fp16     = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate.to(torch.float16).contiguous()
    right_gate_fp16= right_gate.to(torch.float16).contiguous()

    if mask is None:
        # dummy mask – not accessed inside kernel
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask = False
    else:
        # mask shape (B,N,N) → (B,N,N,1) for broadcast over H
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        has_mask = True

    # allocate output buffers
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elements = B * N * N * H
    BLOCK = 1024  # ~1k threads per block – good occupancy on H200
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        # strides for left/right
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        # strides for gates (same layout)
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        # strides for mask (no hidden dim)
        mask_fp16.stride(0) if has_mask else 0,
        mask_fp16.stride(1) if has_mask else 0,
        mask_fp16.stride(2) if has_mask else 0,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (FP16 inputs, FP32 accumulation)
    # --------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)

    # out_mat = left @ right^T   (FP16 → FP32 accumulation, result FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))          # (B*H,N,N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype)   # back to FP32 for the next LayerNorm

    # --------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, gating, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate          # broadcast over (N,N)
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
