"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣  Layer‑norm + four linear projections (left, right, left‑gate,
    right‑gate, out‑gate) are cheap – they stay in PyTorch.

2️⃣  A single Triton kernel fuses:
        * mask broadcast (or a tensor of ones when nomask)
        * element‑wise multiplication of the left/right tensors by the mask
        * multiplication by the sigmoid‑gates
   This is the mandatory “use a kernel” part.  The kernel works on the
   contiguous FP16 layout (B,N,N,H) and is completely memory‑bound.

3️⃣  The cubic O(N³) contraction
        out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
   is performed with a batched GEMM using torch.bmm (Tensor‑core
   accelerated on Hopper).  The tensors are reshaped to (B·H, N, N) and the
   right operand is transposed on the second axis.

4️⃣  Post‑processing (layer‑norm, out‑gate, final linear) is again done in
   PyTorch.  The final result is returned in float32.

The kernel is deliberately simple – it only touches the 4‑D tensors,
avoids shared‑memory tiling and uses a large 1‑D block (1024) to keep the
occupancy high on H200.  All strides are passed explicitly, making the
kernel work for any contiguous layout.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into left/right tensors
# -------------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *mut fp16  (B,N,N,H) – overwritten in‑place
    right_ptr,         # *mut fp16  (B,N,N,H)
    left_gate_ptr,     # *const fp16 (B,N,N,H)
    right_gate_ptr,    # *const fp16 (B,N,N,H)
    mask_ptr,          # *const fp16 (B,N,N,1) – broadcasted over H
    B, N, H,           # shape parameters
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_gg0, stride_gg1, stride_gg2, stride_gg3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N * H
    idx = pid * BLOCK + tl.arange(0, BLOCK)
    mask = idx < total

    # ----- linear index → (b,i,j,h) ------------------------------------
    tmp = idx
    h = tmp % H;               tmp = tmp // H
    j = tmp % N;               tmp = tmp // N
    i = tmp % N;               b = tmp // N

    # ----- compute flat pointers ----------------------------------------
    l_ptr = left_ptr  + b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    r_ptr = right_ptr + b * stride_r0 + i * stride_r1 + j * stride_r2 + h * stride_r3
    lg_ptr = left_gate_ptr  + b * stride_gg0 + i * stride_gg1 + j * stride_gg2 + h * stride_gg3
    rg_ptr = right_gate_ptr + b * stride_gg0 + i * stride_gg1 + j * stride_gg2 + h * stride_gg3
    m_ptr = mask_ptr + b * stride_m0 + i * stride_m1 + j * stride_m2          # H‑dim broadcast

    # ----- load -----------------------------------------------------------
    l  = tl.load(l_ptr,  mask=mask, other=0.0)
    r  = tl.load(r_ptr, mask=mask, other=0.0)
    lg = tl.load(lg_ptr, mask=mask, other=0.0)
    rg = tl.load(rg_ptr, mask=mask, other=0.0)
    m  = tl.load(m_ptr,  mask=mask, other=1.0)   # 1.0 when mask is “all‑ones”

    # ----- fused elementwise ---------------------------------------------
    l = l * m * lg
    r = r * m * rg

    # ----- store ----------------------------------------------------------
    tl.store(l_ptr, l, mask=mask)
    tl.store(r_ptr, r, mask=mask)


def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input [B,N,N,dim]  (float32)
        torch.Tensor,                # mask  [B,N,N] (bool/float or None)
        Dict[str, torch.Tensor],     # model weights (float32)
        Dict[str, Any],              # config: {"dim":int, "hidden_dim":int}
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    H = config["hidden_dim"]          # hidden dimension (c_z)

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm + linear projections (remain in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )
    # shape (B,N,N,H) after projection
    left  = F.linear(x, weights["left_proj.weight"].to(dtype))
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # gates (pre‑sigmoid)
    left_gate  = F.linear(x, weights["left_gate.weight"].to(dtype))
    right_gate = F.linear(x, weights["right_gate.weight"].to(dtype))
    out_gate   = F.linear(x, weights["out_gate.weight"].to(dtype))

    # sigmoid
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # -----------------------------------------------------------------
    # 2️⃣ Triton kernel – mask + gate fusion
    # -----------------------------------------------------------------
    # mask tensor of shape (B,N,N,1); all‑ones when nomask
    if mask is None:
        mask_tensor = torch.ones((B, N, N, 1), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(dtype).unsqueeze(-1)          # (B,N,N,1)
        mask_tensor = mask_tensor.to(torch.float16)

    # cast to fp16 for the kernel (memory‑bound, no precision loss for this step)
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    left_gate_fp16  = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()

    total_el = B * N * N * H
    BLOCK = 1024                     # good occupancy for pure elementwise work
    grid = (triton.cdiv(total_el, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_tensor,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        mask_tensor.stride(0), mask_tensor.stride(1),
        mask_tensor.stride(2), mask_tensor.stride(3),
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 3️⃣ Batched GEMM (the O(N³) part) – use Tensor‑cores via torch.bmm
    # -----------------------------------------------------------------
    # reshape to (B·H, N, N).  We keep fp16 for the GEMM.
    left_bh = left_fp16.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B·H,N,N)
    right_bh = right_fp16.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # (B·H,N,N) @ (B·H,N,N)^T → (B·H,N,N)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))

    # back to (B,N,N,H) → (B,N,N,dim) later
    out = out_bh.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype)                     # fp32 for the remaining ops

    # -----------------------------------------------------------------
    # 4️⃣ Post‑processing
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                     # broadcast over N,N
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
