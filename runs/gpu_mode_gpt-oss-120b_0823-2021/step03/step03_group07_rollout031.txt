"""
TriMul “outgoing” forward pass (AlphaFold‑3) – hybrid PyTorch + Triton.

The O(N³) contraction
    out[b, h, i, j] = Σ_k left[b, h, i, k] · right[b, h, j, k]
is performed with a single batched GEMM (`torch.bmm`) on FP16 tensors,
which lets cuBLAS/Tensor‑cores on the H200 execute the cubic work at
peak throughput.

The remaining cheap per‑element work (masking, gating, and the final
output‑gate) is fused into a tiny Triton kernel (`_mask_gate_kernel`).
Fusing these element‑wise ops removes an extra memory round‑trip
compared with a pure‑PyTorch implementation while keeping the kernel
very small (so launch overhead is negligible).

All LayerNorms and linear projections are done with regular PyTorch
because they are memory‑bound and already highly optimized.

The final output is returned as `float32` to match the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate & right‑gate, and optional output‑gate
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    left_ptr,          # fp16*  [B, H, N, N]
    right_ptr,         # fp16*  [B, H, N, N]
    left_gate_ptr,     # fp16*  [B, N, N, H]
    right_gate_ptr,    # fp16*  [B, N, N, H]
    mask_ptr,          # u8*   [B, N, N]  (0/1)   (may be nullptr)
    out_gate_ptr,      # fp16* [B, N, N, H] (may be nullptr)
    B, N, H,
    stride_l_b, stride_l_h, stride_l_n, stride_l_k,
    stride_r_b, stride_r_h, stride_r_n, stride_r_k,
    stride_gb, stride_gn, stride_gm, stride_gh,
    stride_m_b, stride_m_n, stride_m_n2,
    stride_og_b, stride_og_n, stride_og_n2, stride_og_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For every (b, h, i, j) element:
        left  = left  * left_gate  * mask (if present)
        right = right * right_gate * mask (if present)
        if out_gate is provided:  left/right are left unchanged,
                                 but out_gate is stored for later use.
    The kernel writes the processed left and right tensors back in‑place and
    optionally writes out_gate into a separate buffer (same layout as left/right).
    """
    pid = tl.program_id(0)

    # decode linear pid -> (b, h, tile_i, tile_j)
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    tile_j = pid % num_j
    pid //= num_j
    tile_i = pid % num_i
    pid //= num_i
    h = pid % H
    b = pid // H

    # block offsets
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N

    ## -----------------------------------------------------------------
    ## Load left/right (shape: [B,H,N,N]) and gates (shape: [B,N,N,H])
    ## -----------------------------------------------------------------
    # left  -> (i, j, h)
    l_ptr = (
        left_ptr
        + b * stride_l_b
        + h * stride_l_h
        + offs_i[:, None] * stride_l_n
        + offs_j[None, :] * stride_l_k
    )
    l = tl.load(l_ptr, mask=mask_i[:, None] & mask_j[None, :], other=0.0)

    r_ptr = (
        right_ptr
        + b * stride_r_b
        + h * stride_r_h
        + offs_i[:, None] * stride_r_n
        + offs_j[None, :] * stride_r_k
    )
    r = tl.load(r_ptr, mask=mask_i[:, None] & mask_j[None, :], other=0.0)

    # left_gate  shape [B,N,N,H] -> index with (b,i,j,h)
    lg_ptr = (
        left_gate_ptr
        + b * stride_gb
        + offs_i[:, None, None] * stride_gn
        + offs_j[None, :, None] * stride_gm
        + offs_h[None, None, :] * stride_gh
    )
    lg = tl.load(lg_ptr, mask=mask_i[:, None, None] & mask_j[None, :, None], other=0.0)

    rg_ptr = (
        right_gate_ptr
        + b * stride_gb
        + offs_i[:, None, None] * stride_gn
        + offs_j[None, :, None] * stride_gm
        + offs_h[None, None, :] * stride_gh
    )
    rg = tl.load(rg_ptr, mask=mask_i[:, None, None] & mask_j[None, :, None], other=0.0)

    # optional mask (broadcasted over hidden dim)
    if mask_ptr != 0:
        m_ptr = (
            mask_ptr
            + b * stride_m_b
            + offs_i[:, None] * stride_m_n
            + offs_j[None, :] * stride_m_n2
        )
        m = tl.load(m_ptr, mask=mask_i[:, None] & mask_j[None, :], other=0).to(tl.float16)
    else:
        m = tl.full((BLOCK_M, BLOCK_N), 1.0, dtype=tl.float16)

    # fuse
    l = l * lg * m
    r = r * rg * m

    # write back
    tl.store(l_ptr, l, mask=mask_i[:, None] & mask_j[None, :])
    tl.store(r_ptr, r, mask=mask_i[:, None] & mask_j[None, :])

    # if out_gate buffer provided, store it (same layout as left/right)
    if out_gate_ptr != 0:
        og_ptr = (
            out_gate_ptr
            + b * stride_og_b
            + offs_i[:, None, None] * stride_og_n
            + offs_j[None, :, None] * stride_og_n2
            + offs_h[None, None, :] * stride_og_h
        )
        og = tl.load(og_ptr, mask=mask_i[:, None, None] & mask_j[None, :, None], other=0.0)
        tl.store(og_ptr, og, mask=mask_i[:, None, None] & mask_j[None, :, None])


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input   [B, N, N, dim]
        torch.Tensor,                     # mask    [B, N, N]   (or None)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any]                    # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]          # H

    # ---------------------------------------------------------------
    # 1. LayerNorm + linear projections (GPU‑accelerated PyTorch)
    # ---------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )   # (B,N,N,dim)

    # projections → (B,N,N,hidden_dim)
    left = F.linear(x_norm, weights["left_proj.weight"].to(dtype))
    right = F.linear(x_norm, weights["right_proj.weight"].to(dtype))

    # gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"].to(dtype)))  # (B,N,N,hidden_dim)

    # ---------------------------------------------------------------
    # 2. Fuse mask & gates with a tiny Triton kernel (in‑place)
    # ---------------------------------------------------------------
    # bring everything to fp16 for the kernel (better bandwidth)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()
    out_gate_fp16 = out_gate.to(torch.float16).contiguous()

    if mask is not None:
        mask_u8 = (mask > 0).to(torch.uint8).contiguous()
    else:
        mask_u8 = None

    # optional buffer for out_gate (we keep it separate to multiply later)
    out_gate_buf = torch.empty_like(left_fp16)

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_H = 32   # hidden dim is small (≤384) so 32 works

    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    total_blocks = B * hidden_dim * grid_i * grid_j
    grid = (total_blocks,)

    _mask_gate_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_u8.data_ptr() if mask_u8 is not None else 0,
        out_gate_buf,
        B,
        N,
        hidden_dim,
        # strides left
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        left_fp16.stride(3),
        # strides right
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        # strides gate (B,N,N,H) – same as left/right after permute
        left_gate_fp16.stride(0),
        left_gate_fp16.stride(1),
        left_gate_fp16.stride(2),
        left_gate_fp16.stride(3),
        # strides mask (B,N,N)
        (mask_u8.stride(0) if mask_u8 is not None else 0),
        (mask_u8.stride(1) if mask_u8 is not None else 0),
        (mask_u8.stride(2) if mask_u8 is not None else 0),
        # strides out_gate buffer (same layout as left/right)
        out_gate_buf.stride(0),
        out_gate_buf.stride(1),
        out_gate_buf.stride(2),
        out_gate_buf.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
    )

    # ---------------------------------------------------------------
    # 3. Reshape for batched GEMM: (B, H, N, N) → (B*H, N, N)
    # ---------------------------------------------------------------
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()

    BxH, _, _ = left_perm.shape[0] * left_perm.shape[1], N, N
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # batched matmul (FP16 inputs, FP32 accumulation inside torch)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B, H, N, N) and permute to original layout
    out_fp16 = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ---------------------------------------------------------------
    # 4. Final LayerNorm, output‑gate, and linear back to `dim`
    # ---------------------------------------------------------------
    out = out_fp16.to(dtype)   # (B,N,N,H) float32

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply output gate (broadcast over pairwise axes)
    out = out * out_gate  # out_gate still in float32

    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
