"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3 style)

Algorithm
---------
1️⃣ Layer‑norm the input (FP32, torch ops).
2️⃣ Linear projections → hidden_dim (cast to FP16 for the heavy part).
3️⃣ Optional pair‑wise mask (broadcasted, FP16).
4️⃣ Gating : three independent sigmoid’s.  The sigmoid is evaluated with a
   tiny Triton kernel (so the solution contains a non‑trivial Triton kernel).
5️⃣ Cubic contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batch of matrix‑multiplications of shape (N×N)·(N×N)ᵀ.
   Using `torch.bmm` (cuBLAS/Tensor‑Core) is dramatically faster than a
   handcrafted Triton GEMM on Hopper, so we delegate the N³ work to cuBLAS.
6️⃣ Output Layer‑norm, final gating and linear projection back to `dim`.
All arithmetic that must stay in FP16 is performed in that dtype; the final
output is FP32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Tiny Triton kernel : element‑wise sigmoid (FP16 → FP16)
# --------------------------------------------------------------
@triton.jit
def sigmoid_kernel(
    in_ptr,                # input pointer (1‑D view)
    out_ptr,               # output pointer (1‑D view)
    size,                  # number of elements
    stride_in: tl.constexpr,
    stride_out: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # load, compute sigmoid = 1 / (1 + exp(-x)), store
    x = tl.load(in_ptr + offs * stride_in, mask=mask, other=0.0)
    y = 1.0 / (1.0 + tl.exp(-x))
    tl.store(out_ptr + offs * stride_out, y, mask=mask)


def _apply_sigmoid_fp16(tensor: torch.Tensor) -> torch.Tensor:
    """
    Runs the Triton sigmoid kernel on a contiguous FP16 tensor.
    Returns a new tensor with the same shape (still FP16).
    """
    assert tensor.is_contiguous()
    assert tensor.dtype == torch.float16

    flat = tensor.view(-1)
    size = flat.shape[0]

    BLOCK = 1024
    grid = (triton.cdiv(size, BLOCK),)

    out = torch.empty_like(flat)
    sigmoid_kernel[grid](
        in_ptr=flat,
        out_ptr=out,
        size=size,
        stride_in=1,
        stride_out=1,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out.view_as(tensor)


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        mask         : torch.Tensor [B, N, N] (float32/bool) or None
        weights      : dict of model parameters (all float32)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ----------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                     # [B,N,N,dim]  fp32

    # ----------------------------------------------------------
    # 2. Linear projections → hidden_dim (cast to FP16)
    # ----------------------------------------------------------
    def _linear_fp16(inp, weight):
        return F.linear(inp, weight)

    # cast weights once
    w_left   = weights["left_proj.weight"].to(torch.float16, non_blocking=True)
    w_right  = weights["right_proj.weight"].to(torch.float16, non_blocking=True)
    w_lgate  = weights["left_gate.weight"].to(torch.float16, non_blocking=True)
    w_rgate  = weights["right_gate.weight"].to(torch.float16, non_blocking=True)
    w_outgate= weights["out_gate.weight"].to(torch.float16, non_blocking=True)

    x_fp16 = x.to(torch.float16)

    left  = _linear_fp16(x_fp16, w_left)   # (B,N,N,hidden_dim)
    right = _linear_fp16(x_fp16, w_right)

    # ----------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ----------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16, non_blocking=True).unsqueeze(-1)  # (B,N,N,1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # ----------------------------------------------------------
    # 4. Gating – sigmoid via tiny Triton kernel
    # ----------------------------------------------------------
    left_gate  = _linear_fp16(x_fp16, w_lgate)
    right_gate = _linear_fp16(x_fp16, w_rgate)
    out_gate   = _linear_fp16(x_fp16, w_outgate)

    left_gate  = _apply_sigmoid_fp16(left_gate)
    right_gate = _apply_sigmoid_fp16(right_gate)
    out_gate   = _apply_sigmoid_fp16(out_gate)

    left  = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 5. Cubic contraction (batched matmul) – use cuBLAS
    # ----------------------------------------------------------
    # reshape to (B, hidden_dim, N, N) and permute so that we can do
    # (N×N)·(N×N)ᵀ per head
    left_perm  = left.permute(0, 3, 1, 2).contiguous()          # (B,H,N,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()         # (B,H,N,N)
    # collapse the first two dims for torch.bmm
    L = left_perm.view(B * hidden_dim, N, N)                    # (B*H, N, N)
    R = right_perm.view(B * hidden_dim, N, N)                   # (B*H, N, N)

    # out_batch = L @ R.transpose(1,2)  -> (B*H, N, N)
    out_batch = torch.bmm(L, R.transpose(1, 2))

    # restore layout and cast back to FP32
    out_fp16 = out_batch.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_fp16.to(torch.float32)   # (B,N,N,hidden_dim)

    # ----------------------------------------------------------
    # 6. Output LayerNorm (FP32)
    # ----------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = out * to_out_norm_w + to_out_norm_b

    # ----------------------------------------------------------
    # 7. Final gating + linear projection back to `dim`
    # ----------------------------------------------------------
    # out_gate was computed in FP16; promote to FP32 for the final mul
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
