The heavy‑weight cubic contraction  

\[
\text{out}[b,i,j,d]=\sum_{k} \text{left}[b,i,k,d]\;\text{right}[b,j,k,d]
\]

is performed by a custom Triton kernel that treats each `(b, d)` pair (batch × hidden) as an independent GEMM of size `N×N`.  
All cheap work – LayerNorms, linear projections, masking and gating – stays in PyTorch, which on the H200 is orders of magnitude faster than a hand‑written Triton loop over the three‑dimensional index space.

### Why this design works
* **Batched GEMM** – each `(b, head)` slice is a standard matrix‑multiply (`M=N, K=N, N=N`).  Triton efficiently tiles the `(M,N)` output while iterating over `K` in `BLOCK_K`‑chunks, using FP16 arithmetic with FP32 accumulation (Tensor‑Core friendly).  
* **Memory layout** – we permute the tensor to `[B, H, N, N]` and then `view` it as `[BH, N, N]`.  This yields contiguous stride‑1 access along the innermost dimension, which is optimal for the kernel.  
* **Mask handling** – when a mask is supplied we broadcast it once (still in PyTorch) before the GEMM; the kernel never needs to check a mask, keeping the inner loop tight.  
* **Correctness** – the kernel writes the result back as FP16, the host converts to FP32, applies the second LayerNorm, output‑gate and final linear projection exactly as in the reference implementation.

```python
"""
TriMul “outgoing” operator (AlphaFold‑3) – Triton‑accelerated cubic contraction.

The expensive contraction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

is implemented as a batched GEMM (B·H, N, N) using a custom Triton kernel.
All other ops (LayerNorm, linear projections, masking, gating) stay in PyTorch
because they are cheap compared to the O(N³) work.

The kernel works in FP16 with FP32 accumulation (Tensor‑Core friendly) and
writes FP16 back; the wrapper restores FP32 for the remaining PyTorch ops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM for each (batch, head) pair
# -------------------------------------------------------------------------
@triton.jit
def trimul_batched_gemm_kernel(
    left_ptr,          # const fp16*  [BH, N, N]
    right_ptr,         # const fp16*  [BH, N, N] (will be read as transposed)
    out_ptr,           #       fp16*  [BH, N, N]
    BH, N,             # problem size
    stride_lbh, stride_lbn, stride_lbk,   # left strides
    stride_rbh, stride_rbn, stride_rbk,   # right strides
    stride_obh, stride_obn, stride_obk,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a tile (BLOCK_M × BLOCK_N) of the matrix product
        out[bh, i, j] = Σ_k left[bh, i, k] * right[bh, j, k]
    where `right` is accessed as its transpose.
    """
    pid = tl.program_id(0)                     # linear id over (bh, tile_m, tile_n)

    # -------------------------------------------------
    # Decode program id into (bh, tile_m, tile_n)
    # -------------------------------------------------
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    tile_n = pid % num_tiles_n
    pid //= num_tiles_n
    tile_m = pid % num_tiles_m
    pid //= num_tiles_m
    bh = pid                                   # 0 <= bh < BH

    # -------------------------------------------------
    # Offsets for this tile
    # -------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Loop over K dimension in BLOCK_K chunks
    # -------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[bh, i, k]   shape (M, K)
        l_ptr = left_ptr + bh * stride_lbh \
                + offs_m[:, None] * stride_lbn \
                + offs_k[None, :] * stride_lbk
        l = tl.load(l_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[bh, j, k] needed as (K, N) because we multiply
        # left[i,k] * right[j,k] = left[i,k] * (right^T)[k,j]
        r_ptr = right_ptr + bh * stride_rbh \
                + offs_k[:, None] * stride_rbk \
                + offs_n[None, :] * stride_rbn
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # GEMM: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(l, r)

    # -------------------------------------------------
    # Store result back as FP16
    # -------------------------------------------------
    out_ptrs = out_ptr + bh * stride_obh \
               + offs_m[:, None] * stride_obn \
               + offs_n[None, :] * stride_obk
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper: TriMul forward pass (no gradients)
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input : [B, N, N, dim]
        torch.Tensor,                # mask  : [B, N, N] (or None)
        Dict[str, torch.Tensor],     # weights dict
        Dict[str, Any],              # config dict {"dim":..., "hidden_dim":...}
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # H in the paper

    # -----------------------------------------------------------------
    # 1️⃣  LayerNorm + linear projections   (cheap – stay in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 2️⃣  Optional mask (broadcast once)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)            # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 3️⃣  Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4️⃣  Reshape for the batched GEMM and cast to FP16
    # -----------------------------------------------------------------
    # (B,N,N,H) -> (B, H, N, N) -> (BH, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left.view(B * hidden_dim, N, N).to(torch.float16)
    right_fp16 = right.view(B * hidden_dim, N, N).to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)   # (BH, N, N), FP16

    # -----------------------------------------------------------------
    # 5️⃣  Triton launch
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    BH = B * hidden_dim
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_tiles = BH * grid_m * grid_n
    grid = (total_tiles,)

    trimul_batched_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        BH,
        N,
        # strides for left   (BH, N, N)
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2),
        # strides for right  (BH, N, N)
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2),
        # strides for out    (BH, N, N)
        out_fp16.stride(0), out_fp16.stride(1), out_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6️⃣  Convert back to original layout + remaining PyTorch ops
    # -----------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    out = out.to(dtype)   # back to FP32 for LayerNorm etc.

    # to_out_norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # output gate (broadcast over N,N)
    out = out * out_gate

    # final linear back to original dim
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
```
