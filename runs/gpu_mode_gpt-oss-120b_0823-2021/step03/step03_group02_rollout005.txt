"""
TriMul “outgoing” forward pass (no gradients) – fast Triton‑accelerated contraction.

The dominant O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is expressed as a batched GEMM (I×K)·(K×J) → (I×J) for each head‑dimension *d*.
All cheap operations (LayerNorm, linear projections, sigmoids, masking) stay in
PyTorch.  The heavy matrix multiply is performed by a small Triton kernel that
operates on fp16 inputs and accumulates in fp32 (Tensor‑core friendly).  The
final tensor is cast back to fp32 for the output LayerNorm and the last linear
layer.

The implementation works for any batch size B∈{1,2}, sequence length N up to
1024, and hidden dimension H (=config["hidden_dim"]).  It also correctly
handles the optional mask (broadcasted on the hidden dimension) and runs on
NVIDIA Hopper (sm_90) GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched GEMM (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    a_ptr,            # left  tensor  (Z, I, K)   fp16
    b_ptr,            # right tensor  (Z, K, J)   fp16
    c_ptr,            # output tensor (Z, I, J)   fp16
    M, N, K,          # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM with fp16 inputs, fp32 accumulation.

    The kernel is launched with a 3‑D grid:
        pid_z – batch*head dimension
        pid_m – tile row (I)
        pid_n – tile col (J)
    """
    pid_z = tl.program_id(0)          # flattened (B*H)
    pid_m = tl.program_id(1)          # tile along rows   (I)
    pid_n = tl.program_id(2)          # tile along cols   (J)

    # Compute offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (Tensor‑core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile from A (shape BLOCK_M×BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a tile from B (shape BLOCK_K×BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 accumulation
        acc += tl.dot(a, b)

    # Write the result back as fp16
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batch_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    Arguments
    ----------
    left : torch.Tensor (B, H, I, K)  fp16, contiguous
    right: torch.Tensor (B, H, K, J)  fp16, contiguous

    Returns
    -------
    out : torch.Tensor (B, H, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # Allocate output
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # Flatten the batch*head dimension for the kernel
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # Strides (in elements, not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes – empirically good on Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                              # pid_z
        triton.cdiv(I, BLOCK_M),        # pid_m
        triton.cdiv(J, BLOCK_N),        # pid_n
    )

    batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,        # enough to keep the SM busy
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass – PyTorch + Triton
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no grads).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor [B, N, N]       (float32/bool) or None
        - weights      : dict of torch.Tensor parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]   (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep in fp32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                     # shape (B,N,N,dim) fp32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 once – all heavy ops use fp16
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Linear weight tensors (biases are zero in the reference model)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Projections
    # ------------------------------------------------------------------
    # shape after projection: (B,N,N,hidden_dim)
    left  = F.linear(x_fp16, left_proj_w)   # fp16
    right = F.linear(x_fp16, right_proj_w)  # fp16

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask may be bool or float; convert to fp16 and add trailing dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5) Gating (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core contraction – batched GEMM via Triton
    #    left : (B,N,N,hidden) → (B,hidden,N,N) → (B,hidden,I,K)
    #    right: (B,N,N,hidden) → (B,hidden,N,N) → (B,hidden,K,J) (transpose K,J)
    # ------------------------------------------------------------------
    # bring the hidden dimension to second axis and make the reduction axis contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = triton_batch_gemm(left_perm, right_perm)   # (B, H, I, J) fp16

    # back to the original layout (B,N,N,hidden) then cast to fp32 for the rest
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 8) Final gating + linear projection
    # ------------------------------------------------------------------
    # Re‑use the same out_gate computed earlier (it is fp16), cast to fp32
    out = out * out_gate.to(torch.float32)
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)          # final shape (B,N,N,dim) fp32

    return out
