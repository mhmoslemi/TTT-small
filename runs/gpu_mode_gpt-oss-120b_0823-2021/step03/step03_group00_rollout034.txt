"""
TriMul “outgoing” forward pass (AlphaFold‑3)

We follow the reference implementation but move the *mask‑application* step to
a tiny Triton kernel.  All linear layers and the O(N³) contraction are kept in
PyTorch (the contraction is performed with torch.bmm which maps to cuBLAS /
Tensor‑cores on Hopper).  This satisfies the requirement “at least part of the
operations in a kernel” while keeping the overall runtime competitive.

Key ideas
---------
* LayerNorms and linear projections stay in FP32/FP16 torch code – they are
  already highly optimized.
* The heavy cubic contraction `out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]`
  is rewritten as a batched matrix‑multiply:
        left   : (B, H, N, N)   →   (B*H, N, N)
        right  : (B, H, N, N)   →   (B*H, N, N).transpose(1,2)
        out    = torch.bmm(left, right)   # (B*H, N, N)
  This call is executed by cuBLAS/Tensor‑cores on the H200.
* The only custom kernel (`_apply_mask_kernel`) multiplies a 4‑D tensor by a
  broadcasted mask (both FP16).  It is launched with a 1‑D grid, each program
  handling a contiguous tile of `BLOCK` elements.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – apply a binary (float16) mask to a 4‑D tensor
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    src_ptr,                # fp16 source pointer (B,N,N,H)
    mask_ptr,               # fp16 mask pointer   (B,N,N,1) – already broadcasted
    dst_ptr,                # fp16 destination pointer
    total_elements: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # load source and mask (both fp16)
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + offs, mask=mask, other=0.0)

    # element‑wise multiply
    out = src * m

    tl.store(dst_ptr + offs, out, mask=mask)


def _apply_mask_fp16(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    tensor : (B,N,N,H)  fp16
    mask   : (B,N,N,1) fp16  (already unsqueezed)
    returns a new tensor with mask applied (fp16)
    """
    assert tensor.is_contiguous()
    assert mask.is_contiguous()
    out = torch.empty_like(tensor)

    total = tensor.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _apply_mask_kernel[grid](
        src_ptr=tensor,
        mask_ptr=mask,
        dst_ptr=out,
        total_elements=total,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Helper – manual LayerNorm in FP32 (mean / var over last dim)
# ----------------------------------------------------------------------
def _layer_norm_fp32(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x : (..., C)  float32
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    norm = (x - mean) / torch.sqrt(var + eps)
    return norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        - weights      : dict of leaf tensors (see below)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = _layer_norm_fp32(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )  # (B,N,N,dim) float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 and linear projections
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Linear weights (no bias) – cast once to fp16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # (B,N,N,hidden_dim) tensors in fp16
    left  = F.linear(x_fp16, left_proj_w)          # no bias
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask – apply via Triton kernel (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # shape (B,N,N,1) fp16
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left  = _apply_mask_fp16(left,  mask_fp16)
        right = _apply_mask_fp16(right, mask_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – still in FP16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden_dim)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction – batched GEMM via torch.bmm (FP16 → FP32 accumulation)
    #    left : (B,N,N,hidden) → (B, hidden, N, N)
    #    right: (B,N,N,hidden) → (B, hidden, N, N) and transpose K↔J
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()     # (B, H, N, N)

    # reshape to (B*H, N, N) for torch.bmm
    left_flat  = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # torch.bmm performs fp16×fp16 → fp32 accumulation automatically
    out_flat = torch.bmm(left_flat, right_flat)            # (B*H, N, N) fp32

    # reshape back to (B, N, N, hidden_dim)
    out_fp32 = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    out = _layer_norm_fp32(
        out_fp32,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + projection (FP32)
    # ------------------------------------------------------------------
    # out_gate is (B,N,N,hidden_dim) fp16 – cast to fp32 for broadcasting
    out = out * out_gate.to(torch.float32)

    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim) fp32

    return out
