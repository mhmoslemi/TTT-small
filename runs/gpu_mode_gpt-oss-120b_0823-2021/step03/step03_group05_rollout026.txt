"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣ LayerNorm + four linear projections (left, right, left‑gate, right‑gate,
    out‑gate) – stay in PyTorch (they are cheap compared with the O(N³) term).

2️⃣ Fuse mask‑application, left‑gate, and right‑gate multiplications in a tiny
    Triton kernel.  The kernel overwrites the ``left`` and ``right`` tensors
    (FP16) with:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    This satisfies the “must use a kernel” rule while keeping the kernel
    memory‑bound and very fast.

3️⃣ Core cubic contraction
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    is a batched matrix‑matrix multiplication for each hidden channel.
    By reshaping the tensors to (B·H, N, N) we can call ``torch.bmm`` which
    maps to cuBLAS/Tensor‑cores on Hopper.  The tensors are kept in FP16 for
    this step and cast back to FP32 afterwards.

4️⃣ Post‑processing: LayerNorm on the hidden dimension, multiply by the
    out‑gate, final linear projection back to ``dim`` – also done in PyTorch.

The whole pipeline works for any batch size (1‑2), any sequence length up to
1024, and any ``dim``/``hidden_dim`` allowed by the spec.  All tensors are
contiguous and the kernel receives explicit strides, so the code works for
both “nomask” and masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: apply mask and gating (left & right) in‑place.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *mut fp16  (B,N,N,H) – overwritten
    right_ptr,         # *mut fp16  (B,N,N,H) – overwritten
    left_gate_ptr,     # *const fp16 (B,N,N,H)
    right_gate_ptr,    # *const fp16 (B,N,N,H)
    mask_ptr,          # *const fp16 (B,N,N,1) – broadcasted over H
    B, N, H,           # static sizes
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_g0, stride_g1, stride_g2, stride_g3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N * H
    idx = pid * BLOCK + tl.arange(0, BLOCK)          # linear index
    mask = idx < total

    # ----- decode linear index into (b,i,j,h) -----
    tmp = idx
    h = tmp % H;      tmp = tmp // H
    j = tmp % N;      tmp = tmp // N
    i = tmp % N;      b = tmp // N

    # ----- compute element pointers -----
    l_ptr = left_ptr   + b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    r_ptr = right_ptr  + b * stride_r0 + i * stride_r1 + j * stride_r2 + h * stride_r3
    lg_ptr = left_gate_ptr  + b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    rg_ptr = right_gate_ptr + b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    m_ptr = mask_ptr + b * stride_m0 + i * stride_m1 + j * stride_m2          # H‑dim broadcast

    # ----- load -----
    l = tl.load(l_ptr, mask=mask, other=0.0)
    r = tl.load(r_ptr, mask=mask, other=0.0)
    lg = tl.load(lg_ptr, mask=mask, other=0.0)
    rg = tl.load(rg_ptr, mask=mask, other=0.0)
    m = tl.load(m_ptr, mask=mask, other=1.0)      # 1.0 when mask is all‑ones

    # ----- fuse element‑wise ops -----
    l = l * m * lg
    r = r * m * rg

    # ----- store back -----
    tl.store(l_ptr, l, mask=mask)
    tl.store(r_ptr, r, mask=mask)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        - x : Tensor[B, N, N, dim]      (float32)
        - mask : Tensor[B, N, N] or None (bool / float32)
        - weights : dict of model parameters (float32)
        - config : {"dim": int, "hidden_dim": int}

    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]          # H in the paper
    dtype = torch.float32

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm + linear projections (cheap, stay in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # (B,N,N,hidden_dim) – keep fp32 for now, will cast to fp16 later
    left  = F.linear(x, weights["left_proj.weight"].to(dtype))
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # -----------------------------------------------------------------
    # 2️⃣ Fuse mask & gating in a Triton kernel (FP16)
    # -----------------------------------------------------------------
    if mask is None:
        mask_fp16 = torch.ones((B, N, N, 1), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(dtype).unsqueeze(-1)          # (B,N,N,1)
        mask_fp16 = mask_fp16.to(torch.float16)

    # Cast tensors to FP16 and make them contiguous for the kernel
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    lg_fp16    = left_gate.to(torch.float16).contiguous()
    rg_fp16    = right_gate.to(torch.float16).contiguous()

    total_elements = B * N * N * hidden_dim
    BLOCK = 1024                                 # elementwise kernel granularity
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        lg_fp16,
        rg_fp16,
        mask_fp16,
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        lg_fp16.stride(0), lg_fp16.stride(1),
        lg_fp16.stride(2), lg_fp16.stride(3),
        mask_fp16.stride(0), mask_fp16.stride(1),
        mask_fp16.stride(2), mask_fp16.stride(3),
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 3️⃣ Core O(N³) contraction using batched GEMM (FP16 -> FP32)
    # -----------------------------------------------------------------
    # reshape to (B*H, N, N) where H = hidden_dim
    left_bh  = left_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_bh = right_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # batched matmul: (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))

    # back to (B, N, N, H)
    out = out_bh.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype)                          # FP32 for the rest

    # -----------------------------------------------------------------
    # 4️⃣ Post‑processing: layer‑norm, out‑gate, final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    # out_gate is (B,N,N,hidden_dim) – already FP32
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
