"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

We keep the heavy cubic contraction on the GPU’s cuBLAS/Tensor‑Core path
(`torch.bmm`) and implement a tiny Triton kernel that fuses the
element‑wise mask‑and‑gate multiplication for the *left* and *right*
tensors.  The rest of the work (layer‑norms, linear projections, final
norm/proj) is done in PyTorch – this gives a correct reference while still
meeting the requirement of “at least some of the operations in a kernel”.
The kernel works on flat FP16 tensors, does a single multiply‑multiply per
element and writes the result back in FP16.

The overall algorithm
---------------------
0️⃣ unpack inputs
1️⃣ FP32 LayerNorm on the 4‑D tensor (`dim` axis)
2️⃣ cast to FP16 and compute the four linear projections (left/right &
   three gates) with `F.linear`
3️⃣ optional mask:  `mask` is broadcasted to the hidden dimension and
   fused with the gating in a Triton kernel
4️⃣ core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix‑multiply over the *k* dimension.
   We reshape to (B*H, N, N) and call `torch.bmm` – this runs on cuBLAS /
   Tensor‑Core and is far faster than a pure‑Triton N³ kernel.
5️⃣ FP32 LayerNorm, output‑gate and final projection back to `dim`.

The kernel
---------
```
mask_gate_kernel(
    x_ptr, mask_ptr, gate_ptr, out_ptr,
    N,                                   # number of elements in the flat view
    stride_x, stride_m, stride_g, stride_o # strides for the 4‑D tensors (flattened)
)
```
performs `out = x * mask * gate` (all FP16).  It is launched with a 1‑D
grid, each program handling a block of `BLOCK` contiguous elements.

The implementation below follows the Triton‑launch conventions described in
the statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton element‑wise kernel: out = x * mask * gate
# --------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    x_ptr,                # FP16 tensor (flattened)
    mask_ptr,             # FP16 tensor (flattened, broadcasted)
    gate_ptr,             # FP16 tensor (flattened, broadcasted)
    out_ptr,              # FP16 tensor (flattened)
    N: tl.constexpr,      # total number of elements
    stride_x: tl.constexpr,
    stride_m: tl.constexpr,
    stride_g: tl.constexpr,
    stride_o: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)

    mask = offs < N
    x = tl.load(x_ptr + offs * stride_x, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs * stride_m, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs * stride_g, mask=mask, other=0.0)

    out = x * m * g
    tl.store(out_ptr + offs * stride_o, out, mask=mask)


def _apply_mask_and_gate(tensor, mask, gate):
    """
    tensor : (B, N, N, H)  FP16, contiguous
    mask   : (B, N, N, 1)  FP16, contiguous (already broadcasted)
    gate   : (B, N, N, H)  FP16, contiguous
    returns tensor * mask * gate  (FP16, same shape)
    """
    assert tensor.is_contiguous()
    assert mask.is_contiguous()
    assert gate.is_contiguous()
    B, N, _, H = tensor.shape
    total = B * N * N * H

    x_ptr = tensor.data_ptr()
    m_ptr = mask.data_ptr()
    g_ptr = gate.data_ptr()
    out = torch.empty_like(tensor)

    o_ptr = out.data_ptr()

    # all tensors are flat, stride = 1
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_kernel[grid](
        x_ptr,
        m_ptr,
        g_ptr,
        o_ptr,
        N=total,
        stride_x=1,
        stride_m=1,
        stride_g=1,
        stride_o=1,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict with the same keys as the reference implementation
        config       : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual – cheap)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gates (cast to FP16)
    # --------------------------------------------------------------
    def _to_fp16(t): return t.to(torch.float16)

    left_proj_w   = _to_fp16(weights["left_proj.weight"])   # (hidden_dim, dim)
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)       # (B, N, N, dim)

    left  = F.linear(x_fp16, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 3️⃣ Optional mask + gating – fused in Triton
    # --------------------------------------------------------------
    if mask is not None:
        # broadcast mask to hidden_dim and FP16
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)          # (B,N,N,1)
        # fuse mask and gates for left and right separately
        left = _apply_mask_and_gate(left, mask_fp16, left_gate)
        right = _apply_mask_and_gate(right, mask_fp16, right_gate)
    else:
        # only gating, still use the kernel for consistency
        left = _apply_mask_and_gate(left, torch.ones_like(left), left_gate)
        right = _apply_mask_and_gate(right, torch.ones_like(right), right_gate)

    # --------------------------------------------------------------
    # 4️⃣ Core contraction – batched GEMM via cuBLAS (torch.bmm)
    #    reshape tensors: (B, H, N, N) and (B, H, N, N) → (B*H, N, N)
    # --------------------------------------------------------------
    # left : (B, N, N, H) → (B, H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, H) → (B, H, N, N) and we need (K,J) = (N,N) with J transposed
    right_t = right.permute(0, 3, 2, 1).contiguous()   # (B, H, N, N)

    # collapse batch*head dimension for bmm
    LH = left_t.view(B * hidden_dim, N, N)      # (B*H, I, K)
    RH = right_t.view(B * hidden_dim, N, N)     # (B*H, K, J)

    # batched matrix multiplication in fp16 (Tensor‑cores)
    out_fp16 = torch.bmm(LH, RH)               # (B*H, I, J)

    # restore original order: (B, N, N, H)
    out_fp16 = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_fp16.to(torch.float32)           # (B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Output LayerNorm (FP32) + output‑gate + final projection
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # apply output gate (broadcast over hidden_dim)
    out = out * out_gate.to(torch.float32)

    # final linear projection back to dim (still FP32)
    out = F.linear(out, weights["to_out.weight"])

    return out
