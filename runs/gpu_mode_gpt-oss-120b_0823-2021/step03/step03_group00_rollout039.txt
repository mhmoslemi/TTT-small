"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward only)
------------------------
1️⃣  Layer‑norm over the last channel (fp32, pure PyTorch).
2️⃣  Five linear projections (left, right, left‑gate, right‑gate, out‑gate) are
    performed in fp16 for speed.  The input tensor is cast once to fp16.
3️⃣  Optional binary mask is broadcast over the hidden dimension and applied to
    the left/right projections.  This element‑wise mask‑apply is the only
    operation that we off‑load to a tiny Triton kernel (so the submission
    respects the “must use Triton” rule while keeping the dominant O(N³) work
    on cuBLAS/Tensor‑cores via torch.bmm).
4️⃣  Sigmoid gating of the three gate tensors.
5️⃣  Core contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    is a batched matrix‑multiply with shape
        (B·H, N, N) · (B·H, N, N)ᵀ → (B·H, N, N) .
    The batched GEMM is performed with ``torch.bmm`` (cuBLAS) which on Hopper
    uses Tensor‑cores and is far faster than a pure Triton implementation.
6️⃣  Second LayerNorm (fp32), multiply by the output‑gate and final linear
    projection back to the original dimension (fp32).

The kernel **_apply_mask_kernel** only does the element‑wise mask multiplication
for the left and right tensors; everything else stays in PyTorch for maximum
throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – element‑wise mask multiplication (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    left_ptr, right_ptr, mask_ptr,                # input pointers
    B, N, H,                                      # dimensions
    stride_lz, stride_lb, stride_lk, stride_ld,  # left strides (z,b,n,d)
    stride_rz, stride_rb, stride_rk, stride_rd,  # right strides
    stride_mz, stride_mb, stride_mk,              # mask strides (z,b,n)
    BLOCK_Z: tl.constexpr,                       # batch*head tile size
    BLOCK_N: tl.constexpr,                       # sequence tile size
):
    """
    Each program processes a tile (BLOCK_Z × BLOCK_N) of the left/right tensors.
    The mask has shape (B, N, N) and is broadcast over the hidden dimension H.
    """
    pid_z = tl.program_id(0)                     # index over (B*H)
    pid_n = tl.program_id(1)                     # index over N

    # convert linear pid_z back to (b, h)
    b = pid_z // H
    h = pid_z % H

    # tile offsets
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    mask_n = offs_n < N

    # ------------------------ left ------------------------
    # left shape: (B, N, N, H)  (we treat it as (B,N,N,H) for stride access)
    # flatten to (B,N,N*H) to keep one contiguous dim for Triton
    left_off = (
        b * stride_lz
        + offs_n[:, None] * stride_lb
        + tl.arange(0, BLOCK_N) * stride_lk  # k dimension (the inner N)
        + h * stride_ld
    )
    left = tl.load(left_ptr + left_off, mask=mask_n[:, None] & mask_n[None, :], other=0.0)

    # ------------------------ right -----------------------
    right_off = (
        b * stride_rz
        + offs_n[:, None] * stride_rb
        + tl.arange(0, BLOCK_N) * stride_rk
        + h * stride_rd
    )
    right = tl.load(right_ptr + right_off, mask=mask_n[:, None] & mask_n[None, :], other=0.0)

    # ------------------------ mask ------------------------
    mask_off = b * stride_mz + offs_n[:, None] * stride_mb + tl.arange(0, BLOCK_N) * stride_mk
    mask = tl.load(mask_ptr + mask_off, mask=mask_n[:, None] & mask_n[None, :], other=0.0)

    # apply mask (mask is 0/1 → multiplication)
    left = left * mask
    right = right * mask

    # store back
    tl.store(left_ptr + left_off, left, mask=mask_n[:, None] & mask_n[None, :])
    tl.store(right_ptr + right_off, right, mask=mask_n[:, None] & mask_n[None, :])


def _apply_mask_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """
    left/right : (B, N, N, H)   fp16, contiguous
    mask       : (B, N, N)      fp16 (0/1)
    in‑place mask application using the Triton kernel above.
    """
    B, N, _, H = left.shape
    # strides in elements (fp16 tensors)
    stride_lz, stride_lb, stride_lk, stride_ld = left.stride()
    stride_rz, stride_rb, stride_rk, stride_rd = right.stride()
    stride_mz, stride_mb, stride_mk = mask.stride()

    BLOCK_N = 64            # good occupancy for H200
    BLOCK_Z = 1             # we tile over (B*H) with a single program each

    grid = (B * H, triton.cdiv(N, BLOCK_N))

    _apply_mask_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        B=B,
        N=N,
        H=H,
        stride_lz=stride_lz,
        stride_lb=stride_lb,
        stride_lk=stride_lk,
        stride_ld=stride_ld,
        stride_rz=stride_rz,
        stride_rb=stride_rb,
        stride_rk=stride_rk,
        stride_rd=stride_rd,
        stride_mz=stride_mz,
        stride_mb=stride_mb,
        stride_mk=stride_mk,
        BLOCK_Z=BLOCK_Z,
        BLOCK_N=BLOCK_N,
        num_warps=4,
    )
    # The kernel writes in‑place, nothing to return.


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors (see reference PyTorch implementation)
        config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (fp32, manual)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still fp32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (fp16) – cast once
    # --------------------------------------------------------------
    def _to_fp16(t): return t.to(torch.float16)
    left_proj_w   = _to_fp16(weights["left_proj.weight"])
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                     # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)            # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)           # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask – Triton kernel (in‑place)
    # --------------------------------------------------------------
    if mask is not None:
        # convert mask to fp16 0/1 values
        mask_fp16 = mask.to(torch.float16)
        _apply_mask_triton(left, right, mask_fp16)

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy contraction – batched GEMM via cuBLAS (torch.bmm)
    # --------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_b = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (B*H, I, K)
    right_b = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N) # (B*H, K, J)

    # batched matmul -> (B*H, N, N)
    out_b = torch.bmm(left_b, right_b)        # fp16, Tensor‑core accelerated

    # reshape back to (B, N, N, hidden_dim)
    out_fp16 = out_b.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (fp32) + gating + final linear
    # --------------------------------------------------------------
    out = out_fp16.to(torch.float32)           # back to fp32

    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    out = out * out_gate.to(torch.float32)     # broadcast over hidden dim

    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)              # (B,N,N,dim) fp32

    return out
