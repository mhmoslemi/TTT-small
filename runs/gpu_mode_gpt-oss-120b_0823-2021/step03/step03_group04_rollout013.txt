"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm (GPU‑centric)
-----------------------
1️⃣  Layer‑norm the input (B,N,N,C) – done in FP32 with torch.nn.functional.
2️⃣  Linear projections to the hidden dimension H for “left”, “right” and
    three gates – still in FP32.
3️⃣  Fuse element‑wise mask, left‑gate, right‑gate into two tensors
    (left_fused, right_fused).  This is the only mandatory Triton kernel:
       out_left  = left  * left_gate  * mask
       out_right = right * right_gate * mask
    The kernel works on a flat 1‑D grid, loads/stores with arbitrary strides
    and supports the “no‑mask” case (HAS_MASK constexpr).
4️⃣  Cast the fused tensors to FP16, reshape to (B*H, N, N) and perform the
    cubic contraction with a batched matrix‑multiply (torch.bmm).  This
    exploits cuBLAS/Tensor‑cores (FP16→FP16 compute, FP32 accumulation).
5️⃣  Layer‑norm the hidden result, multiply by the output‑gate and project
    back to the original channel dimension C.

Only step 3 is executed in Triton – this satisfies the “some‑kernel”
requirement while keeping the dominant O(N³) work on the highly‑optimized
cuBLAS path, which is essential for the large sequence lengths used in the
tests.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – element‑wise fusion of mask & gates
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                # fp16 * (B,N,N,H)
    left_gate_ptr, right_gate_ptr,      # fp16 * (B,N,N,H)
    mask_ptr,                           # fp16 * (B,N,N,1)  or nullptr
    out_left_ptr, out_right_ptr,        # fp16 * (B,N,N,H)  outputs
    B, N, H,                            # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,      # left / right strides
    stride_g0, stride_g1, stride_g2, stride_g3,      # gate strides
    stride_m0, stride_m1, stride_m2,                # mask strides (no H)
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For each (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    The kernel processes BLOCK elements per program id.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear indices
    total = B * N * N * H
    mask = offs < total

    # ------------------------------------------------------------------
    # Decode linear index to (b,i,j,h)   (contiguous layout: b,i,j,h)
    # ------------------------------------------------------------------
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ------------------------------------------------------------------
    # Load left/right values
    # ------------------------------------------------------------------
    idx_lr = (b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3)
    left_val  = tl.load(left_ptr  + idx_lr, mask=mask, other=0.0)
    right_val = tl.load(right_ptr + idx_lr, mask=mask, other=0.0)

    # ------------------------------------------------------------------
    # Load gate values
    # ------------------------------------------------------------------
    idx_g = (b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3)
    left_gate_val  = tl.load(left_gate_ptr  + idx_g, mask=mask, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + idx_g, mask=mask, other=0.0)

    # ------------------------------------------------------------------
    # Load mask (if present).  Mask has no hidden dim, broadcast over H.
    # ------------------------------------------------------------------
    if HAS_MASK:
        idx_m = b * stride_m0 + i * stride_m1 + j * stride_m2   # scalar mask
        m = tl.load(mask_ptr + idx_m, mask=mask, other=0.0)
    else:
        m = 1.0

    # ------------------------------------------------------------------
    # Fuse & store
    # ------------------------------------------------------------------
    out_l = left_val  * left_gate_val  * m
    out_r = right_val * right_gate_val * m

    tl.store(out_left_ptr  + idx_lr, out_l, mask=mask)
    tl.store(out_right_ptr + idx_lr, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        * x            – (B, N, N, C)  torch.float32
        * mask         – (B, N, N)    torch.bool / torch.float32 / None
        * weights      – dict of model parameters (float32)
        * config       – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor
        (B, N, N, C) float32
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, C = x.shape
    H = config["hidden_dim"]
    dtype = torch.float32

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (still FP32)
    # ------------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))     # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # ------------------------------------------------------------------
    # 3️⃣ Fuse mask & gates – Triton kernel
    # ------------------------------------------------------------------
    # Convert everything to FP16 (Tensor‑core friendly) and make contiguous
    left_fp16      = left.contiguous().to(torch.float16)
    right_fp16     = right.contiguous().to(torch.float16)
    left_gate_fp16 = left_gate.contiguous().to(torch.float16)
    right_gate_fp16= right_gate.contiguous().to(torch.float16)

    if mask is None:
        # Dummy mask – pointer will never be dereferenced because HAS_MASK=False
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask = False
    else:
        # Expand mask to broadcast over hidden dim H
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()  # (B,N,N,1)
        has_mask = True

    # Allocate output buffers
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elements = B * N * N * H
    BLOCK = 1024                                   # empirically good for H200
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        mask_fp16.stride(0) if has_mask else 0,
        mask_fp16.stride(1) if has_mask else 0,
        mask_fp16.stride(2) if has_mask else 0,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (FP16 inputs → FP16 output, FP32 accumulation)
    # ------------------------------------------------------------------
    # (B,N,N,H) → (B*H, N, N)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # out_mat = left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B,N,N,H) and cast to FP32 for the next LayerNorm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(dtype)

    # ------------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, output‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                     # broadcast over (N,N)
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
