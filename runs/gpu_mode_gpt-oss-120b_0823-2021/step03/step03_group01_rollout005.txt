"""
custom_kernel – “outgoing” TriMul forward pass (AlphaFold‑3 style).

The cubic contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a fused batched GEMM Triton kernel that works on
(fp16) left/right and accumulates in fp32 (Tensor‑Core friendly).
All other ops (LayerNorms, linear projections, gating, final Linear) stay
in PyTorch – this gives the best trade‑off between code size, correctness,
and speed on Hopper (sm_90).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (I×K)·(K×J) → (I×J)   (fp16 → fp16, fp32 acc)
# ----------------------------------------------------------------------
@triton.jit
def _batch_matmul_kernel(
    a_ptr,                # left  matrix pointer   (Z, I, K)
    b_ptr,                # right matrix pointer   (Z, K, J)
    c_ptr,                # output matrix pointer  (Z, I, J)
    M, N, K,              # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)          # batch*head dimension
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (Tensor‑Core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over K
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile  (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # accumulate in fp32
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # write back (cast down to fp16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM with shape
        left  : (B, H, I, K)   fp16
        right : (B, H, K, J)   fp16
    Returns
        out   : (B, H, I, J)   fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # output allocation
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse leading dims for the kernel
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides in element units (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Heuristic tiling that fits Tensor‑Cores on Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                                          # batch*head
        triton.cdiv(I, BLOCK_M),                    # rows
        triton.cdiv(J, BLOCK_N),                    # cols
    )
    _batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default on SM90
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask         : torch.Tensor [B, N, N] (float32/bool) or None
        weights      : dict of model parameters (all float32)
        config       : dict with keys
                       - "dim"          : input channel dimension
                       - "hidden_dim"   : projection dimension

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. Input LayerNorm  (FP32 for numeric stability)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b          # [B,N,N,dim]  fp32

    # --------------------------------------------------------------
    # 2. Linear projections – cast to fp16 for the heavy contraction
    # --------------------------------------------------------------
    def _to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w = _to_fp16(weights["left_proj.weight"])
    right_proj_w = _to_fp16(weights["right_proj.weight"])
    left_gate_w = _to_fp16(weights["left_gate.weight"])
    right_gate_w = _to_fp16(weights["right_gate.weight"])
    out_gate_w = _to_fp16(weights["out_gate.weight"])

    # fp16 copy of the normalized tensor
    x_fp16 = x_norm.to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)        # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None:
        # mask may be bool or float; convert to fp16 and add trailing dim
        mask_fp16 = mask.to(torch.float16, non_blocking=True).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) – stay in fp16
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton batched GEMM (the N³ contraction)
    #    left : (B, N, N, hidden)   →  (B, H, I, K)  where H=hidden_dim
    #    right: (B, N, N, hidden)   →  (B, H, K, J)  with K=J=N
    # --------------------------------------------------------------
    # 5a. reshape / permute so that the hidden dimension becomes the “head”
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    # note: einsum uses right[b, j, k, d]; we need (B,H,K,J)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # 5b. call the Triton kernel
    out_fp16 = _triton_batched_gemm(left_perm, right_perm)   # (B,H,I,J)

    # 5c. restore original layout and cast back to fp32 for the rest
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,dim)

    # --------------------------------------------------------------
    # 6. Output LayerNorm  (still fp32)
    # --------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = out * to_out_norm_w + to_out_norm_b

    # --------------------------------------------------------------
    # 7. Final gating + Linear projection
    # --------------------------------------------------------------
    # out_gate is still fp16; cast to fp32 for element‑wise product
    out = out * out_gate.to(torch.float32)
    to_out_w = weights["to_out.weight"]          # (dim, dim) float32
    out = F.linear(out, to_out_w)                # (B,N,N,dim)

    return out
