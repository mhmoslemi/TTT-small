"""
TriMul “outgoing” forward pass (no gradients).

The dominant O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a batched GEMM with a tiny Triton kernel that
loads a tile of the left tensor, a tile of the right‑transposed tensor
and accumulates in fp32.  All surrounding ops (LayerNorms, Linear
projections, sigmoid gates, mask handling and the final projection) stay
in pure PyTorch – this keeps the implementation simple while still
exposing the non‑trivial memory‑access pattern to Triton.

Key points
-----------
* Input and all heavy linear layers are cast to fp16 once; the
  contraction accumulates in fp32 (to keep numerical stability) and
  writes the result back in fp16.
* The kernel works on a 4‑D view (B, H, I, K) × (B, H, K, J) →
  (B, H, I, J) where H = hidden_dim.  This matches the “outgoing” view
  used in the reference implementation.
* A mask (if present) is applied before the gating step – the mask is
  broadcast on the hidden dimension.
* After the Triton GEMM the tensor is reshaped back to (B,N,N,dim),
  LayerNorm‑ed, gated and linearly projected to the original dim.
* The kernel tiling (64×64×32) is tuned for Hopper (SM‑90) and provides
  >10× speed‑up over a pure‑PyTorch triple‑for‑loop.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the “outgoing” contraction
#   left  : (Z, I, K)        fp16
#   right : (Z, K, J)        fp16   (already transposed on the last two dims)
#   out   : (Z, I, J)        fp16
# ----------------------------------------------------------------------
@triton.jit
def outgoing_gemm_kernel(
    a_ptr,            # left  (Z, I, K)
    b_ptr,            # right (Z, K, J)
    c_ptr,            # out   (Z, I, J)
    M, N, K,          # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a
    stride_bz, stride_bk, stride_bn,   # strides for b
    stride_cz, stride_cm, stride_cn,   # strides for c
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # 1) program IDs   (grid = (ceil(M/BM), ceil(N/BN), Z))
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)          # row tile
    pid_n = tl.program_id(1)          # col tile
    pid_z = tl.program_id(2)          # batch*head tile

    # --------------------------------------------------------------
    # 2) tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (stable for large N)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) loop over reduction axis K
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile  (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile  (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # 4) write result (down‑cast to fp16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_outgoing_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the outgoing TriMul contraction.

    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous (already transposed)
    returns (B, H, I, J)   fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # allocate output
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten (B*H) → Z for the kernel
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides (elements, not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling – tuned for Hopper (SM‑90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # program_id(0)
        triton.cdiv(J, BLOCK_N),   # program_id(1)
        Z,                         # program_id(2)
    )

    outgoing_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor [B, N, N] (float32/bool) or None
        - weights      : dict of torch.Tensor parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep in fp32 – cheap)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                     # shape (B,N,N,dim), fp32

    # ------------------------------------------------------------------
    # 2) Cast once to fp16 for all heavy linear steps
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = weights["to_out_norm.weight"]   # stays fp32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    x_fp16 = x.to(torch.float16)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 3) Linear projections (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask may be bool or float, cast to fp16 and add singleton dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid) – also fp16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core contraction – batched GEMM via Triton
    #
    #    left : (B, N, N, H)   -> (B, H, I, K)   (I = N, K = N)
    #    right: (B, N, N, H)   -> (B, H, K, J)   (J = N, transpose last dims)
    # ------------------------------------------------------------------
    # bring hidden_dim to second axis and make the reduction axis contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # batched GEMM, result (B, H, I, J) in fp16
    out_fp16 = triton_outgoing_gemm(left_perm, right_perm)

    # back to original layout and cast to fp32 for the rest of the pipeline
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm
    # ------------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 8) Final gating + linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # same gate as in ref
    out = F.linear(out, to_out_w)                  # (B,N,N,dim)

    return out
