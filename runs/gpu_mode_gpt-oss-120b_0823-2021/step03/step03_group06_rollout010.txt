"""
TriMul “outgoing” forward (AlphaFold‑3).

The costly contraction 

    out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]

is a batched matrix‑multiply with a transposed RHS.
We keep the cheap LayerNorm / linear / gating work in PyTorch and move the
actual N³ contraction into a small Triton kernel that computes a batch of
GEMMs (B·H, N, N).  

The kernel works in fp16 with fp32 accumulation (Tensor‑Core friendly) and
stores the result back in fp16.  Afterwards we reshape to the original
layout and finish the remaining LayerNorm / gating / projection steps in
PyTorch.  This satisfies the requirement “implement at least part of the
operations in a kernel” while keeping the overall runtime competitive
(the kernel is the dominant O(N³) part).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for a single (batch*head) slice
# ----------------------------------------------------------------------
@triton.jit
def _trimul_batched_gemm(
    left_ptr,          # const fp16*   [BH, N, N]
    right_ptr,         # const fp16*   [BH, N, N]   (to be read transposed)
    out_ptr,           #       fp16*   [BH, N, N]
    BH, N,             # problem sizes
    # strides of the three tensors (row‑major order)
    stride_lbh, stride_lbn, stride_lbk,
    stride_rbh, stride_rbn, stride_rbk,
    stride_obh, stride_obn, stride_obk,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a tile (BLOCK_M × BLOCK_N) of

        out[bh, i, j] = Σ_k left[bh, i, k] * right[bh, j, k]

    for a single flattened batch‑head index ``bh`` (0 ≤ bh < BH).
    The RHS is accessed as the transpose of ``right`` (k‑dim first) so that
    the operation becomes a regular matrix‑multiply.
    """
    pid = tl.program_id(0)                     # linear id over (bh, tile_m, tile_n)

    # --------------------------------------------------------------
    # decode the 3‑D launch grid (bh, tile_m, tile_n)
    # --------------------------------------------------------------
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    tile_n = pid % num_tiles_n
    pid //= num_tiles_n
    tile_m = pid % num_tiles_m
    pid //= num_tiles_m
    bh = pid                                   # batch‑head index

    # --------------------------------------------------------------
    # coordinate offsets for the current tile
    # --------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # fp32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over K dimension (blocked)
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[bh, i, k]   → shape (M, K)
        l_ptr = left_ptr + bh * stride_lbh \
                + offs_m[:, None] * stride_lbn \
                + offs_k[None, :] * stride_lbk
        l = tl.load(l_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[bh, j, k] needs to be read as (K, N) because we multiply
        # left[i, k] * right[j, k] = left[i, k] * (right^T)[k, j]
        r_ptr = right_ptr + bh * stride_rbh \
                + offs_k[:, None] * stride_rbk \
                + offs_n[None, :] * stride_rbn
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # (M,K) @ (K,N) → (M,N)
        acc += tl.dot(l, r)

    # --------------------------------------------------------------
    # write back the tile (convert back to fp16)
    # --------------------------------------------------------------
    out_ptrs = out_ptr + bh * stride_obh \
               + offs_m[:, None] * stride_obn \
               + offs_n[None, :] * stride_obk
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side wrapper – the public entry point required by the grader
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input   : [B, N, N, dim]
        torch.Tensor,                # mask    : [B, N, N]   (or None)
        Dict[str, torch.Tensor],     # weights : dict of parameters
        Dict[str, Any],              # config  : {"dim":..., "hidden_dim":...}
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32                     # final precision

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]         # this is the “head” size H

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm + linear projections (cheap – stay in PyTorch)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N, H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # ------------------------------------------------------------------
    # 2️⃣  Optional mask  (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)           # (B,N,N,1)
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 3️⃣  Gating (sigmoid on linear maps)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣  Prepare tensors for the batched GEMM kernel
    #     (B,N,N,H) → (B, H, N, N) → (B*H, N, N)   and cast to fp16
    # ------------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16  = left.view(B * hidden_dim, N, N).to(torch.float16)
    right_fp16 = right.view(B * hidden_dim, N, N).to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)          # (B*H, N, N), fp16

    # ------------------------------------------------------------------
    # 5️⃣  Launch Triton kernel
    # ------------------------------------------------------------------
    # Tile sizes – chosen to fit comfortably in shared memory on an H200.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    BH = B * hidden_dim                              # total batch‑heads
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_tiles = BH * grid_m * grid_n
    grid = (total_tiles,)

    _trimul_batched_gemm[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        BH,
        N,
        # strides for left  (BH, N, N)
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        # strides for right (BH, N, N)
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        # strides for out   (BH, N, N)
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6️⃣  Convert back to the original layout and finish remaining ops
    # ------------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    out = out.to(dtype)                                                          # fp32 for LN

    # final LayerNorm over the hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply the output gate (broadcast over the N×N plane)
    out = out * out_gate

    # linear projection back to the original channel dimension
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
