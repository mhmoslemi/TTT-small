"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) → (B,N,N,C)  (FP32)
2. Linear projections to the hidden dimension H:
      left  = x @ W_left       (B,N,N,H)
      right = x @ W_right
      left_gate  = σ(x @ W_lgate)
      right_gate = σ(x @ W_rgate)
      out_gate   = σ(x @ W_outgate)     (B,N,N,H)
3. Fuse mask (if present) and the two gates together with a tiny Triton
   kernel.  The kernel works on a 1‑D grid and writes two tensors
   `left_fused` and `right_fused` in fp16.
4. Batched GEMM (cuBLAS / Tensor‑cores):
      out_h = left_fused @ right_fusedᵀ   → (B*H,N,N)
   reshaped back to (B,N,N,H) and cast to FP32.
5. Hidden‑dim layer‑norm, multiply by `out_gate` and a final linear
   projection back to C.  The result is the same shape as the reference
   implementation: (B,N,N,C).

Only step 3 is performed in Triton – the rest stays in PyTorch (fast cuBLAS
for the O(N³) contraction).  All heavy arithmetic is in FP16 with FP32
accumulation, while the final output is FP32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – fuse mask and element‑wise gates
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # fp16 * (B,N,N,H)
    right_ptr,              # fp16 *
    left_gate_ptr,          # fp16 *
    right_gate_ptr,         # fp16 *
    mask_ptr,               # fp16 * (B,N,N,1)  or nullptr
    out_left_ptr,           # fp16 *
    out_right_ptr,          # fp16 *
    B, N, H,                # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    The kernel is launched with a 1‑D grid sized
        ceil(B*N*N*H / BLOCK)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # ------- decode linear index → (b,i,j,h) -------
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ------- load tensors -------
    idx = (b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3)
    left  = tl.load(left_ptr + idx,  mask=mask, other=0.0)
    right = tl.load(right_ptr + idx, mask=mask, other=0.0)

    idx_gate = (b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + h * stride_lg3)
    lgate = tl.load(left_gate_ptr + idx_gate,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + idx_gate, mask=mask, other=0.0)

    if HAS_MASK:
        # mask has no hidden dimension, broadcast over H
        idx_mask = (b * stride_m0 + i * stride_m1 + j * stride_m2)
        m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)
    else:
        m = 1.0

    out_l = left  * lgate * m
    out_r = right * rgate * m

    tl.store(out_left_ptr  + idx, out_l, mask=mask)
    tl.store(out_right_ptr + idx, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host wrapper
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : (x, mask, weights, config)
        x       – (B, N, N, C)   torch.float32
        mask    – (B, N, N)       torch.bool/float32/None
        weights – dict of FP32 model parameters
        config  – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor – (B, N, N, C)  float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype_fp32 = torch.float32
    dtype_fp16 = torch.float16

    B, N, _, C = x.shape
    H = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → keep high‑precision for LN)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype_fp32),
        bias=weights["norm.bias"].to(dtype_fp32),
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (FP32 → FP16 for the heavy work)
    # --------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype_fp32))   # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype_fp32))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype_fp32)))
    right_gate= torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype_fp32)))
    out_gate  = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype_fp32)))  # (B,N,N,H)

    # cast to fp16 (contiguous layout)
    left_fp16       = left.to(dtype_fp16).contiguous()
    right_fp16      = right.to(dtype_fp16).contiguous()
    left_gate_fp16  = left_gate.to(dtype_fp16).contiguous()
    right_gate_fp16 = right_gate.to(dtype_fp16).contiguous()
    out_gate_fp16   = out_gate.to(dtype_fp16).contiguous()   # stays on the CPU side for later

    # --------------------------------------------------------------
    # 3️⃣ Fuse mask + gates (Triton)
    # --------------------------------------------------------------
    if mask is None:
        mask_fp16 = torch.empty(0, device=device, dtype=dtype_fp16)
        has_mask = False
    else:
        mask_fp16 = mask.to(dtype_fp16).unsqueeze(-1).contiguous()   # (B,N,N,1)
        has_mask = True

    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elems = B * N * N * H
    BLOCK = 1024                                   # fits well in H200 registers
    grid = (triton.cdiv(total_elems, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1), left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2), mask_fp16.stride(3) if has_mask else 0,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (FP16, FP32 accumulation)
    # --------------------------------------------------------------
    # (B,N,N,H) → (B*H,N,N)   by moving H before N,N
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)

    # out_h = left @ rightᵀ   (cuBLAS/Tensor‑core fast)
    out_h = torch.bmm(left_mat, right_mat.transpose(1, 2))          # (B*H,N,N)

    # back to (B,N,N,H)
    out = out_h.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)
    out = out.to(dtype_fp32)                                         # FP32 for LN

    # --------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, out‑gate and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype_fp32),
        bias=weights["to_out_norm.bias"].to(dtype_fp32),
    )
    # out_gate is still FP16 → cast to FP32 for the multiplication
    out = out * out_gate_fp16.to(dtype_fp32)   # broadcast over (N,N)

    # final linear projection back to C (FP32)
    out = F.linear(out, weights["to_out.weight"].to(dtype_fp32))   # (B,N,N,C)

    return out
