"""
custom_kernel – “outgoing” TriMul forward pass (AlphaFold‑3)

The heavy O(N³) contraction  

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

is performed with a small, blocked GEMM Triton kernel that works on the
re‑ordered tensors (B, H, N, N).  All cheap operations (LayerNorm, the four
linear projections, the three sigmoid gates, the final LayerNorm + linear) are
kept in PyTorch because they are memory‑bound and much faster there than in
custom kernels.

The kernel tiles the M‑ (i) and N‑ (j) dimensions with BLOCK_M×BLOCK_N
and walks the K‑dimension (k) with BLOCK_K.  Each tile is accumulated in FP32
and written back as FP16; the surrounding Python code casts back to FP32 for the
final LayerNorm and output projection.

The implementation is fully generic (supports B∈{1,2}, N up to 1024, any
dim / hidden_dim that fit in memory, optional mask, and both normal / Cauchy
input distributions).  It runs on Hopper (sm_90) and uses only Triton 3.3.1
primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: blocked batched GEMM for the O(N³) contraction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,          # const fp16*  [B, H, N, N]   (i‑k)
    right_ptr,         # const fp16*  [B, H, N, N]   (j‑k) – we read it transposed
    out_ptr,           #       fp16*  [B, H, N, N]   (i‑j)
    B, N, H,           # int metadata
    # strides (row‑major) of the three tensors
    stride_lb, stride_lh, stride_li, stride_lk,   # left  (B,H,N,N)
    stride_rb, stride_rh, stride_rj, stride_rk,   # right (B,H,N,N)
    stride_ob, stride_oh, stride_oi, stride_oj,   # out   (B,H,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute one (BLOCK_M × BLOCK_N) tile of the 4‑D result

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    A single program id corresponds to a (b, h, tile_m, tile_n) block.
    """
    pid = tl.program_id(0)

    # ---------------------------------------------------------------
    # decode pid  ->  b, h, tile_m, tile_n
    # ---------------------------------------------------------------
    num_n = tl.cdiv(N, BLOCK_N)          # tiles along the j‑axis
    num_m = tl.cdiv(N, BLOCK_M)          # tiles along the i‑axis

    tile_n = pid % num_n
    pid //= num_n
    tile_m = pid % num_m
    pid //= num_m
    h = pid % H
    b = pid // H

    # ---------------------------------------------------------------
    # tile offsets and masks
    # ---------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32 (more accurate than pure FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # K‑loop (blocked)
    # ---------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, h, i, k]   -> shape (M, K)
        l_ptr = (left_ptr +
                 b * stride_lb +
                 h * stride_lh +
                 offs_m[:, None] * stride_li +
                 offs_k[None, :] * stride_lk)
        l = tl.load(l_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[b, h, j, k] is read as transposed so we get (K, N)
        #   original layout: (b, h, j, k)
        #   after transposition we want (k, j)
        r_ptr = (right_ptr +
                 b * stride_rb +
                 h * stride_rh +
                 offs_k[:, None] * stride_rk +          # k as row
                 offs_n[None, :] * stride_rj)           # j as col
        r_t = tl.load(r_ptr,
                      mask=mask_k[:, None] & mask_n[None, :],
                      other=0.0).to(tl.float32)

        # (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(l, r_t)

    # ---------------------------------------------------------------
    # write back result (cast to fp16)
    # ---------------------------------------------------------------
    out_ptrs = (out_ptr +
                b * stride_ob +
                h * stride_oh +
                offs_m[:, None] * stride_oi +
                offs_n[None, :] * stride_oj)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host‑side wrapper that stitches everything together
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input   [B, N, N, dim]
        torch.Tensor,                     # mask    [B, N, N]   (or None)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any]                    # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim] and dtype torch.float32
    """
    x, mask, weights, config = data
    device = x.device
    # keep internal computations in float32 (more stable) and cast to fp16 only
    # for the heavy GEMM
    FP32 = torch.float32
    FP16 = torch.float16

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]               # == H

    # ------------------------------------------------------------------
    # 1. Cheap operations in PyTorch (LayerNorm, projections, gates)
    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(FP32),
        bias=weights["norm.bias"].to(FP32),
    )

    # Linear projections (bias=False)
    left  = F.linear(x, weights["left_proj.weight"].to(FP32))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(FP32))

    # optional pair‑wise mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1)                 # (B,N,N,1)
        left  = left * mask_
        right = right * mask_

    # gates (sigmoid after a linear)
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(FP32)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(FP32)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(FP32)))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 2. Rearrange to [B, H, N, N] and cast to fp16 for the kernel
    # ------------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16  = left.to(FP16)
    right_fp16 = right.to(FP16)

    out_fp16 = torch.empty_like(left_fp16)          # (B,H,N,N), fp16

    # ------------------------------------------------------------------
    # 3. Launch the Triton kernel (blocked GEMM)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    # one program per (b, h, tile_m, tile_n)
    total_programs = B * hidden_dim * grid_m * grid_n
    grid = (total_programs,)

    _trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        # strides for left  (B,H,N,N)
        left_fp16.stride(0),   # B
        left_fp16.stride(1),   # H
        left_fp16.stride(2),   # i  (row)
        left_fp16.stride(3),   # k  (col)
        # strides for right (we read it transposed)
        right_fp16.stride(0),  # B
        right_fp16.stride(1),  # H
        right_fp16.stride(2),  # j  (col after transpose)
        right_fp16.stride(3),  # k  (row after transpose)
        # strides for out
        out_fp16.stride(0),    # B
        out_fp16.stride(1),    # H
        out_fp16.stride(2),    # i
        out_fp16.stride(3),    # j
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 4. Post‑process in PyTorch (LayerNorm, gating, final linear)
    # ------------------------------------------------------------------
    # reshape back to (B,N,N,H) and cast to FP32 for the rest of the pipeline
    out = out_fp16.to(FP32).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # final LayerNorm over hidden_dim
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(FP32),
        bias=weights["to_out_norm.bias"].to(FP32),
    )

    # apply output gate (broadcast on the pair‑wise axes)
    out = out * out_gate

    # final linear back to original dim
    out = F.linear(out, weights["to_out.weight"].to(FP32))

    return out
