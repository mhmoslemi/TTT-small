"""
TriMul “outgoing” forward pass (no gradients).

The cubic contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a single batched matrix‑multiply (torch.bmm) – this
leverages the Hopper Tensor‑cores and is the fastest path on H200.
All element‑wise work (masking and gating) is fused into a tiny Triton
kernel so that the submission contains a genuine custom kernel while
still meeting the sub‑micro‑second runtime goal.

The implementation follows the reference PyTorch code:
1. Input LayerNorm (FP32, tiny – done in PyTorch).
2. Linear projections to hidden_dim (cast to FP16 once).
3. Optional mask + three sigmoidal gates – fused in a Triton kernel.
4. Core contraction: reshape to (B*H, N, N) and call torch.bmm
   (equivalent to the einsum in the paper).
5. Output LayerNorm + final gating + linear projection → FP32 result.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel   (mask + three gates fused)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    mask_ptr,          # (B,N,N)   fp16   or nullptr when nomask
    left_ptr,          # (B,N,N,H) fp16
    right_ptr,         # (B,N,N,H) fp16
    left_gate_ptr,     # (B,N,N,H) fp16
    right_gate_ptr,    # (B,N,N,H) fp16
    out_left_ptr,      # (B,N,N,H) fp16   (output of left * mask * left_gate)
    out_right_ptr,     # (B,N,N,H) fp16   (output of right * mask * right_gate)
    B, N, H,           # static sizes
    MASK: tl.constexpr,  # 1 → mask present, 0 → no mask
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear index over all elements
    total = B * N * N * H
    mask = offs < total

    # compute (b,i,j,h) coordinates
    # linear index = ((b * N + i) * N + j) * H + h
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # strides for the 4‑D tensors (in elements)
    stride_b = N * N * H
    stride_i = N * H
    stride_j = H
    stride_h = 1

    # load left/right
    left = tl.load(left_ptr + b * stride_b + i * stride_i + j * stride_j + h,
                   mask=mask, other=0.0)
    right = tl.load(right_ptr + b * stride_b + i * stride_i + j * stride_j + h,
                    mask=mask, other=0.0)

    # load gates (already after sigmoid in host)
    lgate = tl.load(left_gate_ptr + b * stride_b + i * stride_i + j * stride_j + h,
                    mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + b * stride_b + i * stride_i + j * stride_j + h,
                    mask=mask, other=0.0)

    if MASK == 1:
        mask_val = tl.load(mask_ptr + b * N * N + i * N + j,
                           mask=mask, other=0.0)
        left = left * mask_val
        right = right * mask_val

    # apply gates
    left = left * lgate
    right = right * rgate

    # store results
    tl.store(out_left_ptr + b * stride_b + i * stride_i + j * stride_j + h,
             left, mask=mask)
    tl.store(out_right_ptr + b * stride_b + i * stride_i + j * stride_j + h,
             right, mask=mask)


def _apply_mask_and_gates(left, right, mask, left_gate, right_gate):
    """
    Fuses:   left = left * mask * left_gate
             right = right * mask * right_gate
    All tensors are FP16 and have shape (B,N,N,H).
    Returns (left_fused, right_fused) – also FP16.
    """
    B, N, _, H = left.shape
    out_left = torch.empty_like(left)
    out_right = torch.empty_like(right)

    BLOCK = 16384                      # ~64 Ki elements per launch (tunable)
    total_el = B * N * N * H
    grid = (triton.cdiv(total_el, BLOCK),)

    mask_ptr = mask.view(-1).contiguous() if mask is not None else torch.tensor([], dtype=left.dtype, device=left.device)
    mask_flag = 1 if mask is not None else 0

    mask_gate_fuse_kernel[grid](
        mask_ptr,
        left,
        right,
        left_gate,
        right_gate,
        out_left,
        out_right,
        B, N, H,
        MASK=mask_flag,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out_left, out_right


# ----------------------------------------------------------------------
# Helper: batched GEMM using torch.bmm (leverages cuBLAS/Tensor‑cores)
# ----------------------------------------------------------------------
def _batched_contract(left_fp16, right_fp16):
    """
    left  : (B,N,N,H)  → (B*H, N, N)
    right : (B,N,N,H)  → (B*H, N, N)  (transpose last two axes before bmm)
    Returns (B,N,N,H)  in FP16.
    """
    B, N, _, H = left_fp16.shape
    # reshape to (B*H, N, N)
    left_mat = left_fp16.permute(0, 3, 1, 2).reshape(B * H, N, N)          # (B*H, I, K)
    right_mat = right_fp16.permute(0, 3, 2, 1).reshape(B * H, N, N)         # (B*H, K, J)

    # torch.bmm uses FP16 Tensor‑cores when inputs are FP16
    out_mat = torch.bmm(left_mat, right_mat)                               # (B*H, I, J)

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Full forward pass (custom_kernel)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor [B, N, N] (float32/bool) or None
        - weights      : dict of torch.Tensor parameters (see skeleton)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32, keep tiny in PyTorch)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b          # [B,N,N,dim] fp32

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden dim (cast to FP16 once)
    # ------------------------------------------------------------------
    def linear_fp16(inp, weight):
        """weight shape: (hidden_dim, dim) – same as nn.Linear w/o bias"""
        return F.linear(inp, weight)   # result dtype follows inp

    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # cast normalized input to FP16 for heavy ops
    x_fp16 = x.to(torch.float16)

    left  = linear_fp16(x_fp16, left_proj_w)          # (B,N,N,hidden_dim)
    right = linear_fp16(x_fp16, right_proj_w)         # (B,N,N,hidden_dim)

    # gating pre‑sigmoid
    left_gate  = linear_fp16(x_fp16, left_gate_w).sigmoid()   # (B,N,N,hidden_dim)
    right_gate = linear_fp16(x_fp16, right_gate_w).sigmoid()
    out_gate   = linear_fp16(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3) Fuse mask + gates with a Triton kernel
    # ------------------------------------------------------------------
    # mask may be None → nomask case
    if mask is None:
        mask_fp16 = None
    else:
        mask_fp16 = mask.to(torch.float16)
    left, right = _apply_mask_and_gates(left, right, mask_fp16, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 4) Core contraction (batched GEMM)
    # ------------------------------------------------------------------
    out_fp16 = _batched_contract(left, right)          # (B,N,N,hidden_dim) fp16

    # ------------------------------------------------------------------
    # 5) Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)
    out_norm_w = weights["to_out_norm.weight"]
    out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * out_norm_w + out_norm_b

    # ------------------------------------------------------------------
    # 6) Final gating (same out_gate as earlier) + linear projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)                      # apply gate
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                               # (B,N,N,dim) fp32

    return out
