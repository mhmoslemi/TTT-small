"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the (B,N,N,C) input.
2️⃣  Linear projections → (B,N,N,H) for *left*, *right* and three gates.
3️⃣  Fuse element‑wise work (mask, left‑gate, right‑gate) with a tiny Triton
    kernel.  The kernel writes two fused tensors `left_fused` and
    `right_fused` in fp16.
4️⃣  Reshape to (B*H, N, N) and perform the O(N³) contraction with a
    batched matrix‑multiply (`torch.bmm`).  This uses cuBLAS/Tensor‑cores
    (fp16 × fp16 → fp16, accumulation in fp32).
5️⃣  Hidden‑dim layer‑norm, gating by *out_gate* and final linear projection
    back to the original channel dimension C.

Only the element‑wise fusion is performed in Triton – this satisfies the
“some‑kernel” constraint while keeping the dominant cubic work on the highly
optimised cuBLAS path.  All heavy arithmetic is in fp16; the final result is
returned as fp32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – apply mask (optional) and multiply by the two gate tensors
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_and_gate_kernel(
    left_ptr, right_ptr,                 # fp16  (B,N,N,H)
    left_gate_ptr, right_gate_ptr,       # fp16  (B,N,N,H)
    mask_ptr,                            # fp16  (B,N,N)  or nullptr
    out_left_ptr, out_right_ptr,         # fp16  (B,N,N,H)  output buffers
    B, N, H,                             # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_g0, stride_g1, stride_g2, stride_g3,
    stride_m0, stride_m1, stride_m2,     # mask strides (no hidden dim)
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    The kernel is launched with a 1‑D grid that covers B*N*N*H elements.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear ids
    total = B * N * N * H
    mask = offs < total

    # --------------------------------------------------------------
    # decode linear index → (b,i,j,h)
    # layout = contiguous (b,i,j,h)
    # --------------------------------------------------------------
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # --------------------------------------------------------------
    # load tensors
    # --------------------------------------------------------------
    idx_l = (
        b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    )
    left      = tl.load(left_ptr + idx_l,      mask=mask, other=0.0)
    right     = tl.load(right_ptr + idx_l,     mask=mask, other=0.0)

    idx_g = (
        b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    )
    lgate = tl.load(left_gate_ptr  + idx_g, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + idx_g, mask=mask, other=0.0)

    if HAS_MASK:
        idx_m = b * stride_m0 + i * stride_m1 + j * stride_m2
        m = tl.load(mask_ptr + idx_m, mask=mask, other=0.0)
    else:
        m = 1.0

    out_l = left  * lgate * m
    out_r = right * rgate * m

    # --------------------------------------------------------------
    # write results
    # --------------------------------------------------------------
    tl.store(out_left_ptr  + idx_l, out_l, mask=mask)
    tl.store(out_right_ptr + idx_l, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : (x, mask, weights, config)
        x       – (B, N, N, C)    torch.float32
        mask    – (B, N, N)       torch.bool / float32 / None
        weights – dict of learned parameters (float32)
        config  – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor
        (B, N, N, C)   float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, C = x.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (keep in fp32 for now)
    # ------------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # ------------------------------------------------------------------
    # 3️⃣ Cast to fp16 and fuse mask + gates with Triton
    # ------------------------------------------------------------------
    left_fp16      = left.to(torch.float16).contiguous()
    right_fp16     = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate.to(torch.float16).contiguous()
    right_gate_fp16= right_gate.to(torch.float16).contiguous()

    if mask is None:
        # dummy buffer – Triton will never read it because HAS_MASK=0
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask = False
    else:
        # (B,N,N) -> (B,N,N,1) to broadcast over H
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        has_mask = True

    # allocate output buffers
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elements = B * N * N * H
    BLOCK = 1024                     # good occupancy on Hopper
    grid = (triton.cdiv(total_elements, BLOCK),)

    _apply_mask_and_gate_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        # strides for left/right (identical)
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        # strides for gates
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        # strides for mask (no hidden dim)
        mask_fp16.stride(0), mask_fp16.stride(1),
        mask_fp16.stride(2),
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS / Tensor‑cores)
    #    left_fused/right_fused : (B,N,N,H) → (B*H, N, N)
    # ------------------------------------------------------------------
    # bring hidden dim to second axis for the batched matmul
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)      # (B*H,N,N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)     # (B*H,N,N)

    # fp16 × fp16 → fp16 (accumulate in fp32)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))            # (B*H,N,N)

    # reshape back to (B,N,N,H) and cast to fp32 for the next norm
    out = (
        out_mat.view(B, H, N, N)               # (B,H,N,N)
        .permute(0, 2, 3, 1)                   # (B,N,N,H)
        .to(dtype)
    )

    # ------------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                     # broadcast over (N,N)
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
```

### How the lessons were applied
| Lesson | Application |
|--------|-------------|
| **Delegate N³ contraction to cuBLAS** | Steps 4‑5 reshape the tensors and use `torch.bmm` (cuBLAS/Tensor‑cores) for the cubic work. |
| **Match dtypes for batched GEMM** | All tensors entering the batched matmul are cast to `float16`; accumulation stays in FP32 (default cuBLAS behaviour). |
| **Limit shared memory / kernel size** | The Triton kernel only does a simple element‑wise multiplication, using a 1‑D grid with `BLOCK=1024`, well within shared‑memory limits. |
| **Pass exact strides** | Strides are taken directly from the tensors (`tensor.stride(dim)`) and passed explicitly, avoiding mismatched‑stride bugs. |
| **Mask handling** | The kernel has a compile‑time `HAS_MASK` flag; when `mask` is `None` we pass a dummy buffer and set the flag to `0`, avoiding any unnecessary loads. |
