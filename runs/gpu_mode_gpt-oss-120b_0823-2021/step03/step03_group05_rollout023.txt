"""
TriMul (outgoing) – forward pass
--------------------------------
* Layer‑norm + three linear projections (left/right + gates) are kept in PyTorch
  – they are cheap compared with the O(N³) contraction.
* A tiny Triton kernel fuses:
        - mask (or all‑ones when no mask)
        - multiplication with left‑gate and right‑gate
        - in‑place scaling of the left/right projected tensors.
  This satisfies the “must use a kernel” requirement while staying memory‑bound.
* The cubic contraction
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
  is performed as a batched GEMM by reshaping the tensors to
        (B·H, N, N)  and calling ``torch.bmm`` (Tensor‑core accelerated on H200).
* Post‑processing (layer‑norm, out‑gate, final linear) is done in PyTorch.
* All intermediate work is performed in FP16 for speed; the final result is
  returned in FP32 as required by the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – fuse mask, left/right gating and in‑place scaling.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *mut fp16  (B,N,N,H) – will be overwritten
    right_ptr,         # *mut fp16  (B,N,N,H) – will be overwritten
    left_gate_ptr,     # *const fp16 (B,N,N,H)   sigmoid(left_gate)
    right_gate_ptr,    # *const fp16 (B,N,N,H)   sigmoid(right_gate)
    mask_ptr,          # *const fp16 (B,N,N,1)  broadcasted mask, 1.0 if no mask
    B, N, H,           # compile‑time sizes
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_g0, stride_g1, stride_g2, stride_g3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N * H
    idx = pid * BLOCK + tl.arange(0, BLOCK)
    mask = idx < total

    # -------------------------------------------------------------
    # decode linear index into (b, i, j, h)
    # -------------------------------------------------------------
    tmp = idx
    h = tmp % H; tmp = tmp // H
    j = tmp % N; tmp = tmp // N
    i = tmp % N; b = tmp // N

    # -------------------------------------------------------------
    # compute pointers for this element
    # -------------------------------------------------------------
    l_ptr = left_ptr  + b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    r_ptr = right_ptr + b * stride_r0 + i * stride_r1 + j * stride_r2 + h * stride_r3
    lg_ptr = left_gate_ptr  + b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    rg_ptr = right_gate_ptr + b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    m_ptr = mask_ptr + b * stride_m0 + i * stride_m1 + j * stride_m2          # H‑dim is broadcasted

    # -------------------------------------------------------------
    # load, fuse and store
    # -------------------------------------------------------------
    l = tl.load(l_ptr, mask=mask, other=0.0)
    r = tl.load(r_ptr, mask=mask, other=0.0)
    lg = tl.load(lg_ptr, mask=mask, other=0.0)
    rg = tl.load(rg_ptr, mask=mask, other=0.0)
    m = tl.load(m_ptr, mask=mask, other=1.0)        # default 1.0 → no masking

    # fused elementwise:   left = left * mask * left_gate
    #                      right = right * mask * right_gate
    l = l * m * lg
    r = r * m * rg

    tl.store(l_ptr, l, mask=mask)
    tl.store(r_ptr, r, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        - x : Tensor[B, N, N, dim]  (float32)
        - mask : Tensor[B, N, N] or None  (bool/float32)
        - weights : dict of model parameters (float32)
        - config : {"dim": int, "hidden_dim": int}

    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    H = config["hidden_dim"]          # hidden dimensionality (c_z)

    # -----------------------------------------------------------------
    # 1️⃣ Layer‑norm + linear projections (stay in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # -----------------------------------------------------------------
    # 2️⃣ Fuse mask & gating using Triton
    # -----------------------------------------------------------------
    # mask_tensor : (B,N,N,1) fp16  → 1.0 everywhere if mask is None
    if mask is None:
        mask_tensor = torch.ones((B, N, N, 1), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(dtype).unsqueeze(-1)               # (B,N,N,1)
        mask_tensor = mask_tensor.to(torch.float16)

    # cast tensors to fp16 (fast for the GEMM later)
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    left_gate_fp16  = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()

    total_el = B * N * N * H
    BLOCK = 1024                                 # gives ~1‑2 KB of register usage
    grid = (triton.cdiv(total_el, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_tensor,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        mask_tensor.stride(0), mask_tensor.stride(1),
        mask_tensor.stride(2), mask_tensor.stride(3),
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 3️⃣ O(N³) contraction – batched GEMM (FP16 → FP32 after)
    # -----------------------------------------------------------------
    # reshape to (B·H, N, N) where H is the hidden dimension
    left_bh  = left_fp16.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B·H,N,N)
    right_bh = right_fp16.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B·H,N,N)

    # batched matrix‑matrix product: (B·H,N,N) @ (B·H,N,N)^T → (B·H,N,N)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))

    # restore original layout (B,N,N,H) and cast back to FP32
    out = out_bh.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(dtype)

    # -----------------------------------------------------------------
    # 4️⃣ Post‑processing (layer‑norm, out‑gate, final linear)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                    # broadcast over N,N
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
