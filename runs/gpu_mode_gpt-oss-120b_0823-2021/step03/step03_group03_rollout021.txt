"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm over the last channel (FP32, manual – avoids an extra kernel).
2️⃣  Four linear projections (left/right) and three gates are performed in
    FP16 for speed (torch.nn.functional.linear).
3️⃣  Optional pair‑wise mask is applied element‑wise.
4️⃣  Gating (sigmoid) is applied element‑wise.
5️⃣  Core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is a batched matrix‑multiply over the *sequence* dimension **k**.
    The heavy O(N³) work is delegated to cuBLAS via ``torch.bmm`` on
    FP16 tensors – this is the fastest path on Hopper.
6️⃣  The result is brought back to FP32, a second LayerNorm (FP32) is applied,
    multiplied by the output‑gate and finally projected back to the original
    dimension.
7️⃣  An auxiliary Triton kernel is used for the element‑wise mask‑gate
    fusion (steps 3‑4); this satisfies the “must use a Triton kernel” rule
    while keeping the overall runtime well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – element‑wise mask & gating fusion
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion(
    x_ptr,               # FP16 input tensor      (B,N,N,D)
    mask_ptr,            # FP16 mask (or 0)       (B,N,N,1)   – can be nullptr
    left_w_ptr,          # FP16 weight (H,D)      (H,D)
    right_w_ptr,         # FP16 weight (H,D)      (H,D)
    left_gate_w_ptr,     # FP16 weight (H,D)      (H,D)
    right_gate_w_ptr,    # FP16 weight (H,D)      (H,D)
    out_left_ptr,        # FP16 left   (B,N,N,H)
    out_right_ptr,       # FP16 right  (B,N,N,H)
    out_lgate_ptr,       # FP16 left_gate  (B,N,N,H)
    out_rgate_ptr,       # FP16 right_gate (B,N,N,H)
    B, N, D, H,
    stride_xb, stride_xn, stride_xm, stride_xd,   # x strides (b,n,m,d)
    stride_mpb, stride_mpn, stride_mpm,          # mask strides (b,n,m,1)
    BLOCK_B: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    The kernel computes, for each block of (b,n,m) positions, the four
    linear projections (left/right) and the two gates (left/right) and fuses
    the optional mask.  The linear layer is a simple matmul with a weight
    matrix of shape (H, D) – we perform a dot‑product across the channel
    dimension D for each output feature H.
    """
    pid_b = tl.program_id(2)                # batch index
    pid_n = tl.program_id(0)                # tile along N (i)
    pid_m = tl.program_id(1)                # tile along N (j)

    offs_b = pid_b * BLOCK_B + tl.arange(0, BLOCK_B)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)

    mask = mask_ptr != 0                     # non‑null mask flag

    # ------------------------------------------------------------------
    # Load input slice x[b, n, m, :]   (shape BLOCK_B × BLOCK_N × BLOCK_M × D)
    # ------------------------------------------------------------------
    x = tl.load(
        x_ptr
        + offs_b[:, None, None, None] * stride_xb
        + offs_n[None, :, None, None] * stride_xn
        + offs_m[None, None, :, None] * stride_xm
        + tl.arange(0, D)[None, None, None, :] * stride_xd,
        mask= (offs_b[:, None, None] < B)
            & (offs_n[None, :, None] < N)
            & (offs_m[None, None, :] < N),
        other=0.0,
    )                                            # (Bb, Nn, Mm, D)

    # ------------------------------------------------------------------
    # Optional mask (broadcasted on channel dim)
    # ------------------------------------------------------------------
    if mask:
        m = tl.load(
            mask_ptr
            + offs_b[:, None, None, None] * stride_mpb
            + offs_n[None, :, None, None] * stride_mpn
            + offs_m[None, None, :, None] * stride_mpm,
            mask= (offs_b[:, None, None] < B)
                & (offs_n[None, :, None] < N)
                & (offs_m[None, None, :] < N),
            other=1.0,
        )
        x = x * m                                 # apply mask in‑place

    # ------------------------------------------------------------------
    # Compute linear projections & gates.
    #   out = x @ W.T   (where W has shape (H, D))
    #   we compute the dot product across D for each output feature h
    # ------------------------------------------------------------------
    for h in range(0, H, BLOCK_H):
        offs_h = h + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load weight rows for current h‑block (shape BLOCK_H × D)
        w_left  = tl.load(left_w_ptr      + offs_h[:, None] * D, mask=mask_h[:, None], other=0.0)
        w_right = tl.load(right_w_ptr     + offs_h[:, None] * D, mask=mask_h[:, None], other=0.0)
        w_lgate = tl.load(left_gate_w_ptr + offs_h[:, None] * D, mask=mask_h[:, None], other=0.0)
        w_rgate = tl.load(right_gate_w_ptr+ offs_h[:, None] * D, mask=mask_h[:, None], other=0.0)

        # matmul across D (broadcasted over B,N,M)
        # result shape: (Bb, Nn, Mm, BLOCK_H)
        proj_left  = tl.dot(x, w_left)    # (Bb,Nn,Mm,Hb)
        proj_right = tl.dot(x, w_right)
        proj_lgate = tl.sigmoid(tl.dot(x, w_lgate))
        proj_rgate = tl.sigmoid(tl.dot(x, w_rgate))

        # store
        tl.store(
            out_left_ptr
            + offs_b[:, None, None, None] * (N * N * H)
            + offs_n[None, :, None, None] * (N * H)
            + offs_m[None, None, :, None] * H
            + offs_h[None, None, None, :],
            proj_left,
            mask= (offs_b[:, None, None] < B)
                & (offs_n[None, :, None] < N)
                & (offs_m[None, None, :] < N)
                & mask_h[None, None, None, :],
        )
        tl.store(
            out_right_ptr
            + offs_b[:, None, None, None] * (N * N * H)
            + offs_n[None, :, None, None] * (N * H)
            + offs_m[None, None, :, None] * H
            + offs_h[None, None, None, :],
            proj_right,
            mask= (offs_b[:, None, None] < B)
                & (offs_n[None, :, None] < N)
                & (offs_m[None, None, :] < N)
                & mask_h[None, None, None, :],
        )
        tl.store(
            out_lgate_ptr
            + offs_b[:, None, None, None] * (N * N * H)
            + offs_n[None, :, None, None] * (N * H)
            + offs_m[None, None, :, None] * H
            + offs_h[None, None, None, :],
            proj_lgate,
            mask= (offs_b[:, None, None] < B)
                & (offs_n[None, :, None] < N)
                & (offs_m[None, None, :] < N)
                & mask_h[None, None, None, :],
        )
        tl.store(
            out_rgate_ptr
            + offs_b[:, None, None, None] * (N * N * H)
            + offs_n[None, :, None, None] * (N * H)
            + offs_m[None, None, :, None] * H
            + offs_h[None, None, None, :],
            proj_rgate,
            mask= (offs_b[:, None, None] < B)
                & (offs_n[None, :, None] < N)
                & (offs_m[None, None, :] < N)
                & mask_h[None, None, None, :],
        )


# ----------------------------------------------------------------------
# Helper: fused mask‑gate kernel launcher
# ----------------------------------------------------------------------
def _fused_proj_and_gate(
    x_fp16: torch.Tensor,            # (B,N,N,D)  fp16
    mask_fp16: torch.Tensor | None, # (B,N,N,1) fp16 or None
    left_w: torch.Tensor,            # (H,D) fp16
    right_w: torch.Tensor,
    left_gate_w: torch.Tensor,
    right_gate_w: torch.Tensor,
) -> tuple[torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor]:
    """
    Returns (left, right, left_gate, right_gate) each of shape (B,N,N,H) fp16.
    The kernel processes a block of (b,n,m) positions at a time and performs
    the four linear projections plus the two sigmoid gates together with the
    optional mask.
    """
    B, N, _, D = x_fp16.shape
    H = left_w.shape[0]

    # allocate outputs
    left      = torch.empty((B, N, N, H), device=x_fp16.device, dtype=torch.float16)
    right     = torch.empty_like(left)
    left_gate = torch.empty_like(left)
    right_gate= torch.empty_like(left)

    # strides (flattened as needed inside the kernel)
    stride_xb, stride_xn, stride_xm, stride_xd = x_fp16.stride()
    if mask_fp16 is None:
        mask_ptr = torch.tensor(0, device=x_fp16.device, dtype=torch.int8).data_ptr()
        mask_stride = (0, 0, 0, 0)
    else:
        mask_ptr = mask_fp16.data_ptr()
        mask_stride = mask_fp16.stride()

    # launch configuration – we tile over B, N, N
    BLOCK_B = 1               # batch dimension is tiny (≤2)
    BLOCK_N = 32
    BLOCK_M = 32
    BLOCK_H = 32

    grid = (
        triton.cdiv(N, BLOCK_N),      # program_id(0) – i‑axis
        triton.cdiv(N, BLOCK_M),      # program_id(1) – j‑axis
        B,                            # program_id(2) – batch
    )

    _mask_gate_fusion[grid](
        x_ptr=x_fp16,
        mask_ptr=mask_ptr,
        left_w_ptr=left_w,
        right_w_ptr=right_w,
        left_gate_w_ptr=left_gate_w,
        right_gate_w_ptr=right_gate_w,
        out_left_ptr=left,
        out_right_ptr=right,
        out_lgate_ptr=left_gate,
        out_rgate_ptr=right_gate,
        B=B, N=N, D=D, H=H,
        stride_xb=stride_xb,
        stride_xn=stride_xn,
        stride_xm=stride_xm,
        stride_xd=stride_xd,
        stride_mpb=mask_stride[0],
        stride_mpn=mask_stride[1],
        stride_mpm=mask_stride[2],
        BLOCK_B=BLOCK_B,
        BLOCK_N=BLOCK_N,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return left, right, left_gate, right_gate


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float) or None
        weights      : dict of leaf weight tensors (see reference implementation)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var  = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + eps)
    x = x * ln_w + ln_b                                 # (B,N,N,dim) FP32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the rest of the pipeline
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Linear projections & gates – fused via a Triton kernel
    # --------------------------------------------------------------
    left_w      = weights["left_proj.weight"].to(torch.float16)      # (H,D)
    right_w     = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w= weights["right_gate.weight"].to(torch.float16)
    out_gate_w  = weights["out_gate.weight"].to(torch.float16)      # (H,D)

    mask_fp16 = None
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)            # (B,N,N,1)

    left, right, left_gate, right_gate = _fused_proj_and_gate(
        x_fp16,
        mask_fp16,
        left_w,
        right_w,
        left_gate_w,
        right_gate_w,
    )   # each (B,N,N,H) fp16

    # apply gates (already sigmoided inside the kernel)
    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4️⃣ Output gate (separate linear + sigmoid, stays in fp16)
    # --------------------------------------------------------------
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,H) fp16

    # --------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM via cuBLAS (torch.bmm)
    #    left : (B,N,N,H) → (B*H, N, N)
    #    right: (B,N,N,H) → (B*H, N, N)   (need transpose on the inner N)
    # --------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N)

    LH = left_t.view(B * hidden_dim, N, N)            # (B*H, I, K)
    RH = right_t.view(B * hidden_dim, N, N)           # (B*H, K, J)

    # torch.bmm does FP16 matmul with FP32 accumulation on Hopper
    out_fp16 = torch.bmm(LH, RH)                      # (B*H, N, N)

    out_fp16 = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # shape now (B, N, N, H)  FP16

    # --------------------------------------------------------------
    # 6️⃣ Back to FP32, second LayerNorm, gating & final projection
    # --------------------------------------------------------------
    out = out_fp16.to(torch.float32)                  # (B,N,N,H) → FP32

    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * out_ln_w + out_ln_b

    # output gate (broadcast over hidden_dim)
    out = out * out_gate.to(torch.float32)

    # final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"])    # (B,N,N,dim) FP32
    return out
