"""
TriMul “outgoing” forward pass (no gradients).

The expensive N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a batched GEMM that is hand‑written in Triton.
All cheap ops (LayerNorm, Linear, sigmoid, masking, final LayerNorm and
linear projection) stay in PyTorch – they are memory‑bound and much cheaper
than the cubic part.  Mixed‑precision is used: the heavy GEMM works in
fp16 / fp32 (Tensor‑core friendly) and the final result is cast back to fp32.

The kernel below follows the classic “block‑wise” GEMM pattern and is
compatible with Hopper (sm_90).  It processes a batch of matrices
(Z, I, K) × (Z, K, J) → (Z, I, J) where Z = B * hidden_dim.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton batched GEMM (fp16 inputs → fp16 outputs, fp32 acc)
# --------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    a_ptr,          # left  tensor (Z, I, K)   fp16
    b_ptr,          # right tensor (Z, K, J)   fp16
    c_ptr,          # out   tensor (Z, I, J)   fp16
    M, N, K,        # sizes of the matmul
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)   # batch*head index
    pid_m = tl.program_id(1)   # row tile
    pid_n = tl.program_id(2)   # col tile

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension in blocks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load a tile (M x K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load b tile (K x N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 accumulation
        acc += tl.dot(a, b)

    # write back as fp16
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous
    returns (B, H, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten the first two dimensions for the kernel
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides in *elements* (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling – empirically good for Hopper / H200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                     # program_id(0) – batch*head dimension
        triton.cdiv(I, BLOCK_M),   # program_id(1) – rows
        triton.cdiv(J, BLOCK_N),   # program_id(2) – cols
    )

    batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # enough to keep the cores busy on H200
    )
    return out


# --------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor [B, N, N]       (float32/bool) or None
        - weights      : dict of torch.Tensor parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]   (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1) Input LayerNorm (keep in FP32 for stability)
    # --------------------------------------------------------------
    w_ln = weights["norm.weight"]
    b_ln = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + eps)
    x = x * w_ln + b_ln                     # [B,N,N,dim] fp32

    # --------------------------------------------------------------
    # 2) Cast once to fp16 for the heavy path
    # --------------------------------------------------------------
    x_fp16 = x.to(dtype=torch.float16, non_blocking=True)

    # linear weights (fp16)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 3) Projections
    # --------------------------------------------------------------
    left  = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden]
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden]

    # --------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.to(dtype=torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 5) Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()      # (B,N,N,hidden)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6) Core cubic contraction via Triton GEMM
    #    left : (B, N, N, H) → (B, H, I, K)
    #    right: (B, N, N, H) → (B, H, K, J)  (transpose last two dims)
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = triton_batched_gemm(left_perm, right_perm)   # (B, H, I, J) fp16

    # back to (B,N,N,hidden)
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 7) Output LayerNorm (fp32)
    # --------------------------------------------------------------
    w_out_ln = weights["to_out_norm.weight"]
    b_out_ln = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * w_out_ln + b_out_ln

    # --------------------------------------------------------------
    # 8) Final gating + linear projection back to dim
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # same gate as in reference
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
