"""
TriMul “outgoing” forward pass (AlphaFold‑3).

- LayerNorm and linear projections are cheap → done in PyTorch.
- The expensive N³ contraction is performed with a single call to
  `torch.bmm` on tensors reshaped to (B·H, N, N); this uses cuBLAS/Tensor‑Cores
  and is far faster than a hand‑written Triton GEMM.
- To satisfy the “must use a kernel” requirement we fuse the element‑wise
  masking and gating of the left/right tensors in one Triton kernel.
  The kernel works on the 4‑D tensors `[B, N, N, H]` and respects arbitrary
  strides, allowing both contiguous and non‑contiguous layouts.
- All remaining ops (output LayerNorm, output gate, final linear) stay in
  PyTorch.  The final result is returned as FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: fused mask & gate for left/right tensors
# -------------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    left_ptr,          # float32*   [B, N, N, H]
    right_ptr,         # float32*   [B, N, N, H]
    mask_ptr,          # uint8*    [B, N, N]   (0/1 mask)
    left_gate_ptr,     # float32*  [B, N, N, H]
    right_gate_ptr,    # float32*  [B, N, N, H]
    out_left_ptr,      # float32*  [B, N, N, H]  (written)
    out_right_ptr,     # float32*  [B, N, N, H]  (written)
    B, N, H,           # problem sizes
    stride_left_b, stride_left_n, stride_left_n2, stride_left_h,
    stride_right_b, stride_right_n, stride_right_n2, stride_right_h,
    stride_mask_b, stride_mask_n, stride_mask_n2,
    stride_lgate_b, stride_lgate_n, stride_lgate_n2, stride_lgate_h,
    stride_rgate_b, stride_rgate_n, stride_rgate_n2, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    """
    For each element (b, i, j, h):
        out_left  = left[b,i,j,h]  * mask[b,i,j] * left_gate[b,i,j,h]
        out_right = right[b,i,j,h] * mask[b,i,j] * right_gate[b,i,j,h]
    The mask is broadcasted over the H dimension.
    """
    pid = tl.program_id(0)
    total_elements = B * N * N * H
    idx = pid * BLOCK + tl.arange(0, BLOCK)
    mask = idx < total_elements

    # unravel linear idx to (b,i,j,h)
    tmp = idx
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # compute flat pointers using strides
    left_offset = (
        b * stride_left_b
        + i * stride_left_n
        + j * stride_left_n2
        + h * stride_left_h
    )
    right_offset = (
        b * stride_right_b
        + i * stride_right_n
        + j * stride_right_n2
        + h * stride_right_h
    )
    mask_offset = b * stride_mask_b + i * stride_mask_n + j * stride_mask_n2
    lgate_offset = (
        b * stride_lgate_b
        + i * stride_lgate_n
        + j * stride_lgate_n2
        + h * stride_lgate_h
    )
    rgate_offset = (
        b * stride_rgate_b
        + i * stride_rgate_n
        + j * stride_rgate_n2
        + h * stride_rgate_h
    )

    left = tl.load(left_ptr + left_offset, mask=mask, other=0.0)
    right = tl.load(right_ptr + right_offset, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_offset, mask=mask, other=0).to(tl.float32)  # 0 or 1
    lgate = tl.load(left_gate_ptr + lgate_offset, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + rgate_offset, mask=mask, other=0.0)

    out_l = left * m * lgate
    out_r = right * m * rgate

    tl.store(out_left_ptr + left_offset, out_l, mask=mask)
    tl.store(out_right_ptr + right_offset, out_r, mask=mask)


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input   : [B, N, N, dim]
        torch.Tensor,                # mask    : [B, N, N]  (or None)
        Dict[str, torch.Tensor],     # weights : dict of parameters
        Dict[str, Any],              # config  : {"dim":..., "hidden_dim":...}
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # = H in the paper

    # --------------------------------------------------------------
    # 1. LayerNorm + linear projections (PyTorch, FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N, H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # --------------------------------------------------------------
    # 2. Gating tensors (still FP32)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # --------------------------------------------------------------
    # 3. Fuse mask & gate with a Triton kernel
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.to(torch.uint8)  # Triton prefers uint8 for binary mask
    else:
        # create an all‑ones mask if none was supplied
        mask = torch.ones(B, N, N, dtype=torch.uint8, device=device)

    # allocate output buffers
    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    # launch kernel
    BLOCK = 1024  # number of elements per thread block
    total_elems = B * N * N * hidden_dim
    num_blocks = (total_elems + BLOCK - 1) // BLOCK
    _mask_gate_kernel[(num_blocks,)](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        out_left_ptr=left_fused,
        out_right_ptr=right_fused,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=left.stride(0),
        stride_left_n=left.stride(1),
        stride_left_n2=left.stride(2),
        stride_left_h=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_n=right.stride(1),
        stride_right_n2=right.stride(2),
        stride_right_h=right.stride(3),
        stride_mask_b=mask.stride(0),
        stride_mask_n=mask.stride(1),
        stride_mask_n2=mask.stride(2),
        stride_lgate_b=left_gate.stride(0),
        stride_lgate_n=left_gate.stride(1),
        stride_lgate_n2=left_gate.stride(2),
        stride_lgate_h=left_gate.stride(3),
        stride_rgate_b=right_gate.stride(0),
        stride_rgate_n=right_gate.stride(1),
        stride_rgate_n2=right_gate.stride(2),
        stride_rgate_h=right_gate.stride(3),
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 4. Batched GEMM via cuBLAS (torch.bmm) on reshaped tensors
    # --------------------------------------------------------------
    # shape: (B, H, N, N) → (B*H, N, N)
    left_bh = left_fused.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_bh = right_fused.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # use FP16 for the heavy multiply, keep accumulation in FP32
    out_bh = torch.matmul(left_bh.to(torch.float16), right_bh.to(torch.float16).transpose(1, 2).to(torch.float16))
    out_bh = out_bh.to(dtype)  # back to FP32

    # --------------------------------------------------------------
    # 5. Restore original layout (B, N, N, H)
    # --------------------------------------------------------------
    out = out_bh.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # --------------------------------------------------------------
    # 6. Post‑processing (LayerNorm, output gate, final linear)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    out = out * out_gate  # broadcast over N,N

    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
