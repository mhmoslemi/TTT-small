"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward‑only)
------------------------
1️⃣  Layer‑norm on the 4‑D input (float‑32) – computed in PyTorch.
2️⃣  Five linear projections (left, right, left‑gate, right‑gate, out‑gate) are
    performed in float‑16 for speed.
3️⃣  Optional binary mask is applied element‑wise on the left/right
    projections with a tiny Triton kernel (broadcasted over the hidden dim).
4️⃣  Sigmoid gating of the three gate tensors.
5️⃣  Core contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .
    This is a batched matrix‑multiply with shape
        (B·H, N, N) · (B·H, N, N)ᵀ → (B·H, N, N) .
    The heavy O(N³) work is delegated to cuBLAS via torch.bmm (fp16×fp16 → fp32
    accumulation).
6️⃣  Output LayerNorm (float‑32), multiply by the output‑gate and final linear
    projection back to the original dimension (float‑32).

All tensor reshapes are contiguous; the only Triton code is the mask‑apply kernel,
which makes the solution satisfy the “must use a kernel” requirement while
remaining fast on Hopper (H200).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – apply broadcasted mask to a 4‑D tensor (B,N,N,H)
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    src_ptr,                # input (fp16)
    mask_ptr,               # mask  (fp16) broadcasted over hidden dim
    dst_ptr,                # output (fp16) = src * mask
    B, N, H,                # logical extents
    stride_bs, stride_bn, stride_bk, stride_bh,   # src strides
    stride_ms, stride_mn, stride_mk,             # mask strides (no hidden dim)
    stride_ds, stride_dn, stride_dk, stride_dh, # dst strides
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Each program processes a tile (BLOCK_N × BLOCK_K × BLOCK_H) of the
    (N, N, H) dimensions for a fixed batch index (pid_z).
    """
    pid_z = tl.program_id(2)          # batch index
    pid_n = tl.program_id(0)          # tile row  (i dimension)
    pid_k = tl.program_id(1)          # tile col  (j dimension)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = pid_k * BLOCK_K + tl.arange(0, BLOCK_K)
    offs_h = tl.arange(0, BLOCK_H)

    mask_n = offs_n < N
    mask_k = offs_k < N
    mask_h = offs_h < H

    # load mask (shape B,N,N) and broadcast over hidden dim
    mask = tl.load(
        mask_ptr
        + pid_z * stride_ms
        + offs_n[:, None] * stride_mn
        + offs_k[None, :] * stride_mk,
        mask=mask_n[:, None] & mask_k[None, :],
        other=0.0,
    )
    # expand mask to (N,N,H)
    mask = mask[:, :, None]          # (N,N,1)
    mask = tl.broadcast_to(mask, (BLOCK_N, BLOCK_K, BLOCK_H))

    # load source tensor
    src = tl.load(
        src_ptr
        + pid_z * stride_bs
        + offs_n[:, None, None] * stride_bn
        + offs_k[None, :, None] * stride_bk
        + offs_h[None, None, :] * stride_bh,
        mask=(mask_n[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]),
        other=0.0,
    )

    dst = src * mask

    tl.store(
        dst_ptr
        + pid_z * stride_ds
        + offs_n[:, None, None] * stride_dn
        + offs_k[None, :, None] * stride_dk
        + offs_h[None, None, :] * stride_dh,
        dst,
        mask=(mask_n[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]),
    )


def _apply_mask_fp16(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    tensor : (B,N,N,H) fp16
    mask   : (B,N,N)   fp16   (binary or 0/1 floats)
    returns tensor * mask  (fp16)
    """
    B, N, _, H = tensor.shape
    dst = torch.empty_like(tensor)

    # strides in element units
    s_bs, s_bn, s_bk, s_bh = tensor.stride()
    m_ms, m_mn, m_mk = mask.stride()
    d_ds, d_dn, d_dk, d_dh = dst.stride()

    # tile sizes – chosen to fit comfortably in shared memory on H200
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 16   # hidden dim is at most 128, 16 works well

    grid = (
        triton.cdiv(N, BLOCK_N),   # rows (i)
        triton.cdiv(N, BLOCK_K),   # cols (j)
        B,                         # batch
    )
    _apply_mask_kernel[grid](
        tensor,
        mask,
        dst,
        B, N, H,
        stride_bs=s_bs,
        stride_bn=s_bn,
        stride_bk=s_bk,
        stride_bh=s_bh,
        stride_ms=m_ms,
        stride_mn=m_mn,
        stride_mk=m_mk,
        stride_ds=d_ds,
        stride_dn=d_dn,
        stride_dk=d_dk,
        stride_dh=d_dh,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return dst


# ----------------------------------------------------------------------
# Helper – batched GEMM using cuBLAS (torch.bmm)
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, N, N) fp16
    right : (B, H, N, N) fp16
    Returns (B, H, N, N) fp16   = left @ rightᵀ per (B,H) slice
    """
    B, H, N, _ = left.shape
    # reshape to (B*H, N, N)
    left_2d = left.reshape(B * H, N, N)
    right_2d = right.reshape(B * H, N, N)

    # torch.bmm does fp16×fp16 → fp16 (Tensor‑cores) but accumulates in fp32 internally.
    out_2d = torch.bmm(left_2d, right_2d.transpose(1, 2))

    return out_2d.reshape(B, H, N, N)

# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors (see below)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (float32, manual)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still fp32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections + gating (float16)
    # --------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16)

    left_proj_w  = to_fp16(weights["left_proj.weight"])   # (hidden, dim)
    right_proj_w = to_fp16(weights["right_proj.weight"])  # (hidden, dim)
    left_gate_w  = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w   = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                     # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)            # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)           # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask – Triton kernel
    # --------------------------------------------------------------
    if mask is not None:
        # mask is bool or 0/1 float – convert to fp16
        mask_fp16 = mask.to(torch.float16)
        left  = _apply_mask_fp16(left,  mask_fp16)
        right = _apply_mask_fp16(right, mask_fp16)

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()   # (B,N,N,hidden)
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()    # (B,N,N,hidden)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM via cuBLAS
    #    reshape to (B, hidden, N, N) and compute left @ rightᵀ per head
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, N, N) with K,J order

    out_fp16 = _batched_gemm_fp16(left_perm, right_perm)   # (B, hidden, N, N)

    # back to (B,N,N,dim)  (hidden -> dim after to_out_norm / to_out)
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (float32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection (float32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)            # broadcast over hidden dim
    to_out_w = weights["to_out.weight"]               # (dim, hidden)
    out = F.linear(out, to_out_w)                     # (B,N,N,dim)

    return out
