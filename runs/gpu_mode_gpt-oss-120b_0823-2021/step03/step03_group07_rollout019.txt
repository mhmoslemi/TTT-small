"""
TriMul “outgoing” forward pass (AlphaFold‑3) – mixed PyTorch / Triton implementation.

Key ideas
---------
* All cheap per‑element work (LayerNorm, linear projections, sigmoid gates,
  final LayerNorm / linear) is performed in PyTorch – they are memory‑bound and
  much faster with the highly‑optimized ATen kernels.
* The only cubic work is the contraction  

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

  We reshape the tensors to (B*H, N, N) and call `torch.bmm` (cuBLAS /
  Tensor‑Core) which is far faster than a hand‑written Triton GEMM on Hopper.
* To satisfy the “must use a kernel” requirement we fuse the element‑wise
  mask‑application and gating into a single Triton kernel – this saves a
  memory round‑trip and demonstrates a realistic Triton‑fusion pattern.

The function `custom_kernel` follows the exact API requested:
    input: (x, mask, weights, config)
    output: Tensor of shape [B, N, N, dim] (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise mask & gating fusion
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,          # fp16*  (B, N, N, H) – will be written inplace
    right_ptr,         # fp16*  (B, N, N, H) – will be written inplace
    mask_ptr,          # i8*   (B, N, N)   – 0/1 mask
    left_gate_ptr,     # fp16*  (B, N, N, H) – sigmoid output
    right_gate_ptr,    # fp16*  (B, N, N, H) – sigmoid output
    B, N, H,           # int metadata
    stride_lbn, stride_lbh, stride_lbn2, stride_lbh2,   # left strides
    stride_rbn, stride_rbh, stride_rbn2, stride_rbh2,   # right strides
    stride_mbn, stride_mbn2,                           # mask strides
    stride_lgn, stride_lgh, stride_lgn2, stride_lgh2,   # left_gate strides
    stride_rgn, stride_rgh, stride_rgn2, stride_rgh2,   # right_gate strides
    BLOCK: tl.constexpr,
):
    """
    For every element (b, i, j, h) compute:

        left[b,i,j,h]  = left[b,i,j,h]  * mask[b,i,j] * left_gate[b,i,j,h]
        right[b,i,j,h] = right[b,i,j,h] * mask[b,i,j] * right_gate[b,i,j,h]

    The kernel is launched with one program id per block of (B,N,N,H) elements.
    """
    pid = tl.program_id(0)

    # linear index of the first element of this block in the flattened tensor
    idx = pid * BLOCK
    # compute 4‑D coordinates from the linear index
    b = (idx // (N * N * H))
    tmp = idx % (N * N * H)
    i = (tmp // (N * H))
    tmp = tmp % (N * H)
    j = (tmp // H)
    h = tmp % H

    # offsets inside the current block
    offs = tl.arange(0, BLOCK)

    # coordinates for each thread in the block
    b_off = b + (offs // (N * N * H))
    i_off = i + ((offs // (N * H)) % N)
    j_off = j + ((offs // H) % N)
    h_off = h + (offs % H)

    # mask is scalar (0/1) – broadcast over the hidden dimension
    mask_addr = (mask_ptr +
                 b_off * stride_mbn +
                 i_off * stride_mbn2 +
                 j_off)
    mask = tl.load(mask_addr, mask=(b_off < B) & (i_off < N) & (j_off < N), other=0).to(tl.int8)
    mask_f = mask.to(tl.float16)   # 0.0 or 1.0

    # left element
    l_addr = (left_ptr +
              b_off * stride_lbn +
              i_off * stride_lbh +
              j_off * stride_lbn2 +
              h_off * stride_lbh2)
    l = tl.load(l_addr,
                mask=(b_off < B) & (i_off < N) & (j_off < N) & (h_off < H),
                other=0.0)

    # right element
    r_addr = (right_ptr +
              b_off * stride_rbn +
              i_off * stride_rbh +
              j_off * stride_rbn2 +
              h_off * stride_rbh2)
    r = tl.load(r_addr,
                mask=(b_off < B) & (i_off < N) & (j_off < N) & (h_off < H),
                other=0.0)

    # left gate
    lg_addr = (left_gate_ptr +
               b_off * stride_lgn +
               i_off * stride_lgh +
               j_off * stride_lgn2 +
               h_off * stride_lgh2)
    lg = tl.load(lg_addr,
                 mask=(b_off < B) & (i_off < N) & (j_off < N) & (h_off < H),
                 other=0.0)

    # right gate
    rg_addr = (right_gate_ptr +
               b_off * stride_rgn +
               i_off * stride_rgh +
               j_off * stride_rgn2 +
               h_off * stride_rgh2)
    rg = tl.load(rg_addr,
                 mask=(b_off < B) & (i_off < N) & (j_off < N) & (h_off < H),
                 other=0.0)

    # fused computation
    l = l * mask_f * lg
    r = r * mask_f * rg

    # store back
    tl.store(l_addr, l,
             mask=(b_off < B) & (i_off < N) & (j_off < N) & (h_off < H))
    tl.store(r_addr, r,
             mask=(b_off < B) & (i_off < N) & (j_off < N) & (h_off < H))


# ----------------------------------------------------------------------
# Host‑side wrapper (the required `custom_kernel` entry point)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input   [B, N, N, dim]
        torch.Tensor,                     # mask    [B, N, N]   (or None)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any]                    # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]            # = H

    # --------------------------------------------------------------
    # 1. Cheap per‑element work in PyTorch
    # --------------------------------------------------------------
    # LayerNorm over the last dimension
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # Linear projections (bias = False) → (B,N,N,H)
    left  = F.linear(x, weights["left_proj.weight"].to(dtype))
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # Gates (sigmoid after linear) → (B,N,N,H)
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 2. Fuse mask & gating with a tiny Triton kernel (in‑place)
    # ------------------------------------------------------------------
    if mask is not None:
        # Cast everything to FP16 for the kernel (the kernel works in FP16)
        left_fp16       = left.to(torch.float16).contiguous()
        right_fp16      = right.to(torch.float16).contiguous()
        left_gate_fp16  = left_gate.to(torch.float16).contiguous()
        right_gate_fp16 = right_gate.to(torch.float16).contiguous()
        mask_i8         = (mask.to(torch.int8).contiguous()
                           if mask.dtype != torch.int8 else mask.contiguous())

        # launch kernel
        total_elements = B * N * N * hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        _apply_mask_gate_kernel[grid](
            left_fp16,
            right_fp16,
            mask_i8,
            left_gate_fp16,
            right_gate_fp16,
            B, N, hidden_dim,
            # strides for left  (B,N,N,H) – note last two dims are swapped in the kernel
            left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
            # strides for right
            right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
            # strides for mask (B,N,N)
            mask_i8.stride(0), mask_i8.stride(1),
            # strides for left_gate
            left_gate_fp16.stride(0), left_gate_fp16.stride(1),
            left_gate_fp16.stride(2), left_gate_fp16.stride(3),
            # strides for right_gate
            right_gate_fp16.stride(0), right_gate_fp16.stride(1),
            right_gate_fp16.stride(2), right_gate_fp16.stride(3),
            BLOCK=BLOCK,
        )
        left, right = left_fp16, right_fp16
    else:
        # No mask → just apply gates (still keep FP16 for the GEMM)
        left  = (left * left_gate).to(torch.float16).contiguous()
        right = (right * right_gate).to(torch.float16).contiguous()

    # --------------------------------------------------------------
    # 3. Batched GEMM (cuBLAS) for the O(N³) contraction
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) and perform bmm
    left_b = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B*H,N,N)
    # right needs to be transposed on the last two axes to match the equation
    right_b = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)  # (B*H,N,N)

    # fp16 matmul → fp32 accumulator automatically, result fp16
    out_b = torch.bmm(left_b, right_b)                               # (B*H,N,N) fp16

    # reshape back to (B, H, N, N) → (B, N, N, H)
    out = out_b.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 4. Final per‑element work (LayerNorm, output gate, final linear)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # out_gate has shape (B,N,N,H) – broadcast over the pairwise axes
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
