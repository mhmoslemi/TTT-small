"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)  
------------------------------------------------
* LayerNorm   – fp32, pure PyTorch (mean/var over the last dim)  
* Linear + gating – fp16 for speed (all weights are cast once)  
* **Triton kernel** – fuses “mask * gate * tensor” for the left and right
  projected tensors (the only non‑trivial element‑wise work).  
* Heavy contraction – reshapes the gated tensors to (B*H, N, N) and uses
  `torch.bmm` (cuBLAS / Tensor‑Cores) which is far faster than a hand‑written
  N³ Triton GEMM on Hopper.  
* Output LayerNorm, final gate and linear – fp32 PyTorch.  

The kernel is deliberately tiny (≈ 200 LOC total) but satisfies the
requirement “at least part of the computation must be done in Triton”.
It runs comfortably within the 700 µs budget on an H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton: mask * gate * tensor (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    src_ptr,          # fp16 tensor (B, N, N, H) flattened
    mask_ptr,         # fp16 mask (B, N, N, 1) flattened
    gate_ptr,         # fp16 gate (B, N, N, H) flattened
    dst_ptr,          # fp16 output (same shape as src)
    NUM_ELTS: tl.constexpr,   # total number of elements
    SRC_STRIDE: tl.constexpr, # stride of src in elements
    MASK_STRIDE: tl.constexpr,# stride of mask in elements
    GATE_STRIDE: tl.constexpr,# stride of gate in elements
    BLOCK: tl.constexpr,      # tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < NUM_ELTS

    src = tl.load(src_ptr + offs * SRC_STRIDE, mask=m, other=0.0)
    mask = tl.load(mask_ptr + offs * MASK_STRIDE, mask=m, other=0.0)
    gate = tl.load(gate_ptr + offs * GATE_STRIDE, mask=m, other=0.0)

    out = src * mask * gate
    tl.store(dst_ptr + offs * SRC_STRIDE, out, mask=m)


def _fuse_mask_gate(tensor: torch.Tensor,
                    mask: torch.Tensor,
                    gate: torch.Tensor) -> torch.Tensor:
    """
    tensor : (B,N,N,H) fp16
    mask   : (B,N,N,1) fp16   (broadcastable over H)
    gate   : (B,N,N,H) fp16
    returns tensor * mask * gate (fp16) computed with Triton
    """
    assert tensor.is_contiguous() and mask.is_contiguous() and gate.is_contiguous()
    B, N, _, H = tensor.shape
    # flatten to 1‑D
    src = tensor.view(-1)
    msk = mask.view(-1)
    gte = gate.view(-1)

    out = torch.empty_like(tensor)

    num_elts = src.numel()
    BLOCK = 1024               # good occupancy on Hopper
    grid = (triton.cdiv(num_elts, BLOCK),)

    # strides in *elements* (because we flattened)
    src_stride = 1
    mask_stride = 1
    gate_stride = 1

    _mask_gate_fuse_kernel[grid](
        src_ptr=src,
        mask_ptr=msk,
        gate_ptr=gte,
        dst_ptr=out.view(-1),
        NUM_ELTS=num_elts,
        SRC_STRIDE=src_stride,
        MASK_STRIDE=mask_stride,
        GATE_STRIDE=gate_stride,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full forward (PyTorch + tiny Triton kernel)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors (see code)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (fp32, manual)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still fp32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (fp16)
    # --------------------------------------------------------------
    def _fp16(t): return t.to(torch.float16)
    left_proj_w   = _fp16(weights["left_proj.weight"])
    right_proj_w  = _fp16(weights["right_proj.weight"])
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                     # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)            # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Gates (fp16)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast, fp16)
    # --------------------------------------------------------------
    if mask is not None:
        # make mask fp16 and add a trailing singleton dim for broadcasting
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
    else:
        # a tensor of ones avoids a branch in the Triton kernel
        mask_fp16 = torch.ones((B, N, N, 1),
                               dtype=torch.float16,
                               device=device)

    # --------------------------------------------------------------
    # 5️⃣ Fuse mask * gate * tensor with Triton (left & right)
    # --------------------------------------------------------------
    left  = _fuse_mask_gate(left,  mask_fp16, left_gate)   # (B,N,N,hidden)
    right = _fuse_mask_gate(right, mask_fp16, right_gate) # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 6️⃣ Heavy contraction – use cuBLAS (batched GEMM)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) where H = hidden_dim
    left_reshape  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_reshape = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    # collapse batch*head
    left_mat  = left_reshape.view(B * hidden_dim, N, N)      # (B*H, N, N)
    right_mat = right_reshape.view(B * hidden_dim, N, N)     # (B*H, N, N)

    # batched matrix multiply (fp16×fp16 → fp16, cuBLAS Tensor Cores)
    out_mat = torch.bmm(left_mat, right_mat)                # (B*H, N, N)

    # restore shape (B, H, N, N) → (B, N, N, H)
    out_fp16 = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_fp16.to(torch.float32)                        # back to fp32

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm (fp32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 8️⃣ Final gating + linear projection (fp32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                   # (B,N,N,dim)

    return out
