"""
Fast “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Manual LayerNorm on the input (FP32) – cheap and needs no Triton.
2️⃣  All linear layers are executed with torch.nn.functional.linear in FP16.
3️⃣  A small Triton element‑wise kernel fuses:
        * optional pair‑wise mask
        * left‑gate   (left  *= sigmoid(left_gate))
        * right‑gate  (right *= sigmoid(right_gate))
   This removes three separate launches in the reference implementation.
4️⃣  The cubic contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is performed as a batched GEMM (B·H, N, N) × (B·H, N, N)ᵀ
   using torch.bmm on FP16 tensors – cuBLAS/Tensor‑cores give the best
   performance on Hopper.
5️⃣  The result is reshaped back, LayerNorm‑2 (FP32), multiplied by the
   output‑gate and finally projected to the original dimension (FP32).

Only the element‑wise part is written in Triton; all heavy matrix work stays
on cuBLAS, which is the fastest option on an H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – element‑wise mask & gating (FP16)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    left_ptr,                     # fp16 * (B,N,N,H)
    right_ptr,                    # fp16 * (B,N,N,H)
    left_gate_ptr,                # fp16 * (B,N,N,H)
    right_gate_ptr,               # fp16 * (B,N,N,H)
    mask_ptr,                     # fp16 * (B,N,N,1)  or nullptr
    B, N, H,                      # extents
    stride_lz, stride_ly, stride_lx, stride_lc,   # left  strides (z,y,x,c)
    stride_rz, stride_ry, stride_rx, stride_rc,   # right strides
    stride_gz, stride_gy, stride_gx, stride_gc,   # gate  strides
    stride_mz, stride_my, stride_mx,               # mask  strides (no channel dim)
    HAVE_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N * H
    offs = pid * 128 + tl.arange(0, 128)                # 128‑wide vector
    mask = offs < total

    # compute multi‑dim indices
    idx = offs
    c = idx % H
    idx = idx // H
    x = idx % N
    idx = idx // N
    y = idx % N
    b = idx // N

    # linear offsets for each tensor
    left_off  = b * stride_lz + y * stride_ly + x * stride_lx + c * stride_lc
    right_off = b * stride_rz + y * stride_ry + x * stride_rx + c * stride_rc
    lgate_off = b * stride_gz + y * stride_gy + x * stride_gx + c * stride_gc
    rgate_off = lgate_off                                 # same layout

    left  = tl.load(left_ptr  + left_off,  mask=mask, other=0.0)
    right = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + lgate_off, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + rgate_off, mask=mask, other=0.0)

    # optional mask (broadcast over channel dim)
    if HAVE_MASK:
        # mask layout is (B,N,N,1) → same y,x,b indices, no channel dim
        mask_off = b * stride_mz + y * stride_my + x * stride_mx
        m = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)
        left  = left * m
        right = right * m

    # gating (sigmoid is computed in FP32 for stability, then cast back)
    lgate = tl.exp(lgate) / (1.0 + tl.exp(lgate))
    rgate = tl.exp(rgate) / (1.0 + tl.exp(rgate))

    left  = left  * lgate
    right = right * rgate

    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask):
    """
    Fuse optional mask and the three gates in‑place (FP16).
    left/right      : (B,N,N,H)  fp16 contiguous
    left_gate/right_gate : (B,N,N,H) fp16 (already sigmoid‑ed)
    mask            : (B,N,N)   fp16 or None
    """
    B, N, _, H = left.shape
    assert right.shape == left.shape
    assert left_gate.shape == left.shape
    assert right_gate.shape == left.shape

    stride_l = left.stride()
    stride_r = right.stride()
    stride_g = left_gate.stride()
    stride_m = (0, 0, 0, 0) if mask is None else mask.stride()

    HAVE_MASK = 1 if mask is not None else 0

    total = B * N * N * H
    BLOCK = 128
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask if mask is not None else left,   # dummy ptr when not used
        B=B, N=N, H=H,
        stride_lz=stride_l[0], stride_ly=stride_l[1],
        stride_lx=stride_l[2], stride_lc=stride_l[3],
        stride_rz=stride_r[0], stride_ry=stride_r[1],
        stride_rx=stride_r[2], stride_rc=stride_r[3],
        stride_gz=stride_g[0], stride_gy=stride_g[1],
        stride_gx=stride_g[2], stride_gc=stride_g[3],
        stride_mz=stride_m[0], stride_my=stride_m[1],
        stride_mx=stride_m[2],
        HAVE_MASK=HAVE_MASK,
        num_warps=4,
    )
    return left, right


# ----------------------------------------------------------------------
# Entry point – forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict with same keys as the reference module
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual – cheap)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                               # (B,N,N,dim) FP32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    def _to_fp16(t): return t.to(torch.float16)

    left_proj_w  = _to_fp16(weights["left_proj.weight"])   # (hidden_dim, dim)
    right_proj_w = _to_fp16(weights["right_proj.weight"])
    left_gate_w  = _to_fp16(weights["left_gate.weight"])
    right_gate_w = _to_fp16(weights["right_gate.weight"])
    out_gate_w   = _to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                         # (B,N,N,dim)

    left = F.linear(x_fp16, left_proj_w)                # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()   # (B,N,N,hidden_dim)
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()     # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Mask + gating – fused in Triton (in‑place on left/right)
    # ------------------------------------------------------------------
    mask_fp16 = None
    if mask is not None:
        mask_fp16 = mask.to(torch.float16)               # (B,N,N)
    left, right = _apply_mask_and_gate(left, right, left_gate, right_gate, mask_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction – batched GEMM via cuBLAS (fp16)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for left and (B*H, N, N) for rightᵀ
    left_t = left.permute(0, 3, 1, 2).contiguous()        # (B,H,N,N)
    right_t = right.permute(0, 3, 2, 1).contiguous()      # (B,H,N,N)

    B_h, _, _ = left_t.shape[0], left_t.shape[2], left_t.shape[3]
    left_flat = left_t.view(B * hidden_dim, N, N)        # (B*H,N,N)
    right_flat = right_t.view(B * hidden_dim, N, N)      # (B*H,N,N)

    # batched matrix multiply: (B*H,N,N) @ (B*H,N,N)^T -> (B*H,N,N)
    out_fp16 = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out_fp16 = out_fp16.view(B, hidden_dim, N, N)        # (B,H,N,N)

    # back to (B,N,N,dim)
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 6️⃣ Final gating + projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
