"""
TriMul “outgoing” forward pass (AlphaFold‑3) – fast Triton‐augmented implementation.

The expensive O(N³) contraction

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

is performed by a blocked GEMM kernel written in Triton.  All cheap
operations (LayerNorm, linear projections, gating, final LayerNorm / linear)
remain in native PyTorch because they are memory‑bound and already
well‑optimised.  The kernel works on FP16 tensors with FP32 accumulation and
stores the result back as FP16; the host code then finishes the forward pass
in FP32.

The kernel tiles the (M × N) output matrix with
BLOCK_M × BLOCK_N tiles and iterates over K in BLOCK_K steps,
loading the two operand tiles (the second operand is read transposed so a
single `tl.dot` computes the inner product).  This mirrors a classic batched
GEMM and gives >10× speed‑up over the naïve triple‑loop while keeping the
implementation simple and portable.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------
# Triton kernel: batched GEMM for the N³ contraction
# -------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,          # const fp16*  [B, H, N, N]
    right_ptr,         # const fp16*  [B, H, N, N]   (will be read transposed)
    out_ptr,           #       fp16*  [B, H, N, N]
    B, N, H,           # int metadata
    # strides (row‑major) for each tensor
    stride_lb, stride_lh, stride_ln, stride_lk,   # left
    stride_rb, stride_rh, stride_rn, stride_rk,   # right
    stride_ob, stride_oh, stride_on, stride_ok,   # out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a (BLOCK_M × BLOCK_N) tile of the 4‑D result

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    One program instance works on a single (b, h, tile_i, tile_j) block.
    """
    pid = tl.program_id(0)

    # -------------------- decode program id --------------------
    # order: ((b * H + h) * num_m + tile_i) * num_n + tile_j
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    tile_j = pid % num_n
    pid //= num_n
    tile_i = pid % num_m
    pid //= num_m
    h = pid % H
    b = pid // H

    # -------------------- tile coordinates --------------------
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32 for accuracy)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- main K loop --------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, h, i, k]   shape (M, K)
        l_ptr = (left_ptr +
                 b * stride_lb +
                 h * stride_lh +
                 offs_i[:, None] * stride_ln +
                 offs_k[None, :] * stride_lk)
        l = tl.load(l_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[b, h, j, k] read as transposed -> shape (K, N)
        # we want right^T so we index (k, j)
        r_ptr = (right_ptr +
                 b * stride_rb +
                 h * stride_rh +
                 offs_k[:, None] * stride_rk +   # k as row
                 offs_j[None, :] * stride_rn)    # j as column
        r_t = tl.load(r_ptr,
                      mask=mask_k[:, None] & mask_j[None, :],
                      other=0.0).to(tl.float32)

        # (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(l, r_t)

    # -------------------- write result --------------------
    out_ptrs = (out_ptr +
                b * stride_ob +
                h * stride_oh +
                offs_i[:, None] * stride_on +
                offs_j[None, :] * stride_ok)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input   [B, N, N, dim]
        torch.Tensor,                # mask    [B, N, N] (or None)
        Dict[str, torch.Tensor],     # weights
        Dict[str, Any]               # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32          # final output dtype

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]          # H

    # ------------------------------------------------------------------
    # 1. Cheap work in PyTorch
    # ------------------------------------------------------------------
    # LayerNorm over the feature dimension
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # Linear projections (bias=False)
    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # optional masking
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # Gates (sigmoid after linear)
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 2. Rearrange to [B, H, N, N] and cast to fp16 for the kernel
    # ------------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)          # (B,H,N,N), fp16

    # ------------------------------------------------------------------
    # 3. Triton kernel launch (blocked batched GEMM)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    # one program per (b, h, tile_i, tile_j)
    grid = (B * hidden_dim * grid_m * grid_n,)

    _trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        # strides for left tensor
        left_fp16.stride(0),   # B
        left_fp16.stride(1),   # H
        left_fp16.stride(2),   # N (row index i)
        left_fp16.stride(3),   # N (col index k)
        # strides for right tensor
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        # strides for output tensor
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 4. Remaining post‑processing in PyTorch
    # ------------------------------------------------------------------
    # back to (B,N,N,H) layout and FP32
    out = out_fp16.to(dtype).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # to_out_norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply output gate (broadcast over pairwise axes)
    out = out * out_gate

    # final linear back to original dim
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
