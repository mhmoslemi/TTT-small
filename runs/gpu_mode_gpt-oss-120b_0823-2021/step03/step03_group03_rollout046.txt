"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  Manual LayerNorm on the input (FP32, cheap, no extra kernel).
2️⃣  Linear projections + gate‑pre‑activations are performed in FP16 with
    torch.nn.functional.linear (uses cuBLAS/Tensor‑cores).
3️⃣  A tiny Triton element‑wise kernel fuses
        • optional pair‑wise mask,
        • left‑gate  = sigmoid(left_gate_pre)  * left,
        • right‑gate = sigmoid(right_gate_pre) * right,
        • out_gate   = sigmoid(out_gate_pre)   (saved for later).
   This kernel touches every element exactly once and therefore gives a
   measurable speed‑up over a pure‑Python implementation.
4️⃣  The cubic contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is reshaped to a batched GEMM (B·H, N, N) and executed with
    torch.bmm, i.e. cuBLAS/Tensor‑cores (fastest on H200).
5️⃣  The result is transposed back, LayerNorm’ed, multiplied by the
    output‑gate and finally projected back to the original dimension.

Only the element‑wise mask‑gate fusion is written in Triton – all other
operations stay in PyTorch for simplicity and maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel – elementwise mask & gating fusion
# -------------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # FP16  (B, N, N, H)
    right_ptr,         # FP16  (B, N, N, H)
    left_gate_ptr,     # FP16  (B, N, N, H) – pre‑sigmoid
    right_gate_ptr,    # FP16  (B, N, N, H) – pre‑sigmoid
    out_gate_ptr,      # FP16  (B, N, N, H) – pre‑sigmoid (saved)
    mask_ptr,          # FP16  (B, N, N, 1) or nullptr when no mask
    B, N, H,
    stride_left_b, stride_left_n1, stride_left_n2, stride_left_h,
    stride_right_b, stride_right_n1, stride_right_n2, stride_right_h,
    stride_lgate_b, stride_lgate_n1, stride_lgate_n2, stride_lgate_h,
    stride_rgate_b, stride_rgate_n1, stride_rgate_n2, stride_rgate_h,
    stride_ogate_b, stride_ogate_n1, stride_ogate_n2, stride_ogate_h,
    stride_mask_b, stride_mask_n1, stride_mask_n2, stride_mask_h,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N * H
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offset < total

    # linear index → (b, i, j, h)
    tmp = offset
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    i = (tmp // N) % N
    b = tmp // (N * N)

    # compute pointers
    off_left = (
        b * stride_left_b
        + i * stride_left_n1
        + j * stride_left_n2
        + h * stride_left_h
    )
    off_right = (
        b * stride_right_b
        + i * stride_right_n1
        + j * stride_right_n2
        + h * stride_right_h
    )
    off_lgate = (
        b * stride_lgate_b
        + i * stride_lgate_n1
        + j * stride_lgate_n2
        + h * stride_lgate_h
    )
    off_rgate = (
        b * stride_rgate_b
        + i * stride_rgate_n1
        + j * stride_rgate_n2
        + h * stride_rgate_h
    )
    off_ogate = (
        b * stride_ogate_b
        + i * stride_ogate_n1
        + j * stride_ogate_n2
        + h * stride_ogate_h
    )
    if HAS_MASK:
        off_mask = (
            b * stride_mask_b
            + i * stride_mask_n1
            + j * stride_mask_n2
            + 0 * stride_mask_h
        )
        mask_val = tl.load(mask_ptr + off_mask, mask=mask, other=0.0)
    else:
        mask_val = 1.0

    # load values
    left   = tl.load(left_ptr   + off_left,   mask=mask, other=0.0)
    right  = tl.load(right_ptr  + off_right,  mask=mask, other=0.0)
    lgate  = tl.load(left_gate_ptr  + off_lgate,   mask=mask, other=0.0)
    rgate  = tl.load(right_gate_ptr + off_rgate,  mask=mask, other=0.0)
    ogate  = tl.load(out_gate_ptr   + off_ogate,   mask=mask, other=0.0)

    # apply mask
    left  = left * mask_val
    right = right * mask_val

    # gating
    left  = left  * tl.sigmoid(lgate)
    right = right * tl.sigmoid(rgate)
    ogate = tl.sigmoid(ogate)

    # store back (in‑place)
    tl.store(left_ptr   + off_left,   left,  mask=mask)
    tl.store(right_ptr  + off_right,  right, mask=mask)
    tl.store(out_gate_ptr + off_ogate, ogate, mask=mask)


def _fuse_mask_and_gates(left, right, left_gate_pre, right_gate_pre,
                         out_gate_pre, mask):
    """
    left, right, left_gate_pre, right_gate_pre, out_gate_pre : FP16 tensors
        shape (B, N, N, H)
    mask : FP16 tensor (B, N, N, 1) or None
    Returns
    -------
    left, right : FP16 tensors after mask & gate
    out_gate    : FP16 tensor (B, N, N, H) – sigmoid(out_gate_pre)
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 256
    grid = (triton.cdiv(total, BLOCK),)

    # strides
    stride_left = left.stride()
    stride_right = right.stride()
    stride_lgate = left_gate_pre.stride()
    stride_rgate = right_gate_pre.stride()
    stride_ogate = out_gate_pre.stride()
    if mask is None:
        HAS_MASK = False
        mask_ptr = torch.tensor([], dtype=torch.float16, device=left.device)
        stride_mask = (0, 0, 0, 0)
    else:
        HAS_MASK = True
        mask_ptr = mask
        stride_mask = mask.stride()

    _mask_gate_fusion_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate_pre,
        right_gate_ptr=right_gate_pre,
        out_gate_ptr=out_gate_pre,
        mask_ptr=mask_ptr,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left[0],
        stride_left_n1=stride_left[1],
        stride_left_n2=stride_left[2],
        stride_left_h=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_n1=stride_right[1],
        stride_right_n2=stride_right[2],
        stride_right_h=stride_right[3],
        stride_lgate_b=stride_lgate[0],
        stride_lgate_n1=stride_lgate[1],
        stride_lgate_n2=stride_lgate[2],
        stride_lgate_h=stride_lgate[3],
        stride_rgate_b=stride_rgate[0],
        stride_rgate_n1=stride_rgate[1],
        stride_rgate_n2=stride_rgate[2],
        stride_rgate_h=stride_rgate[3],
        stride_ogate_b=stride_ogate[0],
        stride_ogate_n1=stride_ogate[1],
        stride_ogate_n2=stride_ogate[2],
        stride_ogate_h=stride_ogate[3],
        stride_mask_b=stride_mask[0],
        stride_mask_n1=stride_mask[1],
        stride_mask_n2=stride_mask[2],
        stride_mask_h=stride_mask[3],
        HAS_MASK=HAS_MASK,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # out_gate_pre now holds sigmoid(out_gate_pre)
    return left, right, out_gate_pre


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradient).

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool or float)
        weights      : dict of leaf weight tensors
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                                   # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gate pre‑activations (FP16)
    # --------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16)

    # projection weights
    w_left  = to_fp16(weights["left_proj.weight"])   # (hidden, dim)
    w_right = to_fp16(weights["right_proj.weight"])

    # gate weights
    w_lgate = to_fp16(weights["left_gate.weight"])
    w_rgate = to_fp16(weights["right_gate.weight"])
    w_ogate = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                      # (B,N,N,dim)

    left      = F.linear(x_fp16, w_left)              # (B,N,N,hidden)
    right     = F.linear(x_fp16, w_right)

    left_gate_pre  = F.linear(x_fp16, w_lgate)        # (B,N,N,hidden)
    right_gate_pre = F.linear(x_fp16, w_rgate)
    out_gate_pre   = F.linear(x_fp16, w_ogate)

    # --------------------------------------------------------------
    # 3️⃣ Mask + gating fusion (Triton)
    # --------------------------------------------------------------
    if mask is None:
        mask_fp16 = None
    else:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)

    left, right, out_gate = _fuse_mask_and_gates(
        left, right, left_gate_pre, right_gate_pre, out_gate_pre, mask_fp16
    )   # all remain FP16

    # --------------------------------------------------------------
    # 4️⃣ Cubic contraction: batched GEMM via torch.bmm (cuBLAS)
    # --------------------------------------------------------------
    # reshape to (B*hidden, N, N)
    left_  = left.permute(0, 3, 1, 2).contiguous()      # (B, H, I, K)
    right_ = right.permute(0, 3, 2, 1).contiguous()     # (B, H, K, J)

    BxH, I, K = left_.shape
    _, K2, J = right_.shape
    assert K == K2

    left_flat  = left_.view(BxH, I, K)
    right_flat = right_.view(BxH, K, J)

    # batched matmul in FP16, accumulate in FP32 inside torch
    out_flat = torch.bmm(left_flat, right_flat)          # (B*H, I, J) fp16

    out_fp16 = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_fp16.to(torch.float32)                     # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 5️⃣ Output LayerNorm (FP32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    m2 = out.mean(-1, keepdim=True)
    v2 = out.var(-1, unbiased=False, keepdim=True)
    out = (out - m2) / torch.sqrt(v2 + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 6️⃣ Final gating + projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)              # broadcast over hidden
    out = F.linear(out, weights["to_out.weight"])       # (B,N,N,dim)

    return out
