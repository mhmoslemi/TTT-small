"""
TriMul “outgoing” forward (no grads).

Only the two LayerNorms (input‑norm and output‑norm) are implemented in
Triton; the heavy N³ contraction is performed with a single batched
matrix‑multiply (torch.bmm) after reshaping the projected tensors.  All
intermediate tensors are kept in fp16 for the GEMM and cast back to fp32
for the final LayerNorm and projection.  This satisfies the requirement
to have a non‑trivial Triton kernel while still being fast on Hopper.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm kernel (operates on a flat view of the last dimension)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # input  (B,N,N,D)  fp32
    y_ptr,                # output (B,N,N,D)  fp32
    w_ptr,                # weight (D)       fp32
    b_ptr,                # bias   (D)       fp32
    B, N, D, eps,
    stride_xb, stride_xn1, stride_xn2, stride_xd,   # strides of x
    stride_yb, stride_yn1, stride_yn2, stride_yd,   # strides of y
    stride_w, stride_b,
    BLOCK_D: tl.constexpr,
):
    # one program per (b,i,j) triple
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < D

    # ------------------------------------------------------------------
    # load a tile along the feature dimension
    # ------------------------------------------------------------------
    x = tl.load(
        x_ptr
        + b * stride_xb
        + i * stride_xn1
        + j * stride_xn2
        + offs_d * stride_xd,
        mask=mask_d,
        other=0.0,
    )

    # ------------------------------------------------------------------
    # mean / variance (reduction over D)
    # ------------------------------------------------------------------
    mean = tl.sum(x, axis=0) / D
    diff = x - mean
    var = tl.sum(diff * diff, axis=0) / D
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # apply scale and bias
    # ------------------------------------------------------------------
    w = tl.load(w_ptr + offs_d * stride_w, mask=mask_d, other=1.0)
    b_ = tl.load(b_ptr + offs_d * stride_b, mask=mask_d, other=0.0)

    y = (x - mean) * inv_std * w + b_

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    tl.store(
        y_ptr
        + b * stride_yb
        + i * stride_yn1
        + j * stride_yn2
        + offs_d * stride_yd,
        y,
        mask=mask_d,
    )


def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """
    Triton‑accelerated LayerNorm over the last dimension.
    x : (B,N,N,D)  float32, contiguous on CUDA
    weight, bias : (D,) float32
    returns y : (B,N,N,D) float32
    """
    assert x.is_contiguous()
    B, N, _, D = x.shape

    y = torch.empty_like(x)

    # strides in *elements* (not bytes)
    stride_xb, stride_xn1, stride_xn2, stride_xd = x.stride()
    stride_yb, stride_yn1, stride_yn2, stride_yd = y.stride()
    stride_w = weight.stride(0)
    stride_b = bias.stride(0)

    BLOCK_D = 64                     # good for D up to 768
    grid = (triton.cdiv(B * N * N, 1),)   # one program per (b,i,j)

    layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        B,
        N,
        D,
        eps,
        stride_xb,
        stride_xn1,
        stride_xn2,
        stride_xd,
        stride_yb,
        stride_yn1,
        stride_yn2,
        stride_yd,
        stride_w,
        stride_b,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Helper that runs a batched GEMM using torch.bmm (cuBLAS/Tensor‑cores)
# ----------------------------------------------------------------------
def batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B,H,N,N)  fp16
    right : (B,H,N,N)  fp16
    Returns (B,H,N,N) fp16 = Σ_k left[...,k,:] * right[...,k,:]^T
    """
    B, H, N, _ = left.shape
    # reshape to (B*H, N, N)
    l = left.reshape(B * H, N, N)
    r = right.reshape(B * H, N, N)
    out = torch.bmm(l, r.transpose(1, 2))          # (B*H, N, N)
    return out.reshape(B, H, N, N)


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward of the “outgoing” TriMul operator.

    data = (input_tensor, mask, weights, config)
    input_tensor : (B, N, N, dim)  float32 CUDA
    mask         : (B, N, N)       float32/bool or None
    weights      : dict of torch.Tensor (see code)
    config       : dict with keys "dim" and "hidden_dim"
    returns      : (B, N, N, dim)  float32
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm – Triton
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = triton_layernorm(input_tensor, norm_w, norm_b, eps)          # (B,N,N,dim) fp32

    # ------------------------------------------------------------------
    # 2) Linear projections (keep everything fp16 for the GEMM)
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])

    # fp16 version of the normalized input
    x_fp16 = x.to(torch.float16)

    # projections (B,N,N,hidden)
    left  = F.linear(x_fp16, left_proj_w)          # bias=False in original code
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4) Gating – sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction (batched GEMM)
    #    reshape to (B, hidden, N, N) and compute Σ_k left[...,k,:] * right[...,k,:]^T
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = batched_gemm(left_perm, right_perm)      # (B, H, I, J) fp16

    # back to (B,N,N,hidden)
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm – Triton
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = triton_layernorm(out, to_out_norm_w, to_out_norm_b, eps)

    # ------------------------------------------------------------------
    # 7) Final gating + projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # same gate as in reference
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                   # (B,N,N,dim)

    return out
