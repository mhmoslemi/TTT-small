"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm (GPU‑friendly)
------------------------
1️⃣  Layer‑norm the 4‑D input  (B,N,N,C)   →   FP32.
2️⃣  Linear projections for *left*, *right* and three gates.
3️⃣  Fuse three element‑wise ops in a tiny Triton kernel:
        left_fused  = left  * left_gate  * mask
        right_fused = right * right_gate * mask
    (mask is optional).  The kernel works on a 1‑D grid and keeps the data
    in FP16, which is the format required by the downstream batched GEMM.
4️⃣  Reshape ``left_fused`` / ``right_fused`` to (B*H, N, N) and perform the
    O(N³) contraction with a single batched matrix‑multiply
        out[b,h] = left_fused[b,h] @ right_fused[b,h].T
    This call uses cuBLAS/Tensor‑cores (FP16→FP16, accumulation in FP32).
5️⃣  Layer‑norm the hidden result (dim = H), multiply by ``out_gate`` and
    project back to the original channel dimension C with a final linear.

Only the element‑wise fusion (step 3) is performed in Triton – this satisfies
the “some‑kernel” requirement while keeping the dominant cubic work on the
highly‑optimised cuBLAS path, which is essential for the large sequence
lengths used in AlphaFold‑3.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – apply mask (if present) and gate tensors
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # fp16 * (B,N,N,H)
    right_ptr,              # fp16 * (B,N,N,H)
    left_gate_ptr,          # fp16 * (B,N,N,H)
    right_gate_ptr,         # fp16 * (B,N,N,H)
    mask_ptr,               # fp16 * (B,N,N,1)   – may be nullptr
    out_left_ptr,           # fp16 * (B,N,N,H)
    out_right_ptr,          # fp16 * (B,N,N,H)
    B, N, H,                # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    The kernel processes a flattened view of the 4‑D tensor.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # ---- decode linear index (b,i,j,h) ----
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ---- load left / right ----
    idx = (
        b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    )
    left = tl.load(left_ptr + idx, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + idx, mask=mask, other=tl.float16(0.0))

    # ---- load gates ----
    idx_gate = (
        b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + h * stride_lg3
    )
    lgate = tl.load(left_gate_ptr + idx_gate, mask=mask, other=tl.float16(0.0))
    rgate = tl.load(right_gate_ptr + idx_gate, mask=mask, other=tl.float16(0.0))

    # ---- load mask (if any) ----
    if HAS_MASK:
        idx_mask = b * stride_m0 + i * stride_m1 + j * stride_m2  # no H dim
        m = tl.load(mask_ptr + idx_mask, mask=mask, other=tl.float16(1.0))
    else:
        m = tl.float16(1.0)

    # ---- fuse ----
    out_l = left * lgate * m
    out_r = right * rgate * m

    # ---- store ----
    tl.store(out_left_ptr + idx, out_l, mask=mask)
    tl.store(out_right_ptr + idx, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        x          – (B, N, N, C)    torch.float32
        mask       – (B, N, N)       torch.bool/float32/None
        weights    – dict of FP32 parameters (see body)
        config     – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor
        (B, N, N, C)   torch.float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, C = x.shape
    H = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → keep numerical stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (still FP32 for now)
    # --------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))   # (B,N,N,H)

    # --------------------------------------------------------------
    # 3️⃣ Cast to FP16 and fuse mask+gates with Triton
    # --------------------------------------------------------------
    left_fp16      = left.to(torch.float16).contiguous()
    right_fp16     = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate.to(torch.float16).contiguous()
    right_gate_fp16= right_gate.to(torch.float16).contiguous()

    if mask is None:
        # dummy mask – kernel will ignore it (HAS_MASK=False)
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask = False
    else:
        # broadcast mask over the hidden dimension H
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        has_mask = True

    # allocate output buffers
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elements = B * N * N * H
    BLOCK = 1024                       # good trade‑off on H200
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        # mask strides – last dim is 1 (broadcasted) when present
        mask_fp16.stride(0) if has_mask else 0,
        mask_fp16.stride(1) if has_mask else 0,
        mask_fp16.stride(2) if has_mask else 0,
        mask_fp16.stride(3) if has_mask else 0,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM on the hidden dimension (FP16 → FP16, acc=FP32)
    # --------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N).T   →  (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype)   # back to FP32 for the next LayerNorm

    # --------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    # out_gate has shape (B,N,N,H) already in FP32; broadcast over (N,N)
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
```

**Explanation of design choices**

* **Why only one Triton kernel?**  
  The cubic contraction `∑_k left[...,k,:] * right[...,k,:]` dominates runtime.  
  Dispatching this to cuBLAS (batched `bmm`) gives Tensor‑core utilization on the
  Hopper H200 and far outperforms any handcrafted Triton GEMM.  
  The problem statement only requires “some‑kernel”, so the element‑wise mask‑gate
  fusion is a perfect, low‑overhead candidate.

* **Mask handling** – The kernel receives a pointer to a mask tensor of shape
  `(B,N,N,1)`. When `HAS_MASK=False` the pointer may be a zero‑size tensor; the
  kernel simply uses `m = 1.0`, eliminating any extra loads.

* **Strides** – All tensors are contiguous after the `contiguous()` calls, thus
  the stride arguments are directly the `tensor.stride(dim)` values. The kernel
  decodes the flat index into `(b,i,j,h)` exactly matching PyTorch’s default
  contiguous layout (`b` fastest, then `i`, then `j`, then `h`), guaranteeing
  correctness irrespective of the input shape.

* **Precision** – All linear layers and LayerNorms are performed in FP32 for
  numerical stability, matching the reference implementation.  The heavy matrix
  multiplication uses FP16 inputs (the fused tensors) with FP32 accumulation
  (`torch.bmm` does this automatically on Hopper), which is the fastest path.

* **Performance** –  
  * `BLOCK=1024` yields ~1 Mi elements per kernel launch, saturating the H200’s
    memory bandwidth while keeping register pressure low.  
  * The batched GEMM is a single `torch.bmm`, so the kernel launch overhead is
    negligible compared with the O(N³) work.

The function `custom_kernel` adheres to the required signature and returns a
float‑32 tensor of shape `[bs, seqlen, seqlen, dim]`.
