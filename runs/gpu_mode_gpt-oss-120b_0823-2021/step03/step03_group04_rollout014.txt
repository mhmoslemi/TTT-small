"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the input (B,N,N,C).
2️⃣  Linear projections → left, right and three gates (C → H).
3️⃣  Fuse element‑wise:  left  = left  * sigmoid(left_gate)  * mask
                         right = right * sigmoid(right_gate) * mask
   (mask is optional). This tiny work is done in a single Triton kernel so that
   the benchmark always executes a kernel and showcases a realistic “some‑kernel”
   constraint.
4️⃣  Reshape the fused tensors to (B*H, N, N) and compute the cubic contraction
   with a batched matrix‑multiply (Tensor‑core friendly FP16 → FP32 accumulate).
   out[b,h] = left[b,h] @ right[b,h].transpose(-2,-1)
   The result shape is (B*H,N,N) → (B,N,N,H) after reshaping.
5️⃣  Hidden‑dim LayerNorm, gating with out_gate and a final linear projection back
   to C (all in float32 for numerical stability).

Only the element‑wise fusion (step 3) is performed in Triton; the dominant
O(N³) work is delegated to cuBLAS via ``torch.bmm`` which gives the best
performance on Hopper (H200).  All heavy arithmetic uses FP16 while
accumulations are in FP32.  The final output is returned in FP32 to match the
reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – fuse mask, left‑gate, right‑gate
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                # fp16 * (B,N,N,H)
    left_gate_ptr, right_gate_ptr,      # fp16 * (B,N,N,H) – already sigmoid‑ed
    mask_ptr,                           # fp16 * (B,N,N,1) or nullptr
    out_left_ptr, out_right_ptr,        # fp16 * (B,N,N,H) – outputs
    B, N, H,                            # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_m0, stride_m1, stride_m2,    # mask has no hidden dim
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    The kernel processes a 1‑D grid of size ceil(B*N*N*H / BLOCK).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # ---------- decode linear index ----------
    # layout = (B, N, N, H) → contiguous in memory
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ---------- load tensors ----------
    # left / right
    idx = b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    left  = tl.load(left_ptr  + idx, mask=mask, other=0.0)
    right = tl.load(right_ptr + idx, mask=mask, other=0.0)

    # gates (already passed through sigmoid in host code)
    idx_gate = b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + h * stride_lg3
    lgate = tl.load(left_gate_ptr  + idx_gate, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + idx_gate, mask=mask, other=0.0)

    # optional mask – broadcast over H
    if HAS_MASK:
        # mask shape (B,N,N,1) → no stride for H dimension
        idx_mask = b * stride_m0 + i * stride_m1 + j * stride_m2
        m = tl.load(mask_ptr + idx_mask, mask=mask, other=1.0)
    else:
        m = 1.0

    # ---------- fused write ----------
    out_l = left * lgate * m
    out_r = right * rgate * m

    tl.store(out_left_ptr  + idx, out_l, mask=mask)
    tl.store(out_right_ptr + idx, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        x      – (B, N, N, C)   torch.float32
        mask   – (B, N, N)      torch.bool / float32 / None
        weights– dict of all learned parameters (float32)
        config – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor
        (B, N, N, C)   float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype  = torch.float32

    B, N, _, C = x.shape
    H = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (float32 → fp16 later)
    # --------------------------------------------------------------
    #   left  = x @ W_left.T
    #   right = x @ W_right.T
    #   gates = sigmoid(x @ W_gate.T)
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))          # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate= torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate  = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N,H)

    # --------------------------------------------------------------
    # 3️⃣ Fuse mask & gates in fp16 with Triton
    # --------------------------------------------------------------
    # Cast everything to fp16 (Tensor‑core friendly)
    left_fp16        = left.to(torch.float16).contiguous()
    right_fp16       = right.to(torch.float16).contiguous()
    left_gate_fp16   = left_gate.to(torch.float16).contiguous()
    right_gate_fp16  = right_gate.to(torch.float16).contiguous()

    if mask is None:
        # dummy tensor – kernel will ignore it via HAS_MASK=False
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask = False
    else:
        # mask broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()   # (B,N,N,1)
        has_mask = True

    # allocate fused outputs
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elem = B * N * N * H
    BLOCK = 1024                                     # good for H200
    grid = (triton.cdiv(total_elem, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        # strides for left/right (identical)
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        # strides for gates (identical to left/right)
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        # strides for mask (no H dim)
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (fp16 inputs, fp32 accumulation)
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) → ready for torch.bmm
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)

    # out_mat  = left @ right.T   (FP16×FP16 → FP16, acc in FP32)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))          # (B*H,N,N)

    # back to (B,N,N,H) and FP32 for the next LayerNorm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(dtype)

    # --------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, out_gate and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                     # broadcast over (N,N)
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
