"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (high‑level)
----------------------
1️⃣  Manual FP32 LayerNorm on the last dimension.
2️⃣  Cast the normalized tensor to FP16 and compute the four linear layers
    (two projections + three gates) with torch.nn.functional.linear.
3️⃣  Fuse *mask* and *gating* (left‑gate, right‑gate) in a tiny Triton kernel
    that works on the contiguous layout (B, N, N, H).  This avoids an extra
    launch for each tensor and keeps the element‑wise work on‑GPU.
4️⃣  Reshape the gated tensors so that the cubic contraction becomes a batched
    matrix‑multiply:
        left  : (B, H, N, N)   → (B*H, N, N)
        right : (B, H, N, N)ᵀ → (B*H, N, N)
    The heavy O(N³) work is performed by torch.bmm (cuBLAS/Tensor‑cores).
5️⃣  Convert the result back to FP32, apply a second FP32 LayerNorm,
    multiply by the output‑gate and finish with the final linear projection.
    The final tensor has shape (B, N, N, dim) and dtype float32.

Only step 3 is executed inside a Triton kernel – this satisfies the
requirement “must implement at least part of the operations in a kernel” while
keeping the overall runtime dominated by the highly‑optimized cuBLAS GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask and gating (element‑wise)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,          # input tensors   (B,N,N,H) fp16
    left_gate_ptr, right_gate_ptr,# gate tensors    (B,N,N,H) fp16
    mask_ptr,                     # mask (or 1.0)   (B,N,N,1) fp16
    out_left_ptr, out_right_ptr, # outputs         (B,N,N,H) fp16
    B, N, H,                      # extents
    BLOCK: tl.constexpr,          # block size (1‑D)
):
    """Element‑wise:  out_left  = left  * mask * left_gate
                     out_right = right * mask * right_gate
    All tensors are contiguous in memory with shape (B,N,N,H) except the mask
    which has a trailing singleton dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # flat index
    total = B * N * N * H

    mask = offs < total

    # compute flat offsets for each tensor (contiguous layout)
    # index = ((b * N + i) * N + j) * H + h
    #   = ((b * N + i) * N + j) * H + h
    #   = (((b * N) + i) * N + j) * H + h
    # we let Triton compute the linear address directly
    left  = tl.load(left_ptr  + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)

    # gate tensors have the same layout
    left_gate  = tl.load(left_gate_ptr  + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # mask has shape (B,N,N,1) -> stride = H (since last dim is 1)
    # we can reuse the same linear index because the last dim is 1.
    # The mask value is broadcasted over H.
    m = tl.load(mask_ptr + (offs // H), mask=mask, other=1.0)

    out_l = left  * m * left_gate
    out_r = right * m * right_gate

    tl.store(out_left_ptr  + offs, out_l, mask=mask)
    tl.store(out_right_ptr + offs, out_r, mask=mask)


# ----------------------------------------------------------------------
# Helper – launch the element‑wise Triton kernel
# ----------------------------------------------------------------------
def _apply_mask_and_gate(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor | None,
) -> tuple[torch.Tensor, torch.Tensor]:
    """
    All tensors are fp16, contiguous, shape (B,N,N,H).
    mask is optional, shape (B,N,N) or (B,N,N,1).  If None a mask of 1.0 is used.
    Returns gated left/right tensors (still fp16, contiguous).
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024  # good occupancy for 1‑D launches

    # output buffers
    out_left = torch.empty_like(left)
    out_right = torch.empty_like(right)

    # ensure mask has shape (B,N,N,1) and is fp16
    if mask is None:
        mask_tensor = torch.full((B, N, N, 1), 1.0, dtype=torch.float16, device=left.device)
    else:
        mask_tensor = mask.to(torch.float16).reshape(B, N, N, 1)

    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_tensor,
        out_left_ptr=out_left,
        out_right_ptr=out_right,
        B=B,
        N=N,
        H=H,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out_left, out_right


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf weight tensors (see reference implementation)
        config       : dict containing keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual – cheap, no extra kernel)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gates (FP16)
    # --------------------------------------------------------------
    # cast weights once
    def _fp16(t): return t.to(torch.float16)

    left_proj_w   = _fp16(weights["left_proj.weight"])    # (hidden_dim, dim)
    right_proj_w  = _fp16(weights["right_proj.weight"])
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)  # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 3️⃣ Fuse mask + gating (Triton)
    # --------------------------------------------------------------
    left, right = _apply_mask_and_gate(
        left, right, left_gate, right_gate, mask
    )   # still fp16, shape (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Core contraction – batched GEMM via cuBLAS (fast)
    # --------------------------------------------------------------
    # reshape to (B, H, N, N) then collapse batch & head
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, K) with K=N
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J) with J=N

    BH, _, _, _ = left_t.shape
    left_flat  = left_t.view(B * hidden_dim, N, N)          # (B*H, N, N)
    right_flat = right_t.view(B * hidden_dim, N, N)         # (B*H, N, N)

    # batched matrix multiply: (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    # torch.bmm is heavily tuned on Hopper (Tensor‑cores)
    out_fp16 = torch.bmm(left_flat, right_flat)            # (B*H, N, N)

    # restore original layout (B, N, N, hidden_dim)
    out_fp16 = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_fp16.to(torch.float32)                       # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Output LayerNorm (FP32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 6️⃣ Final gating + projection (FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)            # broadcast over dim
    out = F.linear(out, weights["to_out.weight"])    # (B,N,N,dim)

    return out
