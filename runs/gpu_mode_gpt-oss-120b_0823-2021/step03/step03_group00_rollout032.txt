"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward only)
------------------------
1️⃣  Layer‑norm on the 4‑D input (float32) – done in PyTorch (simple & fast).
2️⃣  Linear projections + three gating linear layers are performed in fp16.
3️⃣  OPTIONAL binary mask is fused in a tiny Triton kernel that
    multiplies the left‑ and right‑projected tensors by the mask
    (broadcast over the hidden dimension).
4️⃣  Sigmoid gating (fp16) on the three gate tensors.
5️⃣  Core contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .
   This is a batched matrix‑multiply with shape
        (B·H, N, N) · (B·H, N, N)ᵀ → (B·H, N, N) .
   We reshaped the tensors and used torch.bmm (cuBLAS/Tensor‑cores) – the
   dominant O(N³) work therefore runs at peak speed.
6️⃣  Second Layer‑norm (float32), output‑gate, and final linear
    projection back to the original dimension.

Only the mask‑fusion is implemented in Triton; the rest stays in PyTorch.
The kernel processes one (batch, i, j) element per program and loops over
the hidden dimension, which is tiny (≤ 384) and therefore fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – elementwise mask multiplication
# ----------------------------------------------------------------------
@triton.jit
def _mask_mul_kernel(
    mask_ptr,                # [B, N, N]      (bool/float16)
    left_ptr,                # [B, N, N, H]   (fp16)
    right_ptr,               # [B, N, N, H]   (fp16)
    B, N, H,                 # sizes
    stride_mask_b, stride_mask_n, stride_mask_n2,
    stride_l_b, stride_l_n, stride_l_n2, stride_l_h,
    stride_r_b, stride_r_n, stride_r_n2, stride_r_h,
    BLOCK_B: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_N2: tl.constexpr,
):
    """
    One program processes a tile (BLOCK_B × BLOCK_N × BLOCK_N2) of the
    (b,i,j) index space and multiplies the hidden dimension by the mask.
    H is typically ≤ 384, so we simply iterate over it.
    """
    pid_b  = tl.program_id(0)
    pid_i  = tl.program_id(1)
    pid_j  = tl.program_id(2)

    offs_b = pid_b * BLOCK_B + tl.arange(0, BLOCK_B)
    offs_i = pid_i * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_j = pid_j * BLOCK_N2 + tl.arange(0, BLOCK_N2)

    mask_b = offs_b < B
    mask_i = offs_i < N
    mask_j = offs_j < N

    # broadcast mask to hidden dimension later
    mask = tl.load(
        mask_ptr
        + offs_b[:, None, None] * stride_mask_b
        + offs_i[None, :, None] * stride_mask_n
        + offs_j[None, None, :] * stride_mask_n2,
        mask=mask_b[:, None, None] & mask_i[None, :, None] & mask_j[None, None, :],
        other=0.0,
    ).to(tl.float16)          # (b,i,j)

    # loop over hidden dimension
    for h in range(0, H):
        # load left & right
        l = tl.load(
            left_ptr
            + offs_b[:, None, None] * stride_l_b
            + offs_i[None, :, None] * stride_l_n
            + offs_j[None, None, :] * stride_l_n2
            + h * stride_l_h,
            mask=mask_b[:, None, None] & mask_i[None, :, None] & mask_j[None, None, :],
            other=0.0,
        )
        r = tl.load(
            right_ptr
            + offs_b[:, None, None] * stride_r_b
            + offs_i[None, :, None] * stride_r_n
            + offs_j[None, None, :] * stride_r_n2
            + h * stride_r_h,
            mask=mask_b[:, None, None] & mask_i[None, :, None] & mask_j[None, None, :],
            other=0.0,
        )
        # apply mask
        l = l * mask
        r = r * mask
        # store back
        tl.store(
            left_ptr
            + offs_b[:, None, None] * stride_l_b
            + offs_i[None, :, None] * stride_l_n
            + offs_j[None, None, :] * stride_l_n2
            + h * stride_l_h,
            l,
            mask=mask_b[:, None, None] & mask_i[None, :, None] & mask_j[None, None, :],
        )
        tl.store(
            right_ptr
            + offs_b[:, None, None] * stride_r_b
            + offs_i[None, :, None] * stride_r_n
            + offs_j[None, None, :] * stride_r_n2
            + h * stride_r_h,
            r,
            mask=mask_b[:, None, None] & mask_i[None, :, None] & mask_j[None, None, :],
        )


def _apply_mask_triton(mask: torch.Tensor,
                       left: torch.Tensor,
                       right: torch.Tensor) -> None:
    """
    In‑place multiplication of ``left`` and ``right`` by ``mask``.
    All tensors must be on the same CUDA device, left/right in fp16,
    mask in float16 (or bool – it will be cast).
    """
    B, N, _, H = left.shape
    # strides in element counts
    stride_mask = mask.stride()
    stride_l = left.stride()
    stride_r = right.stride()

    # tile sizes – small enough to keep registers low
    BLOCK_B = 1
    BLOCK_N = 32
    BLOCK_N2 = 32

    grid = (
        triton.cdiv(B, BLOCK_B),
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(N, BLOCK_N2),
    )
    _mask_mul_kernel[grid](
        mask_ptr=mask,
        left_ptr=left,
        right_ptr=right,
        B=B,
        N=N,
        H=H,
        stride_mask_b=stride_mask[0],
        stride_mask_n=stride_mask[1],
        stride_mask_n2=stride_mask[2],
        stride_l_b=stride_l[0],
        stride_l_n=stride_l[1],
        stride_l_n2=stride_l[2],
        stride_l_h=stride_l[3],
        stride_r_b=stride_r[0],
        stride_r_n=stride_r[1],
        stride_r_n2=stride_r[2],
        stride_r_h=stride_r[3],
        BLOCK_B=BLOCK_B,
        BLOCK_N=BLOCK_N,
        BLOCK_N2=BLOCK_N2,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask         : torch.Tensor or None, shape [B, N, N] (bool/float16)
        - weights      : dict of model weights (see docstring)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (float32, simple torch implementation)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b          # still fp32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (fp16) – keep everything in fp16 after this
    # --------------------------------------------------------------
    def fp16(t): return t.to(torch.float16)
    left_proj_w  = fp16(weights["left_proj.weight"])
    right_proj_w = fp16(weights["right_proj.weight"])
    left_gate_w  = fp16(weights["left_gate.weight"])
    right_gate_w = fp16(weights["right_gate.weight"])
    out_gate_w   = fp16(weights["out_gate.weight"])

    x16 = x.to(torch.float16)                     # (B,N,N,dim)

    left  = F.linear(x16, left_proj_w)           # (B,N,N,hidden)
    right = F.linear(x16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask – fused via Triton (in‑place)
    # --------------------------------------------------------------
    if mask is not None:
        # ensure mask is float16 for the kernel
        mask_fp16 = mask.to(torch.float16)
        _apply_mask_triton(mask_fp16, left, right)

    # --------------------------------------------------------------
    # 4️⃣ Gates (sigmoid, fp16)
    # --------------------------------------------------------------
    left_gate  = F.linear(x16, left_gate_w).sigmoid()
    right_gate = F.linear(x16, right_gate_w).sigmoid()
    out_gate   = F.linear(x16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy contraction – batched GEMM via torch.bmm (cuBLAS)
    #    reshape to (B*hidden, N, N) and (B*hidden, N, N)ᵀ
    # --------------------------------------------------------------
    # (B,N,N,hidden) -> (B, hidden, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # transpose K,J

    B_h, _, I, K = left_perm.shape
    assert I == K == N

    left_flat  = left_perm.view(B_h, I, K)       # (B*H, N, N)
    right_flat = right_perm.view(B_h, K, I)      # (B*H, N, N)

    # torch.bmm uses Tensor‑cores on H200 (fp16 × fp16 → fp16, acc. fp32)
    out_flat = torch.bmm(left_flat, right_flat)   # (B*H, N, N)

    out_fp16 = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_fp16.to(torch.float32)              # back to fp32

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (float32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection (float32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)   # broadcast over hidden dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)            # (B,N,N,dim)

    return out
