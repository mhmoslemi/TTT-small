"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) in FP32.
2. Linear projections (FP32 → FP16) producing tensors
      left, right, left_gate, right_gate, out_gate
   each with shape (B,N,N,H).
3. Fuse mask (if present) and the two gates with a tiny Triton kernel.
   This kernel does *only* element‑wise work, guaranteeing a GPU launch.
4. Reshape the fused tensors to (B*H, N, N) and perform a batched
   matrix‑multiply (FP16 inputs, FP32 accumulation) using torch.bmm,
   which calls cuBLAS/Tensor‑cores and handles the O(N³) contraction.
5. Layer‑norm the hidden result, multiply by out_gate and project back
   to the original channel dimension C with a final linear layer.
All heavy arithmetic is done in FP16 for speed; the final output is
converted back to FP32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – element‑wise fusion of mask & gates
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_and_gate_kernel(
    left_ptr, right_ptr,                     # fp16 * (B,N,N,H)
    left_gate_ptr, right_gate_ptr,           # fp16 * (B,N,N,H)
    mask_ptr,                                # fp16 * (B,N,N,1) or nullptr
    out_left_ptr, out_right_ptr,             # fp16 * (B,N,N,H) – outputs
    B, N, H,                                 # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h):
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    The kernel launches on a 1‑D grid covering B*N*N*H elements.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # ----- decode linear index into (b,i,j,h) -----
    idx = offs
    h = idx % H
    idx = idx // H
    j = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # ----- compute flat offsets for each tensor -----
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    off_rg = b * stride_rg0 + i * stride_rg1 + j * stride_rg2 + h * stride_rg3
    off_lg = b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + h * stride_lg3

    left  = tl.load(left_ptr  + off_l,  mask=mask, other=0.0)
    right = tl.load(right_ptr + off_rg, mask=mask, other=0.0)

    lgate = tl.load(left_gate_ptr  + off_lg, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + off_lg, mask=mask, other=0.0)

    if HAS_MASK:
        # mask has shape (B,N,N,1) → no hidden dimension
        off_m = b * stride_m0 + i * stride_m1 + j * stride_m2
        m = tl.load(mask_ptr + off_m, mask=mask, other=0.0)
    else:
        m = 1.0

    out_l = left  * lgate * m
    out_r = right * rgate * m

    tl.store(out_left_ptr  + off_l,  out_l, mask=mask)
    tl.store(out_right_ptr + off_rg, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        x          – (B, N, N, C)   torch.float32
        mask       – (B, N, N)      torch.bool / float32 / None
        weights    – dict with all learned parameters (float32)
        config     – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor
        (B, N, N, C) float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, C = x.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (FP32 → FP16)
    # ------------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate= torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate  = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N,H)

    # Cast everything to FP16 for the upcoming GEMM
    left_fp16       = left.to(torch.float16).contiguous()
    right_fp16      = right.to(torch.float16).contiguous()
    left_gate_fp16  = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()
    out_gate_fp16   = out_gate.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 3️⃣ Fuse mask & gates with Triton
    # ------------------------------------------------------------------
    if mask is None:
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)  # dummy buffer
        has_mask = False
    else:
        # (B,N,N) → (B,N,N,1) for broadcasting over H
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        has_mask = True

    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elements = B * N * N * H
    # BLOCK of 2048 gives good occupancy on H200 while keeping register pressure low
    BLOCK = 2048
    grid = (triton.cdiv(total_elements, BLOCK),)

    _apply_mask_and_gate_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        # strides for left/right (identical layout)
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        # right strides (same as left – passed for completeness)
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        # gate strides (same layout)
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        # mask strides (if present)
        mask_fp16.stride(0) if has_mask else 0,
        mask_fp16.stride(1) if has_mask else 0,
        mask_fp16.stride(2) if has_mask else 0,
        mask_fp16.stride(3) if has_mask else 0,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS / Tensor‑cores)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)

    # FP16 @ FP16^T → FP16 output, accumulation in FP32
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))          # (B*H,N,N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype)          # back to FP32 for LayerNorm

    # ------------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    # out_gate has shape (B,N,N,H) already – broadcast over (B,N,N,H)
    out = out * out_gate_fp16.to(dtype)

    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)
    return out
