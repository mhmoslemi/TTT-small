"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward only)
------------------------
1️⃣  Layer‑norm the 4‑D input (float32, manual).
2️⃣  Linear projections + three sigmoid gates – performed in fp16 for speed.
3️⃣  A tiny Triton kernel fuses:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
   (the mask is optional; if not supplied the kernel receives a mask of 1s).
4️⃣  Heavy cubic contraction  
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .
   We reshape the tensors to (B·H, N, N) and call   torch.bmm   – this lets
   cuBLAS/Tensor‑cores on the H200 do the O(N³) work efficiently.
5️⃣  Convert back to fp32, apply a second LayerNorm, multiply by the output‑gate
   and a final linear projection to the original dimension.
All steps preserve the original API: the function receives a tuple
(input, mask, weights, config) and returns a float‑32 tensor of shape
[bs, seq_len, seq_len, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton element‑wise kernel – mask & gate fusion
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr, right_ptr,                # fp16 tensors (B,H,N,N)
    left_gate_ptr, right_gate_ptr,      # fp16 tensors (B,H,N,N)
    mask_ptr,                           # fp16 tensor (B,1,N,N) or 1s
    stride_lz, stride_lh, stride_li, stride_lj,   # left  strides
    stride_rz, stride_rh, stride_ri, stride_rj,   # right strides
    stride_gz, stride_gh, stride_gi, stride_gj,   # gate  strides
    stride_mz, stride_mh, stride_mi, stride_mj,   # mask  strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
):
    """
    One program processes a tile (BLOCK_I × BLOCK_J) for a given (batch,head)
    pair (pid_z).  The kernel multiplies left/right by the mask and the
    corresponding gate, all in‑place.
    """
    pid_z = tl.program_id(2)                     # batch*head index
    pid_i = tl.program_id(0)                     # tile row   (i)
    pid_j = tl.program_id(1)                     # tile col   (j)

    i_off = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i_off < tl.load(tl.constexpr_ptr(0))   # placeholder, will be replaced
    mask_j = j_off < tl.load(tl.constexpr_ptr(1))

    # Load tiles (use masks to stay in‑bounds)
    left = tl.load(
        left_ptr
        + pid_z * stride_lz
        + i_off[:, None] * stride_li
        + j_off[None, :] * stride_lj,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )
    right = tl.load(
        right_ptr
        + pid_z * stride_rz
        + i_off[:, None] * stride_ri
        + j_off[None, :] * stride_rj,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )
    left_gate = tl.load(
        left_gate_ptr
        + pid_z * stride_gz
        + i_off[:, None] * stride_gi
        + j_off[None, :] * stride_gj,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )
    right_gate = tl.load(
        right_gate_ptr
        + pid_z * stride_gz
        + i_off[:, None] * stride_gi
        + j_off[None, :] * stride_gj,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )
    mask = tl.load(
        mask_ptr
        + pid_z * stride_mz
        + i_off[:, None] * stride_mi
        + j_off[None, :] * stride_mj,
        mask=mask_i[:, None] & mask_j[None, :],
        other=1.0,
    )

    # Fuse: * mask * gate
    left = left * mask * left_gate
    right = right * mask * right_gate

    # Store back
    tl.store(
        left_ptr
        + pid_z * stride_lz
        + i_off[:, None] * stride_li
        + j_off[None, :] * stride_lj,
        left,
        mask=mask_i[:, None] & mask_j[None, :],
    )
    tl.store(
        right_ptr
        + pid_z * stride_rz
        + i_off[:, None] * stride_ri
        + j_off[None, :] * stride_rj,
        right,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _apply_mask_gate(left, right, left_gate, right_gate, mask):
    """
    In‑place fusion of mask and gating using Triton.
    All tensors are fp16, contiguous, shape (B, H, N, N).
    ``mask`` has shape (B, N, N) or None.
    Returns the (modified) left and right tensors.
    """
    B, H, N, _ = left.shape
    # reshape mask to (B,1,N,N) and broadcast over heads
    if mask is None:
        mask_tile = torch.full((B, 1, N, N), 1.0, dtype=torch.float16, device=left.device)
    else:
        mask_tile = mask.to(torch.float16).unsqueeze(1)   # (B,1,N,N)

    # launch configuration
    BLOCK_I = 64
    BLOCK_J = 64
    grid = (
        triton.cdiv(N, BLOCK_I),          # i‑tiles
        triton.cdiv(N, BLOCK_J),          # j‑tiles
        B * H,                            # batch*head
    )
    # strides (in elements)
    stride_lz, stride_lh, stride_li, stride_lj = left.stride()
    stride_rz, stride_rh, stride_ri, stride_rj = right.stride()
    stride_gz, stride_gh, stride_gi, stride_gj = left_gate.stride()   # same as right_gate
    stride_mz, stride_mh, stride_mi, stride_mj = mask_tile.stride()

    _apply_mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_tile,
        stride_lz=stride_lz,
        stride_lh=stride_lh,
        stride_li=stride_li,
        stride_lj=stride_lj,
        stride_rz=stride_rz,
        stride_rh=stride_rh,
        stride_ri=stride_ri,
        stride_rj=stride_rj,
        stride_gz=stride_gz,
        stride_gh=stride_gh,
        stride_gi=stride_gi,
        stride_gj=stride_gj,
        stride_mz=stride_mz,
        stride_mh=stride_mh,
        stride_mi=stride_mi,
        stride_mj=stride_mj,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        num_warps=4,
    )
    return left, right


# ----------------------------------------------------------------------
# Forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors (see code)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (fp32, manual)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still fp32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections + gating (fp16)
    # --------------------------------------------------------------
    # Cast weights once to fp16
    to_fp16 = lambda t: t.to(torch.float16)
    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                     # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)            # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 3️⃣ Fuse mask & gating with a tiny Triton kernel
    # --------------------------------------------------------------
    # reshape to (B, H, N, N) for the kernel
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_gate  = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous()

    left, right = _apply_mask_gate(left, right, left_gate, right_gate, mask)

    # --------------------------------------------------------------
    # 4️⃣ Heavy contraction – batched GEMM via torch.bmm (fp16)
    # --------------------------------------------------------------
    # left : (B, H, N, N)  → (B*H, N, N)
    # right: (B, H, N, N) → (B*H, N, N) and we need (K,J) = (N,N) transposed
    BH = B * hidden_dim
    left_flat  = left.view(BH, N, N)                     # (B*H, I, K)
    right_flat = right.view(BH, N, N).transpose(1, 2)   # (B*H, K, J)

    # fp16 matrix multiply, accumulation in fp32 inside torch
    out_flat = torch.bmm(left_flat, right_flat)        # (B*H, I, J) fp16

    # reshape back to (B, H, N, N)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)                         # now (B,N,N,hidden)

    # --------------------------------------------------------------
    # 5️⃣ Output LayerNorm (fp32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 6️⃣ Final gating + projection (fp32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over hidden dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                  # (B,N,N,dim)

    return out
