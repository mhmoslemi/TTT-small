"""
TriMul “outgoing” forward (no gradients)

We keep the cheap element‑wise work (LayerNorm, Linear projections, gating) in
pure PyTorch and accelerate the only cubic contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

by reshaping the tensors to a batched matrix‑multiply and launching a tiny
Triton kernel that does the same work as torch.bmm but avoids the Python
overhead of the loop over the hidden dimension.  The kernel computes a
batched GEMM on fp16 inputs with fp32 accumulation and writes fp16 results.
All other operations stay in fp32 (or fp16 while the GEMM is running) and the
final output is cast back to fp32.

The kernel works for any batch size B∈{1,2}, sequence length N up to 1024 and
hidden dimension H (called ``hidden_dim`` in the config).  A mask, when present,
is broadcast over the hidden dimension before the contraction.

The implementation follows the “outgoing” variant from AlphaFold‑3:
    left  = x·W_left   (projected along the i‑axis)
    right = x·W_right  (projected along the j‑axis)
    out   = Σ_k left[...,i,k,:] * right[...,j,k,:]
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (I×K)·(K×J) → (I×J)   (fp16→fp16, fp32 accumulator)
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    a_ptr,                 # left  tensor   (Z, I, K)   fp16
    b_ptr,                 # right tensor   (Z, K, J)   fp16
    c_ptr,                 # output tensor  (Z, I, J)   fp16
    M, N, K,               # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_z = tl.program_id(0)          # batch*head dimension
    pid_m = tl.program_id(1)          # tile row   (I)
    pid_n = tl.program_id(2)          # tile col   (J)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kb in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # a tile: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # b tile: (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 accumulation
        acc += tl.dot(a, b)

    # write fp16 result
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batch_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the kernel above.

    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous
    returns (B, H, I, J)   fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # output allocation
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten the (B,H) dimensions – Triton sees a 3‑D tensor
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides (in elements, not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling parameters – tuned for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                                   # program_id(0) – batch*head
        triton.cdiv(I, BLOCK_M),             # program_id(1) – rows
        triton.cdiv(J, BLOCK_N),             # program_id(2) – cols
    )

    batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor [B, N, N]       (float32/bool) or None
        - weights      : dict of torch.Tensor parameters
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor [B, N, N, dim]   (float32)
    """
    # ------------------------------------------------------------------
    # 0) unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep in FP32)
    # ------------------------------------------------------------------
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    # manual layer‑norm (avoid nn.Module overhead)
    mean = inp.mean(dim=-1, keepdim=True)
    var = inp.var(dim=-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + eps)
    x = x * w_norm + b_norm        # shape (B,N,N,dim), still fp32

    # ------------------------------------------------------------------
    # 2) Linear projections  (cast once to FP16 for the heavy part)
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w  = to_fp16(weights["left_proj.weight"])   # (dim, hidden)
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w  = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w   = to_fp16(weights["out_gate.weight"])

    to_out_norm_w = weights["to_out_norm.weight"]   # stay fp32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # FP16 copy of the normalized tensor (used for all matmuls/gates)
    x_fp16 = x.to(torch.float16)

    # projections → (B,N,N,hidden)
    left  = F.linear(x_fp16, left_proj_w)    # bias is None in the reference
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure same dtype and broadcast
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction – batched GEMM via Triton
    #    left  : (B, hidden, N, N)  → (B, hidden, I, K)
    #    right : (B, hidden, N, N)  → (B, hidden, K, J)  (transpose last two)
    # ------------------------------------------------------------------
    # bring hidden dimension to axis 1 so we can treat it as “head”
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # batched GEMM → (B, H, I, J)  fp16
    out_fp16 = triton_batch_gemm(left_perm, right_perm)

    # back‑to (B,N,N,hidden) and cast to FP32 for the remaining layers
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (still hidden dim)
    # ------------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7) Final gating + projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # same gate as in step 4
    out = F.linear(out, to_out_w)                  # (B,N,N,dim)

    return out
```

**Explanation of design choices**

1. **LayerNorm & Linear ops in PyTorch** – they are cheap (O(B·N²·C)) and
   benefit from highly‑optimized kernels already in PyTorch. Keeping them in
   Python avoids the overhead of writing many small Triton kernels.

2. **Mixed‑precision** – The dominant cubic work is performed in **fp16** with
   **fp32** accumulation to preserve numerical stability, matching the best
   practice on Hopper (tensor‑cores).

3. **Batched GEMM kernel** – Instead of a naïve N³ triple loop we reshape the
   tensors to `(B·hidden, N, N)` and compute a batched matrix multiplication.
   The Triton kernel mirrors `torch.bmm` but eliminates the Python dispatch per
   head and lets us tune block sizes for the H200.  `BLOCK_M/N/K = 64/64/32`
   is a good trade‑off for the tested sequence lengths (256‑1024) on SM90.

4. **Mask handling** – The mask is optional; when present it is simply broadcast
   over the hidden dimension before the contraction, exactly as in the reference
   implementation.

5. **Output pipeline** – After the contraction we restore the original layout,
   apply the second LayerNorm, the output gate (shared with the input gate), and
   the final linear projection, all in **fp32** as required by the original model.

The function `custom_kernel` follows the requested signature and can be called
as

```python
out = custom_kernel((x, mask, weights, cfg))
```

producing a tensor of shape `[B, N, N, dim]` with dtype `torch.float32`.
