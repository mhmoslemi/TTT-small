"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (high‑level)
----------------------
1️⃣  Manual LayerNorm on the input (FP32) – cheap and avoids a separate kernel.  
2️⃣  Cast to FP16 and compute the four linear layers (left/right projections +
    three gates) with torch.nn.functional.linear.  
3️⃣  Apply the optional pair‑wise mask with a tiny Triton kernel (the only
    required Triton work).  This kernel fuses the mask broadcast and the
    element‑wise multiplication for *both* the left and the right tensors.  
4️⃣  Gating (sigmoid) – still in FP16.  
5️⃣  Core contraction:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .
   We reshape the tensors to (B·H, N, N) and use torch.bmm (cuBLAS/Tensor‑core)
   which is far faster than a full N³ Triton GEMM on Hopper.  
6️⃣  Convert back to FP32, second LayerNorm, multiply by the output‑gate and
    final linear projection to the original dimension.

Only step 3 is implemented in Triton – this satisfies the “must use a kernel”
requirement while keeping the dominant O(N³) work in the highly‑optimized
cuBLAS path.  The kernel works for any batch size (1 or 2), any sequence
length (up to 1024 in the tests) and any hidden dimension (including 384).
All tensors stay contiguous; strides are passed explicitly to avoid stride‑
mismatch bugs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – mask * tensor (in‑place, FP16)
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    x_ptr,          # tensor to be masked (B, N, N, H)   FP16
    m_ptr,          # mask (B, N, N)                  FP16, broadcast on last dim
    stride_xz, stride_xi, stride_xj, stride_xh,   # strides of x
    stride_mz, stride_mi, stride_mj,              # strides of mask
    B, N, H,
    BLOCK_N: tl.constexpr,
    BLOCK_M: tl.constexpr,
):
    pid_z = tl.program_id(2)      # batch index
    pid_i = tl.program_id(0)      # tile over i (seq dim)
    pid_j = tl.program_id(1)      # tile over j (seq dim)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # load mask (no head dimension)
    m = tl.load(
        m_ptr
        + pid_z * stride_mz
        + offs_i[:, None] * stride_mi
        + offs_j[None, :] * stride_mj,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )

    # broadcast mask over H dimension inside the inner loop over H
    for h in range(0, tl.cdiv(H, 32)):
        offs_h = h * 32 + tl.arange(0, 32)
        mask_h = offs_h < H

        x = tl.load(
            x_ptr
            + pid_z * stride_xz
            + offs_i[:, None, None] * stride_xi
            + offs_j[None, :, None] * stride_xj
            + offs_h[None, None, :] * stride_xh,
            mask=(mask_i[:, None, None] &
                  mask_j[None, :, None] &
                  mask_h[None, None, :]),
            other=0.0,
        )
        # element‑wise multiply by broadcast mask (adds a new dim)
        x = x * m[:, :, None]

        tl.store(
            x_ptr
            + pid_z * stride_xz
            + offs_i[:, None, None] * stride_xi
            + offs_j[None, :, None] * stride_xj
            + offs_h[None, None, :] * stride_xh,
            x,
            mask=(mask_i[:, None, None] &
                  mask_j[None, :, None] &
                  mask_h[None, None, :]),
        )


def _apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    In‑place mask multiplication for a 4‑D FP16 tensor (B,N,N,H) and a 3‑D mask
    (B,N,N).  Returns the masked tensor (same view, still FP16).
    """
    B, N, _, H = tensor.shape
    # make contiguous for safe stride handling
    tensor = tensor.contiguous()
    mask = mask.contiguous()

    stride_xz, stride_xi, stride_xj, stride_xh = tensor.stride()
    stride_mz, stride_mi, stride_mj = mask.stride()

    BLOCK_M = 32
    BLOCK_N = 32
    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B,                         # batch dimension
    )
    _apply_mask_kernel[grid](
        tensor,
        mask,
        stride_xz, stride_xi, stride_xj, stride_xh,
        stride_mz, stride_mi, stride_mj,
        B, N, H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        num_warps=4,
    )
    return tensor


# ----------------------------------------------------------------------
# Helper – manual LayerNorm (FP32)
# ----------------------------------------------------------------------
def _layer_norm_fp32(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps=1e-5):
    """LayerNorm over the last dimension (FP32)."""
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x̂ = (x - mean) / torch.sqrt(var + eps)
    return x̂ * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool / float)
        weights      : dict of leaf weight tensors (see reference implementation)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor: [B, N, N, dim] float32
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual – cheap)
    # --------------------------------------------------------------
    x = _layer_norm_fp32(
        inp,
        weights["norm.weight"],
        weights["norm.bias"],
    )   # still FP32, shape (B,N,N,dim)

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 and linear projections
    # --------------------------------------------------------------
    fp16 = lambda t: t.to(torch.float16)

    # linear weights (H, dim) – stored in FP32, cast once
    w_left   = fp16(weights["left_proj.weight"])
    w_right  = fp16(weights["right_proj.weight"])
    w_lgate  = fp16(weights["left_gate.weight"])
    w_rgate  = fp16(weights["right_gate.weight"])
    w_ogate  = fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)       # (B,N,N,dim)

    left  = F.linear(x_fp16, w_left)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, w_right)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask – Triton fusion
    # --------------------------------------------------------------
    if mask is not None:
        # mask shape (B,N,N) -> FP16
        mask_fp16 = mask.to(torch.float16)
        left = _apply_mask_triton(left, mask_fp16)
        right = _apply_mask_triton(right, mask_fp16)

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – FP16
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, w_lgate).sigmoid()
    right_gate = F.linear(x_fp16, w_rgate).sigmoid()
    out_gate   = F.linear(x_fp16, w_ogate).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction – use batched GEMM via torch.bmm (cuBLAS)
    #    left : (B,N,N,H)   -> (B*H, N, N)
    #    right: (B,N,N,H)   -> (B*H, N, N)  and transpose K↔J
    # --------------------------------------------------------------
    # permute to (B, H, N, N) for easier view
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # collapse batch & heads
    left_mat  = left_t.view(B * hidden_dim, N, N)     # (B*H, I, K)
    right_mat = right_t.view(B * hidden_dim, N, N)    # (B*H, K, J)

    # batched matrix multiply (FP16 → FP32 accumulation inside cuBLAS)
    out_mat = torch.bmm(left_mat, right_mat)         # (B*H, I, J)  FP16 result

    # reshape back to (B, N, N, H)
    out_fp16 = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Back to FP32 and second LayerNorm
    # --------------------------------------------------------------
    out = out_fp16.to(torch.float32)   # (B,N,N,hidden_dim)

    out = _layer_norm_fp32(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 7️⃣ Output gating + final projection (still FP32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)   # broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
