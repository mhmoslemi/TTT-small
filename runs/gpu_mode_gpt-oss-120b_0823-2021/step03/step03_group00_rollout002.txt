"""
TriMul “outgoing” forward (AlphaFold‑3)

- LayerNorm over the last channel (fp32)
- Five fp16 linear projections (left/right/value, three sigmoid gates)
- Optional binary mask broadcasted on the hidden dimension
- Core contraction   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
  implemented as a batched GEMM (fp16 × fp16 → fp32) in a tiny Triton kernel.
- Second LayerNorm + output‑gate + final fp32 linear.
The kernel processes a (BLOCK_M×BLOCK_N) tile of the (I×J) matrix for every
(batch,head) pair, accumulating in fp32 for numerical safety.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (I×K)·(K×J) → (I×J) for one (batch,head)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr, b_ptr, c_ptr,           # pointers
    M, N, K,                       # extents
    stride_az, stride_am, stride_ak,   # a strides (z,m,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,n)
    stride_cz, stride_cm, stride_cn,   # c strides (z,m,n)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    One Triton program works on a tile (BLOCK_M × BLOCK_N) of the output for a
    particular (batch, head) index (pid_z).  Operates in fp16 and accumulates in
    fp32.
    """
    pid_z = tl.program_id(2)          # collapsed batch*head dimension
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile column

    # ---- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K --------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile : (BLOCK_M × BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile : (BLOCK_K × BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp16×fp16 → fp32 dot product
        acc += tl.dot(a, b)

    # write back (fp16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM used by TriMul.
    Arguments
    ---------
    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous
    Returns
    -------
    out   : (B, H, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse (B*H) → single launch dimension
    batch = B * H
    a = left.reshape(batch, I, K)
    b = right.reshape(batch, K, J)
    c = out.reshape(batch, I, J)

    # strides (in elements)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tuning parameters – give good occupancy on Hopper (SM‑90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # rows
        triton.cdiv(J, BLOCK_N),   # cols
        batch,                     # one program per (B,H)
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # fp16 friendly on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward – entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors (see below)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over last channel (fp32)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    # manual LN to avoid extra kernel launch
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                         # still fp32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections + gating (fp16)
    # --------------------------------------------------------------
    # cast once
    def to_fp16(t): return t.to(torch.float16)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)       # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        # ensure mask is float16 and has a trailing dim
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction via Triton batched GEMM
    #    left : (B,N,N,hidden) → (B, hidden, N, N)
    #    right: (B,N,N,hidden) → (B, hidden, N, N) and transpose K,J
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, J)

    # batched GEMM returns fp16 (B, H, I, J)
    out_fp16 = _triton_batched_gemm(left_perm, right_perm)

    # reshape back to (B,N,N,hidden) then to fp32 for the next LN
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (fp32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 7️⃣ Output gate + final linear (fp32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                  # (B,N,N,dim)

    return out

### Why this meets the requirements
1. **Correctness** – Implements exactly the operations from the reference PyTorch
   `TriMul` (LayerNorms, linear projections, masks, sigmoid gates, the
   `einsum` contraction, second LayerNorm, output‑gate and final linear).  
2. **Triton involvement** – The expensive cubic contraction is performed by the
   `_batch_gemm_kernel`, a genuine Triton kernel that tiles the GEMM and
   accumulates in fp32 for stability.  
3. **Performance considerations**
   * Mixed‑precision: all heavy matrix work is in FP16 with FP32 accumulation,
     matching the Tensor‑Core friendly path on Hopper.
   * Tiling (`BLOCK_M=64, BLOCK_N=64, BLOCK_K=32`) gives good occupancy on
     SM‑90 and fits in shared memory.
   * The rest of the work (LayerNorm, linear, gating) stays in PyTorch; these
     are memory‑bound elementwise ops and not the bottleneck.
4. **Flexibility** – Supports both masked and un‑masked runs, any batch size
   (`B∈{1,2}`), sequence lengths up to 1024, and the allowed channel dimensions
   (`dim`, `hidden_dim`). All tensors are assumed to be on the same CUDA device.
