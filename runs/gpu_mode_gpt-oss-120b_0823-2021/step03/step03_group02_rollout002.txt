"""
TriMul “outgoing” forward pass (no backward).

The computational bottleneck is the N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
We treat the hidden dimension *d* as “heads”.  For each head we have a
batched matrix‑multiply (I×K)·(K×J) → (I×J).  The multiplication itself is
performed by a tiny Triton kernel that simply loads a block of the two
matrices, accumulates in fp32 and writes the result in fp16.  All surrounding
operations (LayerNorms, Linear projections, sigmoid gates, final LayerNorm and
linear) stay in PyTorch – this keeps the implementation simple, numerically
stable and fast while still satisfying the “must‑use‑Triton‑somewhere”
requirement.

The kernel works on tensors shaped (Z, I, K) and (Z, K, J) where Z = B * H
(B = batch size, H = hidden_dim).  The surrounding code reshapes the tensors
accordingly, casts to fp16, applies the optional mask and the gating
coefficients, calls the kernel, and finally restores the original layout and
dtype (float32) for the remaining LayerNorm / linear layers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (fp16 inputs, fp32 accumulation, fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr,  # left  tensor   (Z, I, K)  fp16
    b_ptr,  # right tensor   (Z, K, J)  fp16
    c_ptr,  # output tensor  (Z, I, J)  fp16
    M, N, K,               # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids (tile coordinates)
    pid_z = tl.program_id(0)          # batch*head dimension
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # start offsets of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M × K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile (K × N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fused fp32 accumulation
        acc += tl.dot(a, b)

    # write the result (fp16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    left : (B, H, I, K)   fp16, contiguous
    right: (B, H, K, J)   fp16, contiguous
    returns (B, H, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten batch*head dimension for the kernel launch
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides in elements (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling – values that work well on Hopper/H200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                              # program_id(0) → batch*head
        triton.cdiv(I, BLOCK_M),        # program_id(1) → rows
        triton.cdiv(J, BLOCK_N),        # program_id(2) → cols
    )

    _batched_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # good default for 64×64 tiles on H200
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        mask         : torch.Tensor [B, N, N]       (float32/bool) or None
        weights      : dict of torch.Tensor parameters
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]   (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep in FP32 for numeric stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual LN – avoids nn.Module overhead
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b               # shape (B,N,N,dim), fp32

    # ------------------------------------------------------------------
    # 2) Linear projections & gating weights (cast once to fp16)
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = weights["to_out_norm.weight"]   # stays fp32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # fp16 copy of normalized input for all heavy ops
    x_fp16 = x.to(torch.float16)

    # projections → (B,N,N,hidden)
    left  = F.linear(x_fp16, left_proj_w)   # fp16
    right = F.linear(x_fp16, right_proj_w)  # fp16

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()   # (B,N,N,hidden)
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()    # (B,N,N,hidden)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction – batched GEMM via Triton
    #    left : (B, H, I, K)   = (B, hidden, N, N)
    #    right: (B, H, K, J)   = (B, hidden, N, N) with last two axes swapped
    # ------------------------------------------------------------------
    # bring hidden dim to second axis and make the reduction dimension contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # batched matmul in fp16, accumulation in fp32
    out_fp16 = _batched_gemm_fp16(left_perm, right_perm)   # (B, H, I, J)

    # restore original layout (B,N,N,hidden) and cast back to fp32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (still in fp32)
    # ------------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7) Final gating + linear projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)      # same out_gate as used earlier
    out = F.linear(out, to_out_w)               # (B,N,N,dim), fp32

    return out
```

**Explanation of design choices**

1. **What the Triton kernel does** – It only performs the N³ contraction,
   which is the only part that truly benefits from custom GPU tiling.  The
   kernel uses fp16 inputs (the projected tensors) and accumulates in fp32
   for numeric stability, then writes fp16 output that is later cast back to
   fp32.

2. **Why other ops stay in PyTorch** – LayerNorm, sigmoid gating and the final
   linear projection are cheap element‑wise operations compared to the O(N³)
   work.  Keeping them in PyTorch avoids unnecessary data movement and lets us
   reuse highly‑optimized fused kernels already present in the PyTorch
   library.

3. **Mask handling** – The mask is applied in fp16 after the projections; the
   extra broadcast dimension is cheap and fits naturally into the existing
   tensor shapes.

4. **Tiling parameters** – `BLOCK_M = BLOCK_N = 64` and `BLOCK_K = 32` match the
   sweet spot for Hopper/H200 tensor‑core usage (64×64 output tile, 32‑wide
   reduction).  `num_warps=4` gives enough parallelism without exceeding shared
   memory limits.

5. **Memory layout** – We `permute` to `(B, H, I, K)` and `(B, H, K, J)` and
   call `.contiguous()` so that the last two dimensions are dense, which is
   required for the kernel’s stride calculations.

6. **Correctness** – All operations respect the original PyTorch semantics:
   LayerNorm uses unbiased‑False variance, sigmoid gates are applied to the
   *same* normalized input as in the reference implementation, and the final
   `out_gate` is reused before the last linear projection.

The function `custom_kernel` therefore complies with the required signature,
uses a Triton kernel for the dominant cubic contraction, and returns a
float‑32 tensor of shape `[B, N, N, dim]` matching the reference PyTorch module.
