"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C)  – PyTorch (FP32).
2. Linear projections → (B,N,N,H) for left, right and three gates – PyTorch.
3. Fuse mask, left‑gate and right‑gate with a small Triton kernel
   (the only required “some‑kernel” part).  All tensors are kept in FP16
   for the subsequent GEMM.
4. Batched GEMM using cuBLAS/Tensor‑cores:
        out[b,h] = left[b,h] @ right[b,h].transpose(-2,-1)
   Reshape (B,N,N,H) → (B*H,N,N) before the matmul.
5. Layer‑norm over the hidden dimension H, multiply by out_gate and
   final linear projection back to C – PyTorch (FP32).

The heavy O(N³) work is delegated to cuBLAS (torch.bmm) which is
orders‑of‑magnitude faster than any pure‑Triton implementation.
Only the element‑wise fusion (mask × gate) is performed in Triton,
ensuring at least some work runs on the GPU as required by the
challenge.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – fuse mask and gating for left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,                   # fp16 * (B,N,N,H)
    right_ptr,                  # fp16 * (B,N,N,H)
    left_gate_ptr,              # fp16 * (B,N,N,H)
    right_gate_ptr,             # fp16 * (B,N,N,H)
    mask_ptr,                   # fp16 * (B,N,N)   or nullptr
    out_left_ptr, out_right_ptr,# fp16 * (B,N,N,H)   output buffers
    B, N, H,                    # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_m0, stride_m1, stride_m2,
    HAS_MASK: tl.constexpr,
    BLOCK:   tl.constexpr,
):
    """
    For every element (b,i,j,h) compute
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    The kernel is launched on a 1‑D grid covering B*N*N*H elements.
    """
    pid   = tl.program_id(0)
    offs  = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask  = offs < total

    # linear → (b,i,j,h)
    tmp = offs
    h   = tmp % H
    tmp = tmp // H
    j   = tmp % N
    tmp = tmp // N
    i   = tmp % N
    b   = tmp // N

    # offsets for left/right tensors (same layout)
    idx = (b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3)
    left   = tl.load(left_ptr   + idx, mask=mask, other=0.0)
    right  = tl.load(right_ptr  + idx, mask=mask, other=0.0)

    # offsets for gate tensors (same layout)
    idx_g = (b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + h * stride_lg3)
    lgate = tl.load(left_gate_ptr  + idx_g, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + idx_g, mask=mask, other=0.0)

    # optional mask (broadcast over H)
    if HAS_MASK:
        idx_m = (b * stride_m0 + i * stride_m1 + j * stride_m2)
        m = tl.load(mask_ptr + idx_m, mask=mask, other=0.0)
    else:
        m = 1.0

    out_l = left  * lgate * m
    out_r = right * rgate * m

    tl.store(out_left_ptr  + idx, out_l, mask=mask)
    tl.store(out_right_ptr + idx, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : (x, mask, weights, config)
        x        – (B, N, N, C)          torch.float32
        mask     – (B, N, N) or None     torch.bool/float32
        weights  – dict of FP32 parameters
        config   – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor  (B, N, N, C)   float32
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, C = x.shape
    H = config["hidden_dim"]
    dtype = torch.float32

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (still FP32)
    # --------------------------------------------------------------
    left       = F.linear(x, weights["left_proj.weight"].to(dtype))      # (B,N,N,H)
    right      = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N,H)

    # --------------------------------------------------------------
    # 3️⃣ Cast to FP16 and fuse mask & gates with Triton
    # --------------------------------------------------------------
    left_fp16       = left .to(torch.float16).contiguous()
    right_fp16      = right.to(torch.float16).contiguous()
    left_gate_fp16  = left_gate .to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()

    if mask is None:
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)   # dummy
        has_mask = False
    else:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()   # (B,N,N,1)
        has_mask = True

    # allocate output buffers for the fused tensors
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elements = B * N * N * H
    BLOCK = 1024                         # good occupancy on H200
    grid  = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (FP16 inputs → FP16 output, accum FP32)
    #    Shape: (B*H, N, N)
    # --------------------------------------------------------------
    # (B,N,N,H) → (B,H,N,N) → (B*H,N,N)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H,N,N) FP16

    # reshape back to (B,N,N,H) and promote to FP32 for LayerNorm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(dtype)

    # --------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, out‑gate multiplication, final proj
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                       # broadcast over (N,N)
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
