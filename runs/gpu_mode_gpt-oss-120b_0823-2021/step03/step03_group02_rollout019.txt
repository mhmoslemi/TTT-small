"""
TriMul “outgoing” forward pass (no gradients).

The dominant N³ contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is executed as a batched GEMM in fp16 with a tiny Triton kernel.
All other ops (layer‑norms, linear projections, sigmoid gates) stay in
PyTorch – they are cheap compared with the cubic work and avoid a
large‑scale Triton rewrite.  The kernel loads the optional mask together
with the left/right tiles, so no extra PyTorch kernel is required for
masking.

The implementation follows the reference PyTorch code exactly, but
uses mixed‑precision (fp16 for the heavy GEMM) and a custom Triton kernel
to keep the whole pipeline on‑device and avoid the overhead of a Python
loop.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (H×I)·(H×J) → (H×I×J)   (fp16 input, fp32 acc, fp16 out)
#   a : (Z, I, K)   – left  tensor, already permuted to (B, hidden, N, N)
#   b : (Z, K, J)   – right tensor, permuted to (B, hidden, N, N) and
#                    transposed on the last two dimensions
#   c : (Z, I, J)   – result
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    a_ptr, b_ptr, c_ptr,           # pointers
    M, N, K,                       # matrix sizes
    stride_az, stride_am, stride_ak,   # a strides (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides (z,i,j)
    MASK_PTR: tl.constexpr,            # optional mask pointer, 0 = no mask
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)           # batch*head dimension
    pid_m = tl.program_id(1)           # tile over rows   (I)
    pid_n = tl.program_id(2)           # tile over cols   (J)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile (M×K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile (K×N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # optional mask: when MASK_PTR != 0 we also load a mask tile
        # mask shape is (Z, M, K) – same layout as a; we broadcast it to K
        if MASK_PTR != 0:
            mask_tile = tl.load(
                MASK_PTR
                + pid_z * stride_az
                + offs_m[:, None] * stride_am
                + offs_k[None, :] * stride_ak,
                mask=mask_m[:, None] & mask_k[None, :],
                other=1.0,
            )
            a = a * mask_tile   # element‑wise mask multiplication

        acc += tl.dot(a, b)

    # write result (fp16) – mask out‑of‑bounds entries
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batch_gemm(left: torch.Tensor,
                      right: torch.Tensor,
                      mask: torch.Tensor | None = None) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    Parameters
    ----------
    left : torch.Tensor (B, H, I, K)   fp16, contiguous
    right: torch.Tensor (B, H, K, J)   fp16, contiguous
    mask : torch.Tensor (B, H, I, K)   fp16 or None (broadcasted same shape as left)

    Returns
    -------
    out : torch.Tensor (B, H, I, J)    fp16
    """
    B, H, I, K = left.shape
    _, _, _, J = right.shape
    assert left.is_contiguous() and right.is_contiguous()
    if mask is not None:
        assert mask.shape == left.shape

    # output allocation
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten batch*head for the kernel
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides (in elements)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # optional mask flattening
    if mask is not None:
        mask_ptr = mask.view(Z, I, K).data_ptr()
    else:
        mask_ptr = 0

    # tiling – tuned for Hopper (SM90) tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                                 # program_id(0) – batch*head
        triton.cdiv(I, BLOCK_M),           # program_id(1) – rows
        triton.cdiv(J, BLOCK_N),           # program_id(2) – cols
    )

    batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        MASK_PTR=mask_ptr,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # enough warps to hide latency on H200
        num_stages=3,
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        mask         : torch.Tensor [B, N, N]       (float32/bool) or None
        weights      : dict of torch.Tensor parameters (see skeleton)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]   (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm  (keep in fp32 for numeric stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                     # still fp32

    # ------------------------------------------------------------------
    # 2) Cast once to fp16 for the heavy part
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (fp16) – no bias per reference code
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5) Gating (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core contraction – batched GEMM via Triton
    #    left : (B, hidden, N, N)   → (B*hidden, N, N)
    #    right: (B, hidden, N, N)   → (B*hidden, N, N) and transposed
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # fuse mask inside the kernel if a mask exists (same layout as left)
    mask_for_kernel = None
    if mask is not None:
        # mask shape (B,N,N) → broadcast to hidden dim
        mask_for_kernel = mask.unsqueeze(1).repeat(1, hidden_dim, 1, 1).contiguous()
    out_fp16 = triton_batch_gemm(left_perm, right_perm, mask_for_kernel)

    # back to (B,N,N,hidden) layout and cast to fp32 for the remaining layers
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm  (fp32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 8) Final gating + linear projection (fp32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # reuse gate computed earlier
    to_out_w = weights["to_out.weight"]             # (dim, dim) in fp32
    out = F.linear(out, to_out_w)                  # (B,N,N,dim)

    return out
