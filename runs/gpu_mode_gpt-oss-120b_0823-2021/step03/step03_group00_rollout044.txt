"""
custom_kernel – outgoing TriMul (AlphaFold‑3)

Algorithm (forward‑only)
------------------------
1️⃣ Layer‑norm the 4‑D input (float32, manual reduction over the channel dim).
2️⃣ Linear projections + three sigmoid gates are performed in fp16 for speed.
3️⃣ Optional binary mask (broadcast over the hidden dimension) is applied to the
   left/right projections.
4️⃣ Core cubic contraction

        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

   is a batched matrix‑multiply with shape
        (B, H, I, K) · (B, H, K, J) → (B, H, I, J)
   (H = hidden_dim, I = J = N, K = N).  
   The heavy work is delegated to a tiny Triton GEMM kernel that tiles the
   (I×K)·(K×J) product for each (batch, head) pair, using fp16×fp16 → fp32
   accumulation and Tensor‑cores on Hopper (SM‑90).

5️⃣ The result is brought back to float32, a second LayerNorm is applied,
   multiplied by the output‑gate, and finally projected back to the original
   channel dimension.

Only the cubic contraction is written in Triton; all other steps stay in
native PyTorch (they are cheap compared with the O(N³) work).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM – (I×K)·(K×J) → (I×J) for each (batch, head) pair
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr, b_ptr, c_ptr,        # pointers
    M, N, K,                    # matrix extents
    stride_az, stride_am, stride_ak,   # a strides (z,m,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,n)
    stride_cz, stride_cm, stride_cn,   # c strides (z,m,n)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    One program processes a tile (BLOCK_M×BLOCK_N) of the output for a given
    (batch, head) index (pid_z).  Operates in fp16, accumulates in fp32.
    """
    pid_z = tl.program_id(2)          # batch*head
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col

    # --------------------------------------------------------------
    # Tile coordinates
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in blocks
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile  (BLOCK_M × BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile  (BLOCK_K × BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp16×fp16 → fp32 dot
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back (fp32 → fp16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM used by TriMul.
    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous
    returns (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse batch*head for kernel launch
    batch = B * H
    a = left.view(batch, I, K)
    b = right.view(batch, K, J)
    c = out.view(batch, I, J)

    # strides in *elements* (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tiling – empirically good on Hopper (SM‑90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),      # rows
        triton.cdiv(J, BLOCK_N),      # cols
        batch,                        # one program per (B,H)
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                 # good fp16 occupancy on H200
    )
    return out


# ----------------------------------------------------------------------
# Entry point – public API expected by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool / float)
        weights      : dict of leaf tensors (see code for required keys)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (float32)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # (B,N,N,dim) float32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections + gating (fp16)
    # --------------------------------------------------------------
    # convert weights once
    def _fp16(t): return t.to(torch.float16)

    left_proj_w   = _fp16(weights["left_proj.weight"])
    right_proj_w  = _fp16(weights["right_proj.weight"])
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                     # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)            # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)     # (B,N,N,1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy contraction – Triton batched GEMM
    #    reshape to (B, H, I, K) and (B, H, K, J)
    # --------------------------------------------------------------
    # left : (B,N,N,hidden) → (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B,N,N,hidden) → (B, hidden, N, N) then swap K,J
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # GEMM in fp16, result (B, hidden, N, N)
    out_fp16 = _triton_batched_gemm(left_perm, right_perm)

    # back to original layout and fp32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (float32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection (float32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)   # broadcast over dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)            # (B,N,N,dim) float32

    return out
