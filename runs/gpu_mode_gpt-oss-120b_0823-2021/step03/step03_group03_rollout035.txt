"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------

1. **LayerNorm (FP32)** – manual mean/var on the last dimension.
2. **Linear projections & gates** – cast the normalized tensor to FP16 and
   apply the five linear layers (left/right proj, three gates) with `torch.nn.functional.linear`.
3. **Optional mask** – broadcast mask (FP16) and multiply it into the left/right
   tensors.
4. **Gating** – sigmoid on the three gate tensors and element‑wise multiply.
5. **Core contraction** – the heavy N³ work (`out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]`)
   is performed as a batched matrix‑multiply in **FP16** using a tiny Triton
   kernel that tiles the (I×K)·(K×J) GEMM.  The kernel works on the view
   `(B*H, N, N)` where `H = hidden_dim`.
6. **Output LayerNorm (FP32)** – same manual implementation as step 1.
7. **Final gating & linear projection** – FP32 gating, then a last linear layer
   back to the original `dim`.

Only the matrix‑multiply of step 5 is written in Triton; everything else stays
in PyTorch to keep the implementation simple and portable.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched GEMM (I×K)·(K×J) → (I×J) for a single head
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,            # left  tensor  (B*H, I, K)   – FP16
    b_ptr,            # right tensor  (B*H, K, J)   – FP16
    c_ptr,            # output tensor (B*H, I, J)   – FP16
    M, N, K,          # extents
    stride_az, stride_am, stride_ak,   # a strides (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Each program instance computes a BLOCK_M×BLOCK_N tile of the output
    matrix for a single (batch,head) pair.
    """
    pid_z = tl.program_id(2)          # collapsed batch*head index
    pid_m = tl.program_id(0)          # tile along I (row)
    pid_n = tl.program_id(1)          # tile along J (col)

    # offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in BLOCK_K sized chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M×K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (K×N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16 dot product, accumulation in FP32
        acc += tl.dot(a, b)

    # Store the result (convert back to FP16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the core contraction.
    left  : (B, H, I, K)  FP16, contiguous
    right : (B, H, K, J)  FP16, contiguous
    returns: (B, H, I, J) FP16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten the (B,H) dimension – kernel expects a 3‑D tensor
    a = left.view(B * H, I, K)
    b = right.view(B * H, K, J)
    c = out.view(B * H, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes tuned for Hopper/SM‑90 (fp16, 4 warps)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),          # tiles along I
        triton.cdiv(J, BLOCK_N),          # tiles along J
        B * H,                            # one program per (batch,head)
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool or float)
        weights      : dict of leaf weight tensors (see reference implementation)
        config       : dict containing keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32, manual – cheap)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    # mean / var over last dim (dim)
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gate weights (cast to FP16)
    # --------------------------------------------------------------
    def _to_fp16(t): return t.to(torch.float16)

    left_proj_w   = _to_fp16(weights["left_proj.weight"])    # (hidden_dim, dim)
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)        # (B, N, N, dim) in FP16

    left  = F.linear(x_fp16, left_proj_w)    # (B, N, N, hidden_dim)
    right = F.linear(x_fp16, right_proj_w)   # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        # mask may be bool or float; convert to FP16 and broadcast
        m = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()   # (B,N,N,hidden_dim)
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()    # (B,N,N,hidden_dim)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction – Triton batched GEMM
    # --------------------------------------------------------------
    # reshape to (B, H, I, K) / (B, H, K, J) where H = hidden_dim, I = J = N, K = N
    # left  : (B, N, N, H) → (B, H, N, N)
    # right : (B, N, N, H) → (B, H, N, N) and we need (K,J) = (N,N) with J = N
    left_t  = left.permute(0, 3, 1, 2).contiguous()         # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()        # (B, H, K, J)

    # batched GEMM in fp16 via Triton → (B, H, N, N)
    out_fp16 = _triton_batched_gemm(left_t, right_t)

    # back‑to original layout (B, N, N, hidden_dim) then cast to FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + projection (FP32)
    # --------------------------------------------------------------
    # out_gate is still FP16; bring to FP32 for broadcasting
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
