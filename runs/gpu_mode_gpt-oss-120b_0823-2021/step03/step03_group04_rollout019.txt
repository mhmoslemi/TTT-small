"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm (high‑level)
----------------------
1️⃣  Layer‑norm the input tensor x ∈ (B,N,N,C)  (FP32 → FP32).
2️⃣  Linear projections of x → left, right ∈ (B,N,N,H) and three gating
    tensors (left_gate, right_gate, out_gate) ∈ (B,N,N,H)   (FP32).
3️⃣  Fuse element‑wise:   left  ← left  * left_gate  * mask
                         right ← right * right_gate * mask
    This tiny work is performed by a dedicated Triton kernel (the only
    required “some‑kernel” constraint).  All tensors are cast to FP16
    before the kernel to profit from Tensor‑cores.
4️⃣  Batched matrix‑multiply (cuBLAS/Tensor‑cores):
        out[b,h] = left_fused[b,h] @ right_fused[b,h].T
    where left_fused/right_fused are reshaped to (B*H,N,N).  The result
    has shape (B,N,N,H) and is kept in FP32 for the next LayerNorm.
5️⃣  Hidden‑dim LayerNorm, gating with out_gate and final linear projection
    back to C channels (FP32 → FP32).

Only step‑3 is written in Triton; the rest uses PyTorch which already
dispatches to highly‑optimized kernels on the H200.  The implementation
handles the optional mask, arbitrary batch size (1 or 2) and any
supported sequence length, channel dimensions and hidden dimensions.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel : element‑wise mask + gate fusion
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,               # fp16 * (B,N,N,H)
    right_ptr,              # fp16 * (B,N,N,H)
    left_gate_ptr,          # fp16 * (B,N,N,H)
    right_gate_ptr,         # fp16 * (B,N,N,H)
    mask_ptr,               # fp16 * (B,N,N) or nullptr
    out_left_ptr,           # fp16 * (B,N,N,H)  – result buffers
    out_right_ptr,          # fp16 * (B,N,N,H)
    B, N, H,                # shape scalars
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_m0, stride_m1, stride_m2,  # mask is 3‑D
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h):
        out_left  = left  * left_gate  * mask   (mask=1 when absent)
        out_right = right * right_gate * mask
    """
    pid   = tl.program_id(0)
    offs  = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # -------- decode linear index → (b,i,j,h) --------
    tmp = offs
    h   = tmp % H                # fastest dimension in memory layout
    tmp = tmp // H
    j   = tmp % N
    tmp = tmp // N
    i   = tmp % N
    b   = tmp // N

    # -------- load tensors --------
    idx = (b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3)
    left_val  = tl.load(left_ptr      + idx, mask=mask, other=0.0)
    right_val = tl.load(right_ptr     + idx, mask=mask, other=0.0)

    idx_gate = (b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + h * stride_lg3)
    left_gate  = tl.load(left_gate_ptr  + idx_gate, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + idx_gate, mask=mask, other=0.0)

    if HAS_MASK:
        idx_mask = (b * stride_m0 + i * stride_m1 + j * stride_m2)
        mask_val = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)
    else:
        mask_val = 1.0

    # -------- fuse --------
    out_l = left_val  * left_gate  * mask_val
    out_r = right_val * right_gate * mask_val

    # -------- store --------
    tl.store(out_left_ptr  + idx, out_l, mask=mask)
    tl.store(out_right_ptr + idx, out_r, mask=mask)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        x       – (B, N, N, C)   torch.float32
        mask    – (B, N, N)      torch.bool / torch.float32 / None
        weights – dict of all learned parameters (float32)
        config  – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor – (B, N, N, C)  float32
    """
    x, mask, weights, config = data
    device = x.device
    dtype  = torch.float32

    B, N, _, C = x.shape
    H = config["hidden_dim"]        # hidden dimension (c_z)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (FP32 → FP16 later)
    # --------------------------------------------------------------
    # x is (B,N,N,C)
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))        # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))       # (B,N,N,H)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))   # (B,N,N,H)
    right_gate= torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))  # (B,N,N,H)
    out_gate  = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))     # (B,N,N,H)

    # --------------------------------------------------------------
    # 3️⃣ Fuse mask & gates → Triton kernel (FP16)
    # --------------------------------------------------------------
    # Cast everything to FP16 (Tensor‑core friendly) and make them contiguous
    left_fp16       = left.to(torch.float16).contiguous()
    right_fp16      = right.to(torch.float16).contiguous()
    left_gate_fp16  = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()

    if mask is None:
        # dummy mask – kernel will treat HAS_MASK=False
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask  = False
    else:
        # mask shape (B,N,N) → broadcast over H
        mask_fp16 = mask.to(torch.float16).contiguous()
        has_mask  = True

    # Allocate output buffers
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    total_elements = B * N * N * H
    BLOCK = 1024                                 # 1‑D grid granularity
    grid  = (triton.cdiv(total_elements, BLOCK),)

    _apply_mask_gate_kernel[grid](
        # pointers
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        left_fused,
        right_fused,
        # shape scalars
        B, N, H,
        # strides (fp16 tensors)
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1), left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        # mask strides – only three dimensions (B,N,N)
        mask_fp16.stride(0) if has_mask else 0,
        mask_fp16.stride(1) if has_mask else 0,
        mask_fp16.stride(2) if has_mask else 0,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS/Tensor‑cores)
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) for a single bmm per hidden channel
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)

    # out_mat = left @ right.T   (FP16 inputs → FP16 output, acc in FP32)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))          # (B*H,N,N)

    # reshape back to (B,N,N,H) and cast to FP32 for the following LayerNorm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)
    out = out.to(dtype)  # FP32

    # --------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, gating, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    # out_gate is FP32, broadcast over (N,N)
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
