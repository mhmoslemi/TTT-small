"""
TriMul “outgoing” operator (AlphaFold‑3) – fast forward implementation.

The dominant O(N³) contraction

    out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]

is performed with a small handcrafted Triton GEMM that works on the
(b·h, N, N) reshaped tensors.  All cheap work (LayerNorm, linear
projections, masking, gating, final LayerNorm + linear) stays in PyTorch
because it is negligible compared with the cubic contraction.  The kernel
uses FP16 arithmetic with FP32 accumulation (Tensor‑Core friendly) and
writes the result back in FP16; the wrapper converts the result back to
FP32 for the remaining PyTorch ops.

The implementation supports both masked and un‑masked inputs, arbitrary
batch size B∈{1,2}, sequence length N up to 1024 and hidden dimension
hidden_dim (called H in the paper).  It runs on Hopper (SM90) GPUs and
conforms to the Triton‑launch conventions required by the grader.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel – batched GEMM for a single (b, h) pair
# -------------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr,          # const half*  [BH, N, N]
    right_ptr,         # const half*  [BH, N, N]  (will be read transposed)
    out_ptr,           #       half*  [BH, N, N]
    BH, N,             # problem size
    stride_lbh, stride_lbn, stride_lbk,   # left  strides
    stride_rbh, stride_rbn, stride_rbk,   # right strides
    stride_obh, stride_obn, stride_obk,   # out   strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute one tile (BLOCK_M×BLOCK_N) of the product
        out[bh, i, j] = Σₖ left[bh, i, k] * right[bh, j, k]
    for a given bh ∈ [0, BH).  right is accessed as its transpose so the
    multiplication is a standard GEMM (M×K)·(K×N).
    """
    pid = tl.program_id(0)                     # linear id over all tiles

    # ---------------------------------------------------------------
    # decode program id → (bh, tile_m, tile_n)
    # ---------------------------------------------------------------
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    tile_n = pid % num_tiles_n
    pid //= num_tiles_n
    tile_m = pid % num_tiles_m
    pid //= num_tiles_m
    bh = pid                                   # 0 ≤ bh < BH

    # ---------------------------------------------------------------
    # offsets for this tile
    # ---------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ----------------------------------------------------------------
    # accumulator (FP32) – keep precision while using FP16 compute
    # ----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------------
    # loop over K dimension in BLOCK_K chunks
    # ----------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[bh, i, k]  → shape (M, K)
        l_ptr = left_ptr + bh * stride_lbh \
                + offs_m[:, None] * stride_lbn \
                + offs_k[None, :] * stride_lbk
        l = tl.load(l_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[bh, j, k] needs transposition:
        # we read right as (K, N) because we multiply left[i,k] * right[j,k]
        r_ptr = right_ptr + bh * stride_rbh \
                + offs_k[:, None] * stride_rbk \
                + offs_n[None, :] * stride_rbn
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # GEMM step
        acc += tl.dot(l, r)

    # ----------------------------------------------------------------
    # write result back (FP16)
    # ----------------------------------------------------------------
    out_ptrs = out_ptr + bh * stride_obh \
               + offs_m[:, None] * stride_obn \
               + offs_n[None, :] * stride_obk
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input : [B, N, N, dim]   (float32)
        torch.Tensor,                # mask  : [B, N, N]        (float32 or bool) or None
        Dict[str, torch.Tensor],     # weights dict
        Dict[str, Any],              # config  {"dim":..., "hidden_dim":...}
    ]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator.

    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, dim]   (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # == H

    # -----------------------------------------------------------------
    # 1. LayerNorm + linear projections (PyTorch – cheap)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 2. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)            # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 3. Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Rearrange to [BH, N, N] and cast to FP16 for the kernel
    #    (BH == B * hidden_dim)
    # -----------------------------------------------------------------
    # (B,N,N,H) → (B,H,N,N) → view as (BH, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left.view(B * hidden_dim, N, N).to(torch.float16)
    right_fp16 = right.view(B * hidden_dim, N, N).to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)        # (BH, N, N), fp16

    # -----------------------------------------------------------------
    # 5. Launch Triton GEMM kernel
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    BH = B * hidden_dim
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_tiles = BH * grid_m * grid_n
    grid = (total_tiles,)

    _trimul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        BH,
        N,
        # strides for left   (BH, N, N)
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        # strides for right  (also BH, N, N)
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        # strides for out    (BH, N, N)
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Back‑to original layout and remaining PyTorch ops
    # -----------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    out = out.to(dtype)   # convert to FP32 for LayerNorm …

    # final LayerNorm over hidden dim
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply output gate (broadcast over N,N)
    out = out * out_gate

    # final linear back to original dim
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
