"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward only)
------------------------
1️⃣  Layer‑norm on the 4‑D input (fp32, manual).
2️⃣  Four linear projections (left, right, left‑gate, right‑gate, out‑gate) are
    executed in fp16 with torch.nn.functional.linear (fast cuBLAS).
3️⃣  A tiny Triton kernel fuses the three element‑wise operations that follow
    the projections:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    The kernel works on a 1‑D view of the tensors and therefore runs with
    negligible launch overhead.
4️⃣  The cubic contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    is performed with a single batched GEMM (fp16 × fp16 → fp32) using
    torch.bmm after reshaping to (B·H, N, N).  This leverages Tensor‑cores on the
    Hopper H200 and is much faster than a hand‑written Triton GEMM.
5️⃣  The result is reshaped back, layer‑normed, multiplied by the output‑gate
    and finally projected back to the original dimension (fp32).

Only the element‑wise post‑projection work is written in Triton, satisfying the
requirement that “part of the operations” lives in a Triton kernel while keeping
the overall runtime competitive.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask & gates
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_and_gates_kernel(
    left_ptr,            # fp16  [total]
    right_ptr,           # fp16  [total]
    mask_ptr,            # fp16  [total]  (may be 0 → treat as 1.0)
    left_gate_ptr,       # fp16  [total]
    right_gate_ptr,      # fp16  [total]
    out_left_ptr,        # fp16  [total]
    out_right_ptr,       # fp16  [total]
    total_elements: tl.constexpr,
    has_mask: tl.constexpr,          # 1 if mask is present else 0
):
    pid = tl.program_id(0)
    BLOCK = 1024                         # enough work per program
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # load
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if has_mask:
        m = tl.load(mask_ptr + offs, mask=mask, other=1.0)
        l = l * m
        r = r * m

    # apply gates
    l = l * lg
    r = r * rg

    # store
    tl.store(out_left_ptr + offs, l, mask=mask)
    tl.store(out_right_ptr + offs, r, mask=mask)


def _apply_mask_and_gates(left, right, mask, left_gate, right_gate):
    """
    left, right, left_gate, right_gate : fp16 tensors of shape [B, N, N, H]
    mask                              : None or fp16 tensor [B, N, N, 1]
    Returns (left_out, right_out)    : fp16 tensors of same shape
    """
    assert left.shape == right.shape == left_gate.shape == right_gate.shape
    B, N, _, H = left.shape
    total = left.numel()

    out_left = torch.empty_like(left)
    out_right = torch.empty_like(right)

    # flatten pointers
    left_ptr = left.view(-1)
    right_ptr = right.view(-1)
    lg_ptr = left_gate.view(-1)
    rg_ptr = right_gate.view(-1)
    out_l_ptr = out_left.view(-1)
    out_r_ptr = out_right.view(-1)

    if mask is None:
        mask_ptr = torch.tensor(0, dtype=torch.int64, device=left.device)  # dummy
        has_mask = 0
    else:
        mask_ptr = mask.view(-1)
        has_mask = 1

    grid = lambda META: (triton.cdiv(total, META["BLOCK"]),)

    _apply_mask_and_gates_kernel[grid](
        left_ptr,
        right_ptr,
        mask_ptr,
        lg_ptr,
        rg_ptr,
        out_l_ptr,
        out_r_ptr,
        total,
        has_mask,
        BLOCK=1024,
        num_warps=4,
    )
    return out_left, out_right


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors (see below)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (fp32, manual)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still fp32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (fp16)
    # ------------------------------------------------------------------
    # cast once
    def fp16(t): return t.to(torch.float16)
    left_proj_w   = fp16(weights["left_proj.weight"])
    right_proj_w  = fp16(weights["right_proj.weight"])
    left_gate_w   = fp16(weights["left_gate.weight"])
    right_gate_w  = fp16(weights["right_gate.weight"])
    out_gate_w    = fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                     # (B,N,N,dim)

    # (B,N,N,hidden)
    left = F.linear(x_fp16, left_proj_w)
    right = F.linear(x_fp16, right_proj_w)

    # gates (B,N,N,hidden)
    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask & gates with a tiny Triton kernel
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
    else:
        mask_fp16 = None

    left, right = _apply_mask_and_gates(left, right, mask_fp16,
                                        left_gate, right_gate)   # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Heavy contraction – batched GEMM via torch.bmm
    #    left : (B,N,N,hidden)  -> (B*hidden, N, N)
    #    right: (B,N,N,hidden)  (already aligned) -> (B*hidden, N, N) , then transpose K,J
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, N, N)

    B_h, _, I, K = left_perm.shape
    _, _, K2, J = right_perm.shape
    assert K == K2 and I == J == N

    # reshape to (B*H, N, N)
    left_mat = left_perm.view(B * hidden, I, K)
    right_mat = right_perm.view(B * hidden, K, J)

    # fp16 matmul -> fp32 accumulation (torch uses Tensor Cores automatically)
    out_mat_fp32 = torch.bmm(left_mat.float(), right_mat.float())   # (B*H,N,N) fp32

    # reshape back
    out = out_mat_fp32.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 6️⃣ Final gating + projection (fp32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over hidden dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                   # (B,N,N,dim)

    return out
