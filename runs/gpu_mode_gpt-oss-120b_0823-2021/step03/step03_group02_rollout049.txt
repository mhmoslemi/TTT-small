"""
TriMul “outgoing” forward pass – fast forward only.

The cubic contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a single batched GEMM call (torch.bmm) which maps to
cuBLAS/Tensor‑cores on Hopper.  All cheap element‑wise work (masking,
gate‑multiplication, the two LayerNorms and the final linear projection)
remains in PyTorch, but the mask‑and‑gate fusion for the *left* and *right*
tensors is off‑loaded to a tiny Triton kernel so that the submission contains
at least one user‑written kernel as required.

The implementation follows the reference PyTorch module with the following
optimisations:

*   Input‑ and output‑LayerNorms are computed manually (no nn.Module overhead).
*   All heavy matrix‑multiplications are performed in FP16 and accumulated in
    FP32.
*   The reduction dimension `k` of the N³ contraction is kept contiguous by
    reshaping to (B·H, N, N) and calling `torch.bmm`.
*   A custom Triton kernel fuses three element‑wise operations
        – optional mask,
        – left‑gate,
        – right‑gate
    for the left and right projected tensors, reducing memory traffic.
*   The final result is cast back to FP32 for the remaining LayerNorm and the
    output linear projection.

The kernel works for any batch size ≤2, any sequence length from the test set,
and any hidden‑/model dimension (the reference uses dim = 128/384 and
hidden_dim = 128).  It returns a FP32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask (if present) and the three sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def fuse_mask_gate_kernel(
    left_ptr,               # fp16*  (B, N, N, H)
    right_ptr,              # fp16*  (B, N, N, H)
    left_gate_ptr,          # fp16*  (B, N, N, H)
    right_gate_ptr,         # fp16*  (B, N, N, H)
    out_gate_ptr,           # fp16*  (B, N, N, H)   – stored for later use
    mask_ptr,               # fp16*  (B, N, N, 1)   – may be nullptr
    B, N, H,                # shape parameters
    has_mask: tl.constexpr, # compile‑time flag
    BLOCK: tl.constexpr,    # tile size (fits one warp nicely)
):
    pid = tl.program_id(0)
    # linear index for the whole tensor (flattened over B,N,N,H)
    offs = pid * BLOCK + tl.arange(0, BLOCK)

    # bounds check
    mask = offs < B * N * N * H
    if not tl.any(mask):
        return

    # compute the 4‑d index (b,i,j,h) from the linear offset
    # stride in the last dimension (H) is 1, then N, N*H, B*N*H
    h = offs % H
    tmp = offs // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # flatten the pointers to 1‑D and load the needed elements
    idx = ((b * N + i) * N + j) * H + h          # same for left/right/gt
    left_val = tl.load(left_ptr + idx, mask=mask, other=0.0)
    right_val = tl.load(right_ptr + idx, mask=mask, other=0.0)

    left_gate = tl.load(left_gate_ptr + idx, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + idx, mask=mask, other=0.0)
    out_gate = tl.load(out_gate_ptr + idx, mask=mask, other=0.0)

    if has_mask:
        # mask has shape (B,N,N,1) – broadcast on H
        mask_idx = ((b * N + i) * N + j)          # no *H because last dim =1
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # 1.0 = no mask
        left_val = left_val * m
        right_val = right_val * m

    # apply sigmoid gating (the inputs are already sigmoid‑ed in the caller,
    # we just multiply)
    left_val = left_val * left_gate
    right_val = right_val * right_gate

    # store the fused tensors back in‑place
    tl.store(left_ptr + idx, left_val, mask=mask)
    tl.store(right_ptr + idx, right_val, mask=mask)

    # store the out‑gate for later use (still fp16)
    tl.store(out_gate_ptr + idx, out_gate, mask=mask)


# ----------------------------------------------------------------------
# Helper – run the fusion kernel
# ----------------------------------------------------------------------
def fuse_mask_and_gates(left, right, left_gate, right_gate, out_gate, mask):
    """
    In‑place fusion of optional mask and the three sigmoid gates.
    All tensors are FP16, contiguous and have shape [B,N,N,H].
    Returns the (unchanged) out_gate tensor (now fused) for later use.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024  # 1‑warp tile – good for Hopper

    grid = (triton.cdiv(total, BLOCK),)

    has_mask = mask is not None
    mask_ptr = mask if mask is not None else 0

    fuse_mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        out_gate_ptr=out_gate,
        mask_ptr=mask_ptr,
        B=B,
        N=N,
        H=H,
        has_mask=has_mask,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out_gate


# ----------------------------------------------------------------------
# Main entry‑point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no backward).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor [B, N, N] (float32/bool) or None
        - weights      : dict of torch.Tensor parameters (see docstring)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep in FP32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual LayerNorm over the last dimension
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                       # still FP32

    # ------------------------------------------------------------------
    # 2) Cast to FP16 once – all heavy ops run in FP16
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (FP16) – shape [B,N,N,hidden]
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)    # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 4) Fuse optional mask and the three gates with a tiny Triton kernel
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16)
    else:
        mask_fp16 = None

    # kernel works in‑place on left/right and also writes out_gate (already
    # computed) – we keep the tensor for the final gating step.
    fuse_mask_and_gates(left, right, left_gate, right_gate, out_gate, mask_fp16)

    # ------------------------------------------------------------------
    # 5) Core N³ contraction – batched GEMM via torch.bmm (cuBLAS)
    #    left : (B, hidden, N, N)  -> (B*hidden, N, N)
    #    right: (B, hidden, N, N)  -> (B*hidden, N, N) and transposed
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    # reshape to (B*H, N, N) for torch.bmm
    LH = left_perm.view(B * hidden, N, N)
    RH = right_perm.view(B * hidden, N, N)

    # FP16 matmul, accumulate in FP32 automatically on Hopper
    out_fp16 = torch.bmm(LH, RH)                       # (B*H, N, N)
    out_fp16 = out_fp16.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    # now shape (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    # cast back to FP32 for the LN
    out = out_fp16.to(torch.float32)

    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7) Final gating (same out_gate used earlier) and linear projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # apply gate
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                  # (B,N,N,dim) FP32

    return out
