"""
TriMul “outgoing” forward pass (no gradients).

The N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a single torch.bmm (leveraging cuBLAS/Tensor‑cores).
All other heavy element‑wise work (masking + the three gates) is fused into
a tiny Triton kernel, keeping the whole pipeline in fp16 until the final
LayerNorms and output projection, which run in fp32 for numerical stability.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused mask & gating
# Input tensors are fp16, output tensors are fp16.
#   left  : (B, N, N, H)   – already linearly projected
#   right : (B, N, N, H)   – already linearly projected
#   mask  : (B, N, N, 1)   – broadcasted on H (may be None → all ones)
#   lg    : (B, N, N, H)   – left‑gate (sigmoid already applied)
#   rg    : (B, N, N, H)   – right‑gate (sigmoid already applied)
#   out_left  , out_right : (B, N, N, H)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    left_ptr, right_ptr, mask_ptr,
    lg_ptr, rg_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    MASKED: tl.constexpr,          # 1 → mask present, 0 → treat as all‑ones
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                 # flatten all elements
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear offsets
    mask = offs < total

    # decode linear offset → (b,i,j,h)
    tmp = offs
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # pointer arithmetic
    left = tl.load(
        left_ptr
        + b * stride_left_b
        + i * stride_left_i
        + j * stride_left_j
        + h * stride_left_h,
        mask=mask,
        other=0.0,
    )
    right = tl.load(
        right_ptr
        + b * stride_right_b
        + i * stride_right_i
        + j * stride_right_j
        + h * stride_right_h,
        mask=mask,
        other=0.0,
    )
    lg = tl.load(
        lg_ptr
        + b * stride_lg_b
        + i * stride_lg_i
        + j * stride_lg_j
        + h * stride_lg_h,
        mask=mask,
        other=0.0,
    )
    rg = tl.load(
        rg_ptr
        + b * stride_rg_b
        + i * stride_rg_i
        + j * stride_rg_j
        + h * stride_rg_h,
        mask=mask,
        other=0.0,
    )
    # mask (if present) is a scalar per (b,i,j) → broadcast on H
    if MASKED == 1:
        m = tl.load(
            mask_ptr
            + b * stride_mask_b
            + i * stride_mask_i
            + j * stride_mask_j,
            mask=mask,
            other=1.0,                     # missing entries → 1 (no effect)
        )
        left = left * m * lg
        right = right * m * rg
    else:
        left = left * lg
        right = right * rg

    tl.store(
        out_left_ptr
        + b * stride_out_b
        + i * stride_out_i
        + j * stride_out_j
        + h * stride_out_h,
        left,
        mask=mask,
    )
    tl.store(
        out_right_ptr
        + b * stride_out_b
        + i * stride_out_i
        + j * stride_out_j
        + h * stride_out_h,
        right,
        mask=mask,
    )


def fuse_mask_and_gates(left, right, mask, left_gate, right_gate):
    """
    left, right, left_gate, right_gate : (B,N,N,H) fp16
    mask                               : (B,N,N) fp16 or None
    Returns fused left, right tensors (B,N,N,H) fp16.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024

    out_left = torch.empty_like(left)
    out_right = torch.empty_like(right)

    # strides in element units (not bytes)
    sL = left.stride()
    sR = right.stride()
    sLG = left_gate.stride()
    sRG = right_gate.stride()
    sO = out_left.stride()

    if mask is None:
        mask_ptr = torch.tensor(0, device=left.device).data_ptr()  # dummy
        MASKED = 0
        sM = (0, 0, 0)  # not used
    else:
        mask_ptr = mask.data_ptr()
        MASKED = 1
        sM = mask.stride()

    grid = (triton.cdiv(total, BLOCK),)

    mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask_ptr,
        lg_ptr=left_gate,
        rg_ptr=right_gate,
        out_left_ptr=out_left,
        out_right_ptr=out_right,
        B=B,
        N=N,
        H=H,
        stride_left_b=sL[0],
        stride_left_i=sL[1],
        stride_left_j=sL[2],
        stride_left_h=sL[3],
        stride_right_b=sR[0],
        stride_right_i=sR[1],
        stride_right_j=sR[2],
        stride_right_h=sR[3],
        stride_mask_b=sM[0] if MASKED else 0,
        stride_mask_i=sM[1] if MASKED else 0,
        stride_mask_j=sM[2] if MASKED else 0,
        stride_lg_b=sLG[0],
        stride_lg_i=sLG[1],
        stride_lg_j=sLG[2],
        stride_lg_h=sLG[3],
        stride_rg_b=sRG[0],
        stride_rg_i=sRG[1],
        stride_rg_j=sRG[2],
        stride_rg_h=sRG[3],
        stride_out_b=sO[0],
        stride_out_i=sO[1],
        stride_out_j=sO[2],
        stride_out_h=sO[3],
        MASKED=MASKED,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out_left, out_right


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]  (float32, CUDA)
        mask         : torch.Tensor [B, N, N]      (float32/bool) or None
        weights      : dict of torch.Tensor parameters
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]   (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm  (keep in fp32 for stability)
    # ------------------------------------------------------------------
    w_ln = weights["norm.weight"]
    b_ln = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * w_ln + b_ln          # (B,N,N,dim) fp32

    # ------------------------------------------------------------------
    # 2) Linear projections (cast once to fp16)
    # ------------------------------------------------------------------
    # weight matrices are stored as (out_features, in_features) → same as nn.Linear
    def to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w  = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w  = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w   = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden) fp16
    right = F.linear(x_fp16, right_proj_w)  # (B,N,N,hidden) fp16

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 4) Fuse mask & gates with a tiny Triton kernel
    #    (mask may be None → treat as all‑ones)
    # ------------------------------------------------------------------
    left_f, right_f = fuse_mask_and_gates(left, right,
                                          mask.float() if mask is not None else None,
                                          left_gate, right_gate)

    # ------------------------------------------------------------------
    # 5) Core contraction – batched GEMM using torch.bmm (fast cuBLAS)
    #    left_f : (B, N, N, hidden) → (B*hidden, N, N)
    #    right_f: (B, N, N, hidden) → (B*hidden, N, N)  (transpose last 2 dims)
    #    result : (B*hidden, N, N) → reshape back → (B,N,N,hidden)
    # ------------------------------------------------------------------
    LH = left_f.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    RH = right_f.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B, H, _, _ = LH.shape
    LH_flat = LH.view(B * H, N, N)        # (B*H, N, N)
    RH_flat = RH.view(B * H, N, N)        # (B*H, N, N)

    # matmul over the reduction dimension N
    out_flat = torch.bmm(LH_flat, RH_flat)   # (B*H, N, N) fp16
    out_fp16 = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)   # back to fp32 for LN
    w_out_ln = weights["to_out_norm.weight"]
    b_out_ln = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * w_out_ln + b_out_ln

    # ------------------------------------------------------------------
    # 7) Final gating + projection (fp32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # same out_gate as before
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
