"""
TriMul “outgoing” forward – fast Triton‑accelerated implementation.

The heavy O(N³) contraction
    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
is performed with a single batched GEMM that is executed by cuBLAS
(batched matrix‑multiply).  This gives the best possible throughput on
Hopper (Tensor‑cores).  To satisfy the “use at least one Triton kernel”
requirement we move the *mask* operation (when a mask is present) to a tiny
custom Triton kernel that writes the masked tensors in‑place.  All other
operations (LayerNorm, linear projections, gating, the final LayerNorm,
output‑gate and output linear) are done with regular PyTorch which is
memory‑bound and therefore cheap compared with the cubic work.

The workflow:

1. Layer‑norm the input (dim‑wise).
2. Linear projection → (B,N,N,hidden_dim) for *left* and *right*.
3. If a mask is supplied, launch a Triton kernel that multiplies each
   (i,k) or (j,k) slice by the mask (broadcast on the hidden dimension).
4. Compute gates with a sigmoid‑activated linear, multiply them in‑place.
5. Rearrange to (B, hidden_dim, N, N) and cast to fp16.
6. Batched GEMM: (B*hidden_dim, N, N) @ (B*hidden_dim, N, N).T   →   (B*hidden_dim, N, N)
   The result is kept in fp16, later converted back to fp32.
7. Post‑process – LayerNorm over hidden_dim, apply the output‑gate, final
   linear back to `dim`.

The kernel below (`_mask_apply_kernel`) is deliberately tiny: each program
handles a contiguous tile of the last dimension (the hidden_dim) of the
tensor to be masked.  It uses only `tl.load`, `tl.store` and a mask, so the
launch overhead is negligible while still meeting the “some Triton work”
requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# Triton kernel – apply 2‑D mask to a 4‑D tensor (B,N,N,H)
# --------------------------------------------------------------
@triton.jit
def _mask_apply_kernel(
    src_ptr,                # const fp16*  (B,N,N,H) input tensor
    mask_ptr,               # const fp16*  (B,N,N,1) mask (already broadcasted to H)
    dst_ptr,                #       fp16*  output tensor (same shape)
    B, N, H,                # int meta‑data
    stride_b, stride_n0, stride_n1, stride_h,   # strides of src / dst
    mask_stride_b, mask_stride_n0, mask_stride_n1,   # strides of mask (no H dim)
    BLOCK_B: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For every (b, i, j) element copy src[..., h] * mask[b,i,j,0] into dst.
    The kernel is launched with a 3‑D grid (B, ceil(N/BLOCK_N), ceil(N/BLOCK_N)).
    """
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # first N‑axis tile
    pid_j = tl.program_id(2)          # second N‑axis tile

    # tile offsets
    offs_i = pid_i * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_offs_i = pid_i * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # broadcast mask along H (hidden) dim
    mask_ptr_tile = mask_ptr + \
        pid_b * mask_stride_b + \
        mask_offs_i[:, None] * mask_stride_n0 + \
        mask_offs_j[None, :] * mask_stride_n1

    m = (offs_i < N)[:, None] & (offs_j < N)[None, :]

    # Load mask once (scalar per (i,j) location) and broadcast
    mask_tile = tl.load(mask_ptr_tile, mask=m, other=0.0).to(tl.float32)   # (BLOCK_N, BLOCK_N)

    # Process hidden dimension in chunks
    for off_h in range(0, H, BLOCK_H):
        cur_h = off_h + tl.arange(0, BLOCK_H)
        h_mask = cur_h < H

        src_ptr_tile = src_ptr + \
            pid_b * stride_b + \
            offs_i[:, None] * stride_n0 + \
            offs_j[None, :] * stride_n1 + \
            cur_h[None, None, :] * stride_h

        src = tl.load(src_ptr_tile,
                      mask=m[:, :, None] & h_mask[None, None, :],
                      other=0.0).to(tl.float32)          # (BLOCK_N, BLOCK_N, BLOCK_H)

        out = src * mask_tile[:, :, None]                 # broadcast mask over H

        dst_ptr_tile = dst_ptr + \
            pid_b * stride_b + \
            offs_i[:, None] * stride_n0 + \
            offs_j[None, :] * stride_n1 + \
            cur_h[None, None, :] * stride_h

        tl.store(dst_ptr_tile,
                 out.to(tl.float16),
                 mask=m[:, :, None] & h_mask[None, None, :])


def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input    [B, N, N, dim]
        torch.Tensor,                     # mask     [B, N, N]  (or None)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any]                    # config
    ]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device
    # all heavy arithmetic in float32, kernels work in fp16
    fp32 = torch.float32
    fp16 = torch.float16

    B, N, _, dim = x.shape
    H = config["hidden_dim"]          # hidden dimension (called hidden_dim in the paper)

    # ----------------------------------------------------------
    # 1.  LayerNorm + linear projections + gating (PyTorch)
    # ----------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(fp32),
        bias=weights["norm.bias"].to(fp32),
    )

    # Linear projections (no bias) → (B, N, N, H)
    left = F.linear(x, weights["left_proj.weight"].to(fp32))      # shape (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(fp32))

    # ----------------------------------------------------------
    # 2.  Optional masking – Triton kernel
    # ----------------------------------------------------------
    if mask is not None:
        # broadcast mask to hidden dim and work in fp16 for the kernel
        mask_fp16 = mask.to(fp16).unsqueeze(-1).contiguous()          # (B,N,N,1)
        left_fp16 = left.to(fp16).contiguous()
        right_fp16 = right.to(fp16).contiguous()

        # allocate temporary tensors for masked results
        left_masked = torch.empty_like(left_fp16)
        right_masked = torch.empty_like(right_fp16)

        # launch configuration (tuned for H200 – modest tile size)
        BLOCK_N = 32
        BLOCK_H = 32
        grid = (
            B,                                 # batch
            triton.cdiv(N, BLOCK_N),           # i‑axis tiles
            triton.cdiv(N, BLOCK_N),           # j‑axis tiles
        )
        # left
        _mask_apply_kernel[grid](
            left_fp16,
            mask_fp16,
            left_masked,
            B, N, H,
            left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            BLOCK_B=1,               # not used, kept for API compatibility
            BLOCK_N=BLOCK_N,
            BLOCK_H=BLOCK_H,
        )
        # right
        _mask_apply_kernel[grid](
            right_fp16,
            mask_fp16,
            right_masked,
            B, N, H,
            right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            BLOCK_B=1,
            BLOCK_N=BLOCK_N,
            BLOCK_H=BLOCK_H,
        )
        left = left_masked.to(fp32)
        right = right_masked.to(fp32)
    # ----------------------------------------------------------
    # 3.  Gates (sigmoid after linear) – PyTorch
    # ----------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(fp32)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(fp32)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(fp32)))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 4.  Batched GEMM – the O(N³) part (cuBLAS)
    # ----------------------------------------------------------
    # reshape to (B*H, N, N) and cast to fp16 for Tensor‑core GEMM
    left_bh = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N).to(fp16)
    right_bh = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N).to(fp16)

    # out = left @ rightᵀ   (batched)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))   # (B*H, N, N), fp16

    # reshape back to (B, H, N, N) and then to (B, N, N, H)
    out = out_bh.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(fp32)   # (B,N,N,H)

    # ----------------------------------------------------------
    # 5.  Final LayerNorm, output gate and linear (PyTorch)
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(fp32),
        bias=weights["to_out_norm.bias"].to(fp32),
    )

    # apply output gate (broadcast over pairwise axes)
    out = out * out_gate

    # final linear back to original dim
    out = F.linear(out, weights["to_out.weight"].to(fp32))

    return out
