"""
TriMul “outgoing” forward pass (AlphaFold‑3) – mixed PyTorch / Triton implementation.

The heavy O(N³) contraction
    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
is performed by a small, highly‑tuned Triton GEMM kernel that works on
FP16 data with FP32 accumulation.  All cheap element‑wise work (LayerNorm,
linear projections, gating, final LayerNorm / linear) stays in regular
PyTorch.  This gives a fast end‑to‑end kernel while still satisfying the
requirement that *some* work is done inside Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ---------------------------------------------------------
# Triton kernel: batched GEMM for the (B,H,N,N) contraction
# ---------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr,          # const fp16*  [B, H, N, N]   (row‑major)
    right_ptr,         # const fp16*  [B, H, N, N]   (row‑major, will be read transposed)
    out_ptr,           #       fp16*  [B, H, N, N]
    B, N, H,           # meta‑data
    stride_lb, stride_lh, stride_ln, stride_lk,   # left  strides
    stride_rb, stride_rh, stride_rn, stride_rk,   # right strides
    stride_ob, stride_oh, stride_on, stride_ok,   # out   strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a (BLOCK_M × BLOCK_N) tile of the 4‑D result

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    One program instance handles a single (b, h, tile_m, tile_n) tile.
    """
    pid = tl.program_id(0)

    # -------------------------------------------------
    # decode the 1‑D pid into (b, h, tile_m, tile_n)
    # -------------------------------------------------
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    tile_n = pid % num_n
    pid //= num_n
    tile_m = pid % num_m
    pid //= num_m
    h = pid % H
    b = pid // H

    # -------------------------------------------------
    # tile offsets and masks
    # -------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32 (more numerically stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # loop over K dimension in tiles
    # -------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, h, i, k]   → shape (M, K)
        l_ptr = (left_ptr +
                 b * stride_lb +
                 h * stride_lh +
                 offs_m[:, None] * stride_ln +
                 offs_k[None, :] * stride_lk)
        l = tl.load(l_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[b, h, j, k] ^T → we need (K, N)
        # original layout: (b, h, j, k)
        # after transpose: (k, j)
        r_ptr = (right_ptr +
                 b * stride_rb +
                 h * stride_rh +
                 offs_k[:, None] * stride_rk +          # k dimension (now row)
                 offs_n[None, :] * stride_rn)           # j dimension (now col)
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # matrix multiply tile
        acc += tl.dot(l, r)

    # -------------------------------------------------
    # write back the tile
    # -------------------------------------------------
    out_ptrs = (out_ptr +
                b * stride_ob +
                h * stride_oh +
                offs_m[:, None] * stride_on +
                offs_n[None, :] * stride_ok)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ---------------------------------------------------------
# Host wrapper – the public entry point
# ---------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input tensor      [B, N, N, dim]
        torch.Tensor,                # mask tensor       [B, N, N]  (or None)
        Dict[str, torch.Tensor],     # model weights
        Dict[str, Any]               # config dict with dim / hidden_dim etc.
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the same layout as the
        reference PyTorch implementation.
    """
    x, mask, weights, config = data
    device = x.device
    # all heavy work will be done in FP16, but keep the final dtype FP32
    out_dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]      # = H

    # -----------------------------------------------------------------
    # 1) cheap element‑wise ops – stay in PyTorch (LayerNorm, linear, gates)
    # -----------------------------------------------------------------
    # LayerNorm over last dimension
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(out_dtype),
        bias=weights["norm.bias"].to(out_dtype),
    )

    # Linear projections (bias is omitted in the reference code)
    left = F.linear(x, weights["left_proj.weight"].to(out_dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(out_dtype))

    # optional mask – broadcast on the hidden dimension
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # gating – sigmoid after a linear
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(out_dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(out_dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(out_dtype)))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 2) reshape to [B, H, N, N] and cast to fp16 for the kernel
    # -----------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output buffer (fp16) – same layout as left/right
    out_fp16 = torch.empty_like(left_fp16)

    # -----------------------------------------------------------------
    # 3) launch Triton kernel (blocked batched GEMM)
    # -----------------------------------------------------------------
    # tile sizes – experimentally good for Hopper (sm90) with H200 memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    # total programs = B * H * grid_m * grid_n
    grid = (B * hidden_dim * grid_m * grid_n,)

    _trimul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        # strides for left  [B,H,N,N]
        left_fp16.stride(0),   # B
        left_fp16.stride(1),   # H
        left_fp16.stride(2),   # N (row i)
        left_fp16.stride(3),   # N (col k)
        # strides for right [B,H,N,N]
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        # strides for out   [B,H,N,N]
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 4) post‑processing in PyTorch (LayerNorm, out‑gate, final linear)
    # -----------------------------------------------------------------
    # back to original layout: (B,H,N,N) -> (B,N,N,H)
    out = out_fp16.to(out_dtype).permute(0, 2, 3, 1).contiguous()

    # final LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(out_dtype),
        bias=weights["to_out_norm.bias"].to(out_dtype),
    )

    # apply output gate (broadcast on pair‑wise axes)
    out = out * out_gate

    # final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"].to(out_dtype))

    return out
