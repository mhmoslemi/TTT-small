"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward only)
------------------------
1️⃣  Layer‑norm the input 4‑D tensor (float32) – done by a small Triton kernel
    that computes mean/variance per vector (‑dim axis) and applies the
    learned weight/bias.

2️⃣  Cast to fp16 once and perform all linear projections / gating with
    PyTorch F.linear for simplicity (the heavy work is the N³ contraction).

3️⃣  Optional binary mask is applied in fp16 (broadcast over the hidden dim).

4️⃣  Gating – sigmoid on the three gate tensors.

5️⃣  Core contraction  
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .  
    This is a batched matrix‑multiply with shape  
       (B·H, N, N) × (B·H, N, N)ᵀ → (B·H, N, N) .  
    We reshape the tensors and call ``torch.bmm`` which uses cuBLAS/Tensor‑Core
    on Hopper (H200), giving the best performance.

6️⃣  Second Layer‑norm (again Triton), gate, and final linear projection to the
    original dimension.  The final output is float32.

Only step 1 and step 6 are written in Triton – they are pure elementwise
operations that benefit from the massive parallelism of the H200 while keeping
the code simple and robust.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (float32) – operates on the last dimension of a
# contiguous tensor of shape (..., D).  The weight and bias are 1‑D
# tensors of length D (float32).  The kernel is launched with a 1‑D grid
# over the outer elements.
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,                # input pointer (float32)
    y_ptr,                # output pointer (float32)
    weight_ptr,           # gamma (float32)
    bias_ptr,             # beta  (float32)
    N,                    # size of the normalized dimension
    eps,                  # epsilon (float32, constexpr)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)                     # one program per outer element
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

    mask = offs < N

    # ------------------------------------------------------------------
    # Load a block of the vector, compute mean & variance in fp32.
    # ------------------------------------------------------------------
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    # mean
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / N

    # variance
    diff = x - mean
    sum_sq = tl.sum(diff * diff, axis=0)
    var = sum_sq / N

    # ------------------------------------------------------------------
    # Normalize, apply weight/bias.
    # ------------------------------------------------------------------
    inv_std = 1.0 / tl.sqrt(var + eps)
    norm = (x - mean) * inv_std

    gamma = tl.load(weight_ptr + offs, mask=mask, other=1.0)
    beta  = tl.load(bias_ptr   + offs, mask=mask, other=0.0)

    y = norm * gamma + beta
    tl.store(y_ptr + offs, y, mask=mask)


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Triton implementation of LayerNorm over the last dimension.
    x, weight, bias must be contiguous float32 tensors.
    """
    assert x.is_contiguous() and weight.is_contiguous() and bias.is_contiguous()
    B, *rest, D = x.shape
    outer = x.numel() // D                # number of independent vectors
    y = torch.empty_like(x)

    BLOCK = 128                           # enough for D up to 1024, power‑of‑2
    grid = (triton.cdiv(outer, BLOCK),)

    # reinterpret the tensors as flat 1‑D pointers; the kernel treats each
    # program as a contiguous block of D elements.
    x_ptr = x.view(-1)
    y_ptr = y.view(-1)
    w_ptr = weight.view(-1)
    b_ptr = bias.view(-1)

    _layernorm_kernel[grid](
        x_ptr,
        y_ptr,
        w_ptr,
        b_ptr,
        N=D,
        eps=eps,
        BLOCK_SIZE=BLOCK,
        num_warps=4,                     # fp32 friendly on SM90
    )
    return y


# ----------------------------------------------------------------------
# Main entry point – forward only
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool or float)
        weights      : dict of leaf tensors (see code below)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (float32) – Triton
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"].to(torch.float32).contiguous()
    ln_b = weights["norm.bias"  ].to(torch.float32).contiguous()
    x = triton_layernorm(input_tensor, ln_w, ln_b)               # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2️⃣ Cast once to fp16 and compute all linear projections
    # ------------------------------------------------------------------
    # weights (fp16) – they are stored as float32 in the checkpoint
    def fp16(t): return t.to(torch.float16).contiguous()
    left_proj_w   = fp16(weights["left_proj.weight"])
    right_proj_w  = fp16(weights["right_proj.weight"])
    left_gate_w   = fp16(weights["left_gate.weight"])
    right_gate_w  = fp16(weights["right_gate.weight"])
    out_gate_w    = fp16(weights["out_gate.weight"])
    to_out_w      = fp16(weights["to_out.weight"])

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction – batched GEMM via torch.bmm (fast cuBLAS)
    #    reshape to (B*H, N, N) and (B*H, N, N)ᵀ
    # ------------------------------------------------------------------
    # left : (B,N,N,hidden) → (B, hidden, N, N) → (B*hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # transpose K,J

    B_H, _, _ = left_perm.shape[0] * left_perm.shape[1], N, N   # just for readability
    left_mat = left_perm.view(B * hidden_dim, N, N)   # (B*H, I, K)
    right_mat = right_perm.view(B * hidden_dim, N, N) # (B*H, K, J)

    # batched matrix multiply – fp16×fp16→fp16 (Tensor‑cores)
    out_mat = torch.bmm(left_mat, right_mat)          # (B*H, I, J)

    # reshape back to (B, N, N, hidden)
    out_fp16 = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_fp16.to(torch.float32)                  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (float32) – Triton
    # ------------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"].to(torch.float32).contiguous()
    out_ln_b = weights["to_out_norm.bias"  ].to(torch.float32).contiguous()
    out = triton_layernorm(out, out_ln_w, out_ln_b)

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over hidden dim
    out = F.linear(out, to_out_w)                   # (B,N,N,dim), still fp32

    return out
