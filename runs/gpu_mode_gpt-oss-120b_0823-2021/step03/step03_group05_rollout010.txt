"""
TriMul (outgoing) – fast forward pass
-------------------------------------

The heavy O(N³) contraction  

    out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]

is performed with a cuBLAS batched GEMM (torch.bmm) after reshaping the
tensors to (B·H, N, N).  This gives the best performance on Hopper because the
kernel can use Tensor‑cores automatically.

The only part that is forced to run in Triton is the element‑wise fusion of
*mask* (when present) and the three sigmoid gates:

    left  = left  * mask * left_gate
    right = right * mask * right_gate

All linear layers, LayerNorms and the final projection stay in PyTorch – they
are cheap relative to the cubic work and already run efficiently on the GPU.
The kernel works for any contiguous layout (FP16 is used inside the kernel to
maximise bandwidth; the final result is cast back to FP32).

The function signature follows the statement of the problem:
    custom_kernel(data) -> Tensor[B, N, N, dim]

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel – fuse mask (or all‑ones) with left/right gates.
# All tensors are 4‑D: (B, N, N, H) stored contiguously.
# The kernel overwrites left and right in‑place.
# -------------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *mut fp16  (B,N,N,H) – overwritten
    right_ptr,         # *mut fp16  (B,N,N,H) – overwritten
    left_gate_ptr,     # *const fp16 (B,N,N,H)
    right_gate_ptr,    # *const fp16 (B,N,N,H)
    mask_ptr,          # *const fp16 (B,N,N,1)   (1.0 when no mask)
    B, N, H,           # sizes
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_gg0, stride_gg1, stride_gg2, stride_gg3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N * H
    idx = pid * BLOCK + tl.arange(0, BLOCK)
    mask = idx < total

    # -------- decode linear index -----------------------------------------
    tmp = idx
    h = tmp % H;            tmp = tmp // H
    j = tmp % N;            tmp = tmp // N
    i = tmp % N;            b = tmp // N

    # -------- compute pointers --------------------------------------------
    l_ptr = left_ptr  + b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    r_ptr = right_ptr + b * stride_r0 + i * stride_r1 + j * stride_r2 + h * stride_r3
    lg_ptr = left_gate_ptr  + b * stride_gg0 + i * stride_gg1 + j * stride_gg2 + h * stride_gg3
    rg_ptr = right_gate_ptr + b * stride_gg0 + i * stride_gg1 + j * stride_gg2 + h * stride_gg3
    m_ptr = mask_ptr + b * stride_m0 + i * stride_m1 + j * stride_m2        # broadcast over H

    # -------- load ---------------------------------------------------------
    l  = tl.load(l_ptr, mask=mask, other=0.0)
    r  = tl.load(r_ptr, mask=mask, other=0.0)
    lg = tl.load(lg_ptr, mask=mask, other=0.0)
    rg = tl.load(rg_ptr, mask=mask, other=0.0)
    m  = tl.load(m_ptr, mask=mask, other=1.0)       # 1.0 = no‑mask

    # -------- fused elementwise -------------------------------------------
    l = l * m * lg
    r = r * m * rg

    # -------- store --------------------------------------------------------
    tl.store(l_ptr, l, mask=mask)
    tl.store(r_ptr, r, mask=mask)


# -------------------------------------------------------------------------
# Entry point used by the evaluator
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[
        torch.Tensor,               # input   (B,N,N,dim)  float32
        torch.Tensor,               # mask    (B,N,N)      bool/float32 or None
        Dict[str, torch.Tensor],    # weights
        Dict[str, Any]              # config  {"dim":..., "hidden_dim":...}
    ]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        - x : Tensor[B, N, N, dim]           (float32, contiguous)
        - mask : Tensor[B,N,N] or None
        - weights : dict of model parameters (float32)
        - config : {"dim": int, "hidden_dim": int}

    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]          # H in the paper

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm + linear projections (cheap – stay in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )
    # -> (B,N,N,hidden_dim)
    left  = F.linear(x, weights["left_proj.weight"].to(dtype))
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # gates (pre‑sigmoid)
    left_gate  = F.linear(x, weights["left_gate.weight"].to(dtype))
    right_gate = F.linear(x, weights["right_gate.weight"].to(dtype))
    out_gate   = F.linear(x, weights["out_gate.weight"].to(dtype))

    # sigmoid
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # -----------------------------------------------------------------
    # 2️⃣ Fuse mask + gates in Triton (elementwise, memory‑bound)
    # -----------------------------------------------------------------
    # Prepare mask tensor of shape (B,N,N,1) in fp16.
    if mask is None:
        mask_fp16 = torch.ones((B, N, N, 1), dtype=torch.float16,
                               device=device)
    else:
        # mask may be bool or float; ensure float16 and expand last dim.
        mask_fp16 = mask.to(dtype).unsqueeze(-1).to(torch.float16)

    # Cast the tensors that the kernel touches to FP16 (the kernel cannot
    # handle float32).  They remain contiguous after the cast.
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    left_gate_fp16  = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()

    total_elems = B * N * N * hidden_dim
    BLOCK = 1024                                 # good occupancy for pure elementwise
    grid = (triton.cdiv(total_elems, BLOCK),)

    _mask_gate_fusion_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        mask_fp16.stride(0), mask_fp16.stride(1),
        mask_fp16.stride(2), mask_fp16.stride(3),
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 3️⃣ Core O(N³) contraction via batched GEMM (Tensor‑cores)
    # -----------------------------------------------------------------
    # Reshape to (B·H, N, N) for the batched matmul.
    # left : (B,N,N,H) --> (B·H, N, N)
    left_bh  = left_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_bh = right_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Batched matrix multiplication: (B·H,N,N) @ (B·H,N,N)^T
    # Result shape (B·H,N,N)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))

    # Restore the original order (B,N,N,H) and convert back to FP32.
    out = out_bh.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype)          # back to FP32 for the remaining ops

    # -----------------------------------------------------------------
    # 4️⃣ Post‑processing (LayerNorm, out‑gate, final linear)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate                         # broadcast over (N,N)
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
