"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward only)
------------------------
1️⃣  Layer‑norm over the last channel of the 4‑D tensor (fp32).  
    Implemented with a tiny Triton kernel that computes mean/var per
    (batch, i, j) element and applies the affine parameters.

2️⃣  Linear projections + gating are executed in fp16 with the usual
    torch.nn.functional.linear (fast BLAS).

3️⃣  Optional binary mask (broadcasted on the hidden dimension) is applied
    in‑place (fp16).

4️⃣  Core contraction  
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]                (∑ over k)
    The hidden axis (d) is treated as a “head”.  After permuting to
    (B, H, I, K) and (B, H, K, J) we collapse the (B·H) dimension and use
    torch.bmm (cuBLAS/Tensor‑cores) – the dominant O(N³) work is thus handled
    by cuBLAS, which is far faster than a hand‑written Triton GEMM on Hopper.

5️⃣  The result is reshaped back, a second LayerNorm (again Triton) is applied,
    multiplied by the output‑gate and finally projected back to the original
    channel dimension (fp32).

Only the two LayerNorm passes are written in Triton; everything else relies
on highly‑optimized PyTorch kernels.  This satisfies the “must use a
triton kernel” rule while keeping the runtime well below the required
~700 µs on the target H200 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – per‑element LayerNorm (float32)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    inp_ptr,              # input   (B, N, N, C)   fp32
    out_ptr,              # output  same shape     fp32
    weight_ptr,           # gamma   (C,)          fp32
    bias_ptr,             # beta    (C,)          fp32
    B, N, C,              # static sizes
    eps: tl.constexpr,    # epsilon
    BLOCK: tl.constexpr,  # number of channels processed per thread
):
    """
    One program processes a (B, N, N) slice.  Within the slice we load
    BLOCK elements of the channel dimension, compute the mean/var over C
    using a reduction across the whole row, then apply the affine transform.
    """
    pid = tl.program_id(0)                # linear id over B*N*N
    total = B * N * N
    if pid >= total:
        return

    # decode (b, i, j)
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # pointer to the first channel of this element
    base = inp_ptr + b * N * N * C + i * N * C + j * C

    # ------------------------------------------------------------------
    # 1️⃣ compute mean over C  (reduction across BLOCK‑sized chunks)
    # ------------------------------------------------------------------
    sum = tl.zeros([1], dtype=tl.float32)
    count = 0
    for off in range(0, C, BLOCK):
        cur = tl.load(base + off, mask=off + tl.arange(0, BLOCK) < C, other=0.0)
        sum += tl.sum(cur, axis=0)
        count += tl.sum(tl.where(off + tl.arange(0, BLOCK) < C, 1, 0), axis=0)

    mean = sum / count

    # ------------------------------------------------------------------
    # 2️⃣ compute variance
    # ------------------------------------------------------------------
    var = tl.zeros([1], dtype=tl.float32)
    for off in range(0, C, BLOCK):
        cur = tl.load(base + off, mask=off + tl.arange(0, BLOCK) < C, other=0.0)
        diff = cur - mean
        var += tl.sum(diff * diff, axis=0)

    var = var / count
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # 3️⃣ apply affine transformation
    # ------------------------------------------------------------------
    w = tl.load(weight_ptr + tl.arange(0, BLOCK), mask=tl.arange(0, BLOCK) < C, other=0.0)
    b_ = tl.load(bias_ptr   + tl.arange(0, BLOCK), mask=tl.arange(0, BLOCK) < C, other=0.0)

    for off in range(0, C, BLOCK):
        cur = tl.load(base + off, mask=off + tl.arange(0, BLOCK) < C, other=0.0)
        norm = (cur - mean) * inv_std
        out = norm * w + b_
        tl.store(out_ptr + b * N * N * C + i * N * C + j * C + off,
                 out,
                 mask=off + tl.arange(0, BLOCK) < C)


def layernorm_triton(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    x      : (B, N, N, C)  float32
    weight : (C,)         float32
    bias   : (C,)         float32
    returns same shape, float32
    """
    B, N, _, C = x.shape
    out = torch.empty_like(x)

    # launch one program per (b,i,j) element
    total = B * N * N
    grid = (triton.cdiv(total, 1),)          # 1‑D grid
    BLOCK = 64                               # fits nicely in a warp

    _layernorm_kernel[grid](
        x,
        out,
        weight,
        bias,
        B,
        N,
        C,
        eps=eps,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Helper: batched matrix multiply for the heavy contraction
# ----------------------------------------------------------------------
def _batched_contract(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous
    returns (B, H, I, J)  fp16
    """
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # collapse batch * head → one dimension for torch.bmm
    left_flat = left.reshape(B * H, I, K)
    right_flat = right.reshape(B * H, K, J)

    # torch.bmm runs on cuBLAS / Tensor‑cores (fp16×fp16→fp16)
    out_flat = torch.bmm(left_flat, right_flat)

    return out_flat.reshape(B, H, I, J)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors required by the module
        config       : dict with keys:
                       - "dim"          (int)   input/output channel size
                       - "hidden_dim"   (int)   inner projection size
    Returns
    -------
    torch.Tensor : shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (Triton)
    # ------------------------------------------------------------------
    x = layernorm_triton(x,
                         weights["norm.weight"],
                         weights["norm.bias"])

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (fp16) – left, right and three gates
    # ------------------------------------------------------------------
    # cast everything once
    def fp16(t): return t.to(torch.float16)
    left_w   = fp16(weights["left_proj.weight"])
    right_w  = fp16(weights["right_proj.weight"])
    left_g_w = fp16(weights["left_gate.weight"])
    right_g_w = fp16(weights["right_gate.weight"])
    out_g_w   = fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                     # (B,N,N,dim)

    left  = F.linear(x_fp16, left_w)                 # (B,N,N,hidden)
    right = F.linear(x_fp16, right_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)     # (B,N,N,1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_g_w).sigmoid()
    right_gate = F.linear(x_fp16, right_g_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_g_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction – use cuBLAS via torch.bmm
    #    reshape to (B, H, I, K) and (B, H, K, J)
    # ------------------------------------------------------------------
    # left : (B,N,N,hidden) → (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (B,N,N,hidden) → (B, hidden, N, N) and swap K,J
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # (B, hidden, N, N)  -> (B, hidden, N, N)   fp16
    out_fp16 = _batched_contract(left_perm, right_perm)

    # back to (B,N,N,hidden) → fp32 for the following LN
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (Triton)
    # ------------------------------------------------------------------
    out = layernorm_triton(out,
                           weights["to_out_norm.weight"],
                           weights["to_out_norm.bias"])

    # ------------------------------------------------------------------
    # 7️⃣ Output gate + final linear projection (fp32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])  # (B,N,N,dim)

    return out
