"""
TriMul “outgoing” forward (AlphaFold‑3).

The expensive N³ contraction
    out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
is performed with a single batched GEMM call (torch.bmm) which
leverages cuBLAS/Tensor‑cores.  The only part that lives inside a
Triton kernel is the element‑wise fusion of masking, gating and the
linear‑projection results – this is the hot region that would otherwise
require three separate PyTorch kernels and many memory passes.

Algorithm
---------
1. Layer‑norm + two linear projections (left/right) – done in PyTorch
   because they are cheap compared to the O(N³) work.
2. Fuse:   left = left * mask * sigmoid(left_gate)
           right = right * mask * sigmoid(right_gate)
   with a single Triton kernel that processes all (B,N,N,hidden) elements.
3. Reshape to (B·H, N, N) and run a batched matrix‑multiply
   out = torch.bmm(left, right.transpose(1,2))   → (B·H, N, N)
4. Remaining post‑processing (norm, out‑gate, final linear) – PyTorch.

The kernel works in FP16 with FP32 accumulation for the element‑wise
operations; the final output is cast back to FP32 so that the following
LayerNorms match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# Triton kernel: fuse mask, gating and element‑wise multiplication
# --------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16*  [B, N, N, H]
    right_ptr,         # fp16*  [B, N, N, H]
    left_gate_ptr,     # fp16*  [B, N, N, H]
    right_gate_ptr,    # fp16*  [B, N, N, H]
    mask_ptr,          # i1*    [B, N, N]  (or nullptr when no mask)
    out_left_ptr,      # fp16*  (same shape as left)
    out_right_ptr,     # fp16*  (same shape as right)
    B, N, H,           # problem sizes
    stride_left_b, stride_left_n, stride_left_n2, stride_left_h,
    stride_right_b, stride_right_n, stride_right_n2, stride_right_h,
    stride_gate_b, stride_gate_n, stride_gate_n2, stride_gate_h,
    stride_mask_b, stride_mask_n, stride_mask_n2,
    stride_outl_b, stride_outl_n, stride_outl_n2, stride_outl_h,
    stride_outr_b, stride_outr_n, stride_outr_n2, stride_outr_h,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For every element (b, i, j, h) compute
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    where mask is 1 when not provided.
    """
    pid = tl.program_id(0)
    total = B * N * N * H
    idx = pid * BLOCK + tl.arange(0, BLOCK)
    mask = idx < total

    # decode linear idx -> (b,i,j,h)
    tmp = idx
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # compute flat pointers
    left = left_ptr + b * stride_left_b + i * stride_left_n + j * stride_left_n2 + h * stride_left_h
    right = right_ptr + b * stride_right_b + i * stride_right_n + j * stride_right_n2 + h * stride_right_h
    left_gate = left_gate_ptr + b * stride_gate_b + i * stride_gate_n + j * stride_gate_n2 + h * stride_gate_h
    right_gate = right_gate_ptr + b * stride_gate_b + i * stride_gate_n + j * stride_gate_n2 + h * stride_gate_h

    l = tl.load(left, mask=mask, other=0.0).to(tl.float32)
    r = tl.load(right, mask=mask, other=0.0).to(tl.float32)
    lg = tl.load(left_gate, mask=mask, other=0.0).to(tl.float32)
    rg = tl.load(right_gate, mask=mask, other=0.0).to(tl.float32)

    # sigmoid = 1/(1+exp(-x)) – kept in fp32 for stability
    lg_sig = 1.0 / (1.0 + tl.exp(-lg))
    rg_sig = 1.0 / (1.0 + tl.exp(-rg))

    if HAS_MASK:
        m = tl.load(mask_ptr + b * stride_mask_b + i * stride_mask_n + j * stride_mask_n2,
                    mask=mask, other=1).to(tl.float32)
        l = l * lg_sig * m
        r = r * rg_sig * m
    else:
        l = l * lg_sig
        r = r * rg_sig

    # store back as fp16
    outl = out_left_ptr + b * stride_outl_b + i * stride_outl_n + j * stride_outl_n2 + h * stride_outl_h
    outr = out_right_ptr + b * stride_outr_b + i * stride_outr_n + j * stride_outr_n2 + h * stride_outr_h

    tl.store(outl, l.to(tl.float16), mask=mask)
    tl.store(outr, r.to(tl.float16), mask=mask)


# --------------------------------------------------------------
# Host wrapper
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input   : [B, N, N, dim]
        torch.Tensor,                # mask    : [B, N, N]  (or None)
        Dict[str, torch.Tensor],     # weights : dict of parameters
        Dict[str, Any],              # config  : {"dim":..., "hidden_dim":...}
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # = H in the paper

    # --------------------------------------------------------------
    # 1. LayerNorm + linear projections (PyTorch, FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    left_gate = F.linear(x, weights["left_gate.weight"].to(dtype))
    right_gate = F.linear(x, weights["right_gate.weight"].to(dtype))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # --------------------------------------------------------------
    # 2. Fuse mask & gating with a Triton kernel (FP16)
    # --------------------------------------------------------------
    # Cast to fp16 for the kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    left_gate_fp16 = left_gate.to(torch.float16)
    right_gate_fp16 = right_gate.to(torch.float16)

    out_left = torch.empty_like(left_fp16)
    out_right = torch.empty_like(right_fp16)

    # strides
    s_l = left_fp16.stride()
    s_r = right_fp16.stride()
    s_lg = left_gate_fp16.stride()
    s_rg = right_gate_fp16.stride()

    if mask is None:
        has_mask = 0
        mask_ptr = torch.tensor(0, device=device).data_ptr()
        s_mask = (0, 0, 0)
    else:
        has_mask = 1
        mask = mask.to(torch.int32)          # i1 not supported, use int32 as bool
        mask_ptr = mask.data_ptr()
        s_mask = mask.stride()

    BLOCK = 128
    total_elems = B * N * N * hidden_dim
    grid = (triton.cdiv(total_elems, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_ptr,
        out_left,
        out_right,
        B, N, hidden_dim,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_lg[0], s_lg[1], s_lg[2], s_lg[3],
        s_rg[0], s_rg[1], s_rg[2], s_rg[3],
        s_mask[0] if has_mask else 0,
        s_mask[1] if has_mask else 0,
        s_mask[2] if has_mask else 0,
        out_left.stride(0), out_left.stride(1), out_left.stride(2), out_left.stride(3),
        out_right.stride(0), out_right.stride(1), out_right.stride(2), out_right.stride(3),
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 3. Batched GEMM (torch.bmm) on (B*H, N, N)
    # --------------------------------------------------------------
    left_bh = out_left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_bh = out_right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_bh.view(B * hidden_dim, N, N)          # (B*H, N, N)
    right_mat = right_bh.view(B * hidden_dim, N, N)

    # right needs to be transposed on the last two dims
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # --------------------------------------------------------------
    # 4. Reshape back + post‑processing (LayerNorm, out‑gate, final linear)
    # --------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    out = out.to(dtype)   # back to FP32 for following ops

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    out = out * out_gate   # broadcast over N,N

    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
