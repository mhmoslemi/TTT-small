"""
TriMul “outgoing” operator (AlphaFold‑3) – fast forward pass.

The expensive N³ contraction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

is performed with a single batched GEMM (torch.bmm) on FP16 tensors,
which gives Tensor‑Core speed on Hopper.  
The only work moved to Triton is the element‑wise “mask‑gate‑apply” step,
which fuses three element‑wise ops (mask, left‑gate, right‑gate) into one
kernel and writes the result back in FP16, ready for the GEMM.

All LayerNorms, linear projections, final gating and output Linear remain
in PyTorch because they are cheap compared to the O(N³) mat‑mul.
The kernel works for any batch size (1 or 2), any sequence length up to
1024 and any hidden dimension (including 128, 384, 768).  The final
output is returned in FP32 as required by the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel – apply mask (if present) and per‑position gates
# -------------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    x_ptr,                # const float32*   [B, N, N, D]
    mask_ptr,             # const float32*   [B, N, N] (or nullptr)
    left_gate_ptr,        # const float32*   [B, N, N, D]
    right_gate_ptr,       # const float32*   [B, N, N, D]
    left_out_ptr,         # float16*         [B, N, N, D]
    right_out_ptr,        # float16*         [B, N, N, D]
    B, N, D,              # problem sizes
    has_mask: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,d) computes

        left  = x * mask * left_gate
        right = x * mask * right_gate

    (mask is 1.0 when absent).  The results are stored as FP16 for the
    subsequent batched GEMM.
    """
    pid = tl.program_id(0)

    # flatten the 4‑D tensor to 1‑D for simple indexing
    total = B * N * N * D
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)

    # mask out-of‑bounds threads
    mask = offs < total

    # decode linear offset to (b,i,j,d)
    #  idx = ((b * N + i) * N + j) * D + d
    tmp = offs // D
    d   = offs % D
    tmp2 = tmp // N
    j   = tmp % N
    i   = tmp2 % N
    b   = tmp2 // N

    # compute raw pointers
    x_ptr   = x_ptr   + ((b * N + i) * N + j) * D + d
    lg_ptr  = left_gate_ptr   + ((b * N + i) * N + j) * D + d
    rg_ptr  = right_gate_ptr  + ((b * N + i) * N + j) * D + d
    lo_ptr  = left_out_ptr    + ((b * N + i) * N + j) * D + d
    ro_ptr  = right_out_ptr   + ((b * N + i) * N + j) * D + d

    # optional mask
    if has_mask:
        mask_ptr = mask_ptr + (b * N + i) * N + j
        m = tl.load(mask_ptr, mask=mask, other=1.0).to(tl.float32)
    else:
        m = 1.0

    # load values
    x   = tl.load(x_ptr,   mask=mask, other=0.0).to(tl.float32)
    lg  = tl.load(lg_ptr,  mask=mask, other=0.0).to(tl.float32)
    rg  = tl.load(rg_ptr,  mask=mask, other=0.0).to(tl.float32)

    # apply mask and gates
    left_val  = x * m * lg
    right_val = x * m * rg

    # store FP16 results
    tl.store(lo_ptr, left_val.to(tl.float16), mask=mask)
    tl.store(ro_ptr, right_val.to(tl.float16), mask=mask)


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input  : [B, N, N, dim]   (float32)
        torch.Tensor,                # mask   : [B, N, N]        (float32) or None
        Dict[str, torch.Tensor],     # weights: dict of parameters
        Dict[str, Any],              # config : {"dim":..., "hidden_dim":...}
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # = H in the paper

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (cheap – keep in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension (still in PyTorch)
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right_proj = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 3. Gates (pre‑activate with sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(dtype))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(dtype))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(dtype))
    )   # (B,N,N,H)

    # -----------------------------------------------------------------
    # 4. Fuse mask + gates with a Triton kernel and cast to FP16
    # -----------------------------------------------------------------
    # allocate FP16 buffers for the two projected tensors
    left_fp16  = torch.empty_like(left_proj, dtype=torch.float16)
    right_fp16 = torch.empty_like(right_proj, dtype=torch.float16)

    # launch parameters
    BLOCK = 1024  # 1‑K threads per program – good occupancy on Hopper

    total_elements = B * N * N * hidden_dim
    num_blocks = (total_elements + BLOCK - 1) // BLOCK
    grid = (num_blocks,)

    # pointer arguments (use .data_ptr() for speed)
    mask_ptr = mask.data_ptr() if mask is not None else 0

    mask_gate_kernel[grid](
        x.data_ptr(),
        mask_ptr,
        left_gate.data_ptr(),
        right_gate.data_ptr(),
        left_fp16.data_ptr(),
        right_fp16.data_ptr(),
        B, N, hidden_dim,
        has_mask=mask is not None,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 5. Batched GEMM: out = left @ rightᵀ  (shape (B*H, N, N))
    #    Use FP16 operands, FP32 accumulator (torch.bmm does this)
    # -----------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat  = left_fp16.view(B * hidden_dim, N, N)
    right_mat = right_fp16.view(B * hidden_dim, N, N)

    # rightᵀ: take transpose on the last two dims
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H,N,N), fp16 -> fp32 accumulator

    # reshape back to (B, N, N, H) and convert to FP32 for the remaining ops
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(dtype)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply the output gate (broadcast over N,N)
    out = out * out_gate

    # final linear back to original dim
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
