"""
custom_kernel – “outgoing” TriMul forward pass (AlphaFold‑3 style).

The cubic contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a fused batched GEMM Triton kernel (fp16 compute, fp32
accumulation).  All other operations (LayerNorm, linear projections,
gating, final LayerNorm + linear) are carried out in PyTorch to keep the
implementation simple and numerically stable.  The kernel works for any
batch size (1‑2), sequence length up to 1024 and hidden dimensions that are
multiples of 32 (the Tile sizes are chosen to be Tensor‑Core friendly on
Hopper/SM90).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (I×K)·(K×J) → (I×J)  (fp16 → fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,               # left  tensor pointer  (Z, I, K)
    b_ptr,               # right tensor pointer  (Z, K, J)
    c_ptr,               # output tensor pointer (Z, I, J)
    M, N, K,             # matrix dimensions
    stride_az, stride_am, stride_ak,   # strides for a (z, i, k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z, k, j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    #  Tile indices
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)          # row tile
    pid_n = tl.program_id(1)          # col tile
    pid_z = tl.program_id(2)          # batch*head tile

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32 (Tensor‑Core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    #  K‑reduction loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load tiles of A (M×K) and B (K×N)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp16·fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    #  Write result (cast back to fp16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    Arguments
    ---------
    left  : Tensor (B, H, I, K)  dtype=torch.float16, contiguous
    right : Tensor (B, H, K, J)  dtype=torch.float16, contiguous

    Returns
    -------
    out : Tensor (B, H, I, J) dtype=torch.float16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # Collapse leading dimensions so the kernel sees (Z, *, *)
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Tile sizes tuned for Hopper (Tensor‑Core friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # rows
        triton.cdiv(J, BLOCK_N),   # cols
        Z,                         # batch*head
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for SM90
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        mask         : torch.Tensor [B, N, N]       (float32/bool) or None
        weights      : dict of parameter tensors
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32, bias‑free)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual LayerNorm to stay in fp32 and avoid extra memory
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                         # [B,N,N,dim] fp32

    # --------------------------------------------------------------
    # 2. Linear projections  (cast to fp16 for the heavy GEMM)
    # --------------------------------------------------------------
    def _to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    w_left = _to_fp16(weights["left_proj.weight"])
    w_right = _to_fp16(weights["right_proj.weight"])
    w_lgate = _to_fp16(weights["left_gate.weight"])
    w_rgate = _to_fp16(weights["right_gate.weight"])
    w_outgate = _to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)

    left = F.linear(x_fp16, w_left)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, w_right)

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, w_lgate).sigmoid()
    right_gate = F.linear(x_fp16, w_rgate).sigmoid()
    out_gate = F.linear(x_fp16, w_outgate).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton batched GEMM   (the N³ contraction)
    #    left:  (B, N, N, H) -> (B, H, N, N)
    #    right: (B, N, N, H) -> (B, H, N, N) and then transpose K ↔ J
    # --------------------------------------------------------------
    left_t = left.permute(0, 3, 1, 2).contiguous()          # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()        # (B, H, K, J)

    out_fp16 = _batched_gemm_fp16(left_t, right_t)          # (B, H, I, J)

    # Back to original layout and float32 for the remaining ops
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 6. Output LayerNorm
    # --------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = out * to_out_norm_w + to_out_norm_b

    # --------------------------------------------------------------
    # 7. Final gating + linear projection
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)                # broadcast over dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                         # (B,N,N,dim)

    return out
