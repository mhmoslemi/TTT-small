"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)  
================================================

The O(N³) contraction is performed with a single batched matrix‑multiply
(`torch.bmm`) which runs on cuBLAS / Tensor‑cores and is therefore far faster
than a hand‑written Triton GEMM.  The only part that actually needs a Triton
kernel is the element‑wise fusion of the mask, the three sigmoid gates and the
left/right projections – this saves a separate launch for each tensor and
removes an expensive memory‑to‑memory copy.

The implementation follows the reference PyTorch module:

1️⃣ Layer‑norm on the input (FP32, manual).  
2️⃣ Linear projections (`left`, `right`, three gates) are done in FP16.  
3️⃣ A fused Triton kernel applies the optional mask and multiplies the
   projections with their respective sigmoid gates (`left_gate`,
   `right_gate`, `out_gate`).  
4️⃣ The core contraction  
   `out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]`  
   is a batched GEMM: after permuting the tensors we obtain
   `(B*H, N, N)·(B*H, N, N)ᵀ → (B*H, N, N)`.  `torch.bmm` (cuBLAS) handles this
   efficiently on the H200.  
5️⃣ Second Layer‑norm, output‑gate multiplication and final linear projection
   (all FP32) produce the result `[B, N, N, dim]`.

Only the fused mask‑gate kernel is written in Triton; the rest stays in
PyTorch for clarity and maximal performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused mask + gate application (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,               # fp16 tensors, shape (B,N,N,H) flattened
    left_gate_ptr, right_gate_ptr,     # fp16 tensors, same shape
    mask_ptr,                          # fp16 mask, shape (B,N,N,1) flattened (or None)
    total_elts: tl.constexpr,          # total number of FP16 elements per tensor
    BLOCK: tl.constexpr,
):
    """In‑place:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elts

    # load
    left      = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    # mask may be absent → treat as 1.0
    if mask_ptr:
        m = tl.load(mask_ptr + (offs // (tl.constexpr(1))//0), mask=mask, other=1.0)  # broadcasted mask already flattened
    else:
        m = 1.0

    left  = left  * left_gate  * m
    right = right * right_gate * m

    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_mask_and_gate(left, right, left_gate, right_gate, mask):
    """
    Fused in‑place mask+gate operation using Triton.
    All tensors must be contiguous FP16.
    """
    B, N, N2, H = left.shape
    assert N == N2
    total = left.numel()                     # B*N*N*H

    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    mask_ptr = mask.contiguous().view(-1).data_ptr() if mask is not None else 0

    _mask_gate_fuse_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        left_gate.data_ptr(),
        right_gate.data_ptr(),
        mask_ptr,
        total,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return left, right, left_gate, right_gate


# ----------------------------------------------------------------------
# Helper: manual LayerNorm (FP32)
# ----------------------------------------------------------------------
def _layer_norm_fp32(x, weight, bias, eps=1e-5):
    mean = x.mean(-1, keepdim=True)
    var  = x.var(-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + eps)
    return x * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype=float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors (see reference code)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype=float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = _layer_norm_fp32(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )                                   # [B,N,N,dim] FP32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # weights (FP16)
    def to_fp16(t): return t.to(torch.float16)
    w_left   = to_fp16(weights["left_proj.weight"])
    w_right  = to_fp16(weights["right_proj.weight"])
    w_lgate  = to_fp16(weights["left_gate.weight"])
    w_rgate  = to_fp16(weights["right_gate.weight"])
    w_outgate= to_fp16(weights["out_gate.weight"])

    left   = F.linear(x_fp16, w_left)          # (B,N,N,hidden_dim)
    right  = F.linear(x_fp16, w_right)

    left_gate  = F.linear(x_fp16, w_lgate).sigmoid()
    right_gate = F.linear(x_fp16, w_rgate).sigmoid()
    out_gate   = F.linear(x_fp16, w_outgate).sigmoid()   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse mask & gates with a Triton kernel
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
    else:
        mask_fp16 = None

    left, right, left_gate, right_gate = _apply_mask_and_gate(
        left, right, left_gate, right_gate, mask_fp16
    )
    # out_gate stays separate (used later)

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction – batched GEMM via cuBLAS
    #    left  : (B,N,N,hidden) -> (B,hidden,N,N)
    #    right : (B,N,N,hidden) -> (B,hidden,N,N) then transpose K,J
    # ------------------------------------------------------------------
    left  = left.permute(0, 3, 1, 2).contiguous()       # (B, H, I, K)
    right = right.permute(0, 3, 2, 1).contiguous()      # (B, H, K, J)

    # reshape batch*head for torch.bmm
    BH, I, K = left.shape[0] * left.shape[1], left.shape[2], left.shape[3]
    left_flat  = left.view(BH, I, K)                 # (B*H, N, N)
    right_flat = right.view(BH, K, I)                # (B*H, N, N) transposed

    # batched matrix multiply (fp16 × fp16 → fp16, accumulated in fp32 internally)
    out_flat = torch.bmm(left_flat, right_flat)      # (B*H, N, N)

    # reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    out = _layer_norm_fp32(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final gating + linear projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)            # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])    # (B,N,N,dim)

    return out
