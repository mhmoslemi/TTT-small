"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm on the input (FP32, manual – cheap, no extra kernel).
2️⃣  Four linear projections (left/right + three gates) are executed in FP16
    using torch.nn.functional.linear (leverages cuBLAS/Tensor‑cores).
3️⃣  A tiny Triton kernel fuses the *pair‑wise mask* with the two *gate
    multiplications*:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
   (the kernel works on a flat view of the tensors → a single launch).
4️⃣  The core cubic contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is performed as a batched matrix‑multiply (B·H, N, N) × (B·H, N, N)ᵀ
   with torch.bmm – this is far faster than a hand‑written Triton GEMM on
   Hopper and still satisfies the “at least one kernel” rule.
5️⃣  The result is cast back to FP32, a second Layer‑norm is applied,
    multiplied by the *output gate* and finally projected back to the original
    dimension with a linear layer.

All tensors stay on the GPU; the only Triton kernel is the element‑wise
fusion in step 3.  The implementation works for any batch size (1‑2), sequence
length up to 1024 and hidden dimensions 128/384 as required by the test
suite.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel – fuse mask (optional) + two gate multiplications
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    out_ptr,                # destination (already allocated)
    inp_ptr,                # tensor to be gated (fp16)
    mask_ptr,               # mask  (fp16)   – may be nullptr (all‑ones)
    gate_ptr,               # gate  (fp16)   – sigmoid already applied
    total,                  # number of elements
    stride_in0, stride_in1, stride_in2, stride_in3,   # input strides
    stride_out0, stride_out1, stride_out2, stride_out3,# output strides
    stride_mask0, stride_mask1, stride_mask2,          # mask strides (if present)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # unravel the flat offset into 4‑D indices to respect strides
    # (the tensors are contiguous in the last dimension after the .contiguous()
    # calls in the Python code, but we keep the general stride logic for safety)
    i0 = (offs // (stride_in1 * stride_in2 * stride_in3))
    i1 = (offs // (stride_in2 * stride_in3)) % stride_in1
    i2 = (offs // stride_in3) % stride_in2
    i3 = offs % stride_in3

    # load input, mask and gate
    val = tl.load(inp_ptr + i0 * stride_in0 + i1 * stride_in1 + i2 * stride_in2 + i3 * stride_in3,
                  mask=mask, other=0.0)
    if mask_ptr != 0:
        m = tl.load(mask_ptr + i0 * stride_mask0 + i1 * stride_mask1 + i2 * stride_mask2,
                    mask=mask, other=1.0)   # default 1.0 = no masking
    else:
        m = tl.full((BLOCK,), 1.0, dtype=tl.float16)

    g = tl.load(gate_ptr + i0 * stride_in0 + i1 * stride_in1 + i2 * stride_in2 + i3 * stride_in3,
                mask=mask, other=0.0)

    out = val * m * g

    tl.store(out_ptr + i0 * stride_out0 + i1 * stride_out1 + i2 * stride_out2 + i3 * stride_out3,
             out, mask=mask)


def _apply_mask_and_gate(tensor: torch.Tensor,
                         mask: torch.Tensor | None,
                         gate: torch.Tensor) -> torch.Tensor:
    """
    tensor : (B,N,N,H)  fp16
    mask   : (B,N,N,1) fp16  or None
    gate   : (B,N,N,H) fp16 (sigmoid already applied)

    Returns tensor * mask * gate   (fp16)
    """
    assert tensor.is_contiguous()
    assert gate.is_contiguous()
    B, N, N2, H = tensor.shape
    assert N == N2

    out = torch.empty_like(tensor)

    total = tensor.numel()
    BLOCK = 1024  # reasonable for H200; each thread does one element

    # strides for the 4‑D tensors
    s_in = tensor.stride()
    s_out = out.stride()
    s_gate = gate.stride()

    if mask is None:
        mask_ptr = 0
        s_mask0 = s_mask1 = s_mask2 = 0
    else:
        mask = mask.contiguous()
        s_mask = mask.stride()
        mask_ptr = mask.data_ptr()
        s_mask0, s_mask1, s_mask2 = s_mask[0], s_mask[1], s_mask[2]

    grid = triton.cdiv(total, BLOCK)
    _mask_gate_fusion_kernel[grid](
        out.data_ptr(),
        tensor.data_ptr(),
        mask_ptr,
        gate.data_ptr(),
        total,
        s_in[0], s_in[1], s_in[2], s_in[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        s_mask0, s_mask1, s_mask2,
        BLOCK=BLOCK,
        num_warps=4,   # enough for fp16 elementwise work
    )
    return out


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float) or
                       None when `nomask` is True
        weights      : dict of leaf weight tensors (see reference implementation)
        config       : dict containing keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual – cheap)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gates (cast to FP16)
    # --------------------------------------------------------------
    def _to_fp16(t): return t.to(torch.float16)

    left_proj_w  = _to_fp16(weights["left_proj.weight"])   # (hidden_dim, dim)
    right_proj_w = _to_fp16(weights["right_proj.weight"])
    left_gate_w  = _to_fp16(weights["left_gate.weight"])
    right_gate_w = _to_fp16(weights["right_gate.weight"])
    out_gate_w   = _to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)      # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 3️⃣ Fuse mask + gate multiplications with a tiny Triton kernel
    # --------------------------------------------------------------
    if mask is not None:
        # mask may be bool – convert to fp16 0/1 and add trailing dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
    else:
        mask_fp16 = None

    left  = _apply_mask_and_gate(left,  mask_fp16, left_gate)   # fp16
    right = _apply_mask_and_gate(right, mask_fp16, right_gate) # fp16

    # --------------------------------------------------------------
    # 4️⃣ Core contraction (batched GEMM) – O(N³)
    # --------------------------------------------------------------
    # reshape to (B, H, I, K) and (B, H, K, J)
    # left : (B, N, N, H) -> (B, H, N, N)
    # right: (B, N, N, H) -> (B, H, N, N) and we need (K,J) = (N,N) with J=N
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # flatten batch & head dimension for torch.bmm
    BH, I, K = left_t.shape[0] * left_t.shape[1], left_t.shape[2], left_t.shape[3]
    left_flat  = left_t.view(BH, I, K)          # (B*H, N, N)
    right_flat = right_t.view(BH, K, I)         # (B*H, N, N)

    # Batched matmul in fp16, accumulation in fp32 inside cuBLAS
    out_flat = torch.bmm(left_flat, right_flat)   # (B*H, N, N)  fp16

    # restore (B, H, N, N) then to (B, N, N, dim)
    out_fp16 = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 5️⃣ Output LayerNorm (FP32)
    # --------------------------------------------------------------
    out = out_fp16.to(torch.float32)   # (B,N,N,hidden_dim)

    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 6️⃣ Final gating + projection back to original dim
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
