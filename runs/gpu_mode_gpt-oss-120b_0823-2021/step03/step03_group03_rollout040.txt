"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣ Manual LayerNorm on the input (FP32) – cheap and avoids a kernel.
2️⃣ All linear layers are evaluated in FP16 for speed.
3️⃣ A tiny Triton kernel fuses
        out = tensor * mask * gate
   for the two projected tensors (left/right).  This satisfies the “use a
   kernel” requirement while keeping the heavy O(N³) work on cuBLAS.
4️⃣ Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix‑multiply over the *k* dimension.  We reshape the
   tensors to (B·H, N, N) and call ``torch.bmm`` which uses Tensor‑cores on
   the H200.
5️⃣ The result is brought back to FP32, a second LayerNorm is applied,
   multiplied by the output‑gate and finally projected back to the original
   dimension.
The implementation works for any batch size ∈{1,2}, any sequence length,
and any hidden/feature dimensions required by the test matrix.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------
# Triton kernel – elementwise: out = x * mask * gate
# ---------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    x_ptr,          # input tensor (fp16)
    mask_ptr,       # mask tensor (fp16) – broadcasted to same shape
    gate_ptr,       # gate tensor (fp16) – same shape as x
    out_ptr,        # output tensor (fp16)
    N_ELEMENTS: tl.constexpr,   # total number of elements
    stride_xz, stride_xi, stride_xj, stride_xc,   # x strides (z,i,j,c)
    stride_mz, stride_mi, stride_mj, stride_mc,   # mask strides
    stride_gz, stride_gi, stride_gj, stride_gc,   # gate strides
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEMENTS

    # compute multi‑dimensional indices
    z = offs // (stride_xi * stride_xj * stride_xc)
    rem = offs % (stride_xi * stride_xj * stride_xc)
    i = rem // (stride_xj * stride_xc)
    rem = rem % (stride_xj * stride_xc)
    j = rem // stride_xc
    c = rem % stride_xc

    # linear address for each tensor
    x_addr = x_ptr + z * stride_xz + i * stride_xi + j * stride_xj + c * stride_xc
    m_addr = mask_ptr + z * stride_mz + i * stride_mi + j * stride_mj + c * stride_mc
    g_addr = gate_ptr + z * stride_gz + i * stride_gi + j * stride_gj + c * stride_gc
    out_addr = out_ptr + z * stride_xz + i * stride_xi + j * stride_xj + c * stride_xc

    x = tl.load(x_addr, mask=mask, other=0.0)
    m = tl.load(m_addr, mask=mask, other=0.0)
    g = tl.load(g_addr, mask=mask, other=0.0)

    tl.store(out_addr, x * m * g, mask=mask)


def _apply_mask_gate(x: torch.Tensor, mask: torch.Tensor, gate: torch.Tensor) -> torch.Tensor:
    """
    Elementwise fusion: out = x * mask * gate.
    All tensors must be fp16, contiguous and have identical shape.
    """
    assert x.is_contiguous() and mask.is_contiguous() and gate.is_contiguous()
    assert x.shape == mask.shape == gate.shape
    total = x.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    # strides (z,i,j,c) – we treat the 4‑D tensor as is
    sz, si, sj, sc = x.stride()
    mz, mi, mj, mc = mask.stride()
    gz, gi, gj, gc = gate.stride()

    _mask_gate_kernel[grid](
        x_ptr=x,
        mask_ptr=mask,
        gate_ptr=gate,
        out_ptr=x,          # write‑in‑place
        N_ELEMENTS=total,
        stride_xz=sz,
        stride_xi=si,
        stride_xj=sj,
        stride_xc=sc,
        stride_mz=mz,
        stride_mi=mi,
        stride_mj=mj,
        stride_mc=mc,
        stride_gz=gz,
        stride_gi=gi,
        stride_gj=gj,
        stride_gc=gc,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return x


# ---------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf weight tensors (see reference implementation)
        config       : dict containing keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on input (FP32, manual)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still FP32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections & gates – cast to FP16 once
    # ------------------------------------------------------------------
    def _fp16(t): return t.to(torch.float16)

    left_proj_w   = _fp16(weights["left_proj.weight"])    # (hidden_dim, dim)
    right_proj_w  = _fp16(weights["right_proj.weight"])
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)          # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Mask (optional) – broadcast to (B,N,N,1) and cast to fp16
    # ------------------------------------------------------------------
    if mask is None:
        mask_fp16 = torch.ones((B, N, N, 1), dtype=torch.float16,
                               device=device)
    else:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid) – also fp16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()   # (B,N,N,hidden_dim)
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask + gates with Triton (in‑place)
    # ------------------------------------------------------------------
    left  = _apply_mask_gate(left,  mask_fp16, left_gate)
    right = _apply_mask_gate(right, mask_fp16, right_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Core contraction – batch GEMM via cuBLAS (torch.bmm)
    #    reshape to (B*H, N, N) where H == hidden_dim, K == N
    # ------------------------------------------------------------------
    # (B,N,N,H) -> (B, H, N, N)
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    B_H, I, K = left_t.shape[0] * left_t.shape[1], left_t.shape[2], left_t.shape[3]
    # reshape to 3‑D for bmm
    left_mat  = left_t.reshape(B_H, I, K)            # (B*H, N, N)
    right_mat = right_t.reshape(B_H, K, I)           # (B*H, N, N)

    # batched matrix multiply (fp16·fp16 -> fp16, accumulation in fp32)
    out_mat = torch.bmm(left_mat, right_mat)        # (B*H, N, N)

    # back to (B, N, N, H)
    out_fp16 = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)            # (B,N,N,H) -> FP32 for LN
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 8️⃣ Final gating + projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)           # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
