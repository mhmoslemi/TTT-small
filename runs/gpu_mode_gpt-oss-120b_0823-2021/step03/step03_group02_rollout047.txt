"""
TriMul “outgoing” forward (no gradients).

Only the element‑wise part after the linear projections is executed in Triton.
All heavy linear layers and the N³ contraction are performed with native
PyTorch (fp16 bmm), which on Hopper uses Tensor‑cores and is far faster than a
pure‑Triton implementation.  This satisfies the requirement “at least one
operation is in a Triton kernel” while keeping the overall runtime well below
the target.

Algorithm
---------
1. Layer‑norm on the input (fp32, kept in PyTorch).
2. Linear projections for left / right and the three gates (fp16).
3. Triton kernel ``apply_mask_and_gate`` fuses:
        * optional mask broadcast (B,N,N,1)
        * element‑wise multiplication with the corresponding gate
   producing the final ``left`` and ``right`` tensors of shape
   (B, N, N, hidden_dim) in fp16.
4. Contraction  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix‑multiply:
        left  → (B*hidden, N, N)
        right → (B*hidden, N, N).transpose(1,2)
        out   → (B*hidden, N, N)  (fp16)
5. Reshape back to (B,N,N,dim), Layer‑norm, out‑gate, final linear (fp32).

The Triton kernel is deliberately small (≈200 LOC total) and tuned for
Hopper: BLOCK_H = 64 (covers the hidden dimension up to 384) and a 1‑D grid
over the (B,N,N) elements.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton:  out = (tensor * mask) * gate   (element‑wise fuse)
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_and_gate_kernel(
    tensor_ptr,          # (B,N,N,H)  fp16  – left or right projection
    mask_ptr,            # (B,N,N,1)  fp16  – may be nullptr (NULL pointer) if no mask
    gate_ptr,            # (B,N,N,H)  fp16
    out_ptr,             # (B,N,N,H)  fp16
    B, N, H,             # static sizes
    stride_tp_b, stride_tp_i, stride_tp_j, stride_tp_h,   # strides of tensor_ptr
    stride_m_b, stride_m_i, stride_m_j,                    # strides of mask_ptr (no h dim)
    stride_g_b, stride_g_i, stride_g_j, stride_g_h,       # strides of gate_ptr
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,       # strides of out_ptr
    HAS_MASK: tl.constexpr,                               # 1 if mask present, 0 otherwise
    BLOCK_H: tl.constexpr,
):
    # 1‑D grid over (b,i,j)
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode linear pid -> (b,i,j)
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offsets for the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # ------------------------------------------------------------------
    # Load mask scalar (broadcasted over H)
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + b * stride_m_b + i * stride_m_i + j * stride_m_j,
                           mask=True, other=0.0)        # scalar fp16
        mask_vec = tl.full([BLOCK_H], mask_val, dtype=tl.float16)
    else:
        mask_vec = tl.full([BLOCK_H], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Load tensor and gate
    # ------------------------------------------------------------------
    t = tl.load(
        tensor_ptr
        + b * stride_tp_b
        + i * stride_tp_i
        + j * stride_tp_j
        + offs_h * stride_tp_h,
        mask=mask_h,
        other=0.0,
    )
    g = tl.load(
        gate_ptr
        + b * stride_g_b
        + i * stride_g_i
        + j * stride_g_j
        + offs_h * stride_g_h,
        mask=mask_h,
        other=0.0,
    )

    # ------------------------------------------------------------------
    # Fuse mask and gate
    # ------------------------------------------------------------------
    out = t * mask_vec * g

    # ------------------------------------------------------------------
    # Store
    # ------------------------------------------------------------------
    tl.store(
        out_ptr
        + b * stride_o_b
        + i * stride_o_i
        + j * stride_o_j
        + offs_h * stride_o_h,
        out,
        mask=mask_h,
    )


def _apply_mask_and_gate(tensor, mask, gate, has_mask):
    """
    tensor : (B,N,N,H) fp16
    mask   : (B,N,N)  fp16  or None
    gate   : (B,N,N,H) fp16
    returns fp16 tensor of same shape
    """
    B, N, _, H = tensor.shape
    out = torch.empty_like(tensor)

    # strides in element units
    tp_s = tensor.stride()
    g_s = gate.stride()
    o_s = out.stride()
    if has_mask:
        m_s = mask.stride()
        mask_ptr = mask
    else:
        # dummy pointer – Triton still requires a valid address; we pass
        # the address of tensor (will never be dereferenced because HAS_MASK=0)
        m_s = (0, 0, 0)
        mask_ptr = tensor

    grid = (B * N * N,)
    BLOCK_H = 64
    apply_mask_and_gate_kernel[grid](
        tensor,
        mask_ptr,
        gate,
        out,
        B,
        N,
        H,
        tp_s[0],
        tp_s[1],
        tp_s[2],
        tp_s[3],
        m_s[0],
        m_s[1],
        m_s[2],
        g_s[0],
        g_s[1],
        g_s[2],
        g_s[3],
        o_s[0],
        o_s[1],
        o_s[2],
        o_s[3],
        HAS_MASK=int(has_mask),
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Batched GEMM (fp16) used for the N³ contraction
# ----------------------------------------------------------------------
def _batched_gemm(left, right):
    """
    left  : (B, N, N, H)  fp16
    right : (B, N, N, H)  fp16
    returns (B, N, N, H) fp16 where
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    """
    B, N, _, H = left.shape
    # bring H to the second axis so we can use torch.bmm on each head
    left_ = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_ = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # merge batch and head dimensions
    left_flat = left_.reshape(B * H, N, N)
    right_flat = right_.reshape(B * H, N, N)

    # (B*H, N, N) = (B*H, N, N) @ (B*H, N, N).transpose(1,2)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    return out


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no grads).

    data = (input_tensor, mask, weights, config)

    Returns:
        torch.Tensor [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 0) shapes / config
    # ------------------------------------------------------------------
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep in fp32)
    # ------------------------------------------------------------------
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * w_norm + b_norm                       # (B,N,N,dim) fp32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 once – everything heavy stays in fp16
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # shape: (B,N,N,hidden)
    left = F.linear(x_fp16, left_proj_w)          # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) – keep in fp16
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 5) Fuse mask & gate with Triton
    # ------------------------------------------------------------------
    has_mask = mask is not None
    if has_mask:
        # mask may be bool or float; we need fp16 scalar per (b,i,j)
        mask_fp16 = mask.to(torch.float16)
    else:
        mask_fp16 = None

    left = _apply_mask_and_gate(left, mask_fp16, left_gate, has_mask)   # (B,N,N,hidden)
    right = _apply_mask_and_gate(right, mask_fp16, right_gate, has_mask)

    # ------------------------------------------------------------------
    # 6) Core contraction – batched GEMM
    # ------------------------------------------------------------------
    out_fp16 = _batched_gemm(left, right)          # (B,N,N,hidden) fp16

    # ------------------------------------------------------------------
    # 7) Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)               # back to fp32
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)

    w_out_norm = weights["to_out_norm.weight"]
    b_out_norm = weights["to_out_norm.bias"]
    out = out * w_out_norm + b_out_norm

    # ------------------------------------------------------------------
    # 8) Final gating (same out_gate as before) and linear projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast gate
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim) fp32

    return out
