"""
TriMul “outgoing” forward pass (no gradients).

The dominant N³ contraction is expressed as a batched matrix‑multiply
   out[b,i,j] = Σₖ left[b,i,k] * right[b,j,k]
which is equivalent to a batched GEMM (B·H, N, N) × (B·H, N, N)ᵀ.
We let cuBLAS (via torch.bmm) handle this work because it is already
highly‑optimised on Hopper.  The only part we accelerate with Triton is the
element‑wise combination of the mask and the three sigmoid‑gates, which
is a 4‑D broadcasted operation that would otherwise be a memory‑bound bottleneck.

All heavy linear layers are performed in fp16, the GEMM runs in fp16 → fp32
(accumulation), and the final LayerNorm / projection steps are done back in
fp32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: apply mask (if present) and three sigmoid‑gates
# --------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    x_ptr,                     # normalized input      (B,N,N,D)   fp16
    mask_ptr,                  # mask (optional)       (B,N,N,1)   fp16
    left_proj_w_ptr,           # (D, H)                fp16
    right_proj_w_ptr,          # (D, H)                fp16
    left_gate_w_ptr,           # (D, H)                fp16
    right_gate_w_ptr,          # (D, H)                fp16
    out_gate_w_ptr,            # (D, H)                fp16
    left_ptr,                  # out left   (B,N,N,H)  fp16
    right_ptr,                 # out right  (B,N,N,H)  fp16
    left_gate_ptr,             # out left gate   (B,N,N,H) fp16
    right_gate_ptr,            # out right gate  (B,N,N,H) fp16
    out_gate_ptr,              # out out gate    (B,N,N,H) fp16
    B, N, D, H,               # static shape parameters
    HAS_MASK: tl.constexpr,    # 1 if mask is provided, else 0
    BLOCK: tl.constexpr,       # tile size along the flattened (B,N,N) dimension
):
    pid = tl.program_id(0)
    # flatten the first three axes so we can launch a 1‑D grid
    total = B * N * N
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # map flat index back to (b,i,j)
    b = offs // (N * N)
    ij = offs % (N * N)
    i = ij // N
    j = ij % N

    # strides (in elements) for the input tensor
    stride_xb, stride_xi, stride_xj, stride_xd = tl.load(tl.make_block_ptr(
        x_ptr, shape=(B, N, N, D), strides=(N * N * D, N * D, D, 1), block_shape=(1, 1, 1, 1)
    ).offsets)

    # load a slice x[b,i,j,:] (size D)
    x = tl.load(
        x_ptr + b * stride_xb + i * stride_xi + j * stride_xj,
        mask=mask[:, None],
        other=0.0,
        eviction_policy='evict_last',
    )  # (BLOCK, D)

    # optional mask
    if HAS_MASK:
        mask_val = tl.load(
            mask_ptr + b * stride_xb + i * stride_xi + j * stride_xj,
            mask=mask[:, None],
            other=0.0,
        )
        x = x * mask_val

    # linear projections:  (B,N,N,D) @ (D,H) -> (B,N,N,H)
    # we perform them with a fused matmul‑like dot product per element
    left  = tl.dot(x, tl.load(left_proj_w_ptr, mask=tl.arange(0, D)[:, None] < D, other=0.0))
    right = tl.dot(x, tl.load(right_proj_w_ptr, mask=tl.arange(0, D)[:, None] < D, other=0.0))

    # gates (sigmoid of a linear projection)
    left_gate  = tl.dot(x, tl.load(left_gate_w_ptr,  mask=tl.arange(0, D)[:, None] < D, other=0.0))
    right_gate = tl.dot(x, tl.load(right_gate_w_ptr, mask=tl.arange(0, D)[:, None] < D, other=0.0))
    out_gate   = tl.dot(x, tl.load(out_gate_w_ptr,   mask=tl.arange(0, D)[:, None] < D, other=0.0))

    left_gate  = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate))
    out_gate   = 1.0 / (1.0 + tl.exp(-out_gate))

    # apply gates
    left  = left  * left_gate
    right = right * right_gate

    # store results (flattened index → proper strides)
    stride_lbz, stride_lbi, stride_lbj, stride_lbh = tl.load(tl.make_block_ptr(
        left_ptr, shape=(B, N, N, H), strides=(N * N * H, N * H, H, 1), block_shape=(1, 1, 1, 1)
    ).offsets)

    tl.store(
        left_ptr + b * stride_lbz + i * stride_lbi + j * stride_lbj,
        left,
        mask=mask[:, None],
    )
    tl.store(
        right_ptr + b * stride_lbz + i * stride_lbi + j * stride_lbj,
        right,
        mask=mask[:, None],
    )
    tl.store(
        left_gate_ptr + b * stride_lbz + i * stride_lbi + j * stride_lbj,
        left_gate,
        mask=mask[:, None],
    )
    tl.store(
        right_gate_ptr + b * stride_lbz + i * stride_lbi + j * stride_lbj,
        right_gate,
        mask=mask[:, None],
    )
    tl.store(
        out_gate_ptr + b * stride_lbz + i * stride_lbi + j * stride_lbj,
        out_gate,
        mask=mask[:, None],
    )


# --------------------------------------------------------------
# Helper: launch the element‑wise kernel
# --------------------------------------------------------------
def apply_mask_and_gates(
    x_fp16: torch.Tensor,               # (B,N,N,D) fp16
    mask: torch.Tensor | None,          # (B,N,N)  fp16 or None
    weights: dict,
    dim: int,
    hidden_dim: int,
):
    """
    Returns:
        left, right, out_gate  each (B,N,N,hidden_dim) fp16
    """
    B, N, _, D = x_fp16.shape
    device = x_fp16.device

    # allocate outputs
    left = torch.empty((B, N, N, hidden_dim), device=device, dtype=torch.float16)
    right = torch.empty_like(left)
    # we only need out_gate later (same shape as left/right)
    out_gate = torch.empty_like(left)

    # dummy buffers for gate tensors (not used after kernel, but required for launch)
    left_gate = torch.empty_like(left)
    right_gate = torch.empty_like(left)

    # flatten grid
    total = B * N * N
    BLOCK = 128
    grid = (triton.cdiv(total, BLOCK),)

    HAS_MASK = 0 if mask is None else 1

    if mask is None:
        # allocate a dummy mask tensor (won’t be read because HAS_MASK==0)
        mask_ptr = torch.empty((1,), device=device, dtype=torch.float16).data_ptr()
    else:
        mask_fp16 = mask.to(torch.float16)
        mask_ptr = mask_fp16.data_ptr()

    gate_mask_kernel[grid](
        x_ptr=x_fp16.data_ptr(),
        mask_ptr=mask_ptr,
        left_proj_w_ptr=weights["left_proj.weight"].to(torch.float16).contiguous().data_ptr(),
        right_proj_w_ptr=weights["right_proj.weight"].to(torch.float16).contiguous().data_ptr(),
        left_gate_w_ptr=weights["left_gate.weight"].to(torch.float16).contiguous().data_ptr(),
        right_gate_w_ptr=weights["right_gate.weight"].to(torch.float16).contiguous().data_ptr(),
        out_gate_w_ptr=weights["out_gate.weight"].to(torch.float16).contiguous().data_ptr(),
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        left_gate_ptr=left_gate.data_ptr(),
        right_gate_ptr=right_gate.data_ptr(),
        out_gate_ptr=out_gate.data_ptr(),
        B=B,
        N=N,
        D=D,
        H=hidden_dim,
        HAS_MASK=HAS_MASK,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return left, right, out_gate


# --------------------------------------------------------------
# Batched GEMM using torch.bmm (cuBLAS) – the N³ part
# --------------------------------------------------------------
def batch_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left , right : (B, N, N, hidden) fp16
    Returns out    : (B, N, N, hidden) fp16  where
          out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_r = left.permute(0, 3, 1, 2).reshape(B * H, N, N)          # (B*H, N, N)
    right_r = right.permute(0, 3, 2, 1).reshape(B * H, N, N)        # (B*H, N, N)  (K dimension is the middle one)

    # batched matmul, fp16×fp16→fp32 accumulation performed by cuBLAS
    out_r = torch.bmm(left_r, right_r)                              # (B*H, N, N)  fp16 result (actually fp16)

    # reshape back
    out = out_r.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    return out


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Arguments
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]  (float32, CUDA)
        mask         : torch.Tensor [B, N, N]      (float32/bool) or None
        weights      : dict of torch.Tensor parameters (see skeleton)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]   (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1) Input LayerNorm (keep in PyTorch – cheap)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                     # still fp32

    # --------------------------------------------------------------
    # 2) Cast to fp16 for the heavy part
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Linear projections + gates + optional mask – done in one Triton kernel
    # --------------------------------------------------------------
    left, right, out_gate = apply_mask_and_gates(
        x_fp16, mask, weights, dim, hidden_dim
    )   # all are fp16, shape (B,N,N,hidden)

    # --------------------------------------------------------------
    # 4) Core contraction (N³) – cuBLAS batched GEMM
    # --------------------------------------------------------------
    out_fp16 = batch_gemm(left, right)          # (B,N,N,hidden) fp16

    # --------------------------------------------------------------
    # 5) Bring back to (B,N,N,dim) – the hidden dim → dim path
    # --------------------------------------------------------------
    out = out_fp16.to(torch.float32)            # fp32 for the remaining ops

    # --------------------------------------------------------------
    # 6) Output LayerNorm
    # --------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # --------------------------------------------------------------
    # 7) Final gating (reuse out_gate computed earlier) + projection
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim) fp32

    return out
