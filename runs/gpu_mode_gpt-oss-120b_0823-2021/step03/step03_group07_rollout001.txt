```python
"""
TriMul outgoing forward pass (AlphaFold‑3 style).

The dominant O(N³) operation is a batched matrix multiplication
    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
where `left`/`right` have shape (B, H, N, N).  This kernel evaluates the
contraction using a classic blocked GEMM tiling (BLOCK_M × BLOCK_N × BLOCK_K)
with fp16 inputs and fp32 accumulation.  All surrounding linear /
LayerNorm / gating work is performed with regular PyTorch, which is
memory‑bound and cheap compared with the cubic term.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ---------------------------------------------------------
# Triton kernel: batched GEMM for the triangle contraction
# ---------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # const fp16*  [B, H, N, N]
    right_ptr,         # const fp16*  [B, H, N, N]  (will be accessed transposed)
    out_ptr,           #       fp16*  [B, H, N, N]
    B, N, H,           # int metadata
    # strides (row‑major) for the three tensors
    stride_lb, stride_lh, stride_ln, stride_lk,   # left  (B,H,N,N)
    stride_rb, stride_rh, stride_rn, stride_rk,   # right (B,H,N,N)
    stride_ob, stride_oh, stride_on, stride_ok,   # out   (B,H,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a (BLOCK_M × BLOCK_N) tile of

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    Each program instance is responsible for one (b, h, tile_m, tile_n) tile.
    """
    pid = tl.program_id(0)

    # -------------------------------------------------
    # Decode program id  -> (b, h, tile_m, tile_n)
    # -------------------------------------------------
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    tile_n = pid % num_n
    pid //= num_n
    tile_m = pid % num_m
    pid //= num_m
    h = pid % H
    b = pid // H

    # -------------------------------------------------
    # Tile offsets + masks
    # -------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Main K loop (blocked)
    # -------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, h, i, k]   shape (M, K)
        l_ptr = (left_ptr +
                 b * stride_lb +
                 h * stride_lh +
                 offs_m[:, None] * stride_ln +
                 offs_k[None, :] * stride_lk)
        l = tl.load(
            l_ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right[b, h, j, k]  we need it transposed -> (K, N)
        #   original layout: (b, h, j, k)
        #   after transpose we treat k as row, j as column
        r_ptr = (right_ptr +
                 b * stride_rb +
                 h * stride_rh +
                 offs_k[:, None] * stride_rk +   # k‑dim becomes row
                 offs_n[None, :] * stride_rn)    # j‑dim becomes col
        r_t = tl.load(
            r_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(l, r_t)

    # -------------------------------------------------
    # Write back the tile
    # -------------------------------------------------
    out_ptrs = (out_ptr +
                b * stride_ob +
                h * stride_oh +
                offs_m[:, None] * stride_on +
                offs_n[None, :] * stride_ok)

    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ---------------------------------------------------------
# Host‑side wrapper that respects the required function sign
# ---------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input    [B, N, N, dim]
        torch.Tensor,                     # mask     [B, N, N]   (or None)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any]                    # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (dtype = float32, same device as input).
    """
    x, mask, weights, config = data
    device = x.device
    # keep a high‑precision dtype for the final output
    f32 = torch.float32
    f16 = torch.float16

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]          # H

    # -----------------------------------------------------------------
    # 1️⃣  Cheap element‑wise work (LayerNorm, linear proj, gating)
    # -----------------------------------------------------------------
    # LayerNorm over the last dimension
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(f32),
        bias=weights["norm.bias"].to(f32),
    )

    # Linear projections (bias=False)
    left = F.linear(x, weights["left_proj.weight"].to(f32))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(f32))

    # Optional mask (broadcast on the hidden dimension)
    if mask is not None:
        mask_ = mask.unsqueeze(-1)        # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # Gating (sigmoid after linear)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(f32)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(f32)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(f32)))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 2️⃣  Rearrange to [B, H, N, N] and cast to fp16 for the kernel
    # --------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left.to(f16)
    right_fp16 = right.to(f16)

    out_fp16 = torch.empty_like(left_fp16)   # (B,H,N,N), fp16

    # --------------------------------------------------------------
    # 3️⃣  Launch Triton kernel (blocked GEMM)
    # --------------------------------------------------------------
    # Tile size – chosen to fit comfortably into shared memory on Hopper.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Number of program instances = B * H * ceil(N/M) * ceil(N/N)
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_progs = B * hidden_dim * grid_m * grid_n
    grid = (total_progs,)

    _trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        # strides for left
        left_fp16.stride(0),   # B
        left_fp16.stride(1),   # H
        left_fp16.stride(2),   # N (i dimension)
        left_fp16.stride(3),   # N (k dimension)
        # strides for right
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        # strides for out
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 4️⃣  Post‑processing back to the original layout
    # --------------------------------------------------------------
    # (B,H,N,N) -> (B,N,N,H) and cast to fp32 for the remaining ops
    out = out_fp16.to(f32).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # Final LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(f32),
        bias=weights["to_out_norm.bias"].to(f32),
    )

    # Apply output gate (broadcast on pairwise axes)
    out = out * out_gate

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"].to(f32))

    return out
```
