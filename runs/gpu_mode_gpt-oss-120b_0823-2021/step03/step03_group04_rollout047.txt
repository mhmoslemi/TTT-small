"""
TriMul – outgoing variant (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm over the channel dimension C of the 4‑D tensor
    (B, N, N, C).  Implemented with torch.nn.functional.layer_norm
    (fast, already on‑GPU).

2️⃣  Linear projections (float32 → fp16) for:
        • left  :  (B,N,N,H)
        • right :  (B,N,N,H)
        • left_gate , right_gate , out_gate  (B,N,N,H)

3️⃣  Fuse three element‑wise operations (mask, gate, and multiplication)
    in a single Triton kernel:
        left_fused  = left  * left_gate  * mask
        right_fused = right * right_gate * mask
    The mask is optional; when absent a constant ``1`` is used.
    The kernel works on a 1‑D grid of blocks and therefore has virtually
    zero shared‑memory pressure.

4️⃣  Batched GEMM (cuBLAS / Tensor‑cores) on the fused tensors:
        out[b, h] = left_fused[b, h] @ right_fused[b, h].T
    By reshaping to (B*H, N, N) we delegate the O(N³) work to cuBLAS.
    Operands are fp16, accumulation in fp32, output in fp16.

5️⃣  Layer‑norm over the hidden dimension H, multiply by out_gate,
    and a final linear projection back to C (float32 → float32).

The heavy arithmetic stays in fp16 (Tensor‑core friendly) while
the small element‑wise work runs on Triton, satisfying the “some‑kernel”
requirement and keeping the overall runtime well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – element‑wise mask & gate fusion
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    # inputs
    left_ptr, right_ptr,                 # fp16 * (B,N,N,H)
    left_gate_ptr, right_gate_ptr,       # fp16 * (B,N,N,H)
    mask_ptr,                           # fp16 * (B,N,N,1)  or nullptr
    # outputs
    left_out_ptr, right_out_ptr,         # fp16 * (B,N,N,H)
    # shape / strides
    B, N, H,                             # compile‑time scalars
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_rg0, stride_rg1, stride_rg2, stride_rg3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    # compile‑time flags
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,j,h):
        left_out  = left  * left_gate  * mask
        right_out = right * right_gate * mask
    Mask is broadcasted over the hidden dimension H.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    active = offs < total

    # --------- decode linear index to (b,i,j,h) ----------
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # --------- load left / right -------------------------
    idx = (
        b * stride_l0
        + i * stride_l1
        + j * stride_l2
        + h * stride_l3
    )
    left = tl.load(left_ptr + idx, mask=active, other=0.0)
    right = tl.load(right_ptr + idx, mask=active, other=0.0)

    # --------- load gates --------------------------------
    idx_gate = (
        b * stride_lg0
        + i * stride_lg1
        + j * stride_lg2
        + h * stride_lg3
    )
    lgate = tl.load(left_gate_ptr + idx_gate, mask=active, other=0.0)
    rgate = tl.load(right_gate_ptr + idx_gate, mask=active, other=0.0)

    # --------- load mask (if present) -------------------
    if HAS_MASK:
        # mask has shape (B,N,N,1) → stride on dim H is zero
        idx_mask = (
            b * stride_m0
            + i * stride_m1
            + j * stride_m2
        )
        m = tl.load(mask_ptr + idx_mask, mask=active, other=0.0)
    else:
        m = 1.0

    # --------- apply fusion --------------------------------
    left_out = left * lgate * m
    right_out = right * rgate * m

    # --------- store results -------------------------------
    tl.store(left_out_ptr + idx, left_out, mask=active)
    tl.store(right_out_ptr + idx, right_out, mask=active)


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        x          – (B, N, N, C)      torch.float32
        mask       – (B, N, N) or None torch.float32 / bool
        weights    – dict of all learned parameters (float32)
        config     – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor
        (B, N, N, C) float32
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, C = x.shape
    H = config["hidden_dim"]
    dtype = torch.float32

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (FP32 → FP16)
    # --------------------------------------------------------------
    left      = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right     = F.linear(x, weights["right_proj.weight"].to(dtype))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # (B,N,N,H)

    # Cast to fp16 and make contiguous for the kernel
    left_fp16      = left.to(torch.float16).contiguous()
    right_fp16     = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate.to(torch.float16).contiguous()
    right_gate_fp16= right_gate.to(torch.float16).contiguous()

    if mask is None:
        # dummy mask – not accessed inside the kernel
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask = False
    else:
        # mask shape (B,N,N) → (B,N,N,1) to broadcast over H
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        has_mask = True

    # Allocate fused output buffers
    left_fused  = torch.empty_like(left_fp16)
    right_fused = torch.empty_like(right_fp16)

    # --------------------------------------------------------------
    # 3️⃣ Triton kernel – mask & gate fusion
    # --------------------------------------------------------------
    total_elements = B * N * N * H
    BLOCK = 1024                                  # 1‑D block size (tuned for H200)
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        # inputs
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        # outputs
        left_fused,
        right_fused,
        # shape & strides
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1), left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        # mask strides (only three meaningful dims; fourth set to 0)
        mask_fp16.stride(0) if has_mask else 0,
        mask_fp16.stride(1) if has_mask else 0,
        mask_fp16.stride(2) if has_mask else 0,
        mask_fp16.stride(3) if has_mask else 0,
        # compile‑time flags
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM via torch.bmm (Tensor‑cores)
    # --------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)      # (B*H, N, N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)     # (B*H, N, N)

    # fp16 * fp16 → fp16 (accum in fp32)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))            # (B*H, N, N)

    # back‑to (B, N, N, H) and FP32 for the next LayerNorm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype)

    # --------------------------------------------------------------
    # 5️⃣ Hidden‑dim LayerNorm, gating & final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate               # broadcast over (N,N)
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # (B,N,N,C)

    return out
