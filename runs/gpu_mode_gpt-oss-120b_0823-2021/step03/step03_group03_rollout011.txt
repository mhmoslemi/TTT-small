"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (FP32, manual – cheap and avoids an extra kernel).
2️⃣  Four linear projections (left/right) and three gating linear layers are
    performed in FP16 for speed.
3️⃣  Optional pair‑wise mask is applied.
4️⃣  Sigmoid gating is applied.
5️⃣  **Core contraction** – the Einstein‑summation
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is a batched matrix‑multiply over the *sequence* dimension **k**.
    We launch a tiny Triton kernel that computes a tiled GEMM
        (B*H, N, N) · (B*H, N, N)ᵀ → (B*H, N, N)
    in FP16 (accumulate in FP32).  This kernel is the only Triton component;
    the rest of the pipeline lives in PyTorch.
6️⃣  Convert the result back to FP32, apply a second LayerNorm, multiply by the
    output‑gate and finish with a linear projection to the original dimension.

The Triton kernel is tuned for Hopper/SM‑90:
    BLOCK_M = BLOCK_N = 64   # tile size in the i/j dimensions
    BLOCK_K = 32             # tile size in the reduction (k) dimension
    4 warps, fp16 → fp32 accumulation
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched GEMM (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,            # left  tensor  (B*H, I, K)
    b_ptr,            # right tensor  (B*H, K, J)
    c_ptr,            # output tensor (B*H, I, J)
    M, N, K,          # extents
    stride_az, stride_am, stride_ak,   # a strides (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel for (B*H, I, K) @ (B*H, K, J) → (B*H, I, J)."""
    pid_z = tl.program_id(2)          # index over batch*head
    pid_m = tl.program_id(0)          # tile along I
    pid_n = tl.program_id(1)          # tile along J

    # tile start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (I × K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile (K × J)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp16 dot → fp32 accumulation
        acc += tl.dot(a, b)

    # write result (convert back to fp16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Performs (left @ right) where left.shape = (B, H, I, K) and
    right.shape = (B, H, K, J).  All tensors must be FP16 and contiguous.
    Returns a FP16 tensor of shape (B, H, I, J).
    """
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2, "inner dimensions must match"

    # output allocation
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # view as 3‑D tensors for the kernel
    a = left.view(B * H, I, K)
    b = right.view(B * H, K, J)
    c = out.view(B * H, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),          # tiles along I
        triton.cdiv(J, BLOCK_N),          # tiles along J
        B * H,                            # one program per (batch, head)
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                     # good default for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool or float)
        weights      : dict of leaf weight tensors (see reference implementation)
        config       : dict containing keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype torch.float32
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual – avoids an extra kernel)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    # mean & var over the last dimension (dim)
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                       # still FP32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections & gates (cast to FP16)
    # ------------------------------------------------------------------
    def _fp16(t): return t.to(torch.float16)

    left_proj_w   = _fp16(weights["left_proj.weight"])   # (hidden_dim, dim)
    right_proj_w  = _fp16(weights["right_proj.weight"])
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                     # (B,N,N,dim)

    # linear layers (bias = False in the reference)
    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Masking (optional)
    # ------------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction via Triton
    #    left  : (B, N, N, H) → (B, H, N, K)   with K == N
    #    right : (B, N, N, H) → (B, H, K, N)   (transpose K↔J)
    # ------------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)  where I=K=N
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)  J=N

    # batched GEMM in fp16 via Triton
    out_fp16 = _triton_batched_gemm(left_t, right_t)  # (B, H, I, J)

    # reshape back to (B, N, N, dim) and convert to FP32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)            # broadcast over dim
    out = F.linear(out, weights["to_out.weight"])    # (B,N,N,dim)

    return out
