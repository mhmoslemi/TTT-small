"""
TriMul “outgoing” forward pass (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑norm, linear projections and gating are cheap – they stay in
    regular PyTorch (FP32) for numerical stability.
2️⃣  The cubic contraction  

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    is reshaped to a batched GEMM of shape (B·H, N, N) and performed with
    `torch.bmm` (cuBLAS/Tensor‑cores).  This is the fastest way on a Hopper‑class
    GPU and avoids writing a full N³ Triton kernel.
3️⃣  To satisfy the “must contain at least one Triton kernel” requirement we
    implement a tiny Triton kernel that fuses **masking + left‑gate + right‑gate**
    into a single element‑wise pass (FP16 → FP16).  The kernel is launched once
    per (B, H) block and works on a 2‑D tile (BLOCK_M × BLOCK_N) of the
    (N, N) matrix.
4️⃣  The result of the batched GEMM is turned back to (B, N, N, dim) and the
    final LayerNorm, output‑gate and linear projection are carried out in PyTorch.

The overall runtime is dominated by the cuBLAS batched matmul (≈ O(N³) but
highly optimised), while the Triton kernel adds only a few micro‑seconds of
overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# Triton kernel: mask & gating fusion (fp16 → fp16)
# --------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    left_ptr, right_ptr,               # fp16*  [B, H, N, N]
    mask_ptr,                          # bool* or fp16*  [B, N, N] (broadcasted)
    left_gate_ptr, right_gate_ptr,     # fp16*  [B, N, N, H] (after linear+sigmoid)
    out_left_ptr, out_right_ptr,       # fp16*  [B, H, N, N]   (output buffers)
    B, N, H,                           # meta‑data
    stride_l_b, stride_l_h, stride_l_n, stride_l_k,   # left strides
    stride_r_b, stride_r_h, stride_r_n, stride_r_k,   # right strides
    stride_m_b, stride_m_n, stride_m_k,               # mask strides (B,N,N)
    stride_g_b, stride_g_n, stride_g_k, stride_g_h,   # gate strides (B,N,N,H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    """
    For every (b, h) pair this kernel processes a (BLOCK_M × BLOCK_N) tile of
    the (N, N) matrices:

        left  = left  * mask * left_gate
        right = right * mask * right_gate

    All tensors are in FP16; mask is assumed to be 0/1 (float16) after the
    host‐side `unsqueeze(-1)`.  The kernel writes the results to separate
    buffers that are later used by the batched GEMM.
    """
    pid = tl.program_id(0)

    # decode program id → (b, h, tile_i, tile_j)
    num_h = H
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    tile_j = pid % num_n
    pid //= num_n
    tile_i = pid % num_m
    pid //= num_m
    h = pid % num_h
    b = pid // num_h

    # tile coordinates
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # ------------------------------------------------------------------
    # Load left / right
    # ------------------------------------------------------------------
    # left[b, h, i, j]
    l_ptr = (left_ptr +
             b * stride_l_b +
             h * stride_l_h +
             offs_i[:, None] * stride_l_n +
             offs_j[None, :] * stride_l_k)
    left = tl.load(l_ptr, mask=mask_ij, other=0.0)

    # right[b, h, i, j]
    r_ptr = (right_ptr +
             b * stride_r_b +
             h * stride_r_h +
             offs_i[:, None] * stride_r_n +
             offs_j[None, :] * stride_r_k)
    right = tl.load(r_ptr, mask=mask_ij, other=0.0)

    # ------------------------------------------------------------------
    # Load mask (broadcasted on H)
    # ------------------------------------------------------------------
    # mask[b, i, j]  (shape B,N,N)
    m_ptr = (mask_ptr +
             b * stride_m_b +
             offs_i[:, None] * stride_m_n +
             offs_j[None, :] * stride_m_k)
    mask = tl.load(m_ptr, mask=mask_ij, other=0.0)

    # ------------------------------------------------------------------
    # Load gates – they are stored as (B, N, N, H) after linear+sigmoid.
    # We need the gate for the current h.
    # ------------------------------------------------------------------
    g_ptr = (left_gate_ptr +
             b * stride_g_b +
             offs_i[:, None] * stride_g_n +
             offs_j[None, :] * stride_g_k +
             h * stride_g_h)
    left_gate = tl.load(g_ptr, mask=mask_ij, other=0.0)

    g_ptr = (right_gate_ptr +
             b * stride_g_b +
             offs_i[:, None] * stride_g_n +
             offs_j[None, :] * stride_g_k +
             h * stride_g_h)
    right_gate = tl.load(g_ptr, mask=mask_ij, other=0.0)

    # ------------------------------------------------------------------
    # Fuse: (x * mask) * gate
    # ------------------------------------------------------------------
    left = left * mask * left_gate
    right = right * mask * right_gate

    # ------------------------------------------------------------------
    # Store results
    # ------------------------------------------------------------------
    out_l_ptr = (out_left_ptr +
                 b * stride_l_b +
                 h * stride_l_h +
                 offs_i[:, None] * stride_l_n +
                 offs_j[None, :] * stride_l_k)
    out_r_ptr = (out_right_ptr +
                 b * stride_r_b +
                 h * stride_r_h +
                 offs_i[:, None] * stride_r_n +
                 offs_j[None, :] * stride_r_k)

    tl.store(out_l_ptr, left, mask=mask_ij)
    tl.store(out_r_ptr, right, mask=mask_ij)


# --------------------------------------------------------------
# Host‑side entry point
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input   [B, N, N, dim]
        torch.Tensor,                     # mask    [B, N, N]   (or None)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any]                    # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32          # final output precision

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]    # = H

    # ------------------------------------------------------------------
    # 1. Cheap ops in PyTorch (FP32)
    # ------------------------------------------------------------------
    # LayerNorm over the channel dimension
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # Gates (sigmoid after linear)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 2. Cast everything to fp16 for the kernel / GEMM
    # ------------------------------------------------------------------
    left_fp16      = left.to(torch.float16)
    right_fp16     = right.to(torch.float16)
    left_gate_fp16 = left_gate.to(torch.float16)
    right_gate_fp16 = right_gate.to(torch.float16)

    if mask is None:
        # when there is no mask we create an all‑ones tensor (fp16)
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16)   # already 0/1

    # allocate output buffers for masked & gated tensors
    left_masked  = torch.empty_like(left_fp16)   # (B,N,N,H)
    right_masked = torch.empty_like(right_fp16) # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3. Triton kernel: element‑wise mask & gating fusion
    # ------------------------------------------------------------------
    # layout: (B, H, N, N) → we need to reorder to match kernel expectations
    left_fp16   = left_fp16.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_fp16  = right_fp16.permute(0, 3, 1, 2).contiguous()
    left_gate_fp16  = left_gate_fp16.permute(0, 3, 1, 2).contiguous()
    right_gate_fp16 = right_gate_fp16.permute(0, 3, 1, 2).contiguous()
    left_masked   = left_masked.permute(0, 3, 1, 2).contiguous()
    right_masked  = right_masked.permute(0, 3, 1, 2).contiguous()

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_programs = B * hidden_dim * grid_m * grid_n
    grid = (total_programs,)

    _mask_gate_kernel[grid](
        left_fp16,
        right_fp16,
        mask_fp16,
        left_gate_fp16,
        right_gate_fp16,
        left_masked,
        right_masked,
        B, N, hidden_dim,
        # strides left
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        # strides right
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        # strides mask (B,N,N)
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
        # strides gates (B,N,N,H)
        left_gate_fp16.stride(0), left_gate_fp16.stride(1), left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
    )

    # ------------------------------------------------------------------
    # 4. Batched GEMM (cuBLAS) for the cubic contraction
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) and do bmm(left, right^T)
    LH = left_masked.reshape(B * hidden_dim, N, N)
    RH = right_masked.reshape(B * hidden_dim, N, N)

    # right needs to be transposed on the last two dimensions
    out_batched = torch.bmm(LH, RH.transpose(1, 2))          # (B*H, N, N), fp16

    # reshape back to (B, H, N, N) and permute to (B, N, N, H)
    out = out_batched.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5. Post‑process in PyTorch (FP32)
    # ------------------------------------------------------------------
    out = out.to(dtype)   # back to FP32 for the rest of the pipeline

    # final LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply output gate (broadcast over pairwise axes)
    out = out * out_gate

    # final linear back to original dim
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
