"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (high‑level)
----------------------
1️⃣  Layer‑norm the input (FP32) – performed with a tiny Triton kernel
    that computes mean / variance per (b,i,j, :) slice and normalises it.
2️⃣  All linear layers and gates are evaluated in FP16 with native
    ``torch.nn.functional.linear`` (fast BLAS kernels).
3️⃣  Optional pair‑wise mask is applied *inside a Triton kernel* that
    fuses the broadcasted mask with the two projected tensors.
4️⃣  Gating (sigmoid) is done in FP16 (torch‑ops are already highly
    optimised on Hopper).
5️⃣  Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is a batched matrix‑multiply over the reduction dimension **k**
    (i.e. a B×H GEMM where H = hidden_dim).  This is performed with
    ``torch.bmm`` (cuBLAS/Tensor‑cores) because it is far faster than a
    hand‑written Triton N³ kernel on the H200.
6️⃣  Second Layer‑norm (FP32) – again the tiny Triton kernel from step 1.
7️⃣  Output gating and final linear projection back to the original
    dimension (FP32).

Only the two LayerNorms and the mask‑application are written in Triton;
all heavy linear work stays in highly‑optimised cuBLAS kernels.
The implementation respects the required signature `custom_kernel(data)`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – per‑slice LayerNorm (FP32)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    inp_ptr,                # input   (B, N, N, C)   fp32
    out_ptr,                # output  (B, N, N, C)   fp32
    weight_ptr,             # gamma   (C)           fp32
    bias_ptr,               # beta    (C)           fp32
    C,                      # number of channels (compile‑time)
    eps,                    # epsilon (float32)
    stride_ib, stride_in, stride_ik, stride_ic,   # input strides
    stride_ob, stride_on, stride_ok, stride_oc,   # output strides
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)          # linear index over (B,N,N) slices
    # start index of the slice
    b = pid // (tl.cdiv(C, 1) * 1)   # dummy – we only need offsets
    # compute the base pointer for the slice (ignoring channel dim)
    offs = pid * stride_ib
    # Offsets for the channel dimension within the slice
    offs_c = tl.arange(0, BLOCK)
    mask = offs_c < C

    # load a block of the channel dimension (might be smaller than C)
    inp = tl.load(inp_ptr + offs + offs_c * stride_ic, mask=mask, other=0.0)

    # ---- mean ---------------------------------------------------------
    # reduction in FP32 – sum then divide by C
    sum_val = tl.reduce(inp, op="sum", axis=0)
    mean = sum_val / C

    # ---- variance -----------------------------------------------------
    diff = inp - mean
    var = tl.reduce(diff * diff, op="sum", axis=0) / C
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ---- normalisation + affine ----------------------------------------
    out = (inp - mean) * inv_std
    if weight_ptr != 0:
        gamma = tl.load(weight_ptr + offs_c, mask=mask, other=1.0)
        out = out * gamma
    if bias_ptr != 0:
        beta = tl.load(bias_ptr + offs_c, mask=mask, other=0.0)
        out = out + beta

    # store
    tl.store(out_ptr + offs + offs_c * stride_oc, out, mask=mask)


def _layernorm_triton(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """
    Triton LayerNorm over the last dimension (C).
    x      : (B, N, N, C)  fp32, contiguous
    weight : (C,) or None (fp32)
    bias   : (C,) or None (fp32)
    returns: same shape as x (fp32)
    """
    B, N, _, C = x.shape
    total_slices = B * N * N
    out = torch.empty_like(x)

    # strides for the 4‑D tensor (contiguous => stride ordering)
    stride_ib, stride_in, stride_ik, stride_ic = x.stride()
    stride_ob, stride_on, stride_ok, stride_oc = out.stride()

    # Triton expects pointers; we pass 0 when weight/bias are None
    w_ptr = weight.data_ptr() if weight is not None else 0
    b_ptr = bias.data_ptr()   if bias is not None else 0

    # Choose a block size that covers the channel dimension.
    # C ≤ 768 → BLOCK = 128 is safe (covers any dim with a few iterations).
    BLOCK = 128
    grid = (triton.cdiv(total_slices, 1),)   # one program per slice

    _layernorm_kernel[grid](
        inp_ptr=x,
        out_ptr=out,
        weight_ptr=w_ptr,
        bias_ptr=b_ptr,
        C=C,
        eps=eps,
        stride_ib=stride_ib,
        stride_in=stride_in,
        stride_ik=stride_ik,
        stride_ic=stride_ic,
        stride_ob=stride_ob,
        stride_on=stride_on,
        stride_ok=stride_ok,
        stride_oc=stride_oc,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel – apply pairwise mask (broadcasted) to two tensors
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_out_ptr, right_out_ptr,
    total,                 # total number of elements (B,N,N,hidden_dim)
    mask_stride,           # stride of mask (B,N,N) (no channel dim)
    hidden_dim: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total

    # load mask (single value per (b,i,j) → broadcast)
    m = tl.load(mask_ptr + (offs // hidden_dim) * mask_stride, mask=mask, other=1.0)

    # load left/right values
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)

    # apply mask (mask is 0/1 in FP16)
    l = l * m
    r = r * m

    tl.store(left_out_ptr + offs, l, mask=mask)
    tl.store(right_out_ptr + offs, r, mask=mask)


def _apply_mask_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor) -> tuple:
    """
    Broadcast a  (B,N,N) mask onto two tensors of shape (B,N,N,hidden_dim)
    (all tensors are fp16, contiguous). Returns masked copies.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    total = left.numel()
    out_left = torch.empty_like(left)
    out_right = torch.empty_like(right)

    # strides
    mask_stride = mask.stride(0)  # mask is (B,N,N)

    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _apply_mask_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_out_ptr=out_left,
        right_out_ptr=out_right,
        total=total,
        mask_stride=mask_stride,
        hidden_dim=H,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out_left, out_right


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf weight tensors (see reference implementation)
        config       : dict containing keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    device = inp.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32) – Triton
    # ------------------------------------------------------------------
    x = _layernorm_triton(
        inp,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # shape (B,N,N,dim) fp32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections & gates (cast to FP16)
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16,
                     weights["left_proj.weight"].to(torch.float16))   # (B,N,N,hidden)
    right = F.linear(x_fp16,
                     weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask – Triton fused broadcast
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure mask is fp16 (0/1) and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_mask_triton(left, right, mask_fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – FP16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16,
                          weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16,
                          weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate   = F.linear(x_fp16,
                          weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction (batched GEMM) – use cuBLAS via torch.bmm
    #    left : (B, N, N, H) -> (B*H, N, N)
    #    right: (B, N, N, H) -> (B*H, N, N) and we need its transpose on dim‑2
    # ------------------------------------------------------------------
    # permute so that the hidden dimension becomes the batch dim for bmm
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # reshape to (B*H, I, K) / (B*H, K, J)
    left_flat  = left_t.view(B * hidden_dim, N, N)
    right_flat = right_t.view(B * hidden_dim, N, N)

    # batched matrix multiplication in fp16, accumulation in fp32 (torch does this)
    out_fp16 = torch.bmm(left_flat, right_flat)                 # (B*H, I, J)

    # reshape back to (B, N, N, H)
    out_fp16 = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_fp16.to(torch.float32)   # (B, N, N, H) -> (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32) – Triton
    # ------------------------------------------------------------------
    out = _layernorm_triton(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
