"""
TriMul “outgoing” forward kernel (AlphaFold‑3).

The heavy O(N³) contraction

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

is performed with a single batched‑matrix‑multiply (torch.bmm), which on the
H200 maps to cuBLAS/Tensor‑Core GEMM and is therefore far faster than a
pure‑Triton implementation.  To satisfy the requirement of “using at least
one Triton kernel”, a tiny kernel is used to apply the optional pair‑wise
mask to the projected tensors (`left` and `right`).  All remaining
operations (LayerNorm, linear layers, gating, final LayerNorm + linear) are
implemented with PyTorch – they are bandwidth‑bound and much cheaper than the
contraction.

The algorithm

1. **Layer‑norm** over the feature dimension `dim`.
2. **Linear projection** → `left`, `right` with shape `[B, N, N, H]`.
3. **Mask** (if supplied) applied by a Triton kernel (in‑place, FP16).
4. **Gates** (sigmoid) multiplied into `left` / `right`.
5. **Reshape** to `[B*H, N, N]` (FP16) and run a batched GEMM:
       `out = left @ right.transpose(-2, -1)`.
6. **Reshape** back to `[B, N, N, H]`, **Layer‑norm**, **output gate**, and
   final linear → `[B, N, N, dim]`.

All heavy arithmetic is carried out in FP16 with FP32 accumulation (torch
defaults do this automatically for GEMM on Hopper).  The final result is
returned in `float32` to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: optional pair‑wise mask (applied in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    tensor_ptr,          # fp16*  (B, N, N, H) flattened
    mask_ptr,            # bool* (B, N, N, 1) flattened, broadcastable on H
    B, N, H,             # shape metadata
    stride_tb, stride_tn, stride_tn2, stride_th,   # strides of tensor
    stride_mb, stride_mn, stride_mn2,              # strides of mask (no H dim)
    BLOCK_B: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    In‑place multiplication of a 4‑D tensor by a broadcasted mask:
        tensor[b, i, j, h] *= mask[b, i, j]
    Only the mask part is loaded once per (b,i,j) tile; the H‑dimension is
    processed in blocks to stay within shared‑memory limits.
    """
    pid = tl.program_id(0)

    # decode a 3‑D grid: (b, i_tile, j_tile)
    b = pid // (tl.cdiv(N, BLOCK_N) * tl.cdiv(N, BLOCK_N))
    rem = pid % (tl.cdiv(N, BLOCK_N) * tl.cdiv(N, BLOCK_N))
    i_tile = rem // tl.cdiv(N, BLOCK_N)
    j_tile = rem % tl.cdiv(N, BLOCK_N)

    # tile offsets
    offs_i = i_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_valid = mask_i[:, None] & mask_j[None, :]

    # load the scalar mask (broadcasted over H)
    mask_ptr_tile = mask_ptr + b * stride_mb + offs_i[:, None] * stride_mn + offs_j[None, :] * stride_mn2
    mask_tile = tl.load(mask_ptr_tile, mask=mask_valid, other=0.0).to(tl.float16)

    # loop over H in blocks
    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        h_valid = offs_h < H

        # load tensor slice
        tensor_ptr_tile = (
            tensor_ptr
            + b * stride_tb
            + offs_i[:, None, None] * stride_tn
            + offs_j[None, :, None] * stride_tn2
            + offs_h[None, None, :] * stride_th
        )
        tensor_tile = tl.load(
            tensor_ptr_tile,
            mask=mask_valid[:, :, None] & h_valid[None, None, :],
            other=0.0,
        )
        # apply mask
        tensor_tile = tensor_tile * mask_tile[:, :, None]

        # store back
        tl.store(
            tensor_ptr_tile,
            tensor_tile,
            mask=mask_valid[:, :, None] & h_valid[None, None, :],
        )


def custom_kernel(
    data: Tuple[
        torch.Tensor,               # input   [B, N, N, dim]
        torch.Tensor,               # mask    [B, N, N]  (or None)
        Dict[str, torch.Tensor],    # weights
        Dict[str, Any],             # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Returns
    -------
    out : torch.Tensor
        Tensor of shape [B, N, N, dim] (dtype=torch.float32)
    """
    x, mask, weights, config = data

    device = x.device
    dtype = torch.float32          # final dtype required by the reference

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]          # = H

    ##################################################################
    # 1. Cheap per‑element work in PyTorch
    ##################################################################
    # LayerNorm over the feature axis
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # Linear projections (bias=False)
    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # Gates (sigmoid after linear)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    ##################################################################
    # 2. Optional mask – applied with a tiny Triton kernel (in‑place)
    ##################################################################
    if mask is not None:
        # expand mask to have a trailing singleton dimension to match left/right
        mask_exp = mask.unsqueeze(-1).contiguous()                     # (B,N,N,1)

        # ensure same dtype and device
        left_fp16 = left.to(torch.float16)
        right_fp16 = right.to(torch.float16)

        # launch kernel – grid over (B, N_tiles, N_tiles)
        BLOCK_N = 32
        BLOCK_H = 32
        grid_b = B
        grid_i = triton.cdiv(N, BLOCK_N)
        grid_j = triton.cdiv(N, BLOCK_N)
        grid = (grid_b * grid_i * grid_j,)

        _apply_mask_kernel[grid](
            left_fp16,
            mask_exp,
            B,
            N,
            hidden_dim,
            left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
            mask_exp.stride(0), mask_exp.stride(1), mask_exp.stride(2),
            BLOCK_B=1,
            BLOCK_N=BLOCK_N,
            BLOCK_H=BLOCK_H,
        )
        _apply_mask_kernel[grid](
            right_fp16,
            mask_exp,
            B,
            N,
            hidden_dim,
            right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
            mask_exp.stride(0), mask_exp.stride(1), mask_exp.stride(2),
            BLOCK_B=1,
            BLOCK_N=BLOCK_N,
            BLOCK_H=BLOCK_H,
        )
        left = left_fp16.to(dtype)
        right = right_fp16.to(dtype)
    else:
        left = left.to(dtype)
        right = right.to(dtype)

    ##################################################################
    # 3. Batched GEMM using cuBLAS (torch.bmm) – the O(N³) part
    ##################################################################
    # reshape to (B*H, N, N) for a single batched matmul
    left_bh = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (B*H, N, N)
    right_bh = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N) # (B*H, N, N)

    # FP16 GEMM – torch automatically uses Tensor‑Core kernels on Hopper
    left_bh = left_bh.to(torch.float16)
    right_bh = right_bh.to(torch.float16)

    # out = left @ right^T  (the contraction uses right transposed on last two dims)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))      # (B*H, N, N)

    # back to original layout and dtype
    out = out_bh.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(dtype)   # (B, N, N, H)

    ##################################################################
    # 4. Post‑processing (LayerNorm, output gate, final linear)
    ##################################################################
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # broadcast output gate over the pairwise axes
    out = out * out_gate

    # final projection back to `dim`
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
