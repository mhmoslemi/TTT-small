"""
TriMul “outgoing” forward pass (AlphaFold‑3) – Triton‑accelerated N³ contraction.

The heavy work of the module is the tensor contraction

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

where `left` and `right` are the gated linear projections of shape
[batch, hidden_dim, N, N].  The surrounding element‑wise work (layer‑norms,
gate‑sigmoids, final linear) is cheap and stays in PyTorch.

A tiny Triton kernel evaluates the above contraction as a batched‑GEMM
with classic 2‑D tiling (BLOCK_M × BLOCK_N) and a K‑loop tiled with
BLOCK_K.  All tensors are cast to FP16 for the kernel (FP32 accumulation)
and the result is written back as FP16, then converted to FP32 for the
remaining PyTorch post‑processing.
"""

import torch
import torch.nn.functional as F
from typing import Tuple, Dict, Any

import triton
import triton.language as tl


# ------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul contraction
# ------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # const fp16*  [B, H, N, N] (i‑k)
    right_ptr,         # const fp16*  [B, H, N, N] (j‑k) – we read it transposed
    out_ptr,           #       fp16*  [B, H, N, N] (i‑j)
    B, N, H,           # int metadata
    # strides (row‑major) for the three tensors
    stride_lb, stride_lh, stride_li, stride_lk,   # left  (B,H,N,N)
    stride_rb, stride_rh, stride_rj, stride_rk,   # right (B,H,N,N)
    stride_ob, stride_oh, stride_oi, stride_oj,   # out   (B,H,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute a (BLOCK_M × BLOCK_N) tile of

           out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    The kernel is launched with one program id per (b, h, tile_i, tile_j)
    combination.
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # decode the 1‑D pid → (b, h, tile_i, tile_j)
    # ------------------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    tile_j = pid % num_j
    pid //= num_j
    tile_i = pid % num_i
    pid //= num_i
    h = pid % H
    b = pid // H

    # ------------------------------------------------------------------
    # tile offsets and masks
    # ------------------------------------------------------------------
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (more numerically stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop (tiling)
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, h, i, k]  → (M, K)
        l_ptr = (left_ptr +
                 b * stride_lb +
                 h * stride_lh +
                 offs_i[:, None] * stride_li +
                 offs_k[None, :] * stride_lk)
        l = tl.load(l_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[b, h, j, k] is read transposed so we get (K, N)
        #   right[b, h, j, k] (original layout) → we want the K‑dimension as rows
        r_ptr = (right_ptr +
                 b * stride_rb +
                 h * stride_rh +
                 offs_k[:, None] * stride_rk +       # K as row
                 offs_j[None, :] * stride_rj)        # J as column
        r_t = tl.load(r_ptr,
                      mask=mask_k[:, None] & mask_j[None, :],
                      other=0.0).to(tl.float32)

        # matrix multiply (M,K) @ (K,N) → (M,N)
        acc += tl.dot(l, r_t)

    # ------------------------------------------------------------------
    # write result back
    # ------------------------------------------------------------------
    out_ptrs = (out_ptr +
                b * stride_ob +
                h * stride_oh +
                offs_i[:, None] * stride_oi +
                offs_j[None, :] * stride_oj)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ------------------------------------------------------------
# Host‑side entry point
# ------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input   [B, N, N, dim]
        torch.Tensor,                     # mask    [B, N, N] (or None)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any]                    # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]          # = H

    # ------------------------------------------------------------
    # 1. Cheap element‑wise work in PyTorch
    # ------------------------------------------------------------
    # layer‑norm over the channel dimension
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(x.dtype),
        bias=weights["norm.bias"].to(x.dtype),
    )

    # linear projections (bias‑free)
    left  = F.linear(x, weights["left_proj.weight"].to(x.dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(x.dtype))

    # optional masking (mask may be None)
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # (B,N,N,1)
        left  = left * mask_
        right = right * mask_

    # gates (sigmoid after linear)
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(x.dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(x.dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(x.dtype)))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------
    # 2. Rearrange to [B, H, N, N] and cast to FP16 for the kernel
    # ------------------------------------------------------------
    # (B,N,N,H) → (B,H,N,N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output buffer (same shape as left/right)
    out_fp16 = torch.empty_like(left_fp16)    # (B,H,N,N), fp16

    # ------------------------------------------------------------
    # 3. Triton kernel launch (blocked GEMM)
    # ------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # total number of program IDs = B * H * ceil(N/BLOCK_M) * ceil(N/BLOCK_N)
    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    total_progs = B * hidden_dim * grid_i * grid_j
    grid = (total_progs,)

    _trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        # strides for left  (B,H,N,N)
        left_fp16.stride(0),   # B
        left_fp16.stride(1),   # H
        left_fp16.stride(2),   # i (row)
        left_fp16.stride(3),   # k (col)
        # strides for right (B,H,N,N)
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),   # j (row) – we will read it transposed
        right_fp16.stride(3),   # k (col)
        # strides for out (B,H,N,N)
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------
    # 4. Post‑processing back in PyTorch
    # ------------------------------------------------------------
    # reshape back to (B,N,N,H) and cast to original dtype (float32)
    out = out_fp16.to(x.dtype).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # final layer‑norm over hidden_dim
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(x.dtype),
        bias=weights["to_out_norm.bias"].to(x.dtype),
    )

    # apply output gate (broadcast on pairwise axes)
    out = out * out_gate

    # final linear back to original dim
    out = F.linear(out, weights["to_out.weight"].to(x.dtype))

    return out
```

The implementation follows the reference PyTorch version exactly:

* **LayerNorm / Linear / Sigmoid gates** are performed with native PyTorch
  (`torch.nn.functional`), which is memory‑bound and cheap.
* The cubic contraction is offloaded to the Triton kernel `_trimul_outgoing_kernel`,
  which treats each `(b, h)` pair as a separate GEMM of shape `(N, N) @ (N, N)ᵀ`,
  tiled with `BLOCK_M × BLOCK_N × BLOCK_K`.  The kernel uses FP16 data and
  FP32 accumulation for speed and numerical stability.
* The result is reshaped back, normalized, gated and projected to the original
  dimension, yielding the final output tensor of shape `[B, N, N, dim]`
  (float32).

The kernel meets all constraints (supports optional masking, works for the
