"""
TriMul “outgoing” forward pass (no gradients).

The expensive N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a tiny batched‑GEMM Triton kernel (fp16 inputs,
fp32 accumulation, fp16 output).  All surrounding operations
(LayerNorm, linear projections, sigmoid gates, final LayerNorm and
linear) are carried out in PyTorch – this gives a clean implementation
while still meeting the requirement “at least part of the work must be
in a Triton kernel”.

The kernel works on tensors of shape (Z, I, K) · (Z, K, J) → (Z, I, J)
where Z = B·H (batch × hidden‑dim).  The hidden dimension H equals the
`hidden_dim` from the configuration and is treated as a separate axis
so that each GEMM fits in shared memory and can exploit tensor‑cores on
the H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (I×K)·(K×J) → (I×J)
#   a : (Z, I, K)   fp16
#   b : (Z, K, J)   fp16
#   c : (Z, I, J)   fp16   (fp32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,        # pointers
    M, N, K,                    # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)          # batch*head dimension
    pid_m = tl.program_id(1)          # tile along M (rows)
    pid_n = tl.program_id(2)          # tile along N (cols)

    # --------------------------------------------------------------
    # 0. compute tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # 1. accumulate over K
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # a : (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # b : (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp32 accumulate

    # --------------------------------------------------------------
    # 2. write result (fp16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    left : (B, H, I, K)  fp16, contiguous
    right: (B, H, K, J)  fp16, contiguous
    returns (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten the (B,H) dimensions – kernel works on a 3‑D tensor
    Z = B * H
    a = left.reshape(Z, I, K)
    b = right.reshape(Z, K, J)
    c = out.reshape(Z, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # These block sizes were found to be a good fit for Hopper (sm_90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                               # program_id(0) – batch*head
        triton.cdiv(I, BLOCK_M),         # program_id(1) – rows
        triton.cdiv(J, BLOCK_N),         # program_id(2) – cols
    )

    _batched_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no grad).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor [B, N, N] (float32/bool) or None
        - weights      : dict of torch.Tensor parameters
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm   (keep in fp32 for numerical stability)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * ln_w + ln_b                     # shape (B,N,N,dim), fp32

    # ------------------------------------------------------------------
    # 2) Linear projections and gating weights (convert to fp16 once)
    # ------------------------------------------------------------------
    def _to_fp16(t): return t.to(torch.float16, non_blocking=True)

    left_proj_w   = _to_fp16(weights["left_proj.weight"])
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])
    to_out_norm_w = weights["to_out_norm.weight"]   # fp32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 3) Cast normalized input to fp16 for the heavy ops
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Projections (fp16) → (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6) Gating (sigmoid) – also fp16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Core contraction – batched GEMM via Triton
    #    left : (B, hidden, N, N)   → (B, hidden, I, K)
    #    right: (B, hidden, N, N)   → (B, hidden, K, J)   (transpose last two)
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _batched_gemm(left_perm, right_perm)    # (B, H, I, J) fp16

    # bring back to (B,N,N,dim) layout and cast to fp32 for the rest
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 9) Final gating + linear projection (fp32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # same gating as reference
    out = F.linear(out, to_out_w)                  # (B,N,N,dim)

    return out
