"""
TriMul “outgoing” forward pass (no gradients).

The dominant N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a batched GEMM with a small Triton kernel.
All cheap ops (LayerNorm, Linear, gating, final LayerNorm + projection)
stay in PyTorch and use fp16 where possible.
The kernel works on tensors of shape (Z, I, K)·(Z, K, J) → (Z, I, J)
with Z = B * hidden_dim, I = J = seq_len.
It is written for Hopper (sm_90) but works on any recent GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    a_ptr,            # left  tensor   (Z, I, K)   fp16
    b_ptr,            # right tensor   (Z, K, J)   fp16
    c_ptr,            # output tensor  (Z, I, J)   fp16
    M, N, K,          # matrix dimensions
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a batch of GEMMs:
        C[z, i, j] = Σ_k A[z, i, k] * B[z, k, j]

    Block‑wise tiling is used so that each program processes a
    (BLOCK_M × BLOCK_N) tile of the output matrix.  The reduction over K
    is performed in fp32 for numerical stability, then cast back to fp16.
    """
    pid_z = tl.program_id(2)          # batch * hidden dimension
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # K‑loop
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile: (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Store result (cast back to fp16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batch_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    Arguments
    ---------
    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous

    Returns
    -------
    out : (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # output allocation
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten the leading batch*head dimension for the kernel
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides in elements (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # simple tiling that works well on Hopper (can be tuned later)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # program_id(0) – rows
        triton.cdiv(J, BLOCK_N),   # program_id(1) – cols
        Z,                         # program_id(2) – batch*head
    )

    batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32, CUDA)
        mask         : torch.Tensor [B, N, N]       (bool/float or None)
        weights      : dict of torch.Tensor parameters (fp32)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]  (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1) Input LayerNorm  (keep in fp32 for stability)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                      # [B,N,N,dim]  fp32

    # --------------------------------------------------------------
    # 2) Cast to fp16 for the heavy path
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Linear weights (fp16 for speed)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 3) Projections
    # --------------------------------------------------------------
    # shape after projection: [B,N,N,hidden_dim]
    left  = F.linear(x_fp16, left_proj_w)          # fp16
    right = F.linear(x_fp16, right_proj_w)         # fp16

    # --------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 5) Gating (sigmoid, still fp16)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6) Core N³ contraction via batched GEMM
    #    left : (B, N, N, H)  → (B, H, N, N)
    #    right: (B, N, N, H)  → (B, H, N, N) and transpose K/J axes
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # batched GEMM, result (B, H, I, J) in fp16
    out_fp16 = triton_batch_gemm(left_perm, right_perm)

    # back to [B, N, N, hidden] and cast to fp32 for the rest
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 7) Output LayerNorm (fp32)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # --------------------------------------------------------------
    # 8) Final gating + linear projection (keep fp32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # reuse the same gate
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                  # (B,N,N,dim)

    return out
```

**Explanation of design choices**

* **Kernel responsibility** – The only N³‑scale work (`Σ_k left * right`) is delegated to a small Triton kernel that performs a batched GEMM. This satisfies the “must‑use‑Triton” requirement while keeping the kernel simple and efficient.

* **Mixed‑precision strategy** – All element‑wise work (LayerNorm, Linear projections, gating, final LayerNorm & projection) stays in FP32 except for the heavy linear layers and the GEMM, which run in FP16 with fp32 accumulation inside the kernel. This gives the speed of Tensor‑cores while preserving final FP32 accuracy.

* **Memory layout** – By permuting to `(B, H, I, K)` and `(B, H, K, J)` we make the reduction dimension `K` contiguous for each batch, enabling the kernel to load tiles with stride‑1 accesses.

* **Tiling parameters** – `BLOCK_M = BLOCK_N = 64` and `BLOCK_K = 32` are a good default on Hopper; they fit nicely into shared memory and give high occupancy. They are defined on the host and passed as `tl.constexpr` parameters.

* **No fallback** – The implementation always launches the Triton kernel; if it fails the error will be raised, making debugging straightforward.

* **Correctness** – The sequence of operations mirrors the reference PyTorch implementation, including the identical gating (`out_gate`) applied after the output LayerNorm, and the final linear projection. The function returns a float‑32 tensor of shape `[B, N, N, dim]` for all supported configurations.
