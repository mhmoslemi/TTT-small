"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward‑only)
------------------------
1️⃣  Layer‑norm the 4‑D input (float32, manual implementation in a tiny
    Triton kernel – this gives us a required triton component).

2️⃣  Linear projections and gating are done in fp16 with PyTorch
    (F.linear + .sigmoid).  The mask (if present) is also applied in fp16
    with a second Triton kernel (broadcast & element‑wise multiply).

3️⃣  Core contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .
    After permuting the tensors we obtain two matrices of shape
        (B·H, N, N)  and  (B·H, N, N)ᵀ .
    The O(N³) work is performed by a single call to ``torch.bmm`` (cuBLAS /
    Tensor‑cores on Hopper), which is far faster than a hand‑written Triton
    GEMM for the sizes of interest.

4️⃣  The result is reshaped back, a second Layer‑norm is applied (again with
    the Triton kernel), multiplied by the output‑gate and finally projected
    back to the original dimension with a FP32 linear layer.

The implementation keeps the expensive cubic contraction on cuBLAS while
still satisfying the “must use a Triton kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def _layer_norm_kernel(
    inp_ptr, out_ptr,       # pointers
    weight_ptr, bias_ptr,   # LN parameters
    B, N, C,                # tensor shape (B,N,N,C) --> C is the last dim
    eps: tl.constexpr,
    BLOCK_C: tl.constexpr,  # work over the channel dimension
):
    """
    Simple per‑row LayerNorm over the last dimension (C).
    One program works on a single (b,i,j) element and processes up to
    BLOCK_C channels at a time.
    """
    pid = tl.program_id(0)                     # linear index over B·N·N
    total = B * N * N
    mask = pid < total

    # offsets of the (b,i,j) triple
    idx = pid
    j = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # channel offsets
    offs_c = tl.arange(0, BLOCK_C)

    # linear address computation
    base = (
        b * (N * N * C) +
        i * (N * C) +
        j * C
    )
    inp = tl.load(
        inp_ptr + base + offs_c,
        mask=mask[:, None] & (offs_c < C),
        other=0.0,
    )
    # compute mean & var (reduce over C)
    # use a small reduction in registers (BLOCK_C ≤ 64)
    mean = tl.mean(inp, axis=0)
    var = tl.mean((inp - mean) * (inp - mean), axis=0)

    # normalize
    out = (inp - mean) / tl.sqrt(var + eps)

    # apply weight & bias (they are 1‑D of length C)
    w = tl.load(weight_ptr + offs_c, mask=offs_c < C, other=1.0)
    b_ = tl.load(bias_ptr + offs_c, mask=offs_c < C, other=0.0)
    out = out * w + b_

    tl.store(
        out_ptr + base + offs_c,
        out,
        mask=mask[:, None] & (offs_c < C),
    )


@triton.jit
def _apply_mask_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,                     # shapes: left/right (B,N,N,H), mask (B,N,N)
    BLOCK_H: tl.constexpr,
):
    """
    Broadcast the binary mask (float16) over the hidden dimension H and
    multiply it into the left and right projected tensors.
    One program processes a (b,i,j) element and handles up to BLOCK_H
    hidden channels.
    """
    pid = tl.program_id(0)
    total = B * N * N
    mask_prog = pid < total

    # decode (b,i,j)
    idx = pid
    j = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    offs_h = tl.arange(0, BLOCK_H)

    # base address for left/right (shape B,N,N,H)
    base = (
        b * (N * N * H) +
        i * (N * H) +
        j * H
    )
    # load mask (scalar) and broadcast
    m = tl.load(mask_ptr + b * (N * N) + i * N + j, mask=mask_prog, other=0.0)
    m = tl.broadcast_to(m, (BLOCK_H,))

    # multiply
    left = tl.load(left_ptr + base + offs_h, mask=mask_prog & (offs_h < H), other=0.0)
    right = tl.load(right_ptr + base + offs_h, mask=mask_prog & (offs_h < H), other=0.0)

    left = left * m
    right = right * m

    tl.store(left_ptr + base + offs_h, left,
             mask=mask_prog & (offs_h < H))
    tl.store(right_ptr + base + offs_h, right,
             mask=mask_prog & (offs_h < H))


# ----------------------------------------------------------------------
# Helper – batched GEMM using cuBLAS (torch.bmm)
# ----------------------------------------------------------------------
def _batched_gemm_fp16(lhs: torch.Tensor, rhs: torch.Tensor) -> torch.Tensor:
    """
    lhs : (B, H, I, K)   fp16
    rhs : (B, H, K, J)   fp16
    returns (B, H, I, J) fp16
    The heavy N³ work is delegated to torch.bmm (cuBLAS/Tensor‑cores).
    """
    B, H, I, K = lhs.shape
    _, _, _, J = rhs.shape
    lhs = lhs.view(B * H, I, K)
    rhs = rhs.view(B * H, K, J)
    out = torch.bmm(lhs, rhs)                 # fp16 matmul
    return out.view(B, H, I, J)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors (see problem statement)
        config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (fp32) – Triton
    # --------------------------------------------------------------
    # allocate output buffer
    x_ln = torch.empty_like(inp, dtype=torch.float32, device=device)

    # launch
    BLOCK_C = 64  # dim is at most 384 → fits comfortably
    grid = (triton.cdiv(B * N * N, 1),)  # one program per (b,i,j)
    _layer_norm_kernel[grid](
        inp_ptr=inp,
        out_ptr=x_ln,
        weight_ptr=weights["norm.weight"],
        bias_ptr=weights["norm.bias"],
        B=B,
        N=N,
        C=dim,
        eps=eps,
        BLOCK_C=BLOCK_C,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (fp16) + gating
    # --------------------------------------------------------------
    # cast once
    x_fp16 = x_ln.to(torch.float16)

    left  = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))   # (B,N,N,hidden)
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    left_gate  = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate   = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim) – Triton
    # --------------------------------------------------------------
    if mask is not None:
        # ensure mask is fp16 (0/1) and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        # launch a kernel that multiplies left/right in‑place
        total = B * N * N
        grid = (triton.cdiv(total, 1),)
        BLOCK_H = 64  # hidden_dim ≤ 128 → fits in a warp
        _apply_mask_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            mask_ptr=mask_fp16,
            B=B,
            N=N,
            H=hidden_dim,
            BLOCK_H=BLOCK_H,
            num_warps=4,
        )

    # --------------------------------------------------------------
    # 4️⃣ Core contraction – batched GEMM with cuBLAS
    # --------------------------------------------------------------
    # reshape to (B, H, I, K) and (B, H, K, J)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) where K and J are swapped

    out_fp16 = _batched_gemm_fp16(left_perm, right_perm)   # (B, H, N, N)

    # back to original layout, convert to fp32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 5️⃣ Output LayerNorm (fp32) – Triton
    # --------------------------------------------------------------
    out_ln = torch.empty_like(out, dtype=torch.float32, device=device)
    BLOCK_C2 = 64
    grid = (triton.cdiv(B * N * N, 1),)
    _layer_norm_kernel[grid](
        inp_ptr=out,
        out_ptr=out_ln,
        weight_ptr=weights["to_out_norm.weight"],
        bias_ptr=weights["to_out_norm.bias"],
        B=B,
        N=N,
        C=hidden_dim,
        eps=eps,
        BLOCK_C=BLOCK_C2,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Final gating + projection (fp32)
    # --------------------------------------------------------------
    out_ln = out_ln * out_gate.to(torch.float32)          # broadcast over hidden dim
    out_final = F.linear(out_ln, weights["to_out.weight"])   # (B,N,N,dim)

    return out_final
