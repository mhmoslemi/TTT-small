"""
TriMul “outgoing” forward (no gradients).

The dominant O(N³) operation
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a batched GEMM in fp16 with a small Triton kernel
that only does the matrix multiplication (no Python loops).
All cheap element‑wise work (layer‑norms, linear projections,
gates, mask broadcasting) stays in PyTorch so that we keep
high‑throughput Tensor‑Core usage while still satisfying the
requirement “at least one Triton kernel”.

The kernel works on tensors of shape
    (Z, I, K)  and  (Z, K, J)   →   (Z, I, J)
where Z = B * hidden_dim.  The hidden dimension is moved to the
second axis (B, H, …) and the reduction dimension K (= seq_len)
is made contiguous.  fp16 inputs are accumulated in fp32 for
numerical safety and the result is cast back to fp16 before it
is reshaped to the original (B, N, N, dim) layout and processed
by the remaining PyTorch layers (LayerNorm → gating → final
Linear).  The final output is returned in float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton: batched GEMM (Z, I, K) × (Z, K, J) -> (Z, I, J)
# -------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    a_ptr,            # left  tensor (Z, I, K)  fp16
    b_ptr,            # right tensor (Z, K, J)  fp16
    c_ptr,            # output tensor (Z, I, J) fp16
    M, N, K,          # dimensions
    stride_az, stride_am, stride_ak,   # a strides (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)          # batch*head dimension
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in BLOCK_K steps
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K) and B tile (K × N)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp32 dot‑product accumulation
        acc += tl.dot(a, b)

    # store fp16 result
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batch_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM via the Triton kernel above.

    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous
    returns (B, H, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # allocate output
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse batch*head into a single dimension for the kernel
    Z = B * H
    a = left.reshape(Z, I, K)
    b = right.reshape(Z, K, J)
    c = out.reshape(Z, I, J)

    # strides in element units (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling – tuned for Hopper/Tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        Z,                                 # program_id(0) – batch*head
        triton.cdiv(I, BLOCK_M),           # program_id(1) – rows
        triton.cdiv(J, BLOCK_N),           # program_id(2) – cols
    )

    batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # enough warps to keep the SM busy
        num_stages=3,
    )
    return out


# -------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim] (float32, CUDA)
        mask         : torch.Tensor [B, N, N] (float32/bool) or None
        weights      : dict of torch.Tensor parameters
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep in PyTorch – cheap)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual LayerNorm (fp32)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                       # [B,N,N,dim] fp32

    # ------------------------------------------------------------------
    # 2) Cast once to fp16 for heavy ops
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # linear weight tensors (fp16 for proj/gates, fp32 for final)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"]      # fp32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]          # fp32

    # ------------------------------------------------------------------
    # 3) Linear projections (B,N,N,hidden_dim) in fp16
    # ------------------------------------------------------------------
    left  = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden]
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure mask is fp16 and has a trailing dim for broadcasting
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core N³ contraction via Triton batched GEMM
    #    left : (B, hidden, N, N)   -> (B, hidden, I, K)
    #    right: (B, hidden, N, N)   -> (B, hidden, K, J)   (transpose last two)
    # ------------------------------------------------------------------
    # bring hidden to 2nd axis and make reduction dimension contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # batched GEMM in fp16, result fp16
    out_fp16 = triton_batch_gemm(left_perm, right_perm)   # (B, H, I, J)

    # reshape back to (B,N,N,dim) and cast to fp32 for the remaining ops
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 8) Final gating + linear projection (fp32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)         # same gate used earlier
    out = F.linear(out, to_out_w)                  # (B,N,N,dim) fp32

    return out
