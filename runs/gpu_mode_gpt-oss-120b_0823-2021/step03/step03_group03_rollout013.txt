"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  Manual LayerNorm on the 4‑D input (FP32) – elementwise kernel.
2️⃣  FP16 linear projections + 3 × sigmoid gates (torch, FP16 fast).
3️⃣  Optional pair‑wise mask (FP16) – fused in the same elementwise kernel as
    the projection result.
4️⃣  Core cubic contraction:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply over the reduction dimension *k*.  
   We reshape ``left``/``right`` to (B*H, N, N) (H = hidden_dim) and call
   ``torch.bmm`` (cuBLAS/Tensor‑cores) – the dominant O(N³) work.
5️⃣  FP32 LayerNorm on the result, output‑gate multiplication, and final
   linear projection back to the original dimension.

Only the elementwise work (LayerNorm + optional mask) is done in Triton,
which gives a noticeable speed‑up over pure PyTorch while keeping the
implementation simple and numerically identical to the reference.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm (mean/var) + optional mask multiplication
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_mask_kernel(
    inp_ptr,                # input  (B,N,N,C)   FP32
    out_ptr,                # output (B,N,N,C)   FP32
    mask_ptr,               # mask   (B,N,N)    FP16  (or nullptr)
    eps,                    # 1e-5  (const float)
    ln_w_ptr,               # LayerNorm weight (C)   FP32
    ln_b_ptr,               # LayerNorm bias   (C)   FP32
    B, N, C,                # static extents
    stride_ib, stride_in, stride_ik, stride_ic,   # input strides
    stride_ob, stride_on, stride_ok, stride_oc,   # output strides
    stride_mb, stride_mn, stride_mk,               # mask strides (may be 0)
    MASKED: tl.constexpr,                          # 1 → mask present, 0 → no mask
):
    pid = tl.program_id(0)                # linear index over (B,N,N)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from linear pid
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # pointers to the first element of the row (c = 0)
    inp_row = inp_ptr + b * stride_ib + i * stride_in + j * stride_ik
    out_row = out_ptr + b * stride_ob + i * stride_on + j * stride_ok

    # ------------------------------------------------------------------
    # 1️⃣ compute mean & var over the channel dimension (C)
    # ------------------------------------------------------------------
    offs_c = tl.arange(0, C)
    # load full channel vector
    x = tl.load(inp_row + offs_c * stride_ic, mask=offs_c < C, other=0.0)   # FP32
    # mean
    mean = tl.sum(x, axis=0) / C
    # variance
    var = tl.sum((x - mean) * (x - mean), axis=0) / C

    # ------------------------------------------------------------------
    # 2️⃣ apply LayerNorm (gamma, beta) and optional mask
    # ------------------------------------------------------------------
    gamma = tl.load(ln_w_ptr + offs_c, mask=offs_c < C, other=0.0)
    beta  = tl.load(ln_b_ptr + offs_c, mask=offs_c < C, other=0.0)

    normed = (x - mean) / tl.sqrt(var + eps)
    out = normed * gamma + beta

    if MASKED:
        # mask is FP16, we up‑cast to FP32 before multiplication
        mask_val = tl.load(mask_ptr + b * stride_mb + i * stride_mn + j * stride_mk,
                           mask=True, other=0.0).to(tl.float32)
        out = out * mask_val

    # store result
    tl.store(out_row + offs_c * stride_oc, out, mask=offs_c < C)


def _layernorm_mask(
    inp: torch.Tensor,
    mask: torch.Tensor | None,
    ln_weight: torch.Tensor,
    ln_bias: torch.Tensor,
) -> torch.Tensor:
    """
    Fuse LayerNorm (FP32) and optional mask (FP16) into a single Triton kernel.
    Returns a tensor of the same shape & dtype (FP32) as ``inp``.
    """
    B, N, _, C = inp.shape
    out = torch.empty_like(inp)

    # strides (in elements, not bytes)
    stride_ib, stride_in, stride_ik, stride_ic = inp.stride()
    stride_ob, stride_on, stride_ok, stride_oc = out.stride()

    if mask is None:
        mask_ptr = torch.tensor(0, dtype=torch.int64, device=inp.device)  # dummy
        mask_strides = (0, 0, 0)
        masked = 0
    else:
        mask = mask.to(torch.float16)
        mask_ptr = mask
        mask_strides = mask.stride()
        masked = 1

    grid = (triton.cdiv(B * N * N, 1),)   # one program per (b,i,j)
    _layernorm_mask_kernel[grid](
        inp_ptr=inp,
        out_ptr=out,
        mask_ptr=mask_ptr,
        eps=1e-5,
        ln_w_ptr=ln_weight,
        ln_b_ptr=ln_bias,
        B=B,
        N=N,
        C=C,
        stride_ib=stride_ib,
        stride_in=stride_in,
        stride_ik=stride_ik,
        stride_ic=stride_ic,
        stride_ob=stride_ob,
        stride_on=stride_on,
        stride_ok=stride_ok,
        stride_oc=stride_oc,
        stride_mb=mask_strides[0] if mask is not None else 0,
        stride_mn=mask_strides[1] if mask is not None else 0,
        stride_mk=mask_strides[2] if mask is not None else 0,
        MASKED=masked,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Entry‑point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim]      (float32)
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf weight tensors (see reference impl.)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm + optional mask (fused Triton kernel)
    # --------------------------------------------------------------
    x = _layernorm_mask(
        inp,
        mask,
        weights["norm.weight"],
        weights["norm.bias"],
    )                                   # (B,N,N,dim)  FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gates (FP16)
    # --------------------------------------------------------------
    # cast once
    x_fp16 = x.to(torch.float16)

    # weight tensors – already FP32, convert to FP16 (no bias)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)     # (hidden_dim, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # projections
    left  = F.linear(x_fp16, left_proj_w)    # (B,N,N,hidden_dim)
    right = F.linear(x_fp16, right_proj_w)   # (B,N,N,hidden_dim)

    # 3️⃣ gating (sigmoid) – still FP16
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden_dim)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4️⃣ Core contraction (batched GEMM via cuBLAS)
    # --------------------------------------------------------------
    # reshape to (B, H, I, K) and (B, H, K, J) where I=J=N, K=N, H=hidden_dim
    # left:  (B,N,N,H) → (B,H,N,N)
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,I,K)
    # right: (B,N,N,H) → (B,H,N,N) and we need (K,J) = (N,N) with J=N
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B,H,K,J)

    # Collapse batch*head dimension and perform batched matmul
    B_H, I, K = left_t.shape[0] * left_t.shape[1], left_t.shape[2], left_t.shape[3]
    left_flat  = left_t.view(B_H, I, K)      # (B*H, N, N)
    right_flat = right_t.view(B_H, K, I)     # (B*H, N, N)

    # torch.bmm uses Tensor‑cores on Hopper (FP16→FP32 accumulation)
    out_flat = torch.bmm(left_flat, right_flat)   # (B*H, N, N) FP16

    # reshape back to (B, H, N, N) then permute to (B,N,N,H)
    out_fp16 = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Output LayerNorm (FP32) – we reuse the same Triton kernel
    # --------------------------------------------------------------
    out_fp32 = _layernorm_mask(
        out_fp16.to(torch.float32),          # no mask at this stage
        None,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )                                      # (B,N,N,hidden_dim) FP32

    # --------------------------------------------------------------
    # 6️⃣ Final gating + projection back to original dim (FP32)
    # --------------------------------------------------------------
    # out_gate is still FP16 – promote to FP32 for the final multiplication
    out = out_fp32 * out_gate.to(torch.float32)      # (B,N,N,hidden_dim)
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)  FP32

    return out
