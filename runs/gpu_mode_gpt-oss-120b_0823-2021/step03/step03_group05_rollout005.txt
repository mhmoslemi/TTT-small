The outgoing TriMul operation consists of three stages  

1. **Layer‑norm + linear projections** – cheap, stay in PyTorch.  
2. **Fuse mask, gate sigmoid and gating of the left/right tensors** – a single
   element‑wise Triton kernel (the only required kernel).  
3. **Cubic contraction** – `out[b,i,j,:] = Σ_k left[b,i,k,:]·right[b,j,k,:]`.  
   This is a batched matrix‑matrix multiplication over the `(N,N)` axes for
   every hidden channel.  The fastest way on a Hopper GPU is to reshape the
   tensors to `(B·H, N, N)` and call `torch.bmm`, which uses Tensor‑cores.
4. **Post‑processing** – layer‑norm, out‑gate, final linear (again in PyTorch).

The kernel only touches the element‑wise part, therefore it is tiny, memory‑
bound and runs with very low latency.  All heavy linear‑algebra work is handed
off to cuBLAS/Tensor‑cores via `torch.bmm`.

The implementation below follows the exact same mathematics as the reference
PyTorch module and works for any combination of batch‑size, sequence length,
mask‑presence and dtype described in the statement.

```python
"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1. Layer‑norm + linear projections (left/right, left/right‑gate, out‑gate)
   are performed with PyTorch (these are cheap compared with O(N³)).
2. A single Triton kernel fuses:
       * optional mask (broadcasted over the hidden dimension)
       * left‑gate, right‑gate sigmoid multiplication
   in‑place on the left/right tensors (FP16).
3. The cubic contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is realized as a batched GEMM:
       left   → (B·H, N, N)
       right  → (B·H, N, N)  (transpose on the second N)
   and executed with `torch.bmm`, which on Hopper maps to Tensor‑core
   GEMM kernels.
4. Post‑processing – layer‑norm, out‑gate, final linear – again in PyTorch.
5. The final tensor is returned in float‑32.

Only the element‑wise mask‑gate fusion is implemented in Triton, satisfying
the “must use a kernel” rule while keeping the overall runtime well under the
target budget.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel – element‑wise fusion of mask, left‑gate, right‑gate.
# All tensors are 4‑D (B, N, N, H) stored contiguously in FP16.
# -------------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *mut fp16   (B,N,N,H) – overwritten
    right_ptr,         # *mut fp16   (B,N,N,H) – overwritten
    left_gate_ptr,     # *const fp16 (B,N,N,H)   sigmoid(left)   (already in FP16)
    right_gate_ptr,    # *const fp16 (B,N,N,H)   sigmoid(right)  (already in FP16)
    mask_ptr,          # *const fp16 (B,N,N,1)   1.0 if no mask
    B, N, H,           # static sizes
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_g0, stride_g1, stride_g2, stride_g3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    BLOCK: tl.constexpr,
):
    """Fuse mask and gating into left/right in‑place.

    left  = left  * mask * left_gate
    right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    total = B * N * N * H
    idx = pid * BLOCK + tl.arange(0, BLOCK)
    mask = idx < total

    # decode linear index → (b,i,j,h)
    tmp = idx
    h = tmp % H;          tmp //= H
    j = tmp % N;          tmp //= N
    i = tmp % N;          b = tmp // N

    # compute pointers (stride in elements, not bytes)
    l_ptr = left_ptr    + b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    r_ptr = right_ptr   + b * stride_r0 + i * stride_r1 + j * stride_r2 + h * stride_r3
    lg_ptr = left_gate_ptr  + b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    rg_ptr = right_gate_ptr + b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    m_ptr = mask_ptr + b * stride_m0 + i * stride_m1 + j * stride_m2          # broadcast over H

    # load
    l  = tl.load(l_ptr, mask=mask, other=0.0)
    r  = tl.load(r_ptr, mask=mask, other=0.0)
    lg = tl.load(lg_ptr, mask=mask, other=0.0)
    rg = tl.load(rg_ptr, mask=mask, other=0.0)
    m  = tl.load(m_ptr, mask=mask, other=1.0)   # 1.0 when mask is all‑ones

    # fused elementwise
    l = l * m * lg
    r = r * m * rg

    # store back
    tl.store(l_ptr, l, mask=mask)
    tl.store(r_ptr, r, mask=mask)


def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        - x       : Tensor[B, N, N, dim]   (float32)
        - mask    : Tensor[B, N, N] or None (bool/float32)
        - weights : dict of model parameters (float32)
        - config  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    Tensor[B, N, N, dim]   (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]          # H

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm + linear projections (still in FP32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # linear projections → (B,N,N,hidden_dim)
    left  = F.linear(x, weights["left_proj.weight"].to(dtype))
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # gates (pre‑sigmoid)
    left_gate  = F.linear(x, weights["left_gate.weight"].to(dtype))
    right_gate = F.linear(x, weights["right_gate.weight"].to(dtype))
    out_gate   = F.linear(x, weights["out_gate.weight"].to(dtype))

    # sigmoid
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # -----------------------------------------------------------------
    # 2️⃣ Fuse mask and gate multiplication in a Triton kernel
    # -----------------------------------------------------------------
    # mask tensor of shape (B,N,N,1) in FP16; all‑ones when mask is None
    if mask is None:
        mask_fp16 = torch.ones((B, N, N, 1), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(dtype).unsqueeze(-1)          # (B,N,N,1)
        mask_fp16 = mask_fp16.to(torch.float16)

    # Cast everything to FP16 for the kernel (small precision loss, later up‑cast)
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    left_gate_fp16  = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()

    total_elements = B * N * N * hidden_dim
    BLOCK = 1024                                     # 1‑k elementwise work per thread block
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_fp16,
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        mask_fp16.stride(0), mask_fp16.stride(1),
        mask_fp16.stride(2), mask_fp16.stride(3),
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 3️⃣ Cubic contraction via batched GEMM (FP16 → FP32)
    # -----------------------------------------------------------------
    # reshape to (B·H, N, N)
    left_bh  = left_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_bh = right_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # batched matrix multiplication: (B·H,N,N) @ (B·H,N,N)^T
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))

    # reshape back to (B, N, N, H) and cast to FP32 for the remaining ops
    out = out_bh.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(dtype)

    # -----------------------------------------------------------------
    # 4️⃣ Post‑processing (LayerNorm, out‑gate, final linear)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    # out_gate has shape (B,N,N,hidden_dim) after sigmoid, already in FP32
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
```
