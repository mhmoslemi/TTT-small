"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
0️⃣ Unpack arguments.
1️⃣ Manual LayerNorm on the last dimension (FP32) – cheap and avoids a kernel.
2️⃣ Cast to FP16 once and compute all four Linear layers (projections & gates) via
   torch.nn.functional.linear (still in FP16 for speed).
3️⃣ Apply the pair‑wise mask (if present) and the three sigmoid gates **inside a
   tiny Triton element‑wise kernel**.  This keeps the whole elementwise work on
   the GPU and saves a costly Python loop.
4️⃣ Core cubic contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix‑multiply over the reduction dimension *k*.  We reshape
   ``left`` and ``right`` to (B*H, N, N) and invoke ``torch.bmm`` which uses
   cuBLAS/Tensor‑cores on Hopper – far faster than a full Triton GEMM.
5️⃣ Bring the result back to (B,N,N,dim) and LayerNorm (FP32) again.
6️⃣ Final output gate (sigmoid) and linear projection back to the original
   ``dim`` (FP32).

Only the element‑wise part is done with Triton; the heavy N³ work relies on the
high‑performance cuBLAS implementation already available in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – elementwise multiply three tensors
# ----------------------------------------------------------------------
@triton.jit
def _elem_mul3_kernel(
    a_ptr,          # tensor A
    b_ptr,          # tensor B (may be broadcast)
    c_ptr,          # tensor C (may be broadcast)
    out_ptr,        # output
    N: tl.constexpr,               # total number of elements
    stride_a0, stride_a1, stride_a2, stride_a3,
    stride_b0, stride_b1, stride_b2, stride_b3,
    stride_c0, stride_c1, stride_c2, stride_c3,
    stride_o0, stride_o1, stride_o2, stride_o3,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N

    # compute multi‑dimensional indices from linear offset
    # we assume tensors are contiguous in the last three dims (N,N,hidden)
    # and have shape (B, N, N, H).  The stride arguments are passed from host.
    idx0 = offs // (stride_a1 * stride_a2 * stride_a3)          # B
    rem  = offs %  (stride_a1 * stride_a2 * stride_a3)
    idx1 = rem // (stride_a2 * stride_a3)                       # N
    rem  = rem %  (stride_a2 * stride_a3)
    idx2 = rem // stride_a3                                    # N
    idx3 = rem %  stride_a3                                    # H

    a = tl.load(a_ptr + idx0*stride_a0 + idx1*stride_a1 + idx2*stride_a2 + idx3*stride_a3,
                mask=mask, other=0.0)
    b = tl.load(b_ptr + idx0*stride_b0 + idx1*stride_b1 + idx2*stride_b2 + idx3*stride_b3,
                mask=mask, other=0.0)
    c = tl.load(c_ptr + idx0*stride_c0 + idx1*stride_c1 + idx2*stride_c2 + idx3*stride_c3,
                mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + idx0*stride_o0 + idx1*stride_o1 + idx2*stride_o2 + idx3*stride_o3,
             out, mask=mask)


def _apply_mask_and_gate(tensor, mask, gate):
    """
    tensor : (B,N,N,H) fp16
    mask   : (B,N,N,1) fp16  or None
    gate   : (B,N,N,H) fp16
    returns tensor * mask * gate   (fp16)
    """
    if mask is None:
        # broadcast mask = 1
        mask = torch.ones_like(tensor[:, :, :, :1])
    # flatten to 1‑D view for the kernel
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024

    out = torch.empty_like(tensor)

    # strides for each tensor
    s_a0, s_a1, s_a2, s_a3 = tensor.stride()
    s_b0, s_b1, s_b2, s_b3 = mask.stride()
    s_c0, s_c1, s_c2, s_c3 = gate.stride()
    s_o0, s_o1, s_o2, s_o3 = out.stride()

    grid = (triton.cdiv(total, BLOCK),)
    _elem_mul3_kernel[grid](
        a_ptr=tensor,
        b_ptr=mask,
        c_ptr=gate,
        out_ptr=out,
        N=total,
        stride_a0=s_a0, stride_a1=s_a1, stride_a2=s_a2, stride_a3=s_a3,
        stride_b0=s_b0, stride_b1=s_b1, stride_b2=s_b2, stride_b3=s_b3,
        stride_c0=s_c0, stride_c1=s_c1, stride_c2=s_c2, stride_c3=s_c3,
        stride_o0=s_o0, stride_o1=s_o1, stride_o2=s_o2, stride_o3=s_o3,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf weight tensors
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b      # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gates (cast to FP16 once)
    # --------------------------------------------------------------
    def _to_fp16(t): return t.to(torch.float16)

    left_proj_w   = _to_fp16(weights["left_proj.weight"])    # (hidden, dim)
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)   # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)    # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Mask & gating – Triton elementwise kernel
    # --------------------------------------------------------------
    # mask: (B,N,N) -> (B,N,N,1) fp16
    if mask is None:
        mask_fp16 = None
    else:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)

    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B,N,N,hidden)

    left  = _apply_mask_and_gate(left,  mask_fp16, left_gate)   # fp16
    right = _apply_mask_and_gate(right, mask_fp16, right_gate)

    # --------------------------------------------------------------
    # 4️⃣ Core contraction – batched GEMM via torch.bmm (cuBLAS)
    # --------------------------------------------------------------
    # reshape to (B*hidden, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)   # (B*H,N,N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden, N, N)  # (B*H,N,N)

    # (B*H,N,N) @ (B*H,N,N)^T  -> (B*H,N,N)
    out_mat = torch.bmm(left_mat, right_mat)   # fp16, uses Tensor‑cores

    # back to (B,N,N,hidden)
    out_fp16 = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Output LayerNorm (FP32)
    # --------------------------------------------------------------
    out_fp32 = out_fp16.to(torch.float32)
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out_fp32.mean(-1, keepdim=True)
    var_o  = out_fp32.var(-1, unbiased=False, keepdim=True)
    out_fp32 = (out_fp32 - mean_o) / torch.sqrt(var_o + 1e-5)
    out_fp32 = out_fp32 * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 6️⃣ Final gating + projection (FP32)
    # --------------------------------------------------------------
    out_fp32 = out_fp32 * out_gate.to(torch.float32)          # broadcast over hidden
    out_fp32 = F.linear(out_fp32, weights["to_out.weight"])  # (B,N,N,dim)

    return out_fp32
