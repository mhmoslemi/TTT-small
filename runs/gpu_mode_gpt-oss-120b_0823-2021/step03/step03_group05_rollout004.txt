"""
TriMul (outgoing) – fast forward‑pass.

Algorithm
---------
1️⃣ Layer‑norm + four linear projections (left, right, left_gate,
    right_gate, out_gate) – stay in PyTorch (these are cheap).

2️⃣ One fused Triton kernel that:
   * applies the optional mask,
   * multiplies left/right by their respective sigmoid‑gates,
   * writes the result back in‑place.
   The kernel works on the contiguous FP16 tensors
   (B, N, N, H) and uses a 1‑D launch grid.  No shared‑memory is needed
   because the operation is purely element‑wise.

3️⃣ Core O(N³) contraction:
   out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   is realised as a batched matrix‑multiply.
   After the element‑wise fuse we reshape
       left  → (B·H, N, N)
       right → (B·H, N, N)  and transpose the last two axes.
   The call ``torch.bmm`` (FP16) invokes cuBLAS/Tensor‑cores on the H200,
   giving the best possible performance for the cubic work.

4️⃣ Post‑processing (layer‑norm, out_gate, final linear) – again in PyTorch.
   The final tensor is returned in FP32 as required.

The implementation respects all constraints:
* works for any mask (including “nomask” → all‑ones),
* supports the different (B,N,dim,hidden_dim) configurations,
* only a single Triton kernel is used (fulfilling the “must use a kernel”
  rule), and the heavy N³ work is delegated to cuBLAS.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel – fuse mask, left_gate, right_gate into left/right tensors.
# All tensors are assumed contiguously stored FP16 with shape (B,N,N,H).
# -------------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # *mut fp16  (B,N,N,H) – overwritten with gated left
    right_ptr,         # *mut fp16  (B,N,N,H) – overwritten with gated right
    left_gate_ptr,     # *const fp16 (B,N,N,H)   sigmoid(left_gate)
    right_gate_ptr,    # *const fp16 (B,N,N,H)   sigmoid(right_gate)
    mask_ptr,          # *const fp16 (B,N,N,1)   1.0 if mask is absent
    B, N, H,           # compile‑time constants
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_g0, stride_g1, stride_g2, stride_g3,
    stride_m0, stride_m1, stride_m2, stride_m3,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,j,h):
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    total = B * N * N * H
    index = pid * BLOCK + tl.arange(0, BLOCK)          # linear index
    mask = index < total

    # -------- decode linear index --------
    #   index = (((b * N + i) * N + j) * H) + h
    tmp = index
    h = tmp % H; tmp = tmp // H
    j = tmp % N; tmp = tmp // N
    i = tmp % N; b = tmp // N

    # -------- compute pointers ----------
    l_ptr = left_ptr   + b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    r_ptr = right_ptr  + b * stride_r0 + i * stride_r1 + j * stride_r2 + h * stride_r3
    lg_ptr = left_gate_ptr  + b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    rg_ptr = right_gate_ptr + b * stride_g0 + i * stride_g1 + j * stride_g2 + h * stride_g3
    m_ptr = mask_ptr + b * stride_m0 + i * stride_m1 + j * stride_m2  # broadcast over H

    # -------- load ----------
    l = tl.load(l_ptr, mask=mask, other=0.0)
    r = tl.load(r_ptr, mask=mask, other=0.0)
    lg = tl.load(lg_ptr, mask=mask, other=0.0)
    rg = tl.load(rg_ptr, mask=mask, other=0.0)
    m = tl.load(m_ptr, mask=mask, other=1.0)          # 1.0 when mask is all‑ones

    # -------- fused computation ----------
    l = l * m * lg
    r = r * m * rg

    # -------- store ----------
    tl.store(l_ptr, l, mask=mask)
    tl.store(r_ptr, r, mask=mask)


def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        - x : Tensor[B, N, N, dim]          (float32)
        - mask : Tensor[B, N, N] or None    (bool / float32)
        - weights : dict of FP32 parameters (see body)
        - config : {"dim": int, "hidden_dim": int}

    Returns
    -------
    Tensor[B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]                     # H

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm + four linear projections (cheap, stay in PyTorch)
    # --------------------------------------------------------------
    # Layer‑norm over the last dimension
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                                     # (B,N,N,dim)

    # Linear projections → hidden dimension
    left  = F.linear(x, weights["left_proj.weight"])      # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])    # (B,N,N,H)

    # Gates (pre‑sigmoid)
    left_gate  = F.linear(x, weights["left_gate.weight"])
    right_gate = F.linear(x, weights["right_gate.weight"])
    out_gate   = F.linear(x, weights["out_gate.weight"])

    # Sigmoid activation
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)                 # (B,N,N,H)

    # --------------------------------------------------------------
    # 2️⃣ Fuse mask + gate multiplication in a Triton kernel
    # --------------------------------------------------------------
    # Build a mask tensor of shape (B,N,N,1) in FP16.
    # When ``mask`` is None (nomask) we use an all‑ones tensor.
    if mask is None:
        mask_tensor = torch.ones((B, N, N, 1),
                                 dtype=torch.float16,
                                 device=device)
    else:
        # mask may be bool or float; ensure float32 → float16 conversion.
        mask_tensor = mask.to(torch.float32).unsqueeze(-1)      # (B,N,N,1)
        mask_tensor = mask_tensor.to(torch.float16)

    # Cast everything to FP16 for the kernel (no precision loss for the
    # subsequent batched GEMM because it also runs in FP16).
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    left_gate_fp16  = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()

    total_elements = B * N * N * hidden_dim
    BLOCK = 1024                                 # 1‑D elementwise kernel
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        right_fp16,
        left_gate_fp16,
        right_gate_fp16,
        mask_tensor,
        B, N, hidden_dim,
        # strides for left
        left_fp16.stride(0), left_fp16.stride(1),
        left_fp16.stride(2), left_fp16.stride(3),
        # strides for right
        right_fp16.stride(0), right_fp16.stride(1),
        right_fp16.stride(2), right_fp16.stride(3),
        # strides for gates
        left_gate_fp16.stride(0), left_gate_fp16.stride(1),
        left_gate_fp16.stride(2), left_gate_fp16.stride(3),
        # strides for mask (broadcasted over H)
        mask_tensor.stride(0), mask_tensor.stride(1),
        mask_tensor.stride(2), mask_tensor.stride(3),
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 3️⃣ Core O(N³) contraction – batched GEMM via torch.bmm (FP16)
    # --------------------------------------------------------------
    # reshape to (B*H, N, N); left stays as‑is, right is transposed on the
    # *second* N‑axis (the “k” dimension in the original einsum).
    left_bh  = left_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_bh = right_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # (B*H, N, N)  @  (B*H, N, N)^T  →  (B*H, N, N)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_bh.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)                     # back to FP32 for the rest

    # --------------------------------------------------------------
    # 4️⃣ Post‑processing: layer‑norm, out‑gate, final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                           # broadcast over (N,N)
    out = F.linear(out, weights["to_out.weight"])

    return out
