"""
TriMul “outgoing” forward pass (AlphaFold‑3).

The expensive O(N³) operation is the contraction

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

where ``left`` and ``right`` are the linearly‑projected, gated tensors
of shape [B, H, N, N] (H = hidden_dim).  All cheap work (LayerNorm,
linear projections, gating, final LayerNorm / linear) is performed with
standard PyTorch, which is memory‑bound and fast on the H200.

The contraction itself is a batched matrix‑multiply between a (N×N)
matrix and the *transpose* of another (N×N) matrix for every (b, h)
pair.  We implement it with a single Triton kernel that uses a classic
blocked GEMM tiling (BLOCK_M × BLOCK_N × BLOCK_K).  The kernel works in
FP16 with FP32 accumulation, stores the result back as FP16 and finally
converts to FP32 for the rest of the pipeline.

The kernel is launched with one program‑id per (batch, head, tile_m,
tile_n) block, i.e. a total of B·H·ceil(N/BLOCK_M)·ceil(N/BLOCK_N) blocks.
This layout yields near‑perfect occupancy on Hopper (SM‑90) while keeping
the shared‑memory usage well below the limit.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # const fp16*  [B, H, N, N]
    right_ptr,         # const fp16*  [B, H, N, N]   (read transposed)
    out_ptr,           #       fp16*  [B, H, N, N]
    B, N, H,           # int meta‑data
    # strides (row‑major) for the three tensors
    stride_lb, stride_lh, stride_ln, stride_lk,   # left
    stride_rb, stride_rh, stride_rn, stride_rk,   # right
    stride_ob, stride_oh, stride_on, stride_ok,   # out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute a (BLOCK_M × BLOCK_N) tile of

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    The kernel iterates over the K‑dimension in tiles and accumulates in
    FP32.  ``right`` is accessed as the transpose so that a single
    ``tl.dot`` computes the inner product.
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Decode the program id into (b, h, tile_m, tile_n)
    # ------------------------------------------------------------------
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    tile_n = pid % num_n
    pid //= num_n
    tile_m = pid % num_m
    pid //= num_m
    h = pid % H
    b = pid // H

    # ------------------------------------------------------------------
    # Compute the absolute offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, h, i, k]   shape (M, K)
        l_ptr = (left_ptr +
                 b * stride_lb +
                 h * stride_lh +
                 offs_m[:, None] * stride_ln +
                 offs_k[None, :] * stride_lk)
        l = tl.load(l_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[b, h, j, k] is read as right^T  -> shape (K, N)
        r_ptr = (right_ptr +
                 b * stride_rb +
                 h * stride_rh +
                 offs_k[:, None] * stride_rk +   # k dimension becomes row
                 offs_n[None, :] * stride_rn)    # n dimension becomes col
        r_t = tl.load(r_ptr,
                      mask=mask_k[:, None] & mask_n[None, :],
                      other=0.0).to(tl.float32)

        # (M, K) @ (K, N) -> (M, N)
        acc += tl.dot(l, r_t)

    # ------------------------------------------------------------------
    # Store the result back
    # ------------------------------------------------------------------
    out_ptrs = (out_ptr +
                b * stride_ob +
                h * stride_oh +
                offs_m[:, None] * stride_on +
                offs_n[None, :] * stride_ok)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Host side entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input  [B, N, N, dim]
        torch.Tensor,                     # mask   [B, N, N]  (or None)
        Dict[str, torch.Tensor],          # model weights
        Dict[str, Any]                    # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]   # = H

    # ------------------------------------------------------------------
    # 1️⃣  Cheap ops in PyTorch
    # ------------------------------------------------------------------
    # LayerNorm over the last dimension
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # Linear projections (bias = False)
    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # Optional masking
    if mask is not None:
        # mask shape (B,N,N) → (B,N,N,1) for broadcast
        mask_ = mask.unsqueeze(-1).to(dtype)
        left = left * mask_
        right = right * mask_

    # Gates (sigmoid after linear)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 2️⃣  Rearrange to [B, H, N, N] and cast to fp16 for the kernel
    # ------------------------------------------------------------------
    # (B,N,N,H) → (B,H,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)   # (B,H,N,N), fp16

    # ------------------------------------------------------------------
    # 3️⃣  Triton kernel launch (blocked GEMM)
    # ------------------------------------------------------------------
    # Tile sizes – tuned for Hopper (SM‑90).  They keep shared‑memory < 48 KB.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    # total blocks = B * H * grid_m * grid_n
    grid = (B * hidden_dim * grid_m * grid_n,)

    _trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        # strides for left (row‑major)
        left_fp16.stride(0),   # B
        left_fp16.stride(1),   # H
        left_fp16.stride(2),   # N (row)
        left_fp16.stride(3),   # N (col)
        # strides for right
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        # strides for out
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 4️⃣  Post‑process in PyTorch
    # ------------------------------------------------------------------
    # Back to (B,N,N,H) layout and cast to float32 for remaining ops
    out = out_fp16.to(dtype).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # Final LayerNorm (over hidden_dim)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # Apply output gate (broadcast over pairwise dimensions)
    out = out * out_gate

    # Final linear back to the original dimensionality
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
