```python
"""
custom_kernel – “outgoing” TriMul forward pass (AlphaFold‑3 style).

Algorithm
---------
1. Layer‑norm the 4‑D input (B,N,N,D) – done in PyTorch (FP32) because
   the mean/var reduction over the channel dimension is cheap compared
   to the cubic contraction.
2. Linear projections + gating are also kept in PyTorch, but the heavy
   O(N³) contraction

        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

   is performed by a fused batched GEMM written in Triton.  The tensors
   are reshaped to (B·H, I, K) and (B·H, K, J) where H = hidden_dim,
   I = J = N, K = N.  The kernel computes a FP16 mat‑mul with an FP32
   accumulator (Tensor‑Core friendly) and writes the result back in
   FP16.  The remaining post‑processing (output LayerNorm, gating and
   final linear) is performed in PyTorch.

3. All casts are explicit so that the final result has dtype ``torch.float32``.

The kernel follows the Triton launch convention (grid = (ceil(I/BM),
ceil(J/BN), B·H)).  Tiling (BM=64, BN=64, BK=32) is a good
Tensor‑Core configuration on Hopper/SM90 and stays well within the
shared‑memory budget.

Only the O(N³) contraction is moved to Triton – this satisfies the
requirement “implement at least part of the operations in a kernel” while
keeping the surrounding code simple and correct.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel : batched GEMM (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    a_ptr,               # left  matrix pointer   (Z, I, K)
    b_ptr,               # right matrix pointer   (Z, K, J)
    c_ptr,               # output matrix pointer  (Z, I, J)
    M, N, K,             # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # tile identifiers
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)          # row tile
    pid_n = tl.program_id(1)          # col tile
    pid_z = tl.program_id(2)          # batch*head tile

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (higher precision for the sum)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # reduction over K
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile  (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # accumulate
        acc += tl.dot(a, b)

    # store back in FP16 (Tensor‑Core output type)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batch_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.

    Parameters
    ----------
    left  : Tensor (B, H, I, K)  dtype=torch.float16, contiguous
    right : Tensor (B, H, K, J)  dtype=torch.float16, contiguous

    Returns
    -------
    out : Tensor (B, H, I, J) dtype=torch.float16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse the leading two dims -> Z = B*H
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # Hopper‑friendly tile sizes (Tensor‑Core friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # rows
        triton.cdiv(J, BLOCK_N),   # cols
        Z,                         # batch*head
    )
    batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default on SM90
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask         : torch.Tensor [B, N, N] (float32/bool) or None
        weights      : dict of parameter tensors
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                         # [B,N,N,dim] FP32

    # ------------------------------------------------------------------
    # 2. Linear projections (convert to FP16 for the heavy part)
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w = to_fp16(weights["left_proj.weight"])   # (hidden_dim, dim)
    right_proj_w = to_fp16(weights["right_proj.weight"])

    left_gate_w = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w = to_fp16(weights["out_gate.weight"])

    # FP16 copy of the normalized tensor
    x_fp16 = x.to(torch.float16)

    # (B,N,N,hidden_dim)
    left = F.linear(x_fp16, left_proj_w)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton batched GEMM (the cubic contraction)
    #    out_fp16[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, k, j]
    # ------------------------------------------------------------------
    # reshape to (B, H, I, K) and (B, H, K, J)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, J)

    out_fp16 = triton_batch_matmul(left_perm, right_perm)   # (B, H, I, J)

    # back to original layout and cast to FP32 for the remaining ops
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7. Final gating + linear projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)               # broadcast over dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                        # (B,N,N,dim)

    return out
```

**Explanation of lessons & what was applied**

| Lesson | Applied? | Reason |
|--------|----------|--------|
| 1 Limit shared memory usage | Yes | Tile sizes (64×64×32) keep shared‑memory < 64 KB per block on SM90. |
| 2 Delegate N³ contraction to cuBLAS | Partially | cuBLAS `bmm` would be fast, but the problem explicitly asks for *some* Triton work. The cubic contraction is implemented as a fused batched GEMM in Triton, which is comparable in performance because we use FP16/Tensor‑Core tiles. |
| 3 Match dtypes for batched GEMM | Yes | Both operands are cast to FP16 before the Triton GEMM; accumulation is FP32, then cast back. |
| Stride unpacking must match rank | Yes | Strides are obtained from the *view‑flattened* 3‑D tensors (`a`, `b`, `c`) that the kernel expects; the number of dimensions matches exactly. |
| Avoid excessive host‑side work | Yes | Only light LayerNorm/gating ops stay in PyTorch; the dominant O(N³) work is on the device. |

The resulting kernel meets the correctness requirements, runs entirely on the GPU, and respects the Triton‑launch conventions for the target Hopper (SM90) hardware.
