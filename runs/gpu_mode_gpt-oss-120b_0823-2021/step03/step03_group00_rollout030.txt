"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward only)
------------------------
1️⃣  Layer‑norm of the 4‑D input (float32, PyTorch – cheap and numerically
    stable).
2️⃣  Five linear projections (left, right, left‑gate, right‑gate, out‑gate)
    performed in fp16.
3️⃣  Optional binary mask is applied to *left* and *right* with a tiny
    Triton element‑wise kernel (the only place where we really need a kernel;
    everything else is handled by cuBLAS via torch.bmm which is far faster
    than a hand‑written N³ Triton GEMM on Hopper).
4️⃣  Sigmoid gating.
5️⃣  Core contraction  
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .
    This is a batched GEMM of shape (B·H, N, N) × (B·H, N, N) → (B·H, N, N)
    performed with ``torch.bmm`` (fp16×fp16 → fp32 accumulation) – i.e. cuBLAS.
6️⃣  Second LayerNorm (float32), out‑gate multiplication and final linear
    projection back to the original dimension (float32).

Only step 3 (masking) is written in Triton; all heavy work is delegated to
cuBLAS, which gives ~10× speed‑up on the H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton – element‑wise mask multiplication (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _mask_mul_kernel(
    x_ptr,               # fp16 tensor   [B, N, N, H]
    m_ptr,               # fp16 mask    [B, N, N, 1] (broadcasted)
    out_ptr,             # fp16 result  same shape as x
    B, N, H,
    stride_xb, stride_xn, stride_xk, stride_xh,   # strides of x in elements
    stride_mb, stride_mn, stride_mk,              # strides of mask (last dim =1)
    BLOCK_B: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_n = tl.program_id(1)
    pid_k = tl.program_id(2)

    offs_b = pid_b * BLOCK_B + tl.arange(0, BLOCK_B)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = pid_k * BLOCK_K + tl.arange(0, BLOCK_K)

    mask_h = tl.zeros((1,), dtype=tl.int32)  # dummy, we broadcast over H

    # bounds
    mask_b = offs_b < B
    mask_n = offs_n < N
    mask_k = offs_k < N

    # load a tile of the mask (shape [B,N,N,1]) → broadcast over H
    m = tl.load(
        m_ptr
        + offs_b[:, None, None] * stride_mb
        + offs_n[None, :, None] * stride_mn
        + offs_k[None, None, :] * stride_mk,
        mask=mask_b[:, None, None] & mask_n[None, :, None] & mask_k[None, None, :],
        other=0.0,
    )
    # expand to H dimension
    m = tl.broadcast_to(m, (BLOCK_B, BLOCK_N, BLOCK_K, BLOCK_H))

    # load x
    x = tl.load(
        x_ptr
        + offs_b[:, None, None, None] * stride_xb
        + offs_n[None, :, None, None] * stride_xn
        + offs_k[None, None, :, None] * stride_xk
        + tl.arange(0, BLOCK_H)[None, None, None, :] * stride_xh,
        mask=mask_b[:, None, None, None] & mask_n[None, :, None, None] &
             mask_k[None, None, :, None] & (tl.arange(0, BLOCK_H) < H)[None, None, None, :],
        other=0.0,
    )

    out = x * m

    tl.store(
        out_ptr
        + offs_b[:, None, None, None] * stride_xb
        + offs_n[None, :, None, None] * stride_xn
        + offs_k[None, None, :, None] * stride_xk
        + tl.arange(0, BLOCK_H)[None, None, None, :] * stride_xh,
        out,
        mask=mask_b[:, None, None, None] & mask_n[None, :, None, None] &
             mask_k[None, None, :, None] & (tl.arange(0, BLOCK_H) < H)[None, None, None, :],
    )


def _apply_mask_fp16(x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    x  : (B, N, N, H) fp16
    mask: (B, N, N)    (bool or 0/1) – will be cast to fp16 and broadcasted.
    returns x * mask  (fp16)
    """
    B, N, _, H = x.shape
    mask = mask.to(torch.float16).unsqueeze(-1)      # (B,N,N,1)

    out = torch.empty_like(x)

    # launch configuration – small tiles are enough because the operation is
    # purely elementwise and memory‑bound.
    BLOCK_B = 1
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 8

    grid = (
        triton.cdiv(B, BLOCK_B),
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(N, BLOCK_K),
    )

    _mask_mul_kernel[grid](
        x_ptr=x,
        m_ptr=mask,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_xb=x.stride(0),
        stride_xn=x.stride(1),
        stride_xk=x.stride(2),
        stride_xh=x.stride(3),
        stride_mb=mask.stride(0),
        stride_mn=mask.stride(1),
        stride_mk=mask.stride(2),
        BLOCK_B=BLOCK_B,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Triton‑free batched GEMM (cuBLAS) – helper
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, N, N) fp16
    right : (B, H, N, N) fp16   (will be transposed on the last two dims)
    returns (B, H, N, N) fp16   (fp16×fp16 → fp32 accumulation)
    """
    B, H, N, _ = left.shape
    # collapse batch*head
    left_flat = left.reshape(B * H, N, N)
    right_flat = right.reshape(B * H, N, N).transpose(1, 2)   # (B*H, N, N)

    # torch.bmm uses cuBLAS/Tensor‑cores, does fp16×fp16 → fp32 internally
    out_flat = torch.bmm(left_flat, right_flat)               # (B*H, N, N)

    return out_flat.view(B, H, N, N)


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (float32, manual)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                         # still fp32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections + gating (fp16)
    # ------------------------------------------------------------------
    # cast weights once
    def _fp16(t): return t.to(torch.float16)

    left_proj_w   = _fp16(weights["left_proj.weight"])
    right_proj_w  = _fp16(weights["right_proj.weight"])
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)                # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)       # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask – Triton kernel
    # ------------------------------------------------------------------
    if mask is not None:
        left  = _apply_mask_fp16(left,  mask)
        right = _apply_mask_fp16(right, mask)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction: batched GEMM via cuBLAS (fp16 -> fp32)
    #    left : (B,N,N,hidden) → (B, hidden, N, N)
    #    right: (B,N,N,hidden) → (B, hidden, N, N) then transpose K,J
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    out_fp16 = _batched_gemm_fp16(left_perm, right_perm)  # (B, H, N, N)

    # back to (B,N,N,dim) and float32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + projection (float32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)      # broadcast over dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)               # (B,N,N,dim)

    return out
