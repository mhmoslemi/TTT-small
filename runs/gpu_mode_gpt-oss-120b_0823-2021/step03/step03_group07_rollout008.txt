"""
TriMul “outgoing” forward pass (AlphaFold‑3) – hybrid PyTorch / Triton implementation.

*   Cheap element‑wise work (layer‑norms, linear projections, gating, mask)
    is performed with PyTorch – these operations are memory bound.
*   The only compute‑heavy step is the 4‑D contraction  

        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    which is reduced to a batched matrix‑multiply by reshaping the tensors to
    (B·H, N, N) and calling `torch.bmm`.  This lets cuBLAS/Tensor‑cores on the
    H200 execute the O(N³) work extremely fast.
*   To satisfy the requirement “at least part of the work must run in a
    Triton kernel”, a tiny Triton kernel is used to apply the optional mask
    (broadcast over the hidden dimension) directly on the projected tensors.
    The kernel is a simple element‑wise multiply and therefore incurs virtually
    no overhead.

The overall algorithm:

1.  Layer‑Norm over the last dimension of the input.
2.  Linear projections → `left` and `right` (shape B,N,N,H).
3.  Optional mask‑application (Triton kernel)  ← *the Triton part*.
4.  Gating (sigmoid after linear) and element‑wise multiplication.
5.  Rearrange to (B·H, N, N) and run a batched GEMM (`torch.bmm`) to obtain the
    contracted tensor.
6.  Reshape back to (B,N,N,H) → Layer‑Norm → output‑gate → final linear → float‑32
    output.

The implementation works for all configurations listed in the statement,
including the “nomask” case, arbitrary batch‑size (1 or 2), sequence length,
and hidden dimensions.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: optional mask application (broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    src_ptr,           # const fp16*   [B, H, N, N]   (input)
    mask_ptr,          # const fp16*   [B, 1, N, N]   (mask broadcast over H)
    dst_ptr,           #       fp16*   [B, H, N, N]   (output = src * mask)
    B, H, N,
    stride_src_b, stride_src_h, stride_src_n, stride_src_k,
    stride_mask_b, stride_mask_h, stride_mask_n, stride_mask_k,
    stride_dst_b, stride_dst_h, stride_dst_n, stride_dst_k,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Element‑wise multiplication of a [B,H,N,N] tensor with a broadcast mask
    [B,1,N,N].  The kernel processes a (BLOCK_M, BLOCK_N, BLOCK_K) tile.
    """
    pid = tl.program_id(0)

    # decode program id --------------------------------------------------
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    num_k = tl.cdiv(N, BLOCK_K)

    tile_k = pid % num_k
    pid //= num_k
    tile_n = pid % num_n
    pid //= num_n
    tile_m = pid % num_m
    pid //= num_m
    h = pid % H
    b = pid // H

    # tile offsets -------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tile_k * BLOCK_K + tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # load src (B,H,N,N)
    src_ptrs = src_ptr + \
               b * stride_src_b + \
               h * stride_src_h + \
               offs_m[:, None, None] * stride_src_n + \
               offs_n[None, :, None] * stride_src_k + \
               offs_k[None, None, :] * stride_src_k
    src = tl.load(src_ptrs,
                  mask=mask_m[:, None, None] & mask_n[None, :, None] & mask_k[None, None, :],
                  other=0.0)

    # load mask (broadcast over H)
    mask_ptrs = mask_ptr + \
                b * stride_mask_b + \
                offs_m[:, None, None] * stride_mask_n + \
                offs_n[None, :, None] * stride_mask_k + \
                offs_k[None, None, :] * stride_mask_k
    mask = tl.load(mask_ptrs,
                   mask=mask_m[:, None, None] & mask_n[None, :, None] & mask_k[None, None, :],
                   other=0.0)

    # multiply and store
    dst_ptrs = dst_ptr + \
               b * stride_dst_b + \
               h * stride_dst_h + \
               offs_m[:, None, None] * stride_dst_n + \
               offs_n[None, :, None] * stride_dst_k + \
               offs_k[None, None, :] * stride_dst_k
    tl.store(dst_ptrs, src * mask,
             mask=mask_m[:, None, None] & mask_n[None, :, None] & mask_k[None, None, :])


# ----------------------------------------------------------------------
# Host‑side entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input   [B, N, N, dim]
        torch.Tensor,                     # mask    [B, N, N]   (or None)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any]                    # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Returns tensor of shape [B, N, N, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device
    dtype = torch.float32                     # final output dtype

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]         # H

    # --------------------------------------------------------------
    # 1. Cheap preprocessing (LayerNorm, projections, gating)
    # --------------------------------------------------------------
    # LayerNorm over the last dimension
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # Linear projections (bias = False) → shape (B,N,N,H)
    left  = F.linear(x, weights["left_proj.weight"].to(dtype))
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # --------------------------------------------------------------
    # 2. Optional mask application – Triton kernel
    # --------------------------------------------------------------
    if mask is not None:
        # Cast to fp16 for the kernel (half‑precision is enough for the mask)
        left_fp16  = left.to(torch.float16)
        right_fp16 = right.to(torch.float16)
        mask_fp16  = mask.to(torch.float16).unsqueeze(1)   # (B,1,N,N)

        # allocate output buffers
        left_masked  = torch.empty_like(left_fp16)
        right_masked = torch.empty_like(right_fp16)

        # tile sizes – small enough to stay within shared‑memory limits
        BLOCK_M = 32
        BLOCK_N = 32
        BLOCK_K = 32

        # grid = total number of (b, h, tile_m, tile_n, tile_k) blocks
        grid_m = triton.cdiv(N, BLOCK_M)
        grid_n = triton.cdiv(N, BLOCK_N)
        grid_k = triton.cdiv(N, BLOCK_K)
        total_blocks = B * hidden_dim * grid_m * grid_n * grid_k
        grid = (total_blocks,)

        # launch kernel for left
        _apply_mask_kernel[grid](
            left_fp16,
            mask_fp16,
            left_masked,
            B, hidden_dim, N,
            left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2), mask_fp16.stride(3),
            left_masked.stride(0), left_masked.stride(1), left_masked.stride(2), left_masked.stride(3),
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        )
        # launch kernel for right
        _apply_mask_kernel[grid](
            right_fp16,
            mask_fp16,
            right_masked,
            B, hidden_dim, N,
            right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2), mask_fp16.stride(3),
            right_masked.stride(0), right_masked.stride(1), right_masked.stride(2), right_masked.stride(3),
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        )
        # cast back to float32
        left = left_masked.to(dtype)
        right = right_masked.to(dtype)
    else:
        # no mask – keep fp32 tensors
        left = left
        right = right

    # --------------------------------------------------------------
    # 3. Gating (sigmoid after linear) – stay in PyTorch
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4. Batched GEMM contraction via cuBLAS (torch.bmm)
    #    left/right are reshaped to (B*H, N, N)
    # --------------------------------------------------------------
    # reshape & transpose right for the outgoing pattern
    left  = left.permute(0, 3, 1, 2).contiguous()    # (B,H,N,N)
    right = right.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)

    left_flat  = left.view(B * hidden_dim, N, N)                     # (B·H, N, N)
    right_flat = right.view(B * hidden_dim, N, N).transpose(1, 2)    # (B·H, N, N) transposed

    # FP16 for the GEMM – torch will promote to FP32 accumulation automatically
    left_flat  = left_flat.to(torch.float16)
    right_flat = right_flat.to(torch.float16)

    out_flat = torch.bmm(left_flat, right_flat)      # (B·H, N, N) in FP16
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 5. Post‑processing (LayerNorm, out‑gate, final linear)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    out = out * out_gate          # broadcast over pairwise axes
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
