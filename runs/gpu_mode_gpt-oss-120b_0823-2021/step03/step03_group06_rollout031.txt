"""
TriMul “outgoing” kernel (AlphaFold‑3).

The expensive contraction  

    out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]          (right is “transposed”)

is executed as a batched GEMM (B·H, N, N) inside a single Triton kernel.
All cheap operations – LayerNorms, linear projections, masking and gating –
stay in PyTorch because they are memory‑bound and inexpensive compared with the
O(N³) work.  

The Triton kernel tiles the output matrix into 2‑D blocks (BLOCK_M×BLOCK_N) and
accumulates in FP32 while the inputs are FP16 (Tensor‑Core friendly).  After
the kernel finishes the result is reshaped back to the original layout and the
remaining PyTorch post‑processing (to_out_norm, out_gate, to_out) is applied.

Key design choices (derived from the “lessons”):
* The N³ contraction is the only part moved to Triton – this gives the biggest
  speed‑up while keeping the code simple.
* Inputs are cast to FP16 before the kernel; accumulation uses FP32 to retain
  numeric quality.
* Strides are passed explicitly from the host; the kernel never assumes a fixed
  tensor rank.
* No fallback to a pure‑Python implementation – a failure is loud and easy to
  debug.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for a single (b, h) pair
# -------------------------------------------------------------------------
@triton.jit
def _trimul_batched_gemm_kernel(
    left_ptr,          # const float16*  [BH, N, N]
    right_ptr,         # const float16*  [BH, N, N]   (will be read transposed)
    out_ptr,           #       float16*  [BH, N, N]
    BH, N,             # problem sizes
    stride_lbh, stride_lbn, stride_lbk,   # left strides
    stride_rbh, stride_rbn, stride_rbk,   # right strides
    stride_obh, stride_obn, stride_obk,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute one tile (BLOCK_M × BLOCK_N) of the product

        out[bh, i, j] = Σ_k left[bh, i, k] * right[bh, j, k]

    The right operand is accessed as its transpose, i.e. (k, j).  Accumulation is
    performed in FP32 and the final result is stored as FP16.
    """
    pid = tl.program_id(0)                     # linear id over (bh, tile_m, tile_n)

    # ---------- decode linear program‑id ----------
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    tile_n = pid % num_tiles_n
    pid //= num_tiles_n
    tile_m = pid % num_tiles_m
    pid //= num_tiles_m
    bh = pid                                   # 0 ≤ bh < BH

    # ---------- tile coordinates ----------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ---------- accumulator ----------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------- loop over K ----------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[bh, i, k]   shape (M, K)
        l_ptr = left_ptr + bh * stride_lbh \
                + offs_m[:, None] * stride_lbn \
                + offs_k[None, :] * stride_lbk
        l = tl.load(l_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[bh, j, k]  we need (k, j) because multiplication uses right^T
        r_ptr = right_ptr + bh * stride_rbh \
                + offs_k[:, None] * stride_rbk \
                + offs_n[None, :] * stride_rbn
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # GEMM: (M,K) @ (K,N) → (M,N)
        acc += tl.dot(l, r)

    # ---------- write back ----------
    out_ptrs = out_ptr + bh * stride_obh \
               + offs_m[:, None] * stride_obn \
               + offs_n[None, :] * stride_obk
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper (entry point required by the grader)
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input   : [B, N, N, dim]
        torch.Tensor,                # mask    : [B, N, N]  (or None)
        Dict[str, torch.Tensor],     # weights : dict of parameters
        Dict[str, Any],              # config  : {"dim":..., "hidden_dim":...}
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: float32 tensor of shape [B, N, N, dim]
            - mask:        float32 tensor of shape [B, N, N] or None
            - weights: dict with all linear / layer‑norm parameters
            - config: {"dim": dim, "hidden_dim": hidden_dim}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # = H in the paper

    # -----------------------------------------------------------------
    # 1. LayerNorm + linear projections (still in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N, H)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 2. Optional mask (broadcasted over hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 3. Gating (sigmoid activations)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Rearrange to (B·H, N, N) and cast to FP16 for the kernel
    # -----------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left.view(B * hidden_dim, N, N).to(torch.float16)
    right_fp16 = right.view(B * hidden_dim, N, N).to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)     # (B·H, N, N), FP16

    # -----------------------------------------------------------------
    # 5. Launch Triton kernel
    # -----------------------------------------------------------------
    # Tunable tile sizes – chosen to fit comfortably in shared memory on H200.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    BH = B * hidden_dim
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    # One program per (bh, tile_m, tile_n); flatten to 1‑D grid
    total_programs = BH * grid_m * grid_n
    grid = (total_programs,)

    _trimul_batched_gemm_kernel[grid](
        # pointers
        left_fp16,
        right_fp16,
        out_fp16,
        # problem sizes
        BH,
        N,
        # strides for left (BH, N, N)
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        # strides for right (BH, N, N)
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        # strides for out (BH, N, N)
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        # compile‑time constants
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Reshape back to (B, N, N, H) and continue in PyTorch
    # -----------------------------------------------------------------
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    out = out.to(dtype)      # back to float32 for the remaining ops

    # to_out_norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )

    # apply output gate (broadcast over N,N)
    out = out * out_gate

    # final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
```

### Brief explanation of the “lessons” and what was applied

| Lesson | Relevance | Action taken |
|--------|-----------|--------------|
| **1️⃣ Limit shared memory** | Crucial for large `N` (up to 1024). | Chose modest tile sizes `64×64×32`; this keeps the per‑SM shared‑memory footprint well below the H200 limit. |
| **2️⃣ Delegate N³ to cuBLAS** | Using cuBLAS would be fast, but the task explicitly requires at least one custom Triton kernel. | Only the cubic contraction is moved to Triton; everything else stays in PyTorch (cheap). |
| **3️⃣ Match dtypes for batched GEMM** | Mixing FP16/FP32 inside the kernel would error. | Both left/right are cast to FP16 before the kernel; accumulation is FP32, then stored as FP16. |
| **4️⃣ Delegate cubic contraction to cuBLAS** | Same as (2) – we keep the kernel but follow the spirit: do the heavy work inside a single GEMM‑style kernel rather than Python loops. | Implemented a batched GEMM kernel that does exactly the needed contraction. |
| **Fail‑case: stride unpacking** | The kernel must not assume a fixed rank. | Strides are explicitly passed from the host (`tensor.stride(i)`) and used directly; no hard‑coded assumptions. |

The resulting implementation satisfies all constraints, executes the dominant O(N³) work inside a highly‑parallel Triton kernel, and returns a float‑32 tensor ready for downstream layers.
