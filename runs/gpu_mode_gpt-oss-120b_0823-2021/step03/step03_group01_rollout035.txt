"""
custom_kernel – “outgoing” TriMul forward pass (AlphaFold‑3 style).

The cubic contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is performed by a fused batched GEMM written in Triton.  All other
operations (LayerNorm, linear projections, gating and the final
LayerNorm+linear) stay in PyTorch because they are cheap compared
to the N³ work.

The kernel works on FP16 tensors (Tensor‑Core friendly) and accumulates
in FP32 for numerical stability.  The final result is cast back to FP32
so that the surrounding PyTorch code can keep the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,            # left  tensor   (Z, I, K)
    b_ptr,            # right tensor   (Z, K, J)
    c_ptr,            # output tensor  (Z, I, J)
    M, N, K,          # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ Batched GEMM kernel.
    The launch grid is (ceil(M/BM), ceil(N/BN), Z) where Z = B*H.
    """
    pid_m = tl.program_id(0)          # tile row  index
    pid_n = tl.program_id(1)          # tile col  index
    pid_z = tl.program_id(2)          # batch*head index

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑reduction
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile  (BM, BK)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile (BK, BN)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # multiply‑accumulate (FP16×FP16 → FP32)
        acc += tl.dot(a, b)

    # write back (store as FP16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for tensors with shape (B, H, I, K) and (B, H, K, J).

    Both inputs must be contiguous, dtype=torch.float16.
    Returns tensor (B, H, I, J) dtype=torch.float16.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse the first two dims → Z = B*H
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling tuned for Hopper/SM90 (Tensor‑Core friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # rows
        triton.cdiv(J, BLOCK_N),   # cols
        Z,                         # batch*head
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # SM90 likes many warps
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask         : torch.Tensor [B, N, N] (bool/float) or None
        weights      : dict of parameter tensors (all float32)
        config       : dict with keys
                       - "dim"         : input channel dimension
                       - "hidden_dim"  : intermediate hidden dimension

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                      # [B,N,N,dim] FP32

    # ------------------------------------------------------------------
    # 2. Linear projections (cast to FP16 for the heavy part)
    # ------------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16, non_blocking=True)

    left_proj_w  = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w  = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w   = to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)   # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 3. Optional mask – broadcast over hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton batched GEMM   (the N³ contraction)
    #    left : (B, N, N, hidden)  → (B, hidden, N, N)
    #    right: (B, N, N, hidden)  → (B, hidden, N, N)   (notice swap of axes 2/3)
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _triton_batched_gemm(left_perm, right_perm)   # (B, H, I, J)

    # back to original layout, cast to FP32 for the remaining ops
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7. Final gating + linear projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)             # broadcast over dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)                      # (B,N,N,dim)

    return out
