"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm (forward‑only)
------------------------
1️⃣  Layer‑norm over the last dimension (float32, done in PyTorch for
    numerical stability).
2️⃣  Five linear projections (left, right, left‑gate, right‑gate, out‑gate)
    are performed in fp16 for speed.
3️⃣  Optional binary/float mask is broadcast over the hidden dimension and
    applied to the left/right projections.
4️⃣  Sigmoid gating of the three gate tensors.
5️⃣  Core cubic contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .
    This is a batched matrix‑multiply with shape
        (B·H, N, N) · (B·H, N, N)ᵀ → (B·H, N, N)
    (H = hidden_dim).  The heavy O(N³) work is delegated to a tiny Triton
    kernel that performs a tiled GEMM in fp16 → fp32 accumulation.
6️⃣  Convert back to fp32, apply a second layer‑norm, multiply by the
    output‑gate and finish with a final linear projection to the original
    dimension.
The kernel implements only the batched GEMM (step 5); the rest is kept in
PyTorch for clarity.  The kernel works for any batch size (1‑2), any sequence
length up to 1024, and hidden dimensions 128‑384.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched GEMM (I×K)·(K×J) → (I×J) for many heads
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,                      # left  tensor   (B*H, I, K)
    b_ptr,                      # right tensor   (B*H, K, J)
    c_ptr,                      # output tensor  (B*H, I, J)
    M, N, K,                    # matrix extents
    stride_az, stride_am, stride_ak,   # a strides  (z,m,k)
    stride_bz, stride_bk, stride_bn,   # b strides  (z,k,n)
    stride_cz, stride_cm, stride_cn,   # c strides  (z,m,n)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Each program processes a tile (BLOCK_M × BLOCK_N) of the output for a
    particular (batch*head) index (pid_z).  Operates in fp16, accumulates in
    fp32, and writes back in fp16.
    """
    pid_z = tl.program_id(2)               # batch*head index
    pid_m = tl.program_id(0)               # tile row
    pid_n = tl.program_id(1)               # tile col

    # tile start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # fp32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (BLOCK_M × BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile (BLOCK_K × BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp16×fp16 → fp32 dot product
        acc += tl.dot(a, b)

    # write back in fp16
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[:, None],
    )


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM used by TriMul.
    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous
    returns (B, H, I, J)   fp16
    """
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2, "inner dimensions must match"

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse batch*head so the kernel gets a single program per pair
    batch = B * H
    a = left.view(batch, I, K)
    b = right.view(batch, K, J)
    c = out.view(batch, I, J)

    # stride in elements (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling tuned for Hopper (SM‑90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),   # rows
        triton.cdiv(J, BLOCK_N),   # cols
        batch,                     # one program per (B,H)
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # fp16 friendly on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim]   (float32)
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of leaf tensors required by the module
        config       : dict with keys
                       * "dim"        – input channel dimension
                       * "hidden_dim" – projection dimension

    Returns
    -------
    torch.Tensor
        shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (fp32, manual)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + 1e-5)
    x_norm = x_norm * ln_w + ln_b                     # (B,N,N,dim) fp32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections in fp16
    # --------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])

    x_fp16 = x_norm.to(torch.float16)                # (B,N,N,dim)

    left  = F.linear(x_fp16, left_proj_w)           # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_w)          # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)    # (B,N,N,1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy cubic contraction via Triton GEMM
    #   left  : (B,N,N,hidden) → (B, hidden, N, N)
    #   right : (B,N,N,hidden) → (B, hidden, N, N) and then transpose K,J
    # --------------------------------------------------------------
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # batched GEMM, fp16 → fp16 (accumulated in fp32 inside the kernel)
    out_fp16 = _batched_gemm_fp16(left_t, right_t)   # (B, H, I, J)

    # back to (B,N,N,dim) and fp32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (fp32, manual)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection (fp32)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)           # broadcast over dim
    out = F.linear(out, weights["to_out.weight"])    # (B,N,N,dim) fp32

    return out
