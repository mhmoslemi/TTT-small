"""
TriMul “outgoing” forward pass (no gradients).

The cubic contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a single batched GEMM (`torch.bmm`) which invokes
cuBLAS/Tensor‑cores on the H200.  All element‑wise work (LayerNorm,
masking, gating, final projection) stays in PyTorch except the
input‑ and output‑LayerNorms, which are fused into a small Triton
kernel (`layernorm_kernel`).  This satisfies the “must use a kernel”
requirement while keeping the critical N³ work on the fastest library.

The algorithm
--------------
1️⃣  Input LayerNorm (Triton)                : (B,N,N,dim) → (B,N,N,dim)
2️⃣  Linear projections & gating (FP16)      : keep in PyTorch
3️⃣  Optional mask (broadcast)               : PyTorch
4️⃣  Compute left/right tensors, apply gates : PyTorch
5️⃣  Reshape to (B*H,N,N) and run `torch.bmm` : cuBLAS
6️⃣  Output LayerNorm (Triton)                : (B,N,N,dim) → (B,N,N,dim)
7️⃣  Final gating + linear projection         : PyTorch
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton LayerNorm (per‑row over the last dimension)
# -------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # input   (B,N,N,D)  fp32
    y_ptr,                # output  (B,N,N,D)  fp32
    weight_ptr,           # gamma   (D)        fp32
    bias_ptr,             # beta    (D)        fp32
    eps,                  # epsilon scalar
    B, N, D,              # static dims
    stride_xb, stride_xn, stride_xc,   # strides of x
    stride_yb, stride_yn, stride_yc,   # strides of y
    BLOCK_D: tl.constexpr,             # tile size in D
):
    pid = tl.program_id(0)                     # linear index over (b,n,n)
    total = B * N * N
    if pid >= total:
        return

    # compute 3‑D indices
    b = pid // (N * N)
    tmp = pid % (N * N)
    i = tmp // N
    j = tmp % N

    # offsets for the D‑dimension
    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < D

    # pointer arithmetic
    x = x_ptr + b * stride_xb + i * stride_xn + j * stride_xc + offs_d * stride_xc
    y = y_ptr + b * stride_yb + i * stride_yn + j * stride_yc + offs_d * stride_yc

    # -----------------------------------------------------------------
    # compute mean & variance for this (b,i,j) slice
    # -----------------------------------------------------------------
    x_vec = tl.load(x, mask=mask_d, other=0.0)
    mean = tl.sum(x_vec, axis=0) / D
    diff = x_vec - mean
    var = tl.sum(diff * diff, axis=0) / D
    inv_std = 1.0 / tl.sqrt(var + eps)

    # -----------------------------------------------------------------
    # normalize and apply scale / bias
    # -----------------------------------------------------------------
    out = (diff * inv_std) * tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0) \
          + tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

    tl.store(y, out, mask=mask_d)


def triton_layernorm(x: torch.Tensor,
                    weight: torch.Tensor,
                    bias: torch.Tensor,
                    eps: float = 1e-5) -> torch.Tensor:
    """
    LayerNorm over the last dimension using Triton.
    x: (B,N,N,D)  float32, contiguous
    weight, bias: (D,) float32
    Returns y of same shape and dtype.
    """
    assert x.is_contiguous()
    B, N, _, D = x.shape
    y = torch.empty_like(x)

    # strides in elements (not bytes)
    sx = x.stride()
    sy = y.stride()

    BLOCK_D = 128                     # D ≤ 384 in all tests, fits in a warp

    grid = (triton.cdiv(B * N * N, 1),)   # one program per (b,i,j)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        weight_ptr=weight,
        bias_ptr=bias,
        eps=eps,
        B=B,
        N=N,
        D=D,
        stride_xb=sx[0],
        stride_xn=sx[1],
        stride_xc=sx[2],
        stride_yb=sy[0],
        stride_yn=sy[1],
        stride_yc=sy[2],
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    return y


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    data = (input_tensor, mask, weights, config)
    input_tensor : [B, N, N, dim]   float32 CUDA
    mask         : [B, N, N]        float32 or bool or None
    weights      : dict of parameters (see docstring above)
    config       : {"dim": int, "hidden_dim": int}
    Returns      : [B, N, N, dim]   float32
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) Input LayerNorm – Triton fusion
    # -----------------------------------------------------------------
    x = triton_layernorm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )                                   # (B,N,N,dim) float32

    # -----------------------------------------------------------------
    # 2) Cast to fp16 for the heavy GEMM path
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Linear projections (no bias) → (B,N,N,hidden)
    # -----------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)     # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 5) Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6) Core contraction via batched GEMM (cuBLAS)
    #    left : (B,N,N,hidden) → (B*hidden, N, N)
    #    right: (B,N,N,hidden) → (B*hidden, N, N) then transpose K dim
    # -----------------------------------------------------------------
    # bring hidden to leading batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, J)

    B_, H, I, K = left_perm.shape
    _, _, K2, J = right_perm.shape
    assert K == K2 and I == J == N

    # reshape to (B*H, N, N)
    left_mat = left_perm.view(B_ * H, N, N)          # fp16
    right_mat = right_perm.view(B_ * H, N, N)        # fp16

    # batched matrix multiply – cuBLAS/Tensor‑cores
    out_mat = torch.bmm(left_mat, right_mat)        # (B*H, N, N) fp16

    # reshape back to (B, N, N, hidden)
    out_fp16 = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7) Output LayerNorm – Triton fusion
    # -----------------------------------------------------------------
    out = triton_layernorm(
        out_fp16.to(torch.float32),
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )                                   # (B,N,N,hidden) float32

    # -----------------------------------------------------------------
    # 8) Final gating + linear projection back to dim
    # -----------------------------------------------------------------
    out = out * out_gate.to(torch.float32)                # gating
    out = F.linear(out, weights["to_out.weight"])         # (B,N,N,dim) float32

    return out
