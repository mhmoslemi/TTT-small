"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  LayerNorm on the last dimension (FP32, done in PyTorch – cheap & avoids a
    custom reduction kernel).
2️⃣  Four linear projections (left, right, three gates) are evaluated in FP16
    for speed.  The input is cast once to FP16.
3️⃣  Optional pair‑wise mask is applied to the left/right tensors (FP16).
4️⃣  Sigmoid‑gates are applied (still FP16).
5️⃣  Core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    – This is a batched matrix‑multiply over the *k* dimension.  We implement a
      small Triton kernel that treats each (batch, head) pair as an
      independent GEMM  (N×K)·(K×N) → (N×N).  The kernel works in FP16,
      accumulates in FP32 and writes FP16.
6️⃣  Convert back to FP32, apply a second LayerNorm, multiply by the output‑gate
    and finish with a linear projection to the original dimension.
The kernel is tuned for Hopper/SM‑90:
    BLOCK_M = BLOCK_N = 64   # tile size in the i/j dimensions
    BLOCK_K = 32             # tile size in the reduction (k) dimension
    4 warps, fp16→fp32 accumulation
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched GEMM (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr,            # left  tensor  (B*H, I, K)
    b_ptr,            # right tensor  (B*H, K, J)
    c_ptr,            # output tensor (B*H, I, J)
    M, N, K,          # extents
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM used by TriMul.
    One program (pid_z) processes a whole (batch, head) pair.
    Within each program the tile grid (pid_m, pid_n) covers the output
    I‑J matrix.
    """
    pid_z = tl.program_id(2)          # index over (batch * hidden_dim)
    pid_m = tl.program_id(0)          # tile index along I
    pid_n = tl.program_id(1)          # tile index along J

    # tile start coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (to keep precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension K in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load a tile (M×K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load b tile (K×N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16 dot → FP32 accumulation
        acc += tl.dot(a, b)

    # write result (convert back to FP16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM used by the outgoing TriMul.
    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous
    returns: (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # collapse the (B,H) dimensions – the kernel expects a 3‑D tensor
    a = left.view(B * H, I, K)
    b = right.view(B * H, K, J)
    c = out.view(B * H, I, J)

    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tuned tile sizes for Hopper (fp16)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),          # tiles along I
        triton.cdiv(J, BLOCK_N),          # tiles along J
        B * H,                            # one program per (batch,head)
    )
    _batch_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                     # good default for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        mask         : torch.Tensor or None, shape [B, N, N] (bool or float)
        weights      : dict of leaf weight tensors (see reference implementation)
        config       : dict containing keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor with shape [B, N, N, dim], dtype torch.float32
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]          # called `hidden_dim` in the paper

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32, manual – cheap)
    # --------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    # mean / var over the last dimension
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + 1e-5)
    x = x * ln_w + ln_b                     # still FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections & gates (cast to FP16)
    # --------------------------------------------------------------
    # helper to cast weight matrices once
    def _to_fp16(t): return t.to(torch.float16)

    left_proj_w   = _to_fp16(weights["left_proj.weight"])    # (hidden_dim, dim)
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])

    x_fp16 = x.to(torch.float16)      # (B, N, N, dim)

    # linear layers without bias (as in the reference)
    left  = F.linear(x_fp16, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x_fp16, right_proj_w)  # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 3️⃣ Masking (optional)
    # --------------------------------------------------------------
    if mask is not None:
        # mask shape (B, N, N) -> (B, N, N, 1)
        m = mask.to(torch.float16).unsqueeze(-1)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – still FP16
    # --------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()   # (B, N, N, hidden_dim)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core contraction via Triton batched GEMM
    # --------------------------------------------------------------
    #   left  : (B, N, N, H)  -> (B, H, N, N)   (I=N, K=N)
    #   right : (B, N, N, H)  -> (B, H, N, N)   we need (K=N, J=N) with J = N
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    # right needs to be transposed on the last two dims to match the
    # contraction formula: left[..., i, k, d] * right[..., j, k, d]
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # Batched GEMM in fp16 using the Triton kernel above.
    out_fp16 = _triton_batched_gemm(left_t, right_t)  # (B, H, I, J)

    # reshape back to (B, N, N, hidden_dim) and cast to FP32 for the remaining ops
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (FP32, manual)
    # --------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * out_ln_w + out_ln_b

    # --------------------------------------------------------------
    # 7️⃣ Final gating + projection (FP32)
    # --------------------------------------------------------------
    # out_gate was computed in FP16; cast it to FP32 before broadcasting.
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim)

    return out
