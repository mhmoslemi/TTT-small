"""
TriMul “outgoing” kernel (AlphaFold‑3) – fast forward pass.

The heavy O(N³·H) reduction is performed with a fused
batched‑matrix‑multiply (B×H independent GEMMs) that is
implemented in Triton.  The rest of the pipeline
(LayerNorm, Linear projections, gating, final LayerNorm
and output projection) stays in PyTorch – these are
memory‑bound and benefit from the highly‑optimised
torch.nn.functional.linear implementation.

The kernel works for both masked and un‑masked inputs.
Masking is applied inside the same Triton kernel so that we
avoid an extra memory pass.
All arithmetic is carried out in FP16 inside the kernel and
cast back to FP32 for the final output (the model weights are
FP32, therefore we convert them to FP16 before the launch).

Supported shapes:
    B ∈ {1,2}, N ∈ {128,256,512,768,1024},
    dim ∈ {128,384}, hidden_dim (=H) ∈ {128,384,768}
The kernel is written for sm_90 (H200) and tuned for
BLOCK_N = 64, BLOCK_K = 32, BLOCK_M = 64 – values that give
good occupancy for the above ranges.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with optional mask
# ----------------------------------------------------------------------
@triton.jit
def outgoing_trimul_kernel(
    # pointers
    left_ptr,          # *fp16  [B, N, N, H]
    right_ptr,         # *fp16  [B, N, N, H]
    mask_ptr,          # *fp16  [B, N, N]  (0/1) – may be all‑ones
    out_ptr,           # *fp16  [B, N, N, H]

    # dimensions
    B, N, H,           # i64
    stride_left_bn, stride_left_nk, stride_left_kh,
    stride_right_bn, stride_right_nk, stride_right_kh,
    stride_out_bn,   stride_out_nj,   stride_out_jh,
    stride_mask_bn,  stride_mask_nn,

    # compile‑time constants
    BLOCK_M: tl.constexpr,   # i‑dimension (i)
    BLOCK_N: tl.constexpr,   # j‑dimension (j)
    BLOCK_K: tl.constexpr,   # reduction k
    USE_MASK: tl.constexpr,  # 0/1 compile‑time flag
):
    """
    Compute for each batch b and head h:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h] * mask
    where mask = mask[b, i, k] if USE_MASK else 1.
    """
    pid = tl.program_id(0)                 # (b, h) flattened
    num_h = tl.cdiv(H, BLOCK_M) * B        # total programs per head‑batch

    # decode batch and head from pid
    b = pid // (tl.cdiv(H, BLOCK_M))
    pid_h = pid % (tl.cdiv(H, BLOCK_M))
    h_offset = pid_h * BLOCK_M

    # block offsets
    offs_i = tl.arange(0, BLOCK_M)
    offs_j = tl.arange(0, BLOCK_N)
    i = h_offset + offs_i
    j = tl.program_id(1) * BLOCK_N + offs_j   # second grid dim -> j‑blocks

    # mask for out‑of‑bounds
    mask_i = i < N
    mask_j = j < N

    # pointers for the output tile
    out_ptrs = out_ptr + \
        b * stride_out_bn + \
        i[:, None] * stride_out_nj + \
        j[None, :] * stride_out_jh + \
        h_offset

    # initialise accumulator in fp32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over reduction axis k
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # load left block  [i, k, h]
        left = tl.load(
            left_ptr + b * stride_left_bn
                       + i[:, None] * stride_left_nk
                       + k_offsets[None, :] * stride_left_kh
                       + h_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # load right block [j, k, h] – note the transposition on k
        right = tl.load(
            right_ptr + b * stride_right_bn
                        + j[:, None] * stride_right_nk
                        + k_offsets[None, :] * stride_right_kh
                        + h_offset,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0
        )
        # optional mask (broadcast over h)
        if USE_MASK:
            m = tl.load(
                mask_ptr + b * stride_mask_bn
                           + i[:, None] * stride_mask_nn
                           + k_offsets[None, :],
                mask=mask_i[:, None] & mask_k[None, :],
                other=1.0
            )
            left = left * m

        # multiply‑accumulate in fp32
        acc += tl.dot(left, right, acc)
    # ------------------------------------------------------------------

    # write back (still fp16)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel for a full tensor
# ----------------------------------------------------------------------
def triton_outgoing_trimul(left: torch.Tensor,
                           right: torch.Tensor,
                           mask: torch.Tensor,
                           use_mask: bool) -> torch.Tensor:
    """
    left, right:  [B, N, N, H]  fp16
    mask:        [B, N, N]     fp16 (0/1)
    returns      out:         [B, N, N, H]  fp16
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()
    stride_mask = mask.stride()

    # tile sizes – tuned for Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * tl.cdiv(H, BLOCK_M), tl.cdiv(N, BLOCK_N))

    left_ptr = left.data_ptr()
    right_ptr = right.data_ptr()
    mask_ptr = mask.data_ptr()
    out_ptr = out.data_ptr()

    outgoing_trimul_kernel[grid](
        left_ptr, right_ptr, mask_ptr, out_ptr,
        B, N, H,
        stride_left[0], stride_left[2], stride_left[3],
        stride_right[0], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[3],
        stride_mask[0], stride_mask[1],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        USE_MASK=1 if use_mask else 0,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Main entry‑point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ----------
    data : (input, mask, weights, config)
        input  : [B, N, N, dim]            torch.float32
        mask   : [B, N, N]                 torch.float32 (0/1) – may be all‑ones
        weights: dict of model parameters (all fp32 tensors)
        config : {'dim': int, 'hidden_dim': int, 'nomask': bool}

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = x.device
    # ------------------------------------------------------------------
    # LayerNorm (manual – avoids extra nn.Module overhead)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + 1e-5)
    x_norm = x_norm * norm_w + norm_b

    # ------------------------------------------------------------------
    # Linear projections (fp16 for the heavy path)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"]).to(torch.float16)
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # Mask handling – broadcast mask to [...,1] and cast to fp16
    # ------------------------------------------------------------------
    if nomask:
        mask_fp16 = torch.ones_like(mask, dtype=torch.float16, device=device)
        use_mask = False
    else:
        mask_fp16 = mask.to(torch.float16)
        use_mask = True

    # ------------------------------------------------------------------
    # Gates (still in fp32 for stability)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate.to(torch.float16)
    right = right * right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # Core O(N³·H) reduction – Triton fused GEMM with mask
    # ------------------------------------------------------------------
    out = triton_outgoing_trimul(left, right, mask_fp16, use_mask)   # fp16

    # ------------------------------------------------------------------
    # Final LayerNorm, out‑gate and output projection (back to fp32)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
