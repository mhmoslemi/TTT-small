"""
TriMul “outgoing” kernel (AlphaFold‑3) – fast Triton‑accelerated forward pass.

*   A custom Triton kernel performs the per‑vector LayerNorm that appears
    twice in the module (the input‑norm and the output‑norm).  The kernel
    computes the mean / variance **per‑last‑dimension** using block‑wise
    reductions, applies the learned scale / bias and writes the result
    back in FP32.

*   A second tiny Triton kernel applies the optional pair‑wise mask
    (shape [B,N,N]) to the projected tensors (shape [B,N,N,H]) – this
    operation is memory‑bound and benefits from Triton’s explicit masking.

*   The expensive O(N³·H) reduction is executed with ``torch.einsum`` which,
    on Hopper, is internally lowered to a series of GEMM calls and is
    already close to peak performance.

The implementation keeps the rest of the pipeline in PyTorch for numerical
stability (gating, linear projections and the final output projection) while
still satisfying the “at least part of the work in Triton” rule.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# Triton kernel: per‑vector LayerNorm (no bias term in the mean/var
# computation – we only apply the learned *weight* and *bias* after
# normalisation).  Works for any leading shape, normalises over the
# last dimension C.
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    inp_ptr,          # const float*  (input  [*,C])
    out_ptr,          # float*        (output [*,C])
    weight_ptr,       # const float*  (LayerNorm weight, length C)
    bias_ptr,         # const float*  (LayerNorm bias,   length C)
    N,                # i64           total number of vectors (product of leading dims)
    C,                # i64           channel dimension
    eps,              # tl.constexpr float
    BLOCK_C: tl.constexpr,   # compile‑time tile size for C
):
    pid = tl.program_id(0)                         # one program per vector
    if pid >= N:
        return

    # offsets for this vector
    offs_c = tl.arange(0, BLOCK_C)                 # [0, BLOCK_C)
    mask_c = offs_c < C

    # pointer to the start of this vector
    inp = tl.load(inp_ptr + pid * C + offs_c, mask=mask_c, other=0.0)
    # ---------- mean ----------
    # reduction in float32 to keep precision
    sum_ = tl.sum(inp, axis=0)                     # scalar
    mean = sum_ * (1.0 / C)

    # ---------- variance ----------
    diff = inp - mean
    var = tl.sum(diff * diff, axis=0) * (1.0 / C)

    # normalise
    inv_std = 1.0 / tl.sqrt(var + eps)
    norm = (inp - mean) * inv_std

    # apply weight / bias
    w = tl.load(weight_ptr + offs_c, mask=mask_c, other=1.0)
    b = tl.load(bias_ptr   + offs_c, mask=mask_c, other=0.0)
    out = norm * w + b

    tl.store(out_ptr + pid * C + offs_c, out, mask=mask_c)


def layernorm_triton(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Layer‑norm over the **last** dimension of ``x`` using the supplied
    ``weight`` and ``bias`` (both 1‑D tensors of size C).

    Returns a new tensor with the same shape as ``x`` (dtype FP32).
    """
    assert x.is_cuda and weight.is_cuda and bias.is_cuda
    B, *lead, C = x.shape
    N = int(torch.prod(torch.tensor(lead)))        # number of vectors to normalise
    out = torch.empty_like(x)

    # Choose a block size that divides C nicely; 128 works well for C≤768.
    BLOCK_C = 128
    grid = (N,)

    layernorm_kernel[grid](
        x.view(-1).contiguous(),
        out.view(-1).contiguous(),
        weight,
        bias,
        N,
        C,
        eps,
        BLOCK_C=BLOCK_C,
    )
    return out


# --------------------------------------------------------------
# Triton kernel: apply a broadcast mask [B,N,N] to a tensor
#               [B,N,N,H] → mask[...,None] * tensor
# --------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    src_ptr,      # const float*   (input  [B,N,N,H])
    mask_ptr,     # const float*   (mask   [B,N,N])
    dst_ptr,      # float*         (output [B,N,N,H])
    total_el,     # i64            total elements = B*N*N*H
    H,            # i64            hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_el

    # read input
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # each H‑slice shares the same mask element → divide index by H
    m = tl.load(mask_ptr + (offs // H), mask=mask, other=1.0)

    tl.store(dst_ptr + offs, src * m, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    ``tensor`` : [B,N,N,H]   (float32)
    ``mask``   : [B,N,N]    (float32, 0/1)

    Returns ``tensor * mask[...,None]`` computed in Triton.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024

    src = tensor.contiguous().view(-1)
    dst = torch.empty_like(src)
    mask_f = mask.contiguous().view(-1)          # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)
    apply_mask_kernel[grid](
        src,
        mask_f,
        dst,
        total,
        H,
        BLOCK=BLOCK,
    )
    return dst.view(B, N, N, H)


# --------------------------------------------------------------
# Entrypoint – custom kernel expected by the grader
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : tuple (input, mask, weights, config)
        input   – [B, N, N, dim]   (float32, on CUDA)
        mask    – [B, N, N]        (float32, 0/1) – may be all‑ones when ``nomask=True``
        weights – dict of pretrained parameters (see docstring above)
        config  – dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Input LayerNorm (Triton)
    # ------------------------------------------------------------------
    x = layernorm_triton(
        x,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias) – kept in PyTorch for stability
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask – Triton kernel (memory‑bound)
    # ------------------------------------------------------------------
    if not nomask:
        left  = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # Gating (sigmoid) – PyTorch
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·H) reduction – torch.einsum leverages cuBLAS GEMM on Hopper
    # ------------------------------------------------------------------
    # result shape: [B, N, N, hidden_dim]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # Output LayerNorm + gating + final projection – Triton + PyTorch
    # ------------------------------------------------------------------
    out = layernorm_triton(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=1e-5,
    )

    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
