"""
TriMul “outgoing” kernel (AlphaFold‑3) – fast forward pass.

The O(N³·C) reduction is performed with a custom Triton kernel that
computes, for every batch *b*, output channel *h* and pair (i,j),

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

The kernel tiles the *k* dimension (the only large reduction dimension)
and uses `tl.dot` on 16‑element vectors, which lets the Hopper tensor
cores execute the multiply‑accumulate in FP16 while the accumulation is
done in FP32 for numerical stability.  All surrounding LayerNorm,
Linear projections and gating are kept in PyTorch – they are cheap
relative to the N³ work and benefit from the highly‑optimised cuBLAS /
torch kernels.

A tiny Triton kernel is also used to apply the optional pair‑wise mask
(in‑place) because masking is purely memory‑bound and illustrates the
required “some work in Triton”.  When ``nomask=True`` the mask kernel is
skipped.

The implementation supports the full range of test shapes (B∈{1,2},
N∈{128,…,1024}, dim∈{128,384}, hidden_dim=128) and runs entirely in
float32 (the final output) while using mixed‑precision internally.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# 1️⃣ Helper: apply a broadcast mask with Triton (optional)
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*  (input tensor B,N,N,H)
    mask_ptr,         # const float*  (mask tensor B,N,N, broadcasted)
    dst_ptr,          # float*        (output tensor)
    B, N, H,          # i32 scalars
    total_el: tl.constexpr,  # total number of elements (B*N*N*H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_el

    # load the data element
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # each hidden‑dim slice shares the same mask element;
    # the mask index is (b * N * N + i * N + j)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(dst_ptr + offs, src * m, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    tensor : [B, N, N, H]   (float32)
    mask   : [B, N, N]      (float32, 0/1)
    returns tensor * mask[...,None] (float32)
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024

    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_f = mask.view(-1)                # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    mask_apply_kernel[grid](
        src,
        mask_f,
        dst,
        B,
        N,
        H,
        total,
        BLOCK=BLOCK,
    )
    return dst.view(B, N, N, H)


# ----------------------------------------------------------------------
# 2️⃣ Core O(N³·C) reduction – Triton implementation
# ----------------------------------------------------------------------
# The kernel works on a 4‑D grid:
#   pid0 -> batch    (0 … B-1)
#   pid1 -> out_chan (0 … H-1)
#   pid2 -> i block  (0 … ceil(N/BLOCK_M)-1)
#   pid3 -> j block  (0 … ceil(N/BLOCK_N)-1)
#
# Within each (i,j) tile we iterate over k in BLOCK_K sized chunks and
# accumulate with tl.dot on 16‑element vectors (FP16).  Accumulation is
# FP32 for stability.
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const float*  [B, N, N, H]
    right_ptr,         # const float*  [B, N, N, H]
    out_ptr,           # float*        [B, N, N, H]
    B, N, H,           # i32
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,   # number of channels processed per program (usually 1)
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_i = tl.program_id(2)
    pid_j = tl.program_id(3)

    # block start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    h0 = pid_h * BLOCK_H

    # bounds
    i_mask = i0 + tl.arange(0, BLOCK_M) < N
    j_mask = j0 + tl.arange(0, BLOCK_N) < N
    h_mask = h0 + tl.arange(0, BLOCK_H) < H

    # initialise accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # iterate over k
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.arange(0, BLOCK_K)
        k_valid = k_start + cur_k < N

        # ------------------------------
        # load left[b, i, k, h]
        # shape (BLOCK_M, BLOCK_K, BLOCK_H)
        # ------------------------------
        left_i = i0 + tl.arange(0, BLOCK_M)
        left_k = k_start + cur_k
        left_h = h0 + tl.arange(0, BLOCK_H)

        left_ptrs = (
            left_ptr
            + pid_b * stride_left_b
            + left_i[:, None, None] * stride_left_i
            + left_k[None, :, None] * stride_left_k
            + left_h[None, None, :] * stride_left_h
        )
        left = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_valid[None, :] & h_mask[None, None, :], other=0.0)

        # ------------------------------
        # load right[b, j, k, h]
        # shape (BLOCK_N, BLOCK_K, BLOCK_H)
        # ------------------------------
        right_j = j0 + tl.arange(0, BLOCK_N)
        right_k = k_start + cur_k
        right_h = h0 + tl.arange(0, BLOCK_H)

        right_ptrs = (
            right_ptr
            + pid_b * stride_right_b
            + right_j[:, None, None] * stride_right_i   # note: right uses i‑stride for the “j” index
            + right_k[None, :, None] * stride_right_k
            + right_h[None, None, :] * stride_right_h
        )
        # right needs to be accessed as (j,k,h) but the reduction uses
        # right[j, k, h] where k is the same as left's k.
        right = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_valid[None, :] & h_mask[None, None, :], other=0.0)

        # ------------------------------
        # dot product over k (BLOCK_K) for every (i,j,h)
        # We broadcast the k dimension and perform a dot on 16‑elem
        # vectors.  Since BLOCK_K is not necessarily a multiple of 16 we
        # let tl.dot handle the tail safely (masking above already zero‑ed
        # out‑of‑range entries).
        # ------------------------------
        # reshape to (M, K/16, 16) etc.  Using tl.dot on the innermost 16.
        # For simplicity we let tl.dot handle the whole vector.
        for kk in range(0, BLOCK_K, 16):
            k_slice = tl.arange(0, 16) + kk
            k_valid_slice = k_slice < BLOCK_K
            # load 16‑wide slices
            l = tl.load(
                left_ptrs + k_slice[None, None, :] * stride_left_k,
                mask=i_mask[:, None, None] & k_valid_slice[None, None, :] & h_mask[None, None, :],
                other=0.0,
            )
            r = tl.load(
                right_ptrs + k_slice[None, None, :] * stride_right_k,
                mask=j_mask[:, None, None] & k_valid_slice[None, None, :] & h_mask[None, None, :],
                other=0.0,
            )
            # convert to fp16 for the dot (tensorcores)
            l_fp16 = l.to(tl.float16)
            r_fp16 = r.to(tl.float16)
            # dot over the 16‑element vector
            acc += tl.dot(l_fp16, r_fp16).to(tl.float32)

    # store result
    out_i = i0 + tl.arange(0, BLOCK_M)
    out_j = j0 + tl.arange(0, BLOCK_N)
    out_h = h0 + tl.arange(0, BLOCK_H)

    out_ptrs = (
        out_ptr
        + pid_b * stride_out_b
        + out_i[:, None, None] * stride_out_i
        + out_j[None, :, None] * stride_out_j
        + out_h[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float32),
        mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
    )


# ----------------------------------------------------------------------
# 3️⃣ Top‑level entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : (input, mask, weights, config)
        input  : [B, N, N, dim]                float32
        mask   : [B, N, N]                     float32 (0/1) – may be all‑ones
        weights: dict of pretrained parameters (see code)
        config : dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (manual – avoids extra PyTorch op)
    # --------------------------------------------------------------
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # mean / var over last dim
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias                     # shape [B,N,N,dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑less)
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask – Triton path (memory‑bound)
    # --------------------------------------------------------------
    if not nomask:
        left = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·C) reduction – Triton kernel
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    H = hidden_dim

    # allocate output buffer (FP32)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # strides (in elements, not bytes)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # launch configuration (tuned for Hopper)
    BLOCK_M = 64    # i tile
    BLOCK_N = 64    # j tile
    BLOCK_K = 32    # reduction tile
    BLOCK_H = 1     # we process one channel per program (grid over H)

    grid = (
        B,                         # batch dimension
        (H + BLOCK_H - 1) // BLOCK_H,   # channel dimension
        (N + BLOCK_M - 1) // BLOCK_M,   # i tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j tiles
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # --------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gating and output projection (PyTorch)
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    mean_out = out.mean(-1, keepdim=True)
    var_out = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_out) / torch.sqrt(var_out + eps)
    out = out * to_out_norm_weight + to_out_norm_bias

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
