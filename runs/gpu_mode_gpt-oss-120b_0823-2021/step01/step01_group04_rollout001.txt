"""
TriMul “outgoing” forward pass (AlphaFold‑3).

The heavy N³ part is a set of independent matrix‑multiplications

    out_h[b,i,j] = Σ_k left_h[b,i,k] * right_h[b,j,k]

which can be written as   out_h = left_h @ right_hᵀ
for every hidden channel *h* and batch *b*.  By reshaping the tensors we can
use a single torch.bmm (cuBLAS) – this is much faster than an explicit
einsum on H200.

What remains expensive are the point‑wise operations that touch the full
4‑D tensor (layer‑norm, linear projections, gating, mask).  These are fused in a
tiny Triton kernel (`tri_mul_fuse_kernel`).  The kernel

* loads a block of elements from *left*, *right*, the mask and the two gates,
* applies  mask × gate to each tensor,
* writes the results back in‑place.

Only the N³ multiplication is delegated to PyTorch; all other work is kept
on‑device and runs in fp32 (the model weights are fp32).  The final
normalisation, output‑gate and linear projection are also performed in
PyTorch – they are cheap compared with the N³ work.

The implementation works for any batch size (≤2 in the tests), any sequence
length (128 – 1024), any model dimension *dim* (128 or 384) and hidden dimension
*hidden_dim* (always 128).  A mask may be omitted (`nomask=True`) – in that
case a tensor of ones is used.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel – fuse mask, left‑gate and right‑gate into the projected
# tensors.  The kernel works completely in‑place.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_fuse_kernel(
    left_ptr,          # [B, N, N, H]   fp32
    right_ptr,         # [B, N, N, H]   fp32
    mask_ptr,          # [B, N, N, 1]   fp32  (broadcasted mask)
    left_gate_ptr,     # [B, N, N, H]   fp32
    right_gate_ptr,    # [B, N, N, H]   fp32
    B, N, H,           # static sizes (int32)
    # strides for the 4‑D tensors (in elements, not bytes)
    stride_left_b, stride_left_n1, stride_left_n2, stride_left_h,
    stride_right_b, stride_right_n1, stride_right_n2, stride_right_h,
    stride_mask_b, stride_mask_n1, stride_mask_n2,
    stride_lgate_b, stride_lgate_n1, stride_lgate_n2, stride_lgate_h,
    stride_rgate_b, stride_rgate_n1, stride_rgate_n2, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # linear program id
    total = B * N * N * H                       # total number of scalar elements
    offs = pid * BLOCK + tl.arange(0, BLOCK)    # linear offsets this thread will touch
    mask = offs < total

    # ----- decode linear offset -----
    # layout: ((b * N + i) * N + k) * H + h
    b  = offs // (N * N * H)
    rest = offs % (N * N * H)
    i  = rest // (N * H)
    rest = rest % (N * H)
    k  = rest // H
    h  = rest % H

    # ----- pointer arithmetic (in elements) -----
    left_off = (b * stride_left_b +
                i * stride_left_n1 +
                k * stride_left_n2 +
                h * stride_left_h)

    right_off = (b * stride_right_b +
                 i * stride_right_n1 +      # note: same i as for left (outgoing version)
                 k * stride_right_n2 +
                 h * stride_right_h)

    mask_off = (b * stride_mask_b +
                i * stride_mask_n1 +
                k * stride_mask_n2)          # no channel dim

    lgate_off = (b * stride_lgate_b +
                 i * stride_lgate_n1 +
                 k * stride_lgate_n2 +
                 h * stride_lgate_h)

    rgate_off = (b * stride_rgate_b +
                 i * stride_rgate_n1 +
                 k * stride_rgate_n2 +
                 h * stride_rgate_h)

    # ----- load, fuse and store back -----
    left  = tl.load(left_ptr + left_off,   mask=mask, other=0.0)
    right = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    m     = tl.load(mask_ptr + mask_off,   mask=mask, other=1.0)   # default 1 if no mask
    lgate = tl.load(left_gate_ptr + lgate_off,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + rgate_off, mask=mask, other=0.0)

    left  = left  * m * lgate
    right = right * m * rgate

    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left, right, mask, left_gate, right_gate):
    """
    In‑place fusion of mask and per‑position gates using the Triton kernel.
    All tensors are assumed contiguous and on the same CUDA device.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                                   # 1 k threads – good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    # strides in *elements* (not bytes)
    sL = left.stride()
    sR = right.stride()
    sM = mask.stride()
    sLG = left_gate.stride()
    sRG = right_gate.stride()

    tri_mul_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B,
        N=N,
        H=H,
        stride_left_b=sL[0], stride_left_n1=sL[1], stride_left_n2=sL[2], stride_left_h=sL[3],
        stride_right_b=sR[0], stride_right_n1=sR[1], stride_right_n2=sR[2], stride_right_h=sR[3],
        stride_mask_b=sM[0], stride_mask_n1=sM[1], stride_mask_n2=sM[2],
        stride_lgate_b=sLG[0], stride_lgate_n1=sLG[1], stride_lgate_n2=sLG[2], stride_lgate_h=sLG[3],
        stride_rgate_b=sRG[0], stride_rgate_n1=sRG[1], stride_rgate_n2=sRG[2], stride_rgate_h=sRG[3],
        BLOCK=BLOCK,
    )
    return left, right


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

        * input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        * mask         : torch.Tensor or None, shape [B, N, N] (float32)
        * weights_dict : dict of torch.Tensors – exactly the names used in the
          reference PyTorch implementation
        * config_dict  : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (fp32)
    # ------------------------------------------------------------------
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * w_norm + b_norm                       # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gates (sigmoid) – still in FP32
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4️⃣ Mask handling – convert to broadcastable shape
    # ------------------------------------------------------------------
    if mask is None:
        # “nomask” case – all‑ones mask, same dtype/device as the tensors
        mask = torch.ones(x.shape[:-1], dtype=x.dtype, device=x.device)
    mask = mask.unsqueeze(-1)          # (B,N,N,1)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask & gates (Triton) – in‑place on left/right
    # ------------------------------------------------------------------
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 6️⃣ N³ einsum → batched matrix multiplication
    #    left_h[b,i,k] @ right_h[b,j,k]ᵀ  (for each hidden channel h)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H, N, N)

    # Batched GEMM (cuBLAS) – very fast on Hopper
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm
    # ------------------------------------------------------------------
    w_outnorm = weights["to_out_norm.weight"]
    b_outnorm = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * w_outnorm + b_outnorm

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to model dimension
    # ------------------------------------------------------------------
    out_final = F.linear(out_norm, weights["to_out.weight"])

    # Ensure fp32 output (weights are fp32, but keep the cast explicit)
    return out_final.to(torch.float32)
