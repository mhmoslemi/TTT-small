*Only the element‑wise mask‑gate fusion is done in Triton; the heavy

```python
"""
TriMul “outgoing” kernel (AlphaFold‑3)

Algorithm
---------
1. Layer‑norm over the last dimension (`dim`).
2. Linear projections → left/right tensors of shape (B,N,N,H)   (no bias).
3. Linear gates (left, right, out) → sigmoid activations, same shape.
4. Fuse mask (if present) and the three gates into the left/right tensors
   *in‑place* with a tiny Triton kernel.  The out‑gate is returned separately.
5. Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is exactly a batched matrix‑multiply for every hidden dimension:
        L  = left.permute(0,3,1,2)   # (B,H,N,K)
        R  = right.permute(0,3,1,2)  # (B,H,N,K)
        out = torch.matmul(L, R.transpose(-1,-2))   # (B,H,N,N)
   Permute back to (B,N,N,H).
6. Final Layer‑norm over the hidden dimension, scale by the out‑gate and
   project back to `dim` with a linear layer.

Only step 4 is written in Triton; all other steps stay in PyTorch because
they are already highly optimized on the H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: mask + left/right gates + out‑gate fusion
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                # [B,N,N,H]      (float32)
    mask_ptr,                           # [B,N,N,1]      (float32) – may be all‑ones
    lgate_ptr, rgate_ptr, ogate_ptr,   # [B,N,N,H]      (float32) – already sigmoid‑ed
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """
    In‑place:   left  = left * mask * left_gate
                right = right * mask * right_gate
    Returns:    out_gate (already sigmoid) – copied to a temporary buffer.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # flat index inside tensor
    total = B * N * N * H
    mask = offs < total

    # ------------------------------------------------------------------
    # decode flat index → (b,i,k,d)
    # ------------------------------------------------------------------
    tmp = offs
    d = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ------------------------------------------------------------------
    # compute pointers
    # ------------------------------------------------------------------
    left  = left_ptr  + b * stride_l_b  + i * stride_l_i  + k * stride_l_k  + d * stride_l_h
    right = right_ptr + b * stride_r_b  + i * stride_r_i  + k * stride_r_k  + d * stride_r_h
    maskp = mask_ptr  + b * stride_m_b  + i * stride_m_i  + k * stride_m_k
    lgate = lgate_ptr + b * stride_lg_b + i * stride_lg_i + k * stride_lg_k + d * stride_lg_h
    rgate = rgate_ptr + b * stride_rg_b + i * stride_rg_i + k * stride_rg_k + d * stride_rg_h
    ogate = ogate_ptr + b * stride_og_b + i * stride_og_i + k * stride_og_k + d * stride_og_h

    # ------------------------------------------------------------------
    # load, fuse and store
    # ------------------------------------------------------------------
    l = tl.load(left,  mask=mask, other=0.0)
    r = tl.load(right, mask=mask, other=0.0)

    m = tl.load(maskp, mask=mask, other=1.0)          # 1 if no mask
    lg = tl.load(lgate, mask=mask, other=1.0)         # sigmoid already applied
    rg = tl.load(rgate, mask=mask, other=1.0)
    og = tl.load(ogate, mask=mask, other=1.0)

    l = l * m * lg
    r = r * m * rg

    tl.store(left,  l, mask=mask)
    tl.store(right, r, mask=mask)
    # write out‑gate to its own buffer (we cannot modify it in‑place because
    # the caller may still need the original tensor shape)
    tl.store(ogate, og, mask=mask)


def _apply_mask_and_gates(left, right, mask, left_gate, right_gate, out_gate):
    """
    Launches the Triton kernel that fuses mask and three gates.
    Returns the (already‑sigmoid) out_gate tensor (unchanged shape).
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                     # fits easily in registers on H200
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        mask_ptr=mask.data_ptr(),
        lgate_ptr=left_gate.data_ptr(),
        rgate_ptr=right_gate.data_ptr(),
        ogate_ptr=out_gate.data_ptr(),
        B=B, N=N, H=H,
        stride_l_b=left.stride(0), stride_l_i=left.stride(1),
        stride_l_k=left.stride(2), stride_l_h=left.stride(3),
        stride_r_b=right.stride(0), stride_r_i=right.stride(1),
        stride_r_k=right.stride(2), stride_r_h=right.stride(3),
        stride_m_b=mask.stride(0), stride_m_i=mask.stride(1),
        stride_m_k=mask.stride(2), stride_m_1=mask.stride(3),
        stride_lg_b=left_gate.stride(0), stride_lg_i=left_gate.stride(1),
        stride_lg_k=left_gate.stride(2), stride_lg_h=left_gate.stride(3),
        stride_rg_b=right_gate.stride(0), stride_rg_i=right_gate.stride(1),
        stride_rg_k=right_gate.stride(2), stride_rg_h=right_gate.stride(3),
        stride_og_b=out_gate.stride(0), stride_og_i=out_gate.stride(1),
        stride_og_k=out_gate.stride(2), stride_og_h=out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Public entry point required by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, dim]   (float32 / bf16)
        mask_tensor  : torch.Tensor [B, N, N]        (0/1 float) or None
        weights_dict : dict with the following keys (all float32):
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        config_dict  : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor [B, N, N, dim]  (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, w, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → (B,N,N,hidden)
    # ------------------------------------------------------------------
    left  = F.linear(x, w["left_proj.weight"])
    right = F.linear(x, w["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gates (sigmoid) → same shape
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, w["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask handling
    # ------------------------------------------------------------------
    if mask is None:
        # all‑ones mask, shape (B,N,N,1) to broadcast over hidden dim
        mask = torch.ones(B, N, N, 1, dtype=inp.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(inp.dtype)   # (B,N,N,1)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask + gates into left/right (in‑place) with Triton
    # ------------------------------------------------------------------
    out_gate = _apply_mask_and_gates(
        left, right, mask, left_gate, right_gate, out_gate
    )   # left/right now contain mask*gate*proj; out_gate unchanged

    # ------------------------------------------------------------------
    # 6️⃣ Core contraction – batched matmul over the K dimension
    # ------------------------------------------------------------------
    # reshape to (B, hidden, N, K)
    left_t  = left.permute(0, 3, 1, 2)          # (B, H, N, K)
    right_t = right.permute(0, 3, 1, 2)         # (B, H, N, K)

    # (B, H, N, N) = (B, H, N, K) @ (B, H, K, N)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))

    # back to (B, N, N, H)
    out = out_t.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7️⃣ Final Layer‑norm, out‑gate scaling, projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, w["to_out.weight"])

    # The reference implementation returns float32 regardless of input dtype
    return out.to(torch.float32)
```

| *Handle optional mask* | If `mask` is `None` we create an all‑ones mask; otherwise we broadcast it to `(B,N,N,1)`.  This matches the reference behaviour. |
