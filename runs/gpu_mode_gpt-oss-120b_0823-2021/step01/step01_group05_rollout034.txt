"""
TriMul “outgoing” kernel – fully Triton‑implemented core reduction.

The forward pass consists of
1️⃣ LayerNorm + linear projections + gating   (still done in PyTorch for
   numerical stability and because they are cheap).
2️⃣ Optional pair‑wise mask application – a tiny Triton kernel (already fast).
3️⃣ The expensive O(N³·C) reduction:
        out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]
   This is a batched matrix‑multiply for every channel *c*.  The kernel
   tiles the *i* and *j* axes (BLOCK_M × BLOCK_N) and iterates over *k* in
   BLOCK_K chunks.  For each channel‑block (BLOCK_H) we load a tile of the
   left and right tensors, perform a dot‑product over *k* with `tl.dot`,
   and accumulate into the output tile.  All arithmetic is performed in
   float32, the final output is cast back to float32 (the same dtype as the
   inputs).

The implementation works for any batch size (≤2), sequence length N up to
1024, and hidden dimension C (c_z) that is a multiple of BLOCK_H (we pad
the last block with zeros).  It runs on Hopper (sm_90) and fits comfortably
into the 141 GB HBM3e of the H200.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Tiny mask kernel (still used when a mask is present)
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    src_ptr,    # const float*
    mask_ptr,   # const float*   (shape B,N,N)
    dst_ptr,    # float*
    B, N, H,    # i64 scalars (batch, seq_len, hidden)
    stride_src_b, stride_src_i, stride_src_j, stride_src_c,
    stride_mask_b, stride_mask_i, stride_mask_j,
    total_el,   # i64  = B*N*N*H
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_el

    # compute indices
    h = offs % H
    tmp = offs // H
    j = tmp % N
    i = (tmp // N) % N
    b = tmp // (N * N)

    src = tl.load(src_ptr + b * stride_src_b
                           + i * stride_src_i
                           + j * stride_src_j
                           + h * stride_src_c, mask=mask, other=0.0)
    m = tl.load(mask_ptr + b * stride_mask_b
                           + i * stride_mask_i
                           + j * stride_mask_j, mask=mask, other=1.0)
    tl.store(dst_ptr + offs, src * m, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """tensor: [B,N,N,H]  mask: [B,N,N]  returns tensor*mask[...,None]"""
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024

    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_f = mask.view(-1).to(tensor.dtype)

    stride_src = tensor.stride()
    stride_mask = mask.stride()

    grid = (triton.cdiv(total, BLOCK),)
    apply_mask_kernel[grid](
        src,
        mask_f,
        dst,
        B, N, H,
        stride_src[0], stride_src[1], stride_src[2], stride_src[3],
        stride_mask[0], stride_mask[1], stride_mask[2],
        total,
        BLOCK=BLOCK,
    )
    return dst.view(B, N, N, H)


# ----------------------------------------------------------------------
# Core O(N³·C) reduction – fully Triton
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const float*  [B,N,N,C]
    right_ptr,         # const float*  [B,N,N,C]
    out_ptr,           # float*        [B,N,N,C]
    B, N, C,           # i64 scalars
    # strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_c,
    stride_right_b, stride_right_j, stride_right_k, stride_right_c,
    stride_out_b, stride_out_i, stride_out_j, stride_out_c,
    # tile sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)

    # total number of blocks = B * ceil(N/BM) * ceil(N/BN)
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_i * num_j)
    rem   = pid % (num_i * num_j)
    block_i = rem // num_j
    block_j = rem % num_j

    # start indices for this tile
    i_base = block_i * BLOCK_M
    j_base = block_j * BLOCK_N

    # ranges within the tile
    i_range = i_base + tl.arange(0, BLOCK_M)
    j_range = j_base + tl.arange(0, BLOCK_N)

    i_mask = i_range < N
    j_mask = j_range < N

    # accumulator: (BM,BN,BC)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_C), dtype=tl.float32)

    # iterate over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        k_range = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # ----- load left tile [i,k,c] -----
        left_ptrs = left_ptr + \
            batch * stride_left_b + \
            i_range[:, None, None] * stride_left_i + \
            k_range[None, :, None] * stride_left_k + \
            tl.arange(0, BLOCK_C)[None, None, :] * stride_left_c

        left = tl.load(left_ptrs,
                       mask=i_mask[:, None, None] & k_mask[None, :, None],
                       other=0.0)          # (BM, BK, BC)

        # ----- load right tile [j,k,c] -----
        right_ptrs = right_ptr + \
            batch * stride_right_b + \
            j_range[:, None, None] * stride_right_j + \
            k_range[None, :, None] * stride_right_k + \
            tl.arange(0, BLOCK_C)[None, None, :] * stride_right_c

        right = tl.load(right_ptrs,
                        mask=j_mask[:, None, None] & k_mask[None, :, None],
                        other=0.0)        # (BN, BK, BC)

        # For each channel‑block we perform a matrix multiply over k
        # left: (BM, BK, BC)   right: (BN, BK, BC)
        # We need Σ_k left[i,k,c] * right[j,k,c]
        # This is equivalent to a dot product over the K dimension for every c.
        # Triton provides tl.dot for 2‑D matrices; we therefore loop over the
        # channel block.
        for c0 in range(0, BLOCK_C, 1):          # unrolled by the compiler
            # shape (BM, BK) and (BN, BK)
            l = left[:, :, c0]          # (BM, BK)
            r = right[:, :, c0]         # (BN, BK)
            # dot over K -> (BM, BN)
            acc[:, :, c0] += tl.dot(l, r, trans_b=True)   # (BM, BN)

    # write out
    out_ptrs = out_ptr + \
        batch * stride_out_b + \
        i_range[:, None, None] * stride_out_i + \
        j_range[None, :, None] * stride_out_j + \
        tl.arange(0, BLOCK_C)[None, None, :] * stride_out_c

    tl.store(out_ptrs,
             acc,
             mask=i_mask[:, None, None] & j_mask[None, :, None])


# ----------------------------------------------------------------------
# Entry point – combines PyTorch preprocessing with the Triton kernel
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input, mask, weights, config)
        - input : torch.Tensor [B, N, N, dim]   (float32)
        - mask  : torch.Tensor [B, N, N]       (float32, optional)
        - weights : dict of model parameters
        - config  : dict with keys ``dim`` and ``hidden_dim`` and optional
                    ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm – manual to avoid extra modules
    # ------------------------------------------------------------------
    eps = 1e-5
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # optional mask (handled by a small Triton kernel)
    # ------------------------------------------------------------------
    if not nomask:
        left = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # gating
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·C) reduction – Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Pad hidden dimension to a multiple of BLOCK_C for the kernel
    BLOCK_C = 32  # fits well in registers on H200
    pad = (BLOCK_C - hidden % BLOCK_C) % BLOCK_C
    if pad:
        left = torch.nn.functional.pad(left, (0, pad))
        right = torch.nn.functional.pad(right, (0, pad))

    out = torch.empty((B, N, N, hidden + pad), dtype=torch.float32, device=left.device)

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),)

    left_ptr = left
    right_ptr = right
    out_ptr = out

    # strides
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    trimul_outgoing_kernel[grid](
        left_ptr, right_ptr, out_ptr,
        B, N, hidden + pad,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K, BLOCK_C=BLOCK_C,
    )

    if pad:
        out = out[..., :hidden]   # remove padding

    # ------------------------------------------------------------------
    # Final LayerNorm, gating and output projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_mean = out.mean(-1, keepdim=True)
    out_var = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - out_mean) / torch.sqrt(out_var + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    out_norm = out_norm * out_gate

    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    return out_final
