"""
TriMul “outgoing” forward pass (AlphaFold‑3).

*   LayerNorm, Linear projections and the final LayerNorm/Projection are
    performed with PyTorch – they are cheap compared with the O(N³·C) core
    and benefit from the highly‑optimised cuBLAS kernels.
*   The expensive reduction  
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]               (1)
    is a batched matrix‑multiply (left_h @ right_hᵀ) for every hidden
    channel *h*.  This is implemented in pure Triton with a classic
    GEMM tiling (BLOCK_M × BLOCK_N × BLOCK_K) and a 2‑D thread‑block grid.
    The kernel works for any batch size *B*, sequence length *N* and hidden
    dimension *H* (the “hidden_dim” in the paper).  All arithmetic is
    performed in FP32; the final tensor is returned as float‑32.

The host‑side ``custom_kernel`` prepares the inputs, runs the Triton GEMM,
applies the optional mask (also with a tiny Triton kernel) and then finishes
the pipeline in PyTorch.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def mask_apply_kernel(
    src_ptr, mask_ptr, dst_ptr,  #
    total_el, H,  #
    BLOCK: tl.constexpr,
):
    """Broadcast a [B,N,N] mask to a [B,N,N,H] tensor."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_el

    src = tl.load(src_ptr + offs, mask=mask, other=0.0)
    # each hidden‑dim slice shares the same mask element → integer division
    m = tl.load(mask_ptr + (offs // H), mask=mask, other=1.0)

    tl.store(dst_ptr + offs, src * m, mask=mask)


@triton.jit
def trigmul_gemm_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # tensor metadata
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # tiling parameters
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,
):
    """
    Compute, for every batch b and hidden channel h,
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a standard GEMM tiling (M=i, N=j, K=k).  The kernel is launched
    with a 3‑D grid: (grid_m, grid_n, batch*H).  Inside each program we
    further split the batch*H dimension into groups (GROUP_SIZE_M) to keep
    the number of blocks per SM reasonable.
    """
    pid = tl.program_id(0)                     # block row
    pid_n = tl.program_id(1)                   # block col
    pid_bh = tl.program_id(2)                  # (batch, hidden) linear index

    # --------------------------------------------------------------
    # Decode (b, h) from pid_bh
    # --------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # --------------------------------------------------------------
    # Compute the block start coordinates
    # --------------------------------------------------------------
    block_i = (pid // GROUP_SIZE_M) * BLOCK_M
    block_j = (pid_n) * BLOCK_N

    # --------------------------------------------------------------
    # Offsets within the block
    # --------------------------------------------------------------
    offs_i = block_i + tl.arange(0, BLOCK_M)
    offs_j = block_j + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to the start of the current block
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # --------------------------------------------------------------
    # Accumulator
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles of size BLOCK_K
    # --------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        # ----------------------------------------------------------
        # Load a tile of left  : [M, K]
        # ----------------------------------------------------------
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        left_ptrs = left_base + (offs_i[:, None] * stride_left_i
                                 + offs_k[None, :] * stride_left_k)
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # ----------------------------------------------------------
        # Load a tile of right : [N, K]   (note the transposition)
        # ----------------------------------------------------------
        right_ptrs = right_base + (offs_j[:, None] * stride_right_j
                                   + offs_k[None, :] * stride_right_k)
        # right needs to be transposed → we load as (K, N) and later dot
        right_tile = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :],
                             other=0.0)

        # ----------------------------------------------------------
        # GEMM accumulate:  acc[i,j] += Σₖ left[i,k] * right[j,k]
        # ----------------------------------------------------------
        acc += tl.dot(left_tile, right_tile, trans_b=True)

    # --------------------------------------------------------------
    # Write back the result
    # --------------------------------------------------------------
    out_ptrs = out_base + (offs_i[:, None] * stride_out_i
                           + offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# Helper functions
# ----------------------------------------------------------------------


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """Apply mask[...,None] to a [B,N,N,H] tensor using the tiny kernel."""
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024

    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.float().view(-1)          # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)
    mask_apply_kernel[grid](
        src,
        mask_flat,
        dst,
        total,
        H,
        BLOCK=BLOCK,
    )
    return dst.view(B, N, N, H)


def trigmul_gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the core TriMul reduction.
    left, right : [B, N, N, H] (float32)
    Returns      : [B, N, N, H] (float32)
    """
    B, N, _, H = left.shape

    # Strides (in elements) for each dimension
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()

    # Output tensor
    out = torch.empty_like(left)

    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Tiling parameters – these work well on Hopper SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    GROUP_SIZE_M = 8   # groups to improve load‑balance across SMs

    # Grid dimensions
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid_bh = B * H                     # one program per (batch, hidden)

    grid = (grid_m, grid_n, grid_bh)

    trigmul_gemm_kernel[grid](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
    )
    return out

# Entry point
# ----------------------------------------------------------------------


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input, mask, weights, config)
        input  : [B, N, N, dim]    torch.float32
        mask   : [B, N, N]        torch.float32 (0/1) – may be all‑ones.
        weights: dict of model parameters (see docstring above).
        config : dict with keys ``dim`` and ``hidden_dim`` and optional
                 ``nomask`` (True → ignore mask).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm (manual to avoid extra Python calls)
    # ------------------------------------------------------------------
    eps = 1e-5
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b

    # ------------------------------------------------------------------
    # Linear projections (bias‑less)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask – cheap Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        left = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # Gates
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·C) reduction – Triton GEMM
    # ------------------------------------------------------------------
    # left/right shape: [B, N, N, hidden_dim]
    # result shape   : [B, N, N, hidden_dim]
    out = trigmul_gemm_triton(left, right)

    # ------------------------------------------------------------------
    # Final LayerNorm + gating + output projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
