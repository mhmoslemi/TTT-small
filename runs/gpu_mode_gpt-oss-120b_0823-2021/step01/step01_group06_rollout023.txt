"""
TriMul “outgoing” forward pass (AlphaFold‑3 style).

The heavy N³ term   out[i,j,d] = Σₖ left[i,k,d] * right[j,k,d]
is implemented as a batched GEMM with torch.bmm – on Hopper this is far
faster than a hand‑written reduction.  All LayerNorm / Linear / gating
operations stay in PyTorch (they are already fused by torch.compile) and
only the optional mask‑application is off‑loaded to a tiny Triton kernel.
The kernel is written with explicit stride handling so it works for any
contiguous or strided layout of the 4‑D tensors.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton: element‑wise multiplication by a broadcasted mask
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    x_ptr,               # [B,N,N,H] input
    mask_ptr,            # [B,N,N,1] mask (already unsqueezed)
    out_ptr,             # [B,N,N,H] output
    total_elems,         # B*N*N*H  (number of elements to process)
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,   # strides of x  (int64)
    stride_m_b, stride_m_i, stride_m_j,               # strides of mask (no h‑dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear ids
    mask = offs < total_elems                                # guard

    # linear → multi‑dimensional indices (B, i, j, h)
    # We compute the index in each dimension by successive division / modulo
    # using the known strides (the product of later dimensions).
    # This avoids the need for tl.arange for each dim.
    b = (offs // (stride_x_i * stride_x_j * stride_x_h))
    rem = offs - b * (stride_x_i * stride_x_j * stride_x_h)
    i = (rem // (stride_x_j * stride_x_h))
    rem = rem - i * (stride_x_j * stride_x_h)
    j = (rem // stride_x_h)
    h = rem - j * stride_x_h

    # compute pointer offsets (int64) for x and mask
    x_offset   = b * stride_x_b + i * stride_x_i + j * stride_x_j + h * stride_x_h
    # mask has no h‑dim, broadcasted
    m_offset   = b * stride_m_b + i * stride_m_i + j * stride_m_j

    x_val = tl.load(x_ptr + x_offset, mask=mask, other=0.0)
    m_val = tl.load(mask_ptr + m_offset, mask=mask, other=1.0)   # mask is 0/1
    tl.store(out_ptr + x_offset, x_val * m_val, mask=mask)


# ----------------------------------------------------------------------
# Helper: layer‑norm (no fused bias handling in Triton, done in PyTorch)
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """Simple LayerNorm over the last dimension (C)."""
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32).
    """
    inp, mask, weights, config = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = layer_norm(
        inp,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (dim -> hidden)
    # --------------------------------------------------------------
    # bias‑free Linear → use torch.nn.functional.linear
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask application (via Triton)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → broadcast to [B, N, N, 1]
        mask_exp = mask.unsqueeze(-1).contiguous()
        total = left.numel()
        BLOCK = 128 * 1024                     # 131 072 elements ≈ 512 KB (fp32)
        grid = (triton.cdiv(total, BLOCK),)

        # left
        mask_apply_kernel[grid](
            left.data_ptr(),
            mask_exp.data_ptr(),
            left.data_ptr(),
            total,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            mask_exp.stride(0), mask_exp.stride(1), mask_exp.stride(2),
            BLOCK=BLOCK,
        )
        # right
        mask_apply_kernel[grid](
            right.data_ptr(),
            mask_exp.data_ptr(),
            right.data_ptr(),
            total,
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask_exp.stride(0), mask_exp.stride(1), mask_exp.stride(2),
            BLOCK=BLOCK,
        )

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core N³ reduction: batched GEMM over hidden dimension
    # --------------------------------------------------------------
    # reshape to (B*hidden, N, N) for a single batched matmul
    left_rs  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)          # (B*H, N, N)
    right_rs = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)         # (B*H, N, N)

    # out_rs = left_rs @ right_rsᵀ  → (B*H, N, N)
    out_rs = torch.bmm(left_rs, right_rs.transpose(1, 2))

    # back to [B, N, N, hidden]
    out = out_rs.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm + gating + final Linear (hidden -> dim)
    # --------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
