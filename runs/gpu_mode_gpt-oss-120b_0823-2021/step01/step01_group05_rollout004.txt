"""
TriMul “outgoing” forward (AlphaFold‑3).

*   LayerNorm, Linear projections and the O(N³·C) reduction are carried out
    with PyTorch – these ops are already highly‑optimised on Hopper.
*   The only part moved to Triton is the optional pair‑wise mask‑application.
    A tiny kernel multiplies a [B, N, N, H] tensor by a broadcast
    [B, N, N] mask (the mask is 0/1).  Mask‑application is memory‑bound,
    so the explicit vectorised loads/stores of Triton give a noticeable
    speed‑up over a naïve PyTorch multiplication, especially for the large
    N‑values used in the tests.
*   All tensor‑shapes and strides are passed explicitly to the kernel – this
    avoids the “stride‑unpack” pitfalls observed in earlier attempts.
*   The final output is float‑32, matching the reference implementation.

The function `custom_kernel` has the required signature and can be called
directly from a torch‑scripted model or a `torch.compile` pipeline.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: apply a broadcast mask of shape [B, N, N] to a tensor
#               of shape [B, N, N, H] (float32).
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*  (input tensor)
    mask_ptr,         # const float*  (mask tensor)
    dst_ptr,          # float*        (output tensor)
    B, N, H,          # i32 – dimensions
    stride_src_b, stride_src_n1, stride_src_n2, stride_src_h,   # strides of src
    stride_mask_b, stride_mask_n1, stride_mask_n2,               # strides of mask
    total_elements,  # i64 – total number of elements in src (B*N*N*H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear index
    mask = offs < total_elements

    # ----- decode linear index into (b,i,k,h) -----
    #   linear = ((b*N + i)*N + k)*H + h
    #   we recover each coordinate by successive div/mod with known sizes
    tmp = offs // H
    h   = offs % H
    k   = tmp % N
    tmp = tmp // N
    i   = tmp % N
    b   = tmp // N

    # ----- load values -------------------------------------------------
    src = tl.load(src_ptr + b * stride_src_b
                         + i * stride_src_n1
                         + k * stride_src_n2
                         + h * stride_src_h,
                  mask=mask, other=0.0)

    mask_val = tl.load(mask_ptr + b * stride_mask_b
                               + i * stride_mask_n1
                               + k * stride_mask_n2,
                      mask=mask, other=1.0)          # mask is 0/1 → default 1

    # ----- write result -------------------------------------------------
    tl.store(dst_ptr + offs, src * mask_val, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` (B,N,N,H) by ``mask[...,None]`` (B,N,N) using Triton.
    Both tensors must be on CUDA and dtype float32.
    """
    assert tensor.is_cuda and mask.is_cuda
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                     # reasonable occupancy on H200

    # flatten pointers for the kernel
    src_ptr  = tensor.view(-1).data_ptr()
    dst      = torch.empty_like(tensor)
    dst_ptr  = dst.view(-1).data_ptr()
    mask_ptr = mask.view(-1).data_ptr()

    # strides (in elements, not bytes) for the original tensors
    s_b, s_i, s_k, s_h = tensor.stride()
    m_b, m_i, m_k      = mask.stride()

    grid = (triton.cdiv(total, BLOCK),)

    mask_apply_kernel[grid](
        src_ptr,
        mask_ptr,
        dst_ptr,
        B, N, H,
        s_b, s_i, s_k, s_h,
        m_b, m_i, m_k,
        total,
        BLOCK=BLOCK,
    )
    return dst


# ----------------------------------------------------------------------
# Entry‑point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input, mask, weights, config)
        - input : torch.Tensor[B, N, N, dim]   (float32, CUDA)
        - mask  : torch.Tensor[B, N, N]        (float32, 0/1, CUDA)
        - weights : dict of model parameters (all float32 CUDA tensors)
        - config  : dict with keys ``dim`` and ``hidden_dim`` (int)

    Returns
    -------
    torch.Tensor[B, N, N, dim]   (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    # Manual LayerNorm to avoid pulling in torch.nn (keeps everything on the
    # fast torch path and mirrors the reference implementation).
    eps = 1e-5
    mean = x.mean(-1, keepdim=True)
    var  = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask – done with Triton (the only custom kernel)
    # ------------------------------------------------------------------
    if not nomask:
        left  = apply_mask_triton(left,  mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise modulation
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·C) reduction – einsum (leverages cuBLAS GEMM internally)
    # ------------------------------------------------------------------
    # out shape: [B, N, N, hidden_dim]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gating and output projection
    # ------------------------------------------------------------------
    mean = out.mean(-1, keepdim=True)
    var  = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean) / torch.sqrt(var + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    out_norm = out_norm * out_gate

    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out_final.to(torch.float32)
