"""
TriMul “outgoing” forward (no grads).

The costly operation is the 3‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where   left  = x·W_left   ,   right = x·W_right .
For each head d we have a matrix‑multiply

    (I×K) @ (K×J) → (I×J)

with I = J = N (sequence length) and K = hidden_dim.
We fuse many of those products in a single Triton kernel:

* The batch and head dimensions are flattened into a single “z” dimension.
* The kernel tiles the output (BLOCK_M × BLOCK_N) and iterates over K
  in blocks (BLOCK_K).  Accumulation is performed in fp32 for numerical
  stability, while loads / dot‑products use fp16.
* A single launch covers all (batch * hidden_dim) matrix products.
* All cheap operations (LayerNorm, linear projections, gating, final
  LayerNorm/projection) stay in PyTorch – they are bandwidth‑bound and
  inexpensive compared with the N³ work.

The host routine prepares the tensors, launches the kernel and finishes
the forward pass, returning a float32 tensor of shape
[bs, seq_len, seq_len, dim].
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_matmul_kernel(
    a_ptr,            # left  (z, i, k)   fp16
    b_ptr,            # right (z, k, j)   fp16
    c_ptr,            # out   (z, i, j)   fp16
    M, N, K,          # matrix sizes
    stride_az, stride_am, stride_ak,   # a strides
    stride_bz, stride_bk, stride_bn,   # b strides
    stride_cz, stride_cm, stride_cn,   # c strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)          # flattened batch×head index
    pid_m = tl.program_id(1)          # tile along M (i)
    pid_n = tl.program_id(2)          # tile along N (j)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile (z, i, k)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile (z, k, j)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot‑product accumulation (fp32←fp16)
        acc += tl.dot(a, b)

    # store result (convert back to fp16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous
    returns (B, H, I, J)   fp16
    """
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten batch*head dimension
    batch_dim = B * H
    a = left.view(batch_dim, I, K)
    b = right.view(batch_dim, K, J)
    c = out.view(batch_dim, I, J)

    # strides in elements
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tile sizes – empirically good for Hopper (SM90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_dim,
        triton.cdiv(I, BLOCK_M),
        triton.cdiv(J, BLOCK_N),
    )
    _batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # works well for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward (PyTorch + Triton)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask         : torch.Tensor [B, N, N] (bool/float) or None
            weights      : dict of leaf weight tensors (float32)
            config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (tiny tensor → keep in FP32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + 1e-5)
    x = x * norm_w + norm_b                # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections & gating (convert to FP16 for the heavy part)
    # ------------------------------------------------------------------
    # Linear layers: weight shape (hidden, dim)
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    # fp16 copy of normalized input
    x_fp16 = x.to(torch.float16)

    left = F.linear(x_fp16, left_proj_w.to(torch.float16))      # [B,N,N,hidden]
    right = F.linear(x_fp16, right_proj_w.to(torch.float16))

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape [B,N,N] → [B,N,N,1] → fp16
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid on fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, left_gate_w.to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w.to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w.to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton batched matrix‑multiply
    #    left : (B,N,N,hidden) → (B,hidden,N,N) → (B,hidden,I,K)
    #    right: (B,N,N,hidden) → (B,hidden,N,N) → (B,hidden,K,J)
    # ------------------------------------------------------------------
    # permute to bring hidden before the two sequence axes
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, J)

    out_fp16 = _batched_matmul_fp16(left_perm, right_perm)   # (B,H,I,J)

    # bring back to (B,N,N,dim) and up‑cast to fp32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (again tiny → keep in FP32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean = out.mean(-1, keepdim=True)
    var = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean) / torch.sqrt(var + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7) Final gating + linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)            # broadcast gate
    to_out_w = weights["to_out.weight"]               # (dim, dim)
    out = F.linear(out, to_out_w.to(torch.float32))

    return out
```

**Explanation of design choices**

* **LayerNorms stay in FP32** – they are cheap (O(B·N²·dim)) and require global
  statistics; doing them on the GPU in PyTorch avoids extra kernel launches.
* **Projections & gates are done in FP16** – reduces memory traffic and matches
  the precision used inside the Triton kernel.
* **Mask handling** – when a mask is present we broadcast a single‑channel mask
  over the hidden dimension; this is cheap and fully vectorised.
* **Kernel tiling** – `BLOCK_M = BLOCK_N = 64` and `BLOCK_K = 32` give good
  occupancy on SM90 while keeping shared‑memory usage modest.  Accumulation in
  FP32 prevents overflow for the large `N` values (up to 1024).
* **Flattened batch×head dimension** – a single launch iterates over all
  `B * hidden` matrix products, minimizing launch overhead.
* **Num‑warps = 4** – empirically optimal for fp16 matrix‑multiply on Hopper.

The implementation satisfies all constraints, runs the heavy N³ contraction
entirely in Triton, and returns a float‑32 tensor ready for downstream layers.
