"""
TriMul “outgoing” kernel (AlphaFold3) – fast Triton implementation.

The heavy part of the forward pass is the N‑by‑N matrix multiplication
for every (batch, hidden) pair:

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

which is exactly (left_h) @ (right_h).T.
The kernel computes this for a block of rows × cols while iterating over the
K‑dimension in tiles.  All arithmetic is done in FP16 with FP32 accumulation.
LayerNorm / Linear / Gating are kept in PyTorch – they are cheap compared
to the N³ work.

Key design choices (based on the “lessons” from earlier attempts):
 * ✅ Tile sizes (BLOCK_M/N/K) are chosen to keep shared‑memory usage < 64 KB
   (the default limit on Hopper) and to give good occupancy.
 * ✅ Strides are passed explicitly from the host – we never assume a fixed
   number of dimensions, avoiding the “match stride unpacking” pitfall.
 * ✅ The program‑id maps to (batch, hidden, block_row, block_col) in a
   deterministic way; no extra arithmetic is needed.
 * ✅ The kernel only touches the multiplication, so the launch grid is
   (B * H * ceil(N/BLOCK_M) * ceil(N/BLOCK_N)), a single‑dim grid as required
   by Triton’s launch convention.
 * ✅ We avoid any Python control‑flow inside the kernel (no `continue`,
   no `break`), and we never use ops that are unavailable in Triton
   (`tl.mean`, etc.).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: batched matmul for each hidden channel (outgoing direction)
# -------------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # const fp16  [B, H, N, N]   (i,k)
    right_ptr,         # const fp16  [B, H, N, N]   (j,k)
    out_ptr,           #   mut fp16  [B, H, N, N]   (i,j)
    B, N, H,           # tensor dimensions
    # strides for left  (b,h,i,k)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    # strides for right (b,h,j,k)
    stride_r_b, stride_r_h, stride_r_j, stride_r_k,
    # strides for out   (b,h,i,j)
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a tile of the outgoing TriMul:
        out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
    The tile covers BLOCK_M rows (i) and BLOCK_N cols (j) and iterates over K
    in blocks of size BLOCK_K.
    """
    pid = tl.program_id(0)

    # -----------------------------------------------------------------
    # Decode linear pid -> (b, h, block_i, block_j)
    # -----------------------------------------------------------------
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)

    block_j = pid % num_blocks_j
    pid //= num_blocks_j
    block_i = pid % num_blocks_i
    pid //= num_blocks_i
    h = pid % H
    b = pid // H

    # -----------------------------------------------------------------
    # Offsets for this tile
    # -----------------------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (precision‑safety)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K dimension
    # -----------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load left[i,k] ------------------------------------------------
        # shape (BLOCK_M, BLOCK_K)
        l_ptr = (left_ptr
                 + b * stride_l_b
                 + h * stride_l_h
                 + offs_i[:, None] * stride_l_i
                 + offs_k[None, :] * stride_l_k)
        l = tl.load(l_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ---- load right[j,k] and treat as transposed -----------------------
        # we need right_T[k,j] = right[j,k]
        # shape (BLOCK_K, BLOCK_N)
        r_ptr = (right_ptr
                 + b * stride_r_b
                 + h * stride_r_h
                 + offs_k[:, None] * stride_r_k
                 + offs_j[None, :] * stride_r_j)
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # ---- dot -----------------------------------------------------------------
        # (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(l, r)

    # -----------------------------------------------------------------
    # Write back the tile
    # -----------------------------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_o_b
                + h * stride_o_h
                + offs_i[:, None] * stride_o_i
                + offs_j[None, :] * stride_o_j)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Only the N³ matrix multiplication is off‑loaded to Triton;
    the surrounding LayerNorm / Linear / gating stays in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape                     # input: [B,N,N,dim]
    hidden_dim = config["hidden_dim"]                     # c_z from the paper

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm + Linear projections (PyTorch)
    # -----------------------------------------------------------------
    # LayerNorm over the last dimension
    norm_w = weights["norm.weight"].to(dtype)
    norm_b = weights["norm.bias"].to(dtype)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b)

    # Linear projections (bias = False in the reference implementation)
    left_proj_w  = weights["left_proj.weight"].to(dtype)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(dtype)

    # shape after linear: [B,N,N,hidden]
    left = F.linear(x, left_proj_w)        # (B,N,N,hidden)
    right = F.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 2️⃣ Optional mask
    # -----------------------------------------------------------------
    if mask is not None:
        # mask: [B,N,N] -> [B,N,N,1] broadcasting over hidden dim
        m = mask.unsqueeze(-1).to(dtype)
        left = left * m
        right = right * m

    # -----------------------------------------------------------------
    # 3️⃣ Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(dtype)
    right_gate_w = weights["right_gate.weight"].to(dtype)
    out_gate_w   = weights["out_gate.weight"].to(dtype)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ Rearrange to [B, hidden, N, N] and cast to fp16 for the kernel
    # -----------------------------------------------------------------
    # (B,N,N,hidden) -> (B,hidden,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # output buffer (same layout as left/right)
    out_fp16 = torch.empty_like(left_fp16)

    # -----------------------------------------------------------------
    # 5️⃣ Triton launch parameters
    # -----------------------------------------------------------------
    # Tile sizes tuned for Hopper: they keep shared memory < 64KB and give
    # good occupancy for the N values in the test suite (256‑1024).
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: one 1‑D dimension, each element processes a (b,h) tile grid.
    num_blocks_i = triton.cdiv(N, BLOCK_M)
    num_blocks_j = triton.cdiv(N, BLOCK_N)
    total_tiles = B * hidden_dim * num_blocks_i * num_blocks_j
    grid = (total_tiles,)

    # Strides (int64) – note the order matches the kernel signature
    l_strides = left_fp16.stride()   # (B,H,N,N)
    r_strides = right_fp16.stride()
    o_strides = out_fp16.stride()

    _trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        # left strides
        l_strides[0], l_strides[1], l_strides[2], l_strides[3],
        # right strides
        r_strides[0], r_strides[1], r_strides[2], r_strides[3],
        # out strides
        o_strides[0], o_strides[1], o_strides[2], o_strides[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Convert back to (B,N,N,hidden) and finish the PyTorch pipeline
    # -----------------------------------------------------------------
    out = out_fp16.to(dtype).permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden)

    # Final LayerNorm over the hidden dimension
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # Multiply by out_gate (broadcast over the two N axes)
    out = out * out_gate

    # Linear back to original dim
    to_out_w = weights["to_out.weight"].to(dtype)   # (dim, hidden)
    out = F.linear(out, to_out_w)

    return out
