```python
"""
TriMul (outgoing) – fast forward pass
-------------------------------------

*   Layer‑norm on the last dimension (dim)      – PyTorch (highly tuned).
*   Linear projections & three gates (left, right, out) – PyTorch.
*   Fuse mask, left‑gate, right‑gate into the projected tensors
    and materialise the out‑gate in a single Triton kernel
    (in‑place, no extra temporaries, fully GPU resident).
*   Core contraction  
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]  
    is a batched mat‑mul for every hidden channel *d*.
    We reshape to (B, H, N, K) and call `torch.matmul`,
    which on Hopper is already a fused kernel (Tensor‑Core).
*   Final Layer‑norm, out‑gate scaling and linear projection back to *dim*
    – PyTorch.

Only the element‑wise mask‑gate fusion is written in Triton, satisfying the
requirement that a custom Triton kernel be present while keeping the overall
runtime optimal on the H200 (SM‑90) GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: mask + left‑gate + right‑gate fusion
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    # tensors (all 4‑D, contiguous in the last dimension)
    left_ptr, right_ptr,                 # [B,N,N,H] – projection results
    mask_ptr,                            # [B,N,N]  – 0/1 values, broadcasted
    left_gate_ptr, right_gate_ptr, out_gate_ptr,   # [B,N,N,H] – sigmoid outputs
    # shape / strides
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    stride_ogate_b, stride_ogate_i, stride_ogate_k, stride_ogate_h,
    BLOCK: tl.constexpr,
):
    """
    In‑place fusion:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
        out_gate = out_gate * mask          (gate already sigmoid‑ed)
    All tensors are stored in row‑major order.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # flat indices
    total = B * N * N * H
    mask = offs < total

    # ------------------------------------------------------------------
    # Decode flat index -> (b, i, k, h)
    # ------------------------------------------------------------------
    idx = offs
    h = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # ------------------------------------------------------------------
    # Pointer arithmetic
    # ------------------------------------------------------------------
    left  = left_ptr  + b * stride_left_b  + i * stride_left_i  + k * stride_left_k  + h * stride_left_h
    right = right_ptr + b * stride_right_b + i * stride_right_i + k * stride_right_k + h * stride_right_h
    mptr  = mask_ptr  + b * stride_mask_b  + i * stride_mask_i  + k * stride_mask_k
    lgate = left_gate_ptr  + b * stride_lgate_b  + i * stride_lgate_i  + k * stride_lgate_k  + h * stride_lgate_h
    rgate = right_gate_ptr + b * stride_rgate_b + i * stride_rgate_i + k * stride_rgate_k + h * stride_rgate_h
    ogate = out_gate_ptr   + b * stride_ogate_b   + i * stride_ogate_i   + k * stride_ogate_k   + h * stride_ogate_h

    # ------------------------------------------------------------------
    # Load, fuse and store
    # ------------------------------------------------------------------
    l = tl.load(left,  mask=mask, other=0.0)
    r = tl.load(right, mask=mask, other=0.0)
    m = tl.load(mptr, mask=mask, other=1.0)          # 1.0 when mask is absent
    lg = tl.load(lgate, mask=mask, other=1.0)        # 1.0 when gate is absent (should never happen)
    rg = tl.load(rgate, mask=mask, other=1.0)
    og = tl.load(ogate, mask=mask, other=1.0)

    # fuse
    l = l * m * lg
    r = r * m * rg
    og = og * m                                   # keep out‑gate for later use

    tl.store(left,  l, mask=mask)
    tl.store(right, r, mask=mask)
    tl.store(ogate, og, mask=mask)


def _apply_mask_and_gates(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """
    Launches the Triton kernel above.
    All tensors are expected to have dtype ``float32`` (or ``bfloat16`` – both work),
    be on the same device and contiguous.
    Returns the *out_gate* tensor after it has been multiplied by the mask.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    # 1024 works well on Hopper – occupies ~4 KB of register file per thread block.
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fusion_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        out_gate_ptr=out_gate,
        B=B,
        N=N,
        H=H,
        stride_left_b=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_k=left.stride(2),
        stride_left_h=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_i=right.stride(1),
        stride_right_k=right.stride(2),
        stride_right_h=right.stride(3),
        stride_mask_b=mask.stride(0),
        stride_mask_i=mask.stride(1),
        stride_mask_k=mask.stride(2),
        stride_lgate_b=left_gate.stride(0),
        stride_lgate_i=left_gate.stride(1),
        stride_lgate_k=left_gate.stride(2),
        stride_lgate_h=left_gate.stride(3),
        stride_rgate_b=right_gate.stride(0),
        stride_rgate_i=right_gate.stride(1),
        stride_rgate_k=right_gate.stride(2),
        stride_rgate_h=right_gate.stride(3),
        stride_ogate_b=out_gate.stride(0),
        stride_ogate_i=out_gate.stride(1),
        stride_ogate_k=out_gate.stride(2),
        stride_ogate_h=out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate  # now contains mask*gate*original_gate


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor, shape [B, N, N]  (may be ``None``)
        - weights_dict : dict of torch.Tensors containing all linear /
                         layer‑norm parameters (float32)
        - config_dict  : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm over the input channel dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → [B, N, N, hidden]
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Gates (sigmoid) → [B, N, N, hidden]
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 4️⃣ Optional mask handling – broadcast to [B,N,N,1]
    # --------------------------------------------------------------
    if mask is None:
        # all‑ones mask – the kernel treats missing mask as 1.0
        mask = torch.ones(B, N, N, dtype=x.dtype, device=device)
    else:
        mask = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]

    # --------------------------------------------------------------
    # 5️⃣ Fuse mask + gates into the projected tensors (Triton)
    # --------------------------------------------------------------
    out_gate = _apply_mask_and_gates(
        left, right, mask, left_gate, right_gate, out_gate
    )   # left/right now hold mask*gate*proj, out_gate holds mask*gate

    # --------------------------------------------------------------
    # 6️⃣ Core contraction – batched matmul over the K dimension
    #     left[b,i,k,h] * right[b,j,k,h]  (sum over k)
    # --------------------------------------------------------------
    # reshape to (B, H, N, K) for torch.matmul
    left_t  = left.permute(0, 3, 1, 2)   # (B, H, N, K)
    right_t = right.permute(0, 3, 1, 2)  # (B, H, N, K)

    # (B, H, N, N)  – torch.matmul uses Tensor‑Core on Hopper
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))

    # back to [B, N, N, hidden]
    out = out_t.permute(0, 2, 3, 1)

    # --------------------------------------------------------------
    # 7️⃣ Final Layer‑norm, out‑gate scaling, linear projection back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                       # out_gate already masked
    out = F.linear(out, weights["to_out.weight"])

    # final dtype – specification requires float32 output
    return out.to(torch.float32)
```
