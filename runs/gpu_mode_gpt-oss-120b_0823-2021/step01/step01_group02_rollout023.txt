"""
TriMul “outgoing” forward (AlphaFold‑3)

Algorithm
---------
1. Layer‑norm over the input channel dimension (dim).
2. Linear projections → left / right (shape [B,N,N,H]).
3. Linear gates → left_gate / right_gate / out_gate (shape [B,N,N,H]) and
   apply sigmoid.
4. Fuse mask (if present) and the three element‑wise gates into *left*,
   *right* and *out_gate* with a single Triton kernel (in‑place).  This
   keeps the whole pipeline on the GPU and satisfies the “must contain a
   Triton kernel” requirement.
5. Core tensor contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equal to a batched matrix‑multiply for each hidden channel
   d:
        L = left.permute(0,3,1,2)   # (B,H,N,K)
        R = right.permute(0,3,1,2)  # (B,H,N,K)
        O = torch.matmul(L, R.transpose(-1,-2))   # (B,H,N,N)
   The result is permuted back to (B,N,N,H).
6. Final Layer‑norm over the hidden dimension, multiply with the
   out‑gate and a linear projection back to the original channel size.

Only step 4 is performed in Triton; all other steps are delegated to
highly‑optimized PyTorch kernels.  The implementation supports optional
masking, arbitrary batch size (1 or 2), and any sequence length up to
1024 on an NVIDIA H200 (SM‑90).  Mixed‑precision (float16/bfloat16) is
handled automatically by PyTorch – the final result is returned as
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional) + three sigmoid gates into the
#                projected tensors.  The kernel works on a flat view of
#                the (B,N,N,H) tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                # [B,N,N,H] projected tensors
    mask_ptr,                           # [B,N,N,1] mask (1.0 if no mask)
    left_gate_ptr, right_gate_ptr, out_gate_ptr,   # [B,N,N,H] sigmoid gates
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)                # flat offsets
    total = B * N * N * H
    mask = offs < total

    # decode flat index → (b, i, k, h)
    idx = offs
    h = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # base pointers for each tensor
    l_ptr = left_ptr   + b * stride_l_b   + i * stride_l_i   + k * stride_l_k   + h * stride_l_h
    r_ptr = right_ptr  + b * stride_r_b   + i * stride_r_i   + k * stride_r_k   + h * stride_r_h
    m_ptr = mask_ptr    + b * stride_m_b   + i * stride_m_i   + k * stride_m_k                # mask is single‑channel
    lg_ptr = left_gate_ptr  + b * stride_lg_b + i * stride_lg_i + k * stride_lg_k + h * stride_lg_h
    rg_ptr = right_gate_ptr + b * stride_rg_b + i * stride_rg_i + k * stride_rg_k + h * stride_rg_h
    og_ptr = out_gate_ptr   + b * stride_og_b + i * stride_og_i + k * stride_og_k + h * stride_og_h

    # load (mask defaults to 1.0, gates default to 1.0 if something goes out of bounds)
    l = tl.load(l_ptr, mask=mask, other=0.0)
    r = tl.load(r_ptr, mask=mask, other=0.0)
    m = tl.load(m_ptr, mask=mask, other=1.0)
    lg = tl.load(lg_ptr, mask=mask, other=1.0)
    rg = tl.load(rg_ptr, mask=mask, other=1.0)
    og = tl.load(og_ptr, mask=mask, other=1.0)

    # fuse:   left  = left * mask * left_gate
    #         right = right * mask * right_gate
    # out_gate is left untouched, we just store the sigmoid value back
    l = l * m * lg
    r = r * m * rg

    tl.store(l_ptr, l, mask=mask)
    tl.store(r_ptr, r, mask=mask)
    tl.store(og_ptr, og, mask=mask)


def _apply_mask_and_gates(left, right, mask, left_gate, right_gate, out_gate):
    """Launch the Triton kernel that fuses mask & the three sigmoid gates.

    The operation is performed in‑place on *left* and *right*; *out_gate*
    is returned (already contains sigmoid values).
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                # good trade‑off on Hopper (≈4 KB per warp)
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left, right,
        mask,
        left_gate, right_gate, out_gate,
        B, N, H,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2), mask.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor of shape [B, N, N]  (may be None)
        - weights_dict : dict with the exact same keys used in the reference
                         PyTorch implementation
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → [B,N,N,hidden]
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gates (sigmoid) → [B,N,N,hidden]
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast to 1‑channel)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(B, N, N, 1, dtype=input_tensor.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(input_tensor.dtype)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask & gates into left/right (in‑place) with Triton
    # ------------------------------------------------------------------
    out_gate = _apply_mask_and_gates(left, right, mask, left_gate, right_gate, out_gate)

    # ------------------------------------------------------------------
    # 6️⃣ Core contraction (batched matmul over the K dimension)
    #        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    #   (B,N,N,H) → (B,H,N,K)
    left_t  = left.permute(0, 3, 1, 2)   # (B, H, N, K)
    right_t = right.permute(0, 3, 1, 2)  # (B, H, N, K)

    # Batched matmul (float32 – torch chooses the best kernel on Hopper)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))  # (B, H, N, N)

    # Back to original layout
    out = out_t.permute(0, 2, 3, 1)    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ Final Layer‑norm over the hidden dimension, out‑gate scaling,
    #     and projection back to *dim*
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # Ensure the output is float32 as required by the spec
    return out.to(torch.float32)
