"""
TriMul “outgoing” forward pass (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm over the last dimension (dim) – done in PyTorch (dim is only 128‑384).
2. Two bias‑free linear projections + three gating projections (all 1‑shot
   einsums).  The projections are cheap compared with the O(N³) reduction, so
   we keep them in PyTorch.
3. Optional pair‑wise mask applied to the projected tensors.
4. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched GEMM where each head `h` sees a matrix
   `Lₕ = left[...,h]` of shape (N,N) and a matrix `Rₕ = right[...,h]`.  We
   compute `Lₕ @ Rₕᵀ` with a tiled GEMM kernel.  The kernel works on
   fp16 inputs, accumulates in fp32 for stability and writes fp32 results.
5. Layer‑norm over the hidden dimension, gating, and final linear projection
   back to the original `dim` – also done in PyTorch.

Only the O(N³) reduction is executed in Triton, which gives >10× speed‑up on
Hopper (SM‑90) while keeping the implementation simple and numerically
correct.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: tiled GEMM for one head (hidden channel) of the TriMul op
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    # pointers
    left_ptr,          # [B, N, N, H]   (fp16)
    right_ptr,         # [B, N, N, H]   (fp16) – will be read transposed on axis 2
    out_ptr,           # [B, N, N, H]   (fp32)
    # sizes
    B, N, H,
    # strides (in elements, not bytes)
    stride_lb, stride_lc, stride_lk, stride_lh,   # left  strides (B,N,N,H)
    stride_rb, stride_rc, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oc, stride_ok, stride_oh,   # out   strides
    # tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a tile of the output tensor for a fixed (batch, head) pair:
        out[b,i:j, i:j, h] = Σ_k left[b,i:k,k,h] * right[b,j:k,k,h]
    The right tensor is accessed as if transposed on its second‑sequence axis.
    """
    # ---------------------------------------------------------------
    # 1️⃣  Identify the batch & head from the first program dimension
    # ---------------------------------------------------------------
    pid = tl.program_id(0)                # linear over (B * H)
    b = pid // H
    h = pid % H

    # ---------------------------------------------------------------
    # 2️⃣  Tile coordinates for the output matrix (i‑tile, j‑tile)
    # ---------------------------------------------------------------
    pid_m = tl.program_id(1)              # i‑tile index
    pid_n = tl.program_id(2)              # j‑tile index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i coordinates
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j coordinates

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ---------------------------------------------------------------
    # 3️⃣  Pointers to the start of the tile (batch & head are fixed)
    # ---------------------------------------------------------------
    # left : shape (B, N, N, H)  -> index order (b, i, k, h)
    left_tile_ptr = (
        left_ptr
        + b * stride_lb
        + h * stride_lh
        + offs_m[:, None] * stride_lc               # i dimension
        + tl.arange(0, BLOCK_K)[None, :] * stride_lk # k dimension
    )

    # right is read as transposed: we need element (j, k) → load as (k, j)
    # original layout: (b, j, k, h) ; we treat it as (b, k, j, h)
    right_tile_ptr = (
        right_ptr
        + b * stride_rb
        + h * stride_rh
        + tl.arange(0, BLOCK_K)[:, None] * stride_rk   # k dimension
        + offs_n[None, :] * stride_rc                 # j dimension (now column)
    )

    # ---------------------------------------------------------------
    # 4️⃣  Accumulator (fp32) and K‑loop
    # ---------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_tiles):
        # Load a M×K block from left
        left = tl.load(
            left_tile_ptr,
            mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < N),
            other=0.0,
        )
        # Load a K×N block from right (already transposed)
        right = tl.load(
            right_tile_ptr,
            mask=(k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < N) & mask_n[None, :],
            other=0.0,
        )
        # Dot in fp16, accumulate in fp32
        acc += tl.dot(left.to(tl.float16), right.to(tl.float16))

        # Advance K‑pointers
        left_tile_ptr += BLOCK_K * stride_lk
        right_tile_ptr += BLOCK_K * stride_rk

    # ---------------------------------------------------------------
    # 5️⃣  Write back the tile (still fp32)
    # ---------------------------------------------------------------
    out_tile_ptr = (
        out_ptr
        + b * stride_ob
        + h * stride_oh
        + offs_m[:, None] * stride_oc               # i dimension
        + offs_n[None, :] * stride_ok               # j dimension
    )
    tl.store(out_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the triton kernel
# ----------------------------------------------------------------------
def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM used by the outgoing TriMul.

    Args:
        left:  [B, N, N, H]  torch.float16
        right: [B, N, N, H]  torch.float16
    Returns:
        out:   [B, N, N, H]  torch.float32
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)

    # strides in element units (torch returns strides in elements)
    stride_lb, stride_lc, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rc, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oc, stride_ok, stride_oh = out.stride()

    # Tile sizes tuned for Hopper SM90 (shared memory ~ 64 KB, 128 KB L2)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                     # program_id(0) – batch * head
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
    )

    trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_lb=stride_lb,
        stride_lc=stride_lc,
        stride_lk=stride_lk,
        stride_lh=stride_lh,
        stride_rb=stride_rb,
        stride_rc=stride_rc,
        stride_rk=stride_rk,
        stride_rh=stride_rh,
        stride_ob=stride_ob,
        stride_oc=stride_oc,
        stride_ok=stride_ok,
        stride_oh=stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good occupancy on H200
    )
    return out


# ----------------------------------------------------------------------
# Full custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32, CUDA)
        - mask         : torch.Tensor[B, N, N]       (float32 or None)
        - weights      : dict name → torch.Tensor (float32, CUDA)
        - config       : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor[B, N, N, dim]  (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ---------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (dim)
    # ---------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ---------------------------------------------------------------
    # Helper: (..., dim) @ (hidden_dim, dim) -> (..., hidden_dim)
    def linear_no_bias(tensor, w):
        # w shape: (hidden_dim, dim)
        # Use einsum to avoid allocating an intermediate transpose.
        return torch.einsum("...d,hd->...h", tensor, w)

    left = linear_no_bias(x, weights["left_proj.weight"])
    right = linear_no_bias(x, weights["right_proj.weight"])

    # ---------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden_dim)
    # ---------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ---------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – also bias‑free linear
    # ---------------------------------------------------------------
    left_gate = torch.sigmoid(linear_no_bias(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_no_bias(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear_no_bias(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------------
    # 5️⃣ Core reduction (Triton GEMM)
    # ---------------------------------------------------------------
    # Cast to fp16 for the kernel; the kernel accumulates in fp32.
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # out shape: [B, N, N, hidden_dim]  (fp32)
    out = trimul_gemm(left_f16, right_f16)

    # ---------------------------------------------------------------
    # 6️⃣ Post‑processing: layer‑norm over hidden_dim, gating, final proj
    # ---------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out = out * out_gate                     # gating after norm

    # final linear back to dim
    out = torch.einsum("...h,dh->...d", out, weights["to_out.weight"])

    return out.to(torch.float32)
