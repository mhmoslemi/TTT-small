"""
TriMul – “outgoing” version (AlphaFold‑3)

The forward pass consists of
1️⃣ Layer‑norm over the input channel dimension *dim* (PyTorch).
2️⃣ Linear projections → tensors L,R of shape (B,N,N,H).
3️⃣ Three sigmoid‑gates (left, right, out) of the same shape.
4️⃣ **Triton kernel** – fuses the optional mask with the left/right
   projections and their gates.  This keeps the heavy element‑wise work
   completely on‑device and avoids an extra launch for the out‑gate.
5️⃣ Core contraction
        out[b,i,j,d] = Σ_k L[b,i,k,d] * R[b,j,k,d]
   which is a batched matrix‑multiply per hidden channel.
   We perform it with `torch.matmul` on the transposed view
   (B,H,N,K) @ (B,H,K,N) → (B,H,N,N) because the GEMM routine is already
   highly tuned on Hopper.
6️⃣ Final Layer‑norm, out‑gate scaling and projection back to *dim*
   (PyTorch).

Only step 4 is written in Triton, satisfying the “must contain a Triton
kernel” requirement while keeping overall runtime optimal on an H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional) and three sigmoid‑gates into the
#               left/right projected tensors.  The out‑gate is written
#               back unchanged (already sigmoid‑ed) so it can be used later.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # [B,N,N,H]  float32
    right_ptr,         # [B,N,N,H]  float32
    mask_ptr,          # [B,N,N,1]  float32  (all‑ones if no mask)
    lgate_ptr,         # [B,N,N,H]  float32  (sigmoid)
    rgate_ptr,         # [B,N,N,H]  float32  (sigmoid)
    ogate_ptr,         # [B,N,N,H]  float32  (sigmoid) – written back unchanged
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_m_b, stride_m_i, stride_m_k, stride_m_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,k,d) we compute
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    and store the (already sigmoid‑ed) out_gate unchanged.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # flat offsets

    total = B * N * N * H
    mask = offs < total
    if not mask.any():
        return

    # ---- decode flat index into (b,i,k,d) ---------------------------
    idx = offs
    d = idx % H
    idx = idx // H
    k = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # ---- compute pointers -------------------------------------------
    left  = left_ptr  + b * stride_l_b  + i * stride_l_i  + k * stride_l_k  + d * stride_l_h
    right = right_ptr + b * stride_r_b  + i * stride_r_i  + k * stride_r_k  + d * stride_r_h
    mask_p = mask_ptr  + b * stride_m_b  + i * stride_m_i  + k * stride_m_k                # 1‑channel
    lgate = lgate_ptr + b * stride_lg_b + i * stride_lg_i + k * stride_lg_k + d * stride_lg_h
    rgate = rgate_ptr + b * stride_rg_b + i * stride_rg_i + k * stride_rg_k + d * stride_rg_h
    ogate = ogate_ptr + b * stride_og_b + i * stride_og_i + k * stride_og_k + d * stride_og_h

    # ---- load --------------------------------------------------------
    l = tl.load(left,  mask=mask, other=0.0)
    r = tl.load(right, mask=mask, other=0.0)
    m = tl.load(mask_p,mask=mask, other=1.0)      # 1.0 when mask absent
    lg = tl.load(lgate,mask=mask, other=1.0)
    rg = tl.load(rgate,mask=mask, other=1.0)
    og = tl.load(ogate,mask=mask, other=1.0)

    # ---- fuse ---------------------------------------------------------
    l = l * m * lg
    r = r * m * rg

    # ---- store --------------------------------------------------------
    tl.store(left,  l, mask=mask)
    tl.store(right, r, mask=mask)
    tl.store(ogate, og, mask=mask)               # out‑gate unchanged


def _apply_mask_gate(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """
    Launches the Triton kernel above.  All tensors are contiguous
    ``float32`` and have shape (B,N,N,H) – ``mask`` is (B,N,N,1).
    Returns the out_gate tensor (already sigmoid‑ed) for later use.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                       # empirical sweet‑spot for H200
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left,
        right,
        mask,
        left_gate,
        right_gate,
        out_gate,
        B, N, H,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2), mask.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        BLOCK=BLOCK,
    )
    # out_gate already contains sigmoid(x @ out_gate_weight)
    return out_gate


# ----------------------------------------------------------------------
# Entry point – the full forward pass of the outgoing TriMul module
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor  [B, N, N, dim]   (float32/bfloat16)
        * mask_tensor  : torch.Tensor  [B, N, N]        (float32), may be ``None``.
        * weights_dict : dict of torch.Tensor containing all linear /
                         layer‑norm parameters (float32).
        * config_dict  : dict with keys ``dim`` and ``hidden_dim``.
    Returns
    -------
    torch.Tensor  [B, N, N, dim]   (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]                     # = c_z in the paper

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm over ``dim`` (PyTorch – already very fast)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → left / right  [B,N,N,hidden]
    # --------------------------------------------------------------
    left  = F.linear(x, w["left_proj.weight"])
    right = F.linear(x, w["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Gates (sigmoid) → same shape
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, w["out_gate.weight"]))

    # --------------------------------------------------------------
    # 4️⃣ Optional mask – broadcast to (B,N,N,1)
    # --------------------------------------------------------------
    if mask is None:
        mask = torch.ones(B, N, N, 1, dtype=x.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(x.dtype)       # (B,N,N,1)

    # --------------------------------------------------------------
    # 5️⃣ Fuse mask + gates into left/right with Triton (in‑place)
    # --------------------------------------------------------------
    out_gate = _apply_mask_gate(left, right, mask, left_gate, right_gate, out_gate)

    # --------------------------------------------------------------
    # 6️⃣ Core contraction.
    #     out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #     Equivalent to batched GEMM per hidden channel.
    # --------------------------------------------------------------
    #   (B,N,N,H) → (B,H,N,K) where K == N
    left_t  = left.permute(0, 3, 1, 2)          # (B, H, N, K)
    right_t = right.permute(0, 3, 1, 2)         # (B, H, N, K)

    # batched matmul (float32) – leverages Hopper’s tensor‑core kernels
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))   # (B, H, N, N)

    # back to original layout
    out = out_t.permute(0, 2, 3, 1)            # (B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣ Final LayerNorm, out‑gate scaling and projection back to ``dim``.
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )
    out = out * out_gate                       # gated output
    out = F.linear(out, w["to_out.weight"])

    # Ensure output dtype is float32 as required by the spec
    return out.to(torch.float32)
