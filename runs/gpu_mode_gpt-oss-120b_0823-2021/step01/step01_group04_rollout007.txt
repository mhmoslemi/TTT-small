"""
TriMul “outgoing” forward pass (AlphaFold3).

The heavy N³ work (einsum) is turned into a batch of matrix‑multiplications:
    out_h = left_h @ right_hᵀ   (B·H, N, N) × (B·H, N, N)ᵀ
which is executed with a single torch.bmm (cuBLAS).

The element‑wise part (LayerNorm, linear projections, gating, mask) is kept in
FP32 PyTorch for numerical stability, **except** the three fused element‑wise
operations (mask * left_gate * left, mask * right_gate * right).  These are
performed in‑place by a tiny Triton kernel that processes the 4‑D tensors as a
flat array, avoiding an extra memory pass and letting the compiler fuse the
loads/stores.

All tensors stay on the CUDA device; the final result is returned as FP32.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask, left‑gate and right‑gate into the projected
# tensors.  The kernel works completely in‑place on `left` and `right`.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_fuse_kernel(
    left_ptr,            # float32 * [B, N, N, H]
    right_ptr,           # float32 * [B, N, N, H]
    mask_ptr,            # float32 * [B, N, N, 1]   (broadcasted mask)
    left_gate_ptr,       # float32 * [B, N, N, H]
    right_gate_ptr,      # float32 * [B, N, N, H]

    B, N, H,             # static tensor sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,

    BLOCK: tl.constexpr,
):
    """Fuse mask and gates into left/right tensors (in‑place)."""
    pid = tl.program_id(0)                       # 1‑D launch
    total = B * N * N * H                         # total number of elements
    offs = pid * BLOCK + tl.arange(0, BLOCK)      # linear indices for this thread
    mask = offs < total

    # ----- Decode linear offset into (b,i,k,h) -----
    # linear = ((b * N + i) * N + k) * H + h
    b = offs // (N * N * H)
    rest = offs % (N * N * H)
    i = rest // (N * H)
    rest = rest % (N * H)
    k = rest // H
    h = rest % H

    # ----- Compute pointer arithmetic (element strides) -----
    left_off = (b * stride_left_b +
                i * stride_left_i +
                k * stride_left_k +
                h * stride_left_h)

    right_off = (b * stride_right_b +
                 i * stride_right_i +
                 k * stride_right_k +
                 h * stride_right_h)

    mask_off = (b * stride_mask_b +
                i * stride_mask_i +
                k * stride_mask_k)               # mask has no channel dim

    lgate_off = (b * stride_lgate_b +
                 i * stride_lgate_i +
                 k * stride_lgate_k +
                 h * stride_lgate_h)

    rgate_off = (b * stride_rgate_b +
                 i * stride_rgate_i +
                 k * stride_rgate_k +
                 h * stride_rgate_h)

    # ----- Load, fuse and store back -----
    left   = tl.load(left_ptr   + left_off,   mask=mask, other=0.0)
    right  = tl.load(right_ptr  + right_off,  mask=mask, other=0.0)
    m      = tl.load(mask_ptr   + mask_off,   mask=mask, other=0.0)
    lgate  = tl.load(left_gate_ptr + lgate_off, mask=mask, other=0.0)
    rgate  = tl.load(right_gate_ptr+ rgate_off, mask=mask, other=0.0)

    left  = left  * m * lgate
    right = right * m * rgate

    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left: torch.Tensor,
                         right: torch.Tensor,
                         mask: torch.Tensor,
                         left_gate: torch.Tensor,
                         right_gate: torch.Tensor) -> None:
    """
    Launch the Triton kernel that fuses:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    The operation is performed in‑place.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                     # good occupancy on Hopper
    grid = (triton.cdiv(total, BLOCK),)

    # Tensor strides are expressed in *elements* (not bytes)
    sl = left.stride()
    sr = right.stride()
    sm = mask.stride()               # mask has shape (B,N,N,1)
    slg = left_gate.stride()
    srg = right_gate.stride()

    _tri_mul_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=sl[0], stride_left_i=sl[1], stride_left_k=sl[2], stride_left_h=sl[3],
        stride_right_b=sr[0], stride_right_i=sr[1], stride_right_k=sr[2], stride_right_h=sr[3],
        stride_mask_b=sm[0], stride_mask_i=sm[1], stride_mask_k=sm[2],
        stride_lgate_b=slg[0], stride_lgate_i=slg[1], stride_lgate_k=slg[2], stride_lgate_h=slg[3],
        stride_rgate_b=srg[0], stride_rgate_i=srg[1], stride_rgate_k=srg[2], stride_rgate_h=srg[3],
        BLOCK=BLOCK,
    )
    # In‑place, nothing to return.


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

        * input_tensor : torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
        * mask         : torch.Tensor of shape [B, N, N]   (bool/float, CUDA)
        * weights      : dict with all linear / LN parameters
        * config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (FP32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]          # (dim,)
    norm_b = weights["norm.bias"]            # (dim,)
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b          # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"]  # (hidden, dim)

    left  = torch.nn.functional.linear(x_norm, left_proj_w)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid) – keep as FP32 tensors
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()   # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 4. Apply mask and gating with a Triton kernel (in‑place)
    # ------------------------------------------------------------------
    # mask is (B,N,N); expand to (B,N,N,1) for broadcasting inside the kernel
    if mask is None:
        # no mask → treat as all‑ones
        mask_tensor = torch.ones_like(left[..., :1])
    else:
        mask_tensor = mask.unsqueeze(-1).to(left.dtype)   # (B,N,N,1)

    _fuse_mask_and_gates(left, right, mask_tensor, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 5. N³ einsum via batched matrix multiply
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    ⇔ for each head h: left_h @ right_hᵀ
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, N, N, H) → (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Batched GEMM (cuBLAS) – highly optimized on Hopper
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # Restore original layout (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)   # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + output gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    out_norm = out_norm * out_gate   # apply output gate (broadcasted over hidden)

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]    # (dim, hidden)
    out_final = torch.nn.functional.linear(out_norm, to_out_w)   # (B,N,N,dim)

    return out_final.to(torch.float32)
```

### What the implementation does
1. **LayerNorm & linear layers** – kept in PyTorch for numerical stability.
2. **Mask & gating fusion** – a custom Triton kernel (`_tri_mul_fuse_kernel`) processes the 4‑D tensors as a flat array, applying `mask * gate` and storing the result back in‑place. This eliminates a separate memory‑bound pass and is the required Triton component.
3. **N³ contraction** – reshaped to a batch of matrix multiplications and executed with a single `torch.bmm`, which on Hopper uses the highly‑tuned cuBLAS kernels.
4. **Output normalisation, gate, and final linear** – back in PyTorch.

The kernel respects all launch‑convention rules, works for any supported `B, N, dim, hidden_dim`, and returns a float‑32 tensor on the same CUDA device.
