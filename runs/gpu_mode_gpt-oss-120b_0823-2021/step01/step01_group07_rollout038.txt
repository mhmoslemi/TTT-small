```python
"""
TriMul outgoing kernel – fast Triton implementation of the
N³ reduction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where
    left  = (x·W_left)   * sigmoid(x·W_left_gate)
    right = (x·W_right)  * sigmoid(x·W_right_gate)

The surrounding LayerNorm / gating / linear projections are kept in
PyTorch (they are cheap compared to the O(N³) work).  The core
reduction is a tiled GEMM implemented in fp16 with fp32 accumulation.
The kernel is launched with three‑dimensional grid:
    program_id(0) → (batch, head) pair
    program_id(1) → i‑tile
    program_id(2) → j‑tile
Tile sizes (64×64×32) were chosen to fit the SM‑90 shared‑memory budget
and give high occupancy on the H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for one hidden channel
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    # pointers
    left_ptr,          # [B, N, N, H]   fp16
    right_ptr,         # [B, N, N, H]   fp16
    out_ptr,           # [B, N, N, H]   fp32
    # problem size
    B, N, H,
    # strides (in elements, not bytes)
    stride_lb, stride_lc, stride_lk, stride_lh,   # left
    stride_rb, stride_rc, stride_rk, stride_rh,   # right (original layout)
    stride_ob, stride_oc, stride_ok, stride_oh,   # out
    # tile sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute one (i‑tile, j‑tile) of the output for a fixed (batch, head)
    pair.
    Right tensor is logically accessed as right[b, j, k, h] → we load it as
    right[b, k, j, h] (i.e. transposed on the second axis) so the inner
    product becomes a regular matrix‑multiply.
    """
    pid0 = tl.program_id(0)          # linear index over (B, H)
    b = pid0 // H                    # batch index
    h = pid0 % H                     # head (hidden) index

    pid_m = tl.program_id(1)         # i‑tile
    pid_n = tl.program_id(2)         # j‑tile

    # ------------------------------------------------------------------
    # 1) tile coordinates in the output matrix
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑axis
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑axis

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # 2) Pointers to the current tile (batch & head are fixed)
    # ------------------------------------------------------------------
    # left:   shape (N, N, H) → we need slice (i, k, h)
    left_tile_ptr = (left_ptr
        + b * stride_lb
        + h * stride_lh
        + offs_m[:, None] * stride_lc          # i dimension
        + tl.arange(0, BLOCK_K)[None, :] * stride_lk)   # k dimension

    # right is read transposed: we need slice (j, k, h) → load as (k, j)
    right_tile_ptr = (right_ptr
        + b * stride_rb
        + h * stride_rh
        + tl.arange(0, BLOCK_K)[:, None] * stride_rk   # k dimension
        + offs_n[None, :] * stride_rc)                 # j dimension

    # accumulator (fp32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) Loop over the reduction dimension K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_idx in range(0, num_k_tiles):
        # masks for the current K‑tile
        cur_k = k_idx * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left block  (M × K)
        left = tl.load(left_tile_ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # right block (K × N) – already transposed
        right = tl.load(right_tile_ptr,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        # fp16 dot, accumulate in fp32
        left  = left.to(tl.float16)
        right = right.to(tl.float16)
        acc += tl.dot(left, right)          # (M,K)·(K,N) → (M,N)

        # advance K‑pointers
        left_tile_ptr  += BLOCK_K * stride_lk
        right_tile_ptr += BLOCK_K * stride_rk

    # ------------------------------------------------------------------
    # 4) Write back the tile
    # ------------------------------------------------------------------
    out_tile_ptr = (out_ptr
        + b * stride_ob
        + h * stride_oh
        + offs_m[:, None] * stride_oc          # i dimension
        + offs_n[None, :] * stride_ok)         # j dimension

    tl.store(out_tile_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : [B, N, N, H]   fp16
    returns    : [B, N, N, H]   fp32
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # element‑wise strides (not bytes)
    s_lb, s_lc, s_lk, s_lh = left.stride()
    s_rb, s_rc, s_rk, s_rh = right.stride()
    s_ob, s_oc, s_ok, s_oh = out.stride()

    # tile sizes – empirically good for SM‑90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                     # program_id(0) → batch * head
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
    )

    trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=H,
        stride_lb=s_lb, stride_lc=s_lc, stride_lk=s_lk, stride_lh=s_lh,
        stride_rb=s_rb, stride_rc=s_rc, stride_rk=s_rk, stride_rh=s_rh,
        stride_ob=s_ob, stride_oc=s_oc, stride_ok=s_ok, stride_oh=s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,               # works well on Hopper
    )
    return out


# ----------------------------------------------------------------------
# Full custom‑kernel entry point (Python side)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]  torch.float32, CUDA
        - mask         : [B, N, N] or None (float32/bool)
        - weights_dict : mapping name → torch.Tensor (fp32)
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    x, mask, weights, cfg = data
    dim, hidden_dim = cfg["dim"], cfg["hidden_dim"]
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (over dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # -> fp32

    # ------------------------------------------------------------------
    # Helper: bias‑free linear (einsum) – keep it in fp32
    # ------------------------------------------------------------------
    def linear_no_bias(tensor, w):
        # tensor: [..., dim]   w: [hidden_dim, dim]
        return torch.einsum("...d,hd->...h", tensor, w)

    # ------------------------------------------------------------------
    # 2) Projections
    # ------------------------------------------------------------------
    left  = linear_no_bias(x, weights["left_proj.weight"])
    right = linear_no_bias(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (mask shape [B,N,N] → broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)           # [..., N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(linear_no_bias(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_no_bias(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear_no_bias(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core Triton reduction
    # ------------------------------------------------------------------
    # Cast to fp16 for the GEMM (small loss, keeps memory traffic low)
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # out has shape [B, N, N, hidden_dim] and dtype fp32
    out = trimul_gemm(left_f16, right_f16)

    # ------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm over hidden_dim, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                               # gate after norm

    # final projection back to original dim
    out = torch.einsum("...h,dh->...d", out, weights["to_out.weight"])

    return out.to(torch.float32)
```

### How the lessons were applied
| Lesson | Relevance & Action |
|--------|--------------------|
| **Limit shared‑memory usage** | Tile sizes (64×64×32) keep the per‑SM shared memory ≈ 64 KB × 2 (left + right) < 96 KB limit on SM‑90, providing high occupancy. |
| **Match stride unpacking to tensor rank** | Strides are extracted exactly from the 4‑D tensors (`left.stride()` etc.) and passed to the kernel; no hard‑coded assumptions about dimension count are made. |
| **Do not copy whole kernel verbatim** | The kernel is written from‑scratch while following the same algorithmic pattern (tiled GEMM) but with explicit masks, transposed right‑load, and fp16/ fp32 handling as required. |
| **Avoid overly generic fallback** | No Python fallback is provided – if the kernel cannot launch the error will surface, ensuring correctness is not silently hidden. |

The implementation satisfies all constraints:
* Works for any `B ∈ {1,2}`, `N` up to 1024, and both `dim` / `hidden_dim` values.
* Handles presence or absence of a mask.
* Performs the O(N³) reduction in Triton, the rest in PyTorch.
* Final output is `float32` on the same CUDA device.
