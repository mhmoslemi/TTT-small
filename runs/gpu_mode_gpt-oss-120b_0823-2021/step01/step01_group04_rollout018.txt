"""
TriMul “outgoing” forward pass (AlphaFold‑3 style).

The heavy N³ contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

is a batch of independent matrix‑multiplications for each head ``h``.
We therefore reshape the ``(B,N,N,H)`` tensors to ``(B*H, N, N)`` and
use a single ``torch.bmm`` (cuBLAS) – this gives essentially optimal
throughput on Hopper.  The only part that benefits from Triton is the
point‑wise “mask × gate” fusion:

    left  = (x @ W_left )  * mask * sigmoid(x @ W_lgate)
    right = (x @ W_right)  * mask * sigmoid(x @ W_rgate)

All other cheap linear / LayerNorm steps stay in PyTorch.  The kernel
operates in‑place on ``left`` and ``right`` so no extra memory
allocations are needed.

The implementation works for any batch size (1‑2), sequence length up to
1024, and hidden dimensions 128 (or any multiple of 32).  It produces a
float‑32 output identical to the reference PyTorch module.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel – fuses mask, left‑gate and right‑gate elementwise ops
# ----------------------------------------------------------------------
@triton.jit
def _fuse_mask_gate_kernel(
    left_ptr,            # float32 * (B,N,N,H)
    right_ptr,           # float32 * (B,N,N,H)
    mask_ptr,            # float32 * (B,N,N,1) – broadcasted mask
    left_gate_ptr,       # float32 * (B,N,N,H)
    right_gate_ptr,      # float32 * (B,N,N,H)
    B, N, H,             # static sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    """In‑place fusion of (mask * gate) onto left/right tensors."""
    pid = tl.program_id(0)
    total_elements = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # ---- decode linear offset ----
    # layout is (b, i, k, h)
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem = rem % (N * H)
    k = rem // H
    h = rem % H

    # ---- compute pointer offsets (element‑wise) ----
    left_off  = (b * stride_left_b  + i * stride_left_i  + k * stride_left_k  + h * stride_left_h)
    right_off = (b * stride_right_b + i * stride_right_i + k * stride_right_k + h * stride_right_h)

    mask_off  = (b * stride_mask_b  + i * stride_mask_i  + k * stride_mask_k)               # no h dim
    lgate_off = (b * stride_lgate_b + i * stride_lgate_i + k * stride_lgate_k + h * stride_lgate_h)
    rgate_off = (b * stride_rgate_b + i * stride_rgate_i + k * stride_rgate_k + h * stride_rgate_h)

    # ---- load values (mask broadcasted to H) ----
    left   = tl.load(left_ptr  + left_off,  mask=mask, other=0.0)
    right  = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    m      = tl.load(mask_ptr  + mask_off,  mask=mask, other=0.0)
    lgate  = tl.load(left_gate_ptr  + lgate_off, mask=mask, other=0.0)
    rgate  = tl.load(right_gate_ptr + rgate_off, mask=mask, other=0.0)

    # ---- fused multiply ----
    left  = left  * m * lgate
    right = right * m * rgate

    # ---- store back in‑place ----
    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left: torch.Tensor,
                         right: torch.Tensor,
                         mask: torch.Tensor,
                         left_gate: torch.Tensor,
                         right_gate: torch.Tensor) -> None:
    """
    Launch ``_fuse_mask_gate_kernel`` to apply
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    in‑place.  All tensors are contiguous float32 on the same CUDA device.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                     # good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    # strides in element counts (not bytes)
    sl = left.stride()
    sr = right.stride()
    sm = mask.stride()
    slg = left_gate.stride()
    srg = right_gate.stride()

    _fuse_mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        # left strides
        stride_left_b=sl[0], stride_left_i=sl[1], stride_left_k=sl[2], stride_left_h=sl[3],
        # right strides
        stride_right_b=sr[0], stride_right_i=sr[1], stride_right_k=sr[2], stride_right_h=sr[3],
        # mask strides (no h)
        stride_mask_b=sm[0], stride_mask_i=sm[1], stride_mask_k=sm[2],
        # left‑gate strides
        stride_lgate_b=slg[0], stride_lgate_i=slg[1], stride_lgate_k=slg[2], stride_lgate_h=slg[3],
        # right‑gate strides
        stride_rgate_b=srg[0], stride_rgate_i=srg[1], stride_rgate_k=srg[2], stride_rgate_h=srg[3],
        BLOCK=BLOCK,
    )
    # kernel mutates left/right in‑place; nothing to return.


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # always 128 in the benchmark set

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dim (the reference uses nn.LayerNorm)
    # ------------------------------------------------------------------
    eps = 1e-5
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    # keep everything in fp32
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * w_norm + b_norm

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x_norm, left_proj_w)           # (B,N,N,hidden_dim)
    right = F.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid) – still in FP32
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    # ------------------------------------------------------------------
    # 4. Optional mask handling
    # ------------------------------------------------------------------
    if mask is None:
        # a tensor of ones has the same shape as the pairwise mask
        mask = torch.ones(x.shape[:3], device=x.device, dtype=x.dtype)
    # broadcast to channel dim for the kernel
    mask = mask.unsqueeze(-1)                     # (B,N,N,1)

    # ------------------------------------------------------------------
    # 5. Fuse mask & gates with a tiny Triton kernel (in‑place)
    # ------------------------------------------------------------------
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 6. N³ contraction via batched matrix multiply
    #    out_h = left_h @ right_hᵀ   for each head h
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # (B, N, N, H) -> (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # cuBLAS batched GEMM, result (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm
    # ------------------------------------------------------------------
    eps_o = 1e-5
    w_outnorm = weights["to_out_norm.weight"]
    b_outnorm = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps_o)
    out_norm = out_norm * w_outnorm + b_outnorm

    # ------------------------------------------------------------------
    # 8. Apply output gate (broadcasted)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]           # (dim, hidden_dim)
    out_final = F.linear(out_norm, to_out_w)      # (B,N,N,dim)

    return out_final.to(torch.float32)
