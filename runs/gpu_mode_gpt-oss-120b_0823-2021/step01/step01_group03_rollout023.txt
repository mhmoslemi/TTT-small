"""
custom_kernel
==============

Forward implementation of the “outgoing” TriMul operator (AlphaFold‑3).

The costly N³ reduction (einsum) stays in PyTorch (torch.einsum on Hopper is
already highly optimized).  The memory‑bound mask‑application – a tensor of
shape (B, N, N, H) multiplied by (B, N, N) – is moved into a small Triton kernel
that works in‑place on both the left and right projected tensors.  All other
operations (layer‑norm, linear projections, gating, final linear) are kept in
PyTorch for simplicity and speed.

The kernel follows the launch conventions required by Triton 3.3.1 and
targets the NVIDIA H200 (SM 90).  Block sizes are chosen to be multiples of 32
so that the hidden dimension (always a multiple of 32 in the test suite) is
covered without shared‑memory pressure.
"""

import torch
import triton
import triton.language as tl
import math
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: apply a (B,N,N) mask (broadcast on the hidden dimension H)
# to a 4‑D tensor (B,N,N,H) in‑place.
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    x_ptr,          # ptr to tensor to be masked   (B,N,N,H)
    mask_ptr,       # ptr to mask                 (B,N,N)
    B, N, H,
    stride_x_b, stride_x_i, stride_x_k, stride_x_h,
    stride_m_b, stride_m_i, stride_m_k,
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    """
    One program per (batch, i‑tile).  Inside each program we iterate over the
    full K‑dimension in blocks, loading the mask once per (i,k) pair and
    broadcasting it over H.
    """
    pid = tl.program_id(0)

    # batch index and i‑tile index
    batch = pid // tl.cdiv(N, BLOCK_I)
    i_tile = pid % tl.cdiv(N, BLOCK_I)

    # offsets inside the tile
    offs_i = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_k = tl.arange(0, BLOCK_K)
    offs_h = tl.arange(0, BLOCK_H)

    # bounds
    mask_i = offs_i < N
    mask_k = offs_k < N
    mask_h = offs_h < H

    # ------------------------------------------------------------------
    # Load a (i,k) slice of the mask – shape (BLOCK_I, BLOCK_K)
    # ------------------------------------------------------------------
    mask = tl.load(
        mask_ptr
        + batch * stride_m_b
        + offs_i[:, None] * stride_m_i
        + offs_k[None, :] * stride_m_k,
        mask=mask_i[:, None] & mask_k[None, :],
        other=1.0,                     # when mask is all‑ones the kernel is a NOP
    )                               # (BLOCK_I, BLOCK_K)

    # ------------------------------------------------------------------
    # Broadcast mask over H and load the data slice (i,k,h)
    # ------------------------------------------------------------------
    x = tl.load(
        x_ptr
        + batch * stride_x_b
        + offs_i[:, None, None] * stride_x_i
        + offs_k[None, :, None] * stride_x_k
        + offs_h[None, None, :] * stride_x_h,
        mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
        other=0.0,
    )                               # (BLOCK_I, BLOCK_K, BLOCK_H)

    # ------------------------------------------------------------------
    # Apply mask (broadcasted on H) and store back
    # ------------------------------------------------------------------
    x = x * mask[:, :, None]        # broadcast mask to (i,k,1)
    tl.store(
        x_ptr
        + batch * stride_x_b
        + offs_i[:, None, None] * stride_x_i
        + offs_k[None, :, None] * stride_x_k
        + offs_h[None, None, :] * stride_x_h,
        x,
        mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
    )


# ----------------------------------------------------------------------
# Helper: fused LayerNorm for a 4‑D tensor (B,N,N,C) using weight/bias.
# ----------------------------------------------------------------------
def _layer_norm_4d(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # mean / var over the channel dimension
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0. unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dtype = torch.float32

    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    x = _layer_norm_4d(
        inp,
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Linear projections → (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # bias=False
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # ------------------------------------------------------------------
    # 3. Optional mask – moved to Triton.
    # ------------------------------------------------------------------
    if mask is None:
        # all‑ones mask – the kernel becomes a no‑op
        mask_f = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask_f = mask.to(dtype)

    # launch parameters – hidden_dim is always a multiple of 32 in the test suite
    BLOCK_I = 32
    BLOCK_K = 32
    BLOCK_H = 32

    stride_x_b, stride_x_i, stride_x_k, stride_x_h = left.stride()
    mask_stride_b, mask_stride_i, mask_stride_k = mask_f.stride()

    grid = (B * triton.cdiv(N, BLOCK_I),)   # one program per (batch, i‑tile)

    # left
    _apply_mask_kernel[grid](
        left,
        mask_f,
        B,
        N,
        hidden_dim,
        stride_x_b=stride_x_b,
        stride_x_i=stride_x_i,
        stride_x_k=stride_x_k,
        stride_x_h=stride_x_h,
        stride_m_b=mask_stride_b,
        stride_m_i=mask_stride_i,
        stride_m_k=mask_stride_k,
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    # right (same launch, same strides)
    _apply_mask_kernel[grid](
        right,
        mask_f,
        B,
        N,
        hidden_dim,
        stride_x_b=stride_x_b,
        stride_x_i=stride_x_i,
        stride_x_k=stride_x_k,
        stride_x_h=stride_x_h,
        stride_m_b=mask_stride_b,
        stride_m_i=mask_stride_i,
        stride_m_k=mask_stride_k,
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – cheap, stay in PyTorch
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(dtype)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(dtype)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(dtype)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ reduction via torch.einsum.
    #    Cast to bf16 for throughput, cast back to fp32 afterwards.
    # ------------------------------------------------------------------
    out = torch.einsum(
        "... i k d, ... j k d -> ... i j d",
        left.to(torch.bfloat16),
        right.to(torch.bfloat16),
    ).to(dtype)   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 6. Final LayerNorm, output gating and linear projection back to dim
    # ------------------------------------------------------------------
    out = _layer_norm_4d(
        out,
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
