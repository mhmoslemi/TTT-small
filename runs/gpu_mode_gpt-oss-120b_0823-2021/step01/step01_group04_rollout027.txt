"""
TriMul “outgoing” kernel (forward‑only).

The heavy N³ part of the TriMul module is a batched matrix‑multiply:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is exactly (left_h @ right_hᵀ) for each head h.
We therefore:
  1. Apply LayerNorm, the four linear projections and the three sigmoids in
     regular PyTorch (these are cheap and already well‑optimised).
  2. Fuse the per‑element mask, left‑gate and right‑gate multiplications with a
     tiny Triton kernel – this removes three separate element‑wise kernels and
     saves a memory round‑trip.
  3. Reshape the tensors so that every (batch, head) pair becomes one batch
     dimension and perform the N³ work with a single `torch.bmm`.  cuBLAS on the
     H200 is far faster than trying to write the N³ loop in Triton.
  4. Apply the final LayerNorm, output‑gate and linear projection in PyTorch.

Only the element‑wise fusion (step 2) is written in Triton; all other ops stay
in PyTorch which already uses mixed‑precision kernels where appropriate.
The final tensor is returned as float‑32 on the same device as the input.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel – fuse mask, left‑gate and right‑gate into the left/right
# tensors in‑place.  The kernel works on a flat index that spans
#   B * N * N * H  elements.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_fuse_kernel(
    left_ptr,          # [B, N, N, H]   float32
    right_ptr,         # [B, N, N, H]   float32
    mask_ptr,          # [B, N, N, 1]   float32
    left_gate_ptr,     # [B, N, N, H]   float32
    right_gate_ptr,    # [B, N, N, H]   float32
    B, N, H,           # compile‑time sizes
    stride_left_b, stride_left_n1, stride_left_n2, stride_left_h,
    stride_right_b, stride_right_n1, stride_right_n2, stride_right_h,
    stride_mask_b, stride_mask_n1, stride_mask_n2,
    stride_lgate_b, stride_lgate_n1, stride_lgate_n2, stride_lgate_h,
    stride_rgate_b, stride_rgate_n1, stride_rgate_n2, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # -------- decode linear offset into (b,i,k,h) --------
    # linear = (((b * N) + i) * N + k) * H + h
    b   = offs // (N * N * H)
    rest = offs % (N * N * H)
    i   = rest // (N * H)
    rest = rest % (N * H)
    k   = rest // H
    h   = rest % H

    # -------- compute pointer offsets (element‑wise) --------
    left_off   = (b * stride_left_b   +
                  i * stride_left_n1  +
                  k * stride_left_n2  +
                  h * stride_left_h)
    right_off  = (b * stride_right_b  +
                  i * stride_right_n1 +
                  k * stride_right_n2 +
                  h * stride_right_h)
    mask_off   = (b * stride_mask_b   +
                  i * stride_mask_n1  +
                  k * stride_mask_n2)          # mask has no channel dim
    lgate_off  = (b * stride_lgate_b  +
                  i * stride_lgate_n1 +
                  k * stride_lgate_n2 +
                  h * stride_lgate_h)
    rgate_off  = (b * stride_rgate_b  +
                  i * stride_rgate_n1 +
                  k * stride_rgate_n2 +
                  h * stride_rgate_h)

    # -------- load, fuse and store back --------
    left   = tl.load(left_ptr   + left_off,   mask=mask, other=0.0)
    right  = tl.load(right_ptr  + right_off,  mask=mask, other=0.0)
    m      = tl.load(mask_ptr   + mask_off,   mask=mask, other=0.0)
    lgate  = tl.load(left_gate_ptr + lgate_off, mask=mask, other=0.0)
    rgate  = tl.load(right_gate_ptr+ rgate_off, mask=mask, other=0.0)

    left   = left  * m * lgate
    right  = right * m * rgate

    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left: torch.Tensor,
                         right: torch.Tensor,
                         mask: torch.Tensor,
                         left_gate: torch.Tensor,
                         right_gate: torch.Tensor) -> None:
    """
    In‑place fusion of mask, left‑gate and right‑gate into ``left`` and ``right``.
    All tensors must be contiguous and on the same CUDA device.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024               # good occupancy on Hopper
    grid = (triton.cdiv(total, BLOCK),)

    # strides in element units (not bytes)
    sl = left.stride()
    sr = right.stride()
    sm = mask.stride()
    sg_l = left_gate.stride()
    sg_r = right_gate.stride()

    _tri_mul_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=sl[0], stride_left_n1=sl[1], stride_left_n2=sl[2], stride_left_h=sl[3],
        stride_right_b=sr[0], stride_right_n1=sr[1], stride_right_n2=sr[2], stride_right_h=sr[3],
        stride_mask_b=sm[0], stride_mask_n1=sm[1], stride_mask_n2=sm[2],
        stride_lgate_b=sg_l[0], stride_lgate_n1=sg_l[1], stride_lgate_n2=sg_l[2], stride_lgate_h=sg_l[3],
        stride_rgate_b=sg_r[0], stride_rgate_n1=sg_r[1], stride_rgate_n2=sg_r[2], stride_rgate_h=sg_r[3],
        BLOCK=BLOCK,
    )
    # kernel runs in‑place – nothing to return


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]  (float32)
        - mask         : torch.Tensor of shape [B, N, N]      (bool/float)
        - weights_dict : mapping of weight name → torch.Tensor.
        - config_dict  : contains ``dim`` and ``hidden_dim`` entries.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on the same device as input).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim, hidden_dim = cfg["dim"], cfg["hidden_dim"]
    device = x.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (no bias‑fusion needed)
    # ------------------------------------------------------------------
    norm_w = w["norm.weight"]
    norm_b = w["norm.bias"]
    # manual LN to avoid an extra kernel call; keep everything in fp32
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x_norm, w["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x_norm, w["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gates (sigmoid) – still in PyTorch; cheap compared to N³ work
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, w["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, w["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x_norm, w["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask + gates with a tiny Triton kernel
    # ------------------------------------------------------------------
    # mask shape -> (B,N,N,1) for broadcasting
    if mask is None:
        # when ``nomask`` is True the harness passes a tensor full of ones,
        # otherwise a real mask.  For safety we create a ones mask.
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2],
                          dtype=left.dtype, device=device)
    mask = mask.unsqueeze(-1)                     # (B,N,N,1)
    # ensure tensors are contiguous – Triton expects contiguous memory layout
    left      = left.contiguous()
    right     = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate= right_gate.contiguous()
    mask      = mask.contiguous()
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 5️⃣ N³ contraction via batched matrix‑multiply
    #    out_h = left_h @ right_hᵀ   for each head h
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # reshape to (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # cuBLAS batched GEMM (fp32); extremely fast on H200
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore original layout: (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm, output‑gate and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = w["to_out_norm.weight"]
    to_out_norm_b = w["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # apply output gate (broadcast over hidden dim)
    out_norm = out_norm * out_gate

    # final projection back to original dim
    out_final = F.linear(out_norm, w["to_out.weight"])

    # ensure float32 output (weights are float32, but keep the cast explicit)
    return out_final.to(torch.float32)
