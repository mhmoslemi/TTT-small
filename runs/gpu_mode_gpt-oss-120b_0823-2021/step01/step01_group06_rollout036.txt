"""
TriMul “outgoing” forward (no gradients).

* LayerNorm, Linear projections and gating are performed with PyTorch
  (they are memory‑bound and already fused by torch.compile).
* The only element‑wise operation that cannot be fused efficiently is the
  mask‑application on the two projected tensors.  We implement it with a
  tiny Triton kernel that works for any stride layout (contiguous or
  transposed) and respects the optional mask flag.
* The O(N³) contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
  is a batched matrix‑multiply per hidden dimension d.  On H200 a
  regular torch.bmm (FP32) is faster than a hand‑written reduction, so we
  keep it in PyTorch.

The kernel operates on a flattened view of a 4‑D tensor.  Given the
strides of the tensor (B,N,N,H) we can compute the address of a
linear index without having to decode every dimension on the host.
The mask is broadcasted along the hidden dimension H.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton: apply a broadcasted mask of shape [B,N,N,1] to a tensor of
# shape [B,N,N,H].  The kernel works for any contiguous or strided layout.
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    x_ptr,          # ptr to source tensor [B,N,N,H]
    mask_ptr,       # ptr to mask tensor   [B,N,N,1]
    out_ptr,        # ptr to destination   [B,N,N,H]
    total,          # total number of elements in x (B*N*N*H)
    mask_stride_b,  # stride of mask in batch dim (in elements)
    mask_stride_n,  # stride of mask in seq   dim (in elements)
    x_stride_b, x_stride_i, x_stride_j, x_stride_h,  # strides of x
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    mask = offs < total                                # guard

    # ----- compute multi‑dimensional indices for x -----
    # linear index -> (b,i,j,h)
    tmp = offs
    h = tmp % x_stride_h
    tmp = tmp // x_stride_h
    j = tmp % x_stride_j
    tmp = tmp // x_stride_j
    i = tmp % x_stride_i
    b = tmp // x_stride_i

    # address of the source element
    x_addr = x_ptr + b * x_stride_b + i * x_stride_i + j * x_stride_j + h

    # ----- compute corresponding mask address (mask has no h dim) -----
    # mask index = (b,i,j,0)  -> linear offset using its strides
    mask_addr = mask_ptr + b * mask_stride_b + i * mask_stride_n + j * mask_stride_n

    x_val   = tl.load(x_addr,   mask=mask, other=0.0)
    mask_val = tl.load(mask_addr, mask=mask, other=1.0)   # broadcasted along H

    tl.store(out_ptr + offs, x_val * mask_val, mask=mask)


# ----------------------------------------------------------------------
# Helper LayerNorm (bias + weight) – same as torch.nn.LayerNorm
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = layer_norm(inp, weights["norm.weight"], weights["norm.bias"])

    # ------------------------------------------------------------------
    # 2. Linear projections (dim -> hidden, bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask application – Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] for broadcasting
        mask_exp = mask.unsqueeze(-1).contiguous()
        total = left.numel()
        BLOCK = 128 * 1024                     # ~512 Ki elements (FP32)

        grid = (triton.cdiv(total, BLOCK),)

        # left
        apply_mask_kernel[grid](
            left.data_ptr(),
            mask_exp.data_ptr(),
            left.data_ptr(),          # in‑place store (allowed)
            total,
            mask_exp.stride(0), mask_exp.stride(1),   # mask strides (B,N)
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            BLOCK=BLOCK,
        )
        # right
        apply_mask_kernel[grid](
            right.data_ptr(),
            mask_exp.data_ptr(),
            right.data_ptr(),
            total,
            mask_exp.stride(0), mask_exp.stride(1),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            BLOCK=BLOCK,
        )

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – still in PyTorch
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM per hidden dimension.
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N)
    left_rs  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_rs = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_rs = torch.bmm(left_rs, right_rs.transpose(1, 2))

    # back to [B, N, N, hidden]
    out = out_rs.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear (hidden -> dim)
    # ------------------------------------------------------------------
    out = layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
