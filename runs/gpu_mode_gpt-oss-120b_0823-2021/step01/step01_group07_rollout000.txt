"""
TriMul “outgoing’’ forward pass (AlphaFold‑3).

*   Layer‑norm + linear / gating are kept in PyTorch (dim ≤ 384 ⇒ cheap).
*   The O(N³) reduction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is executed by a fused Triton GEMM‑like kernel.
*   left/right are cast to fp16, the kernel accumulates in fp32 for
    numerical stability and returns fp32.
*   The rest of the post‑processing (layer‑norm, gating, final linear) is
    performed again in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: tiled GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    # pointers
    left_ptr,          # [B, N, N, H]   fp16
    right_ptr,         # [B, N, N, H]   fp16 (will be read transposed on axis 2)
    out_ptr,           # [B, N, N, H]   fp32
    # problem size
    B, N, H,
    # strides (in elements)
    stride_lb, stride_lc, stride_lk, stride_lh,   # left  strides
    stride_rb, stride_rc, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oc, stride_ok, stride_oh,   # out   strides
    # tile sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute a tile of the output for a fixed (batch, head) pair.

    left  : [B, N, N, H]   (i,k)
    right : [B, N, N, H]   (j,k) → we load it as (k,j) to obtain the required
                                 contraction over k.
    out   : [B, N, N, H]   (i,j)
    """
    pid = tl.program_id(0)                 # linear index over (B, H)
    b = pid // H
    h = pid % H

    pid_m = tl.program_id(1)               # i‑tile
    pid_n = tl.program_id(2)               # j‑tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the current tile (left is [i,k], right is stored as [j,k]
    # but we want to read it as [k,j]).
    # ------------------------------------------------------------------
    left_tile_ptr = left_ptr + \
        b * stride_lb + \
        h * stride_lh + \
        offs_m[:, None] * stride_lc + \
        tl.arange(0, BLOCK_K)[None, :] * stride_lk

    right_tile_ptr = right_ptr + \
        b * stride_rb + \
        h * stride_rh + \
        tl.arange(0, BLOCK_K)[:, None] * stride_rk + \
        offs_n[None, :] * stride_rc

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # mask for the K dimension (out‑of‑bounds → zero)
        mask_k = cur_k < N

        # left:  (M,K)
        left = tl.load(
            left_tile_ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right: (K,N) – already transposed
        right = tl.load(
            right_tile_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # fp16 * fp16 → fp32 accumulation
        left = left.to(tl.float16)
        right = right.to(tl.float16)
        acc += tl.dot(left, right)          # (M,K)·(K,N) → (M,N)

        # advance pointers to next K‑block
        left_tile_ptr += BLOCK_K * stride_lk
        right_tile_ptr += BLOCK_K * stride_rk

    # ------------------------------------------------------------------
    # Store result (still fp32)
    # ------------------------------------------------------------------
    out_tile_ptr = out_ptr + \
        b * stride_ob + \
        h * stride_oh + \
        offs_m[:, None] * stride_oc + \
        offs_n[None, :] * stride_ok
    tl.store(
        out_tile_ptr,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]   fp16
    returns      : [B, N, N, H]   fp32
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in *elements* (not bytes)
    stride_lb, stride_lc, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rc, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oc, stride_ok, stride_oh = out.stride()

    # Tile sizes – tuned for SM‑90 (H200/H100)
    BLOCK_M = 64   # rows   of the output tile
    BLOCK_N = 64   # cols   of the output tile
    BLOCK_K = 32   # reduction size per iteration

    grid = (
        B * H,                     # program_id(0) → batch * head
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
    )

    _trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_lb=stride_lb,
        stride_lc=stride_lc,
        stride_lk=stride_lk,
        stride_lh=stride_lh,
        stride_rb=stride_rb,
        stride_rc=stride_rc,
        stride_rk=stride_rk,
        stride_rh=stride_rh,
        stride_ob=stride_ob,
        stride_oc=stride_oc,
        stride_ok=stride_ok,
        stride_oh=stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # 4 warps give good occupancy on Hopper
    )
    return out


# ----------------------------------------------------------------------
# Full custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing’’ TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

        - input_tensor : [B, N, N, dim]   torch.float32  (CUDA)
        - mask         : [B, N, N]       torch.float32 or None (0/1)
        - weights_dict : mapping name → torch.Tensor (float32)
          Expected keys:
              norm.weight, norm.bias,
              left_proj.weight, right_proj.weight,
              left_gate.weight, right_gate.weight, out_gate.weight,
              to_out_norm.weight, to_out_norm.bias,
              to_out.weight
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim, hidden_dim = cfg["dim"], cfg["hidden_dim"]
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 1. Input layer‑norm (dim is small → cheap in torch)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    #    we use a manual einsum to avoid allocating a full nn.Linear
    # ------------------------------------------------------------------
    def linear_no_bias(inp, w):
        # inp : [..., dim]   w : [hidden_dim, dim]
        return torch.einsum("...d,hd->...h", inp, w)

    left  = linear_no_bias(x, weights["left_proj.weight"])
    right = linear_no_bias(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional pair‑wise mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [..., N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(linear_no_bias(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_no_bias(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear_no_bias(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core reduction – Triton GEMM
    # ------------------------------------------------------------------
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # out : [B, N, N, hidden_dim]  (fp32)
    out = _trimul_gemm(left_f16, right_f16)

    # ------------------------------------------------------------------
    # 6. Post‑processing
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                         # gating after norm
    out = torch.einsum("...h,dh->...d", out, weights["to_out.weight"])

    # final tensor is already fp32
    return out
