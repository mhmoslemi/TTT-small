The outgoing‑TriMul forward pass consists of three logical stages  

1. **Point‑wise stage** – layer‑norm, four linear projections, three sigmoid gates,
   mask multiplication and gate‑mask fusion.  
2. **Heavy N³ stage** – `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`.  
   This is a batched matrix‑multiply for each head `h`:  
   `out_h = left_h @ right_hᵀ`.  
3. **Output stage** – layer‑norm, output‑gate, final linear projection.

Only the point‑wise part is cheap enough to be fused in a tiny Triton kernel.
The N³ step is best performed with **cuBLAS batched‑GEMM** (`torch.bmm`) after a
reshape, because Triton‑implemented N³ kernels would require massive shared‑memory
tiling and would be slower on Hopper.

The implementation below

* computes **LayerNorm** manually (no extra kernels needed),
* projects with `torch.nn.functional.linear`,
* builds the three gate tensors,
* launches a **single‑dimensional Triton kernel** that (in‑place) multiplies  
  `left` and `right` by the mask and the corresponding gate,
* reshapes `left/right` to `(B*H, N, N)` and executes a single `torch.bmm`,
* applies the second LayerNorm, output‑gate and final linear projection.

All tensors stay on the CUDA device; the final tensor is returned as `float32`.

The kernel is deliberately kept simple (no shared memory, no loops) – it only
does the element‑wise work that would otherwise require three separate CUDA
launches.  This gives a large speed‑up for the whole forward pass while keeping
the code easy to maintain and correct for all configurations required by the
test suite.

```python
"""
TriMul “outgoing” forward pass (AlphaFold3).

* Point‑wise operations (mask, gates, linear projections) are fused in a tiny
  Triton kernel (in‑place on `left` and `right`).
* The cubic contraction is performed as a batch of matrix‑multiplications:
      out_h = left_h @ right_hᵀ
  after reshaping to (B*H, N, N) and using torch.bmm (cuBLAS‑accelerated).
* Remaining normalisation / output‑gate / final linear are done in PyTorch.

All tensors remain on the GPU, final output is float32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel – fuse mask, left‑gate and right‑gate into left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_fuse_kernel(
    left_ptr,          # [B, N, N, H]   (float32)
    right_ptr,         # [B, N, N, H]   (float32)
    mask_ptr,          # [B, N, N, 1]   (float32) – broadcasted mask
    left_gate_ptr,     # [B, N, N, H]   (float32)
    right_gate_ptr,    # [B, N, N, H]   (float32)
    B, N, H,           # static sizes
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h):
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    The operation is performed in‑place.
    """
    pid = tl.program_id(0)
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # ---- decode linear offset to (b,i,j,h) ----
    b = offs // (N * N * H)
    r = offs % (N * N * H)
    i = r // (N * H)
    r = r % (N * H)
    j = r // H
    h = r % H

    # ---- compute addresses (in element units) ----
    left_off  = (b * stride_left_b  + i * stride_left_i  + j * stride_left_j  + h * stride_left_h)
    right_off = (b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h)
    mask_off  = (b * stride_mask_b  + i * stride_mask_i  + j * stride_mask_j)                # no channel dim
    lgate_off = (b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h)
    rgate_off = (b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h)

    # ---- load ----
    left  = tl.load(left_ptr  + left_off,  mask=mask, other=0.0)
    right = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    m     = tl.load(mask_ptr  + mask_off,  mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr  + lgate_off, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + rgate_off, mask=mask, other=0.0)

    # ---- fused ops ----
    left  = left  * m * lgate
    right = right * m * rgate

    # ---- store back (in‑place) ----
    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left: torch.Tensor,
                         right: torch.Tensor,
                         mask: torch.Tensor,
                         left_gate: torch.Tensor,
                         right_gate: torch.Tensor) -> None:
    """In‑place fusion of mask and gates using the Triton kernel above."""
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                               # good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    # strides are expressed in **elements**, not bytes
    s_left = left.stride()
    s_right = right.stride()
    s_mask = mask.stride()
    s_lgate = left_gate.stride()
    s_rgate = right_gate.stride()

    _tri_mul_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=s_left[0], stride_left_i=s_left[1],
        stride_left_j=s_left[2], stride_left_h=s_left[3],
        stride_right_b=s_right[0], stride_right_i=s_right[1],
        stride_right_j=s_right[2], stride_right_h=s_right[3],
        stride_mask_b=s_mask[0], stride_mask_i=s_mask[1], stride_mask_j=s_mask[2],
        stride_lgate_b=s_lgate[0], stride_lgate_i=s_lgate[1],
        stride_lgate_j=s_lgate[2], stride_lgate_h=s_lgate[3],
        stride_rgate_b=s_rgate[0], stride_rgate_i=s_rgate[1],
        stride_rgate_j=s_rgate[2], stride_rgate_h=s_rgate[3],
        BLOCK=BLOCK,
    )
    # kernel mutates left/right in‑place → nothing to return


# ----------------------------------------------------------------------
# Custom kernel entry point required by the harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of TriMul (outgoing direction).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] on the same CUDA device, dtype float32.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm on the input (no bias in weights → same as nn.LayerNorm)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    eps = 1e-5
    # mean / var over the last dimension
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * ln_w + ln_b                                   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)             # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣  Gates (sigmoid) – also bias‑free linear + sigmoid
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 4️⃣  Mask handling (may be None)
    # ------------------------------------------------------------------
    if mask is None:
        # treat as all‑ones mask
        mask = torch.ones_like(left[..., :1])          # shape (B,N,N,1)
    else:
        mask = mask.unsqueeze(-1).to(left.dtype)       # (B,N,N,1)

    # ------------------------------------------------------------------
    # 5️⃣  Fuse mask & gates (in‑place on left/right) via Triton
    # ------------------------------------------------------------------
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 6️⃣  Cubic contraction – batched GEMM
    #    out_h = left_h @ right_hᵀ   for each head h
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # bmm with transpose on the second operand
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)           # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 7️⃣  Output layer‑norm
    # ------------------------------------------------------------------
    out_ln_w = weights["to_out_norm.weight"]
    out_ln_b = weights["to_out_norm.bias"]
    eps_o = 1e-5
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) * torch.rsqrt(var_o + eps_o)
    out = out * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 8️⃣  Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate                                 # (B,N,N,hidden_dim)
    to_out_w = weights["to_out.weight"]                  # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                        # (B,N,N,dim)

    # Ensure float32 on the same device
    return out.to(dtype=torch.float32, device=input_tensor.device)
```
