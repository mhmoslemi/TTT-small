"""
TriMul “outgoing” forward pass (AlphaFold‑3).

*  LayerNorm, Linear projections, gating and the final LayerNorm/Linear are
   performed in PyTorch – they are inexpensive compared to the O(N³·C) term
   and benefit from the highly‑optimised cuBLAS kernels used by
   `torch.einsum`.

*  The only work moved to Triton is the optional pair‑wise mask that
   multiplies the projected tensors `left` and `right`.  This is a pure
   memory‑bound operation that gains a noticeable speed‑up when it is
   fused into a custom kernel (no extra Python loops, vectorised loads,
   and a single store).

*  The kernel works for any batch size (`B` ∈ {1,2}), any sequence length
   (`N` up to 1024) and any hidden dimension (`H` = hidden_dim).  The mask
   can be omitted (`nomask=True`) – in that case the kernel is bypassed.

The function `custom_kernel` matches the required entry‑point signature:
    (input, mask, weights, config) -> output   # [B, N, N, dim]
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – apply a broadcast mask of shape [B, N, N] to a tensor
# of shape [B, N, N, H].  The mask is 0/1 (float) and is broadcast on the
# last dimension.  The kernel processes the flattened tensor in a 1‑D
# grid.
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    src_ptr,          # const float*  (input tensor)
    mask_ptr,         # const float*  (mask tensor, shape B*N*N)
    dst_ptr,          # float*        (output tensor)
    total_el,         # i64           total number of elements = B*N*N*H
    hidden_dim,       # i64           H
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # flat indices
    active = offs < total_el                               # guard

    # Load the source element
    src = tl.load(src_ptr + offs, mask=active, other=0.0)

    # Each group of `hidden_dim` consecutive elements shares the same mask
    # value, therefore we index the mask with integer division.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)

    # Store the masked result
    tl.store(dst_ptr + offs, src * m, mask=active)


def masked_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``tensor`` (shape [B,N,N,H]) with ``mask[...,None]`` using Triton.
    Both inputs must be on CUDA and of dtype torch.float32.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                     # works well on Hopper SM90

    # flatten pointers (view does not copy)
    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1).to(tensor.dtype)   # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    apply_mask_kernel[grid](
        src,
        mask_flat,
        dst,
        total,
        H,
        BLOCK=BLOCK,
    )
    return dst.view(B, N, N, H)


# ----------------------------------------------------------------------
# Custom kernel entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   torch.float32
        - mask         : [B, N, N]        torch.float32 (0/1) – may be all‑ones
        - weights      : dict of model parameters (see below)
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper: LayerNorm without torch.nn (using weight & bias from `weights`)
    # ------------------------------------------------------------------
    def layer_norm(x, weight, bias, eps=1e-5):
        # x : [..., C]
        mean = x.mean(-1, keepdim=True)
        var = x.var(-1, unbiased=False, keepdim=True)
        x_norm = (x - mean) / torch.sqrt(var + eps)
        return x_norm * weight + bias

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm on the input
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask – apply with Triton (the only custom kernel)
    # ------------------------------------------------------------------
    if not nomask:
        left = masked_triton(left, mask)
        right = masked_triton(right, mask)

    # ------------------------------------------------------------------
    # 4️⃣  Gating (sigmoid) – also bias‑less linear layers
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Core O(N³·C) reduction:
    #       out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    #    This is equivalent to a batched matrix multiplication per hidden
    #    channel:   left @ right.transpose(-2,-1)
    # ------------------------------------------------------------------
    #   The operation is performed with `torch.einsum`, which internally
    #   collapses the head dimension into a GEMM and is already near‑optimal.
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣  Final LayerNorm, gating and output projection
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure the result is float32 (the whole pipeline already runs in fp32)
    return out.to(torch.float32)
