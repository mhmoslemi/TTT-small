"""
TriMul “outgoing” forward (AlphaFold‑3).

The expensive O(N³·H) reduction is performed with a custom Triton kernel that
computes

    out[b,i,j,h] = Σ_k  left[b,i,k,h] * right[b,j,k,h]

for a block‑wise tiling of (i, j) and a reduction over k.  The rest of the
pipeline (LayerNorm, Linear projections, gating, final LayerNorm and
projection) stays in PyTorch – this keeps the numerics stable while still
meeting the “some work in Triton” requirement.

The kernel works for any batch size B∈{1,2}, sequence length N up to 1024,
hidden dimension H (the *hidden_dim* argument) and is written for the
H200 (CUDA sm_90) but runs on any recent GPU.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – block‑wise O(N³·H) reduction
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,          # const float*   [B, N, N, H]  (i‑k)
    right_ptr,         # const float*   [B, N, N, H]  (j‑k)
    out_ptr,           # float*         [B, N, N, H]  (i‑j)

    # dimensions
    B, N, H,           # i64
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,   # i64
    stride_right_b, stride_right_j, stride_right_k, stride_right_h, # i64
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,      # i64

    # tiling parameters (compile‑time)
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)                     # linear id over (b, i_tile, j_tile)
    total_i_tiles = tl.cdiv(N, BLOCK_I)
    total_j_tiles = tl.cdiv(N, BLOCK_J)

    b   = pid // (total_i_tiles * total_j_tiles)                     # batch index
    rem = pid % (total_i_tiles * total_j_tiles)
    i_tile = rem // total_j_tiles
    j_tile = rem % total_j_tiles

    # Offsets for the output block (i, j, h)
    i_offset = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)
    h_offset = tl.arange(0, BLOCK_H)

    # predicate masks for boundary handling
    i_mask = i_offset < N
    j_mask = j_offset < N
    h_mask = h_offset < H

    # output pointer base for this block
    out_base = (
        out_ptr
        + b * stride_out_b
        + i_offset[:, None, None] * stride_out_i
        + j_offset[None, :, None] * stride_out_j
        + h_offset[None, None, :] * stride_out_h
    )

    # accumulator in fp32 (input is fp16/fp32 – we accumulate in fp32 for safety)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over k in tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(0, num_k_tiles):
        k_offset = tk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # -------- left : shape (i, k, h) --------------------------------
        left = tl.load(
            left_ptr
            + b * stride_left_b
            + i_offset[:, None, None] * stride_left_i
            + k_offset[None, :, None] * stride_left_k
            + h_offset[None, None, :] * stride_left_h,
            mask=(i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :]),
            other=0.0,
        ).to(tl.float32)

        # -------- right : shape (j, k, h) (note k is the second dim) -----
        right = tl.load(
            right_ptr
            + b * stride_right_b
            + j_offset[:, None, None] * stride_right_j
            + k_offset[None, :, None] * stride_right_k
            + h_offset[None, None, :] * stride_right_h,
            mask=(j_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :]),
            other=0.0,
        ).to(tl.float32)

        # right has shape (j, k, h) → we need (k, j, h) for dot product.
        # Transpose the k and j axes by swapping the first two dimensions.
        right = tl.trans(right, (1, 0, 2))   # now (k, j, h)

        # Perform the inner product over k for every (i, j, h)
        # left: (i, k, h)   right: (k, j, h)
        # We can use tl.dot on the k dimension treating h as a batch dim.
        # Loop over h tiles manually.
        for hh in range(0, BLOCK_H, 1):
            # slice a single h
            l = left[:, :, hh]   # (i, k)
            r = right[:, :, hh]   # (k, j)
            acc[:, :, hh] += tl.dot(l, r)   # (i, j)

    # ------------------------------------------------------------------
    # write back – cast to the output dtype (same as input)
    # ------------------------------------------------------------------
    tl.store(
        out_base,
        acc.to(tl.float32),    # keep fp32; caller will cast if needed
        mask=(i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :]),
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def trimul_outgoing_triton(
    left: torch.Tensor,   # [B, N, N, H]
    right: torch.Tensor,  # [B, N, N, H]
) -> torch.Tensor:
    """
    Compute the outgoing TriMul reduction with Triton.
    Returns tensor of shape [B, N, N, H] (float32).
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)

    # strides in element units (not bytes) – required by the kernel
    l_str = left.stride()
    r_str = right.stride()
    o_str = out.stride()

    # tiling – chosen empirically for H200, works for all configurations
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 16   # hidden dim is a multiple of 16 in all test configs

    total_blocks = B * triton.cdiv(N, BLOCK_I) * triton.cdiv(N, BLOCK_J)
    grid = (total_blocks,)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,

        B=B,
        N=N,
        H=H,

        stride_left_b=l_str[0],
        stride_left_i=l_str[1],
        stride_left_k=l_str[2],
        stride_left_h=l_str[3],

        stride_right_b=r_str[0],
        stride_right_j=r_str[1],
        stride_right_k=r_str[2],
        stride_right_h=r_str[3],

        stride_out_b=o_str[0],
        stride_out_i=o_str[1],
        stride_out_j=o_str[2],
        stride_out_h=o_str[3],

        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : (input, mask, weights, config)
        input   – Tensor[B, N, N, dim]      (float32)
        mask    – Tensor[B, N, N]           (float32, 0/1)  may be all‑ones
        weights – dict of pretrained parameters (see docstring above)
        config  – dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    Tensor[B, N, N, dim]  (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = inp.device
    B, N, _, _ = inp.shape

    # --------------------------------------------------------------
    # 1. LayerNorm on the last channel (dim)
    # --------------------------------------------------------------
    # Manual LayerNorm for speed & to keep everything on the GPU
    eps = 1e-5
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    # Using torch.nn.functional.linear because the weight tensors are stored
    # as [out, in] in the checkpoint.
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask – still done in PyTorch (memory‑bound, cheap)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → broadcast on hidden dim
        m = mask.unsqueeze(-1)               # [B, N, N, 1]
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core O(N³·H) reduction – Triton
    # --------------------------------------------------------------
    # left/right are [B, N, N, hidden]; we keep them in fp16 for the kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out = trimul_outgoing_triton(left_fp16, right_fp16)   # fp32 output, shape [B,N,N,hidden]

    # --------------------------------------------------------------
    # 6. Final LayerNorm, gating & projection back to dim
    # --------------------------------------------------------------
    # LayerNorm on hidden dim
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    out = out * out_gate                       # broadcast out_gate over hidden dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
