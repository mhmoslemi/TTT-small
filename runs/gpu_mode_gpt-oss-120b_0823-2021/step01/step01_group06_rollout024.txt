"""
TriMul “outgoing” kernel (forward only).

Algorithm
---------
1. Layer‑norm on the raw pair representation   (torch, fused)
2. Linear projections to the hidden dimension   (torch)
3. Optional mask application – a tiny element‑wise Triton kernel
4. Gating (sigmoid) on the projected tensors   (torch)
5. Core N³ reduction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply for every hidden channel d.
   It is implemented in pure Triton using a classic WMMA‑style tiling
   (BLOCK_M × BLOCK_N output tile, BLOCK_K reduction tile) and
   tl.dot for the inner product.  The kernel works on the view
   (B*hidden, N, N) to keep the launch grid 3‑D.
6. Output layer‑norm, gate, and final linear projection   (torch)

All tensors are kept in float32 (the H200 is bandwidth‑rich, FP32 GEMM is
fast enough and avoids loss‑of‑precision).  Only the two kernels below
are written in Triton; the rest uses highly‑optimised torch kernels that
are already fused by torch.compile().
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# 1️⃣ Element‑wise mask kernel  (used only when nomask == False)
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    x_ptr,          # [B,N,N,H]   – input tensor
    mask_ptr,       # [B,N,N,1]   – broadcastable mask
    out_ptr,        # [B,N,N,H]   – output tensor
    total_elems,    # B*N*N*H
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Linear address arithmetic – the stride information is baked into the
    # pointer passed from the host.  Because mask has a trailing singleton
    # dimension, its address is simply offs // H (integer division).
    H = tl.constexpr(tl.load(x_ptr + 0, mask=tl.zeros((1,), dtype=tl.int32)))  # dummy to satisfy compiler
    # The above line is never executed; it only tells Triton that H is a compile‑time
    # constant.  We therefore compute the mask pointer manually.
    # Compute offset for mask (broadcast over hidden dim)
    mask_offs = offs // tl.program_id(0).dtype.itemsize  # integer division by hidden size is safe because hidden is constexpr in host

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)
    tl.store(out_ptr + offs, x * m, mask=mask)


# ----------------------------------------------------------------------
# 2️⃣ Batched matrix‑multiply kernel for the N³ reduction
# ----------------------------------------------------------------------
@triton.jit
def trmul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                       # pointers
    N,                                          # sequence length
    stride_a_batch, stride_a_i, stride_a_k,    # a: [B*H, N, N]   (i,k)
    stride_b_batch, stride_b_j, stride_b_k,    # b: [B*H, N, N]   (j,k) – we will read transposed
    stride_c_batch, stride_c_i, stride_c_j,    # c: output [B*H, N, N] (i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_batch = tl.program_id(0)          # B*H dimension
    pid_m = tl.program_id(1)              # i‑tile
    pid_n = tl.program_id(2)              # j‑tile

    # Compute the start offsets of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out‑of‑bounds (only needed for the final partial tiles)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load tiles of A and B (B is accessed as transposed: [j, k])
        a = tl.load(
            a_batch_ptr
            + (offs_m[:, None] * stride_a_i)
            + (offs_k[None, :] * stride_a_k),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_batch_ptr
            + (offs_n[:, None] * stride_b_j)
            + (offs_k[None, :] * stride_b_k),
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0,
        )
        # dot product over K
        acc += tl.dot(a, b)

    # Write back result
    c = acc.to(tl.float32)
    tl.store(
        c_batch_ptr
        + (offs_m[:, None] * stride_c_i)
        + (offs_n[None, :] * stride_c_j),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# 3️⃣ Helper: fused LayerNorm (no bias handling inside – we add bias manually)
# ----------------------------------------------------------------------
def fused_layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: [..., C]   float32
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# 4️⃣ Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # --------------------------------------------------------------
    x = fused_layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (dim -> hidden)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim) – Triton kernel
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N,1]
        mask_exp = mask.unsqueeze(-1).contiguous()
        total = left.numel()
        BLOCK = 131072  # ~512 KiB per launch
        grid = (triton.cdiv(total, BLOCK),)

        left_masked = torch.empty_like(left)
        right_masked = torch.empty_like(right)

        apply_mask_kernel[grid](
            left.data_ptr(),
            mask_exp.data_ptr(),
            left_masked.data_ptr(),
            total,
            BLOCK=BLOCK,
        )
        apply_mask_kernel[grid](
            right.data_ptr(),
            mask_exp.data_ptr(),
            right_masked.data_ptr(),
            total,
            BLOCK=BLOCK,
        )
        left, right = left_masked, right_masked

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – still in PyTorch
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core N³ reduction via Triton batched GEMM
    # --------------------------------------------------------------
    # reshape to (B*hidden, N, N) for the kernel
    left_view = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_view = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_view.reshape(B * hidden, N, N)
    right_flat = right_view.reshape(B * hidden, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # launch parameters – tuned for H200 (64‑bit registers, shared memory)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden,                         # batch*hidden dimension
        triton.cdiv(N, BLOCK_M),           # i‑tiles
        triton.cdiv(N, BLOCK_N),           # j‑tiles
    )

    trmul_matmul_kernel[grid](
        left_flat.data_ptr(),
        right_flat.data_ptr(),
        out_flat.data_ptr(),
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(2),   # note: we need stride for j (dim 2) because we treat right as transposed
        right_flat.stride(1),   # stride over k
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B,N,N,hidden]
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm, gate, final linear
    # --------------------------------------------------------------
    out = fused_layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
