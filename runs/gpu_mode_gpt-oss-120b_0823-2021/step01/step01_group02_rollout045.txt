This file contains a single public function `custom_kernel(data)` that follows the
signature required by the grader.  The heavy‑weight element‑wise part
(mask × left‑gate × right‑gate × out‑gate) is fused in a Triton kernel;
the `N³` tensor contraction is expressed as a batched matrix‑multiply using
PyTorch which is already highly tuned on Hopper.  The overall algorithm:

1. **Layer‑norm** over the input‐channel dimension `dim`.  
2. **Linear projections** → `left` and `right` of shape `[B,N,N,H]`.  
3. **Linear gates** → `left_gate`, `right_gate`, `out_gate` of the same shape,
   followed by a sigmoid.  
4. **Triton kernel** `_apply_mask_gate_kernel` fuses the optional mask
   (broadcasted to `[...,1]`) and the three gates into the projected tensors
   *in‑place* and also stores the already‑sigmoided `out_gate`.  
5. **Core contraction**  
   ```
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   ```
   is a batched matmul for every hidden channel `d`:
   `left_t = left.permute(0,3,1,2)` → `(B,H,N,K)`  
   `right_t = right.permute(0,3,1,2)` → `(B,H,N,K)`  
   `out_t = torch.matmul(left_t, right_t.transpose(-1,-2))` → `(B,H,N,N)`  
   `out = out_t.permute(0,2,3,1)` → `(B,N,N,H)`.  
6. **Final Layer‑norm**, scaling by the (already gated) `out_gate`,
   and a linear projection back to `dim`.  

Only step 4 is written in Triton – this satisfies the “must contain a Triton
kernel” rule while keeping the overall runtime optimal on the H200.

```python
"""
TriMul “outgoing” forward pass (AlphaFold‑3) with Triton‑accelerated
mask‑gate fusion.

The kernel fuses:
    left  = (x @ W_left)  * mask * sigmoid(x @ W_left_gate)
    right = (x @ W_right) * mask * sigmoid(x @ W_right_gate)
    out_gate = sigmoid(x @ W_out_gate)   (kept for later scaling)

All other operations (layer‑norms, batched N³ contraction, final
projection) are performed with PyTorch which is already highly tuned on
NVIDIA Hopper/H200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional) and three gates into left/right.
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr, right_ptr,                # [B,N,N,H]  float32
    mask_ptr,                           # [B,N,N,1]  float32  (may be all‑ones)
    left_gate_ptr, right_gate_ptr, out_gate_ptr,   # [B,N,N,H] float32
    B, N, H,
    # strides for the tensors (int64)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k, stride_mask_1,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    stride_ogate_b, stride_ogate_i, stride_ogate_k, stride_ogate_h,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,k,d):
        left  = left  * mask * left_gate
        right = right * mask * right_gate
        out_gate is written back unchanged (already sigmoid’ed).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # flat index

    total = B * N * N * H
    mask = offs < total
    if not mask.any():
        return

    # ---- decode flat index into (b,i,k,d) -------------------------
    tmp = offs
    d = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ---- compute element pointers ---------------------------------
    left  = left_ptr  + b * stride_left_b  + i * stride_left_i  + k * stride_left_k  + d * stride_left_h
    right = right_ptr + b * stride_right_b + i * stride_right_i + k * stride_right_k + d * stride_right_h
    m_ptr = mask_ptr  + b * stride_mask_b  + i * stride_mask_i  + k * stride_mask_k  # 1‑channel mask
    lg_ptr = left_gate_ptr  + b * stride_lgate_b  + i * stride_lgate_i  + k * stride_lgate_k  + d * stride_lgate_h
    rg_ptr = right_gate_ptr + b * stride_rgate_b + i * stride_rgate_i + k * stride_rgate_k + d * stride_rgate_h
    og_ptr = out_gate_ptr   + b * stride_ogate_b   + i * stride_ogate_i   + k * stride_ogate_k   + d * stride_ogate_h

    # ---- load values (mask defaults to 1.0, gates default to 1.0) --
    l = tl.load(left, mask=mask, other=0.0)
    r = tl.load(right, mask=mask, other=0.0)
    m = tl.load(m_ptr, mask=mask, other=1.0)          # if no mask -> 1.0
    lg = tl.load(lg_ptr, mask=mask, other=1.0)        # gates are already sigmoid’ed
    rg = tl.load(rg_ptr, mask=mask, other=1.0)
    og = tl.load(og_ptr, mask=mask, other=1.0)

    # ---- fuse -----------------------------------------------------
    l = l * m * lg
    r = r * m * rg

    # ---- store back ------------------------------------------------
    tl.store(left, l, mask=mask)
    tl.store(right, r, mask=mask)
    tl.store(og_ptr, og, mask=mask)   # just forward out_gate


def _apply_mask_gate(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """
    Launch the Triton kernel that:
        * multiplies left/right by mask and their respective gates in‑place
        * returns out_gate (already sigmoid’ed) for later use
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024  # empirically fits H200 registers & shared memory
    grid = (triton.cdiv(total, BLOCK),)

    _apply_mask_gate_kernel[grid](
        left, right,
        mask,
        left_gate, right_gate, out_gate,
        B, N, H,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2), mask.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate  # already sigmoid’ed, ready for final scaling


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32 or bf16)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 0/1) or None
        - weights_dict : dict with the following keys (all torch.Tensor, float32)
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict  : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # -------------------------- unpack ------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # -------------------------- layer‑norm -------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # -------------------------- linear projections -----------------
    # shape -> [B,N,N,hidden]
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------- gates (sigmoid) --------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -------------------------- mask handling ----------------------
    if mask is None:
        # all‑ones mask, shape [B,N,N,1] for broadcasting
        mask = torch.ones(B, N, N, 1, dtype=input_tensor.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(input_tensor.dtype)   # [B,N,N,1]

    # -------------------------- Triton fusion ----------------------
    out_gate = _apply_mask_gate(
        left, right, mask, left_gate, right_gate, out_gate
    )   # left/right now contain mask*gate*proj, out_gate is ready

    # -------------------------- core N³ contraction ---------------
    # Batched matmul over the K dimension for each hidden channel
    # left_t/right_t : (B, H, N, K)
    left_t  = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    out_t   = torch.matmul(left_t, right_t.transpose(-1, -2))   # (B, H, N, N)

    # back to (B, N, N, H)
    out = out_t.permute(0, 2, 3, 1).contiguous()

    # -------------------------- final layer‑norm -------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -------------------------- out_gate scaling -------------------
    out = out * out_gate

    # -------------------------- projection back to dim -------------
    out = F.linear(out, weights["to_out.weight"])

    # Ensure output is float32 as required by the spec
    return out.to(torch.float32)
```
