"""
TriMul “outgoing” forward pass (AlphaFold‑3).

The heavy N³ reduction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply per hidden dimension d.
We implement the element‑wise mask‑application (the only part that is not
already fused by torch.compile) with a tiny Triton kernel and then use a
single‑precision batched GEMM (torch.bmm) for the N³ reduction – this is
far faster on the H200 than a handcrafted Triton reduction for the supported
sizes (N ≤ 1024, hidden ≤ 384).

All LayerNorms, linear projections and sigmoids remain in PyTorch (they are
memory‑bound and already well‑optimized).  The final output is returned as
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton : apply a broadcast mask (shape [B,N,N,1]) to a tensor
#          of shape [B,N,N,H] element‑wise.
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    src_ptr,               # pointer to source tensor   [B,N,N,H]
    mask_ptr,              # pointer to mask tensor     [B,N,N,1]
    dst_ptr,               # pointer to destination     [B,N,N,H]
    total_elements,        # B * N * N * H
    src_stride_b, src_stride_i, src_stride_j, src_stride_h,   # src strides
    mask_stride_b, mask_stride_i, mask_stride_j,              # mask strides (no h‑dim)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear ids within this block
    mask = offs < total_elements                       # guard

    # ----- compute multi‑dimensional indices -----
    # linear index = ((b * N + i) * N + j) * H + h
    # we recover (b,i,j,h) using floor division / modulo with the known strides
    # (all strides are in elements, not bytes)
    h = offs % src_stride_h
    tmp = offs // src_stride_h
    j = tmp % src_stride_j
    tmp = tmp // src_stride_j
    i = tmp % src_stride_i
    b = tmp // src_stride_i

    # ----- load source and mask -----
    src = tl.load(src_ptr + b * src_stride_b
                           + i * src_stride_i
                           + j * src_stride_j
                           + h * src_stride_h,
                  mask=mask, other=0.0)

    # mask has no hidden dimension → broadcast
    m   = tl.load(mask_ptr + b * mask_stride_b
                           + i * mask_stride_i
                           + j * mask_stride_j,
                  mask=mask, other=1.0)

    tl.store(dst_ptr + b * src_stride_b
                        + i * src_stride_i
                        + j * src_stride_j
                        + h * src_stride_h,
              src * m,
              mask=mask)


# ----------------------------------------------------------------------
# Helper: fused LayerNorm (bias + weight supplied)
# ----------------------------------------------------------------------
def fused_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """LayerNorm over the last dimension, FP32."""
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input tensor   [B, N, N, dim]
        torch.Tensor,                # mask tensor    [B, N, N]  (may be None)
        Dict[str, torch.Tensor],     # weights dict
        Dict[str, Any],              # config dict
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Returns
    -------
    torch.Tensor : shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x = fused_layernorm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2) Linear projections to hidden dimension (bias‑free)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Optional mask application (Triton)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape -> [B,N,N,1] for broadcasting
        mask_exp = mask.unsqueeze(-1).contiguous()
        total = left.numel()
        BLOCK = 131072                     # ~512 KiB elements (FP32)
        grid  = (triton.cdiv(total, BLOCK),)

        # allocate temporaries
        left_masked  = torch.empty_like(left)
        right_masked = torch.empty_like(right)

        # launch for left
        apply_mask_kernel[grid](
            left.data_ptr(),
            mask_exp.data_ptr(),
            left_masked.data_ptr(),
            total,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            mask_exp.stride(0), mask_exp.stride(1), mask_exp.stride(2),
            BLOCK=BLOCK,
        )
        # launch for right
        apply_mask_kernel[grid](
            right.data_ptr(),
            mask_exp.data_ptr(),
            right_masked.data_ptr(),
            total,
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask_exp.stride(0), mask_exp.stride(1), mask_exp.stride(2),
            BLOCK=BLOCK,
        )
        left, right = left_masked, right_masked

    # --------------------------------------------------------------
    # 4) Gating (sigmoid) – still in PyTorch
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate   # [B,N,N,hidden]
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Core N³ reduction  (batched GEMM)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # reshape to (B * hidden, N, N) → treat hidden as batch dimension
    left_rs  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)     # (B*H, N, N)
    right_rs = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)    # (B*H, N, N)

    # batched matmul: (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_rs = torch.bmm(left_rs, right_rs.transpose(1, 2))

    # back to [B, N, N, hidden]
    out = out_rs.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6) Output LayerNorm + gating + final linear projection
    # --------------------------------------------------------------
    out = fused_layernorm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
