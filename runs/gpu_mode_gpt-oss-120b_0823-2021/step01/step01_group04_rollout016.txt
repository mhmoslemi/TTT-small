"""
TriMul “outgoing” forward pass (AlphaFold3 style).

The heavy N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batch of independent matrix multiplications for each hidden channel:
    out_h = left_h @ right_hᵀ   (B×H, N, N) @ (B×H, N, N)ᵀ
We therefore reshape the tensors so that a single cuBLAS `bmm` (or Triton‑based
batched GEMM) can do all the work.

The element‑wise part (LayerNorm, linear proj, gating, mask) is cheap but
memory‑bound.  We fuse the mask, left‑gate and right‑gate multiplications in a
tiny Triton kernel that works **in‑place** on `left` and `right`.  This avoids
extra temporaries and improves bandwidth utilisation on the H200.

The whole forward pass stays on GPU, uses fp32 for numerical stability and
returns a float‑32 tensor of shape `[B, N, N, dim]`.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel – fuse mask, left‑gate and right‑gate into left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_fuse_kernel(
    left_ptr,          # [B, N, N, H]  float32
    right_ptr,         # [B, N, N, H]  float32
    mask_ptr,          # [B, N, N, 1]  float32 (broadcasted)
    left_gate_ptr,     # [B, N, N, H]  float32
    right_gate_ptr,    # [B, N, N, H]  float32
    B, N, H,           # static sizes
    # strides (in elements, not bytes)
    stride_left_b, stride_left_n, stride_left_k, stride_left_h,
    stride_right_b, stride_right_n, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_n, stride_mask_k,
    stride_lgate_b, stride_lgate_n, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_n, stride_rgate_k, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    """In‑place: left *= mask * left_gate ;  right *= mask * right_gate"""
    pid = tl.program_id(0)                     # 1‑D launch
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear indices this program works on
    mask = offs < total

    # ---- decode linear index into (b, i, k, h) ----
    # layout: ((b * N + i) * N + k) * H + h
    b = offs // (N * N * H)
    rest = offs % (N * N * H)
    i = rest // (N * H)
    rest = rest % (N * H)
    k = rest // H
    h = rest % H

    # ----- compute pointer offsets -----
    left_off  = (b * stride_left_b  + i * stride_left_n  + k * stride_left_k  + h * stride_left_h)
    right_off = (b * stride_right_b + i * stride_right_n + k * stride_right_k + h * stride_right_h)
    mask_off  = (b * stride_mask_b  + i * stride_mask_n  + k * stride_mask_k)                # no channel dim
    lgate_off = (b * stride_lgate_b + i * stride_lgate_n + k * stride_lgate_k + h * stride_lgate_h)
    rgate_off = (b * stride_rgate_b + i * stride_rgate_n + k * stride_rgate_k + h * stride_rgate_h)

    # ----- load values (mask is broadcasted to the channel dim) -----
    left  = tl.load(left_ptr  + left_off,  mask=mask, other=0.0)
    right = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    m     = tl.load(mask_ptr  + mask_off,  mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr  + lgate_off, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + rgate_off, mask=mask, other=0.0)

    # ----- fused element‑wise ops -----
    left  = left  * m * lgate
    right = right * m * rgate

    # ----- store back in‑place -----
    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left: torch.Tensor,
                         right: torch.Tensor,
                         mask: torch.Tensor,
                         left_gate: torch.Tensor,
                         right_gate: torch.Tensor) -> None:
    """
    Apply ``left *= mask * left_gate`` and ``right *= mask * right_gate`` in‑place
    using a 1‑D Triton kernel.
    All tensors are expected to be contiguous and on the same CUDA device.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                      # 1 k threads – good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    # strides in **elements**, not bytes
    sl = left.stride()
    sr = right.stride()
    sm = mask.stride()
    slg = left_gate.stride()
    srg = right_gate.stride()

    _tri_mul_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=sl[0], stride_left_n=sl[1], stride_left_k=sl[2], stride_left_h=sl[3],
        stride_right_b=sr[0], stride_right_n=sr[1], stride_right_k=sr[2], stride_right_h=sr[3],
        stride_mask_b=sm[0], stride_mask_n=sm[1], stride_mask_k=sm[2],
        stride_lgate_b=slg[0], stride_lgate_n=slg[1], stride_lgate_k=slg[2], stride_lgate_h=slg[3],
        stride_rgate_b=srg[0], stride_rgate_n=srg[1], stride_rgate_k=srg[2], stride_rgate_h=srg[3],
        BLOCK=BLOCK,
    )
    # No return – tensors are modified in‑place


# ----------------------------------------------------------------------
# Helper: manual LayerNorm (fp32, bias & weight supplied)
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + eps)
    return x * weight + bias


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, dim]  (float32, cuda)
        - mask         : torch.Tensor [B, N, N]      (bool or float, cuda)
        - weights_dict : mapping of the nine parameter tensors
        - config_dict  : must contain keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, cuda)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # called ``c_z`` in the paper

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (pre‑norm)
    # ------------------------------------------------------------------
    x = _layer_norm(x,
                    weight=weights["norm.weight"],
                    bias=weights["norm.bias"],
                    eps=1e-5)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    # torch.nn.functional.linear does y = x @ W.T  (W shape [out, in])
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gating logits → sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask + gates – fuse into a single Triton kernel
    # ------------------------------------------------------------------
    # mask may be None (nomask) – treat as all‑ones
    if mask is None:
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2],
                         dtype=x.dtype, device=x.device)
    mask = mask.unsqueeze(-1)          # (B,N,N,1) – broadcastable to channel dim
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 5️⃣ N³ contraction via batched matrix multiplication
    #    left_h @ right_hᵀ for every hidden channel h
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # reshape to (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched GEMM – cuBLAS fast path, works in fp16/fp32 automatically
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # restore shape (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm + output gate
    # ------------------------------------------------------------------
    out = _layer_norm(out,
                      weight=weights["to_out_norm.weight"],
                      bias=weights["to_out_norm.bias"],
                      eps=1e-5)

    out = out * out_gate                       # broadcast over hidden dim

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    # ensure FP32 output (all upstream ops stayed FP32)
    return out.to(torch.float32)
