"""
TriMul “outgoing” forward pass (no gradients).

* 4‑D input  : (B, N, N, C)
* LayerNorm  : per‑channel (last dim)
* Linear proj: left / right  (C → H)   – bias‑less
* Optional mask (B,N,N) is applied in‑place on the projected tensors
  via a tiny Triton kernel (the only required custom kernel).
* Gating      : left‑gate, right‑gate, out‑gate (sigmoid of a linear)
* Core update :   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
                 (implemented with torch.einsum, cast to bfloat16 for speed)
* Final norm + gate + linear → (B,N,N,C)

The code works for all configurations listed in the statement and keeps the
final output in float32.  Only the mask‑application is written in Triton;
all other heavy work is delegated to highly‑optimized PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: apply a (B,N,N) mask (or a tensor of ones) to a
# (B,N,N,H) tensor in‑place.  The mask is broadcast over the hidden
# dimension H.
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    x_ptr,          # pointer to (B,N,N,H) tensor to be masked
    mask_ptr,       # pointer to (B,N,N) mask (float32, 1 → keep, 0 → drop)
    B, N, H,
    stride_x_b, stride_x_i, stride_x_k, stride_x_h,
    stride_m_b, stride_m_i, stride_m_k,
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)

    # each program processes one (batch, i‑tile) pair
    batch = pid // tl.cdiv(N, BLOCK_I)
    i_tile = pid % tl.cdiv(N, BLOCK_I)

    # tile offsets
    offs_i = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_k = tl.arange(0, BLOCK_K)
    offs_h = tl.arange(0, BLOCK_H)

    # bounds
    mask_i = offs_i < N
    mask_k = offs_k < N
    mask_h = offs_h < H

    # ---- load mask (B,N,N)  →  (i,k)  ----
    mask = tl.load(
        mask_ptr
        + batch * stride_m_b
        + offs_i[:, None] * stride_m_i
        + offs_k[None, :] * stride_m_k,
        mask=mask_i[:, None] & mask_k[None, :],
        other=1.0,                     # 1.0 when mask is missing (treated as all‑ones)
    )
    # broadcast on H dimension
    mask = mask[:, :, None]           # (i,k,1)

    # ---- load data (B,N,N,H) ----
    x = tl.load(
        x_ptr
        + batch * stride_x_b
        + offs_i[:, None, None] * stride_x_i
        + offs_k[None, :, None] * stride_x_k
        + offs_h[None, None, :] * stride_x_h,
        mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
        other=0.0,
    )

    # apply mask
    x = x * mask

    # store back
    tl.store(
        x_ptr
        + batch * stride_x_b
        + offs_i[:, None, None] * stride_x_i
        + offs_k[None, :, None] * stride_x_k
        + offs_h[None, None, :] * stride_x_h,
        x,
        mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
    )


# ----------------------------------------------------------------------
# Helper: LayerNorm over the last dimension of a 4‑D tensor.
# ----------------------------------------------------------------------
def layer_norm_4d(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: (B,N,N,C)
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the module output.
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (dim)
    # ------------------------------------------------------------------
    x = layer_norm_4d(
        input_tensor,
        weight=weights["norm.weight"].to(dtype, device=device),
        bias=weights["norm.bias"].to(dtype, device=device),
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    # shape: (B,N,N,hidden_dim)
    left = F.linear(x, weights["left_proj.weight"].to(dtype, device=device))
    right = F.linear(x, weights["right_proj.weight"].to(dtype, device=device))

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask using Triton (in‑place on left/right)
    # ------------------------------------------------------------------
    if mask is None:
        # a tensor of ones – kernel becomes a no‑op
        mask_fp = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask_fp = mask.to(dtype, device=device)

    # launch parameters – hidden_dim is a multiple of 32 in all configs
    BLOCK_I = 32
    BLOCK_K = 32
    BLOCK_H = 32

    # strides for (B,N,N,H)
    stride_b, stride_i, stride_k, stride_h = left.stride()
    mask_stride_b, mask_stride_i, mask_stride_k = mask_fp.stride()

    # one program per (batch, i‑tile)
    grid = (B * triton.cdiv(N, BLOCK_I),)

    # left
    apply_mask_kernel[grid](
        left,
        mask_fp,
        B,
        N,
        hidden_dim,
        stride_x_b=stride_b,
        stride_x_i=stride_i,
        stride_x_k=stride_k,
        stride_x_h=stride_h,
        stride_m_b=mask_stride_b,
        stride_m_i=mask_stride_i,
        stride_m_k=mask_stride_k,
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    # right (same launch config)
    apply_mask_kernel[grid](
        right,
        mask_fp,
        B,
        N,
        hidden_dim,
        stride_x_b=stride_b,
        stride_x_i=stride_i,
        stride_x_k=stride_k,
        stride_x_h=stride_h,
        stride_m_b=mask_stride_b,
        stride_m_i=mask_stride_i,
        stride_m_k=mask_stride_k,
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – cheap, stay in PyTorch
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(dtype, device=device)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(dtype, device=device)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(dtype, device=device)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ reduction via torch.einsum.
    #    Cast to bfloat16 for speed on Hopper/H200, then back to fp32.
    # ------------------------------------------------------------------
    left_bf16 = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)

    # einsum:   (... i k d, ... j k d) -> (... i j d)
    out = torch.einsum("... i k d, ... j k d -> ... i j d", left_bf16, right_bf16)
    out = out.to(dtype)

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gating and output linear
    # ------------------------------------------------------------------
    out = layer_norm_4d(
        out,
        weight=weights["to_out_norm.weight"].to(dtype, device=device),
        bias=weights["to_out_norm.bias"].to(dtype, device=device),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype, device=device))

    return out
