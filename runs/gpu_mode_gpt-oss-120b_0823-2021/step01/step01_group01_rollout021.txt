```python
"""
TriMul outgoing kernel (AlphaFold‑3 style).

The expensive part is the 3‑D “outer‑product” over the sequence dimension

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

for every batch `b` and hidden channel `h`.  This is exactly a batched
matrix‑multiply `left_h @ right_h.T`.  The implementation below tiles the
M‑, N‑ and K‑axes (i, j, k) with a modest `BLOCK_M/N = 64` and `BLOCK_K = 32`
so that each block fits comfortably in the H200 shared‑memory budget while
still providing enough arithmetic intensity to keep the Hopper tensor‑cores
busy.

* data layout   : (B, H, N, N) in FP16 (contiguous)
* accumulation : FP32 (to keep the final result numerically stable)
* write‑back    : FP16, later up‑cast to FP32 in PyTorch.

The rest of the TriMul pipeline (LayerNorm, Linear, gating, final
 LayerNorm + Linear) stays in PyTorch – moving those to Triton would give
little extra speed but would complicate the implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: batched mat‑mul for each hidden channel
# -------------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,          # const fp16   [B, H, N, N]
    right_ptr,         # const fp16   [B, H, N, N]
    out_ptr,           #      fp16   [B, H, N, N]
    B, N, H,           # sizes (int32)
    # strides
    stride_lb, stride_lh, stride_ln, stride_lk,
    stride_rb, stride_rh, stride_rn, stride_rk,
    stride_ob, stride_oh, stride_on, stride_ok,
    # tile sizes (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute   out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

    The kernel is launched with a 1‑D grid where each program instance
    handles one (b, h, tile_i, tile_j) block.
    """
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1) decode pid → (b, h, tile_i, tile_j)
    # --------------------------------------------------------------
    num_tile_i = tl.cdiv(N, BLOCK_M)
    num_tile_j = tl.cdiv(N, BLOCK_N)

    tile_j = pid % num_tile_j
    pid //= num_tile_j
    tile_i = pid % num_tile_i
    pid //= num_tile_i
    h = pid % H
    b = pid // H

    # --------------------------------------------------------------
    # 2) compute global offsets for the current tile
    # --------------------------------------------------------------
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) loop over K dimension in tiles
    # --------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left:   [i, k]   (row i, column k)
        l_ptr = (left_ptr
                 + b * stride_lb
                 + h * stride_lh
                 + offs_i[:, None] * stride_ln
                 + offs_k[None, :] * stride_lk)
        l = tl.load(l_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right transposed: we need (k, j) because the formula uses right[b,h,j,k]
        r_ptr = (right_ptr
                 + b * stride_rb
                 + h * stride_rh
                 + offs_k[:, None] * stride_rk
                 + offs_j[None, :] * stride_rn)
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # accumulate dot product over K
        acc += tl.dot(l, r)          # (M,K) @ (K,N) -> (M,N)

    # --------------------------------------------------------------
    # 4) write result back (FP16)
    # --------------------------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_ob
                + h * stride_oh
                + offs_i[:, None] * stride_on
                + offs_j[None, :] * stride_ok)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper that glues everything together
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Only the N³ “outer‑product” is executed in Triton; all LayerNorm,
    Linear, and gating operations stay in PyTorch (FP32).  The function
    returns a tensor of shape [B, N, N, dim] (float32).
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, cfg = data
    device = input_tensor.device
    dtype  = torch.float32

    B, N, _, dim = input_tensor.shape          # x is [B,N,N,dim]
    hidden = cfg["hidden_dim"]                 # = c_z in the spec

    # -----------------------------------------------------------------
    # 1) LayerNorm + Linear projections (still in PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"].to(dtype),
                     bias=weights["norm.bias"].to(dtype))

    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 2) Optional mask (mask may be None)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # (B,N,N,1)
        left  = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 3) Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4) Rearrange to [B, H, N, N] and cast to fp16 for the kernel
    # -----------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output buffer in fp16
    out_fp16 = torch.empty_like(left_fp16, device=device)

    # -----------------------------------------------------------------
    # 5) Triton launch parameters
    # -----------------------------------------------------------------
    BLOCK_M = 64   # rows  tile
    BLOCK_N = 64   # cols  tile
    BLOCK_K = 32   # reduction tile

    # grid dimensions: one program per (b, h, tile_i, tile_j)
    num_tile_i = triton.cdiv(N, BLOCK_M)
    num_tile_j = triton.cdiv(N, BLOCK_N)
    total_progs = B * hidden * num_tile_i * num_tile_j
    grid = (total_progs,)

    _trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden,
        # strides (B,H,N,N) layout
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        out_fp16.stride(0), out_fp16.stride(1), out_fp16.stride(2), out_fp16.stride(3),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6) Back to (B,N,N,hidden) FP32 and finish the pipeline in PyTorch
    # -----------------------------------------------------------------
    out = out_fp16.to(dtype).permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden)

    # final LayerNorm over hidden dimension
    out = F.layer_norm(out,
                       normalized_shape=(hidden,),
                       weight=weights["to_out_norm.weight"].to(dtype),
                       bias=weights["to_out_norm.bias"].to(dtype))

    # multiply by out_gate (broadcast over i,j)
    out = out * out_gate

    # linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
```

### Why this version is faster than the previous one
| Lesson | How it is applied / why it does **not** apply |
|-------|-----------------------------------------------|
| **Limit shared‑memory usage** | Tile sizes (`64×64×32`) keep the per‑SM shared‑memory well below the H200 limit (≈ 64 KB). No extra `num_stages` is needed. |
| **Match stride unpacking** | Strides are unpacked exactly for the 4‑D layout (`B, H, N, N`). No assumption of a 3‑D tensor; this avoids the *“incorrect number of strides”* runtime error seen in earlier attempts. |
| **Avoid unnecessary fp32‑to‑fp16 casts inside the loop** | We load directly as fp16, promote once to fp32 (`.to(tl.float32)`) and keep the accumulator in fp32. This is the minimal per‑iteration work. |
| **Launch grid based on tile count** | The grid is `B * H * ceil(N/64) * ceil(N/64)`. Earlier code built a grid with a single dimension that was far larger than needed and caused launch‑overhead and low occupancy. |
| **Keep LayerNorm / Linear in PyTorch** | Those operations are memory‑bandwidth bound and already highly optimized in PyTorch/ATen; moving them to Triton would not give a measurable speedup but would increase kernel complexity. |
| **Mask handling** | Mask is applied **before** the Triton call (in FP32), avoiding an extra branch inside the kernel. When `nomask=True` the extra multiplication is a no‑op. |

The kernel now respects all Triton constraints (no `continue`, proper `tl.constexpr` usage, correct stride handling) and returns a **float32** tensor ready for downstream layers.
