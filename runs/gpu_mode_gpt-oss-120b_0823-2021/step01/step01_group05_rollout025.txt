"""
TriMul “outgoing” forward (AlphaFold‑3).

The expensive O(N³·H) reduction

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

is performed in a fused Triton kernel.  All cheap
operations (layer‑norm, linear projections, sigmoidal gates and the final
layer‑norm + output projection) stay in PyTorch for numerical stability and
speed.  The kernel works for any batch size (≤2), any sequence length N and
hidden dimension H up to 384.  It processes one (i,j) pair per thread,
accumulates the dot‑product over k in blocks and writes a vector of length
H (vectorised via `tl.arange`).  Mixed‑precision (FP16 accumulation) is used
internally, the final result is cast back to FP32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# Triton kernel: batched outer‑product reduction over the k‑axis
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const float*  [B,N,N,H]
    right_ptr,         # const float*  [B,N,N,H]
    out_ptr,           # float*        [B,N,N,H]
    B, N, H,           # i64 scalars
    stride_left_bn, stride_left_bk, stride_left_bh,   # strides for left  (b,n,k,h)
    stride_right_bn, stride_right_bj, stride_right_bh,   # strides for right (b,j,k,h)
    stride_out_bn, stride_out_bi, stride_out_bj, stride_out_bh,  # out (b,i,j,h)
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Compute `out = Σₖ left[...,k,:] * right[...,k,:]`."""
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # i‑block
    pid_j = tl.program_id(2)          # j‑block

    # thread coordinates within the (i,j) block
    i = pid_i * BLOCK_K + tl.arange(0, BLOCK_K)
    j = pid_j * BLOCK_K + tl.arange(0, BLOCK_K)

    # mask out‑of‑bounds (supports any N)
    mask_i = i < N
    mask_j = j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # offsets for the (i,j) pair we are responsible for
    # we compute one (i0, j0) per thread; here we let each thread compute a
    # *single* i and j (no inner block) – this keeps register pressure low.
    i0 = tl.program_id(1) * BLOCK_K + tl.arange(0, BLOCK_K)[:1]   # scalar broadcast
    j0 = tl.program_id(2) * BLOCK_K + tl.arange(0, BLOCK_K)[:1]

    # early‑exit for out‑of‑range threads
    if i0 >= N or j0 >= N:
        return

    # offsets to the first element of the (i0,j0) vector we will write
    out_offset = (
        pid_b * stride_out_bn
        + i0 * stride_out_bi
        + j0 * stride_out_bj
    )
    # initialise accumulator in FP16 (faster) then cast later
    acc = tl.zeros([BLOCK_H], dtype=tl.float16)

    # loop over the k‑dimension in blocks
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        k_mask = cur_k < N

        # load left[b, i0, k, :]  -> shape (BLOCK_K, BLOCK_H)
        left_offset = (
            pid_b * stride_left_bn
            + i0 * stride_left_bk
            + cur_k[:, None] * stride_left_bh
        )
        left = tl.load(left_ptr + left_offset,
                       mask=k_mask[:, None],
                       other=0.0).to(tl.float16)          # (K, H)

        # load right[b, j0, k, :] -> shape (BLOCK_K, BLOCK_H)
        right_offset = (
            pid_b * stride_right_bn
            + j0 * stride_right_bj
            + cur_k[:, None] * stride_right_bh
        )
        right = tl.load(right_ptr + right_offset,
                        mask=k_mask[:, None],
                        other=0.0).to(tl.float16)         # (K, H)

        # element‑wise multiply and sum over k
        prod = left * right                         # (K, H)
        acc += tl.sum(prod, axis=0)                 # (H,)

    # write result (cast back to FP32)
    tl.store(out_ptr + out_offset,
             acc.to(tl.float32),
             mask=mask_i & mask_j)


# --------------------------------------------------------------
# Helper to launch the kernel
# --------------------------------------------------------------
def trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]  (float32)
    returns      : out   : [B, N, N, H]  (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # choose block sizes (tuned for H≤384, N up to 1024)
    BLOCK_K = 32               # k‑tile
    BLOCK_H = 128              # hidden‑dim tile (must divide H or be rounded)

    # stride unpacking (torch uses row‑major)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    grid = (
        B,                                 # batch
        triton.cdiv(N, BLOCK_K),           # i‑blocks
        triton.cdiv(N, BLOCK_K),           # j‑blocks
    )
    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=H,
        stride_left_bn=stride_left[0],
        stride_left_bk=stride_left[2],
        stride_left_bh=stride_left[3],
        stride_right_bn=stride_right[0],
        stride_right_bj=stride_right[2],
        stride_right_bh=stride_right[3],
        stride_out_bn=stride_out[0],
        stride_out_bi=stride_out[1],
        stride_out_bj=stride_out[2],
        stride_out_bh=stride_out[3],
        BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
    )
    return out


# --------------------------------------------------------------
# Entry‑point used by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : (input, mask, weights, config)
        input   – Tensor[B, N, N, dim]      (float32)
        mask    – Tensor[B, N, N]          (float32, 0/1) – may be all‑ones.
        weights – dict containing all pretrained params.
        config  – dict with keys ``dim`` and ``hidden_dim`` as in the
                  reference implementation.

    Returns
    -------
    Tensor[B, N, N, dim]  (float32)
    """
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (mirrors nn.LayerNorm)
    # ------------------------------------------------------------------
    eps = 1e-5
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) and optional mask
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    if not nomask:
        # broadcast mask on the hidden dimension
        mask_b = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_b
        right = right * mask_b

    # ------------------------------------------------------------------
    # 3️⃣ Gating (sigmoid) – also applied before the reduction
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Core O(N³·H) reduction in Triton
    # ------------------------------------------------------------------
    # left/right shape: [B, N, N, hidden]
    out = trimul_outgoing_triton(left, right)          # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 5️⃣ Final LayerNorm, gate and output linear
    # ------------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
