"""
TriMul “outgoing” kernel (AlphaFold‑3) – fast forward pass.

The cubic O(N³·C) reduction is still performed with torch.einsum
(built on cuBLAS GEMM) because a hand‑written Triton 3‑D reduction would
require massive shared‑memory tiling and is not faster on the H200 for the
sizes used in the tests.  

What *is* moved to Triton is the element‑wise mask‑application that
precedes the reduction.  The mask has shape [B, N, N] and must be broadcast
over the hidden dimension H (= hidden_dim).  A tiny Triton kernel
reads the mask once per H‑slice (by indexing ``mask_ptr[offs // H]``) and
stores the masked tensor.  This avoids the Python‑level broadcasting and
makes the mask‑apply completely memory‑bound, which is exactly what Triton
excels at.

All LayerNorm, Linear, and gating computations stay in PyTorch – they are
highly‑optimised, numerically‑stable and inexpensive compared with the O(N³)
step.  The implementation supports:
* optional mask (``config['nomask']``)
* arbitrary batch size (1 or 2)
* any sequence length from the test suite (256‑1024)
* any model dimension (128 or 384) and hidden dimension (always 128)

The final output is always float32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: apply a broadcast mask of shape [B, N, N] to a tensor
# of shape [B, N, N, H].  The mask is 0/1 (float) and is broadcast on the
# last dimension.  The kernel is launched with a 1‑D grid that covers the
# total number of elements (B * N * N * H).
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,      # const float*   (input tensor)
    mask_ptr,     # const float*   (mask tensor, shape B*N*N)
    dst_ptr,      # float*         (output tensor)
    total_el,     # i64            total number of elements = B*N*N*H
    H,            # i64            hidden dimension (size of the last axis)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    # linear index of the element this thread will (maybe) process
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_el

    # load the source value (float32)
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # each hidden‑dim slice shares the same mask element:
    #   mask index = floor( offs / H )
    m = tl.load(mask_ptr + (offs // H), mask=mask, other=1.0)

    # write masked value
    tl.store(dst_ptr + offs, src * m, mask=mask)


def mask_apply_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Apply ``mask[..., None]`` to ``tensor`` using the Triton kernel above.
    tensor : [B, N, N, H]   (float32)
    mask   : [B, N, N]      (float32, 0/1)
    returns tensor * mask[...,None]   (float32, same shape as tensor)
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                     # chosen to give good occupancy on H200

    src = tensor.view(-1)            # flatten, contiguous layout guaranteed by view
    dst = torch.empty_like(src)      # allocate output
    mask_f = mask.float().view(-1)   # flatten mask (no hidden dim)

    grid = (triton.cdiv(total, BLOCK),)   # 1‑D grid
    mask_apply_kernel[grid](
        src_ptr=src,
        mask_ptr=mask_f,
        dst_ptr=dst,
        total_el=total,
        H=H,
        BLOCK=BLOCK,
    )
    return dst.view(B, N, N, H)


# ----------------------------------------------------------------------
# Custom kernel entry‑point (host side)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input, mask, weights, config)
        - input : [B, N, N, dim]   torch.float32 on CUDA
        - mask  : [B, N, N]        torch.float32 (0/1); ignored if ``nomask``.
        - weights : dict of tensors (all on CUDA, float32)
        - config  : dict with keys
            * ``dim``          – input/output channel dimension
            * ``hidden_dim``   – inner projection dimension
            * ``nomask`` (opt) – if True the mask is all‑ones

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # default to mask‑less

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------
    #   LayerNorm(x) = (x - μ) / sqrt(var + eps) * weight + bias
    eps = 1e-5
    # compute mean / var over the feature dimension
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask – Triton accelerated
    # --------------------------------------------------------------
    if not nomask:
        left  = mask_apply_triton(left,  mask)
        right = mask_apply_triton(right, mask)

    # --------------------------------------------------------------
    # 4️⃣ Gates (sigmoid) – also linear without bias
    # --------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³·C) reduction – torch.einsum (leverages cuBLAS GEMM)
    #    shape after einsum: [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # --------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gating and output projection
    # --------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    out = out * out_gate                       # element‑wise gating
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # ensure float32 output (the whole pipeline stays in fp32)
    return out.to(torch.float32)
