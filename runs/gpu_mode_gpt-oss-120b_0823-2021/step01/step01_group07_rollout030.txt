"""
TriMul “outgoing” forward pass (AlphaFold‑3) – fast Triton implementation.

Algorithm
---------
1️⃣ Layer‑norm on the last (dim) channel – done in PyTorch (dim ≤ 768,
   cheap and avoids a custom reduction).
2️⃣ Two bias‑free linear projections (left / right) and four gating
   projections – also in PyTorch (tiny matrix‑multiplies).
3️⃣ Optional pair‑wise mask – element‑wise mul in PyTorch.
4️⃣ **Core O(N³) reduction**:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM where the right operand is transposed on its
   second sequence axis.  We implement it as a tiled GEMM in Triton:
   * each program_id(0) processes one (batch, head) pair,
   * program_id(1) / program_id(2) tile the i‑ and j‑dimensions.
   * fp16 arithmetic is used for the dot‑product, accumulation in fp32
     (numerical‑stable and fast on Hopper SM‑90).
5️⃣ Layer‑norm over the hidden dimension, out‑gate, final linear –
   back in PyTorch.

Only the N³ reduction is in Triton – this gives a huge speed‑up while
keeping the code simple and correct for all supported shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – tiled GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    # pointers
    left_ptr,          # [B, N, N, H]   (fp16)
    right_ptr,         # [B, N, N, H]   (fp16) – will be read transposed on dim‑2
    out_ptr,           # [B, N, N, H]   (fp32)
    # problem sizes
    B, N, H,
    # strides (in elements, not bytes)
    stride_lb, stride_lc, stride_lk, stride_lh,   # left  : B, i, k, h
    stride_rb, stride_rc, stride_rk, stride_rh,   # right : B, j, k, h  (k is 2nd seq dim)
    stride_ob, stride_oc, stride_ok, stride_oh,   # out   : B, i, j, h
    # tile sizes (constexpr)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute a single (i‑tile, j‑tile) of the output for a fixed (batch, head).
    The right tensor is accessed as if it were transposed on its two
    sequence axes (j ↔ k).  Accumulation is performed in fp32.
    """
    pid = tl.program_id(0)                 # linear index over (B, H)
    b = pid // H
    h = pid % H

    pid_m = tl.program_id(1)               # i‑tile index
    pid_n = tl.program_id(2)               # j‑tile index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑coordinates
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑coordinates

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the beginning of the tile for the current (b,h)
    # ------------------------------------------------------------------
    # left:   [b, i, k, h]
    left_tile_ptr = left_ptr + b * stride_lb \
                               + h * stride_lh \
                               + offs_m[:, None] * stride_lc \
                               + tl.arange(0, BLOCK_K)[None, :] * stride_lk

    # right is logically [b, j, k, h] but we need (k, j) → load as (k, j)
    # right:  [b, k, j, h] after the transposition
    right_tile_ptr = right_ptr + b * stride_rb \
                                 + h * stride_rh \
                                 + tl.arange(0, BLOCK_K)[:, None] * stride_rk \
                                 + offs_n[None, :] * stride_rc

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑reduction loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # ---- load left (M×K) -------------------------------------------------
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N
        left = tl.load(left_tile_ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # ---- load right (K×N) – already transposed ----------------------------
        right = tl.load(right_tile_ptr,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        # fp16 dot, fp32 accumulation
        left  = left.to(tl.float16)
        right = right.to(tl.float16)
        acc += tl.dot(left, right)          # (M,K)·(K,N) → (M,N)

        # advance K‑dimension pointers
        left_tile_ptr  += BLOCK_K * stride_lk
        right_tile_ptr += BLOCK_K * stride_rk

    # ------------------------------------------------------------------
    # Store the result (still fp32)
    # ------------------------------------------------------------------
    out_tile_ptr = out_ptr + b * stride_ob \
                               + h * stride_oh \
                               + offs_m[:, None] * stride_oc \
                               + offs_n[None, :] * stride_ok
    tl.store(out_tile_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM used by TriMul.

    Arguments
    ---------
    left, right : torch tensors of shape [B, N, N, H] and dtype torch.float16.
                   `right` is interpreted with its second sequence axis
                   transposed (i.e. we compute left @ right.transpose(1,2)).
    Returns
    -------
    out : torch.float32 tensor of shape [B, N, N, H]
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # element‑wise strides (in elements)
    stride_lb, stride_lc, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rc, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oc, stride_ok, stride_oh = out.stride()

    # Tile sizes – tuned for Hopper SM‑90 (shared‑mem ~ 64 KB per SM)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                     # program_id(0) → (batch, head)
        triton.cdiv(N, BLOCK_M),  # i‑tiles
        triton.cdiv(N, BLOCK_N),  # j‑tiles
    )

    _trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=H,
        stride_lb=stride_lb, stride_lc=stride_lc, stride_lk=stride_lk, stride_lh=stride_lh,
        stride_rb=stride_rb, stride_rc=stride_rc, stride_rk=stride_rk, stride_rh=stride_rh,
        stride_ob=stride_ob, stride_oc=stride_oc, stride_ok=stride_ok, stride_oh=stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,               # 4 warps gives good occupancy on H200
    )
    return out


# ----------------------------------------------------------------------
# Full custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        * input_tensor : [B, N, N, dim]  torch.float32 on CUDA
        * mask         : [B, N, N]       torch.float32 (0/1) or ``None``
        * weights_dict : mapping name → torch.Tensor (fp32)
        * config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the last channel (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free) → shape [B,N,N,hidden_dim]
    # --------------------------------------------------------------
    # Helper for a weight matrix of shape (hidden_dim, dim)
    def linear_no_bias(t, w):
        # t : [..., dim] ; w : [hidden_dim, dim]
        return torch.einsum("...d,hd->...h", t, w)

    left  = linear_no_bias(x, weights["left_proj.weight"])
    right = linear_no_bias(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden_dim)
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)            # [..., N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – also bias‑free linear → shape [B,N,N,hidden_dim]
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(linear_no_bias(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_no_bias(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear_no_bias(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core N³ reduction – Triton GEMM
    #    Cast to fp16 for the kernel; result stays fp32.
    # --------------------------------------------------------------
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    out = _trimul_gemm(left_f16, right_f16)   # [B,N,N,hidden_dim] fp32

    # --------------------------------------------------------------
    # 6️⃣ Post‑processing: LayerNorm over hidden_dim, out‑gate, final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out = out * out_gate                      # gating after norm

    # final projection back to original dim
    out = torch.einsum("...h,dh->...d", out, weights["to_out.weight"])

    # keep dtype float32 (already is)
    return out
