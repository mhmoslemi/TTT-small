"""
TriMul – outgoing variant (forward only)

Algorithm
---------
1. LayerNorm on the 4‑D input (B,N,N,D)                – PyTorch
2. Linear projections to hidden dimension H            – PyTorch (bias‑free)
3. Optional mask application (broadcasted)            – PyTorch (cheap)
4. Gating (sigmoid) + element‑wise multiplication      – PyTorch
5. **Core N³ reduction** : for every head h and batch b  
   compute      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply per head and is implemented
   in Triton (block‑tiled GEMM, fp32 accumulation, fp16 storage).
6. Output LayerNorm + final linear projection           – PyTorch
   (the result is cast back to float32).

Only step 5 is performed in Triton, satisfying the “some work in a kernel”
requirement while keeping the overall runtime well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# Triton kernel : batched GEMM for the N³ reduction
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                         # [B,N,N,H] left  tensor
    right_ptr,                        # [B,N,N,H] right tensor
    out_ptr,                          # [B,N,N,H] output
    B, N, H,                          # int scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,            # tile size in i dimension
    BLOCK_N: tl.constexpr,            # tile size in j dimension
    BLOCK_K: tl.constexpr,            # tile size in reduction k
):
    """
    Compute, for a fixed (b,h) pair:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a classic tiled GEMM algorithm.
    """
    pid = tl.program_id(0)          # encodes (b,h, tile_i)
    pid_n = tl.program_id(1)        # encodes tile_j

    # ----------------------------------------------------------
    # decode batch, head and tile coordinates
    # ----------------------------------------------------------
    tiles_i = tl.cdiv(N, BLOCK_M)          # number of i‑tiles per (b,h)
    bn = B * H                              # total (batch,head) combos
    bn_idx = pid // tiles_i                 # which (b,h) we are processing
    tile_i = pid % tiles_i

    b = bn_idx // H                         # batch index
    h = bn_idx % H                          # head index

    # global coordinates of the tile
    i_offset = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n   * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------
    # reduction over k
    # ----------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left:  [i,k] slice   (stride: i -> stride_left_i, k -> stride_left_k)
        left = tl.load(
            left_ptr
            + b * stride_left_b
            + i_offset[:, None] * stride_left_i
            + k_offset[None, :] * stride_left_k
            + h * stride_left_h,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right: [j,k] slice   (stride: j -> stride_right_j, k -> stride_right_k)
        right = tl.load(
            right_ptr
            + b * stride_right_b
            + j_offset[:, None] * stride_right_j
            + k_offset[None, :] * stride_right_k
            + h * stride_right_h,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # dot over the reduction dimension
        acc += tl.dot(left, right)

    # ----------------------------------------------------------
    # write back
    # ----------------------------------------------------------
    tl.store(
        out_ptr
        + b * stride_out_b
        + i_offset[:, None] * stride_out_i
        + j_offset[None, :] * stride_out_j
        + h * stride_out_h,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# --------------------------------------------------------------
# Helper : simple (weight,bias) LayerNorm (no fused bias inside)
# --------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """LayerNorm over the last dimension (no broadcasting)."""
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, D = x.shape
    H = cfg["hidden_dim"]                     # hidden_dim (called “hidden” in the paper)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = _layer_norm(x, weights["norm.weight"], weights["norm.bias"])

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    # shape: [B,N,N,H]  (FP32)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted) – cheap PyTorch path
    # ------------------------------------------------------------------
    if not cfg.get("nomask", True):
        # mask shape [B,N,N,1] -> broadcast on hidden dim
        mask_exp = mask.unsqueeze(-1).to(x.dtype)          # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) – bias‑free linear + sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ reduction – Triton batched GEMM
    # ------------------------------------------------------------------
    # Allocate output buffer (fp16 for lower memory traffic, will be cast later)
    out = torch.empty_like(left, dtype=torch.float16, device=device)

    # strides (int64) – convert to Python ints for the kernel
    l_str = left.stride()
    r_str = right.stride()
    o_str = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # grid: (B*H * ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    tiles_i = triton.cdiv(N, BLOCK_M)
    grid0 = (B * H * tiles_i, triton.cdiv(N, BLOCK_N))

    tri_mul_kernel[grid0](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=l_str[0],
        stride_left_i=l_str[1],
        stride_left_k=l_str[2],
        stride_left_h=l_str[3],
        stride_right_b=r_str[0],
        stride_right_j=r_str[1],
        stride_right_k=r_str[2],
        stride_right_h=r_str[3],
        stride_out_b=o_str[0],
        stride_out_i=o_str[1],
        stride_out_j=o_str[2],
        stride_out_h=o_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # back to fp32 for the remaining layers
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    out = _layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
