```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

## Triton launch convention

Every failure mode below has appeared in this search. Read this before writing.

A `@triton.jit` function is NOT a Python function. It is launched with a grid
subscript, and its `tl.constexpr` parameters exist only inside it:

```python
@triton.jit
def my_kernel(x_ptr, y_ptr, n, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < n
    tl.store(y_ptr + offs, tl.load(x_ptr + offs, mask=m, other=0.0), mask=m)

def custom_kernel(data):
    ...
    BLOCK = 128                                   # a host variable YOU define
    grid = (triton.cdiv(n, BLOCK),)               # host-side, use triton.cdiv
    my_kernel[grid](x, y, n, BLOCK=BLOCK)         # note the [grid] subscript
```

- `my_kernel(...)` without `[grid]` raises "Cannot call @triton.jit'd outside
  the scope of a kernel".
- `BLOCK`, `BLOCK_M`, `BLOCK_N` are not defined on the host unless you define
  them there. Referencing one you only declared as a `tl.constexpr` parameter
  raises NameError.
- `tl.cdiv` is device-side. On the host use `triton.cdiv`.
- Define the kernel at module top level, never nested inside `custom_kernel`.

You do not have to move everything into Triton. A correct submission that keeps
most of the computation in PyTorch and moves one hot region into a kernel scores;
one that fails to launch scores zero.

```python
"""
TriMul “outgoing” kernel (AlphaFold‑3).

The expensive O(N³·C) reduction is performed with PyTorch’s highly‑optimised
einsum (it internally uses cuBLAS gemm‑based tricks).  To satisfy the
requirement of “at least part of the work in Triton”, a tiny Triton kernel
applies the optional pair‑wise mask to the projected tensors – this is a
memory‑bound operation that benefits from Triton’s explicit masking and
vectorised stores.

The rest of the pipeline (LayerNorm, Linear projections, gating, the final
LayerNorm and output projection) is kept in PyTorch for numerical stability
and to keep the implementation short while still being fast on the H200.

All tensors are expected on CUDA; the final output is float‑32.
"""

import torch
import triton
import triton.language as tl
from torch import nn
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: apply a broadcast mask of shape [B, N, N] to a tensor
# of shape [B, N, N, H].  The mask is 0/1 (float) and is broadcast on the
# last dimension.  This kernel is launched with a 1‑D grid that covers
# the total number of elements.
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,   # const float*   (input)
    mask_ptr,  # const float*   (mask)
    dst_ptr,   # float*         (output)
    total_el,  # i64           total number of elements (B*N*N*H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_el

    # load input and mask (mask is broadcasted, we read the same value for
    # each hidden‑dim element)
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + (offs // tl.constexpr(0)), mask=mask, other=1.0)  # dummy idx, will be replaced in Python

    # multiply and store
    tl.store(dst_ptr + offs, src * m, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Apply ``mask[..., None]`` to ``tensor`` using Triton.
    tensor : [B, N, N, H]   (float32)
    mask   : [B, N, N]      (float32, 0/1)
    returns tensor * mask[...,None]
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024

    # flatten pointers
    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_f = mask.float().view(-1)  # [B,N,N]

    # launch grid
    grid = (triton.cdiv(total, BLOCK),)

    # Triton does not support broadcasting inside the kernel, so we expand the
    # mask manually: each element of the hidden dimension reads the same mask
    # value.  We achieve this by passing the mask pointer to the kernel and
    # indexing it with ``offs // H``.
    @triton.jit
    def masked_kernel(
        src_ptr, mask_ptr, dst_ptr,
        total_el, H, BLOCK: tl.constexpr,
    ):
        pid = tl.program_id(0)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < total_el

        src = tl.load(src_ptr + offs, mask=mask, other=0.0)
        # each hidden‑dim slice corresponds to the same mask element
        m = tl.load(mask_ptr + (offs // H), mask=mask, other=1.0)

        tl.store(dst_ptr + offs, src * m, mask=mask)

    masked_kernel[grid](
        src,
        mask_f,
        dst,
        total,
        H,
        BLOCK=BLOCK,
    )
    return dst.view(B, N, N, H)


# ----------------------------------------------------------------------
# Custom kernel entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : tuple
        (input, mask, weights, config)
        - input  : [B, N, N, dim]   float32
        - mask   : [B, N, N]        float32 (0/1) – may be all‑ones when ``nomask``.
        - weights: dict of pretrained parameters.
        - config : dict with keys ``dim`` and ``hidden_dim``.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm + Linear projections (PyTorch – fast and numerically stable)
    # ------------------------------------------------------------------
    # norm
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(
        input_tensor.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    x = x * norm_weight + norm_bias

    # projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask – apply with Triton (the only part performed on GPU via
    # custom kernel)
    # ------------------------------------------------------------------
    if not nomask:
        left  = apply_mask_triton(left,  mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # Gates
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·C) reduction – torch.einsum (uses cuBLAS GEMM under‑the‑hood)
    # ------------------------------------------------------------------
    # shape after einsum: [B, N, N, hidden_dim]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # Final LayerNorm, gating and output projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(
        out.var(-1, unbiased=False, keepdim=True) + 1e-5
    )
    out = out * to_out_norm_weight + to_out_norm_bias

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
```
