"""
TriMul “outgoing” forward pass (no gradients).

The most expensive part is the N³ reduction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be expressed as a batched matrix‑multiply per hidden dimension.
We keep that part in PyTorch (torch.bmm) because the H200 already provides a
highly‑optimised GEMM for the supported sizes (N≤1024, hidden≤384).

What we move into Triton is the element‑wise work that is *not* fused by
torch.compile:
  * mask broadcasting
  * gating (sigmoid) and the element‑wise multiplication with the linear
    projections.

The kernel `masked_proj_gate` receives the projected tensor (`proj`), the
pre‑sigmoid gate tensor (`gate_pre`), and the binary mask (shape
[B, N, N, 1] after unsqueeze).  It writes back
    proj * mask * sigmoid(gate_pre)
for every element.  This merges two separate element‑wise passes into a
single fused kernel, reduces memory traffic and satisfies the requirement
that “some of the operations are performed in a Triton kernel”.

All other operations (LayerNorm, the linear layers that produce the
pre‑sigmoid gates, the final output LayerNorm & linear) are carried out
with regular PyTorch (they are memory‑bound and already efficiently fused
by torch.compile on the target hardware).

The overall algorithm:
1. Layer‑norm the input.
2. Linear projections to hidden dim (left/right) and linear gates
   (pre‑sigmoid).
3. Triton kernel → mask, sigmoid gating, and element‑wise multiplication.
4. Batched GEMM for the N³ reduction.
5. Output LayerNorm, multiply by out‑gate, final linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: mask + sigmoid(gate) + element‑wise multiply
# ----------------------------------------------------------------------
@triton.jit
def masked_proj_gate(
    proj_ptr,               # [B,N,N,H] projected tensor (left or right)
    gate_ptr,               # [B,N,N,H] pre‑sigmoid gate tensor
    mask_ptr,               # [B,N,N,1] binary mask (broadcasted along H)
    out_ptr,                # [B,N,N,H] output buffer
    total_elems,            # B*N*N*H  (int)
    PROJ_STRIDE_H: tl.constexpr,   # stride of hidden dimension (should be 1)
    MASK_STRIDE_H: tl.constexpr,   # stride of mask hidden dim (0, because mask is 1‑wide)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load projected value and gate (both contiguous in memory)
    proj_val = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask offset: divide by stride of hidden dim (normally 1)
    # The mask tensor has shape [...,1] → its hidden‑dim stride is zero,
    # therefore the same mask value is used for all H positions.
    mask_off = offs // PROJ_STRIDE_H
    mask_val = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)   # default 1 (no‑mask)

    # fused computation : proj * mask * sigmoid(gate)
    gated = proj_val * mask_val * tl.sigmoid(gate_val)

    tl.store(out_ptr + offs, gated, mask=mask)


# ----------------------------------------------------------------------
# Helper: simple LayerNorm (no bias fusion – bias added afterwards)
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """
    x: [..., C]  (float32)
    weight, bias: [C]
    Returns same shape as x, dtype float32.
    """
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                      # input tensor [B, N, N, dim]
        torch.Tensor,                      # mask tensor [B, N, N] (may be None)
        Dict[str, torch.Tensor],           # weights dict
        Dict[str, Any],                    # config dict
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]          # e.g. 128
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = layer_norm(
        inp,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (dim -> hidden) – bias‑free
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden]
    left_proj = F.linear(x, weights["left_proj.weight"])      # (hidden, dim) weight
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Linear gates (pre‑sigmoid) – also dim -> hidden, bias‑free
    # ------------------------------------------------------------------
    left_gate_pre = F.linear(x, weights["left_gate.weight"])
    right_gate_pre = F.linear(x, weights["right_gate.weight"])
    out_gate_pre = F.linear(x, weights["out_gate.weight"])

    # ------------------------------------------------------------------
    # 4. Apply mask + sigmoid gate in one Triton kernel (if mask is present)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N,1] for broadcasting
        mask_exp = mask.unsqueeze(-1).contiguous()
        total = left_proj.numel()                     # B*N*N*hidden
        BLOCK = 131072                                 # 128 Ki elements ~ 512 KB (FP32)

        grid = (triton.cdiv(total, BLOCK),)

        # allocate output buffers
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # strides (last dimension stride is 1 for a contiguous tensor)
        proj_stride_h = left_proj.stride(-1)           # should be 1
        mask_stride_h = 0                               # broadcasted dim

        # left
        masked_proj_gate[grid](
            left_proj.data_ptr(),
            left_gate_pre.data_ptr(),
            mask_exp.data_ptr(),
            left.data_ptr(),
            total,
            PROJ_STRIDE_H=proj_stride_h,
            MASK_STRIDE_H=mask_stride_h,
            BLOCK=BLOCK,
        )
        # right
        masked_proj_gate[grid](
            right_proj.data_ptr(),
            right_gate_pre.data_ptr(),
            mask_exp.data_ptr(),
            right.data_ptr(),
            total,
            PROJ_STRIDE_H=proj_stride_h,
            MASK_STRIDE_H=mask_stride_h,
            BLOCK=BLOCK,
        )
    else:
        # no mask → just apply sigmoid gate in‑place
        left = left_proj * torch.sigmoid(left_gate_pre)
        right = right_proj * torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 5. Core N³ reduction  (batched GEMM per hidden dimension)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N)
    left_rs = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)      # (B*H, N, N)
    right_rs = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)    # (B*H, N, N)

    # batched matmul: (B*H, N, N) @ (B*H, N, N)^T  → (B*H, N, N)
    out_rs = torch.bmm(left_rs, right_rs.transpose(1, 2))

    # reshape back to [B, N, N, hidden]
    out = (
        out_rs.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate and final projection back to dim
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out_gate = torch.sigmoid(out_gate_pre)          # shape [B,N,N,hidden]
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])   # (dim, hidden) weight

    return out.to(torch.float32)
