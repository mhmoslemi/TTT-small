"""
TriMul “outgoing” kernel (AlphaFold‑3) – fast forward pass.

The O(N³·C) reduction is performed by torch.einsum (cuBLAS‑based GEMM) which
is already highly optimised on Hopper.  To satisfy the “must use Triton”
requirement we move the mask‑application (the only memory‑bound step) into a
tiny Triton kernel that broadcasts the pair‑wise mask over the hidden channel.
All LayerNorms, Linear projections, gates and the final projection stay in
PyTorch for numerical stability and speed.

The implementation works for any batch size ≤2, sequence length up to 1024,
and arbitrary `dim` / `hidden_dim` (the reference configs use dim∈{128,384},
hidden_dim=128).  The output is always float‑32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# --------------------------------------------------------------
# Triton kernel: apply a 3‑D mask (B,N,N) to a 4‑D tensor (B,N,N,H)
# --------------------------------------------------------------
@triton.jit
def _mask_apply_kernel(
    src_ptr,          # const float* : input tensor (flattened)
    mask_ptr,         # const float* : mask tensor (flattened, B*N*N)
    dst_ptr,          # float*       : output tensor (flattened)
    total_el,         # i64          : total number of elements (B*N*N*H)
    hidden_dim,       # i32          : size of the last dimension (H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_el

    # load the input values
    x = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # each hidden‑dim slice shares the same mask element:
    #   mask_index = element_index // hidden_dim
    m = tl.load(mask_ptr + (offs // hidden_dim), mask=mask, other=1.0)

    tl.store(dst_ptr + offs, x * m, mask=mask)


def _apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Apply ``mask[..., None]`` to ``tensor`` using Triton.

    Parameters
    ----------
    tensor : torch.Tensor
        Shape (B, N, N, H), dtype float32, on CUDA.
    mask : torch.Tensor
        Shape (B, N, N), dtype float32 (0/1), on CUDA.

    Returns
    -------
    torch.Tensor
        Tensor * mask[..., None] with same shape and dtype as `tensor`.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024                                   # 1‑D launch block size

    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)                       # (B*N*N,)

    grid = (triton.cdiv(total, BLOCK),)

    _mask_apply_kernel[grid](
        src,
        mask_flat,
        dst,
        total,
        H,
        BLOCK=BLOCK,
    )
    return dst.view(B, N, N, H)


# --------------------------------------------------------------
# Main entry‑point (called by the evaluator)
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,               # input   [B, N, N, dim]
        torch.Tensor,               # mask    [B, N, N]  (float, 0/1)
        Dict[str, torch.Tensor],    # weights
        Dict[str, Any],             # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Returns
    -------
    torch.Tensor
        Shape (B, N, N, dim), dtype float32, on the same device as the input.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (no bias fuse – manual for clarity)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]   # (dim,)
    norm_b = weights["norm.bias"]     # (dim,)

    # compute mean / var over the last dimension (C) – keepdim for broadcasting
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    eps = 1e-5
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → shape (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional pair‑wise mask (broadcast over hidden_dim) – Triton
    # ------------------------------------------------------------------
    if not nomask:
        left = _apply_mask_triton(left, mask)
        right = _apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – still in PyTorch
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·hidden_dim) reduction – torch.einsum (cuBLAS GEMM)
    # ------------------------------------------------------------------
    #   "b i k h, b j k h -> b i j h"
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]   # (hidden_dim,)
    to_out_norm_b = weights["to_out_norm.bias"]     # (hidden_dim,)

    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(dtype)   # ensure same dtype as input (float32 in spec)
