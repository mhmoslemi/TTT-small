"""
TriMul “outgoing” forward pass (AlphaFold3).

We keep the cheap element‑wise work (LayerNorm, linear projections, gating,
masking) in PyTorch and move the only “fusion‑critical’’ part – the
mask‑and‑gate multiplication – into a tiny Triton kernel.  The expensive
N³ einsum is rewritten as a batched matrix‑multiply:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is exactly a (B·H)‑batch of (N×N) @ (N×N)ᵀ products.
CuBLAS (torch.bmm) handles this at near‑peak bandwidth, while the Triton
kernel eliminates an extra memory pass over the four tensors.

The kernel works in‑place on the projected tensors `left` and `right`,
so the host code only needs to launch it once.  All tensors stay on the
CUDA device and the final result is returned as float‑32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# --------------------------------------------------------------
# Triton kernel – fuse mask, left‑gate and right‑gate into the
# projected tensors.  Operates on a flat 1‑D view of the 4‑D
# tensor [B, N, N, H].
# --------------------------------------------------------------
@triton.jit
def _tri_mul_fuse_mask_gate_kernel(
    left_ptr,            # [B, N, N, H]  (float32)
    right_ptr,           # [B, N, N, H]  (float32)
    mask_ptr,            # [B, N, N, 1]  (float32) – broadcasted mask
    left_gate_ptr,       # [B, N, N, H]  (float32)
    right_gate_ptr,      # [B, N, N, H]  (float32)
    B, N, H,             # static sizes
    stride_left_b, stride_left_n1, stride_left_n2, stride_left_h,
    stride_right_b, stride_right_n1, stride_right_n2, stride_right_h,
    stride_mask_b, stride_mask_n1, stride_mask_n2,
    stride_lgate_b, stride_lgate_n1, stride_lgate_n2, stride_lgate_h,
    stride_rgate_b, stride_rgate_n1, stride_rgate_n2, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # ---- decode linear offset into (b, i, j, h) -----------------
    # linear = ((b * N + i) * N + j) * H + h
    b = offs // (N * N * H)
    rest = offs % (N * N * H)
    i = rest // (N * H)
    rest = rest % (N * H)
    j = rest // H
    h = rest % H

    # ---- compute pointer offsets ---------------------------------
    left_off  = (b * stride_left_b  + i * stride_left_n1 +
                 j * stride_left_n2 + h * stride_left_h)
    right_off = (b * stride_right_b + i * stride_right_n1 +
                 j * stride_right_n2 + h * stride_right_h)
    mask_off  = (b * stride_mask_b  + i * stride_mask_n1 +
                 j * stride_mask_n2)               # mask has no channel dim
    lgate_off = (b * stride_lgate_b + i * stride_lgate_n1 +
                 j * stride_lgate_n2 + h * stride_lgate_h)
    rgate_off = (b * stride_rgate_b + i * stride_rgate_n1 +
                 j * stride_rgate_n2 + h * stride_rgate_h)

    # ---- load, fuse and store ------------------------------------
    left  = tl.load(left_ptr  + left_off,  mask=mask, other=0.0)
    right = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    m     = tl.load(mask_ptr  + mask_off,  mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + lgate_off, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + rgate_off, mask=mask, other=0.0)

    left  = left  * m * lgate
    right = right * m * rgate

    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left: torch.Tensor,
                         right: torch.Tensor,
                         mask: torch.Tensor,
                         left_gate: torch.Tensor,
                         right_gate: torch.Tensor) -> None:
    """
    In‑place Triton fusion of mask and gate multiplications.
    All tensors must be contiguous, same dtype (float32) and on CUDA.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                     # enough threads per program for H200
    grid = (triton.cdiv(total, BLOCK),)

    # strides in element counts (not bytes)
    s_left  = left.stride()
    s_right = right.stride()
    s_mask  = mask.stride()
    s_lgate = left_gate.stride()
    s_rgate = right_gate.stride()

    _tri_mul_fuse_mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=s_left[0],
        stride_left_n1=s_left[1],
        stride_left_n2=s_left[2],
        stride_left_h=s_left[3],
        stride_right_b=s_right[0],
        stride_right_n1=s_right[1],
        stride_right_n2=s_right[2],
        stride_right_h=s_right[3],
        stride_mask_b=s_mask[0],
        stride_mask_n1=s_mask[1],
        stride_mask_n2=s_mask[2],
        stride_lgate_b=s_lgate[0],
        stride_lgate_n1=s_lgate[1],
        stride_lgate_n2=s_lgate[2],
        stride_lgate_h=s_lgate[3],
        stride_rgate_b=s_rgate[0],
        stride_rgate_n1=s_rgate[1],
        stride_rgate_n2=s_rgate[2],
        stride_rgate_h=s_rgate[3],
        BLOCK=BLOCK,
    )
    # No return – tensors are modified in‑place


# ------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor  – shape [B, N, N, dim], dtype float32, on the same device.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps_ln = 1e-5          # LayerNorm eps
    eps_out = 1e-5         # to_out_norm eps

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    w_ln = weights["norm.weight"]
    b_ln = weights["norm.bias"]
    # keep everything in fp32 for numerical stability
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps_ln)
    x = x * w_ln + b_ln                     # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]     # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]   # (hidden_dim, dim)

    # torch.nn.functional.linear does: y = x @ W.t()
    left = F.linear(x, left_proj_w)      # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Gating values (sigmoid) – also linear, no bias
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 4) Mask handling – shape (B, N, N) → (B, N, N, 1)
    # ------------------------------------------------------------------
    if mask is None:
        # “nomask” case – a tensor of ones
        mask = torch.ones_like(input_tensor[..., 0], dtype=input_tensor.dtype,
                               device=input_tensor.device)
    mask = mask.unsqueeze(-1)        # (B, N, N, 1)

    # ------------------------------------------------------------------
    # 5) Fuse mask & gates with Triton (in‑place on left/right)
    # ------------------------------------------------------------------
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 6) N³ einsum → batched matmul
    #    out_h = left_h @ right_hᵀ   for each hidden channel h
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, N, N, H) → (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # torch.bmm uses cuBLAS and is highly optimized on H200
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm
    # ------------------------------------------------------------------
    w_out_ln = weights["to_out_norm.weight"]
    b_out_ln = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps_out)
    out = out * w_out_ln + b_out_ln

    # ------------------------------------------------------------------
    # 8) Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate                                 # (B, N, N, hidden_dim)
    to_out_w = weights["to_out.weight"]                  # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                        # (B, N, N, dim)

    # Ensure float32 output
    return out.to(torch.float32)
```

**Explanation of the lessons and why the implementation works**

| Lesson | Relevance / Action |
|-------|--------------------|
| *Limit shared memory usage* | The kernel only loads five scalars per element and stores two – no shared memory is used, so the guideline is satisfied automatically. |
| *Match stride unpacking to tensor rank* | The kernel explicitly receives a stride for every dimension of each 4‑D tensor (`stride_*_b, _n1, _n2, _h`).  This avoids the “assume‑3‑stride’’ bug that caused crashes in earlier attempts. |
| *Don’t copy whole kernels verbatim* | The kernel is a minimal, purpose‑built fusion (mask × gate × projection) rather than a copy of a generic matmul kernel, respecting the rule. |
| *You may keep heavy work in PyTorch* | The N³ part is expressed as a batched `torch.bmm`.  cuBLAS on the H200 is far faster than any hand‑written Triton GEMM, and it satisfies the “at least part in a kernel’’ requirement. |

The overall solution therefore meets correctness, runs fully on‑device, fuses the only memory‑bound region with Triton, and leverages the highly‑optimized cuBLAS path for the cubic‑complexity computation.
