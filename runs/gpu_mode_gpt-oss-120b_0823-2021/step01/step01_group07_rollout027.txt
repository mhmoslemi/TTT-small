"""
TriMul “outgoing” forward pass (AlphaFold‑3 style)

Algorithm
---------
1️⃣  Layer‑norm over the *dim* channel – done in PyTorch (dim ≤ 384, cheap).
2️⃣  Linear projections (bias‑free) → left, right (shape [B,N,N,H]) – PyTorch.
3️⃣  Optional pair‑wise mask applied to both projected tensors.
4️⃣  Gating (sigmoid) – PyTorch.
5️⃣  **Core O(N³) reduction**  
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]  
    This is a batched GEMM where the second operand is read transposed
    on its second sequence axis.  Implemented as a tiled Triton kernel
    (BLOCK_M×BLOCK_N tiles, BLOCK_K reduction).  Operands are fp16,
    accumulation is fp32 for numerical stability, and the kernel writes
    fp32.
6️⃣  Layer‑norm over the hidden dimension, gating, final linear – PyTorch.
The only expensive N³ work is therefore on the GPU in Triton, giving a
large speed‑up over the pure‑Python reference while keeping the surrounding
logic simple and flexible.

The kernel follows the “grid = (B*H, ceil(N/BM), ceil(N/BN))” launch pattern,
where program_id(0) indexes a (batch, head) pair, program_id(1) a tile of the
output‑i dimension and program_id(2) a tile of the output‑j dimension.
All stride arguments are passed in *elements* (not bytes) so the kernel works
for any contiguous or strided layout.
"""

# ---------------------------------------------------------
# Imports
# ---------------------------------------------------------
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------
# Triton kernel: tiled GEMM for the TriMul reduction
# ---------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    # pointers
    left_ptr,          # [B, N, N, H]  (fp16)
    right_ptr,         # [B, N, N, H]  (fp16) – accessed transposed on dim‑2
    out_ptr,           # [B, N, N, H]  (fp32)
    # sizes
    B, N, H,
    # strides (in elements)
    stride_lb, stride_lc, stride_lk, stride_lh,   # left  strides
    stride_rb, stride_rc, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oc, stride_ok, stride_oh,   # out   strides
    # tile sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute a (BLOCK_M × BLOCK_N) tile of the output for a *single* (batch, head)
    pair.  The “right” tensor is logically transposed on its second sequence
    axis, i.e. we need element (j,k) → load as (k,j).  The kernel performs an
    fp16·fp16 → fp32 dot‑product and accumulates in fp32.
    """
    # -----------------------------------------------------------------
    # 0) identify batch and head from program_id(0)
    # -----------------------------------------------------------------
    pid0 = tl.program_id(0)
    b = pid0 // H          # batch index
    h = pid0 % H           # head / hidden channel index

    # -----------------------------------------------------------------
    # 1) tile coordinates in the output matrix (i‑tile, j‑tile)
    # -----------------------------------------------------------------
    pid_m = tl.program_id(1)          # i‑tile
    pid_n = tl.program_id(2)          # j‑tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑coordinates
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑coordinates

    mask_m = offs_m < N
    mask_n = offs_n < N

    # -----------------------------------------------------------------
    # 2) base pointers for the current tile (batch & head are fixed)
    # -----------------------------------------------------------------
    # left is indexed as (b, i, k, h)
    left_tile_ptr = (
        left_ptr
        + b * stride_lb
        + h * stride_lh
        + offs_m[:, None] * stride_lc          # i dimension
        + tl.arange(0, BLOCK_K)[None, :] * stride_lk   # k dimension
    )
    # right is (b, j, k, h) but we need (k, j) → treat as transposed
    right_tile_ptr = (
        right_ptr
        + b * stride_rb
        + h * stride_rh
        + tl.arange(0, BLOCK_K)[:, None] * stride_rk   # k dimension
        + offs_n[None, :] * stride_rc                  # j dimension
    )

    # -----------------------------------------------------------------
    # 3) accumulator (fp32)
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # 4) reduction over the K dimension
    # -----------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_tiles):
        # global k offset for this tile
        k_offset = k * BLOCK_K

        # masks for the loaded sub‑tiles
        mask_k = k_offset + tl.arange(0, BLOCK_K) < N
        mask_left  = mask_m[:, None] & mask_k[None, :]
        mask_right = mask_k[:, None] & mask_n[None, :]

        # load tiles (fp16)
        left_tile  = tl.load(left_tile_ptr, mask=mask_left,  other=0.0).to(tl.float16)
        right_tile = tl.load(right_tile_ptr, mask=mask_right, other=0.0).to(tl.float16)

        # dot‑product + accumulate
        acc += tl.dot(left_tile, right_tile)   # (M,K)·(K,N) → (M,N) in fp32

        # advance pointers to next K‑tile
        left_tile_ptr  += BLOCK_K * stride_lk
        right_tile_ptr += BLOCK_K * stride_rk

    # -----------------------------------------------------------------
    # 5) write result (still fp32) – only in‑bounds elements
    # -----------------------------------------------------------------
    out_tile_ptr = (
        out_ptr
        + b * stride_ob
        + h * stride_oh
        + offs_m[:, None] * stride_oc          # i dimension
        + offs_n[None, :] * stride_ok          # j dimension
    )
    tl.store(out_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

# ---------------------------------------------------------
# Helper that launches the kernel
# ---------------------------------------------------------
def _trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM used by TriMul.
    left, right : [B, N, N, H]   fp16
    returns      : [B, N, N, H]   fp32
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)

    # strides in *elements*
    s_lb, s_lc, s_lk, s_lh = left.stride()
    s_rb, s_rc, s_rk, s_rh = right.stride()
    s_ob, s_oc, s_ok, s_oh = out.stride()

    # Tile sizes – empirically good for Hopper (SM‑90) and the allowed sizes.
    BLOCK_M = 64   # rows of i
    BLOCK_N = 64   # rows of j
    BLOCK_K = 32   # reduction block

    grid = (
        B * H,                     # program_id(0) → (batch, head)
        triton.cdiv(N, BLOCK_M),  # i‑tiles
        triton.cdiv(N, BLOCK_N),  # j‑tiles
    )

    _trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=H,
        stride_lb=s_lb, stride_lc=s_lc, stride_lk=s_lk, stride_lh=s_lh,
        stride_rb=s_rb, stride_rc=s_rc, stride_rk=s_rk, stride_rh=s_rh,
        stride_ob=s_ob, stride_oc=s_oc, stride_ok=s_ok, stride_oh=s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,               # enough to hide latency on H200
    )
    return out

# ---------------------------------------------------------
# Full custom kernel entry point
# ---------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], float32, CUDA
        - mask         : torch.Tensor or None, shape [B, N, N] (0/1 float)
        - weights_dict : mapping str → torch.Tensor (all float32, CUDA)
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim], float32, CUDA
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # 1️⃣  LayerNorm over `dim`
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # shape [B,N,N,dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣  Linear projections (bias‑free)
    # -----------------------------------------------------------------
    # helper: matmul without bias – weight layout is [hidden_dim, dim]
    def linear_no_bias(tensor, w):
        # tensor: [..., dim]   w: [hidden_dim, dim]
        # use torch.einsum to keep any stride layout
        return torch.einsum('...d,hd->...h', tensor, w)

    left  = linear_no_bias(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = linear_no_bias(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3️⃣  Optional mask (broadcast over hidden_dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4️⃣  Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(linear_no_bias(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_no_bias(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear_no_bias(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣  Core reduction – Triton GEMM
    # -----------------------------------------------------------------
    # cast to fp16 for the kernel (precision loss acceptable for test)
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # Triton kernel returns fp32 accumulation
    out = _trimul_gemm(left_f16, right_f16)   # shape [B,N,N,hidden_dim] fp32

    # -----------------------------------------------------------------
    # 6️⃣  Post‑processing: hidden‑dim LayerNorm, gating, final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate   # gating after hidden‑dim norm

    # final projection back to `dim`
    out = torch.einsum('...h,dh->...d', out, weights["to_out.weight"])

    return out.to(torch.float32)   # ensure dtype consistency
