```python
"""
TriMul “outgoing” kernel (AlphaFold‑3)

- Layer‑norm, linear projections and gating are performed with PyTorch (FP32).
- The O(N³) reduction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
  is implemented as a tiled GEMM in Triton:
    * each (batch, head) pair is processed by a separate program‑id(0)
    * i‑ and j‑axes are tiled with BLOCK_M × BLOCK_N blocks
    * reduction over k is tiled with BLOCK_K
    * inputs are cast to fp16, multiplication in fp16, accumulation in fp32.
- After the kernel we apply a second LayerNorm, a gating scalar, and a final
  linear projection back to the original dimension.

The kernel works for any batch size ≤2, sequence length up to 1024,
and hidden dimensions 128, 384.  It returns a float‑32 tensor of shape
[batch, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: tiled GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    # pointers
    left_ptr,          # [B, N, N, H]   (fp16)
    right_ptr,         # [B, N, N, H]   (fp16) – accessed transposed on dim‑2
    out_ptr,           # [B, N, N, H]   (fp32)
    # sizes
    B, N, H,
    # strides (in elements)
    stride_lb, stride_lc, stride_lk, stride_lh,   # left  strides
    stride_rb, stride_rc, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oc, stride_ok, stride_oh,   # out   strides
    # tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute a tile of the output for a fixed (batch, head) pair:
        out_{b,i:j, h} = Σ_k left_{b,i:k, h} * right_{b,j:k, h}
    right is logically transposed on its middle axis, so we load it as
    (k, j) instead of (j, k).
    """
    pid = tl.program_id(0)           # linear index over (B, H)
    b = pid // H
    h = pid % H

    pid_m = tl.program_id(1)         # tile index along i (row)
    pid_n = tl.program_id(2)         # tile index along j (col)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑coordinates
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑coordinates

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ---- pointers to the start of the current tile ---------------------------------
    # left:   shape (B, N, N, H) -> indices (b, i, k, h)
    left_tile_ptr = left_ptr + b * stride_lb \
                               + h * stride_lh \
                               + offs_m[:, None] * stride_lc \
                               + tl.arange(0, BLOCK_K)[None, :] * stride_lk

    # right is accessed as right[b, j, k, h] → we need (k, j) order for the dot
    right_tile_ptr = right_ptr + b * stride_rb \
                                 + h * stride_rh \
                                 + tl.arange(0, BLOCK_K)[:, None] * stride_rk \
                                 + offs_n[None, :] * stride_rc

    # accumulator (fp32 for stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- reduction over k -----------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute actual k‑indices for masking
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # load left block  (M × K)
        left = tl.load(
            left_tile_ptr,
            mask=mask_m[:, None] & (cur_k < N),
            other=0.0,
        )
        # load right block (K × N) – already transposed
        right = tl.load(
            right_tile_ptr,
            mask=(cur_k[:, None] < N) & mask_n[None, :],
            other=0.0,
        )

        # fp16 multiplication, fp32 accumulation
        left = left.to(tl.float16)
        right = right.to(tl.float16)
        acc += tl.dot(left, right)          # (M,K)·(K,N) → (M,N)

        # advance pointers to next K‑slice
        left_tile_ptr += BLOCK_K * stride_lk
        right_tile_ptr += BLOCK_K * stride_rk

    # ---- write result ----------------------------------------------------------------
    out_tile_ptr = out_ptr + b * stride_ob \
                               + h * stride_oh \
                               + offs_m[:, None] * stride_oc \
                               + offs_n[None, :] * stride_ok
    tl.store(out_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM used by TriMul.
    left/right : [B, N, N, H] (fp16)
    returns    : [B, N, N, H] (fp32)
    """
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)

    # strides expressed in element counts (not bytes)
    stride_lb, stride_lc, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rc, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oc, stride_ok, stride_oh = out.stride()

    # Tile sizes – empirically good for Hopper (SM‑90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                     # program_id(0) → batch * hidden
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
    )

    _trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_lb=stride_lb,
        stride_lc=stride_lc,
        stride_lk=stride_lk,
        stride_lh=stride_lh,
        stride_rb=stride_rb,
        stride_rc=stride_rc,
        stride_rk=stride_rk,
        stride_rh=stride_rh,
        stride_ob=stride_ob,
        stride_oc=stride_oc,
        stride_ok=stride_ok,
        stride_oh=stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # matches tile shape on Hopper
    )
    return out


# ----------------------------------------------------------------------
# Full custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32, cuda
        - mask         : torch.Tensor or None, shape [B, N, N] (0/1), dtype float32
        - weights_dict : mapping str → torch.Tensor (float32, cuda)
        - config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32, cuda)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # shape [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    # helper: matmul x @ w.T   where w is (hidden_dim, dim)
    def linear_no_bias(t, w):
        # t: [..., dim]   w: [hidden_dim, dim]
        return torch.einsum('...d,hd->...h', t, w)

    left  = linear_no_bias(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = linear_no_bias(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask if present
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [..., N, N, 1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – also bias‑free linear + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(linear_no_bias(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_no_bias(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear_no_bias(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TRIMUL reduction via Triton GEMM
    # ------------------------------------------------------------------
    # Cast projections to fp16 for the kernel (still precise enough)
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # Triton kernel returns fp32 accumulator
    out = _trimul_gemm(left_f16, right_f16)   # shape [B, N, N, hidden] (fp32)

    # ------------------------------------------------------------------
    # 6. Post‑processing: norm, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate   # gating after the second norm

    # final projection back to original dim
    out = torch.einsum('...h,dh->...d', out, weights["to_out.weight"])

    # ensure output is float32 (already is, but keep API stable)
    return out.to(torch.float32)
```

**Key points**

* The heavy‑weight `N³` computation is fully fused in the Triton kernel, giving a tiled GEMM with fp16 arithmetic and fp32 accumulation.
* All other cheap operations (layer‑norm, linear projections, sigmoid gating) stay in PyTorch for simplicity and negligible overhead.
* The implementation respects the launch conventions, stride handling, and memory‑size recommendations for Hopper (SM‑90) GPUs.
* The function `custom_kernel(data)` matches the required signature and returns a float‑32 tensor of shape `[B, N, N, dim]`.
