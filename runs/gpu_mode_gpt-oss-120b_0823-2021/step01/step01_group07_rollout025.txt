The outgoing TriMul operator consists of three stages that are cheap and can stay in
PyTorch (LayerNorm, linear projections & gating).  
The heavy‑weight part is the `einsum('... i k d, ... j k d -> ... i j d')`,
i.e. a batched `N×N` matrix‑multiply for every hidden channel `d`.  
We implement this reduction with a tiled GEMM kernel in Triton:

* each program processes one `(batch, hidden)` pair (`pid0`);
* two‑dimensional grid (`pid1`,`pid2`) walks over the `(i,j)` output tiles;
* the `k`‑dimension is reduced in blocks (`BLOCK_K`);
* the right operand is read transposed (`right[b, j, k, h]`) so the kernel can use
  the classic `M×K * K×N` pattern;
* inputs are cast to **fp16** (good for throughput on Hopper) while the
  accumulator lives in **fp32** for numerical stability;
* after the kernel we finish with the second LayerNorm, the output‑gate and the
  final linear projection – all in PyTorch.

The kernel is deliberately small (64 × 64 × 32 tiles, 4 warps) so that shared‑
memory fits comfortably on the H200 (SM‑90) and occupancy stays high.  
The implementation works for any batch size `B∈{1,2}`, any sequence length
`N∈{128,…,1024}`, and any hidden dimension `H` (the hidden dimension is the
`hidden_dim` from the config).

### Triton kernel

```python
import triton
import triton.language as tl

@triton.jit
def trimul_gemm_kernel(
    # pointers
    left_ptr,          # [B, N, N, H] (fp16)
    right_ptr,         # [B, N, N, H] (fp16) – will be read transposed on dim 2
    out_ptr,           # [B, N, N, H] (fp32)
    # sizes
    B, N, H,
    # strides (in elements, not bytes)
    stride_lb, stride_lc, stride_lk, stride_lh,
    stride_rb, stride_rc, stride_rk, stride_rh,
    stride_ob, stride_oc, stride_ok, stride_oh,
    # tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    One program = one (b, h) pair.
    """
    pid = tl.program_id(0)               # linear index over (B, H)
    b = pid // H
    h = pid % H

    pid_m = tl.program_id(1)             # i‑tile
    pid_n = tl.program_id(2)             # j‑tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the beginning of the tile (batch & hidden are fixed)
    left_tile_ptr = left_ptr + b * stride_lb \
                               + h * stride_lh \
                               + offs_m[:, None] * stride_lc \
                               + tl.arange(0, BLOCK_K)[None, :] * stride_lk

    # right is accessed as (k, j) → we load a (K, N) block
    right_tile_ptr = right_ptr + b * stride_rb \
                                 + h * stride_rh \
                                 + tl.arange(0, BLOCK_K)[:, None] * stride_rk \
                                 + offs_n[None, :] * stride_rc

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Load left block
        left = tl.load(
            left_tile_ptr,
            mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < N),
            other=0.0,
        )
        # Load right block (already transposed)
        right = tl.load(
            right_tile_ptr,
            mask=(k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < N) & mask_n[None, :],
            other=0.0,
        )
        # fp16 dot, fp32 accumulation
        acc += tl.dot(left.to(tl.float16), right.to(tl.float16))

        # advance pointers to next K‑block
        left_tile_ptr  += BLOCK_K * stride_lk
        right_tile_ptr += BLOCK_K * stride_rk

    # write result (fp32)
    out_tile_ptr = out_ptr + b * stride_ob \
                               + h * stride_oh \
                               + offs_m[:, None] * stride_oc \
                               + offs_n[None, :] * stride_ok
    tl.store(out_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])
```

### Helper that launches the kernel

```python
def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right : [B, N, N, H]   (fp16)
    returns    : [B, N, N, H]   (fp32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides (in elements)
    stride_lb, stride_lc, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rc, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oc, stride_ok, stride_oh = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                              # program_id(0)
        triton.cdiv(N, BLOCK_M),           # i‑tiles
        triton.cdiv(N, BLOCK_N),           # j‑tiles
    )

    trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_lb=stride_lb,
        stride_lc=stride_lc,
        stride_lk=stride_lk,
        stride_lh=stride_lh,
        stride_rb=stride_rb,
        stride_rc=stride_rc,
        stride_rk=stride_rk,
        stride_rh=stride_rh,
        stride_ob=stride_ob,
        stride_oc=stride_oc,
        stride_ok=stride_ok,
        stride_oh=stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # works well on SM‑90
    )
    return out
```

### Full `custom_kernel` entry‑point

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel & helper (same as above) – placed at module level
# --------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_lc, stride_lk, stride_lh,
    stride_rb, stride_rc, stride_rk, stride_rh,
    stride_ob, stride_oc, stride_ok, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    b = pid // H
    h = pid % H
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    left_tile_ptr = left_ptr + b * stride_lb + h * stride_lh \
                    + offs_m[:, None] * stride_lc \
                    + tl.arange(0, BLOCK_K)[None, :] * stride_lk
    right_tile_ptr = right_ptr + b * stride_rb + h * stride_rh \
                     + tl.arange(0, BLOCK_K)[:, None] * stride_rk \
                     + offs_n[None, :] * stride_rc

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        left = tl.load(
            left_tile_ptr,
            mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < N),
            other=0.0,
        )
        right = tl.load(
            right_tile_ptr,
            mask=(k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < N) & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(left.to(tl.float16), right.to(tl.float16))
        left_tile_ptr += BLOCK_K * stride_lk
        right_tile_ptr += BLOCK_K * stride_rk

    out_tile_ptr = out_ptr + b * stride_ob + h * stride_oh \
                   + offs_m[:, None] * stride_oc \
                   + offs_n[None, :] * stride_ok
    tl.store(out_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

def trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    stride_lb, stride_lc, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rc, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oc, stride_ok, stride_oh = out.stride()

    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_lb=stride_lb,
        stride_lc=stride_lc,
        stride_lk=stride_lk,
        stride_lh=stride_lh,
        stride_rb=stride_rb,
        stride_rc=stride_rc,
        stride_rk=stride_rk,
        stride_rh=stride_rh,
        stride_ob=stride_ob,
        stride_oc=stride_oc,
        stride_ok=stride_ok,
        stride_oh=stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out

# --------------------------------------------------------------
# custom_kernel – the public entry point used by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]   torch.float32 on CUDA
        - mask         : [B, N, N]        torch.float32 (0/1) or None
        - weights_dict : name → torch.Tensor (fp32)
        - config_dict  : {"dim": int, "hidden_dim": int}
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # shape [B,N,N,dim]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    def linear_no_bias(t, w):
        # t: [..., dim] , w: [hidden_dim, dim]
        return torch.einsum("...d,hd->...h", t, w)

    left  = linear_no_bias(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = linear_no_bias(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)            # [...,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(linear_no_bias(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_no_bias(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear_no_bias(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ reduction – Triton GEMM
    # ------------------------------------------------------------------
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # out_tmp : [B,N,N,hidden] fp32
    out_tmp = trimul_gemm(left_f16, right_f16)

    # ------------------------------------------------------------------
    # 6️⃣ Post‑process: LayerNorm over hidden_dim, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_tmp,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                         # gating after norm

    # final projection back to original dim
    out = torch.einsum("...h,dh->...d", out, weights["to_out.weight"])

    return out  # already float32
```

**Key points**

* The only `N³` work (the double‑sum over `k`) lives in the tiled GEMM kernel,
  giving orders‑of‑magnitude speed‑up over the naïve PyTorch loops.
* All other operations (LayerNorm, linear layers, sigmoid, final projection)
  are kept in PyTorch – they are inexpensive compared to the `N³` term and
  avoid unnecessary kernel complexity.
* The kernel works for both masked and un‑masked inputs, any batch size ≤ 2,
  any sequence length allowed by the test suite, and any hidden dimension.
* Output is returned as `float32`, matching the reference implementation.
