"""
TriMul “outgoing” operator – forward pass only.

The operator works on a 4‑D tensor `x` of shape [B, N, N, dim] and performs:
  1. Layer‑norm over the last dimension (dim).
  2. Two bias‑free linear projections (left/right) into a hidden dimension H.
  3. Optional pair‑wise mask applied to the projected tensors.
  4. Gating (sigmoid) on the projections.
  5. Core O(N³) reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
     implemented as a batched GEMM with the right tensor transposed on the
     second sequence axis.  The reduction is executed in a custom Triton kernel
     that tiles the computation (BLOCK_M×BLOCK_N×BLOCK_K) and accumulates in
     fp32 while loading in fp16.
  6. Layer‑norm over the hidden dimension, gating, and a final linear
     projection back to `dim`.

Only the reduction (step 5) lives in Triton; everything else stays in
PyTorch for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: tiled GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    # pointers
    left_ptr,          # [B, N, N, H] (fp16)
    right_ptr,         # [B, N, N, H] (fp16) – accessed transposed on dim‑2
    out_ptr,           # [B, N, N, H] (fp32)

    # problem sizes
    B, N, H,
    # strides (in elements, NOT bytes)
    stride_lb, stride_lc, stride_lk, stride_lh,   # left: B,N,i,k,H
    stride_rb, stride_rc, stride_rk, stride_rh,   # right: B,N,j,k,H
    stride_ob, stride_oc, stride_ok, stride_oh,   # out:   B,N,i,j,H

    # tile sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute a tile of the output for a fixed (batch, head) pair:
        out[b, i:i+M, j:j+N, h] = Σ_k left[b, i:i+M, k, h] * right[b, j:j+N, k, h]

    `right` is logically transposed on its second sequence dimension,
    therefore we load it as (k, j) instead of (j, k).
    """
    pid = tl.program_id(0)                # linear index over (B, H)
    b = pid // H                           # batch index
    h = pid % H                            # head index

    pid_m = tl.program_id(1)               # i‑tile
    pid_n = tl.program_id(2)               # j‑tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑coordinates
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑coordinates

    mask_m = offs_m < N
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Pointers to the start of the current tile (batch & head are fixed)
    # --------------------------------------------------------------
    left_tile_ptr = (
        left_ptr
        + b * stride_lb
        + h * stride_lh
        + offs_m[:, None] * stride_lc                # i dimension
        + tl.arange(0, BLOCK_K)[None, :] * stride_lk # k dimension
    )
    # right is accessed as (k, j) -> we swap the axes when forming the pointer
    right_tile_ptr = (
        right_ptr
        + b * stride_rb
        + h * stride_rh
        + tl.arange(0, BLOCK_K)[:, None] * stride_rk # k dimension
        + offs_n[None, :] * stride_rc                # j dimension (as column)
    )

    # accumulator in fp32 (more stable than fp16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load left tile (M x K)
        left = tl.load(
            left_tile_ptr,
            mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < N),
            other=0.0,
        )
        # load right tile (K x N) – already transposed
        right = tl.load(
            right_tile_ptr,
            mask=(k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < N)
                 & mask_n[None, :],
            other=0.0,
        )

        # cast to fp16 for the dot, accumulate in fp32
        left = left.to(tl.float16)
        right = right.to(tl.float16)
        acc += tl.dot(left, right)   # (M,K)·(K,N) → (M,N)

        # advance to the next k‑slice
        left_tile_ptr += BLOCK_K * stride_lk
        right_tile_ptr += BLOCK_K * stride_rk

    # --------------------------------------------------------------
    # Write the tile back
    # --------------------------------------------------------------
    out_tile_ptr = (
        out_ptr
        + b * stride_ob
        + h * stride_oh
        + offs_m[:, None] * stride_oc   # i dimension
        + offs_n[None, :] * stride_ok   # j dimension
    )
    tl.store(out_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


def _trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for TriMul reduction.

    Arguments
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H] in fp16.
    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H] in fp32.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # strides in number of elements (not bytes)
    stride_lb, stride_lc, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rc, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oc, stride_ok, stride_oh = out.stride()

    # Tile sizes chosen for Hopper (SM‑90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                              # program_id(0)  → batch * head
        triton.cdiv(N, BLOCK_M),            # i‑tiles
        triton.cdiv(N, BLOCK_N),            # j‑tiles
    )

    _trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_lb=stride_lb,
        stride_lc=stride_lc,
        stride_lk=stride_lk,
        stride_lh=stride_lh,
        stride_rb=stride_rb,
        stride_rc=stride_rc,
        stride_rk=stride_rk,
        stride_rh=stride_rh,
        stride_ob=stride_ob,
        stride_oc=stride_oc,
        stride_ok=stride_ok,
        stride_oh=stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,        # enough for our tile size on H200
    )
    return out


# ----------------------------------------------------------------------
# Full custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        * input_tensor : torch.Tensor [B, N, N, dim] (float32, CUDA)
        * mask         : torch.Tensor [B, N, N] (float32 0/1) or None
        * weights_dict : mapping from weight name to torch.Tensor (float32)
        * config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last (dim) dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2. Bias‑free linear projections to hidden_dim
    #    Using einsum is fast and keeps the code simple.
    # ------------------------------------------------------------------
    def linear_no_bias(t, w):
        # t : [..., dim]   w : [hidden_dim, dim]
        return torch.einsum("...d,hd->...h", t, w)

    left  = linear_no_bias(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = linear_no_bias(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional pair‑wise mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [..., N, N, 1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – also bias‑free linear + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(linear_no_bias(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(linear_no_bias(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(linear_no_bias(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) reduction via Triton GEMM
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (accuracy is fine for the test suite)
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Triton kernel returns fp32
    out = _trimul_gemm(left_fp16, right_fp16)   # shape [B, N, N, H] fp32

    # ------------------------------------------------------------------
    # 6. Post‑processing: hidden‑dim LayerNorm, gating, final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    out = out * out_gate   # gating after norm

    # final projection back to original dim
    out = torch.einsum("...h,dh->...d", out, weights["to_out.weight"])

    # Ensure the dtype matches the reference implementation
    return out.to(torch.float32)
