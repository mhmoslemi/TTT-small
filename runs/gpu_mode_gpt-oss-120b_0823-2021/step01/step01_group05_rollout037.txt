"""
TriMul “outgoing” kernel (AlphaFold‑3).

The expensive O(N³·C) reduction is performed entirely in a single Triton
kernel.  For every batch * hidden‑dim (BH) we compute

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which is equivalent to a batched matrix multiplication
    out_h = left_h @ right_hᵀ   with   left_h, right_h ∈ ℝ^{B·N×N}.

The kernel tiles the k‑dimension (BLOCK_K) and the output i‑j plane
(BLOCK_I × BLOCK_J).  All arithmetic is carried out in fp32 (the
input tensors are fp32) and the final tensor is returned in the same layout
as the reference PyTorch implementation.

Only the reduction is done in Triton – the surrounding LayerNorm,
Linear‑projections, gating and the final LayerNorm / output projection are
implemented with ordinary PyTorch calls (these are cheap compared to the
O(N³·C) work and benefit from the highly‑optimised cuBLAS kernels).  The mask,
when present, is also applied with a tiny Triton helper (memory‑bound).

The implementation respects the launch‑convention of Triton 3.3.1 and runs
efficiently on Hopper GPUs (H200/H100).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Tiny kernel to apply the optional pair‑wise mask (broadcasted on the last dim)
# -------------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    src_ptr,          # const float*   (input  [B,N,N,H])
    mask_ptr,         # const float*   (mask   [B,N,N])
    dst_ptr,          # float*         (output [B,N,N,H])
    B, N, H,
    stride_src_b, stride_src_i, stride_src_j, stride_src_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    total_el: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_el

    # linear index → (b,i,j,h)
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    src = tl.load(src_ptr + b * stride_src_b
                         + i * stride_src_i
                         + j * stride_src_j
                         + h * stride_src_h, mask=mask, other=0.0)

    m = tl.load(mask_ptr + b * stride_mask_b
                         + i * stride_mask_i
                         + j * stride_mask_j, mask=mask, other=1.0)

    tl.store(dst_ptr + offs, src * m, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """tensor: [B,N,N,H] , mask: [B,N,N]  →  tensor*mask[...,None]"""
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024

    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_f = mask.view(-1)

    # strides for the flattened view
    stride_src_b, stride_src_i, stride_src_j, stride_src_h = tensor.stride()
    stride_mask_b, stride_mask_i, stride_mask_j = mask.stride()

    grid = (triton.cdiv(total, BLOCK),)

    apply_mask_kernel[grid](
        src,
        mask_f,
        dst,
        B, N, H,
        stride_src_b, stride_src_i, stride_src_j, stride_src_h,
        stride_mask_b, stride_mask_i, stride_mask_j,
        total,
        BLOCK=BLOCK,
    )
    return dst.view_as(tensor)


# -------------------------------------------------------------------------
# Core O(N³·C) reduction kernel
# -------------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const float*   [B,N,N,H]
    right_ptr,         # const float*   [B,N,N,H]
    out_ptr,           # float*         [B,N,N,H]
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_k, stride_r_j, stride_r_h,   # note swapped j/k for transpose
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_i = tl.program_id(0)   # i‑block
    pid_j = tl.program_id(1)   # j‑block
    pid_bh = tl.program_id(2)  # batch * hidden‑dim index

    # derive b and h from pid_bh
    b = pid_bh // H
    h = pid_bh % H

    # tile start coordinates
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k‑tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K

        # compute actual tile sizes (boundary handling)
        i_max = tl.minimum(i_base + BLOCK_I, N)
        j_max = tl.minimum(j_base + BLOCK_J, N)
        k_max = tl.minimum(k_base + BLOCK_K, N)

        # offsets within the tile
        i_off = tl.arange(0, BLOCK_I)[:, None]
        j_off = tl.arange(0, BLOCK_J)[None, :]
        k_off = tl.arange(0, BLOCK_K)[None, :]

        # mask for out‑of‑bounds rows/cols
        i_mask = i_base + i_off < i_max
        j_mask = j_base + j_off < j_max
        k_mask = k_base + k_off < k_max

        # -------- load left slice  (i, k) ----------
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + (i_base + i_off) * stride_l_i
            + (k_base + k_off) * stride_l_k
            + h * stride_l_h,
            mask=i_mask & k_mask,
            other=0.0,
        )                      # (BLOCK_I, BLOCK_K)

        # -------- load right slice (j, k) -------------
        # we need right transposed: right[b, j, k, h] → we want (k, j)
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + (k_base + k_off) * stride_r_k   # k index becomes first
            + (j_base + j_off) * stride_r_j   # j index becomes second
            + h * stride_r_h,
            mask=j_mask & k_mask,
            other=0.0,
        )                      # (BLOCK_K, BLOCK_J)

        # dot product over k
        # left: (I,K) , right: (K,J) → result (I,J)
        acc += tl.dot(left, right)   # fp32 accumulation

    # store result
    out = acc.to(tl.float32)

    # write back with masks for trailing edges
    i_store_mask = (i_base + tl.arange(0, BLOCK_I)[:, None]) < N
    j_store_mask = (j_base + tl.arange(0, BLOCK_J)[None, :]) < N
    store_mask = i_store_mask & j_store_mask

    tl.store(
        out_ptr
        + b * stride_o_b
        + (i_base + tl.arange(0, BLOCK_I)[:, None]) * stride_o_i
        + (j_base + tl.arange(0, BLOCK_J)[None, :]) * stride_o_j
        + h * stride_o_h,
        out,
        mask=store_mask,
    )


# -------------------------------------------------------------------------
# Entry‑point called by the test harness
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input, mask, weights, config)
        - input : torch.Tensor [B, N, N, dim]      (float32)
        - mask  : torch.Tensor [B, N, N]            (float32, 0/1) – may be all‑ones
        - weights: dict of model parameters (see code below)
        - config : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper: LayerNorm without torch.nn (to keep everything on‑device)
    # ------------------------------------------------------------------
    def layer_norm(tensor, w, b, eps=1e-5):
        mean = tensor.mean(-1, keepdim=True)
        var = tensor.var(-1, unbiased=False, keepdim=True)
        return (tensor - mean) * torch.rsqrt(var + eps) * w + b

    # ------------------------------------------------------------------
    # LayerNorm + Linear projections (PyTorch – fast & stable)
    # ------------------------------------------------------------------
    x = layer_norm(
        x,
        weights["norm.weight"],
        weights["norm.bias"],
    )

    # Linear layers – biasless
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (apply with Triton – cheap compared to O(N³))
    # ------------------------------------------------------------------
    if not nomask:
        left = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # Gates (PyTorch)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"]
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"]
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·C) reduction – Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    H = hidden_dim

    # allocate output
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # strides (contiguous layout)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    # right will be accessed as (b, k, j, h) → need strides for k and j swapped
    stride_r_b, stride_r_i, stride_r_j, stride_r_h = right.stride()
    # treat i ↔ k for transposed access
    stride_r_k = stride_r_i
    stride_r_j = stride_r_j

    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tile sizes – tuned for Hopper (shared‑memory friendly)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_I),           # i‑blocks
        triton.cdiv(N, BLOCK_J),           # j‑blocks
        B * H,                             # batch * hidden‑dim
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_k, stride_r_j, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Final LayerNorm, gating and output projection (PyTorch)
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
