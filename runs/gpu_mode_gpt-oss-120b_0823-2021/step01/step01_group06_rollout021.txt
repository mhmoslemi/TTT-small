"""
TriMul “outgoing” kernel (forward only).

The heavy N³ operation
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply per hidden dimension `d`.  We keep the
LayerNorm / Linear / gating steps in PyTorch (they are memory‑bound and
already fused by torch.compile) and move the only non‑fused element‑wise
work – the optional mask‑application – into a tiny Triton kernel.
The reduction itself is performed with a single‑precision batched
`torch.bmm`, which on an H200 is far faster than a hand‑written Triton
reduction for the supported sizes (N ≤ 1024, hidden ≤ 384).

The Triton kernel works with arbitrary tensor strides and uses a 1‑D
grid.  It multiplies a tensor of shape `[B,N,N,H]` by a broadcasted mask
`[B,N,N,1]` (or a scalar mask when `nomask=True`).  All other operations
remain in PyTorch for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: apply a 4‑D mask (broadcasted over the hidden dimension)
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    x_ptr,               # pointer to input tensor   [B,N,N,H]
    mask_ptr,            # pointer to mask tensor    [B,N,N,1]  (or scalar)
    out_ptr,             # pointer to output tensor  [B,N,N,H]
    total_elems,         # B*N*N*H  (number of elements we will process)
    MASK_STRIDE_B,       # stride of mask in batch dim (in elements)
    MASK_STRIDE_N,       # stride of mask in seq dim   (in elements)
    X_STRIDE_B, X_STRIDE_N1, X_STRIDE_N2, X_STRIDE_H,  # strides of x
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    mask = offs < total_elems

    # ------------------------------------------------------------------
    # Load the data element
    # ------------------------------------------------------------------
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)

    # ------------------------------------------------------------------
    # Compute the offset of the corresponding mask element.
    #   mask has shape [B,N,N,1] → we need to drop the hidden dimension.
    #   The linear offset of a mask element is the same as the
    #   linear offset of x divided by the hidden‑stride.
    # ------------------------------------------------------------------
    mask_offs = offs // X_STRIDE_H
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)   # 1.0 = no‑mask

    tl.store(out_ptr + offs, x * m, mask=mask)


# ----------------------------------------------------------------------
# Helper: fused LayerNorm (bias + weight are supplied)
# ----------------------------------------------------------------------
def fused_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """
    LayerNorm over the last dimension, without the extra
    mean/var buffers that torch.nn.LayerNorm creates.
    """
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point (called by the evaluator)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (dim → dim)
    # ------------------------------------------------------------------
    x = fused_layernorm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (dim → hidden) – bias‑free
    # ------------------------------------------------------------------
    # F.linear expects shape [..., C] and weight shape [out, in]
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask‑application (mask shape [B,N,N]) – Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N]; we need [B,N,N,1] for broadcasting
        mask_exp = mask.unsqueeze(-1).contiguous()
        total = left.numel()                     # B * N * N * hidden
        BLOCK = 131072                           # ~512 KiB of fp32 per block
        grid = (triton.cdiv(total, BLOCK),)

        # allocate temporaries
        left_masked = torch.empty_like(left)
        right_masked = torch.empty_like(right)

        # launch kernel for left
        mask_apply_kernel[grid](
            left.data_ptr(),
            mask_exp.data_ptr(),
            left_masked.data_ptr(),
            total,
            mask_exp.stride(0), mask_exp.stride(1),           # mask strides (B,N)
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            BLOCK=BLOCK,
        )
        # launch kernel for right
        mask_apply_kernel[grid](
            right.data_ptr(),
            mask_exp.data_ptr(),
            right_masked.data_ptr(),
            total,
            mask_exp.stride(0), mask_exp.stride(1),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            BLOCK=BLOCK,
        )
        left, right = left_masked, right_masked

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – keep FP32
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,hidden]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B,N,N,hidden]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ reduction:
    #     out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    We reshape to (B*hidden, N, N) and use torch.bmm which on the
    #    H200 maps to a highly‑tuned GEMM.
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N)
    left_rs = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_rs = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N) # (B*H, N, N)

    # batched matrix multiplication  (B*H, N, N) @ (B*H, N, N)^T
    out_rs = torch.bmm(left_rs, right_rs.transpose(1, 2))          # (B*H, N, N)

    # restore shape [B, N, N, hidden]
    out = (
        out_rs.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (hidden → hidden) and final projection
    # ------------------------------------------------------------------
    out = fused_layernorm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to original dim

    # guarantee float32 output (all ops above used fp32)
    return out.to(torch.float32)
