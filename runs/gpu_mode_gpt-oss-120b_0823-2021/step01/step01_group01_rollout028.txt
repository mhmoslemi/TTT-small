**What remains to improve?**

1. **Correct grid definition** – a 3‑D grid `(B, H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))` eliminates the massive `grid = (triton.cdiv(...),)` used before.
2. **Better launch parameters** – choose tile sizes that keep the per‑CTA shared‑memory footprint low while still providing enough work. For `N ≤ 512` we use `64×64×32`; for larger `N` we fall back to `32×32×64` which reduces the number of tiles and improves cache reuse.

```python
"""
TriMul “outgoing” kernel (AlphaFold3) – optimized launch.

The heavy part of the module is the batched matrix multiplication
for each hidden dimension:

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]

which is exactly (left_h) @ (right_h).T for every (b, h) pair.
The kernel computes a tile of rows × cols while iterating over the K‑dimension
in tiles.  Computation is performed in FP16 with FP32 accumulation and the
result is written back as FP16.  All LayerNorm / Linear / gating work stays
in PyTorch – we only move the N³ multiplication to Triton.

Key optimisations
-----------------
* 3‑D launch grid (batch, head, tile‑M, tile‑N) instead of a flattened 1‑D grid.
* Two static tile configurations; a tiny runtime “autotune” picks the
  configuration that best fits the current sequence length.
* Block sizes ≤ 64 keep shared‑memory usage < 8 KB per CTA on Hopper.
* No extra mask handling inside the kernel – the mask is applied in the
  PyTorch pre‑processing step (already cheap).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -------------------------------------------------------------------------
# Triton kernel: batched matmul for each hidden dimension
# -------------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # const *fp16   [B, H, N, N]
    right_ptr,         # const *fp16   [B, H, N, N]
    out_ptr,           # mut   *fp16   [B, H, N, N]
    B, N, H,           # sizes
    stride_lb, stride_lh, stride_ln, stride_lk,   # left strides (B,H,N,N)
    stride_rb, stride_rh, stride_rn, stride_rk,   # right strides
    stride_ob, stride_oh, stride_on, stride_ok,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute for a single (batch, head) tile:
        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
    """
    pid_b = tl.program_id(0)   # batch dimension
    pid_h = tl.program_id(1)   # head dimension
    pid_m = tl.program_id(2)   # tile row
    pid_n = tl.program_id(3)   # tile col

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over K dimension in tiles
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k]  shape (M, K)
        l_ptr = (left_ptr
                 + pid_b * stride_lb
                 + pid_h * stride_lh
                 + offs_m[:, None] * stride_ln
                 + offs_k[None, :] * stride_lk)
        l = tl.load(l_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[j,k] we need the transpose (k, N) → (k, j)
        r_ptr = (right_ptr
                 + pid_b * stride_rb
                 + pid_h * stride_rh
                 + offs_k[:, None] * stride_rk   # (K, 1)
                 + offs_n[None, :] * stride_rn)  # (1, N)
        r = tl.load(r_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # dot-product over K
        acc += tl.dot(l, r)   # (M,K) @ (K,N) -> (M,N)

    # --------------------------------------------------------------
    # write back
    # --------------------------------------------------------------
    out_ptrs = (out_ptr
                + pid_b * stride_ob
                + pid_h * stride_oh
                + offs_m[:, None] * stride_on
                + offs_n[None, :] * stride_ok)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------------------
# Host‑side wrapper
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    The N³ multiplication is executed by `trimul_kernel`.
    Everything else (LayerNorm, Linear, gating, final projection) stays in
    native PyTorch for simplicity and because it is not the performance bottleneck.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. LayerNorm + Linear projections (PyTorch)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype)
    )

    # Linear projections (bias=False)
    left  = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # -----------------------------------------------------------------
    # 2. Apply mask (optional) – broadcast over hidden dimension
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)               # (B,N,N,1)
        left  = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 3. Gating (sigmoid linear)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Rearrange to [B, H, N, N] and cast to fp16 for the kernel
    # -----------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)   # (B,H,N,N) fp16

    # -----------------------------------------------------------------
    # 5. Choose tile configuration (tiny runtime autotune)
    # -----------------------------------------------------------------
    # For very large N we reduce BLOCK_M/N to keep the number of tiles moderate.
    if N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 32, 32, 64

    # Grid: (batch, head, tiles_m, tiles_n)
    grid = (
        B,
        hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # -----------------------------------------------------------------
    # 6. Launch kernel
    # -----------------------------------------------------------------
    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        out_fp16.stride(0), out_fp16.stride(1), out_fp16.stride(2), out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7. Back to (B,N,N,hidden) and continue in PyTorch
    # -----------------------------------------------------------------
    out = out_fp16.to(dtype).permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden)

    # to_out_norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype)
    )

    # multiply by out_gate (broadcast over N,N)
    out = out * out_gate

    # final linear back to dim
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
```
