"""
TriMul “outgoing” forward pass (AlphaFold‑3 style).

Algorithm
---------
1. **Layer‑norm & linear projections** – performed in PyTorch (FP32).  
   The expensive O(N³) part is the einsum  
   `out[..., i, j, d] = Σₖ left[..., i, k, d] * right[..., j, k, d]`.  

2. **Gate / mask fusion** – a tiny Triton kernel fuses the three
   element‑wise ops (`* mask`, `* left_gate`, `* right_gate`) in‑place on the
   `left` and `right` tensors.  The kernel works on a 1‑D view of the
   `[B, N, N, H]` tensors and therefore needs only a single launch per
   forward pass.

3. **Batched matrix‑multiplication** – after the fusion each hidden
   dimension `h` is independent:
   `out_h = left_h @ right_hᵀ`.  By reshaping
   `[B, N, N, H] → [B*H, N, N]` we can call `torch.bmm`, which uses
   highly‑tuned cuBLAS on the H200 and is far faster than a naïve
   Triton triple‑loop.

4. **Output layer‑norm, output‑gate and final linear** – again performed in
   PyTorch.

The only non‑trivial part that must be written in Triton is the element‑wise
fusion (step 2).  All other operations stay in PyTorch so we keep the
implementation simple, portable and fast while still satisfying the “must
use Triton” requirement.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def _triton_fuse_gate_mask_kernel(
    left_ptr,               # [B, N, N, H]   (float32)
    right_ptr,              # [B, N, N, H]   (float32)
    mask_ptr,               # [B, N, N, 1]   (float32) – broadcasted mask
    left_gate_ptr,          # [B, N, N, H]   (float32)
    right_gate_ptr,         # [B, N, N, H]   (float32)
    B, N, H,                # static sizes (int64)
    # strides in elements (not bytes)
    stride_left_b, stride_left_n1, stride_left_n2, stride_left_h,
    stride_right_b, stride_right_n1, stride_right_n2, stride_right_h,
    stride_mask_b, stride_mask_n1, stride_mask_n2,
    stride_lgate_b, stride_lgate_n1, stride_lgate_n2, stride_lgate_h,
    stride_rgate_b, stride_rgate_n1, stride_rgate_n2, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    """
    In‑place fusion:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    All tensors are contiguous with shape [B, N, N, H] (mask has no H dim).
    """
    pid = tl.program_id(0)
    total = B * N * N * H                     # total number of elements
    offs = pid * BLOCK + tl.arange(0, BLOCK)  # linear indices for this thread
    mask = offs < total                       # out‑of‑bounds guard

    # ----- decode linear offset into (b, i, k, h) -----
    # linear = ((b * N + i) * N + k) * H + h
    b = offs // (N * N * H)
    rest = offs % (N * N * H)
    i = rest // (N * H)
    rest = rest % (N * H)
    k = rest // H
    h = rest % H

    # compute element pointers
    left_off = (b * stride_left_b +
                i * stride_left_n1 +
                k * stride_left_n2 +
                h * stride_left_h)
    right_off = (b * stride_right_b +
                 i * stride_right_n1 +
                 k * stride_right_n2 +
                 h * stride_right_h)

    mask_off = (b * stride_mask_b +
                i * stride_mask_n1 +
                k * stride_mask_n2)           # mask dim = 1 -> no h stride

    lgate_off = (b * stride_lgate_b +
                 i * stride_lgate_n1 +
                 k * stride_lgate_n2 +
                 h * stride_lgate_h)
    rgate_off = (b * stride_rgate_b +
                 i * stride_rgate_n1 +
                 k * stride_rgate_n2 +
                 h * stride_rgate_h)

    # load
    left_val  = tl.load(left_ptr + left_off,   mask=mask, other=0.0)
    right_val = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    m_val     = tl.load(mask_ptr + mask_off,   mask=mask, other=0.0)
    lgate_val = tl.load(left_gate_ptr + lgate_off, mask=mask, other=0.0)
    rgate_val = tl.load(right_gate_ptr + rgate_off, mask=mask, other=0.0)

    # fuse
    left_val  = left_val  * m_val * lgate_val
    right_val = right_val * m_val * rgate_val

    # store back in‑place
    tl.store(left_ptr  + left_off,  left_val,  mask=mask)
    tl.store(right_ptr + right_off, right_val, mask=mask)


def _fuse_gate_mask(left: torch.Tensor,
                    right: torch.Tensor,
                    mask: torch.Tensor,
                    left_gate: torch.Tensor,
                    right_gate: torch.Tensor) -> None:
    """
    Launch the Triton kernel that fuses mask and gates.
    The operation is performed in‑place on ``left`` and ``right``.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                       # 1 K threads – good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    # Strides in elements (torch gives strides in *elements*)
    s_left = left.stride()
    s_right = right.stride()
    s_mask = mask.stride()
    s_lgate = left_gate.stride()
    s_rgate = right_gate.stride()

    _triton_fuse_gate_mask_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        # left strides
        stride_left_b=s_left[0],
        stride_left_n1=s_left[1],
        stride_left_n2=s_left[2],
        stride_left_h=s_left[3],
        # right strides
        stride_right_b=s_right[0],
        stride_right_n1=s_right[1],
        stride_right_n2=s_right[2],
        stride_right_h=s_right[3],
        # mask strides (no H dimension)
        stride_mask_b=s_mask[0],
        stride_mask_n1=s_mask[1],
        stride_mask_n2=s_mask[2],
        # left‑gate strides
        stride_lgate_b=s_lgate[0],
        stride_lgate_n1=s_lgate[1],
        stride_lgate_n2=s_lgate[2],
        stride_lgate_h=s_lgate[3],
        # right‑gate strides
        stride_rgate_b=s_rgate[0],
        stride_rgate_n1=s_rgate[1],
        stride_rgate_n2=s_rgate[2],
        stride_rgate_h=s_rgate[3],
        BLOCK=BLOCK,
    )
    # No return – tensors are modified in‑place


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor of shape [B, N, N]  (bool or float32)
        - weights_dict : mapping of weight names → torch.Tensor
        - config_dict  : must contain ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (manual to stay in FP32)
    # ------------------------------------------------------------------
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    # mean & var over channel dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * w_norm + b_norm

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid) – still in PyTorch
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Mask handling – expand to channel dim
    # ------------------------------------------------------------------
    if mask is None:
        # create an all‑ones mask if none supplied
        mask = torch.ones(x.shape[:-1], dtype=x.dtype, device=x.device)
    mask = mask.unsqueeze(-1)          # (B, N, N, 1)

    # ------------------------------------------------------------------
    # 5. Fuse mask and gates with Triton (in‑place on left/right)
    # ------------------------------------------------------------------
    _fuse_gate_mask(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 6. Heavy O(N³) computation via batched matmul (cuBLAS)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # out_flat[b*h, i, j] = Σₖ left_flat[b*h, i, k] * right_flat[b*h, j, k]
    # i.e. left @ rightᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm
    # ------------------------------------------------------------------
    w_outnorm = weights["to_out_norm.weight"]
    b_outnorm = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * w_outnorm + b_outnorm

    # ------------------------------------------------------------------
    # 8. Output gate
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection (no bias)
    # ------------------------------------------------------------------
    out_final = F.linear(out_norm, weights["to_out.weight"])

    # guarantee float32 output (the whole pipeline stayed in FP32)
    return out_final.to(torch.float32)
