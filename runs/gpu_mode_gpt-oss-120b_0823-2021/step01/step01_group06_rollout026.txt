"""
TriMul “outgoing” kernel (forward only).

The expensive N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply for every hidden dimension `d`.
We fuse this reduction in a single Triton kernel that
 * loads tiles of the left‑matrix (i,k) and the right‑matrix (j,k) (transposed)
 * computes a dot‑product tile with tl.dot
 * accumulates in fp32 (the input is fp32, we keep fp32 for accuracy)
 * writes the result tile to the output.

All other operations (LayerNorm, Linear projections, gating) are left to
PyTorch – they are memory bound and already highly tuned by torch.compile.
The kernel therefore satisfies the “must contain at least one Triton region”
requirement while giving a large speed‑up over the naïve Python loops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def trimul_out_kernel(
    left_ptr,          # [B*H, N, N]   fp32
    right_ptr,         # [B*H, N, N]   fp32
    out_ptr,           # [B*H, N, N]   fp32
    N,                 # sequence length (int)
    stride_lbh,        # stride over batch*hidden dimension
    stride_lb_n,       # stride over first N (row)
    stride_lb_k,       # stride over second N (col / reduction axis)
    stride_rbh,        # same for right
    stride_rb_n,
    stride_rb_k,
    stride_obh,        # same for out
    stride_ob_n,
    stride_ob_k,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Each program works on a tile (m, n) of the output matrix for a fixed
    batch‑hidden index b_h = program_id(0).  The reduction axis k is tiled
    in a loop.
    """
    pid_bh = tl.program_id(0)          # batch * hidden index
    pid_m = tl.program_id(1)           # tile row
    pid_n = tl.program_id(2)           # tile col

    # starting offsets of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers for the current batch/hidden slice
    left_base = left_ptr + pid_bh * stride_lbh
    right_base = right_ptr + pid_bh * stride_rbh
    out_base = out_ptr + pid_bh * stride_obh

    # initialise accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile   [M, K]
        left_tile = tl.load(
            left_base
            + offs_m[:, None] * stride_lb_n
            + offs_k[None, :] * stride_lb_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load right tile  [N, K]  (note: we need right transposed, so we load
        # rows = offs_n, cols = offs_k and later dot with left)
        right_tile = tl.load(
            right_base
            + offs_n[:, None] * stride_rb_n
            + offs_k[None, :] * stride_rb_k,
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0,
        )
        # compute dot product over K dimension
        acc += tl.dot(left_tile, right_tile, allow_tf32=False)

    # write result
    out_ptrs = out_base + offs_m[:, None] * stride_ob_n + offs_n[None, :] * stride_ob_k
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: fused LayerNorm (bias‑free version, bias added later)
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (dim → hidden, bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])     # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – also bias‑free linear
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ reduction using Triton
    #    We reshape to (B*H, N, N) and run the kernel.
    # ------------------------------------------------------------------
    # reshape: [B, N, N, H] -> [B*H, N, N]
    left_rs = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_rs = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    out_rs = torch.empty_like(left_rs)

    # launch parameters (tuned for H200, N ≤ 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(B * hidden, 1),               # batch*hidden dimension
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    trimul_out_kernel[grid](
        left_rs,
        right_rs,
        out_rs,
        N,
        left_rs.stride(0), left_rs.stride(1), left_rs.stride(2),
        right_rs.stride(0), right_rs.stride(1), right_rs.stride(2),
        out_rs.stride(0), out_rs.stride(1), out_rs.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden]
    out = out_rs.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating + final projection
    # ------------------------------------------------------------------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # ensure float32 output
    return out.to(torch.float32)
