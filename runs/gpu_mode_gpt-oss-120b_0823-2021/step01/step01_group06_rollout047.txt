"""
TriMul “outgoing” kernel (forward only).

* The heavy N³ reduction is performed with a batched GEMM
  (torch.bmm) after reshaping the tensors – this is far faster on
  Hopper than a hand‑written Triton reduction for the supported sizes
  (N ≤ 1024, hidden_dim ≤ 384).

* The only operation moved to Triton is the element‑wise mask application
  (when a mask is present).  The kernel works with an arbitrary
  contiguous or strided layout and uses a single 1‑D grid.  It multiplies
  a tensor of shape [B,N,N,H] by a broadcasted mask of shape [B,N,N,1].
  The kernel is deliberately simple, safe and respects the stride
  arguments to avoid the “stride‑unpacking” bugs seen in earlier attempts.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# Triton kernel: apply mask (broadcasted on the hidden dimension)
# --------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    src_ptr,                # [B,N,N,H] source tensor
    mask_ptr,               # [B,N,N,1] mask tensor (broadcasted)
    dst_ptr,                # [B,N,N,H] destination tensor
    total_elements,        # B*N*N*H  (scalar)
    src_stride_b, src_stride_n1, src_stride_n2, src_stride_h,   # strides of src
    mask_stride_b, mask_stride_n1, mask_stride_n2,               # strides of mask (no H stride)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # ------------------------------------------------------------------
    # Linear index → multi‑dimensional indices (b,i,j,k)
    # ------------------------------------------------------------------
    # Compute the index of the hidden dimension (k) first because mask does
    # not have that dimension, then derive the mask linear offset.
    k = offs % src_stride_h                     # offset inside hidden dim (0..H-1)
    idx_without_h = offs // src_stride_h        # linear index in [B,N,N] space
    # derive mask offset (same as idx_without_h because mask has no hidden dim)
    mask_off = idx_without_h * 1                # *1 for clarity

    # Load values
    src = tl.load(src_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)

    tl.store(dst_ptr + offs, src * m, mask=mask)


# --------------------------------------------------------------
# Helper: fused LayerNorm (bias + weight, eps=1e-5)
# --------------------------------------------------------------
def fused_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x : [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim → dim)
    # ------------------------------------------------------------------
    x = fused_layernorm(inp, weights["norm.weight"], weights["norm.bias"])

    # ------------------------------------------------------------------
    # 2. Linear projections (dim → hidden, bias‑free)
    # ------------------------------------------------------------------
    # shape after projection: [B,N,N,hidden]
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask application (broadcasted on hidden dim) via Triton
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → [B,N,N,1] for broadcasting
        mask_exp = mask.unsqueeze(-1).contiguous()
        total = left.numel()
        BLOCK = 131072                     # ~128 Ki elements (fits L2)
        grid = (triton.cdiv(total, BLOCK),)

        # left
        apply_mask_kernel[grid](
            left.data_ptr(),
            mask_exp.data_ptr(),
            left.data_ptr(),               # in‑place is fine
            total,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            mask_exp.stride(0), mask_exp.stride(1), mask_exp.stride(2),
            BLOCK=BLOCK,
        )
        # right (same strides)
        apply_mask_kernel[grid](
            right.data_ptr(),
            mask_exp.data_ptr(),
            right.data_ptr(),
            total,
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask_exp.stride(0), mask_exp.stride(1), mask_exp.stride(2),
            BLOCK=BLOCK,
        )

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – also bias‑free linear layers
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ reduction: out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    #    We reshape to (B*hidden, N, N) and use torch.bmm which maps to a
    #    highly‑tuned GEMM on Hopper.
    # ------------------------------------------------------------------
    # (B,N,N,hidden) → (B*hidden, N, N)
    left_rs  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_rs = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # Batched matmul: (B*H, N, N) @ (B*H, N, N)^T → (B*H, N, N)
    out_rs = torch.bmm(left_rs, right_rs.transpose(1, 2))

    # Restore original shape [B, N, N, hidden]
    out = out_rs.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden → hidden) + gating + final linear
    # ------------------------------------------------------------------
    out = fused_layernorm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # Ensure final dtype is float32 (all ops were in FP32)
    return out.to(torch.float32)
