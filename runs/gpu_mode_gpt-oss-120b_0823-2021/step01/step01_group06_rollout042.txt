"""
TriMul “outgoing” forward (no gradients).

The only part that really benefits from a custom Triton kernel is the
mask‑application (when a mask is present).  All other ops – LayerNorm,
linear projections, gating and the N³ reduction – are either memory‑bound
or map directly to a highly‑optimized batched GEMM (torch.bmm).  This
implementation keeps those in PyTorch (they are already fused by
torch.compile) and uses a small, 1‑D Triton kernel to multiply a
[B,N,N,H] tensor by a broadcasted mask [B,N,N,1] with correct stride
handling.

The algorithm
--------------
1. Layer‑norm the input.
2. Linear projections → left / right (dim → hidden).
3. If a mask is supplied, launch `apply_mask_kernel` on left and right.
4. Compute gating values (sigmoid) and apply them.
5. N³ core:   out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d] .
   This is a batched matrix‑multiply per hidden dimension and is performed
   with a single `torch.bmm` on tensors of shape (B*hidden, N, N).
6. Output LayerNorm + gating + final linear projection.

The kernel is deliberately simple: it loads a linear index, converts it
to a 4‑D address using the provided strides, loads the data and the mask,
multiplies and stores the result.  All strides are passed explicitly so
there is no assumption about tensor rank.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise mask multiplication (broadcast mask on last dim)
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    x_ptr,                # ptr to tensor [B,N,N,H]
    mask_ptr,             # ptr to mask   [B,N,N,1]
    out_ptr,              # ptr to output [B,N,N,H]
    total_elems,          # B*N*N*H  (int)
    stride_x_b, stride_x_n1, stride_x_n2, stride_x_h,   # strides of x
    stride_m_b, stride_m_n1, stride_m_n2,                # strides of mask (no H stride, broadcasted)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear offsets
    mask = offs < total_elems

    # ----- decode linear offset to (b, i, j, h) -----
    # offset = b*stride_x_b + i*stride_x_n1 + j*stride_x_n2 + h*stride_x_h
    # we recover each index by successive division/modulo.
    # Because strides are monotonic (b > i > j > h) we can use integer div.
    off = offs
    b = off // stride_x_b
    off = off - b * stride_x_b
    i = off // stride_x_n1
    off = off - i * stride_x_n1
    j = off // stride_x_n2
    h = off - j * stride_x_n2

    # compute addresses
    x_addr   = x_ptr   + b*stride_x_b   + i*stride_x_n1   + j*stride_x_n2   + h*stride_x_h
    mask_addr = mask_ptr + b*stride_m_b  + i*stride_m_n1  + j*stride_m_n2

    x_val   = tl.load(x_addr,   mask=mask, other=0.0)
    m_val   = tl.load(mask_addr,mask=mask, other=1.0)   # broadcasted over h
    tl.store(out_ptr + offs, x_val * m_val, mask=mask)


# ----------------------------------------------------------------------
# Helper: fused layer‑norm (bias + weight supplied)
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor,
                eps: float = 1e-5) -> torch.Tensor:
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,            # input tensor [B,N,N,dim]
        torch.Tensor,            # mask tensor [B,N,N]   (may be None)
        Dict[str, torch.Tensor],# model weights
        Dict[str, Any]           # config dict
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (dim → dim)
    # --------------------------------------------------------------
    x = _layer_norm(
        inp,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (dim → hidden), bias‑free
    # --------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask application (Triton)
    # --------------------------------------------------------------
    if not nomask:
        # broadcast mask to [B,N,N,1]
        mask_exp = mask.unsqueeze(-1).contiguous()
        total = left.numel()
        BLOCK = 131072                     # ~512 Ki elements (fp32)
        grid = (triton.cdiv(total, BLOCK),)

        # left
        apply_mask_kernel[grid](
            left.data_ptr(),
            mask_exp.data_ptr(),
            left.data_ptr(),               # in‑place store
            total,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            mask_exp.stride(0), mask_exp.stride(1), mask_exp.stride(2),
            BLOCK=BLOCK,
        )
        # right
        apply_mask_kernel[grid](
            right.data_ptr(),
            mask_exp.data_ptr(),
            right.data_ptr(),
            total,
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask_exp.stride(0), mask_exp.stride(1), mask_exp.stride(2),
            BLOCK=BLOCK,
        )

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) – dim → hidden
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core N³ reduction via batched GEMM
    #    out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    # --------------------------------------------------------------
    # reshape to (B*hidden, N, N)
    left_rs  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_rs = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_rs = torch.bmm(left_rs, right_rs.transpose(1, 2))

    # back to [B, N, N, hidden]
    out = out_rs.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm + out‑gate + final linear (hidden → dim)
    # --------------------------------------------------------------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
