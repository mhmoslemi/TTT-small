"""
TriMul “outgoing” forward pass (AlphaFold‑3 style).

The heavy N³ contraction 
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]          (d = hidden_dim)
is expressed as a batched matrix‑multiply per hidden channel and carried out
with torch.bmm (highly tuned on Hopper).  The only element‑wise work – the
optional mask‑application – is off‑loaded to a small Triton kernel.  LayerNorm,
Linear projections and the sigmoid gates remain in PyTorch (they are already
memory‑bound and benefit from torch.compile fusion).

The Triton kernel (`apply_mask_kernel`) multiplies a tensor of shape
[B, N, N, H] by a broadcasted mask of shape [B, N, N, 1] using a 1‑D grid and
pure‑linear indexing; no stride‑decoding is performed inside the kernel, which
keeps the launch overhead negligible.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton: element‑wise mask multiplication (supports nomask=False)
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    x_ptr,          # const float*  -> input tensor (B,N,N,H)   (row‑major)
    mask_ptr,       # const float*  -> mask tensor   (B,N,N,1)   (row‑major)
    out_ptr,        # float*        -> output tensor (B,N,N,H)
    total_elem,     # i32           -> number of elements in x (B*N*N*H)
    BLOCK: tl.constexpr,
):
    """Multiply a 4‑D tensor by a broadcasted mask (mask has an extra singleton dim)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    mask = offs < total_elem

    # Load the value from the input tensor.
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)

    # The mask is stored without the hidden dimension, i.e. shape (B,N,N,1).
    # To broadcast, we drop the last stride by integer‑division.
    mask_val = tl.load(mask_ptr + (offs // tl.broadcast_to(1, 1)), mask=mask, other=1.0)

    tl.store(out_ptr + offs, x * mask_val, mask=mask)


# ----------------------------------------------------------------------
# Helper: LayerNorm (bias + weight supplied)
# ----------------------------------------------------------------------
def layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """Fused LayerNorm, operates on the last dimension."""
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : (input, mask, weights, config)
        input   – torch.Tensor of shape [B, N, N, dim]    (float32)
        mask    – torch.Tensor of shape [B, N, N]        (float32 or bool)
        weights – dict containing the eleven weight / bias tensors
        config  – dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm  (dim → dim)
    # --------------------------------------------------------------
    x = layernorm(inp, weights["norm.weight"], weights["norm.bias"])

    # --------------------------------------------------------------
    # 2. Linear projections  (dim → hidden) – bias‑free
    # --------------------------------------------------------------
    # [B,N,N,H]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask application (Triton)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape -> [B,N,N,1] for broadcasting
        mask_exp = mask.unsqueeze(-1).contiguous()
        total = left.numel()
        BLOCK = 131072                     # ~0.5 MiB per launch, works for all test sizes
        grid = (triton.cdiv(total, BLOCK),)

        left_masked = torch.empty_like(left)
        right_masked = torch.empty_like(right)

        # left
        apply_mask_kernel[grid](
            left.data_ptr(),
            mask_exp.data_ptr(),
            left_masked.data_ptr(),
            total,
            BLOCK=BLOCK,
        )
        # right
        apply_mask_kernel[grid](
            right.data_ptr(),
            mask_exp.data_ptr(),
            right_masked.data_ptr(),
            total,
            BLOCK=BLOCK,
        )
        left, right = left_masked, right_masked

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) – bias‑free linear + sigmoid
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate          # [B,N,N,H]
    right = right * right_gate       # [B,N,N,H]

    # --------------------------------------------------------------
    # 5. Core N³ contraction via batched GEMM
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    We reshape to (B*H, N, N) and use torch.bmm which maps to a
    #    highly‑optimized GEMM on Hopper.
    # --------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    left_rs = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_rs = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # batched matrix multiply: (B*H,N,N) @ (B*H,N,N)^T → (B*H,N,N)
    out_rs = torch.bmm(left_rs, right_rs.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_rs.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm (hidden → hidden) and final linear projection
    # --------------------------------------------------------------
    out = layernorm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])

    # Ensure float32 output (all ops were in fp32)
    return out.to(torch.float32)
