"""
TriMul “outgoing” forward pass (AlphaFold3).

* Layer‑norm on the input (dim).
* Linear projections → left / right (no bias).
* Optional mask (B,N,N) is applied to the projected tensors in‑place by a
  tiny Triton kernel (the only required custom kernel).
* Gating (sigmoid) is done in PyTorch (cheap).
* Core N³ update:   out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]   (einsum)
  – executed with torch.einsum on bfloat16 for speed.
* Final Layer‑norm, gating and output linear.

All tensors are kept on the GPU; the final result is float32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: element‑wise mask × tensor  (B,N,N,H)
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    x_ptr,               # ptr to tensor (B,N,N,H)
    mask_ptr,            # ptr to mask   (B,N,N) – broadcast on H
    B, N, H,
    stride_x_b, stride_x_i, stride_x_k, stride_x_h,
    stride_m_b, stride_m_i, stride_m_k,
    BLOCK_I: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    batch = pid // tl.cdiv(N, BLOCK_I)          # batch index
    i_tile = pid % tl.cdiv(N, BLOCK_I)         # i‑tile within batch

    # ---------------------- tile coordinates -------------------------
    offs_i = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_k = tl.arange(0, BLOCK_K)
    offs_h = tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_k = offs_k < N
    mask_h = offs_h < H

    # ---------------------- load mask (B,N,N) ------------------------
    # shape after load: (i_tile, k_tile)
    m = tl.load(
        mask_ptr
        + batch * stride_m_b
        + offs_i[:, None] * stride_m_i
        + offs_k[None, :] * stride_m_k,
        mask=mask_i[:, None] & mask_k[None, :],
        other=1.0,                 # when mask is absent we feed a tensor of ones
    )
    # broadcast to hidden dim
    m = m[:, :, None]               # (i,k,1)

    # ---------------------- load data (B,N,N,H) ----------------------
    x = tl.load(
        x_ptr
        + batch * stride_x_b
        + offs_i[:, None, None] * stride_x_i
        + offs_k[None, :, None] * stride_x_k
        + offs_h[None, None, :] * stride_x_h,
        mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
        other=0.0,
    )

    # ---------------------- apply mask -------------------------------
    x = x * m

    # ---------------------- store back --------------------------------
    tl.store(
        x_ptr
        + batch * stride_x_b
        + offs_i[:, None, None] * stride_x_i
        + offs_k[None, :, None] * stride_x_k
        + offs_h[None, None, :] * stride_x_h,
        x,
        mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
    )


# ----------------------------------------------------------------------
# Helper: Layer‑norm for a 4‑D tensor (B,N,N,C) using supplied weight/bias.
# ----------------------------------------------------------------------
def layer_norm_4d(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: (B,N,N,C)
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point (called by the evaluator)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on the same device as input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------
    x = layer_norm_4d(
        input_tensor,
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )

    # --------------------------------------------------------------
    # 2. Linear projections → left / right  (dim → hidden_dim)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # --------------------------------------------------------------
    # 3. Apply optional mask (broadcast on hidden dim) via Triton
    # --------------------------------------------------------------
    if mask is None:
        # tensor of ones – kernel becomes a no‑op
        mask_f = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask_f = mask.to(dtype)

    # launch parameters – hidden_dim is guaranteed to be a multiple of 32
    BLOCK_I = 32
    BLOCK_K = 32
    BLOCK_H = 32

    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_m_b, stride_m_i, stride_m_k = mask_f.stride()

    grid = (B * triton.cdiv(N, BLOCK_I),)   # one program per (batch, i‑tile)

    # left
    apply_mask_kernel[grid](
        left,
        mask_f,
        B,
        N,
        hidden_dim,
        stride_x_b=stride_l_b,
        stride_x_i=stride_l_i,
        stride_x_k=stride_l_k,
        stride_x_h=stride_l_h,
        stride_m_b=stride_m_b,
        stride_m_i=stride_m_i,
        stride_m_k=stride_m_k,
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    # right
    apply_mask_kernel[grid](
        right,
        mask_f,
        B,
        N,
        hidden_dim,
        stride_x_b=stride_l_b,
        stride_x_i=stride_l_i,
        stride_x_k=stride_l_k,
        stride_x_h=stride_l_h,
        stride_m_b=stride_m_b,
        stride_m_i=stride_m_i,
        stride_m_k=stride_m_k,
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) – cheap, kept in PyTorch
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(dtype)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(dtype)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(dtype)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core N³ reduction (einsum) – use bfloat16 for speed
    # --------------------------------------------------------------
    left_bf16 = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)

    # out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    out = torch.einsum("... i k d, ... j k d -> ... i j d", left_bf16, right_bf16)
    out = out.to(dtype)        # back to FP32 for the following layers

    # --------------------------------------------------------------
    # 6. Final LayerNorm, gating and output linear
    # --------------------------------------------------------------
    out = layer_norm_4d(
        out,
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
