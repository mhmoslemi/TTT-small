"""
TriMul “outgoing” forward – Triton‑accelerated core reduction.

The expensive O(N³·C) step
    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
is equivalent to a batched matrix‑multiply for every channel c:
    A_{b,c} = left[b,:, :,c]      (N×N)
    B_{b,c} = right[b,:, :,c]     (N×N)
    out_{b,c} = A_{b,c} @ B_{b,c}ᵀ   (N×N)

We therefore reshape the tensors to a combined batch dimension
    (B·C, N, N) and run a single high‑throughput Triton GEMM.
All other ops (LayerNorm, Linear, gating, final LayerNorm & projection)
are performed with PyTorch – they are cheap compared with the N³ work and
benefit from the highly‑optimised ATen kernels.

The kernel is a classic tiled GEMM that:
* loads a BLOCK_M×BLOCK_K tile of A and a BLOCK_K×BLOCK_N tile of Bᵀ,
* accumulates with tl.dot into a fp32 accumulator,
* writes a BLOCK_M×BLOCK_N tile of C.
It works for any N up to 1024 (the test suite) and any hidden dimension C
that is a multiple of the chosen BLOCK_K (we enforce a small pad when needed).

The public entry‑point `custom_kernel` follows the signature required by the
grader.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton GEMM: C = A @ Bᵀ   (all tensors are fp32)
# A : (batch, M, K)      – here M = N, K = N
# B : (batch, N, K)      – we pass B already transposed so its layout is (batch, N, K)
# C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def matmul_tiled_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    batch, M, N, K,                     # logical sizes
    stride_ab, stride_ak, stride_akb,   # strides for A   (batch, M, K)
    stride_bb, stride_bk, stride_bkn,   # strides for Bᵀ  (batch, N, K)
    stride_cb, stride_cm, stride_cn,    # strides for C   (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # linear batch index + tile indices
    batch_idx = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    tile_idx  = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = tile_idx // tl.cdiv(N, BLOCK_N)   # tile row
    pid_n = tile_idx %  tl.cdiv(N, BLOCK_N)   # tile col

    # compute the start offsets of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out-of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this batch
    a_batch_ptr = a_ptr + batch_idx * stride_ab
    b_batch_ptr = b_ptr + batch_idx * stride_bb
    c_batch_ptr = c_ptr + batch_idx * stride_cb

    # accumulators
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load A tile  (BLOCK_M × BLOCK_K)
        a = tl.load(
            a_batch_ptr
            + (offs_m[:, None] * stride_ak)      # M stride
            + (offs_k[None, :] * stride_akb),    # K stride
            mask=mask_m[:, None],
            other=0.0,
        )
        # load Bᵀ tile (BLOCK_N × BLOCK_K) – note B is already transposed,
        # so we treat it as (N, K)
        b = tl.load(
            b_batch_ptr
            + (offs_n[:, None] * stride_bk)      # N stride (row)
            + (offs_k[None, :] * stride_bkn),    # K stride (col)
            mask=mask_n[:, None],
            other=0.0,
        )
        # inner‑product over K
        acc += tl.dot(a, b)   # (M,K)·(N,K)ᵀ → (M,N)

        # advance pointers
        a_batch_ptr += BLOCK_K * stride_akb
        b_batch_ptr += BLOCK_K * stride_bkn

    # store result
    c = acc.to(tl.float32)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper that launches the GEMM for a (B, N, N, C) tensor pair
# ----------------------------------------------------------------------
def batched_matmul_channels(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : (B, N, N, C)  float32
    returns      : (B, N, N, C)  float32  where
        out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
    """
    B, N, _, C = left.shape
    # reshape to (B*C, N, N)
    left_ = left.permute(0, 3, 1, 2).reshape(B * C, N, N)        # (B*C, M, K)
    right_ = right.permute(0, 3, 1, 2).reshape(B * C, N, N)      # (B*C, N, K)   (we will treat it as transposed)

    # make sure K is a multiple of BLOCK_K – pad if needed (very cheap)
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    K = N
    pad_k = (BLOCK_K - (K % BLOCK_K)) % BLOCK_K
    if pad_k:
        left_ = torch.nn.functional.pad(left_, (0, pad_k, 0, 0))   # pad last dim (K)
        right_ = torch.nn.functional.pad(right_, (0, pad_k, 0, 0))

    batch, M, K = left_.shape
    N = right_.shape[1]

    # strides (in elements) for A, Bᵀ, C
    stride_ab = left_.stride(0)
    stride_ak = left_.stride(1)
    stride_akb = left_.stride(2)

    stride_bb = right_.stride(0)
    stride_bk = right_.stride(1)
    stride_bkn = right_.stride(2)

    # output allocation
    out_ = torch.empty_like(left_)

    stride_cb = out_.stride(0)
    stride_cm = out_.stride(1)
    stride_cn = out_.stride(2)

    # launch grid: one program per (batch, tile_m, tile_n)
    grid = (
        batch * triton.cdiv(M, BLOCK_M) * triton.cdiv(N, BLOCK_N),
    )

    matmul_tiled_kernel[grid](
        left_.contiguous(),
        right_.contiguous(),
        out_,
        batch,
        M,
        N,
        K,
        stride_ab,
        stride_ak,
        stride_akb,
        stride_bb,
        stride_bk,
        stride_bkn,
        stride_cb,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to (B, N, N, C)
    out = out_.reshape(B, C, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input, mask, weights, config)
        - input : torch.Tensor[bs, seq_len, seq_len, dim]   (float32)
        - mask  : torch.Tensor[bs, seq_len, seq_len]       (float32, 0/1)
        - weights : dict of pretrained parameters (see paper)
        - config  : {'dim': int, 'hidden_dim': int, 'nomask': bool}

    Returns
    -------
    torch.Tensor[bs, seq_len, seq_len, dim]   (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm (manual to stay on‑device)
    # ------------------------------------------------------------------
    eps = 1e-5
    w = weights["norm.weight"]
    b = weights["norm.bias"]
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + eps)
    x = x * w + b

    # ------------------------------------------------------------------
    # Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask – broadcast over hidden dimension
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.float().unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·C) reduction – Triton GEMM over the hidden dimension
    # ------------------------------------------------------------------
    out = batched_matmul_channels(left, right)          # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # Final LayerNorm, gating and output projection
    # ------------------------------------------------------------------
    w_out = weights["to_out_norm.weight"]
    b_out = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * w_out + b_out

    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
