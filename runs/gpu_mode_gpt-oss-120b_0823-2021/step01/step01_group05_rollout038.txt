"""
TriMul “outgoing” kernel (AlphaFold‑3) – Triton‑accelerated forward pass.

*   LayerNorm, Linear projections and gating are kept in PyTorch (they are
    memory‑bound, numerically‑stable and already highly‑optimised).
*   The two expensive GPU kernels are
        1.   mask_apply – broadcast a pair‑wise mask [B,N,N] onto a 4‑D tensor
               [B,N,N,H] (optional, `nomask==False`);
        2.   tri_mul   – compute the O(N³·H) reduction
               out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
               using a tiled batched‑matrix‑multiply (the “einsum” in the
               reference implementation).  The kernel works on FP16 for the
               inner accumulation and promotes the result back to FP32.
*   The final LayerNorm, gating and output‑projection are again performed in
    PyTorch to avoid numerical‑stability issues.

The implementation works for the full range of test configurations described
in the problem statement (variable batch size, sequence length, feature‑dim,
optional mask and different input distributions).  All tensors are expected
to be on CUDA and the returned tensor is float‑32.
"""

import torch
import triton
import triton.language as tl
from torch import nn
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# 1️⃣  Mask‑apply kernel (broadcast mask over the hidden dimension)
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*  – input tensor  [B,N,N,H] flattened
    mask_ptr,         # const float*  – mask          [B,N,N]   flattened
    dst_ptr,          # float*        – output tensor [B,N,N,H] flattened
    total_el,         # i64           – total number of elements (B*N*N*H)
    H,                # i32           – hidden dimension (H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_el

    # load the tensor element
    x = tl.load(src_ptr + offs, mask=mask, other=0.0)

    # the corresponding mask element is at index offs // H
    m = tl.load(mask_ptr + (offs // H), mask=mask, other=1.0)

    tl.store(dst_ptr + offs, x * m, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    tensor : [B, N, N, H]   (float32)
    mask   : [B, N, N]      (float32, 0/1)
    returns tensor * mask[...,None]  (float32)
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024

    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.view(-1)                     # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)

    mask_apply_kernel[grid](
        src,
        mask_flat,
        dst,
        total,
        H,
        BLOCK=BLOCK,
    )
    return dst.view(B, N, N, H)


# ----------------------------------------------------------------------
# 2️⃣  Tri‑mul kernel – batched matrix multiplication for each head H
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # const half*  – [B,N,N,H] flattened (FP16)
    right_ptr,              # const half*  – [B,N,N,H] flattened (FP16)
    out_ptr,                # float*       – [B,N,N,H] flattened (FP32)
    B, N, H,
    # strides for the 4‑D tensors (in elements, not bytes)
    stride_bl_i, stride_bl_k, stride_bl_h,   # left  : (B,N,N,H)
    stride_br_j, stride_br_k, stride_br_h,   # right : (B,N,N,H)
    stride_bo_i, stride_bo_j, stride_bo_h,   # out   : (B,N,N,H)
    BLOCK_M: tl.constexpr,   # tile size in the i‑dimension
    BLOCK_N: tl.constexpr,   # tile size in the j‑dimension
    BLOCK_K: tl.constexpr,   # tile size in the reduction dimension k
    GROUP_SIZE_M: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    #  Compute a 2‑D grid of tiles (i, j).  The reduction dimension k
    #  is processed inside the kernel.
    # --------------------------------------------------------------
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    # map linear pid -> (tile_m, tile_n)
    tile_m = (pid // num_tiles_n) % num_tiles_m
    tile_n = pid % num_tiles_n

    # group tiles along M to improve L2 reuse (standard Triton trick)
    pid_z = pid // (num_tiles_m * num_tiles_n)
    tile_m = (pid_z * GROUP_SIZE_M + tile_m) % num_tiles_m

    # -----------------------------------------
    #  Offsets for the current tile
    # -----------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # iterate over batches
    for b in range(B):
        # pointers to the start of the current batch
        left_batch_ptr  = left_ptr  + b * stride_bl_i * N * N
        right_batch_ptr = right_ptr + b * stride_br_j * N * N
        out_batch_ptr   = out_ptr   + b * stride_bo_i * N * N

        # accumulator in FP32 (more stable than FP16 accumulation)
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # ------------------------------------------------------------------
            #  Load a block of left  : shape (BLOCK_M, BLOCK_K, H)
            #  and a block of right : shape (BLOCK_N, BLOCK_K, H) – note the
            #  reduction dimension is the second one for both tensors.
            # ------------------------------------------------------------------
            # left[b, i, k, h]
            left_ptrs = (left_batch_ptr
                         + offs_m[:, None] * stride_bl_i
                         + offs_k[None, :] * stride_bl_k
                         + tl.arange(0, H)[None, None, :] * stride_bl_h)
            left_block = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

            # right[b, j, k, h]   (note we need the *k* index as the second axis)
            right_ptrs = (right_batch_ptr
                         + offs_n[:, None] * stride_br_j
                         + offs_k[None, :] * stride_br_k
                         + tl.arange(0, H)[None, None, :] * stride_br_h)
            right_block = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

            # ------------------------------------------------------------------
            #  Compute the inner product over k and H.
            #  We first do a dot over k for each head, then sum over heads.
            # ------------------------------------------------------------------
            # left_block: (M, K, H)   right_block: (N, K, H)
            # transpose right to (K, N, H) for dot product
            right_block_t = tl.trans(right_block, (1, 0, 2))

            # element‑wise multiply => (M, K, H) * (K, N, H) -> (M, N, H)
            prod = left_block[:, :, None, :] * right_block_t[None, :, :, :]

            # sum over K and H  (K reduction first, then H)
            acc += tl.sum(prod, axis=(1, 3))

        # ------------------------------------------------------------------
        #  Store the result
        # ------------------------------------------------------------------
        out_ptrs = (out_batch_ptr
                    + offs_m[:, None] * stride_bo_i
                    + offs_n[None, :] * stride_bo_j
                    + tl.arange(0, H)[None, None, :] * stride_bo_h)

        tl.store(out_ptrs, acc[:, :, None].to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    left/right : [B,N,N,H]  (float16 or float32)
    returns    : [B,N,N,H]  (float32)
    """
    # work in FP16 inside the kernel for bandwidth, promote to FP32 for output
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)

    # strides in elements (not bytes)
    stride_bl_i, stride_bl_k, stride_bl_h = left.stride()
    stride_br_j, stride_br_k, stride_br_h = right.stride()
    stride_bo_i, stride_bo_j, stride_bo_h = out.stride()

    # choose tile sizes – these work well on Hopper (sm_90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    GROUP_SIZE_M = 8

    total_tiles = B * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N)
    grid = (total_tiles,)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_bl_i, stride_bl_k, stride_bl_h,
        stride_br_j, stride_br_k, stride_br_h,
        stride_bo_i, stride_bo_j, stride_bo_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
    )
    return out

# 3️⃣  Full forward – glue everything together
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input, mask, weights, config)
        - input : [B, N, N, dim]  float32
        - mask  : [B, N, N]       float32 (0/1) – may be all‑ones when ``nomask``.
        - weights : dict of pretrained parameters.
        - config : dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``.

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32).
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣  LayerNorm (manual – no need for torch.nn.LayerNorm)
    # --------------------------------------------------------------
    eps = 1e-5
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # --------------------------------------------------------------
    # 2️⃣  Linear projections (bias‑free)
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣  Optional mask – Triton implementation
    # --------------------------------------------------------------
    if not nomask:
        left = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # --------------------------------------------------------------
    # 4️⃣  Gating
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣  Core O(N³·H) reduction – Triton kernel
    # --------------------------------------------------------------
    # left/right are [B,N,N,hidden_dim] in FP32; the kernel expects FP16 inside.
    out = tri_mul_triton(left, right)                     # [B,N,N,hidden_dim] FP32

    # --------------------------------------------------------------
    # 6️⃣  Final LayerNorm, gating and output projection (PyTorch)
    # --------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
