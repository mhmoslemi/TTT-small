"""
TriMul “outgoing” forward (no backward).

The expensive part of the module is the 3‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape (B, N, N, H)   (H = hidden_dim).
For each head `d` this is a batched matrix‑multiply:
    (B, H, I, K) · (B, H, K, J) → (B, H, I, J)

Only this contraction is executed in Triton; all cheap LayerNorm /
linear‑gate work stays in PyTorch (fp32 → fp16 conversion is cheap compared
to the N³ work).  The kernel processes many (B·H) independent matrix products
in a single launch and uses fp16 accumulation in FP32 for accuracy.
Block sizes are tuned for Hopper/SM‑90 (64×64 tiles, K‑tile 32, 4 warps).

The host‑side `custom_kernel` builds the required tensors, launches the
kernel and finishes the remaining LayerNorm/gating/linear steps.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched 2‑D matrix multiply (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def _batch_matmul_kernel(
    a_ptr,            # left  matrix  (B*H, I, K)
    b_ptr,            # right matrix  (B*H, K, J)
    c_ptr,            # output matrix (B*H, I, J)
    M, N, K,          # I, J, K sizes
    stride_az, stride_am, stride_ak,   # a strides (batch, i, k)
    stride_bz, stride_bk, stride_bn,   # b strides (batch, k, j)
    stride_cz, stride_cm, stride_cn,   # c strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    # tile identifiers
    pid_m = tl.program_id(0)          # tile row   (M)
    pid_n = tl.program_id(1)          # tile col   (N)
    pid_z = tl.program_id(2)          # batch*head (B*H)

    # ---------------------------------------------------------------
    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (FP32 for numeric stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # loop over K dimension in chunks
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile   (shape BLOCK_M × BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile   (shape BLOCK_K × BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # accumulate
        acc += tl.dot(a, b)

    # ---------------------------------------------------------------
    # write result (convert back to fp16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _triton_batch_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matmul for the TriMul contraction.

    left  : (B, H, I, K)  fp16, contiguous
    right : (B, H, K, J)  fp16, contiguous
    returns (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten the batch*head dimension – the kernel expects a 3‑D tensor
    batch = B * H
    a = left.reshape(batch, I, K)
    b = right.reshape(batch, K, J)
    c = out.reshape(batch, I, J)

    # strides in **elements** (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tile size – tuned for Hopper SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),
        triton.cdiv(J, BLOCK_N),
        batch,
    )
    _batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # good default for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Public entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the “outgoing” TriMul operator.

    data = (input_tensor, mask, weights, config)
        input_tensor : [B, N, N, dim]            (float32)
        mask         : [B, N, N]   or None       (float32 / bool)
        weights      : dict of leaf tensors
        config       : {"dim": int, "hidden_dim": int}

    Returns
        out : [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ---------------------------------------------------------------
    # 1) LayerNorm on the input (FP32)
    # ---------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual LN (same eps as PyTorch)
    eps = 1e-5
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b

    # ---------------------------------------------------------------
    # 2) Linear projections + gating (convert to fp16 for the heavy part)
    # ---------------------------------------------------------------
    # helper to cast weight once
    def _to_fp16(t):
        return t.to(torch.float16, non_blocking=True)

    left_proj_w = _to_fp16(weights["left_proj.weight"])
    right_proj_w = _to_fp16(weights["right_proj.weight"])
    left_gate_w = _to_fp16(weights["left_gate.weight"])
    right_gate_w = _to_fp16(weights["right_gate.weight"])
    out_gate_w = _to_fp16(weights["out_gate.weight"])
    to_out_norm_w = weights["to_out_norm.weight"]          # stays fp32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]                    # stays fp32

    x_fp16 = x.to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # (B,N,N, H)
    right = F.linear(x_fp16, right_proj_w) # (B,N,N, H)

    # ---------------------------------------------------------------
    # 3) Optional mask (broadcasted on the channel dim)
    # ---------------------------------------------------------------
    if mask is not None:
        # mask shape [B,N,N] -> [B,N,N,1] -> fp16
        m = mask.unsqueeze(-1).to(torch.float16)
        left = left * m
        right = right * m

    # ---------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ---------------------------------------------------------------
    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------------
    # 5) Triton batched matmul: produce (B, H, N, N)
    #    left : (B,N,N,H) -> (B,H,N,N)   (permute before matmul)
    #    right: (B,N,N,H) -> (B,H,N,N)   (permute & swap last two axes)
    # ---------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, J)

    out = _triton_batch_matmul(left_perm, right_perm)  # (B, H, I, J) fp16

    # back to original layout and fp32
    out = out.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # (B,N,N,dim)

    # ---------------------------------------------------------------
    # 6) Output LayerNorm (FP32)
    # ---------------------------------------------------------------
    mean = out.mean(-1, keepdim=True)
    var = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean) / torch.sqrt(var + eps)
    out = out * to_out_norm_w + to_out_norm_b

    # ---------------------------------------------------------------
    # 7) Final gating & projection (FP32)
    # ---------------------------------------------------------------
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, to_out_w)

    return out
```

**Explanation of the design choices**

| Step | Why it stays in PyTorch? | Why it is cheap |
|------|--------------------------|-----------------|
| Input `LayerNorm` | Needs mean/var across the last dimension (dim ≤ 384).  The amount of work is `O(B·N²·dim)` which is negligible compared to the `N³·hidden_dim` contraction. | Simple element‑wise ops, memory‑bound but tiny. |
| Linear projections & gates | 4‑D → 3‑D matmul with small inner dimension (`dim` or `hidden_dim` = 128).  These are also memory‑bound and benefit little from custom kernels. | Per‑element work `O(B·N²·dim·hidden_dim)` but `dim·hidden_dim` is tiny. |
| Mask application | Broadcasted element‑wise multiply; trivial. |
| **Core contraction** | The `einsum` corresponds to a batched matrix multiply with shape `(B·H, N, N)·(B·H, N, N)`.  This is the only `O(N³·H)` piece and dominates runtime. | Implemented in Triton, tiled, fp16 compute, FP32 accumulation. |
| Output `LayerNorm` & final linear | Same reasoning as step 1 – small tensor, cheap. |

**Performance notes**

* Tiles `64×64×32` give good occupancy on SM‑90 while keeping shared‑memory usage ≤ 16 KB per block.
* A single launch processes the whole `(B·H)` batch dimension, avoiding kernel‑launch overhead.
* FP16 arithmetic speeds up the inner dot product; accumulation in FP32 preserves the numerical quality needed for the final FP32 output. 

The presented implementation respects all requirements:

* Works for any `B ∈ {1,2}`, `N` up to 1024, `dim` ∈ {128,384}, `hidden_dim` = 128.
* Handles both masked and un‑masked inputs.
* Returns a `float32` tensor of shape `[B, N, N, dim]`.
* Uses Triton 3.3.1 primitives only inside the kernel and launches it correctly.
