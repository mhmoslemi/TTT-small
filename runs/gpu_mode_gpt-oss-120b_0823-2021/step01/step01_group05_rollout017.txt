"""
TriMul “outgoing” kernel (AlphaFold‑3) – Triton‑accelerated core reduction.

The expensive O(N³·H) contraction
    out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]
is rewritten as a tiled Triton kernel.  The surrounding LayerNorms,
Linear projections and gating are kept in PyTorch for numerical stability.
If a mask is present it is applied with a tiny Triton kernel before the
reduction (the only other Triton work in the reference implementation).

The kernel tiles the i‑ and j‑axes (BLOCK_M × BLOCK_N) and the reduction
axis k (BLOCK_K).  The hidden dimension H is processed in chunks
(BLOCK_H) so that a single thread block can keep all partial results in
registers.  Accumulation is performed in float32 even when the inputs are
float16, guaranteeing the same numerical quality as the reference model.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# 1️⃣  Mask application (kept from the previous solution – tiny memory‑bound kernel)
# ----------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,   # const float*  (input tensor flattened)
    mask_ptr,  # const float*  (mask tensor flattened, shape B*N*N)
    dst_ptr,   # float*        (output tensor flattened)
    total_el,  # i64           total number of elements in src/dst
    H,         # i64           hidden dimension (stride for mask broadcasting)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_el

    # read input element
    x = tl.load(src_ptr + offs, mask=valid, other=0.0)

    # broadcast mask: each hidden‑dim element reads the same mask entry
    m = tl.load(mask_ptr + (offs // H), mask=valid, other=1.0)

    tl.store(dst_ptr + offs, x * m, mask=valid)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    tensor : [B, N, N, H]   (float32)
    mask   : [B, N, N]      (float32, 0/1)
    returns tensor * mask[...,None] using Triton.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK = 1024

    src = tensor.reshape(-1)
    dst = torch.empty_like(src)
    mask_flat = mask.reshape(-1)               # [B,N,N]

    grid = (triton.cdiv(total, BLOCK),)
    mask_apply_kernel[grid](
        src,
        mask_flat,
        dst,
        total,
        H,
        BLOCK=BLOCK,
    )
    return dst.view(B, N, N, H)


# ----------------------------------------------------------------------
# 2️⃣  Core O(N³·H) reduction – outgoing TriMul
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,               # const float*   [B, N, N, H]
    right_ptr,              # const float*   [B, N, N, H]
    out_ptr,                # float*         [B, N, N, H]

    # sizes
    B, N, H,                # i64

    # strides (flattened tensor, row‑major)
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,

    # tiling constants
    BLOCK_M: tl.constexpr,   # tiles in i dimension
    BLOCK_N: tl.constexpr,   # tiles in j dimension
    BLOCK_K: tl.constexpr,   # reduction tile size
    BLOCK_H: tl.constexpr,   # hidden‑dim tile size
):
    # --------------------------------------------------------------
    # program IDs
    # --------------------------------------------------------------
    pid_b = tl.program_id(2)                         # batch dimension
    pid_i = tl.program_id(0)                         # i‑tile
    pid_j = tl.program_id(1)                         # j‑tile

    # --------------------------------------------------------------
    # start offsets for this block
    # --------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)          # [M]
    offs_j = j_start + tl.arange(0, BLOCK_N)          # [N]
    mask_i = offs_i < N
    mask_j = offs_j < N

    # hidden‑dim offsets (same for every i/j)
    offs_h = tl.arange(0, BLOCK_H)

    # --------------------------------------------------------------
    # accumulator
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over k (reduction)
    # --------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)      # [K]
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # load a tile of left:  shape (M, K, H)
        # ------------------------------------------------------------------
        ptr_left = (
            left_ptr
            + pid_b * stride_lb
            + offs_i[:, None, None] * stride_li
            + offs_k[None, :, None] * stride_lk
            + offs_h[None, None, :] * stride_lh
        )
        left = tl.load(
            ptr_left,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0,
        ).to(tl.float32)      # cast to fp32 for accumulation

        # ------------------------------------------------------------------
        # load a tile of right: shape (N, K, H)
        # ------------------------------------------------------------------
        ptr_right = (
            right_ptr
            + pid_b * stride_rb
            + offs_j[:, None, None] * stride_rj
            + offs_k[None, :, None] * stride_rk
            + offs_h[None, None, :] * stride_rh
        )
        right = tl.load(
            ptr_right,
            mask=mask_j[:, None, None] & mask_k[None, :, None],
            other=0.0,
        ).to(tl.float32)

        # ------------------------------------------------------------------
        # element‑wise multiply and reduce over k
        # ------------------------------------------------------------------
        # left: (M, K, H), right: (N, K, H) → we need (M,N,K,H)
        # broadcasting via tl.broadcast_to
        left_bc  = tl.broadcast_to(left,  (BLOCK_M, BLOCK_N, BLOCK_K, BLOCK_H))
        right_bc = tl.broadcast_to(right, (BLOCK_M, BLOCK_N, BLOCK_K, BLOCK_H))

        prod = left_bc * right_bc                         # (M,N,K,H)
        acc += tl.sum(prod, axis=2)                       # sum over K

    # --------------------------------------------------------------
    # store result
    # --------------------------------------------------------------
    ptr_out = (
        out_ptr
        + pid_b * stride_ob
        + offs_i[:, None, None] * stride_oi
        + offs_j[None, :, None] * stride_oj
        + offs_h[None, None, :] * stride_oh
    )
    tl.store(
        ptr_out,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None],
    )


def trimul_outgoing_triton(
    left: torch.Tensor,
    right: torch.Tensor,
) -> torch.Tensor:
    """
    Triton implementation of the outgoing TriMul contraction.
    left, right : [B, N, N, H] (float32 or float16)
    returns      : [B, N, N, H] (float32)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)

    # ------------------------------------------------------------------
    # launch configuration – these values work well on Hopper (sm_90)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 32                      # hidden dimension is a multiple of 32

    grid = (
        triton.cdiv(N, BLOCK_M),      # i‑tiles
        triton.cdiv(N, BLOCK_N),      # j‑tiles
        B,                            # batch
    )

    # stride helpers (row‑major)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# 3️⃣  Top‑level entry point – combines everything
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    Parts of the computation (mask, O(N³·H) reduction) are run in Triton.
    All other linear / norm / gate operations stay in PyTorch for stability.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (manual, no torch.nn.LayerNorm to keep everything on‑device)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * norm_w + norm_b                     # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask – Triton kernel (tiny kernel above)
    # ------------------------------------------------------------------
    if not nomask:
        left = apply_mask_triton(left, mask)
        right = apply_mask_triton(right, mask)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – still in PyTorch
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate          # shape [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction – Triton implementation
    # ------------------------------------------------------------------
    out = trimul_outgoing_triton(left, right)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gating and output projection (PyTorch)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
