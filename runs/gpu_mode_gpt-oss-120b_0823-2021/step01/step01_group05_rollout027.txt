"""
TriMul “outgoing” kernel (AlphaFold‑3) – Triton‑accelerated core reduction.

The expensive O(N³·H) reduction  
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]                 (1)

is a batched matrix multiplication for every hidden dimension *h*:
    out_h = left_h @ right_hᵀ .                                        (2)

The surrounding LayerNorm, Linear projections and gating are kept in
PyTorch for numerical stability.  The kernel below computes (2) for an
arbitrary batch size B, sequence length N and hidden dimension H
(`hidden_dim`).  It works with fp32 tensors and is launched with a
2‑D grid (blocks over the (i,j) tiles and over the batch dimension).

The implementation follows the classic tiled GEMM pattern:
  * each program processes a BLOCK_M × BLOCK_N tile of the output,
  * loops over K in BLOCK_K‑sized chunks,
  * loads the full H‑vector for the tile (shape BLOCK_M×BLOCK_K×BLOCK_H)
    and accumulates across K with `tl.dot` for every hidden slice.

The kernel is written for Triton 3.3.1 and targets NVIDIA Hopper (sm_90);
it has been tuned for the configurations used in the test harness
(SeqLen up to 1024, hidden_dim ≤ 128).  The final output is cast back to
float32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM over the hidden dimension (core TriMul step)
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const float*  [B,N,N,H]
    right_ptr,         # const float*  [B,N,N,H]
    out_ptr,           # float*        [B,N,N,H]
    B, N, H,           # i32 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b,   stride_out_i,   stride_out_j,   stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Compute out = left @ rightᵀ for every hidden slice."""
    pid = tl.program_id(0)          # tile over (i,j)
    pid_b = tl.program_id(1)        # batch dimension

    # ----- tile coordinates ------------------------------------------------
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ----- accumulator ------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ----- main K‑loop ------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,i,k,h]   shape (M, K, H)
        left = tl.load(
            left_ptr
            + pid_b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_left_h,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right[b,j,k,h]  shape (N, K, H)
        right = tl.load(
            right_ptr
            + pid_b * stride_right_b
            + offs_j[:, None, None] * stride_right_j
            + offs_k[None, :, None] * stride_right_k
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_right_h,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )

        # accumulate over K for every hidden slice
        for dh in range(0, BLOCK_H):
            acc[:, :, dh] += tl.dot(
                left[:, :, dh],            # (M, K)
                right[:, :, dh].T          # (K, N) -> (N, K) transposed
            )

    # ----- write result -----------------------------------------------------
    out = acc.to(tl.float32)
    out_ptrs = out_ptr \
        + pid_b * stride_out_b \
        + offs_i[:, None, None] * stride_out_i \
        + offs_j[None, :, None] * stride_out_j \
        + tl.arange(0, BLOCK_H)[None, None, :] * stride_out_h

    mask = mask_i[:, None, None] & mask_j[None, :, None]
    tl.store(out_ptrs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: launch the Triton kernel for the core reduction
# ----------------------------------------------------------------------
def trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H] (float32)
    returns      : [B, N, N, H] (float32) = left @ rightᵀ per hidden slice
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # block sizes – chosen to fit Hopper SM shared memory comfortably
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 32                     # H ≤ 128 → 4 passes when H=128

    # grid: (tiles over (i,j), batch)
    grid = (triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N), B)

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (PyTorch + Triton core)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : (input, mask, weights, config)
        input :  [B, N, N, dim]   float32
        mask  :  [B, N, N]        float32 (0/1) – may be all‑ones when ``nomask``.
        weights : dict of pretrained parameters.
        config  : dict with keys ``dim`` and ``hidden_dim`` and optional
                  ``nomask`` (bool).

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # LayerNorm (manual, no torch.nn.LayerNorm to stay in pure torch)
    # --------------------------------------------------------------
    eps = 1e-5
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # --------------------------------------------------------------
    # Linear projections (bias‑free)
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # Optional mask – cheap Triton kernel (vectorised across hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask[...,None] broadcasted on hidden dimension; we reuse the same
        # small kernel for both tensors.
        B, N, _, Hproj = left.shape
        total = B * N * N * Hproj
        BLOCK = 1024

        @triton.jit
        def mask_kernel(src_ptr, mask_ptr, dst_ptr, total_el, Hproj, BLOCK: tl.constexpr):
            pid = tl.program_id(0)
            offs = pid * BLOCK + tl.arange(0, BLOCK)
            m = offs < total_el
            src = tl.load(src_ptr + offs, mask=m, other=0.0)
            # each hidden entry shares the same mask value
            mk = tl.load(mask_ptr + (offs // Hproj), mask=m, other=1.0)
            tl.store(dst_ptr + offs, src * mk, mask=m)

        src = left.view(-1)
        dst = torch.empty_like(src)
        mask_f = mask.float().view(-1)          # [B,N,N]
        grid = (triton.cdiv(total, BLOCK),)
        mask_kernel[grid](src, mask_f, dst, total, Hproj, BLOCK=BLOCK)
        left = dst.view_as(left)

        src = right.view(-1)
        dst = torch.empty_like(src)
        mask_kernel[grid](src, mask_f, dst, total, Hproj, BLOCK=BLOCK)
        right = dst.view_as(right)

    # --------------------------------------------------------------
    # Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Core O(N³·H) reduction – Triton GEMM over hidden dimension
    # --------------------------------------------------------------
    # left/right : [B,N,N,hidden_dim]
    # after kernel : out_mid : [B,N,N,hidden_dim]
    out_mid = trimul_outgoing_triton(left, right)

    # --------------------------------------------------------------
    # Final LayerNorm, gating and output projection
    # --------------------------------------------------------------
    mean = out_mid.mean(-1, keepdim=True)
    var = out_mid.var(-1, unbiased=False, keepdim=True)
    out = (out_mid - mean) / torch.sqrt(var + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
