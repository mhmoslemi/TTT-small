"""
TriMul “outgoing” kernel (forward only).

The heavy N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is expressed as a batched GEMM per hidden dimension d.
All cheap element‑wise work (LayerNorm, linear projections, gating) stays in
PyTorch; the only operation moved to Triton is the optional mask‑application
(which is the only non‑fused element‑wise step required by the spec).

The Triton kernel multiplies a 4‑D tensor [B,N,N,H] by a broadcasted mask
[B,N,N] (expanded to [...,1]) using a 1‑D grid.  Strides are passed from the
host so the kernel works for any contiguous or strided layout.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ------------------------------------------------------------
# Triton: mask * tensor  (supports any stride layout)
# ------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    src_ptr,               # src tensor  [B,N,N,H]
    mask_ptr,              # mask tensor [B,N,N,1] (broadcasted)
    dst_ptr,               # destination tensor
    total_elem,            # total number of elements in src (B*N*N*H)
    # strides of src (4‑D)
    SRC_STRIDE_B, SRC_STRIDE_N1, SRC_STRIDE_N2, SRC_STRIDE_H,
    # strides of mask (3‑D, last dim is 1)
    MASK_STRIDE_B, MASK_STRIDE_N1, MASK_STRIDE_N2,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)            # linear offsets
    mask = offs < total_elem

    # ------------------ decode linear offset to 4‑D indices ------------------
    # src layout: (b, i, j, d)
    tmp = offs
    d = tmp % SRC_STRIDE_H
    tmp = tmp // SRC_STRIDE_H
    j = tmp % SRC_STRIDE_N2
    tmp = tmp // SRC_STRIDE_N2
    i = tmp % SRC_STRIDE_N1
    b = tmp // SRC_STRIDE_N1

    # compute mask offset (mask has no hidden dim)
    mask_off = (
        b * MASK_STRIDE_B +
        i * MASK_STRIDE_N1 +
        j * MASK_STRIDE_N2
    )

    src = tl.load(src_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)   # broadcasted
    tl.store(dst_ptr + offs, src * m, mask=mask)


# ----------------------------------------------------------------------
# Helper: fused LayerNorm (bias + weight supplied)
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] (float32)
            - mask_tensor : [B, N, N] (bool/float) – may be all‑ones when nomask=True
            - weights_dict: contains all linear / norm parameters
            - config_dict : {"dim":..., "hidden_dim":..., "nomask":bool}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = _layer_norm(inp, weights["norm.weight"], weights["norm.bias"])

    # --------------------------------------------------------------
    # 2. Linear projections (dim -> hidden) – bias‑free
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcasted along hidden dim) – Triton kernel
    # --------------------------------------------------------------
    if not nomask:
        # expand mask to shape [...,1] for broadcasting
        mask_exp = mask.unsqueeze(-1).contiguous()
        total = left.numel()
        BLOCK = 131072                     # ~512 KiB (FP32) per thread block
        grid = (triton.cdiv(total, BLOCK),)

        left_masked = torch.empty_like(left)
        right_masked = torch.empty_like(right)

        # launch for left
        _apply_mask_kernel[grid](
            left.data_ptr(),
            mask_exp.data_ptr(),
            left_masked.data_ptr(),
            total,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            mask_exp.stride(0), mask_exp.stride(1), mask_exp.stride(2),
            BLOCK=BLOCK,
        )
        # launch for right (reuse same mask)
        _apply_mask_kernel[grid](
            right.data_ptr(),
            mask_exp.data_ptr(),
            right_masked.data_ptr(),
            total,
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask_exp.stride(0), mask_exp.stride(1), mask_exp.stride(2),
            BLOCK=BLOCK,
        )
        left, right = left_masked, right_masked

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) – bias‑free linear + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core N³ reduction via batched GEMM per hidden dim
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # reshape to (B * hidden, N, N) and use torch.bmm which maps to a
    # highly‑optimized GEMM on Hopper.
    left_rs = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)          # (B*H, N, N)
    right_rs = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)       # (B*H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T → (B*H, N, N)
    out_rs = torch.bmm(left_rs, right_rs.transpose(1, 2))

    # reshape back to [B, N, N, hidden]
    out = out_rs.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear (hidden -> dim)
    # --------------------------------------------------------------
    out = _layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
