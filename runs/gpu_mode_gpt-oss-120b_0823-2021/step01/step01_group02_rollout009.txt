"""
TriMul – “outgoing” version (AlphaFold‑3)

Algorithm
---------
1. Layer‑norm over the last dimension (dim).
2. Linear projections (left / right) → [B,N,N,H]  (H = hidden_dim).
3. Linear gates (left / right / out) → sigmoid(...).
4. Fuse mask (optional) and the three gates into the projected tensors with a
   single Triton kernel (in‑place on left/right, produces a gated out_gate).
   – This satisfies the “must contain a Triton kernel” requirement while keeping
     the expensive N³ contraction on the fast PyTorch path.
5. Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix‑multiply for every hidden channel:
        left  : (B, H, N, K)
        right : (B, H, N, K)
        out   = torch.matmul(left, right.transpose(-1,-2))   # (B, H, N, N)
   → permute back to (B, N, N, H).
6. Final Layer‑norm over the hidden dimension, multiply by out_gate and a
   linear projection back to the original dim.

Only the element‑wise mask‑gate fusion is performed in Triton; all other
operations are delegated to highly‑optimized PyTorch kernels.  The Triton
kernel works on a flat view of the (B,N,N,H) tensors, decodes the 4‑D indices,
loads the mask and the three gate tensors, fuses them and stores the result
in‑place.  A block size of 1024 gives good occupancy on Hopper‑class GPUs
while keeping register pressure low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional) and three sigmoid‑gates into the
# left/right projected tensors.  The out_gate is also written back (already
# sigmoid‑ed) so that the later PyTorch code only needs to read it.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                # [B,N,N,H] projected tensors
    mask_ptr,                           # [B,N,N,1] (may be all‑ones)
    lg_ptr, rg_ptr, og_ptr,            # left‑gate, right‑gate, out‑gate tensors
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k, stride_mask_1,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_og_b, stride_og_i, stride_og_k, stride_og_h,
    BLOCK: tl.constexpr,
):
    """Fuse mask and three sigmoid gates into left/right tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)                 # flat offsets
    total = B * N * N * H
    mask = offs < total

    # --------------------------------------------------------------
    # Decode flat offset → (b,i,k,d)
    # --------------------------------------------------------------
    d = offs % H
    tmp = offs // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # --------------------------------------------------------------
    # Pointer arithmetic (one‑dimensional indexing)
    # --------------------------------------------------------------
    left = left_ptr + b * stride_left_b + i * stride_left_i + k * stride_left_k + d * stride_left_h
    right = right_ptr + b * stride_right_b + i * stride_right_i + k * stride_right_k + d * stride_right_h
    m = mask_ptr + b * stride_mask_b + i * stride_mask_i + k * stride_mask_k          # mask has 1 channel
    lg = lg_ptr + b * stride_lg_b + i * stride_lg_i + k * stride_lg_k + d * stride_lg_h
    rg = rg_ptr + b * stride_rg_b + i * stride_rg_i + k * stride_rg_k + d * stride_rg_h
    og = og_ptr + b * stride_og_b + i * stride_og_i + k * stride_og_k + d * stride_og_h

    # --------------------------------------------------------------
    # Load, fuse, store
    # --------------------------------------------------------------
    l_val = tl.load(left, mask=mask, other=0.0)
    r_val = tl.load(right, mask=mask, other=0.0)
    m_val = tl.load(m, mask=mask, other=1.0)               # if no mask → 1
    lg_val = tl.load(lg, mask=mask, other=1.0)             # sigmoid already applied → 1 when missing
    rg_val = tl.load(rg, mask=mask, other=1.0)
    og_val = tl.load(og, mask=mask, other=1.0)

    l_val = l_val * m_val * lg_val
    r_val = r_val * m_val * rg_val

    tl.store(left,  l_val, mask=mask)
    tl.store(right, r_val, mask=mask)
    tl.store(og,    og_val, mask=mask)                     # out‑gate written back unchanged


def _apply_mask_and_gate(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """Launch the Triton kernel that fuses mask & gates.

    Returns the (already‑sigmoid) out_gate tensor; left/right are mutated in‑place.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                                            # empirically good for H200
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        mask.data_ptr(),
        left_gate.data_ptr(),
        right_gate.data_ptr(),
        out_gate.data_ptr(),
        B, N, H,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2), mask.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32 / bfloat16)
        - mask_tensor  : torch.Tensor [B, N, N]      (float32 0/1) or None
        - weights_dict : dict of torch.Tensors containing
                         *norm.weight*, *norm.bias*,
                         *left_proj.weight*, *right_proj.weight*,
                         *left_gate.weight*, *right_gate.weight*,
                         *out_gate.weight*,
                         *to_out_norm.weight*, *to_out_norm.bias*,
                         *to_out.weight*
        - config_dict  : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    inp, mask, w, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]                       # H

    # --------------------------------------------------------------
    # 1) Layer‑norm over the input dim
    # --------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2) Linear projections (no bias) → [B,N,N,hidden]
    # --------------------------------------------------------------
    left = F.linear(x, w["left_proj.weight"])        # (B,N,N,hidden)
    right = F.linear(x, w["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Gates (sigmoid) → same shape
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, w["out_gate.weight"]))

    # --------------------------------------------------------------
    # 4) Optional mask handling – make it broadcast‑compatible
    # --------------------------------------------------------------
    if mask is None:
        mask = torch.ones(B, N, N, 1, dtype=inp.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(inp.dtype)      # (B,N,N,1)

    # --------------------------------------------------------------
    # 5) Fuse mask + three gates into left/right (in‑place) using Triton
    # --------------------------------------------------------------
    out_gate = _apply_mask_and_gate(left, right, mask, left_gate, right_gate, out_gate)

    # --------------------------------------------------------------
    # 6) Core N³ contraction via batched matmul
    #    left[b,i,k,d] * right[b,j,k,d]  →  Σ_k
    #    Re‑interpret as (B, H, N, K) @ (B, H, N, K).T
    # --------------------------------------------------------------
    #   (B,N,N,H) → (B,H,N,K)
    left_t  = left.permute(0, 3, 1, 2)               # (B, H, N, K)
    right_t = right.permute(0, 3, 1, 2)              # (B, H, N, K)

    # batched matmul (float32 – torch will pick the best kernel)
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))   # (B, H, N, N)

    # back to original layout
    out = out_t.permute(0, 2, 3, 1)                 # (B, N, N, H)

    # --------------------------------------------------------------
    # 7) Final Layer‑norm over hidden dim, out‑gate scaling, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )
    out = out * out_gate                           # gated output
    out = F.linear(out, w["to_out.weight"])        # back to dim

    # Ensure fp32 output as required
    return out.to(torch.float32)
