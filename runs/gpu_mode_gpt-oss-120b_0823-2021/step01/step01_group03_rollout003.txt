"""
TriMul “outgoing” forward pass (AlphaFold3).

The expensive N³ reduction (einsum) stays in torch‑einsum (Hopper’s
high‑performance kernel).  The memory‑bound part – applying the pairwise
mask *and* the per‑position left/right gates – is fused into a single
triton kernel that processes the tensors (B,N,N,H) in‑place.

Only the fused mask‑gate kernel is written in Triton; everything else is
standard torch (layer‑norm, linear layers, final layer‑norm/gate/out‑proj).
The final result is returned in float‑32 as required.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: apply the (B,N,N) binary mask and the sigmoid gates
# (both broadcasted on the hidden dimension H) to a tensor X of shape
# (B,N,N,H) IN‑PLACE.
#   X <- X * mask * gate
# The mask and gate are given as FP32 tensors; the kernel works in FP32.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    x_ptr,               # pointer to X (B,N,N,H)
    mask_ptr,            # pointer to mask (B,N,N) – broadcast on H
    gate_ptr,            # pointer to gate (B,N,N,H) – already sigmoided
    B, N, H,
    stride_x_b, stride_x_i, stride_x_k, stride_x_h,
    stride_m_b, stride_m_i, stride_m_k,
    stride_g_b, stride_g_i, stride_g_k, stride_g_h,
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)

    # one program per (batch, i‑tile)
    batch = pid // tl.cdiv(N, BLOCK_I)
    i_tile = pid % tl.cdiv(N, BLOCK_I)

    # tile offsets
    offs_i = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_k = tl.arange(0, BLOCK_K)
    offs_h = tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_k = offs_k < N
    mask_h = offs_h < H

    # ------------------------------------------------------------------
    # Load mask (B,N,N) → (i,k) and broadcast on H
    # ------------------------------------------------------------------
    mask = tl.load(
        mask_ptr
        + batch * stride_m_b
        + offs_i[:, None] * stride_m_i
        + offs_k[None, :] * stride_m_k,
        mask=mask_i[:, None] & mask_k[None, :],
        other=1.0,                     # when mask is omitted we pass a tensor of ones
    )
    mask = mask[:, :, None]           # (i,k,1)

    # ------------------------------------------------------------------
    # Load gate (already sigmoided) – shape (B,N,N,H)
    # ------------------------------------------------------------------
    gate = tl.load(
        gate_ptr
        + batch * stride_g_b
        + offs_i[:, None, None] * stride_g_i
        + offs_k[None, :, None] * stride_g_k
        + offs_h[None, None, :] * stride_g_h,
        mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
        other=0.0,
    )

    # ------------------------------------------------------------------
    # Load X, apply (mask * gate) and store back
    # ------------------------------------------------------------------
    x = tl.load(
        x_ptr
        + batch * stride_x_b
        + offs_i[:, None, None] * stride_x_i
        + offs_k[None, :, None] * stride_x_k
        + offs_h[None, None, :] * stride_x_h,
        mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
        other=0.0,
    )
    x = x * mask * gate
    tl.store(
        x_ptr
        + batch * stride_x_b
        + offs_i[:, None, None] * stride_x_i
        + offs_k[None, :, None] * stride_x_k
        + offs_h[None, None, :] * stride_x_h,
        x,
        mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
    )


# ----------------------------------------------------------------------
# Helper: 4‑D LayerNorm (B,N,N,C)
# ----------------------------------------------------------------------
def layer_norm_4d(x: torch.Tensor, w: torch.Tensor, b: torch.Tensor, eps: float = 1e-5):
    # mean & var over the last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x̂ = (x - mean) * torch.rsqrt(var + eps)
    return x̂ * w + b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack ------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dtype = torch.float32

    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]          # H in the paper

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------
    x = layer_norm_4d(
        inp,
        weights["norm.weight"].to(dtype),
        weights["norm.bias"].to(dtype),
    )

    # ------------------------------------------------------------------
    # 2. Linear projections → (B,N,N,hidden)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # no bias
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # ------------------------------------------------------------------
    # 3. Compute sigmoid gates (still in torch – cheap)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(dtype)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(dtype)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(dtype)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask * gate into a single Triton kernel
    #    If mask is None → create a tensor of ones (no‑op mask)
    # ------------------------------------------------------------------
    if mask is None:
        mask_fp = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask_fp = mask.to(dtype)

    # launch configuration (H is always a multiple of 32 in the test set)
    BLOCK_I = 32
    BLOCK_K = 32
    BLOCK_H = 32

    # strides for the tensors we will pass to the kernel
    sx_b, sx_i, sx_k, sx_h = left.stride()
    sm_b, sm_i, sm_k = mask_fp.stride()
    sg_b, sg_i, sg_k, sg_h = left_gate.stride()   # same layout as left/right

    # one program per (batch, i‑tile)
    grid = (B * triton.cdiv(N, BLOCK_I),)

    # left
    mask_gate_kernel[grid](
        left, mask_fp, left_gate,
        B, N, hidden,
        stride_x_b=sx_b, stride_x_i=sx_i, stride_x_k=sx_k, stride_x_h=sx_h,
        stride_m_b=sm_b, stride_m_i=sm_i, stride_m_k=sm_k,
        stride_g_b=sg_b, stride_g_i=sg_i, stride_g_k=sg_k, stride_g_h=sg_h,
        BLOCK_I=BLOCK_I, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
    )
    # right
    mask_gate_kernel[grid](
        right, mask_fp, right_gate,
        B, N, hidden,
        stride_x_b=sx_b, stride_x_i=sx_i, stride_x_k=sx_k, stride_x_h=sx_h,
        stride_m_b=sm_b, stride_m_i=sm_i, stride_m_k=sm_k,
        stride_g_b=sg_b, stride_g_i=sg_i, stride_g_k=sg_k, stride_g_h=sg_h,
        BLOCK_I=BLOCK_I, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 5. Core N³ multiplication via torch.einsum.
    #    Cast to bfloat16 for speed on Hopper/H200, then back.
    # ------------------------------------------------------------------
    left_bf16 = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)

    out = torch.einsum("... i k d, ... j k d -> ... i j d", left_bf16, right_bf16)
    out = out.to(dtype)                     # back to FP32

    # ------------------------------------------------------------------
    # 6. Final LayerNorm, gating and output projection
    # ------------------------------------------------------------------
    out = layer_norm_4d(
        out,
        weights["to_out_norm.weight"].to(dtype),
        weights["to_out_norm.bias"].to(dtype),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype))

    return out
