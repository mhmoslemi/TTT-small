"""
TriMul “outgoing” forward pass (AlphaFold‑3).

*   LayerNorm, linear projections and gating are cheap – they stay in
    PyTorch (FP32) for numerical stability.
*   The only compute‑intensive part is the N³ einsum

        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

    which factorises per hidden dimension `d` into a batched matrix‑multiply

        out_h = left_h @ right_hᵀ          (B, N, N) × (B, N, N) → (B, N, N)

    We therefore reshape the tensors so that every (batch, hidden) pair
    becomes a single batch dimension and let cuBLAS (`torch.bmm`) do the
    heavy lifting.

*   The element‑wise mask, left‑gate and right‑gate multiplications are
    fused in a tiny Triton kernel (`tri_mul_fuse_kernel`).  The kernel works
    **in‑place** on `left` and `right` to avoid extra temporary buffers.
    It processes the 4‑D tensors as a 1‑D linear index, decodes the indices,
    loads the values, applies `mask * gate` and stores the result back.

*   Finally we apply the output LayerNorm, output gate and the last linear
    projection in PyTorch.

The implementation supports both masked and un‑masked inputs, any batch size
(`B∈{1,2}`), sequence length up to 1024 and hidden dimensions 128/384.  All
tensors remain on the CUDA device and the returned tensor is `float32`.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel – fuses mask, left‑gate and right‑gate element‑wise ops
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_fuse_kernel(
    left_ptr,          # [B, N, N, H]   (float32)
    right_ptr,         # [B, N, N, H]   (float32)
    mask_ptr,          # [B, N, N, 1]   (float32) – broadcasted mask
    lgate_ptr,         # [B, N, N, H]   (float32)
    rgate_ptr,         # [B, N, N, H]   (float32)
    B, N, H,           # static sizes (int)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    """
    For each element ``(b,i,k,h)`` compute

        left  = left  * mask[b,i,k,0] * lgate[b,i,k,h]
        right = right * mask[b,i,k,0] * rgate[b,i,k,h]

    The kernel runs on a 1‑D grid, each thread processes ``BLOCK`` contiguous
    linear indices.
    """
    pid = tl.program_id(0)
    total = B * N * N * H                     # total number of elements
    offs = pid * BLOCK + tl.arange(0, BLOCK)  # linear offsets processed by this thread
    mask = offs < total

    # ---- decode linear offset into (b,i,k,h) -------------------------
    # linear = ((b * N + i) * N + k) * H + h
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem = rem % (N * H)
    k = rem // H
    h = rem % H

    # ---- compute raw pointers -----------------------------------------
    left  = left_ptr  + b * stride_left_b  + i * stride_left_i  + k * stride_left_k  + h * stride_left_h
    right = right_ptr + b * stride_right_b + i * stride_right_i + k * stride_right_k + h * stride_right_h
    mask_ = mask_ptr  + b * stride_mask_b  + i * stride_mask_i  + k * stride_mask_k
    lgate = lgate_ptr + b * stride_lgate_b + i * stride_lgate_i + k * stride_lgate_k + h * stride_lgate_h
    rgate = rgate_ptr + b * stride_rgate_b + i * stride_rgate_i + k * stride_rgate_k + h * stride_rgate_h

    # ---- load, fuse and store -----------------------------------------
    l = tl.load(left,  mask=mask, other=0.0)
    r = tl.load(right, mask=mask, other=0.0)
    m = tl.load(mask_, mask=mask, other=0.0)
    lg = tl.load(lgate, mask=mask, other=0.0)
    rg = tl.load(rgate, mask=mask, other=0.0)

    l = l * m * lg
    r = r * m * rg

    tl.store(left,  l, mask=mask)
    tl.store(right, r, mask=mask)


def _fuse_mask_and_gates(left, right, mask, lgate, rgate):
    """
    Launch ``tri_mul_fuse_kernel``.  The kernel works in‑place on `left`
    and `right` and returns them for convenience.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                         # 1 k threads → good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    # strides are expressed in number of elements (not bytes)
    sl = left.stride()
    sr = right.stride()
    sm = mask.stride()
    sg_l = lgate.stride()
    sg_r = rgate.stride()

    tri_mul_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=lgate,
        rgate_ptr=rgate,
        B=B, N=N, H=H,
        stride_left_b=sl[0], stride_left_i=sl[1], stride_left_k=sl[2], stride_left_h=sl[3],
        stride_right_b=sr[0], stride_right_i=sr[1], stride_right_k=sr[2], stride_right_h=sr[3],
        stride_mask_b=sm[0], stride_mask_i=sm[1], stride_mask_k=sm[2],
        stride_lgate_b=sg_l[0], stride_lgate_i=sg_l[1], stride_lgate_k=sg_l[2], stride_lgate_h=sg_l[3],
        stride_rgate_b=sg_r[0], stride_rgate_i=sg_r[1], stride_rgate_k=sg_r[2], stride_rgate_h=sg_r[3],
        BLOCK=BLOCK,
    )
    return left, right


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor  [B, N, N, dim] (float32, CUDA)
        - mask_tensor  : torch.Tensor  [B, N, N]    (float32, CUDA)  or None
        - weights_dict : dict of torch.Tensor containing all linear / LN params
        - config_dict  : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------------
    # 1) Input LayerNorm   (B,N,N,dim) → (B,N,N,dim)
    # -------------------------------------------------------
    ln_w = w["norm.weight"]
    ln_b = w["norm.bias"]
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * ln_w + ln_b                         # shape (B,N,N,dim)

    # -------------------------------------------------------
    # 2) Linear projections (no bias) → hidden dim
    # -------------------------------------------------------
    left = F.linear(x_norm, w["left_proj.weight"])       # (B,N,N,hidden)
    right = F.linear(x_norm, w["right_proj.weight"])

    # -------------------------------------------------------
    # 3) Gates (sigmoid) – also in FP32
    # -------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, w["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, w["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, w["out_gate.weight"]))

    # -------------------------------------------------------
    # 4) Fuse mask & gates with Triton
    # -------------------------------------------------------
    if mask is None:
        # no mask → use a tensor of ones, shape (B,N,N,1)
        mask_fused = torch.ones(x.shape[0], x.shape[1], x.shape[2], 1,
                                dtype=x.dtype, device=x.device)
    else:
        mask_fused = mask.unsqueeze(-1)                   # (B,N,N,1)

    # Triton kernel works in‑place
    left, right = _fuse_mask_and_gates(left, right, mask_fused,
                                       left_gate, right_gate)

    # -------------------------------------------------------
    # 5) N³ contraction via batched matrix‑multiply
    #    left_h @ right_hᵀ  for each hidden channel h
    # -------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched GEMM (cuBLAS) – very fast on H200
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B, N, N, hidden)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)

    # -------------------------------------------------------
    # 6) Output LayerNorm
    # -------------------------------------------------------
    ln2_w = w["to_out_norm.weight"]
    ln2_b = w["to_out_norm.bias"]
    mean2 = out.mean(-1, keepdim=True)
    var2 = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - mean2) / torch.sqrt(var2 + eps)
    out_norm = out_norm * ln2_w + ln2_b

    # -------------------------------------------------------
    # 7) Apply output gate and final linear projection
    # -------------------------------------------------------
    out_norm = out_norm * out_gate
    out_final = F.linear(out_norm, w["to_out.weight"])   # (B,N,N,dim)

    # ensure fp32 output (the rest of the pipeline already uses fp32)
    return out_final.to(torch.float32)
```

### How the lessons were applied
| Lesson | Relevance | Action taken |
|--------|-----------|--------------|
| **Limit shared memory** | Not a bottleneck – the kernel only loads/stores a few scalars per thread, so shared memory is negligible. | No shared memory usage, thus nothing to tune. |
| **Match stride unpacking** | Critical – earlier attempts assumed a fixed number of strides, causing crashes when tensors were non‑contiguous. | Strides are fetched directly from each tensor (`tensor.stride()`) and passed explicitly to the kernel; the kernel decodes exactly four dimensions for the 4‑D tensors and three for the mask. |
| **Avoid unnecessary tiling** | The heavy N³ work is delegated to cuBLAS (`torch.bmm`), which already implements optimal tiling for the target GPU. | No custom tiling for the mat‑mul part; only a 1‑D launch with a modest `BLOCK=1024` for the element‑wise fuse kernel. |
| **Use in‑place fusion** | Reduces memory traffic and allocation pressure. | The Triton kernel overwrites `left` and `right` directly. |
| **Mixed‑precision** | Not required for correctness; staying in FP32 avoids extra casts and keeps the LayerNorm numerics stable. | All operations are FP32; the kernel works on FP32 data. |

The resulting implementation keeps the memory footprint low, leverages the highly‑optimised cuBLAS path for the cubic‑cost contraction, and fuses the cheap point‑wise work in a custom Triton kernel, meeting both correctness and the required runtime target on the H200.
