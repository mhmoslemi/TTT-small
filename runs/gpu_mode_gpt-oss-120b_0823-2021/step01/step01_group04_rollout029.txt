"""
TriMul “outgoing” forward pass (AlphaFold‑3 style).

The most expensive step is the N³ pairwise multiplication
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]      (d = hidden_dim)
which is exactly a batched matrix‑multiply per hidden dimension:
    out_h = left_h @ right_hᵀ
We therefore reshape (B,N,N,hidden) → (B*hidden, N, N) and use a single
torch.bmm (cuBLAS) for the N³ work.

All cheap element‑wise work (LayerNorm, linear projections, gating,
masking) is kept in PyTorch except the fusion of mask·left·left_gate and
mask·right·right_gate.  This fusion is performed by a tiny 1‑D Triton
kernel that reads the four tensors, applies the three multiplicative
factors and writes the results back in‑place, saving a full pass over the
large tensors.

The implementation respects the required signature:
    custom_kernel(data) → torch.Tensor
where
    data = (input, mask, weights, config)

Only the forward pass is implemented; gradients are not needed.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel – fuse mask, left‑gate and right‑gate into the projected
# tensors.  This saves three separate element‑wise passes.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,        # [B, N, N, H]  (float32)
    right_ptr,       # [B, N, N, H]  (float32)
    mask_ptr,        # [B, N, N, 1]  (float32)
    lgate_ptr,       # [B, N, N, H]  (float32)
    rgate_ptr,       # [B, N, N, H]  (float32)

    # static sizes
    B, N, H,

    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,

    BLOCK: tl.constexpr,
):
    """Fuse mask and two gates into left/right tensors (in‑place)."""
    pid = tl.program_id(0)
    total = B * N * N * H                     # total elements to process
    offs = pid * BLOCK + tl.arange(0, BLOCK)  # linear offsets for this thread
    mask = offs < total

    # ---- decode linear offset (b,i,k,h) ----
    b = offs // (N * N * H)
    rest = offs % (N * N * H)
    i = rest // (N * H)
    rest = rest % (N * H)
    k = rest // H
    h = rest % H

    # ---- compute pointers ----
    left_off  = (b * stride_left_b  + i * stride_left_i  +
                 k * stride_left_k  + h * stride_left_h)
    right_off = (b * stride_right_b + i * stride_right_i +
                 k * stride_right_k + h * stride_right_h)
    mask_off  = (b * stride_mask_b  + i * stride_mask_i  + k * stride_mask_k)
    lgate_off = (b * stride_lgate_b + i * stride_lgate_i +
                 k * stride_lgate_k + h * stride_lgate_h)
    rgate_off = (b * stride_rgate_b + i * stride_rgate_i +
                 k * stride_rgate_k + h * stride_rgate_h)

    # ---- load ----
    left  = tl.load(left_ptr + left_off,  mask=mask, other=0.0)
    right = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    m     = tl.load(mask_ptr + mask_off,   mask=mask, other=0.0)   # broadcasted
    lgate = tl.load(lgate_ptr + lgate_off, mask=mask, other=0.0)
    rgate = tl.load(rgate_ptr + rgate_off, mask=mask, other=0.0)

    # ---- fuse ----
    left  = left  * m * lgate
    right = right * m * rgate

    # ---- store back in‑place ----
    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _fuse_mask_and_gates(left: torch.Tensor,
                         right: torch.Tensor,
                         mask: torch.Tensor,
                         left_gate: torch.Tensor,
                         right_gate: torch.Tensor) -> None:
    """
    In‑place fusion of mask and gates using a 1‑D Triton launch.
    All tensors are contiguous, float32, on the same CUDA device.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                     # good occupancy on Hopper
    grid = (triton.cdiv(total, BLOCK),)

    # strides in element counts
    s_left = left.stride()
    s_right = right.stride()
    s_mask = mask.stride()
    s_lgate = left_gate.stride()
    s_rgate = right_gate.stride()

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        lgate_ptr=left_gate,
        rgate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=s_left[0], stride_left_i=s_left[1],
        stride_left_k=s_left[2], stride_left_h=s_left[3],
        stride_right_b=s_right[0], stride_right_i=s_right[1],
        stride_right_k=s_right[2], stride_right_h=s_right[3],
        stride_mask_b=s_mask[0], stride_mask_i=s_mask[1], stride_mask_k=s_mask[2],
        stride_lgate_b=s_lgate[0], stride_lgate_i=s_lgate[1],
        stride_lgate_k=s_lgate[2], stride_lgate_h=s_lgate[3],
        stride_rgate_b=s_rgate[0], stride_rgate_i=s_rgate[1],
        stride_rgate_k=s_rgate[2], stride_rgate_h=s_rgate[3],
        BLOCK=BLOCK,
    )
    # kernels work in‑place – nothing to return


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]               # input & output channel dim
    hidden_dim = config["hidden_dim"] # inner hidden dimension (c_z)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (no bias in the original code)
    # ------------------------------------------------------------------
    eps = 1e-5
    w_ln = weights["norm.weight"]
    b_ln = weights["norm.bias"]
    # mean / var along channel dim (last)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * w_ln + b_ln                     # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)             # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Gating (sigmoid) – also linear without bias
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 4️⃣ Mask handling – unsqueeze to broadcast over hidden dim
    # ------------------------------------------------------------------
    if mask is None:
        # if no mask is supplied we treat everything as valid
        mask = torch.ones_like(input_tensor[..., 0:1])  # shape [B,N,N,1]
    else:
        mask = mask.unsqueeze(-1)                        # [B,N,N,1]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask & gates into left/right (in‑place) via Triton
    # ------------------------------------------------------------------
    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 6️⃣ The N³ einsum reduced to batched matmul per hidden dimension:
    #       out_h = left_h @ right_hᵀ
    #    We reshape (B,N,N,hidden) → (B*hidden, N, N) and use torch.bmm.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matrix multiply: (B*H, N, N) @ (B*H, N, N)^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, hidden]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (hidden -> hidden)
    # ------------------------------------------------------------------
    eps_o = 1e-5
    w_out_ln = weights["to_out_norm.weight"]
    b_out_ln = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps_o)
    out = out * w_out_ln + b_out_ln

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection hidden -> dim (no bias)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)               # [B,N,N,dim]

    # Ensure float32 output (all ops are float32 already)
    return out
