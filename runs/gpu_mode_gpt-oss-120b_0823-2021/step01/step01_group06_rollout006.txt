"""
TriMul “outgoing” implementation (forward only).

The expensive N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply per hidden dimension d.
We keep the LayerNorm / Linear / gating logic in PyTorch (already fused by
torch.compile) and offload the batched GEMM to a custom Triton kernel.
The kernel computes a fused batched GEMM for all (batch * hidden) matrices
simultaneously, using a 2‑D program grid (M‑tiles × N‑tiles) and a 1‑D
program id for the (batch, hidden) pair.

This layout avoids any extra memory copies: left/right are reshaped to
    (B*H, N, N)   (row‑major) and the kernel writes the result into
    out (same shape).  All arithmetic is performed in fp32, the final
    projection back to `dim` is done in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B^T) for many (B*H) matrices
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_outgoing_kernel(
    A_ptr,          # left  tensor  [BH, N, N]
    B_ptr,          # right tensor  [BH, N, N]
    C_ptr,          # output tensor [BH, N, N]
    BH, N,
    stride_ah, stride_ai, stride_aj,   # strides of A (batch, row, col)
    stride_bh, stride_bi, stride_bj,   # strides of B
    stride_ch, stride_ci, stride_cj,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    FP16_ACC: tl.constexpr,
):
    """C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]   (note B is transposed)."""
    pid = tl.program_id(0)                # which batch*hidden index
    pid_m = tl.program_id(1)              # tile row
    pid_n = tl.program_id(2)              # tile col

    bh = pid
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows/cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointer offsets for the current batch*hidden matrix
    a_ptrs = A_ptr + bh * stride_ah + offs_m[:, None] * stride_ai + tl.arange(0, BLOCK_K)[None, :] * stride_aj
    b_ptrs = B_ptr + bh * stride_bh + offs_n[:, None] * stride_bi + tl.arange(0, BLOCK_K)[None, :] * stride_bj  # B is NOT transposed yet

    # accumulator
    if FP16_ACC:
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)
    else:
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        # load A tile [M, K]
        a = tl.load(a_ptrs + k_offset * stride_aj, mask=mask_m[:, None] & (k_offset + tl.arange(0, BLOCK_K) < N), other=0.0)
        # load B tile [N, K]  (we need B^T, so we load rows of B and later dot with a)
        b = tl.load(b_ptrs + k_offset * stride_bj, mask=mask_n[:, None] & (k_offset + tl.arange(0, BLOCK_K) < N), other=0.0)
        # compute dot product: a @ b^T   (a: MxK, b: NxK) → MxN
        if FP16_ACC:
            acc += tl.dot(a, b, allow_tf32=False).to(tl.float16)
        else:
            acc += tl.dot(a, b, allow_tf32=False)

    # write back
    c_ptrs = C_ptr + bh * stride_ch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: fused LayerNorm (bias + weight) – identical to torch.nn.LayerNorm
# ----------------------------------------------------------------------
def fused_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """LayerNorm over last dimension (no cheap GPU kernel, use PyTorch)."""
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, seq_len, seq_len, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = fused_layernorm(x, weights["norm.weight"], weights["norm.bias"])

    # ------------------------------------------------------------------
    # 2. Linear projections (dim -> hidden) – bias‑free
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – also dim -> hidden
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ reduction via Triton batched GEMM
    #    reshape to (BH, N, N) where BH = B * hidden
    # ------------------------------------------------------------------
    left_rs = left.permute(0, 3, 1, 2).contiguous().reshape(B * hidden, N, N)   # (BH, N, N)
    right_rs = right.permute(0, 3, 1, 2).contiguous().reshape(B * hidden, N, N)

    out_rs = torch.empty_like(left_rs)   # allocate output

    # launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32                         # small enough to fit in shared memory
    grid = (
        B * hidden,                      # program_id(0) → different matrix
        triton.cdiv(N, BLOCK_M),         # program_id(1) → tile row
        triton.cdiv(N, BLOCK_N),         # program_id(2) → tile col
    )

    batched_gemm_outgoing_kernel[grid](
        left_rs,
        right_rs,
        out_rs,
        B * hidden,
        N,
        left_rs.stride(0), left_rs.stride(1), left_rs.stride(2),
        right_rs.stride(0), right_rs.stride(1), right_rs.stride(2),
        out_rs.stride(0), out_rs.stride(1), out_rs.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        FP16_ACC=0,                      # keep fp32 accumulator (more accurate)
    )

    # reshape back to [B, N, N, hidden]
    out = out_rs.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear (hidden -> dim)
    # ------------------------------------------------------------------
    out = fused_layernorm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # ensure float32 output
    return out.to(torch.float32)
