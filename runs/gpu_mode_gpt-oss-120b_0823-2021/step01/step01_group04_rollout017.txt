"""
TriMul “outgoing” forward pass (AlphaFold3).

The expensive O(N³) contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

is a batch of independent matrix‑matrix multiplications:
    left_h   : (B, N, N, H) → (B*H, N, N)
    right_hᵀ : (B, N, N, H) → (B*H, N, N).transpose(-2,-1)

so we can compute it with a single torch.bmm (cuBLAS) after a cheap reshape.

All element‑wise work (LayerNorm, linear projections, gating, masking) is
performed in fp32 with a lightweight Triton kernel that fuses:

    left  = (x @ W_leftᵀ)   * mask * sigmoid(x @ W_lgateᵀ)
    right = (x @ W_rightᵀ)  * mask * sigmoid(x @ W_rgateᵀ)

The kernel works **in‑place** on the projected tensors, keeping memory traffic
low.  The remaining normalisations and the final linear projection are done
in PyTorch (these are tiny compared with the N³ work).

The implementation supports both masked and un‑masked inputs, arbitrary batch
size B∈{1,2}, sequence length N up to 1024, and hidden dimensions 128‑384.
All tensors stay on the CUDA device; the final result is returned as float32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel – fuses mask, left/right projections and gating
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_fuse_kernel(
    # pointers (contiguous tensors, 4‑D)
    left_ptr, right_ptr,                # [B, N, N, H]  fp32
    mask_ptr,                           # [B, N, N, 1]  fp32 (broadcasted)
    left_gate_ptr, right_gate_ptr,      # [B, N, N, H]  fp32
    # static sizes
    B, N, H,                            # compile‑time ints
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    BLOCK: tl.constexpr,
):
    """In‑place: left = left * mask * left_gate ; right = right * mask * right_gate."""
    pid = tl.program_id(0)                           # 1‑D launch
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)         # linear indices this thread handles
    mask = offs < total

    # ----- decode linear offset into (b,i,k,h) -----
    b = (offs // (N * N * H))
    rest = offs % (N * N * H)
    i = (rest // (N * H))
    rest = rest % (N * H)
    k = (rest // H)
    h = rest % H

    # ----- compute element pointers -----
    left_off  = b * stride_left_b  + i * stride_left_i  + k * stride_left_k  + h * stride_left_h
    right_off = b * stride_right_b + i * stride_right_i + k * stride_right_k + h * stride_right_h
    mask_off  = b * stride_mask_b  + i * stride_mask_i  + k * stride_mask_k                    # mask has no channel dim
    lgate_off = b * stride_lgate_b + i * stride_lgate_i + k * stride_lgate_k + h * stride_lgate_h
    rgate_off = b * stride_rgate_b + i * stride_rgate_i + k * stride_rgate_k + h * stride_rgate_h

    # ----- load, fuse and store back -----
    left  = tl.load(left_ptr  + left_off,  mask=mask, other=0.0)
    right = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    m     = tl.load(mask_ptr  + mask_off,  mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr  + lgate_off, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + rgate_off, mask=mask, other=0.0)

    left  = left * m * lgate
    right = right * m * rgate

    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)


def _launch_fuse_kernel(left, right, mask, left_gate, right_gate):
    """
    Calls the Triton kernel above.  All tensors must be contiguous and on CUDA.
    Returns the (still contiguous) left and right tensors after in‑place fusion.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                                    # good occupancy on H200
    grid = (triton.cdiv(total, BLOCK),)

    # strides in elements (torch returns strides in number of elements, not bytes)
    sl = left.stride()
    sr = right.stride()
    sm = mask.stride()
    sg = left_gate.stride()
    rg = right_gate.stride()

    _tri_mul_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        stride_left_b=sl[0], stride_left_i=sl[1], stride_left_k=sl[2], stride_left_h=sl[3],
        stride_right_b=sr[0], stride_right_i=sr[1], stride_right_k=sr[2], stride_right_h=sr[3],
        stride_mask_b=sm[0], stride_mask_i=sm[1], stride_mask_k=sm[2],
        stride_lgate_b=sg[0], stride_lgate_i=sg[1], stride_lgate_k=sg[2], stride_lgate_h=sg[3],
        stride_rgate_b=rg[0], stride_rgate_i=rg[1], stride_rgate_k=rg[2], stride_rgate_h=rg[3],
        BLOCK=BLOCK,
    )
    return left, right


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of TriMul “outgoing” (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        * input_tensor : torch.Tensor [B, N, N, dim] (float32, CUDA)
        * mask         : torch.Tensor [B, N, N]    (float32 or bool, CUDA) or None
        * weights_dict : dict containing all Linear/LayerNorm weights
        * config_dict  : {"dim": int, "hidden_dim": int}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm on the last dimension (fp32)
    # ------------------------------------------------------------------
    ln_w = w["norm.weight"]
    ln_b = w["norm.bias"]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * ln_w + ln_b          # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free) → [B,N,N,hidden]
    # ------------------------------------------------------------------
    left_proj_w  = w["left_proj.weight"]   # (hidden, dim)
    right_proj_w = w["right_proj.weight"]

    left  = F.linear(x_norm, left_proj_w)   # (B,N,N,hidden)
    right = F.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Gates (bias‑free) → sigmoid(...)
    # ------------------------------------------------------------------
    left_gate_w  = w["left_gate.weight"]
    right_gate_w = w["right_gate.weight"]
    out_gate_w   = w["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_w))

    # ------------------------------------------------------------------
    # 4) Mask handling – broadcast to channel dimension
    # ------------------------------------------------------------------
    if mask is None:
        # no mask → a tensor of ones with correct shape
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2], 1,
                          dtype=x.dtype, device=x.device)
    else:
        mask = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]

    # ------------------------------------------------------------------
    # 5) Fuse mask and gates with Triton (in‑place on left/right)
    # ------------------------------------------------------------------
    left, right = _launch_fuse_kernel(left.contiguous(),
                                      right.contiguous(),
                                      mask.contiguous(),
                                      left_gate.contiguous(),
                                      right_gate.contiguous())

    # ------------------------------------------------------------------
    # 6) The O(N³) contraction via batched matmul
    #    left_h  : (B,N,N,hidden) → (B*hidden, N, N)
    #    right_hᵀ: (B,N,N,hidden) → (B*hidden, N, N).T
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    left_flat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)          # (B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)         # (B*H, N, N)

    # batched matmul (cuBLAS, fp32)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))        # (B*H, N, N)

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)                # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm
    # ------------------------------------------------------------------
    out_ln_w = w["to_out_norm.weight"]
    out_ln_b = w["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * out_ln_w + out_ln_b

    # ------------------------------------------------------------------
    # 8) Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate           # gate broadcasted over hidden dim
    to_out_w = w["to_out.weight"]            # (dim, hidden)
    out_final = F.linear(out_norm, to_out_w) # (B,N,N,dim)

    return out_final.to(torch.float32)
