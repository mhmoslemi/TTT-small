"""
TriMul “outgoing” kernel (AlphaFold‑3) – fast Triton implementation.

The expensive O(N³·C) reduction is realised as a batched matrix‑multiply:
for every hidden dimension `h` (and batch `b`)

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
                = (left_h @ right_hᵀ)[i,j]

Thus we reshape the tensors to a single batch dimension B·C and run a
standard GEMM A @ Bᵀ with a Triton matmul kernel.  All other
operations (LayerNorm, linear projections, gating, final LayerNorm and
projection) stay in PyTorch – they are cheap compared with the O(N³·C)
work and benefit from the highly‑optimised cuBLAS kernels.

The kernel works on float32, but the main compute can be performed in
float16 for speed (the host code casts back to float32 before returning).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (batched GEMM)
# A : (batch, M, K)   – left projection
# B : (batch, N, K)   – right projection  (we will pass its transpose)
# C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,        # pointers
    M, N, K,                    # matrix dims
    stride_az, stride_am, stride_ak,   # strides for A
    stride_bz, stride_bn, stride_bk,   # strides for B (already transposed)
    stride_cz, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)          # batch index (B·C)
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block col

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the block
    a_ptrs = a_ptr + pid_z * stride_az + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_z * stride_bz + offs_n[None, :] * stride_bn + tl.arange(0, BLOCK_K)[:, None] * stride_bk

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load blocks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k * BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[None, :] & (tl.arange(0, BLOCK_K)[:, None] < K - k * BLOCK_K), other=0.0)

        # dot product
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c_ptrs = c_ptr + pid_z * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward of the “outgoing” TriMul operator.

    Args
    ----
    data : (input, mask, weights, config)
        input  : [B, N, N, dim]          float32
        mask   : [B, N, N]               float32 (0/1) – may be all‑ones.
        weights: dict of model parameters (see skeleton).
        config : dict with ``dim`` and ``hidden_dim`` (and optional ``nomask``).

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. LayerNorm (manual, no torch.nn to keep everything in pure torch)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    eps = 1e-5
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask – applied with a tiny Triton kernel (vector‑wise mul)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N,1] broadcast over hidden dim
        mask_ = mask.unsqueeze(-1).to(dtype)
        B, N, _, H = left.shape

        BLOCK = 1024
        total = B * N * N * H

        @triton.jit
        def mask_kernel(src_ptr, mask_ptr, dst_ptr, total_el, H, BLOCK: tl.constexpr):
            pid = tl.program_id(0)
            offs = pid * BLOCK + tl.arange(0, BLOCK)
            mask = offs < total_el

            src = tl.load(src_ptr + offs, mask=mask, other=0.0)
            m = tl.load(mask_ptr + (offs // H), mask=mask, other=1.0)
            tl.store(dst_ptr + offs, src * m, mask=mask)

        src = left.view(-1)
        dst = torch.empty_like(src)
        mask_flat = mask_.view(-1)       # [B,N,N]
        grid = (triton.cdiv(total, BLOCK),)
        mask_kernel[grid](src, mask_flat, dst, total, H, BLOCK=BLOCK)
        left = dst.view_as(left)

        src = right.view(-1)
        dst = torch.empty_like(src)
        grid = (triton.cdiv(total, BLOCK),)
        mask_kernel[grid](src, mask_flat, dst, total, H, BLOCK=BLOCK)
        right = dst.view_as(right)

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·C) reduction – batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    # we need rightᵀ for the multiplication
    right_t = right_flat.transpose(1, 2).contiguous()

    out_flat = torch.empty_like(left_flat)       # (B*H, N, N)

    # launch parameters – these work well on Hopper (sm_90)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(B * H, 1),            # batch dimension (z)
        triton.cdiv(N, BLOCK_M),          # row blocks
        triton.cdiv(N, BLOCK_N),          # col blocks
    )

    tri_mul_kernel[grid](
        left_flat,
        right_t,
        out_flat,
        N,
        N,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_t.stride(0),   right_t.stride(1),   right_t.stride(2),
        out_flat.stride(0),  out_flat.stride(1),  out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 6. Final LayerNorm, gating and output projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean = out.mean(-1, keepdim=True)
    var = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean) / torch.sqrt(var + eps)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.type(dtype)
