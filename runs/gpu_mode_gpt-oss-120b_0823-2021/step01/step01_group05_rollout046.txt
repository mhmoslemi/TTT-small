"""
TriMul “outgoing” kernel (AlphaFold‑3) – Triton implementation.

The expensive O(N³·C) reduction

    out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]

is carried out by a tiled batched‑matrix‑multiply kernel written in Triton.
All other operations (LayerNorm, Linear projections, gating, the final
LayerNorm/projection) stay in PyTorch for numerical stability.  A tiny
mask‑application kernel is also provided for the optional pair‑wise mask.

The kernel works for any batch size B (≤2), sequence length N (≤1024) and
hidden dimension C (the “hidden_dim” in the model, typically 128).  It uses
fp16 arithmetic inside the reduction and casts the result back to fp32,
which is safe because the surrounding PyTorch code follows the reference
implementation’s precision.

The launch grid is 1‑D and encodes (batch, i‑tile, j‑tile) in a single
program id, which avoids the “stride‑mismatch” pitfalls seen in earlier
attempts.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# Tiny kernel that applies the optional mask (broadcast on the
# hidden dimension).  This is the only part that *must* run in Triton
# according to the task statement.
# --------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # const float*  (shape B,N,N,C)
    mask_ptr,         # const float*  (shape B,N,N)  broadcasted
    dst_ptr,          # float*        (output)
    total_el,         # i64  total number of elements in src
    H,                # i64 hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_el

    # load source value
    src = tl.load(src_ptr + offs, mask=active, other=0.0)

    # each hidden‑dim element shares the same mask value → divide index by H
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)

    tl.store(dst_ptr + offs, src * m, mask=active)


def triton_mask(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    tensor : [B, N, N, C]  (fp32)
    mask   : [B, N, N]     (fp32 0/1)
    returns tensor * mask[...,None] computed with Triton.
    """
    B, N, _, C = tensor.shape
    total = B * N * N * C
    BLOCK = 1024

    src = tensor.view(-1)
    dst = torch.empty_like(src)
    mask_f = mask.view(-1).to(tensor.dtype)

    grid = (triton.cdiv(total, BLOCK),)
    mask_apply_kernel[grid](
        src,
        mask_f,
        dst,
        total,
        C,
        BLOCK=BLOCK,
    )
    return dst.view(B, N, N, C)


# --------------------------------------------------------------
# Core O(N³·C) kernel – tiled batched matrix multiply with a hidden
# dimension loop.  We compute:
#   out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const float*  (B,N,N,C)   left after proj+gate+mask
    right_ptr,         # const float*  (B,N,N,C)   right after proj+gate+mask
    out_ptr,           # float*        (B,N,N,C)   output
    B, N, C,           # i64 scalar dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_c,
    stride_right_b, stride_right_j, stride_right_k, stride_right_c,
    stride_out_b, stride_out_i, stride_out_j, stride_out_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_C: tl.constexpr,
):
    """
    Tiling parameters (chosen for Hopper):
        BLOCK_M = BLOCK_N = 32   (tiles over i, j)
        BLOCK_K = 32             (tile over reduction axis k)
        BLOCK_C = 32             (process 32 hidden channels at a time)
    """
    pid = tl.program_id(0)

    # ----------------------------------------------------------
    # Decode pid -> (batch, i_tile, j_tile)
    # ----------------------------------------------------------
    num_i_tiles = tl.cdiv(N, BLOCK_M)
    num_j_tiles = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_i_tiles * num_j_tiles

    batch_id = pid // tiles_per_batch
    tile_id = pid % tiles_per_batch
    i_tile = tile_id // num_j_tiles
    j_tile = tile_id % num_j_tiles

    # ------------------------------------------------------------------
    # Pointers to the start of the current tiles
    # ------------------------------------------------------------------
    i_off = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # each hidden‑dim block we will accumulate separately
    c_off = tl.arange(0, BLOCK_C)

    # accumulator in fp32 for stability
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_C), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k tiles
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # ---------------------------
        # Load left tile  (M x K x C)
        # ---------------------------
        left_ptrs = (
            left_ptr
            + batch_id * stride_left_b
            + i_off[:, None, None] * stride_left_i
            + k_off[None, :, None] * stride_left_k
            + c_off[None, None, :] * stride_left_c
        )
        left = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None], other=0.0)

        # ---------------------------
        # Load right tile (N x K x C) and transpose K ↔ N
        # ---------------------------
        right_ptrs = (
            right_ptr
            + batch_id * stride_right_b
            + j_off[:, None, None] * stride_right_j
            + k_off[None, :, None] * stride_right_k
            + c_off[None, None, :] * stride_right_c
        )
        # shape after load: (BLOCK_N, BLOCK_K, BLOCK_C)
        right = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_mask[None, :, None], other=0.0)

        # -------------------------------------------------
        # Inner product over K for each channel block.
        # Use tl.dot which returns a scalar – we apply it
        # for every (i,j,c) combination.
        # -------------------------------------------------
        for c in range(0, BLOCK_C):
            # left[:, :, c] : (M, K)
            # right[:, :, c] : (N, K)
            # we need (M,K) dot (N,K)ᵀ → (M,N)
            l = left[:, :, c]          # (M,K)
            r = right[:, :, c]         # (N,K)
            # tl.dot works on 1‑D vectors; we perform a matmul by
            # broadcasting the dot over the outer dimensions.
            # The trick: compute outer product of rows via dot inside
            # two nested loops (M,N are ≤32, so this is cheap).
            for mi in range(0, BLOCK_M):
                li = l[mi, :]           # (K)
                for nj in range(0, BLOCK_N):
                    rj = r[nj, :]       # (K)
                    prod = tl.dot(li, rj)   # scalar fp32
                    acc[mi, nj, c] += prod

    # ------------------------------------------------------------------
    # Store the result (only for valid i/j positions)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch_id * stride_out_b
        + i_off[:, None, None] * stride_out_i
        + j_off[None, :, None] * stride_out_j
        + c_off[None, None, :] * stride_out_c
    )
    mask_store = i_mask[:, None, None] & j_mask[None, :, None]
    tl.store(out_ptrs, acc, mask=mask_store)


# --------------------------------------------------------------
# Helper that launches the Triton kernel
# --------------------------------------------------------------
def trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, C]  (float32)
    returns out  : [B, N, N, C]  (float32)
    """
    B, N, _, C = left.shape
    out = torch.empty_like(left)

    # strides (in elements, not bytes) – assume contiguous layout
    s_b, s_i, s_k, s_c = left.stride()
    s_br, s_jr, s_kr, s_cr = right.stride()
    s_bo, s_io, s_jo, s_co = out.stride()

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_C = 32                     # hidden dim is a multiple of 32 (128)

    # total number of program ids = B * ceil(N/BLOCK_M) * ceil(N/BLOCK_N)
    grid = (
        B
        * triton.cdiv(N, BLOCK_M)
        * triton.cdiv(N, BLOCK_N),
    )
    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        C,
        s_b,
        s_i,
        s_k,
        s_c,
        s_br,
        s_jr,
        s_kr,
        s_cr,
        s_bo,
        s_io,
        s_jo,
        s_co,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_C=BLOCK_C,
    )
    return out


# --------------------------------------------------------------
# Main entry‑point used by the grader
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : tuple
        (input, mask, weights, config)
        - input : [B, N, N, dim]  torch.float32
        - mask  : [B, N, N]       torch.float32 (0/1) – may be all‑ones
        - weights : dict of model parameters
        - config  : dict with keys ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor  [B, N, N, dim] float32
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # LayerNorm (manual version – no torch.nn.LayerNorm to keep everything on‑device)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * norm_w + norm_b

    # ------------------------------------------------------------------
    # Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask – Triton‑accelerated element‑wise multiplication
    # ------------------------------------------------------------------
    if not nomask:
        left = triton_mask(left, mask)
        right = triton_mask(right, mask)

    # ------------------------------------------------------------------
    # Gates (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·C) reduction – Triton implementation
    # ------------------------------------------------------------------
    out = trimul_outgoing_triton(left, right)          # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Final LayerNorm + gating + output projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(dtype)   # ensure same dtype as input (float32)
