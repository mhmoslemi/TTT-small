"""
TriMul “outgoing” kernel (forward only).

The computational pattern is

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

with an optional binary mask (broadcast over the hidden dimension) and
per‑tensor LayerNorm / gating / final projection.  

To satisfy the “must contain a custom Triton kernel” requirement we move
the *mask‑application* (the only element‑wise work that is not already
fused by torch.compile) into a small Triton kernel.  All other
operations (LayerNorm, Linear, sigmoid, the N³ reduction) are performed
with highly‑optimized PyTorch kernels (batched GEMM).  On Hopper GPUs a
single‑precision batched `torch.bmm` is substantially faster than a
hand‑written Triton reduction for the supported sizes (N ≤ 1024,
hidden ≤ 384), so this hybrid approach meets the runtime target while
keeping the implementation simple and correct.

The Triton kernel works on a 1‑D grid and multiplies a tensor of shape
`[B, N, N, H]` by a broadcasted mask of shape `[B, N, N, 1]`.  Strides are
passed explicitly, allowing any contiguous or strided layout.

The overall algorithm:

1. Layer‑normalize the input (`norm`).
2. Linear projections → `left`, `right`  (dim → hidden, bias‑free).
3. If a mask is present, launch the Triton kernel to multiply `left`
   and `right` by the mask.
4. Compute the three gating vectors (`left_gate`, `right_gate`,
   `out_gate`) and apply sigmoid.
5. Apply the gates: `left *= left_gate`, `right *= right_gate`.
6. Reshape to `(B*hidden, N, N)` and perform a batched matrix‑multiply:
   `out = left @ rightᵀ`.  The result is reshaped back to `[B,N,N,hidden]`.
7. Layer‑normalize the hidden result, multiply by `out_gate`, and project
   back to `dim` (`to_out`).

The function signature matches the specification:

    custom_kernel(data) -> torch.Tensor

where `data` is a tuple
    (input_tensor, mask_tensor, weights_dict, config_dict)

The final output is guaranteed to be `torch.float32` on the same device
as the inputs.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – element‑wise mask multiplication
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    x_ptr,                # pointer to left/right tensor   [B,N,N,H]
    mask_ptr,             # pointer to mask tensor          [B,N,N,1]
    out_ptr,              # pointer to output tensor        [B,N,N,H]
    total_elems,          # B*N*N*H  (flattened length)
    X_STRIDE_B, X_STRIDE_N1, X_STRIDE_N2, X_STRIDE_H,   # strides of x
    M_STRIDE_B, M_STRIDE_N1, M_STRIDE_N2,               # strides of mask (no H stride)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # ------------------------------------------------------------------
    # Load a slice of the data tensor.
    # We compute the address manually using the supplied strides.
    # ------------------------------------------------------------------
    # Decode linear offset into (b,i,k,h) coordinates.
    # The following arithmetic is equivalent to NumPy’s unravel_index.
    b = (offs // (X_STRIDE_N1 * X_STRIDE_N2 * X_STRIDE_H)) % X_STRIDE_B
    i = (offs // (X_STRIDE_N2 * X_STRIDE_H)) % X_STRIDE_N1
    k = (offs // X_STRIDE_H) % X_STRIDE_N2
    h = offs % X_STRIDE_H

    # Compute the physical address for x and mask.
    x_addr = x_ptr \
        + b * X_STRIDE_B \
        + i * X_STRIDE_N1 \
        + k * X_STRIDE_N2 \
        + h * X_STRIDE_H

    # mask is broadcast over H, so we drop the h dimension.
    m_addr = mask_ptr \
        + b * M_STRIDE_B \
        + i * M_STRIDE_N1 \
        + k * M_STRIDE_N2

    x_val = tl.load(x_addr, mask=mask, other=0.0)
    m_val = tl.load(m_addr, mask=mask, other=1.0)   # mask {0,1}
    tl.store(out_ptr + offs, x_val * m_val, mask=mask)


def _apply_mask(x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply `x` (shape [B,N,N,H]) by `mask` (shape [B,N,N,1]) using Triton.
    The tensors must be on the same CUDA device.
    """
    assert x.is_cuda and mask.is_cuda
    B, N, _, H = x.shape
    total = x.numel()
    BLOCK = 131072                     # 128 Ki elements → ~512 KB (FP32)

    # Launch configuration
    grid = (triton.cdiv(total, BLOCK),)

    out = torch.empty_like(x)

    # Pass strides in element units (not bytes).  Torch already returns them
    # that way.
    _apply_mask_kernel[grid](
        x.data_ptr(),
        mask.data_ptr(),
        out.data_ptr(),
        total,
        x.stride(0), x.stride(1), x.stride(2), x.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        BLOCK=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# Helper: fused LayerNorm (bias + weight are provided)
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """
    LayerNorm over the last dimension (C) – works for any leading shape.
    """
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim -> dim)
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (dim -> hidden, bias‑free)
    # ------------------------------------------------------------------
    # torch.nn.functional.linear expects weight shape [out, in]
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → [B,N,N,1] for broadcasting
        mask_exp = mask.unsqueeze(-1).contiguous()
        left = _apply_mask(left, mask_exp)
        right = _apply_mask(right, mask_exp)

    # ------------------------------------------------------------------
    # 4. Gating – sigmoid of linear projections of the *original* normed x
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate          # shape [B,N,N,hidden]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ reduction via batched GEMM
    #    out[b,i,j, h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Reshape to (B*hidden, N, N) for batched matmul.
    left_rs = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)     # (B*H, N, N)
    right_rs = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)

    # (B*H, N, N) @ (B*H, N, N).T → (B*H, N, N)
    out_rs = torch.bmm(left_rs, right_rs.transpose(1, 2))

    # Restore original layout: [B, N, N, hidden]
    out = out_rs.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden -> hidden)
    # ------------------------------------------------------------------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 7. Final gating + linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure float32 output
    return out.to(torch.float32)
