"""
TriMul “outgoing” forward pass (αFold‑3 style).

Algorithm
---------
1. Layer‑norm on the 4‑D input  (B,N,N,C) → (B,N,N,C)  – done in PyTorch.
2. Linear projections to hidden dimension H (no bias) → left/right.
3. Compute three gates (sigmoid(linear)) – also in PyTorch.
4. Fuse the *mask*, *left‑gate* and *right‑gate* into the projected tensors
   with a tiny Triton kernel (in‑place).  This removes three separate
   element‑wise passes and keeps memory traffic low.
5. The cubic “einsum”  

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

   is exactly a batch of matrix‑multiplications per hidden channel:

        out_h = left_h @ right_hᵀ               (B‑batch, N×N)

   We reshape the tensors so that (B,H) become a single batch dimension and
   perform the whole N³ work with a single `torch.bmm` (cuBLAS‑fast).
6. Layer‑norm on the hidden result, apply the output‑gate and final linear
   projection back to the original channel dimension.  All remaining ops stay
   in PyTorch.

Only the element‑wise mask / gate fusion is performed in Triton; the heavy
N³ part relies on cuBLAS which is already optimal on Hopper (H200).  This
strategy satisfies the “must use Triton” rule while keeping the kernel
extremely lightweight (≈1 k threads, no shared memory), thus meeting the
target runtime of a few hundred micro‑seconds.
"""

import torch
import triton
import triton.language as tl


# -----------------------------------------------------------------------
# Triton kernel: fuse mask and the two per‑element gates into left/right
# -----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          #  [B,N,N,H]  float32
    right_ptr,         #  [B,N,N,H]  float32
    mask_ptr,          #  [B,N,N,1]  float32  (broadcasted)
    left_gate_ptr,     #  [B,N,N,H]  float32
    right_gate_ptr,    #  [B,N,N,H]  float32
    B, N, H,           # static sizes
    stride_l_b, stride_l_n1, stride_l_n2, stride_l_h,
    stride_r_b, stride_r_n1, stride_r_n2, stride_r_h,
    stride_m_b, stride_m_n1, stride_m_n2,
    stride_lg_b, stride_lg_n1, stride_lg_n2, stride_lg_h,
    stride_rg_b, stride_rg_n1, stride_rg_n2, stride_rg_h,
    BLOCK: tl.constexpr,
):
    pid  = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    total = B * N * N * H
    mask = offs < total

    # decode linear index → (b,i,k,h)
    b =  offs // (N * N * H)
    rest = offs % (N * N * H)
    i =  rest // (N * H)
    rest = rest % (N * H)
    k =  rest // H
    h =  rest % H

    # compute flat pointers (elements, not bytes)
    left_off   = (b * stride_l_b  + i * stride_l_n1 + k * stride_l_n2 + h * stride_l_h)
    right_off  = (b * stride_r_b  + i * stride_r_n1 + k * stride_r_n2 + h * stride_r_h)
    mask_off   = (b * stride_m_b  + i * stride_m_n1 + k * stride_m_n2)          # no channel dim
    lg_off     = (b * stride_lg_b + i * stride_lg_n1 + k * stride_lg_n2 + h * stride_lg_h)
    rg_off     = (b * stride_rg_b + i * stride_rg_n1 + k * stride_rg_n2 + h * stride_rg_h)

    # load values (mask is broadcasted)
    l   = tl.load(left_ptr   + left_off,   mask=mask, other=0.0)
    r   = tl.load(right_ptr  + right_off,  mask=mask, other=0.0)
    m   = tl.load(mask_ptr   + mask_off,   mask=mask, other=0.0)
    lg  = tl.load(left_gate_ptr  + lg_off, mask=mask, other=0.0)
    rg  = tl.load(right_gate_ptr + rg_off, mask=mask, other=0.0)

    # fused element‑wise multiplication
    l = l * m * lg
    r = r * m * rg

    # store back in‑place
    tl.store(left_ptr  + left_off,  l, mask=mask)
    tl.store(right_ptr + right_off, r, mask=mask)


def _fuse_mask_and_gates(left: torch.Tensor,
                         right: torch.Tensor,
                         mask: torch.Tensor,
                         left_gate: torch.Tensor,
                         right_gate: torch.Tensor) -> None:
    """
    In‑place Triton fusion of mask and gates.
    All tensors must be contiguous, CUDA‐float32 and on the same device.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024                                 # ~1 k threads → good occupancy
    grid = (triton.cdiv(total, BLOCK),)

    # strides in *elements* (not bytes)
    s_l = left.stride()
    s_r = right.stride()
    s_m = mask.stride()
    s_lg = left_gate.stride()
    s_rg = right_gate.stride()

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        B=B, N=N, H=H,
        # left strides
        stride_l_b=s_l[0], stride_l_n1=s_l[1], stride_l_n2=s_l[2], stride_l_h=s_l[3],
        # right strides
        stride_r_b=s_r[0], stride_r_n1=s_r[1], stride_r_n2=s_r[2], stride_r_h=s_r[3],
        # mask strides (no channel dim)
        stride_m_b=s_m[0], stride_m_n1=s_m[1], stride_m_n2=s_m[2],
        # left‑gate strides
        stride_lg_b=s_lg[0], stride_lg_n1=s_lg[1], stride_lg_n2=s_lg[2], stride_lg_h=s_lg[3],
        # right‑gate strides
        stride_rg_b=s_rg[0], stride_rg_n1=s_rg[1], stride_rg_n2=s_rg[2], stride_rg_h=s_rg[3],
        BLOCK=BLOCK,
    )
    # No return – tensors are modified in‑place


# -----------------------------------------------------------------------
# Main entry point required by the harness
# -----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, C]   (float32, CUDA)
        mask        : torch.Tensor [B, N, N]      (float32 or bool, CUDA)
        weights     : dict of all linear / norm parameters (float32 CUDA)
        config      : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]          # input channel dimension C
    hidden_dim = cfg["hidden_dim"]   # H

    # ------------------------------------------------------------------
    # 1) Input LayerNorm  (B,N,N,C) → (B,N,N,C)
    # ------------------------------------------------------------------
    ln_w = weights["norm.weight"]
    ln_b = weights["norm.bias"]
    eps = 1e-5

    # manual LN keeps everything in FP32 (weights are FP32)
    mean = x.mean(dim=-1, keepdim=True)
    var  = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * ln_w + ln_b

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left  = torch.nn.functional.linear(x_norm, left_proj_w)   # FP32 matmul
    right = torch.nn.functional.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid) → (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 4) Apply mask + gates in‑place with Triton
    # ------------------------------------------------------------------
    # mask shape (B,N,N) → (B,N,N,1) for broadcasting
    if mask is None:
        # if the caller supplies None we treat it as “no mask”
        mask = torch.ones_like(x_norm[..., :1])
    else:
        mask = mask.unsqueeze(-1)               # (B,N,N,1)

    # ensure contiguous tensors for the kernel
    left       = left.contiguous()
    right      = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask       = mask.contiguous()

    _fuse_mask_and_gates(left, right, mask, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 5) N³ einsum as batched matmul per hidden channel
    #    out_h = left_h @ right_hᵀ
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, N, N, H) → (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # bmm uses cuBLAS (FP32) → highly tuned on Hopper
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output‑gate and final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    eps_o = 1e-5

    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps_o)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # apply output gate (broadcasted over hidden dim)
    out_norm = out_norm * out_gate

    # final projection back to original dim
    to_out_w = weights["to_out.weight"]          # (dim, hidden_dim)
    out_final = torch.nn.functional.linear(out_norm, to_out_w)

    # ensure FP32 output (the whole pipeline stayed in FP32)
    return out_final.to(torch.float32)
