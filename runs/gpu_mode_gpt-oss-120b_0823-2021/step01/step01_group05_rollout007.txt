"""
tri_mul_outgoing.py
-------------------

Triton‑accelerated **outgoing** TriMul (AlphaFold‑3) forward pass.

The expensive O(N³·C) reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a blocked batched GEMM in Triton.  All
LayerNorms, Linear projections and gating are kept in PyTorch
(for numerical stability and to keep the code short).  The
kernel works for any batch size B∈{1,2}, sequence length N,
and hidden dimension H (the “hidden_dim” of the module).  It
expects the tensors to be contiguous in memory and on the
CUDA device.

The kernel processes a 2‑D tile of the output (M×N) for a
subset of the hidden channels (h‑block).  Each tile iterates
over K‑tiles, performing an fp16 dot‑product accumulation and
finally writes the result in float32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# Triton kernel: blocked batched GEMM for the TriMul reduction.
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,               # const float*   [B, N, N, H]
    right_ptr,              # const float*   [B, N, N, H]
    out_ptr,                # float*         [B, N, N, H]

    # tensor sizes
    B, N, H,                # i32
    # strides (row‑major layout)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,

    # tiling parameters (compile‑time constants)
    BLOCK_M: tl.constexpr,  # rows   of left   (i dimension)
    BLOCK_N: tl.constexpr,  # cols   of right  (j dimension)
    BLOCK_K: tl.constexpr,  # reduction dimension k
    H_BLOCK: tl.constexpr,  # hidden‑channel block size
):
    """
    Compute a tile of
        out[b, i, j, h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for a fixed batch `b` and a fixed hidden‑channel block
    `h_start … h_start+H_BLOCK`.
    """
    pid = tl.program_id(0)                # 0 … B* (H/H_BLOCK)-1

    # ----------------------------------------------------------
    # 1) Decode program‑id → batch index and hidden‑block offset
    # ----------------------------------------------------------
    h_block_id = pid % (H // H_BLOCK)     # which hidden block
    b_id       = pid // (H // H_BLOCK)    # which batch

    h_start = h_block_id * H_BLOCK

    # ----------------------------------------------------------
    # 2) Offsets for the first element of this tile
    # ----------------------------------------------------------
    i_offset = tl.program_id(1) * BLOCK_M
    j_offset = tl.program_id(2) * BLOCK_N

    # compute masks for out‑of‑bounds tiles
    i_mask = i_offset + tl.arange(0, BLOCK_M) < N
    j_mask = j_offset + tl.arange(0, BLOCK_N) < N

    # ----------------------------------------------------------
    # 3) Accumulators (float32 for precision, we accumulate in fp16
    #    then cast; this is faster on Hopper)
    # ----------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, H_BLOCK), dtype=tl.float32)

    # ----------------------------------------------------------
    # 4) Main K‑loop
    # ----------------------------------------------------------
    for k_outer in range(0, tl.cdiv(N, BLOCK_K)):
        # k‑range inside this tile
        k_offset = k_outer * BLOCK_K
        k_mask   = k_offset + tl.arange(0, BLOCK_K) < N

        # ------------------------------------------------------
        # Load left  : [M, K, H_BLOCK]
        # ------------------------------------------------------
        left_ptrs = left_ptr + \
            b_id   * stride_l_b + \
            (i_offset + tl.arange(0, BLOCK_M)[:, None]) * stride_l_i + \
            (k_offset + tl.arange(0, BLOCK_K)[None, :]) * stride_l_k + \
            (h_start + tl.arange(0, H_BLOCK)[None, None, :]) * stride_l_h
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float16)

        # ------------------------------------------------------
        # Load right : [N, K, H_BLOCK]  (note the transpose on k)
        # ------------------------------------------------------
        right_ptrs = right_ptr + \
            b_id   * stride_r_b + \
            (j_offset + tl.arange(0, BLOCK_N)[:, None]) * stride_r_i + \
            (k_offset + tl.arange(0, BLOCK_K)[None, :]) * stride_r_k + \
            (h_start + tl.arange(0, H_BLOCK)[None, None, :]) * stride_r_h
        right = tl.load(right_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float16)

        # ------------------------------------------------------
        # GEMM on the K dimension for every hidden channel
        # ------------------------------------------------------
        for h in range(H_BLOCK):
            # (M,K) dot (N,K)ᵀ → (M,N)
            acc[:, :, h] += tl.dot(left[:, :, h], right[:, :, h].T, accumulate_dtype=tl.float32)

    # ----------------------------------------------------------
    # 5) Store the result
    # ----------------------------------------------------------
    out_ptrs = out_ptr + \
        b_id   * stride_o_b + \
        (i_offset + tl.arange(0, BLOCK_M)[:, None]) * stride_o_i + \
        (j_offset + tl.arange(0, BLOCK_N)[None, :]) * stride_o_j + \
        (h_start + tl.arange(0, H_BLOCK)[None, None, :]) * stride_o_h

    tl.store(out_ptrs, acc, mask=i_mask[:, None, None] & j_mask[None, :, None])


# --------------------------------------------------------------
# Helper: launch the Triton kernel for the full tensor.
# --------------------------------------------------------------
def trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : [B, N, N, H]  (float32, contiguous, CUDA)
    returns      : out  [B, N, N, H]  (float32)
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # ------------------------------------------------------------------
    # Tiling parameters – tuned for Hopper (SM90) and the given sizes.
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    H_BLOCK = 16                      # hidden‑dim tile, must divide H

    # sanity
    assert H % H_BLOCK == 0, "hidden_dim must be a multiple of H_BLOCK (16)"

    # grid: (batch * (H/H_BLOCK),  ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        triton.cdiv(B * (H // H_BLOCK), 1),          # pid 0
        triton.cdiv(N, BLOCK_M),                    # pid 1 (i‑tile)
        triton.cdiv(N, BLOCK_N),                    # pid 2 (j‑tile)
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        H_BLOCK=H_BLOCK,
    )
    return out


# --------------------------------------------------------------
# Entry‑point required by the evaluation harness.
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input, mask, weights, config)
        - input : [B, N, N, dim]        float32 CUDA tensor
        - mask  : [B, N, N]             float32 (0/1) – may be all‑ones
        - weights : dict of layer‑norm and linear params
        - config  : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # 1) unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) LayerNorm (manual, no torch.nn.LayerNorm to keep everything on
    #    the same device and avoid extra Python overhead)
    # ------------------------------------------------------------------
    eps = 1e-5
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # ------------------------------------------------------------------
    # 3) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 4) Optional mask – broadcast on the hidden dimension
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → [B,N,N,1] for broadcasting
        mask_f = mask.float().unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Gates
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core O(N³·C) reduction – Triton GEMM implementation
    # ------------------------------------------------------------------
    # left/right : [B,N,N,hidden_dim] (float32)
    # out        : [B,N,N,hidden_dim] (float32)
    out = trimul_outgoing_triton(left, right)

    # ------------------------------------------------------------------
    # 7) Final LayerNorm + gating + output projection
    # ------------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
