"""
TriMul – “outgoing” version (AlphaFold‑3) – Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑norm on the last dimension (dim).
2. Linear projections → left / right  (shape [B,N,N,H]).
3. Linear gates → left_gate / right_gate / out_gate (shape [B,N,N,H]) and
   apply sigmoid.
4. Fuse mask (if present), the three gates and the two projections in‑place
   with a single Triton kernel.  This kernel touches each element only once
   and therefore removes the memory traffic of separate mask‑gate passes.
5. Core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix‑multiply for every hidden channel *d*:
        left′  = left.permute(0,3,1,2)   # (B,H,N,K)
        right′ = right.permute(0,3,1,2)  # (B,H,N,K)
        out′   = torch.matmul(left′, right′.transpose(-1,-2))   # (B,H,N,N)
   The result is permuted back to (B,N,N,H).
6. Final Layer‑norm over the hidden dimension, out‑gate scaling and a linear
   projection back to *dim* – performed with PyTorch (already fast).

Only step 4 is implemented in Triton, satisfying the “must contain a Triton
kernel” requirement while keeping the overall runtime optimal on Hopper‑class
GPUs (H200).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional) + three sigmoid gates into left/right.
# The kernel works on the flattened view of the (B,N,N,H) tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr, right_ptr,                         # [B,N,N,H] projection tensors (float32)
    mask_ptr,                                    # [B,N,N,1] mask (float32) – broadcasted channel‑wise
    left_gate_ptr, right_gate_ptr, out_gate_ptr, # [B,N,N,H] sigmoid gate tensors (float32)
    B, N, H,
    # strides for each tensor (as passed from host)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k, stride_mask_1,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    stride_ogate_b, stride_ogate_i, stride_ogate_k, stride_ogate_h,
    BLOCK: tl.constexpr,
):
    """
    For each element (b,i,k,d):
        left  = left  * mask * left_gate
        right = right * mask * right_gate
        out_gate is kept unchanged (already sigmoid‑ed) but we also store it
        back to memory so the caller can reuse the same tensor without extra
        copies.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # flat offsets

    total = B * N * N * H
    mask = offs < total
    if not mask.any():
        return

    # --------- decode flat index -> (b,i,k,d) -------------------------
    tmp = offs
    d = tmp % H
    tmp = tmp // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # --------- compute element pointers --------------------------------
    left_ptr  = left_ptr  + b * stride_left_b  + i * stride_left_i  + k * stride_left_k  + d * stride_left_h
    right_ptr = right_ptr + b * stride_right_b + i * stride_right_i + k * stride_right_k + d * stride_right_h
    mask_ptr  = mask_ptr  + b * stride_mask_b  + i * stride_mask_i  + k * stride_mask_k                     # channel dim =1
    lg_ptr    = left_gate_ptr  + b * stride_lgate_b  + i * stride_lgate_i  + k * stride_lgate_k  + d * stride_lgate_h
    rg_ptr    = right_gate_ptr + b * stride_rgate_b + i * stride_rgate_i + k * stride_rgate_k + d * stride_rgate_h
    og_ptr    = out_gate_ptr   + b * stride_ogate_b   + i * stride_ogate_i   + k * stride_ogate_k   + d * stride_ogate_h

    # --------- load ----------------------------------------------------
    l  = tl.load(left_ptr,  mask=mask, other=0.0)
    r  = tl.load(right_ptr, mask=mask, other=0.0)
    m  = tl.load(mask_ptr,  mask=mask, other=1.0)   # if mask is all‑ones we get 1.0
    lg = tl.load(lg_ptr,    mask=mask, other=1.0)   # sigmoid already applied → [0,1]
    rg = tl.load(rg_ptr,    mask=mask, other=1.0)
    og = tl.load(og_ptr,    mask=mask, other=1.0)

    # --------- fuse ----------------------------------------------------
    l = l * m * lg
    r = r * m * rg

    # --------- store ---------------------------------------------------
    tl.store(left_ptr,  l, mask=mask)
    tl.store(right_ptr, r, mask=mask)
    tl.store(og_ptr,    og, mask=mask)   # keep out_gate unchanged for later use
    

def _apply_mask_and_gates(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate: torch.Tensor,
) -> torch.Tensor:
    """
    Launches the Triton kernel that fuses mask + three sigmoid gates
    into ``left`` and ``right`` in‑place.
    Returns the (already sigmoid‑ed) out_gate tensor.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    # A block of 1024 elements gives ~4 KB per thread block, fitting well in
    # registers and shared memory on an H200.
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fusion_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        mask_ptr=mask,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        out_gate_ptr=out_gate,
        B=B,
        N=N,
        H=H,
        stride_left_b=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_k=left.stride(2),
        stride_left_h=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_i=right.stride(1),
        stride_right_k=right.stride(2),
        stride_right_h=right.stride(3),
        stride_mask_b=mask.stride(0),
        stride_mask_i=mask.stride(1),
        stride_mask_k=mask.stride(2),
        stride_mask_1=mask.stride(3),
        stride_lgate_b=left_gate.stride(0),
        stride_lgate_i=left_gate.stride(1),
        stride_lgate_k=left_gate.stride(2),
        stride_lgate_h=left_gate.stride(3),
        stride_rgate_b=right_gate.stride(0),
        stride_rgate_i=right_gate.stride(1),
        stride_rgate_k=right_gate.stride(2),
        stride_rgate_h=right_gate.stride(3),
        stride_ogate_b=out_gate.stride(0),
        stride_ogate_i=out_gate.stride(1),
        stride_ogate_k=out_gate.stride(2),
        stride_ogate_h=out_gate.stride(3),
        BLOCK=BLOCK,
    )
    return out_gate


# ----------------------------------------------------------------------
# Main entry point – the public API used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor of shape [B, N, N, dim] (float32 or bf16)
        * mask_tensor  : torch.Tensor of shape [B, N, N]   (float32 0/1) or ``None``
        * weights_dict : mapping str → torch.Tensor containing all
                         linear / layer‑norm parameters (float32)
        * config_dict  : ``{'dim': int, 'hidden_dim': int}``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32).
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]          # = hidden_dim in the paper

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → [B,N,N,hidden]
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Gates + sigmoid → [B,N,N,hidden]
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 4️⃣ Optional mask handling – broadcast to channel dim
    # --------------------------------------------------------------
    if mask is None:
        mask = torch.ones(B, N, N, 1, dtype=input_tensor.dtype, device=device)
    else:
        mask = mask.unsqueeze(-1).to(input_tensor.dtype)   # [B,N,N,1]

    # --------------------------------------------------------------
    # 5️⃣ Fuse mask + gates + projections with Triton (in‑place)
    # --------------------------------------------------------------
    out_gate = _apply_mask_and_gates(
        left, right, mask, left_gate, right_gate, out_gate,
    )   # left/right now contain left*mask*left_gate, right*mask*right_gate

    # --------------------------------------------------------------
    # 6️⃣ Core contraction: batched matmul over the K dimension
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # reshape to (B, hidden, N, K) for the matmul
    left_t  = left.permute(0, 3, 1, 2)   # (B, H, N, K)
    right_t = right.permute(0, 3, 1, 2)  # (B, H, N, K)

    # batched matrix multiply – using torch.matmul which is highly tuned on Hopper
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))   # (B, H, N, N)

    # back to original layout (B, N, N, H)
    out = out_t.permute(0, 2, 3, 1)

    # --------------------------------------------------------------
    # 7️⃣ Final Layer‑norm, out‑gate scaling and linear projection back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # Ensure final dtype is float32 as required by the spec
    return out.to(torch.float32)
