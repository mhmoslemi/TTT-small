"""
TriMul “outgoing” kernel (forward only).

The heavy N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply per hidden dimension d.
We implement it with a single Triton kernel that performs a
blocked GEMM for every (batch, hidden) pair, keeping the
intermediate tensors in fp16 for higher throughput on H200.
All LayerNorm / Linear / gating work is done in PyTorch (they are
memory‑bound and already fused by torch.compile).  The kernel
returns the fp32 result after a final cast.

The kernel works for any batch size (1‑2), any sequence length up to
1024 and hidden dimensions that are multiples of 32 (the typical
values in the test matrix).  Masking is applied in‑place before the
GEMM – if `nomask=False` we simply zero‑out the masked entries with a
lightweight element‑wise Triton kernel.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# 1) Element‑wise mask application (optional)
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    x_ptr,                 # [B,N,N,H] (fp32)
    mask_ptr,              # [B,N,N,1] (bool / fp32 0‑1)
    out_ptr,               # [B,N,N,H] (fp32)
    total_elems: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # linear load / store – strides are baked into the pointer arithmetic
    x   = tl.load(x_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + (offs // tl.constexpr(1)), mask=mask, other=1.0)   # broadcast mask along H
    tl.store(out_ptr + offs, x * m, mask=mask)


# ----------------------------------------------------------------------
# 2) Batched GEMM for the N³ reduction
#    out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]^T
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,               # [B*H, N, K]   (fp16)
    right_ptr,              # [B*H, N, K]   (fp16)
    out_ptr,                # [B*H, N, N]   (fp16)
    B_H, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_m, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr,  # rows of A / out per program
    BLOCK_N: tl.constexpr,  # cols of B^T / out per program
    BLOCK_K: tl.constexpr,  # reduction dimension tile
    GROUP_SIZE_M: tl.constexpr,
):
    pid = tl.program_id(0)

    # ----------- 2‑d grid of blocks (m,n) ------------------------------
    # we tile the (M,N) output space; each pid maps to a (block_m, block_n)
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # group blocks to improve L2 reuse
    group_id = pid // (GROUP_SIZE_M * num_blocks_n)
    first_pid_in_group = group_id * GROUP_SIZE_M * num_blocks_n
    pid_in_group = pid - first_pid_in_group

    block_m = (pid_in_group // num_blocks_n) * GROUP_SIZE_M + (pid_in_group % GROUP_SIZE_M)
    block_n = pid_in_group % num_blocks_n

    # if out‑of‑range we early‑exit
    if block_m >= num_blocks_m or block_n >= num_blocks_n:
        return

    # offset pointers to the start of the tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # linear index of the (batch, hidden) pair that this program works on
    batch_hidden = pid // (num_blocks_m * num_blocks_n)

    # pointers for this batch/hidden
    left_tile_ptr  = left_ptr  + batch_hidden * stride_l_batch
    right_tile_ptr = right_ptr + batch_hidden * stride_r_batch
    out_tile_ptr   = out_ptr   + batch_hidden * stride_o_batch

    # initialise accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # load A and B^T tiles (fp16 -> fp32 for accumulation)
        a = tl.load(left_tile_ptr + (offs_m[:, None] * stride_l_m
                                   + k_offset[None, :] * stride_l_k),
                    mask=(offs_m[:, None] < N) & (k_offset[None, :] < K),
                    other=0.0).to(tl.float32)

        b = tl.load(right_tile_ptr + (offs_n[:, None] * stride_r_m
                                    + k_offset[None, :] * stride_r_k),
                    mask=(offs_n[:, None] < N) & (k_offset[None, :] < K),
                    other=0.0).to(tl.float32)

        # dot‑product along K
        acc += tl.dot(a, b)

    # write result back (fp16)
    out = acc.to(tl.float16)
    tl.store(out_tile_ptr + (offs_m[:, None] * stride_o_m
                           + offs_n[None, :] * stride_o_n),
             out,
             mask=(offs_m[:, None] < N) & (offs_n[None, :] < N))


# ----------------------------------------------------------------------
# Helper: simple LayerNorm (bias + weight) – pure PyTorch
# ----------------------------------------------------------------------
def layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x = layer_norm(inp, weights["norm.weight"], weights["norm.bias"])

    # --------------------------------------------------------------
    # 2) Linear projections (dim -> hidden) – bias‑free
    # --------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Optional mask (element‑wise multiplication)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] -> [B,N,N,1] (fp32)
        mask_exp = mask.unsqueeze(-1).to(torch.float32).contiguous()
        left_masked  = torch.empty_like(left)
        right_masked = torch.empty_like(right)

        total = left.numel()
        BLOCK = 131072                     # 128 Ki elements ≈ 512 KB (fp32)
        grid = (triton.cdiv(total, BLOCK),)

        # left
        apply_mask_kernel[grid](
            left.data_ptr(),
            mask_exp.data_ptr(),
            left_masked.data_ptr(),
            total,
            BLOCK=BLOCK,
        )
        # right
        apply_mask_kernel[grid](
            right.data_ptr(),
            mask_exp.data_ptr(),
            right_masked.data_ptr(),
            total,
            BLOCK=BLOCK,
        )
        left, right = left_masked, right_masked

    # --------------------------------------------------------------
    # 4) Gating (sigmoid) – bias‑free linear + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate   # [B,N,N,H]
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Core N³ reduction using the Triton GEMM kernel
    #    We reshape to (B*H, N, N) and run a batched GEMM.
    #    fp16 is used inside the kernel for higher arithmetic throughput.
    # --------------------------------------------------------------
    # reshape & permute to (B, H, N, N) → (B*H, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # cast to fp16 for the kernel
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # output buffer (fp16)
    out_f16 = torch.empty_like(left_f16)

    # launch parameters (tuned for H200, N ≤ 1024, hidden ≤ 384)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    GROUP_SIZE_M = 8

    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)
    total_programs = (B * hidden) * num_blocks_m * num_blocks_n

    grid = (total_programs,)

    tri_mul_gemm_kernel[grid](
        left_f16.data_ptr(),
        right_f16.data_ptr(),
        out_f16.data_ptr(),
        B * hidden, N, N,
        left_f16.stride(0), left_f16.stride(1), left_f16.stride(2),
        right_f16.stride(0), right_f16.stride(1), right_f16.stride(2),
        out_f16.stride(0), out_f16.stride(1), out_f16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=GROUP_SIZE_M,
    )

    # reshape back to [B,N,N,hidden] and cast to fp32
    out = out_f16.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear
    # --------------------------------------------------------------
    out = layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
