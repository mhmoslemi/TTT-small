"""
TriMul “outgoing” forward – mixed‑precision Triton implementation
=================================================================

The heavy part of the TriMul forward pass is the 3‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication for every head *d* :

    left  : (B, H, I, K)   = (batch, hidden_dim, seq_len, seq_len)
    right : (B, H, K, J)

For each (b,h) we need an I×K × K×J → I×J product.  The kernel below
computes many of those products in a single launch using tiling in the
M, N and K dimensions (BLOCK_M, BLOCK_N, BLOCK_K).  Accumulation is done
in FP32 for numeric stability, while the inner dot‑product uses FP16
(Triton automatically casts to the larger dtype when required).

Only the LayerNorms, linear projections and gating are kept in PyTorch –
they are cheap compared to the O(N³) work of the einsum.  The kernel
operates on FP16 tensors, the result is cast back to FP32 and the final
LayerNorm / linear projection are applied.

The implementation works for both “nomask” and masked inputs, arbitrary
batch size (1‑2), sequence length up to 1024 and hidden dimensions up to
384.  All strides are passed explicitly, so the kernel is agnostic to the
contiguity layout of the input tensors.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    a_ptr,                # left  matrix   (batch_head, I, K)
    b_ptr,                # right matrix   (batch_head, K, J)
    c_ptr,                # output matrix  (batch_head, I, J)
    M, N, K,              # shapes
    stride_az, stride_am, stride_ak,   # a strides (z,i,k)
    stride_bz, stride_bk, stride_bn,   # b strides (z,k,j)
    stride_cz, stride_cm, stride_cn,   # c strides (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 3‑D program ids
    pid_m = tl.program_id(0)               # tile along M (I)
    pid_n = tl.program_id(1)               # tile along N (J)
    pid_z = tl.program_id(2)               # batch × head dimension

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # a: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # b: (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # accumulate (FP32)
        acc += tl.dot(a, b)

    # write FP16 result (later cast to FP32 in Python)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def triton_batch_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiply for the TriMul core.

    Arguments
    ----------
    left : torch.Tensor  (B, H, I, K)   fp16, contiguous
    right: torch.Tensor  (B, H, K, J)   fp16, contiguous

    Returns
    -------
    torch.Tensor of shape (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten (B,H) → Z for the kernel
    Z = B * H
    a = left.view(Z, I, K)
    b = right.view(Z, K, J)
    c = out.view(Z, I, J)

    # strides in elems
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tiling – values tuned for Hopper (SM90) but work for all target sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),
        triton.cdiv(J, BLOCK_N),
        Z,
    )
    batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,            # good default for fp16 on H200
        num_stages=2,           # fits within shared‑memory budget
    )
    return out


# ----------------------------------------------------------------------
# Full forward – mixed precision, only the O(N³) part in Triton
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim]   (float32)
        mask         : torch.Tensor, shape [B, N, N]       (float32/bool or None)
        weights      : dict of leaf tensors (LayerNorm, Linear weights, etc.)
        config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor  shape [B, N, N, dim]   (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # LayerNorm  (manual implementation to avoid an extra nn.Module)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * norm_w + norm_b                         # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections (float32 -> fp16)
    # ------------------------------------------------------------------
    # weight shapes are (hidden_dim, dim)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    x_fp16 = x.to(torch.float16)                    # (B,N,N,dim) fp16

    left = torch.nn.functional.linear(x_fp16, left_proj_w)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x_fp16, right_proj_w) # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask is broadcastable to [B,N,N,1]
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched matmul:
    #   left  : (B, H, I, K)   -> permute to (B, H, I, K)
    #   right : (B, H, J, K)   -> permute to (B, H, K, J)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, J)

    # ------------------------------------------------------------------
    # Triton‑accelerated batched matmul (I×K)*(K×J) -> (I×J) for each head
    # ------------------------------------------------------------------
    out_mat = triton_batch_matmul(left_perm, right_perm)   # (B, H, I, J) fp16

    # bring back to original layout and cast to fp32
    out = out_mat.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # Output LayerNorm
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # Final gating and projection
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]
    out = out * out_gate.to(torch.float32)                # gating
    out = torch.nn.functional.linear(out, to_out_w)       # (B,N,N,dim) fp32

    return out
