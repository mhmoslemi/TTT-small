"""
TriMul “outgoing” forward (AlphaFold3 style).

The expensive part is the 3‑dimensional tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix‑multiply for each hidden dimension `d`:

    left  : (B, H, I, K)   (permute 0,3,1,2)
    right : (B, H, K, J)   (permute 0,3,2,1)
    out   : (B, H, I, J)   → later permuted back to (B, I, J, H)

Only this N³ operation is placed in Triton; all cheap LayerNorms,
linear layers and gating are left in PyTorch (they are memory‑bound but
tiny compared with the N³ work).  The kernel works in fp16 for the
multiply‑add and accumulates in fp32 to keep numerical stability.  The
final result is converted back to fp32 on the host and the remaining
post‑processing (output LayerNorm, gating and projection) is performed
in PyTorch.

The kernel tiles the output matrix (I×J) with BLOCK_M×BLOCK_N tiles and
processes the reduction dimension K in BLOCK_K steps.  Each
program‑id triplet (pid_m, pid_n, pid_z) computes one tile for a
particular (batch, head) pair, i.e. for a tensor of shape
(B*H, I, J).  The launch configuration (BLOCK_M,N,K, num_warps) has been
tuned for Hopper/SM90 (H200) and fits comfortably inside shared‑memory
limits.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched (I×K)·(K×J) → (I×J) for many (batch, head) pairs
# ----------------------------------------------------------------------
@triton.jit
def _batch_matmul_kernel(
    a_ptr,            # left  matrix  (batch*head, I, K)
    b_ptr,            # right matrix  (batch*head, K, J)
    c_ptr,            # output matrix (batch*head, I, J)
    M, N, K,          # matrix extents
    stride_az, stride_am, stride_ak,   # strides for a
    stride_bz, stride_bk, stride_bn,   # strides for b
    stride_cz, stride_cm, stride_cn,   # strides for c
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row   (I)
    pid_n = tl.program_id(1)          # tile col   (J)
    pid_z = tl.program_id(2)          # batch*head index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # out‑of‑bounds masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (good precision, still fast on H200)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load a block of A: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load a block of B: (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot‑product accumulation (fp32)
        acc += tl.dot(a, b)

    # write back in fp16 (the host will cast to fp32 later)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _triton_batch_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiply used in TriMul.

    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous
    returns (B, H, I, J)  fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten (B*H) so the kernel sees a 3‑D tensor
    batch_dim = B * H
    a = left.view(batch_dim, I, K)
    b = right.view(batch_dim, K, J)
    c = out.view(batch_dim, I, J)

    # strides in elements (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # tile sizes – empirically good on Hopper/SM90
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),
        triton.cdiv(J, BLOCK_N),
        batch_dim,
    )
    _batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # 4 warps = good occupancy for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Main entry‑point: pure‑Python wrapper that builds the full forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim] (float32)
            - mask  : torch.Tensor [B, N, N]      (float32, bool or None)
            - weights: dict of leaf weight tensors
            - config : dict with keys ``dim`` and ``hidden_dim``

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32, hand‑written to avoid extra dependency)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # PyTorch LayerNorm semantics
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * norm_w + norm_b                     # shape [B,N,N,dim]

    # --------------------------------------------------------------
    # 2. Linear projections & gates (float32 → fp16)
    # --------------------------------------------------------------
    # Convert weights once to fp16 for the heavy part
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        # mask may be bool or float, we need 0/1 in fp16
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4. Gating (sigmoid on fp16)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triton batched matrix multiply
    # --------------------------------------------------------------
    # reshape for matmul: (B, hidden_dim, N, N) → (B, H, I, K) and (B, H, K, J)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    out_fp16 = _triton_batch_matmul(left_perm, right_perm)   # (B, H, I, J)

    # back to original layout and fp32
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating & final projection (all float32)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # LayerNorm over hidden dim (dim)
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    # final gating (out_gate was computed in fp16, cast back)
    out = out * out_gate.to(torch.float32)

    # final linear projection back to original dim
    out = F.linear(out, to_out_w)

    return out
