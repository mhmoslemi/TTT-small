"""
TriMul “outgoing” kernel (forward only).

The heavy N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply per hidden dimension d.
We keep the LayerNorm / Linear / gating stages in PyTorch (they are
memory‑bound and already fused by torch.compile) and move the N³
reduction to a custom Triton kernel.  The kernel processes one
(batch, hidden) pair at a time, tiled with BLOCK_M/N/K = 64 and
uses tl.dot for the inner‑product.  The launch grid is
   (B*hidden, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
so every program instance writes a (BLOCK_M × BLOCK_N) tile of the
output matrix for a given hidden channel.

The rest of the pipeline (mask application, final LayerNorm,
output projection) stays in PyTorch, which on the H200 is
memory‑bound and already very fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the N³ reduction
# ----------------------------------------------------------------------
@triton.jit
def trmul_outgoing_kernel(
    left_ptr,          # [B, N, N, H]  (contiguous)
    right_ptr,         # [B, N, N, H]  (contiguous)
    out_ptr,           # [B, N, N, H]  (contiguous)
    B, N, H,           # static sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_batch_hidden = tl.program_id(0)      # indexes (b, h) flattened
    pid_m = tl.program_id(1)                 # tile row
    pid_n = tl.program_id(2)                 # tile col

    # recover batch and hidden channel
    b = pid_batch_hidden // H
    h = pid_batch_hidden % H

    # start offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out‑of‑bounds (N is at most 1024 → fits in 16‑bit)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the first element of the tile (i.e. k = 0)
    left_tile_ptr  = left_ptr  + b * stride_left_b  \
                                 + h * stride_left_h  \
                                 + offs_m[:, None] * stride_left_i \
                                 + 0 * stride_left_k
    right_tile_ptr = right_ptr + b * stride_right_b \
                                 + h * stride_right_h \
                                 + offs_n[None, :] * stride_right_i \
                                 + 0 * stride_right_k

    # accumulator in fp32 (even if inputs are fp16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in tiles
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        # compute real k offsets for this iteration
        k_offs = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offs < N

        # load a tile from left: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_tile_ptr + k_offs[None, :] * stride_left_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load a tile from right: shape (BLOCK_N, BLOCK_K) → need transpose
        right = tl.load(
            right_tile_ptr + k_offs[None, :] * stride_right_k,
            mask=mask_n[None, :] & mask_k[:, None],
            other=0.0,
        )
        # right currently (BLOCK_N, BLOCK_K), we want (BLOCK_K, BLOCK_N)
        right = tl.trans(right)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(left, right)

        # advance pointers to next k‑tile
        left_tile_ptr  += BLOCK_K * stride_left_k
        right_tile_ptr += BLOCK_K * stride_right_k

    # write result
    out_tile_ptr = out_ptr + b * stride_out_b \
                              + h * stride_out_h \
                              + offs_m[:, None] * stride_out_i \
                              + offs_n[None, :] * stride_out_j
    tl.store(
        out_tile_ptr,
        acc.to(tl.float32),                     # keep fp32 for stability
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper: fused LayerNorm (no bias in the reference implementation)
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, w: torch.Tensor, b: torch.Tensor, eps: float = 1e-5):
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * w + b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,               # input tensor [B,N,N,C]
        torch.Tensor,               # mask tensor [B,N,N]
        Dict[str, torch.Tensor],    # weights
        Dict[str, Any],             # config
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, C = inp.shape
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = _layer_norm(inp, weights["norm.weight"], weights["norm.bias"])

    # --------------------------------------------------------------
    # 2. Linear projections (dim -> hidden)
    # --------------------------------------------------------------
    # shape: [B,N,N,hidden]
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcasted over hidden)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N,1] → broadcast in the hidden dim
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left  = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate   # [B,N,N,hidden]
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core N³ reduction using Triton
    # --------------------------------------------------------------
    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # strides (in elements) for the 4‑D tensors
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_i, sR_k, sR_h = right.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    # tile sizes – 64 works well for N ≤ 1024 on Hopper
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden,                         # batch‑hidden dimension
        triton.cdiv(N, BLOCK_M),            # tile rows
        triton.cdiv(N, BLOCK_N),            # tile cols
    )

    trmul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden,
        sL_b, sL_i, sL_k, sL_h,
        sR_b, sR_i, sR_k, sR_h,
        sO_b, sO_i, sO_j, sO_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gate, final linear
    # --------------------------------------------------------------
    out = _layer_norm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
