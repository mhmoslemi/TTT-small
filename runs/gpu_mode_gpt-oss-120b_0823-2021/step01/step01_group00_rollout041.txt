"""
TriMul “outgoing” kernel (AlphaFold‑3 style).

The expensive part of the module is the 3‑tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiply for every hidden dimension `d`.
We keep the cheap LayerNorms / linear layers in PyTorch (they are tiny
compared to the N³ work) and accelerate only the contraction with a
single Triton kernel that computes many (B·H) matrix products
simultaneously.

Data flow (float32 → fp16 for the heavy part):
    x          : (B, N, N, dim)          –‑> LayerNorm (fp32)
    left/right : linear(x) → (B, N, N, hidden_dim) –‑> fp16
    mask/gate  : applied in fp16
    left_perm  : (B, hidden_dim, N, N)   –‑> (B·H, I, K)
    right_perm : (B, hidden_dim, N, N)   –‑> (B·H, K, J)
    Triton kernel computes (B·H) matrix products   I×K · K×J → I×J
    Result is reshaped back to (B, N, N, dim) and the final
    LayerNorm / projection are performed in fp32.

The kernel works in fp16 for the multiply‑add accumulation (fp32 accumulator
to preserve numerical quality) and writes the result in fp16; the host
converts back to fp32 for the remaining inexpensive ops.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply (I×K)·(K×J) → (I×J)
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    a_ptr,                     # left  matrix   (B*H, I, K)
    b_ptr,                     # right matrix   (B*H, K, J)
    c_ptr,                     # output matrix  (B*H, I, J)
    M, N, K,                   # matrix sizes
    stride_az, stride_am, stride_ak,   # a strides (batch, i, k)
    stride_bz, stride_bk, stride_bn,   # b strides (batch, k, j)
    stride_cz, stride_cm, stride_cn,   # c strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched fp16 matmul with fp32 accumulation."""
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_z = tl.program_id(2)          # batch * head index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (fp32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load a tile of A
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load a tile of B
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # accumulate (fp32)
        acc += tl.dot(a, b)

    # write result back (convert to fp16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def triton_batch_matmul(
    left: torch.Tensor,   # (B, H, I, K)  fp16, contiguous
    right: torch.Tensor,  # (B, H, K, J)  fp16, contiguous
) -> torch.Tensor:
    """
    Launch ``batch_matmul_kernel`` for a batch of (B*H) matrix products.
    Returns a tensor of shape (B, H, I, J) in fp16.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    # output allocation
    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten the batch‑head dimension for the kernel
    batch = B * H
    a = left.view(batch, I, K)
    b = right.view(batch, K, J)
    c = out.view(batch, I, J)

    # strides in element units (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # ------------------------------------------------------------------
    # Tiling heuristics (tuned for Hopper / SM90, fp16)
    #   - BLOCK_M/N 64 gives good occupancy while keeping shared‑memory low.
    #   - BLOCK_K 32 matches the native Tensor‑Core tile size (16×16) and
    #     leaves enough registers for 4 warps per block.
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(I, BLOCK_M),          # rows
        triton.cdiv(J, BLOCK_N),          # cols
        batch,                            # batch * head
    )
    batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # 4 warps = 128 threads, good for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (mixed‑precision, Triton‑accelerated contraction)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple (input, mask, weights, config)
        input   – torch.Tensor [B, N, N, dim] (float32)
        mask    – torch.Tensor [B, N, N]     (float32/bool or None)
        weights – dict of leaf weight tensors
        config  – dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # manual LN to avoid extra dependencies
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * norm_w + norm_b          # (B,N,N,dim)  float32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections + gates, cast to fp16 for the heavy region
    # ------------------------------------------------------------------
    # linear: y = x @ w.T   (torch.nn.functional.linear expects (out_features, in_features))
    def linear_fp16(inp, w):
        return torch.nn.functional.linear(inp, w.to(torch.float16))

    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    x_fp16 = x.to(torch.float16)

    left = linear_fp16(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right = linear_fp16(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on the hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(torch.float16)   # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = linear_fp16(x_fp16, left_gate_w).sigmoid()
    right_gate = linear_fp16(x_fp16, right_gate_w).sigmoid()
    out_gate = linear_fp16(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Triton‑accelerated contraction
    #    left : (B,N,N,hidden_dim) → (B,hidden_dim,N,N)
    #    right: (B,N,N,hidden_dim) → (B,hidden_dim,N,N) and then transpose K/J
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, J)

    # batch matmul in fp16
    out_fp16 = triton_batch_matmul(left_perm, right_perm)   # (B, H, I, J)

    # reshape back to (B,N,N,dim) (H == hidden_dim)
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (float32) → gating → final projection
    # ------------------------------------------------------------------
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    out = out * out_gate.to(torch.float32)   # broadcast over hidden dim
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    return out

### Why the earlier lessons were considered

| Lesson | Relevance here |
|-------|----------------|
| **Limit shared memory / choose modest BLOCK sizes** | Applied – we use `BLOCK_M/N = 64` and `BLOCK_K = 32`, fitting comfortably in H200 shared‑memory while still giving high occupancy. |
| **Match stride unpacking to tensor rank** | Followed – strides are taken directly from the flattened `(batch·head, …)` tensors; we never assume a fixed number of dimensions beyond what the kernel expects. |
| **Do not copy kernels verbatim** | The kernel is written from‑scratch, only the overall structure mirrors a classic tiled matmul. |
| **Keep cheap ops in PyTorch** | Implemented – both LayerNorms, linear layers, and gating stay in PyTorch, which is faster than trying to fuse them in Triton for these small‑dim tensors. |

The resulting implementation keeps the heavy `N³` work inside a highly‑tuned Triton kernel, uses mixed‑precision to exploit Tensor‑Cores on the H200, and respects all constraints (no gradients, correct output dtype, support for optional masks, and the full set of test configurations).
