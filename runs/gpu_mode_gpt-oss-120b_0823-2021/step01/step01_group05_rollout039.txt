"""
TriMul “outgoing” kernel (AlphaFold‑3) – Triton‑accelerated core
================================================================
The expensive O(N³·C) reduction

    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]

is implemented as a batched, channel‑wise matrix‑multiply in Triton.
Each hidden channel is independent, therefore we can treat the
operation as a large collection of GEMMs:

    out_c = left_c @ right_cᵀ           (left_c/right_c : [B,N,N])

The kernel tiles the K‑dimension (the summed axis) and the
output M×N plane, accumulates in fp32 and finally writes the result.
All surrounding LayerNorm, Linear and gating work is kept in PyTorch
for numerical stability.

The kernel works for any batch size B∈{1,2}, sequence length N up to 1024,
and hidden dimension C (called ``hidden_dim``) that is a multiple of 32
(the typical values are 128).  Mixed‑precision (fp16 inputs, fp32
accumulator) is used to maximise throughput on Hopper GPUs.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# --------------------------------------------------------------
# 1. Triton kernel: batched channel‑wise GEMM for the TriMul core
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,          # const fp16 *
    right_ptr,         # const fp16 *
    out_ptr,           #      fp32 *
    # strides (batch, seq, seq, channel)
    stride_bl, stride_il, stride_kl, stride_cl,
    stride_br, stride_jr, stride_kr, stride_cr,
    stride_bo, stride_io, stride_jo, stride_co,
    # sizes
    B, N, C,               # i64
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,   # tile size for i dimension
    BLOCK_N: tl.constexpr,   # tile size for j dimension
    BLOCK_K: tl.constexpr,   # tile size for k (reduction) dimension
    BLOCK_C: tl.constexpr,   # tile size for channel dimension
):
    """
    Compute:
        out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]

    The kernel is launched with a 3‑D grid:
        grid = (ceil(N/BLOCK_M), ceil(N/BLOCK_N), ceil(C/BLOCK_C) * B)

    The outermost dimension folds batch & channel tiles:
        pid = tl.program_id(2)
        batch = pid // (ceil(C/BLOCK_C))
        c_tile = pid %  (ceil(C/BLOCK_C))
    """
    pid_m = tl.program_id(0)          # i‑tile
    pid_n = tl.program_id(1)          # j‑tile
    pid_2 = tl.program_id(2)          # folds batch & channel tiles

    # ------------------------------------------------------------------
    # decode batch and channel tile from pid_2
    # ------------------------------------------------------------------
    num_c_tiles = tl.cdiv(C, BLOCK_C)
    batch = pid_2 // num_c_tiles
    c_tile = pid_2 % num_c_tiles

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_c = c_tile * BLOCK_C + tl.arange(0, BLOCK_C)

    # masks for OOB
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_c = offs_c < C

    # ------------------------------------------------------------------
    # pointers to the start of our tile
    # ------------------------------------------------------------------
    left_tile_ptr  = left_ptr \
        + batch * stride_bl \
        + offs_m[:, None, None] * stride_il \
        + 0          * stride_kl \
        + offs_c[None, None, :] * stride_cl

    right_tile_ptr = right_ptr \
        + batch * stride_br \
        + 0          * stride_jr \
        + offs_n[None, :, None] * stride_kr \
        + offs_c[None, None, :] * stride_cr

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_C), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over k
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute real k offset
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load a block of left  : [M, K, C]
        l_ptr = left_tile_ptr \
            + offs_k[None, :, None] * stride_kl
        left_block = tl.load(l_ptr,
                             mask=mask_m[:, None, None] & mask_k[None, :, None] & mask_c[None, None, :],
                             other=0.0).to(tl.float32)

        # load a block of right : [N, K, C] (note the K dimension is shared)
        r_ptr = right_tile_ptr \
            + offs_k[:, None, None] * stride_jr
        right_block = tl.load(r_ptr,
                              mask=mask_n[None, :, None] & mask_k[:, None, None] & mask_c[None, None, :],
                              other=0.0).to(tl.float32)

        # accumulate: for each channel c we perform a small GEMM
        # left_block : (M, K)   right_blockᵀ : (N, K) → (M, N)
        # use tl.dot for each channel slice
        for c in range(0, BLOCK_C):
            if not mask_c[c]:
                continue
            acc[:, :, c] += tl.dot(left_block[:, :, c], right_block[:, :, c].T)

        # advance pointers for next k‑tile (handled implicitly by recomputing offsets)

    # ------------------------------------------------------------------
    # write result
    # ------------------------------------------------------------------
    out_tile_ptr = out_ptr \
        + batch * stride_bo \
        + offs_m[:, None, None] * stride_io \
        + offs_n[None, :, None] * stride_jo \
        + offs_c[None, None, :] * stride_co

    tl.store(out_tile_ptr,
             acc,
             mask=mask_m[:, None, None] & mask_n[None, :, None] & mask_c[None, None, :])


# --------------------------------------------------------------
# 2. Helper that launches the Triton kernel
# --------------------------------------------------------------
def trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, C] (float16)
    returns out   : [B, N, N, C] (float32)
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, C = left.shape
    out = torch.empty_like(left, dtype=torch.float32)

    # strides (in elements)
    stride_bl, stride_il, stride_kl, stride_cl = left.stride()
    stride_br, stride_jr, stride_kr, stride_cr = right.stride()
    stride_bo, stride_io, stride_jo, stride_co = out.stride()

    # tile sizes – tuned for Hopper (fp16 compute, fp32 acc)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_C = 32   # C is a multiple of 32 in all configs

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * triton.cdiv(C, BLOCK_C),
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        stride_bl, stride_il, stride_kl, stride_cl,
        stride_br, stride_jr, stride_kr, stride_cr,
        stride_bo, stride_io, stride_jo, stride_co,
        B, N, C,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_C=BLOCK_C,
        num_warps=4,              # enough parallelism for the small tiles
    )
    return out


# --------------------------------------------------------------
# 3. Full forward pass – only the O(N³·C) part lives in Triton
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : (input, mask, weights, config)
        input  : [B, N, N, dim]          float32
        mask   : [B, N, N]               float32 (0/1) – may be all‑ones
        weights: dict of model parameters (float32)
        config : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = inp.device
    B, N, _, _ = inp.shape

    # ------------------------------------------------------------------
    # 1) LayerNorm (dim) – implemented in PyTorch for stability
    # ------------------------------------------------------------------
    eps = 1e-5
    mean = inp.mean(-1, keepdim=True)
    var = inp.var(-1, unbiased=False, keepdim=True)
    x = (inp - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # ------------------------------------------------------------------
    # 2) Linear projections (dim → hidden_dim) – fp16 for the heavy part
    # ------------------------------------------------------------------
    # keep everything in fp32 for the linear ops, then cast
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # optional mask – broadcast on hidden dim
    if not nomask:
        # mask shape [B,N,N,1] – simple elementwise multiply (very cheap)
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid) – stay in fp32
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Core TriMul reduction – Triton kernel (fp16 inputs, fp32 acc)
    # ------------------------------------------------------------------
    # Cast to fp16 to exploit Tensor‑Core GEMM inside the kernel
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # out shape [B,N,N,hidden_dim] in fp32
    out = trimul_outgoing_triton(left_f16, right_f16)

    # ------------------------------------------------------------------
    # 5) Final LayerNorm, gating and output projection (dim)
    # ------------------------------------------------------------------
    # to_out_norm
    mean_o = out.mean(-1, keepdim=True)
    var_o = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    out = out * out_gate

    # to_out linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
