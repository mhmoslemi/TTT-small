"""
TriMul “outgoing” implementation (forward only).

The heavy N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply per hidden dimension `d`.  We reshape the
tensors to (B*H, N, N) and invoke `torch.bmm`, which on the H200 is
implemented by cuBLAS and far faster than a hand‑written Triton reduction.

The only part we move to Triton is the element‑wise mask application
(required when `nomask=False`).  The kernel multiplies a tensor of shape
[B, N, N, H] by a broadcasted mask of shape [B, N, N, 1] using a 1‑D
grid.  Strides are passed explicitly so the kernel works for any
contiguous or strided layout.

All other operations (LayerNorm, Linear, gating, final LayerNorm and
projection) are performed in PyTorch – they are memory‑bound and already
fused by `torch.compile`.  The final output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise mask multiplication (supports nomask=False)
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    x_ptr,          # ptr to tensor [B,N,N,H]
    mask_ptr,       # ptr to mask   [B,N,N,1] (broadcast along H)
    out_ptr,        # ptr to output tensor [B,N,N,H]
    total_elems,    # B*N*N*H  (int)
    stride_mask_B,  # mask stride in B dimension (in elements)
    stride_mask_N,  # mask stride in N dimension (in elements)
    stride_x_B, stride_x_N1, stride_x_N2, stride_x_H,  # strides of x (in elements)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # load the data element
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)

    # compute the corresponding mask offset.
    # The mask has no H dimension, so we drop the lowest stride (H).
    #   idx = offs // stride_x_H   gives linear index in the [B,N,N] tensor.
    mask_offs = offs // stride_x_H
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    tl.store(out_ptr + offs, x * m, mask=mask)


# ----------------------------------------------------------------------
# Helper: fused LayerNorm (bias & weight supplied)
# ----------------------------------------------------------------------
def fused_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """
    LayerNorm over the last dimension.  Operates in float32.
    """
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim]  (float32)
            - mask_tensor  : [B, N, N] (bool/float) – may be omitted when nomask=True
            - weights_dict : model parameters (see docstring above)
            - config_dict  : {"dim":int, "hidden_dim":int, "nomask":bool, ...}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim -> dim)
    # ------------------------------------------------------------------
    x = fused_layernorm(
        inp,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (dim -> hidden), bias‑free
    # ------------------------------------------------------------------
    # shape: [B, N, N, hidden]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask application (mask shape [B,N,N])
    # ------------------------------------------------------------------
    if not nomask:
        # Expand mask to [B,N,N,1] for broadcasting
        mask_exp = mask.unsqueeze(-1).contiguous()
        total = left.numel()
        BLOCK = 131072  # 128 Ki elements ≈ 512 KB (FP32)
        grid = (triton.cdiv(total, BLOCK),)

        # allocate temporaries
        left_masked = torch.empty_like(left)
        right_masked = torch.empty_like(right)

        # launch kernel for left
        apply_mask_kernel[grid](
            left.data_ptr(),
            mask_exp.data_ptr(),
            left_masked.data_ptr(),
            total,
            mask_exp.stride(0),          # mask B stride
            mask_exp.stride(1),          # mask N stride (same for both N dims)
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            BLOCK=BLOCK,
        )
        # launch kernel for right
        apply_mask_kernel[grid](
            right.data_ptr(),
            mask_exp.data_ptr(),
            right_masked.data_ptr(),
            total,
            mask_exp.stride(0),
            mask_exp.stride(1),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            BLOCK=BLOCK,
        )
        left, right = left_masked, right_masked

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – same input `x`
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ reduction using batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N)
    left_rs = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_rs = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N) # (B*H, N, N)

    # batched matmul: (B*H, N, N) @ (B*H, N, N)^T → (B*H, N, N)
    out_rs = torch.bmm(left_rs, right_rs.transpose(1, 2))

    # back to [B, N, N, hidden]
    out = out_rs.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden -> hidden) and final linear projection
    # ------------------------------------------------------------------
    out = fused_layernorm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # ensure float32 output
    return out.to(torch.float32)
