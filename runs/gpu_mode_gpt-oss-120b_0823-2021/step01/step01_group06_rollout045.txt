"""
custom_kernel
==============

Implementation of the *outgoing* TriMul operator (AlphaFold‑3) with a
hand‑written Triton kernel for the element‑wise mask‑application.
All LayerNorms, linear projections, gating and the N³ reduction are kept
in PyTorch because on the H200 a single‑precision batched GEMM
(`torch.bmm`) is already heavily optimized and far faster than a
pure‑Triton reduction for the supported sizes (N ≤ 1024, hidden ≤ 384).

The Triton kernel does one useful amount of work that cannot be fused
by `torch.compile` – it multiplies a 4‑D tensor `[B,N,N,H]` with a
broadcasted mask `[B,N,N,1]` in a single pass, saving an extra memory
read/write pass that would otherwise be performed in Python.  The rest
of the forward pass follows the reference PyTorch implementation
exactly.

The function signature matches the specification:

    custom_kernel(data) -> torch.Tensor

where `data` is a tuple
    (input_tensor, mask_tensor, weight_dict, config_dict)

and the returned tensor has shape `[B, N, N, dim]` (float‑32).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: apply a binary mask (broadcasted on the hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    src_ptr,                 # pointer to source tensor  [B,N,N,H]
    mask_ptr,                # pointer to mask tensor    [B,N,N,1]
    dst_ptr,                 # pointer to destination   [B,N,N,H]
    total_elements,          # B*N*N*H  (number of elements to process)
    src_stride_b, src_stride_n1, src_stride_n2, src_stride_h,   # src strides
    mask_stride_b, mask_stride_n1, mask_stride_n2,              # mask strides (no H stride)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Linear offset → multi‑dimensional indices
    #   idx = ((b * N + i) * N + j) * H + h
    # We recover b,i,j,h using integer division / modulo.
    # All strides are in elements (not bytes) because we operate on fp32.
    h = offs % src_stride_h
    tmp = offs // src_stride_h
    j = tmp % src_stride_n2
    tmp = tmp // src_stride_n2
    i = tmp % src_stride_n1
    b = tmp // src_stride_n1

    # compute mask linear offset (mask has no H dimension)
    mask_off = (b * mask_stride_b) + (i * mask_stride_n1) + (j * mask_stride_n2)

    src = tl.load(src_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)   # 1.0 = keep, 0.0 = drop
    tl.store(dst_ptr + offs, src * m, mask=mask)


def _apply_mask(src: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Multiply ``src`` of shape [B,N,N,H] with ``mask`` of shape [B,N,N]
    (broadcasted on the hidden dimension) using a Triton kernel.
    """
    assert src.is_contiguous()
    assert mask.is_contiguous()
    B, N, _, H = src.shape
    total = src.numel()
    BLOCK = 131072                     # 128 Ki elements ≈ 512 KiB (fp32)

    grid = (triton.cdiv(total, BLOCK),)

    dst = torch.empty_like(src)

    _apply_mask_kernel[grid](
        src.data_ptr(),
        mask.data_ptr(),
        dst.data_ptr(),
        total,
        src.stride(0), src.stride(1), src.stride(2), src.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        BLOCK=BLOCK,
    )
    return dst


def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """
    Fused LayerNorm (no bias inside the norm – bias added afterwards).
    Operates in fp32 and returns the same dtype as ``x``.
    """
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input, mask, weights, config)
        - input : torch.Tensor, shape [B, N, N, dim]
        - mask  : torch.Tensor, shape [B, N, N]   (may be all‑ones when ``nomask=True``)
        - weights : dict of tensors as described in the prompt
        - config  : dict containing ``dim``, ``hidden_dim`` and ``nomask`` flag

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0. unpack
    # ------------------------------------------------------------------
    inp, mask, w, cfg = data
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = _layer_norm(inp,
                    weight=w["norm.weight"],
                    bias=w["norm.bias"])

    # ------------------------------------------------------------------
    # 2. Linear projections (dim → hidden) – bias‑free
    # ------------------------------------------------------------------
    # shape: [B,N,N,hidden]
    left  = torch.nn.functional.linear(x, w["left_proj.weight"])
    right = torch.nn.functional.linear(x, w["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask application (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # ``mask`` is [B,N,N]; we keep it as‑is because the kernel
        # treats it as a flat tensor and broadcasts internally.
        left  = _apply_mask(left,  mask)
        right = _apply_mask(right, mask)

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – also bias‑free linear layers
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, w["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, w["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, w["out_gate.weight"]))

    left  = left  * left_gate   # [B,N,N,hidden]
    right = right * right_gate  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 5. Core N³ reduction:
    #    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
    #    Implemented as a batched matrix‑multiply per hidden dimension.
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N) for each operand
    left_rs  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_rs = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)  # (B*H, N, N)

    # batched GEMM: (B*H, N, N) @ (B*H, N, N)^T → (B*H, N, N)
    out_rs = torch.bmm(left_rs, right_rs.transpose(1, 2))

    # back to [B,N,N,hidden]
    out = out_rs.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = _layer_norm(out,
                      weight=w["to_out_norm.weight"],
                      bias=w["to_out_norm.bias"])
    out = out * out_gate
    out = torch.nn.functional.linear(out, w["to_out.weight"])

    # final dtype must be float32
    return out.to(torch.float32)
