"""
TriMul “outgoing” forward pass (AlphaFold‑3 style).

The heavy N³ einsum is rewritten as a batched matrix‑multiply
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
For each head *d* we need a matrix product (I×K)·(K×J) → (I×J).
A single Triton kernel computes many such products in one launch,
working in fp16 for the inner MACs and accumulating in fp32.
All cheap operations (LayerNorm, linear projections, gating, final
LayerNorm / projection) stay in PyTorch.  The kernel is written for
Hopper (SM‑90) and tuned with modest block sizes that keep shared‑memory
footprint low while still providing high occupancy.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched (I×K)·(K×J) → (I×J) for many (batch,head) pairs
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr,            # left  matrix,  shape (B*H, I, K)
    b_ptr,            # right matrix,  shape (B*H, K, J)
    c_ptr,            # output matrix, shape (B*H, I, J)
    M, N, K,          # matrix sizes
    stride_az, stride_am, stride_ak,   # strides for a (z,i,k)
    stride_bz, stride_bk, stride_bn,   # strides for b (z,k,j)
    stride_cz, stride_cm, stride_cn,   # strides for c (z,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)      # batch*head index
    pid_m = tl.program_id(1)      # tile along M (rows)
    pid_n = tl.program_id(2)      # tile along N (cols)

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (good numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K in blocks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load a tile of A
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load a tile of B
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot‑product accumulation
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back (convert to fp16 for storage, later cast to fp32 in PyTorch)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left  : (B, H, I, K)   fp16, contiguous
    right : (B, H, K, J)   fp16, contiguous
    returns: (B, H, I, J) fp16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, H, I, K = left.shape
    _, _, K2, J = right.shape
    assert K == K2

    out = torch.empty((B, H, I, J), device=left.device, dtype=torch.float16)

    # flatten the leading batch*head dimension for the kernel
    a = left.view(B * H, I, K)
    b = right.view(B * H, K, J)
    c = out.view(B * H, I, J)

    # strides in elements (not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # ------------------------------------------------------------------------------
    # Tile sizes – these work well on Hopper (SM90) for the dimensions we encounter.
    # They keep shared memory < 64 KB per block and give ~4 warps per block.
    # -------------------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                     # pid_z
        triton.cdiv(I, BLOCK_M),   # pid_m
        triton.cdiv(J, BLOCK_N),   # pid_n
    )
    _batched_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=I,
        N=J,
        K=K,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for fp16 on H200
    )
    return out


# ----------------------------------------------------------------------
# Full forward (pure PyTorch + Triton acceleration)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data : tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (float32/bool) or None
            - weights      : dict of leaf tensors (see docstring above)
            - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the incoming 4‑D tensor (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # manual LN (stateless, cheap)
    mean = input_tensor.mean(-1, keepdim=True)
    var  = input_tensor.var(-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + 1e-5)
    x = x * norm_weight + norm_bias                     # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections & gating (float32 → fp16)
    # ------------------------------------------------------------------
    # helper to cast weight once
    def _to_fp16(t): return t.to(torch.float16)

    left_proj_w  = _to_fp16(weights["left_proj.weight"])
    right_proj_w = _to_fp16(weights["right_proj.weight"])
    left_gate_w  = _to_fp16(weights["left_gate.weight"])
    right_gate_w = _to_fp16(weights["right_gate.weight"])
    out_gate_w   = _to_fp16(weights["out_gate.weight"])
    to_out_norm_w = weights["to_out_norm.weight"]       # stay fp32 (tiny)
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]           # final proj stays fp32

    x_fp16 = x.to(torch.float16)

    left  = torch.nn.functional.linear(x_fp16, left_proj_w)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x_fp16, right_proj_w) # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask (broadcast over hidden)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape → (B,N,N,1) for broadcasting
        mask_ = mask.to(torch.float16).unsqueeze(-1)
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Rearrange for batched matmul: (B,hidden,N,N) → (B,hidden,I,K) & (B,hidden,K,J)
    # ------------------------------------------------------------------
    # left  : (B,N,N,hidden) → (B,hidden,N,N) → (B,hidden,I,K)
    # right : (B,N,N,hidden) → (B,hidden,N,N) → (B,hidden,K,J)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    # ------------------------------------------------------------------
    # 6️⃣ Heavy N³ part – Triton batched matmul
    # ------------------------------------------------------------------
    out_fp16 = _batched_matmul(left_perm, right_perm)  # (B, H, I, J) fp16

    # ---------------------------------------------------------------
    # 7️⃣ Back‑to original layout & final processing (float32)
    # ---------------------------------------------------------------
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # (B,N,N,dim)

    # 7a️⃣ Output LayerNorm (tiny, stays in fp32)
    mean_o = out.mean(-1, keepdim=True)
    var_o  = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_o) / torch.sqrt(var_o + 1e-5)
    out = out * to_out_norm_w + to_out_norm_b

    # 7b️⃣ Final gating & projection
    out = out * out_gate.to(torch.float32)
    out = torch.nn.functional.linear(out, to_out_w)

    return out
