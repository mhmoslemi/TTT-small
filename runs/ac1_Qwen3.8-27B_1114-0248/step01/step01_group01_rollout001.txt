```python
"""
Search for nonnegative sequences minimizing
    2*n*max(conv(a,a))/(sum(a)^2).

Approach
--------
- Normalize every candidate to sum = sqrt(n), making coefficient values O(1).
- Generate several analytic/random starts (central-binomial, reversed, mixed,
  power laws, low-frequency modulated, noisy, and resampled previous best).
- The main local optimizer is a family of linear-programming steps:
    * max-sum LP: given current a, maximize sum(g) s.t. a*g <= max(a*a)
      (with optional slack and active-constraint reductions).  This moves
      mass into under-constrained regions of the convolution.
    * Chebyshev/first-order LP: minimize the linearized maximum of
      conv(a + delta, a + delta) over active constraints, with sum(delta)=0.
- After each LP direction, do an exact line search on the original objective.
- Add small smoothed multiplicative noise when a basin stalls.
- Keep the best valid sequence found before the deadline and return it.
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


TARGET_UPPER_BOUND = 1.5027
UB_FINAL = 1000.0


def clean_array(a):
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return np.zeros(1, dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=UB_FINAL, neginf=0.0)
    arr = np.clip(arr, 0.0, UB_FINAL)
    return arr


def central_seq(n):
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (i + 0.5) / (i + 1.0)
    return a


def score_array(a):
    arr = clean_array(a)
    n = arr.size
    if n == 0:
        return float("inf")
    s = float(np.sum(arr))
    if not math.isfinite(s) or s < 0.01:
        return float("inf")
    c = np.convolve(arr, arr)
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    if not math.isfinite(m):
        return float("inf")
    val = float(2.0 * n * m / (s * s))
    if not math.isfinite(val):
        return float("inf")
    return val


def normalize_sum(a, n=None):
    arr = clean_array(a)
    if n is None:
        n = arr.size
    else:
        n = int(n)
        if arr.size != n:
            arr = np.asarray(interpolate_sequence(arr, n), dtype=np.float64).ravel()
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    s = float(np.sum(arr))
    if not math.isfinite(s) or s <= 1e-15:
        return central_seq(n).astype(np.float64)
    target = math.sqrt(float(max(1, n)))
    out = arr * (target / s)
    out = np.clip(out, 0.0, UB_FINAL)
    # Re-normalize after clipping, in clipping had any effect.
    s2 = float(np.sum(out))
    if s2 > 1e-15:
        out = out * (target / s2)
    return out.astype(np.float64)


def interpolate_sequence(a, n):
    arr = clean_array(a)
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    if arr.size == 0:
        return central_seq(n).astype(np.float64)
    if arr.size == 1:
        out = np.full(n, float(arr[0]), dtype=np.float64)
        return normalize_sum(out, n)
    old = np.linspace(0.0, 1.0, arr.size, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, arr)
    out = np.maximum(out, 0.0)
    if float(np.sum(out)) <= 1e-15:
        return central_seq(n).astype(np.float64)
    return normalize_sum(out, n)


def score_key(item):
    return item[0]


def update_best(best_a, best_score, a):
    a = clean_array(a)
    if a.size == 0:
        return best_a, best_score
    sc = score_array(a)
    if math.isfinite(sc) and sc < best_score:
        return a.astype(np.float64), float(sc)
    return best_a, best_score


def add_candidate(starts, a, tag, n):
    a = clean_array(a)
    n = int(n)
    if a.size != n or n <= 0:
        return starts
    a = normalize_sum(a, n)
    sc = score_array(a)
    if math.isfinite(sc):
        starts.append((sc, a, str(tag)))
    return starts


def build_A(a):
    """
    Matrix A with rows for convolution lags and columns for indices j.
    A[k,j] = a[k-j] if 0 <= k-j < len(a), else 0.
    For nonnegative vectors x, A @ x = convolve(a, x).
    It is also the gradient-core for conv(a, a) wrt a.
    """
    a = clean_array(a)
    n = a.size
    if n == 0:
        return np.zeros((0, 0), dtype=np.float64)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def select_top_rows(C, M, frac):
    C = np.asarray(C, dtype=np.float64)
    if frac is None or frac >= 1.0:
        return np.arange(C.size, dtype=np.int64)
    k = max(1, int(round(C.size * max(0.1, min(1.0, frac)))))
    if k >= C.size:
        return np.arange(C.size, dtype=np.int64)
    # Return the indices of the k largest C values.
    return np.argpartition(C, -k)[-k:]


def solve_lp(c_vec, A_ub, b_ub, bounds, A_eq, b_eq, lp_time):
    if linprog is None:
        return None
    opts = {"time_limit": max(0.1, float(lp_time))}
    try:
        return linprog(
            c_vec,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            A_eq=A_eq,
            b_eq=b_eq,
            method="highs",
            options=opts,
        )
    except Exception:
        pass
    try:
        return linprog(
            c_vec,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            A_eq=A_eq,
            b_eq=b_eq,
            method="highs",
        )
    except Exception:
        return None


def add_noise(a, rng, sigma, smooth_width=16):
    a = clean_array(a)
    if a.size == 0:
        return a
    loga = np.log(np.maximum(a, 1e-15))
    noise = rng.normal(0.0, float(sigma), a.size)
    if smooth_width and smooth_width > 1:
        w = int(max(1, min(smooth_width, a.size)))
        kernel = np.ones(w, dtype=np.float64) / float(w)
        noise = np.convolve(noise, kernel, mode="same")
    out = np.exp(loga + noise)
    out = np.nan_to_num(out, nan=0.0, posinf=UB_FINAL, neginf=0.0)
    out = np.clip(out, 0.0, UB_FINAL)
    return normalize_sum(out, a.size)


def lp_maxsum_step(a, lp_time=3.0, top_frac=1.0, cap_scale=1.0):
    a0 = normalize_sum(a)
    n = a0.size
    in_score = score_array(a0)
    if n < 2 or not math.isfinite(in_score):
        return a0, in_score, False

    # A positive floor avoids numerically singular columns.  It changes the
    # current point only negligibly.
    a_lp = np.maximum(a0, 1e-12)
    C = np.convolve(a_lp, a_lp)
    M = float(np.max(C))
    if not math.isfinite(M) or M <= 0.0:
        return a0, in_score, False

    A = build_A(a_lp)
    rows = select_top_rows(C, M, top_frac)
    A_sub = A[rows]
    Mcap = M * max(0.5, float(cap_scale))
    b = np.full(len(rows), Mcap, dtype=np.float64)

    S_lp = float(np.sum(a_lp))
    a_fm_pos = max(float(a_lp[0]), 1e-12)
    ub = max(50.0, 5.0 * max(1.0, S_lp), 10.0 * Mcap / a_fm_pos)
    ub = min(ub, 10000.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = np.column_stack((np.zeros(n, dtype=np.float64), np.full(n, ub, dtype=np.float64)))
    res = solve_lp(c_obj, A_sub, b, bounds, None, None, lp_time)
    if res is None or getattr(res, "x", None) is None:
        return a0, in_score, False

    g = np.asarray(res.x, dtype=np.float64).ravel()
    if g.size != n:
        return a0, in_score, False
    g = np.nan_to_num(g, nan=0.0, posinf=ub, neginf=0.0)
    g = np.clip(g, 0.0, ub)
    G = float(np.sum(g))
    if not math.isfinite(G) or G <= S_lp * (1.0 + 1e-7):
        return a0, in_score, False

    h = g * (float(np.sum(a0)) / G)
    h = np.clip(h, 0.0, UB_FINAL)

    best_a = a0
    best_score = in_score
    steps = (1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.45, 0.4, 0.35, 0.3,
             0.25, 0.2, 0.15, 0.12, 0.1, 0.08, 0.06, 0.04, 0.03, 0.02, 0.01, 0.005)
    for s in steps:
        cand = a0 + float(s) * (h - a0)
        if not np.all(np.isfinite(cand)):
            continue
        cand = np.clip(cand, 0.0, UB_FINAL)
        cand = normalize_sum(cand, n)
        sc = score_array(cand)
        if math.isfinite(sc) and sc < best_score:
            best_a = cand
            best_score = sc
    improved = math.isfinite(best_score) and best_score < in_score - 1e-12
    return best_a, float(best_score), bool(improved)


def lp_chebyshev_step(a, lp_time=3.0, top_frac=1.0):
    a0 = normalize_sum(a)
    n = a0.size
    in_score = score_array(a0)
    if n < 2 or not math.isfinite(in_score):
        return a0, in_score, False

    C = np.convolve(a0, a0)
    M = float(np.max(C))
    if not math.isfinite(M) or M <= 0.0:
        return a0, in_score, False

    A = build_A(a0)
    rows = select_top_rows(C, M, top_frac)
    A_sub = A[rows]
    C_sub = C[rows]
    m = len(rows)

    # Minimize t subject to C_k + 2*(a_{k-j} delta_j) <= t.
    Cmat = np.empty((m, n + 1), dtype=np.float64)
    Cmat[:, :n] = 2.0 * A_sub
    Cmat[:, -1] = -1.0
    b = -C_sub

    c_vec = np.zeros(n + 1, dtype=np.float64)
    c_vec[-1] = 1.0

    A_eq = np.ones(n + 1, dtype=np.float64)
    A_eq[-1] = 0.0
    b_eq = np.array([0.0], dtype=np.float64)

    lower = -0.49 * a0
    upper = np.minimum(5.0, 1.2 * np.maximum(a0, 1.0) + 0.5)
    upper = np.maximum(upper, lower + 1e-12)

    bounds = np.empty((n + 1, 2), dtype=np.float64)
    bounds[:n, 0] = lower
    bounds[:n, 1] = upper
    bounds[-1, 0] = -10.0 * max(M, 1e-12)
    bounds[-1, 1] = 10.0 * max(M, 1e-12) + 1.0

    res = solve_lp(c_vec, Cmat, b, bounds, A_eq, b_eq, lp_time)
    if res is None or getattr(res, "x", None) is None:
        return a0, in_score, False

    x = np.asarray(res.x, dtype=np.float64).ravel()
    if x.size != n + 1:
        return a0, in_score, False
    delta = x[:n]
    t_opt = float(x[-1])
    delta = np.nan_to_num(delta, nan=0.0, posinf=0.0, neginf=0.0)

    if not math.isfinite(t_opt):
        return a0, in_score, False

    # If no first-order decrease, a line search is unlikely to help.
    if t_opt >= M - 1e-9:
        return a0, in_score, False

    # Enforce sum-zero more accurately.
    delta -= float(np.mean(delta))
    dmax = float(np.max(np.abs(delta)))
    if not math.isfinite(dmax) or dmax <= 1e-12:
        return a0, in_score, False

    # Largest step that should maintain nonnegativity, allowing a little slack.
    max_step = 1.0
    neg = delta < -1e-14
    if np.any(neg):
        ratios = a0[neg] / (-delta[neg])
        valid = ratios[np.isfinite(ratios) & (ratios > 0.0)]
        if valid.size > 0:
            max_step = min(max_step, float(np.min(valid)))
    max_step = min(max_step, 1.0)
    if not math.isfinite(max_step) or max_step <= 1e-12:
        return a0, in_score, False

    best_a = a0
    best_score = in_score
    base_steps = (1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.35, 0.3,
                  0.25, 0.2, 0.15, 0.1, 0.08, 0.06, 0.04, 0.03, 0.02, 0.01, 0.005, 0.002)
    for bs in base_steps:
        s = float(bs) * max_step
        if s <= 0.0:
            continue
        cand = a0 + s * delta
        if not np.all(np.isfinite(cand)):
            continue
        if np.min(cand) < -1e-10:
            continue
        cand = np.clip(cand, 0.0, UB_FINAL)
        cand = normalize_sum(cand, n)
        sc = score_array(cand)
        if math.isfinite(sc) and sc < best_score:
            best_a = cand
            best_score = sc
    improved = math.isfinite(best_score) and best_score < in_score - 1e-12
    return best_a, float(best_score), bool(improved)


def polish_candidate(a, budget_s, rng, verbose=False):
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s))

    a0 = clean_array(a)
    if a0.size == 0:
        fb = central_seq(128)
        return fb, score_array(fb)
    a0 = normalize_sum(a0, a0.size)
    best_a = a0.astype(np.float64).copy()
    best_score = score_array(a0)

    no_success = 0
    it = 0
    # Cycles: max-sum full, max-sum full with small cap, Chebyshev full,
    # max-sum active rows with larger cap, Chebyshev active rows.
    stage_types = (
        (0, 1.0, 1.0),
        (0, 1.0, 1.05),
        (1, 1.0, 1.0),
        (0, 0.75, 1.10),
        (1, 0.75, 1.0),
    )

    while time.monotonic() < deadline - 1.0:
        rem = deadline - time.monotonic()
        if rem <= 1.5:
            break
        lp_time = max(0.7, min(4.5, rem * 0.10))

        cur = best_a
        if no_success > 0 and rng is not None:
            sigma = 0.015 + 0.01 * min(4, no_success // 4)
            cur = add_noise(best_a, rng, sigma)

        improved = False
        typ, frac, cap = stage_types[it % len(stage_types)]

        if linprog is not None:
            # Use only as much time as is reasonably available to a single LP.
            if time.monotonic() + lp_time + 0.5 < deadline:
                if typ == 0:
                    na, ns, imp = lp_maxsum_step(cur, lp_time, frac, cap)
                else:
                    na, ns, imp = lp_chebyshev_step(cur, lp_time, frac)
                if math.isfinite(ns) and ns < best_score:
                    best_a = na.astype(np.float64)
                    best_score = float(ns)
                    improved = True
                    no_success = 0
        else:
            cand = add_noise(best_a, rng, 0.05)
            sc = score_array(cand)
            if math.isfinite(sc) and sc < best_score:
                best_a = cand
                best_score = sc
                improved = True
                no_success = 0

        if not improved:
            no_success += 1
            # Cheap direct hill-climbing around the current best point.
            for _ in range(25):
                cand = add_noise(best_a, rng, 0.045, smooth_width=8)
                sc = score_array(cand)
                if math.isfinite(sc) and sc < best_score:
                    best_a = cand
                    best_score = sc
            if no_success >= 12:
                break
            if verbose and (no_success % 3 == 0):
                print(
                    "polish stall",
                    "n=%d" % int(best_a.size),
                    "score=%.10f" % float(best_score),
                    "it=%d" % it,
                )
        else:
            if verbose and (it % 4 == 0):
                print(
                    "polish it=%d" % it,
                    "n=%d" % int(best_a.size),
                    "score=%.10f" % float(best_score),
                )

        it += 1

    return best_a.astype(np.float64), float(best_score)


def generate_candidate_list(n, rng, prev_seq=None):
    n = int(n)
    starts = []
    if n <= 0:
        return starts

    base = central_seq(n)
    starts = add_candidate(starts, base, "central", n)
    starts = add_candidate(starts, base[::-1], "rev", n)

    i_arr = np.arange(n, dtype=np.float64)
    idx = np.arange(1, n + 1, dtype=np.float64)
    t = i_arr / float(max(1, n - 1))

    # Mix forward and reverse central profiles.
    for alpha in (0.08, 0.15, 0.25, 0.35, 0.50, 0.70):
        a = (1.0 - alpha) * base + alpha * base[::-1]
        starts = add_candidate(starts, a, "mix%.2f" % alpha, n)

    # Single-side power laws.
    for p in (0.20, 0.30, 0.40, 0.50, 0.60, 0.80, 1.00):
        a = np.power(idx, -float(p))
        starts = add_candidate(starts, a, "pow%.2f" % p, n)

    # Two-sided power laws.
    for p in (0.30, 0.50, 0.80):
        a = np.power(idx, -float(p)) + np.power(n - idx + 1.0, -float(p))
        starts = add_candidate(starts, a, "pow2%.2f" % p, n)

    # Mild linear tapering of central.
    for beta in (-0.40, -0.20, 0.20, 0.40, 0.80):
        a = base * (1.0 + float(beta) * t)
        if np.any(a <= 0.0):
            a = np.maximum(a, 1e-12)
        starts = add_candidate(starts, a, "taper%.2f" % beta, n)

    # Low-frequency oscillating modulation of central.
    for k in (1, 2, 4, 8):
        for ph in (0.0, 1.57079632679):
            a = base * (1.0 + 0.18 * np.cos((2.0 * math.pi * float(k)) * t + float(ph)))
            if np.any(a <= 0.0):
                a = np.maximum(a, 1e-12)
            starts = add_candidate(starts, a, "wave_%d_%s" % (k, int(ph)), n)

    # Smooth multiplicative noise.
    if rng is not None:
        for sigma in (0.05, 0.10, 0.20, 0.40):
            a = base * np.exp(rng.normal(0.0, float(sigma), n))
            starts = add_candidate(starts, a, "noise%.2f" % sigma, n)

    # A few simple envelopes.
    starts = add_candidate(starts, np.ones(n, dtype=np.float64), "uniform", n)
    starts = add_candidate(starts, np.exp(-i_arr / (n / 4.0)), "exp4", n)
    starts = add_candidate(starts, np.exp(-i_arr / (n / 8.0)), "exp8", n)
    a = np.exp(-i_arr / (n / 4.0)) + np.exp(-(n - 1.0 - i_arr) / (n / 4.0))
    starts = add_candidate(starts, a, "two_exp", n)

    # Previous known construction, resampled if available.
    if prev_seq is not None:
        try:
            p = clean_array(prev_seq)
        except Exception:
            p = None
        if p is not None and p.size > 0:
            starts = add_candidate(starts, interpolate_sequence(p, n), "prev", n)

    return starts


def propose_candidate(seed=42, budget_s=400):
    """Return the best nonnegative sequence found within the time budget."""
    start_time = time.monotonic()
    total = max(1.0, float(budget_s))
    # Keep a reserve for final validation and return overhead.
    deadline = start_time + max(1.0, total - 22.0)

    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    best_a = central_seq(1024).astype(np.float64)
    best_score = score_array(best_a)

    # Baselines: central constructions for several lengths and previous data.
    for n in (128, 256, 512, 1024):
        c = central_seq(n).astype(np.float64)
        best_a, best_score = update_best(best_a, best_score, c)

    prev = None
    try:
        prev = height_sequence_1
    except Exception:
        prev = None

    if prev is not None:
        best_a, best_score = update_best(best_a, best_score, prev)

    # Build a short list of promising initial candidates.
    probe_list = []

    for n, top_count in ((1024, 6), (512, 3)):
        starts = generate_candidate_list(n, rng, prev)
        if not starts:
            continue
        starts.sort(key=score_key)
        # Also include the currently best normalized shape of this length as a start.
        cur_best = normalize_sum(best_a, n)
        starts = add_candidate(starts, cur_best, "best_interp", n)
        starts.sort(key=score_key)
        # Keep only candidates reasonably close to the current global bound.
        cutoff = float(best_score) + 0.35
        for item in starts[:top_count]:
            if item[0] <= cutoff:
                probe_list.append(item)

    if probe_list:
        probe_list.sort(key=score_key)
        # Deduplicate by length and near-score to avoid polishing the same basin repeatedly.
        dedup = []
        n_values = []
        for item in probe_list:
            n = int(item[1].size)
            duplicate = False
            for j, nn in enumerate(n_values):
                if nn == n and item[0] <= dedup[j][0] + 1e-6:
                    duplicate = True
                    break
            if not duplicate:
                dedup.append(item)
                n_values.append(n)
        probe_list = dedup[:9]

        # First pass: short polishing of multiple promising starts.
        for idx, item in enumerate(probe_list):
            rem = deadline - time.monotonic()
            if rem < 40.0:
                break
            left = max(1, len(probe_list) - idx)
            budget = min(25.0, max(5.0, (rem - 30.0) / float(left)))
            if budget <= 1.0:
                continue
            sc0, a0, tag = item
            print(
                "probe start",
                "n=%d" % int(a0.size),
                "tag=%s" % tag,
                "score=%.10f" % float(sc0),
                "budget=%.1f" % float(budget),
            )
            pa, ps = polish_candidate(a0, budget, rng, verbose=True)
            best_a, best_score = update_best(best_a, best_score, pa)
            print(
                "probe done",
                "n=%d" % int(pa.size),
                "score=%.10f" % float(ps),
                "global=%.10f" % float(best_score),
            )

    # If the best sequence is shorter than 1024, attempt one longer resample;
    # often the asymptotic limit helps.
    rem = deadline - time.monotonic()
    if rem > 90.0 and best_a.size != 1024:
        a1024 = interpolate_sequence(best_a, 1024)
        pa, ps = polish_candidate(a1024, min(50.0, rem - 70.0), rng, verbose=True)
        best_a, best_score = update_best(best_a, best_score, pa)
        print("resample to 1024 score", "%.10f" % float(ps), "global", "%.10f" % float(best_score))

    # Final concentrated polish on the best sequence found.
    rem = deadline - time.monotonic()
    if rem > 35.0 and math.isfinite(best_score):
        budget = min(rem - 18.0, 260.0)
        if budget > 8.0:
            pa, ps = polish_candidate(best_a, budget, rng, verbose=True)
            best_a, best_score = update_best(best_a, best_score, pa)
            print("final polish score", "%.10f" % float(ps), "global", "%.10f" % float(best_score))

    # Final safety: guarantee a valid, finite, nonempty sequence.
    best_a = clean_array(best_a)
    if best_a.size == 0 or not math.isfinite(best_score) or float(np.sum(best_a)) < 0.01:
        best_a = central_seq(1024).astype(np.float64)
        best_score = score_array(best_a)

    # Authoritative check, if available in the environment.
    try:
        ext = float(evaluate_sequence([float(v) for v in best_a]))
        if math.isfinite(ext):
            best_score = min(best_score, ext)
    except Exception:
        pass

    print("FINAL returned score estimate", "%.10f" % float(best_score), "length", int(best_a.size))
    return [float(v) for v in best_a]
```
