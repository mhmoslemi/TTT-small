def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def central_seq(n):
    """Return coefficients of (1-z)^(-1/2), truncated to length n.

    These have convolution coefficients exactly 1 on the first half and are a strong
    nonnegative starting construction.
    """
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (i + 0.5) / (i + 1.0)
    return a


def score_array(a):
    """Fast authoritative-compatible score for a numpy array."""
    try:
        arr = np.asarray(a, dtype=np.float64)
    except Exception:
        return float("inf")
    if arr.ndim != 1 or arr.size == 0:
        return float("inf")
    if not np.all(np.isfinite(arr)):
        return float("inf")
    arr = np.nan_to_num(arr, nan=0.0, posinf=UB_FINAL, neginf=0.0)
    arr = np.clip(arr, 0.0, UB_FINAL)

    if c.size == 0:

    m = float(np.max(c))
    if not math.isfinite(m):
        return float("inf")
    val = float(2.0 * arr.size * m / (s * s))
    if not math.isfinite(val):
        return float("inf")
    return val

def smooth_obj(x, beta):
    """Smooth approximation of the anchored objective and analytic gradient."""
    x = np.asarray(x, dtype=np.float64).ravel()
    m = int(x.size)
    if m < 0:
        m = 0
    n = m + 1
    if not np.all(np.isfinite(x)):
        return 1e100, np.zeros(max(m, 0), dtype=np.float64)

    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    if m > 0:
        a[1:] = x
    a = np.nan_to_num(a, nan=0.0, posinf=UB_FINAL, neginf=0.0)
    a = np.clip(a, 0.0, UB_FINAL)

    s = float(np.sum(a))
    if s <= 0.0 or not math.isfinite(s):
        return 1e100, np.zeros(max(m, 0), dtype=np.float64)

    c = np.convolve(a, a)
    if c.size == 0:
        return 1e100, np.zeros(max(m, 0), dtype=np.float64)
    cn = float(np.max(c))
    if not math.isfinite(cn):
        return 1e100, np.zeros(max(m, 0), dtype=np.float64)

    beta = float(beta)
    if not math.isfinite(beta) or beta <= 0.0:
        beta = 1.0

    z = np.minimum(beta * (c - cn), 0.0)
    z = np.clip(z, -745.0, 0.0)
    ez = np.exp(z)
    great = float(np.sum(ez))
    if great <= 0.0 or not math.isfinite(great):
        return 1e100, np.zeros(max(m, 0), dtype=np.float64)

    L = cn + math.log(great) / beta
    w = ez / great

    # h_i = sum_j a_j w_{i+j}, i=0..n-1
    conv = np.convolve(a, w[::-1])
    idx = np.arange(n)
    h = conv[2 * n - 2 - idx]
    dL = 2.0 * h

    C = 2.0 * float(n)
    G = C * (dL - (2.0 * L * a) / (s * s))
    val = C * L / (s * s)

    grad = G[1:]
    if not math.isfinite(val):
        return 1e100, np.zeros(max(m, 0), dtype=np.float64)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    if not np.all(np.isfinite(grad)):
        return 1e100, np.zeros(max(m, 0), dtype=np.float64)
    return float(val), grad


def run_lbfgs(n, x0, beta, maxiter):
    """Run one bounded L-BFGS-B stage, returning a finite candidate on failure."""
    x0 = np.asarray(x0, dtype=np.float64).ravel().copy()
    if x0.size != n - 1:
        x0 = np.zeros(max(0, n - 1), dtype=np.float64)
    ub = ub_for_length(n)
    x0 = np.nan_to_num(x0, nan=0.0, posinf=ub, neginf=0.0)
    x0 = np.clip(x0, 0.0, ub)

    if minimize is None:
        return x0
    if x0.size == 0:
        return x0

    bounds = [(0.0, ub)] * x0.size
    maxiter = int(max(1, maxiter))
    try:
        res = minimize(
            fun=smooth_obj,
            x0=x0,
            args=(float(beta),),
            method="L-BFGS-B",
            bounds=bounds,
            options={
                "maxiter": maxiter,
                "maxls": 50,
                "ftol": 1e-12,
                "gtol": 1e-5,
                "disp": False,
            },
        )
        xx = np.asarray(res.x, dtype=np.float64).ravel()
    except Exception:
        xx = x0

    if xx.size != x0.size:
        xx = x0
    xx = np.nan_to_num(xx, nan=0.0, posinf=ub, neginf=0.0)
    return np.clip(xx, 0.0, ub)


def polish_anchored(n, x0, maxiter1, maxiter2):
    """Multi-stage local polish for one anchored start."""
    n = int(n)
    x0 = np.asarray(x0, dtype=np.float64).ravel().copy()
    if x0.size != n - 1:
        x0 = np.zeros(max(0, n - 1), dtype=np.float64)
    ub = ub_for_length(n)
    x0 = np.nan_to_num(x0, nan=0.0, posinf=ub, neginf=0.0)
    x0 = np.clip(x0, 0.0, ub)

    best_x = x0.copy()
    best_score = float(score_array(anchored_array(n, best_x)))

    if minimize is not None:
        M = float(max_conv_array(anchored_array(n, best_x)))
        if math.isfinite(M) and M > 0.0:
            beta1 = 20.0 / max(M, 1e-8)
            x1 = run_lbfgs(n, best_x, beta1, maxiter1)
            s1 = float(score_array(anchored_array(n, x1)))
            if math.isfinite(s1) and s1 < best_score:
                best_x = x1
                best_score = s1

            M1 = float(max_conv_array(anchored_array(n, best_x)))
            if math.isfinite(M1) and M1 > 0.0:
                beta2 = 50.0 / max(M1, 1e-8)
                x2 = run_lbfgs(n, best_x, beta2, maxiter2)
                s2 = float(score_array(anchored_array(n, x2)))
                if math.isfinite(s2) and s2 < best_score:
                    best_x = x2
                    best_score = s2

                if n <= 64:
                    M2 = float(max_conv_array(anchored_array(n, best_x)))
                    if math.isfinite(M2) and M2 > 0.0:
                        beta3 = 100.0 / max(M2, 1e-8)
                        x3 = run_lbfgs(n, best_x, beta3, max(8, maxiter2 // 2))
                        s3 = float(score_array(anchored_array(n, x3)))
                        if math.isfinite(s3) and s3 < best_score:
                            best_x = x3
                            best_score = s3

    # Exact derivative-free refinement only for truly small problems.
    if n <= 10 and minimize is not None:
        m = max(0, n - 1)
        bounds = [(0.0, ub)] * m
        try:
            res = minimize(
                fun=true_obj_anchored,
                x0=best_x,
                method="Powell",
                bounds=bounds,
                options={
                    "maxiter": 800,
                    "xtol": 1e-7,
                    "ftol": 1e-10,
                    "disp": False,
                },
            )
            xx = np.asarray(res.x, dtype=np.float64).ravel()
            if xx.size == m:
                xx = np.nan_to_num(xx, nan=0.0, posinf=ub, neginf=0.0)
                xx = np.clip(xx, 0.0, ub)
                s = float(score_array(anchored_array(n, xx)))
                if math.isfinite(s) and s < best_score:
                    best_x = xx
                    best_score = s
        except Exception:
            pass

    return best_x, best_score


def make_initial(n, kind, param, rng):
    """Generate a deterministic or rng-generated anchored initial vector."""
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    b = central_seq(n)
    if n == 1:
        return np.array([], dtype=np.float64)

    kind = str(kind)
    p = float(param)

    if kind == "central":
        a = b.astype(np.float64).copy()
    elif kind == "rev":
        a = b[::-1].astype(np.float64).copy()
    elif kind == "sym":
        a = (b + b[::-1]).astype(np.float64).copy()
    elif kind == "alpha":
        idx = np.arange(1, n + 1, dtype=np.float64)
        if p == 0.0:
            a = np.ones(n, dtype=np.float64)
        else:
            a = np.power(idx, -p)
    elif kind == "alpha_mirror":
        idx1 = np.arange(1, n + 1, dtype=np.float64)
        idx2 = n - idx1 + 1.0
        a = np.power(idx1, -p) + np.power(idx2, -p)
    elif kind == "trend":
        t = np.arange(n - 1, dtype=np.float64) / float(max(1, n - 1))
        a = b.astype(np.float64).copy()
        a[1:] *= (1.0 + p * t)
        a = np.clip(a, 0.0, 1e6)
    elif kind == "mirror":
        a = b.astype(np.float64) + p * b[::-1].astype(np.float64)
    elif kind == "wave":
        t = np.arange(n, dtype=np.float64) / float(max(1, n - 1))
        mult = 1.0 + p * np.cos(2.0 * math.pi * t)
        a = b.astype(np.float64) * mult
        a = np.clip(a, 0.0, 1e6)
    elif kind == "noise":
        sigma = 0.15 + 0.15 * abs(p)
        a = b.astype(np.float64).copy()
        a[1:] *= np.exp(rng.normal(0.0, sigma, n - 1))
        a = np.clip(a, 0.0, 1e6)
    elif kind == "rand":
        base = 0.25 + 0.45 * np.abs(rng.standard_normal(n))
        decay = np.exp(-np.arange(n, dtype=np.float64) / float(max(1, n // 4)))
        a = base * decay
    else:
        a = b.astype(np.float64).copy()

    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=UB_FINAL, neginf=0.0)
    if a.size != n or not np.all(np.isfinite(a)) or a[0] <= 1e-15:
        a = b.astype(np.float64).copy()

    a = a / a[0]
    ub = ub_for_length(n)
    return np.clip(a[1:], 0.0, ub)

def interpolate_anchored(a, n):
    """Resample an old sequence to a new anchored length."""
    n = int(n)
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0 or n <= 0:
        return np.zeros(max(0, n - 1), dtype=np.float64)
    if a.size == 1:
        return np.zeros(max(0, n - 1), dtype=np.float64)
    old = np.linspace(0.0, 1.0, a.size)
    new = np.linspace(0.0, 1.0, n)
    na = np.interp(new, old, a)
    if not np.all(np.isfinite(na)) or na[0] <= 1e-15:
        na = central_seq(n)
    na = na / na[0]
    ub = ub_for_length(n)
    return np.clip(na[1:], 0.0, ub)


def make_noisy_candidate(a, rng, sigma):
    """Block-multiplicative perturbation for final hill-climbing / annealing."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = a.size
    if n == 0:
        return a
    a = np.nan_to_num(a, nan=0.0, posinf=UB_FINAL, neginf=0.0)
    a = np.clip(a, 0.0, UB_FINAL)

    nblocks = int(rng.integers(1, 3))
    for _ in range(nblocks):
        max_width = max(1, n // 4)
        width = int(rng.integers(1, max(2, max_width + 1)))
        width = max(1, min(n, width))
        start = int(rng.integers(0, max(1, n - width + 1)))
        scale = float(np.exp(rng.normal(0.0, sigma, width)))
        a[start:start + width] *= scale

    if rng.random() < 0.30:
        a = 0.5 * (a + a[::-1])

    a = np.nan_to_num(a, nan=0.0, posinf=UB_FINAL, neginf=0.0)
    return np.clip(a, 0.0, UB_FINAL)


def anneal_polish(a, budget_s, rng):
    """Light stochastic refinement of the current best sequence."""
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.nan_to_num(a, nan=0.0, posinf=UB_FINAL, neginf=0.0)
    a = np.clip(a, 0.0, UB_FINAL)

    cur = a.copy()
    cur_score = float(score_array(cur))
    best = a.copy()
    best_score = cur_score

    if not math.isfinite(best_score):
        fb = central_seq(128)
        best = fb
        cur = fb
        best_score = float(score_array(fb))
        cur_score = best_score

    deadline = time.monotonic() + max(0.0, float(budget_s))
    sigma = 0.12
    T = 0.02
    no_improve = 0
    step = 0

    while time.monotonic() < deadline:
        step += 1
        cand = make_noisy_candidate(cur, rng, sigma)
        sc = float(score_array(cand))

        if math.isfinite(sc):
            improve_cur = sc <= cur_score
            if not improve_cur:
                diff = sc - cur_score
                if diff < 10.0 * max(T, 1e-9):
                    pacc = math.exp(-diff / max(T, 1e-9))
                    improve_cur = bool(rng.random() < pacc)

            if improve_cur:
                cur = cand
                cur_score = sc
                if sc < best_score:
                    best = cand.copy()
                    best_score = sc
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve >= 120:
            cur = best.copy()
            cur_score = best_score
            no_improve = 0
            sigma = max(0.02, 0.7 * sigma)
            T = max(0.002, 0.7 * T)

        if step % 40 == 0:
            sigma = max(0.02, 0.97 * sigma)
            T = max(0.001, 0.97 * T)

        if step % 400 == 0:
            print(f"anneal step={step} best_score={best_score:.10f} sigma={sigma:.4f}")

    return best, float(best_score)


def lp_refine(a):
    """One state-of-the-art style linearization move, used only for short enough n."""
    if linprog is None:
        return a, float(score_array(a))

    a0 = np.asarray(a, dtype=np.float64).ravel()
    if a0.size < 2 or a0.size > 128:
        return a0, float(score_array(a0))

    s = float(np.sum(a0))
    if s <= 0.01 or not math.isfinite(s):
        fb = central_seq(64)
        return fb, float(score_array(fb))

    f = a0 / s
    n = int(f.size)
    c = np.convolve(f, f)
    M = float(np.max(c))
    if not math.isfinite(M) or M <= 0.0:
        return f, float(score_array(f))

    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        if start <= end:
            js = np.arange(start, end + 1)
            A[k, js] = f[k - js]

    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    ub = 100.0
    bounds = [(0.0, ub)] * n
    c_obj = np.ones(n, dtype=np.float64)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    except Exception:
        return f, float(score_array(f))

    if not getattr(res, "success", False) or res.x is None:
        return f, float(score_array(f))

    g = np.asarray(res.x, dtype=np.float64).ravel()
    g = np.nan_to_num(g, nan=0.0, posinf=ub, neginf=0.0)
    g = np.clip(g, 0.0, ub)
    gs = float(np.sum(g))
    if gs <= 0.0 or not math.isfinite(gs):
        return f, float(score_array(f))

    gres = g / gs
    best = f.copy()
    best_score = float(score_array(f))
    gs_score = float(score_array(gres))

                best_score = sc

        pass

        if n > 1024 and (work_deadline - time.monotonic()) < 35.0:
            break
        if n > 512 and (work_deadline - time.monotonic()) < 25.0:
            break

        starts = []

        # Carry forward the currently best shape by interpolation/resampling.
        x_interp = interpolate_anchored(carry, n)
        if x_interp.size == n - 1:
            a = anchored_array(n, x_interp)
            sc = float(score_array(a))
            if math.isfinite(sc):
                if sc < best_score:
                    best_a = a.copy()
                    best_score = sc

            kind, param = spec
            x = make_initial(n, kind, param, rng)
            if x.size != n - 1:
                continue
            a = anchored_array(n, x)
            sc = float(score_array(a))
            if math.isfinite(sc):
                if sc < best_score:
                    best_a = a.copy()
                    best_score = sc

        max_st = max(1, max_starts_for(n))

        elif n <= 24:
            mi1, mi2 = 220, 70
        elif n <= 48:
            mi1, mi2 = 150, 50
        elif n <= 128:
            mi1, mi2 = 90, 35
        elif n <= 256:
            mi1, mi2 = 50, 20
        elif n <= 512:
            mi1, mi2 = 30, 12
        else:
            mi1, mi2 = 18, 8

        local_best_a = None
        local_best_score = float("inf")

                if sc < best_score:
                    best_a = a.copy()
                    best_score = sc
                if sc < local_best_score:
                    local_best_a = a.copy()
                    local_best_score = sc

                ssc = float(score_array(sym))

                        best_score = ssc

                        local_best_score = ssc

                    local_best_a = anchored_array(n, x0).copy()
                    local_best_score = sc0

        if local_best_a is not None and local_best_score < float(score_array(best_a)) and local_best_score < best_score - 1e-12:
            pass

        if local_best_a is not None and local_best_score < best_score:
            best_a = local_best_a.copy()
            best_score = local_best_score

        if local_best_a is not None:
            carry = local_best_a.copy()
        elif best_a.size != n:
            carry = best_a.copy()

        print(f"length n={n:4d} after local search best_score={best_score:.10f}")

    # Optional LP move on the best short sequence.
    if linprog is not None and best_a.size >= 2 and best_a.size <= 128:
        if best_score > TARGET_UPPER_BOUND and (deadline - time.monotonic()) > 90.0:
            try:
                a_lp, sc_lp = lp_refine(best_a)
                if math.isfinite(sc_lp) and sc_lp < best_score:
                    best_a = a_lp.astype(np.float64).copy()
                    best_score = sc_lp
                    print(f"LP refinement best_score={best_score:.10f} n={best_a.size}")
            except Exception:
                pass

    # Final annealing / hill climbing.
    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 3.0:
        if best_a.size <= 128:
            anneal_budget = min(remaining - 1.0, 70.0)
        elif best_a.size <= 384:
            anneal_budget = min(remaining - 1.0, 40.0)
        elif best_a.size <= 1024:
            anneal_budget = min(remaining - 1.0, 20.0)
        else:
            anneal_budget = min(remaining - 1.0, 8.0)

        if anneal_budget > 1.0:
            try:
                a_ann, sc_ann = anneal_polish(best_a, anneal_budget, rng)
                if math.isfinite(sc_ann) and sc_ann < best_score:
                    best_a = a_ann.astype(np.float64).copy()
                    best_score = sc_ann
                    print(f"annealed best_score={best_score:.10f} n={best_a.size}")
            except Exception:
                pass

    # Final validation and fallback.
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=UB_FINAL, neginf=0.0)
    best_a = np.clip(best_a, 0.0, UB_FINAL)

    if best_a.size == 0 or not math.isfinite(float(score_array(best_a))) or float(np.sum(best_a)) < 0.01:
        best_a = central_seq(128).astype(np.float64)
        best_score = float(score_array(best_a))

    # Optional final check against the authoritative evaluator, if present.
    try:
        ext = float(evaluate_sequence([float(v) for v in best_a]))
        if math.isfinite(ext) and ext < best_score:
            best_score = ext
    except Exception:
        pass

    print(f"FINAL best_score={best_score:.10f} n={best_a.size}")
    return [float(v) for v in best_a]
