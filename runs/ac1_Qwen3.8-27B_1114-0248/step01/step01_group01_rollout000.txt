```python
"""
Nonnegative convolution peak-factor optimizer.

Key ideas in this version:
- The problem is scale invariant, so we search with length-n beta-cell
  densities a_i = P(i/n <= X < (i+1)/n) for X on [0,1].
- For large n, score ~ 2 * max_y (p*p)(y) for density p.
- Densities with mild endpoint concentration Beta(A,A), where A is just above
  0.5 in typical optimal regimes, empirically beat the classical central
  construction substantially.
- For robustness we also evaluate central/previous candidates, do a coarse
  (A,B) beta search over several lengths, run a bounded low-dimensional
  refinement for beta parameters, a full bounded L-BFGS-B smooth-max polish
  for medium lengths, and a conservative stochastic block-annealer finish.

No file/network IO.  All helper functions are top-level, no lambdas/closures.
"""

import math
import time

import numpy as np

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None

try:
    from scipy.special import betainc
except Exception:
    betainc = None


TARGET_UPPER_BOUND = 1.5027
UB_FINAL = 1000.0


def central_seq(n):
    """Return coefficients of (1-z)^(-1/2), truncated to length n."""
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (i + 0.5) / (i + 1.0)
    return a


def sanitize_sequence(x):
    """Force a 1D float64 nonnegative finite vector clipped to evaluator range."""
    arr = np.asarray(x, dtype=np.float64).ravel()
    if arr.size == 0:
        return arr
    arr = np.nan_to_num(arr, nan=0.0, posinf=UB_FINAL, neginf=0.0)
    arr = np.clip(arr, 0.0, UB_FINAL)
    return arr


def linear_convolution(x, force_fft=False):
    """Linear convolution x*x.  Uses FFT for longer sequences."""
    x = np.asarray(x, dtype=np.float64).ravel()
    n = int(x.size)
    if n == 0:
        return np.array([], dtype=np.float64)

    need = 2 * n - 1
    if force_fft or n >= 3000:
        m = 1
        while m < need:
            m <<= 1
        X = np.fft.rfft(x, m)
        out = np.fft.irfft(X * X, m)[:need]
        # Remove possible tiny negative roundoff around zero for max safety.
        out = np.where(out < 0.0, 0.0, out)
        return out

    return np.convolve(x, x, mode="full")


def score_array(x, force_fft=False, exact=False):
    """
    Authoritative-compatible score:
        2 * n * max(convolve(a,a)) / (sum(a)^2)
    with the same clipping as the evaluator.
    """
    arr = sanitize_sequence(x)
    if arr.size == 0:
        return float("inf")

    s = float(np.sum(arr))
    if s < 0.01 or not math.isfinite(s):
        return float("inf")

    use_fft = (not exact) and (bool(force_fft) or arr.size >= 3000)
    c = linear_convolution(arr, force_fft=use_fft)
    if c.size == 0:
        return float("inf")

    m = float(np.max(c))
    if not math.isfinite(m):
        return float("inf")

    val = float(2.0 * arr.size * m / (s * s))
    if not math.isfinite(val):
        return float("inf")
    return val


def beta_sequence(n, A, B):
    """
    Cell probabilities of an unnormalized Beta density:
        p(x) ∝ x^(A-1) (1-x)^(B-1),  x in [0,1].
    Returns a nonnegative length-n sequence summing to 1.
    """
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)

    A = float(A)
    B = float(B)

    if not (math.isfinite(A) and math.isfinite(B)):
        A, B = 0.7, 0.7

    # We only want integrable, nonpathological endpoint behavior.
    A = min(max(A, 0.5005), 1.30)
    B = min(max(B, 0.5005), 1.30)

    m = None
    if betainc is not None:
        try:
            edges = np.linspace(0.0, 1.0, n + 1, dtype=np.float64)
            cum = betainc(A, B, edges)
            m = np.diff(np.real(cum))
            if m.size != n:
                m = None
        except Exception:
            m = None

    if m is None:
        # Fallback midpoint rule.  This is robust even without scipy.special.
        x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
        with np.errstate(divide="ignore", invalid="ignore", over="ignore"):
            m = np.exp((A - 1.0) * np.log(x) + (B - 1.0) * np.log1p(-x))
        m = np.nan_to_num(m, nan=0.0, posinf=0.0, neginf=0.0)
        m = np.clip(m, 0.0, None)

    m = np.nan_to_num(m, nan=0.0, posinf=0.0, neginf=0.0)
    m = np.clip(m, 0.0, 1.0)
    s = float(np.sum(m))
    if s <= 1e-300:
        m = np.ones(n, dtype=np.float64) / float(n)
    else:
        m = m / s

    return m


def beta_objective(x, n, use_fft):
    """Objective for scipy.minimize over (A, B)."""
    try:
        A = float(x[0])
        B = float(x[1])
        if not (0.5005 <= A <= 1.25 and 0.5005 <= B <= 1.25):
            return 1e6
        a = beta_sequence(int(n), A, B)
        val = score_array(a, force_fft=bool(use_fft))
        if not math.isfinite(val):
            return 1e6
        return float(val)
    except Exception:
        return 1e6


def refine_lowdim_beta(n, A0, B0, maxiter=120):
    """Short bounded(Nelder-Mead) refinement of Beta parameters."""
    A0 = float(A0)
    B0 = float(B0)
    if minimize is None or maxiter <= 5:
        return A0, B0

    x0 = np.array([A0, B0], dtype=np.float64)
    use_fft = int(n) >= 2048
    initial = beta_objective(x0, n, use_fft)

    try:
        res = minimize(
            beta_objective,
            x0,
            args=(int(n), use_fft),
            method="Nelder-Mead",
            options={
                "maxiter": int(maxiter),
                "xatol": 1e-3,
                "fatol": 1e-7,
            },
        )
    except Exception:
        return A0, B0

    try:
        xx = np.asarray(res.x, dtype=np.float64).ravel()
        if xx.size != 2 or not np.all(np.isfinite(xx)):
            return A0, B0
        val = beta_objective(xx, n, use_fft)
        if math.isfinite(val) and val < initial - 1e-12:
            return float(xx[0]), float(xx[1])
    except Exception:
        pass

    return A0, B0


def vars_from_array(a, anchor):
    """
    Convert an array to optimizer variables by fixing a[anchor] = 1.
    Variable order is natural index order, skipping anchor.
    """
    a = sanitize_sequence(a)
    n = int(a.size)
    if n <= 1:
        return np.array([], dtype=np.float64)

    anchor = int(anchor)
    anchor = max(0, min(n - 1, anchor))
    scale = float(a[anchor])
    if scale <= 1e-15:
        scale = 1.0

    if anchor == 0:
        x = a[anchor + 1:] / scale
    elif anchor == n - 1:
        x = a[:anchor] / scale
    else:
        x = np.concatenate((a[:anchor], a[anchor + 1:])) / scale

    x = np.nan_to_num(x, nan=0.0, posinf=UB_FINAL, neginf=0.0)
    return np.clip(x, 0.0, UB_FINAL)


def build_from_vars(x, anchor):
    """Inverse of vars_from_array, with fixed anchor value 1."""
    x = sanitize_sequence(x)
    n = int(x.size) + 1
    a = np.empty(n, dtype=np.float64)
    anchor = int(anchor)
    anchor = max(0, min(n - 1, anchor))

    if anchor == 0:
        a[0] = 1.0
        a[1:] = x
    elif anchor == n - 1:
        a[:anchor] = x
        a[anchor] = 1.0
    else:
        a[:anchor] = x[:anchor]
        a[anchor] = 1.0
        a[anchor + 1:] = x[anchor:]

    a = np.nan_to_num(a, nan=0.0, posinf=UB_FINAL, neginf=0.0)
    return np.clip(a, 0.0, UB_FINAL)


def smooth_obj(x, anchor, beta):
    """
    Smooth log-sum-exp approximation of the anchored objective, with gradient.

    a[anchor] is fixed to 1.  This removes scale invariance while preserving
    the objective exactly up to the smoothing parameter beta.
    """
    x = np.asarray(x, dtype=np.float64).ravel()
    m = int(x.size)
    n = m + 1

    zero_grad = np.zeros(max(m, 0), dtype=np.float64)
    if not np.all(np.isfinite(x)):
        return 1e100, zero_grad

    anchor = int(anchor)
    a = build_from_vars(x, anchor)
    if a.size != n or not np.all(np.isfinite(a)):
        return 1e100, zero_grad

    s = float(np.sum(a))
    if s <= 0.0 or not math.isfinite(s):
        return 1e100, zero_grad

    # For medium-sized polish we use direct convolution.
    c = np.convolve(a, a, mode="full")
    if c.size == 0:
        return 1e100, zero_grad

    cm = float(np.max(c))
    if not math.isfinite(cm):
        return 1e100, zero_grad

    beta = float(beta)
    if not math.isfinite(beta) or beta <= 0.0:
        beta = 1.0

    z = beta * (c - cm)
    z = np.minimum(z, 0.0)
    z = np.clip(z, -745.0, 0.0)
    ez = np.exp(z)
    great = float(np.sum(ez))
    if great <= 0.0 or not math.isfinite(great):
        return 1e100, zero_grad

    L = cm + math.log(great) / beta
    w = ez / great

    # Correct smooth derivative:
    # For a_i, dL/da_i = 2 * sum_j a_j w_{i+j}.
    conv = np.convolve(a[::-1], w, mode="full")
    h = conv[n - 1:2 * n - 1]
    dL = 2.0 * h

    C = 2.0 * float(n)
    scale2 = s * s
    G = C * (dL - (2.0 * L * a) / scale2)
    val = C * L / scale2

    if anchor == 0:
        grad = G[1:]
    elif anchor == n - 1:
        grad = G[:anchor]
    else:
        grad = np.concatenate((G[:anchor], G[anchor + 1:]))

    if not math.isfinite(val):
        return 1e100, zero_grad

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    if not np.all(np.isfinite(grad)):
        return 1e100, zero_grad

    return float(val), grad


def lbfgs_refine(a, maxiter=40):
    """
    Bounded L-BFGS-B polish using smooth max approximation.
    Returns (sequence, score).
    """
    a0 = sanitize_sequence(a)
    n = int(a0.size)
    s0 = score_array(a0, force_fft=(n >= 3000))
    best = a0.copy()
    best_score = s0

    if n < 2 or minimize is None or not math.isfinite(best_score):
        return best, best_score

    # Choose a positive anchor.  The largest coefficient is usually best.
    try:
        anchors = [int(np.argmax(a0)), 0, n - 1, n // 2]
    except Exception:
        anchors = [0]

    anchors = []
    seen = set()
    for idx in anchors:
        idx = int(max(0, min(n - 1, idx)))
        if idx not in seen:
            seen.add(idx)
            anchors.append(idx)

    max_anchors = 1 if n > 700 else 2 if n > 200 else 3
    anchors = anchors[:max_anchors]

    ub = 100.0

    for anchor in anchors:
        if math.isfinite(best_score):
            pass

        base = a0 if math.isfinite(s0) else best
        val0 = float(base[anchor])
        if val0 <= 1e-14:
            continue

        x0 = vars_from_array(base, anchor)
        if x0.size <= 0:
            continue
        x0 = np.clip(x0, 0.0, ub)

        # Estimate current max for good log-sum-exp sharpness.
        c0 = np.convolve(base, base, mode="full")
        M = float(np.max(c0)) if c0.size else 0.0
        if not math.isfinite(M) or M <= 1e-300:
            M = 1e-12
        beta = max(1.0, 50.0 / M)

        bounds = [(0.0, ub)] * x0.size
        try:
            res = minimize(
                smooth_obj,
                x0,
                args=(anchor, float(beta)),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": int(max(10, maxiter)),
                    "maxls": 30,
                    "ftol": 1e-12,
                    "gtol": 1e-6,
                    "disp": False,
                },
            )
            xx = np.asarray(res.x, dtype=np.float64).ravel()
        except Exception:
            continue

        if xx.size != x0.size:
            continue
        xx = np.nan_to_num(xx, nan=0.0, posinf=ub, neginf=0.0)
        xx = np.clip(xx, 0.0, ub)

        cand = build_from_vars(xx, anchor)
        sc = score_array(cand, force_fft=(n >= 3000))
        if math.isfinite(sc) and sc < best_score:
            best = cand
            best_score = float(sc)

    return best, float(best_score)


def make_noisy_candidate(a, rng, sigma):
    """
    Stochastic block multiplier candidate.  Useful for escaping tiny local
    minima in the exact nonsmooth objective.
    """
    a = sanitize_sequence(a)
    n = int(a.size)
    if n == 0:
        return a

    out = a.copy()
    nblocks = 1
    if rng.random() < 0.45:
        nblocks = 2

    max_width = max(1, n // 8)
    for _ in range(nblocks):
        width = int(rng.random() * max_width) + 1
        width = max(1, min(n, width))
        start = int(rng.random() * max(1, n - width + 1))
        delta = rng.normal(0.0, sigma, width)
        delta = np.clip(delta, -3.0, 3.0)
        out[start:start + width] *= np.exp(delta)

    if rng.random() < 0.10:
        out = 0.5 * (out + out[::-1])

    out = np.nan_to_num(out, nan=0.0, posinf=UB_FINAL, neginf=0.0)
    return np.clip(out, 0.0, UB_FINAL)


def anneal_polish(a, budget_s, rng):
    """
    Conservative stochastic beam/annealing polish on the exact score.
    """
    a = sanitize_sequence(a)
    if a.size == 0:
        a = central_seq(512).astype(np.float64)

    cur = a.copy()
    cur_score = score_array(cur, force_fft=(cur.size >= 2500))
    best = cur.copy()
    best_score = cur_score

    # Quick symmetrization attempt.
    try:
        sym = 0.5 * (best + best[::-1])
        sym_score = score_array(sym, force_fft=(sym.size >= 2500))
        if math.isfinite(sym_score) and sym_score < best_score:
            best = sym.copy()
            best_score = float(sym_score)
            cur = best.copy()
            cur_score = best_score
    except Exception:
        pass

    if not math.isfinite(best_score):
        fb = central_seq(512).astype(np.float64)
        best = fb
        cur = fb
        best_score = score_array(fb, force_fft=False)
        cur_score = best_score

    deadline = time.monotonic() + max(0.0, float(budget_s))
    sigma = 0.08
    T = 0.004
    no_improve = 0
    step = 0

    while time.monotonic() < deadline:
        step += 1
        cand = make_noisy_candidate(cur, rng, sigma)
        sc = score_array(cand, force_fft=(cand.size >= 2500))

        if math.isfinite(sc):
            accept = (sc <= cur_score)
            if not accept and (sc - cur_score) < 25.0 * max(T, 1e-12):
                accept = bool(rng.random() < math.exp(-(sc - cur_score) / max(T, 1e-12)))

            if accept:
                cur = cand
                cur_score = float(sc)
                if sc < best_score:
                    best = cand.copy()
                    best_score = float(sc)
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve >= 160:
            cur = best.copy()
            cur_score = best_score
            no_improve = 0
            sigma = max(0.015, 0.65 * sigma)
            T = max(0.0005, 0.6 * T)

        if step % 32 == 0:
            sigma = max(0.012, 0.985 * sigma)
            T = max(0.0004, 0.985 * T)

        if step % 800 == 0:
            print(f"anneal step={step} best_score={best_score:.10f} sigma={sigma:.4f}")

    return best, float(best_score)


def propose_candidate(seed=42, budget_s=400):
    """
    Return the best nonnegative sequence found within budget_s.
    """
    start_time = time.monotonic()
    total_budget = max(1.0, float(budget_s))
    deadline = start_time + max(0.0, total_budget * 0.982)

    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    # ------------------------------------------------------------------
    # Safe baselines.
    # ------------------------------------------------------------------
    best_a = central_seq(512).astype(np.float64)
    best_score = score_array(best_a, force_fft=False)

    print(f"central(512) score={best_score:.10f}")

    for n in (256, 512, 1024, 2048, 4096):
        try:
            a = central_seq(n).astype(np.float64)
            sc = score_array(a, force_fft=(n >= 3000))
            if math.isfinite(sc) and sc < best_score:
                best_a = a
                best_score = float(sc)
                print(f"central({n}) score={best_score:.10f}")
        except Exception:
            pass

    # Use any previous global construction if available.
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if hs.size > 0:
            hs = sanitize_sequence(hs)
            sc = score_array(hs, force_fft=(hs.size >= 3000), exact=(hs.size <= 4096))
            if math.isfinite(sc) and sc < best_score:
                best_a = hs
                best_score = float(sc)
                print(f"previous SOTA score={best_score:.10f} n={best_a.size}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Primary construction: beta-cell densities.
    # ------------------------------------------------------------------
    A_small = [0.58, 0.61, 0.64, 0.67, 0.70, 0.73, 0.77, 0.82]
    A_large = [
        0.55, 0.57, 0.59, 0.61, 0.63, 0.65, 0.67, 0.69,
        0.71, 0.73, 0.76, 0.80, 0.85, 0.90, 1.00,
    ]

    sym_best_score_by_n = {}
    sym_best_A_by_n = {}

    best_small_score = float("inf")
    best_small_a = None

    def update_candidate(a, sc, n):
        """Local helper is not a function closure; this is just nested logic."""
        return a, sc

    # Quick small-length beta scan.
    for n in (256, 512, 1024):
        if time.monotonic() > deadline - 30.0:
            break
        for A in A_small:
            try:
                a = beta_sequence(n, A, A)
                sc = score_array(a, force_fft=(n >= 3000))
                if math.isfinite(sc):
                    if sc < best_score:
                        best_a = a.copy()
                        best_score = float(sc)
                        print(f"beta({n}, A={A:.3f}) score={best_score:.10f}")
                    cur_n = sym_best_score_by_n.get(n, float("inf"))
                    if sc < cur_n:
                        sym_best_score_by_n[n] = float(sc)
                        sym_best_A_by_n[n] = float(A)
                    if n <= 1024 and sc < best_small_score:
                        best_small_score = float(sc)
                        best_small_a = a.copy()
            except Exception:
                pass

    # Larger-length symmetric beta scan.
    large_lengths = [512, 1024, 2048, 4096]
    if deadline - time.monotonic() > 130.0:
        large_lengths.append(8192)
    if deadline - time.monotonic() > 300.0:
        large_lengths.append(16384)

    for n in large_lengths:
        if time.monotonic() > deadline - 70.0:
            break
        # Very large lengths are expensive; when time is tight test only a partial grid.
        if n >= 8192 and (deadline - time.monotonic()) < 90.0:
            A_use = [0.60, 0.65, 0.70, 0.75, 0.80]
        else:
            A_use = A_large

        for A in A_use:
            try:
                a = beta_sequence(n, A, A)
                sc = score_array(a, force_fft=(n >= 2048))
                if math.isfinite(sc):
                    if sc < best_score:
                        best_a = a.copy()
                        best_score = float(sc)
                        print(f"beta({n}, A={A:.3f}) score={best_score:.10f}")
                    cur_n = sym_best_score_by_n.get(n, float("inf"))
                    if sc < cur_n:
                        sym_best_score_by_n[n] = float(sc)
                        sym_best_A_by_n[n] = float(A)
                    if n <= 1024 and sc < best_small_score:
                        best_small_score = float(sc)
                        best_small_a = a.copy()
            except Exception:
                pass

    # ------------------------------------------------------------------
    # Low-dimensional and local 2D refinement over Beta parameters.
    # ------------------------------------------------------------------
    top_lengths = []
    for n in sorted(large_lengths, reverse=True):
        if len(top_lengths) >= 3:
            break
        if n >= 8192 and (deadline - time.monotonic()) < 140.0:
            continue
        if n >= 4096 and (deadline - time.monotonic()) < 80.0:
            continue
        top_lengths.append(n)

    if not top_lengths:
        top_lengths = [1024, 2048]

    for n in top_lengths:
        if time.monotonic() > deadline - 80.0:
            break

        A0 = sym_best_A_by_n.get(n, 0.68)
        B0 = A0
        n_int = int(n)

        if minimize is not None and (deadline - time.monotonic()) > 50.0:
            mi = 90 if n_int <= 2048 else 60
            try:
                A1, B1 = refine_lowdim_beta(n_int, A0, B0, maxiter=mi)
                a = beta_sequence(n_int, A1, B1)
                sc = score_array(a, force_fft=(n_int >= 2048))
                if math.isfinite(sc):
                    if sc < best_score:
                        best_a = a.copy()
                        best_score = float(sc)
                        print(f"lowdim-beta({n}, {A1:.4f}, {B1:.4f}) score={best_score:.10f}")
                    cur_n = sym_best_score_by_n.get(n, float("inf"))
                    if sc < cur_n:
                        sym_best_score_by_n[n] = float(sc)
                        sym_best_A_by_n[n] = float(A1)
                    if n_int <= 1024 and sc < best_small_score:
                        best_small_score = float(sc)
                        best_small_a = a.copy()
                    A0, B0 = float(A1), float(B1)
            except Exception:
                pass

        # Around the current best symmetric A.
        if (deadline - time.monotonic()) > 25.0:
            center = float(sym_best_A_by_n.get(n, A0))
            vals = np.linspace(center - 0.08, center + 0.08, 7)

            for A in vals:
                if not (0.52 <= A <= 1.15):
                    continue
                for B in vals:
                    if not (0.52 <= B <= 1.15) or (deadline - time.monotonic()) > 25.0:
                        continue
                    try:
                        a = beta_sequence(n_int, A, B)
                        sc = score_array(a, force_fft=(n_int >= 2048))
                        if math.isfinite(sc):
                            if sc < best_score:
                                best_a = a.copy()
                                best_score = float(sc)
                                print(f"2D-beta({n}, A={A:.4f}, B={B:.4f}) score={best_score:.10f}")
                            cur_n = sym_best_score_by_n.get(n, float("inf"))
                            if sc < cur_n:
                                sym_best_score_by_n[n] = float(sc)
                                sym_best_A_by_n[n] = float(A)
                            if n_int <= 1024 and sc < best_small_score:
                                best_small_score = float(sc)
                                best_small_a = a.copy()
                    except Exception:
                        pass

    # ------------------------------------------------------------------
    # Optional full polynomial/beta local polish for medium lengths.
    # ------------------------------------------------------------------
    local_a = None
    if best_a.size <= 1024 and best_a.size >= 16:
        local_a = best_a
    elif best_small_a is not None and best_small_a.size >= 16:
        local_a = best_small_a
    else:
        local_a = central_seq(1024).astype(np.float64)

    if local_a is not None and (deadline - time.monotonic()) > 90.0:
        try:
            n_local = int(local_a.size)
            mi = 35 if n_local > 512 else 60
            a1, sc1 = lbfgs_refine(local_a, maxiter=mi)
            if math.isfinite(sc1) and sc1 < best_score:
                best_a = a1.copy()
                best_score = float(sc1)
                print(f"L-BFGS polish score={best_score:.10f} n={best_a.size}")
            # Symmetrized variant can sometimes remove asymmetry noise.
            sym = 0.5 * (best_a + best_a[::-1])
            scs = score_array(sym, force_fft=(sym.size >= 3000), exact=(sym.size <= 4096))
            if math.isfinite(scs) and scs < best_score:
                best_a = sym.copy()
                best_score = float(scs)
                print(f"sym score={best_score:.10f} n={best_a.size}")
        except Exception:
            pass

    # Another local polish on the currently best small length, if it is different.
    if best_small_a is not None and (deadline - time.monotonic()) > 70.0:
        try:
            if not np.array_equal(best_small_a, best_a):
                ni = int(best_small_a.size)
                mi = 35 if ni > 512 else 50
                a1, sc1 = lbfgs_refine(best_small_a, maxiter=mi)
                if math.isfinite(sc1) and sc1 < best_score:
                    best_a = a1.copy()
                    best_score = float(sc1)
                    print(f"small-LBFGS score={best_score:.10f} n={best_a.size}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stochastic finishing.
    # ------------------------------------------------------------------
    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 18.0:
        try:
            anneal_budget = remaining - 2.5
            # Large final sequences can be polished with FFT scoring; cap budget.
            if best_a.size > 4096:
                anneal_budget = min(anneal_budget, 70.0)
            elif best_a.size > 1500:
                anneal_budget = min(anneal_budget, 55.0)
            else:
                anneal_budget = min(anneal_budget, 45.0)

            if anneal_budget > 3.0:
                a_ann, sc_ann = anneal_polish(best_a, anneal_budget, rng)
                if math.isfinite(sc_ann) and sc_ann < best_score:
                    best_a = a_ann.copy()
                    best_score = float(sc_ann)
                    print(f"annealed score={best_score:.10f} n={best_a.size}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Final scaling / validation.
    # ------------------------------------------------------------------
    final_a = sanitize_sequence(best_a)
    n_final = int(final_a.size)
    if n_final <= 0:
        final_a = central_seq(512).astype(np.float64)
        n_final = int(final_a.size)

    s_final = float(np.sum(final_a))
    if s_final > 0.0 and math.isfinite(s_final):
        target_sum = math.sqrt(2.0 * max(1, n_final))
        max_coeff = float(np.max(final_a))
        if max_coeff > 0.0:
            scale_limit = (0.9 * UB_FINAL) / max_coeff
            scale = min(target_sum / s_final, scale_limit)
        else:
            scale = target_sum / s_final
        final_a = final_a * float(scale)
        final_a = sanitize_sequence(final_a)
    else:
        final_a = central_seq(512).astype(np.float64)
        target_sum = math.sqrt(2.0 * final_a.size)
        ss = float(np.sum(final_a))
        if ss > 0:
            final_a = final_a * (target_sum / ss)
        final_a = sanitize_sequence(final_a)

    final_score = score_array(final_a, force_fft=(final_a.size >= 3000), exact=(final_a.size <= 4096))
    if not math.isfinite(final_score):
        final_a = central_seq(512).astype(np.float64)
        final_score = score_array(final_a, force_fft=False)

    best_score = min(best_score, final_score)

    # Optional authoritative check, if the evaluator symbol is present.
    try:
        ext = float(evaluate_sequence([float(v) for v in final_a]))
        if math.isfinite(ext):
            best_score = min(best_score, ext)
        else:
            # Extremely defensive fallback.
            fb = central_seq(512).astype(np.float64)
            fb_score = score_array(fb, force_fft=False)
            if math.isfinite(fb_score):
                final_a = fb
                best_score = fb_score
    except Exception:
        pass

    print(f"FINAL score={best_score:.10f} n={final_a.size}")
    return [float(v) for v in final_a]
```
