"""
Search for nonnegative step heights minimizing
    2*n*max(conv(a,a)) / sum(a)^2.

This implementation uses probability-normalized sequences p (sum 1) and
optimizes them with mirror descent in logit space.  Convolution and gradients
are computed with FFTs.  It combines several descent directions:
  - flattening/low-pass directions based on c - mean(c);
  - peak softmax directions emphasizing current largest convolution lags;
  - orientation-bias directions to move mass toward weaker one-sided lags;
  - mixture modes.

It explores several natural initial shapes (sqrt-binomial, tapers, symmetric
variants, tilted variants, random perturbations) at multiple lengths, does a
short staged exploration, then polishes the best found sequence within the
time budget.
"""

import numpy as np
import time


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _normalize_probability(p):
    p = np.asarray(p, dtype=np.float64)
    if p.ndim != 1 or p.size == 0:
        return np.array([], dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(p.size, 1.0 / p.size, dtype=np.float64)
    return p / s


def _center_clip_z(z, max_range=50.0):
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return z
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.mean(z))
    mn = float(np.min(z))
    mx = float(np.max(z))
    rng = mx - mn
    if not np.isfinite(rng) or rng <= 0.0:
        return np.zeros_like(z, dtype=np.float64)
    if rng > max_range:
        z = z * (max_range / rng)
    return z


def _softmax_p(z):
    z = np.asarray(z, dtype=np.float64)
    n = int(z.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    m = float(np.max(z))
    ex = np.exp(z - m)
    s = float(ex.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    p = ex / s
    p = np.maximum(p, 0.0)
    s2 = float(p.sum(dtype=np.float64))
    if not np.isfinite(s2) or s2 <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s2


def _shape_to_p(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def _make_sqrt_series(n):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * float(2 * k - 1) / float(2 * k)
    return a


def _conv_p(p, nsize):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    L = 2 * n - 1
    P = np.fft.rfft(p, nsize)
    c = np.fft.irfft(P * P, nsize)
    c = c[:L]
    c = np.maximum(c, 0.0)
    c = np.where(np.isfinite(c), c, 0.0)
    return c


def _score_p(p):
    p = _normalize_probability(p)
    n = int(p.size)
    if n == 0:
        return np.inf
    nsize = _next_pow2(2 * n - 1)
    c = _conv_p(p, nsize)
    cm = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cm):
        return np.inf
    return float(2 * n * cm)


def _normalize_weights(w):
    w = np.asarray(w, dtype=np.float64)
    if w.size == 0:
        return None
    w = np.where(np.isfinite(w), w, 0.0)
    m = float(np.max(np.abs(w)))
    if not np.isfinite(m) or m <= 1e-16:
        return None
    return w / m


def _peak_weights_raw(c, temp):
    c = np.asarray(c, dtype=np.float64)
    L = int(c.size)
    if L == 0:
        return np.array([], dtype=np.float64)
    score = float(np.max(c))
    if not np.isfinite(score) or score <= 1e-14:
        return np.full(L, 1.0 / L, dtype=np.float64)

    if temp <= 1e-12:
        w = (c >= score * 0.999).astype(np.float64)
        s = float(w.sum(dtype=np.float64))
        if s <= 1e-14:
            return np.full(L, 1.0 / L, dtype=np.float64)
        return w / s

    u = (c - score) / temp
    u -= float(np.max(u))
    w = np.exp(u)
    w += 1e-15
    s = float(w.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 1e-14:
        return np.full(L, 1.0 / L, dtype=np.float64)
    return w / s


def _make_weights(c, mode):
    c = np.asarray(c, dtype=np.float64)
    L = int(c.size)
    if L == 0:
        return None
    c = np.where(np.isfinite(c), c, 0.0)
    score = float(np.max(c))
    if not np.isfinite(score) or score <= 1e-14:
        return None
    mean_c = float(np.mean(c))

    if mode == 0:
        w = c - mean_c
        return _normalize_weights(w)

    if mode == 1:
        d = c - mean_c
        w = np.sign(d) * (np.abs(d) ** 3)
        return _normalize_weights(w)

    if mode == 2:
        w = _peak_weights_raw(c, max(1e-9, score * 0.25))
        return _normalize_weights(w)

    if mode == 3:
        w = _peak_weights_raw(c, max(1e-9, score * 0.10))
        return _normalize_weights(w)

    if mode == 4:
        w = _peak_weights_raw(c, max(1e-9, score * 0.04))
        return _normalize_weights(w)

    if mode == 5:
        wf = c - mean_c
        wp = _peak_weights_raw(c, max(1e-9, score * 0.06))
        wf_n = _normalize_weights(wf)
        wp_n = _normalize_weights(wp)
        if wf_n is None:
            return wp_n
        if wp_n is None:
            return wf_n
        return _normalize_weights(0.5 * wf_n + 0.5 * wp_n)

    if mode == 6:
        idx = np.linspace(0.0, 1.0, L, dtype=np.float64) if L > 1 else np.zeros(max(L, 1), dtype=np.float64)
        w = (c - mean_c) - 0.25 * idx[:L]
        return _normalize_weights(w)

    if mode == 7:
        idx = np.linspace(0.0, 1.0, L, dtype=np.float64) if L > 1 else np.zeros(max(L, 1), dtype=np.float64)
        w = (c - mean_c) + 0.25 * idx[:L]
        return _normalize_weights(w)

    w = c - mean_c
    return _normalize_weights(w)


def _gradient_direction(p, w, nsize):
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(p.size)
    L = 2 * n - 1
    if w.size != L:
        return None

    rw = w[::-1].copy()
    P = np.fft.rfft(p, nsize)
    W = np.fft.rfft(rw, nsize)
    conv = np.fft.irfft(P * W, nsize)

    # H[m] = sum_i p[i] * w[i + m]
    H = conv[L - n:L][::-1]
    H = np.where(np.isfinite(H), H, 0.0)

    mu = float(np.dot(p, H))
    h = H - mu
    scale = float(np.max(np.abs(h))) if h.size else 0.0
    if not np.isfinite(scale) or scale <= 1e-13:
        return None
    h = h / scale
    h = np.where(np.isfinite(h), h, 0.0)
    return h


def _run_mode(p_init, seconds, mode, rng, label):
    p = _normalize_probability(p_init)
    n = int(p.size)
    if n == 0:
        return p, np.inf, np.array([], dtype=np.float64)
    if seconds <= 0.0:
        sc = _score_p(p)
        z = _center_clip_z(np.log(np.maximum(p, 1e-300)))
        return p, sc, z

    nsize_conv = _next_pow2(2 * n - 1)
    nsize_grad = _next_pow2(3 * n - 2)

    z = _center_clip_z(np.log(np.maximum(p, 1e-300)))
    p = _softmax_p(z)
    c = _conv_p(p, nsize_conv)
    score = float(2 * n * np.max(c))

    best_p = p.copy()
    best_z = z.copy()
    best_score = score
    step = 0.55
    fail = 0
    it = 0
    start = time.monotonic()
    deadline = start + float(seconds)
    last_print = start

    if n > 16000:
        factors = (0.25, 0.5, 1.0, 2.0)
    else:
        factors = (0.25, 0.5, 1.0, 2.0, 3.0)

    while time.monotonic() < deadline:
        it += 1
        cmax = float(np.max(c))
        w = _make_weights(c, mode)
        if w is None:
            break

        h = _gradient_direction(p, w, nsize_grad)
        if h is None:
            z = z + 0.04 * rng.standard_normal(n)
            z = _center_clip_z(z)
            p = _softmax_p(z)
            c = _conv_p(p, nsize_conv)
            score = float(2 * n * np.max(c))
            step = 0.35
            fail += 1
            if fail > 10:
                z = best_z + 0.05 * rng.standard_normal(n)
                z = _center_clip_z(z)
                p = _softmax_p(z)
                c = _conv_p(p, nsize_conv)
                score = float(2 * n * np.max(c))
                step = 0.35
                fail = 4
            continue

        best_cand = None
        best_cand_score = score
        best_step = step

        for fac in factors:
            st = step * float(fac)
            if st <= 1e-4:
                continue
            zt = z - st * h
            if fail > 6 and fac == 1.0 and rng.random() < 0.25:
                zt = zt + 0.03 * rng.standard_normal(n)
            zt = _center_clip_z(zt)
            pt = _softmax_p(zt)
            ct = _conv_p(pt, nsize_conv)
            sc = float(2 * n * np.max(ct))
            if np.isfinite(sc) and sc < best_cand_score - 1e-11:
                best_cand_score = sc
                best_cand = (zt, pt, ct)
                best_step = st

        if best_cand is not None:
            z, p, c = best_cand
            score = best_cand_score
            if score < best_score:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                fail = 0
                now = time.monotonic()
                if now - last_print > 5.0:
                    print(f"[{label}] mode={mode} score={best_score:.9f} t={now - start:.1f}/{seconds:.1f}")
                    last_print = now
            step = min(2.5, best_step * 1.15)
        else:
            step *= 0.62
            fail += 1
            if step < 0.04:
                step = 0.55
                fail = 0
            if fail >= 12:
                noise = 0.04 + 0.01 * min(1.0, float(fail) / 40.0)
                z = best_z + noise * rng.standard_normal(n)
                z = _center_clip_z(z)
                p = _softmax_p(z)
                c = _conv_p(p, nsize_conv)
                score = float(2 * n * np.max(c))
                step = 0.40
                fail = 5

    return best_p, best_score, best_z

def _run_schedule(p_init, seconds, schedule, rng, label, deadline, initial_score=None):

    now = time.monotonic()
    for mode, frac in schedule:
        rem = float(deadline - now) if deadline is not None else np.inf
        t = float(seconds) * float(frac)
        t = min(t, rem)

            p = best_p.copy()
        now = time.monotonic()

    if np.isfinite(score) and score < best_score:

def _make_shapes(n, rng):
    jobs = []
    n = int(n)
    if n <= 0:
        return jobs

    r = _make_sqrt_series(n)
    rev = r[::-1].copy()
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    sx = np.sin(np.pi * x)
    sx = np.maximum(sx, 1e-300)

    _add_shape(jobs, r, 'sqrt')
    _add_shape(jobs, rev, 'sqrt_rev')
    _add_shape(jobs, r + rev, 'sym_add')
    _add_shape(jobs, np.sqrt(np.maximum(r * rev, 0.0) + 1e-12), 'sym_mul')
    _add_shape(jobs, r * (0.20 + 0.80 * x), 'taper_dec')
    _add_shape(jobs, (r + rev) * (sx ** 0.5), 'taper_sym')
    _add_shape(jobs, np.ones(n, dtype=np.float64), 'uniform')
    _add_shape(jobs, np.maximum(x, 1e-6) ** (-0.10), 'dec_m10')
    _add_shape(jobs, np.maximum(x, 1e-6) ** 0.10, 'inc_p10')
    _add_shape(jobs, np.maximum(x * (1.0 - x) + 1e-6, 1e-6) ** (-0.10), 'central_m10')
    _add_shape(jobs, sx ** 0.5, 'sin05')
    _add_shape(jobs, sx, 'sin1')
    _add_shape(jobs, 1.0 - np.abs(2.0 * x - 1.0), 'tri')
    _add_shape(jobs, np.exp(-6.0 * (x - 0.5) ** 2), 'gauss6')
    _add_shape(jobs, r * np.exp(0.8 * x), 'tilt_up')
    _add_shape(jobs, r * np.exp(-0.8 * x), 'tilt_down')
    _add_shape(jobs, r * np.exp(0.25 * rng.standard_normal(n)), 'rnd1')
    _add_shape(jobs, r * np.exp(0.40 * (x - 0.5)), 'linear_tilt')

    if len(jobs) > 12:
        jobs.sort(key=_score_dict_key)
        jobs = jobs[:12]
    return jobs


def _resample_p(p, m):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    m = int(m)
    if n <= 0 or m <= 0:
        return None
    if n == 1:
        return np.full(m, 1.0 / m, dtype=np.float64)
    x0 = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    x1 = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
    q = np.interp(x1, x0, p)
    q = np.maximum(q, 1e-15)
    s = float(q.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return q / s

    except Exception:
        return None
    if arr.ndim != 1 or arr.size == 0 or arr.size > 40000:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return arr / s


def _make_output(p):
    p = _normalize_probability(p)
    n = int(p.size)
    if n == 0:
        n = 8192
        p = np.ones(n, dtype=np.float64) / float(n)
    target = float(np.sqrt(2.0 * n))
    out = p * target
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)
    out = np.minimum(out, 1000.0)
    s = float(out.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 1e-9:
        out = np.ones(n, dtype=np.float64)
    return [float(v) for v in out]


def propose_candidate(seed=42, budget_s=400, **kwargs):
    try:
        rng = np.random.default_rng(int(seed))
        raw_budget = float(budget_s)
        time_budget = max(1.0, raw_budget - 5.0)
        start = time.monotonic()
        deadline = start + time_budget

        # Fallback.
        n_fb = 8192
        p_global = _shape_to_p(_make_sqrt_series(n_fb))
        s_global = _score_p(p_global)

        jobs = []
        height_p = _get_height_sequence()
        height_job = None

        if height_p is not None:
            hs = _score_p(height_p)
            if np.isfinite(hs):
                height_job = {'p': height_p.copy(), 'score': hs, 'n': int(height_p.size), 'label': 'height'}
                jobs.append(height_job)
                if hs < s_global:
                    s_global = hs
                    p_global = height_p.copy()

        n_list = [4096, 8192, 16384]
        if raw_budget > 220.0:
            n_list.append(24576)

        for n in n_list:
            shapes = _make_shapes(n, rng)
            shapes.sort(key=_score_dict_key)
            top = 4 if n >= 16384 else 3
            for item in shapes[:top]:
                _add_job(jobs, item['p'], item['score'], f"n{n}_{item['label']}")

            if height_p is not None and height_p.size != n:
                pr = _resample_p(height_p, n)
                if pr is not None:
                    sr = _score_p(pr)
                    _add_job(jobs, pr, sr, f"n{n}_height_interp")

        if not jobs:
            _add_job(jobs, p_global, s_global, 'fallback')

        if raw_budget < 25.0:
            jobs.sort(key=_score_dict_key)
            if len(jobs) > 0 and np.isfinite(jobs[0]['score']) and jobs[0]['score'] < s_global:
                s_global = jobs[0]['score']
                p_global = jobs[0]['p'].copy()
            try:
                print(f"short budget final length={p_global.size} score={s_global:.9f}")
            except Exception:
                pass
            return _make_output(p_global)

        # Stage 1: brief exploration of promising starts.
        jobs.sort(key=_score_dict_key)
        stage1_jobs = []
        count_n = {}
        for job in jobs:
            nn = job['n']
            if count_n.get(nn, 0) >= 3:
                continue
            stage1_jobs.append(job)
            count_n[nn] = count_n.get(nn, 0) + 1
            if len(stage1_jobs) >= 12:
                break

        in_stage = False
        if height_job is not None:
            for item in stage1_jobs:
                if item is height_job:
                    in_stage = True
                    break
        if height_job is not None and not in_stage and len(stage1_jobs) < 16:
            stage1_jobs.append(height_job)

        polish_frac = 0.18
        if s_global < 1.53:
            s1_frac = 0.16
        elif s_global < 1.55:
            s1_frac = 0.25
        else:
            s1_frac = 0.35
        s2_frac = max(0.30, 1.0 - s1_frac - polish_frac)

        alloc = max(1.0, time_budget - 8.0)
        stage1_time = alloc * (s1_frac / max(1e-12, s1_frac + s2_frac))
        stage2_time = alloc * (s2_frac / max(1e-12, s1_frac + s2_frac))

        stage1_time = min(stage1_time, 120.0)
        stage2_time = min(stage2_time, 220.0)
        stage1_time = max(4.0, min(stage1_time, float(deadline - start) - 10.0))

        results = []
        per1 = stage1_time / max(1, len(stage1_jobs))
        for job in stage1_jobs:
            now = time.monotonic()
            rem = float(deadline - now)
            t = min(per1, 25.0, max(2.0, rem))
            if t <= 2.0:
                continue
            sched = _quick_schedule(t, job['n'])
            p2, sc2 = _run_schedule(job['p'], t, sched, rng, job['label'] + '_s1', deadline, job['score'])
            if p2 is not None and p2.size > 0 and np.isfinite(sc2):
                job['p'] = p2
                job['score'] = sc2
            results.append(job)
            if np.isfinite(sc2) and sc2 < s_global:
                s_global = sc2
                p_global = job['p'].copy()
                try:
                    print(f"stage1 global update [{job['label']}] score={s_global:.9f} n={job['n']}")
                except Exception:
                    pass

        # Stage 2: longer runs on the best representatives.
        results.sort(key=_score_dict_key)
        selected = []
        used_n = set()
        for cand in results:
            if len(selected) >= 3:
                break
            if cand['n'] in used_n:
                continue
            selected.append(cand)
            used_n.add(cand['n'])

        has_global_n = False
        for item in selected:
            if item['n'] == p_global.size:
                has_global_n = True
                break
        if not has_global_n:
            selected.append({'p': p_global.copy(), 'score': s_global, 'n': int(p_global.size), 'label': 'global'})

        max_target_n = p_global.size
        for job in jobs:
            if job['n'] > max_target_n:
                max_target_n = job['n']

        now = time.monotonic()
        if p_global.size < max_target_n and float(deadline - now) > 70.0:
            q = _resample_p(p_global, max_target_n)
            if q is not None:
                qscore = _score_p(q)
                res_job = {'p': q.copy(), 'score': qscore, 'n': int(max_target_n), 'label': 'resample_best'}
                has_res_n = False
                for item in selected:
                    if item['n'] == max_target_n:
                        has_res_n = True
                        break
                if not has_res_n:
                    selected.append(res_job)

        rem2 = float(deadline - now)
        stage2_time = min(stage2_time, max(5.0, rem2 - 10.0))
        per2 = stage2_time / max(1, len(selected))
        for job in selected:
            nowr = time.monotonic()
            remr = float(deadline - nowr)
            t = min(per2, 80.0, max(3.0, remr - 5.0))
            if t <= 3.0:
                continue
            if t > 18.0:
                sched = _full_schedule()
            else:
                sched = _quick_schedule(t, job['n'])
            p2, sc2 = _run_schedule(job['p'], t, sched, rng, job['label'] + '_s2', deadline, job['score'])
            if p2 is not None and p2.size > 0 and np.isfinite(sc2):
                job['p'] = p2
                job['score'] = sc2
                if sc2 < s_global:
                    s_global = sc2
                    p_global = p2.copy()
                    try:
                        print(f"stage2 global update [{job['label']}] score={s_global:.9f} n={job['n']}")
                    except Exception:
                        pass

        # Final polish.
        now = time.monotonic()
        rem = float(deadline - now)
        if rem > 10.0:
            t = min(rem * 0.95, rem)
            p2, sc2 = _run_schedule(p_global, t, _polish_schedule(), rng, 'polish', deadline, s_global)
            if p2 is not None and p2.size > 0 and np.isfinite(sc2) and sc2 < s_global:
                s_global = sc2
                p_global = p2.copy()
                try:
                    print(f"polish global score={s_global:.9f} n={p_global.size}")
                except Exception:
                pass

        try:
            print(f"final length={p_global.size} final local score={s_global:.9f}")
        except Exception:
            pass

        return _make_output(p_global)

    except Exception as exc:
        try:
            print('search failed, using fallback:', exc)
        except Exception:
            pass
        p = _shape_to_p(_make_sqrt_series(8192))
        return _make_output(p)
