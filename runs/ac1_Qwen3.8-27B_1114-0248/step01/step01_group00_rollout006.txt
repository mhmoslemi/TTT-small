```python
"""
Search for nonnegative step-height sequences minimizing

    2 * n * max(conv(a, a)) / sum(a)**2

The objective is scale invariant.  This program searches mainly in the family

    a_i ~ x_i**(-p) * (1 - x_i)**r,   x_i in [0,1],

which is closely related to the finite truncations of (1-x)**(-1/2), but with
two free endpoint-exponent parameters.  It scouts this low-dimensional family
with FFT-based convolution scores, then refines the best candidates with a
multiplicative / mirror-descent local search using:
  * an L2 flattening direction for all convolution coefficients,
  * smooth active-peak subgradient directions, and
  * occasional random smooth perturbations to escape stalls.
"""

import numpy as np
import time


def next_pow2(m):
    """Smallest power of two >= m."""
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def target_for_n(n):
    """Normalization sum sqrt(2n)."""
    return float(np.sqrt(2.0 * float(n)))


def normalize_to_target(a, target):
    """Scale nonnegative vector to have the given sum."""
    a = np.asarray(a, dtype=np.float64).copy()
    if a.ndim != 1:
        a = a.ravel()
    n = a.size
    if n == 0:
        return a
    a = np.clip(a, 0.0, 1000.0)
    a = np.where(np.isfinite(a), a, 0.0)
    s = float(a.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, target / float(n), dtype=np.float64)
    return a * (target / s)


def center_clip_z(z, max_range=60.0):
    """Center logits in a safe dynamic range."""
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return z
    z = np.where(np.isfinite(z), z, 0.0).copy()
    z = z - float(z.mean())
    mn = float(np.min(z))
    mx = float(np.max(z))
    rng = mx - mn
    if not np.isfinite(rng):
        return np.zeros_like(z)
    if rng > max_range:
        z = z * (max_range / rng)
    return z


def stable_softmax_target(z, target):
    """Stable normalized softmax: returned vector has sum target."""
    z = np.asarray(z, dtype=np.float64)
    n = z.size
    if n == 0:
        return z
    z = np.where(np.isfinite(z), z, -1e300)
    z = z - float(np.max(z))
    ex = np.exp(z)
    s = float(ex.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, target / float(n), dtype=np.float64)
    return target * ex / s


def convolution_fft(a):
    """Full linear convolution a*a using a real FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)
    return c[:2 * n - 1]


def score_from(a, c):
    """Score from sequence a and its full convolution c."""
    n = a.size
    s = float(a.sum(dtype=np.float64))
    if n == 0 or not np.isfinite(s) or s <= 1e-300:
        return float("inf")
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    if m < 0.0:
        m = 0.0
    return float(2.0 * n * m / (s * s))


def score_only(a):
    """FFT-based objective value."""
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return float("inf")
    a = np.clip(a, 0.0, 1000.0)
    a = np.where(np.isfinite(a), a, 0.0)
    s = float(a.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 1e-300:
        return float("inf")
    c = convolution_fft(a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    if m < 0.0:
        m = 0.0
    return float(2.0 * a.size * m / (s * s))


def gradient_cross(weights, a, grad_size=None):
    """
    Gradient of sum_k weights[k] * (a*a)[k] with respect to a.

    derivative_m = 2 * sum_k weights[k] * a[k-m],
    where a[k-m] is valid only within [0, n-1].
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1
    w = np.asarray(weights, dtype=np.float64)
    if w.size < L:
        w = np.pad(w, (0, L - w.size), mode="constant")
    elif w.size > L:
        w = w[:L]
    w = np.where(np.isfinite(w), w, 0.0)
    if grad_size is None:
        grad_size = next_pow2(3 * n - 2)
    ar = a[::-1]
    W = np.fft.rfft(w, grad_size)
    AR = np.fft.rfft(ar, grad_size)
    conv = np.fft.irfft(W * AR, grad_size)
    start = n - 1
    g = 2.0 * conv[start:start + n]
    g = np.where(np.isfinite(g), g, 0.0)
    return g


def make_profile(n, p, r, shift=0.5):
    """
    Profile on [0,1]: g(x) = x**(-p) * (1-x)**r.

    Uses x_i = (i+shift)/n to avoid endpoint singularity problems.
    """
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    x = (np.arange(n, dtype=np.float64) + float(shift)) / float(n)
    y = 1.0 - x
    raw = np.ones(n, dtype=np.float64)
    with np.errstate(divide="ignore", invalid="ignore", over="ignore"):
        if abs(float(p)) > 1e-14:
            raw = raw * np.power(x, -float(p))
        if abs(float(r)) > 1e-14:
            raw = raw * np.power(y, float(r))
    raw = np.where(np.isfinite(raw), raw, 0.0)
    raw = np.maximum(raw, 0.0)
    s = float(raw.sum(dtype=np.float64))
    target = target_for_n(n)
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, target / float(n), dtype=np.float64)
    return raw * (target / s)


def make_sqrt_profile(n):
    """Central binomial square-root coefficients, normalized."""
    n = int(n)
    a = np.empty(n, dtype=np.float64)
    if n == 0:
        return a
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * float(2 * k - 1) / float(2 * k)
    return normalize_to_target(a, target_for_n(n))


def make_smooth_noise(rng, n):
    """Low-pass random field, normalized to zero mean unit variance."""
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    noise = rng.standard_normal(n)
    F = np.fft.rfft(noise)
    k = max(2, int(np.sqrt(n)))
    if k > F.size:
        k = F.size
    F2 = np.zeros_like(F)
    F2[:k] = F[:k]
    out = np.fft.irfft(F2, n)
    out = out - float(out.mean())
    sd = float(out.std())
    if np.isfinite(sd) and sd > 0.0:
        out = out / sd
    else:
        out = np.zeros(n, dtype=np.float64)
    return out


def make_perturbation(base, rng, sigma):
    """Multiply a base sequence by a smooth positive random factor."""
    base = np.asarray(base, dtype=np.float64)
    noise = make_smooth_noise(rng, base.size)
    cand = base * np.exp(float(sigma) * noise)
    cand = np.where(np.isfinite(cand), cand, 0.0)
    cand = np.maximum(cand, 0.0)
    target = target_for_n(base.size)
    return normalize_to_target(cand, target)


def parameter_grids():
    """Parameter grid for endpoint exponents."""
    p_extra = np.array([0.0, 0.05, 0.1, 0.15, 0.2, 0.25, 0.3], dtype=np.float64)
    p_main = np.linspace(0.30, 0.50, 41, dtype=np.float64)
    p_vals = np.unique(np.concatenate((p_extra, p_main)))
    r_vals = np.array(
        [-0.20, -0.10, 0.0, 0.05, 0.10, 0.15, 0.20, 0.30, 0.50],
        dtype=np.float64,
    )
    return p_vals, r_vals


def build_param_sweep(n, rng, top_count, deadline):
    """Evaluate profile grid at length n and return best params/sequences."""
    n = int(n)
    p_vals, r_vals = parameter_grids()
    scores = []
    seqs = []
    params = []

    for p in p_vals:
        if time.monotonic() > deadline:
            break
        for r in r_vals:
            if time.monotonic() > deadline:
                break
            seq = make_profile(n, float(p), float(r))
            sc = score_only(seq)
            if np.isfinite(sc):
                scores.append(float(sc))
                seqs.append(seq)
                params.append((float(p), float(r)))

    seq = make_sqrt_profile(n)
    sc = score_only(seq)
    if np.isfinite(sc):
        scores.append(float(sc))
        seqs.append(seq)
        params.append((0.5, 0.0))

    if not scores:
        return [], []

    scores = np.asarray(scores, dtype=np.float64)
    order = np.argsort(scores)
    top_params = []
    top_seqs = []
    for idx in order:
        if len(top_seqs) >= int(top_count):
            break
        idx = int(idx)
        top_params.append(params[idx])
        top_seqs.append((float(scores[idx]), seqs[idx]))
    return top_params, top_seqs


def build_refined_candidates(n, base_params, deadline, top_count):
    """Small local grid around best parameters at a longer length."""
    n = int(n)
    scores = []
    seqs = []
    params = []
    dp_vals = np.array([-0.03, -0.015, 0.0, 0.015, 0.03], dtype=np.float64)
    dr_vals = np.array([-0.10, -0.05, 0.0, 0.05, 0.10], dtype=np.float64)

    for p0, r0 in base_params[:6]:
        if time.monotonic() > deadline:
            break
        for dp in dp_vals:
            if time.monotonic() > deadline:
                break
            for dr in dr_vals:
                if time.monotonic() > deadline:
                    break
                p = float(p0) + float(dp)
                r = float(r0) + float(dr)
                if p < -0.05 or p > 0.80 or r < -0.80:
                    continue
                seq = make_profile(n, p, r)
                sc = score_only(seq)
                if np.isfinite(sc):
                    scores.append(float(sc))
                    seqs.append(seq)
                    params.append((p, r))

    if not scores:
        return [], []

    scores = np.asarray(scores, dtype=np.float64)
    order = np.argsort(scores)
    top_params = []
    top_seqs = []
    for idx in order:
        if len(top_seqs) >= int(top_count):
            break
        idx = int(idx)
        top_params.append(params[idx])
        top_seqs.append((float(scores[idx]), seqs[idx]))
    return top_params, top_seqs


def sort_pool(pool):
    """Sort candidate pool by score; discard invalid/infinite scores."""
    if not pool:
        return []
    scores = np.empty(len(pool), dtype=np.float64)
    for i in range(len(pool)):
        v = float(pool[i][0])
        scores[i] = v
    order = np.argsort(scores)
    out = []
    for idx in order:
        idx = int(idx)
        if np.isfinite(scores[idx]):
            out.append(pool[idx])
    return out


def sequences_close(a, b):
    """Cheap duplicate check for candidate sequences."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    if a.shape != b.shape:
        return False
    amax = float(np.max(np.abs(a))) if a.size else 0.0
    bmax = float(np.max(np.abs(b))) if b.size else 0.0
    scale = max(1.0, amax, bmax)
    diff = np.max(np.abs(a - b)) if a.size else 0.0
    return bool(diff <= 1e-8 * scale)


def select_unique_top(pool, k):
    """Take top k unique candidates."""
    sorted_pool = sort_pool(pool)
    selected = []
    for item in sorted_pool:
        if len(selected) >= int(k):
            break
        seq = np.asarray(item[1], dtype=np.float64)
        dup = False
        for sel in selected:
            if sequences_close(seq, sel[1]):
                dup = True
                break
        if not dup:
            selected.append(item)
    return selected


def local_iteration(a, z, c, score, mode, q, step, rng, n, conv_size, grad_size, target, noise_amp):
    """
    One line-search step of mirror descent.

    Returns:
      (a_new, z_new, c_new, score_new, step_chosen, improved)
    """
    if mode == "l2":
        s = float(a.sum(dtype=np.float64))
        avg = (s * s) / float(max(1, c.size))
        weights = c - avg
        weights = np.where(np.isfinite(weights), weights, 0.0)
    elif mode == "active":
        cmax = float(np.max(c)) if c.size else 0.0
        if not np.isfinite(cmax) or cmax <= 0.0:
            weights = np.ones_like(c, dtype=np.float64)
        else:
            ratio = np.clip(c / cmax, 0.0, 1.0)
            weights = np.power(ratio, float(q))
            weights = np.where(np.isfinite(weights), weights, 0.0)
    elif mode == "multi":
        cmax = float(np.max(c)) if c.size else 0.0
        if not np.isfinite(cmax) or cmax <= 0.0:
            weights = np.ones_like(c, dtype=np.float64)
        else:
            thr = cmax * 0.95
            w = np.maximum(c - thr, 0.0)
            w = np.power(w, float(max(1, int(q))))
            ws = float(w.sum(dtype=np.float64))
            if np.isfinite(ws) and ws > 0.0:
                w = w / ws
            else:
                w = np.ones_like(c, dtype=np.float64)
            weights = w
    else:
        return a, z, c, score, step, False

    wscale = float(np.max(np.abs(weights))) if weights.size else 0.0
    if not np.isfinite(wscale) or wscale <= 1e-300:
        return a, z, c, score, step, False

    g = gradient_cross(weights, a, grad_size)
    gscale = float(np.max(np.abs(g))) if g.size else 0.0
    if not np.isfinite(gscale) or gscale <= 1e-300:
        return a, z, c, score, step, False

    d = g / gscale
    d = d - float(d.mean())
    dscale = float(np.max(np.abs(d))) if d.size else 0.0
    if not np.isfinite(dscale) or dscale <= 1e-300:
        return a, z, c, score, step, False
    d = d / dscale

    if mode == "l2":
        factors = (0.10, 0.25, 0.50, 1.00, 2.00)
    else:
        factors = (0.25, 0.50, 1.00, 2.00, 4.00)

    best_a = a
    best_z = z
    best_c = c
    best_sc = score
    best_st = step
    improved = False

    for fac in factors:
        st = float(step) * float(fac)
        if st <= 0.0:
            continue
        zt = z - st * d
        if noise_amp > 0.0:
            zt = zt + float(noise_amp) * rng.standard_normal(n)
        zt = center_clip_z(zt)
        at = stable_softmax_target(zt, target)
        ct = convolution_fft(at)
        sc = score_from(at, ct)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_a = at
            best_z = zt
            best_c = ct
            best_sc = sc
            best_st = st
            improved = True

    if improved:
        return best_a, best_z, best_c, best_sc, best_st, True
    return a, z, c, score, step, False


def local_search(a_init, max_seconds, rng, hard_deadline, label, base_score):
    """
    Hybrid local search for one fixed length.

    Mixes L2 flatten and active-peak directions.
    """
    a = normalize_to_target(np.asarray(a_init, dtype=np.float64), target_for_n(a_init.size))
    n = int(a.size)
    target = target_for_n(n)
    if n == 0 or max_seconds <= 0.0:
        return a, score_only(a)

    conv_size = next_pow2(2 * n - 1)
    grad_size = next_pow2(3 * n - 2)

    a_safe = np.maximum(a, 1e-300)
    z = np.log(a_safe)
    z = center_clip_z(z)
    a = stable_softmax_target(z, target)

    c = convolution_fft(a)
    score = score_from(a, c)
    if not np.isfinite(score):
        return a, float("inf")

    best_a = a.copy()
    best_z = z.copy()
    best_score = float(score)

    # phases: (mode, q, fraction)
    phases = (
        ("l2", 1, 0.24),
        ("active", 3, 0.28),
        ("multi", 6, 0.16),
        ("active", 8, 0.20),
        ("l2", 1, 0.12),
    )

    now_print = 0.0
    try:
        print(f"[{label}] local start n={n} score={score:.9f} base={float(base_score):.9f}")
    except Exception:
        pass

    for mode, q, frac in phases:
        phase_budget = float(max_seconds) * float(frac)
        if phase_budget < 1.0:
            continue
        phase_end = min(float(hard_deadline) - 0.75, time.monotonic() + phase_budget)
        if phase_end <= time.monotonic():
            break

        if mode == "l2":
            step = 0.16
        elif mode == "multi":
            step = 0.24
        else:
            step = 0.32
            if int(q) >= 8:
                step = 0.22

        fail = 0
        it = 0

        while time.monotonic() < phase_end and time.monotonic() < float(hard_deadline):
            it += 1
            noise_amp = 0.0
            if fail > 18 and rng.random() < 0.25:
                noise_amp = 0.025

            (
                a_new,
                z_new,
                c_new,
                sc_new,
                st_new,
                improved,
            ) = local_iteration(
                a,
                z,
                c,
                score,
                mode,
                int(q),
                step,
                rng,
                n,
                conv_size,
                grad_size,
                target,
                noise_amp,
            )

            if improved:
                a = a_new
                z = z_new
                c = c_new
                score = sc_new
                step = min(2.2, 1.10 * float(st_new))
                fail = 0

                if score < best_score - 1e-10:
                    best_a = a.copy()
                    best_z = z.copy()
                    best_score = float(score)
                    now = time.monotonic()
                    if now - now_print > 5.0:
                        try:
                            print(f"[{label}] {mode} q={int(q)} improved best={best_score:.9f}")
                        except Exception:
                            pass
                        now_print = now
            else:
                step *= 0.70
                fail += 1
                if step < 0.02:
                    step = 0.16 if mode == "l2" else (0.24 if mode == "multi" else 0.30)

                if fail > 25:
                    # Reheat from best state by adding a smooth-ish random kick.
                    noise = 0.04 + 0.02 * min(1.0, float(fail) / 100.0)
                    zz = best_z.copy() + noise * rng.standard_normal(n)
                    zz = center_clip_z(zz)
                    aa = stable_softmax_target(zz, target)
                    cc = convolution_fft(aa)
                    ss = score_from(aa, cc)
                    if np.isfinite(ss):
                        a = aa
                        z = zz
                        c = cc
                        score = ss
                        step = 0.16 if mode == "l2" else (0.24 if mode == "multi" else 0.30)
                        fail = 0
                    else:
                        # Recover best and leave this phase early.
                        a = best_a.copy()
                        z = best_z.copy()
                        c = convolution_fft(a)
                        score = score_from(a, c)
                        break

    try:
        print(f"[{label}] local best n={best_a.size} score={best_score:.9f}")
    except Exception:
        pass
    return best_a, best_score


def get_height_sequence(max_len=16384):
    """Safely retrieve optional global height_sequence_1."""
    try:
        h = height_sequence_1
    except NameError:
        return None
    except Exception:
        return None

    try:
        arr = np.asarray(h, dtype=np.float64)
    except Exception:
        return None

    if arr.ndim != 1 or arr.size == 0:
        return None

    arr = np.clip(arr, 0.0, 1000.0)
    arr = np.where(np.isfinite(arr), arr, 0.0)

    if arr.size > int(max_len):
        old = np.linspace(0.0, 1.0, arr.size, dtype=np.float64)
        new = np.linspace(0.0, 1.0, int(max_len), dtype=np.float64)
        arr = np.interp(new, old, arr)
        arr = np.where(np.isfinite(arr), arr, 0.0)

    return arr


def safe_exact_score(sequence):
    """Call the externally defined evaluate_sequence safely."""
    try:
        items = [float(x) for x in sequence]
    except Exception:
        return float("inf")
    try:
        val = evaluate_sequence(items)
    except Exception:
        return float("inf")
    try:
        val = float(val)
    except Exception:
        return float("inf")
    if not np.isfinite(val):
        return float("inf")
    return val


def propose_candidate(seed=42, budget_s=400, **kwargs):
    """
    Search for a low-score nonnegative sequence within the time budget.
    """
    rng = np.random.default_rng(int(seed))
    start = time.monotonic()
    budget = float(budget_s)
    if budget <= 0.0 or not np.isfinite(budget):
        budget = 400.0
    hard = start + max(1.0, budget - 5.0)

    # Immediate fallback if teardown/exceptions occur.
    try:
        fallback = make_profile(4096, 0.45, 0.10)
        fallback = normalize_to_target(fallback, target_for_n(4096))
    except Exception:
        fallback = np.ones(1024, dtype=np.float64)

    pool = []

    try:
        try:
            print(f"budget={budget:.2f}s hard={max(0.0, hard - start):.2f}s")
        except Exception:
            pass

        # Optional previously found SOTA.
        h = get_height_sequence(max_len=16384)
        if h is not None:
            if h.size > 12000:
                old = np.linspace(0.0, 1.0, h.size, dtype=np.float64)
                new_len = 8192
                new = np.linspace(0.0, 1.0, new_len, dtype=np.float64)
                h = np.interp(new, old, h)
            h = normalize_to_target(h, target_for_n(h.size))
            sc = score_only(h)
            if np.isfinite(sc):
                pool.append((sc, h, "height"))
                try:
                    print(f"height candidate n={h.size} score={sc:.9f}")
                except Exception:
                    pass

        # Length selection.
        if budget > 220.0:
            n_scout = 4096
            n_final = 8192
        elif budget > 90.0:
            n_scout = 2048
            n_final = 4096
        else:
            n_scout = 1024
            n_final = 2048

        # Keep enough time for local search and final exact checks.
        gen_bank = min(60.0, 0.18 * budget)
        gen_deadline = min(hard, start + max(5.0, gen_bank))

        # Scout the profile family.
        top_params, top_seqs = build_param_sweep(n_scout, rng, top_count=14, deadline=gen_deadline)
        for sc, seq in top_seqs:
            pool.append((sc, seq, "scout"))

        # Promote promising parameters to the larger length.
        # A small refinement grid helps if the scout optimum is not captured exactly.
        if time.monotonic() < gen_deadline and top_params:
            ref_params, ref_seqs = build_refined_candidates(
                int(n_final), top_params, deadline=gen_deadline, top_count=12
            )
            for sc, seq in ref_seqs:
                pool.append((sc, seq, "refined"))

        if top_params:
            for p0, r0 in top_params[:12]:
                seq = make_profile(int(n_final), float(p0), float(r0))
                sc = score_only(seq)
                if np.isfinite(sc):
                    pool.append((sc, seq, "final_profile"))

        # Classic square-root profile at final length.
        seq = make_sqrt_profile(int(n_final))
        sc = score_only(seq)
        if np.isfinite(sc):
            pool.append((sc, seq, "sqrt_final"))

        # Smooth perturbations of the current best candidates.
        pool_sorted = sort_pool(pool)
        if pool_sorted:
            base_candidates = select_unique_top(pool_sorted, 3)
            for sc, seq, label in base_candidates:
                for sigma in (0.05, 0.10, 0.20):
                    cand = make_perturbation(seq, rng, sigma)
                    sc2 = score_only(cand)
                    if np.isfinite(sc2):
                        pool.append((sc2, cand, f"perturb_{label}_{sigma}"))

        pool_sorted = sort_pool(pool)
        if not pool_sorted:
            seq = fallback
            try:
                print("search produced no candidates; using fallback")
            except Exception:
                pass
            out = normalize_to_target(seq, target_for_n(seq.size))
            return [float(x) for x in out]

        # Local search top few candidates.
        local_reserve = min(25.0, 0.10 * budget)
        local_hard = max(gen_deadline, hard - local_reserve)
        top_local = select_unique_top(pool, 5)

        if top_local and time.monotonic() + 25.0 < local_hard:
            fractions = (0.38, 0.27, 0.20, 0.13, 0.06)
            local_total = max(0.0, min(local_hard - time.monotonic(), hard - 12.0 - time.monotonic()))
            for i, (sc, seq, label) in enumerate(top_local):
                if i >= len(fractions):
                    break
                t = local_total * float(fractions[i])
                t = min(t, local_hard - time.monotonic() - 1.0)
                if t <= 5.0:
                    break
                try:
                    new_seq, new_score = local_search(
                        seq,
                        float(t),
                        rng,
                        local_hard,
                        f"{label}_local",
                        float(sc),
                    )
                except Exception:
                    continue
                if np.isfinite(new_score):
                    pool.append((new_score, new_seq, f"local_{label}"))
                    pool = sort_pool(pool)

        # Choose a few top candidates for exact evaluation.
        pool_sorted = sort_pool(pool)
        exact_candidates = pool_sorted[:8]
        best_exact_seq = None
        best_exact_score = float("inf")
        exact_done = 0
        max_exact_long = 4 if len(exact_candidates) > 0 and np.asarray(exact_candidates[0][1]).size > 8000 else 6

        for sc, seq, label in exact_candidates:
            if time.monotonic() > hard - 2.0:
                break
            if exact_done >= max_exact_long:
                break
            val = safe_exact_score(seq)
            exact_done += 1
            if val < best_exact_score:
                best_exact_score = val
                best_exact_seq = np.asarray(seq, dtype=np.float64).copy()
                try:
                    print(f"exact {label} n={seq.size} score={val:.9f}")
                except Exception:
                    pass

        # Final sequence: exact-evaluated best if available, otherwise best FFT score.
        if best_exact_seq is not None and best_exact_score < float("inf"):
            out = best_exact_seq
        else:
            if pool_sorted:
                out = np.asarray(pool_sorted[0][1], dtype=np.float64).copy()
            else:
                out = fallback

        # Final sanitation and normalization.
        out = np.asarray(out, dtype=np.float64).ravel()
        if out.size == 0:
            out = np.ones(1024, dtype=np.float64)
        out = np.clip(out, 0.0, 1000.0)
        out = np.where(np.isfinite(out), out, 0.0)
        out = normalize_to_target(out, target_for_n(out.size))
        out = np.clip(out, 0.0, 1000.0)

        try:
            fs = score_only(out)
            print(f"final length={out.size} fft_score={fs:.9f} exact_best={best_exact_score:.9f}")
        except Exception:
            pass

        return [float(x) for x in out]

    except Exception:
        try:
            print("propose_candidate error; returning fallback")
        except Exception:
            pass
        out = normalize_to_target(fallback, target_for_n(fallback.size))
        return [float(x) for x in out]
```
