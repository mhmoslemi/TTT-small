def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:

        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero

        return np.inf

def _next_p2(m):
    """Smallest power of two >= m."""
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _center_clip_z(z):
    """Center logits and prevent excessive dynamic range."""
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return z
    z = z - np.mean(z)
    mn = float(np.min(z))
    mx = float(np.max(z))
    rng = mx - mn
    if not np.isfinite(rng):
        return np.zeros_like(z)
    if rng > 60.0:
        z = z * (60.0 / rng)

def _softmax_target(z, target_sum):
    """Stable softmax scaled to sum to target_sum."""
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return z
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, -1e300)
    z = z - np.max(z)
    ex = np.exp(z)
    s = float(ex.sum(dtype=np.float64))
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.shape, target_sum / z.size, dtype=np.float64)
    return target_sum * ex / s


def _make_sqrt_series(n):

    Coefficients of (1-x)**(-1/2):

    Their infinite convolution is 1 + x + x^2 + ..., so truncated versions

    if a.size == 0:
        return a
    a[0] = 1.0
    for k in range(1, a.size):

def _conv_fft(a, size):
    """Linear convolution a*a via FFT."""
    A = np.fft.rfft(a, size)
    C = A * A
    out = np.fft.irfft(C, size)
    return out[:2 * int(a.size) - 1]


def _score_from_c(a, c):
    """Exact objective from a and its convolution coefficients."""
    if a.size == 0 or c.size == 0:
        return np.inf
    S = float(a.sum(dtype=np.float64))
    S2 = S * S
    if not np.isfinite(S2) or S2 <= 0.0:
        return np.inf
    cmax = float(np.max(c))
    if not np.isfinite(cmax):
        return np.inf
    if cmax <= 0.0:
        cmax = 0.0
    return float(2.0 * a.size * cmax / S2)

def _power_weights(c, q):
    """
    Smooth peak weights p_k proportional to (max-c ratio)^q.
    p sums to 1 and is nonnegative.
    """
    c = np.maximum(c, 0.0)
    L = int(c.size)
    if L == 0:
        return np.array([], dtype=np.float64)
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.full(L, 1.0 / L, dtype=np.float64)
    scaled = c / cmax
    q = max(1, int(q))
    w = scaled ** q
    ws = float(w.sum(dtype=np.float64))
    if not np.isfinite(ws) or ws <= 0.0:
        idx = int(np.argmax(c))
        w = np.zeros(L, dtype=np.float64)
        w[idx] = 1.0
        return w
    w /= ws
    return w


def _gradient(a, weights, grad_size, allow_negative=False):
    """
    Gradient of sum_k weights[k] * (a*a)[k] with respect to a.

    For J = sum_k w_k c_k:
        dJ/da_m = 2 * sum_j a_j w_{m+j}.

    Computed as a real convolution with reversed weights.
    """
    n = int(a.size)
    L = int(weights.size)
    rw = weights[::-1]
    A = np.fft.rfft(a, grad_size)
    P = np.fft.rfft(rw, grad_size)
    conv = np.fft.irfft(A * P, grad_size)

    start = L - 1
    stop = start - n
    r = conv[start:stop:-1]
    if r.size != n:
        # Fallback for unusual length interactions.
        r = conv[max(0, L - n):L]
        r = r[-n:]

    g = 2.0 * r
    if not allow_negative:
        g = np.maximum(g, 0.0)
    g = np.where(np.isfinite(g), g, 0.0)
    return g

def _run_job(a_init, max_seconds, mode_schedule, rng, label, hard_deadline, baseline_score):
    """
    Run several phases on the same job, keeping only improvements.

    mode_schedule: list of (mode, fraction), e.g. [(-1, 0.2), (8, 0.5), (32, 0.3)]
    """
    seq = np.asarray(a_init, dtype=np.float64).copy()
    best_score = baseline_score

    for mode, frac in mode_schedule:
        t = float(max_seconds) * float(frac)
        t = min(t, float(hard_deadline - time.monotonic()))
        if t <= 1.0:
            continue
        seq_new, score_new = _optimize_fixed_mode(seq, t, int(mode), rng, label)
        if np.isfinite(score_new) and score_new < best_score:
            best_score = score_new
            seq = np.asarray(seq_new, dtype=np.float64).copy()
    return seq, best_score
