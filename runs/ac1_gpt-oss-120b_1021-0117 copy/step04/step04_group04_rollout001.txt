import time
import numpy as np

def _uniform_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence of given length with sum equal 1."""
    if length == 0:
        return np.array([], dtype=np.float64)
    return np.full(length, 1.0 / length, dtype=np.float64)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients that (approximately)
    minimises the AC1 bound evaluated by ``evaluate_sequence``.
    A simple stochastic hill‑climb starting from a uniform distribution is used.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.4)

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 0.0:
            cur = a0 / s                 # normalise to sum = 1
        else:
            cur = _uniform_sequence(a0.size)
    else:
        n = 2000
        cur = _uniform_sequence(n)

    n = cur.size
    # ---- evaluate initial candidate ----------------------------------
    best = cur.copy()
    best_score = evaluate_sequence(best.tolist())
    if not np.isfinite(best_score):
        best = _uniform_sequence(n)
        best_score = evaluate_sequence(best.tolist())
    cur = best.copy()
    cur_score = best_score

    # ---- simple stochastic hill‑climb -------------------------------
    it = 0
    no_improve = 0
    # parameters controlling the search intensity
    max_no_improve = 8000
    transfer_frac = 0.45   # maximum fraction of mass moved in one step

    while time.monotonic() < deadline:
        it += 1
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while j == i:
            j = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        # move a random fraction of mass from i to j
        delta = cur[i] * rng.random() * transfer_frac
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluate candidate
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                no_improve = 0
        else:
            no_improve += 1

        # if we are stuck, perform a random perturbation of the whole vector
        if no_improve > max_no_improve:
            # add small Gaussian noise, keep non‑negative, renormalise
            noise = rng.normal(scale=0.02, size=n)
            cand = cur + noise
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0.0:
                cand /= s
            cand_score = evaluate_sequence(cand.tolist())
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
            no_improve = 0

        # occasional symmetrisation (helps for many autocorrelation problems)
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = np.maximum(sym, 0.0)
            s = sym.sum()
            if s > 0.0:
                sym /= s
            sym_score = evaluate_sequence(sym.tolist())
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score

    # return the best vector found as plain Python floats
    return [float(x) for x in best]
