import time

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for sum(a)=1."""
    n = a.shape[0]

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

                    no_improve = 0
                else:
                    # restart from a fresh random point
                    cur = rng.random(n)
                    cur = np.maximum(cur, 1e-12)
                    cur /= cur.sum()
                    cur_sc = _score(cur)
                    no_improve = 0

        # ---- early exit ------------------------------------------------
        if best_sc < target:
            break

    # Return plain Python list (float) as required
    return [float(x) for x in best_vec]
