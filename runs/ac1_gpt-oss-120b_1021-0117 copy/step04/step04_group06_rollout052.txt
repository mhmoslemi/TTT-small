import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast score (sum of coefficients is assumed to be 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, conv: np.ndarray | None = None) -> float:
    n = a.shape[0]
    if conv is None:
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv))

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative coefficient list with a low AC1 bound.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # -------------------- initial vector --------------------

        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: exponential decay, length 1024
        n_fallback = 1024
        a = np.exp(-np.arange(n_fallback, dtype=np.float64) / 20.0)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                     # simplex projection (sum=1)

    n = a.shape[0]
    # FFT length sufficient for a*a and a*w convolutions (3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    # -------------------- bookkeeping --------------------
    best_vec = a.copy()
    best_sc = _score(best_vec)
    cur = a.copy()
    cur_sc = best_sc

    # hyper‑parameters
    step0 = 0.8
    eps0 = 0.5
    eps_decay = 0.999
    eps_min = 1e-8
    beta = 0.9                     # momentum coefficient
    m = np.zeros_like(a)           # momentum accumulator
    no_improve = 0
    it = 0

    # -------------------- optimisation loop --------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current iterate -----
        a_fft = np.fft.rfft(cur, n=L)

        # self‑convolution (objective)
        conv_self = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_self.max()

        # ----- smoothed weights (soft‑max) -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv_self - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:

        else:

        sym_sc = _score(sym)

    # Return a plain Python list
    return [float(x) for x in best_vec]
