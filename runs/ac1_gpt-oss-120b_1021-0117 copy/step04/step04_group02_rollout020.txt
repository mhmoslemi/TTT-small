import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound for a sequence with sum≈1
# ------------------------------------------------------------
def _score_vec(v: np.ndarray, L: int, n: int) -> float:
    """Return 2*n*max(convolution) where sum(v)=1."""
    af = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------

    """Return a non‑negative step‑function (list of floats) with low AC1 bound."""
    rng = np.random.default_rng(seed)

    if "height_sequence_1" in globals():

    else:

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                         # now sum(a)=1

    n = a.shape[0]
    # FFT length for convolutions (next power of two >= 2n-1)
    L = 1 << ((2 * n - 1).bit_length())

    cur = a.copy()

    best_sc = cur_sc

    # ------------------------------------------------------------

    # return as plain Python list
    return [float(x) for x in best_vec]
