import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Helper: evaluate the AC1 bound (identical to the external scorer)
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    """2 * n * max(conv(vec,vec))  (vec is assumed to sum to 1)."""
    n = vec.size
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    af = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 upper bound."""
    start = time.monotonic()
    # keep a tiny safety margin so we can return before the sandbox kills us
    deadline = start + max(0.0, float(budget_s) - 0.3)

    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # initialise – use the provided parent if available,
    # otherwise start from a uniform distribution.
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.clip(cur, 0.0, 1000.0)
        cur = _project_simplex(cur)          # enforce sum = 1
    else:
        n = 1024
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    n = cur.size
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # --------------------------------------------------------
    # tiny stochastic refinement (matrix‑free, O(n log n) per step)
    # --------------------------------------------------------
    it = 0
    while time.monotonic() < deadline and it < 3000:
        # pick two distinct indices and move a random fraction of mass
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] == 0.0:
            it += 1
            continue
        delta = cur[i] * rng.random() * 0.25
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)

        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
        if cand_sc < _score(cur):
            cur = cand
        it += 1

        # occasional symmetrisation (helps for Toeplitz structure)
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()
                cur = sym

    # --------------------------------------------------------
    # Return the best candidate found (plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
