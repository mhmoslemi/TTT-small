import time

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast objective evaluation (assumes a is a real vector)
# ------------------------------------------------------------
def _score_fast(a: np.ndarray) -> float:
    """Return 2*n*max(conv(a))/sum(a)^2 using FFT (no Python loops)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    maxb = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * maxb / (s * s))
