import time
import math
import numpy as np

# ------------------------------------------------------------
# Fast AC1 upper‑bound (sum is assumed to be 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2·n·max(conv(a,a)) for a normalized to sum = 1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative coefficient list with a low AC1 bound.
    The routine works by repeatedly moving a small amount of mass from the
    entry that contributes most to the current maximal convolution value to a
    random other entry, accepting improvements (and occasional uphill moves)
    and gradually shrinking the step size.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    # -------------------- initial vector --------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / a.size)
    else:
        n = 1024
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.shape[0]
    # tiny random perturbation to break symmetry, then renormalise
    a += 1e-6 * (rng.random(n) - 0.5)
    a = np.clip(a, 0.0, None)
    a = a / a.sum()

    # FFT length sufficient for a full linear convolution
    L = 1 << ((2 * n - 1).bit_length())

    # bookkeeping
    cur = a.copy()
    cur_sc = _score(cur, L)
    best = cur.copy()
    best_sc = cur_sc

    # hyper‑parameters
    step_frac = 0.5          # fraction of mass moved each step
    min_frac = 1e-5
    no_improve = 0
    it = 0

    # -------------------- optimisation loop --------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and the lag achieving the maximum -----
        af = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_idx = int(conv.argmax())

        # ----- contributions of each pair to that lag -----
        i_start = max(0, max_idx - (n - 1))
        i_end = min(max_idx, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = max_idx - idx
        contrib = cur[idx] * cur[partner]

        # ----- choose a heavy contributor and a random target -----
        i_max = idx[int(contrib.argmax())]
        j = rng.integers(0, n)
        while j == i_max:
            j = rng.integers(0, n)

        # ----- move a fraction of mass from i_max to j -----
        delta = cur[i_max] * step_frac
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i_max] -= delta
        cand[j] += delta

        # ----- evaluate candidate -----
        cand_sc = _score(cand, L)

        # ----- accept if improvement, else occasional Metropolis step -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) /
                            (step_frac * 1e-3 + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- slowly shrink the step size -----
        step_frac = max(min_frac, step_frac * 0.9999)

        # ----- escape from stagnation with a random mass transfer -----
        if no_improve > 3000:
            i0 = rng.integers(0, n)
            i1 = rng.integers(0, n)
            if i0 != i1:
                delta2 = cur[i0] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i0] -= delta2
                tmp[i1] += delta2
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < 1.50274349:
            break

    # -------------------- final jitter refinement --------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = cur[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp_sc = _score(tmp, L)
        if tmp_sc < best_sc:
            best, best_sc = tmp.copy(), tmp_sc
            cur, cur_sc = tmp, tmp_sc

    # return plain Python list
    return [float(x) for x in best]
