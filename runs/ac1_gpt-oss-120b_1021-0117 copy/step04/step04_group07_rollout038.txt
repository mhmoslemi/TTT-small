import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score  = 2 * n * max(conv(a,a))   (a is assumed to sum to 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int, n: int) -> float:
    """Fast evaluation of the AC1 bound for a vector summing to 1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of the smoothed max (soft‑max) of the convolution.
# Returns (max_conv, gradient w.r.t. a).
# ------------------------------------------------------------
def _softmax_grad(a: np.ndarray, L: int, n: int, eps: float):
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    maxc = conv.max()
    # soft‑max weights
    w = np.exp((conv - maxc) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient via convolution with reversed weights
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return maxc, g

    if s <= 0.0:

        if sc_add < sc_mul:
            cand, cand_sc = cand_add, sc_add
        else:
            cand, cand_sc = cand_mul, sc_mul

        # ----- accept/reject (simple downhill + occasional uphill) -----
        cur_sc = _score(cur, L, n)
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12)):
            cur = cand
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            no_improve += 1

        # ----- occasional symmetrisation (problem is nearly symmetric) -----
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = sym / sym.sum()
            sym_sc = _score(sym, L, n)
            if sym_sc < cur_sc:
                cur = sym
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta_mass = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta_mass
                tmp[j] += delta_mass
                tmp = tmp / tmp.sum()
                tmp_sc = _score(tmp, L, n)
                if tmp_sc < cur_sc:
                    cur = tmp
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if target already met -----
        if best_sc < target:
            break

    # return the best vector found (as a plain Python list)
    return [float(x) for x in best]
