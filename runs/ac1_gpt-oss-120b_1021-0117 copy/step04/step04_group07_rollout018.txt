import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Compute the AC1 bound (sum is assumed to be 1)
# ------------------------------------------------------------
def _score_and_conv(a: np.ndarray, L: int) -> tuple[float, np.ndarray]:
    """Return (score, full convolution) for a (sum(a)=1)."""
    n = a.size
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max()), conv


# ------------------------------------------------------------
# Local optimizer (Adam + smoothing) starting from a0
# ------------------------------------------------------------
def _local_opt(a0: np.ndarray, L: int, target: float,
               deadline: float, max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """Run a short Adam‑style descent from a0, stop early if target met."""
    n = a0.size
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.12

    # smoothing for the soft‑max approximation of max(conv)
    eps = 0.01
    eps_min = 1e-8
    eps_decay = 0.9995

    # keep the best found inside this run
    best_local = a.copy()
    best_sc = _score_and_conv(a, L)[0]

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # convolution of a with itself
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # gradient of the smoothed max
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
