# ------------------------------------------------------------
# Simple renormalisation onto the probability simplex (sum=1)
# ------------------------------------------------------------
def _simplex_normalise(x: np.ndarray) -> np.ndarray:
    """Clip negatives, then renormalise to have sum = 1."""
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s <= 0.0:
        # fallback uniform
        n = x.shape[0]
        return np.full_like(x, 1.0 / n)
    return x / s

def _score(a: np.ndarray, L: int = None) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
