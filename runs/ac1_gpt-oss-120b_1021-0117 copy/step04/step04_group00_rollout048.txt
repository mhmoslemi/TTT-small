import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Scoring function used by the evaluator (sum is fixed to 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, conv_max: float = None) -> float:
    """Return 2 * n * max(convolution(a,a)) for a simplex‑normalised a."""
    n = a.shape[0]
    if conv_max is None:
        L = 1 << ((3 * n - 2).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        conv_max = conv.max()
    return float(2.0 * n * conv_max)


# ------------------------------------------------------------
# Gradient of the smoothed max (soft‑max temperature = eps)
# ------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, eps: float, L: int):
    """Compute gradient of the soft‑max approximation of max(conv)."""
    n = a.size
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    # soft‑max weights
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum
    # gradient = 2 * (a * w) convolution, efficiently via FFT
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g, max_val

    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # modest length if no parent is supplied
        a0 = rng.random(2048, dtype=np.float64)
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)          # enforce sum = 1
    cur = a0.copy()
    n = cur.size
    L = 1 << ((3 * n - 2).bit_length())   # FFT length (power of two)

    # --------------------------------------------------------
    # Optimisation state
    # --------------------------------------------------------
    best_vec = cur.copy()
    # compute once the max convolution for the initial vector
    af0 = np.fft.rfft(cur, n=L)
    conv0 = np.fft.irfft(af0 * af0, n=L)[: 2 * n - 1]
    best_sc = _score(cur, conv0.max())
    cur_sc = best_sc

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # Soft‑max temperature schedule
    eps0 = 1.0
    eps_decay = 0.9995
    eps_min = 1e-8

    # Misc. control
    no_improve = 0
    it = 0
    target = 1.50274349
    step0 = 0.8          # base learning‑rate

    # --------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iter)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- gradient of the current smoothed objective ----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        g, conv_max = _smooth_gradient(cur, eps, L)

        # ---- Adam moment updates ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ---- tentative step (with line‑search) ----
        alpha = step0 / math.sqrt(it)     # simple decay
        # Adam‑style direction
        direction = m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = None
        cand_sc = None
        # up to 8 back‑tracking attempts
        for _ in range(8):
            cand_tmp = cur - alpha * direction
            cand_tmp = _project_simplex(cand_tmp)
            # evaluate candidate
            af = np.fft.rfft(cand_tmp, n=L)
            conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
            cand_tmp_sc = _score(cand_tmp, conv.max())
            if cand_tmp_sc <= cur_sc:
                cand, cand_sc = cand_tmp, cand_tmp_sc
                break
            alpha *= 0.5
        if cand is None:
            # accept the last tried candidate even if not improving
            cand, cand_sc = cand_tmp, cand_tmp_sc

        # ---- accept / reject ----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            # occasional uphill move (Metropolis style)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ---- periodic symmetrisation (helps due to Toeplitz structure) ----
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            af = np.fft.rfft(sym, n=L)
            conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
            sym_sc = _score(sym, conv.max())
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # ---- random mass‑transfer when stuck ----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                af = np.fft.rfft(tmp, n=L)
                conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
                tmp_sc = _score(tmp, conv.max())
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0
                else:
                    no_improve = 0

        # ---- early exit if target reached ----
        if best_sc < target:
            break

    # Return best candidate (as plain Python list)
    return [float(x) for x in best_vec]
