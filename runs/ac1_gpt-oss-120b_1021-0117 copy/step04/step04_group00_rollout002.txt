def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    # initialize
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2048
        a = rng.random(n, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)

    n = a.size
    L = 1 << ((2*n - 1).bit_length())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    base_lr = 0.4
    eps0 = 0.5
    eps_decay = 0.9997
    eps_min = 1e-4

    best_a = a.copy()
    best_sc = _score(best_a)
    cur_sc = best_sc
    iter = 0
    stagnation = 0

    while time.monotonic() < deadline:
        iter += 1
        # compute convolution and max
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # compute softmax weights
        eps = max(eps_min, eps0 * (eps_decay ** iter))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum
        # gradient via convolution with w_rev
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = grad_full[:2*n-1]
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**iter)
        v_hat = v / (1 - beta2**iter)
        step = -base_lr / math.sqrt(iter) * m_hat / (np.sqrt(v_hat) + adam_eps)

        cand = a + step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        if cand_sc < cur_sc - 1e-12:
            a, cur_sc = cand, cand_sc
            stagnation = 0
            if cand_sc < best_sc:
                best_a, best_sc = cand.copy(), cand_sc
        else:
            stagnation += 1

        # occasional symmetry
        if iter % 300 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                a, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_a, best_sc = sym.copy(), sym_sc
                stagnation = 0

        # random perturbations after stagnation
        if stagnation > 2000:
            # apply several random moves
            for _ in range(5):
                i = rng.integers(0, n)
                if a[i] <= 0:
                    continue
                delta = a[i] * rng.random() * 0.3
                j = rng.integers(0, n)
                # ensure j != i
                while j == i:
                    j = rng.integers(0, n)
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    a, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_a, best_sc = tmp.copy(), tmp_sc
                    stagnation = 0
                    break
            else:
                stagnation = 0  # reset anyway

        # time check early exit
        if best_sc < 1.50274349:
            break

    # final fallback ensure sum>=0.01; but already normalized
    return [float(x) for x in best_a]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast evaluation of the AC1 upper bound (sum is assumed to be 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    The algorithm uses an Adam‑style projected gradient descent with
    soft‑max smoothing of the max‑convolution, occasional symmetrisation,
    and random mass‑transfer moves to escape plateaus.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin before the hard sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a random start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a modest length when no parent is supplied
        n0 = 2048
        a = rng.random(n0, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)                # enforce sum = 1, non‑negative
    n = a.size

    # ------------------------------------------------------------
    # FFT planning – length at least 2n‑1 (next power of two)
    # ------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------
    # Optimisation hyper‑parameters
    # ------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999          # Adam moments
    adam_eps = 1e-8
    base_lr = 0.4                     # base learning rate (scaled by 1/√t)
    eps0 = 0.5                        # initial soft‑max temperature
    eps_decay = 0.9997
    eps_min = 1e-4

    m = np.zeros_like(a)              # first moment
    v = np.zeros_like(a)              # second moment

    best_vec = a.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    it = 0
    stagn = 0
    target = 1.50274349

    # ------------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution a*a and its max -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weights (smooth approximation of the max) -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective: 2 * a * w -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = grad_full[: 2 * n - 1]
        idx = (2 * n - 2) - np.arange(n)          # reverse slice to align
        g = 2.0 * grad[idx]

        # ----- Adam update (projected) -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = -base_lr / math.sqrt(it) * m_hat / (np.sqrt(v_hat) + adam_eps)

        cand = a + step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        if cand_sc < cur_sc - 1e-12:
            a, cur_sc = cand, cand_sc
            stagn = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            stagn += 1

        # ----- occasional symmetrisation (helps with Toeplitz structure) -----
        if it % 300 == 0:

            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                a, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                stagn = 0

        if stagn > 2000:

            improved = False
            for _ in range(5):
                i = rng.integers(0, n)
                if a[i] <= 0.0:

                    improved = True
                    stagn = 0
                    break
            if not improved:
                stagn = 0   # reset counter even if no improvement

        # ----- early exit if target achieved -----
        if best_sc < target:
            break

    # Return the best candidate discovered as plain Python floats
    return [float(x) for x in best_vec]
