import time, math, numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Compute auto‑convolution (full) and its maximum via FFT
# ------------------------------------------------------------
def _conv_max_and_fft(a: np.ndarray, L: int):
    """Return (max(conv), a_fft, full_conv) for a*a."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """Return a non‑empty, non‑negative sequence with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for cleanup
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------ initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: modest uniform vector
        n_fallback = 2000
        a0 = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s

    n = a0.size
    # symmetrise start (the problem is almost symmetric)
    a0 = (a0 + a0[::-1]) * 0.5
    a0 = _project_simplex(a0)

    # FFT length enough for convolution of length n with length 2n‑1
    L = 1 << ((3 * n - 2).bit_length())

    # ------ optimiser state ------------------------------------------------
    cur = a0.copy()
    maxc, cur_fft, cur_conv = _conv_max_and_fft(cur, L)
    cur_sc = float(2 * n * maxc)

    best = cur.copy()
    best_sc = cur_sc

    # Adam moments
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.5

    # smoothing for soft‑max
    eps = 0.1
    eps_min = 1e-9
    eps_decay = 0.9995

    it = 0
    no_improve = 0
    sym_interval = 250   # how often to enforce symmetry
    restart_patience = 8000

    while time.monotonic() < deadline:
        it += 1

        # ----- gradient (soft‑max or sub‑gradient) -------------------------
        maxc, a_fft, conv = _conv_max_and_fft(cur, L)

        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)          # slice for gradient
            g = 2.0 * grad_full[idx]
            eps = max(eps_min, eps * eps_decay)
        else:
            # sub‑gradient from all maximal indices
            max_idxs = np.where(conv == maxc)[0]
            g = np.zeros_like(cur)
            for k in max_idxs:
                start = max(0, k - (n - 1))
                end = min(k + 1, n)
                idx_range = np.arange(start, end)
                g[idx_range] += 2.0 * cur[k - idx_range]

        # ----- Adam update -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)

        # occasional symmetrisation (keeps search in the symmetric subspace)
        if it % sym_interval == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        cand_maxc, _, _ = _conv_max_and_fft(cand, L)
        cand_sc = float(2 * n * cand_maxc)

        # ----- acceptance ---------------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:

            T = max(0.01, 1.0 - it / 50000.0)

                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic global symmetrisation of current iterate ----------
        if it % (sym_interval * 2) == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_maxc, _, _ = _conv_max_and_fft(sym, L)
            sym_sc = float(2 * n * sym_maxc)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # ----- restart if stuck --------------------------------------------

            rand_vec = rng.random(n) ** 2
            rand_vec = _project_simplex(rand_vec)
            cur = 0.5 * best + 0.5 * rand_vec
            cur = _project_simplex(cur)

            m.fill(0.0)
            v.fill(0.0)
            maxc, _, _ = _conv_max_and_fft(cur, L)
            cur_sc = float(2 * n * maxc)
            no_improve = 0

        # ----- early exit if target reached --------------------------------
        if best_sc < 1.50274349:
            break

    # Return the best sequence found as plain Python list
    return [float(x) for x in best]
