def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total_budget = float(budget_s)
    deadline = start + max(0.0, total_budget - 0.5)

    # build initial candidates
    candidates = []
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.maximum(base, 0.0)
        base = _project_simplex(base)
        candidates.append(base)
    # uniform
    n = candidates[0].size if candidates else 1024
    uniform = np.full(n, 1.0/n)
    candidates.append(uniform)
    # arcsine distribution
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0/np.sqrt((i+0.5)*(n-0.5-i))
    arcsine = arcsine / arcsine.sum()
    candidates.append(arcsine)
    # tent shape
    center = (n-1)/2
    dist = np.abs(np.arange(n) - center)
    tent = np.maximum(0.0, (center - dist))
    tent = tent / tent.sum()
    candidates.append(tent)

    # pick best initial
    best_vec = None
    best_score = float('inf')
    for cand in candidates:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
    cur = best_vec.copy()
    cur_score = best_score

    n = cur.size
    L = 1 << ((2*n-1).bit_length())
    # parameters
    eps = 5.0
    eps_decay = 0.9995
    eps_min = 1e-8
    base_lr = 1.0
    it = 0
    no_improve = 0
    best_overall = cur.copy()
    best_overall_score = cur_score

    while time.monotonic() < deadline:
        it += 1
        # compute conv and max
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()

        # update epsilon
        eps = max(eps_min, eps * eps_decay)

        # softmax weights
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # normalized step
        grad_norm = np.linalg.norm(grad) + 1e-12
        lr = base_lr / math.sqrt(it) / grad_norm

        # try backtracking
        accepted = False
        cand = None
        cand_score = None
        for attempt in range(5):
            cand_tmp = cur - lr * grad
            cand_tmp = _project_simplex(cand_tmp)
            # enforce symmetry
            cand_tmp = (cand_tmp + cand_tmp[::-1]) * 0.5
            cand_tmp = _project_simplex(cand_tmp)

            cand_tmp_score = _score(cand_tmp)
            if cand_tmp_score < cur_score:
                cand = cand_tmp
                cand_score = cand_tmp_score
                accepted = True
                break
            else:
                lr *= 0.5  # shrink step

        if accepted:
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_overall_score:
                best_overall_score = cand_score
                best_overall = cand.copy()
        else:
            no_improve += 1

        # random perturb if stuck
        if no_improve > 3000:
            i_idx = rng.integers(0, n)
            if cur[i_idx] > 0:
                delta = cur[i_idx] * rng.random() * 0.3
                j_idx = rng.integers(0, n)
                if j_idx == i_idx:
                    j_idx = (j_idx + 1) % n
                tmp = cur.copy()
                tmp[i_idx] -= delta
                tmp[j_idx] += delta
                tmp = _project_simplex(tmp)
                # symmetry
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                tmp_score = _score(tmp)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    no_improve = 0
                    if tmp_score < best_overall_score:
                        best_overall_score = tmp_score
                        best_overall = tmp.copy()
                else:
                    no_improve = 0  # reset anyway

        if best_overall_score < 1.50274349:
            break

    # convert to list
    return [float(x) for x in best_overall]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- normalized step size -----
        grad_norm = np.linalg.norm(grad) + 1e-12
        lr = base_lr / math.sqrt(it) / grad_norm

        # ----- backtracking line search (up to 5 attempts) -----
        improved = False
        for _ in range(5):
            cand = cur - lr * grad
            cand = _project_simplex(cand)
            cand = (cand + cand[::-1]) * 0.5          # enforce symmetry
            cand = _project_simplex(cand)
            cand_sc = _score(cand)
            if cand_sc < cur_sc:
                improved = True
                break
            lr *= 0.5

        if improved:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_overall_sc:
                best_overall_sc = cand_sc
                best_overall = cand.copy()
        else:
            no_improve += 1

        # ----- occasional random perturbation when stuck -----
        if no_improve > 3000:
            i_idx = rng.integers(0, n)
            if cur[i_idx] > 0.0:
                delta = cur[i_idx] * rng.random() * 0.3
                j_idx = rng.integers(0, n)
                if j_idx == i_idx:
                    j_idx = (j_idx + 1) % n
                tmp = cur.copy()
                tmp[i_idx] -= delta
                tmp[j_idx] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_overall_sc:
                        best_overall_sc = tmp_sc
                        best_overall = tmp.copy()
                else:
                    no_improve = 0

        # ----- early exit if target reached -----
        if best_overall_sc < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Return the best vector found (plain Python list)
    # ------------------------------------------------------------------
    if best_overall_sc == float("inf"):
        best_overall = np.full(n, 1.0 / n, dtype=np.float64)
    return [float(x) for x in best_overall]
