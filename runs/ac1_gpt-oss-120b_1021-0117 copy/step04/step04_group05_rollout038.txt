import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        # exponential decay fallback (length 1024)
        n_fb = 1024
        cur = np.exp(-np.arange(n_fb, dtype=np.float64) / (n_fb / 10.0))
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)                     # enforce sum == 1

    n = cur.shape[0]

    # FFT parameters (power‑of‑two lengths)
    L_full = 1 << ((3 * n - 2).bit_length())        # for gradient (needs ≥ 3n‑2)
    idx_slice = (2 * n - 2) - np.arange(n)          # mapping used for gradient

    # initial statistics
    cur_fft = np.fft.rfft(cur, n=L_full)
    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
    cur_max = float(cur_conv.max())
    cur_score = 2.0 * n * cur_max                    # exact value returned by evaluator

    best_vec = cur.copy()
    best_score = cur_score

    # ------------------- optimisation parameters ------------
    step0 = 1.2                                       # initial step magnitude
    eps_rel = 0.015                                   # for smoothed max
    eps_min = 1e-7
    it = 0
    no_improve = 0
    target = 1.50274349                               # known benchmark

    # ------------------- main loop ---------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- smoothed‑max weights (soft max) -------------
        eps = max(eps_min, cur_max * eps_rel)
        w = np.exp((cur_conv - cur_max) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective ----------
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L_full)
        grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L_full)
        g = 2.0 * grad_full[idx_slice]

        # ----- line search ---------------------------------
        alpha = step0 / math.sqrt(it)
        accepted = False
        cand = None
        for _ in range(8):                     # up to 8 back‑track attempts
            cand = cur - alpha * g
            cand = _project_simplex(cand)
            cand_fft = np.fft.rfft(cand, n=L_full)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L_full)[: 2 * n - 1]
            cand_max = float(cand_conv.max())
            cand_score = 2.0 * n * cand_max
            if cand_score < cur_score:
                accepted = True
                break
            alpha *= 0.5

        if accepted:
            cur, cur_fft, cur_conv = cand, cand_fft, cand_conv
            cur_max, cur_score = cand_max, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
        else:
            # ----- simulated‑annealing acceptance -------------
            prob = math.exp(-(cand_score - cur_score) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_fft, cur_conv = cand, cand_fft, cand_conv
                cur_max, cur_score = cand_max, cand_score
                no_improve = 0
            else:
                no_improve += 1

        # ----- occasional symmetrisation (helps) ----------
        if it % 3000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L_full)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L_full)[: 2 * n - 1]
            sym_max = float(sym_conv.max())
            sym_score = 2.0 * n * sym_max
            if sym_score < cur_score:
                cur, cur_fft, cur_conv = sym, sym_fft, sym_conv
                cur_max, cur_score = sym_max, sym_score
                no_improve = 0
                if sym_score < best_score:
                    best_score = sym_score
                    best_vec = sym.copy()

        # ----- random mass‑transfer when stuck -------------
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L_full)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L_full)[: 2 * n - 1]
                tmp_max = float(tmp_conv.max())
                tmp_score = 2.0 * n * tmp_max
                if tmp_score < cur_score:
                    cur, cur_fft, cur_conv = tmp, tmp_fft, tmp_conv
                    cur_max, cur_score = tmp_max, tmp_score
                if tmp_score < best_score:
                    best_score = tmp_score
                    best_vec = tmp.copy()
                no_improve = 0

        # ----- restart from best with jitter if very stuck ----
        if no_improve > 12000:
            jitter = rng.normal(scale=0.005, size=n)
            cur = best_vec + jitter
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            cur_fft = np.fft.rfft(cur, n=L_full)
            cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
            cur_max = float(cur_conv.max())
            cur_score = 2.0 * n * cur_max
            no_improve = 0

        # ----- early exit if target achieved -----------------
        if best_score < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found as a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
