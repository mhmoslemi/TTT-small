    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    while time.monotonic() < deadline:
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # tolerance
        tol = max_val * 1e-12 + 1e-14
        mask = conv >= max_val - tol

            w = np.full_like(conv, 1.0/conv.size)
        else:

            w[mask] = 1.0 / count

        w_rev = w[::-1]

import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        if tmp_sc < best_sc:
            best_vec, best_sc = tmp, tmp_sc
            continue
        # occasional symmetry test
        sym = (tmp + tmp[::-1]) * 0.5
        sym = _project_simplex(sym)
        sym_sc = _score(sym)
        if sym_sc < best_sc:
            best_vec, best_sc = sym, sym_sc

    # return plain Python list
    return [float(x) for x in best_vec]
