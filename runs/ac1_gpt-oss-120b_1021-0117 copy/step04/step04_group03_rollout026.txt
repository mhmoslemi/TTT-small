import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex using the O(n) algorithm."""
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast scoring (assumes sum(a)=1, otherwise the factor is applied)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int, n: int) -> float:
    """Return 2 * n * max(convolution) (sum(a) is assumed 1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    start = time.monotonic()
    total_budget = float(budget_s)
    # keep a small safety margin for the harness
    deadline = start + max(0.0, total_budget - 0.5)

    # --------------------------------------------------------

    # explore exponents around -0.5
    p_vals = np.linspace(-1.2, 0.2, 20)

    L = 1 << ((2 * n - 1).bit_length())   # FFT length (power of two >= 2n-1)

    best_vec = base.copy()

        cand_sum = cand.sum()
        if cand_sum == 0.0:

        cand = cand / cand_sum                     # already on simplex

        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()

    cur = best_vec.copy()

    # --------------------------------------------------------
