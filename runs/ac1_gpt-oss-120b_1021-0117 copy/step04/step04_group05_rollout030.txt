import time

# --------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x) = 1})
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    conv = np.fft.irfft(vf * vf, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max()), conv, vf
