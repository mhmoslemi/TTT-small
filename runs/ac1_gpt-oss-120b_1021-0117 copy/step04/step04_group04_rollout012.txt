import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast score based on FFT (2 * n * max convolution)
# ------------------------------------------------------------
def _score_fft(a: np.ndarray, L_conv: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L_conv)
    conv = np.fft.irfft(af * af, n=L_conv)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradiant of the smooth‑max (log‑sum‑exp) of the convolution
# ------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, L_big: int, eps: float) -> np.ndarray:
    """Return ∇(log‑sum‑exp) of a*a convolution."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L_big)
    # full convolution a*a (length 2n‑1) inside the big FFT space
    conv = np.fft.irfft(a_fft * a_fft, n=L_big)[: 2 * n - 1]
    max_val = conv.max()
    w = np.exp((conv - max_val) / eps)
    ws = w.sum()
    if ws == 0.0:
        w[:] = 1.0 / w.size
    else:
        w /= ws
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L_big)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L_big)
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # spare margin

    # ------------------- initialise ------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # simple analytic start – exponential decay
        n0 = 2500
        a = np.exp(-np.arange(n0) / 250.0)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)                     # sum == 1
    n = a.shape[0]

    # FFT lengths
    L_conv = 1 << ((2 * n - 1).bit_length())    # for a*a convolution
    L_big = 1 << ((3 * n - 2).bit_length())    # for gradient (a * w_rev)

    # best solution tracking
    best = a.copy()
    best_sc = _score_fft(best, L_conv)

    cur = best.copy()
    cur_sc = best_sc

    # ------------------- Phase 1: smoothed gradient descent ----
    eps0 = 0.8
    eps_decay = 0.9996
    eps_min = 1e-8
    step0 = 0.5
    it = 0
    # allocate at most 20 % of the total budget for this phase
    gd_deadline = start + 0.2 * (deadline - start)

    while time.monotonic() < gd_deadline:
        it += 1
        eps = max(eps_min, eps0 * (eps_decay ** it))
        g = _smooth_gradient(cur, L_big, eps)

        # additive update with diminishing step size
        grad_norm = np.linalg.norm(g) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm
        cand = cur - alpha * g

        # enforce non‑negativity and simplex constraint
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        # optional symmetrisation (helps stability)
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_sc = _score_fft(cand, L_conv)

        # acceptance rule (pure descent with occasional uphill)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            # Metropolis‑like acceptance with temperature tied to eps
            prob = math.exp(-(cand_sc - cur_sc) / (eps + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # early exit if target is already beaten
        if best_sc < 1.50274349:
            return [float(x) for x in best]

    # ------------------- Phase 2: simulated annealing ------------
    # Temperature schedule
    T = 0.02
    T_end = 1e-6
    decay = 0.99995
    it_sa = 0

    while time.monotonic() < deadline:
        # propose a small mass transfer
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] <= 0.0:
            continue
        # move a random fraction (up to 40 %) of mass from i to j
        delta = cur[i] * rng.random() * 0.4
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # enforce symmetry (averaging with reversed copy)
        cand = (cand + cand[::-1]) * 0.5
        # clip tiny negatives and renormalise to keep sum≈1
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        cand_sc = _score_fft(cand, L_conv)

        # acceptance
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc

        # cool down
        T = max(T_end, T * decay)
        it_sa += 1

        # occasional global perturbation to escape deep plateaus
        if it_sa % 200000 == 0:
            # random Dirichlet‑like restart (uniform + simplex projection)
            tmp = rng.random(n)
            tmp = _project_simplex(tmp)
            tmp_sc = _score_fft(tmp, L_conv)
            if tmp_sc < best_sc:
                best, best_sc = tmp.copy(), tmp_sc
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc

        # early termination if we already beat the target
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------
    return [float(x) for x in best]
