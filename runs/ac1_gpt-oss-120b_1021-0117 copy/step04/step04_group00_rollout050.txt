        base = np.maximum(base, 0.0)
        base = _project_simplex(base)
        best_vec = base.copy()
        best_sc = _score(best_vec)

        # Pre-compute FFT length
        L = 1 << ((3 * n - 2).bit_length())

        # Adam hyperparameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-8
        lr = 0.2   # base learning rate

        # schedule parameters
        eps0 = 5.0
        eps_decay = 0.9998
        eps_min = 1e-6

        # main optimization loop until time out
        it = 0
        m = np.zeros_like(base)
        v = np.zeros_like(base)
        cur = base.copy()
        cur_sc = best_sc
        no_improve = 0
        target = 1.50274349

        while time.monotonic() < deadline:
            it += 1

            # compute convolution
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()

            # compute softmax weights
            eps = max(eps_min, eps0 * (eps_decay ** it))
            # avoid overflow/underflow
            shifted = conv - max_val
            # if eps is too small may cause overflow in exp; clip
            w = np.exp(shifted / eps)
            w_sum = w.sum()

            else:
                w /= w_sum

            # gradient
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]

            m = beta1 * m + (1.0 - beta1) * g

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score used during optimisation (assumes sum = 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a, a)) for a summing to 1."""
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())
