def evaluate_sequence(sequence: list) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

## Mathematical Specification
### Representation
Sequences are represented as non-negative step-function sequences of length \(n\). Each sequence is defined by at most \(k\) consecutive intervals, where each interval contains a constant value (the height of the step). The sequence is formally defined by:
- Breakpoints: indices \( p_0, p_1, \ldots, p_{k-1} \) dividing the domain into intervals.
- Heights: values \( h_0, h_1, \ldots, h_{k-1} \) assigned to each interval.

The sequence can be flattened into a 1D array by repeating each interval's value for the sublength of the interval.

\[
\text{AC1} = \frac{2n \cdot \max(\text{convolve}(a, a))}{(\sum a)^2}
