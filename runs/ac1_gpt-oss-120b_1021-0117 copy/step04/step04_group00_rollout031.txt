import time, math, numpy as np

def _project_simplex(v):
    """Project onto simplex (>=0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """2 * n * max(convolution(a,a)). Assumes sum(a)=1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    rng = np.random.default_rng(seed)
    start = time.monotonic()

    else:
        # fallback length 1024 random
        a = rng.random(1024, dtype=np.float64)
    a = np.maximum(a, 0.0)
    # ensure positivity to avoid log(0)
    a = np.maximum(a, 1e-15)
    a = _project_simplex(a)

    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())

    best_vec = a.copy()
    best_sc = _score(best_vec)

    # algorithm parameters
    eps0 = 2.0
    eps_decay = 0.9998
    eps_min = 1e-6
    eta0 = 0.9  # base step for exponentiated gradient
    target = 1.50274349
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:

        else:
            w /= w_sum

        w_rev = w[::-1]

        eta = eta0 / math.sqrt(it)

        # stabilize exponentials

import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]

    else:

        theta = css[rho] / (rho + 1)
