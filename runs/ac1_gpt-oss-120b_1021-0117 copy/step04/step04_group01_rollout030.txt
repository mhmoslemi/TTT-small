import time
import math
import numpy as np

# ----- Euclidean projection onto the probability simplex -----
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ----- AC1 upper‑bound (without the denominator) -----
def _score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())

# ----- Main optimisation routine -----
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # geometric decay fallback (length ≈2000)
        n_fallback = 2000
        r = 0.998
        a = np.power(r, np.arange(n_fallback), dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size, dtype=np.float64)
    else:
        a = a / s                     # normalise to sum = 1

    n = a.size
    L = 1 << ((3 * n - 2).bit_length())   # FFT length covering all convolutions

    # ---- bookkeeping ------------------------------------------------
    cur = a.copy()
    cur_sc = _score(cur, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ---- hyper‑parameters -------------------------------------------
    step0 = 0.8                      # base step size
    backtrack_factor = 0.7
    max_backtrack = 15
    no_improve = 0
    it = 0
    idx_slice = (2 * n - 2) - np.arange(n)   # indices to extract gradient entries

    # ---- main optimisation loop ------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution of current iterate
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # active shifts where the maximum is attained
        active = np.where(np.isclose(conv, max_val, atol=1e-12))[0]
        # construct weight vector w (average over active shifts)
        w = np.zeros(2 * n - 1, dtype=np.float64)

        w_fft = np.fft.rfft(w_rev, n=L)

        g = 2.0 * grad_full[idx_slice]

        # step size schedule
