import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Exact AC1 score (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Subgradient of max(conv) w.r.t. a (sum(a)=1)
# ------------------------------------------------------------
def _subgradient(a: np.ndarray, L: int, eps_mask: float = 1e-12) -> np.ndarray:
    """
    Returns 2 * sum_{k in Argmax} shift(a, k) using FFTs.
    L must be a power of two >= 3*len(a)-2.
    """
    n = a.size
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    # mask of all (near) maximal entries
    mask = np.where(conv >= max_val - eps_mask, 1.0, 0.0)

    # FFTs for the gradient
    mf = np.fft.rfft(mask, n=L)
    a_rev = a[::-1]
    a_rev_f = np.fft.rfft(a_rev, n=L)
    grad_full = np.fft.irfft(mf * a_rev_f, n=L)

    # first n entries, reversed, times 2
    grad = 2.0 * grad_full[:n][::-1]
    return grad
