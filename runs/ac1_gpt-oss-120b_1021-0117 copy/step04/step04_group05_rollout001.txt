def propose_candidate(seed=42, budget_s=1100, **kwargs):
    set start, deadline = start + budget_s - safety.

    create rng.

    if height_sequence_1 in globals():
        cur = np.array(height_sequence_1, dtype=float64)
    else:
        # else small fallback: maybe use exponential.
        n_fallback = 1024
        cur = np.exp(-np.arange(n_fallback)/ (n_fallback/10))

    cur = np.maximum(cur, 0)
    cur = _project_simplex(cur)

    n = cur.shape[0]
    L_full = 1 << ((3*n - 2).bit_length())
    # performance: compute FFT lengths maybe huge (e.g., n=2535 => 3*n-2 = 7603; next pow2 = 8192). So L_full = 8192.

    cur_fft = np.fft.rfft(cur, n=L_full)

    # compute conv and score
    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[:2*n-1]
    cur_max = cur_conv.max()
    cur_score = 2.0 * n * cur_max

    best_vec = cur.copy()
    best_score = cur_score

    # optimizer parameters
    step0 = 0.5   # base learning rate scaling; may adjust
    eps_rel = 0.02
    eps_min = 1e-8

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    t_iter = 0
    no_improve = 0

    # loop
    while time.monotonic() < deadline:
        t_iter += 1

        # smoothing epsilon
        eps = max(eps_min, cur_max * eps_rel)

        # compute weight softmax
        w = np.exp((cur_conv - cur_max) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # compute gradient via convolution
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L_full)
        grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L_full)
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # update Adam moments
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g**2)
        # bias correction
        m_hat = m / (1 - beta1**t_iter)
        v_hat = v / (1 - beta2**t_iter)

        # step size
        # use decreasing base learning rate: step0 / sqrt(t)
        lr = step0 / math.sqrt(t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # candidate update
        cand = cur - step
        # enforce nonneg and sum to 1 via projection onto simplex
        cand = _project_simplex(cand)

        # symmetrize (optional)
        if t_iter % 1000 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        # evaluate candidate
        cand_fft = np.fft.rfft(cand, n=L_full)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L_full)[:2*n-1]
        cand_max = cand_conv.max()
        cand_score = 2.0 * n * cand_max

        # accept if better
        if cand_score < cur_score:
            cur = cand
            cur_fft = cand_fft
            cur_conv = cand_conv
            cur_max = cand_max
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
        else:
            # maybe accept with small probability (simulated annealing)
            # compute temperature decaying with iter time
            temp = max(0.01, eps)
            prob = math.exp(-(cand_score - cur_score) / temp)
            if rng.random() < prob:
                cur = cand
                cur_fft = cand_fft
                cur_conv = cand_conv
                cur_max = cand_max
                cur_score = cand_score
                no_improve = 0
            else:
                no_improve += 1

        # occasional random mass transfers when stuck
        if no_improve > 2000:
            # choose random i,j with i != j
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i: j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L_full)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L_full)[:2*n-1]
                tmp_score = 2.0 * n * tmp_conv.max()
                if tmp_score < cur_score:
                    cur = tmp
                    cur_fft = tmp_fft
                    cur_conv = tmp_conv
                    cur_max = tmp_conv.max()
                    cur_score = tmp_score
                if tmp_score < best_score:
                    best_score = tmp_score
                    best_vec = tmp.copy()
                no_improve = 0

        # occasionally enforce symmetry
        if t_iter % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L_full)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L_full)[:2*n-1]
            sym_score = 2.0 * n * sym_conv.max()
            if sym_score < cur_score:
                cur = sym
                cur_fft = sym_fft
                cur_conv = sym_conv
                cur_max = sym_conv.max()
                cur_score = sym_score
                no_improve = 0
                if sym_score < best_score:
                    best_score = sym_score
                    best_vec = sym.copy()

        # early exit if target reached
        if best_score < 1.50274349:
            break

    # return list
    return [float(x) for x in best_vec]

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
