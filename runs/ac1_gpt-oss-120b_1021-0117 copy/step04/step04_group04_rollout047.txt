def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initial a
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        a0 = np.exp(-np.arange(n0)/200.0)

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)
    n = a0.shape[0]

    L = 1 << ((3*n - 2).bit_length())

    # helpers for convolution
    def full_conv(x):
        xf = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(xf*xf, n=L)[:2*n-1]
        return conv

    def grad_from_weights(a, w):
        # w length 2n-1
        w_rev = w[::-1]
        a_fft = np.fft.rfft(a, n=L)
        w_fft = np.fft.rfft(w_rev, n=L)
        prod = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n-2) - np.arange(n)
        return 2.0 * prod[idx]

    # evaluate using fast compute: using a_fft conv then compute max.
    # but external evaluate_sequence is used for final scoring; internal compute faster.
    def score_fast(a):
        conv = full_conv(a)
        return float(2.0 * n * conv.max())

    # initialize best:
    cur = a0.copy()
    best = cur.copy()
    best_sc = score_fast(best)
    cur_sc = best_sc

    # hyperparams
    eps0 = 1.5
    eps_decay = 0.9999
    eps_min = 1e-10

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.3   # base learning rate
    iter_count = 0
    no_improve = 0

    # main loop
    while time.monotonic() < deadline:
        iter_count += 1

        # compute conv
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]

        max_val = conv.max()
        # compute smoothed weights
        eps = max(eps_min, eps0 * (eps_decay ** iter_count))
        # subtract max for numerical stability
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        grad = grad_from_weights(cur, w)

        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad * grad)
        # bias correction
        m_hat = m / (1 - beta1**iter_count)
        v_hat = v / (1 - beta2**iter_count)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = score_fast(cand)
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best = cand.copy()
                best_sc = cand_sc
        else:
            no_improve += 1
            # occasional acceptance of worse for exploration, but not necessary.
            # we could random accept with small prob
            if rng.random() < 0.001:
                cur = cand
                cur_sc = cand_sc

        # periodic symmetrisation
        if iter_count % 3000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = score_fast(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best = sym.copy()
                    best_sc = sym_sc

        # random mass transfer when stuck
        if no_improve > 5000:
            # random perturbation: swap small mass between two indices
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                if j == i:
                    j = (j + 1) % n
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = score_fast(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best = tmp.copy()
                    best_sc = tmp_sc
                no_improve = 0

        # restart when stuck far longer
        if iter_count % 20000 == 0 and no_improve > 10000:
            # reinitialize around best with noise
            noise = rng.normal(scale=0.01, size=n)
            cand = best + noise
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            cur = cand
            cur_sc = score_fast(cur)
            m.fill(0)
            v.fill(0)
            no_improve = 0

        # early exit if target achieved
        if best_sc < 1.50274349:
            break

    return [float(x) for x in best]

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            noise = rng.normal(scale=0.01, size=n)
            cand = best + noise
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            cur = cand
            cur_sc = _fast_score(cur, L, n)
            m.fill(0)
            v.fill(0)
            no_improve = 0

        # early exit if we have already beaten the target
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Return the best found sequence as a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best]
