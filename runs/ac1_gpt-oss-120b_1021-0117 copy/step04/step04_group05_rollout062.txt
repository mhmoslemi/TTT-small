import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function whose AC1 bound is small."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # margin for safety

    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

        n0 = 1500
        x = np.arange(n0, dtype=np.float64)
        cur = np.exp(-x / (n0 / 10.0))

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)       # enforce sum == 1
    n = cur.shape[0]

    # ---- FFT helper lengths -----------------------------------------
    # convolution length = 2*n-1; we use a power of two a little larger.
    L = 1 << ((2 * n - 1).bit_length() + 1)

    # pre‑compute FFT of current vector
    cur_fft = np.fft.rfft(cur, n=L)
    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]

    cur_sc = 2.0 * n * cur_max          # sum == 1 ⇒ denominator = 1
    best_vec = cur.copy()

    # ---- optimisation parameters ------------------------------------
