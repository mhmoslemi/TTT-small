                    best_vec = tmp.copy()

    return [float(x) for x in best_vec]

import time, math, numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, >=0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# One‑step LP refinement (sparse convolution constraints)
# ------------------------------------------------------------
def _lp_refine(a: np.ndarray, max_val: float, deadline: float):
    """Solve max Σg subject to (a*g) ≤ max_val, g≥0. Return scaled g and its score."""
    n = a.shape[0]
    # build sparse Toeplitz matrix: each diagonal d (0..n-1) has constant a[d]
    data = [np.full(n, a[d]) for d in range(n)]
    offsets = np.arange(n, dtype=int)
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    ub = np.full(2 * n - 1, max_val)
    c = -np.ones(n)                         # minimise -Σg  <=> maximise Σg
    try:
        # HiGHS handles sparse problems efficiently
        res = opt.linprog(
            c,
            A_ub=A,
            b_ub=ub,
            bounds=(0, None),
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
        )
    except Exception:
        return None, None
    if not res.success:
        return None, None
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return None, None
    g_scaled = g / sum_g
    # score of the scaled vector (sum(g_scaled)=1)
    conv = np.convolve(g_scaled, g_scaled)
    new_sc = 2.0 * n * conv.max()
    return g_scaled, new_sc

    best_vec = cur.copy()
