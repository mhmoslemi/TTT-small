import time
import math
import numpy as np

# ------------------------------------------------------------
# Utility: projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound for a vector that sums to 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(conv) where conv = a * a (full convolution)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total_budget = float(budget_s)
    deadline = start + max(0.0, total_budget - 0.5)   # small safety margin

    # --------------------------------------------------------
    # Base vector (parent if available, otherwise uniform)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base = np.maximum(base, 0.0)
    else:
        n = 1024
        base = np.full(n, 1.0 / n, dtype=np.float64)

    base = _project_simplex(base)          # ensure sum = 1
    n = base.size

    # --------------------------------------------------------
    # Construct a few analytic basis shapes
    # --------------------------------------------------------
    i = np.arange(n, dtype=np.float64)

    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcsine = arcsine / arcsine.sum()

    half_cos = np.sin(np.pi * (i + 0.5) / n)
    half_cos = half_cos / half_cos.sum()

    tri = np.minimum(i + 1, n - i)          # symmetric triangular window
    tri = tri / tri.sum()

    basis = [uniform, arcsine, half_cos, tri, base]

    # --------------------------------------------------------
    # Quick initial scan of the basis vectors
    # --------------------------------------------------------
    best_vec = None
    best_sc = float("inf")
    for vec in basis:
        sc = _score(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()

    # --------------------------------------------------------
    # Random convex combinations of the basis vectors
    # --------------------------------------------------------
    K = len(basis)
    B = np.vstack(basis)                    # shape (K, n)
    combo_deadline = start + total_budget * 0.07
    while time.monotonic() < combo_deadline:
        w = rng.random(K)
        w /= w.sum()
        cand = w @ B
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()
            if best_sc < 1.50274349:       # early stop if target reached
                break

    # --------------------------------------------------------
    # Fallback safety
    # --------------------------------------------------------
    if best_vec is None:
        best_vec = uniform.copy()
        best_sc = _score(best_vec)

    # --------------------------------------------------------
    # Smoothed‑max gradient descent
    # --------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc

    best_overall = cur.copy()
    best_overall_sc = cur_sc

    L = 1 << ((2 * n - 1).bit_length())    # FFT length for convolutions
    eps = 1.0
    eps_decay = 0.9999
    eps_min = 1e-8
    step0 = 1.0
    beta = 0.9                               # momentum coefficient
    vel = np.zeros_like(cur)

    it = 0
    no_improve = 0
    while time.monotonic() < deadline:
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # smoothed weights for the current max
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        vel = beta * vel + (1.0 - beta) * g
        grad_norm = np.linalg.norm(vel) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm

        cand = cur - alpha * vel
        cand = _project_simplex(cand)

        # occasional symmetrisation
        if it % 400 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_overall_sc:
                best_overall, best_overall_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            if no_improve > 5000:
                # random perturbation around the best known vector
                tmp = best_overall + 0.02 * rng.normal(size=n) * best_overall
                tmp = np.clip(tmp, 0.0, None)
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_overall_sc:
                    best_overall, best_overall_sc = tmp.copy(), tmp_sc

        # decay smoothing parameter
        eps = max(eps_min, eps * eps_decay)

        # early exit if we already beat the target
        if best_overall_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # Return the best vector found (as a plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_overall]
