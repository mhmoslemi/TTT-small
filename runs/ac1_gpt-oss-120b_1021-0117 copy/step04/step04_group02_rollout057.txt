import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound for a sequence with sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence whose sum is ≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Subgradient of the max‑convolution term at index k*
# ------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Subgradient of (a*a)[k] w.r.t. a.
    For each j, grad_j = 2*a[k-j] iff 0<=k-j<n else 0.
    """
    n = a.size
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[idx[mask]]
    return g

            eps = max(eps * eps_decay, min_eps)

        # ----- compute gradient -----
        if eps > max_val * 1e-3:            # smoothed gradient
            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]
        else:                               # exact subgradient of the max term
            g = _subgradient_maxconv(cur, max_idx)

        # ----- step size (decaying) -----
        g_norm = np.linalg.norm(g) + 1e-12
        alpha = base_alpha / (math.sqrt(it) * g_norm)

        # ----- momentum‑augmented update (Nesterov style) -----
        v_new = mu * v - alpha * g
        cand = cur + v_new
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        improved = False

        if cand_sc < cur_sc:          # accept momentum step
            cur, cur_sc, v = cand, cand_sc, v_new
            improved = True
        else:
            # ----- back‑tracking line‑search without momentum -----
            alpha_try = alpha
            for _ in range(5):
                cand2 = cur - alpha_try * g
                cand2 = _project_simplex(cand2)
                cand2_sc = _score(cand2)
                if cand2_sc < cur_sc:
                    cur, cur_sc = cand2, cand2_sc
                    v = np.zeros_like(cur)   # reset momentum
                    improved = True
                    break
                alpha_try *= 0.5

        # ----- occasional Metropolis acceptance (rare) -----
        if not improved:
            # try a tiny stochastic move
            if rng.random() < 0.01:
                delta = 0.005 * rng.random()
                dir_vec = rng.normal(size=n)
                dir_vec -= dir_vec.mean()   # keep sum ≈ 0
                cand3 = cur + delta * dir_vec
                cand3 = _project_simplex(cand3)
                cand3_sc = _score(cand3)
                if cand3_sc < cur_sc:
                    cur, cur_sc = cand3, cand3_sc
                    v = np.zeros_like(cur)
                    improved = True

        # ----- bookkeeping -----
        if improved:
            no_improve = 0
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_vec = cur.copy()
        else:
            no_improve += 1

        # ----- periodic symmetrisation (helps a lot) -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                v = np.zeros_like(cur)
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    v = np.zeros_like(cur)
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- early exit if target achieved -----
        if best_sc < improve_target:
            break

    # Return as plain list of Python floats
    return [float(x) for x in best_vec]
