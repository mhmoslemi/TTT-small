import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients with a low AC1 bound.
    The routine prefers the supplied parent `height_sequence_1` if it exists,
    otherwise falls back to a uniform step function of the same length.
    A tiny local search then tries to improve the candidate within the time budget.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine problem size and obtain a parent candidate (if any)
    # ------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if parent is not None:
        a_parent = np.asarray(parent, dtype=np.float64).copy()
        a_parent = np.clip(a_parent, 0.0, 1000.0)
        n = a_parent.size
    else:
        # fallback length – modest but enough to be non‑trivial
        n = 1024
        a_parent = None

    # ------------------------------------------------------------------
    # Uniform candidate (sum = 1) – mathematically optimal for the given bound
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_vec = uniform
    best_sc = evaluate_sequence(best_vec.tolist())

        if parent_sc < best_sc:
            best_vec = a_parent
            best_sc = parent_sc

    # ------------------------------------------------------------------
    # Simple symmetrisation – often reduces the peak of the autocorrelation
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        sym = (best_vec + best_vec[::-1]) * 0.5
        sym_sc = evaluate_sequence(sym.tolist())
        if sym_sc < best_sc:

            best_sc = sym_sc

    if time.monotonic() < deadline:
