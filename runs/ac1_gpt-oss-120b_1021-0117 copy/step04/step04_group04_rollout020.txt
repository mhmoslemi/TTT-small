def propose_candidate(seed=42, budget_s=1100, **kwargs):

    import time, numpy as np

    start = time.time()

    n = None

    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2000  # default length

    # Uniform distribution
    
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # optionally we can attempt simple local modifications for a few iterations given time.

    best = a.copy()
    best_score = evaluate_sequence(best.tolist())

    # Simple random hill climbing: small perturbations of two positions preserving sum.

    rng = np.random.default_rng(seed)
    deadline = start + max(0.0, float(budget_s) - 0.2)  # some safety

    # We'll allow up to maybe 20000 attempts or time budget.
    
    attempts = 0
    while time.time() < deadline and attempts < 20000:
        attempts += 1
        # pick two indices i, j distinct.
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a small amount delta from i to j
        delta = rng.random() * 0.1 * best[i]  # up to 10% of i
        if delta == 0.0:
            continue
        if best[i] < delta:
            delta = best[i] * 0.5
        candidate = best.copy()
        candidate[i] -= delta
        candidate[j] += delta
        # never negative, but ensure clip
        if candidate[i] < 0:
            continue
        # evaluate
        score = evaluate_sequence(candidate.tolist())
        if score < best_score:
            best = candidate
            best_score = score

    # Return best as list
    return [float(x) for x in best]

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex (sum == 1, non‑negative)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
