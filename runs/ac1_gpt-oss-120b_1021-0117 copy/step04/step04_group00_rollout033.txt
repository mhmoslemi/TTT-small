import time
import numpy as np

# ----------------------------------------------------------------------
# Simple helper: fast evaluation of the objective (sum is assumed to be 1)
# ----------------------------------------------------------------------
def _objective(vec: np.ndarray) -> float:
    """Return 2 * n * max(conv(vec,vec))  (vec is non‑negative, sum≈1)."""
    n = vec.size
    # length for a linear convolution via FFT (next power of two)
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence that (approximately) minimises
    the AC1 upper bound defined by the evaluator.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin before the sandbox kills the process
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # Initialise – use the provided parent if present, otherwise a random start
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # modest length, random positive entries
        a = rng.random(1024, dtype=np.float64)

    # ensure non‑negativity
    a = np.maximum(a, 0.0)

    # normalise to unit sum (the objective is scale‑invariant)
    total = a.sum()
    if total < 1e-12:                # pathological case – fall back to uniform
        a[:] = 1.0 / a.size
    else:
        a /= total

    cur = a.copy()
    cur_score = _objective(cur)

    best = cur.copy()
    best_score = cur_score

    # --------------------------------------------------------------
    # Simple stochastic hill‑climbing with an annealing schedule
    # --------------------------------------------------------------
    temperature = 1.0
    temp_decay = 0.9995
    min_temp = 1e-4
    iteration = 0

    # pre‑compute the FFT length (constant for a given n)
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    while time.monotonic() < deadline:
        iteration += 1

        # ----- propose a small mass transfer between two positions -----
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue  # nothing to move
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # move at most 10 % of the source mass
        delta = cur[i] * rng.random() * 0.10
        proposal = cur.copy()
        proposal[i] -= delta
        proposal[j] += delta

        # renormalise (keeps the sum exactly 1)
        proposal_sum = proposal.sum()
        if proposal_sum == 0.0:
            continue
        proposal /= proposal_sum

        # ----- evaluate -----
        prop_score = _objective(proposal)

        # ----- accept if better, otherwise probabilistic uphill move -----
        accept = False
        if prop_score < cur_score:
            accept = True
        else:
            # Metropolis‑type acceptance
            prob = np.exp(-(prop_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = proposal
            cur_score = prop_score
            if prop_score < best_score:
                best = proposal.copy()
                best_score = prop_score

        # ----- temperature schedule -----
        temperature = max(min_temp, temperature * temp_decay)

        # ----- occasional symmetrisation (helps with the Toeplitz structure) -----
        if iteration % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            # re‑normalise after the averaging
            sym_sum = sym.sum()
            if sym_sum > 0:
                sym /= sym_sum
                sym_score = _objective(sym)
                if sym_score < cur_score:
                    cur, cur_score = sym, sym_score
                    if sym_score < best_score:
                        best, best_score = sym.copy(), sym_score

    # ------------------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
