import numpy as np

def _softmax_logits(p: np.ndarray) -> np.ndarray:
    """Return softmax(p) ensuring sum = 1."""
    shift = np.max(p)
    e = np.exp(p - shift)
    s = e.sum()
    if s == 0.0:
        # uniform fallback
        return np.full_like(p, 1.0 / p.size)
    return e / s

def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full linear convolution a*a using FFT length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv
