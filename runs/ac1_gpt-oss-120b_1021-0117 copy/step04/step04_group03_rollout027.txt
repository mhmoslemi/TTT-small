import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Score (assuming sum(v)=1)
# ------------------------------------------------------------
def _score(v: np.ndarray) -> float:
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(vf * vf, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
