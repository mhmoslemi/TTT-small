import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection of a vector onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Exact score (assumes sum == 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.maximum(base, 0.0)
        base = _project_simplex(base)
    else:
        # fallback: moderate size uniform
        n_fallback = 2000
        base = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    n = base.size
    i = np.arange(n, dtype=np.float64)

    # arcsine‑type candidate
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcsine = arcsine / arcsine.sum()

    # half‑cosine candidate (non‑negative part only)
    cosine = np.cos(np.pi * (i - (n - 1) / 2) / n)
    cosine = np.maximum(cosine, 0.0)
    cosine = cosine / cosine.sum()

    # triangular candidate
    half = (n + 1) // 2
    tri = np.concatenate([np.linspace(0, 1, half, endpoint=False),
                          np.linspace(1, 0, n - half, endpoint=False)])
    tri = tri / tri.sum()

    candidates = [base, arcsine, cosine, tri]
    best_vec = None
    best_sc = float("inf")
    for cand in candidates:
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # --------------------------------------------------------
    # Parameters for the smoothed‑max gradient descent
    # --------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())          # FFT length (>= 3n)
    init_eps = 5.0
    eps = init_eps
    eps_decay = 0.9995
    eps_min = 1e-7
    step0 = 1.0
    it = 0
    no_improve = 0
    target = 1.50274349

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution and its maximum
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # smoothed soft‑max weights
        eps = max(eps_min, eps * eps_decay)
        w = np.exp((conv - max_val) / eps)        # shift by max_val to avoid overflow
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # backtracking line search on the projected gradient step
        grad_norm = np.linalg.norm(g) + 1e-12
        base_alpha = step0 / math.sqrt(it) / grad_norm
        improved = False
        for factor in (1.0, 0.5, 0.25):
            cand = cur - base_alpha * factor * g
            # enforce symmetry (the optimum is expected to be symmetric)
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)
            cand_sc = _score(cand)
            if cand_sc < cur_sc - 1e-12:
                cur, cur_sc = cand, cand_sc
                improved = True
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve > 2000:
                # random mass transfer to escape a flat region
                i_idx = rng.integers(0, n)
                if cur[i_idx] > 0.0:
                    delta = cur[i_idx] * rng.random() * 0.5
                    j_idx = rng.integers(0, n)
                    if j_idx == i_idx:
                        j_idx = (j_idx + 1) % n
                    tmp = cur.copy()
                    tmp[i_idx] -= delta
                    tmp[j_idx] += delta
                    tmp = (tmp + tmp[::-1]) * 0.5
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        if tmp_sc < best_sc:
                            best_sc, best_vec = tmp_sc, tmp.copy()
                        no_improve = 0

        # occasional explicit symmetrisation (helps numerical drift)
        if (it % 300) == 0:
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)
            cur_sc = _score(cur)
            if cur_sc < best_sc:
                best_sc, best_vec = cur_sc, cur.copy()

        # early exit if the target is reached
        if best_sc < target:
            break

    # --------------------------------------------------------
    # Return the best vector found as a plain Python list
    # --------------------------------------------------------
    if best_vec is None:
        best_vec = cur
    return [float(x) for x in best_vec]
