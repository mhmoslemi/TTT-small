def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    # choose rng with seed for reproducibility
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1) # keep some margin

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 1024
        a = rng.random(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())

    # Score function
    def _score_vec(v):
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())

    cur = a.copy()
    cur_score = _score_vec(cur)
    best = cur.copy()
    best_score = cur_score

    # momentum
    m = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.8
    itr = 0
    no_imp = 0

    while time.monotonic() < deadline:
        itr += 1
        # convolution
        af = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        max_val = conv.max()
        k = int(conv.argmax())

        # gradient of max at index k
        g = np.zeros_like(cur)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g[idx] = 2.0 * cur[k - idx]

        # momentum update
        m = beta * m + (1.0 - beta) * g

        # step size
        base_alpha = step0 / math.sqrt(itr)
        # normalize direction to unit norm
        norm_m = np.linalg.norm(m)
        if norm_m == 0:
            continue
        dir_vec = m / norm_m

        # line search
        alpha = base_alpha
        cand = None
        cand_score = None
        for _ in range(10):
            cand_tmp = cur - alpha * dir_vec
            cand_tmp = _project_simplex(cand_tmp)
            cand_tmp_score = _score_vec(cand_tmp)
            if cand_tmp_score < cur_score:
                cand = cand_tmp
                cand_score = cand_tmp_score
                break
            alpha *= 0.5
        if cand is None:
            # no improvement, maybe try small random move
            no_imp += 1
        else:
            cur = cand
            cur_score = cand_score
            no_imp = 0
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score

        # periodic random perturbation if stagnated
        if no_imp > 2000:
            # random mass transfer
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score_vec(tmp)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score
                no_imp = 0

        # early exit if beat target
        if best_score < 1.50274349:
            break

    return [float(x) for x in best]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score (2 * n * max(conv)) – assumes sum = 1
# ------------------------------------------------------------
def _score_vec(a: np.ndarray, L: int) -> float:
    """Return 2 * n * max(a*a) for a non‑negative vector with unit sum."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of max convolution at a given lag index k
# ------------------------------------------------------------
def _grad_max_at_k(a: np.ndarray, k: int) -> np.ndarray:
    """Gradient of max_{lag} (a*a) w.r.t. a when the maximum occurs at lag k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    g[idx] = 2.0 * a[k - idx]
    return g

# Main entry point
# ------------------------------------------------------------

    rng = np.random.default_rng(seed)

    # FFT length: next power of two ≥ 2n‑1

    best = cur.copy()
    best_sc = cur_sc

    step0 = 0.8         # base step magnitude

    while time.monotonic() < deadline:
