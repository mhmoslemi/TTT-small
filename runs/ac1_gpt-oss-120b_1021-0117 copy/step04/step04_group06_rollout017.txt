import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast evaluation (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(conv) where conv = a * a (linear convolution)."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative coefficient list with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # small safety margin

    # --------------------------------------------------------
    # Initialise the sequence
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: short exponential decay
        n_fallback = 512
        a = np.exp(-np.arange(n_fallback, dtype=np.float64) / 20.0)

    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / total                     # normalise to sum = 1 (simplex)

    n = a.shape[0]
    # FFT length – power of two ≥ 2n‑1 (sufficient for both convolutions)
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # Optimisation state
    # --------------------------------------------------------
    best_vec = a.copy()
    best_sc = _score(best_vec, L)

    # Adam‑style moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # hyper‑parameters
    eps0 = 0.5                # initial smoothing for soft‑max
    eps_decay = 0.9995        # exponential decay each iteration
