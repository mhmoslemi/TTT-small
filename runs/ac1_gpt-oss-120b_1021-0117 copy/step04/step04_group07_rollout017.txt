def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound for a vector that already satisfies sum=1
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    """Return 2*n*max(a*a) assuming sum(a)=1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # initialise
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 2000
        a0 = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s0                     # simplex normalisation

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())   # power of two >= 3n-2

    cur = a0.copy()
    best = cur.copy()
    cur_sc = _fast_score(cur, L, n)
    best_sc = cur_sc

    # --------------------------------------------------------
    # algorithmic parameters
    # --------------------------------------------------------
    step0 = 1.2                # base step magnitude
    eps = 0.1                  # initial smoothing for soft‑max
    eps_min = 1e-7
    eps_decay = 0.9998
    momentum = np.zeros_like(cur)
    beta = 0.9                 # momentum factor
    it = 0
    no_improve = 0

    # --------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # ----- gradient (smooth or sub‑gradient) -----
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            # binary mask on (almost) maximal entries
            mask = conv >= maxc - 1e-12
            w = mask.astype(float)
            w_sum = w.sum()
            if w_sum > 0.0:
                w /= w_sum

        # gradient via convolution with the (reversed) weight vector
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- step and projection -----
        step = step0 / math.sqrt(it) / (np.linalg.norm(momentum) + 1e-12)
        cand = cur - step * momentum
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_sc = _fast_score(cand, L, n)

        # ----- accept / reject -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # occasional uphill move (simulated‑annealing flavour)
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation -----
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, L, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # ----- random perturbation when stuck -----
        if no_improve > 3000:
            # add small centred noise and re‑project
            sigma = max(0.001, 0.1 / math.sqrt(it))
            pert = cur + sigma * (rng.random(n) - 0.5)
            pert = _project_simplex(pert)
            pert_sc = _fast_score(pert, L, n)
            if pert_sc < cur_sc:
                cur, cur_sc = pert, pert_sc
                if pert_sc < best_sc:
                    best, best_sc = pert.copy(), pert_sc
                    no_improve = 0
            else:
                no_improve = 0

        # ----- epsilon decay -----
        eps = max(eps_min, eps * eps_decay)

        # ----- early exit if target reached -----
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # short random local search around best (if time permits)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        # decrease perturbation magnitude over time
        sigma = 0.05 * (0.5 ** (it * 1e-4))
        cand = best + sigma * (rng.random(n) - 0.5)
        cand = _project_simplex(cand)
        cand_sc = _fast_score(cand, L, n)
        if cand_sc < best_sc:
            best, best_sc = cand.copy(), cand_sc
            it += 1
            # break early if we already beat the target
            if best_sc < 1.50274349:
                break
        else:
            it += 1
        # avoid infinite loop if sigma becomes too tiny
        if sigma < 1e-6:
            break

    # --------------------------------------------------------
    # Return best sequence found
    # --------------------------------------------------------
    return [float(x) for x in best]
