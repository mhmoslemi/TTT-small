import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x>=0, sum=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Initialise sequence (use parent if present, else random exponential)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.clip(cur, 0.0, 1000.0)
    else:
        n_fallback = 1024
        cur = np.exp(-np.arange(n_fallback, dtype=np.float64) /
                     (n_fallback / 10.0))
    n = cur.shape[0]
    cur = _project_simplex(cur)          # normalise sum == 1

    # ------------------------------------------------------------------
    # FFT sizes (power‑of‑two for speed)
    # ------------------------------------------------------------------
    L_full = 1 << ((3 * n - 2).bit_length())   # for gradient (3n‑2)
    L_conv = 1 << ((2 * n - 1).bit_length())   # for scoring (2n‑1)

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    cur_fft = np.fft.rfft(cur, n=L_full)
    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
    cur_max = cur_conv.max()
    cur_sc = 2.0 * n * cur_max
    best_vec = cur.copy()
    best_sc = cur_sc

    # parameters
    step0 = 0.8
    eps_rel = 0.02
    eps_min = 1e-7
    it = 0
    no_improve = 0
    restart_threshold = 5000
    backtrack_limit = 6
    sym_interval = 2500

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- smoothed‑max weights -----
        eps = max(eps_min, cur_max * eps_rel)
        w = np.exp((cur_conv - cur_max) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smoothed objective -----
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L_full)
