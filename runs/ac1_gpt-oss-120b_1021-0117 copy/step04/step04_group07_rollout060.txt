import time, math, numpy as np

def _project_simplex(v):
   n = v.size
   u = np.sort(v)[::-1]
   css = np.cumsum(u) - 1.0
   rho_arr = u - css / np.arange(1, n+1)
   rho = np.where(rho_arr > 0)[0]
   if rho.size == 0:
       theta = 0.0
   else:
       rho = rho[-1]
       theta = css[rho] / (rho+1)
   w = np.maximum(v - theta, 0.0)
   return w

import time
import math
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
