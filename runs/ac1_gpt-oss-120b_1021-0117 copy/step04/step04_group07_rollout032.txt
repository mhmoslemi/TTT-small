    total = a.sum()

    else:
        a = a / total

    n = a.size
    L = 1 << ((3 * n - 2).bit_length()) # power of two >= 3n-2

    # main loop
    cur = a.copy()
    best = cur.copy()
    best_score = 2 * n * np.max(np.convolve(best, best))
    # as sum=1

    # hyperparameters:
    eps = 0.01  # start smoothing parameter
    eps_min = 1e-9
    eps_decay = 0.9995

    it = 0
    momentum = np.zeros_like(cur)
    beta = 0.9
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        # FFT of cur
        A = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(A * A, n=L)[:2 * n - 1]

            w = np.exp((conv - maxc) / eps)

import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max of a*a convolution (full) for a sum‑1 vector."""
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * a.size - 1]
    return float(conv.max())


def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """Return a non‑empty non‑negative sequence with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small head‑room for cleanup
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # initialise
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: decreasing geometric shape
        n_fallback = 2500
        a0 = np.geomspace(1.0, 0.1, n_fallback, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                       # normalize to simplex

    n = a0.size
    # FFT length >= 3n‑2 (full convolution of a with weight vector)
    L = 1 << ((3 * n - 2).bit_length())

    cur = a0.copy()
    best = cur.copy()
    cur_score = 2.0 * n * _conv_max(cur, L)
    best_score = cur_score

    # hyper‑parameters
    eps = 0.05                # start temperature for soft‑max
    eps_min = 1e-9
    eps_decay = 0.9998
    step0 = 0.8               # base step length
    beta = 0.9                # momentum mixing factor
    momentum = np.zeros_like(cur)
    it = 0
    no_improve = 0

    # --------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- FFT of current vector -----
        A = np.fft.rfft(cur, n=L)

        # ----- convolution and its maximum -----
        conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # ----- weight vector (smooth max) -----
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:

            else:
                w /= w_sum
            eps = max(eps_min, eps * eps_decay)

            w[int(conv.argmax())] = 1.0

        # ----- gradient through weighted convolution -----
