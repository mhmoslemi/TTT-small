from scipy.optimize import linprog

def solve_lp(a, M):
    n=len(a)
    A_ub = build_A(a)   # shape (2n-1, n)
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    # We solve maximize sum(g) => minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)]*n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 10})
    if res.success:
        g = res.x
        return g
    else:
        return None

# ---- helper: project onto probability simplex (sum=1, non‑negative) ----
def _simplex_proj(v: np.ndarray) -> np.ndarray:
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

        # exact sub‑gradient of max(conv) at index k
        idx = k - np.arange(n)
        grad = np.where((0 <= idx) & (idx < n), 2.0 * cur[idx], 0.0)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step * m_hat
        cand = _simplex_proj(cand)

        cand_M = _conv_max(cand, L)
        cand_score = 2.0 * n * cand_M

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score

        # occasional Gaussian jitter to escape plateaus
        if it % 5000 == 0:
            noise = rng.normal(scale=0.01, size=n)
            tmp = cur + noise
            tmp = _simplex_proj(tmp)
            tmp_M = _conv_max(tmp, L)
            tmp_score = 2.0 * n * tmp_M
            if tmp_score < cur_score:
                cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score

        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Phase 3 – random sampling while time permits (simple Dirichlet draws)
    while time.monotonic() < deadline - 0.1:
        rnd = rng.random(n)
        rnd = _simplex_proj(rnd)
        rnd_M = _conv_max(rnd, L)
        rnd_score = 2.0 * n * rnd_M
        if rnd_score < best_score:
            best, best_score = rnd, rnd_score
            if best_score < 1.50274349:
                break

    # ------------------------------------------------------------------
    return [float(x) for x in best]
