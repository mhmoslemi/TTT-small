import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 bound for a vector already satisfying sum=1
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty, non‑negative sequence with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # initial vector (use parent if present, otherwise triangle)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # construct a simple triangular shape if no parent is given
        n_fallback = 2000
        a0 = np.arange(n_fallback, dtype=np.float64)
        center = (n_fallback - 1) / 2.0
        a0 = 1.0 - np.abs(a0 - center) / center
        a0 = np.maximum(a0, 0.0)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s  # normalize to simplex

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())  # next power of two ≥ 3n‑2

    # --------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------
    eps = 0.01            # smoothing for soft‑max
    eps_min = 1e-6
    eps_decay = 0.9995
    step0 = 0.6
    beta = 0.9            # momentum coefficient
    momentum = np.zeros_like(a0)

    cur = a0.copy()
    cur_sc = _fast_score(cur, L, n)

    best = cur.copy()
    best_sc = cur_sc

    it = 0
    no_improve = 0

    # --------------------------------------------------------
    # optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # ----- gradient (smooth or sub‑gradient) -----
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)            # soft‑max weights
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]
            eps = max(eps_min, eps * eps_decay)
        else:
            k = int(conv.argmax())
            g = np.zeros_like(cur)
            start_idx = max(0, k - (n - 1))
            end_idx = min(k + 1, n)
            rng_idx = np.arange(start_idx, end_idx)
            g[rng_idx] = 2.0 * cur[k - rng_idx]

        # ----- step size -----
        grad_norm = np.linalg.norm(g) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm

        # ----- momentum update -----
        momentum = beta * momentum - alpha * g
        cand = cur + momentum

        # ----- projection onto simplex -----
        cand = _project_simplex(cand)

        # ----- enforce symmetry (helps lower max autocorrelation) -----
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)   # re‑project after averaging

        # ----- evaluate candidate -----
        cand_sc = _fast_score(cand, L, n)

        # ----- accept or reject -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # Metropolis‑style uphill move
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- occasional random symmetric perturbation when stuck -----
        if no_improve > 2000:
            tmp = best.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            delta = tmp[i] * rng.random() * 0.2
            tmp[i] -= delta
            tmp[j] += delta
            # keep symmetry
            i_sym = n - 1 - i
            j_sym = n - 1 - j
            if i_sym != i:
                tmp[i_sym] -= delta
                tmp[j_sym] += delta
            tmp = _project_simplex(tmp)
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp = _project_simplex(tmp)
            tmp_sc = _fast_score(tmp, L, n)
            cur, cur_sc = tmp, tmp_sc
            momentum = np.zeros_like(cur)
            no_improve = 0

        # ----- periodic full symmetrisation check -----
        if it % 3000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, L, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # Return the best vector found (as plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best]
