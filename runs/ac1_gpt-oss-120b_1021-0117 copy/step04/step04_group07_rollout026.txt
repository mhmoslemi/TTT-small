import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound assuming sum(a)==1
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    """Return 2*n*max(a*a) for a normalized to sum=1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin before the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise (use parent if present)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: flat Dirichlet sample
        n_fallback = 2500
        a0 = rng.random(n_fallback)
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                     # normalise to sum = 1

    n = a0.size
    # FFT length, power of two ≥ 3n‑2 (safety margin for convolutions)
    L = 1 << ((3 * n - 2).bit_length())

    cur = a0.copy()
    cur_sc = _fast_score(cur, L, n)
    best = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # Optimisation hyper‑parameters
    # --------------------------------------------------------
    eps = 0.12               # initial smoothing for soft‑max
    eps_min = 1e-6
    eps_decay = 0.9994

    beta1, beta2 = 0.9, 0.999          # Adam moments
    adam_eta = 0.12                    # base learning‑rate
    adam_eps = 1e-8

    m = np.zeros(n)                   # first moment
    v = np.zeros(n)                   # second moment
    it = 0
    no_improve = 0
    target = 1.50274349

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # ----- soft‑max weights (smooth approximation of the max) -----
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w[:] = 1.0 / w.size
            else:
                w /= w_sum
            eps = max(eps_min, eps * eps_decay)
        else:
            w = np.zeros_like(conv)
            w[int(conv.argmax())] = 1.0

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # map to length‑n gradient
        g = 2.0 * grad_full[idx]

        # ----- Adam update (projected) -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = adam_eta * m_hat / (np.sqrt(v_hat) + adam_eps)

        cand = cur - step
        cand = _project_simplex(cand)

        # ----- occasional random mass‑transfer to escape flat regions -----
        if no_improve > 5000 and (it & 7) == 0:
            i = rng.integers(0, n)
            if cand[i] > 0.0:
                delta = cand[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                cand[i] -= delta
                cand[j] += delta
                cand = _project_simplex(cand)

        # ----- evaluate candidate (fast) -----
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_sc = float(2.0 * n * cand_conv.max())

        # ----- Acceptance test -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            # occasional uphill move (simulated‑annealing flavour)
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- Periodic anti‑symmetrisation attempt (mix with reverse) -----
        if it % 2000 == 0:
            rev = cur[::-1]
            mixed = 0.75 * cur + 0.25 * rev
            mixed = _project_simplex(mixed)
            mixed_sc = _fast_score(mixed, L, n)
            if mixed_sc < cur_sc:
                cur, cur_sc = mixed, mixed_sc
                if mixed_sc < best_sc:
                    best, best_sc = mixed.copy(), mixed_sc

        # ----- Early exit if target reached -----
        if best_sc < target:
            break

    # Return the best vector found as a plain Python list
    return [float(x) for x in best]
