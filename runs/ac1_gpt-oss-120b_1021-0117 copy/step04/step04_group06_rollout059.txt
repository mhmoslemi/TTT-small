import time, math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(a: np.ndarray, conv: np.ndarray | None = None) -> float:
    """Fast evaluation of 2·n·max(a*a) (sum is assumed to be 1)."""
    n = a.shape[0]
    if conv is None:
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative coefficient list with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 1024
        a = np.exp(-np.arange(n0, dtype=np.float64) / 20.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s
        a = _project_simplex(a)

    n = a.shape[0]
    # FFT length enough for 3·n‑2 linear convolution
    L = 1 << ((3 * n - 2).bit_length())

    # ----- bookkeeping ----------------------------------------------------
    cur = a.copy()
    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # ----- hyper‑parameters ------------------------------------------------
    base_step = 0.6
    beta = 0.9                     # momentum decay
    momentum = np.zeros_like(cur)
    eps0 = 0.5
    eps_decay = 0.995
    eps_min = 1e-8
    no_improve = 0
    it = 0

    # ----- main optimisation loop -----------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution a*a (full)
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        mx = conv.max()

        # soft‑max weights on convolution entries
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - mx) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g
        norm_m = np.linalg.norm(momentum) + 1e-12
        step = base_step / (math.sqrt(it) * norm_m)

        cand = cur - step * momentum
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # accept if improvement
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # occasional simulated‑annealing acceptance
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (step + 1e-12)):
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # periodic symmetrisation (helps autocorrelation)
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # random perturbation when stuck
        if no_improve > 2000:
            pert = rng.normal(scale=0.02, size=n)
            cand2 = cur + pert
            cand2 = _project_simplex(cand2)
            cand2_sc = _score(cand2)
            if cand2_sc < cur_sc:
                cur, cur_sc = cand2, cand2_sc
                if cand2_sc < best_sc:
                    best, best_sc = cand2.copy(), cand2_sc
            no_improve = 0

        # early exit if we already beat the target
        if best_sc < 1.50274349:
            break

    # ----- final local refinement -----------------------------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = best[i] * rng.random() * 0.03
        if delta <= 0:
            continue
        tmp = best.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < best_sc:
            best, best_sc = tmp, tmp_sc
            continue
        # try symmetrising the perturbation
        sym = (tmp + tmp[::-1]) * 0.5
        sym = _project_simplex(sym)
        sym_sc = _score(sym)
        if sym_sc < best_sc:
            best, best_sc = sym, sym_sc

    # return plain Python list
    return [float(x) for x in best]
