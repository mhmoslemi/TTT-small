def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a vector with sum = 1 (returns 2*n*max(conv))
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Fast L-infinity autocorrelation score (sum(a)=1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # -----------------------------------------------------------------
    # Initialise a feasible point on the simplex
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: exponential decay, length 2000
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 100.0)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
        s = 1.0
    a0 = a0 / s                     # now sum(a0)=1

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length >= 2n-1

    cur = a0.copy()
    cur_sc = _score(cur, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------
    vel = np.zeros_like(cur)          # momentum vector
    base_alpha = 0.5
    beta = 0.9                        # momentum decay
    sym_interval = 400                # symmetrisation frequency
    improve_target = 1.50274349
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and max index(ices) -----
        cur_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        # tolerance for multiple maximisers
        tol = max(1e-12, max_val * 1e-12)
        max_inds = np.where(conv >= max_val - tol)[0]

        # ----- sub‑gradient of max(conv) (average over all maximisers) -----
        grad = np.zeros_like(cur)
        for k in max_inds:
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1) + 1         # exclusive upper bound
            idx = np.arange(j_start, j_end, dtype=np.int64)
            grad[idx] += 2.0 * cur[k - idx]
        if max_inds.size > 1:
            grad /= max_inds.size

        # ----- step size and momentum update -----
        g_norm = np.linalg.norm(grad) + 1e-12
        alpha = base_alpha / math.sqrt(it) / g_norm
        vel = beta * vel + alpha * grad

        # ----- candidate and projection onto simplex -----
        cand = cur - vel
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_sc = _score(cand, L)

        # ----- acceptance (downhill or occasional uphill) -----
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation (often beneficial) -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < improve_target:
            break

    # return as plain list of Python floats
    return [float(x) for x in best_vec]
