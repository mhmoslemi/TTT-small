def generate_sidon_set(n, target_m):
    positions = []
    used_sums = set()
    for x in range(n):
        conflict = False
        for y in positions:
            if (x + y) in used_sums:
                conflict = True
                break
        if not conflict and (2*x) not in used_sums:
            # Because self sum counts as well, ensure x+x not used.
            # Actually we will add sums after adding x
            positions.append(x)
            for y in positions:
                used_sums.add(x + y)
        if len(positions) >= target_m:
            break
    return positions

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score used by the evaluator – assumes sum(a)=1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2 * n * max(convolution(a,a)) – a is assumed to sum to 1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    The algorithm is a smoothed gradient descent on the
    log‑sum‑exp approximation of the max‑convolution,
    with simplex projection and occasional random perturbations.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin to avoid hard timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a random start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a modest length when no parent is supplied
        a0 = rng.random(2000, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total == 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / total                     # enforce sum = 1

    cur = a0.copy()
    n = cur.size
    cur_sc = _score(cur)

    # ------------------------------------------------------------
    # FFT planning – length at least 3n‑2 (power of two)
    # ------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    best_vec = cur.copy()
    best_sc = cur_sc

    # algorithmic parameters (tuned for quicker convergence)
    step0 = 0.85               # base step magnitude
    eps0 = 1.5                 # initial smoothing constant
    eps_decay = 0.9997         # exponential decay of smoothing
    eps_min = 1e-9
    beta = 0.93                # momentum factor
    momentum = np.zeros_like(cur)

    it = 0
    no_improve = 0
    target = 1.50274349        # target bound to beat

    # ------------------------------------------------------------
    # Optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- smoothed soft‑max weights -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)

        if w_sum == 0.0:

            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
