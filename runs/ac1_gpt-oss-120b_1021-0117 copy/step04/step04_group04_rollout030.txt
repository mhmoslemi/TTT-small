def _project_simplex_sum(v, S):

    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - S
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the simplex {x >= 0, sum(x)=S}
# ------------------------------------------------------------
def _project_simplex_sum(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with sum=S."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - S
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast AC1 bound (scale–invariant) using FFT
# ------------------------------------------------------------
def _score_fast(a: np.ndarray) -> float:
    n = a.shape[0]
    s = a.sum()
    if s == 0.0:
        return float("inf")
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

# ------------------------------------------------------------
# One gradient (soft‑max) descent step
# ------------------------------------------------------------
def _gradient_step(a: np.ndarray, Lfft: int, eps: float, step0: float,
                   it: int, target_sum: float, rng: np.random.Generator):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=Lfft)

    conv = np.fft.irfft(a_fft * a_fft, n=Lfft)[: 2 * n - 1]
    max_val = conv.max()

    # soft‑max weights on the convolution values
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum

    # gradient of the smoothed max
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=Lfft)
    grad_full = np.fft.irfft(a_fft * w_fft, n=Lfft)
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]

    # adaptive step size
    grad_norm = np.linalg.norm(g) + 1e-12
    alpha = step0 / math.sqrt(it) / grad_norm
    cand = a - alpha * g
    cand = _project_simplex_sum(cand, target_sum)
    cand_sc = _score_fast(cand)
    cur_sc = _score_fast(a)

    # simple Metropolis acceptance
    if cand_sc < cur_sc:
        a = cand
    else:
        prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
        if rng.random() < prob:
            a = cand

    # occasional symmetrisation
    if it % 2000 == 0:
        sym = (a + a[::-1]) * 0.5
        a = _project_simplex_sum(sym, target_sum)

    return a, cand_sc, cur_sc

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    a = _project_simplex_sum(a, target_sum)

    best = a.copy()
    best_sc = _score_fast(best)

    # ----- parameters -------------------------------------------------------
    eps0 = 0.6                # initial smoothing for soft‑max
    eps_decay = 0.998
    eps_min = 1e-9
    step0 = 0.5
    it = 0

    # FFT length sufficient for all convolutions (3n‑2 ≥ 2n‑1)
    Lfft = 1 << ((3 * n - 2).bit_length())

    # try a few LP refinements if cvxpy is available
    use_cvx = False

    except Exception:
        use_cvx = False
    lp_steps = 12

            break

    # return the best sequence found (as plain Python list)
    return [float(x) for x in best]
