import time, math, numpy as np

# ------------------------------------------------------------
# Projection of a vector onto the probability simplex {x>=0, sum=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 score for a vector already normalized to sum=1
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of max_k (a*a)[k] w.r.t. a (sub‑gradient at the max index)
# ------------------------------------------------------------
def _subgradient(a: np.ndarray, max_idx: int) -> np.ndarray:
    n = a.size
    g = np.zeros_like(a)
    start = max(0, max_idx - (n - 1))
    end = min(max_idx, n - 1)
    # slice a[start:end+1] reversed gives a_{max_idx-i}
    seg = a[start : end + 1]
    g[start : end + 1] = 2.0 * seg[::-1]
    return g


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: uniform vector of moderate size
        n0 = 2000
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                                    # normalize to sum=1

    n = a.size
    L = 1 << ((3 * n - 2).bit_length())            # FFT length >= 3n‑2

    # ----- bookkeeping ----------------------------------------------------
    cur = a.copy()
    cur_sc = _fast_score(cur, L, n)
    best = cur.copy()
    best_sc = cur_sc

    # algorithmic parameters
    base_step = 0.8
    beta = 0.9                     # momentum coefficient
    momentum = np.zeros_like(cur)
    it = 0
    no_improve = 0
    sym_interval = 2500
    mass_transfer_thresh = 4000

    # ------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- compute convolution & locate maximal index -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()
        max_idx = int(conv.argmax())

        # ----- sub‑gradient at the active index -----
        g = _subgradient(cur, max_idx)

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- step size (slowly decreasing) -----
        step = base_step / (math.sqrt(it) + 1.0)

        # ----- two candidate proposals (additive & multiplicative) -----
        # additive
        cand_add = cur - step * momentum
        cand_add = _project_simplex(cand_add)

        # multiplicative (exponentiated gradient)
        exp_factor = np.exp(-step * momentum)
        cand_mul = cur * exp_factor
        sm = cand_mul.sum()
        if sm > 0.0:
            cand_mul = cand_mul / sm
        else:
            cand_mul = cur.copy()

        # ----- evaluate both candidates -----
        sc_add = _fast_score(cand_add, L, n)
        sc_mul = _fast_score(cand_mul, L, n)

        # pick the better one
        if sc_add <= sc_mul:
            cand, cand_sc = cand_add, sc_add
        else:
            cand, cand_sc = cand_mul, sc_mul

        # ----- accept if improvement -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # occasional uphill move (simulated‑annealing flavour)
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (step + 1e-12)):
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation (helps due to near‑symmetry) -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, L, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > mass_transfer_thresh:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _fast_score(tmp, L, n)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------
    # Return best sequence found as plain Python list
    # ------------------------------------------------------------
    return [float(x) for x in best]
