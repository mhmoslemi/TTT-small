import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise a feasible vector --------------------------------------
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # simple fallback: exponential decay
        n_fallback = 1024
        cur = np.exp(-np.arange(n_fallback, dtype=np.float64) / (n_fallback / 10.0))

    cur = np.maximum(cur, 0.0)
    # enforce symmetry – the optimum is expected to be symmetric
    cur = (cur + cur[::-1]) * 0.5
    # normalise to sum == 1 via simplex projection (handles rounding)
    cur = _project_simplex(cur)

    n = cur.shape[0]

    # ------------------------------------------------------------------
    # FFT preparation ---------------------------------------------------
    # ------------------------------------------------------------------
    L_full = 1 << ((3 * n - 2).bit_length())   # length for gradient convolutions
    idx = (2 * n - 2) - np.arange(n)            # mapping indices for gradient slice

    cur_fft = np.fft.rfft(cur, n=L_full)
    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
    cur_max = cur_conv.max()
    cur_sc = 2.0 * n * cur_max

    best_vec = cur.copy()

    # ------------------------------------------------------------------

    step0 = 2.0          # base step size

    while time.monotonic() < deadline:
