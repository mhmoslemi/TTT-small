import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Duchi et al.)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast evaluation of 2*n*max(conv) (sum of a is 1)
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # small head‑room

    # --------------------------------------------------------
    # initial vector (use parent if present, else uniform)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / a.size)
    else:
        n_fallback = 2000
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    n = a.size
    # FFT length sufficient for linear convolution (next power of two)
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------
    cur = a.copy()
    cur_sc = _fast_score(cur, L, n)
    best = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # optimizer hyper‑parameters
    # --------------------------------------------------------
    eps = 0.05                # initial soft‑max temperature
    eps_min = 1e-6
    eps_decay = 0.9995

    beta1, beta2 = 0.9, 0.999
    beta1_pow = 1.0
    beta2_pow = 1.0
    m = np.zeros_like(cur)    # 1st moment
    v = np.zeros_like(cur)    # 2nd moment
    lr = 0.05                  # base learning rate for Adam
    adam_eps = 1e-8

    it = 0
    last_improve = 0
    stagnation_limit = 5000   # iterations without improvement before a restart

    target = 1.50274349

    # --------------------------------------------------------
    # optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # ----- soft‑max weights (smooth approximation of arg‑max) -----
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            # exact max: distribute weight uniformly among all maxima
            mask = (conv == maxc)
            w = mask.astype(float)
            cnt = w.sum()
            if cnt == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= cnt

        # ----- gradient via convolution with reversed weights -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)

        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        cand = cur - step
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_sc = _fast_score(cand, L, n)

        # keep the global best

        # ----- temperature decay -----

        # ----- occasional symmetrisation (helps for near‑symmetric optima) -----
        if it % 2500 == 0:

            cur, cur_sc = new, new_sc
            last_improve = it  # reset counter

        # ----- early exit if target reached -----
        if best_sc < target:

    # --------------------------------------------------------
    # Return best vector found (plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best]
