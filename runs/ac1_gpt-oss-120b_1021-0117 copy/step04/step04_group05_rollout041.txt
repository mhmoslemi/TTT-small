import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact scoring (identical to evaluator for sum==1)
# ----------------------------------------------------------------------
def _score_vec(vec: np.ndarray) -> float:
    """Return 2*n*max(conv) / (sum(vec)**2) using FFT."""
    n = vec.shape[0]
    S = vec.sum()
    if S < 1e-12:
        return float("inf")
    L = 2 * n - 1
    f = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(f * f, n=L)[:L]
    return float(2.0 * n * conv.max() / (S * S))


# ----------------------------------------------------------------------
# Core optimiser for a single start vector
# ----------------------------------------------------------------------
def _optimize_start(start_vec: np.ndarray,
                    rng: np.random.Generator,
                    deadline: float):
    """Run a gradient‑descent / annealing loop until the deadline."""
    n = start_vec.shape[0]
    L = 2 * n - 1

    cur = start_vec.copy()

    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L)[:L]
    cur_max = cur_conv.max()
    cur_sc = 2.0 * n * cur_max / (cur.sum() ** 2)

    best_vec = cur.copy()
    best_sc = cur_sc

    step0 = 1.0          # initial step magnitude
    eps_rel = 0.01       # relative smoothing for the soft‑max
    eps_min = 1e-9
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # ----- smoothed max weights -----
        eps = max(eps_min, cur_max * eps_rel)
        w = np.exp((cur_conv - cur_max) / eps)

            w /= w_sum

                                 np.fft.rfft(w_rev, n=L), n=L)

        g = 2.0 * grad_full[idx]

        # ----- back‑tracking line search -----
