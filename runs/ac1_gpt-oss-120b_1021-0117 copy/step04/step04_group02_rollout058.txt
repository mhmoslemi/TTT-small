import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 score for a vector that already satisfies sum = 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2·n·max(convolution) for a non‑negative sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative sequence (list of Python floats) that aims
    to minimise the AC1 upper bound evaluated by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # -----------------------------------------------------------------
    # Initialise a feasible point on the simplex
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
    else:
        # simple exponential decay if no parent is present
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        cur = np.exp(-t / 100.0)

    # project onto the simplex (sum = 1)
    cur = _project_simplex(cur)

    # enforce symmetry – it empirically helps the bound
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)

    n = cur.shape[0]
    # length sufficient for convolutions a*a and a*w_rev
    L = 1 << ((3 * n - 2).bit_length())

    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------
    # Adam‑style optimisation parameters
    # ------------------------------------------------------------
    m = np.zeros_like(cur)          # first moment
    v = np.zeros_like(cur)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = 0.5

    eps = None                      # smoothing parameter for soft‑max
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:

        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]

        if eps is None:

    # Return the best vector found as a plain Python list
