def _optimize(start_vec, deadline, target):

    eps_min = 1e-7
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        # conv and max
        a_fft = np.fft.rfft(cur, n=L)

        max_val = conv.max()

import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection – sum == 1, non‑negative
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # enforce exact sum=1 (tiny numerical drift)
    s = w.sum()
    if s > 0:
        w /= s
    return w


# ------------------------------------------------------------
# Score used by the evaluator (sum == 1 → 2·n·max(conv))
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# One optimisation run (projected‑gradient with soft‑max smoothing)
# ------------------------------------------------------------
def _optimize(start_vec: np.ndarray, deadline: float, rng: np.random.Generator,
              target: float) -> tuple[np.ndarray, float]:
    cur = start_vec.copy()
    n = cur.size
    L = 1 << ((3 * n - 2).bit_length())          # enough for gradient conv
    best = cur.copy()
    best_sc = _score(best)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)
    beta = 0.90                                    # momentum factor
    step0 = 1.0                                    # base step size
    eps0 = 1.0                                     # initial soft‑max temperature
    eps_decay = 0.9997
    eps_min = 1e-8
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of a with itself & its maximum -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weights -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)           # slice giving ∂/∂a
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- back‑tracking line search -----
        alpha = step0 / math.sqrt(it)
        cand = None
        cand_sc = None
        for _ in range(10):
            tmp = cur - alpha * momentum
            tmp = _project_simplex(tmp)
            sc = _score(tmp)
            if sc <= cur_sc:                       # accept a descent step
                cand, cand_sc = tmp, sc
                break
            alpha *= 0.5
        if cand is None:                           # no improvement → take last

        # ----- accept / possibly uphill move -----
