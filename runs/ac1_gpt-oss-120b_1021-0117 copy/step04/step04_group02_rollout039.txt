Implementation of isotonic regression decreasing:

def _isotonic_decreasing(v):
    n = len(v)
    # PAVA algorithm: start with each value as level.
    levels = v.copy()
    weights = np.ones_like(v)
    i = 0
    while i < n-1:
        if levels[i] < levels[i+1] - 1e-12: # not decreasing (we need level_i >= level_{i+1})
            # merge levels i and i+1
            total_weight = weights[i] + weights[i+1]
            avg = (weights[i]*levels[i] + weights[i+1]*levels[i+1]) / total_weight
            levels[i] = avg
            weights[i] = total_weight
            # remove i+1
            levels = np.delete(levels, i+1)
            weights = np.delete(weights, i+1)
            n = len(levels)
            # move back if possible
            i = max(i-1, 0)
        else:
            i += 1
    # Expand back to original length
    result = np.empty(len(v))
    idx = 0
    for level, w in zip(levels, weights):
        result[idx: idx+int(w)] = level

# ------------------------------------------------------------
# Projection of a vector onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        if not improved:
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score_fft(cand_try, n, L)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, n, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_fft(tmp, n, L)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                no_improve = 0

        # ----- keep a safety margin for the harness -----
        if time.monotonic() > deadline - 0.2:
            break

    # return plain Python list
    return [float(x) for x in best_vec]
