# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = best[i] * rng.random() * 0.03
        if delta <= 0.0:
            continue
        tmp = best.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_max = _self_conv(tmp).max()
        tmp_score = 2.0 * n * tmp_max
        if tmp_score < best_score:
            best = tmp.copy()
            best_score = tmp_score
        # occasional symmetry test
        if rng.random() < 0.08:
            sym = (tmp + tmp[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_max = _self_conv(sym).max()
            sym_score = 2.0 * n * sym_max
            if sym_score < best_score:
                best = sym.copy()
                best_score = sym_score

    # ------------------------------------------------------------
    # return as plain Python list
    # ------------------------------------------------------------
    return [float(x) for x in best]
