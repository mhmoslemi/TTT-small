import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Exact AC1 bound for a sequence with sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------
    it = 0
    no_improve = 0
    base_alpha = 0.5
    eps = None                     # smoothing parameter for soft‑max
    eps_decay = 0.9995
    p = 2.0                         # start p‑norm exponent
    p_max = 20.0
    p_inc = 0.001
    mix_iter = 3000                 # iterations to transition from soft‑max to p‑norm
    sym_interval = 300
    improve_target = 1.50274349

    # main optimisation loop
    while time.monotonic() < deadline:
        it += 1

        # ---------- current convolution ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ---------- soft‑max weights ----------
        if eps is None:
            eps = max(max_val * 0.2, 1e-9)
        else:
            eps = max(eps * eps_decay, 1e-9)
        w_soft = np.exp(np.clip((conv_full - max_val) / eps, -700, 700))
        wsum = w_soft.sum()
        if wsum == 0.0:
            w_soft = np.full_like(w_soft, 1.0 / w_soft.size)
        else:
            w_soft /= wsum

        # ---------- p‑norm weights ----------
        eps_small = 1e-12
        w_pow = np.power(np.maximum(conv_full, eps_small), p - 1.0)

        # ---------- gradients ----------
        idx = (2 * n - 2) - np.arange(n)          # mapping back to original indices

        # soft‑max gradient
        w_soft_rev = w_soft[::-1]
        w_soft_fft = np.fft.rfft(w_soft_rev, n=L)
        grad_soft_full = np.fft.irfft(a_fft * w_soft_fft, n=L)[: 2 * n - 1]
        g_soft = 2.0 * grad_soft_full[idx]

        # p‑norm gradient
        w_pow_rev = w_pow[::-1]
        w_pow_fft = np.fft.rfft(w_pow_rev, n=L)
        grad_pow_full = np.fft.irfft(a_fft * w_pow_fft, n=L)[: 2 * n - 1]
        g_pow = 2.0 * grad_pow_full[idx]

        # ----- blend the two directions -----
        mix = min(1.0, it / mix_iter)
        g = (1.0 - mix) * g_soft + mix * g_pow

        # ----- step size -----
        g_norm = np.linalg.norm(g) + 1e-12
        alpha = base_alpha / (math.sqrt(it) * g_norm)

        # ----- backtracking line‑search -----
        improved = False
        cand = None
        cand_sc = None
        for _ in range(8):
            cand_try = cur - alpha * g
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5

        # ----- occasional Metropolis acceptance -----
        if not improved:
            # use the last tried candidate (alpha already reduced)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- anneal p -----
        if p < p_max:
            p = min(p_max, p + p_inc)

        # ----- symmetrisation (helps a lot) -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- stop if target reached -----
        if best_sc < improve_target:
            break

        # ----- time guard -----
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------
    # Final short local search using tiny random transfers
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if best_vec[i] <= 0.0:
            continue
        delta = best_vec[i] * rng.random() * 0.3
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        tmp = best_vec.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < best_sc:
            best_sc = tmp_sc
            best_vec = tmp

    # return as plain list of Python floats
    return [float(x) for x in best_vec]
