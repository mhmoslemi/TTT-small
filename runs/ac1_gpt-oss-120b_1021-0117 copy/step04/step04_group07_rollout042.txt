def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # leave 0.5s margin

    # initial vector
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 2000
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two >= 2n-1

    # parameters
    eps = 0.001
    eps_min = 1e-9
    eps_decay = 0.9995
    beta1 = 0.9
    beta2 = 0.999
    lr0 = 0.05
    step_max = 0.2   # max L2 norm of step
    target_score = 1.50274349
    # optimizer state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # compute initial score
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    maxc = float(conv_full.max())
    cur_score = 2.0 * n * maxc
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    maxc_cur = maxc

    it = 0
    no_improve = 0
    sym_interval = 1500
    # main loop
    while time.monotonic() < deadline:
        it += 1

        # compute convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = float(conv.max())
        # compute smoothing weight
        if eps > eps_min:
            # stable softmax
            shift = maxc  # for numerical stability
            w = np.exp((conv - shift) / eps)
            w_sum = w.sum()
            if w_sum == 0.0 or np.isinf(w_sum) or np.isnan(w_sum):
                # fallback to one-hot at max index
                w = np.zeros_like(conv)
                w[int(conv.argmax())] = 1.0
                w_sum = 1.0
            w /= w_sum
        else:
            # pure subgradient: one-hot at max index
            w = np.zeros_like(conv)
            w[int(conv.argmax())] = 1.0

        # gradient via convolution with reversed weights
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)   # length L
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr0 * m_hat / (np.sqrt(v_hat) + 1e-8)

        # optional clipping of step norm
        step_norm = np.linalg.norm(step)
        if step_norm > step_max:
            step = step * (step_max / step_norm)

        cand = cur - step
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_fft = np.fft.rfft(cand, n=L)
        conv_cand = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        maxc_cand = float(conv_cand.max())
        cand_score = 2.0 * n * maxc_cand

        improved = False
        if cand_score < cur_score:
            cur, cur_score, maxc_cur = cand, cand_score, maxc_cand
            improved = True
        else:
            # occasional uphill acceptance
            # temperature decreasing with iterations
            temp = 0.001 * (1.0 / math.sqrt(it))
            if rng.random() < math.exp(-(cand_score - cur_score) / (temp + 1e-12)):
                cur, cur_score, maxc_cur = cand, cand_score, maxc_cand
                improved = True

        if cand_score < best_score:
            best, best_score = cand.copy(), cand_score
            no_improve = 0
        else:
            if not improved:
                no_improve += 1

        # decay epsilon
        eps = max(eps_min, eps * eps_decay)

        # symmetrisation occasionally
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            conv_sym = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_score = 2.0 * n * float(conv_sym.max())
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score
                    no_improve = 0

        # random mass transfer when stuck
        if no_improve > 5000:
            # pick donor with some mass
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                conv_tmp = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_score = 2.0 * n * float(conv_tmp.max())
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score
                no_improve = 0

        # early exit if target reached
        if best_score < target_score:
            break

    # final refinement via mixing with uniform and reversed
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    rev = best[::-1]
    for alpha in [0.0, 0.005, 0.01, 0.02, 0.05, 0.1]:
        for target in (uni, rev):
            combo = (1.0 - alpha) * best + alpha * target
            # combo already sums to 1
            combo_fft = np.fft.rfft(combo, n=L)
            conv_combo = np.fft.irfft(combo_fft * combo_fft, n=L)[: 2 * n - 1]
            combo_score = 2.0 * n * float(conv_combo.max())
            if combo_score < best_score:
                best, best_score = combo.copy(), combo_score

    # final safety projection (just in case rounding errors)
    best = _project_simplex(best)

    # return as list of floats
    return [float(x) for x in best]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    # Project v onto simplex {x >= 0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
