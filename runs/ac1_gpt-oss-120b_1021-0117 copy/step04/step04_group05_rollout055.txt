import time, math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(v: np.ndarray, L: int) -> float:
    """Score = 2*n*max(conv) for sum(v)=1 (the denominator of the evaluator collapses)."""
    n = v.shape[0]
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    m = np.zeros_like(cur)                # momentum buffer
    step0 = 1.0
    beta = 0.7                             # momentum decay
    eps_rel_start = 0.02
    eps_rel_end = 0.001
    eps_min = 1e-9
    no_improve = 0
    it = 0
    sym_interval = 300                     # how often to force symmetry
    transfer_thresh = 2500                 # attempts before a random mass transfer

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- smoothed‑max weights (log‑sum‑exp) -----
        eps_rel = max(eps_rel_end,
                      eps_rel_start * (0.995 ** (it // 1000)))
        eps = max(eps_min, cur_max * eps_rel)
        w = np.exp((cur_conv - cur_max) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # map to each coordinate
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        m = beta * m + (1.0 - beta) * g

        # ----- step size -----
        alpha = step0 / math.sqrt(it)

        # ----- candidate generation -----
        cand = cur - alpha * m
        cand = _project_simplex(cand)

        # ----- evaluation -----
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_max = cand_conv.max()
        cand_sc = 2.0 * n * cand_max

        # ----- accept if better, otherwise occasional probabilistic acceptance -----
        if cand_sc < cur_sc:
            cur = cand
            cur_fft = cand_fft
            cur_conv = cand_conv
            cur_max = cand_max
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_fft = cand_fft
                cur_conv = cand_conv
                cur_max = cand_max
                cur_sc = cand_sc
                no_improve = 0

        # ----- enforce (near) symmetry periodically -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < cur_sc:
                cur = sym
                cur_fft = sym_fft
                cur_conv = sym_conv
                cur_max = sym_conv.max()
                cur_sc = sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()

        # ----- random mass‑transfer when stuck -----
        if no_improve > transfer_thresh:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_fft = tmp_fft
                    cur_conv = tmp_conv
                    cur_max = tmp_conv.max()
                    cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                no_improve = 0

        # ----- early exit if target met -----
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Return the best vector found (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
