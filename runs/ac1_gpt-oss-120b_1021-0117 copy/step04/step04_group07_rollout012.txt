import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound for a vector that already
# satisfies sum(a)=1 (or any sum – the formula is homogeneous).
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    maxc = conv.max()
    s = a.sum()
    if s == 0.0:
        return float("inf")
    return float(2.0 * n * maxc / (s * s))


# ------------------------------------------------------------
# Smoothed gradient of the max‑convolution using a soft‑max.
# ------------------------------------------------------------
def _grad_smooth(a: np.ndarray, eps: float, L: int) -> np.ndarray:
    """Return gradient of smooth max with parameter eps."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)

    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    maxc = conv.max()

    w = np.exp((conv - maxc) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)

    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g


# ------------------------------------------------------------
# Sub‑gradient at the active (max) index.
# ------------------------------------------------------------
def _grad_sub(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=1 << ((2 * n - 1).bit_length()))
    conv = np.fft.irfft(a_fft * a_fft, n=a_fft.size * 2 - 2)[: 2 * n - 1]
    k = int(conv.argmax())
    g = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(k + 1, n)
    idx = np.arange(start, end)
    g[idx] = 2.0 * a[k - idx]
    return g

# ------------------------------------------------------------
def _optimise_start(a0: np.ndarray, L: int, deadline: float, rng: np.random.Generator):
    n = a0.shape[0]
    a = a0.copy()
    v = np.zeros_like(a)               # momentum buffer
    eps = 0.05
    eps_min = 1e-7
    eps_decay = 0.9995
    step0 = 0.7
    mu = 0.9                           # momentum coefficient
    it = 0
    no_imp = 0
    cur_sc = _fast_score(a, L)
    best = a.copy()
    best_sc = cur_sc

    while time.monotonic() < deadline:
        it += 1

        # ----------------------------------------------------
        # gradient (smooth or sub‑gradient)
        # ----------------------------------------------------
        if eps > eps_min:
            g = _grad_smooth(a, eps, L)
            eps = max(eps_min, eps * eps_decay)
        else:
            g = _grad_sub(a)

        # ----------------------------------------------------
        # momentum step + projection onto simplex
        # ----------------------------------------------------
        grad_norm = np.linalg.norm(g) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm
        v = mu * v - alpha * g
        cand = a + v
        cand = _project_simplex(cand)

        # enforce symmetry (the problem is almost symmetric)
        cand = (cand + cand[::-1]) * 0.5

        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        return a

    def _init_uniform():
        a = np.full(n, 1.0 / n, dtype=np.float64)
        return a

    def _init_dirichlet():
        a = rng.exponential(scale=1.0, size=n)
        a = a / a.sum()
        return a

    def _init_gaussian():
        x = np.arange(n, dtype=np.float64)
        centre = (n - 1) / 2.0
        sigma = rng.uniform(0.1 * n, 0.4 * n)
        a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        a = a / a.sum()
        return a

    init_fns = [_init_from_parent, _init_uniform, _init_dirichlet, _init_gaussian]

    # ----------------------------------------------------
    # Multi‑start optimisation
    # ----------------------------------------------------
    best_overall = None
    best_score = float("inf")
    target = 1.50274349

    for init_fn in init_fns:
        if time.monotonic() >= deadline:
            break
        a0 = init_fn()
        # ensure simplex feasibility (numeric safety)
        a0 = _project_simplex(a0)
        a0 = (a0 + a0[::-1]) * 0.5

        cur_best, cur_score = _optimise_start(a0, L, deadline, rng)

        if cur_score < best_score:
            best_overall, best_score = cur_best, cur_score
            if best_score < target:
                break

    # ----------------------------------------------------
    # Final polishing on the best found vector (if time permits)
    # ----------------------------------------------------
    if best_overall is not None and time.monotonic() < deadline:
        # a short extra run with a very small epsilon
        eps_orig = 1e-7
        a_polished, polished_score = _optimise_start(best_overall, L, deadline, rng)
        if polished_score < best_score:
            best_overall, best_score = a_polished, polished_score

    # ----------------------------------------------------
    # Return as plain Python list
    # ----------------------------------------------------
    if best_overall is None:
        # fallback safety net: uniform vector
        best_overall = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_overall]
