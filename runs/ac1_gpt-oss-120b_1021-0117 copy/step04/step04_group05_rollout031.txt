def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time,math, numpy as np
    # maybe use local imports for speed may not matter.
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0) # with margin 1 sec
    # initialize cur
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 1024
        cur = np.exp(-np.arange(n_fallback) / (n_fallback/10.0))
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)
    n = cur.shape[0]
    # FFT length L >= 3*n
    L = 1 << ((3*n - 2).bit_length())
    # precompute cur_fft
    cur_fft = np.fft.rfft(cur, n=L)
    # compute initial score
    conv = np.fft.irfft(cur_fft * cur_fft, n=L)[:2*n-1]
    cur_max = conv.max()
    cur_sc = 2.0 * n * cur_max
    best_vec = cur.copy()
    best_sc = cur_sc
    # Adam variables
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.3  # learning rate
    # smoothing parameters
    eps_rel = 0.02
    eps_min = 1e-7
    # counters
    it = 0
    no_improve = 0
    # main loop
    while time.monotonic() < deadline:
        it += 1
        # compute smoothed weights
        eps = max(eps_min, cur_max * eps_rel)
        w = np.exp((conv - cur_max) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0/w.size)
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        cand = cur - step
        cand = _project_simplex(cand)
        # evaluate candidate
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[:2*n-1]
        cand_max = cand_conv.max()
        cand_sc = 2.0 * n * cand_max
        # acceptance
        if cand_sc < cur_sc:
            cur = cand
            cur_fft = cand_fft
            conv = cand_conv
            cur_max = cand_max
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            # possibly accept probabilistically to escape local minima
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_fft = cand_fft
                conv = cand_conv
                cur_max = cand_max
                cur_sc = cand_sc
                no_improve = 0
            else:
                no_improve += 1
        # occasional symmetrisation
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[:2*n-1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < cur_sc:
                cur = sym
                cur_fft = sym_fft
                conv = sym_conv
                cur_max = sym_conv.max()
                cur_sc = sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
        # random mass transfer when stuck
        if no_improve > 5000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[:2*n-1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_fft = tmp_fft
                    conv = tmp_conv
                    cur_max = tmp_conv.max()
                    cur_sc = tmp_sc
                    no_improve = 0
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
        # early stop if target achieved
        if best_sc < 1.50274349:
            break
    # return best vector as list
    return [float(x) for x in best_vec]

import time

import numpy as np

# --------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
