import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------------------
    # Initialise – use the provided parent if present, otherwise a simple start
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a modest length fallback
        a0 = rng.random(1024, dtype=np.float64)

    # normalise to a probability simplex (sum = 1, non‑negative)
    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s0

    n = a0.size
    # FFT length: smallest power of two ≥ 3n‑2 (covers all needed convolutions)
    L = 1 << ((3 * n - 2).bit_length())

    # ----------------------------------------------------------------------
    # Helper utilities (vector‑wise, no closures)
    # ----------------------------------------------------------------------
    def _normalize(v: np.ndarray) -> np.ndarray:
        """Set negatives to zero and renormalise to sum 1."""
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            return np.full_like(v, 1.0 / v.size)
        return v / s

    def _score(v: np.ndarray) -> float:
        """Exact AC1 bound for a vector whose sum is 1."""
        a_fft = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    def _grad_and_max(v: np.ndarray, eps: float):
        """Gradient of the soft‑max of the convolution and the current max."""
        a_fft = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        # soft‑max weights
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # convolution of a with the reversed weight vector
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        return 2.0 * grad_full[idx], max_val

    # ----------------------------------------------------------------------
    cur = _normalize(a0)
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    eps_decay = 0.9998         # smoothing decay per iteration
    eps_min = 1e-9
    step0 = 0.8                # base step magnitude
    beta = 0.85                # momentum factor
    momentum = np.zeros_like(cur)
    no_improve = 0
    it = 0
    target = 1.50274349        # stop early if reached

    # ----------------------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # gradient of the smoothed objective
