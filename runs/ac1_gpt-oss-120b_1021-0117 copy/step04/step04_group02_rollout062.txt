import time
import math
import numpy as np

# ------------------------------------------------------------
# Greedy construction of a sparse set with bounded ordered sums
# ------------------------------------------------------------
def _build_positions(n: int, max_ord: int, rng: np.random.Generator):
    """
    Return (positions, sum_counts) where positions is a list of indices
    and sum_counts[s] counts ordered pairs (i,j) with i+j == s.
    No ordered sum count exceeds max_ord.
    """
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)
    positions = []
    order = rng.permutation(n)
    for i in order:
        # self‑pair (i,i) would add 1 to count at 2*i
        if sum_counts[2 * i] >= max_ord:
            continue
        ok = True
        for j in positions:
            s = i + j
            # adding (i,j) and (j,i) would increase count by 2
            if sum_counts[s] + 2 > max_ord:
                ok = False
                break
        if ok:
            # record the ordered contributions
            for j in positions:
                s = i + j
                sum_counts[s] += 2
            sum_counts[2 * i] += 1
            positions.append(i)
    return positions, sum_counts
