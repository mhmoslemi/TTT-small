import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    cond = u - css / np.arange(1, n + 1) > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound for a vector with sum=1
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    """Return 2*n*max(a*a) assuming sum(a)=1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------

                     budget_s: float = 1100,

    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.5)

        a0[:] = 1.0 / a0.size
    else:
        a0 = _project_simplex(a0)               # ensures sum==1

    n = a0.size

    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------

    cur = a0.copy()
    cur_sc = _fast_score(cur, L, n)
    best = cur.copy()

    # algorithmic parameters

    eps_min = 1e-9
    eps_decay = 0.9995
    step0 = 0.5
    it = 0
    no_improve = 0

    # Adam moments (optional, can be disabled by setting beta1=0)
    beta1, beta2 = 0.9, 0.999

    while time.monotonic() < deadline:
