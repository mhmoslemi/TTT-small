import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast AC1 bound for a sequence with sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence whose sum is 1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        if eps is None:
            eps = max(max_val * 0.2, 1e-9)
        else:
            eps = max(eps * eps_decay, 1e-12)
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)[: 2 * n - 1]
        idx = (2 * n - 2) - np.arange(n)          # map to coordinates of a
        g = 2.0 * grad_full[idx]
        g_norm = np.linalg.norm(g) + 1e-12

        # ----- step size -----
        alpha = base_alpha / math.sqrt(it) / g_norm

        # ----- attempt multiplicative (mirror) update -----
        new_a = cur * np.exp(-alpha * g)
        new_a_sum = new_a.sum()
        if new_a_sum <= 0.0:
            new_a = np.full_like(cur, 1.0 / n)
        else:
            new_a /= new_a_sum
        cand_sc = _score(new_a)

        improved = False
        cand = None

        if cand_sc < cur_sc:
            cand = new_a
            improved = True
        else:
            # ----- fallback: projected additive step with backtracking -----
            alpha_f = alpha
            for _ in range(5):
                cand_try = cur - alpha_f * g
                cand_try = _project_simplex(cand_try)
                cand_sc_try = _score(cand_try)
                if cand_sc_try < cur_sc:
                    cand = cand_try
                    cand_sc = cand_sc_try
                    improved = True
                    break
                alpha_f *= 0.5

        # ----- occasional Metropolis acceptance if still no improvement -----
        if not improved:
            cand_try = cur - alpha * g
            cand_try = _project_simplex(cand_try)
            cand_sc_try = _score(cand_try)
            prob = math.exp(-(cand_sc_try - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                cand_sc = cand_sc_try
                improved = True

        # ----- accept if improved -----
        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- targeted random mass transfer when stuck -----
        if no_improve > stagnation_limit:
            # try to reduce the contribution at the current worst shift
            k = max_idx
            i_arr = np.arange(n, dtype=np.int64)
            j_arr = k - i_arr
            mask = (j_arr >= 0) & (j_arr < n)
            if mask.any():
                weights = cur[i_arr[mask]] * cur[j_arr[mask]]
                total_w = weights.sum()
                if total_w > 0:
                    probs = weights / total_w
                    i_sel = rng.choice(i_arr[mask], p=probs)
                    # choose a distant target index
                    t_sel = rng.integers(0, n)
                    while t_sel == i_sel:
                        t_sel = rng.integers(0, n)
                    delta = cur[i_sel] * rng.random() * 0.45
                    tmp = cur.copy()
                    tmp[i_sel] -= delta
                    tmp[t_sel] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp)
                    if tmp_sc < cur_sc:
                        cur = tmp
                        cur_sc = tmp_sc
                        if tmp_sc < best_sc:
                            best_sc = tmp_sc
                            best_vec = tmp.copy()
                        no_improve = 0
                        continue  # skip generic random move

            # generic random move as a fallback
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                    continue
            no_improve = 0  # reset after a try

        # ----- random restart if hopelessly stuck -----
        if no_improve > 20000:
            cur = np.full(n, 1.0 / n)
            cur_sc = _score(cur)
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < improve_target:
            break

    # Return the best sequence found (as plain Python floats)
    return [float(x) for x in best_vec]
