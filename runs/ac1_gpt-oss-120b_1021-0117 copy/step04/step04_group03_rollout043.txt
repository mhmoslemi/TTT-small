    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.5)  # safety margin

    rng = np.random.default_rng(seed)

    # get base vector
    if 'height_sequence_1' in globals():
         base = np.asarray(height_sequence_1, dtype=np.float64)
    else:
         base = None

    # choose n (if base present, use its length; else default)
    if base is not None:
         n = base.size
    else:
         n = 2048
    # define helper to normalize to simplex
    def normalize(v):
         v = np.maximum(v, 0.0)
         s = v.sum()
         if s == 0:
             v = np.full_like(v, 1.0 / v.size)
         else:
             v = v / s
         return v

    # Build initial candidates
    candidates = []
    # Parent (projected)
    if base is not None:
        cand = normalize(base)
        candidates.append(cand)
    # uniform
    uniform = np.full(n, 1.0/n)
    candidates.append(uniform)
    # arcsine shape as previous
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0/np.sqrt((i+0.5)*(n-0.5-i))
    arcsine = arcsine / arcsine.sum()
    candidates.append(arcsine)
    # triangular (peak in center)
    tri = np.minimum(np.arange(n), np.arange(n)[::-1]) + 1
    tri = tri.astype(np.float64)
    tri = tri / tri.sum()
    candidates.append(tri)
    # cosine window
    coswin = 0.5 - 0.5*np.cos(np.pi * i / (n-1))
    coswin = coswin / coswin.sum()
    candidates.append(coswin)

    # Evaluate each using our _score_normed: compute approx conv max then *2*n
    best_vec = None
    best_score = float('inf')
    for cand in candidates:
        sc = _score_normed(cand)  # assume sum=1
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    # Setup variables for gradient descent
    cur = best_vec.copy()
    cur_score = best_score
    best_overall_vec = cur.copy()
    best_overall_score = cur_score

    # Gradient descent parameters
    L = 1 << ((2*n - 1).bit_length())
    eps = 0.5
    eps_decay = 0.9995  # slower decay to keep smoothing; maybe 0.9995
    eps_min = 1e-7
    step0 = 0.5
    beta = 0.9
    momentum = np.zeros_like(cur)

    iter = 0
    no_improve = 0

    while time.monotonic() < deadline:
        iter += 1

        # compute gradient and other stuff
        # compute conv and max
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()

        # compute softmax weights
        eps = max(eps_min, eps * eps_decay)
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1 - beta) * grad

        # step size
        grad_norm = np.linalg.norm(momentum) + 1e-12
        alpha = step0 / (math.sqrt(iter) * grad_norm)

        # candidate update
        cand = cur - alpha * momentum
        # enforce symmetry optionally every few iter
        if (iter % 200) == 0:
            cand = (cand + cand[::-1]) * 0.5

        cand = _project_simplex(cand)

        cand_score = _score_normed(cand)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_overall_score:
                best_overall_score = cand_score
                best_overall_vec = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        # If stagnation, random perturbation
        if no_improve > 3000:
            # random mass transfer
            i_idx = rng.integers(0, n)
            if cur[i_idx] > 0.0:
                delta = cur[i_idx] * rng.random() * 0.3
                j_idx = rng.integers(0, n)
                if j_idx == i_idx:
                    j_idx = (j_idx + 1) % n
                tmp = cur.copy()
                tmp[i_idx] -= delta
                tmp[j_idx] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score_normed(tmp)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    if tmp_score < best_overall_score:
                        best_overall_score = tmp_score
                        best_overall_vec = tmp.copy()
                no_improve = 0

        # early exit if target reached
        if best_overall_score < 1.50274349:
            break

    # return best found as list
    if best_overall_vec is None:
        best_overall_vec = cur
    return [float(x) for x in best_overall_vec]

def _score_normed(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

Now gradient computation:

def _gradient(a, eps):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    max_val = conv.max()
    # softmax weights
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # compute gradient using reversed weights
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * w_fft, n=L)
    idx = (2*n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad

import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (L2 Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Score for a vector (sum is assumed 1)
# ------------------------------------------------------------
def _score(v: np.ndarray) -> float:
    """2 * n * max(full convolution)  (v sums to 1)."""
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(F * F, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Gradient of the smoothed max (soft‑max temperature eps)
# ------------------------------------------------------------
def _grad(v: np.ndarray, eps: float, L: int):
    """Return gradient of the soft‑max approximated max(conv)."""
    n = v.shape[0]
    F = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(F * F, n=L)[: 2 * n - 1]
    mx = conv.max()
    w = np.exp((conv - mx) / eps)
    s = w.sum()
    w = w / s if s != 0.0 else np.full_like(w, 1.0 / w.size)
    w_rev = w[::-1]
    W = np.fft.rfft(w_rev, n=L)
    Gfull = np.fft.irfft(F * W, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * Gfull[idx]
