import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_score(a: np.ndarray, n: int, L: int) -> float:
    """Fast 2·n·max(a*a) assuming sum(a)=1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function vector with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- initialise ----
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(1024, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # sum = 1
    n = cur.size
    # ensure symmetry from the start (optimal solutions are symmetric)
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)

    # FFT length: at least 3n‑2, power of two
    L = 1 << ((3 * n - 2).bit_length())

    # ---- Adam state ----
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    alpha0 = 0.15            # base step size
    eps_opt = 1e-8

    # ---- smoothing for soft‑max ----
    eps0, eps_decay, eps_min = 0.5, 0.9995, 1e-9

    # ---- bookkeeping ----
    best = cur.copy()
    best_sc = _fft_score(best, n, L)
    cur_sc = best_sc
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and soft‑max weights -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w /= w.sum()                     # soft‑max probabilities

        # ----- gradient (sub‑gradient of max) -----
        w_rev = w[::-1]
