def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    return w

    # ---- problem size and FFT length ----
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000  # fallback length
    L = 1 << ((3 * n - 2).bit_length())   # enough for full gradient computation

    # ---- initial candidates (normalized to sum=1) ----
    init_cands = []

    # 1) provided construction
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 0.0:
            a0 /= s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        init_cands.append(a0)

    # 2) reversed version
    if init_cands:
        init_cands.append(init_cands[0][::-1].copy())

    # 3) geometric decay
    r = 0.998
    geo = np.power(r, np.arange(n), dtype=np.float64)
    geo /= geo.sum()
    init_cands.append(geo)

    # 4) random Dirichlet (gamma) sample
    rnd = rng.gamma(1.0, 1.0, size=n)
    rnd /= rnd.sum()
    init_cands.append(rnd)

    # ---- pick best of the initial pool ----
    best_vec = None
    best_sc = np.inf
    for vec in init_cands:
        sc = _score_vec(vec, L, n)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # ---- optimisation hyper‑parameters ----
    step0 = 1.0
    max_backtrack = 12
    backtrack_factor = 0.5
    no_improve = 0
    it = 0
    target = 1.50274349

    # ---- main optimisation loop ----
    while time.monotonic() < deadline:
        it += 1

        # exact subgradient of the max‑convolution
        g, _ = _subgradient(cur, L, n)

        # back‑tracking line search
        step = step0 / math.sqrt(it)
        improved = False
        for _ in range(max_backtrack):
            cand = cur - step * g
            cand = _project_simplex(cand)
            cand_sc = _score_vec(cand, L, n)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best_vec, best_sc = cand.copy(), cand_sc
                    no_improve = 0
                improved = True
                break
            step *= backtrack_factor

        if not improved:
            no_improve += 1

        # occasional symmetrisation (helps with the even‑odd structure)
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym, L, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # random perturbation when progress stalls
        if no_improve >= 5000:
            # small Gaussian jitter around the current best
            delta = rng.normal(scale=0.02, size=n)
            cand = best_vec + delta
            cand = _project_simplex(cand)
            cand_sc = _score_vec(cand, L, n)
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
                cur, cur_sc = cand, cand_sc
                no_improve = 0
            else:
                # random restart from a fresh Dirichlet sample
                cur = rng.gamma(1.0, 1.0, size=n)
                cur /= cur.sum()
                cur_sc = _score_vec(cur, L, n)
                no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

    # ---- final fine‑tuning with remaining time ----
    while time.monotonic() < deadline - 0.05:
        delta = rng.normal(scale=0.005, size=n)
        cand = best_vec + delta
        cand = _project_simplex(cand)
        cand_sc = _score_vec(cand, L, n)
        if cand_sc < best_sc:
            best_vec, best_sc = cand, cand_sc

    # ---- return as plain Python list ----
    return [float(x) for x in best_vec]
