    no_improve = 0

    maxc = np.max(np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1])

import time, math
import numpy as np

def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------
    # 1️⃣  Initialise a feasible vector (sum = 1, non‑negative)
    # ------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # exponential decay on a moderate size if no parent is present
        n0 = 2000
        a0 = np.exp(-np.arange(n0) / 12.0)
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                     # normalize to the simplex

    n = a0.size
    # FFT length ≥ 3n‑2 (convolution of two length‑n vectors)
    L = 1 << ((3 * n - 2).bit_length())

    # work in the log‑domain: u such that softmax(u) = a
    u = np.log(a0 + 1e-30)

    # optimizer state
    v = np.zeros_like(u)                # momentum buffer
    beta = 0.9                          # momentum factor
    step0 = 0.5                         # base step size
    step_decay = 5e-4                   # slow decay of the step size
    eps = 0.02                          # start smoothing for the max
    eps_min = 1e-9
    eps_decay = 0.9995
    sym_period = 2000                  # how often to enforce symmetry
    no_improve = 0
    best = a0.copy()
    # initial score
    a_fft = np.fft.rfft(best, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
    best_score = 2.0 * n * conv.max()
    cur_score = best_score

    it = 0
    # ------------------------------------------------------------
    # 2️⃣  Main optimisation loop
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------
        #   current a and its convolution (already known from last iter)
        # --------------------------------------------------------
        a = np.exp(u - np.max(u))
        a = a / a.sum()

        # --------------------------------------------------------
        #   gradient of the smoothed max‑convolution
        # --------------------------------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        maxc = conv.max()

        # soft‑max weights
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            eps = max(eps_min, eps * eps_decay)
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # gradient w.r.t. a
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * grad_full[idx]

        # chain‑rule to the log‑domain (softmax)
        dot = np.dot(grad_a, a)
        grad_u = a * (grad_a - dot)

        # --------------------------------------------------------
        #   momentum update in the log‑domain
        # --------------------------------------------------------
        lr = step0 / (1.0 + it * step_decay)
        v = beta * v - lr * grad_u
        u = u + v

        # --------------------------------------------------------
        #   new a (after the update) and optional symmetrisation
        # --------------------------------------------------------
        a = np.exp(u - np.max(u))
        a = a / a.sum()
        if it % sym_period == 0:
            a = (a + a[::-1]) * 0.5
            a = a / a.sum()
            u = np.log(a + 1e-30)

        # --------------------------------------------------------
        #   evaluate the new candidate
        # --------------------------------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        cur_score = 2.0 * n * conv.max()

        # --------------------------------------------------------
        #   bookkeeping
        # --------------------------------------------------------
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
        else:
            no_improve += 1

        # --------------------------------------------------------
        #   escape from stagnation
        # --------------------------------------------------------
        if no_improve > 5000:
            # random perturbation around the current best
            noise = rng.normal(scale=0.08, size=u.shape)
            u = np.log(best + 1e-30) + noise
            v = np.zeros_like(u)
            no_improve = 0

        # --------------------------------------------------------
        #   early exit if target reached
        # --------------------------------------------------------
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------
    # 3️⃣  Return the best vector found (as plain Python floats)
    # ------------------------------------------------------------
    return [float(x) for x in best]
