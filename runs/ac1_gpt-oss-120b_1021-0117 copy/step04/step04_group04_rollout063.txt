import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (L2 onto {x>=0, sum=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Score = 2 * n * max(convolution)   (sum is normalized to 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int = None) -> float:
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Core optimisation routine
# ------------------------------------------------------------
def _optimise(start: np.ndarray, deadline: float, rng: np.random.Generator):
    a = _project_simplex(start.astype(np.float64, copy=False))
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())          # length for gradient convolutions
    best = a.copy()
    best_sc = _score(best, L)
    cur = best.copy()
    cur_sc = best_sc

    # hyper‑parameters
    eps0 = 1.0
    eps_decay = 0.9996
    eps_min = 1e-7
    step0 = 0.6
    momentum = 0.9
    v = np.zeros_like(a)
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & smoothed max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        ws = w.sum()
        if ws == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= ws

        # ----- gradient of soft‑max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- momentum step & projection -----
        v = momentum * v + (1.0 - momentum) * g
        gn = np.linalg.norm(v) + 1e-12
        alpha = step0 / math.sqrt(it) / gn
        cand = cur - alpha * v
        cand = _project_simplex(cand)

        cand_sc = _score(cand, L)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                stall = 0
        else:

            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc
            stall += 1

        trial[i] -= delta
        trial[j] += delta
        trial = _project_simplex(trial)
        trial_sc = _score(trial, L)
        if trial_sc < best_sc:
            best, best_sc = trial, trial_sc

    return best


# ------------------------------------------------------------
# Public entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # tiny safety margin

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # simple analytic start: exponential decay, length ≈2000
        n0 = 2000
        a0 = np.exp(-np.arange(n0) / 200.0)

    # ----- run optimiser ------------------------------------------------
    best = _optimise(a0, deadline, rng)

    # ----- return as plain Python list ---------------------------------
    return [float(x) for x in best]
