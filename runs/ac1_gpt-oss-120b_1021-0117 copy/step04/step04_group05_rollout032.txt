def optimize_start(init_vec, rng, deadline):
    cur = init_vec.copy()
    n = len(cur)
    L_full = 1 << ((3*n - 2).bit_length())
    L_conv = 1 << ((2*n - 1).bit_length())
    cur_fft = np.fft.rfft(cur, n=L_full)
    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[:2*n-1]
    cur_max = cur_conv.max()
    cur_sc = 2.0 * n * cur_max
    best_vec = cur.copy()
    best_sc = cur_sc
    step0 = 0.8
    eps_rel = 0.02
    eps_min = 1e-7
    it = 0
    no_improve = 0
    m_dir = np.zeros_like(cur)  # momentum direction
    beta = 0.9

    while time.monotonic() < deadline:
        it += 1
        eps = max(eps_min, cur_max * eps_rel)
        w = np.exp((cur_conv - cur_max) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient as convolution of cur and reversed w
        w_rev = w[::-1]
        # ensure both FFT length same L_full
        w_rev_fft = np.fft.rfft(w_rev, n=L_full)
        grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L_full)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # momentum
        m_dir = beta * m_dir + (1.0 - beta) * g

        # step size schedule
        alpha = step0 / math.sqrt(it)
        # direction is m_dir (instead of g)
        direction = m_dir

        accepted = False
        cand = None
        cand_sc = None
        for _ in range(6):
            cand_vec = cur - alpha * direction
            cand_vec = _project_simplex(cand_vec)
            cand_fft = np.fft.rfft(cand_vec, n=L_full)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L_full)[:2*n-1]
            cand_max = cand_conv.max()
            cand_sc = 2.0 * n * cand_max
            if cand_sc < cur_sc:
                accepted = True
                cand = cand_vec
                cand_fft = cand_fft
                cand_conv = cand_conv
                cand_max = cand_max
                break
            alpha *= 0.5

        if not accepted:
            # occasional stochastic acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                accepted = True
                cand = cand_vec
                cand_fft = cand_fft
                cand_conv = cand_conv
                cand_max = cand_max
        if accepted:
            cur = cand
            cur_fft = cand_fft
            cur_conv = cand_conv
            cur_max = cand_max
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            no_improve += 1

        # symmetrization every 2000 iters
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L_full)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L_full)[:2*n-1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < cur_sc:
                cur = sym
                cur_fft = sym_fft
                cur_conv = sym_conv
                cur_max = sym_conv.max()
                cur_sc = sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
        # random mass transfer when stuck
        if no_improve > 2000:
            # random perturbation
            idx_i = rng.integers(0, n)
            idx_j = rng.integers(0, n)
            if idx_i != idx_j:
                delta = cur[idx_i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[idx_i] -= delta
                tmp[idx_j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L_full)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L_full)[:2*n-1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_fft = tmp_fft
                    cur_conv = tmp_conv
                    cur_max = tmp_conv.max()
                    cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                no_improve = 0

        # early exit if target achieved
        if best_sc < 1.50274349:
            break

    return best_vec, best_sc

    # generate a random smooth vector with some variability
    # generate exponential random values, then smooth with variable kernel.
    rand_vals = rng.exponential(scale=1.0, size=n)
    # smoothing kernel size random between 5 and 200
    k = rng.integers(5, 200)
    kernel = np.ones(k) / k
    smooth = np.convolve(rand_vals, kernel, mode='same')
    # Ensure nonnegative
    smooth = np.maximum(smooth, 0)
    # Project to simplex
    return _project_simplex(smooth)

import time, math
import numpy as np

# ----------------------------------------------------------------------
# Project onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random smooth initialization (few Gaussian bumps)
# ----------------------------------------------------------------------
def _random_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a smooth, non‑negative vector of length n and project to simplex."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    vec = np.zeros(n, dtype=np.float64)
    modes = rng.integers(1, 4)                # 1‑3 bumps
    for _ in range(modes):
        centre = rng.random()
        width = rng.random() * 0.3 + 0.05      # 0.05 … 0.35
        height = rng.random()
        vec += height * np.exp(-((x - centre) ** 2) / (2.0 * width ** 2))
    # tiny noise to avoid exact symmetry artefacts
    vec += 0.01 * rng.random(n)
    vec = np.maximum(vec, 0.0)
    return _project_simplex(vec)

def _optimize(start_vec: np.ndarray, rng: np.random.Generator, deadline: float):
    """Run the iterative improvement on a single starting vector."""
    n = start_vec.shape[0]
    # FFT length that comfortably contains a convolution of length 3n‑2
    L_full = 1 << ((3 * n - 2).bit_length())

    cur = start_vec.copy()
    cur_fft = np.fft.rfft(cur, n=L_full)
    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
    cur_max = cur_conv.max()
    cur_score = 2.0 * n * cur_max

    best_vec = cur.copy()
    best_score = cur_score

    step0 = 0.8
    eps_rel = 0.02
    eps_min = 1e-7
    it = 0
    no_improve = 0
    momentum = np.zeros_like(cur)
    beta = 0.9

    while time.monotonic() < deadline:
        it += 1

        # ------- smoothed max weights (soft‑max) -------
        eps = max(eps_min, cur_max * eps_rel)
        w = np.exp((cur_conv - cur_max) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ------- gradient of smoothed objective -------
        w_rev = w[::-1]

        idx = (2 * n - 2) - np.arange(n)            # map to coordinate order
        g = 2.0 * grad_full[idx]

        direction = momentum

        alpha = step0 / math.sqrt(it)

            if cand_score < cur_score:

                break
            alpha *= 0.5

                cand = cand_vec

            cur_score = cand_score
            no_improve = 0
