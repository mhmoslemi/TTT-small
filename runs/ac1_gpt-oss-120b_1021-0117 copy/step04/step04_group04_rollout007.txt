def _score(a):
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af*af, n=L)[:2*n-1]

def _softmax(y):
    y_max = np.max(y)
    exps = np.exp(y - y_max)
    return exps / np.sum(exps)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (L2 Euclidean) – returns vector with sum == 1, x >= 0
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Softmax – guaranteed sum == 1, non‑negative
# ------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ------------------------------------------------------------
# Objective (same as evaluator when sum == 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # -----------------------------------------------------------------
        # occasional bookkeeping
        # -----------------------------------------------------------------
        if it % 30 == 0:
            sc = float(2.0 * n * max_val)   # identical to _score(a) because sum(a)=1
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                if best_sc < target:
                    break

        if it % 5000 == 0:
            # enforce (approximate) symmetry – often helps
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            y = np.log(sym + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            sc_sym = _score(sym)
            if sc_sym < best_sc:
                best_sc = sc_sym
                best_a = sym.copy()
                if best_sc < target:
                    break

    # -----------------------------------------------------------------
    # simple local random search – cheap hill‑climbing if we are still above target
    # -----------------------------------------------------------------
    if best_sc >= target and time.monotonic() < deadline:
        cur = best_a.copy()
        cur_sc = best_sc
        max_local_iters = 20000
        local_it = 0
        while time.monotonic() < deadline and local_it < max_local_iters:
            local_it += 1
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if cur[i] <= 0.0:
                continue
            delta = cur[i] * rng.random() * 0.3
            nxt = cur.copy()
            nxt[i] -= delta
            nxt[j] += delta
            sc_nxt = _score(nxt)
            if sc_nxt < cur_sc:
                cur, cur_sc = nxt, sc_nxt
                if sc_nxt < best_sc:
                    best_sc, best_a = sc_nxt, nxt.copy()
                    if best_sc < target:
                        break

    # -----------------------------------------------------------------
    # final symmetry polishing
    # -----------------------------------------------------------------
    sym_final = (best_a + best_a[::-1]) * 0.5
    sym_final = _project_simplex(sym_final)
    sc_sym_final = _score(sym_final)
    if sc_sym_final < best_sc:
        best_a = sym_final
        best_sc = sc_sym_final

    # -----------------------------------------------------------------
    # return plain Python list
    # -----------------------------------------------------------------
    return [float(v) for v in best_a]
