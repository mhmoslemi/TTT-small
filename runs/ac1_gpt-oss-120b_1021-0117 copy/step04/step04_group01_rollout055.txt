            w = np.exp((conv - conv.max()) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum

            # ----- gradient of the weighted sum -----
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(af * w_fft, n=L)   # length >= 3n-2
            idx = (2 * n - 2) - np.arange(n)            # map to gradient entries
            grad = 2.0 * grad_full[idx]

            # ----- line‑search with momentum -----
            step = step0 / math.sqrt(it)
            improved = False
            for _ in range(max_backtrack):
                vel = mu * vel - step * grad
                cand = a + vel
                cand = _project_simplex(cand)
                cand_sc, _, _ = _score_and_fft(cand)
                if cand_sc < sc:
                    a = cand
                    improved = True
                    break
                # backtrack
                step *= back_factor
                vel.fill(0.0)

            if not improved:
                no_improve += 1
            else:
                no_improve = 0

            # ----- occasional random perturbation when stuck -----
            if no_improve > patience:
                i = rng.integers(0, n)
                if a[i] > 1e-12:
                    delta = a[i] * rng.random() * 0.3
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    a[i] -= delta
                    a[j] += delta
                    a = _project_simplex(a)
                no_improve = 0

            # ----- periodic symmetrisation (helps even‑odd pattern) -----
            if it % 2000 == 0:
                sym = (a + a[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc, _, _ = _score_and_fft(sym)
                if sym_sc < sc:
                    a = sym
                    vel.fill(0.0)

    # ----- start from the supplied parent if present ------------------
    if "height_sequence_1" in globals():
        _run_opt(a0)

    # ----- random restarts ------------------------------------------------
    while time.monotonic() < deadline - 1.0:
        rnd = rng.gamma(1.0, 1.0, size=n)
        rnd = rnd / rnd.sum()
        _run_opt(rnd)

    # ----- final fine‑tuning with tiny Gaussian noise --------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    while time.monotonic() < deadline - 0.05:
        delta = rng.normal(scale=0.001, size=n)
        cand = best_vec + delta
        cand = _project_simplex(cand)
        cand_sc, _, _ = _score_and_fft(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()

    # return plain Python list
    return [float(v) for v in best_vec]
