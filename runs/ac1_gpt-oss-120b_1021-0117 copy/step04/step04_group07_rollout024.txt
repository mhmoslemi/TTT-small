import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection of a vector onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound for a vector already on the simplex
# ------------------------------------------------------------
def _score_simplex(a: np.ndarray, L: int, n: int) -> float:
    """Return 2*n*max(conv)/(sum(a)^2) with sum(a)≈1."""
    s = a.sum()
    if s <= 0.0:
        return np.inf
    fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

    n = a.size
