import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto the probability simplex)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Max of self‑convolution (using FFT, returns max and auxiliary data)
# ------------------------------------------------------------
def _maxconv(a: np.ndarray, L: int):
    """Return max(a*a) and also the full convolution and FFT of a."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * a.shape[0] - 1]
    return conv.max(), conv, a_fft

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a random start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2048, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    n = cur.size
    target_sum = math.sqrt(2.0 * n)           # sum that makes the bound = max(conv)

    s = cur.sum()
    if s == 0.0:
        cur = np.full(n, target_sum / n, dtype=np.float64)
    else:
        cur = cur * (target_sum / s)

    # ------------------------------------------------------------
    # FFT planning – length at least 3n‑2 (power of two)
    # ------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    best_vec = cur.copy()
    best_sc, _, _ = _maxconv(best_vec, L)   # best score = current max(conv)
    cur_sc = best_sc

    # algorithmic parameters
    step0 = 0.8
    beta = 0.9
    eps0 = 1.0
    eps_decay = 0.9995
    eps_min = 1e-8
    momentum = np.zeros_like(cur)
    no_improve = 0
    it = 0
    target = 1.50274349

    # LP‑related parameters
    lp_interval = 200          # frequency of LP calls
    lp_K = 120                 # number of active convolution constraints
    mix_factor = 0.5           # initial mixing weight for LP update

    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        M, conv, a_fft = _maxconv(cur, L)
        cur_sc = M

        # ----- keep best -----
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # --------------------------------------------------------
        # Smoothed‑soft‑max gradient step (matrix‑free)
        # --------------------------------------------------------
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - M) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- back‑tracking line search -----
        base_alpha = step0 / math.sqrt(it)
        alpha = base_alpha
        cand = None
        cand_sc = None
        for _ in range(8):
            cand_tmp = cur - alpha * momentum
            cand_tmp = _project_simplex(cand_tmp) * target_sum
            cand_tmp_sc, _, _ = _maxconv(cand_tmp, L)
            if cand_tmp_sc <= cur_sc:
                cand, cand_sc = cand_tmp, cand_tmp_sc
                break
            alpha *= 0.5
        if cand is None:
            cand = cand_tmp
            cand_sc = cand_tmp_sc

        # ----- accept / reject -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
            no_improve = 0
        else:
            # occasional Metropolis uphill move
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # --------------------------------------------------------
        # Linear‑programming boost (sparse, few active constraints)
        # --------------------------------------------------------
        if it % lp_interval == 0:
            K = min(lp_K, conv.size)
            top_idx = np.argpartition(conv, -K)[-K:]
            rows, cols, data = [], [], []
            for ridx, k in enumerate(top_idx):
                i_start = max(0, k - (n - 1))
                i_end = min(k, n - 1)
                for i in range(i_start, i_end + 1):
                    rows.append(ridx)
                    cols.append(k - i)
                    data.append(cur[i])
            A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(K, n))
            b_ub = np.full(K, M, dtype=np.float64)
            c = -np.ones(n, dtype=np.float64)  # maximise sum(g)

            try:
                lp_res = scipy.optimize.linprog(
                    c, A_ub=A_ub, b_ub=b_ub,
                    bounds=[(0, None)] * n,
                    method='highs',
                    options={'disp': False}
                )
                if lp_res.success and lp_res.x is not None:
                    g_lp = lp_res.x
                    sum_g = g_lp.sum()
                    if sum_g > 0.0:
                        g_lp = g_lp * (target_sum / sum_g)

                        # --- try full replacement ---
                        repl_sc, _, _ = _maxconv(g_lp, L)
                        if repl_sc < cur_sc:
                            cur, cur_sc = g_lp, repl_sc
                            if repl_sc < best_sc:
                                best_sc = repl_sc
                                best_vec = g_lp.copy()
                        else:
                            # --- otherwise blend ---
                            t = max(0.1, mix_factor * (0.97 ** (it // lp_interval)))
                            cand_lp = (1.0 - t) * cur + t * g_lp
                            cand_lp = cand_lp * (target_sum / cand_lp.sum())
                            cand_lp_sc, _, _ = _maxconv(cand_lp, L)
                            if cand_lp_sc < cur_sc:
                                cur, cur_sc = cand_lp, cand_lp_sc
                                if cand_lp_sc < best_sc:
                                    best_sc = cand_lp_sc
                                    best_vec = cand_lp.copy()
            except Exception:
                pass   # silently ignore LP failures

        # --------------------------------------------------------
        # Periodic symmetrisation (helps with Toeplitz structure)
        # --------------------------------------------------------
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym) * target_sum
            sym_sc, _, _ = _maxconv(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # --------------------------------------------------------
        # Random mass‑transfer when stuck
        # --------------------------------------------------------
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp) * target_sum
                tmp_sc, _, _ = _maxconv(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate found (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
