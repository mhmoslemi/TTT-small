def _project_simplex(v):
    # same
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css/np.arange(1,n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

    n = a.shape[0]

    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Exact score (sum of coefficients is assumed to be 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv))


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that lowers the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise a feasible vector
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=float)
        a = np.maximum(a, 0.0)
    else:
        n = 1024
        a = np.full(n, 1.0 / n, dtype=float)

    a = _project_simplex(a)
    n = a.size

    # construct a few simple starters
    candidates = [a]
    candidates.append(np.full(n, 1.0 / n, dtype=float))
    i = np.arange(n, dtype=float)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcs = arcs / arcs.sum()
    candidates.append(arcs)

    # pick the best starter according to the exact score
    best_vec = None
    best_sc = float("inf")
    for cand in candidates:
        cand = _project_simplex(cand)
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    # --------------------------------------------------------
    # Parameters for smoothed‑max gradient descent
    # --------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())       # FFT length (next power of two)
    eps = 1.0
    eps_decay = 0.9999
    eps_min = 1e-8
    step0 = 1.0
    it = 0
    no_improve = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution and its maximum
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # smooth approximation of the arg‑max
        eps = max(eps_min, eps * eps_decay)
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # projected step
        grad_norm = np.linalg.norm(g) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm
        cand = cur - alpha * g
        cand = _project_simplex(cand)

        # enforce symmetry (helps a lot for this problem)
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # accept if we improve
        if cand_sc + 1e-12 < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < 1.50274349:
                    break
        else:
            no_improve += 1

        # occasional random mass transfer to escape plateaus
        if no_improve > 4000:
            i_idx = rng.integers(0, n)
            if cur[i_idx] > 0.0:
                delta = cur[i_idx] * rng.random() * 0.3
                j_idx = rng.integers(0, n)
                if j_idx == i_idx:
                    j_idx = (j_idx + 1) % n
                tmp = cur.copy()
                tmp[i_idx] -= delta
                tmp[j_idx] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc + 1e-12 < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0
            else:
                # if the chosen index is empty, just jitter a bit
                cur = (cur + rng.random(n) * 0.01) * 0.5
                cur = _project_simplex(cur)
                no_improve = 0

    # --------------------------------------------------------
    # Return the best vector seen (as a plain list of floats)
    # --------------------------------------------------------
    if best_vec is None or best_vec.size == 0:
        best_vec = cur
    return [float(x) for x in best_vec]
