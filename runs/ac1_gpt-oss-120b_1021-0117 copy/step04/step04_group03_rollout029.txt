def exact_grad(v, max_idx):
    n = v.size
    grad = np.zeros_like(v)
    s = max_idx
    if s < n:
        grad[:s+1] = 2 * v[s::-1]

        offset = s - (n-1)

    return grad

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Convolution‑based AC1 score (sum normalized to 1)
# ------------------------------------------------------------
def _compute_score(v: np.ndarray, L: int, n: int) -> float:
    """Return 2·n·max(conv(v,v)) assuming sum(v)=1."""
    a_fft = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Symmetric beta‑type shape a_i ∝ ((i+½)(n‑½‑i))**p
# ------------------------------------------------------------
def _beta_shape(p: float, i: np.ndarray, n: int) -> np.ndarray:
    a = ((i + 0.5) * (n - 0.5 - i)) ** p
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / a_sum
    a = (a + a[::-1]) * 0.5
    a = a / a.sum()
    return a
