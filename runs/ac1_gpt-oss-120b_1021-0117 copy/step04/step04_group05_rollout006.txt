import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # ------------------------------------------------------------------
    # Initialise vector (use provided parent if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 1024
        cur = np.exp(-np.arange(n_fallback, dtype=np.float64) / (n_fallback / 10.0))
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)               # enforce sum == 1, non‑negative
    n = cur.shape[0]

    # ------------------------------------------------------------------
    # FFT configuration (single length works for both convolution and gradient)
    # ------------------------------------------------------------------
    L_fft = 1 << ((3 * n - 2).bit_length())   # ≥ 3n‑2, power of two
    cur_fft = np.fft.rfft(cur, n=L_fft)

    # current full convolution and its maximum
    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_fft)[: 2 * n - 1]
    cur_max = cur_conv.max()
    cur_score = 2.0 * n * cur_max               # sum == 1 ⇒ denominator = 1
    best_vec = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Adam optimiser state
    # ------------------------------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    alpha0 = 0.1                               # base learning rate

    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- smoothed‑max weights (softmax approximation) -----
        eps_rel = max(1e-12, cur_max * 0.02)
        w = np.exp((cur_conv - cur_max) / eps_rel)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L_fft)
        grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L_fft)
        idx = (2 * n - 2) - np.arange(n)          # extract relevant slice
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha0 / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step * m_hat
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_fft = np.fft.rfft(cand, n=L_fft)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L_fft)[: 2 * n - 1]
        cand_max = cand_conv.max()
        cand_score = 2.0 * n * cand_max

        if cand_score < cur_score:                # monotone improvement
            cur = cand
            cur_fft = cand_fft
            cur_conv = cand_conv
            cur_max = cand_max
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:

                best_vec = cand.copy()
        else:                                      # occasional uphill move

        if best_score < target:
            break

    # ------------------------------------------------------------------
    # Return the best-found candidate as a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
