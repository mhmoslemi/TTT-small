import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Convolution (self) via FFT, length 2*n-1
# ------------------------------------------------------------
def _self_conv(x: np.ndarray, L: int) -> np.ndarray:
    """Return convolution of x with itself, truncated to 2*len-1."""
    fft = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * x.size - 1]
    return conv

# Score for a vector (exact evaluator formula)
# ------------------------------------------------------------
def _score_vec(x: np.ndarray, Lconv: int) -> float:
    """2 * n * max(conv) / (sum(x)^2).  Used for local search."""
    n = x.size
    conv = _self_conv(x, Lconv)
    maxb = conv.max()
    s = x.sum()
    # sum(x) is kept ≈1 in the algorithm but we retain the exact formula
    return float(2.0 * n * maxb / (s * s))


# ------------------------------------------------------------
# Candidate families
# ------------------------------------------------------------
def _make_power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-0.5-i))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - 0.5 - i)
    a = np.power(denom, -p, where=denom > 0, out=np.ones_like(denom))
    a /= a.sum()
    return a


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- base vector -------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base = np.maximum(base, 0.0)
        base = _project_simplex(base)
    else:
        n = 2048
        base = np.full(n, 1.0 / n, dtype=np.float64)

    n = base.size
    Lconv = 1 << ((2 * n - 1).bit_length())          # FFT length for self‑conv
    Lgrad = 1 << ((3 * n - 2).bit_length())          # FFT length for grad conv

    # ----- generate cheap candidates ------------------------------------
    cand_list = [base]

    # uniform
    cand_list.append(np.full(n, 1.0 / n, dtype=np.float64))

    # arcsine (p=0.5) and a few nearby powers
    for p in np.linspace(0.0, 0.5, 7):
        cand_list.append(_make_power_shape(n, p))

    # a few random Dirichlet draws (α=0.8 gives slightly edge‑heavy)
    for _ in range(4):
        rand = rng.dirichlet(np.full(n, 0.8))
        cand_list.append(rand)

    best_vec = None
    best_sc = float("inf")
    for c in cand_list:
        sc = _score_vec(c, Lconv)
        if sc < best_sc:
            best_sc = sc
            best_vec = c.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    # ----- smoothed‑max gradient descent -------------------------------
    eps = 0.2               # initial smoothing
    eps_decay = 0.9995
    eps_min = 1e-8
    step0 = 1.0
    momentum = np.zeros_like(cur)
    beta = 0.85
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # convolution of current vector
        a_fft = np.fft.rfft(cur, n=Lgrad)
        conv = np.fft.irfft(a_fft * a_fft, n=Lgrad)[: 2 * n - 1]
        M = conv.max()

        # smoothing weights (soft‑max)
        eps = max(eps_min, eps * eps_decay)
        w = np.exp((conv - M) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient = 2 * (a * w_rev)   (via convolution)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=Lgrad)
        grad_full = np.fft.irfft(a_fft * w_fft, n=Lgrad)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g
        d = -momentum

        # step size (scale‑invariant)
        grad_norm = np.linalg.norm(d) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm

        cand = cur + alpha * d
        cand = _project_simplex(cand)

        # enforce symmetry (helps the symmetric problem)
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_sc = _score_vec(cand, Lconv)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            if no_improve > 2500:
                # random mass transfer to escape stagnation
                i = rng.integers(0, n)
                if cur[i] > 1e-12:
                    delta = cur[i] * rng.random() * 0.25
                    j = rng.integers(0, n)
                    if j == i:
                        j = (j + 1) % n
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp = (tmp + tmp[::-1]) * 0.5
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score_vec(tmp, Lconv)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        if tmp_sc < best_sc:
                            best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0

        # early exit when target met
        if best_sc < 1.50274349:
            break

    # ----- optional final linear‑program improvement --------------------
    # If time permits, run a cheap LP using the current vector as bound.
    time_left = deadline - time.monotonic()
    if time_left > 30.0:
        try:
            import cvxpy as cp

            # build sparse convolution matrix for current cur
            rows = np.empty(n * n, dtype=np.int32)
            cols = np.empty(n * n, dtype=np.int32)
            data = np.empty(n * n, dtype=np.float64)
            pos = 0
            for j in range(n):
                rows[pos : pos + n] = np.arange(j, j + n, dtype=np.int32)
                cols[pos : pos + n] = j
                data[pos : pos + n] = cur
                pos += n
            A = cp.Parameter((2 * n - 1, n), value=sp.csr_matrix((data, (rows, cols)),
                                                                 shape=(2 * n - 1, n)),
                             is_sparse=True)

            M = _self_conv(cur, Lconv).max()
            g = cp.Variable(n, nonneg=True)
            constraints = [A @ g <= M, g <= 1000.0]
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False,
                       abstol=1e-8, reltol=1e-8, feastol=1e-8, max_iters=5000)

            if g.value is not None:
                g_opt = np.asarray(g.value).flatten()
                if g_opt.sum() > 1e-12:
                    g_opt /= g_opt.sum()
                    t = 0.15
                    new_cur = (1.0 - t) * cur + t * g_opt
                    new_cur = _project_simplex(new_cur)
                    new_cur = (new_cur + new_cur[::-1]) * 0.5
                    new_cur = _project_simplex(new_cur)
                    new_sc = _score_vec(new_cur, Lconv)
                    if new_sc < best_sc:
                        best_sc, best_vec = new_sc, new_cur.copy()
        except Exception:
            pass  # LP optional – ignore any failure

    # ------------------------------------------------------------
    # Return result as a plain list
    # ------------------------------------------------------------
    if best_vec is None:
        best_vec = cur
    return [float(v) for v in best_vec]
