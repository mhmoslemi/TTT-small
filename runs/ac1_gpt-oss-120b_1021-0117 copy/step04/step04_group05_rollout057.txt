# --------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    w = np.exp((conv - cur_max) / eps)
    s = w.sum()
    if s == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= s
    return w

                cand = _project_simplex(cand)
                cand_fft = np.fft.rfft(cand, n=L_full)
                cand_conv = _self_conv_fft(cand_fft, L_full, n)
                cand_max = cand_conv.max()
                cand_score = _score_from_max(cand_max)
                if cand_score < cur_score:
                    accepted = True
                    break
                alpha *= 0.5

            if not accepted:
                # occasional probabilistic jump
                prob = math.exp(-(cand_score - cur_score) / (alpha + 1e-12))
                if rng.random() < prob:
                    accepted = True

            if accepted:
                cur, cur_fft, cur_conv = cand, cand_fft, cand_conv
                cur_max, cur_score = cand_max, cand_score
                no_improve = 0
                if cur_score < best_score:
                    best_score, best_vec = cur_score, cur.copy()
            else:
                no_improve += 1

            # ----- periodic symmetrisation -----
            if it % 1000 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_fft = np.fft.rfft(sym, n=L_full)
                sym_conv = _self_conv_fft(sym_fft, L_full, n)
                sym_max = sym_conv.max()
                sym_score = _score_from_max(sym_max)
                if sym_score < cur_score:
                    cur, cur_fft, cur_conv = sym, sym_fft, sym_conv
                    cur_max, cur_score = sym_max, sym_score
                    no_improve = 0
                if sym_score < best_score:
                    best_score, best_vec = sym_score, sym.copy()

            # ----- random mass transfer when stuck -----
            if no_improve > 3000:
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    delta = cur[i] * rng.random() * 0.3
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_fft = np.fft.rfft(tmp, n=L_full)
                    tmp_conv = _self_conv_fft(tmp_fft, L_full, n)
                    tmp_max = tmp_conv.max()
                    tmp_score = _score_from_max(tmp_max)
                    if tmp_score < cur_score:
                        cur, cur_fft, cur_conv = tmp, tmp_fft, tmp_conv
                        cur_max, cur_score = tmp_max, tmp_score
                    if tmp_score < best_score:
                        best_score, best_vec = tmp_score, tmp.copy()
                    no_improve = 0

            # ----- early exit if target reached -----
            if best_score < 1.50274349:
                break

    # fallback safety
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    # final conversion to plain Python list
    return [float(x) for x in best_vec]
