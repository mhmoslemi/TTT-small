def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(vec: np.ndarray, L: int) -> float:
    """Return 2*n*max(conv(vec,vec)) for a vector with sum=1."""
    n = vec.shape[0]
    af = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())
