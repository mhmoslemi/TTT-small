import time

# ------------------------------------------------------------
# Project onto the probability simplex (sum=1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

            w = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)
        else:
            active = np.where(conv >= max_val - 1e-12)[0]
            w = np.zeros_like(conv)
            if active.size:
                w[active] = 1.0 / active.size

        # ----- gradient of the (smoothed) objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = step0 / math.sqrt(it + 1.0)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur = cand

        # ----- random perturbation when stuck -----
        if no_improve > 5000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                    no_improve = 0
                cur = tmp

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------
    # Return the best candidate found
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
