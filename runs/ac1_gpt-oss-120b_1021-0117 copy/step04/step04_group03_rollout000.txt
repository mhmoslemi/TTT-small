import time

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast score (sum is not necessarily 1)
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(conv) / (sum(a)^2) using FFT, L is power‑of‑2 >= 2n‑1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    return 2.0 * n * max_val / (s * s)

# ------------------------------------------------------------

    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = _project_simplex(a)    # normalise to sum=1
    return a
