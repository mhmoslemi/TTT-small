import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Full AC1 score (exact, used for bookkeeping)
# ------------------------------------------------------------
def _full_score(a: np.ndarray) -> float:
    n = a.shape[0]
    if a.sum() < 0.01:
        return float("inf")
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (a.sum() ** 2))

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----------- initial sequence ---------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fall‑back: decaying exponential, length ~2000
        n0 = 2000
        a0 = np.exp(-np.arange(n0) / 200.0)

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)          # enforce sum = 1
    n = a0.shape[0]

    # FFT length that comfortably covers a * w_rev convolution
    L = 1 << ((3 * n - 2).bit_length())

    # Adam hyper‑parameters
    alpha = 0.05                # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    m = np.zeros_like(a0)       # first moment
    v = np.zeros_like(a0)       # second moment

    # smoothing for soft‑max max‑approximation
    eps0 = 0.6
    eps_decay = 0.9996
    eps_min = 1e-9

    best = a0.copy()
    best_score = _full_score(best)

    cur = best.copy()
    cur_score = best_score

    no_improve = 0
    it = 0

    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution --------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]

        # ----- soft‑max weights for the max -----------------------
        max_val = conv.max()
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:

        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----------------
        w_rev = w[::-1]
