        max_val = conv.max()

        # ---- soft‑max weights (log‑sum‑exp approximation of the max) ------------
        w = np.exp((conv - max_val) / eps)
        ws = w.sum()
        if ws == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= ws

        # ---- gradient of the smoothed objective ---------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[idx]

        # ---- Adam moment updates -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v_sq = beta2 * v_sq + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v_sq / (1.0 - beta2 ** it)

        # ---- parameter update (gradient descent) ---------------------------------
        step = -alpha * m_hat / (np.sqrt(v_hat) + 1e-8)
        cand = cur + step
        cand = _clip_and_normalise(cand)

        # ---- score of the candidate ---------------------------------------------
        cand_score = _score_fft(cand)

        # ---- accept if improvement ------------------------------------------------
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
        else:
            no_improve += 1
            # occasional probabilistic acceptance to escape shallow traps
            if rng.random() < 0.001:
                cur, cur_score = cand, cand_score

        # ---- maintain the best ever ------------------------------------------------
        if cand_score < best_score:
            best, best_score = cand.copy(), cand_score
            no_improve = 0

        # ---- periodic symmetrisation (helps the autocorrelation shape) ----------
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _clip_and_normalise(sym)
            sym_score = _score_fft(sym)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
            if sym_score < best_score:
                best, best_score = sym.copy(), sym_score

        # ---- random mass transfer if stuck for a while ---------------------------
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _clip_and_normalise(tmp)
                tmp_score = _score_fft(tmp)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score
                no_improve = 0

        # ---- full restart with a fresh sparse random vector if hopeless ------------
        if no_improve > 20000:
            # sparse random initialise: ~5 % non‑zero entries
            k = max(2, n // 20)
            idxs = rng.choice(n, size=k, replace=False)
            vals = rng.random(k)
            cur = np.zeros(n)
            cur[idxs] = vals
            cur = _clip_and_normalise(cur)
            cur_score = _score_fft(cur)
            m.fill(0.0)
            v_sq.fill(0.0)
            eps = 0.5
            no_improve = 0

        # ---- early exit if we have already beaten the target ----------------------
        if best_score < 1.50274349:
            break

    # return the best‑found sequence as plain Python floats
    return [float(x) for x in best]
