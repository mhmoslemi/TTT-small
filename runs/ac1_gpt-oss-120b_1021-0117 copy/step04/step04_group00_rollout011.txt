import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Return a non‑negative vector with sum = 1."""
    s = v.sum()
    if s == 0.0:
        # uniform fallback – should never happen for a sensible start
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _self_conv_max(a: np.ndarray, L: int):
    """Return max of a*a (linear convolution) and the FFT of a."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _softmax_weights(conv: np.ndarray, max_val: float, eps: float) -> np.ndarray:
    """Soft‑max over the convolution values, temperature = eps."""
    # (conv‑max)/eps ≤ 0, so exponentials do not overflow
    w = np.exp((conv - max_val) / eps)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size, dtype=np.float64)
    return w / s


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of the smoothed max‑conv objective.
    g_i = 2 * (a * w_rev)_ { (2n‑2)‑i } .
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)          # slice yielding length‑n vector
    return 2.0 * grad_full[idx]


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative list of coefficients whose
    AC1 bound (2 n·max(a*a)/sum(a)^2) is as low as possible.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin before the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initial vector – use the parent if it exists
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
    else:
        # a modest random start when no parent is supplied
        cur = rng.random(1024, dtype=np.float64)

    cur = _normalize(cur)
    n = cur.size

    # FFT length – must handle convolution of lengths n and 2n‑1
    L = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2

    # --------------------------------------------------------
    # Algorithmic parameters
    # --------------------------------------------------------
    eps0 = 0.5                       # initial soft‑max temperature
    eps_decay = 0.9992               # exponential decay per iteration
    eps_min = 1e-12
    step0 = 0.35                     # base step size for exponentiated update
    beta = 0.9                       # momentum factor
    momentum = np.zeros_like(cur)    # initialise momentum
    no_improve = 0
    best_vec = cur.copy()
    # initial score (sum is 1, thus score = 2·n·max_conv)
    best_max, a_fft, _ = _self_conv_max(cur, L)
    best_score = 2.0 * n * best_max
    target_score = 1.50274349
    it = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- current convolution and max ----
        cur_max, a_fft, conv = _self_conv_max(cur, L)

        # ---- soft‑max weights over the convolution ----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = _softmax_weights(conv, cur_max, eps)

        # ---- gradient (matrix‑free via FFT) ----
        grad = _gradient(a_fft, w, n, L)

        # ---- candidate generation (exponentiated gradient) ----
