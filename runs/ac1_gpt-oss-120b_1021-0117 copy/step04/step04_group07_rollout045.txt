import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 bound for a vector already on the simplex (sum = 1)
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    """Return 2*n*max(conv) where conv = a*a (linear convolution)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())

def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """Return a non‑empty non‑negative sequence with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for cleanup
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # --------------------------------------------------------
    # Initial vector (use parent if present)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        if s <= 0.0:
            cur = np.full_like(cur, 1.0 / cur.size)
        else:
            cur = cur / s
    else:
        # fallback: exponentially decaying distribution
        n_fallback = 2500
        x = np.arange(n_fallback, dtype=np.float64)
        cur = np.exp(-x / (n_fallback / 8.0))
        cur = cur / cur.sum()

    n = cur.size
    # FFT length enough for linear convolution of length 2n-1
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------
    cur_sc = _fast_score(cur, L, n)
    best = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # optimizer hyper‑parameters
    # --------------------------------------------------------

    step0 = 0.8        # base step size
    beta = 0.9         # momentum factor
    momentum = np.zeros_like(cur)

    it = 0
    no_improve = 0

    # --------------------------------------------------------

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & max (used also for score) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()
        cur_sc = 2.0 * n * maxc

        # ----- gradient (soft‑max or sub‑gradient) -----
