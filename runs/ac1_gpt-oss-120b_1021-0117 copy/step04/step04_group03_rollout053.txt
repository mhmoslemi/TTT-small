    candidates.append(uniform)
    # arcsine distribution:

    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcsine = arcsine / arcsine.sum()
    candidates.append(arcsine)
    # random Dirichlet:
    random_dirichlet = rng.dirichlet(np.ones(n))
    candidates.append(random_dirichlet)
    # maybe extra random combos: weighted mix of first two

    best_score = float('inf')
    best_a = None

    # Evaluate each candidate with exact score using conv via FFT
    L_exact = 1 << ((2*n - 1).bit_length())
    for a in candidates:
        a_fft = np.fft.rfft(a, n=L_exact)
        conv = np.fft.irfft(a_fft * a_fft, n=L_exact)[:2*n-1]
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    # Optimization loop using Adam on softmax parametrization.
    # We'll start from best_a.

    no_improve = 0

    while time.monotonic() < deadline:

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----------------------------------------------------------------------
    # Determine length and basic initial candidates
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        base = None

    if base is None:
        # modest fallback size
        n = 1024
        base = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = base.size
        base = np.maximum(base, 0.0)

    # normalise onto simplex (sum = 1)
    base = _project_simplex(base)

    # Additional deterministic starters
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcsine = arcsine / arcsine.sum()
    random_dir = rng.dirichlet(np.ones(n))

    candidates = [base, uniform, arcsine, random_dir]

    # ----------------------------------------------------------------------
    # Evaluate candidates with the exact AC1 score (via FFT)
    # ----------------------------------------------------------------------
    L_fft = 1 << ((2 * n - 1).bit_length())   # next power of two >= 2n-1
    best_score = float("inf")
    best_a = None
    for cand in candidates:
        a_fft = np.fft.rfft(cand, n=L_fft)
        conv = np.fft.irfft(a_fft * a_fft, n=L_fft)[: 2 * n - 1]
        sc = float(2.0 * n * conv.max())
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()

    # ----------------------------------------------------------------------
    # Adam optimisation on the soft‑max (log‑it) parametrisation
    # ----------------------------------------------------------------------
    eps_soft = 1e-12
    u = np.log(best_a + eps_soft)               # logits, unconstrained
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    beta1_pow = 1.0
    beta2_pow = 1.0
    eps_adam = 1e-8
    step0 = 0.5                                 # base learning‑rate scale
    it = 0
    no_improve = 0

    # smoothing for the 'soft max' – we start relatively smooth and decay
    eps_smooth = 1.0
    eps_decay = 0.9995
    eps_min = 1e-6

    # main loop
    while time.monotonic() < deadline:
        it += 1
        # ----- obtain probability vector from logits -----
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)               # stable softmax
        a = exp_u / exp_u.sum()

        # ----- convolution and exact score -----
        a_fft = np.fft.rfft(a, n=L_fft)
        conv = np.fft.irfft(a_fft * a_fft, n=L_fft)[: 2 * n - 1]
        cur_score = float(2.0 * n * conv.max())

        # ----- keep best globally -----
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # early exit if we already beat the target
        if best_score < 1.50274349:
            break

        # ----- sub‑gradient of the max‑convolution -----
        max_val = conv.max()
        # indices attaining (near) the maximum
        argmax_idx = np.flatnonzero(conv >= max_val - 1e-14)
        g = np.zeros_like(a)                    # gradient wrt a
        for s in argmax_idx:
            # contribution of a shifted by s
            start = max(0, s - (n - 1))
            end = min(s, n - 1)
            i_vals = np.arange(start, end + 1)
            j_vals = s - i_vals
            g[i_vals] += 2.0 * a[j_vals]
        g /= max(1, len(argmax_idx))            # average if multiple maximisers

        dot_ga = np.dot(g, a)
        gu = a * (g - dot_ga)                    # ∂L/∂u

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * gu

        beta1_pow *= beta1

    # ----------------------------------------------------------------------

    return [float(x) for x in best_a.tolist()]
