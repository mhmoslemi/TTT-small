    cur_score = 2.0 * n * cur_max
    best_vec = cur.copy()
    best_score = cur_score

    while time.monotonic() < deadline:
        it += 1
        # schedule epsilon factor linearly decreasing with iteration count
        eps_rel = eps_rel_end + (eps_rel_start - eps_rel_end) * max(0.0, (deadline - time.monotonic()) / (budget_s - 0.5))

        if w_sum == 0.0:

            cand = cur * exp_factor
            cand = _project_simplex(cand)
            # evaluate

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (L2 Euclidean projection onto {x >= 0, sum(x)=1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
