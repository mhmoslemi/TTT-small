import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project the vector v onto the simplex.  Returns a new array."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Compute the evaluator's objective (identical to evaluate_sequence
# but avoids Python list conversion for speed).
# ------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2 * n * max(convolution(a,a)) / (sum(a)**2)."""
    n = a.shape[0]
    # Choose a power‑of‑two length long enough for a linear convolution
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:                     # mirrors the guard in evaluate_sequence
        return np.inf
    return float(2.0 * n * max_b / (s * s))
