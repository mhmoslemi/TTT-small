import time
import math
import numpy as np

# --------------------------------------------------------------
# Helpers
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max(a: np.ndarray, fft_len: int) -> float:
    """Maximum of the linear convolution a*a using an FFT of given length."""
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max())


# --------------------------------------------------------------
# Main search routine
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) that seeks to
    minimize the AC1 upper bound as defined by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin before the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------
    # Initial vector (use parent if present)
    # ----------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: exponentially decaying sequence of moderate length
        n_fallback = 1024
        cur = np.exp(-np.arange(n_fallback, dtype=np.float64) / (n_fallback / 10.0))

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)               # enforce sum == 1, non‑negative
    n = cur.shape[0]

    # FFT lengths (powers of two for speed)
    L_full = 1 << ((3 * n - 2).bit_length())   # for gradient convolutions
    L_conv = 1 << ((2 * n - 1).bit_length())   # for scoring (unused directly)

    # ----------------------------------------------------------
    # Book‑keeping
    # ----------------------------------------------------------
    cur_fft = np.fft.rfft(cur, n=L_full)
    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
    cur_max = cur_conv.max()
    cur_sc = 2.0 * n * cur_max
    best_vec = cur.copy()
    best_sc = cur_sc

    # ----------------------------------------------------------
    # Hyper‑parameters
    # ----------------------------------------------------------
    step0 = 1.0                # initial step magnitude
    eps_rel_start = 0.02       # initial smoothing factor (relative)
    eps_rel_min = 0.003        # lower bound for smoothing
    eps_min = 1e-9
    mu = 0.85                  # momentum coefficient
    v = np.zeros_like(cur)     # momentum buffer
    it = 0
    no_improve = 0
    target = 1.50274349

    # ----------------------------------------------------------
    # Main optimisation loop
    # ----------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- adapt smoothing (cooling schedule) ----
        eps_rel = max(eps_rel_min, eps_rel_start * (0.96 ** (it // 2500)))
        eps = max(eps_min, cur_max * eps_rel)

        # ---- smoothed‑max weights ----
        w = np.exp((cur_conv - cur_max) / eps)   # exponent <= 0, safe
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient (matrix‑free) ----
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L_full)
        grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L_full)
        idx = (2 * n - 2) - np.arange(n)          # map to per‑coordinate gradient
        g = 2.0 * grad_full[idx]

        # ---- momentum update ----
        v = mu * v + g

        # ---- line‑search step ----
        alpha = step0 / math.sqrt(it)            # decreasing step size
        accepted = False
        cand = None
        cand_sc = None
        cand_fft = None
        cand_conv = None

        for _ in range(6):                       # back‑track up to 6 times
            cand = cur - alpha * v
            cand = _project_simplex(cand)
            cand_fft = np.fft.rfft(cand, n=L_full)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L_full)[: 2 * n - 1]
            cand_sc = 2.0 * n * cand_conv.max()
            if cand_sc < cur_sc:
                accepted = True
                break
            alpha *= 0.5

        if accepted:
            cur = cand
            cur_fft = cand_fft
            cur_conv = cand_conv
            cur_max = cand_conv.max()
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            # occasional probabilistic acceptance (simulated‑annealing flavour)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_fft = cand_fft
                cur_conv = cand_conv
                cur_max = cand_conv.max()
                cur_sc = cand_sc
                no_improve = 0
            else:
                no_improve += 1

        # ---- random mass‑transfer when stuck (helps escape plateaus) ----
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L_full)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L_full)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_fft = tmp_fft
                    cur_conv = tmp_conv
                    cur_max = tmp_conv.max()
                    cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                no_improve = 0

        # ---- occasional restart with small noise after long stagnation ----
        if no_improve > 12000:
            # perturb the best known solution to explore a fresh neighbourhood
            cur = best_vec + 0.005 * rng.standard_normal(n)
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            cur_fft = np.fft.rfft(cur, n=L_full)
            cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
            cur_max = cur_conv.max()
            cur_sc = 2.0 * n * cur_max
            v.fill(0.0)
            no_improve = 0

        # ---- early exit if target achieved ----
        if best_sc < target:
            break

    # ----------------------------------------------------------
    # Return the best candidate found (plain Python list)
    # ----------------------------------------------------------
    return [float(x) for x in best_vec]
