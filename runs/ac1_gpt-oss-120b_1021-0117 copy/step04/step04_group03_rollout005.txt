def _smoothed_grad(a, L, eps):
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    max_val = conv.max()
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n-2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val, conv
```

import time, math
import numpy as np

# ------------------------------------------------------------
# Utility: projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast score for a vector already normalized to sum = 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv) assuming sum(a)=1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv))


# ------------------------------------------------------------
# Smoothed max gradient (softmax weight on autocorrelation)
# ------------------------------------------------------------
def _grad_softmax(a: np.ndarray, L: int, eps: float):
    """Return gradient of smoothed max objective, plus max value."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    max_val = conv.max()
    # softmax over conv entries
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum
    # gradient = 2 * (a * reversed(w)) convolution
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * a.size - 2) - np.arange(a.size)
    grad = 2.0 * grad_full[idx]
    return grad, max_val
