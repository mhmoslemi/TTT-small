# ------------------------------------------------------------
# Project onto probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        g = 2.0 * grad_full[idx]

        # ---- Adam update ----
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # ---- accept if improved, else Metropolis ----
        improved = False
        if cand_sc < cur_sc:
            improved = True
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (eps_temp + 1e-12))
            if rng.random() < prob:
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            no_improve += 1

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()

        # ---- periodic symmetrisation ----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ---- random mass‑transfer when stuck ----
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    # bigger shake: perturb whole vector around best
                    sigma = 0.1
                    noisy = best_vec + sigma * rng.standard_normal(n)
                    noisy = _project_simplex(noisy)
                    noisy_sc = _score(noisy)
                    if noisy_sc < cur_sc:
                        cur = noisy
                        cur_sc = noisy_sc
                        if noisy_sc < best_sc:
                            best_sc = noisy_sc
                            best_vec = noisy.copy()
                        no_improve = 0
                    else:
                        no_improve = 0

        if best_sc < target:
            break

    # -----------------------------------------------------------------
    # Ensure a feasible return (sum > 0.01 enforced by evaluator)
    # -----------------------------------------------------------------
    final = best_vec
    if final.sum() < 0.01:
        final = final / final.sum() * 0.01
    return [float(x) for x in final]
