import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 upper‑bound (sum of entries is assumed to be 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution(a, a)) (a sums to 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            no_improve += 1
            # occasional uphill move
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (lr_base + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # ----- random mass‑transfer when stuck -----
        if no_improve >= 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
            # reset Adam state to avoid stale moments
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # --------------------------------------------------------
    # Light local polishing (small random swaps)
    # --------------------------------------------------------
    for _ in range(5000):
        if time.monotonic() >= deadline:
            break
        i = rng.integers(0, n)
        if best[i] == 0.0:
            continue
        delta = best[i] * rng.random() * 0.1
        j = rng.integers(0, n)
        if j == i:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
            if best_sc < target:
                break

    # --------------------------------------------------------
    # Return the best candidate found as plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best]
