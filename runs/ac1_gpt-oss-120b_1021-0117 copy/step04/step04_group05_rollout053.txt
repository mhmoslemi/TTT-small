# Projection onto the probability simplex (sum = 1, >=0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x>=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----- determine length and a fallback base sequence -----
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # exponential fall‑back, length 1024
        n_fallback = 1024
        x = np.arange(n_fallback, dtype=np.float64)
        base = np.exp(-x / (n_fallback / 10.0))
    n = base.shape[0]

    # ----- global best holder -----
    best_vec_holder = [None]            # mutable container
    best_sc_holder = [float('inf')]

    # ----- initialise with the provided parent (if any) -----
    init_parent = _sym_normalise(base)
    _optimise(init_parent, n, deadline, best_vec_holder, best_sc_holder, rng)

    # ----- a few deterministic shapes to diversify -----
    def _sin_shape(p):
        t = np.sin(np.pi * np.arange(n) / (n - 1))
        return (t ** p).astype(np.float64)

    deterministic_inits = [
        _sin_shape(1.0),
        _sin_shape(2.0),
        _sin_shape(0.5),
        np.cos(np.linspace(0, np.pi, n)) ** 2,
        np.ones(n)                       # uniform (will be symmetrised anyway)
    ]

    for vec in deterministic_inits:
        if time.monotonic() >= deadline:
            break
        vec = _sym_normalise(vec)
        _optimise(vec, n, deadline, best_vec_holder, best_sc_holder, rng)
        if best_sc_holder[0] < 1.50274349:
            break

    # ----- random restarts until time runs out -----
    while time.monotonic() < deadline:
        # random Dirichlet‑like start, mixed with base for diversity
        rnd = rng.exponential(scale=1.0, size=n)
        mix = rng.random()
        start_vec = mix * base + (1 - mix) * rnd
        start_vec = _project_simplex(start_vec)
        start_vec = _sym_normalise(start_vec)
        _optimise(start_vec, n, deadline, best_vec_holder, best_sc_holder, rng)
        if best_sc_holder[0] < 1.50274349:
            break

    # ----- fallback if everything failed -----
    if best_vec_holder[0] is None:
        best_vec_holder[0] = _sym_normalise(base)

    # return plain Python list of floats
    return [float(x) for x in best_vec_holder[0]]
