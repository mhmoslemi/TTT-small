import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Scoring (exact, using FFT)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Exact AC1 bound for a (sum assumed = 1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())

# Convolution & smoothed gradient
# ------------------------------------------------------------
def _conv_and_grad(a: np.ndarray, eps: float, L: int):
    """Return (conv, max, softmax weights, gradient) for current a."""
    a_fft = np.fft.rfft(a, n=L)

    # autocorrelation
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    m = conv.max()

    w = np.exp((conv - m) / eps)

    if s == 0.0:

    idx = (2 * a.size - 2) - np.arange(a.size)
    g = 2.0 * grad_full[idx]
