import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast surrogate score (assumes sum(a)=1): 2 * n * max(conv(a,a))
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative list of coefficients that (approximately)
    minimises the AC1 upper bound given by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------------
    # Initialise a starting sequence.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # modest analytic start – exponential decay
        n0 = 2000
        a = np.exp(-np.arange(n0, dtype=np.float64) / 200.0)

    # make non‑negative and normalise to sum = 1 (the scale does not affect the score)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    n = a.shape[0]

    # FFT length sufficient for all convolutions we need (≥ 3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Keep the best we have seen (fallback to the initial vector)
    # --------------------------------------------------------------
    best = a.copy()
    best_sc = _fast_score(best, L)

    # current iterate
    cur = a.copy()
    cur_sc = best_sc

    # --------------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------------
    eps = 0.7                # initial smoothing for log‑sum‑exp
    eps_decay = 0.9997
    eps_min = 1e-8
    step0 = 0.4              # initial step magnitude
    it = 0
    no_improve = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current iterate with itself -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- smoothed max using log‑sum‑exp -----
        cur_eps = max(eps_min, eps * (eps_decay ** it))
        w = np.exp((conv - max_val) / cur_eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]                               # reverse for convolution
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)  # length >= 3n‑2
        idx = (2 * n - 2) - np.arange(n)               # extract the part needed for ∂/∂a
        g = 2.0 * grad_full[idx]                       # gradient vector

        # ----- descent step (normalized) -----
        grad_norm = np.linalg.norm(g) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm
        cand = cur - alpha * g
        cand = _project_simplex(cand)

        # ----- evaluate candidate (fast surrogate) -----
        cand_sc = _fast_score(cand, L)

        # ----- accept / reject -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:

                no_improve = 0
        else:

            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:

            no_improve += 1

        # ----- random mass‑transfer when stuck -----
        if no_improve > 8000:

        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------------
    # Return the best sequence found, as a plain Python list.
    # --------------------------------------------------------------
    return [float(x) for x in best]
