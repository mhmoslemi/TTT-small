    a0 = _project_simplex(a0)

    n = a0.shape[0]
    L = 1 << ((3*n - 2).bit_length())

    # best and cur
    best = a0.copy()
    best_sc = _score(best)
    cur = a0.copy()
    cur_sc = best_sc

    # Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    lr0 = 0.5  # initial learning rate
    eps_adam = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # smoothing schedule for log-sum-exp weight
    eps0 = 0.5
    eps_decay = 0.9995
    eps_min = 1e-6

    it = 0

    while time.monotonic() < deadline:
        it += 1

        # compute convolution of cur with self:
        cur_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2*n - 1]

        max_val = conv.max()
        # compute smoothed weights:
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smoothed objective: 2 * (cur * (w reversed convolution)) from earlier derivation
        # compute conv of cur with w_rev.
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)   # size L
        # grad_full index mapping: convolution length >= 3n-2
        # the gradient for each a_i is 2 * sum_{k} w_{i+k} * cur_k, which is conv of cur and w_rev at appropriate offset.
        # In original code they used idx = (2 * n - 2) - np.arange(n)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update:
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr0 / (np.sqrt(v_hat) + eps_adam) * m_hat

        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:

        else:

# ------------------------------------------------------------
# Simplex projection (L1 = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Fast score (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray, conv: np.ndarray = None) -> float:
    n = a.shape[0]
    if conv is None:
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
