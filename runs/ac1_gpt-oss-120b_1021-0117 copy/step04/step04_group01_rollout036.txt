```
def _score(a, fft_len=None):
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[:2*n-1]
    return float(2.0 * n * conv.max())
```

import time

import numpy as np

# ---------- Euclidean projection onto the probability simplex ----------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x>=0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    cond = u - css / np.arange(1, n + 1) > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

                improved = True
                break
            step *= backtrack_factor

        if not improved:
            # ----- random rescue when stuck -----
            no_improve += 1
            if no_improve > 3000:
                # perturb a fraction of mass between two random positions
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    delta = cur[i] * rng.random() * 0.3
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp, fft_len=L)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0
                    else:
                        no_improve = 0

        # ----- occasional symmetrisation (helps with even/odd structure) -----
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, fft_len=L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ----- final random fine‑tuning (use any remaining time) -----
    while time.monotonic() < deadline - 0.05:
        delta = rng.normal(scale=0.005, size=n)
        cand = best_vec + delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand, fft_len=L)
        if cand_sc < best_sc:
            best_vec, best_sc = cand, cand_sc

    # ----- return as plain Python list -----
    return [float(x) for x in best_vec]
