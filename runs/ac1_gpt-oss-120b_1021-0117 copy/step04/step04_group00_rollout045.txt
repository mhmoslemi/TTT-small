# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12                # base learning rate

    # smoothing for soft‑max
    eps0 = 1.0
    eps_decay = 0.999
    eps_min = 1e-8

    # bookkeeping
    best = cur.copy()
    # initial score (sum(cur)=1 → score = 2*n*max(conv))
    cur_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    best_sc = 2.0 * n * conv.max()
    cur_sc = best_sc

    # control variables
    it = 0
    no_improve = 0
    target = 1.50274349
    sym_interval = 250      # how often to enforce symmetry

    # ------------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- current convolution and its maximum (already have cur_fft) ----
        if it == 1:
            # cur_fft already computed for iteration 1
            pass
        else:
            cur_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val   # sum(cur)=1 by construction

        # ---- smoothing (soft‑max) ----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w_raw = np.exp((conv - max_val) / eps)
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # ---- gradient of the smoothed objective ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- candidate generation and projection ----
        cand = cur - step
        cand = _project_simplex(cand)

        # ---- evaluate candidate ----
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv_full = np.fft.irfft(cand_fft * cand_fft, n=L)
        cand_conv = cand_conv_full[: 2 * n - 1]
        cand_max = cand_conv.max()
        cand_sc = 2.0 * n * cand_max  # sum(cand)=1 after projection

        # ---- acceptance logic ----
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
        elif cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            # occasional uphill move (Metropolis style)
            prob = math.exp(-(cand_sc - cur_sc) / (lr0 + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_sc = cand_sc
            no_improve += 1

        # ---- periodic symmetrisation (helps with the Toeplitz structure) ----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv_full = np.fft.irfft(sym_fft * sym_fft, n=L)
            sym_conv = sym_conv_full[: 2 * n - 1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc
                cur = sym
                cur_sc = sym_sc
                no_improve = 0

        # ---- random mass‑transfer when stuck ----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv_full = np.fft.irfft(tmp_fft * tmp_fft, n=L)
                tmp_conv = tmp_conv_full[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < best_sc:
                    best = tmp.copy()
                    best_sc = tmp_sc
                    cur = tmp
                    cur_sc = tmp_sc
                    no_improve = 0
                else:
                    no_improve = 0

        # ---- early exit if target reached ----
        if best_sc <= target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
