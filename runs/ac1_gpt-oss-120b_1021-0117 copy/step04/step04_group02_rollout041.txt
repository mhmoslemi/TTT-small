import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.size
    # shift to make sum positive
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast AC1 score for a vector with sum ≈ 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence normalized to sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Main search routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # -----------------------------------------------------------------
    # initialise feasible point (simplex, sum = 1)
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s                         # now sum(a0)=1
    else:
        # fallback: exponential decay, length 2000
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()

    n = a0.size
    # length sufficient for a * a  (2n‑1) and a * w_rev (≤3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------
    # optimisation state
    # ------------------------------------------------------------
    max_iters = 200000                     # safety cap
    it = 0
    no_improve = 0
    eps = None
    # Adam‑style moments
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    base_lr = 0.08
    sym_interval = 300
    improve_target = 1.50274349

    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # ----- convolution of current iterate -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- smoothing weight (soft‑max) -----
        if eps is None:
            eps = max(max_val * 0.2, 1e-12)
        else:
            eps = max(eps * 0.9999, 1e-12)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # map to gradient indices
        g = 2.0 * grad_full[idx]                  # sub‑gradient

        # ----- Adam‑style adaptive step -----
        it_f = float(it)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = base_lr / (math.sqrt(it_f) + 1e-12)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # ----- accept if better, else Metropolis -----
        improved = False
        if cand_sc < cur_sc:
            improved = True
        else:

            if rng.random() < prob:
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:

        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- occasional restart from a fresh random point -----
        if no_improve > 8000 and (time.monotonic() - start) > 0.4 * (budget_s - 0.5):
            cur = rng.random(n)
            cur = _project_simplex(cur)
            cur_sc = _score(cur)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            no_improve = 0

        # ----- early exit if target already met -----
        if best_sc < improve_target:
            break

    # return as list of plain Python floats
    return [float(x) for x in best_vec]
