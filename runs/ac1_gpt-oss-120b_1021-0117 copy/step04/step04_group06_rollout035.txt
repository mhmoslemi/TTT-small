import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Convolution max (full linear convolution) via FFT
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.shape[0] - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Soft‑max weights and gradient of the smoothed max
# ------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, eps: float, L: int):
    """Return (gradient, current max of convolution)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)

    # full convolution
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()

    # soft‑max weights
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum

    # gradient of weighted sum  (2 * a * w_rev convolution)
    w_rev = w[::-1]
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g, max_val


# ------------------------------------------------------------
# Scoring function for sum‑1 vectors (exact evaluator up‑to a constant)
# ------------------------------------------------------------
def _score_of(a: np.ndarray, L: int) -> float:
    """2·n·max(conv) where sum(a)=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative coefficient list with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # -------------------- initial vector --------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 0.0:
            a0 = a0 / s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        n_fallback = 2000
        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 20.0)
        a0 = a0 / a0.sum()

    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two FFT length

    # -------------------- Adam hyper‑parameters --------------------
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.5               # base learning rate
    eps0 = 0.5              # initial soft‑max temperature
    eps_decay = 0.999
    min_eps = 1e-8

    # -------------------- state --------------------
    a = a0.copy()
    m = np.zeros_like(a)    # first moment
    v = np.zeros_like(a)    # second moment
    best = a.copy()
    best_sc = _score_of(best, L)
    no_improve = 0
    it = 0
    beta1_pow = beta1
    beta2_pow = beta2

    # -------------------- main optimisation loop --------------------
    while time.monotonic() < deadline:
        it += 1

        # temperature for the soft‑max
        eps = eps0 * (eps_decay ** it)
        if eps < min_eps:
            eps = min_eps

        # gradient of smoothed max
        g, cur_max = _gradient_softmax(a, eps, L)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # step size (decaying with sqrt(it))
        lr = lr0 / math.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = a - step
        cand = _project_simplex(cand)

        sc = _score_of(cand, L)
        if sc + 1e-12 < best_sc:
            best_sc = sc
            best = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = cand

        # ----- occasional random jitter -----
        if it % 1000 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = a[i] * rng.random() * 0.1
            tmp = a.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_of(tmp, L)
            if tmp_sc < best_sc:
                best_sc = tmp_sc
                best = tmp.copy()
                a = tmp
                no_improve = 0

        # ----- escape plateaus via larger mass transfer -----
        if no_improve > 4000:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_of(tmp, L)
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best = tmp.copy()
                a = tmp
                no_improve = 0

        # update bias‑correction powers
        beta1_pow *= beta1
        beta2_pow *= beta2

        # early exit if we already beat the target
        if best_sc < 1.50274349:
            break

    # -------------------- final local refinement --------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = best[i] * rng.random() * 0.02
        if delta <= 0.0:
            continue
        tmp = best.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score_of(tmp, L)
        if tmp_sc < best_sc:
            best_sc = tmp_sc
            best = tmp
            continue
        # optional symmetry check (helps in many cases)
        sym = (tmp + tmp[::-1]) * 0.5
        sym = _project_simplex(sym)
        sym_sc = _score_of(sym, L)
        if sym_sc < best_sc:
            best_sc = sym_sc
            best = sym

    # return plain Python list
    return [float(x) for x in best]
