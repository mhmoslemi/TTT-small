import time
import math
import numpy as np

# ---------- Simplex projection ----------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ---------- Score (full objective) ----------
def _score(a: np.ndarray, fft_len: int = None) -> float:
    """Return 2*n*max(conv) / (sum_a**2) for a simplex‑normalised vector."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    max_b = float(conv.max())
    sum_a = float(a.sum())
    return 2.0 * n * max_b / (sum_a * sum_a)


# ---------- Main optimisation ----------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ----- initialise -----
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: exponential decay
        n_fallback = 2000
        a0 = np.exp(-np.linspace(0, 5, n_fallback, dtype=np.float64))
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                     # simplex normalisation

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length for full convolution

    # ---- bookkeeping ----
    best_vec = a0.copy()
    best_sc = _score(best_vec, fft_len=L)

    cur = a0.copy()
    cur_sc = best_sc

    # ---- hyper‑parameters ----
    eps0 = 0.5                # smoothing start
    eps_decay = 0.9998        # slower decay than reference
    eps_min = 1e-12

    step0 = 1.0               # larger base step
    max_backtrack = 12
    backtrack_factor = 0.5

    beta_mom = 0.9            # momentum coefficient
    mom = np.zeros_like(cur)

    no_improve = 0
    it = 0
    restart_interval = 8000   # force occasional re‑initialisation
    sym_interval = 3000

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current iterate -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weights (smooth approximation of max) -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- gradient of smooth max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        mom = beta_mom * mom + (1.0 - beta_mom) * g

        # ----- line‑search on step size -----
        step_base = step0 / math.sqrt(it)
        step = step_base
        improved = False
        for _ in range(max_backtrack):

            cand = _project_simplex(cand)
            cand_sc = _score(cand, fft_len=L)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best_vec, best_sc = cand.copy(), cand_sc
                    no_improve = 0
                improved = True
                break
            step *= backtrack_factor

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        # ----- occasional symmetrisation (helps even‑odd structure) -----
        if it % sym_interval == 0:

            sym = _project_simplex(sym)
            sym_sc = _score(sym, fft_len=L)
            if sym_sc < cur_sc:

            cur = rng.dirichlet(np.full(n, alpha)).astype(np.float64)

    while time.monotonic() < deadline - 0.05:

        cand = best_vec + delta

        cand = cand / cand.sum()
        cand = _project_simplex(cand)
        cand_sc = _score(cand, fft_len=L)
        if cand_sc < best_sc:
            best_vec, best_sc = cand, cand_sc

    # ----- return result as plain Python list -----
    return [float(x) for x in best_vec]
