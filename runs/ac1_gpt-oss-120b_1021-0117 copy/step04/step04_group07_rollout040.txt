import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean), top‑level helper
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound for sum‑1 vectors (uses FFT)
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Full convolution (linear) and its FFT, top‑level helper
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int):
    """Return full linear convolution (as np.ndarray) and a's FFT."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)
    return conv[: 2 * a.size - 1], a_fft

# ------------------------------------------------------------

              n: int, L: int, eps: float, eps_min: float):

    eps_min = 1e-9
    eps_decay = 0.997

    # learning‑rate schedule
    base_lr = 0.15
    max_step_norm = 0.2

    it = 0
    no_improve = 0
    restart_patience = 3000
    target_score = 1.50274349

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution of current vector
        conv, a_fft = _conv_fft(cur, L)

        # gradient of smoothed max (or sub‑gradient)
        g, maxc = _gradient(cur, a_fft, conv, n, L, eps, eps_min)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step direction
        step = base_lr * (m_hat / (np.sqrt(v_hat) + eps_adam))

        # optional norm clipping (helps stability)
        step_norm = np.linalg.norm(step)
        if step_norm > max_step_norm:
            step = step * (max_step_norm / (step_norm + 1e-12))

        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _fast_score(cand, L, n)

        # accept if improvement; occasional uphill move (simulated annealing)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # temperature decays like 1/sqrt(it)
            T = max(1e-5, 0.2 / math.sqrt(it))
            if rng.random() < math.exp(-(cand_sc - cur_sc) / T):
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # update global best
        if cand_sc < best_sc:
            best, best_sc = cand.copy(), cand_sc
            no_improve = 0

        # decay smoothing parameter
        eps = max(eps_min, eps * eps_decay)

        # restart if stuck for a while
        if no_improve >= restart_patience:
            rand = rng.exponential(scale=1.0, size=n)
            rand = rand / rand.sum()
            cur = _project_simplex(rand)
            cur_sc = _fast_score(cur, L, n)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # early exit if target reached
        if best_sc < target_score:
            break

    # --------------------------------------------------------
    # Return best candidate as plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best]
