import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Score = 2 * n * max(convolution)  (assumes sum == 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Simple greedy Sidon set generator (unique pairwise sums)
# ------------------------------------------------------------
def _sidon_set(n: int, target: int = None) -> list:
    """Return a list of indices with (i+j) distinct for i<=j."""
    if target is None:
        target = int(math.sqrt(2 * n))  # theoretical upper bound
    sel = []
    sums = set()
    for i in range(n):
        # check collisions with already selected indices
        coll = (i + i) in sums
        if not coll:
            for s in sel:
                if (i + s) in sums:
                    coll = True
                    break
        if not coll:
            sel.append(i)
            sums.add(i + i)
            for s in sel[:-1]:
                sums.add(i + s)
        if len(sel) >= target:
            break
    return sel
