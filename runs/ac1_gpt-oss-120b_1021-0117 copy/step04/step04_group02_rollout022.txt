    start = time.monotonic()
    deadline = start + max(0.0,budget_s - 0.5)

    m = np.zeros_like(cur) # momentum

    max_trials = 5
    it = 0
    no_improve = 0

        it += 1

        # compute softmax gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2*n-1]
        max_val = conv_full.max()

        else:

        m = beta * m + (1.0 - beta) * g

        norm_m = np.linalg.norm(m) + 1e-12

        improved = False
        cand = None

        if improved:

            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:

                    best_vec = sym.copy()

                else:
                    no_improve = 0

        if best_sc < improve_target:
            break

import time

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a)≈1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
