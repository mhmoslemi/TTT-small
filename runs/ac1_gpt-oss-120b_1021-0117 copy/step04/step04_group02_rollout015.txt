import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Compute AC1 objective (2*n*max convolution) and its gradient.
# ------------------------------------------------------------
def _obj_and_grad(a: np.ndarray, L: int, n: int):
    """Return (objective, gradient) for a vector a (sum(a)=1)."""
    # convolution via FFT
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv_full.max()
    # indices attaining the maximum (within tiny tolerance)
    eps = 1e-12
    max_idx = np.where(conv_full >= max_b - eps)[0]

    # gradient of the max‑convolution w.r.t a
    g = np.zeros_like(a)
    cnt = len(max_idx)
    idx_arr = np.arange(n)
    for k in max_idx:
        shift = k - idx_arr
        mask = (shift >= 0) & (shift < n)
        g[mask] += 2.0 * a[shift[mask]]
    if cnt > 1:
        g /= cnt

    # objective = 2*n*max_b  (sum(a)=1)
    obj = 2.0 * n * max_b
    # gradient of objective = 2*n * grad(max_b)
    grad = (2.0 * n) * g
    return obj, grad
