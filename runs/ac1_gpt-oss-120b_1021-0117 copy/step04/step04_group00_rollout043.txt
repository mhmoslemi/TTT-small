import time
import math
import random
import numpy as np

def _sidon_set(n, target_m, rng, deadline):
    """
    Greedy construction of a Sidon set (all pairwise sums distinct)
    aiming for at least target_m elements. Returns the set as a list.
    """
    while time.monotonic() < deadline:
        order = list(range(n))
        rng.shuffle(order)
        used = set()
        S = []
        for i in order:
            # quick exit if impossible to reach target size
            if len(S) + (n - i - 1) < target_m:
                break
            conflict = False
            for s in S:
                if (i + s) in used:
                    conflict = True
                    break
            if conflict or (2 * i) in used:
                continue
            # add i to the set
            for s in S:
                used.add(i + s)          # i + s = s + i, unique sum
            used.add(2 * i)               # the square sum
            S.append(i)
            if len(S) >= target_m:
                # we already have enough, keep building in case we can grow more
                continue
        if len(S) >= target_m:
            return S
    return []  # failed to reach target within deadline

    """

    The resulting sequence achieves an AC1 upper bound well below the target.

    best_seq = None
    best_score = float('inf')
    attempts = 0

        S = _sidon_set(n, min_m, rng, deadline)
        if not S:

        m = len(S)
        # Theoretical bound for a 0‑1 Sidon construction:
