import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – keeps sum==1, x>=0
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Core evaluator (max convolution) – assumes sum==1
# ------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max of linear convolution a*a using FFT length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.shape[0] - 1]
    return float(conv.max())
