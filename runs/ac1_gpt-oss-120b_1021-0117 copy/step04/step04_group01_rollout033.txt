import time

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a: np.ndarray, fft_len: int = None) -> float:

    n = a.shape[0]
    if fft_len is None:

    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
