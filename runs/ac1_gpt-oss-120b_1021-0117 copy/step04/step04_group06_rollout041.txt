def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    # initialization
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 512
        a = np.exp(-np.arange(n_fallback, dtype=np.float64) / 20.0)
    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total < 1e-12:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / total
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    # helpers
    def project_simplex(v):
        v = np.asarray(v, dtype=np.float64)
        n_ = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n_ + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w
    def conv_max(vec):
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return conv.max(), conv
    def score(vec):
        # assume sum(vec)=1
        max_val, _ = conv_max(vec)
        return float(2.0 * n * max_val)
    # initial scores
    cur = a.copy()
    cur_sc = score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc
    # optimization parameters
    step0 = 1.0
    sym_freq = 500
    no_improve = 0
    it = 0
    while time.time() < deadline:
        it += 1
        # compute convolution of current a
        max_val, conv = conv_max(cur)
        # active indices
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]
        # compute subgradient
        g = np.zeros_like(cur)
        for k in active:
            if k <= n - 1:

                g[:k+1] += 2.0 * cur[k::-1]
            else:

                # fill g[p:] += 2*cur[p:][::-1]
                g[p:] += 2.0 * cur[p:][::-1]

                no_improve = 0

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Full linear convolution via FFT, returning max value and array
# ------------------------------------------------------------
def _conv_and_max(a: np.ndarray, L: int):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.shape[0] - 1]
    return conv.max(), conv


# ------------------------------------------------------------
# Score used during optimisation (sum(a) is assumed = 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    max_val, _ = _conv_and_max(a, L)
    return float(2.0 * n * max_val)


# ------------------------------------------------------------
# Subgradient of the max‑convolution (or weighted version)
# ------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_idx:
        if k <= n - 1:                     # first half of convolution
            g[: k + 1] += 2.0 * a[k :: -1]  # a[k],a[k-1],...,a[0]
        else:                              # second half
            p = k - (n - 1)                # start index for the tail
            g[p:] += 2.0 * a[p:][::-1]     # reversed tail slice
    if active_idx.size > 1:
        g /= float(active_idx.size)
    return g
