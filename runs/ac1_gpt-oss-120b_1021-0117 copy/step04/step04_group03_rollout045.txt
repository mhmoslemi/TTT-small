import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x>=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a vector (assumes sum=1). Exact evaluator uses the same
# formula, scaling cancels, so this is a cheap proxy.
# ------------------------------------------------------------
def _score(a: np.ndarray, L_fft: int) -> float:
    """Return 2*n*max(conv) for a (sum assumed 1)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(af * af, n=L_fft)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total_budget = float(budget_s)
    # leave a small safety margin for the harness
    deadline = start + max(0.0, total_budget - 0.5)

    # --------------------------------------------------------
    # Initial vector(s)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base = np.maximum(base, 0.0)
    else:
        base = None

    if base is None:
        # modest default length (power of two for FFT convenience)
        n = 2048
        base = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = base.size

    # Normalise to simplex
    base = _project_simplex(base)

    # A collection of cheap deterministic starters
    i = np.arange(n, dtype=np.float64)

    # uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # arcsine distribution (continuous arcsine law)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcsine = _project_simplex(arcsine)

    # sin^2 shape (windowed sine)
    sin_sq = np.sin(np.pi * (i + 0.5) / n) ** 2
    sin_sq = _project_simplex(sin_sq)

    # half‑cosine (raised cosine)
    half_cos = (1.0 - np.cos(2.0 * np.pi * (i + 0.5) / n)) / 2.0
    half_cos = _project_simplex(half_cos)

    # Mix base with arcsine to diversify
    mix = _project_simplex(0.5 * base + 0.5 * arcsine)

    candidates = [base, uniform, arcsine, sin_sq, half_cos, mix]

    # --------------------------------------------------------
    # Choose the best cheap starter
    # --------------------------------------------------------
    L_fft = 1 << ((2 * n - 1).bit_length())
    best_vec = None
    best_sc = float("inf")
    for cand in candidates:
        sc = _score(cand, L_fft)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # --------------------------------------------------------
    # Smoothed‑max gradient descent (with momentum & occasional
    # random perturbations). The loop respects the deadline.
    # --------------------------------------------------------
    eps = 5.0               # start temperature – smooth max heavily
    eps_decay = 0.9995

                    else:
                        no_improve = 0

        # Early termination if target achieved
        if best_sc < 1.50274349:
            break

    # --------------------------------------------------------
    # Return the best vector found as a plain Python list
    # --------------------------------------------------------
    if best_vec is None:
        best_vec = cur
    return [float(x) for x in best_vec]
