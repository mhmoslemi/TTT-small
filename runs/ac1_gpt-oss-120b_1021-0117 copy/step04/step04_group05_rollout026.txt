import time, math, numpy as np

def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_full_fft(vec, L_full, n):
    fft = np.fft.rfft(vec, n=L_full)
    conv = np.fft.irfft(fft * fft, n=L_full)[:2*n-1]
    return conv, fft

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: project onto probability simplex (sum = 1, ≥0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Convolution (full linear) via FFT, also returns FFT of input
# ------------------------------------------------------------
def _conv_full_fft(vec: np.ndarray, L_full: int, n: int):
    """Return (conv, fft(vec)) where conv is the linear convolution vec*vec."""
    fft = np.fft.rfft(vec, n=L_full)
    conv = np.fft.irfft(fft * fft, n=L_full)[: 2 * n - 1]
    return conv, fft


# ------------------------------------------------------------
# Compute AC1 score from max convolution value
# ------------------------------------------------------------
def _score_from_max(m: float, n: int) -> float:
    return 2.0 * n * m
