def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = float(budget_s)
    deadline = start + max(0.0, total - 0.5)

    # Load base if present, else fallback uniform
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base = np.maximum(base, 0.0)
    else:
        # use length maybe 2000, uniform
        n = 2048
        base = np.full(n, 1.0/n, dtype=np.float64)

    # project onto simplex
    base = _project_simplex(base)
    n = base.shape[0]

    # Additional candidate options
    uniform = np.full(n, 1.0/n)
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5)*(n - 0.5 - i))
    arcsine = arcsine / arcsine.sum()

    candidates = [base, uniform, arcsine]
    best_vec = None
    best_score = float('inf')
    for cand in candidates:
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
    cur = best_vec.copy()
    cur_score = best_score

    # Precompute FFT size
    L = 1 << ((2 * n - 1).bit_length())

    # Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t = 0

    eps = 5.0
    eps_decay = 0.9995
    eps_min = 1e-9
    step0 = 0.2

    no_improve = 0
    max_iter = 50000

    while time.monotonic() < deadline and t < max_iter:
        t += 1

        # compute convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # smoothed weights
        eps = max(eps_min, eps * eps_decay)
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_score = _score(cand)

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # sometimes accept if less worse? but we can just reject
            no_improve += 1

        # random perturb if stuck
        if no_improve > 2000:
            # generate small random perturbation preserving sum
            i_idx = rng.integers(0, n)
            if cur[i_idx] > 0:
                delta = cur[i_idx] * rng.random() * 0.2
                j_idx = rng.integers(0, n)
                if j_idx == i_idx:
                    j_idx = (j_idx + 1) % n
                tmp = cur.copy()
                tmp[i_idx] -= delta
                tmp[j_idx] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score(tmp)
                if tmp_score < cur_score:
                    cur = tmp
                    cur_score = tmp_score
                    if tmp_score < best_score:
                        best_score = tmp_score
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # symmetrize occasionally
        if (t % 1000) == 0:
            sym = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(sym)
            cur_score = _score(cur)
            if cur_score < best_score:
                best_score = cur_score
                best_vec = cur.copy()

        # early exit target
        if best_score < 1.50274349:
            break

    if best_vec is None:
        best_vec = cur

    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Score for a vector known to sum to 1 (exact evaluator)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution(a, a)).  Assumes sum(a)=1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total_budget = float(budget_s)
    # keep a small safety margin for the harness
    deadline = start + max(0.0, total_budget - 0.5)

    # --------------------------------------------------------
    # Load a base vector (parent construction if present)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():

    # --------------------------------------------------------

    # --------------------------------------------------------
    base = _project_simplex(base)
    n = base.size

    # --------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)

    arcsine = arcsine / arcsine.sum()

    candidates = [base, uniform, arcsine]
    best_vec = None
    best_sc = float("inf")
    for cand in candidates:
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    # --------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())          # FFT length (power of two)
    beta1, beta2 = 0.9, 0.999

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1_pow, beta2_pow = 1.0, 1.0

    eps_min = 1e-9
    step0 = 0.2                                    # base step scale

    # --------------------------------------------------------

    # --------------------------------------------------------

    if best_vec is None:

    return [float(x) for x in best_vec]
