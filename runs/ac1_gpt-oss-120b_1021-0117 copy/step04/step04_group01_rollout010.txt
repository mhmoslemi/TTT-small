def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # Determine n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    # Helper: projection
    def _project_simplex(v):
        n0 = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n0+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho+1)
        return np.maximum(v - theta, 0.0)

    # Helper: scoring
    def _score_vec(vec):
        # sum already positive check later.
        s = np.sum(vec)
        if s < 0.01:
            return np.inf
        # compute convolution via FFT
        L = 1 << ((2*n - 1).bit_length())
        fft = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(fft * fft, n=L)[:2*n-1]
        maxc = conv.max()
        return 2.0 * n * float(maxc) / (s*s)

    # Initial candidates via param shapes
    best_vec = None
    best_sc = np.inf
    # param sets
    # constant
    cand = np.ones(n, dtype=np.float64)
    cand = cand / cand.sum()
    sc = _score_vec(cand)
    if sc < best_sc:
        best_sc = sc
        best_vec = cand.copy()
    # linear ramp up then down symmetric (triangular)
    # Use symmetrical triangle: values increase to center then decrease
    idx = np.arange(n)
    center = (n-1)/2.0
    tri = 1.0 - np.abs(idx - center)/center  # linearly decreasing from 1 at center to 0 at ends
    # shift to nonnegative
    tri = np.maximum(tri, 0.0)
    tri = tri / tri.sum()
    sc = _score_vec(tri)
    if sc < best_sc:
        best_sc = sc
        best_vec = tri.copy()
    # sinusoidal shape sin^p
    x = np.arange(n) + 0.5
    sin_vals = np.sin(np.pi * x / n)  # zero at ends
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        cand = np.power(sin_vals, p)
        cand = cand / cand.sum()
        sc = _score_vec(cand)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()
    # Gaussian shapes with sigma fractions
    for sigma_frac in [0.25, 0.35, 0.45]:
        sigma = sigma_frac * n
        cand = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
        cand = cand / cand.sum()
        sc = _score_vec(cand)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score_vec(vec: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 bound for a non‑negative vector (no scaling)."""
    s = vec.sum()
    if s < 0.01:
        return np.inf
    fft = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
    return 2.0 * n * float(conv.max()) / (s * s)
