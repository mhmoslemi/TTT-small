import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # -----------------------------------------------------------------
    # Initial vectors (parent, uniform, arcsine)
    # -----------------------------------------------------------------
    candidates = []
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base = np.maximum(base, 0.0)
        base = _project_simplex(base)
        candidates.append(base)
        n = base.size
    else:
        n = 1024                     # fallback length

    # uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    candidates.append(uniform)

    # arcsine shape
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcsine = arcsine / arcsine.sum()
    candidates.append(arcsine)

    # pick best of the cheap starters
    L = 1 << ((2 * n - 1).bit_length())          # FFT length (power of two)
    best_vec = None
    best_score = float("inf")
    for cand in candidates:
        a_fft = np.fft.rfft(cand, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        sc = 2.0 * n * conv.max()               # sum(cand)=1 after projection
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    cur = best_vec.copy()
    cur_score = best_score

    # -----------------------------------------------------------------
    # Gradient descent on smoothed max (additive step, simplex proj)
    # -----------------------------------------------------------------
    step0 = 1.0                     # larger than the previous 0.5
    eps = 0.5                       # start temperature for soft‑max
    eps_decay = 0.9999
    eps_min = 1e-8

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and soft‑max weights -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        eps = max(eps_min, eps * eps_decay)

        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- add‑itive update + projection -----
        grad_norm = np.linalg.norm(g) + 1e-12
        alpha = step0 / (math.sqrt(it) * grad_norm)
        cand = cur - alpha * g
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_score = 2.0 * n * cand_conv.max()

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
                if best_score < 1.50274349:
                    break
        else:
            no_improve += 1

        # ----- occasional symmetrisation -----
        if (it % 500) == 0:
            sym = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(sym)
            # recompute its score (cheap, no extra FFT)
            cur_fft = np.fft.rfft(cur, n=L)
            cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
            cur_score = 2.0 * n * cur_conv.max()
            # reset momentum‑free (helps after projection)
            # (nothing else to reset)

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000:
            # take some mass from a random positive entry and give it to another
            pos = np.where(cur > 1e-12)[0]

                delta = cur[i_idx] * rng.random() * 0.3
                j_idx = rng.integers(0, n)
                if j_idx == i_idx:

    if best_vec is None:
        best_vec = cur
    return [float(x) for x in best_vec]
