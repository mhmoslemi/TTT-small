def gradient_max_conv(a):
    n = a.shape[0]
    L = next power 2 >= 2*n-1 
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    k = conv.argmax()
    max_val = conv[k]
    # compute gradient g: for each i, g_i = 2 * a[k - i] if 0 <= k - i < n else 0
    # Equivalent: g = 0 array length n; start = max(0, k-(n-1)); end = min(n-1, k)

    i_range = np.arange(lo, hi+1)
    g = np.zeros_like(a)
    g[i_range] = 2 * a[k - i_range]
    return max_val, conv, g

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Compute convolution via FFT and gradient of the maximal entry
# ------------------------------------------------------------
def _conv_and_grad(a: np.ndarray, L: int, n: int):
    """Return (max_val, conv, grad) for a*a."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    k = int(conv.argmax())
    max_val = float(conv[k])
    # gradient of conv[k] w.r.t a: g[i] = 2 * a[k-i] if 0<=k-i<n else 0
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return max_val, conv, grad


# ------------------------------------------------------------
# Fast score (sum(a)=1 assumed)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """2 * n * max(conv) for sum(a)=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative coefficient list with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # -------------------- initialise --------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: exponential decay, length 1024
        n_fallback = 1024

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)                       # enforce sum = 1

    n = a.shape[0]
    # FFT length (next power of two >= 2n-1)
    L = 1 << ((2 * n - 1).bit_length())

    # -------------------- bookkeeping --------------------
    best_vec = a.copy()
    best_sc = _score(best_vec, L)
    cur = a.copy()
    cur_sc = best_sc

    # hyper‑parameters
    step0 = 0.9                # base step magnitude
    eps_decay = 0.9995         # smoothing decay (not used for exact gradient)
    no_improve = 0
    it = 0
    momentum = np.zeros_like(a)  # simple momentum term
    beta = 0.9                  # momentum coefficient

    # -------------------- optimisation loop --------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- gradient of the current maximal convolution entry -----
        max_val, conv, grad = _conv_and_grad(cur, L, n)

        # ----- step with momentum and projection -----
        momentum = beta * momentum + (1.0 - beta) * grad
        grad_norm = np.linalg.norm(momentum) + 1e-12
