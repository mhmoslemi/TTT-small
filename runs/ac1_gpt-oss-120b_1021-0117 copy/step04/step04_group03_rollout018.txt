def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., budget_s - 0.5)   # keep margin
    # Determine n and initial vectors
    if 'height_sequence_1' in globals():
        base_arr = np.asarray(height_sequence_1, dtype=np.float64)
        base_arr = np.maximum(base_arr, 0.0)
        base_arr = _project_simplex(base_arr)  # sum=1
        n = base_arr.size
    else:
        n = 1024
        base_arr = np.full(n, 1.0/n)
    # Build candidate list
    candidates = [base_arr]
    # uniform
    candidates.append(np.full(n, 1.0/n))
    # arcsine
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcs = arcs / arcs.sum()
    candidates.append(arcs)
    # random Dirichlet (exponential)
    rand = rng.exponential(scale=1.0, size=n)
    rand = rand / rand.sum()
    candidates.append(rand)
    # choose best
    best_vec = None
    best_score = np.inf
    for cand in candidates:
        # compute score via FFT
        L = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(cand, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        sc = float(2.0 * n * np.max(conv))
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
    cur = best_vec.copy()
    cur_score = best_score
    # descent parameters
    L = 1 << ((2 * n - 1).bit_length())
    # alternatively use bigger L maybe 2^nextpow2(3n) for conv gradient? previous used (3n -2).bit_length() to prevent aliasing? Actually conv needed length up to 2n-1. For gradient we compute w_fft of length L, okay.
    # We'll set L to next power of two >= 2*n.
    L = 1 << ((2 * n - 1).bit_length())
    # initialization for eps, momentum etc.
    eps = max(np.max(np.fft.irfft(np.fft.rfft(cur, n=L) * np.fft.rfft(cur, n=L), n=L)[:2*n-1]) * 0.1, 1e-4)
    eps_min = 1e-8
    eps_decay = 0.9995
    step0 = 2.0
    beta = 0.9
    v = np.zeros_like(cur)
    it = 0
    no_improve = 0
    # main loop
    while time.monotonic() < deadline:
        it += 1
        # compute FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = np.max(conv)
        # compute smoothing weights
        eps = max(eps_min, eps * eps_decay)
        w = np.exp((conv - max_val) / eps)
        w_sum = np.sum(w)
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # momentum update
        v = beta * v + (1 - beta) * g
        # step size
        grad_norm = np.linalg.norm(v) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm
        cand = cur - alpha * v

        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        new_score = float(2.0 * n * np.max(conv_new))
        if new_score < cur_score - 1e-12:
            cur, cur_score = cand, new_score
            if new_score < best_score:
                best_score, best_vec = new_score, cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

            delta = rng.random(len(idxs)) * 0.1
            cand2 = cur.copy()
            # subtract from random indices, add to others
            for di, idx_i in enumerate(idxs):
                amount = delta[di] * cand2[idx_i]
                # pick target index
                tgt = rng.integers(0, n)
                if tgt == idx_i:
                    tgt = (tgt + 1) % n
                cand2[idx_i] = max(0.0, cand2[idx_i] - amount)
                cand2[tgt] += amount
            cand2 = _project_simplex(cand2)
            # compute its score
            a_fft2 = np.fft.rfft(cand2, n=L)
            conv2 = np.fft.irfft(a_fft2 * a_fft2, n=L)[:2*n-1]
            sc2 = float(2.0 * n * np.max(conv2))
            if sc2 < cur_score:

                if sc2 < best_score:
                    best_score, best_vec = sc2, cand2.copy()
            no_improve = 0
        # early exit if target reached
        if best_score < 1.50274349:
            break
    # Return best_vec as Python list
    if best_vec is None:

    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    rand_dir = rng.exponential(scale=1.0, size=n)

    candidates = [base, uniform, arcsine, rand_dir]

    # ------------------- pick best starter --------------------

    for cand in candidates:
