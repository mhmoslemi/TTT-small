def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # leave half sec buffer

    # initial vector
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # generate uniform distribution length 2000 for fallback
        n_fallback = 2000
        a0 = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # enforce nonneg & sum=1
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s < 1e-12:
        a0[:] = 1.0 / a0.size
    else:
        a0 /= s

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())   # power of two >= 3n-2

    cur = a0.copy()
    # compute initial score
    cur_fft = np.fft.rfft(cur, n=L)
    conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
    cur_sc = float(2.0 * n * conv.max())
    best = cur.copy()
    best_sc = cur_sc

    # Adam params
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.05  # base learning rate (will be scaled)
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t = 0

    # smoothing for softmax
    eps = 0.001
    eps_min = 1e-10
    eps_decay = 0.9995

    # other hyperparams
    no_improve = 0

    # main loop
    while time.monotonic() < deadline:
        t += 1
        # compute fft and conv
        cur_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
        maxc = conv_full.max()
        # compute gradient
        if eps > eps_min:
            w = np.exp((conv_full - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            # compute grad via convolution with reversed w
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(cur_fft * w_fft, n=L)   # length L
            # slice indices (2*n-2) - np.arange(n)
            idx = (2 * n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]
            # reduce eps
            eps = max(eps_min, eps * eps_decay)
        else:
            # pure subgradient at max index(es)
            # choose all indices where conv==maxc
            k = np.flatnonzero(conv_full == maxc)
            g = np.zeros_like(cur)
            for kk in k:
                start = max(0, kk - (n - 1))
                end = min(kk, n - 1)  # ensure i within
                # Actually, we need indices i such that 0 <= i < n and 0 <= kk - i < n
                i_start = max(0, kk - (n - 1))
                i_end = min(kk, n - 1)
                # i runs from i_start to i_end inclusive
                idx_range = np.arange(i_start, i_end + 1, dtype=int)
                # contributions: a[kk - idx] for each idx
                g[idx_range] += 2.0 * cur[kk - idx_range]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        # projection onto simplex
        cand = _project_simplex(cand)

        # compute candidate score fast
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_sc = float(2.0 * n * cand_conv.max())

        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            # update best
            if cand_sc < best_sc:
                best = cand.copy()
                best_sc = cand_sc
                no_improve = 0
        else:
            # occasional acceptance with probability based on improvement
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (lr + 1e-12)):
                cur = cand
                cur_sc = cand_sc
            no_improve += 1

        # symmetrisation occasionally
        if t % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[:2*n-1]
            sym_sc = float(2.0 * n * sym_conv.max())
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best = sym.copy()
                    best_sc = sym_sc
                    no_improve = 0

        # random perturb when stuck
        if no_improve > 4000:
            i = rng.integers(n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(n)
                if j == i:
                    j = (j + 1) % n
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                # compute score
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[:2*n-1]
                tmp_sc = float(2.0 * n * tmp_conv.max())
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best = tmp.copy()
                    best_sc = tmp_sc
                no_improve = 0

        # early exit if target reached
        if best_sc < 1.50274349:
            break

    # convert best to python list of floats
    return [float(x) for x in best]

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of heights) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: uniform distribution of modest length
        n_fallback = 2000
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0 / a.size
    else:
        a /= s                                   # normalize to sum‑one simplex

    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())          # FFT length ≥ 3n‑2

    cur = a.copy()
    cur_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
    cur_sc = float(2.0 * n * conv_full.max())

    best = cur.copy()
    best_sc = cur_sc

    # ----- Adam optimizer state -------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.5                                 # decays as 1/√t
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1_pow, beta2_pow = 1.0, 1.0

    # ----- soft‑max smoothing for max ------------------------------------
    eps = 0.001
    eps_min = 1e-12
    eps_decay = 0.9995

    no_improve = 0
    it = 0

    # ----- main optimisation loop -----------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution of current vector with itself
        cur_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
        maxc = conv_full.max()

        # gradient of the (smooth) max‑convolution
        if eps > eps_min:
            w = np.exp((conv_full - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)          # slice for the proper part
            g = 2.0 * grad_full[idx]
            eps = max(eps_min, eps * eps_decay)
        else:
            # true sub‑gradient using all active max positions
            ks = np.flatnonzero(conv_full == maxc)
            g = np.zeros_like(cur)
            for k in ks:
                i_start = max(0, k - (n - 1))
                i_end = min(k, n - 1)
                idx_range = np.arange(i_start, i_end + 1, dtype=int)
                g[idx_range] += 2.0 * cur[k - idx_range]
            g = g / ks.size

        # ----- Adam update (with projection onto the simplex) ---------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        effective_lr = base_lr / math.sqrt(it)

        # ----- symmetrisation (helps because the problem is nearly symmetric)

            if mix_sc < cur_sc:
                cur, cur_sc = mix.copy(), mix_sc

        # ----- random mass‑transfer guided by gradient when stuck ----------
        if no_improve > 3000:
            grad_pos = np.where(g > 0)[0]
            grad_neg = np.where(g < 0)[0]
            if grad_pos.size and grad_neg.size:
                i = rng.choice(grad_pos)
                j = rng.choice(grad_neg)
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)

                no_improve = 0

        # ----- early termination if target reached -------------------------
        if best_sc < 1.50274349:
            break

    # return the best candidate found as plain Python floats
    return [float(x) for x in best]
