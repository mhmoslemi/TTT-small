def compute_grad(a, eps):
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    maxc = conv.max()
    if eps > 1e-12:
        w = np.exp((conv - maxc) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        return g, maxc
    else:
        # subgradient
        k = int(conv.argmax())
        g = np.zeros_like(a)
        start = max(0, k - (n - 1))
        end = min(k + 1, n)
        idx = np.arange(start, end)
        g[idx] = 2.0 * a[k - idx]
        return g, maxc
```

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # Init a0
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 2000
        a0 = np.full(n_fallback, 1.0/n_fallback, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s < 1e-12:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s

    n = a0.size
    L = 1 << ((3 * n - 2).bit_length())

    def project_simplex(v):
        # v: 1-D np.ndarray
        n_ = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n_ + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    def fast_score(a):
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    # bookkeeping
    cur = a0.copy()
    cur_sc = fast_score(cur)
    best = cur.copy()
    best_sc = cur_sc

    vel = np.zeros_like(cur)
    eps = 0.02           # start smoothing
    eps_min = 1e-9
    eps_decay = 0.9995
    step0 = 0.7
    momentum = 0.85
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # compute conv and gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()
        # gradient
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]
            eps = max(eps_min, eps * eps_decay)
        else:
            k = int(conv.argmax())
            g = np.zeros_like(cur)
            start_i = max(0, k - (n - 1))
            end_i = min(k + 1, n)
            rng_idx = np.arange(start_i, end_i)
            g[rng_idx] = 2.0 * cur[k - rng_idx]

        # momentum update
        grad_norm = np.linalg.norm(g) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm
        vel = momentum * vel - alpha * g
        cand = cur + vel
        cand = project_simplex(cand)

        cand_sc = fast_score(cand)

        # accept improvement
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            vel = np.zeros_like(cur)   # reset velocity after successful step maybe
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # occasionally accept worse to escape
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12)):
                cur, cur_sc = cand, cand_sc
                # keep velocity as is
            no_improve += 1

        # symmetrization
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = project_simplex(sym)
            sym_sc = fast_score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                vel = np.zeros_like(cur)
            if sym_sc < best_sc:
                best, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # random mass transfer when stagnated
        if no_improve > 6000:
            # pick random i with some mass, transfer to j
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = project_simplex(tmp)
                tmp_sc = fast_score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    vel = np.zeros_like(cur)
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0
                else:
                    no_improve = 0
        # early exit if target
        if best_sc < 1.50274349:
            break

    return [float(x) for x in best]

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
