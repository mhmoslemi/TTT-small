import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a vector a (assumes sum(a)=1)
# ------------------------------------------------------------
def _score_vec(a: np.ndarray, L_sc: int) -> float:
    af = np.fft.rfft(a, n=L_sc)
    conv = np.fft.irfft(af * af, n=L_sc)[: 2 * a.shape[0] - 1]
    return float(2.0 * a.shape[0] * conv.max())
