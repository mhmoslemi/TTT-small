import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that is a feasible
    candidate for the AC1 bound.  The construction is simply the uniform
    distribution over the available length, which is optimal for the
    non‑negative setting (it minimises the ℓ₂ norm for a fixed ℓ₁ norm).
    """
    # Respect the allotted time budget – we finish immediately after setup.
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # Determine the length of the sequence.
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2000

    # Uniform vector with unit ℓ₁‑norm (sum = 1).
    a = np.full(n, 1.0 / n, dtype=np.float64)

    if time.time() > deadline:
