import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Compute max of linear convolution a*a (using FFT)
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.shape[0] - 1]
    return float(conv.max())

        g = 2.0 * grad_full[idx]

        # ---- normalised gradient step ----------------------------------------
        g_norm = np.linalg.norm(g) + 1e-12
        step_len = step0 / math.sqrt(it)    # decaying step length
        cand = cur - (step_len / g_norm) * g

        # ---- enforce feasibility (simplex) ------------------------------------
        cand = _project_simplex(cand)

        # ---- evaluate candidate -------------------------------------------------
        cand_max = _max_convolution(cand, L_conv)
        cand_sc = 2.0 * n * cand_max / (cand.sum() ** 2)

        # ---- acceptance --------------------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # occasional uphill move (simulated annealing style)
            prob = math.exp(-(cand_sc - cur_sc) * 0.05)
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ---- best‑so‑far update ------------------------------------------------
        if cand_sc < best_sc:
            best, best_sc = cand.copy(), cand_sc
            no_improve = 0

        # ---- random mass‑transfer when stuck -----------------------------------
        if no_improve > 5000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                if j == i:
                    j = (j + 1) % n
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_max = _max_convolution(tmp, L_conv)
                tmp_sc = 2.0 * n * tmp_max / (tmp.sum() ** 2)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
            no_improve = 0

        # ---- occasional symmetrisation (helps stability) ----------------------
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_max = _max_convolution(sym, L_conv)
            sym_sc = 2.0 * n * sym_max / (sym.sum() ** 2)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc

        # ---- early exit if target reached --------------------------------------
        if best_sc < 1.50274349:
            break

        # ---- time check --------------------------------------------------------
        if time.monotonic() >= deadline:
            break

    # --------------------------------------------------------
    # Return result as plain Python list (fallback to best)
    # --------------------------------------------------------
    return [float(x) for x in best]
