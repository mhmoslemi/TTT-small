def max_convolution(a):
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return conv.max()
```

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ensure nonnegative and sum positive

    # project onto simplex (sum=1)
    a = _project_simplex(a)
    n = a.shape[0]
    best = a.copy()
    best_score = _score(best)   # using same _score defined earlier
    cur = a.copy()
    cur_score = best_score
    L = 1 << ((2*n-1).bit_length())
    # parameters
    temperature = 1.0
    temp_decay = 0.9995
    min_temp = 1e-5
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # random move
        i = rng.integers(n)
        # ensure we pick j != i:
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # delta up to fraction of cur[i]
        max_delta = cur[i] * 0.5  # up to half of mass at i
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure nonnegative: already ensured since delta <= cur[i]
        # no need to renorm sum as total unchanged
        # compute score via FFT
        af = np.fft.rfft(cand, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        cand_max = conv.max()
        cand_score = 2.0 * n * cand_max  # sum=1
        # Accept based on temperature
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score)/temperature):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                # If beat target, break
                if best_score < 1.50274349:
                    break
        # temperature schedule
        temperature = max(min_temp, temperature * temp_decay)
        # occasional reset: if not improved for many steps
        if it % 10000 == 0:

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        max_val = conv.max()

        # smoothed max weights (log‑sum‑exp trick)
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        ws = w.sum()
        if ws == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= ws

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)   # length ≥3n‑2
        idx = (2 * n - 2) - np.arange(n)               # slice needed for gradient
        g = 2.0 * grad_full[idx]

        # step size (inverse‑norm scaling)
        grad_norm = np.linalg.norm(g) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm

        # try a few scalar multiples of the basic step
        best_local = None
        best_local_sc = float('inf')
        for scale in (0.5, 1.0, 1.5):
            cand = cur - scale * alpha * g
            cand = _project_simplex(cand)
            cand_sc = _score(cand)
            if cand_sc < best_local_sc:
                best_local = cand
                best_local_sc = cand_sc

        # acceptance
        if best_local_sc < cur_sc:
            cur, cur_sc = best_local, best_local_sc
            no_improve = 0
            if best_local_sc < best_sc:
                best, best_sc = best_local.copy(), best_local_sc
                if best_sc < 1.50274349:          # early exit on success
                    break
        else:
            no_improve += 1

        # occasional random mass‑transfer when stuck
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # symmetrisation every few thousand steps (helps stability)
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc

    # return plain Python list
    return [float(x) for x in best]
