def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # margin

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # generate random start with length 2048 for speed; pad to nearest power of two maybe.
        n = 2048
        a = rng.random(n, dtype=np.float64)

    a = np.maximum(a, 0.0)

    # Normalize to sum=1
    a = _project_simplex(a)

    n = a.shape[0]
    # FFT length for convolution up to 2n-1 => at least 2*n, best power of two
    L = 1 << ((2 * n - 1).bit_length() + 1)  # maybe 2 * next power
    # to avoid extra rounding.
    # Actually we can pick L = 1 << ((2*n - 1).bit_length())
    L = 1 << ((2 * n - 1).bit_length())

    # define scoring function using FFT
    def _score_vec(v):
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    best = a.copy()
    best_sc = _score_vec(best)

    # parameters
    eta0 = 0.7       # step size
    beta = 0.85      # momentum factor
    eps0 = 0.5
    eps_decay = 0.99
    eps_min = 1e-6
    target = 1.50274349
    momentum = np.zeros_like(a)
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        # compute convolution and max
        af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # smoothing weight
        eps = max(eps_min, eps0 * (eps_decay ** it))
        # Compute softmax weights over conv to get smooth gradient approximating max.
        # w_i proportional to exp((conv_i - max_val)/eps)
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of weighted sum = 2 * convolution of a with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * w_fft, n=L)
        # Only need first n entries corresponding to gradient for each a_j:
        # gradient_j = 2 * sum_{m} a_{m} * w_{j+m}?? Actually derived earlier: gradient_j = 2 * sum_{k} a_k * w_{j+k}. This is convolution of a and w at index j + maybe offset.
        # Using conv of a with w but taking appropriate slice.
        # The full convolution of a (size n) with w (size 2n-1) yields length 3n-2. Our grad_full length is L (>=3n-2?). Actually a_fft size L, w_fft size L, product yields length L. irfft gives length L.
        # Using earlier code: idx = (2 * n - 2) - np.arange(n) to extract proper indices.
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size (inverse sqrt schedule)
        alpha = eta0 / math.sqrt(it)

        # candidate update
        cand = a - alpha * momentum
        cand = _project_simplex(cand)

        # symmetrize every few iterations
        if it % 200 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        # evaluate candidate
        cand_sc = _score_vec(cand)

        # accept if improves
        if cand_sc <= cur_sc:
            a = cand
            cur_sc = cand_sc
        else:
            # occasionally accept uphill with small prob
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                a = cand
                cur_sc = cand_sc

        # random mass-tranfer if stagnation
        if no_improve > 3000:
            # random move
            i = rng.integers(0, n)
            if a[i] > 0:
                delta = a[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                a = tmp
                cur_sc = _score_vec(a)
                no_improve = 0

        # early exit
        if best_sc < target:
            break

    return [float(x) for x in best]

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function whose AC1 bound is as low as possible."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # leave a safety margin

    # ---- initial vector ----------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2048, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # sum == 1, non‑negative
    n = cur.shape[0]

    # FFT length large enough for all convolutions (≥3n‑2, power of two)
    L = 1 << ((3 * n - 2).bit_length())

    # scoring function (sum == 1 ⇒ score = 2·n·max(conv))
    def _score_vec(v: np.ndarray) -> float:
        af = np.fft.rfft(v, n=L)

        return float(2.0 * n * conv.max())

    eps_min = 1e-6
