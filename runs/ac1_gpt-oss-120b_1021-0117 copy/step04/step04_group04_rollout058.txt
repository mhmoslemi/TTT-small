def propose_candidate(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s)-0.5)

    # initialization
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        a = np.exp(-np.arange(n0) / 200.0)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)   # ensures sum=1

    n = a.shape[0]

    L = 1 << ((3*n - 2).bit_length())

    best = a.copy()
    best_sc = _score(best)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.05  # base learning rate

    # smoothing parameters
    eps0 = 0.5
    eps_decay = 0.9995
    eps_min = 1e-6

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        # compute FFT
        a_fft = np.fft.rfft(a, n=L)
        # convolution a*a
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        # compute smoothed weights
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        sum_w = w.sum()
        if sum_w == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= sum_w

        # gradient
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)

        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr / (np.sqrt(v_hat) + eps_adam)

            best = a.copy()
            no_improve = 0

import time

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast score for a vector that is already normalised to sum==1
# ------------------------------------------------------------
def _score_normed(a: np.ndarray, conv: np.ndarray = None) -> float:
    n = a.shape[0]
    if conv is None:
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
