    rng = np.random.default_rng(seed)

    else:
        a = a / s

    n = a.shape[0]
    L = 1 << ((3*n - 2).bit_length())

    # Adam vars
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    beta1_pow = 1.0
    beta2_pow = 1.0
    lr = 0.5
    eps_adam = 1e-8

    # smoothing epsilon
    eps = 0.05
    eps_decay = 0.9999
    eps_min = 1e-8

    best_vec = a.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc
    cur = a.copy()

    # iteration
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # FFT of current
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # smoothing weights
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient via FFT
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # occasional acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # symmetrize occasionally
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        if no_improve > 5000:

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast evaluation (sum of coefficients is assumed to be 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(a*a) where * is linear convolution."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
