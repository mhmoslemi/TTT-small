    # --------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------
    best_vec = cur.copy()
    best_score = 2.0 * n * _max_conv(best_vec, L)
    it = 0
    no_improve = 0
    target = 1.50274349

    # --------------------------------------------------------
    # Optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its score -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # ----- smoothed soft‑max weights -----
        eps_temp = max(eps_min, eps_temp * eps_decay)
        w = np.exp((conv - max_val) / eps_temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # reverse slice indices
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_score = 2.0 * n * _max_conv(cand, L)

        # ----- accept if improvement -----
        if cand_score < cur_score:
            cur = cand
            if cand_score < best_score:
                best_vec = cand.copy()
                best_score = cand_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # ----- occasional random mass‑transfer when stuck -----
            if no_improve > 2000:
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    delta = cur[i] * rng.random() * 0.5
                    j = rng.integers(0, n)
                    while j == i:
                        j = rng.integers(0, n)
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_score = 2.0 * n * _max_conv(tmp, L)
                    if tmp_score < cur_score:
                        cur = tmp
                        cur_score = tmp_score
                        if tmp_score < best_score:
                            best_vec = tmp.copy()
                            best_score = tmp_score
                no_improve = 0

        # ----- periodic symmetrisation (helps with Toeplitz structure) -----
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = 2.0 * n * _max_conv(sym, L)
            if sym_score < cur_score:
                cur = sym
                if sym_score < best_score:
                    best_vec = sym.copy()
                    best_score = sym_score

        # ----- early exit if target reached -----
        if best_score < target:
            break

    # --------------------------------------------------------
    # Return the best candidate found
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
