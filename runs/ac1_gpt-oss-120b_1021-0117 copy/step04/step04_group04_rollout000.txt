import time

# ------------------------------------------------------------
# Simplex projection (Euclidean) for non‑negative vectors with sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    best_sc = _score(best, a_fft)
    cur = best.copy()
    cur_sc = best_sc

    # momentum buffer
    m = np.zeros_like(cur)

    # hyper‑parameters
    beta = 0.9               # momentum decay
    step0 = 0.6              # base step size
    eps0 = 1.0               # initial smoothing for log‑sum‑exp
    eps_decay = 0.9995
    eps_min = 1e-8
    it = 0
    no_improve = 0

    # ------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current FFT (re‑used for both conv and gradient)
        a_fft = np.fft.rfft(cur, n=L)

        # full linear convolution a * a  (length 2n‑1)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # smooth approximation of max (log‑sum‑exp)
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv_full - max_val) / eps)      # max becomes 1
        ws = w.sum()
        if ws == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= ws

        # gradient of the smoothed objective:
        # g = 2 * (a * w_rev)  (linear conv, slice to length n)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)  # length L
        idx = (2 * n - 2) - np.arange(n)              # pick proper slice
        g = 2.0 * grad_full[idx]

        # momentum update
        m = beta * m + (1.0 - beta) * g
        norm_m = np.linalg.norm(m) + 1e-12
        alpha = step0 / math.sqrt(it) / norm_m

        # candidate step + projection onto simplex
        cand = cur - alpha * m
        cand = _project_simplex(cand)

        # occasional symmetrisation (helps stability)
        if it % 3000 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # acceptance rule (deterministic improvement, occasional uphill)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # simple simulated‑annealing style probability
            temperature = 0.01 * (1.0 + it / 10000.0)
            prob = math.exp(-(cand_sc - cur_sc) / (temperature + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # random mass‑transfer when stuck for a while
        if no_improve > 5000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # early exit on target
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------
    # Final local refinement (if time permits)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        # try a handful of tiny random transfers
        improved = False
        for _ in range(200):
            i = rng.integers(0, n)
            if best[i] <= 0.0:
                continue
            delta = best[i] * rng.random() * 0.15
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)
            cand_sc = _score(cand)
            if cand_sc < best_sc:
                best, best_sc = cand, cand_sc
                improved = True
                break
        if not improved:
            break

    # ------------------------------------------------------------
    # Return plain Python list
    # ------------------------------------------------------------
    return [float(x) for x in best]
