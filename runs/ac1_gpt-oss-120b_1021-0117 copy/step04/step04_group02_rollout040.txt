import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# 2·n·max(conv)  (sum of a is 1, so denominator =1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact score used by the evaluator (sum(a)=1)."""
    # convolution via FFT (linear convolution)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # -----------------------------------------------------------------
    # Initialise – use the provided construction if available,
    # otherwise fall back to a uniform distribution.
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s                     # simplex (sum = 1)
    else:
        n = 2000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())      # FFT length (power of two)

    cur = a.copy()
    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------
    it = 0
    no_improve = 0
    base_alpha = 0.6
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # ----- autocorrelation (linear convolution) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        idx_max = int(conv.argmax())          # where the max occurs
        max_val = conv[idx_max]

        # ----- gradient of the (unsmoothed) max -----
        # shift k such that conv index = (n-1)+k
        k = idx_max - (n - 1)
        g = np.zeros_like(cur)

        if k >= 0:
            # term a_{j+k}
            g[: n - k] += cur[k:]
            # term a_{j-k}
            g[k:] += cur[: n - k]
        else:
            k = -k
            # term a_{j-k}
            g[k:] += cur[: n - k]
            # term a_{j+k}
            g[: n - k] += cur[k:]

        g *= 2.0
        g_norm = np.linalg.norm(g) + 1e-12

        # ----- step size (decaying) -----
        alpha = base_alpha / math.sqrt(it) / g_norm

        # ----- projected gradient step -----

        cand_sc = _score(cand)
