    # helper functions defined earlier ensure accessible.

    # initialization:
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()

            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s

    n = a0.size
    L = 1 << ((3*n-2).bit_length())
    cur = a0.copy()
    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # parameters
    base_alpha = 0.8
    eps_decay = 0.9995
    eps = None
    mu = 0.9
    v = np.zeros_like(cur)
    it = 0
    no_improve = 0
    sym_interval = 250
    improve_target = 1.50274349
    restart_limit = 2000

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    while time.monotonic() < deadline:
        it += 1

        # compute FFT and convolution
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        # update epsilon
        if eps is None:
            eps = max(max_val * 0.2, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)

        # compute softmax weights
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # compute gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        cand = cur + v

                best_sc = cand_sc

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
