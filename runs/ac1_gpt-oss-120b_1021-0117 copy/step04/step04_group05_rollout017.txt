def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0: theta = 0.0
    else: rho = rho[-1]; theta = css[rho] / (rho+1)
    return np.maximum(v - theta, 0.0)
```

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 1024
        cur = np.exp(-np.arange(n) / (n/10.0))

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.shape[0]
    L = 1 << ((3*n - 2).bit_length())   # ensure >=3n-2
    cur_fft = np.fft.rfft(cur, n=L)
    cur_conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)
    cur_conv = cur_conv_full[:2*n-1]
    cur_max = cur_conv.max()
    cur_sc = 2.0 * n * cur_max

    best_vec = cur.copy()
    best_sc = cur_sc

    step0 = 2.0
    eps_rel = 0.02
    eps_min = 1e-9
    beta = 0.9
    v = np.zeros_like(cur)
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        # compute weights
        eps = max(eps_min, cur_max * eps_rel)
        w = np.exp((cur_conv - cur_max) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        w_rev_fft = np.fft.rfft(w[::-1], n=L)
        grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size
        step = step0 / math.sqrt(it)

        # line search/backtrack
        alpha = step
        cand_sc = None
        for _ in range(4):

            cand = _project_simplex(cand)
            cand_fft = np.fft.rfft(cand, n=L)
            cand_conv_full = np.fft.irfft(cand_fft * cand_fft, n=L)
            cand_conv = cand_conv_full[:2*n-1]

            cur_fft = cand_fft
            cur_conv = cand_conv

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initial vector (parent if present, else simple exponential fallback)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 1024
        cur = np.exp(-np.arange(n_fallback, dtype=np.float64) / (n_fallback / 10.0))
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)                     # sum == 1, non‑negative
    n = cur.shape[0]

    # ------------------------------------------------------------------
    # FFT setup (single length works for both convolution & gradient)
    # ------------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())            # >= 3n‑2, power of two

    # Scoring helpers
    # ------------------------------------------------------------------

    cur_conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)

    cur_max = cur_conv.max()
    cur_sc = 2.0 * n * cur_max

    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    step0 = 2.0                 # initial step magnitude
    eps_rel = 0.02              # relative smoothing
    eps_min = 1e-9
    beta = 0.9                  # momentum coefficient
    v = np.zeros_like(cur)      # momentum buffer
    it = 0
    no_improve = 0
    max_random = 30             # quick random start explorations

    # Quick random sampling before gradient descent
    # ------------------------------------------------------------------
    for _ in range(max_random):
        if time.monotonic() >= deadline:
            break

    while time.monotonic() < deadline:
