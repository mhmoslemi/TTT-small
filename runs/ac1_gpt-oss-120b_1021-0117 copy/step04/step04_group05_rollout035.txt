import time

# --------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L_full)[: 2 * n - 1]
            sym_max = sym_conv.max()
            sym_score = 2.0 * n * sym_max
            if sym_score < cur_score:
                prev_vec = cur
                cur = sym
                cur_fft = sym_fft
                cur_conv = sym_conv
                cur_max = sym_max
                cur_score = sym_score
                no_improve = 0
                if sym_score < best_score:
                    best_score = sym_score
                    best_vec = sym.copy()

        # ---- random mass transfer when stuck ----
        if no_improve > 5000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L_full)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L_full)[: 2 * n - 1]
                tmp_max = tmp_conv.max()
                tmp_score = 2.0 * n * tmp_max
                if tmp_score < cur_score:
                    prev_vec = cur
                    cur = tmp
                    cur_fft = tmp_fft
                    cur_conv = tmp_conv
                    cur_max = tmp_max
                    cur_score = tmp_score
                if tmp_score < best_score:
                    best_score = tmp_score
                    best_vec = tmp.copy()
                no_improve = 0

        # ---- occasional jitter‑restart from the best solution ----
        if no_improve > 20000:
            noise = rng.normal(scale=0.01, size=n)
            cur = best_vec * (1.0 + noise)
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            cur_fft = np.fft.rfft(cur, n=L_full)
            cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
            cur_max = cur_conv.max()
            cur_score = 2.0 * n * cur_max
            no_improve = 0
            prev_vec = cur.copy()

        # ---- early exit if target met ----
        if best_score < 1.50274349:
            break

    # ----------------------------------------------------------
    # Return best candidate as plain Python list
    # ----------------------------------------------------------
    return [float(x) for x in best_vec]
