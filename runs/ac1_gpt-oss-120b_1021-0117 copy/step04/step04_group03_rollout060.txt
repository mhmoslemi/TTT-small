def _subgradient_maxconv(f, conv, max_val, tol=1e-12):

    n = f.shape[0]

    idxs = np.where(conv >= max_val - tol)[0]

    g = np.zeros_like(f)

    for s in idxs:

        # j range start..end inclusive

        start = max(0, s - (n - 1))

        end = min(n - 1, s)

        # j from start to end (inclusive)

        j = np.arange(start, end + 1)

        f_idx = s - j  # vector of same length

        g[j] += 2 * f[f_idx]

    return g

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Convolution via FFT (full length)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv


# ------------------------------------------------------------
# Scoring function (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    conv = _conv_fft(a, L)
    return float(2.0 * a.size * conv.max())


# ------------------------------------------------------------
# Subgradient of max(conv) (using indices near the maximum)
# ------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray, M: float,
                     tol_factor: float = 1e-8) -> np.ndarray:
    """Return subgradient of max(conv) at a."""
    n = a.size
    thresh = max(1e-12, M * tol_factor)
    idxs = np.where(conv >= M - thresh)[0]
    g = np.zeros_like(a)
    for s in idxs:
        start = max(0, s - (n - 1))
        end = min(n - 1, s)
        j = np.arange(start, end + 1, dtype=np.int64)
        aj = s - j
        g[j] += 2.0 * a[aj]
    return g
