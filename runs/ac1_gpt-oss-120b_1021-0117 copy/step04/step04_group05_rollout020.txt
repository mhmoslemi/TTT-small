# ----------------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries non‑positive: fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety – ensure sum is exactly one
    s = w.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / s
