        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Subgradient of max_k (a*a)_k
# ------------------------------------------------------------
def _subgradient(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """Subgradient of the maximum autocorrelation entry."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in max_idxs:
        i_start = k - (n - 1)
        if i_start < 0:
            i_start = 0
        i_end = k
        if i_end > n - 1:
            i_end = n - 1
        idx = np.arange(i_start, i_end + 1)
        # contribution 2 * a_{k-i}
        g[idx] += 2.0 * a[k - idx]
    if max_idxs.size:
        g /= float(max_idxs.size)
    return g

    # start from a symmetric shape (helps)
    a = (a + a[::-1]) * 0.5
    a = _project_simplex(a)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    cur = a.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------
    it = 0
    no_improve = 0
    base_alpha = 0.9          # base step size
    beta = 0.6                # momentum factor
    improve_target = 1.50274349
    sym_interval = 300        # how often to symmetrise
    v = np.zeros_like(cur)    # momentum vector

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current iterate -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- early exit if target already reached -----
        if best_sc <= improve_target:
            break

        # ----- locate all maximal positions -----
        max_idxs = np.where(conv >= max_val - 1e-12)[0]

        # ----- subgradient of the maximum -----
        g = _subgradient(cur, max_idxs)
        g_norm = np.linalg.norm(g) + 1e-12

        # ----- step size (decaying) -----
        alpha = base_alpha / (g_norm * math.sqrt(it))

        # ----- backtracking line‑search (few trials) -----
        improved = False
        cand = None
        cand_sc = None
        trial_alpha = alpha
        for _ in range(5):
            cand_try = cur - trial_alpha * g
            cand_try = _project_simplex(cand_try)
            cand_sc_try = _score(cand_try)
            if cand_sc_try < cur_sc:
                cand = cand_try
                cand_sc = cand_sc_try
                # update momentum after a successful step
                v = beta * v - trial_alpha * g
                improved = True
                break
            trial_alpha *= 0.5

        # ----- fallback: try using momentum alone if gradient step fails -----
        if not improved:
            cand_try = cur + v
            cand_try = _project_simplex(cand_try)
            cand_sc_try = _score(cand_try)
            if cand_sc_try < cur_sc:
                cand = cand_try
                cand_sc = cand_sc_try
                improved = True
            else:
                # reset momentum when stuck
                v = np.zeros_like(cur)

        # ----- random mass‑transfer when stuck for a long time -----
        if not improved and no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cand = tmp
                    cand_sc = tmp_sc
                    improved = True
                # reset counter regardless of success
                no_improve = 0

        # ----- accept if any improvement happened -----
        if improved:
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_vec = cur.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation (helps a lot) -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

    # return as plain list of Python floats
    return [float(x) for x in best_vec]
