import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Initial vector (use parent if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 1024
        cur = np.exp(-np.arange(n_fallback, dtype=np.float64) / (n_fallback / 10.0))
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce non‑negativity & sum == 1
    n = cur.shape[0]

    # ------------------------------------------------------------------
    # FFT preparation
    # ------------------------------------------------------------------
    L_full = 1 << ((3 * n - 2).bit_length())   # power of two >= 3n‑2

    cur_fft = np.fft.rfft(cur, n=L_full)
    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
    cur_max = cur_conv.max()
    cur_sc = 2.0 * n * cur_max
    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    step0 = 0.9
    init_eps_rel = 0.02
    eps_min = 1e-9
    target = 1.50274349
    it = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- adaptive smoothing for the max -----
        eps_rel = max(1e-4, init_eps_rel / math.sqrt(it))
        eps = max(eps_min, cur_max * eps_rel)

        w = np.exp((cur_conv - cur_max) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L_full)
        grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L_full)
        idx = (2 * n - 2) - np.arange(n)          # map to coordinates
        g = 2.0 * grad_full[idx]

        # ----- backtracking line search -----
        alpha = step0 / math.sqrt(it)
        accepted = False
        for _ in range(6):
            cand = cur - alpha * g
            cand = _project_simplex(cand)
            cand_fft = np.fft.rfft(cand, n=L_full)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L_full)[: 2 * n - 1]
            cand_max = cand_conv.max()
            cand_sc = 2.0 * n * cand_max
            if cand_sc < cur_sc:
                accepted = True
                break
            alpha *= 0.5

        if accepted:
            cur = cand
            cur_fft = cand_fft
            cur_conv = cand_conv
            cur_max = cand_max
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            # occasional probabilistic acceptance (simulated annealing flavour)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_fft = cand_fft
                cur_conv = cand_conv
                cur_max = cand_max
                cur_sc = cand_sc
                no_improve = 0
            else:
                no_improve += 1

        # ----- periodic symmetrisation -----
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L_full)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L_full)[: 2 * n - 1]
            sym_max = sym_conv.max()
            sym_sc = 2.0 * n * sym_max
            if sym_sc < cur_sc:
                cur = sym
                cur_fft = sym_fft
                cur_conv = sym_conv
                cur_max = sym_max
                cur_sc = sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()

        # ----- stochastic mass‑transfer when stuck -----
        if no_improve > 2000:
            src = cur if rng.random() < 0.5 else best_vec
            tmp = src.copy()
            i = rng.integers(0, n)
            if tmp[i] > 0.0:
                delta = tmp[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L_full)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L_full)[: 2 * n - 1]
                tmp_max = tmp_conv.max()
                tmp_sc = 2.0 * n * tmp_max
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_fft = tmp_fft
                    cur_conv = tmp_conv
                    cur_max = tmp_max
                    cur_sc = tmp_sc
                no_improve = 0

        # ----- tiny random perturbation (always cheap) -----
        if rng.random() < 0.001:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.2
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L_full)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L_full)[: 2 * n - 1]
                tmp_max = tmp_conv.max()
                tmp_sc = 2.0 * n * tmp_max
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_fft = tmp_fft
                    cur_conv = tmp_conv
                    cur_max = tmp_max
                    cur_sc = tmp_sc
                    no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # Return the best candidate found (as plain Python list)
    return [float(x) for x in best_vec]
