def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math, random
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = float(budget_s)
    deadline = start + max(0.0, total - 0.2)  # small safety

    # Get base sequence if available
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base = np.maximum(base, 0.0)
    else:
        base = None

    n = None
    # generate candidate sequences
    cand_list = []

    # base if not None: simplex project
    if base is not None:
        n = base.size
        # simple projection onto simplex
        base = _project_simplex(base)
        cand_list.append(base.copy())

    # ensure have n; if no base, choose n=2000 or 2500
    if n is None:
        n = 2000
    else:
        n = int(n)

    # Make uniform length n
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    cand_list.append(uniform)

    # arcsine shape
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0/np.sqrt((i+0.5)*(n-0.5 - i))
    arcsine = arcsine / arcsine.sum()
    cand_list.append(arcsine)

    # sine shape
    sine = np.sin(np.pi * (i+0.5) / n)
    sine = sine / sine.sum()
    cand_list.append(sine)

    # semicircle shape
    x = (i - (n-1)/2) / ((n-1)/2)
    semicircle = np.sqrt(np.maximum(0.0, 1 - x**2))
    semicircle = semicircle / semicircle.sum()
    cand_list.append(semicircle)

    # random Dirichlet samples (few)
    for _ in range(3):
        rand = rng.random(n)
        rand = _project_simplex(rand)  # sums to 1
        cand_list.append(rand)

    # Evaluate candidates using fast _score (sum=1)
    best_vec = None
    best_score = np.inf
    for cand in cand_list:
        sc = _score(cand)
        if sc < best_score:
            best_score, best_vec = sc, cand.copy()
    cur = best_vec.copy()
    cur_score = best_score

    # LP improvement (optional) similar to before but as fallback quick
    # can be left out if limited; but can try quick small LP.

    # Precompute FFT length
    L = 1 << ((2*n - 1).bit_length())
    # Optionally bigger length for convolution of weight vector
    # but we need length for smoothing weights: using conv length 2n-1 -> weight w length 2n-1

    eps_decay = 0.9995
    eps_min = 1e-8

    beta = 0.9

    no_improve = 0

        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2*n-1]
        max_val = conv.max()

        eps = max(eps_min, eps * eps_decay)
        # shift for numerical stability
        # Already max_val is max.

import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a vector already normalised to sum==1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total_budget = float(budget_s)
    # small safety margin before hard timeout
    deadline = start + max(0.0, total_budget - 0.3)

    # -----------------------------------------------------------------
    # Determine length and basic candidates
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base = np.maximum(base, 0.0)
        n = base.size
        base = _project_simplex(base)
    else:
        base = None
        n = 2048                     # fallback length

    # generate a handful of diverse starters (all normalised)
    i = np.arange(n, dtype=np.float64)
    candidates = []

    if base is not None:
        candidates.append(base)

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # arcsine (beta ½,½)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    candidates.append(arcsine / arcsine.sum())

    # raised cosine (sin)
    sine = np.sin(np.pi * (i + 0.5) / n)
    candidates.append(sine / sine.sum())

    # semicircle
    x = (i - (n - 1) / 2) / ((n - 1) / 2)
    semicircle = np.sqrt(np.maximum(0.0, 1 - x ** 2))
    candidates.append(semicircle / semicircle.sum())

    # beta‑type families (symmetrical)
    for p in (0.3, 0.6, 0.9, 1.4, 1.8):
        beta = (i + 0.5) ** (p - 1) * (n - i - 0.5) ** (p - 1)
        beta = np.maximum(beta, 0.0)
        if beta.sum() > 0:
            candidates.append(beta / beta.sum())

    # a few random Dirichlet‑like samples
    for _ in range(3):
        rnd = rng.random(n)
        candidates.append(_project_simplex(rnd))

    # -----------------------------------------------------------------
    # Pick the best cheap starter
    # -----------------------------------------------------------------
    best_vec = None
    best_sc = math.inf
    for cand in candidates:
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # -----------------------------------------------------------------
    # Gradient descent on a smoothed max (soft‑max) with momentum
    # -----------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())
    eps = 1.0               # initial temperature
    eps_decay = 0.9995
    eps_min = 1e-9

    step0 = 0.45            # base step size (will be scaled by gradient norm)
    beta_mom = 0.9          # momentum coefficient
    momentum = np.zeros_like(cur)

    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # convolution a*a
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weights
        eps = max(eps_min, eps * eps_decay)
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

        idx = (2 * n - 2) - np.arange(n)          # map to a‑indices

        if best_sc < 1.50274349:

    # -----------------------------------------------------------------
    # Return the best vector found (as plain Python list of floats)
    # -----------------------------------------------------------------
    if best_vec is None:
        best_vec = cur
    return [float(x) for x in best_vec]
