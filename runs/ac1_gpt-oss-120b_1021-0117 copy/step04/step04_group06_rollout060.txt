import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast score (2 * n * max(conv)), assuming non‑negative a
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of the (soft) max of the convolution
# ------------------------------------------------------------
def _grad_max(a: np.ndarray, L: int, eps: float) -> np.ndarray:
    """Return sub‑gradient of max(conv) using a soft‑max weighting."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]

    # soft‑max weights on the convolution entries
    max_val = conv.max()
    if eps > 1e-12:
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum
    else:
        # hard max (uniform over exact maximisers)
        mask = conv == max_val
        w = mask.astype(float)
        w /= w.sum()

    # gradient via convolution of a with reversed weights
    w_rev = w[::-1]
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    idx = (2 * n - 2) - np.arange(n)          # pick n entries
    return 2.0 * grad_full[idx]
