def _project_simplex(v):
    # expects v may not sum to 1.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css/np.arange(1, n+1) > 0)[0]
    if rho.size > 0:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0
    return w

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

# ------------------------------------------------------------
# Fast score (2 * n * max(convolution))  – assumes sum(a)=1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(a*a) using FFT of length L (L >= 2n-1)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # -----------------------------------------------------------------
    # initialise a starting vector (use parent if available)
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2500
        a0 = np.exp(-np.arange(n0) / 200.0)  # simple exponential decay
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)               # enforce sum == 1
    n = a0.shape[0]

    # FFT length – must accommodate convolutions up to length 3n-2
    L = 1 << ((3 * n - 2).bit_length())

    # -----------------------------------------------------------------
    # Adam hyper‑parameters
    # -----------------------------------------------------------------
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a0)
    v = np.zeros_like(a0)
    beta1_pow = beta1
    beta2_pow = beta2

    # -----------------------------------------------------------------
    # Smoothed‑max schedule
    # -----------------------------------------------------------------
    eps0 = 0.2
    eps_decay = 0.9995
    eps_min = 1e-12

    # -----------------------------------------------------------------
    # bookkeeping
    # -----------------------------------------------------------------
    best = a0.copy()
    best_sc = _score(best, L)
    cur = best.copy()
    cur_sc = best_sc
    no_improve = 0
    it = 0

    # -----------------------------------------------------------------
    # optimisation loop
    # -----------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current candidate -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ----- compute smoothed weight vector -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        if eps < 1e-6:                     # effectively a hard max
            w = np.zeros_like(conv)
            w_idx = int(np.argmax(conv))
            w[w_idx] = 1.0
        else:
            # log‑sum‑exp trick for numerical stability
            w = np.exp((conv - max_val) / eps)
            ws = w.sum()
            if ws == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= ws

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_sc = _score(cand, L)

        # ----- acceptance criterion -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # occasional simulated‑annealing style acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- adapt Adam bias‑correction powers -----
        beta1_pow *= beta1
        beta2_pow *= beta2

        # ----- occasional random perturbation when stuck -----
        if no_improve > 5000:
            # perturb around the best known solution
            cur = best + 0.02 * rng.standard_normal(n)
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            cur_sc = _score(cur, L)
            no_improve = 0

        # ----- symmetry enforcement every few thousand steps -----
        if it % 5000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc

        # ----- early exit if target reached -----
        if best_sc < 1.50274349:
            break

    # -----------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # -----------------------------------------------------------------
    return [float(x) for x in best]
