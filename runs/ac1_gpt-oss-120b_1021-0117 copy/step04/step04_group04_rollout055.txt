import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Gradient of the max‑convolution (exact sub‑gradient)
# ------------------------------------------------------------
def _grad_max_convolution(a: np.ndarray, max_idxs) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] w.r.t. a."""
    n = a.shape[0]
    g = np.zeros_like(a)
    cnt = len(max_idxs)
    for k in max_idxs:
        if k < n:
            # indices 0..k
            g[: k + 1] += a[k :: -1]          # a[k], a[k-1], … , a[0]
        else:
            # indices k-n+1 .. n-1
            start = k - n + 1
            g[start:] += a[start:][::-1]      # a[n-1], a[n-2], … , a[start]
    g *= 2.0 / cnt
    return g

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        a0 = np.exp(-np.arange(n0) / 200.0)

    a0 = np.maximum(a0, 0.0)
    cur = _project_simplex(a0)               # sum == 1, non‑negative
    n = cur.shape[0]

    # FFT length sufficient for linear convolution (up to length 3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    # best solution tracking
    cur_fft = np.fft.rfft(cur, n=L)
    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
    cur_max = cur_conv.max()
    cur_score = 2.0 * n * cur_max
    best = cur.copy()
    best_score = cur_score

    # ----- optimisation hyper‑parameters ------------------------------
    step0 = 0.8            # initial step magnitude
    beta = 0.9             # momentum factor
    momentum = np.zeros_like(cur)
    it = 0
    stagnant = 0          # steps without improvement

    # ----- main loop -------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # --- convolution and max index(es) ---------------------------
        cur_fft = np.fft.rfft(cur, n=L)

    return [float(x) for x in best]
