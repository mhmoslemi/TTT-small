def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound for a simplex‑normalised vector
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    """Return 2*n*max(conv(a,a)) assuming sum(a)=1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback: 2000‑point Dirichlet‑like random start
        a0 = rng.random(2000, dtype=np.float64)
    a0 = np.clip(a0, 0.0, None)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / s                     # simplex normalisation

    n = a0.size
    # FFT length for linear convolution of length 2n‑1
    L = 1 << ((3 * n - 2).bit_length())

    # ----- bookkeeping ----------------------------------------------------
    cur = a0.copy()
    cur_sc = _fast_score(cur, L, n)
    best = cur.copy()
    best_sc = cur_sc

    # algorithmic parameters
    eps = 0.2                # soft‑max temperature
    eps_min = 1e-12
    eps_decay = 0.9995
    beta1, beta2 = 0.9, 0.999
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    lr = 0.05                # Adam base learning rate
    no_improve = 0
    target = 1.50274349

    # ----- optimisation loop ----------------------------------------------
    while time.monotonic() < deadline:
        # convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # soft‑max weights for the max
        if eps > eps_min:
            y = (conv - maxc) / eps
            y = np.clip(y, -50, 0)            # avoid overflow
            w = np.exp(y)
            w_sum = w.sum()
            if w_sum == 0.0:
                w[:] = 1.0 / w.size
            else:
                w /= w_sum
            eps = max(eps_min, eps * eps_decay)
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # gradient of max‑conv (2 * a * w_reversed)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update (projected onto simplex)
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-12)
        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _fast_score(cand, L, n)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            # tiny uphill chance to escape tiny plateaus
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # periodic symmetrisation (the problem is essentially symmetric)
        if t % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, L, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # random mass transfer when stuck
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _fast_score(tmp, L, n)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc

        # early exit if target reached
        if best_sc < target:
            break

    # ------------------------------------------------------------
    # Return best found sequence as plain Python list
    # ------------------------------------------------------------
    return [float(x) for x in best]
