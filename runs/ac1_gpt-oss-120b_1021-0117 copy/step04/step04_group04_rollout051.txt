import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Helpers for convolution / score
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max of a*a (linear convolution) using FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _score_from_max(n: int, max_conv: float) -> float:
    """Score = 2 * n * max_conv (sum of a is 1)."""
    return 2.0 * n * max_conv


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a tiny safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        a0 = np.exp(-np.arange(n0) / 200.0)

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)          # enforce sum == 1
    n = a0.shape[0]

    # FFT length (next power of two >= 2*n-1)
    L = 1 << ((2 * n - 1).bit_length())

    # ----- bookkeeping ------------------------------------------------
    best = a0.copy()
    best_max = _conv_max(best, L)
    best_score = _score_from_max(n, best_max)

    cur = best.copy()
    cur_max = best_max
    cur_score = best_score

    # ----- optimisation parameters ------------------------------------
    step0 = 0.8            # initial step size factor
    eps = 0.6              # smoothing for log‑sum‑exp (unused in sub‑gradient mode)
    eps_decay = 0.9995
    no_improve = 0
    it = 0

    # ----- main loop -------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        cur_max = _conv_max(cur, L)

        # ---- sub‑gradient of the max‑convolution --------------------
        kmax = np.argmax(np.fft.irfft(np.fft.rfft(cur, n=L) * np.fft.rfft(cur, n=L), n=L)[: 2 * n - 1])
        idx = np.arange(n)
