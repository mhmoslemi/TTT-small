import time
import math
import numpy as np

# ---------- Euclidean simplex projection ----------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ---------- Convolution helpers ----------
def _conv_and_fft(a: np.ndarray, L: int):
    """Return (full linear convolution a*a, its max, FFT of a)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv, conv.max(), af


def _gradient(a: np.ndarray, w: np.ndarray, af: np.ndarray, L: int):
    """Sub‑gradient of the (soft‑max) max‑convolution w.r.t a."""
    # w is defined on the convolution index set {0,…,2n‑2}
    w_rev = w[::-1]
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    # pick entries that correspond to the gradient of each a_i
    n = a.size
    idx = (2 * n - 2) - np.arange(n)          # decreasing indices
    return 2.0 * grad_full[idx]


# ---------- Main entry ----------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence whose AC1 bound is low.
    The routine runs for at most `budget_s` seconds and keeps the best
    candidate found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # small safety margin

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n0 = base.size
    else:
        n0 = 1500
        base = np.ones(n0, dtype=np.float64)

    # optionally enlarge the problem (more freedom => lower possible bound)
    target_n = max(n0, 4000)          # choose a larger length if feasible
    if target_n > n0:
        a = np.zeros(target_n, dtype=np.float64)
        a[:n0] = base
    else:
        a = base.copy()

    a = np.maximum(a, 0.0)
    # start from a random point on the simplex (helps escaping pathological starts)
    a = rng.random(a.shape)
    a = _project_simplex(a)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two FFT length

    # ----- Adam optimiser state ---------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.2                           # base learning rate
    beta1_pow = 1.0
    beta2_pow = 1.0

    # ----- soft‑max smoothing schedule --------------------------------
    eps0 = 0.5
    eps_decay = 0.9998
    eps_min = 1e-12

    # ----- bookkeeping -------------------------------------------------
    conv, max_val, af = _conv_and_fft(a, L)
    cur_score = 2.0 * n * max_val
    best_vec = a.copy()
    best_score = cur_score
    it = 0
    no_improve = 0
    restart_after = 8000   # if no improvement for this many steps, re‑initialise

    # ----- main optimisation loop -------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # recompute convolution and its max (already have from previous step)
        # but after a potential update we need fresh values
        conv, max_val, af = _conv_and_fft(a, L)

        # soft‑max weights approximating the max
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth max
        g = _gradient(a, w, af, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = a - step
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_conv, cand_max, _ = _conv_and_fft(cand, L)
        cand_score = 2.0 * n * cand_max

        improved = False
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            improved = True
        else:
            # simple back‑track: halve step size a few times
            for factor in (0.5, 0.25, 0.125):
                cand_bt = a - step * factor
                cand_bt = _project_simplex(cand_bt)
                _, cand_bt_max, _ = _conv_and_fft(cand_bt, L)
                cand_bt_score = 2.0 * n * cand_bt_max
                if cand_bt_score < cur_score:
                    a = cand_bt
                    cur_score = cand_bt_score
                    improved = True
                    break

        if improved:
            no_improve = 0
            if cur_score < best_score:
                best_vec = a.copy()
                best_score = cur_score
        else:
            no_improve += 1

        # occasional random perturbation when stuck
        if no_improve >= restart_after:
            # re‑initialise with a fresh random simplex vector but keep the current best as a guide
            a = rng.random(n)
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            no_improve = 0
            continue

        # symmetrisation every few thousand steps (helps the even‑odd pattern)
        if it % 2500 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            _, sym_max, _ = _conv_and_fft(sym, L)
            sym_score = 2.0 * n * sym_max
            if sym_score < cur_score:
                a = sym
                cur_score = sym_score
                if sym_score < best_score:
                    best_vec = sym.copy()
                    best_score = sym_score
                no_improve = 0

        # early exit if target reached
        if best_score < 1.50274349:
            break

    # ----- fine‑tuning with remaining time ----------------------------
    while time.monotonic() < deadline - 0.05:
        delta = rng.normal(scale=0.005, size=n)
        cand = best_vec + delta
        cand = _project_simplex(cand)
        _, cand_max, _ = _conv_and_fft(cand, L)
        cand_score = 2.0 * n * cand_max
        if cand_score < best_score:
            best_vec = cand
            best_score = cand_score

    # ----- return result as plain Python list -------------------------
    return [float(x) for x in best_vec]
