def _proj_simplex(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 1e-15:

    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    cur = cur / s if s > 1e-12 else np.full_like(cur, 1.0 / cur.size)

    n = cur.size
    # FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())
    # indices used to extract the gradient from the full convolution
    grad_idx = (2 * n - 2) - np.arange(n)

    # --------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------
    cur_sc = _fast_score(cur, L)
    best = cur.copy()
    best_sc = cur_sc

    # hyper‑parameters
    step0 = 0.8          # base step size
    mu = 0.5             # momentum coefficient
    eps = 0.001          # smoothing for the soft‑max
    v = np.zeros_like(cur)   # momentum buffer

    it = 0
    no_improve = 0
    target = 1.50274349   # stop when we beat this bound

    # --------------------------------------------------------
    # optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and soft‑max weights -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()
        w = np.exp((conv - maxc) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed maximum -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[grad_idx]

        # ----- step with momentum -----
        grad_norm = np.linalg.norm(grad) + 1e-12
        alpha = step0 / math.sqrt(it) / grad_norm
        v = mu * v - alpha * grad
        cand = cur + v

        # ----- project onto simplex and enforce symmetry -----
        cand = _proj_simplex(cand)
        cand = (cand + cand[::-1]) * 0.5
        cand /= cand.sum()

        cand_sc = _fast_score(cand, L)

        # ----- accept / reject -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            # occasional uphill move (simulated‑annealing flavour)
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12)):
                cur, cur_sc = cand, cand_sc
                no_improve = 0
            else:
                v.fill(0.0)          # reset momentum on rejection
                no_improve += 1

        # ----- random jitter when we get stuck -----
        if no_improve > 5000:
            jitter = rng.random(n) * 0.01
            cur = cur + jitter
            cur = _proj_simplex(cur)
            cur = (cur + cur[::-1]) * 0.5
            cur /= cur.sum()
            cur_sc = _fast_score(cur, L)
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # --------------------------------------------------------
    # Return the best vector found (as a plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best]
