import time
import math
import numpy as np

# ---------- Euclidean projection onto the probability simplex ----------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ---------- Score of a vector (2*n*max(convolution)) ----------
def _score(vec: np.ndarray, fft_len: int) -> float:
    n = vec.size
    af = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())


# ---------- Main optimisation entry ----------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----- initial vector (use parent if present) -----
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 2000
        a = np.exp(-np.arange(n_fallback, dtype=np.float64) / 500.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a / s
    else:
        a = np.full_like(a, 1.0 / a.size)

    n = a.size
    # FFT length sufficient for linear convolutions up to length 3n-2
    L = 1 << ((3 * n - 2).bit_length())

    cur = a.copy()
    cur_sc = _score(cur, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ----- hyper‑parameters -----
