def gradient_descent(a_init, L, deadline):
   cur = a_init.copy()
   best_vec = cur.copy()
   best_sc = _score(cur, L)
   prev = cur.copy()
   momentum = np.zeros_like(cur)
   beta = 0.9
   # hyperparams
   step0 = 1.0
   eps0 = 1.0
   eps_decay=0.9995
   eps_min=1e-12

   it=0
   no_improve=0
   while time.monotonic() < deadline:
        it+=1
        # compute conv and max for gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()   # Not used directly maybe for smoothing
        # smoothing weight:
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val)/eps)
        w_sum = w.sum()
        if w_sum ==0:
            w = np.full_like(w, 1.0/w.size)
        else:
            w /= w_sum
        # gradient: weight reversed conv with a (like previously)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # extract gradient of length n as per earlier idx
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # momentum update:
        momentum = beta*momentum + (1 - beta)*g
        # step size: maybe step0 / sqrt(it)
        step = step0 / math.sqrt(it)
        cand = cur - step * momentum
        # project onto simplex
        cand = _project_simplex(cand)
        cand_sc = _score(cand, L)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            no_improve = 0
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
        else:
            # backtrack
            # try reducing step
            step_bt = step
            improved=False
            for _ in range(5):
                step_bt *= 0.5
                cand = cur - step_bt * momentum
                cand = _project_simplex(cand)
                cand_sc = _score(cand, L)
                if cand_sc < cur_sc:
                    cur, cur_sc = cand, cand_sc
                    improved=True
                    break
            if not improved:
                no_improve+=1
        # random perturb when stuck:
        if no_improve > 2000:

# ---------- Simplex projection ----------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

        consider(a)

    # ----- combined families -----
    for r in (0.95, 0.98, 0.99):
        for p in (0.8, 1.2, 1.6):
            a = (np.power(r, np.arange(n, dtype=np.float64)) /
                 np.power(np.arange(1, n + 1, dtype=np.float64), p))
            consider(a)

    # Fallback if nothing succeeded
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)
        best_sc = _score(best_vec, fft_len=L)

    # ----- gradient‑descent refinement (projected) -----
    def refine(start_vec: np.ndarray, limit: float):
        nonlocal best_vec, best_sc
        cur = start_vec.copy()
        cur_sc = _score(cur, fft_len=L)
        momentum = np.zeros_like(cur)
        beta = 0.9
        step0 = 1.0
        eps0 = 1.0
        eps_decay = 0.9995
        eps_min = 1e-12
        it = 0
        no_improve = 0

        while time.monotonic() < limit:
            it += 1

            a_fft = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
            max_val = conv.max()

            eps = max(eps_min, eps0 * (eps_decay ** it))
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum

            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]

            # momentum update
            momentum = beta * momentum + (1.0 - beta) * g

            step = step0 / math.sqrt(it)
            cand = cur - step * momentum
            cand = _project_simplex(cand)
            cand_sc = _score(cand, fft_len=L)

            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0

            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                continue

            # backtrack if no improvement
            improved = False
            step_bt = step
            for _ in range(5):
                step_bt *= 0.5
                cand = cur - step_bt * momentum
                cand = _project_simplex(cand)
                cand_sc = _score(cand, fft_len=L)
                if cand_sc < cur_sc:
                    cur, cur_sc = cand, cand_sc
                    improved = True
                    break
            if not improved:
                no_improve += 1

            # random shake when stuck
            if no_improve > 2000:
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    delta = cur[i] * rng.random() * 0.5
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp, fft_len=L)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                no_improve = 0

            # occasional symmetrisation
            if it % 1000 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym, fft_len=L)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()

    # allocate half of the remaining time for refinement
    now = time.monotonic()
    time_for_refine = deadline - now - 0.1
    if time_for_refine > 0:
        refine(best_vec, deadline - 0.05)

    # ----- final random fine‑tuning -----
    while time.monotonic() < deadline - 0.05:
        cand = best_vec + rng.normal(scale=0.002, size=n)
        cand = _project_simplex(cand)
        cand_sc = _score(cand, fft_len=L)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()

    # ----- return as plain list -----
    return [float(x) for x in best_vec]
