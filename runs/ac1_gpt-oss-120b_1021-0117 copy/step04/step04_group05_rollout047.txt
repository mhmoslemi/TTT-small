# ------------------------------------------------------------
# Project onto the probability simplex (sum == 1, elements >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
        else:
            # occasional simulated‑annealing acceptance
            prob = math.exp(-(cand_score - cur_score) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_fft, cur_conv, cur_max, cur_score = \
                    cand, cand_fft, cand_conv, cand_max, cand_score
                stagnation = 0
            else:
                stagnation += 1

        # ---------- escape from stagnation ----------
        if stagnation > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta_mass = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta_mass
                tmp[j] += delta_mass
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)

                tmp_fft = np.fft.rfft(tmp, n=L_fft)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L_fft)[: 2 * n - 1]
                tmp_score = 2.0 * n * tmp_conv.max()
                if tmp_score < cur_score:
                    cur, cur_fft, cur_conv, cur_max, cur_score = \
                        tmp, tmp_fft, tmp_conv, tmp_conv.max(), tmp_score
                if tmp_score < best_score:
                    best_score = tmp_score
                    best_vec = tmp.copy()
            stagnation = 0

        # ---------- decay smoothing ----------
        eps_rel = max(eps_rel_min, eps_rel * decay_eps)

        # ---------- early stop ----------
        if best_score < 1.50274349:
            break

    # -----------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # -----------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    return [float(x) for x in best_vec]
