    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + float(budget_s) - 0.1

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a = np.exp(-np.arange(n, dtype=np.float64) / 20.0)

    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total <= 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / total

    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())

    # projection function defined outside.

    # compute initial score using evaluate_sequence
    best = a.copy()
    best_score = float('inf')
    try:
        best_score = evaluate_sequence(best.tolist())
    except Exception:
        best_score = float('inf')
    # set cur to best
    cur = best.copy()
    cur_score = best_score

    # Hyperparameters

    eps_decay = 0.9995
    eps_min = 1e-8
    lr = 0.08
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    it = 0
    no_improve = 0
    target = 1.50274349

    # Main loop
    while time.monotonic() < deadline:
        it += 1
        # compute convolution of cur
        cur_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv  # sum=1

        # direction of gradient via smoothed max
        eps_cur = max(eps_min, eps0 * (eps_decay ** it))

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
