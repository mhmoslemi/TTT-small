import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum x = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast max‑convolution score (assumes sum == 1)
# ------------------------------------------------------------
def _max_conv_score(a: np.ndarray, L: int, n: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        a0 = np.exp(-np.arange(n0) / 200.0)

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)          # sum == 1, non‑negative
    n = a0.shape[0]
    L = 1 << ((3 * n - 2).bit_length())   # FFT length for all convolutions

    # ----- quick parametric sweep for a good start -------------------

    best_sc = _max_conv_score(best, L, n)

    uni = _project_simplex(uni)

    # exponential families
