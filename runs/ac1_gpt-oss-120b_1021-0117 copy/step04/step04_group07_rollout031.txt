def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math, random
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2) # cushion
    
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2500
        a = np.full(n, 1.0/n)
    # ensure nonnegative and clip
    a = np.maximum(a, 0.0)
    a = a / a.sum()  # normalize sum=1
    n = a.size
    # FFT length
    L = 1 << ((2*n - 2).bit_length())
    # parameters
    eps = 0.01
    eps_min = 1e-7
    eps_decay = 0.9995
    lr = 0.4   # base step
    beta = 0.9
    momentum = np.zeros_like(a)
    best = a.copy()
    best_sc = _fast_score(a, L, n)
    cur = a.copy()
    cur_sc = best_sc
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        # compute conv via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        maxc = conv.max()
        # Softmax gradient
        if eps > eps_min:
            w = np.exp((conv - maxc) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2*n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]
            eps = max(eps_min, eps * eps_decay)
        else:
            k = int(conv.argmax())
            g = np.zeros_like(cur)
            start_idx = max(0, k - (n - 1))
            end_idx = min(k + 1, n)
            # contributions
            idx = np.arange(start_idx, end_idx)
            g[idx] = 2.0 * cur[k - idx]
        # momentum update
        momentum = beta * momentum + (1 - beta) * g
        # step update with diminishing lr
        alpha = lr / (1 + 0.001 * it) / (np.linalg.norm(momentum) + 1e-12)
        cand = cur - alpha * momentum
        cand = _project_simplex(cand)
        cand_sc = _fast_score(cand, L, n)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # occasional random acceptance to escape
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc
        # symmetrize every N steps
        if it % 3000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, L, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0
        # occasional random mass transfer when stuck
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _fast_score(tmp, L, n)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0
        # reset no_improve if improvement
        if cur_sc < best_sc * (1 - 1e-6):
            no_improve = 0
    return [float(x) for x in best]

import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 bound for a vector that already satisfies sum=1
# ------------------------------------------------------------
def _fast_score(a: np.ndarray, L: int, n: int) -> float:
    """Score = 2 * n * max(conv) assuming sum(a)=1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Analytic starter: coefficients of (1 - z)^(-1/2)  (binomial 2k choose k / 4^k)
# ------------------------------------------------------------
def _analytic_start(n: int) -> np.ndarray:
    """Return the normalised binomial‑series start vector of length n."""
    b = np.empty(n, dtype=np.float64)
    b[0] = 1.0
    for i in range(1, n):
        # recurrence b_i = b_{i-1} * (2*i-1) / (2*i)
        b[i] = b[i - 1] * (2 * i - 1) / (2 * i)
    b /= b.sum()                     # project onto simplex (sum = 1)
    return b

    else:

    n = a0.size

    # --------------------------------------------------------

    cur = a0.copy()

    # --------------------------------------------------------

    while time.monotonic() < deadline:
