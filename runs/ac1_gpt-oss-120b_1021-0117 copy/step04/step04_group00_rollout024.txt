import time
import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low AC1 bound.
    The routine works with a multiplicative‑gradient descent on the simplex.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- initialise -------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: modest random start
        cur = rng.random(2000, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    if s == 0.0:
        cur = np.full_like(cur, 1.0 / cur.size)
    else:
        cur = cur / s                     # L1‑normalised (sum = 1)

    n = cur.size
    # FFT length = next power of two >= 3n‑2 (covers a (len n) * w (len 2n‑1))
    L = 1 << ((3 * n - 2).bit_length())

    # ---- bookkeeping ------------------------------------------------------
    best_vec = cur.copy()
    # compute initial bound (sum = 1 → bound = 2·n·max(conv))
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    cur_score = 2.0 * n * conv.max()
    best_score = cur_score

    # ---- optimisation parameters -------------------------------------------
    step0 = 0.5               # base step for multiplicative update
    eps_rel0 = 0.2            # relative smoothing factor (w.r.t. max)
    eps_decay = 0.9997
    eps_min = 1e-12
    warmup = 3000             # iterations using soft‑max weights
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution & score ---------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_vec = cur.copy()
            if best_score < 1.50274349:               # target reached
                break

        # ----- weight vector (smooth → indicator) -------------------------
        if it <= warmup:
            eps = max(eps_min, max_val * eps_rel0 * (eps_decay ** it))
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)
        else:
            # hard max indicator (all indices touching the maximum)
            tol = max_val * 1e-12 + 1e-15
            idx_max = np.where(conv >= max_val - tol)[0]
            if idx_max.size == 0:
                idx_max = np.array([conv.argmax()], dtype=int)
            w = np.zeros_like(conv)
            w[idx_max] = 1.0 / idx_max.size

        # ----- gradient of the smoothed max --------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)          # length ≥ 3n‑2
        idx = (2 * n - 2) - np.arange(n)                     # reverse slice
        g = 2.0 * grad_full[idx]                             # sub‑gradient

        # ----- multiplicative update with back‑tracking --------------------
        eta = step0 / math.sqrt(it)
        for _ in range(5):
            cand = cur * np.exp(-eta * g)
            cand_sum = cand.sum()
            if cand_sum == 0.0:
                cand = np.full_like(cand, 1.0 / n)
            else:
                cand = cand / cand_sum
            # evaluate candidate
            cand_fft = np.fft.rfft(cand, n=L)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
            cand_score = 2.0 * n * cand_conv.max()
            if cand_score <= cur_score:
                cur = cand
                cur_score = cand_score
                break
            eta *= 0.5                      # shrink step if not improving

        # ----- occasional symmetrisation -----------------------------------
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            cur = sym / sym.sum()

        # ----- occasional random mass‑transfer (escape) --------------------
        if it % 5000 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.2
                cur[i] -= delta
                cur[j] += delta
                cur = cur / cur.sum()

    # -----------------------------------------------------------------------
    # Return the best vector discovered (as plain Python list)
    # -----------------------------------------------------------------------
    return [float(x) for x in best_vec]
