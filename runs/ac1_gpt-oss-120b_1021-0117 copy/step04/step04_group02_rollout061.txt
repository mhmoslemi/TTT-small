import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    # find last index where condition holds
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score (2*n*max(convolution)) for a vector with sum≈1
# ------------------------------------------------------------
def _score_and_conv(x: np.ndarray, L: int):
    """Return the AC1 bound (2*n*max(conv)) and the full convolution."""
    n = x.shape[0]
    af = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max()), conv


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())        # FFT length
