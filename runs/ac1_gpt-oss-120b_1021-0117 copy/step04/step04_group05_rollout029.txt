import time

# ----------------------------------------------------------------------
# Helper: convolution max via FFT
# ----------------------------------------------------------------------
def _conv_max(vec: np.ndarray) -> float:
    """Maximum entry of the linear convolution of vec with itself."""
    n = vec.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    vfft = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(vfft * vfft, n=L)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Helper: objective used by the evaluator (no clamping)
# ----------------------------------------------------------------------
def _objective(vec: np.ndarray) -> float:
    """Exact AC1 bound for a given coefficient vector."""
    s = vec.sum()
    if s < 0.01:
        return float('inf')
    n = vec.shape[0]
    mx = _conv_max(vec)
    return float(2.0 * n * mx / (s * s))


# ----------------------------------------------------------------------
# Helper: build the sparse convolution matrix A where (A @ b)[k] = (f * b)[k]
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Return sparse matrix A (shape (2n‑1, n)) with rows = shifted copies of f."""
    n = f.shape[0]
    # each column i contains f[0..n‑1] starting at row i
    offsets = np.arange(n, dtype=np.int64)
    data = np.tile(f, (n, 1))                     # shape (n, n)
    return sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
