def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s
    else:
        n0 = 2000
        t = np.arange(n0, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()

    n = a0.size
    L = 1 << ((2*n - 1).bit_length())
    cur = a0.copy()
    v = np.zeros_like(cur)

    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # parameters
    base_alpha = 1.0
    beta = 0.9
    eps_factor = 0.05   # initial epsilon = eps_factor * max_val
    eps_decay = 0.9995
    epsilon = None
    sym_interval = 300
    improve_target = 1.50274349
    no_improve = 0
    it = 0
    # stagnation thresholds
    restart_thresh = 3000

    while time.monotonic() < deadline:
        it += 1

        # convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n - 1]
        max_val = conv.max()

        # epsilon schedule
        if epsilon is None:
            epsilon = max(eps_factor * max_val, 1e-12)
        else:
            epsilon = max(epsilon * eps_decay, 1e-12)

        # softmax weights (smooth max)
        w = np.exp((conv - max_val) / epsilon)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient approximation
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]
        grad_norm = np.linalg.norm(grad) + 1e-12

        # step size
        alpha = base_alpha / (grad_norm * math.sqrt(it))

        # Nesterov-like update
        v = beta * v - alpha * grad
        cand = cur + v
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        improved = False
        if cand_sc < cur_sc:
            improved = True
        else:
            # reject and damp velocity
            v = np.zeros_like(v)

        if improved:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            no_improve += 1

        # symmetrisation
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # random mass transfer if stuck
        if no_improve > restart_thresh:
            # generate a small random perturbation of current vector, preserve sum
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # early exit
        if best_sc < improve_target:
            break

    # Return best candidate as list
    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast score for a normalized sequence (sum = 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv) assuming sum(a) == 1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
