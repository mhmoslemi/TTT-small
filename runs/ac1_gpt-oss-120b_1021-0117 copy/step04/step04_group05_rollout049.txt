import time, math, numpy as np

def _project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(cur_fft_grad * w_rev_fft, n=L_grad)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ---- back‑tracking line search ---------------------------------
        alpha = step0 / math.sqrt(it)
        accepted = False
        cand = None
        cand_sc = None
        for _ in range(8):
            cand = cur - alpha * g
            cand = _project_simplex(cand)
            cand_fft = np.fft.rfft(cand, n=L_conv)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L_conv)[: 2 * n - 1]
            cand_sc = 2.0 * n * cand_conv.max()
            if cand_sc < cur_sc:
                accepted = True
                break
            alpha *= 0.6

        if accepted:
            cur = cand
            cur_fft_grad = np.fft.rfft(cur, n=L_grad)
            cur_fft_conv = cand_fft
            cur_conv = cand_conv
            cur_max = cand_conv.max()
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            # occasional simulated‑annealing acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_fft_grad = np.fft.rfft(cur, n=L_grad)
                cur_fft_conv = cand_fft
                cur_conv = cand_conv
                cur_max = cand_conv.max()
                cur_sc = cand_sc
                no_improve = 0
            else:
                no_improve += 1

        # ---- periodic symmetrisation -----------------------------------
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L_conv)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L_conv)[: 2 * n - 1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < cur_sc:
                cur = sym
                cur_fft_grad = np.fft.rfft(cur, n=L_grad)
                cur_fft_conv = sym_fft
                cur_conv = sym_conv
                cur_max = sym_conv.max()
                cur_sc = sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()

        # ---- random mass transfer when stuck ----------------------------
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L_conv)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L_conv)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_fft_grad = np.fft.rfft(cur, n=L_grad)
                    cur_fft_conv = tmp_fft
                    cur_conv = tmp_conv
                    cur_max = tmp_conv.max()
                    cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                no_improve = 0

        if best_sc < target:
            break

    # ================================================================
    # Fine‑tuning: cheap random moves until time runs out
    # ================================================================
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if best_vec[i] == 0.0:
            continue
        delta = best_vec[i] * rng.random() * 0.2
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_fft = np.fft.rfft(cand, n=L_conv)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L_conv)[: 2 * n - 1]
        cand_sc = 2.0 * n * cand_conv.max()
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()

    return [float(x) for x in best_vec]
