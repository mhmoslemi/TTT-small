        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # slice to length‑n gradient
        g = 2.0 * grad_full[idx]

        # momentum‑augmented step
        momentum = beta * momentum + (1.0 - beta) * g
        grad_norm = np.linalg.norm(momentum) + 1e-12
        alpha = step0 / (math.sqrt(it) * grad_norm)

        cand = cur - alpha * momentum
        cand = _project_simplex(cand)

        cand_sc = _score(cand, L)

        # accept if better, otherwise occasional Metropolis step
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # periodic symmetrisation (helps with autocorrelation)
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # random mass‑transfer when stuck for a while
        if no_improve > mass_transfer_thresh:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                # reset momentum after a shake
                momentum.fill(0.0)
                no_improve = 0

        # early exit if we have reached the target bound
        if best_sc < target:
            break

    # --------------------------------------------------------
    # final local refinement with tiny random jitter
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        tmp = best.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp, L)
        if tmp_sc < best_sc:
            best, best_sc = tmp, tmp_sc
            continue
        # occasional symmetry check
        sym = (tmp + tmp[::-1]) * 0.5
        sym = _project_simplex(sym)
        sym_sc = _score(sym, L)
        if sym_sc < best_sc:
            best, best_sc = sym, sym_sc

    # return plain Python list
    return [float(x) for x in best]
