# ------------------------------------------------------------
# Project onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # reverse slice
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (base_lr / (math.sqrt(it) + 1e-12)) * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = a - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # ----- keep the best seen so far -----
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()

        # ----- accept/reject (simple downhill with occasional uphill) -----
        if cand_sc <= cur_sc:
            a, cur_sc = cand, cand_sc
        else:
            # Metropolis‑style acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (base_lr + 1e-12))
            if rng.random() < prob:
                a, cur_sc = cand, cand_sc

        # ----- occasional random mass‑transfer to escape plateaus -----
        if it % 2000 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = a[i] * rng.random() * 0.3
            tmp = a.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < best_sc:
                best_sc = tmp_sc
                best_vec = tmp.copy()
                a, cur_sc = tmp, tmp_sc
            elif tmp_sc < cur_sc:
                a, cur_sc = tmp, tmp_sc

        # ----- early exit if target already reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
