import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    # based on algorithm by Wang & Carreira-Perpinan (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for final conversion
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialise sequence (normalize to sum == 1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: exponential decay, length 1024
        n_fallback = 1024
        x = np.arange(n_fallback, dtype=np.float64)
        cur = np.exp(-x / (n_fallback / 10.0))
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # sum == 1
    n = cur.shape[0]

    # ------------------------------------------------------------------
    # pre‑compute FFT length (power of two >= 3n-2)
    # ------------------------------------------------------------------
    L_full = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    best_vec = cur.copy()
    # compute initial score
    cur_fft = np.fft.rfft(cur, n=L_full)
    cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
    cur_max = cur_conv.max()
    cur_score = float(2.0 * n * cur_max)
    best_score = cur_score

    # ------------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------------
    # gradient smoothing parameters
    eps_rel_initial = 0.12
    eps_rel_min = 0.002
    eps_decay = 0.99995          # slow exponential decay per iteration
    eps_min = 1e-12

    # step size (initial magnitude) and schedule
    step0 = 2.0                  # larger initial step
    # no‑improvement counter
    no_improve = 0

    # momentum (heavy‑ball)
    mu = 0.7                     # momentum coefficient
    velocity = np.zeros_like(cur)

    # random‑perturbation settings
    perturb_thresh = 5000
    perturb_scale = 0.02        # relative stddev of Gaussian noise

    # symmetrisation interval
    sym_interval = 2500

    it = 0

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # recompute convolution & max (already have from previous iteration,
        #   but we need fresh after any update)
        # ------------------------------------------------------------------
        cur_fft = np.fft.rfft(cur, n=L_full)
        cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
        cur_max = cur_conv.max()
        cur_score = float(2.0 * n * cur_max)

        # ------------------------------------------------------------------
        # smoothed‑max weights
        # ------------------------------------------------------------------
        eps_rel = max(eps_rel_min, eps_rel_initial * (eps_decay ** it))
        eps = max(eps_min, cur_max * eps_rel)
        # exponentials are safe because (conv - cur_max) <= 0
        w = np.exp((cur_conv - cur_max) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ------------------------------------------------------------------
        # gradient (full linear convolution of cur with reversed weights)
        # ------------------------------------------------------------------
        w_rev_fft = np.fft.rfft(w[::-1], n=L_full)
        grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L_full)
        idx = (2 * n - 2) - np.arange(n)          # slice corresponding to each a_i
        grad = 2.0 * grad_full[idx]

        # ------------------------------------------------------------------
        # momentum‑augmented step with back‑tracking line search
        # ------------------------------------------------------------------
        # update velocity (heavy‑ball)
        velocity = mu * velocity - grad   # direction before scaling

        # compute trial step magnitude
        alpha = step0 / math.sqrt(it)   # decreasing magnitude

            cand = _project_simplex(cand)

            cand_fft = np.fft.rfft(cand, n=L_full)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L_full)[: 2 * n - 1]
            cand_max = cand_conv.max()
            cand_score = float(2.0 * n * cand_max)

            if cand_score < cur_score:

                break

            cur_max = cand_max
            # keep velocity as is (already updated)

            if cand_score < best_score:
                best_score = cand_score

                # early exit if meet target

            else:

        # ------------------------------------------------------------------

            sym_max = sym_conv.max()
            sym_score = float(2.0 * n * sym_max)
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                cur_max = sym_max
                # reset momentum to avoid drift
                velocity = np.zeros_like(cur)
                no_improve = 0
                if sym_score < best_score:
                    best_score = sym_score
                    best_vec = sym.copy()
                    if best_score < 1.50274349:
                        break

        # ------------------------------------------------------------------
        # random perturbation when stuck
        # ------------------------------------------------------------------
        if no_improve > perturb_thresh:
            # jitter around the best solution found so far
            noise = rng.normal(scale=perturb_scale, size=n)
            tmp = best_vec + noise
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            # reset momentum
            velocity = np.zeros_like(tmp)
            cur = tmp
            # reset counter
            no_improve = 0
            # recompute score for the new point
            cur_fft = np.fft.rfft(cur, n=L_full)
            cur_conv = np.fft.irfft(cur_fft * cur_fft, n=L_full)[: 2 * n - 1]
            cur_max = cur_conv.max()
            cur_score = float(2.0 * n * cur_max)
            if cur_score < best_score:
                best_score = cur_score
                best_vec = cur.copy()
                if best_score < 1.50274349:
                    break

    # ------------------------------------------------------------------
    # Return best candidate as simple list of floats
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
