# ---------- Simplex projection ----------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v >= 0)."""
    n = v.size
    # shift by mean to improve numerical stability
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, fft_len=L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < 1.50274349:
            break

    # ----- final fine‑tuning while time permits -----
    while time.monotonic() < deadline - 0.05:
        delta = rng.normal(scale=0.001, size=n)
        cand = best_vec + delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand, fft_len=L)
        if cand_sc < best_sc:
            best_vec, best_sc = cand, cand_sc

    # ----- return result as plain Python list -----
    return [float(x) for x in best_vec]
