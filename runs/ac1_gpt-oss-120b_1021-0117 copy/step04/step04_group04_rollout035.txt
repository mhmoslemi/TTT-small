def refine_lp(a, max_val, top_k=200):
    n = a.shape[0]
    # compute conv if needed
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2*n-1]
    # choose top_k indices where conv close to max
    idx = np.argpartition(-conv, top_k)[:top_k]
    # Build sparse matrix
    rows = []
    cols = []
    data = []
    for row_i, k in enumerate(idx):
        # for j in 0..n-1: if 0 <= k - j < n: coefficient = a[k - j]
        start_j = max(0, k - (n - 1))
        end_j = min(n-1, k)
        # For j ranging start_j..end_j, coefficient = a[k-j]
        # that's same as a slice reversed.
        # Could fill iteratively
        for j in range(start_j, end_j+1):
            coeff = a[k - j]
            rows.append(row_i)
            cols.append(j)
            data.append(coeff)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(top_k, n))
    # Define variable
    import cvxpy as cp
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= max_val]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)
    if g.value is None:
        return None
    else:
        g0 = np.maximum(g.value, 0.0)
        return g0

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalise to guard against tiny numerical drift
    s = w.sum()
    if s > 0:
        w /= s
    return w


# ------------------------------------------------------------
# Score = 2 * n * max(convolution)  (assumes sum == 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    # length sufficient for linear convolution of a with itself
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    m = np.zeros(n, dtype=np.float64)
