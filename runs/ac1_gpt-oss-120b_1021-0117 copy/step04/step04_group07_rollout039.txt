# ------------------------------------------------------------
# Project a vector onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

            if rng.random() < math.exp(-(cand_sc - cur_sc) / (1.0 + np.mean(np.abs(step)))):
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ------ epsilon decay --------------------------------
        eps = max(eps_min, eps * eps_decay)

        # ------ periodic symmetrisation (helps) -------------
        if it % 5000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_sc = _score_from_max(sym_conv.max(), sym.sum(), n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                    no_improve = 0

        # ------ random mass‑transfer when stuck -------------
        if no_improve > 8000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_sc = _score_from_max(tmp_conv.max(), tmp.sum(), n)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ------ early exit if target reached -----------------
        if best_sc < 1.50274349:
            break

    # -----------------------------------------------------------------
    # Return the best vector found (as plain Python list of floats)
    # -----------------------------------------------------------------
    return [float(x) for x in best]
