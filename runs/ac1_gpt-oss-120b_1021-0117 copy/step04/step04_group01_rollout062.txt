import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# ------------------------------------------------------------
# Score used during optimisation (proportional to the true score)
def _score(a: np.ndarray, fft_len: int) -> float:
    """Return 2 * n * max(convolution(a,a)).  Uses FFT of length `fft_len`."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.4)   # safety margin

    # -----------------------------------------------------------------
    # initialise a (simplex‑normalised)
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # quick geometric fall‑back
        n_fallback = 1500
        r = 0.995
        a = np.power(r, np.arange(n_fallback), dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s                     # normalise to sum = 1

    n = a.size
    # FFT length sufficient for 3‑n convolution (gradient step)
    L = 1 << ((3 * n - 2).bit_length())

    # -----------------------------------------------------------------
    # bookkeeping
    cur = a.copy()
    cur_fft = np.fft.rfft(cur, n=L)
    cur_score = _score(cur, fft_len=L)
    best_vec = cur.copy()
    best_score = cur_score

    # hyper‑parameters
    step0 = 0.8                   # base step magnitude
    max_backtrack = 15
    backtrack_factor = 0.6
    no_improve = 0
    it = 0
    target = 1.50274349

    # -----------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current iterate -----
        cur_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- active set (indices attaining the maximum) -----
        # use a tiny absolute tolerance to collect ties
        active = np.where(conv_full >= max_val - 1e-12)[0]
        w = np.zeros_like(conv_full)
        if active.size:
            w[active] = 1.0 / active.size
        else:                               # should never happen
            w[:] = 1.0 / w.size

        # ----- gradient of the max (sub‑gradient) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)     # map to gradient entries
        g = 2.0 * grad_full[idx]

        # ----- line search -----

    return [float(x) for x in best_vec]
