import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 score for a vector with sum == 1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(convolution) for a vector already normalized to sum=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # -----------------------------------------------------------------
    # Initialise a feasible point on the simplex
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s
    else:
        # fallback: exponential decay, length 2500
        n_fallback = 2500
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 120.0)
        a0 = a0 / a0.sum()

    n = a0.size
    # FFT length for full convolution (2n-1) – a power of two
    L = 1 << ((2 * n - 1).bit_length())

    cur = a0.copy()
    cur_sc = _score(cur, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------
    base_alpha = 0.8          # base step size
    momentum = 0.9            # momentum coefficient
    sym_interval = 200        # how often to symmetrise
    stagnate_limit = 3000    # when to perform a random jiggle
    improve_target = 1.50274349

    it = 0
    no_improve = 0
    prev_cur = cur.copy()     # used for momentum

    while time.monotonic() < deadline:
        it += 1

        # ---- convolution of current iterate (FFT) ----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ---- gradient: focus on all arg‑max positions ----
        max_idxs = np.where(conv >= max_val - 1e-12)[0]
        g = np.zeros_like(cur)
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(n, k + 1)
            j = np.arange(lo, hi)
            a_idx = k - j
            g[lo:hi] += 2.0 * cur[a_idx]

        # ---- momentum‑augmented direction ----
        diff = cur - prev_cur
        direction = g + momentum * diff
        dnorm = np.linalg.norm(direction) + 1e-12

        # ---- step size (decaying) ----
        alpha = base_alpha / (math.sqrt(it) * dnorm)

        if best_sc < improve_target:
            break

    # Return the best sequence found (as plain Python floats)
    return [float(x) for x in best_vec]
