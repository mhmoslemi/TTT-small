    # ensures v[0] >= v[1] >= ... >= v[-1]
    # Use PAVA on reversed ascending
    w = v.copy()
    n = len(w)

    while i < n:
        level.append([w[i], 1])  # sum, count

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.5
    if "height_sequence_1" in globals():

        a = np.maximum(a,0)
    else:
        n = 2000
        a = np.exp(-np.arange(n)/100.0)
    a = a/ a.sum()
    n = a.size
    L = 1 << ((3*n-2).bit_length())
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps_adam=1e-8
    step0=0.5
    best_vec = a.copy()
    best_sc = _score(a)
    it=0
    while time.monotonic() < deadline:
        it+=1
        # compute conv, gradient
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        # smoothing epsilon
        eps = max_val * 0.15 + 1e-12
        # compute softmax weights
        w = np.exp((conv - max_val)/eps)
        w_sum = w.sum()
        if w_sum==0:
            w = np.full_like(conv, 1/conv.size)
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = step0 / (np.sqrt(v_hat) + eps_adam)
        cand = a - step * m_hat
        # project onto simplex
        cand = _project_simplex(cand)
        # evaluate
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
        a = cand
        # optional symmetrization periodically
        if it % 200 == 0:
            sym = (a + a[::-1])*0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:

                best_vec = sym.copy()

import time, math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
