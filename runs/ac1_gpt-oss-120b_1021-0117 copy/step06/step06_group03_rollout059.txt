        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        if cur[i] <= 0.0:
            continue
        delta = cur[i] * rng.random() * 0.5       # move up to 50 % of mass at i
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluate – the function is cheap (FFT‑based inside the evaluator)
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score - 1e-12:
            best_score = sc
            best = cand.copy()
            cur = cand.copy()
        # simple acceptance rule to keep the walk alive
        elif rng.random() < 0.05:
            cur = cand.copy()

        # the theoretical optimum is 2.0; stop early if we reach it
        if best_score <= 2.0 + 1e-12:
            break

    # -----------------------------------------------------------------
    # return the best candidate found (as plain Python floats)
    # -----------------------------------------------------------------
    return [float(x) for x in best]
```
