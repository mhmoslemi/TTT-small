def _softmax_weights(conv, alpha):
    max_val = conv.max()
    # shift
    shifted = conv - max_val
    # compute exp(alpha * shifted)
    # scaled shifted may still overflow for large alpha; but values <=0 so exp <=1; safe.
    exp_vals = np.exp(alpha * shifted)
    sum_exp = exp_vals.sum()
    if sum_exp == 0:
        # fallback: uniform weights
        return np.full_like(conv, 1.0/conv.size)
    return exp_vals / sum_exp

Because shifted <= 0, exp(alpha*shifted) in (0,1], no overflow.

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Score assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
