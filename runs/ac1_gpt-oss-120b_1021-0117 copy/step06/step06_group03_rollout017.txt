def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np

    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    rng = np.random.default_rng(seed)

    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            n = a0.size
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / s
    else:
        n = 2000
        a0 = np.exp(-np.arange(n) / 100.0)
        a0 = a0 / a0.sum()

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())
    # Precompute FFT size

    def score_vec(v):
        # Compute 2*n*max(conv) where sum(v)=1
        af = np.fft.rfft(v, n=L)
        conv_full = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv_full.max())

    # Determine initial score
    cur = a0.copy()
    cur_sc = score_vec(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 1.0  # base learning rate

    # smoothing temperature schedule
    eps_temp = None
    eps_decay = 0.9995  # slower decay

    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        iteration += 1

        # compute convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()

        if eps_temp is None:
            eps_temp = max_val * 0.2
        else:
            eps_temp = max(eps_temp * eps_decay, 1e-15)

        # softmax weights
        w = np.exp((conv - max_val) / eps_temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        # project onto simplex (nonnegative and sum=1)
        # use simple renorm after clipping negatives
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s <= 0.0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / s

        cand_sc = score_vec(cand)

        # accept if improvement
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # update best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

        # occasional symmetrization
        if iteration % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = np.maximum(sym, 0.0)
            s = sym.sum()
            if s > 0:
                sym = sym / s
            sym_sc = score_vec(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best_vec = cur.copy()
                no_improve = 0

        # random mass transfer when stuck
        if no_improve > 800:
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n-1)
                if j >= i: j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = np.maximum(tmp, 0.0)
                s = tmp.sum()
                if s > 0:
                    tmp = tmp / s
                tmp_sc = score_vec(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    no_improve = 0
                else:
                    no_improve = max(0, no_improve - 200)

        # restart from best if too long no improvement
        if no_improve > 3000:
            # add small random noise to best
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            s = cur.sum()
            if s > 0:
                cur = cur / s
            m.fill(0.0)
            v.fill(0.0)
            eps_temp = None
            no_improve = 0

        # early exit if we meet target
        if best_sc < 1.50274349:
            break

    # Convert best_vec to list of Python floats
    return [float(x) for x in best_vec]

    conv = np.fft.irfft(af * af, n=L)[: 2 * vec.shape[0] - 1]
    return float(2.0 * vec.shape[0] * conv.max())
