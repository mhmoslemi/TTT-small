# Helper: Euclidean projection onto the simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative entries with sum Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]                # descending sort
    css = np.cumsum(u) - Z
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Helper: max of full linear convolution a*a using FFT
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max_k (a*a)[k] using an FFT of length L (>=2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# ------------------------------------------------------------
# Helper: sub‑gradient approximated by the top‑k convolution entries
# ------------------------------------------------------------
def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    n = a.size
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            i = np.arange(i_start, i_end + 1, dtype=int)
            g[i] += 2.0 * a[k - i]
    g /= max(1, top_k)
    return g

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise base sequence
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        base = rng.random(2048, dtype=np.float64)
    base = np.maximum(base, 0.0)
    base = _project_simplex(base)                     # sum = 1

    n = base.size
    L = 1 << ((2 * n - 1).bit_length())               # next power of two >= 2n‑1

    # --------------------------------------------------------
    # Population for Differential Evolution
    # --------------------------------------------------------
    pop_size = 40 if n <= 3000 else 30
    pop = np.empty((pop_size, n), dtype=np.float64)
    for i in range(pop_size):
        # Dirichlet‑like random point (exponential entries, then project)
        v = rng.exponential(scale=1.0, size=n)
        pop[i] = _project_simplex(v)
    # ensure the known parent is present
    pop[0] = base.copy()

    # --------------------------------------------------------
    # Scoring helper (sum is 1, so denominator cancels)
    # --------------------------------------------------------
    def _score_vec(vec):
        return 2.0 * n * _conv_max(vec, L)

    scores = np.empty(pop_size, dtype=np.float64)
    for i in range(pop_size):
        scores[i] = _score_vec(pop[i])

    best_idx = int(np.argmin(scores))
    best = pop[best_idx].copy()
    best_score = scores[best_idx]

    # --------------------------------------------------------
    # DE parameters
    # --------------------------------------------------------
    F = 0.8
    CR = 0.9
    gen = 0
    target = 1.50274349

    # --------------------------------------------------------
    # Differential Evolution loop
    # --------------------------------------------------------
    while time.monotonic() < deadline - 5.0:  # reserve time for refinement
        gen += 1
        for i in range(pop_size):
            # pick two distinct donors (different from i)
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b = rng.choice(idxs, size=2, replace=False)
            # mutation: best + F * (pop[a] - pop[b])
            mutant = best + F * (pop[a] - pop[b])
            # binomial crossover
            cross = rng.random(n) < CR
            if not cross.any():
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])
            # project back onto simplex
            trial = _project_simplex(trial)
            # evaluate
            trial_score = _score_vec(trial)
            # selection
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best = trial.copy()
                    best_score = trial_score
        # occasional symmetrisation of the current best
        if gen % 20 == 0:
            sym = (best + best[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_vec(sym)
            if sym_score < best_score:
                best = sym
                best_score = sym_score
        # early stop if target already reached
        if best_score < target:
            break

    # --------------------------------------------------------
    # Local refinement: projected SGD with top‑k gradient
    # --------------------------------------------------------
    refine_deadline = deadline
    if time.monotonic() < refine_deadline - 0.1:
        momentum = np.zeros_like(best)
        top_k = 12
        step0 = 0.6
        beta = 0.9
        it = 0
        cur_score = best_score
        while time.monotonic() < refine_deadline:
            it += 1
            # convolution of current best
            a_fft = np.fft.rfft(best, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
            # gradient approximation
            g = _grad_topk(best, conv_full, top_k)
            momentum = beta * momentum + (1.0 - beta) * g
            alpha = step0 / math.sqrt(it)
            cand = best - alpha * momentum
            cand = _project_simplex(cand)
            cand_score = _score_vec(cand)
            if cand_score < cur_score:
                best = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
            # periodic symmetrisation
            if it % 500 == 0:
                sym = (best + best[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_score = _score_vec(sym)
                if sym_score < cur_score:
                    best = sym
                    cur_score = sym_score
                    if sym_score < best_score:
                        best_score = sym_score

    # --------------------------------------------------------
    # Return the best found sequence as a plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best]
