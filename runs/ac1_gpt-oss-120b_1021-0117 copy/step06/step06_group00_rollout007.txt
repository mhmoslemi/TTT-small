def _score(v):
    """Score assuming sum(v)=1 (same as evaluate_sequence up to scaling)."""
    a = np.asarray(v, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # small safety margin

    # ------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random positive vector of length ~2500
        cur = rng.random(2500, dtype=np.float64)

    cur = _project_simplex(cur)                # enforce simplex
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # optimizer state (Adam)
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.2

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc
    it = 0
    no_improve = 0

    # control parameters
    sym_interval = 400
    restart_interval = 3000
    target = 1.50274349

    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- convolution & max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- sub‑gradient of the max -----
        tol = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - tol
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(mask, dtype=np.float64)
        else:
            w = mask.astype(np.float64) / cnt

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step)
        cand_sc = _score(cand)

        # ----- accept / reject -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            # simple simulated annealing
            temp = max(0.1, 1.0 / math.sqrt(it))
            prob = math.exp(-(cand_sc - cur_sc) / temp)
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # ----- random restart when stuck -----
        if no_improve > restart_interval:
            # perturb best solution slightly and re‑project
            delta = rng.normal(scale=0.02, size=n)
            tmp = best_vec + delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return best vector as list of plain Python floats
    return [float(x) for x in best_vec]
