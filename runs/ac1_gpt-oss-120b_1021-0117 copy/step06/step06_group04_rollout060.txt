import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_array(a: np.ndarray, L: int):
    """Return max convolution value and the full convolution (length 2n‑1)."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    return float(conv.max()), conv


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s
    else:
        n0 = 2000
        p = 1.2
        a0 = (np.arange(1, n0 + 1, dtype=np.float64) ** -p)
        a0 = a0 / a0.sum()

    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # FFT length

    cur = a0.copy()
    cur_max, cur_conv = _conv_max_and_array(cur, L)
    cur_sc = 2.0 * n * cur_max / (cur.sum() ** 2)

    best_vec = cur.copy()
    best_sc = cur_sc

    # ---- optimisation parameters ----------------------------------------
    beta = 0.85
    base_alpha = 0.42
    max_backtrack = 12
    sym_interval = 180
    transfer_interval = 800
    reinit_interval = 3000
    target_score = 1.50274349

    momentum = np.zeros_like(cur)
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # ---- sub‑gradient of the max convolution -----------------------
        max_val = cur_max
        tol = max_val * 1e-12 + 1e-16
        idxs = np.where(cur_conv >= max_val - tol)[0]
        grad = np.zeros_like(cur)
        for idx in idxs:
            shift = idx - (n - 1)
            if shift >= 0:
                grad[shift:] += 2.0 * cur[: n - shift]
            else:
                grad[: n + shift] += 2.0 * cur[-shift :]
        if len(idxs) > 1:
            grad /= len(idxs)

        # ---- momentum and adaptive step -------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / (math.sqrt(it) * norm_m)

        # ---- line‑search ---------------------------------------------
        improved = False
        cand = None
        cand_sc = None
        alpha_try = alpha
        for _ in range(max_backtrack):
            cand_try = cur - alpha_try * momentum
            cand_try = _project_simplex(cand_try)
            max_try, conv_try = _conv_max_and_array(cand_try, L)
            sc_try = 2.0 * n * max_try / (cand_try.sum() ** 2)
            if sc_try < cur_sc:
                cand, cand_sc = cand_try, sc_try
                cur_max, cur_conv = max_try, conv_try
                improved = True
                break
            alpha_try *= 0.5

        # ---- occasional uphill move ------------------------------------
        if not improved:
            temp = 1.0 / math.sqrt(it + 1.0)
            cand_try = cur - alpha_try * momentum
            cand_try = _project_simplex(cand_try)
            max_try, conv_try = _conv_max_and_array(cand_try, L)
            sc_try = 2.0 * n * max_try / (cand_try.sum() ** 2)
            prob = math.exp(-(sc_try - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                cand, cand_sc = cand_try, sc_try
                cur_max, cur_conv = max_try, conv_try
                improved = True

        if improved:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---- periodic symmetrisation ---------------------------------
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            max_sym, conv_sym = _conv_max_and_array(sym, L)
            sc_sym = 2.0 * n * max_sym / (sym.sum() ** 2)
            if sc_sym < cur_sc:
                cur, cur_sc = sym, sc_sym
                cur_max, cur_conv = max_sym, conv_sym
                if sc_sym < best_sc:
                    best_sc, best_vec = sc_sym, sym.copy()
                no_improve = 0

        # ---- random mass transfer when stagnating ---------------------
        if no_improve > transfer_interval:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                max_tmp, conv_tmp = _conv_max_and_array(tmp, L)
                sc_tmp = 2.0 * n * max_tmp / (tmp.sum() ** 2)
                if sc_tmp < cur_sc:
                    cur, cur_sc = tmp, sc_tmp
                    cur_max, cur_conv = max_tmp, conv_tmp
                    if sc_tmp < best_sc:
                        best_sc, best_vec = sc_tmp, tmp.copy()
                    no_improve = 0
                else:
                    no_improve = max(0, no_improve - 250)

        # ---- restart from best with jitter after long stall ------------
        if no_improve > reinit_interval:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            no_improve = 0

        # ---- early exit ------------------------------------------------
        if best_sc < target_score:
            break

    return [float(x) for x in best_vec]
