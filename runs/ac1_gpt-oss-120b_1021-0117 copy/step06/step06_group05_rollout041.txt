import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Exact AC1 bound (assumes non‑negative entries)
# ------------------------------------------------------------
def _exact_score(a: np.ndarray) -> float:
    """Compute 2*n*max(convolution)/sum(a)^2 exactly."""
    n = a.shape[0]
    b = np.convolve(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # reserve a small margin for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------

    # --------------------------------------------------------

        a0 = np.maximum(a0, 0.0)

        else:
            a0 = a0 / s                     # now sum == 1
    else:                                  # fallback: uniform of length 2000
        n_fallback = 2000
        a0 = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    cur = a0.copy()
