import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a) = 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length() + 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence whose AC1 bound is low."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----- initialise a feasible point on the simplex -----
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s
    else:
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 120.0)
        a0 = a0 / a0.sum()

    # start from a symmetric version (helps the bound)
    a0 = (a0 + a0[::-1]) * 0.5
    a0 = a0 / a0.sum()

    n = a0.shape[0]
    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ----- FFT helper sizes -----
    L = 1 << ((2 * n - 1).bit_length() + 1)   # power‑of‑two for speed
    grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)

    # ----- optimiser state -----
    momentum = np.zeros_like(cur)
    beta = 0.98
    base_alpha = 0.8
    eps = None
    eps_decay = 0.999
    min_eps = 1e-12

    sym_interval = 100
    improve_target = 1.50274349
    no_improve = 0
    it = 0
    max_trials = 12

    while True:
        it += 1

        # ----- convolution & smoothed max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # epsilon for softmax smoothing
        if eps is None:
            eps = max(max_val * 0.5, 1e-12)
        else:
            eps = max(eps * eps_decay, min_eps)

        # softmax weights over the convolution entries
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[grad_idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- step size (decaying) -----
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / math.sqrt(it) / norm_m

        improved = False
        cand_best = None
        cand_best_sc = None

        # ----- try several step lengths (back‑tracking) -----
        alpha = alpha0
        for _ in range(max_trials):
            # SGD style candidate
            cand_sgd = cur - alpha * momentum
            cand_sgd = _project_simplex(cand_sgd)
            sc_sgd = _score(cand_sgd)

            # Exponential‑gradient style candidate
            eta = alpha * 0.5
            g_clip = np.clip(g, -20.0, 20.0)          # avoid overflow
            cand_exp = cur * np.exp(-eta * g_clip)
            cand_exp = cand_exp / cand_exp.sum()
            sc_exp = _score(cand_exp)

            # pick the better of the two
            if sc_sgd < sc_exp:
                cand_try, sc_try = cand_sgd, sc_sgd
            else:
                cand_try, sc_try = cand_exp, sc_exp

            if sc_try < cur_sc:
                cand_best, cand_best_sc = cand_try, sc_try
                improved = True
                break
            alpha *= 0.5

        # ----- occasional uphill acceptance -----
        if not improved:
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            sc_try = _score(cand_try)
            prob = math.exp(-(sc_try - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand_best, cand_best_sc = cand_try, sc_try
                improved = True

        # ----- accept / reject -----
        if improved and cand_best_sc is not None:
            cur = cand_best
            cur_sc = cand_best_sc
            if cand_best_sc < best_sc:
                best_sc = cand_best_sc
                best_vec = cand_best.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- aggressive random mass‑transfer when stuck -----
        if no_improve > 600:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    if rng.random() < 0.05:
                        cur = tmp
                        cur_sc = tmp_sc
                        no_improve = 0

        # ----- restart from best with small noise if hopeless -----
        if no_improve > 2500:
            cur = best_vec + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum.fill(0.0)
            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < improve_target:
            break

        # ----- periodic time check -----
        if it % 200 == 0 and time.monotonic() >= deadline:
            break

    # return a plain Python list
    return [float(x) for x in best_vec]
