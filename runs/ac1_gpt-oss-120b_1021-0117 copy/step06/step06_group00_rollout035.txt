def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    if 'height_sequence_1' in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        init = rng.random(2000, dtype=np.float64)

    cur = _project_simplex(np.maximum(init, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # best
    cur_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    cur_sc = 2.0 * n * conv.max()
    best_vec = cur.copy()
    best_sc = cur_sc

    # optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    lr0 = 0.5  # initial learning rate

    no_improve = 0
    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        t += 1
        # compute conv and gradient with softmax
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2 * n - 1]
        max_val = conv.max()
        # softmax weights
        # subtract max for stability
        exp_arg = 0.5 * (conv - max_val)  # alpha = 0.5 initially maybe adjust later
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr0 / (np.sqrt(v_hat) + eps)
        update = step * m_hat

        cand = _project_simplex(cur - update)
        # compute score
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[:2 * n - 1]
        cand_sc = 2.0 * n * cand_conv.max()

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:

            # occasional uphill acceptance based on temperature
            temp = max(0.01, 0.5 / math.sqrt(t))
            if rng.random() < math.exp(-(cand_sc - cur_sc)/temp):

            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[:2 * n - 1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < cur_sc:

                if sym_sc < best_sc:

            j = rng.integers(0, n)

                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                # evaluate
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[:2 * n - 1]

                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        if best_sc < target:
            break

    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, non‑negative)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical cleanup – guarantees exact sum=1
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    return w

def _score_vec(a, fft_len):
    """Exact evaluator‑compatible score (2·n·max(conv)).  Assumes sum(a)==1."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2 * n - 1]
    return 2.0 * n * conv.max()
