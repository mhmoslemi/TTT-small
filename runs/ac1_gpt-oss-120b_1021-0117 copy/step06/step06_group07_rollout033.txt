def _conv_fft(a, L):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*a.size-1]
    return conv

def _conv_max_and_grad(a, L):
    conv = _conv_fft(a, L)
    max_val = conv.max()

        grad[i] += a[j]
        grad[j] += a[i]
    return max_val, grad

import time, math
import numpy as np
import scipy.sparse as sp

# ------------------------------------------------------------
# Simplex projection (non‑negative, sum = Z)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Convolution via FFT (full linear)
# ------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv

# ------------------------------------------------------------
# Compute max of convolution and its sub‑gradient
# ------------------------------------------------------------
def _conv_max_and_grad(a: np.ndarray, L: int):
    conv = _conv_fft(a, L)
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    idxs = np.where(conv >= max_val - eps)[0]
    n = a.size
    grad = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        grad[i] += a[j]
        grad[j] += a[i]
    return max_val, grad

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise from parent if available, else random

    a = np.maximum(a, 0.0)

        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length >= 2n‑1

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    max_val, _ = _conv_max_and_grad(a, L)
    cur_sc = 2.0 * n * max_val
    best_vec = a.copy()
    best_sc = cur_sc

    # optimisation parameters
    step0 = 0.8
    beta = 0.90
    momentum = np.zeros_like(a)
    sym_interval = 200
    no_improve = 0
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:

        # current max & gradient
        max_val, grad = _conv_max_and_grad(a, L)

        # -------- gradient descent with momentum ----------

        momentum = beta * momentum + (1.0 - beta) * direction
