import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

                        best_sc, best_vec = sym_sc, sym.copy()

            # ----- random redistribution when stuck -----
            if no_improve > 1500:
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    delta = cur[i] * rng.random() * 0.7
                    j = rng.integers(0, n)
                    if j == i:
                        continue
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp)
                    if tmp_sc < cur_sc:
                        prev = cur
                        cur = tmp
                        cur_sc = tmp_sc
                        no_improve = 0
                        if tmp_sc < best_sc:
                            best_sc, best_vec = tmp_sc, tmp.copy()

            # ----- early stop if target reached -----
            if best_sc < target:
                return [float(x) for x in best_vec]

    # -------------------- final local refinement --------------------
    # simple random pair transfers on the best vector found
    if best_vec is None:
        # fallback (should never happen)
        best_vec = init0.copy()
        best_sc = _score(best_vec)

    cur = best_vec.copy()
    cur_sc = best_sc
    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        delta = cur[i] * rng.random() * 0.5
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
            if best_sc < target:
                break

    return [float(x) for x in best_vec]
