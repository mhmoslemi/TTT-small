def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence whose sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _build_active_set(conv: np.ndarray, k: int) -> np.ndarray:
    """Indices of the k largest entries of a convolution array."""
    if k >= conv.size:
        return np.arange(conv.size, dtype=int)
    part = np.argpartition(-conv, k - 1)[:k]
    return np.sort(part)


def _make_constraint_matrix(f: np.ndarray, idx: np.ndarray) -> csr_matrix:
    """
    Build a sparse matrix A such that (A @ g)[r] = (f * g)[idx[r]].
    Row r corresponds to convolution index idx[r].
    """
    n = f.size
    rows, cols, data = [], [], []
    for r, k in enumerate(idx):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        inds = np.arange(i_start, i_end + 1, dtype=int)
        coeffs = f[k - inds]                     # f_i where i = inds
        rows.extend([r] * inds.size)
        cols.extend(inds.tolist())
        data.extend(coeffs.tolist())
    return csr_matrix((data, (rows, cols)), shape=(idx.size, n))
