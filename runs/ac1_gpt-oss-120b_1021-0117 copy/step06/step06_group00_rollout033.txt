    # a is sum=1 vector.
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)

    max_val = conv.max()

    w = mask.astype(np.float64)

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2500
        a0 = rng.random(n, dtype=np.float64)
    n = a0.size

    # Project to simplex
    cur = _project_simplex(np.maximum(a0, 0.0))
    best_vec = cur.copy()
    best_sc = _score(cur)

    # shape candidates
    alphas = [0.3,0.4,0.5,0.6,0.7,0.8,0.9,1.0,1.2,1.5,2.0,3.0,5.0]
    # U-shape exponents p
    p_vals = [0.5,1.0,2.0,3.0,4.0,6.0]

    # generate candidates
    x = (np.arange(n) + 0.5) / n
    for a in alphas:
        if time.monotonic() >= deadline:
            break
        # symmetric Beta shape: alpha=beta=a
        # compute log densities to avoid overflow
        # Use clip
        eps = 1e-12
        x_clipped = np.clip(x, eps, 1-eps)
        loga = (a-1)*np.log(x_clipped) + (a-1)*np.log(1 - x_clipped)
        a_vec = np.exp(loga)
        a_vec = a_vec / a_vec.sum()
        sc = _score(a_vec)
        if sc < best_sc:
            best_sc, best_vec = sc, a_vec.copy()
    # U-shape
    mid = (n-1)/2.0
    idx_arr = np.arange(n)
    dist = np.abs(idx_arr - mid) / mid
    for p in p_vals:
        if time.monotonic() >= deadline:
            break
        a_vec = (1 - dist) ** p
        a_vec = a_vec / a_vec.sum()
        sc = _score(a_vec)
        if sc < best_sc:
            best_sc, best_vec = sc, a_vec.copy()
    # Gradient descent refine
    cur = best_vec.copy()
    cur_sc = best_sc
    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.8
    it = 0
    no_improve = 0
    target = 1.50274349
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1
        grad = _grad(cur)
        momentum = beta * momentum + (1 - beta) * grad
        step = step0 / (it ** 0.6)   # maybe better exponent
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:

            # occasional acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:

        # symmetrization every 400 iters

            if sym_sc < cur_sc:

import time, math, numpy as np

# ----- helpers ---------------------------------------------------------

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# globals that will be set once n is known
_GLOBAL_L = None


def _score(a):
    """Objective 2*n*max(conv) for a vector that already sums to 1."""
    global _GLOBAL_L
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = _GLOBAL_L if _GLOBAL_L is not None else (1 << ((2 * n - 1).bit_length()))
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _grad(a):
    """Sub‑gradient of the max‑convolution (scaled for our objective)."""
    global _GLOBAL_L
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = _GLOBAL_L if _GLOBAL_L is not None else (1 << ((2 * n - 1).bit_length()))
    a_fft = np.fft.rfft(a, n=L)

    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = conv >= max_val - eps
    w = mask.astype(np.float64)
    cnt = w.sum()
    if cnt > 0:
        w /= cnt

    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad
