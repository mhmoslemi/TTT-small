def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv_full.max())

                pert[i] -= delta
                pert[j] += delta
                pert = _project_simplex(pert)
                pert_sc = _score_fft(pert, L)
                if pert_sc < cur_sc:
                    cand, cand_sc = pert, pert_sc
                    improved = True

        # ----- accept candidate -----
        if improved:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----- restart from best if hopelessly stuck -----
        if no_improve > 3000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            eps_temp = None
            no_improve = 0

        # ----- early exit if target met -----
        if best_sc < improve_target:
            break

        # ----- make sure we stop before hard timeout -----
        if time.monotonic() >= deadline:
            break

    # Return the best sequence as plain Python floats
    return [float(x) for x in best_vec]
