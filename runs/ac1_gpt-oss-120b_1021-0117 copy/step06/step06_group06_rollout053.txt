def propose_candidate(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # initialize
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_init = 2500
        a0 = np.exp(-np.arange(n_init, dtype=np.float64) / 100.0)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0:
        a0 = np.full_like(a0, 1.0 / a0.size)
        s = 1.0
    a0 = a0 / s

    n = a0.size
    L = 1 << ((2*n - 1).bit_length())
    a = a0.copy()
    # enforce symmetry to start
    a = (a + a[::-1]) * 0.5

    # Adam params
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.2
    it = 0
    best_vec = a.copy()
    best_sc = _score(a)
    # smoothing temperature
    T = None
    T_decay = 0.97
    no_improve = 0
    sym_interval = 400
    # start main loop
    while time.monotonic() < deadline:
        it += 1

        # FFT of a
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2*n - 1]
        max_val = conv.max()
        # update temperature
        if T is None:
            T = max_val * 0.15 + 1e-12
        else:
            # ensure T not too small
            T = max(T * T_decay, max_val * 1e-12)

        # compute softmax weighting
        # For stability, subtract max_val before exp
        w = np.exp((conv - max_val) / T)
        w_sum = w.sum()
        if w_sum <= 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # Learning rate schedule
        lr = lr0 / np.sqrt(it)

        step = lr * m / (np.sqrt(v) + eps_adam)

        a_candidate = a - step
        # project onto simplex
        a_candidate = _project_simplex(a_candidate)

        # optional symmetry enforcement
        if it % sym_interval == 0:
            a_candidate = (a_candidate + a_candidate[::-1]) * 0.5
            # projection to ensure numerical sum maintained
            a_candidate = _project_simplex(a_candidate)

        cur_sc = _score(a_candidate)

        # Accept if improvement else keep current? We'll accept either but keep best.
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a_candidate.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Use candidate as new a (even if worse) to continue descent
        a = a_candidate

        # Random perturb if stuck
        if no_improve > 2000:
            # add small noise
            noise = rng.normal(scale=0.05, size=n)
            a = a + noise
            a = np.clip(a, 0, None)
            a = _project_simplex(a)
            a = (a + a[::-1]) * 0.5
            no_improve = 0
            # reset Adam moments
            m.fill(0.0)
            v.fill(0.0)
            T = None

        # early exit if target reached
        if best_sc < 1.50274349:
            break

    # Final refinement: try a few random swaps
    while time.monotonic() < deadline - 0.05:
        i = rng.integers(0, n)
        if best_vec[i] <= 0:
            continue
        delta = best_vec[i] * rng.random() * 0.03
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand

    # Return list
    return [float(x) for x in best_vec]

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (L1 = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


# ------------------------------------------------------------
# AC1 upper bound for a vector whose sum is (approximately) 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    # keep a tiny margin for the final return

    if s <= 0.0:

    n = a0.size

    # enforce symmetry from the start (helps the autocorrelation)
    a = (a0 + a0[::-1]) * 0.5

    # ------------------------------------------------------------
