def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # init vector
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        cur = rng.random(n, dtype=np.float64)
    # ensure nonnegative
    cur = np.maximum(cur, 0.0)
    # project to simplex sum=1
    cur = _project_simplex(cur)
    n = cur.size
    # compute L
    L = 1 << ((2 * n - 1).bit_length())
    # initialize best
    best_vec = cur.copy()
    best_sc = _score_normed(best_vec)
    cur_sc = best_sc

    # optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5  # adjust
    it = 0
    no_improve = 0
    target = 1.50274349

    # set softmax temperature k
    k = 10.0

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # compute convolution
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n - 1]  # length 2n-1
        max_val = conv_full.max()

        # compute softmax weights
        w = np.exp(k * (conv_full - max_val))
        ws = w.sum()
        if ws == 0:
            w = np.zeros_like(w)
        else:
            w /= ws

        # gradient via FFT
        # w_rev = reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # pick appropriate slice
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        # bias correction
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = base_lr / np.sqrt(it)  # step decays

        # candidate with Adam direction
        cand = cur - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project to simplex
        cand = _project_simplex(cand)

        cand_sc = _score_normed(cand)

        # accept if better, else try smaller step
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # try halved step size up to 3 times
            improved = False
            for factor in (0.5, 0.25, 0.125):
                cand2 = cur - lr * factor * m_hat / (np.sqrt(v_hat) + eps)
                cand2 = _project_simplex(cand2)
                cand2_sc = _score_normed(cand2)
                if cand2_sc < cur_sc:
                    cur, cur_sc = cand2, cand2_sc
                    no_improve = 0
                    improved = True
                    break
            if not improved:
                # optional simulated annealing acceptance
                no_improve += 1
                # acceptance probability
                prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
                if rng.random() < prob:
                    cur, cur_sc = cand, cand_sc

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w[:] = 1.0 / n
    return w

def _score_normed(a: np.ndarray, L: int) -> float:
    """Score for a vector already normalized to sum=1: 2*n*max(conv)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
