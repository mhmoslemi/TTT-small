# Projection onto the simplex {x >= 0, sum(x) = 1}

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        if best_sc < target:
            break

        # ----- exact sub‑gradient of the max‑convolution -----
        tol = 1e-12
        mask = (np.abs(conv - cur_max) <= tol).astype(np.float64)   # shape (2n‑1,)
        rev_a = a[::-1]
        rev_fft = np.fft.rfft(rev_a, n=L)
        mask_fft = np.fft.rfft(mask, n=L)
        grad_full = np.fft.irfft(rev_fft * mask_fft, n=L)          # length L
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]                  # exact sub‑gradient

        # ----- projected momentum step -----
        momentum = beta * momentum + (1.0 - beta) * grad
        alpha = step0 / math.sqrt(it)
        a_tmp = a - alpha * momentum
        a_tmp = _project_simplex(a_tmp)

        # ----- evaluate candidate -----
        a_tmp_fft = np.fft.rfft(a_tmp, n=L)
        conv_tmp = np.fft.irfft(a_tmp_fft * a_tmp_fft, n=L)[: 2 * n - 1]
        cand_sc = 2.0 * n * conv_tmp.max()

        # ----- accept improvement -----
        if cand_sc < cur_sc:
            a = a_tmp
            cur_sc = cand_sc
            no_improve = 0
        else:
            # occasional random mass‑transfer when stuck
            if no_improve > 1200:
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                if a[i] > 0.0:
                    delta = a[i] * rng.random() * 0.5
                    a[i] -= delta
                    a[j] += delta
                    a = a / a.sum()
                no_improve = 0

        # ----- periodic symmetrisation (helps for many instances) -----
        if it % 250 == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            a_sym_fft = np.fft.rfft(a_sym, n=L)
            conv_sym = np.fft.irfft(a_sym_fft * a_sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * conv_sym.max()
            if sym_sc < cur_sc:
                a = a_sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    no_improve = 0

        # ----- safety break if we run out of time -----
        if time.monotonic() > deadline - 0.2:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
