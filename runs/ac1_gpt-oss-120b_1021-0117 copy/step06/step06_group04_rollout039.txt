import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise to avoid tiny numerical drift
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w[:] = 1.0 / n
    return w


# ----------------------------------------------------------------------
# Helper: convolution via FFT, returning max value, its index, and full conv
# ----------------------------------------------------------------------
def _conv_max_and_index(a: np.ndarray, L: int):
    """Return (max value, index of max, full convolution a*a)."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv
