def _score(a):
    n = a.shape[0]

    af = np.fft.rfft(a, n=L)

def _gradient_sub(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    # compute conv
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    max_val = conv.max()
    # find indices where conv close to max (within tolerance)
    # tolerance: e.g., 1e-12 relative
    idxs = np.where(np.abs(conv - max_val) < 1e-12)[0]  # may be few
    # create weight w of length 2n-1
    w = np.zeros_like(conv)
    if idxs.size > 0:

    # compute gradient via convolution with w_rev

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (no assumption on the sum)
# ------------------------------------------------------------
def _score_vec(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))


# ------------------------------------------------------------
# Gradient (sub‑gradient of the max‑convolution)
# ------------------------------------------------------------
def _subgradient(a: np.ndarray, L: int):
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    # weight 1 over all maximising indices (exact sub‑gradient)
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)
    cnt = w.sum()
    if cnt > 0:
        w /= cnt
    # gradient = 2 * (a * w_rev) (proper alignment)
    w_rev_fft = np.fft.rfft(w[::-1], n=L)
    grad_full = np.fft.irfft(af * w_rev_fft, n=L)[: 2 * n - 1]
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g, max_val

    n = a0.shape[0]

    best_vec = a0.copy()
