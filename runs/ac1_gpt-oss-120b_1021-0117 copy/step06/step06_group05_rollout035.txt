import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------

    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # initialise a feasible point on the simplex
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        cur = a0 / s                                   # now sum = 1
    else:                                               # fallback
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        cur = np.exp(-t / 100.0)
        cur = cur / cur.sum()

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())                # FFT length (power of two)

    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------
    it = 0
    no_improve = 0
    momentum = np.zeros_like(cur)

    eps = None
    eps_factor = 0.15          # initial smoothing as fraction of max
    eps_decay = 0.9995
    base_alpha = 1.0
    beta = 0.85                # momentum
    sym_interval = 200
    rand_move_prob = 0.02
    restart_thresh = 4000

    # temperature for occasional uphill moves (simulated annealing)
    T = cur_sc * 0.1
    T_decay = 0.9995

    # --------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & soft‑max weight -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps is None:
            eps = max(max_val * eps_factor, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max (via FFT) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- step size (decaying) -----
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / math.sqrt(it) / norm_m

        # ----- additive projected‑gradient move -----
        improved = False
        cand = None
        cand_sc = None
        alpha = alpha0
        for _ in range(10):
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5

        # ----- multiplicative (exponentiated) move (occasionally) -----
        if not improved and rng.random() < 0.12:
            mult_step = alpha0
            cand_try = cur * np.exp(-mult_step * g)
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True

        # ----- simulated‑annealing style random transfer -----
        if not improved:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                cand_try = cur.copy()
                cand_try[i] -= delta
                cand_try[j] += delta
                cand_try = _project_simplex(cand_try)
                cand_sc = _score(cand_try)
                # Metropolis acceptance
                prob = math.exp(-(cand_sc - cur_sc) / (T + 1e-12))
                if rng.random() < prob:
                    cand = cand_try
                    improved = True

        # ----- accept if we have a candidate -----
        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation (helps) -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- occasional random move (exploration) -----
        if rng.random() < rand_move_prob:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0

        # ----- restart from best if hopelessly stuck -----
        if no_improve > restart_thresh:
            cur = best_vec + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            eps = None
            no_improve = 0

        # ----- annealing temperature decay -----
        T = max(T * T_decay, 1e-12)

        # ----- early exit if target reached -----
        if best_sc < 1.50274349:
            break

    # -----------------------------------------------------------------
    # ensure we return something sensible even if optimisation failed
    # -----------------------------------------------------------------
    if best_sc == float("inf") or not np.isfinite(best_sc):
        # simple exponential fallback
        t = np.arange(n, dtype=np.float64)
        best_vec = np.exp(-t / 100.0)
        best_vec = best_vec / best_vec.sum()

    # return as plain list of Python floats
    return [float(x) for x in best_vec]
