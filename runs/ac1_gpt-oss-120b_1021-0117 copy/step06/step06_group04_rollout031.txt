def propose_candidate(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)

    # init a
    if 'height_sequence_1' in globals():

        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
    else:
        n = 2000
        a0 = np.ones(n) / n

    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    def max_conv(x):
        af = np.fft.rfft(x, L)
        conv = np.fft.irfft(af * af, L)[:2*n-1]
        return float(conv.max())

    cur = a0.copy()
    cur_max = max_conv(cur)
    cur_score = 2.0 * n * cur_max # sum=1

    best = cur.copy()
    best_score = cur_score

    # SA parameters
    T0 = 0.01
    T = T0
    iteration = 0
    max_iter = 2000000  # huge, but time bound will stop earlier
    # schedule: T = T0 * (1 - iter / max_iter)
    sym_every = 500
    next_sym = sym_every
    # for random mass move frequency
    while time.monotonic() < deadline:
        iteration += 1
        # temperature schedule
        T = T0 * (1.0 - iteration / max_iter)
        if T < 1e-6:
            T = 1e-6

        # propose move
        i = rng.integers(n)
        # ensure i has at least some mass
        if cur[i] <= 0.0:

        j = rng.integers(n-1)
        if j >= i:

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_idx(a: np.ndarray, n: int, L: int):
    """Return (max value, index) of the auto‑convolution a*a."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * n - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx
