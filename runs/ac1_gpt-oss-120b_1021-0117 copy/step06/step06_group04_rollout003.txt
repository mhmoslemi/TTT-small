    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., budget_s - 0.2)

        a0 = np.maximum(a0, 0.0)
        s = a0.sum()

            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
    else:
        # generate parameterized initial sequence
        n_fallback = 2500
        exponent = rng.uniform(0, 2)
        a0 = np.arange(1, n_fallback+1, dtype=np.float64) ** (-exponent)
        s = a0.sum()
    a0 = a0 / a0.sum()  # simplex

    n = a0.shape[0]
    L = 1 << ((2*n-1).bit_length())

    def project_simplex(v):

    def conv_max(a):
        af = np.fft.rfft(a, L)
        conv = np.fft.irfft(af * af, L)[:2*n-1]
        idx = int(conv.argmax())
        maxv = float(conv[idx])
        return maxv, idx, conv

    def score_from_max(maxv, sum_a):
        return 2.0 * n * maxv / (sum_a*sum_a)

    # compute initial score
    maxv, idx, conv = conv_max(a0)
    cur = a0
    cur_score = score_from_max(maxv, cur.sum())
    best_vec = cur.copy()
    best_score = cur_score

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0

    # main loop
    while time.monotonic() < deadline:
        # compute gradient at current max index
        maxv, idx, _ = conv_max(cur)
        # gradient sub: g[i] = 2 * cur[idx-i] if valid
        g = np.zeros_like(cur)
        # compute indexes
        # use vectorized:
        j = idx - np.arange(n)
        mask = (j >= 0) & (j < n)
        g[mask] = 2.0 * cur[j[mask]]
        # Adam update
        t_iter += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t_iter)
        v_hat = v / (1 - beta2**t_iter)

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_idx(a: np.ndarray, L: int):
    """Full convolution a*a via FFT, returns (max, idx, full_conv)."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv
