def _grad(a, eps):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    max_val = conv.max()
    if eps is None:
        eps = max_val * 0.15 + 1e-12
    else:
        eps = max(eps * 0.97, max_val * 1e-12)
    exp_weights = np.exp((conv - max_val) / eps)
    w_sum = exp_weights.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = exp_weights / w_sum
    # gradient as per earlier: grad_full = ifft( a_fft * fft(w_rev) )
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, eps, max_val

def _grad_score(a, eps):

    L = 1 << ((2*n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    max_val = conv.max()
    cur_score = 2.0 * n * max_val
    # set epsilon
    if eps is None:
        eps_new = max_val * 0.15 + 1e-12

    idx = (2*n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, eps_new, cur_score

import time
import math
import numpy as np

# ------------------------------------------------------------
# Project a vector onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a safety margin for the final return
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---------- initial feasible vector (simplex) ----------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:                                   # fallback: exponential decay
        n_fallback = 2000
        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 100.0)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
        s = 1.0
    a0 = a0 / s                         # sum = 1

    n = a0.size
    # FFT length (next power of two >= 2*n-1)
    L = 1 << ((2 * n - 1).bit_length())

    # ---------- optimisation hyper‑parameters ----------
    a = a0.copy()
    eps = None                     # smoothing temperature
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.05                     # base learning rate
    t_iter = 0

    # keep the best solution seen
    # initial score
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    cur_score = 2.0 * n * conv.max()
    best_vec = a.copy()
    best_score = cur_score

    # control variables
    no_improve = 0
    target = 1.50274349
    sym_interval = 800
    restart_patience = 5000

    # ---------- main optimisation loop ----------
    while time.monotonic() < deadline:
        t_iter += 1

        # ----- compute convolution and gradient (matrix‑free) -----
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # temperature update
        if eps is None:
            eps = max_val * 0.15 + 1e-12
        else:
            eps = max(eps * 0.97, max_val * 1e-12)

        # soft‑max (or hard‑max later) weights for the max index
        if eps < max_val * 1e-6 or t_iter > 2500:
            mask = (conv == max_val)
            w = mask.astype(np.float64)
            w_sum = w.sum()
            if w_sum > 0.0:
                w /= w_sum
            else:
                w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum

        # gradient of the smoothed max
        w_rev = w[::-1]

        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        cand = a - step
        cand = _project_simplex(cand)

        # evaluate candidate (one extra FFT)
        cand_fft = np.fft.rfft(cand, n=L)
        conv_cand = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]

        if cand_score < cur_score:
            a = cand

            improved = True

            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
        else:
            no_improve += 1

        # ---------- periodic symmetrisation ----------
        if t_iter % sym_interval == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            conv_sym = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_score = 2.0 * n * conv_sym.max()
            if sym_score < cur_score:
                a = sym
                cur_score = sym_score

                    best_score = sym_score
                    best_vec = sym.copy()
                no_improve = 0

        if no_improve > 1500:
            i = rng.integers(0, n)
            if a[i] > 0.0:

                j = rng.integers(0, n - 1)

                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                conv_tmp = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_score = 2.0 * n * conv_tmp.max()
                if tmp_score < cur_score:
                    a = tmp
                    cur_score = tmp_score
                    if tmp_score < best_score:
                        best_score = tmp_score
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve += 1

        # ---------- restart from best if hopeless ----------
        if no_improve > restart_patience:
            a = best_vec + 0.01 * rng.random(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            m.fill(0.0)
            v.fill(0.0)
            eps = None
            t_iter = 0
            no_improve = 0

        # early exit if target reached
        if best_score < target:
            break

        # safety time check
        if time.monotonic() >= deadline:
            break

    # ---------- final cheap random refinements ----------
    while time.monotonic() < deadline - 0.02:
        i = rng.integers(0, n)
        if best_vec[i] <= 0.0:
            continue
        delta = best_vec[i] * rng.random() * 0.04
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_fft = np.fft.rfft(cand, n=L)
        conv_cand = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_score = 2.0 * n * conv_cand.max()
        if cand_score < best_score:
            best_score = cand_score
            best_vec = cand.copy()

    # return as plain Python list
    return [float(x) for x in best_vec]
