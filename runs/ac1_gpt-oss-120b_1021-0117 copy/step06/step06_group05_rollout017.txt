import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2 * n * max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    cur = a0.copy()
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ----------------------- optimisation state -----------------------
    momentum = np.zeros_like(cur)
    beta = 0.9                # momentum factor
    base_alpha = 0.7
    sym_interval = 500        # how often to symmetrise
    random_move_prob = 0.02
    no_improve = 0
    restart_thresh = 4000
    improve_target = 1.50274349

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_sc = 2.0 * n * max_val

        # ----- sub‑gradient of the max (using exact arg‑max set) -----
        mask = np.isclose(conv_full, max_val, atol=1e-12, rtol=1e-12)
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt == 0:
            w[:] = 1.0
            cnt = w.size
        w = w / cnt
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- step size (simple decay) -----
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / math.sqrt(it) / norm_m

        # ----- backtracking line‑search (up to 5 trials) -----
        improved = False
        cand = None
        cand_sc = None
        for _ in range(5):
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5

        # ----- occasional uphill move (simulated‑annealing style) -----
        if not improved:
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation (helps a lot) -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc

        # ----- occasional random move (mass transfer) -----
        if rng.random() < random_move_prob:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc

        # ----- restart if hopelessly stuck -----
        if no_improve > restart_thresh:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < improve_target:
            break

    # ----------------------- final local refinement -----------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if cur[i] > 0.0:
            delta = cur[i] * rng.random() * 0.2
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < best_sc:
                best_sc = tmp_sc
                best_vec = tmp.copy()
                cur = tmp
            elif tmp_sc < cur_sc:
                cur = tmp

    # return as plain list of floats
    return [float(x) for x in best_vec]
