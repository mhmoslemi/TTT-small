# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length() + 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Initial feasible point on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s
    else:  # fallback: exponential decay
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 150.0)
        a0 = a0 / a0.sum()

    n = a0.shape[0]

    # convolution helper dimensions
    L = 1 << ((2 * n - 1).bit_length() + 1)      # power of two ≥ 2n‑1
    grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)

    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------
    # Adam optimizer parameters (projected onto simplex each step)
    # ------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.25
    adam_eps = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t_iter = 0

    # other control parameters
    no_improve = 0
    eps_factor = 0.5          # initial smoothing factor
    eps_decay = 0.9995        # decay per iteration
    min_eps = 1e-12
    sym_interval = 100
    target_score = 1.50274349
    max_no_improve_restart = 5000

    while time.monotonic() < deadline:
        t_iter += 1

        # --------------------------------------------------------
        # convolution and smoothed max weights
        # --------------------------------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        eps = max(eps_factor * max_val, min_eps)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------
        # gradient of smoothed max (via FFT)
        # --------------------------------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[grad_idx]

        # --------------------------------------------------------
        # Adam update with simplex projection
        # --------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        improved = False
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            improved = True
        else:
            # try a smaller step as a cheap back‑off
            step_small = lr * 0.1 * m_hat / (np.sqrt(v_hat) + adam_eps)
            cand2 = cur - step_small
            cand2 = _project_simplex(cand2)
            cand2_sc = _score(cand2)
            if cand2_sc < cur_sc:
                cur, cur_sc = cand2, cand2_sc
                improved = True

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        # --------------------------------------------------------
        # keep global best
        # --------------------------------------------------------
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

        # --------------------------------------------------------
        # periodic symmetrisation
        # --------------------------------------------------------
        if t_iter % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # --------------------------------------------------------
        # adapt smoothing factor
        # --------------------------------------------------------
        eps_factor = max(eps_factor * eps_decay, 1e-6)

        # --------------------------------------------------------
        # random mass‑transfer when stuck
        # --------------------------------------------------------
        if no_improve > 1000:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0

        # --------------------------------------------------------
        # restart from best with noise if hopeless
        # --------------------------------------------------------
        if no_improve > max_no_improve_restart:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            eps_factor = 0.5
            no_improve = 0

        # --------------------------------------------------------
        # early exit if target reached
        # --------------------------------------------------------
        if best_sc < target_score:
            break

        # --------------------------------------------------------
        # periodic time check
        # --------------------------------------------------------
        if t_iter % 500 == 0 and time.monotonic() >= deadline:
            break

    # return the best found sequence as plain Python floats
    return [float(x) for x in best_vec]
