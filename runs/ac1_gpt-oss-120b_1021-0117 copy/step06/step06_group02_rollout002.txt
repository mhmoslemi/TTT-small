import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex by the algorithm of Wang & Carreira-Perpinan."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ------------------------------------------------------------
# Fast AC1 bound (sum(a) is assumed ≈ 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length() + 1)   # power‑of‑two ≥ 2·(2n‑1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            s = 1.0
        a0 = a0 / s                     # normalise to sum = 1
    else:                               # fallback if parent absent
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 120.0)
        a0 = a0 / a0.sum()

    cur = a0.copy()
    n = cur.shape[0]

    # pre‑computed index map for gradient extraction (reversed convolution)
    grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)

    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------

    # ------------------------------------------------------------
    base_alpha = 0.85          # base step size (will be scaled)
    beta = 0.9                 # momentum damping
    momentum = np.zeros_like(cur)

        max_val = conv_full.max()

            break

        # ---- periodic time check ------------------------------------
        if time.monotonic() >= deadline:
            break

    # return the best found sequence as plain Python floats
    return [float(x) for x in best_vec]
