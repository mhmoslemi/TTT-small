```
def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1,n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

```
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.4)

    # initial vector
    if 'height_sequence_1' in globals():

    # optimizer state

    beta = 0.9
    step0 = 0.6

    temp_start = 0.5
    temp_end = 10.0
    # iteration counters
    it = 0
    no_improve = 0

    best_vec = cur.copy()
    best_sc = cur_sc
    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1
        # FFT of current
        a_fft = np.fft.rfft(cur, n=L)

        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        # temperature for softmax weighting
        t = temp_start + (temp_end - temp_start) * min(1.0, it / 20000.0)
        # softmax weights (stable)
        w = np.exp(t * (conv - max_val))
        w_sum = w.sum()
```

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)

        return v

        theta = 0.0
