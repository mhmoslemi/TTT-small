import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Initialise a (normalise to sum = 1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a[:] = 1.0
            s = a.size
        a = a / s
    else:
        n = 2048
        a = rng.random(n, dtype=np.float64)
        a = a / a.sum()

    n = a.size
    # FFT length (power of two) large enough for 3n‑2
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Adam optimiser state
    # ------------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.2

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    # initial score (sum = 1 ⇒ score = 2 n * max(conv))
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    cur_sc = 2.0 * n * conv.max()
    best_sc = cur_sc
    best_vec = a.copy()

    target = 1.50274349
    lam = 0.5                     # start smoothing parameter
    no_improve = 0
    it = 0

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # increase smoothing gradually
        lam = min(30.0, lam + 0.02)

        # ----- convolution a*a -----
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]

        # ----- smooth approximation of max -----
        exp_arg = lam * conv
        max_exp = exp_arg.max()
        w = np.exp(exp_arg - max_exp)          # shift for stability
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smooth max -----
        w_rev = w[::-1]                         # length 2n‑1
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)        # indices picking the needed slice
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (base_lr / math.sqrt(it + 1e-12)) * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_simplex(a_new, Z=1.0)

        # ----- evaluate candidate -----
        a_fft_new = np.fft.rfft(a_new, n=L)
        conv_new = np.fft.irfft(a_fft_new * a_fft_new, n=L)[: 2 * n - 1]
        sc_new = 2.0 * n * conv_new.max()

        # ----- accept if improvement -----
        if sc_new < cur_sc:

            cur_sc = sc_new
            no_improve = 0

        else:

            # occasional uphill move

            a_sym = _project_simplex(a_sym, Z=1.0)
            a_fft_sym = np.fft.rfft(a_sym, n=L)

                j += 1
            if a[i] > 0.0:

                a = _project_simplex(a, Z=1.0)

                    np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L
                )[: 2 * n - 1].max()
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
