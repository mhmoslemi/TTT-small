import time, math
import numpy as np

# ------------------------------------------------------------
# Project a vector onto the simplex (non‑negative, sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 bound for a vector with sum = 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    for cand in starters[1:]:
        sc = _score(cand)
        if sc < cur_sc:
            cur = cand.copy()
            cur_sc = sc

    best_vec = cur.copy()
    best_sc = cur_sc

    # ---- optimisation parameters ---------------------------------------
    m = np.zeros_like(cur)          # first moment (Adam)
    v = np.zeros_like(cur)          # second moment
    beta1, beta2 = 0.9, 0.999
    lr = 0.5
    it = 0
    t = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        t += 1

        # convolution of cur with itself
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # smoothing temperature (decays with iterations)
        eps = max_val * max(0.15 / math.sqrt(it), 1e-4)

        # soft‑max weights
        diff = conv_full - max_val
        w_exp = np.exp(diff / eps)
        w = w_exp / w_exp.sum()

        # gradient of the smooth max (matrix‑free via FFT)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        improved = False
        if cand_sc < cur_sc - 1e-12:
            cur, cur_sc = cand, cand_sc
            improved = True
        else:
            # occasional uphill move (simulated annealing)
            temp = max(1e-3, 1.0 / math.sqrt(it))
            if rng.random() < math.exp(-(cand_sc - cur_sc) / temp):
                cur, cur_sc = cand, cand_sc
                improved = True

        if improved:
            no_improve = 0
            if cur_sc < best_sc:
                best_sc, best_vec = cur_sc, cur.copy()
        else:
            no_improve += 1

        # ---------- periodic symmetrisation ----------
        if it % 300 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ---------- random mass transfer when stuck ----------
        if no_improve > 1500:
            i, j = rng.integers(0, n, size=2)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                cur[i] -= delta
                cur[j] += delta
            cur = _project_simplex(cur)
            cur_sc = _score(cur)
            no_improve = 0

        # ---------- restart from best if hopeless ----------
        if no_improve > 3000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = _project_simplex(cur)
            cur_sc = _score(cur)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        # early exit if target met
        if best_sc < target:
            break

        # safety time check
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------
    # Final cheap random refinements
    # ------------------------------------------------------------
    while time.monotonic() < deadline - 0.05:
        i, j = rng.integers(0, n, size=2)
        if best_vec[i] <= 0.0:
            continue
        delta = best_vec[i] * rng.random() * 0.05
        tmp = best_vec.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < best_sc:
            best_sc, best_vec = tmp_sc, tmp.copy()

    # Return as plain Python list
    return [float(x) for x in best_vec]
