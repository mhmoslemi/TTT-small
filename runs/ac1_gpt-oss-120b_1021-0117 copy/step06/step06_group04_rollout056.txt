import time

def _conv_max_score(a: np.ndarray) -> float:
    """Exact AC1 upper‑bound score for a non‑negative vector a."""
    n = a.shape[0]
    # length for FFT (next power of two covering full convolution)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * n - 1]
    max_conv = np.max(conv)
    s = np.sum(a)
    if s <= 0.0:
        return float('inf')
    return float(2.0 * n * max_conv / (s * s))
