import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x>=0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv

            if sym_sc < cur_sc:
                cur = sym
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # -------------------------------------------------------------
        # restart from the best point if we are stuck too long
        # -------------------------------------------------------------
        if no_improve > reinit_thresh:
            cur = best_vec + 0.001 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            no_improve = 0

        # early exit if target reached
        if best_sc < target_score:
            break

    # -----------------------------------------------------------------
    # final short local refinement (optional, time‑permitting)
    # -----------------------------------------------------------------
    while time.monotonic() < deadline - 0.02:
        # try a tiny random transfer from the best vector
        i = rng.integers(0, n)
        if best_vec[i] <= 0.0:
            continue
        delta = best_vec[i] * rng.random() * 0.05
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_max, _, _ = _conv_max_and_index(cand, L)
        cand_sc = 2.0 * n * cand_max / (cand.sum() ** 2)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand

    return [float(x) for x in best_vec]
