import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) onto {x >= 0, sum(x)=Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with sum Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Full linear convolution via FFT (no padding beyond needed)
# ------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv

# ------------------------------------------------------------
# Sub‑gradient of max_k (a*a)[k] (average over all maximisers)
# ------------------------------------------------------------
def _subgradient(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    n = a.size
    # indices where the convolution attains its maximum (within tolerance)
    idxs = np.where(np.isclose(conv, max_val, atol=1e-14))[0]
    if idxs.size == 0:
        return np.zeros_like(a)
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # contribution of a_{k-i} to the sub‑gradient at position i
        i = np.arange(i_start, i_end + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    g /= float(idxs.size)          # average over all maximisers
    return g

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients whose
    AC1 upper bound is as small as possible within the time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # -------------------- initialise --------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: moderate‑size random start
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0
        s = a.sum()
    a /= s                               # enforce sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())  # power‑of‑two length for FFT

    # -------------------- bookkeeping --------------------
    conv = _conv_full(a, L)
    max_val = conv.max()
    best_sc = 2.0 * n * max_val
    best_vec = a.copy()

    # optimisation parameters
    step0 = 0.65          # base step magnitude
    beta = 0.90           # momentum factor
    momentum = np.zeros_like(a)
    sym_interval = 300    # how often to symmetrise
    stagnation_limit = 8000
    no_improve = 0
    it = 0

    # -------------------- optimisation loop --------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution & score
        conv = _conv_full(a, L)
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        # keep the best so far
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # sub‑gradient of the current max‑convolution
        grad = _subgradient(a, conv, max_val)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size (diminishing)
        alpha = step0 / math.sqrt(it)

        # projected gradient descent step
        a_tmp = a - alpha * momentum
        a_tmp = _project_simplex(a_tmp)   # stays on simplex

        # evaluate candidate
        conv_tmp = _conv_full(a_tmp, L)
        cand_sc = 2.0 * n * conv_tmp.max()

        # accept if improvement, otherwise occasional uphill move
        if cand_sc < cur_sc:
            a = a_tmp
            cur_sc = cand_sc
            no_improve = 0
        else:
            # Metropolis‑style acceptance (probability decays with step)
            prob = math.exp(-(cand_sc - cur_sc) * 0.05 / (alpha + 1e-12))
            if rng.random() < prob:
                a = a_tmp
                cur_sc = cand_sc
                no_improve = 0

        # periodic symmetrisation (helps to lower the max)
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5

            conv_sym = _conv_full(a_sym, L)
            sym_sc = 2.0 * n * conv_sym.max()

                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                no_improve = 0

            # jitter around the best solution seen so far
            noise = rng.normal(scale=0.005, size=n)
            a = best_vec + noise
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
