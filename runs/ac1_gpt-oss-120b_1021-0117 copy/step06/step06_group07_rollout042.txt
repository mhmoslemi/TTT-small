def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    # optional imports 
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # margin
    
    # initialize a
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # generate random positive
        a = rng.random(2048, dtype=np.float64)
    # make nonnegative and sum to 1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0
        s = a.sum()
    a = a / s
    n = a.size
    
    # FFT length
    L = 1 << ((2*n-1).bit_length())
    
    # projection function
    def project_simplex(v):
        # same as earlier _project_simplex
        nlen = v.shape[0]
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, nlen+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w
    
    # initial best
    # compute initial score
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    cur_score = 2.0 * n * conv.max()
    best_score = cur_score
    best_a = a.copy()
    
    # Adam params
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0
    # other params
    sym_interval = 100   # may be more frequent
    gamma0 = 5.0
    gamma_inc = 0.02  # per iteration increase
    lr0 = 0.5  # base learning rate
    
    # main loop
    while time.monotonic() < deadline:
        t_iter += 1
        # compute convolution and gradient
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_conv = conv.max()
        # compute weights
        gamma = gamma0 + gamma_inc * t_iter
        # shift for stability
        shift = max_conv
        w = np.exp(gamma * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        
        # compute gradient using FFT
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        # convolution of a and w_rev
        s_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
        # take first (n + len(w_rev) -1) length
        s_full = s_full[:n + len(w_rev) - 1]  # length = 3n-2
        # extract S_k
        idx_start = (2*n-2) - np.arange(n)
        S = s_full[idx_start]
        # diagonal term
        diag = w[2*np.arange(n)]
        grad = 2.0 * S - a * diag
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t_iter)
        v_hat = v / (1 - beta2**t_iter)
        step = lr0 / (np.sqrt(v_hat) + eps) * m_hat
        a_new = a - step
        a_new = project_simplex(a_new)
        
        # evaluate new candidate
        a_new_fft = np.fft.rfft(a_new, n=L)
        conv_new = np.fft.irfft(a_new_fft * a_new_fft, n=L)[:2*n-1]
        score_new = 2.0 * n * conv_new.max()
        
        if score_new < cur_score:
            a = a_new
            cur_score = score_new
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
        else:
            # maybe accept occasionally?
            # simple Metropolis with probability exp(-(score_new - cur_score)/temp)
            temp = 0.01 + 0.001 * t_iter  # temperature
            if np.random.rand() < math.exp(-(score_new - cur_score) / temp):
                a = a_new
                cur_score = score_new
        
        # symmetric step
        if t_iter % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = project_simplex(a_sym)
            a_sym_fft = np.fft.rfft(a_sym, n=L)
            conv_sym = np.fft.irfft(a_sym_fft * a_sym_fft, n=L)[:2*n-1]
            score_sym = 2.0 * n * conv_sym.max()
            if score_sym < cur_score:
                a = a_sym
                cur_score = score_sym
                if score_sym < best_score:
                    best_score = score_sym
                    best_a = a_sym.copy()
        
        # random perturb when stuck
        if t_iter % 2000 == 0:
            # small random redistribution
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = min(a[i], a[i] * rng.random() * 0.1)
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                # recompute cur_score after perturbation
                a_fft = np.fft.rfft(a, n=L)
                conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
                cur_score = 2.0 * n * conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
    
    # return best candidate as list of floats
    return [float(x) for x in best_a]

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = Z}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative entries with sum Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
