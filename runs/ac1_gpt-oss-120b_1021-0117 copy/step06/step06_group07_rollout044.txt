# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with sum Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

            a_smooth = np.convolve(a_sym, kernel, mode='same')
            a_sym = _project_simplex(a_smooth, Z=1.0)
            a_fft_sym = np.fft.rfft(a_sym, n=L)
            conv_sym = np.fft.irfft(a_fft_sym * a_fft_sym, n=L)[: 2 * n - 1]
            sym_score = 2.0 * n * conv_sym.max()
            if sym_score < cur_score:
                a, cur_score = a_sym, sym_score
                if sym_score < best_score:
                    best_score, best_vec = sym_score, a_sym.copy()
                    no_improve = 0

        # ----- occasional random mass‑transfer when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, Z=1.0)
            no_improve = 0

    # --------------------------------------------------------
    # Return best candidate found (as plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
