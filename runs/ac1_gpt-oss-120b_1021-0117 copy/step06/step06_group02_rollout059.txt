def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # initialization
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 2000
        a = np.exp(-np.arange(n) / 120.)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length() + 1)
    grad_idx = (2*n - 2) - np.arange(n)

    cur = a.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = 0.6   # step size scaling
    t = 0

    eps = None
    eps_decay = 0.995
    # symmetrization interval maybe 100
    sym_interval = 120

    no_improve = 0

    while time.monotonic() < deadline:
        t += 1

        # compute convolution
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()

        # epsilon
        if eps is None:
            eps = max_val * 0.5
        else:
            eps = max(eps * eps_decay, 1e-12)

        # softmax weights
        w = np.exp((conv - max_val) / eps)
        ws = w.sum()
        if ws == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[grad_idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = base_lr / math.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        # ensure non-negative
        cand = np.maximum(cand, 0.0)
        # project onto simplex: renorm sum to 1
        sum_cand = cand.sum()
        if sum_cand == 0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / sum_cand

        # optional symmetrize
        if t % sym_interval == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = cand / cand.sum()   # renorm to 1

        cand_score = _score(cand)

        # acceptance: if improvement accept, else possibly accept with prob
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            # uphill acceptance with small prob to escape
            prob = math.exp(-(cand_score - cur_score) / (lr + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_score = cand_score
                no_improve = 0
            else:
                no_improve += 1

            break

        # ----- periodic time check ----------------------------------
        if t_iter % 400 == 0 and time.monotonic() >= deadline:
            break

    # return as plain Python floats
    return [float(x) for x in best_vec]
