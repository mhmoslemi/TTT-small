import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of 2*n*max(conv) for a sequence with sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(self‑convolution) for a sequence whose sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())          # FFT length (power of two)

    best_sc = cur_sc

    # ---- optimizer state (Adam) ------------------------------------
    m = np.zeros_like(cur)          # first moment
    v = np.zeros_like(cur)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # ---- smoothing temperature (soft‑max) -------------------------
    eps_temp = None

    sym_interval = 200
    stagnation = 0

        t_iter += 1

        # ----- self‑convolution & max --------------------------------

        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- update smoothing temperature ---------------------------
