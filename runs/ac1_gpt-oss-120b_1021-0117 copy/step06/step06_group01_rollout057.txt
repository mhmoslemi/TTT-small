    else:                                 # simple exponential fallback
        n0 = 2000
        a0 = np.exp(-np.arange(n0, dtype=np.float64) / 100.0)
        a0 = np.clip(a0, 0.0, 1000.0)

    if a0.sum() < 0.01:                 # guard against degenerate start
        a0 = np.full_like(a0, 1.0 / a0.size)

    best = a0.copy()
    best_sc = _fast_score(best)
    n = best.shape[0]

    # --------------------------------------------------------
    # cheap parametric families (dense but inexpensive)
    # --------------------------------------------------------
    # monotone exponential
    for _ in range(200):
        if time.monotonic() >= deadline:
            break
        alpha = 10 ** rng.uniform(-4, 0)          # 1e‑4 … 1
        cand = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand = cand / cand.sum()
        sc = _fast_score(cand)
        if sc < best_sc:
            best_sc, best = sc, cand.copy()

    # symmetric exponential (decay from both ends)
    for _ in range(200):
        if time.monotonic() >= deadline:
            break
        alpha = 10 ** rng.uniform(-4, 0)
        d = np.minimum(np.arange(n, dtype=np.float64),
                       n - 1 - np.arange(n, dtype=np.float64))
        cand = np.exp(-alpha * d)
        cand = cand / cand.sum()
        sc = _fast_score(cand)
        if sc < best_sc:
            best_sc, best = sc, cand.copy()

    # Gaussian‑like profiles
    mid = (n - 1) / 2.0
    for _ in range(200):
        if time.monotonic() >= deadline:
            break
        sigma = rng.uniform(0.5, 50.0)
        cand = np.exp(-((np.arange(n, dtype=np.float64) - mid) ** 2)
                      / (2.0 * sigma * sigma))
        cand = cand / cand.sum()
        sc = _fast_score(cand)
        if sc < best_sc:
            best_sc, best = sc, cand.copy()

    # --------------------------------------------------------
    # random convex combinations with the current best
    # --------------------------------------------------------
    for _ in range(500):
        if time.monotonic() >= deadline:
            break
        # fresh exponential component
        alpha = 10 ** rng.uniform(-4, 0)
        rand = np.exp(-alpha * np.arange(n, dtype=np.float64))
        # blend, keep non‑negativity, renormalise
        t = rng.random()
        cand = (1.0 - t) * best + t * rand
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand / cand.sum()
        sc = _fast_score(cand)
        if sc < best_sc:
            best_sc, best = sc, cand.copy()

    # --------------------------------------------------------
    # final output (plain Python list of floats)
    # --------------------------------------------------------
    return [float(x) for x in best]
