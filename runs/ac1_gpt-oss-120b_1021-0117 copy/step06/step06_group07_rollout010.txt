def subgradient_max_conv(a, conv, max_val):
    # conv length 2n-1
    idxs = np.where(np.abs(conv - max_val) < 1e-12)[0]
    n = a.size
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i ranges i_start..i_end inclusive
        # g[i] += a[k-i]
        # vector of a_sub = a[k - np.arange(i_start,i_end+1)]
        # compute reversed slice perhaps
        i = np.arange(i_start, i_end+1)
        g[i] += a[k - i]
    # Scale (since we are using subgradient of max, no scaling)
    return g
```

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if a0.ndim != 1:
            a0 = a0.ravel()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s < 1e-12:
            a0[:] = 1.0
            s = a0.sum()
        a0 = a0 / s   # sum=1
    else:
        n = 2048
        a0 = rng.random(n, dtype=np.float64)
        a0 = a0 / a0.sum()
    a = a0.copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    # Bookkeeping
    conv = _conv_full_fft(a, L)
    max_val = conv.max()
    cur_score = 2.0 * n * max_val
    best_vec = a.copy()
    best_score = cur_score
    # parameters
    step0 = 0.5
    beta = 0.9
    momentum = np.zeros_like(a)
    it = 0
    no_improve = 0
    sym_interval = 200
    target = 1.50274349
    # main loop
    while time.monotonic() < deadline:
        it += 1
        # compute conv and max
        conv = _conv_full_fft(a, L)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        # early exit
        if best_score < target:
            break
        # compute subgradient of max conv
        grad = _subgradient_max_conv(a, conv, max_val)
        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        # step size
        step = step0 / math.sqrt(it)
        # gradient descent step
        a_candidate = a - step * momentum
        a_candidate = _project_simplex(a_candidate)  # sum=1
        # evaluate
        conv_cand = _conv_full_fft(a_candidate, L)
        cand_score = 2.0 * n * conv_cand.max()
        # accept if better, else maybe accept with prob
        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_vec = a_candidate.copy()
        else:
            # metropolis style acceptance
            prob = math.exp(-(cand_score - cur_score) / (step + 1e-12))
            if rng.random() < prob:
                a = a_candidate
                cur_score = cand_score

            if sym_score < cur_score:

                cur_score = sym_score

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum Z."""
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Full linear convolution via FFT (returns the whole array)
# ------------------------------------------------------------
def _conv_full_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Compute full linear convolution a*a using FFT length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv

# ------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (exact, using all max indices)
# ------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """Sub‑gradient of the maximum entry of conv = a*a."""
    n = a.size
    # indices where conv hits the maximum (within tolerance)
    idxs = np.where(np.isclose(conv, max_val, atol=1e-12))[0]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        g[i] += a[k - i]
    return g

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a[:] = 1.0
            s = a.sum()
        a = a / s                     # enforce sum = 1
    else:
        n = 2048
        a = rng.random(n, dtype=np.float64)
        a = a / a.sum()

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length for convolution

    # ---- bookkeeping ------------------------------------------------
    conv = _conv_full_fft(a, L)
    max_val = conv.max()
    cur_score = 2.0 * n * max_val
    best_vec = a.copy()
    best_score = cur_score

    # optimisation parameters
    step0 = 0.5
    beta = 0.9
    momentum = np.zeros_like(a)
    it = 0
    no_improve = 0
    sym_interval = 100          # how often to symmetrise
    target = 1.50274349

    # ---- main loop -------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution and score
        conv = _conv_full_fft(a, L)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # keep the best so far
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            no_improve = 0
        else:

        # early exit if target reached
        if best_score < target:

    return [float(x) for x in best_vec]
