    else:

    a = (a + a[::-1]) * 0.5
    a /= a.sum()

    def compute_conv_and_max(arr):
        af = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2 * n - 1]
        return conv, conv.max()
    # Evaluate initial
    conv, max_val = compute_conv_and_max(a)
    cur_sc = 2.0 * n * max_val
    best_a = a.copy()
    best_sc = cur_sc

    # parameters

    eps_decay = 0.97
    step_base = 0.5
    max_trials = 5
    no_improve = 0
    target = 1.50274349
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # compute gradient weight
        if eps is None:
            eps = max_val * 0.2 + 1e-12
        else:
            eps = max(eps * eps_decay, max_val * 1e-12)

        if eps < max_val * 1e-6 or it > 2000:
            # hard mask

            w = mask.astype(np.float64)

import time, math

# ------------------------------------------------------------
# Project onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
