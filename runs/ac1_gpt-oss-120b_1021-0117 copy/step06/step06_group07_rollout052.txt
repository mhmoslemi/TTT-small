def compute_score(a, n, sum_a=None):
    a_clipped = np.clip(a, 0.0, 1000.0)
    sum_a = sum_a if sum_a is not None else a_clipped.sum()
    # Use FFT to compute linear convolution:
    L = 1 << ((2*n - 1).bit_length())
    Af = np.fft.rfft(a_clipped, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a**2)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper functions (FFT‑based convolution and scoring)
# ------------------------------------------------------------
def _full_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length L (>=2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv


def _conv_and_score(a: np.ndarray, L: int, sum_a: float) -> tuple[float, np.ndarray]:
    """Return (score, full convolution) for a non‑negative sequence a."""
    conv = _full_convolution(a, L)
    max_b = conv.max()
    score = 2.0 * a.size * max_b / (sum_a * sum_a)
    return float(score), conv


def _contributions_for_k(a: np.ndarray, k: int) -> np.ndarray:
    """Contribution a[i]*a[k-i] for each i (zero where out of range)."""
    n = a.shape[0]
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(a)
    contrib[mask] = a[idx[mask]] * a[j[mask]]
    return contrib

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # Initial sequence (use the provided parent if available)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:
        a[:] = 1.0
        total = a.sum()
    a = a / total                     # enforce sum = 1 (scale does not affect score)
    sum_a = 1.0                        # constant after the normalisation

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two FFT length

    # --------------------------------------------------------

    # --------------------------------------------------------
    cur_score, conv = _conv_and_score(a, L, sum_a)
