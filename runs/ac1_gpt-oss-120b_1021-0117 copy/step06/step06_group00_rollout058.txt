def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

import numpy as np
import time
import math

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x>=0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score_simplex(a):
    """Score 2*n*max(conv(a,a)) for a already on the simplex (sum=1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # --------------------------------------------------------------
    # initial sequence (use parent if present)
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        init = rng.random(2000, dtype=np.float64)

    cur = _project_simplex(np.maximum(init, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # basic candidates: original, sorted, alternating extremes
    best_vec = cur.copy()
    best_sc = _score_simplex(best_vec)
    cur_sc = best_sc

    # sorted descending
    cand = np.sort(cur)[::-1]
    cand_sc = _score_simplex(cand)
    if cand_sc < best_sc:
        best_sc, best_vec = cand_sc, cand.copy()
        cur, cur_sc = cand, cand_sc

    # alternating large/small pattern
    v = np.sort(cur)
    alt = np.empty_like(v)
    half = (n + 1) // 2
    alt[0::2][:half] = v[::-1][:half]       # large on even slots
    alt[1::2][:n // 2] = v[: n // 2]        # small on odd slots
    alt_sc = _score_simplex(alt)
    if alt_sc < best_sc:
        best_sc, best_vec = alt_sc, alt.copy()
        cur, cur_sc = alt, alt_sc

    # --------------------------------------------------------------
    # optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.5
    m = np.zeros_like(cur)
    v_hat = np.zeros_like(cur)
    t = 0
    alpha = 1.0
    alpha_max = 50.0
    alpha_inc = (alpha_max - alpha) / 3000.0   # reach max after ~3000 iters
    target = 1.50274349
    no_improve = 0

    while True:
        if time.monotonic() >= deadline:
            break
        t += 1

        # convolution of current vector with itself
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_c = conv.max()

        # soft‑max weights on the convolution entries
        diff = conv - max_c                # ≤ 0
        w = np.exp(alpha * diff)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the weighted max‑convolution
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update

        v_hat = beta2 * v_hat + (1 - beta2) * (grad * grad)

        cand = _project_simplex(cand)
