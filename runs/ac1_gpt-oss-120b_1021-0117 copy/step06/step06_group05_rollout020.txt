    def _conv(a):
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return conv

    def _score_conv(conv):
        return float(2.0 * n * conv.max())

    def _score(a):
        return _score_conv(_conv(a))

    # compute initial
    conv_cur = _conv(cur)
    cur_sc = _score_conv(conv_cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    no_improve = 0
    eps = None
    eps_factor = 0.02
    eps_decay = 0.996
    base_alpha = 0.6
    beta = 0.9

    sym_interval = 250
    improve_target = 1.50274349

        # compute conv and softmax weights
        a_fft = np.fft.rfft(cur, n=L)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a*a using FFT of length L (power of two)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Return the AC1 bound 2*n*max(conv) assuming sum(a)=1."""
    return float(2.0 * n * conv.max())


def _score(a: np.ndarray, L: int, n: int) -> float:
    """Convenience wrapper."""
    return _score_from_conv(_conv_full(a, L), n)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # Initial feasible point (normalize to sum=1)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        cur = a0 / s
    else:                                   # fallback
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        cur = np.exp(-t / 100.0)
        cur /= cur.sum()

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())      # FFT length (power of two)

    # --------------------------------------------------------
    # Initial scores
    # --------------------------------------------------------
    cur_sc = _score(cur, L, n)
    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------
    it = 0
    no_improve = 0
    eps = None
    eps_factor = 0.02          # sharper soft‑max
    eps_decay = 0.996
    base_alpha = 0.6
    beta = 0.9                 # momentum
    momentum = np.zeros_like(cur)
    sym_interval = 250
    improve_target = 1.50274349
    max_trials = 20
    rand_move_prob = 0.02
    mass_transfer_thresh = 2000
    restart_thresh = 3000

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # --- convolution and soft‑max weights ---
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps is None:
            eps = max(max_val * eps_factor, 1e-15)
        else:
            eps = max(eps * eps_decay, 1e-15)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of soft‑max max (via FFT) ---
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # --- momentum update ---
        momentum = beta * momentum + (1.0 - beta) * g

        # --- step size (decaying) ---
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / math.sqrt(it) / norm_m

        # --- backtracking line‑search (few tries) ---
        improved = False
        cand = None
        cand_sc = None
        alpha = alpha0
        for _ in range(max_trials):
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try, L, n)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5

        # --- occasional uphill move (annealing style) ---
        if not improved:
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try, L, n)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        # --- accept improvement ---
        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # --- periodic symmetrisation ---
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L, n)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # --- occasional random mass transfer ---
        if rng.random() < rand_move_prob:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L, n)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0

        # --- mass‑transfer when stuck ---
        if no_improve > mass_transfer_thresh:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L, n)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                no_improve = 0

        # --- restart if hopeless ---
        if no_improve > restart_thresh:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum.fill(0.0)
            eps = None
            no_improve = 0

        # --- early exit if target reached ---
        if best_sc < improve_target:
            break

    # --------------------------------------------------------
    # Fine‑grained subgradient polishing (time‑limited)
    # --------------------------------------------------------
    while time.monotonic() < deadline - 0.2:
        conv_cur = _conv_full(best_vec, L)
        max_val = conv_cur.max()
        argmax_idx = np.where(np.abs(conv_cur - max_val) <= 1e-12)[0]

        # subgradient (average of shifts corresponding to argmax lags)
        g_sub = np.zeros_like(best_vec)
        for k in argmax_idx:
            idxs = k - np.arange(n)
            mask = (idxs >= 0) & (idxs < n)
            g_sub[mask] += 2.0 * best_vec[idxs[mask]]
        # step (decreasing)
        step = 0.5 / math.sqrt(it + 1)
        cand = best_vec - step * g_sub
        cand = _project_simplex(cand)
        cand_sc = _score(cand, L, n)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand
        else:
            # occasional tiny random jitter to escape flat plateaus
            if rng.random() < 0.05:
                cand = best_vec + 0.001 * rng.standard_normal(n)
                cand = np.maximum(cand, 0.0)
                cand = cand / cand.sum()
                cand_sc = _score(cand, L, n)
                if cand_sc < best_sc:
                    best_sc = cand_sc
                    best_vec = cand
        if best_sc < improve_target:
            break

    # --------------------------------------------------------
    # Return best found sequence (as plain Python floats)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
