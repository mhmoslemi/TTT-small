        conv = np.fft.irfft(af * af, n=L)[:2*n-1]

    eps_decay = 0.999

    eta_decay = 0.9995
    eta = eta0

    it = 0
    # Keep momentum? Not needed for EG.

    while time.monotonic() < deadline:

        # compute convolution and max
        af = np.fft.rfft(a, n=L)

        max_conv = conv.max()
        # update best
        cur_score = float(2.0 * n * max_conv)
        if cur_score < best_score:

            best_a = a.copy()
            # early stop
            if best_score < 1.50274349:
                break

        # temperature schedule
        if eps is None:

        else:

        # weight for smoothed max
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()

        # step size eta decays
        if it == 1:
            eta = eta0
        else:
            eta = max(eta * eta_decay, 1e-6)

        # update

        # re-normalize to sum 1
        a_new_sum = a_new.sum()

            a_new = np.maximum(a, 0)

            a_new = a_new / a_new.sum()

        a = a_new

    # final best
    return [float(x) for x in best_a]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – used only for fall‑back cases
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Core optimisation – exponentiated gradient on a smoothed max
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function height list with a low AC1 bound.
    The routine runs for at most ``budget_s`` seconds and keeps the best
    feasible vector seen so far.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ---- initial feasible vector (sum = 1) ----
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
    else:                                   # fall‑back simple exponential decay
        n = 2000
        t = np.arange(n, dtype=np.float64)
        a = np.exp(-t / 100.0)
        a = a / a.sum()

    n = a.size
    # length for FFTs (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ---- tracking best solution ----
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    cur_score = float(2.0 * n * conv.max())
    best_a = a.copy()
    best_score = cur_score

    # ---- hyper‑parameters ----
    eps_factor = 0.07          # initial smoothing factor (relative to max)
    eps_decay = 0.999          # geometric decay of smoothing
    eta0 = 0.6                 # initial stepsize for exponentiated update
    eta_decay = 0.9995         # decay of stepsize
    sym_interval = 250         # enforce exact symmetry every this many steps
    rand_move_prob = 0.008     # occasional random mass‑transfer
    no_improve = 0
    no_improve_restart = 8000
    eps = None
    eta = eta0

    # ---- main optimisation loop ----
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ----- compute gradient of the smoothed max -----
        max_conv = conv.max()
        # temperature / smoothing
        if eps is None:
            eps = max(max_conv * eps_factor, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)

        # softmax weights over convolution entries
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient (see original implementation)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]               # shape (n,)

        # ----- exponentiated (mirror) update -----
        # decaying stepsize
        eta = max(eta * eta_decay, 1e-6)

        # avoid overflow in the exponential
        upd = -eta * g
        upd = np.clip(upd, -50, 50)
        a_candidate = a * np.exp(upd)

        # enforce exact symmetry periodically
        if it % sym_interval == 0:
            a_candidate = (a_candidate + a_candidate[::-1]) * 0.5

        # re‑normalise to the simplex
        s = a_candidate.sum()
        if s <= 0.0 or not np.isfinite(s):
            a_candidate = a.copy()
            s = a_candidate.sum()
        a_candidate = a_candidate / s

        # ----- evaluate candidate -----
        af_cand = np.fft.rfft(a_candidate, n=L)
        conv_cand = np.fft.irfft(af_cand * af_cand, n=L)[: 2 * n - 1]
        cand_score = float(2.0 * n * conv_cand.max())

        # ----- accept / reject -----
        if cand_score < cur_score:
            accept = True
        else:
            # simulated‑annealing style acceptance
            prob = math.exp(-(cand_score - cur_score) / (eps + 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_candidate
            af = af_cand
            conv = conv_cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- occasional random mass transfer (exploration) -----
        if rng.random() < rand_move_prob:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                a_tmp = a.copy()
                a_tmp[i] -= delta
                a_tmp[j] += delta
                a_tmp = a_tmp / a_tmp.sum()
                # quick evaluation
                af_tmp = np.fft.rfft(a_tmp, n=L)
                conv_tmp = np.fft.irfft(af_tmp * af_tmp, n=L)[: 2 * n - 1]
                tmp_score = float(2.0 * n * conv_tmp.max())
                if tmp_score < cur_score:
                    a = a_tmp
                    af = af_tmp
                    conv = conv_tmp
                    cur_score = tmp_score
                    if tmp_score < best_score:
                        best_score = tmp_score
                        best_a = a.copy()
                    no_improve = 0

        # ----- restart if stuck -----
        if no_improve > no_improve_restart:
            # perturb around the best solution found so far
            noise = 0.01 * rng.random(n)
            a = best_a + noise
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            af = np.fft.rfft(a, n=L)
            conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
            cur_score = float(2.0 * n * conv.max())
            eta = eta0
            eps = None
            no_improve = 0

        # ----- early termination when target reached -----
        if best_score < 1.50274349:
            break

    # return the best vector as plain Python floats
    return [float(x) for x in best_a]
