def propose_candidate(seed=42,budget_s=1100,**kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s)-0.1)

    # init vector
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a0 = rng.random(n, dtype=np.float64)

    # project onto simplex (sum=1)
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.shape[0]
    L = 1 << ((2*n-1).bit_length())
    # ensure symmetry (optional)
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)

    # maintain best
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5   # initial learning rate

    it = 0

    while time.monotonic() < deadline:
        it += 1
        # compute convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

        # gradient
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w = np.zeros_like(conv)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr0 / (np.sqrt(v_hat) + eps) * m_hat

        # update
        cand = cur - step  # gradient descent step (since we want to minimize)
        # ensure nonnegative and project onto simplex
        cand = _project_simplex(cand)

        # enforce symmetry optionally:
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # acceptance: accept if better
        if cand_sc < best_sc:
            best_vec = cand.copy()
            best_sc = cand_sc
            cur = cand
        elif cand_sc < _score(cur):  # accept if better than current else keep
            cur = cand
        else:

            break

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, non‑negative)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    # renormalise to guard against numerical drift
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a):
    """Score 2·n·max(conv) for a vector already normalized to sum=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the return handling
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # initialise
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # smooth sinusoidal start (non‑negative, symmetric)
        n_tmp = 2500
        xs = np.arange(n_tmp, dtype=np.float64)
        a0 = np.sin(np.pi * xs / (n_tmp - 1))
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    # enforce symmetry from the start
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)

    # ------------------------------------------------------------------
    # optimisation state
    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    L = 1 << ((2 * n - 1).bit_length())

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.3          # base learning rate
    beta1_pow, beta2_pow = 1.0, 1.0

    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        # ----- convolution and max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- sub‑gradient of max(conv) -----
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w.fill(0.0)

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        # ----- candidate -----
        cand = cur - step
        cand = _project_simplex(cand)
        # enforce symmetry each iteration
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # ----- accept if improvement -----
        if cand_sc < best_sc:
            best_vec, best_sc = cand.copy(), cand_sc
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        elif cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve += 1
        else:
            no_improve += 1
            # occasional stochastic acceptance (simulated‑annealing flavour)
            temp = lr0 / math.sqrt(it + 1.0)
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (temp + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # ----- random perturbation when stuck -----
        if no_improve > 3000:
            delta = rng.normal(scale=0.02, size=n)
            tmp = cur + delta
            tmp = _project_simplex(tmp)
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp = _project_simplex(tmp)

                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
            no_improve = 0

        # ----- early exit -----
        if best_sc <= target:
            break

    # ------------------------------------------------------------------
    # Return best found vector as plain Python floats
    return [float(x) for x in best_vec]
