import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex {x>=0, sum=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    # Sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a, L=None):
    """Score = 2 * n * max(conv(a,a)).  Uses FFT for speed."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
