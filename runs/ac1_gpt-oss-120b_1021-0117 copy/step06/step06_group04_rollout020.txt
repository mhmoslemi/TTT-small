def _conv_max_and_index(a, L):
    af = np.fft.rfft(a, L)

    idx = int(conv.argmax())

    cur_sc = 2.0*n*max_b/(cur.sum()**2)
    best = cur.copy()
    best_sc = cur_sc

    base_lr = 0.5

    it = 0
    no_improve = 0
    sym_interval = 250
    transfer_interval = 100
    target_score = 1.50274349

    while time.monotonic() < deadline:

        max_b, idx, _ = _conv_max_and_index(cur, L)

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_index(a: np.ndarray, L: int):
    """Return max convolution value, its index, and the full convolution."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative coefficient sequence (as a Python list)
    with a low AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # -----------------------------------------------------------------
    # initialise a feasible point on the simplex
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s0 = a0.sum()
        if s0 <= 0.0:

            s0 = 1.0

    else:
        # simple fallback: decreasing power law, normalized
        n_fallback = 2000

        a0 = (np.arange(n_fallback, 0, -1, dtype=np.float64) ** p)
        a0 = a0 / a0.sum()

    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # FFT length for a*a

    max_b, idx, _ = _conv_max_and_index(cur, L)
