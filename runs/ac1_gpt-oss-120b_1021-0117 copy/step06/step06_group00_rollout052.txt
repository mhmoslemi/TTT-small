import time, math, numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find rho = max {i | u_i - css_i / i > 0}
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Score for a vector that already sums to 1 (i.e. 2*n*max(conv))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main search routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # leave a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ----- initial candidates ------------------------------------------------
    if "height_sequence_1" in globals():

        parent = np.maximum(parent, 0.0)
        parent = _project_simplex(parent)
        init_candidates = [parent]
    else:
        init_candidates = []

    # fallback generic candidates (built algorithmically)
    n_fallback = 2000
    # uniform
    uniform = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
    init_candidates.append(uniform)

    # triangular (peak in the middle)
    n_tri = n_fallback
    idx = np.arange(n_tri)
    tri = np.minimum(idx + 1, n_tri - idx)
    tri = tri.astype(np.float64)
    tri = _project_simplex(tri)
    init_candidates.append(tri)

    # Gaussian‑like
    n_g = n_fallback
    x = np.arange(n_g)
    c = (n_g - 1) / 2.0
    sigma = n_g / 5.0
    gauss = np.exp(-0.5 * ((x - c) / sigma) ** 2)
    gauss = _project_simplex(gauss)
    init_candidates.append(gauss)

    # pick the best of the above as the starting point
    best_vec = None
    best_sc = float('inf')
    for cand in init_candidates:
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()

    cur = best_vec.copy()

    it = 0
    no_improve = 0
    target = 1.50274349
