import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) onto {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project v onto the non‑negative simplex of total mass Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Maximum entry of the linear (full) convolution a*a using FFT
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max_k (a*a)[k] using an FFT of length L (>=2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence that (heuristically) minimizes
    the AC1 upper bound evaluated by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # --------------------------------------------------------
    # Initial vector (use parent if present)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():

    # --------------------------------------------------------
    # Book‑keeping

    best_vec = a.copy()
    best_sc = 2.0 * n * _conv_max(a, L)

    # optimisation parameters
    step0 = 0.9          # base step magnitude
    beta = 0.95          # momentum factor
    momentum = np.zeros_like(a)
    top_k = 32           # number of largest autocorrelation entries used for gradient
    sym_interval = 120   # how often to symmetrise
    stagnation = 0
    it = 0
    target = 1.50274349   # scoreboard target (unreachable mathematically, but used as early stop)

    # --------------------------------------------------------
    # Optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        cur_max = conv_full.max()
        cur_sc = 2.0 * n * cur_max

        # ----- keep the best -----
        if cur_sc < best_sc:

            best_vec = a.copy()

        else:

        # ----- early exit if we somehow beat the target -----

        if top_k >= conv_full.size:
