import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast AC1 bound for a sequence with sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(convolution) for a sequence (sum≈1)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Subgradient for max_{k} (a*a)[k]
# ------------------------------------------------------------
def _subgradient(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """Exact subgradient of max(conv) wrt a (a non‑negative, sum=1)."""
    n = a.shape[0]
    # indices where conv is (practically) maximal
    tol = max_val * 1e-12 + 1e-15
    idx = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(a)
    for k in idx:
        if k < n:
            # contribution for j = 0..k  -> a[k-j]
            g[: k + 1] += 2.0 * a[k::-1]
        else:
            off = k - (n - 1)          # start position in gradient
            length = n - off
            g[off:] += 2.0 * a[:length][::-1]
    return g

# ------------------------------------------------------------
