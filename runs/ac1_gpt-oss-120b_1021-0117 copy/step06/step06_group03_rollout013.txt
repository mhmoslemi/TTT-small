import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (2 * n * max convolution)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        lr = base_lr / math.sqrt(it)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- simplex projection -----
        a_new = _project_simplex(a_new)

        # ----- evaluate candidate -----
        new_sc = _score(a_new)

        # ----- acceptance rule (mostly downhill, occasional uphill) -----
        if new_sc < cur_sc:
            a = a_new
            cur_sc = new_sc
            no_improve = 0
        else:
            prob = math.exp(-(new_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                a = a_new
                cur_sc = new_sc
                no_improve = 0
            else:
                no_improve += 1

        # ----- best‑so‑far tracking -----
        if new_sc < best_sc:
            best_sc = new_sc
            best_vec = a_new.copy()
            no_improve = 0

        # ----- periodic symmetrisation (helps flatten the profile) -----
        if it % sym_interval == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                a = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- restart on prolonged stagnation -----
        if no_improve > 3000:
            a = best_vec + 0.01 * rng.random(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            cur_sc = _score(a)
            m.fill(0.0)
            v.fill(0.0)
            eps = None
            it = 0
            no_improve = 0

    return best_vec, best_sc

# Entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """Return a non‑negative step‑function sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # safety margin

    # ----- obtain problem size and a default feasible vector -----
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 0.0:
            a0 = a0 / s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        n = a0.shape[0]
    else:
        n = 2000
        a0 = np.exp(-np.arange(n, dtype=np.float64) / 100.0)
        a0 = a0 / a0.sum()

    best_vec = a0.copy()
    best_sc = _score(best_vec)

    # ----- a handful of diverse initializations -----
    initials = [a0]

    # uniform
    initials.append(np.full(n, 1.0 / n))

    # simple exponential decay
    r = 0.999
    exp_vec = np.power(r, np.arange(n, dtype=np.float64))
    exp_vec = exp_vec / exp_vec.sum()
    initials.append(exp_vec)

    # symmetric exponential (mirrored decay)
    sym_exp = exp_vec + exp_vec[::-1]
    sym_exp = sym_exp / sym_exp.sum()
    initials.append(sym_exp)

    # ----- run optimizer on each seed while time permits -----
    for init in initials:
        if time.monotonic() >= deadline:
            break
        vec, sc = _optimize(init, deadline, rng)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()

    # ----- random restarts until time runs out -----
    while time.monotonic() < deadline - 0.5:
        rand = rng.random(n)
        rand = rand / rand.sum()
        vec, sc = _optimize(rand, deadline, rng)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()

    # return plain Python floats
    return [float(x) for x in best_vec]
