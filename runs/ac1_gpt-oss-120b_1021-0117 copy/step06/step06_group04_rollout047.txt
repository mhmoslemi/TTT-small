def compute_gradient(a, top_idx):
    n = a.shape[0]
    grad = np.zeros_like(a)
    indices = np.arange(n)
    for idx in top_idx:
        j = idx - indices
        mask = (j >=0) & (j < n)
        grad[mask] += 2.0 * a[j[mask]]
    return grad

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # initialize a
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
    else:
        n_fallback = 2000
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
    # project onto simplex (sum=1)
    a_sum = a.sum()
    if a_sum <= 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a_sum
    # symmetrize
    a = (a + a[::-1]) * 0.5
    a = _project_simplex(a)

    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())

    # adam params
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_vec = a.copy()
    # compute initial score
    conv = _conv_full(a, L)
    max_val = float(conv.max())
    best_sc = 2.0 * n * max_val
    cur_sc = best_sc
    cur = a.copy()

    no_improve = 0
    target_score = 1.50274349
    k_top = 3   # number of top indices for gradient

    while time.monotonic() < deadline:
        # compute conv and top indices
        conv = _conv_full(cur, L)
        max_val = float(conv.max())
        cur_sc = 2.0 * n * max_val
        # top-k indices
        # if many ties, get unique indices maybe
        top_idx = np.argpartition(-conv, k_top-1)[:k_top]

        # gradient
        grad = np.zeros_like(cur)
        indices = np.arange(n)
        for idx in top_idx:
            j = idx - indices
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * cur[j[mask]]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        step = lr / (np.sqrt(v_hat) + eps)
        cand = cur - step * m_hat

        # enforce nonneg and simplex
        cand = _project_simplex(cand)
        # symmetrize
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        # evaluate
        conv_cand = _conv_full(cand, L)
        max_val_cand = float(conv_cand.max())
        cand_sc = 2.0 * n * max_val_cand

        # acceptance
        improved = False
        if cand_sc < cur_sc:
            improved = True
        else:
            # simulated annealing probability
            temp = max(0.01, np.mean(step))  # approximate temperature
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # periodic symmetrisation (already sym)
        # random heavy perturbation
        if no_improve > 2000:
            # heavy random perturbation
            perturb = rng.standard_cauchy(size=n) * 0.01
            cand2 = cur + perturb
            cand2 = np.clip(cand2, 0, None)
            cand2 = _project_simplex(cand2)
            cand2 = (cand2 + cand2[::-1]) * 0.5
            cand2 = _project_simplex(cand2)
            conv2 = _conv_full(cand2, L)
            cand2_sc = 2.0 * n * float(conv2.max())
            if cand2_sc < cur_sc:
                cur = cand2
                cur_sc = cand2_sc
                if cand2_sc < best_sc:
                    best_sc = cand2_sc
                    best_vec = cand2.copy()
                no_improve = 0
            else:
                no_improve = max(0, no_improve - 500)

        # decay learning rate
        lr *= 0.9995

        # early exit
        if best_sc < target_score:
            break

        # check time inside loop maybe continue.

    # return best vector as list
    return [float(x) for x in best_vec]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1) > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.nonzero(rho_arr)[0][-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a using real FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return conv[: 2 * a.shape[0] - 1]
