import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (>=0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (sum normalized to 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a vector whose sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0

        # restart from best with slight noise if hopeless
        if no_improve > 3000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            m.fill(0.0)
            v.fill(0.0)
            eps_smooth = None
            no_improve = 0

    # -------------------- post‑processing local search --------------------
    # Simple hill‑climbing by moving tiny amounts of mass
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if best_vec[i] < 1e-12:
            continue
        delta = best_vec[i] * rng.random() * 0.07
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()

    # Return as plain Python list
    return [float(x) for x in best_vec]
