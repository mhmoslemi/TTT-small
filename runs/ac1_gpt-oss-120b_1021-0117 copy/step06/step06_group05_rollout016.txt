import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a: np.ndarray) -> float:
    """Exact AC1 bound for a non‑negative sequence a."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence that attempts to minimize the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.25)   # safety margin

    # ---- initial feasible point on the simplex ----
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 2000
        a = np.exp(-np.arange(n_fallback, dtype=np.float64) / 100.0)
    a = np.maximum(a, 0.0)
    if a.sum() <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    a = a / a.sum()                     # sum = 1
    n = a.shape[0]

    best_vec = a.copy()
    best_sc = _score(best_vec)

    # ---- try to use a linear‑programming refinement ----
    have_cvx = False
    try:
        import cvxpy as cp
        have_cvx = True
    except Exception:
        have_cvx = False

    if have_cvx:
        # Build a reusable LP: max sum(g) s.t. g≥0, g≤U, conv(f,g) ≤ M
        U = 1000.0                         # hard upper bound (matches evaluator)
        f_param = cp.Parameter(n)
        M_param = cp.Parameter()
        g = cp.Variable(n)
        conv_expr = cp.conv(f_param, g)    # length 2n‑1
        constraints = [g >= 0, g <= U, conv_expr <= M_param]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        # mixing coefficient for convex update
        t = 0.25
        while time.monotonic() < deadline:
            # compute current max convolution
            L = 1 << ((2 * n - 1).bit_length())
            af = np.fft.rfft(a, n=L)
            conv_full = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
            M = float(conv_full.max())

            # solve the LP
            f_param.value = a
            M_param.value = M
            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
            except Exception:
                break
            if g.value is None:
                break
            g_sol = np.maximum(g.value, 0.0)          # enforce non‑negativity
            sum_g = g_sol.sum()
            if sum_g < 1e-12:
                break
            g_scaled = g_sol / sum_g                 # renormalise to sum=1

            # convex combination and projection
            a = (1.0 - t) * a + t * g_scaled
            a = _project_simplex(a)

            cur_sc = _score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_vec = a.copy()

            # adapt step size
            if cur_sc > best_sc * 0.999:
                t = max(t * 0.7, 0.02)
            else:
                t = min(t * 1.1, 0.5)

            # occasional symmetrisation
            if rng.random() < 0.04:
                sym = (a + a[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym)
                if sym_sc < cur_sc:
                    a = sym
                    cur_sc = sym_sc
                    if cur_sc < best_sc:
                        best_sc = cur_sc
                        best_vec = a.copy()

            if best_sc < 1.50274349:
                break
    else:
        # ---- fallback: simple projected gradient descent with momentum ----
        L = 1 << ((2 * n - 1).bit_length())
        cur = a.copy()
        cur_sc = _score(cur)
        momentum = np.zeros_like(cur)
        beta = 0.9
        base_alpha = 0.7
        eps_factor = 0.03
        eps = None
        it = 0

        while time.monotonic() < deadline:
            it += 1
            af = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
            maxv = conv.max()
            if eps is None:
                eps = maxv * eps_factor + 1e-12
            else:
                eps = max(eps * 0.998, maxv * eps_factor * 0.002 + 1e-12)

            # soft‑max weights for smoothed max
            w = np.exp((conv - maxv) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of smoothed max
            w_rev = w[::-1]
            wf = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(af * wf, n=L)
            idx = (2 * n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]

            # momentum update
            momentum = beta * momentum + (1.0 - beta) * g
            norm_m = np.linalg.norm(momentum) + 1e-12
            alpha = base_alpha / math.sqrt(it) / norm_m

            # trial step
            cand = cur - alpha * momentum
            cand = _project_simplex(cand)
            cand_sc = _score(cand)

            if cand_sc < cur_sc:
                cur = cand
                cur_sc = cand_sc
                if cand_sc < best_sc:
                    best_sc = cand_sc
                    best_vec = cand.copy()
            else:
                # occasional random perturbation
                if rng.random() < 0.01:
                    i = rng.integers(0, n)
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    delta = cur[i] * rng.random() * 0.2
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp)
                    if tmp_sc < cur_sc:
                        cur = tmp
                        cur_sc = tmp_sc
                        if tmp_sc < best_sc:
                            best_sc = tmp_sc
                            best_vec = tmp.copy()

            # symmetrisation
            if it % 600 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym)
                if sym_sc < cur_sc:
                    cur = sym
                    cur_sc = sym_sc
                    if sym_sc < best_sc:
                        best_sc = sym_sc
                        best_vec = sym.copy()

            if best_sc < 1.50274349:
                break

    # ---- return the best sequence found ----
    return [float(x) for x in best_vec]
