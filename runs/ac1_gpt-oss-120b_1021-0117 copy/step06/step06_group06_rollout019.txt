import time, math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    # --------------------------------------------------------
    cur = a0.copy()
    cur_sc = _score(cur, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    lr0 = 0.25          # base learning rate
    eps_adam = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # temperature for soft‑max approximation of the max
    temp_factor = 0.15   # initial factor of max(conv)
    min_temp_ratio = 1e-6

    stagnation = 0       # counts iterations without improvement
    target = 1.50274349
    it = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # --- convolution a*a and its maximum ---
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # --- temperature schedule ---
        # Linear decay of the temperature factor, bounded away from zero.
        temp_factor = max(temp_factor * 0.998, 0.001)
        eps_temp = max(max_val * temp_factor, max_val * min_temp_ratio)

        # --- weight vector w approximating subgradient of max ---
        if eps_temp < max_val * 1e-6:
            # exact subgradient: uniform over arg‑max entries
            mask = conv_full == max_val
            w = mask.astype(np.float64)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(conv_full, 1.0 / conv_full.size)
            else:
                w /= w_sum
        else:
            # soft‑max with temperature eps_temp
            diff = (conv_full - max_val) / eps_temp
            # clipping avoids overflow in exponentiation
            np.clip(diff, -700, 0, out=diff)
            w = np.exp(diff)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(conv_full, 1.0 / conv_full.size)
            else:
                w /= w_sum

        # --- gradient of the (smoothed) max ---
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]          # shape (n,)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand, L)

        # --- accept if improvement ---
        if cand_sc < cur_sc - 1e-12:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1

        # --- occasional symmetrisation (helps enforce near‑palindromic shape) ---
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                stagnation = 0

        # --- random perturbation when stuck ---
        if stagnation > 800:
            # blend with a fresh random Dirichlet‑like vector
            r = rng.random(n)
            r = r / r.sum()
            alpha = 0.12 + 0.08 * rng.random()
            cand = (1.0 - alpha) * cur + alpha * r
            cand = _project_simplex(cand)
            cand_sc = _score(cand, L)
            if cand_sc < cur_sc:
                cur = cand
                cur_sc = cand_sc
                if cand_sc < best_sc:
                    best_sc = cand_sc
                    best_vec = cand.copy()
                stagnation = 0
            else:
                stagnation = 0  # reset even if no improvement to avoid endless loops

        # --- restart from best if hopeless ---
        if stagnation > 2500:
            cur = best_vec + 0.015 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            stagnation = 0

        # --- early exit if target reached ---
        if best_sc < target:
            break

    # --------------------------------------------------------
    # Final local refinement (cheap random swaps)
    # --------------------------------------------------------
    while time.monotonic() < deadline - 0.05:
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        delta = cur[i] * rng.random() * 0.07
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand, L)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            cur = cand
            cur_sc = cand_sc

    # Return the best-found vector as a plain Python list
    return [float(x) for x in best_vec]
