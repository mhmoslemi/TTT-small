import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Score used internally (2 * n * max(convolution(a, a))) for sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        tol = max_val * 5e-3 + 1e-15
        mask = conv >= max_val - tol
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        # ---- gradient of the weighted max‑convolution ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---- momentum update ----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ---- step size (decaying) ----
        step = step0 / math.sqrt(it + 1.0)

        # ---- candidate generation ----
        cand = cur - step * momentum
        cand = _project_simplex(cand)
        cand_score = _score(cand)

        # ---- Metropolis‑style acceptance ----
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
        else:
            prob = math.exp(-(cand_score - cur_score) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        # ---- keep the best overall ----
        if cur_score < best_score:
            best_vec, best_score = cur.copy(), cur_score
            if best_score < target:
                break

        # ---- periodic symmetrisation (helps reduce peaks) ----
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score(sym)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if cur_score < best_score:
                    best_vec, best_score = cur.copy(), cur_score
                    if best_score < target:
                        break

        # ---- occasional random mass transfer when stagnating ----
        if it % 500 == 0:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.25
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score(tmp)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    if cur_score < best_score:
                        best_vec, best_score = cur.copy(), cur_score
                        if best_score < target:
                            break

    # Return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
