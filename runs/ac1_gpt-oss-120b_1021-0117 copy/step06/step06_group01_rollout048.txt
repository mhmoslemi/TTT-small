import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
    else:  # fallback: simple exponential decay
        n_fallback = 2000
        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 100.0)
        a0 = a0 / a0.sum()

    n = a0.size
    best_vec = a0.copy()
    best_sc = _score(best_vec)

    # --------------------------------------------------------
    # explore a variety of parametric families (cheap)
    # --------------------------------------------------------
    # exponential decay
    for _ in range(300):
        alpha = 10 ** rng.uniform(-5, 0)  # 1e‑5 … 1
        cand = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand /= cand.sum()
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()

    # symmetric exponential (decay from both ends)
    for _ in range(300):
        alpha = 10 ** rng.uniform(-5, 0)
        d = np.minimum(np.arange(n, dtype=np.float64),
                       n - 1 - np.arange(n, dtype=np.float64))
        cand = np.exp(-alpha * d)
        cand /= cand.sum()
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()

    # Gaussian‑like
    mid = (n - 1) / 2.0
    for _ in range(300):
        sigma = rng.uniform(0.5, n / 2.0)
        cand = np.exp(-((np.arange(n, dtype=np.float64) - mid) ** 2) /
                      (2 * sigma ** 2))
        cand /= cand.sum()
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()

    # cosine‑squared (high at ends, low in centre)
    cand = np.cos(np.pi * np.arange(n, dtype=np.float64) / (n - 1)) ** 2
    cand /= cand.sum()
    sc = _score(cand)
    if sc < best_sc:
        best_sc, best_vec = sc, cand.copy()

    # beta‑shape family (two‑parameter power law)
    for _ in range(300):
        p = rng.uniform(0.1, 5.0)
        q = rng.uniform(0.1, 5.0)
        i = np.arange(n, dtype=np.float64) + 1
        cand = (i ** (p - 1)) * ((n - i + 1) ** (q - 1))
        cand /= cand.sum()
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()

    # random convex combinations of current best and a fresh exponential
    for _ in range(300):
        alpha = 10 ** rng.uniform(-5, 0)
        rand = np.exp(-alpha * np.arange(n, dtype=np.float64))
        rand /= rand.sum()
        t = rng.random()
        cand = (1 - t) * best_vec + t * rand
        cand = _project_simplex(cand)
        sc = _score(cand)
        if sc < best_sc:
            best_sc, best_vec = sc, cand.copy()

    # --------------------------------------------------------
    # Gradient‑based refinement (soft‑max surrogate of the max)
    # --------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc
    momentum = np.zeros_like(cur)

    beta_mom = 0.9                # momentum coefficient
    base_alpha = 0.9              # base step‑size scaling
    eps = None
    eps_decay = 0.994
    max_trials = 12
    it = 0
    no_improve = 0
    target = 1.50274349
    sym_interval = 80
    L = 1 << ((2 * n - 1).bit_length())

    while time.monotonic() < deadline:
        it += 1

        # convolution of cur with itself
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # smoothing parameter epsilon
        if eps is None:
            eps = max(max_val * 0.3, 1e-10)
        else:
            eps = max(eps * eps_decay, 1e-12)

        # soft‑max weights over convolution entries
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed max (using reversal trick)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g

        # step size (decaying)
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / math.sqrt(it) / norm_m

        # backtracking line search
        improved = False
        cand = None
        cand_sc = None
        alpha = alpha0
        for _ in range(max_trials):
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5

        # occasional uphill move (simulated‑annealing style)
        if not improved:
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            delta = cand_sc - cur_sc
            if delta < 0 or rng.random() < math.exp(-delta / (alpha + 1e-12)):
                cand = cand_try
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----------------------------------------------------
        # periodic symmetrisation (helps a lot)
        # ----------------------------------------------------
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----------------------------------------------------
        # occasional smoothing with a small kernel
        # ----------------------------------------------------
        if it % (sym_interval * 2) == 0:
            kernel = np.array([0.25, 0.5, 0.25])
            sm = np.convolve(cur, kernel, mode='same')
            sm = _project_simplex(sm)
            sm_sc = _score(sm)
            if sm_sc < cur_sc:
                cur = sm
                cur_sc = sm_sc
                if sm_sc < best_sc:
                    best_sc, best_vec = sm_sc, sm.copy()
                no_improve = 0

        # ----------------------------------------------------
        # random mass‑transfer when stuck
        # ----------------------------------------------------
        if no_improve > 1500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0
                else:
                    # tiny chance to accept a worsening move
                    if rng.random() < 0.05:
                        cur = tmp
                        cur_sc = tmp_sc
                        no_improve = 0

        # ----------------------------------------------------
        # restart from best with small Gaussian noise if hopeless
        # ----------------------------------------------------
        if no_improve > 4000:
            cur = best_vec + 0.005 * rng.standard_normal(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            eps = None
            no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

    # return as plain Python list
    return [float(x) for x in best_vec]
