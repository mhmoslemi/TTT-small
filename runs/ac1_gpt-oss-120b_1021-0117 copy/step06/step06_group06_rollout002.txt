import time, math

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# Objective for unit‑sum vectors (2*n*max(conv))
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _gradient(a: np.ndarray, L: int, eps: float, hard: bool) -> np.ndarray:
    """Return gradient of smoothed max‑conv w.r.t. a."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()

    if hard:
        w = (conv == max_val).astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)

            w /= w_sum

    # Main multi‑restart loop
    # --------------------------------------------------------
    while time.monotonic() < deadline - 0.2:
        cur = _init_vector()
        momentum = np.zeros_like(cur)
        cur_sc = _score(cur, L)

        # local optimisation parameters
        beta = 0.9
        eps_decay = 0.97
        eps = None
        it = 0
        max_iters = 5000

        while it < max_iters and time.monotonic() < deadline - 0.02:
            it += 1

            # compute current max‑conv
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
            max_val = conv_full.max()

            # temperature schedule
            if eps is None:
                eps = max_val * 0.15 + 1e-12
            else:
                eps = max(eps * eps_decay, max_val * 1e-12)

            # decide whether to use hard max (after enough iterations)
            hard = (it > 2500) and (eps < max_val * 1e-6)

            # gradient
            grad = _gradient(cur, L, eps, hard)

            # momentum update
            momentum = beta * momentum + (1.0 - beta) * grad

            # adaptive step size
            norm_m = np.linalg.norm(momentum) + 1e-12
            alpha = 0.6 / (math.sqrt(it) * norm_m)

            # candidate step with backtracking
            improved = False
            for _ in range(4):
                cand = cur - alpha * momentum
                cand = _project_simplex(cand)
                cand_sc = _score(cand, L)
                if cand_sc < cur_sc:
                    improved = True
                    break
                alpha *= 0.5

            # occasional uphill acceptance (simulated annealing)
            if not improved:
                temp = max(1e-3, 1.0 / math.sqrt(it))
                prob = math.exp(-(cand_sc - cur_sc) / temp)
                if rng.random() < prob:
                    improved = True

            if improved:
                cur = cand
                cur_sc = cand_sc
                if cand_sc < best_sc:
                    best_sc = cand_sc
                    best_vec = cand.copy()
                    if best_sc < target:
                        break
            # ----------------------------------------------------
            # Periodic symmetrisation (helps flatten the convolution)
            # ----------------------------------------------------
            if it % 300 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym, L)
                if sym_sc < cur_sc:
                    cur = sym
                    cur_sc = sym_sc
                    if sym_sc < best_sc:
                        best_sc = sym_sc
                        best_vec = sym.copy()
                        if best_sc < target:
                            break

            # small random mass transfer when stuck
            if it % 1000 == 0:
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    delta = cur[i] * rng.random() * 0.2
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp, L)
                    if tmp_sc < cur_sc:
                        cur = tmp
                        cur_sc = tmp_sc
                        if tmp_sc < best_sc:
                            best_sc = tmp_sc
                            best_vec = tmp.copy()
                            if best_sc < target:
                                break

        # early exit if target reached
        if best_sc < target:
            break

        # short random refinement after each restart
        for _ in range(80):
            if time.monotonic() >= deadline - 0.02:
                break
            i = rng.integers(0, n)
            if cur[i] <= 0.0:
                continue
            delta = cur[i] * rng.random() * 0.04
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)
            cand_sc = _score(cand, L)
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                cur = cand
                cur_sc = cand_sc

    # --------------------------------------------------------
    # Final symmetry enforcement (optional)
    # --------------------------------------------------------
    if best_vec is not None:
        sym = (best_vec + best_vec[::-1]) * 0.5
        sym = _project_simplex(sym)
        if _score(sym, L) < _score(best_vec, L):
            best_vec = sym

    # fallback safety
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
