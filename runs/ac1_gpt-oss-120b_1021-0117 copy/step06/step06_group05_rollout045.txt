import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if ind.size == 0:
        theta = 0.0
    else:
        rho = ind[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a) == 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------

        n0 = 2500
        a0 = np.exp(-np.arange(n0, dtype=np.float64) / 80.0)
        a0 = a0 / a0.sum()

    # enforce (soft) symmetry from the start
    a0 = (a0 + a0[::-1]) * 0.5
    a0 = a0 / a0.sum()

    cur = a0.copy()
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ---- optimisation parameters ------------------------------------
    momentum = np.zeros_like(cur)
    beta = 0.9                     # momentum decay
    base_alpha = 1.0              # base step size
    eps_factor = 0.3              # initial smoothing as fraction of max(conv)
    eps_decay = 0.9998            # smoothing decay
    eps = None
    max_trials = 20               # backtracking attempts
    sym_interval = 120            # how often to enforce symmetry
    rand_move_prob = 0.04
    mass_transfer_thresh = 2000
    restart_thresh = 3000
    no_improve = 0
    it = 0
    target = 1.50274349

    # ---- main optimisation loop ------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution & its max (all via FFT)
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # smoothing parameter (softmax temperature)
        if eps is None:
            eps = max_val * eps_factor
            eps = max(eps, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)

        # soft‑max weights for the smoothed maximum
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size (diminishing with iteration count)
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / (math.sqrt(it) * norm_m)

        # ---- backtracking line search --------------------------------
        improved = False
        cand = None
        cand_sc = None
        step = alpha
        for _ in range(max_trials):
            cand_try = cur - step * momentum
            cand_try = _project_simplex(cand_try)
            # enforce symmetry at candidate level
            cand_try = (cand_try + cand_try[::-1]) * 0.5
            cand_try = _project_simplex(cand_try)

            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            step *= 0.5

        # occasional uphill move (simulated‑annealing style)
        if not improved:
            cand_try = cur - step * momentum
            cand_try = _project_simplex(cand_try)
            cand_try = (cand_try + cand_try[::-1]) * 0.5
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        # ---- accept / reject -------------------------------------------
        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---- periodic symmetrisation -----------------------------------
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ---- occasional random mass transfer ----------------------------
        if rng.random() < rand_move_prob:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0

        # ---- heavy mass‑transfer when stuck -----------------------------
        if no_improve > mass_transfer_thresh:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.6
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                no_improve = 0

        # ---- restart from the best if hopeless -------------------------
        if no_improve > restart_thresh:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)
            momentum = np.zeros_like(cur)
            eps = None
            no_improve = 0

        # ---- early exit if target reached -------------------------------
        if best_sc < target:
            break

    # return as plain Python list
    return [float(x) for x in best_vec]
