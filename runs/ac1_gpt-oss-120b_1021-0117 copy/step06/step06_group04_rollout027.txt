def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """

    It evaluates both candidates with the official scorer and returns the

    # Helper: produce a copy of the parent and enforce the allowed value range
    def parent_seq():
        arr = np.asarray(height_sequence_1, dtype=np.float64).copy()
        # clamp to the evaluator's admissible interval
        np.clip(arr, 0.0, 1000.0, out=arr)
        return arr

    # Helper: uniform distribution of a given length
    def uniform_seq(length):
        u = np.full(length, 1.0 / length, dtype=np.float64)
        return u

    best_vec = None
    best_score = np.inf

    # -----------------------------------------------------------------
    # 1. Try the parent sequence (if it exists)
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = parent_seq()
            # quick validity check before evaluation
            if cand.size and np.all(np.isfinite(cand)):
                sc = evaluate_sequence(cand.tolist())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_vec = cand
        except Exception:
            # any unexpected problem – simply ignore this candidate
            pass

    # -----------------------------------------------------------------
    # 2. Uniform candidate of the same length (or a reasonable default)
    # -----------------------------------------------------------------
    length = best_vec.shape[0] if best_vec is not None else (
