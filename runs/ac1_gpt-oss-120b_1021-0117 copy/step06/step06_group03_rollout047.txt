def propose_candidate(...):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s)-0.2)

    # init
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0)
        s = a.sum()
        if s <= 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
    else:
        n = 2000
        a = np.exp(-np.arange(n)/100.)
        a /= a.sum()

    n = a.size
    L = 1 << ((2*n-1).bit_length())

    # Helper functions:
    # _score (same)
    # _project_simplex (same)
    # _gradient_softmax(a, eps): compute gradient as described
    def gradient_softmax(a, eps):
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af*af, n=L)[:2*n-1]
        max_val = conv.max()
        # Compute softmax weights with temperature eps
        # w = exp((conv - max_val)/eps) , normalized
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient = 2 * (a * w_rev conv)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * w_fft, n=L)[:2*n-1]  # same length as conv
        # slice appropriate part for gradient length n
        # Derive mapping: For i from 0..n-1, gradient_i = 2 * sum_{k} w_k * a_{k-i}
        # Based on conv formula, we can compute via index: grad_i = 2 * sum_{k} w_{k} * a_{k-i}
        # This equals 2 * (a * w reversed) at appropriate shift.
        # earlier code used idx = (2*n-2) - np.arange(n) then g = 2 * grad_full[idx]
        idx = (2*n-2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        return g, max_val

import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound for a sequence with sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence whose sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of the smoothed max via softmax weights
# ------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, eps: float):
    """Compute gradient of the smoothed max and return (grad, max_val)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)

    # full autocorrelation
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()

    # softmax weights (more mass near the peak)
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2 * (a convolve reversed w)
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * w_fft, n=L)[: 2 * n - 1]

    # slice to obtain gradient of length n
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g, max_val
