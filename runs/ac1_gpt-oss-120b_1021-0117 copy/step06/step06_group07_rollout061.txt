import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex of non‑negative entries summing to Z."""
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise a (sum forced to 1)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a reasonable fallback length
        a = rng.random(2048, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0
        s = a.sum()
    a = a / s                     # enforce sum = 1

    n = a.shape[0]

    # --------------------------------------------------------
    # FFT lengths (powers of two, enough for the required convolutions)
    # --------------------------------------------------------
    L1 = 1 << ((2 * n - 2).bit_length())          # >= 2n‑1
    L2 = 1 << ((3 * n - 3).bit_length())          # >= 3n‑2

    # --------------------------------------------------------
    # Optimisation state (Adam‑style)
    # --------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.30                     # base learning rate
    alpha = 5.0                    # soft‑max “temperature”
    alpha_max = 30.0

    best_vec = a.copy()
    best_sc = float('inf')
    it = 0
    no_improve = 0
    target = 1.50274349

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution a*a (full) -----
        a_fft = np.fft.rfft(a, n=L1)
        conv = np.fft.irfft(a_fft * a_fft, n=L1)[: 2 * n - 1]
        max_conv = conv.max()
        cur_sc = 2.0 * n * max_conv

        # ----- keep the best seen -----
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- soft‑max weights over the convolution entries -----
        # weights sum to 1, concentrate on large values as alpha grows
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the (soft‑max) max‑convolution -----
        rev_a = a[::-1]                         # reversed a
        w_fft = np.fft.rfft(w, n=L2)
        rev_a_fft = np.fft.rfft(rev_a, n=L2)
        grad_full = np.fft.irfft(w_fft * rev_a_fft, n=L2)
        # slice gives the n‑length gradient vector (factor 2 from the derivative)
        G = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- Adam update (scaled step) -----
        m = beta1 * m + (1.0 - beta1) * G
        v = beta2 * v + (1.0 - beta2) * (G * G)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (lr0 / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

        # ----- gradient step and projection back to the simplex -----
        a_tmp = a - step
        a_tmp = _project_simplex(a_tmp)          # keep non‑negative, sum = 1
        a = a_tmp

        # ----- gradual increase of the soft‑max temperature -----
        if it % 100 == 0:
            alpha = min(alpha * 1.05, alpha_max)

        # ----- occasional symmetrisation (helps reduce the centre peak) -----
        if it % 500 == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            # quick check whether symmetrisation helped
            a_fft_sym = np.fft.rfft(a_sym, n=L1)
            conv_sym = np.fft.irfft(a_fft_sym * a_fft_sym, n=L1)[: 2 * n - 1]
            sc_sym = 2.0 * n * conv_sym.max()
            if sc_sym < cur_sc:
                a = a_sym
                if sc_sym < best_sc:
                    best_sc = sc_sym
                    best_vec = a_sym.copy()
                    no_improve = 0

        # ----- occasional random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                # reset Adam moments to avoid stale direction
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0

    # ------------------------------------------------------------
    # Return the best candidate found (as a plain Python list)
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
