import time, math, numpy as np

def _project_simplex(v):
    # same as previous but handle dtype
    n=v.size
    u=np.sort(v)[::-1]
    css=np.cumsum(u)-1
    rho=np.where(u - css/np.arange(1,n+1) > 0)[0]
    if rho.size==0:
        theta=0.0
    else:
        rho=rho[-1]
        theta=css[rho]/(rho+1)
    w=np.maximum(v-theta,0)
    return w

def _score(a):
    n=a.shape[0]
    L=1 << ((2*n-1).bit_length())
    af=np.fft.rfft(a,n=L)
    conv=np.fft.irfft(af*af,n=L)[:2*n-1]
    return float(2.0*n*conv.max())

def _gradient(a, eps):
    # compute gradient of smoothed max
    n=a.shape[0]
    L=1 << ((2*n-1).bit_length())
    af=np.fft.rfft(a,n=L)
    conv=np.fft.irfft(af*af,n=L)[:2*n-1]

    w=np.exp((conv - max_val)/eps)

    idx=(2*n-2)-np.arange(n)
    g=2.0*grad_full[idx]
    return g, max_val

        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1 - beta) * g
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / (math.sqrt(it) * norm_m)

        # backtracking line search
        improved = False
        cand = None
        cand_sc = None
        step_alpha = alpha
        for _ in range(max_trials):
            cand_try = cur - step_alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            step_alpha *= 0.5

        # occasional uphill move
        if not improved:
            cand_try = cur - step_alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            prob = math.exp(-(cand_sc - cur_sc) / (step_alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_vec = cur.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # periodic symmetrisation
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # random small mass transfer
        if rng.random() < rand_move_prob:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0

        # deterministic mass‑transfer when stuck
        if no_improve > mass_transfer_thresh:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # restart from best if hopeless
        if no_improve > restart_thresh:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            eps = None
            cur_sc = _score(cur)
            no_improve = 0

        # occasional noisy perturbation around the best solution
        if it % 500 == 0:
            sigma = 0.02
            cand = best_vec + sigma * rng.standard_normal(n)
            cand = np.maximum(cand, 0.0)
            cand = cand / cand.sum()
            cand_sc = _score(cand)
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                cur = cand.copy()
                cur_sc = cand_sc
                no_improve = 0

        if best_sc < target:
            break

    return [float(x) for x in best_vec]
