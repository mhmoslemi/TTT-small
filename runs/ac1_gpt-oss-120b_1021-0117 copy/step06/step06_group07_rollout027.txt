import time, math, numpy as np, scipy.sparse as sp, cvxpy as cp

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_fft_max(a, L):
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)
    return float(conv[:2*a.size-1].max())

def _subgrad_topk(a, conv, k):
    n = a.shape[0]
    if k >= conv.size:
        idx = np.arange(conv.size)
    else:
        idx = np.argpartition(-conv, k-1)[:k]
    g = np.zeros_like(a)
    for kk in idx:
        i0 = max(0, kk-(n-1))
        i1 = min(n-1, kk)
        # vectorized for range
        i = np.arange(i0, i1+1, dtype=int)
        g[i] += 2.0 * a[kk - i]
    if idx.size:
        g /= float(idx.size)
    return g

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise a
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():

        a = np.maximum(a, 0.0)

            a[:] = 1.0
        a = a / a.sum()
    else:
        n = 2000
        a = rng.random(n, dtype=np.float64)
        a = a / a.sum()

    n = a.size
    L = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    def bound_vec(v):

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    best_vec = a.copy()
    best_bound = cur_bound

    sym_interval = 120
    move_interval = 30

    no_improve = 0
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- current bound (reuse if just computed) ----
        # Compute convolution and maximum
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        cur_max = float(conv_full.max())
        cur_bound = 2.0 * n * cur_max

        # ---- keep best ----
        if cur_bound < best_bound:
            best_bound, best_vec = cur_bound, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ---- early exit ----
        if best_bound < target:
            break

        # ---- sub‑gradient (top‑k) ----
        g = _subgrad_topk(a, conv_full, top_k)

        # ---- Exponentiated‑gradient step ----
        eta = eta0 / math.sqrt(it)
        # protect against overflow
        step = -eta * g
        step = np.clip(step, -50, 50)
        a_new = a * np.exp(step)

        if sum_new == 0 or np.isnan(sum_new):
            a_new = a.copy()
            sum_new = a_new.sum()
        a_new = a_new / sum_new

        # ---- evaluate ----
        new_bound = bound_vec(a_new)

            cur_bound = new_bound
        else:

            if rng.random() < prob:

                cur_bound = new_bound

        # ---- occasional random mass transfer ----

                if move_bound < cur_bound:

                    a_mix = np.maximum(a_mix, 0.0)
                    a_mix = a_mix / a_mix.sum()

                        cur_bound = mix_bound
                        if mix_bound < best_bound:

                            no_improve = 0

    # ------------------------------------------------------------------
    # Return best found candidate as plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]

import time
import math
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft_max(a: np.ndarray, L: int) -> float:
    """Maximum entry of the linear convolution a*a computed by FFT."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

def _subgrad_topk(a: np.ndarray, conv: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max‑convolution using the k largest entries."""
    n = a.size
    if k >= conv.size:
        idx = np.arange(conv.size)
    else:
        idx = np.argpartition(-conv, k - 1)[:k]
    g = np.zeros_like(a)
    for kk in idx:
        i0 = max(0, kk - (n - 1))
        i1 = min(n - 1, kk)
        i = np.arange(i0, i1 + 1, dtype=int)
        g[i] += 2.0 * a[kk - i]
    if idx.size:
        g /= float(idx.size)
    return g

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initial vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a[:] = 1.0
        n = a.size
        # blend with a simple cosine‑squared shape to give a good start
        idx = np.arange(n)
        center = (n - 1) / 2.0
        cos_shape = np.cos(np.pi * (idx - center) / (n - 1)) ** 2
        a = (a + cos_shape) * 0.5
        a /= a.sum()
    else:
        n = 2500
        a = rng.random(n, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a /= a.sum()

    L = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Helper for the bound (sum is normalized to 1)
    # ------------------------------------------------------------------
    def bound_vec(v: np.ndarray) -> float:
        return 2.0 * n * _conv_fft_max(v, L)

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    cur_bound = bound_vec(a)
    best_vec = a.copy()
    best_bound = cur_bound

    # optimisation parameters
    eta0 = 1.0               # base step size for exponentiated gradient
    top_k = 32               # gradients from the largest conv entries
    sym_interval = 100       # symmetrise every this many iterations
    move_interval = 30       # random mass‑transfer interval
    move_frac = 0.25         # fraction of mass moved
    no_improve = 0
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution and its max
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        cur_max = float(conv_full.max())
        cur_bound = 2.0 * n * cur_max

        # keep the best ever seen
        if cur_bound < best_bound:
            best_bound, best_vec = cur_bound, a.copy()
            no_improve = 0
        else:

        # early exit if target reached
        if best_bound < target:

        # sub‑gradient of the max‑convolution
        g = _subgrad_topk(a, conv_full, top_k)

        step_base = eta0 / math.sqrt(it)
        factors = (0.5, 1.0, 2.0)
        improved = False
        for f in factors:

            s = a_candidate.sum()
            if s == 0.0 or np.isnan(s):

            a_candidate /= s
            cand_bound = bound_vec(a_candidate)
            if cand_bound < cur_bound:

                cur_bound = cand_bound

        # ------------------------------------------------------------------
        if it % 2000 == 0 and (deadline - time.monotonic()) > 5.0:

    # ------------------------------------------------------------------
