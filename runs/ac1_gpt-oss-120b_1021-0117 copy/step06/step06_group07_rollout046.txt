import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x) = Z}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with sum Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Exact sub‑gradient of max‑convolution
# ----------------------------------------------------------------------
def _grad_exact(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv) w.r.t. a.
    conv = a * a (full linear convolution). Returns gradient of the
    average over all active (maximal) positions.
    """
    max_val = conv.max()
    active = np.where(conv >= max_val - eps)[0]
    n = a.size
    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        # contribution 2*a[k-idx] to each index idx
        g[idx] += 2.0 * a[k - idx]
    if active.size:
        g /= float(active.size)
    return g
