import time, math, numpy as np

def _project_simplex(v, Z=1.0):
    # Euclidean projection onto simplex sum=Z
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    # shift by mean? Use sorting
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a, L, n):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return 2.0 * n * conv.max(), conv

import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum Z."""
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Compute the AC‑1 bound (2·n·max(conv)) and optionally return the full conv
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int, n: int):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max()), conv

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC‑1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    n = a.size

    best_vec = a.copy()

    # -------- optimisation state -----------------------------------------
    m = np.zeros_like(a)           # first‑moment estimate (Adam)

    beta1, beta2 = 0.9, 0.999
