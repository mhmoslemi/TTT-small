import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper: create a uniform step function of length n (sum of entries = 1)
# ----------------------------------------------------------------------
def _uniform_sequence(n: int) -> list:
    """Return a list of length n with all entries equal to 1/n."""
    if n <= 0:
        return [0.0]
    v = 1.0 / n
    return [float(v) for _ in range(n)]


# ----------------------------------------------------------------------
# Optional cheap projection onto the probability simplex (sum = 1, non‑neg.)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, sum x = 1}."""
    if v.size == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core scoring routine (identical to the evaluator’s internal computation)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Fast AC1 bound for a vector that already satisfies sum(a)=1."""
    n = a.shape[0]
    # length for a linear convolution via FFT (next power of two)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list) that attempts to minimise the
    AC1 upper bound.  The routine works completely in‑memory and respects the
    time budget.
    """
    rng = random.Random(seed)
    start = time.monotonic()
    # keep a small safety margin for the final return
    deadline = start + max(0.0, float(budget_s) - 0.15)

    # ------------------------------------------------------------------
    # Determine the problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)      # use the provided length if available
    else:
        n = 2000                       # a reasonable fallback length

    # ------------------------------------------------------------------
    # Initial feasible point: the uniform distribution (sum = 1)
    # ------------------------------------------------------------------
    cur = np.array(_uniform_sequence(n), dtype=np.float64)
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------------
    # Simple stochastic local search – cheap but often effective.
    # ------------------------------------------------------------------
    # Parameters (feel free to tweak)
    max_no_improve = 2500          # restart after this many idle steps
    swap_radius = 0.4              # maximal relative mass to move per step
    no_improve = 0

    while time.monotonic() < deadline:
        # propose a tiny random mass transfer between two distinct indices
        i = rng.randrange(n)
        j = rng.randrange(n - 1)
        if j >= i:
            j += 1

        # copy current vector and move a fraction of mass
        cand = cur.copy()
        delta = cand[i] * rng.random() * swap_radius
        cand[i] -= delta
        cand[j] += delta

        # ensure feasibility (non‑negative, sum = 1)
        if cand[i] < 0.0:
            # reject this move – it would violate the simplex
            no_improve += 1
            if no_improve > max_no_improve:
                # restart from the best point with a random perturbation
                cur = best_vec + 0.01 * np.abs(np.random.randn(n))
                cur = np.maximum(cur, 0.0)
                cur = cur / cur.sum()
                cur_sc = _score(cur)
                no_improve = 0
            continue

        # score the candidate
        cand_sc = _score(cand)

        # accept if it improves the bound
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                # restart from the best seen so far with a tiny jitter
                cur = best_vec + 0.01 * np.abs(np.random.randn(n))
                cur = np.maximum(cur, 0.0)
                cur = cur / cur.sum()
                cur_sc = _score(cur)
                no_improve = 0

        # occasional symmetrisation (helps for many instances)
        if rng.random() < 0.001:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

    # ------------------------------------------------------------------
    # Return the best vector found (converted to plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
