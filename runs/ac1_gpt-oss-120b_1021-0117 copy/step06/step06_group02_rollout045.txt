            conv_sym = np.fft.irfft(af_sym * af_sym, n=L)[: 2 * n - 1]
            sym_score = float(2.0 * n * conv_sym.max())
            if sym_score < cur_score:
                v = np.log(sym + 1e-12)
                cur_score = sym_score
                a = sym
                momentum.fill(0.0)
                if sym_score < best_score:
                    best_score = sym_score
                    best_a = sym.copy()
                    no_improve = 0

        # ---- aggressive random mass‑transfer when stuck -----------------------------
        if no_improve > 2000:
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                a_tmp = a.copy()
                a_tmp[i] -= delta
                a_tmp[j] += delta
                a_tmp = np.maximum(a_tmp, 0.0)
                a_tmp /= a_tmp.sum()
                v = np.log(a_tmp + 1e-12)
                momentum.fill(0.0)
                eps = None
                no_improve = 0

        # ---- restart from best with small noise if hopeless -------------------------
        if no_improve > 5000:
            v = np.log(best_a + 1e-12) + 0.02 * rng.standard_normal(n)
            momentum.fill(0.0)
            eps = None
            no_improve = 0

        # ---- early exit if target reached -----------------------------------------
        if best_score < improve_target:
            break

    # return the best sequence found (as plain Python floats)
    return [float(x) for x in best_a]
