def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum‑one (a feasible simplex point)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        return v / s
    # fallback uniform if everything became zero
    n = v.size
    return np.full_like(v, 1.0 / n)


def _conv_max_and_index(a: np.ndarray, L: int):
    """Return (max value, its index, full convolution) using an FFT of length L."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv

    """
    Sub‑gradient of the maximal convolution entry.
    max_idx is the index in the linear convolution array (length 2n‑1).

        grad[lag:] += a[: n - lag]
    else:
        lag = -lag
        grad[lag:] += a[: n - lag]
        grad[: n - lag] += a[lag:]

    return grad

    cur_max, cur_idx, _ = _conv_max_and_index(cur, L)
    cur_sc = best_sc
    momentum = np.zeros_like(cur)

    # -----------------------------------------------------------------
    # optimisation loop
    # -----------------------------------------------------------------
    beta = 0.9                     # momentum factor
    it = 0
    no_improve = 0
    target_score = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # gradient of the current max‑convolution entry
        grad = _grad_max(cur, cur_idx)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        norm_m = np.linalg.norm(momentum) + 1e-12

        # step size (decays like 1/√it, normalised)
        step_factor = 0.5 / (math.sqrt(it) + 1.0)
        step = step_factor * (momentum / norm_m)

        # trial move (opposite to gradient direction)
        cand = cur - step
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_max, cand_idx, _ = _conv_max_and_index(cand, L)
        cand_sc = 2.0 * n * cand_max / (cand.sum() ** 2)

        improved = False
        if cand_sc < cur_sc:
            improved = True
        else:
            # simulated‑annealing‑style uphill acceptance
            temp = max(0.01, 1.0 / math.sqrt(it))
            if rng.random() < math.exp(-(cand_sc - cur_sc) / temp):
                improved = True

        if improved:
            cur, cur_sc, cur_idx = cand, cand_sc, cand_idx
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1

        # -----------------------------------------------------------------
        # occasional symmetrisation (helps regularise the shape)
        # -----------------------------------------------------------------
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_max, sym_idx, _ = _conv_max_and_index(sym, L)
            sym_sc = 2.0 * n * sym_max / (sym.sum() ** 2)
            if sym_sc < cur_sc:
                cur, cur_sc, cur_idx = sym, sym_sc, sym_idx
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # -----------------------------------------------------------------
        # random mass transfer when progress stalls
        # -----------------------------------------------------------------
        if no_improve > 1500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_max, tmp_idx, _ = _conv_max_and_index(tmp, L)
                tmp_sc = 2.0 * n * tmp_max / (tmp.sum() ** 2)
                if tmp_sc < cur_sc:
                    cur, cur_sc, cur_idx = tmp, tmp_sc, tmp_idx
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                else:
                    no_improve = max(no_improve - 200, 0)

        # -----------------------------------------------------------------
        # restart from the best so far if we have been stuck too long
        # -----------------------------------------------------------------
        if no_improve > 3000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = _project_simplex(cur)
            momentum = np.zeros_like(cur)
            no_improve = 0

        # early exit if we have already reached the target bound
        if best_sc < target_score:
            break

    # return plain Python list
    return [float(x) for x in best_vec]
