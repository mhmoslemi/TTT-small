import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast scoring (AC1 bound) assuming the vector is non‑negative.
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Weighted gradient of a weighted sum of convolutions.
# ------------------------------------------------------------
def _weighted_grad(a: np.ndarray, shifts: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """Gradient of sum_k w_k * (a*a)[k] w.r.t. a."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for sh, w in zip(shifts, weights):
        if sh == 0:
            g += 2.0 * w * a
        elif sh > 0:
            # a_{i+sh}
            g[: n - sh] += w * a[sh:]
            # a_{i-sh}
            g[sh:] += w * a[: n - sh]
        else:  # sh < 0
            s = -sh
            # a_{i-s}
            g[s:] += w * a[: n - s]
            # a_{i+s}
            g[: n - s] += w * a[s:]
    return g
