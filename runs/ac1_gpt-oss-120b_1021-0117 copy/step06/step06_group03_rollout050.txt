    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # Get n
    if 'height_sequence_1' in globals():
        a_init = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a_init = np.maximum(a_init, 0.0)
        s = a_init.sum()
        if s <= 0.0:
            n = a_init.size
            a_init = np.full(n, 1.0 / n)
        else:
            a_init = a_init / s
    else:
        n = 2000
        a_init = np.full(n, 1.0 / n)

    n = a_init.size
    # Validate: length may be 0? not possible.

    # Additional Gaussian init
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n / 8.0
    a_gauss = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    a_gauss = a_gauss / a_gauss.sum()

    # Choose best initial
    cur = a_init if _score(a_init) < _score(a_gauss) else a_gauss
    cur_sc = _score(cur)

    best_vec = cur.copy()
    best_sc = cur_sc

    L = 1 << ((2 * n - 1).bit_length())
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # smoothing temperature
    eps = None
    eps_decay = 0.995

    base_lr = 0.5
    sym_interval = 250
    no_improve = 0

    improve_target = 1.50274349

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # compute convolution
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        max_val = conv_full.max()

        # update smoothing temperature
        if eps is None:
            eps = max(max_val * 0.1, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)

        # softmax weights
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()

            cur = cand
            cur_sc = cand_sc

            break

    return [float(x) for x in best_vec]

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of 2*n*max(conv) for a sequence whose sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
