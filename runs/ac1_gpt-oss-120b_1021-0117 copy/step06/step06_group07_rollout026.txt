def propose_candidate(seed=42,budget_s=1100,**kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2048
        a = rng.random(n, dtype=np.float64)
    a = np.clip(a,0,None)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()

    n = a.size
    L = 1 << ((2*n - 1).bit_length())

    # initial best
    conv = np.fft.irfft(np.fft.rfft(a,n=L) * np.fft.rfft(a,n=L), n=L)[:2*n-1]
    max_conv = float(conv.max())
    best_score = 2.0 * n * max_conv
    best_vec = a.copy()

    # optimisation vars
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    no_improve = 0
    it = 0
    sym_interval = 300
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # compute conv, max
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_conv = float(conv_full.max())
        cur_score = 2.0 * n * max_conv

        # update best
        if cur_score < best_score:
            best_score, best_vec = cur_score, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_score < target:
            break

        # gradient
        g = _grad_max(a, conv_full)
        # momentum update
        momentum = beta * momentum + (1 - beta) * g
        # direction normalization
        norm = np.linalg.norm(momentum)
        if norm > 0:
            direction = momentum / norm
        else:
            direction = momentum

        alpha = step0 / (np.sqrt(it) + 1e-12)
        a_tmp = a - alpha * direction
        a_tmp = _project_simplex(a_tmp)

        # evaluate candidate
        tmp_fft = np.fft.rfft(a_tmp, n=L)
        tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[:2*n-1]
        tmp_score = 2.0 * n * float(tmp_conv.max())

        # acceptance
        if tmp_score < cur_score:
            a = a_tmp
            cur_score = tmp_score
            no_improve = 0
        else:
            # occasional uphill move if improvement not too large
            prob = math.exp(-(tmp_score - cur_score) / (alpha + 1e-12))
            if rng.random() < prob:
                a = a_tmp
                cur_score = tmp_score

        # symmetrisation
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            sym_score = 2.0 * n * _conv_max_fft(a_sym, L)
            if sym_score < cur_score:
                a = a_sym
                cur_score = sym_score
                if sym_score < best_score:
                    best_score, best_vec = sym_score, a_sym.copy()
                    no_improve = 0

        # random mass transfer when stuck
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            if a[i] > 0:
                delta = a[i] * rng.random() * 0.5
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

        # occasional step reduction
        if no_improve > 5000:
            step0 *= 0.5
            no_improve = 0

    # final return best as list
    return [float(x) for x in best_vec]

import time

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative entries that sum to Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Max of full linear convolution via FFT (returns the maximum value)
# ------------------------------------------------------------
def _conv_max_fft(a: np.ndarray, L: int) -> float:
    """Compute max_k (a*a)[k] using an FFT of length L (>=2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

def _grad_max(a: np.ndarray, conv_full: np.ndarray) -> np.ndarray:
    """Return a sub‑gradient of max_k (a*a)[k] w.r.t. a."""

    # each unordered pair appears twice; factor 2 gives the true sub‑gradient
