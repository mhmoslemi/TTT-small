def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a = rng.random(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    # Normalize to sum=1 (simplex)
    a = a / a.sum()

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())  # power of two >= 2n-1

    best_vec = a.copy()
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    best_sc = float(2 * n * conv.max())

    # momentum (for gradient smoothing)
    momentum = np.zeros_like(a)
    beta = 0.9

    step0 = 0.7  # starting step magnitude
    it = 0
    no_improve = 0
    sym_interval = 200  # more frequent symmetrize
    target = 1.50274349
    # Optionally we may adjust step size schedule later

    while time.monotonic() < deadline:
        it += 1

        # Compute convolution and max
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        cur_max = conv.max()
        cur_sc = 2.0 * n * cur_max

        # Update best
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Early exit
        if best_sc < target:
            break

        # Determine max index (maybe one of several)
        k_max = int(np.argmax(conv))

        # Compute subgradient of max term
        # g_j = 2 * a_{k_max - j} if within bounds else 0
        idx = np.arange(n)
        shift = k_max - idx
        valid = (shift >= 0) & (shift < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[shift[valid]]

        # Momentum
        momentum = beta * momentum + (1.0 - beta) * g

        # Step size decaying
        alpha = step0 / math.sqrt(it)

        a_tmp = a - alpha * momentum
        # Project onto simplex sum=1
        a_tmp = _project_simplex(a_tmp)

        cand_sc = 2.0 * n * conv_tmp.max()

        # Accept improvement
        if cand_sc < cur_sc:
            a = a_tmp
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, a_tmp.copy()
        else:
            # maybe occasional uphill move (Metropolis)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))

                a = a_tmp

                a, cur_sc = a_sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    no_improve = 0

        # Random mass transfer when stuck
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

                a[i] -= delta
                a[j] += delta
                a /= a.sum()
                no_improve = 0

    # Return best as list
    return [float(x) for x in best_vec]

def _grad_topk(a, conv, top_k):
    n = a.size
    idxs = np.argpartition(-conv, top_k-1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:

        g[valid] += 2*a[shift[valid]]
    return g / top_k

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x)=Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum Z."""
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Max of the full linear convolution via FFT
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max_k (a*a)[k] using an FFT of length L (>=2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# ------------------------------------------------------------
# Gradient of the active (max) convolution entry
# ------------------------------------------------------------
def _grad_max_entry(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of (a*a)[k_max] w.r.t. a (non‑negative case)."""
    n = a.size
    idx = np.arange(n)
    shift = k_max - idx
    valid = (shift >= 0) & (shift < n)
    g = np.zeros_like(a)
    g[valid] = 2.0 * a[shift[valid]]
    return g

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: a random vector of length ~2000
        a = rng.random(2000, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                       # enforce simplex (sum = 1)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length >= 2n‑1

    # ---- bookkeeping ------------------------------------------------
    best_vec = a.copy()
    best_sc = 2.0 * n * _conv_max(a, L)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.7
    it = 0
    no_improve = 0
    sym_interval = 200
    target = 1.50274349

    # ---- optimisation loop -----------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution & score
        cur_max = _conv_max(a, L)
        cur_sc = 2.0 * n * cur_max

        # keep best
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0

        if best_sc < target:

        # sub‑gradient from the maximal entry

        k_max = int(np.argmax(full_conv))
        g = _grad_max_entry(a, k_max)
