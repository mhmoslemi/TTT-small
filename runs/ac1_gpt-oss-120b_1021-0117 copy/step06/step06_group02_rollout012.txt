import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, all >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


# ------------------------------------------------------------
# Helper: compute 2 * n * max(convolution) (sum is 1)
# ------------------------------------------------------------
def _conv_score(a: np.ndarray, L: int, n: int) -> float:
    """Fast evaluation of 2*n*max(a*a) assuming sum(a)=1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Subgradient of max(convolution) w.r.t a (single index k_max)
# ------------------------------------------------------------
def _subgrad_maxconv(a: np.ndarray, k_max: int) -> np.ndarray:
    """Return a subgradient of max(conv) at the current max index."""
    n = a.shape[0]
    idx = k_max - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[idx[mask]]
    return g
