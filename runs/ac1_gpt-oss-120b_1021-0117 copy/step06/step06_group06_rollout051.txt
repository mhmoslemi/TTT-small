import time

# ------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(a: np.ndarray, n: int) -> float:
    """Exact AC1 bound for a non‑negative vector a (assumes sum(a)=1)."""
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _gradient(a: np.ndarray, n: int, eps: float, hard: bool):
    """
    Gradient of the (soft) max of the self‑convolution.
    Returns (grad, max_val, new_eps).
    """
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()

    # update smoothing parameter
    new_eps = max(eps * 0.97, max_val * 1e-12) if eps is not None else max_val * 0.15 + 1e-12

    # weight vector for (soft) max
    if hard or new_eps < max_val * 1e-6:
        w = (conv_full == max_val).astype(np.float64)
    else:
        w = np.exp((conv_full - max_val) / new_eps)

    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(conv_full, 1.0 / conv_full.size)
    else:
        w /= w_sum

    # gradient via FFT (matrix‑free)
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]

    return grad, max_val, new_eps
