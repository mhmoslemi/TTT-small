    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

        a_fft = np.fft.rfft(cur, n=L)

        max_c = conv.max()

        w = np.exp(alpha * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:

        else:

        # compute gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)

        idx = (2 * n - 2) - np.arange(n)

        m = beta1 * m + (1 - beta1) * grad

        m_hat = m / (1 - beta1 ** t)

import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(a, L):
    """Full convolution a*a via FFT (length L, power of two)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * len(a) - 1]
    return conv

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin before the hard timeout
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # initial vector
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two ≥ 2n‑1

    # best solution seen so far
    best = cur.copy()
    best_score = 2.0 * n * _conv_fft(best, L).max()

    # Adam optimiser state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    no_improve = 0
    target = 1.50274349     # desired bound

    # soft‑max sharpening factor (higher → sharper approximation of max)
    alpha = 12.0

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # --------------------------------------------------------------
        # enforce symmetry each iteration (optimal solutions are symmetric)
        cur = (cur + cur[::-1]) * 0.5

        # convolution of current iterate
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_c = conv.max()

        # soft‑max weights over the convolution entries
        w = np.exp(alpha * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the (soft) max‑convolution
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)        # map to gradient components
        grad = 2.0 * grad_full[idx]

        # ---------------------- Adam update ---------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = 0.12 / (np.sqrt(v_hat) + eps)   # base step size
        cand = cur - step * m_hat

        # projection back onto the simplex (keeps sum = 1, non‑negative)
        cand = _project_simplex(cand)

        # evaluate candidate (cheap upper‑bound estimate, exact because sum=1)
        conv_c = _conv_fft(cand, L)
        cand_score = 2.0 * n * conv_c.max()

        # keep the best vector seen
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

        cur = cand

        # occasional random redistribution when stuck
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = 2.0 * n * _conv_fft(tmp, L).max()
                if tmp_score < best_score:
                    cur, best, best_score = tmp, tmp, tmp_score
                    no_improve = 0

        # early exit if we have reached the target bound
        if best_score < target:
            break

    # Return as plain Python floats
    return [float(x) for x in best]
