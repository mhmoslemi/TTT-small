import time, math, numpy as np, scipy.sparse as sp, cvxpy as cp

def _project_simplex(v, Z=1.0):
    v = np.asarray(v, dtype=np.float64)
    if Z < 0:
        raise ValueError
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max(a, L):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

def _gradient_topk(a, conv, top_k):
    n = a.size
    idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    g /= top_k
    return g

def _build_conv_matrix(a):
    n = a.shape[0]
    data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)
    T = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return T

# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v, Z=1.0):
    v = np.asarray(v, dtype=np.float64)
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Maximum of the linear (full) convolution a*a using FFT
# ----------------------------------------------------------------------
def _conv_max(a, L):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# Sub‑gradient obtained from the top‑k largest convolution entries
# ----------------------------------------------------------------------
def _gradient_topk(a, conv, top_k):
    n = a.size
    idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:

        i1 = min(n - 1, k)
        i = np.arange(i0, i1 + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    g /= top_k
    return g

# ----------------------------------------------------------------------
def _build_conv_matrix(a):
    n = a.shape[0]
    data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)                     # 0 … n‑1
    T = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return T


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # -------------------- initialise --------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # simplex with sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # -------------------- exact score --------------------
    def _score(x):
        M = _conv_max(x, L)
        s = x.sum()
        if s < 1e-12:
            return float("inf")
        return 2.0 * n * M / (s * s)

    # -------------------- cheap LP refinement (once) --------------------
    if (deadline - time.monotonic()) > 20.0:
        M0 = _conv_max(a, L)
        A0 = _build_conv_matrix(a)
        g = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [A0 @ g <= M0 + 1e-12])
        try:
            prob.solve(solver="ECOS", warm_start=True, max_iters=1500, verbose=False)
        except Exception:
            prob = None
        if g.value is not None:
            g_opt = np.maximum(g.value, 0.0)
            a = _project_simplex(0.6 * a + 0.4 * g_opt)

    cur = a.copy()
    best = cur.copy()
    best_sc = _score(best)
    cur_sc = best_sc

    # -------------------- optimisation parameters --------------------
    momentum = np.zeros_like(cur)
    step0 = 1.0
    beta = 0.90
    top_k = min(64, max(8, n // 40))    # ~5 % of length, capped
    sym_interval = 120
    no_improve = 0
    it = 0
    target = 1.50274349

    # -------------------- main loop --------------------
    while time.monotonic() < deadline:
        it += 1

        # full convolution of current iterate
        af = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        cur_max = conv_full.max()
        cur_sc = 2.0 * n * cur_max     # sum(cur)=1 by construction

        if cur_sc < best_sc:
            best_sc, best = cur_sc, cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_sc < target:
            break

        # ------------------ projected SGD step with momentum ------------------
        g = _gradient_topk(cur, conv_full, top_k)
        momentum = beta * momentum + (1.0 - beta) * g
        alpha = step0 / math.sqrt(it)
        cand = cur - alpha * momentum
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12)):
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best = cand_sc, cand.copy()

        # ------------------ periodic symmetrisation ------------------
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best = sym_sc, sym.copy()
                    no_improve = 0

        # ------------------ targeted mass transfer when stuck ------------------
        if no_improve > 1500:
            k_max = int(conv_full.argmax())
            i0 = max(0, k_max - (n - 1))
            i1 = min(n - 1, k_max)
            i_range = np.arange(i0, i1 + 1, dtype=int)
            j_range = k_max - i_range
            prod = cur[i_range] * cur[j_range]
            if prod.size:
                i_bad = i_range[int(prod.argmax())]
                j_bad = rng.integers(n)
                while j_bad == i_bad:
                    j_bad = rng.integers(n)
                delta = cur[i_bad] * rng.random() * 0.3
                cur[i_bad] -= delta
                cur[j_bad] += delta
                cur = _project_simplex(cur)
                no_improve = 0

        # ------------------ tiny jitter to avoid flat regions ------------------
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            cur = _project_simplex(cur + jitter)

    # ------------------------------------------------------------------
    # Return the best vector found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best]
