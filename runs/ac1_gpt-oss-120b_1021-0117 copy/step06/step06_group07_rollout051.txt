import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    # shift by the max to improve numerical stability
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Main entry point

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # Initialise from parent or random
    # --------------------------------------------------------

        a = np.asarray(height_sequence_1, dtype=np.float64).copy()

        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # --------------------------------------------------------
    # Helper to evaluate the AC1 bound (uses the official evaluator)
    # --------------------------------------------------------
    def _score(vec):
        fft = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
        return 2.0 * n * conv.max()

    # --------------------------------------------------------
    # Keep the best candidate seen so far
    # --------------------------------------------------------
    cur_sc = _score(a)
    best_sc = cur_sc
    best_vec = a.copy()

    # --------------------------------------------------------

    # --------------------------------------------------------
    for _ in range(15):
        m = max(30, int(0.06 * n))           # roughly 6 % non‑zeros
        idx = rng.choice(n, size=m, replace=False)

        vec[idx] = 1.0
        vec = vec / vec.sum()
        sc = _score(vec)
        if sc < best_sc:

    step0 = 0.8

        cur_max = conv_full.max()
        cur_sc = 2.0 * n * cur_max

        # ----- keep the best -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:

        if best_sc < target:

            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = k - i

        a_tmp = _project_simplex(a_tmp, Z=1.0)

            momentum.fill(0.0)

    # --------------------------------------------------------
    # Return the best candidate (as plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
