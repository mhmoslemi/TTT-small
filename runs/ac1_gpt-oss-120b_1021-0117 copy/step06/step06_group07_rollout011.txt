import time, math, numpy as np, cvxpy as cp

def _project_simplex(v, Z=1.0):
    # same as earlier but careful about floating point
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_full_fft(a, L):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return conv[: 2 * a.size - 1]

def _grad_topk(a, conv, top_k):
    n = a.size
    # indices of top_k largest entries
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k-1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            i = np.arange(i_start, i_end + 1, dtype=int)
            g[i] += 2.0 * a[k - i]
    g /= max(1, top_k)
    return g

def _build_A_matrix(f, active_idxs):
    # f is array of length n
    n = f.size
    K = len(active_idxs)
    A = np.zeros((K, n), dtype=np.float64)
    for row, k in enumerate(active_idxs):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            i = np.arange(i_start, i_end + 1, dtype=int)
            A[row, i] = f[k - i]
    return A

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., float(budget_s) - 0.5)  # safety margin

    # initialise a
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # generate a random positive vector of moderate length
        n = 2000
        a = rng.random(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()   # project onto simplex sum=1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two for FFT

    # bookkeeping
    best_vec = a.copy()
    conv = _conv_full_fft(a, L)
    best_sc = 2.0 * n * conv.max()
    cur_sc = best_sc

    # optimisation parameters
    step0 = 1.0
    beta = 0.92
    momentum = np.zeros_like(a)
    top_k = 32
    sym_interval = 200
    no_improve = 0
    it = 0
    target = 1.50274349
    # tolerance for LP active set
    lp_eps = 1e-8

    while time.monotonic() < deadline:
        it += 1

        # current convolution and max
        conv = _conv_full_fft(a, L)
        cur_max = conv.max()
        cur_sc = 2.0 * n * cur_max

        # store best
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # early exit
        if best_sc < target:
            break

        # gradient step (projected SGD with momentum)
        g = _grad_topk(a, conv, top_k)
        momentum = beta * momentum + (1.0 - beta) * g
        alpha = step0 / math.sqrt(it)

        a_tmp = _project_simplex(a_tmp)

        # evaluate candidate

        cand_sc = 2.0 * n * conv_tmp.max()

            M = cur_max
            # solve LP: max sum(g) s.t. A @ g <= M + tol, g>=0
            g_var = cp.Variable(n, nonneg=True)
            constraints = [A @ g_var <= M + lp_eps]
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
            try:
                prob.solve(solver="ECOS", warm_start=True, max_iters=1500, verbose=False)
            except Exception:
                prob = None
            if prob is not None and g_var.value is not None:
                g_opt = np.maximum(g_var.value, 0.0)
                # mix with current a
                t = 0.2
                a_mix = (1 - t) * a + t * g_opt
                a_mix = _project_simplex(a_mix)
                conv_mix = _conv_full_fft(a_mix, L)

                if mix_sc < cur_sc:
                    a, cur_sc = a_mix, mix_sc
                    if mix_sc < best_sc:
                        best_sc, best_vec = mix_sc, a_mix.copy()
                        no_improve = 0

    # return best found
    return [float(x) for x in best_vec]

import time
import math
import numpy as np
import cvxpy as cp

def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=Z}."""
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a using an FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return conv[: 2 * a.size - 1]


def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    """Sub‑gradient approximated from the top‑k largest convolution entries."""
    n = a.size
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            i = np.arange(i_start, i_end + 1, dtype=int)
            g[i] += 2.0 * a[k - i]
    g /= max(1, top_k)
    return g


def _build_A_matrix(f: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """Build small dense matrix A where (A @ g)[r] = (f * g)[active_idxs[r]]."""
    n = f.size
    K = active_idxs.size
    A = np.zeros((K, n), dtype=np.float64)
    for r, k in enumerate(active_idxs):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            i = np.arange(i_start, i_end + 1, dtype=int)
            A[r, i] = f[k - i]
    return A

    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
