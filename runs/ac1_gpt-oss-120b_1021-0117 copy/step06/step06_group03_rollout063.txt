def _score_fast(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    return 2.0 * n * max_b / (sum_a * sum_a)
```

import time
import math
import numpy as np

# ------------------------------------------------------------
# Fast objective (same as evaluate_sequence for non‑clipped values)
# ------------------------------------------------------------
def _score_fast(a: np.ndarray) -> float:
    """2 * n * max(convolution) / (sum(a)^2) for a non‑negative vector."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # length for linear convolution
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:                                 # safeguard – should never happen
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    start = time.monotonic()
    # keep a tiny safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # --------------------------------------------------------
    # initialise a feasible point (fallback if parent missing)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 2000
        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 100.0)

    a0 = np.maximum(a0, 0.0)               # enforce non‑negativity
    a0 = np.minimum(a0, 1000.0)             # respect the evaluator’s clamp
    sum_a0 = a0.sum()
    if sum_a0 < 1e-12:                      # degenerate – replace by uniform
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / sum_a0                    # normalise to sum = 1 (scale‑invariant)

    n = a0.size
    best_vec = a0.copy()
    best_sc = _score_fast(best_vec)

    # --------------------------------------------------------
    # 1) Scan simple parametrised families (exponential / power)
    # --------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # exponential families a_i ∝ exp(-c * i / n)
    for c in np.linspace(0.5, 4.0, 71):          # 71 candidates

        a /= a_sum
        sc = _score_fast(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
            if best_sc < 1.50274349:

    # power‑law families a_i ∝ (i+1)^{-p}

        if sc < best_sc:

            if best_sc < 1.50274349:

    # combined families a_i ∝ exp(-c*i/n) * (i+1)^{-p}
