   start = time.monotonic()
   deadline = start + max(0.0, float(budget_s) - 0.1)

        else:
            a0 = a0 / a0.sum()

        a0 = np.exp(-np.arange(n0)/100.0)
        a0 = a0 / a0.sum()

   n = a0.size
   L = 1 << ((2*n - 1).bit_length())
   cur = a0.copy()
   cur_sc = _score(cur)
   best_vec = cur.copy()
   best_sc = cur_sc

   # parameters
   beta = 0.95    # momentum
   momentum = np.zeros_like(cur)
   eps = None
   eps_decay = 0.999
   base_alpha = 0.8   # base step size scale factor
   max_trials = 20
   sym_interval = 300
   no_improve = 0
   improve_target = 1.50274349

   iteration = 0

   while time.monotonic() < deadline:
        iteration += 1

        # FFT conv
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()

        # epsilon schedule
        if eps is None:
            eps = max(max_val * 0.1, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)

        # softmax weights
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()

        else:
            w /= w_sum

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)

        # momentum update
        momentum = beta * momentum + (1 - beta) * grad

        # compute step size: adapt based on gradient norm
        norm_m = np.linalg.norm(momentum) + 1e-12

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound for a sequence whose sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(conv) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
