import time, math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / n)
    return w / s

            eps = max(eps * eps_decay, max_val * 1e-12)

        # compute softmax weights for the smoothed max
        if eps < max_val * 1e-6 or it > 5000:
            mask = (conv_full == max_val)
            w = mask.astype(np.float64)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(conv_full, 1.0 / conv_full.size)
            else:
                w /= w_sum
        else:
            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv_full, 1.0 / conv_full.size)
            else:
                w /= w_sum

        # gradient of the smoothed max (matrix‑free via FFT)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)   # slice corresponding to each a_i
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # adaptive step size
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / (math.sqrt(it) * norm_m)

        # backtracking line‑search
        improved = False
        cand = None
        cand_sc = None
        a_alpha = alpha
        for _ in range(max_backtrack):
            cand_try = cur - a_alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try, L)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            a_alpha *= 0.5

        # occasional uphill move (simulated annealing)
        if not improved:
            cand_try = cur - a_alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try, L)
            temp = max(1e-3, 1.0 / math.sqrt(it))
            prob = math.exp(-(cand_sc - cur_sc) / temp)
            if rng.random() < prob:
                cand = cand_try
                improved = True

        # accept if improved
        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---------- periodic symmetrisation ----------
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ---------- random mass transfer when stuck ----------
        if no_improve > 800:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0

        # ---------- restart from best if hopeless ----------
        if no_improve > 3000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            eps = None
            no_improve = 0

        # early exit if target achieved
        if best_sc < target:
            break

        # safety time check
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------
    # Final refinement: intensive random local moves
    # ------------------------------------------------------------
    while time.monotonic() < deadline - 0.05:
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        delta = cur[i] * rng.random() * 0.07
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand, L)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            cur = cand
            cur_sc = cand_sc

    # Return as plain Python list of floats
    return [float(x) for x in best_vec]
