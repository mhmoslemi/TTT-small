import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalise (numerical safety)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w[:] = 1.0 / n
    return w


def _conv_max_and_index(a: np.ndarray, L: int):
    """Return (max value, index of max, full convolution a*a of length 2n‑1)."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) beats the target bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # small safety margin

    # -----------------------------------------------------------------
    # initialise a feasible point (simplex, sum = 1)
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
    else:
        # fallback: uniform vector of modest length
        n0 = 2000
        a = np.full(n0, 1.0 / n0)

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # -----------------------------------------------------------------
    # first evaluation
    # -----------------------------------------------------------------
    cur_max, cur_idx, cur_conv = _conv_max_and_index(a, L)
    cur_sc = 2.0 * n * cur_max / (a.sum() ** 2)   # sum(a) == 1 -> just 2*n*max
    best_vec = a.copy()
    best_sc = cur_sc

    # -----------------------------------------------------------------
    # optimisation parameters
    # -----------------------------------------------------------------
    beta = 0.82                  # momentum decay
    base_alpha = 1.0             # base step‑size factor
    max_backtrack = 5            # back‑tracking attempts
    sym_interval = 100           # symmetrisation frequency
    transfer_thresh = 400        # when to try a random mass transfer
    reinit_thresh = 2000         # when to restart from the best point
    target_score = 1.50274349
    top_k = 5                    # number of top convolution entries used for sub‑gradient
    indices = np.arange(n)
    momentum = np.zeros_like(a)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # -------------------------------------------------------------
        # sub‑gradient from the largest convolution entries
        # -------------------------------------------------------------
        # extract the indices of the top‑k convolution values
        if top_k == 1:
            ks = np.array([cur_idx], dtype=int)
        else:
            # np.argpartition is linear‑time on average
            ks = np.argpartition(-cur_conv, top_k - 1)[:top_k]

        grad = np.zeros_like(a)
        for k_idx in ks:
            shift = k_idx - indices
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # adaptive step size (norm‑scaled)
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / (math.sqrt(it) * norm_m)

        # -------------------------------------------------------------
        # back‑tracking line‑search
        # -------------------------------------------------------------
        accepted = False
        cand = None
        cand_sc = None
        alpha_try = alpha
        for _ in range(max_backtrack):
            cand_try = a - alpha_try * momentum
            cand_try = _project_simplex(cand_try)

            max_val, _, _ = _conv_max_and_index(cand_try, L)
            sc = 2.0 * n * max_val / (cand_try.sum() ** 2)

            if sc < cur_sc:
                cand, cand_sc = cand_try, sc
                accepted = True
                break
            alpha_try *= 0.5

        # -----------------------------------------------------------------
        # occasional uphill move (simulated annealing style)
        # -----------------------------------------------------------------
        if not accepted:
            # try one more candidate with the smallest step tried
            cand_try = a - alpha_try * momentum
            cand_try = _project_simplex(cand_try)
            max_val, _, _ = _conv_max_and_index(cand_try, L)
            sc = 2.0 * n * max_val / (cand_try.sum() ** 2)
            # temperature decays with iteration count
            temp = 1e-3 + 1e-3 / math.sqrt(it)
            prob = math.exp(-(sc - cur_sc) / max(temp, 1e-12))
            if rng.random() < prob:
                cand, cand_sc = cand_try, sc
                accepted = True

        # -------------------------------------------------------------
        # accept if improvement (or uphill accepted)
        # -------------------------------------------------------------
        if accepted:
            a = cand
            cur_sc = cand_sc
            cur_max, cur_idx, cur_conv = _conv_max_and_index(a, L)

            if cand_sc < best_sc:

        # -------------------------------------------------------------
        # periodic symmetrisation (helps keep the shape regular)

        # -------------------------------------------------------------
        if no_improve > transfer_thresh:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.6
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                max_val, idx, conv_arr = _conv_max_and_index(tmp, L)
                tmp_sc = 2.0 * n * max_val / (tmp.sum() ** 2)
                if tmp_sc < cur_sc:
                    a = tmp
                    cur_sc = tmp_sc
                    cur_max, cur_idx, cur_conv = max_val, idx, conv_arr
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                else:
                    no_improve = max(0, no_improve - 150)

        # -------------------------------------------------------------
        # restart from the best point after long stagnation
        # -------------------------------------------------------------
        if no_improve > reinit_thresh:
            a = best_vec + 0.01 * rng.random(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            momentum = np.zeros_like(a)
            no_improve = 0

        # -------------------------------------------------------------
        # early exit if target already achieved
        # -------------------------------------------------------------
        if best_sc < target_score:
            break

    # Return the best sequence discovered (as plain Python floats)
    return [float(x) for x in best_vec]
