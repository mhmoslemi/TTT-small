def _score_full(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2 * n - 1]
    max_val = conv.max()
    sum_a = v.sum()
    if sum_a == 0:
        return float('inf')
    return float(2.0 * n * max_val / (sum_a * sum_a))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # init vector
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: uniform distribution length 2000
        n0 = 2000
        a0 = rng.random(n0, dtype=np.float64)

    # ensure nonnegative
    a0 = np.maximum(a0, 0.0)
    # project onto simplex (sum to 1)
    cur = _project_simplex(a0)

    n = cur.size
    L = 1 << ((2*n - 1).bit_length())

    # best tracking
    best_vec = cur.copy()
    best_sc = _score_full(best_vec)

    # optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1

    iteration = 0
    no_improve = 0
    k = 2.0
    k_max = 30.0
    k_inc = 0.01  # per iteration increase

    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        iteration += 1

        # compute conv and gradient weights
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()
        # smoothing weight
        # Shift for numerical stability (max)
        w = np.exp(k * (conv_full - max_val))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # gradient via convolution with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]  # subgradient of smoothed max

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        update = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - update
        cand = _project_simplex(cand)

        cand_sc = _score_full(cand)

        # acceptance
        if cand_sc < best_sc:
            best_vec, best_sc = cand.copy(), cand_sc

        if cand_sc < cur_sc := _score_full(cur):
            cur = cand
            no_improve = 0
        else:
            no_improve += 1
            # occasionally accept worse with small prob
            if rng.random() < 0.001:
                cur = cand

        # symmetrisation occasional
        if iteration % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_full(sym)
            if sym_sc < best_sc:
                best_vec, best_sc = sym.copy(), sym_sc
            if sym_sc < cur_sc:
                cur = sym

        # random redistribution when stuck
        if no_improve > 2000:
            # random small perturbation on two coordinates
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = cur[i] * rng.random() * 0.3
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_full(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
            no_improve = 0

        # increase smoothing parameter
        if k < k_max:
            k = min(k_max, k + k_inc)

        if best_sc < target:
            break

    # final fallback in case no improvement
    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_full(a):
    """Exact AC1 bound for a non‑negative vector a."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_val / (s * s))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # initial vector
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(a0, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # best solution tracking
    best_vec = cur.copy()
    best_sc = _score_full(best_vec)

    # Adam optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12

    # smoothing parameter for the max (soft‑max)
    k = 2.0
    k_max = 30.0
    k_inc = 0.01

    # loop control
    it = 0
    no_improve = 0
    target = 1.50274349
    cur_sc = best_sc

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ---- convolution of current iterate ----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_sc = 2.0 * n * max_val  # sum(cur)=1 after projection

        # ---- soft‑max weights over convolution entries ----
        # use max for numerical stability
        w = np.exp(k * (conv_full - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smoothed max ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]        # sub‑gradient vector

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score_full(cand)

        # ---- acceptance ----
        if cand_sc < cur_sc:

            no_improve = 0
        else:

                cur, cur_sc = cand, cand_sc

        # ---- update best ----
        if cand_sc < best_sc:
            best_vec, best_sc = cand.copy(), cand_sc

        # ---- periodic symmetrisation ----
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_full(sym)
            if sym_sc < best_sc:

            if sym_sc < cur_sc:

                no_improve = 0

        if best_sc < target:
