def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s)-0.2)
    # get n
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2500   # assume fallback length; we can compute from later
        a = np.exp(-np.arange(n)/120.0)
    # ensure nonnegative and sum>0.01
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12: 
        a = np.full_like(a, 1.0 / a.size)
        s = 1.0
    a = a / s   # sum = 1
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    # defines scoring functions
    def max_conv(x):
        af = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return conv.max()
    cur_max = max_conv(a)
    cur_score = 2.0 * n * cur_max   # denominator = 1
    best_a = a.copy()
    best_score = cur_score
    # parameters
    total_iters = 200000   # huge but will break on time
    T0 = 0.1
    cooling = 0.9999
    # decide max transfer fraction
    max_transfer_frac = 0.5
    # main loop
    it = 0
    while time.monotonic() < deadline and it < total_iters:
        it += 1
        # choose pair
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i: j += 1
        if a[i] == 0:
            continue
        delta = a[i] * rng.random() * max_transfer_frac   # transfer from i to j
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # eval
        max_new = max_conv(a_new)
        new_score = 2.0 * n * max_new
        # acceptance
        T = T0 * (cooling ** it)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            cur_max = max_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        # occasional symmetrize
        if it % 5000 == 0:
            sym = (a + a[::-1]) * 0.5
            # ensure sum to 1 (still sum 1)
            # but sym may not sum 1 if rounding errors: re-normalize
            sym = sym / sym.sum()
            max_sym = max_conv(sym)
            sym_score = 2.0 * n * max_sym
            if sym_score < cur_score:
                a = sym
                cur_score = sym_score
                cur_max = max_sym
            if sym_score < best_score:
                best_score = sym_score
                best_a = sym.copy()
        # occasional random restart noise
        if it % 20000 == 0:
            # add small Gaussian noise and reproject onto simplex
            noise = 0.01 * rng.normal(size=n)
            a_noise = a + noise
            a_noise = np.maximum(a_noise, 0.0)
            a_noise = a_noise / a_noise.sum()
            nscore = 2.0 * n * max_conv(a_noise)
            if nscore < best_score:
                best_a = a_noise.copy()
                best_score = nscore
            a = a_noise
            cur_score = nscore
    # after loop, return best
    # ensure nonnegative and sum>0.01 (it will be)
    return [float(x) for x in best_a]

    start = time.monotonic()

    # ------------------------------------------------------------------

    if "height_sequence_1" in globals():

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / a.size)
            s = 1.0
        a = a / s                     # now sum(a) == 1
    else:                               # fallback: simple exponential decay
        n_fallback = 2500
        t = np.arange(n_fallback, dtype=np.float64)
        a = np.exp(-t / 120.0)
        a = a / a.sum()

    n = a.shape[0]
    # power‑of‑two length for FFT (covers full convolution length)
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # helper: maximum of the full self‑convolution (no scaling)

    def _max_conv(x: np.ndarray) -> float:
        af = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(conv.max())

    cur_max = _max_conv(a)
    cur_score = 2.0 * n * cur_max          # denominator is 1 because sum(a)=1
