def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    # Determine n
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
    else:
        n_fallback = 2000
        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 100.0)
        a0 = a0 / a0.sum()
    n = a0.shape[0]
    best_vec = a0.copy()
    best_sc = _score(best_vec)

    # Helper to evaluate & keep best
    def _try(v):
        nonlocal best_vec, best_sc
        sc = _score(v)
        if sc < best_sc:
            best_sc = sc
            best_vec = v.copy()
        return sc

    # Parameter families
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    for _ in range(150):
        alpha = 10 ** rng.uniform(-4, 0)
        cand = np.exp(-alpha * idx)
        cand = cand / cand.sum()
        _try(cand)
    for _ in range(150):
        alpha = 10 ** rng.uniform(-4, 0)
        d = np.minimum(idx, n - 1 - idx)
        cand = np.exp(-alpha * d)
        cand = cand / cand.sum()
        _try(cand)
    for _ in range(150):
        sigma = rng.uniform(0.5, 60.0)
        cand = np.exp(-((idx - mid) ** 2) / (2 * sigma ** 2))
        cand = cand / cand.sum()
        _try(cand)
    for _ in range(150):

            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            no_improve = 0

    return [float(x) for x in best_vec]

def _score(a):
    n = a.shape[0]
    if a.sum() <= 0:
        return float('inf')
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (scale‑invariant)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative sequence a."""
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))
