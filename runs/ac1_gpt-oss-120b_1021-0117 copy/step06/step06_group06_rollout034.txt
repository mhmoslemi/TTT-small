import time, math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 score for a vector normalised to sum = 1
# ------------------------------------------------------------
def _score(a: np.ndarray, n: int, L: int, idx_slice: np.ndarray) -> float:
    """Return 2 * n * max(conv) where conv = a * a (full linear convolution)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
