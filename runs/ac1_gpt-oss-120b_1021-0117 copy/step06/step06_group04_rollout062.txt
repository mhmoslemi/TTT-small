def gradient(a, max_idx):
    # compute gradient vector g (size n)
    # g[i] = 2 * a[max_idx - i] if 0 <= max_idx - i < n else 0
    shift = max_idx - np.arange(n)
    mask = (shift >= 0) & (shift < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[shift[mask]]
    return g

def propose_candidate(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)
    # initialization
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
    else:
        # fallback: use Hann window
        n = 2500
        x = np.arange(n)
        a = 0.5 * (1 - np.cos(2 * math.pi * x / (n - 1)))  # Hann
        a = a / a.sum()
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    # compute initial score
    def conv_max_and_idx(v):
        af = np.fft.rfft(v, L)
        conv = np.fft.irfft(af * af, L)[: 2 * n - 1]
        idx = int(conv.argmax())
        return float(conv[idx]), idx, conv
    max_val, max_idx, cur_conv = conv_max_and_idx(a)
    cur_score = 2.0 * n * max_val
    best_vec = a.copy()
    best_score = cur_score
    # parameters
    eta0 = 0.5
    max_line = 8
    it = 0
    no_imp = 0
    # momentum for gradient descent
    momentum = np.zeros_like(a)
    beta = 0.85
    while time.monotonic() < deadline:
        it += 1
        # compute gradient at current max idx
        # gradient of max conv (subgradient) = 2 * a shifted
        shift = max_idx - np.arange(n)
        mask = (shift >= 0) & (shift < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[shift[mask]]
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        # adaptive step size
        norm_m = np.linalg.norm(momentum) + 1e-12
        eta = eta0 / (math.sqrt(it) * norm_m)
        # try gradient descent candidate
        improved = False
        cand = None
        cand_score = None
        eta_try = eta
        for _ in range(max_line):
            # gradient descent update with projection onto simplex
            cand_try = a - eta_try * momentum
            cand_try = _project_simplex(cand_try)
            max_val_t, _, _ = conv_max_and_idx(cand_try)
            score_t = 2.0 * n * max_val_t
            if score_t < cur_score:
                cand, cand_score = cand_try, score_t
                improved = True
                break
            eta_try *= 0.5
        # try mirror descent candidate (exponential weights) if not improved
        if not improved:
            eta_try = eta
            for _ in range(max_line):
                exp_factor = np.exp(-eta_try * momentum)
                cand_try = a * exp_factor
                cand_sum = cand_try.sum()
                if cand_sum > 0:
                    cand_try = cand_try / cand_sum
                else:
                    # fallback uniform
                    cand_try = np.full_like(a, 1.0 / n)
                max_val_t, _, _ = conv_max_and_idx(cand_try)
                score_t = 2.0 * n * max_val_t
                if score_t < cur_score:
                    cand, cand_score = cand_try, score_t
                    improved = True
                    break
                eta_try *= 0.5
        # simulated annealing acceptance for uphill move
        if not improved:

            # use current eta as temperature

            cand_try = a + jitter

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_index(a: np.ndarray, L: int):
    """Return (max value, its index, full convolution a*a)."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv
