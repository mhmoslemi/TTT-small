def _optimize(a0, start_time, deadline):
    cur = a0.copy()
    n = cur.size
    L = 1 << ((2*n-1).bit_length())
    best = cur.copy()
    best_sc = _score(cur)
    momentum = np.zeros_like(cur)
    beta = 0.9
    it = 0
    eps = None
    eps_decay = 0.995
    base_alpha = 1.0
    while time.monotonic() < deadline:
        it += 1
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        # compute epsilon if not set
        if eps is None:
            eps = max_val * 0.1
        else:
            eps = max(eps * eps_decay, 1e-12)
        # weight factor for active gradient
        tau = min(1.0, it / 2000.0)  # gradually increase to 1
        # compute softmax weight
        w_soft = np.exp((conv - max_val) / eps)
        w_soft_sum = w_soft.sum()
        if w_soft_sum == 0:
            w_soft = np.full_like(conv, 1.0 / conv.size)
        else:
            w_soft /= w_soft_sum
        # compute active mask
        tol = max_val * 1e-9 + 1e-12
        w_mask = (conv >= max_val - tol).astype(float)
        mask_sum = w_mask.sum()
        if mask_sum > 0:
            w_mask /= mask_sum
        else:
            w_mask = np.zeros_like(conv)
        # combine
        w = (1 - tau) * w_soft + tau * w_mask
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n-2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # momentum
        momentum = beta * momentum + (1 - beta) * g
        # step size

import time, math, numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of 2*n*max(convolution) for sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
def _optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple:

    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_sc = cur_sc

    momentum = np.zeros_like(cur)
    beta = 0.9
    eps = None
    eps_decay = 0.995
    base_alpha = 1.0
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # smoothing temperature
        if eps is None:
            eps = max_val * 0.1
        else:
            eps = max(eps * eps_decay, 1e-12)

        # mixing factor: move from soft to exact subgradient
        tau = min(1.0, it / 2000.0)

        # softmax weights
        w_soft = np.exp((conv - max_val) / eps)
        w_sum = w_soft.sum()

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # Determine problem size
    if "height_sequence_1" in globals():
        n = int(height_sequence_1.shape[0])
    else:

    target = 1.50274349

    # Run optimisation on each start, allocating remaining time proportionally
    for idx, init in enumerate(init_vecs):
