import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # ensure sum exactly 1 (avoid tiny rounding loss)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a):
    """Score for a vector already normalized to sum=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.4)   # safety margin

    # ------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2000, dtype=np.float64) + 1e-3
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # optimizer state
    momentum = np.zeros_like(cur)
    beta = 0.9                     # momentum coefficient
    step0 = 0.6                    # initial step scale
    it = 0
    no_improve = 0

    # soft‑max temperature schedule for the max
    alpha_max = 80.0
    alpha0 = 1.0

    target = 1.50274349           # desired bound, for early stop

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- FFT of current vector -----
        a_fft = np.fft.rfft(cur, n=L)

        # ----- convolution (auto‑correlation) -----
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weighting of the convolution entries -----
        # alpha grows slowly, focusing more on the true maximum later
        alpha = min(alpha_max, alpha0 * math.log1p(it))
        # shift by max for numerical stability
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0:
            # fallback to uniform mask if overflow occurs
            mask = conv >= max_val - (max_val * 1e-12 + 1e-15)
            w = mask.astype(np.float64)
            cnt = w.sum()
            if cnt > 0:
                w /= cnt
        else:
            w /= w_sum

        # ----- gradient of weighted max‑convolution -----
        # gradient = 2 * (a * rev(w))  via FFT (matrix‑free)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
