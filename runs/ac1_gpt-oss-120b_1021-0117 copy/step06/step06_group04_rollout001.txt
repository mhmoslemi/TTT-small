def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (non‑negative, sum=1)."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_idx(a: np.ndarray, L: int):
    """Return (max value, its index, full convolution) of a with itself."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv

        # ---------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)

        # ---------------------------------------------------------
        # evaluate candidate
        # ---------------------------------------------------------
        cand_max, _, _ = _conv_max_and_idx(cand, L)
        cand_sc = 2.0 * n * cand_max

        # ---------------------------------------------------------
        # acceptance rule (downhill + occasional uphill)
        # ---------------------------------------------------------
        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            # simple temperature schedule
            T = 0.02 * math.exp(-t / 3000.0)
            prob = math.exp(-(cand_sc - cur_sc) / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            no_improve += 1

        # ---------------------------------------------------------
        # periodic symmetrisation (helps regularity)
        # ---------------------------------------------------------
        if t % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_max, _, _ = _conv_max_and_idx(sym, L)
            sym_sc = 2.0 * n * sym_max
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ---------------------------------------------------------
        # random mass transfer when stalled
        # ---------------------------------------------------------
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_max, _, _ = _conv_max_and_idx(tmp, L)
                tmp_sc = 2.0 * n * tmp_max
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = max(0, no_improve - 500)

        # ---------------------------------------------------------
        # hard restart if really stuck
        # ---------------------------------------------------------
        if no_improve > 5000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur /= cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            t = 0

        # ---------------------------------------------------------
        # early exit if target reached
        # ---------------------------------------------------------
        if best_sc < target_score:
            break

        # safety break near deadline
        if time.monotonic() > deadline - 0.1:
            break

    # -------------------------------------------------------------
    # Light polishing: tiny random relocations (few attempts)
    # -------------------------------------------------------------
    attempts = 200
    while attempts > 0 and time.monotonic() < deadline:
        attempts -= 1
        i = rng.integers(0, n)
        if best_vec[i] <= 0.0:
            continue
        delta = best_vec[i] * rng.random() * 0.02
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_max, _, _ = _conv_max_and_idx(cand, L)
        cand_sc = 2.0 * n * cand_max
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()

    # -------------------------------------------------------------
    # Return as plain Python list (float conversion)
    # -------------------------------------------------------------
    return [float(x) for x in best_vec]
