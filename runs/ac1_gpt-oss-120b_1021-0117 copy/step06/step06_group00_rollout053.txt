import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a, L, n):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())
    
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)
    # init
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2500
        cur = rng.random(n, dtype=np.float64)
    cur = _project_simplex(cur)
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    # best
    best_vec = cur.copy()
    best_sc = _score(best_vec, L, n)
    cur_sc = best_sc
    # optimizer
    momentum = np.zeros_like(cur)
    beta_mom = 0.9
    step0 = 0.8
    beta_temp = 1.0
    beta_temp_inc = 1.001
    max_beta_temp = 50.0
    it = 0
    no_improve = 0
    target = 1.50274349
    # loop
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1
        # compute convolution
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_conv = conv.max()
        # softmax weights
        beta_temp = min(max_beta_temp, beta_temp * beta_temp_inc)
        w = np.exp(beta_temp * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.zeros_like(w)
            w[n-1] = 1.0  # put weight at centre
        else:
            w /= w_sum
        # gradient via convolution with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]
        # momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
        # step size
        step = step0 / math.sqrt(it)
        cand = cur - step * momentum
        cand = _project_simplex(cand)
        cand_sc = _score(cand, L, n)
        # accept if better
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            # simulated annealing acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
        # symmetrization occasionally
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0
        # random jitter when stuck
        if no_improve > 3000:
            # perturb a few entries
            idxs = rng.choice(n, size=min(5, n), replace=False)
            perturb = (rng.random(idxs.shape) - 0.5) * 0.1
            tmp = cur.copy()
            tmp[idxs] = np.maximum(tmp[idxs] + perturb, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, L, n)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
        # early exit
        if best_sc < target:
            break
    # return best
    return [float(x) for x in best_vec]
