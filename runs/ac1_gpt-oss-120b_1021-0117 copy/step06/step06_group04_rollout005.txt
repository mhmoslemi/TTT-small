def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math, random
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety
    # Build initial vector a0
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a0 = np.ones(n, dtype=np.float64) / n
    # make nonnegative and sum normalized to 1
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0:
        a0[:] = 1.0 / a0.size
    else:
        a0 = a0 / s
    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    # compute initial conv & score
    cur = a0.copy()
    conv = np.fft.irfft(np.fft.rfft(cur, L) * np.fft.rfft(cur, L), L)[:2*n-1]
    max_val = conv.max()
    cur_sc = 2.0 * n * max_val / (cur.sum() ** 2)   # but sum==1
    best_vec = cur.copy()
    best_sc = cur_sc

    # Optimization parameters
    momentum = np.zeros_like(cur)
    iter_num = 0
    no_improve = 0
    target_score = 1.50274349
    # Determine thresholds for random moves
    RESTART_NO_IMPROVE = 3000
    MASS_TRANSFER_NO_IMPROVE = 1500
    # Use adaptable step factor
    while time.monotonic() < deadline:
        iter_num += 1
        # compute convolution and top indices
        # compute conv (full) using FFT
        af = np.fft.rfft(cur, L)
        conv = np.fft.irfft(af * af, L)[:2*n-1]
        max_val = conv.max()
        # compute score
        cur_sc = 2.0 * n * max_val  # sum=1
        # early exit if already good
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_sc < target_score:
            break

        # Compute gradient at top K indices
        # Get top indices using argpartition
        K = 5
        if K < conv.size:
            topk_idx = np.argpartition(-conv, K-1)[:K]  # get K largest
        else:
            topk_idx = np.arange(conv.size)
        # compute weights
        top_vals = conv[topk_idx]
        # use exponential weighting
        beta_exp = 10.0
        weights_raw = np.exp(beta_exp * (top_vals - max_val))   # max gets 1
        # Normalize
        if weights_raw.sum() == 0:
            weights = np.ones_like(weights_raw) / len(weights_raw)
        else:
            weights = weights_raw / weights_raw.sum()
        # Compute aggregated gradient
        grad = np.zeros_like(cur)
        indices = np.arange(n)
        for idx, w in zip(topk_idx, weights):
            j = idx - indices
            mask = (j >= 0) & (j < n)
            # grad contribution: 2*a[j] for mask
            grad[mask] += w * 2.0 * cur[j[mask]]
        # momentum update
        mom_coef = 0.9
        momentum = mom_coef * momentum + (1 - mom_coef) * grad

        # step size adaptation
        norm_m = np.linalg.norm(momentum) + 1e-12
        # base step decreasing with sqrt(iter)
        step = 0.5 / (math.sqrt(iter_num) * norm_m)

        # backtracking line search
        improved = False
        cand = None
        cand_sc = None
        alpha_try = step
        for _ in range(6):
            cand_try = cur - alpha_try * momentum
            # project onto simplex
            # using projection function defined later
            cand_try = _project_simplex(cand_try)
            # compute score quickly

            conv_c = np.fft.irfft(af_c * af_c, L)[:2*n-1]
            max_c = conv_c.max()
            sc = 2.0 * n * max_c
            if sc < cur_sc:
                cand = cand_try
                cand_sc = sc
                improved = True
                break
            alpha_try *= 0.5

            # try one more step with same alpha_try (maybe small)
            cand_try = cur - alpha_try * momentum
            cand_try = _project_simplex(cand_try)
            af_c = np.fft.rfft(cand_try, L)

import time

import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
