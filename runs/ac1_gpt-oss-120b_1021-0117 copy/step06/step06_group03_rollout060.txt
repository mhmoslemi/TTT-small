import time, math, numpy as np

# ------------------------------------------------------------
# Utility: project onto the probability simplex (used only for fallback)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Compute convolution max and gradient of the smoothed max
# ------------------------------------------------------------
def _conv_and_grad(a: np.ndarray, L: int, eps: float):
    """
    Returns (max_val, gradient, full_conv) where
    max_val is max of a*a, gradient is d/d a of the smoothed max,
    and full_conv is the linear convolution array (length 2*len(a)-1).
    """
    n = a.size
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()

    # soft‑max weights (smooth approximation of the max)
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient: 2 * (a * w_rev)  (circular convolution via FFT)
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad, conv

    best_score = cur_score

    # optimisation hyper‑parameters
    lr0 = 0.8                     # base learning rate for multiplicative update
    eps_decay = 0.9995
    eps = max(cur_max * 0.2, 1e-12)
    sym_interval = 200
    target = 1.50274349
    no_improve = 0
    it = 0

    # first gradient
    cur_max, grad, _ = _conv_and_grad(a, L, eps)

    while time.monotonic() < deadline:
        it += 1

        # ----- exponential‑gradient (mirror‑descent) step -----
        g_centered = grad - grad.mean()
        step = lr0 / math.sqrt(it + 1.0)
        a = a * np.exp(-step * g_centered)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # ----- recompute max‑convolution and gradient for the new iterate -----
        cur_max, grad, _ = _conv_and_grad(a, L, eps)

        # ----- adapt smoothing temperature -----
        eps = max(eps * eps_decay, 1e-12)

        cur_score = 2.0 * n * cur_max

        # ----- keep the best observed candidate -----
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- periodic symmetrisation (helps flatten the profile) -----
        if it % sym_interval == 0:
            sym = (a + a[::-1]) * 0.5
            sym = sym / sym.sum()
            sym_max = _max_conv_val(sym, L)
            sym_score = 2.0 * n * sym_max
            if sym_score < best_score:
                best_score = sym_score
                best_a = sym.copy()
                a = sym
                cur_max, grad, _ = _conv_and_grad(a, L, eps)
                no_improve = 0

        # ----- random mass transfer when progress stalls -----
        if no_improve > 1200:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                cur_max, grad, _ = _conv_and_grad(a, L, eps)
                cur_score = 2.0 * n * cur_max
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                no_improve = 0

        # ----- restart from the best with small noise if hopeless -----
        if no_improve > 3000:
            a = best_a + 0.01 * rng.random(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            cur_max, grad, _ = _conv_and_grad(a, L, eps)
            no_improve = 0

        # ----- early exit if target reached -----
        if best_score < target:
            break

    # Return the best sequence found (as plain Python floats)
    return [float(x) for x in best_a]
