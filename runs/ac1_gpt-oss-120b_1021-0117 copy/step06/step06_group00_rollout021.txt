import time, math, numpy as np

def _project_simplex(v):
    # v re-ordered as numpy array
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    # find rho
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = a0.size
    else:
        n = 2000
        a0 = rng.random(n, dtype=np.float64)

    cur = _project_simplex(np.maximum(a0, 0.0))
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)
    best = cur.copy()
    best_sc = _score(best)

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5   # base learning rate factor

    # scheduling
    it = 0
    no_improve = 0
    target = 1.50274349

    L_fft = 1 << ((2 * n - 1).bit_length())
    # pre-allocate maybe arrays? not necessary.

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L_fft)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L_fft)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature schedule for smoothed max
        # relative temperature decays as 0.05 / sqrt(it)
        temp_rel = max(0.05 / math.sqrt(it), 1e-4)
        T = max_val * temp_rel
        # compute softmax weights
        # shift by max to avoid overflow
        w_raw = np.exp((conv - max_val) / T)
        sum_w = w_raw.sum()
        if sum_w == 0 or np.isnan(sum_w):
            # fallback to uniform mask over exact maxima
            eps_mask = max_val * 1e-12 + 1e-15
            mask = conv >= max_val - eps_mask
            w_raw = mask.astype(np.float64)
            sum_w = w_raw.sum()
        w = w_raw / sum_w

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L_fft)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L_fft)

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Score for a vector already normalized to sum=1 (same as AC1 bound)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
