import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x >= 0, sum x = 1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) whose AC1 bound
    2*n*max(conv(a,a))/sum(a)^2 is as small as possible.
    Returns the best vector found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.perf_counter()
    # leave a small safety margin for cleanup/return
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise a feasible vector (sum = 1, non‑negative)
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2000, dtype=np.float64)
    a = _project_simplex(np.maximum(a, 0.0))

    n = a.size
    # FFT length: must accommodate linear convolution of length 3n‑2
    L = 1 << ((3 * n - 2).bit_length())
    idx = (2 * n - 2) - np.arange(n)          # slice to extract gradient entries

    # Adam optimiser state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.5               # base step size
    smooth_coef = 0.1           # initial coefficient for soft‑max smoothing

    # ------------------------------------------------------------------
    # Best solution seen so far
    best = a.copy()
    best_score = evaluate_sequence([float(x) for x in best])

    target = 1.50274349        # stop early if we beat this bound
    it = 0
    # ------------------------------------------------------------------
    while True:
        if time.perf_counter() >= deadline:
            break
        it += 1

        # ----- convolution a*a (full) -------------------------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]          # linear a*a
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv        # internal score (sum(a)=1)

        # Record best (using the exact evaluator for safety)
        if cur_score < best_score:
            cand_eval = evaluate_sequence([float(x) for x in a])
            if cand_eval < best_score:
                best = a.copy()
                best_score = cand_eval
                if best_score <= target:
                    break

        # ----- smooth approximation of the max (soft‑max) ---------------------------
        smooth_coef = min(20.0, smooth_coef + 0.001)          # slowly increase
        exp_vals = np.exp(smooth_coef * (conv - max_conv))
        w = exp_vals / exp_vals.sum()                        # weights sum to 1

        # ----- gradient of weighted max‑convolution ---------------------------------
        w_rev = w[::-1]                                       # reversed weights
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]                           # shape (n,)

        # ----- Adam update -----------------------------------------------------------
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad ** 2)
        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)
        step = base_lr / (np.sqrt(v_hat) + eps)
        cand = a - step * m_hat

        # ----- project candidate onto simplex ----------------------------------------
        cand = _project_simplex(cand)

        # ----- evaluate candidate (internal score) -----------------------------------
        cand_fft = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_score = 2.0 * n * conv_c.max()

        # ----- accept/reject ---------------------------------------------------------
        if cand_score < cur_score:
            a = cand
        else:
            # occasional probabilistic acceptance (simulated annealing style)
            prob = math.exp(-(cand_score - cur_score) / (base_lr + 1e-12))
            if rng.random() < prob:
                a = cand

        # ----- occasional symmetrisation -------------------------------------------
        if it % 400 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            conv_sym = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]

            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:

                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                a = tmp

    # ------------------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best]
