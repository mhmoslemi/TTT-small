import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative entries with sum Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    w = v.copy()
    # shift by mean to speed up the case where sum already close to Z
    if w.sum() != Z:
        u = np.sort(w)[::-1]
        css = np.cumsum(u) - Z
        rho_arr = u - css / (np.arange(1, len(u) + 1))
        rho = np.nonzero(rho_arr > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(w - theta, 0.0)
    return w

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (power of two)

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    def _obj(vec):
        """Compute the AC1 upper bound for a normalized vector."""
        fft = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
        return 2.0 * n * conv.max()

    best_vec = a.copy()
    best_sc = _obj(a)

    # optimisation parameters
    step0 = 0.9
    beta = 0.85               # momentum factor
    momentum = np.zeros_like(a)
    no_improve = 0
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current objective and convolution -----
        cur_sc = _obj(a)

        # ----- track the best -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_sc < target:
            break

        # ----- gradient from the maximal convolution entries -----
        fft_a = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        max_idx = np.where(conv_full >= max_val - 1e-12)[0]

        g = np.zeros_like(a)
        for k in max_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i = np.arange(i_start, i_end + 1, dtype=int)
            g[i] += 2.0 * a[k - i]

        # ----- momentum and tentative step -----
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        a_cand = _project_simplex(a - step * momentum)

        cand_sc = _obj(a_cand)

        # ----- back‑track line search if needed -----
        if cand_sc >= cur_sc:
            improved = False
            t = step
            for _ in range(5):
                t *= 0.5
                a_tmp = _project_simplex(a - t * momentum)
                tmp_sc = _obj(a_tmp)
                if tmp_sc < cur_sc:
                    a_cand, cand_sc = a_tmp, tmp_sc
                    improved = True
                    break
            if not improved:
                # occasional uphill acceptance
                prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
                if rng.random() < prob:
                    a = a_cand
                    continue
                # else keep current a
                continue

        # ----- accept improvement -----
        a = a_cand

        # ----- periodic symmetrisation -----
        if it % 100 == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            sym_sc = _obj(a_sym)
            if sym_sc < cur_sc:
                a, cur_sc = a_sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    no_improve = 0

        # ----- random mass transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.3
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

        # quick deadline check
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
