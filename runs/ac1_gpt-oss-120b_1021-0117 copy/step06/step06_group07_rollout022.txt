import time, math, numpy as np, cvxpy as cp

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x)=Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with sum Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Compute AC1 bound for a vector (assumes sum=1, but works otherwise)
# ------------------------------------------------------------
def _score_and_conv(a: np.ndarray, L: int):
    """Return (score, max_conv, full_conv) for vector a."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    M = conv_full.max()
    return 2.0 * a.size * M, M, conv_full


# ------------------------------------------------------------
# Exact sub‑gradient of max(conv) for non‑negative a
# ------------------------------------------------------------
def _subgrad_maxconv(a: np.ndarray, conv: np.ndarray, M: float):
    """Return a sub‑gradient of max(conv) at a (scaled by 2)."""
    n = a.size
    # indices attaining the maximum (within tiny tolerance)
    eps = 1e-12
    idxs = np.where(conv >= M - eps)[0]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        g[i_range] += 2.0 * a[k - i_range]
    if idxs.size > 1:
        g /= float(idxs.size)
    return g
