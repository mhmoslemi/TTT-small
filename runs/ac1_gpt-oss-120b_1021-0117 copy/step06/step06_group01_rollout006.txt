# ------------------------------------------------------------
# Simplex projection (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# Fast AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----------------------------------------------------
        # gradient of smoothed max (via convolution)
        # ----------------------------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
        idx = (2 * n - 2) - np.arange(n)          # aligns for gradient
        g = 2.0 * grad_full[idx]

        # ----------------------------------------------------
        # momentum update
        # ----------------------------------------------------
        momentum = beta * momentum + (1.0 - beta) * g

        # ----------------------------------------------------
        # step size (decaying)
        # ----------------------------------------------------
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / math.sqrt(no_improve + 1) / norm_m

        # ----------------------------------------------------
        # backtracking line search
        # ----------------------------------------------------
        improved = False
        cand = None
        cand_sc = None
        alpha = alpha0
        for _ in range(max_trials):
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5

        # ----------------------------------------------------
        # occasional uphill move (probabilistic)
        # ----------------------------------------------------
        if not improved:
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----------------------------------------------------
        # periodic symmetrisation (helps a lot)
        # ----------------------------------------------------
        if (no_improve % sym_interval) == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----------------------------------------------------
        # local pairwise mass‑transfer search when stagnating
        # ----------------------------------------------------
        if no_improve > 1500:
            for _ in range(200):
                i = rng.integers(0, n)
                if cur[i] <= 0.0:
                    continue
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                    break
            else:
                no_improve = max(no_improve - 500, 0)

        # ----------------------------------------------------
        # restart from best with tiny noise if hopeless
        # ----------------------------------------------------
        if no_improve > 5000:
            cur = best_vec + 0.008 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            eps = None
            no_improve = 0

        # ----------------------------------------------------
        # early exit if target reached
        # ----------------------------------------------------
        if best_sc < improve_target:
            break

    # --------------------------------------------------------
    # Return best solution as plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
```
