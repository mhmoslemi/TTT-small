import time, math, numpy as np

def _project_simplex(v):
    # Euclidean projection onto simplex (sum=1, v>=0)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s
    else:
        n = 2000
        a0 = np.exp(-np.arange(n, dtype=np.float64) / 100.0)
        a0 = a0 / a0.sum()

    n = a0.shape[0]
    best_vec = a0.copy()
    best_sc = _score(best_vec)

    # helper
    def _maybe_update(vec):
        nonlocal best_vec, best_sc
        sc = _score(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
        return sc

    # ---- parametric families ----
    # exponential decays
    for _ in range(300):
        alpha = 10 ** rng.uniform(-5, 0)  # 1e-5 .. 1
        cand = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand = cand / cand.sum()
        _maybe_update(cand)

    # symmetric exponential
    for _ in range(300):
        alpha = 10 ** rng.uniform(-5, 0)
        d = np.minimum(np.arange(n, dtype=np.float64),
                       n - 1 - np.arange(n, dtype=np.float64))
        cand = np.exp(-alpha * d)
        cand = cand / cand.sum()
        _maybe_update(cand)

    # Gaussian-like
    mid = (n - 1) / 2.0
    for _ in range(300):
        sigma = rng.uniform(0.5, 80.0)
        cand = np.exp(-((np.arange(n, dtype=np.float64) - mid) ** 2) /
                      (2 * sigma ** 2))
        cand = cand / cand.sum()
        _maybe_update(cand)

    # cosine-squared
    for _ in range(200):
        cand = np.cos(np.pi * np.arange(n, dtype=np.float64) / (n - 1)) ** 2
        cand = cand / cand.sum()
        _maybe_update(cand)

    # triangular (linear up then down)
    for _ in range(200):
        t = np.arange(n, dtype=np.float64)
        mid_i = (n - 1) / 2.0
        cand = np.where(t <= mid_i, t / mid_i, (n - 1 - t) / mid_i)
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        _maybe_update(cand)

    # plateau with variable width
    for _ in range(200):
        w_frac = rng.uniform(0.05, 0.5)  # fraction of length as plateau
        w = int(w_frac * n)
        if w < 1:
            w = 1
        left = (n - w) // 2
        right = left + w
        cand = np.zeros(n, dtype=np.float64)
        cand[left:right] = 1.0
        # taper linearly to edges

            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            a *= 0.5

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)
