def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # init a
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()  # sum=1

    n = a.size
    # FFT length power of two >= 2n-1
    L = 1 << ((2 * n - 1).bit_length())

    # initial score
    cur_max = _conv_max(a, L)
    best_sc = 2.0 * n * cur_max
    best_vec = a.copy()

    # params
    step0 = 0.6
    beta = 0.9
    momentum = np.zeros_like(a)
    it = 0
    sym_interval = 200
    stagnation_counter = 0
    target = 1.50274349

    # iterate
    while time.monotonic() < deadline:
        it += 1
        # compute convolution and max
        af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        cur_max = float(conv_full.max())
        cur_sc = 2.0 * n * cur_max

        # update best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1

        # early exit
        if best_sc < target:
            break

        # gradient of max conv
        grad = _grad_maxconv(a, conv_full)

        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size
        alpha = step0 / math.sqrt(it)

        # projected gradient step
        a_tmp = a - alpha * momentum
        # ensure nonnegative before projection? _project_simplex will clip.
        a_tmp = _project_simplex(a_tmp)

        # evaluate candidate
        cand_max = _conv_max(a_tmp, L)
        cand_sc = 2.0 * n * cand_max

        # accept if improvement
        if cand_sc < cur_sc:
            a = a_tmp
            cur_sc = cand_sc
            # if it's also better than best we will capture above next iteration
        else:
            # sometimes accept uphill
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                a = a_tmp
                cur_sc = cand_sc

        # periodic symmetrisation
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            sym_max = _conv_max(a_sym, L)
            sym_sc = 2.0 * n * sym_max
            if sym_sc < cur_sc:
                a = a_sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = a_sym.copy()
                    stagnation_counter = 0

        # occasional random mass transfer if stuck
        if stagnation_counter > 3000:
            # pick two distinct indices
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0:
                delta = a[i] * rng.random() * 0.5
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                stagnation_counter = 0

    # return best
    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative entries summing to Z."""
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Maximum of the full linear convolution a*a using FFT
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max_k (a*a)[k] using an FFT of length L (≥2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# ------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution functional
# ------------------------------------------------------------
def _grad_maxconv(a: np.ndarray, conv_full: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) at a (vector on simplex)."""
    n = a.size
    max_val = conv_full.max()
    # consider all shifts attaining (near) the maximum
    idx_maxes = np.where(conv_full >= max_val - 1e-12)[0]
    g = np.zeros_like(a)
    for k in idx_maxes:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    if idx_maxes.size > 1:
        g /= idx_maxes.size
    return g

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------

    if "height_sequence_1" in globals():

    else:

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1 (simplex)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    cur_max = _conv_max(a, L)
    best_sc = 2.0 * n * cur_max
    best_vec = a.copy()

    # optimisation parameters
    step0 = 0.6               # base step magnitude
    beta = 0.90               # momentum factor
    momentum = np.zeros_like(a)
    it = 0
    sym_interval = 200        # how often to symmetrise
    stagnation = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        cur_max = float(conv_full.max())
        cur_sc = 2.0 * n * cur_max

        # ----- keep the best -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            stagnation = 0

        grad = _grad_maxconv(a, conv_full)

        a_tmp = a - alpha * momentum
        a_tmp = _project_simplex(a_tmp)        # stay on simplex
