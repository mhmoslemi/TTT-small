def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]

    # --------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length() + 1)
    grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)

    # --------------------------------------------------------
    # Helper to evaluate a raw candidate (normalises it)
    # --------------------------------------------------------
    def _eval(v):
        v = _normalize(v)
        return v, _score(v)

    # --------------------------------------------------------
    # Initialise best solution with simple families
    # --------------------------------------------------------
    best_vec, best_sc = _eval(np.full(n, 1.0 / n))

    # parent if available
    if base is not None:
        vec, sc = _eval(base)
        if sc < best_sc:
            best_vec, best_sc = vec, sc

    # exponential decays
    taus = np.geomspace(5.0, n * 1.5, num=30)
    for tau in taus:
        v = np.exp(-np.arange(n, dtype=np.float64) / tau)
        for cand in (v, v[::-1]):
            vec, sc = _eval(cand)
            if sc < best_sc:
                best_vec, best_sc = vec, sc

    # power‑law tails
    ps = np.linspace(0.1, 2.5, num=30)
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in ps:
        v = 1.0 / (idx ** p)
        for cand in (v, v[::-1]):
            vec, sc = _eval(cand)
            if sc < best_sc:
                best_vec, best_sc = vec, sc

    # linear ramp
    v = np.linspace(1.0, 0.0, n)
    for cand in (v, v[::-1]):
        vec, sc = _eval(cand)
        if sc < best_sc:
            best_vec, best_sc = vec, sc

    # cosine window
    v = 1.0 - np.cos(np.pi * np.arange(n, dtype=np.float64) / (n - 1))
    for cand in (v, v[::-1]):
        vec, sc = _eval(cand)
        if sc < best_sc:
            best_vec, best_sc = vec, sc

    improve_target = 1.50274349
    if best_sc < improve_target:
        return [float(x) for x in best_vec]

    # --------------------------------------------------------
    # Refinement – smoothed‑max gradient descent with Adam steps
    # --------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc
    m = np.zeros_like(cur)          # first moment
    v2 = np.zeros_like(cur)         # second moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.8
    eps_smooth = None
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps_smooth is None:
            eps_smooth = max(max_val * 0.5, 1e-9)
        else:
            eps_smooth = max(eps_smooth * 0.99, 1e-9)

        w = np.exp((conv_full - max_val) / eps_smooth)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[grad_idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v2 = beta2 * v2 + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v2 / (1 - beta2 ** it)
        direction = - m_hat / (np.sqrt(v_hat) + eps_adam)

        # step size (decreasing with sqrt(it) and normalising)
        alpha0 = base_lr / np.sqrt(it) / (np.linalg.norm(direction) + 1e-12)

        # backtracking line‑search
        improved = False
        cand = None
        cand_sc = None
        alpha = alpha0
        for _ in range(12):
            cand_try = cur + alpha * direction
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5

        # fallback random mass‑transfer if line‑search fails
        if not improved:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                cand_try = cur.copy()
                cand_try[i] -= delta
                cand_try[j] += delta
                cand_try = _project_simplex(cand_try)
                cand_sc = _score(cand_try)
                if cand_sc < cur_sc:
                    cand = cand_try
                    improved = True

        if improved:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                no_improve = 0
                if best_sc < improve_target:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional restart from the current best if we stall
        if no_improve > 2500:
            cur = best_vec + 0.01 * rng.random(n)
            cur = _project_simplex(cur)
            m.fill(0.0)
            v2.fill(0.0)
            eps_smooth = None
            it = 0
            no_improve = 0

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
