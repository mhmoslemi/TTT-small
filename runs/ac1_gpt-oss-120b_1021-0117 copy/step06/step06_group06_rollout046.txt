import time, math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 score (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Small mutation operators
# ------------------------------------------------------------
def _mutate(vec: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a perturbed copy of vec (still on the simplex)."""
    n = vec.size
    child = vec.copy()

    # 70% local mass transfer, 30% Gaussian noise
    if rng.random() < 0.7:
        i = rng.integers(0, n)
        if child[i] > 0.0:
            delta = child[i] * rng.random() * 0.35
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            child[i] -= delta
            child[j] += delta
    else:
        noise = rng.normal(scale=0.02, size=n)
        child += noise
        child = np.maximum(child, 0.0)

    # occasional smoothing
    if rng.random() < 0.25:
        kern = np.array([0.25, 0.5, 0.25], dtype=np.float64)
        child = np.convolve(child, kern, mode='same')
        child = np.maximum(child, 0.0)

    return _project_simplex(child)


def _symmetrize(vec: np.ndarray) -> np.ndarray:
    """Force (approximate) symmetry and re‑project onto the simplex."""
    s = (vec + vec[::-1]) * 0.5
    return _project_simplex(s)


# ------------------------------------------------------------
# Local gradient refinement (hard‑max subgradient)
# ------------------------------------------------------------
def _local_refine(vec: np.ndarray, deadline: float) -> np.ndarray:
    cur = vec.copy()
    cur_sc = _score(cur)
    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    base_alpha = 0.25
    it = 0
    while time.monotonic() < deadline - 0.02:
        it += 1
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        mask = (conv == max_val)
        w = mask.astype(np.float64)
        ws = w.sum()
        if ws > 0:
            w /= ws
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        step = base_alpha / (math.sqrt(it) + 1.0)
        cand = cur - step * grad
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
    return cur


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # reserve a tiny safety margin
    deadline = start + max(0.0, float(budget_s) - 0.15)

    # --------------------------------------------------------
    # Determine dimension and basic initial vectors
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        base = np.maximum(base, 0.0)
        n = base.size
    else:
        n = 2000
        base = np.exp(-np.arange(n, dtype=np.float64) / 100.0)

    base = _project_simplex(base)                     # ensure sum == 1

    # a few handcrafted seeds
    seeds = [base]

    # uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential with different decay rates
    for scale in (50.0, 200.0):
        exp_vec = np.exp(-np.arange(n, dtype=np.float64) / scale)
        seeds.append(_project_simplex(exp_vec))

    # Dirichlet draws for diversity
    for alpha in (0.3, 2.0):
        dir_vec = rng.dirichlet(np.full(n, alpha))
        seeds.append(dir_vec)

    # --------------------------------------------------------
    # Build initial population (size popsize)
    # --------------------------------------------------------
    popsize = 12
    population = []
    for v in seeds:
        population.append(v.copy())
        if len(population) >= popsize:
            break
    while len(population) < popsize:
        rand_vec = rng.dirichlet(np.ones(n))
        population.append(rand_vec)

    scores = [_score(v) for v in population]
    best_idx = int(np.argmin(scores))
    best_vec = population[best_idx].copy()
    best_sc = scores[best_idx]

    # --------------------------------------------------------
    # Evolutionary loop
    # --------------------------------------------------------
    no_improve = 0
    target = 1.50274349
    elite_k = 2

    while time.monotonic() < deadline:
        # tournament selection (size 2)
        parents = []
        for _ in range(popsize):
            i, j = rng.integers(0, popsize, size=2)
            parents.append(population[i] if scores[i] < scores[j] else population[j])

        # generate children via mutation
        children = [_mutate(p, rng) for p in parents]
        children_scores = [_score(c) for c in children]

        # combine and keep the best popsize individuals
        combined = population + children
        combined_scores = scores + children_scores
        order = np.argsort(combined_scores)[:popsize]
        population = [combined[i] for i in order]
        scores = [combined_scores[i] for i in order]

        # update global best
        cur_best_idx = int(np.argmin(scores))
        cur_best_sc = scores[cur_best_idx]
        if cur_best_sc + 1e-12 < best_sc:
            best_sc = cur_best_sc
            best_vec = population[cur_best_idx].copy()
            no_improve = 0
        else:
            no_improve += 1

        # occasional symmetrisation of the current best
        if no_improve % 250 == 0 and no_improve > 0:
            sym = _symmetrize(best_vec)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

        # if stagnated, inject a fresh random individual
        if no_improve > 2000:
            rand_vec = rng.dirichlet(np.ones(n))
            rand_sc = _score(rand_vec)
            # replace the worst member
            worst_idx = int(np.argmax(scores))
            population[worst_idx] = rand_vec
            scores[worst_idx] = rand_sc
            no_improve = 0

    # --------------------------------------------------------
    # Final local gradient refinement (use remaining time)
    # --------------------------------------------------------
    if time.monotonic() < deadline - 0.1:
        refined = _local_refine(best_vec, deadline)
        refined_sc = _score(refined)
        if refined_sc < best_sc:
            best_vec, best_sc = refined, refined_sc

    # --------------------------------------------------------
    # Return best found vector as plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
