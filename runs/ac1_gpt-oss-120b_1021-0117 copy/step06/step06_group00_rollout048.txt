def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 2500
        a0 = rng.random(n, dtype=np.float64)

    # ensure nonnegative and project onto simplex
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    # initial best
    best = cur.copy()
    best_sc = _score(cur, n, L)   # compute score function
    cur_sc = best_sc

    # Parameters
    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.8
    beta0 = 0.2  # initial softmax smoothing parameter
    beta_inc = 0.0005    # increase per iteration (example)
    max_beta = 50.0
    no_improve = 0
    target = 1.50274349
    it = 0

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1
        # update smoothing beta
        beta_cur = min(max_beta, beta0 + beta_inc * it)
        # compute FFT of cur
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_conv = conv_full.max()
        # softmax weights
        exps = np.exp(beta_cur * (conv_full - max_conv))
        w = exps / exps.sum()
        # gradient via weighted sum
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        # step size
        step = step0 / math.sqrt(it)
        # candidate
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score(cand, n, L)
        # acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            # SA acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # symmetrize periodically
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc

            i = rng.integers(0, n)
            j = rng.integers(0, n)

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score_vec(a, n, L):
    """Score = 2*n*max(a*a) assuming sum(a)=1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # initial vector (use provided parent if present)
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a0 = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())      # power‑of‑two FFT length

    best = cur.copy()
    best_sc = _score_vec(best, n, L)
    cur_sc = best_sc

    # optimizer state
    momentum = np.zeros_like(cur)
    mom_coeff = 0.9
    step0 = 0.8

    # soft‑max smoothing for the max operator
    beta0 = 0.1          # initial smoothing
    beta_inc = 0.00015   # increase per iteration
    beta_max = 30.0

    idx = (2 * n - 2) - np.arange(n)   # index map for gradient extraction
    no_improve = 0
    target = 1.50274349
    it = 0

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ---- smoothing parameter -------------------------------------------------
        beta_cur = min(beta_max, beta0 + beta_inc * it)

        # ---- convolution of current vector with itself ---------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        max_conv = conv_full.max()

        # ---- soft‑max weights over the convolution entries -----------------------
        w = np.exp(beta_cur * (conv_full - max_conv))
        w /= w.sum()

        # ---- gradient of the smoothed max (weighted sum) ------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # ---- momentum update -----------------------------------------------------
        momentum = mom_coeff * momentum + (1.0 - mom_coeff) * grad

        # ---- step size (decreasing) ---------------------------------------------
        step = step0 / math.sqrt(it)

        # ---- candidate generation -------------------------------------------------
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score_vec(cand, n, L)

        # ---- acceptance ----------------------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ---- periodic symmetrisation --------------------------------------------
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc

        # ---- random redistribution when stuck ------------------------------------
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_vec(tmp, n, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best, best_sc = tmp.copy(), tmp_sc

        # ---- early exit ----------------------------------------------------------
        if best_sc < target:
            break

    # ----------------------------------------------------------------------
    # Return the best found sequence as plain Python floats
    return [float(x) for x in best]
