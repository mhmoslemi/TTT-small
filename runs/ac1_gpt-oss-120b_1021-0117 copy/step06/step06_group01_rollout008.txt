import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (sum of coeffs assumed to be 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(conv(a,a)) for a normalized to sum=1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())


# ------------------------------------------------------------
# Gradient of a smooth approximation to max(conv)
# ------------------------------------------------------------
def _softmax_grad(a: np.ndarray, eps: float, L: int) -> np.ndarray:
    """Return gradient of smoothed max of a*a (convolution)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    max_val = conv.max()
    w = np.exp((conv - max_val) / eps)
    ws = w.sum()
    if ws == 0.0:
        w[:] = 1.0 / w.size
    else:
        w /= ws
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * a.size - 1]
    idx = (2 * a.size - 2) - np.arange(a.size)
    grad = 2.0 * grad_full[idx]
    return grad

        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 100.0)
        a0 = a0 / a0.sum()

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two for FFT

    # ------------------------------------------------------------------
    # Helper to keep the best candidate found so far
    # ------------------------------------------------------------------
    best_vec = a0.copy()
    best_sc = _score(best_vec, L)

    def _try_candidate(vec: np.ndarray):
        nonlocal best_vec, best_sc
        sc = _score(vec, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()

    # ------------------------------------------------------------------
    # Explore several parametric families (cheap, dense search)
    # ------------------------------------------------------------------
    # monotone exponential
    for _ in range(300):
        alpha = 10 ** rng.uniform(-4, 0)          # 1e‑4 … 1
        cand = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand = cand / cand.sum()
        _try_candidate(cand)

    # symmetric exponential (decay from both ends)
    for _ in range(300):
        alpha = 10 ** rng.uniform(-4, 0)
        d = np.minimum(np.arange(n, dtype=np.float64),
                       n - 1 - np.arange(n, dtype=np.float64))
        cand = np.exp(-alpha * d)
        cand = cand / cand.sum()
        _try_candidate(cand)

    # Gaussian‑like
    mid = (n - 1) / 2.0
    for _ in range(300):
        sigma = rng.uniform(0.5, 50.0)
        cand = np.exp(-((np.arange(n, dtype=np.float64) - mid) ** 2)
                      / (2 * sigma ** 2))
        cand = cand / cand.sum()
        _try_candidate(cand)

    # cosine‑squared (high at ends, low in centre)
    for _ in range(300):
        cand = np.cos(np.pi * np.arange(n, dtype=np.float64) / (n - 1)) ** 2
        cand = cand / cand.sum()
        _try_candidate(cand)

    # Laplace‑like (sharper centre)
    for _ in range(300):
        scale = rng.uniform(0.5, 30.0)
        d = np.abs(np.arange(n, dtype=np.float64) - mid)
        cand = 1.0 / (1.0 + d / scale)
        cand = cand / cand.sum()
        _try_candidate(cand)

    # random convex combos of the current best and a fresh exponential
    for _ in range(300):
        alpha = 10 ** rng.uniform(-4, 0)
        rand = np.exp(-alpha * np.arange(n, dtype=np.float64))
        rand = rand / rand.sum()
        t = rng.random()
        cand = (1 - t) * best_vec + t * rand
        cand = _project_simplex(cand)
        _try_candidate(cand)

    # ------------------------------------------------------------------
    # Gradient‑based refinement (soft‑max surrogate of the max)
    # ------------------------------------------------------------------
    target = 1.50274349      # early stop if reached
    max_restarts = 12
    restart = 0
    while time.monotonic() < deadline and restart < max_restarts:
        restart += 1
        # start from best with a small perturbation
        cur = best_vec + 0.02 * rng.standard_normal(n)
        cur = np.maximum(cur, 0.0)
        cur = cur / cur.sum()
        cur_sc = _score(cur, L)

        momentum = np.zeros_like(cur)
        beta = 0.9
        base_alpha = 0.6
        eps = None
        eps_decay = 0.99
        sym_interval = 80
        no_improve = 0
        it = 0
        # limit per restart to keep overall time bounded
        while time.monotonic() < deadline and it < 3500:
            it += 1

            # ----- gradient of smoothed max -----
            a_fft = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
            max_val = conv.max()
            if eps is None:
                eps = max(max_val * 0.2, 1e-9)
            else:
                eps = max(eps * eps_decay, 1e-12)

            w = np.exp((conv - max_val) / eps)
            ws = w.sum()
            if ws == 0.0:
                w[:] = 1.0 / w.size
            else:
                w /= ws

            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # ----- momentum update -----
            momentum = beta * momentum + (1.0 - beta) * grad

            # ----- step size (decaying) -----
            norm_m = np.linalg.norm(momentum) + 1e-12
            alpha = base_alpha / math.sqrt(it) / norm_m

            # ----- candidate step -----
            cand = cur - alpha * momentum
            cand = _project_simplex(cand)

            # ----- periodic symmetrisation (helps a lot) -----
            if it % sym_interval == 0:
                sym = (cand + cand[::-1]) * 0.5
                sym = _project_simplex(sym)
                cand = sym

            cand_sc = _score(cand, L)

            if cand_sc < cur_sc:
                cur = cand
                cur_sc = cand_sc
                no_improve = 0
                _try_candidate(cur)
                if cur_sc < target:
                    break
            else:
                no_improve += 1
                # occasional random mass‑transfer when stuck
                if no_improve % 250 == 0:
                    i = rng.integers(0, n)
                    if cur[i] > 0.0:
                        delta = cur[i] * rng.random() * 0.35
                        j = rng.integers(0, n - 1)
                        if j >= i:
                            j += 1
                        tmp = cur.copy()
                        tmp[i] -= delta
                        tmp[j] += delta
                        tmp = _project_simplex(tmp)
                        tmp_sc = _score(tmp, L)
                        if tmp_sc < cur_sc:
                            cur = tmp
                            cur_sc = tmp_sc
                            no_improve = 0
                            _try_candidate(cur)

            # ----- early global exit -----
            if best_sc < target:
                break

        if best_sc < target:
            break

    # return as plain Python list
    return [float(x) for x in best_vec]
