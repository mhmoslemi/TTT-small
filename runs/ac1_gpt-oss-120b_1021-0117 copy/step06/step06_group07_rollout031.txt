    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0, float(budget_s) - 0.5)  # safety margin

    while time.monotonic() < deadline:

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: maximum of the linear convolution a * a (full)
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max_k (a*a)[k] using FFT of length L (>=2n-1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise from parent if present, otherwise random
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a modest length if no parent is supplied
        a = rng.random(2048, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a[:] = 1.0
    a = a / a.sum()                     # enforce ℓ¹‑norm = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next pow‑2)

    # --------------------------------------------------------
    # Book‑keeping of the best solution seen so far
    # --------------------------------------------------------
    cur_max = _conv_max(a, L)
    cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
    best_vec = a.copy()
    best_score = cur_score

    # --------------------------------------------------------
    # Annealing / hill‑climbing parameters
    # --------------------------------------------------------
    TARGET = 1.50274349
    T0 = 0.08                # initial temperature
    Tmin = 1e-6              # final temperature
    step_start = 0.18        # initial move size (fraction of source mass)
    step_end = 0.001         # final move size
    no_improve = 0
    it = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - start
        # linear schedules
        T = max(Tmin, T0 * (1.0 - elapsed / (deadline - start)))
        step_factor = step_start * (1.0 - elapsed / (deadline - start)) + \
                      step_end * (elapsed / (deadline - start))

        # ----------------------------------------------------
        # Determine the lag with the current maximal autocorrelation
        # ----------------------------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_k = int(conv_full.argmax())

        # choose a source index i that contributes most to this max
        i_low = max(0, max_k - (n - 1))
        i_high = min(n - 1, max_k)
        i_range = np.arange(i_low, i_high + 1, dtype=np.int32)
        j_range = max_k - i_range
        prod = a[i_range] * a[j_range]
        src_i = i_range[int(prod.argmax())]

        if a[src_i] <= 1e-14:
            # fallback to a random source if the chosen one vanished
            src_i = rng.integers(0, n)

        # ----------------------------------------------------
        # Propose a move: shift a small fraction of mass from src_i
        # to a random destination dst_j (different index)
        # ----------------------------------------------------
        delta = a[src_i] * step_factor * rng.random()
        if delta > a[src_i]:
            delta = a[src_i]

        dst_j = rng.integers(0, n - 1)
        if dst_j >= src_i:
            dst_j += 1

        cand = a.copy()
        cand[src_i] -= delta
        cand[dst_j] += delta

        # ----------------------------------------------------
        # Evaluate candidate
        # ----------------------------------------------------
        cand_max = _conv_max(cand, L)
        cand_score = 2.0 * n * cand_max / (cand.sum() ** 2)

        # ----------------------------------------------------
        # Acceptance rule (Metropolis‑style)
        # ----------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # occasional uphill move
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            cur_score = cand_score
            cur_max = cand_max
            no_improve = 0
        else:
            no_improve += 1

        # ----------------------------------------------------
        # Record best solution
        # ----------------------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best_vec = cand.copy()
            no_improve = 0

        # ----------------------------------------------------
        # Periodic symmetrisation (helps to reduce peaks)
        # ----------------------------------------------------
        if it % 500 == 0:
            sym = (a + a[::-1]) * 0.5
            # re‑normalise to keep ℓ¹ = 1
            sym = sym / sym.sum()
            sym_max = _conv_max(sym, L)
            sym_score = 2.0 * n * sym_max / (sym.sum() ** 2)
            if sym_score < best_score:
                best_score, best_vec = sym_score, sym.copy()
            if sym_score < cur_score:
                a, cur_score, cur_max = sym, sym_score, sym_max
                no_improve = 0

        # ----------------------------------------------------
        # Stagnation handling: occasional random reshuffle

        if no_improve > 20000:

        # Early exit if target reached
        # ----------------------------------------------------
        if best_score < TARGET:

    # --------------------------------------------------------
    # Return best vector found (as plain Python list)
    # --------------------------------------------------------
    if best_vec is None or best_vec.size == 0:
        best_vec = a
    return [float(v) for v in best_vec]
