            v_hat.fill(0.0)
            eps_smooth = None
            no_improve = 0

        # ----- early exit if target achieved -----
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------
    # final scaling search exploiting the 0‑1000 clipping
    # ------------------------------------------------------------
    best_scaled = best_vec.copy()
    best_score = best_sc
    if best_vec.max() > 0.0:
        max_scale = 1000.0 / best_vec.max()
        # test a handful of geometric points between 1 and max_scale
        for scale in np.geomspace(1.0, max_scale, num=12):
            cand = best_vec * scale
            cand_list = [float(x) for x in cand]
            sc = evaluate_sequence(cand_list)
            if sc < best_score:
                best_score = sc
                best_scaled = cand.copy()
    # ----------------------------------------------------------------
    # return as plain Python list (clipping is handled by the evaluator)
    # ----------------------------------------------------------------
    return [float(x) for x in best_scaled]
