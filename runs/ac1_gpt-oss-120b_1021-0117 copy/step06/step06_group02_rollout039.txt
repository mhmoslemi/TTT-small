import time, math, numpy as np

# ------------------------------------------------------------
# Project onto the simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

        cand_sc = _fast_score(cand, L, n)

        # ----- accept/reject (Metropolis style) -----
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation -----
        if it % 100 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, L, n)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- aggressive mass transfer when stuck -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.7
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _fast_score(tmp, L, n)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                no_improve = 0

        # ----- restart from the best with small noise if hopeless -----
        if no_improve > 6000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum.fill(0.0)
            eps = None
            no_improve = 0

        # ----- time check -----
        if it % 500 == 0 and time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------
    # Return the best candidate found (as plain Python floats)
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
