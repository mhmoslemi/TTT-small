import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, all >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (assumes sum(a) ≈ 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence a with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        # ------- periodic symmetrisation (helps a lot) -------
        if t_iter % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ------- random perturbation when stuck -------
        if no_improve > random_restart_thresh:
            # small Gaussian noise then re‑project
            noise = 0.02 * rng.standard_normal(n)
            cur = cur + noise
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            # reset Adam moments to avoid stale direction
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            no_improve = 0
            lr = 0.07

        # ------- smooth temperature decay -------
        eps_factor = max(eps_factor * eps_decay, 0.001)

        # early exit if target reached
        if best_sc < 1.50274349:
            break

    # return best found vector as plain Python floats
    return [float(x) for x in best_vec]
