# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray) -> float:
    """AC1 bound for a (assumed non‑negative, sum≈1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _gradient(a: np.ndarray):
    """
    Gradient of a smooth approximation of the max‑convolution.
    Returns (g, max_val) where g has shape (n,).
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    eps = max(max_val * 0.1, 1e-12)
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * w_fft, n=L)[: 2 * n - 1]
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g, max_val

        momentum = beta * momentum + (1.0 - beta) * g
        norm_m = np.linalg.norm(momentum) + 1e-12
        step = base_alpha / math.sqrt(it) / norm_m

        # backtracking line search
        improved = False
        cand = None
        cand_sc = None
        alpha = step
        for _ in range(max_trials):
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # occasional uphill move (Metropolis)
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # periodic symmetrisation (helps convergence)
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # random mass‑transfer when stuck
        if no_improve > 1800:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # restart from best if hopeless
        if no_improve > 4000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            no_improve = 0

        # early exit if target reached
        if best_sc < improve_target:
            break

    # ------------------------------------------------------------------
    # 3) Simple simulated‑annealing fine‑tuning (if time permits)
    # ------------------------------------------------------------------
    if time.monotonic() < deadline - 0.1:
        cur = best_vec.copy()
        cur_sc = best_sc
        T = best_sc * 0.1 + 1e-12
        while time.monotonic() < deadline:
            i = rng.integers(0, n)
            if cur[i] == 0.0:
                continue
            delta = cur[i] * rng.random() * 0.2
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)
            cand_sc = _score(cand)
            dE = cand_sc - cur_sc
            if dE < 0 or rng.random() < math.exp(-dE / (T + 1e-12)):
                cur = cand
                cur_sc = cand_sc
                if cand_sc < best_sc:
                    best_sc = cand_sc
                    best_vec = cand.copy()
            T *= 0.9999
            if T < 1e-12:
                break

    # ------------------------------------------------------------------
    # Final sanity checks & return
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    best_vec = np.minimum(best_vec, 1000.0)
    s = best_vec.sum()
    if s < 0.01:
        best_vec = np.full(n, 1.0 / n)
    else:
        best_vec = best_vec / s
    return [float(x) for x in best_vec]
