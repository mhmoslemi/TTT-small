import time
import math

# ------------------------------------------------------------
# Project onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast AC1 bound for a vector with sum = 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            if mixed_sc < best_sc:
                best_sc = mixed_sc
                best_vec = mixed.copy()
                cur = mixed.copy()
                cur_sc = mixed_sc
                momentum = np.zeros_like(cur)
                eps = None
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # Return as plain Python list of floats
    return [float(x) for x in best_vec]
