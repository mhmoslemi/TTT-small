def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=Z}."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray, L: int) -> float:
    """Maximum entry of the full linear convolution a*a (FFT‑based)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

            break

        # ----- sub‑gradient from the top‑k convolution entries -----
        if top_k >= conv_full.size:
            idxs = np.arange(conv_full.size)
        else:
            idxs = np.argpartition(-conv_full, top_k - 1)[:top_k]

        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # range of i where a[i] participates in (a*a)[k]
            i_slice = slice(i_start, i_end + 1)
            idx_a = k - np.arange(i_start, i_end + 1, dtype=np.int64)
            g[i_slice] += 2.0 * a[idx_a]
        g /= max(1, len(idxs))

        # momentum and step
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        # projected SGD update
        a_tmp = a - step * momentum
        a_tmp = _project_simplex(a_tmp, Z=1.0)

        cand_sc = _score(a_tmp)

        # accept if improvement, else occasional uphill move
        if cand_sc < cur_sc:
            a = a_tmp
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, a_tmp.copy()
                no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                a = a_tmp

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym, Z=1.0)
            sym_sc = _score(a_sym)
            if sym_sc < cur_sc:
                a = a_sym
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    no_improve = 0

        # ----- random mass‑transfer when stagnating -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a, Z=1.0)
                no_improve = 0

        # ----- cheap permutation search (exploration) -----
        if it % 500 == 0 and (deadline - time.monotonic()) > 2.0:
            perm = rng.permutation(a)
            perm_sc = _score(perm)
            if perm_sc < best_sc:
                best_sc, best_vec = perm_sc, perm.copy()
                a = perm
                no_improve = 0

        # ----- occasional LP refinement (if cvxpy available) -----
        if cp is not None and it % 2000 == 0 and (deadline - time.monotonic()) > 5.0:
            M = cur_max
            # build sparse convolution matrix for current a
            data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
            offsets = np.arange(n)
            A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
            g_var = cp.Variable(n, nonneg=True)
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)),
                               [A @ g_var <= M + 1e-12])
            try:
                prob.solve(solver="ECOS", warm_start=True, max_iters=2000)
            except Exception:  # pragma: no cover
                prob = None
            if g_var.value is not None:
                g_opt = np.maximum(g_var.value, 0.0)
                g_opt = _project_simplex(g_opt, Z=1.0)
                t = 0.2
                a_mix = (1.0 - t) * a + t * g_opt
                a_mix = _project_simplex(a_mix, Z=1.0)
                mix_sc = _score(a_mix)
                if mix_sc < cur_sc:
                    a = a_mix
                    if mix_sc < best_sc:
                        best_sc, best_vec = mix_sc, a_mix.copy()
                        no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate found
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
