import time, math, numpy as np

def _project_simplex(v, Z=1.0):
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max(a, L):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initialise
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random positive vector of moderate size
        a = rng.random(2048, dtype=np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()         # enforce sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two

    # initial scores
    cur_max, _ = _conv_max(a, L)
    cur_score = 2.0 * n * cur_max
    best = a.copy()
    best_score = cur_score

    # optimiser parameters
    step0 = 0.7
    beta = 0.9
    momentum = np.zeros_like(a)

    sym_interval = 200          # symmetrise every this many iterations
    no_improve = 0
    it = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # compute convolution and its max
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = float(conv_full.max())
        max_idx = int(conv_full.argmax())
        cur_score = 2.0 * n * max_val

        # update best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # early exit
        if best_score < target:
            break

        # ----- exact subgradient from the maximal entry -----
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        # indices i satisfying i_start <= i <= i_end
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            g[i_range] = 2.0 * a[max_idx - i_range]

        # momentum and step
        momentum = beta * momentum + (1.0 - beta) * g
        alpha = step0 / math.sqrt(it)
        a_tmp = a - alpha * momentum
        a_tmp = _project_simplex(a_tmp)   # keep on simplex sum=1

        # evaluate candidate
        cand_max, _ = _conv_max(a_tmp, L)
        cand_score = 2.0 * n * cand_max

        # accept if improvement
        if cand_score < cur_score:
            a = a_tmp
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best = a_tmp.copy()
        else:
            # occasional uphill acceptance (Metropolis)
            prob = math.exp(-(cand_score - cur_score) / (alpha + 1e-12))
            if rng.random() < prob:

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative entries with sum Z."""
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Return max of full linear convolution a*a using FFT of length L
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv

    a = np.maximum(a, 0.0)

        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two FFT length

    # -------------------- bookkeeping --------------------

    cur_score = 2.0 * n * cur_max
    best = a.copy()

    # optimisation parameters
    step0 = 0.7
    beta = 0.9
    momentum = np.zeros_like(a)
    sym_interval = 200          # how often to symmetrise
    no_improve = 0
    it = 0
    target = 1.50274349

    # -------------------- main loop --------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution and its maximum
        a_fft = np.fft.rfft(a, n=L)

        max_val = float(conv_full.max())
