import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum=1, all >=0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()

    cur = best_vec.copy()
    cur_sc = best_sc
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------
    eps_factor = 0.03          # initial smoothing relative to max(conv)
    eps_decay = 0.999
    beta = 0.9                 # momentum decay
    momentum = np.zeros_like(cur)
    eps = None
    base_alpha = 0.6
    max_trials = 8
    rand_move_prob = 0.02
    sym_interval = 250
    improve_target = 1.50274349
    no_improve = 0
    mass_transfer_thresh = 2000
    restart_thresh = 3000
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of the current vector -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- smoothing temperature -----
        if eps is None:
            eps = max(max_val * eps_factor, 1e-14)
        else:
            eps = max(eps * eps_decay, 1e-14)

        # ----- smoothed weights (soft max) -----
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max via FFT -----
        w_rev = w[::-1]
        grad_full = np.fft.irfft(a_fft * np.fft.rfft(w_rev, n=L), n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- step size (decreasing) -----
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / math.sqrt(it) / norm_m

        # ----- backtracking line search -----
        cand = None
        cand_sc = None
        alpha = alpha0
        for _ in range(max_trials):
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score_raw(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                break
            alpha *= 0.5

        # ----- occasional uphill move -----
        if cand is None:
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score_raw(cand_try)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try

        # ----- accept if improved (or uphill accepted) -----
        if cand is not None:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation (helps) -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_raw(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- random small mass transfer moves -----
        if rng.random() < rand_move_prob:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_raw(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0

        # ----- heavier mass‑transfer when stuck -----
        if no_improve > mass_transfer_thresh:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_raw(tmp)
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                cur = tmp
                cur_sc = tmp_sc
                no_improve = 0

        # ----- restart from the best if hopeless -----
        if no_improve > restart_thresh:
            cur = best_vec + 0.005 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            eps = None
            no_improve = 0

        # ----- early exit if target met -----
        if best_sc < improve_target:
            break

    # Return plain Python floats
    return [float(x) for x in best_vec]
```
