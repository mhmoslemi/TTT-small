# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

                no_improve += 1
        else:
            no_improve += 1

        # ----- occasional symmetrisation (helps flatten the profile) -----
        if (no_improve % 500) == 0 and no_improve > 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score(sym, n, L)
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                if sym_score < best_score:
                    best_score = sym_score
                    best_vec = sym.copy()
                no_improve = 0

        # ----- restart from best with small noise if hopeless -----
        if no_improve > 4000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            no_improve = 0

        # ----- early exit if target reached -----
        if best_score < target:
            break

    return [float(x) for x in best_vec]
