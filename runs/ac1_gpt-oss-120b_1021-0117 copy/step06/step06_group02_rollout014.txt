def _subgradient_maxconv(a):
    # a: np.ndarray length n, sum normalized to 1 (or any)
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length() + 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    max_val = conv.max()
    # tolerance maybe relative
    tol = max_val * 1e-12 + 1e-12
    active = np.where(conv >= max_val - tol)[0]  # indices
    # compute gradient
    g = np.zeros_like(a)
    # second term: each active contributes a_j
    g += len(active) * a
    for m in active:
        # compute shift contributions
        # for j where 0<=m-j<n i.e., j in [max(0, m-(n-1)), min(n-1,m)]
        lo = max(0, m - (n-1))
        hi = min(n-1, m)
        # for j in [lo,hi], a_{m-j} valid
        # vectorized: indices = np.arange(lo, hi+1)
        # contributions = a[m - indices]
        # add to g[indices]
        idx = np.arange(lo, hi+1, dtype=np.int64)
        g[idx] += a[m - idx]
    return g, max_val
```

import time, math, numpy as np

# ------------------------------------------------------------
# Project onto the simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ------------------------------------------------------------
# AC1 bound (assumes sum(a)=1) – returns 2*n*max(convolution)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length() + 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Subgradient of max_i (a*a)[i] (for sum(a)=1)
# ------------------------------------------------------------
def _subgrad_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-12
    active = np.where(conv >= max_val - tol)[0]
    g = len(active) * a  # second term a_j repeats for each active index
    for m in active:
        lo = max(0, m - (n - 1))
        hi = min(n - 1, m)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += a[m - idx]
    return g
