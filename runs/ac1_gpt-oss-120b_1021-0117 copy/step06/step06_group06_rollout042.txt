def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # little margin

    # initial vector
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n_fallback = 2000
        a = np.exp(-np.arange(n_fallback, dtype=np.float64) / 100.0)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0/a.size)
        s = 1.0
    a = a / s   # sum =1

    n = a.size
    L = 1 << ((3*n - 2).bit_length())

    # projection function
    def _project(v):
        # same as _project_simplex
        n_ = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n_+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    # scoring: fast bound given sum~1
    def _score(v):
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())

    cur = a.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # Adam hyperparameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.2
    # smoothing epsilon for softmax for max
    eps_temp = None
    eps_decay = 0.97

    # counters and thresholds
    no_improve = 0
    sym_interval = 200
    max_trials = 5
    target = 1.50274349
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # compute convolution
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()

        # compute temperature for softmax
        if eps_temp is None:
            eps_temp = max_val * 0.15 + 1e-12
        else:
            eps_temp = max(eps_temp * eps_decay, max_val * 1e-12)

        # compute weights
        if eps_temp < max_val * 1e-6:
            mask = (conv_full == max_val)
            w = mask.astype(np.float64)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(conv_full, 1.0/conv_full.size)
            else:
                w /= w_sum
        else:
            w = np.exp((conv_full - max_val) / eps_temp)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv_full, 1.0/conv_full.size)
            else:
                w /= w_sum

        # gradient via FFT
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # try step with backtracking
        factor = 1.0
        improved = False
        cand_sc = None
        for _ in range(max_trials):
            cand = cur - factor * step
            cand = _project(cand)
            cand_sc = _score(cand)
            if cand_sc < cur_sc:
                improved = True
                break
            factor *= 0.5

        if not improved:
            # occasional stochastic acceptance
            cand = cur - factor * step
            cand = _project(cand)
            cand_sc = _score(cand)
            temp = max(1e-3, 1.0 / math.sqrt(it))
            prob = math.exp(-(cand_sc - cur_sc) / temp) if cand_sc > cur_sc else 1.0
            if rng.random() < prob:
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # symmetrization
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # random mass transfer if stagnating
        if no_improve > 1200:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0

        # restart if hopeless
        if no_improve > 3500:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m = np.zeros_like(cur)
            v = np.zeros_like(cur)
            eps_temp = None
            no_improve = 0

        if best_sc < target:
            break

        # time safety
        if time.monotonic() >= deadline:
            break

    # final refinement random swaps
    while time.monotonic() < deadline - 0.02:
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        delta = cur[i] * rng.random() * 0.05
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project(cand)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            cur = cand
            cur_sc = cand_sc

    # return best
    return [float(x) for x in best_vec]

import time, math
import numpy as np

# ------------------------------------------------------------
# Project a vector onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ------------------------------------------------------------

    n = a.shape[0]

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        # ----- softmax weights -----
        w = np.exp((conv_full - max_val) / eps_temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv_full, 1.0 / conv_full.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max (matrix‑free) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr / (1.0 + it * 1e-4)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- back‑tracking line search -----
        factor = 1.0
        improved = False
        cand_sc = None
        cand = None
        for _ in range(8):
            cand_try = cur - factor * step
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            factor *= 0.5

        # ----- occasional uphill move (simulated annealing) -----
        if not improved:
            cand_try = cur - factor * step
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            temp = max(1e-3, 1.0 / math.sqrt(it))
            prob = math.exp(-(cand_sc - cur_sc) / temp) if cand_sc > cur_sc else 1.0
            if rng.random() < prob:
                cand = cand_try
                improved = True

        # ----- accept / reject -----
        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- random mass transfer when stuck -----
        if no_improve > 1200:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0

        # ----- restart from best if hopeless -----
        if no_improve > 3500:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m = np.zeros_like(cur)
            v = np.zeros_like(cur)
            eps_temp = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # safety check in case the loop body overruns
        if time.monotonic() >= deadline:
            break

    # --------------------------------------------------------
    # Random Dirichlet sampling for extra exploration
    # --------------------------------------------------------
    while time.monotonic() < deadline - 0.1:
        alpha = rng.uniform(0.1, 5.0)
        cand = rng.gamma(alpha, 1.0, size=n)
        cand = cand / cand.sum()
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()

    # --------------------------------------------------------
    # Final cheap local refinement (random swaps)
    # --------------------------------------------------------
    while time.monotonic() < deadline - 0.02:
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        delta = cur[i] * rng.random() * 0.05
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            cur = cand
            cur_sc = cand_sc

    # --------------------------------------------------------
    # Return the best vector found (as a plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
```
