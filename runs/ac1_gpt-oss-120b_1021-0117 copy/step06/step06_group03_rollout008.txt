def _refine(cur, deadline, best_sc, best_vec):
    n = cur.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    eps = 0.02
    eps_decay = 0.995
    it = 0
    best_local_sc = best_sc
    best_local_vec = best_vec
    while time.monotonic() < deadline:
        it += 1
        # compute conv
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        # update eps
        eps = max(eps * eps_decay, 1e-6)
        shifted = conv - max_val
        w = np.exp(shifted / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # grad
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n-2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # step size: simple diminishing
        norm_g = np.linalg.norm(g) + 1e-12
        step = 0.5 / np.sqrt(it) / norm_g

        cand = _project_simplex(cand)
        cand_sc = _score_fft(cand)

    dot = np.dot(a, grad_a)
    grad_u = a * (grad_a - dot)
    obj = max_val + eps * np.log(sum_exp)

import time
import math
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Global placeholders for L-BFGS objective (set before use)
# ------------------------------------------------------------
global_n = None
global_L = None
global_eps = None

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Exact AC1 bound (2*n*max(convolution)) via FFT
# ------------------------------------------------------------
def _score_fft(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# L-BFGS objective (smoothed max) and its gradient (w.r.t. u)
# ------------------------------------------------------------
def _lbfgs_obj(u: np.ndarray):
    eps = global_eps
    n = global_n
    L = global_L
    a = np.exp(u)
    a_sum = a.sum()
    a = a / a_sum
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    shifted = conv - max_val
    exp_shift = np.exp(shifted / eps)
    sum_exp = exp_shift.sum()
    w = exp_shift / sum_exp
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad_a = 2.0 * grad_full[idx]
    dot = np.dot(a, grad_a)
    grad_u = a * (grad_a - dot)
    obj = max_val + eps * np.log(sum_exp)
    return obj, grad_u

# ------------------------------------------------------------
# Simple gradient‑descent refinement using the smoothed max
# ------------------------------------------------------------
def _refine(cur: np.ndarray, deadline: float, best_sc: float, best_vec: np.ndarray):
    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    eps = 0.02
    eps_decay = 0.995
    it = 0
    best_local_sc = best_sc
    best_local_vec = best_vec.copy()
    while time.monotonic() < deadline:
        it += 1
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        eps = max(eps * eps_decay, 1e-6)
        shifted = conv - max_val
        w = np.exp(shifted / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        norm_g = np.linalg.norm(g) + 1e-12
        step = 0.5 / math.sqrt(it) / norm_g
        cand = cur - step * g
        cand = _project_simplex(cand)
        cand_sc = _score_fft(cand)
        cur_sc = _score_fft(cur)
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
        if cand_sc < best_local_sc:
            best_local_sc = cand_sc
            best_local_vec = cand.copy()
        if it % 150 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
            if sym_sc < best_local_sc:
                best_local_sc = sym_sc
                best_local_vec = sym.copy()
        if best_local_sc < 1.50274349:
            break
        if time.monotonic() + 0.1 >= deadline:
            break
    return best_local_vec, best_local_sc
