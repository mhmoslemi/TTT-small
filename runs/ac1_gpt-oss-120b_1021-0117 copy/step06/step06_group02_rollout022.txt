        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length() + 1)   # power‑of‑two ≥ 2n
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----- initialise a feasible point on the simplex -----
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
    else:
        n0 = 2000
        t = np.arange(n0, dtype=np.float64)
        a0 = np.exp(-t / 120.0)
        a0 = a0 / a0.sum()

    n = a0.shape[0]
    L = 1 << ((3 * n - 2).bit_length())          # ≥ 3n for the extra convs
    grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)

    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ----- Adam optimizer state -----
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.8                       # base learning rate
    eps_smooth = None               # smoothing for soft‑max
    eps_decay = 0.9995
    no_improve = 0
    improve_target = 1.50274349

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # ----- convolution of current iterate -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- smoothing parameter -----
        if eps_smooth is None:
            eps_smooth = max(max_val * 0.5, 1e-12)
        else:
            eps_smooth = max(eps_smooth * eps_decay, 1e-12)

        # ----- soft‑max weights -----
        w = np.exp((conv_full - max_val) / eps_smooth)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max (via FFT) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[grad_idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = -lr0 / math.sqrt(it) * m_hat / (np.sqrt(v_hat) + eps_adam)
        cand = cur + step
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # ----- accept if improvement, else occasional uphill move -----
        improved = False
        if cand_sc < cur_sc:
            improved = True
        else:
            # simulated‑annealing style acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (lr0 + 1e-12))
            if rng.random() < prob:
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic random mass‑transfer when stuck -----
        if no_improve > 1200:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.7
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc or rng.random() < 0.07:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0

        # ----- restart from best with mild noise if hopeless -----
        if no_improve > 2800:
            cur = best_vec + 0.015 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            eps_smooth = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < improve_target:
            break

        # ----- time check -----
        if it % 400 == 0 and time.monotonic() >= deadline:
            break

    # return as plain list of Python floats
    return [float(x) for x in best_vec]
