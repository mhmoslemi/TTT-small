import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection onto the probability simplex. """
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _conv_max(v: np.ndarray, L: int, n: int) -> float:
    """Linear convolution v*v (length 2n‑1) via FFT, return its maximum value."""
    af = np.fft.rfft(v, L)
    conv = np.fft.irfft(af * af, L)[: 2 * n - 1]
    return float(conv.max())


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry v[i]=v[-i-1] (preserves sum)."""
    return (v + v[::-1]) * 0.5


def _smooth(v: np.ndarray) -> np.ndarray:
    """One‑step three‑point moving‑average (zero‑pad edges)."""
    ker = np.array([0.25, 0.5, 0.25], dtype=v.dtype)
    return np.convolve(v, ker, mode="same")


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # -----------------------------------------------------------------
    # initialise a feasible point (sum = 1, non‑negative, symmetric)
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.ones_like(a0)
        a0 = a0 / a0.sum()
        n = a0.shape[0]
    else:
        n = 2000
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    L = 1 << ((2 * n - 1).bit_length())          # FFT length >= 2n‑1

    # generate a few smart starters (power‑law shapes, both centre‑peak & edge‑peak)
    cand_list = []
    a_sym = _symmetrize(a0)
    a_sym = _project_simplex(a_sym)
    cand_list.append(a_sym)

    idx = np.arange(n)
    centre = (n - 1) / 2.0
    dist_c = np.abs(idx - centre)                # distance from centre
    dist_e = np.minimum(idx, n - 1 - idx)        # distance from nearest edge
    for p in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0):
        a = (dist_c + 1.0) ** (-p)
        a = _project_simplex(a)
        a = _symmetrize(a)
        a = _project_simplex(a)
        cand_list.append(a)

        a = (dist_e + 1.0) ** (-p)
        a = _project_simplex(a)
        a = _symmetrize(a)
        a = _project_simplex(a)
        cand_list.append(a)

    # pick the best starter according to the AC1 bound
    best_vec = None
    best_sc = np.inf
    for a in cand_list:
        max_val = _conv_max(a, L, n)
        sc = 2.0 * n * max_val          # sum(a)==1 → denominator =1
        if sc < best_sc:
            best_sc = sc
            best_vec = a.copy()

    cur = best_vec.copy()
    cur_sc = best_sc
    momentum = np.zeros_like(cur)

    # -----------------------------------------------------------------
    # optimisation parameters
    # -----------------------------------------------------------------
    beta = 0.97
    base_alpha = 0.6
    sym_interval = 100
    transfer_thresh = 400
    reinit_thresh = 1200
    target_score = 1.50274349
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # convolution, max value & index
        af = np.fft.rfft(cur, L)
        conv = np.fft.irfft(af * af, L)[: 2 * n - 1]
        max_val = float(conv.max())
        max_idx = int(conv.argmax())
        cur_sc = 2.0 * n * max_val

        # sub‑gradient of the active lag
        j = max_idx - np.arange(n)
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(cur)
        grad[mask] = 2.0 * cur[j[mask]]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # adaptive step size
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / (math.sqrt(it) * norm_m)

        # -----------------------------------------------------------------
        # back‑tracking line‑search
        # -----------------------------------------------------------------
        improved = False
        cand = None
        cand_sc = None
        alpha_try = alpha
        for _ in range(8):
            cand_try = cur - alpha_try * momentum
            cand_try = _project_simplex(cand_try)
            cand_try = _symmetrize(cand_try)
            if rng.random() < 0.1:               # occasional smoothing
                cand_try = _smooth(cand_try)
                cand_try = _project_simplex(cand_try)
            max_val_c = _conv_max(cand_try, L, n)
            sc_c = 2.0 * n * max_val_c
            if sc_c < cur_sc:
                cand, cand_sc = cand_try, sc_c
                improved = True
                break
            alpha_try *= 0.5

        # -----------------------------------------------------------------
        # uphill move with annealing schedule
        # -----------------------------------------------------------------
        if not improved:
            cand_try = cur - alpha_try * momentum
            cand_try = _project_simplex(cand_try)
            cand_try = _symmetrize(cand_try)
            max_val_c = _conv_max(cand_try, L, n)
            sc_c = 2.0 * n * max_val_c
            T = max(0.01, 0.5 * (0.999 ** it))
            if rng.random() < math.exp(-(sc_c - cur_sc) / T):
                cand, cand_sc = cand_try, sc_c
                improved = True

        # -----------------------------------------------------------------
        # accept improvement
        # -----------------------------------------------------------------
        if improved:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # -----------------------------------------------------------------
        # periodic full symmetrisation (keeps shape regular)
        # -----------------------------------------------------------------
        if it % sym_interval == 0:
            cur = _symmetrize(cur)
            cur = _project_simplex(cur)
            cur_sc = 2.0 * n * _conv_max(cur, L, n)
            if cur_sc < best_sc:
                best_sc, best_vec = cur_sc, cur.copy()
                no_improve = 0

        # -----------------------------------------------------------------
        # random mass transfer when progress stalls
        # -----------------------------------------------------------------
        if no_improve > transfer_thresh:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n)
                if j == i:
                    j = (j + 1) % n
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp = _symmetrize(tmp)
                max_val_t = _conv_max(tmp, L, n)
                sc_t = 2.0 * n * max_val_t
                if sc_t < cur_sc:
                    cur, cur_sc = tmp, sc_t
                    if sc_t < best_sc:
                        best_sc, best_vec = sc_t, tmp.copy()
                    no_improve = 0
                else:
                    no_improve = max(0, no_improve - 150)

        # -----------------------------------------------------------------
        # restart from the best point with a small perturbation
        # -----------------------------------------------------------------
        if no_improve > reinit_thresh:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            cur = _symmetrize(cur)
            momentum = np.zeros_like(cur)
            no_improve = 0

        # -----------------------------------------------------------------
        # early exit if target reached
        # -----------------------------------------------------------------
        if best_sc < target_score:
            break

    # return as plain Python list
    return [float(x) for x in best_vec]
