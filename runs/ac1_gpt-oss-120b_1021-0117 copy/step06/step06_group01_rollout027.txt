import time
import math
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Softmax (z -> a) guaranteeing a_i >= 0 and sum(a)=1
# ------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


# ------------------------------------------------------------
# Objective & gradient for a given smoothing epsilon
# ------------------------------------------------------------
class _ObjGrad:
    """Callable returning (objective, gradient) w.r.t. the softmax logits."""
    def __init__(self, n: int):
        self.n = n
        self.L = 1 << ((2 * n - 1).bit_length())
        self.eps = 0.1  # placeholder, will be overwritten

    def set_eps(self, eps: float):
        self.eps = eps

    def __call__(self, z: np.ndarray):
        a = _softmax(z)                     # a >=0, sum=1
        a_fft = np.fft.rfft(a, n=self.L)

        # full convolution a*a
        conv = np.fft.irfft(a_fft * a_fft, n=self.L)[: 2 * self.n - 1]
        max_val = conv.max()

        # soft‑max weights (stable)
        w = np.exp((conv - max_val) / self.eps)
        w_sum = w.sum()
        w /= w_sum

        # gradient of the smoothed max w.r.t. a
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=self.L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=self.L)[: 2 * self.n - 1]
        idx = (2 * self.n - 2) - np.arange(self.n)
        grad_a = 2.0 * grad_full[idx]                     # d smoothed_max / d a

        # full objective (2 n * smoothed_max)
        obj = 2.0 * self.n * (max_val + self.eps * math.log(w_sum))

        # chain rule through softmax: dz = a * (dg - <a,dg>)
        dg = 2.0 * self.n * grad_a
        dot = np.dot(a, dg)
        grad_z = a * (dg - dot)

        return obj, grad_z

# Main entry point

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----- initialise a feasible point (sum = 1) -----
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s
    else:                                   # fallback exponential
        n0 = 2000
        a0 = np.exp(-np.arange(n0, dtype=np.float64) / 100.0)
        a0 = a0 / a0.sum()

    n = a0.size
    best_vec = a0.copy()
    best_sc = _score(best_vec)

    # ----- cheap exploratory families (few samples) -----
    def _try(v):
        nonlocal best_vec, best_sc
        sc = _score(v)
        if sc < best_sc:
            best_sc, best_vec = sc, v.copy()

    for _ in range(80):
        alpha = 10 ** rng.uniform(-4, 0)
        cand = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand = cand / cand.sum()
        _try(cand)

    for _ in range(80):
        alpha = 10 ** rng.uniform(-4, 0)
        d = np.minimum(np.arange(n, dtype=np.float64),
                       n - 1 - np.arange(n, dtype=np.float64))
        cand = np.exp(-alpha * d)
        cand = cand / cand.sum()
        _try(cand)

    mid = (n - 1) / 2.0
    for _ in range(80):
        sigma = rng.uniform(0.5, 40.0)
        cand = np.exp(-((np.arange(n, dtype=np.float64) - mid) ** 2) /
                      (2 * sigma ** 2))
        cand = cand / cand.sum()
        _try(cand)

    for _ in range(80):
        cand = np.cos(np.pi * np.arange(n, dtype=np.float64) / (n - 1)) ** 2
        cand = cand / cand.sum()
        _try(cand)

    # ----- random Dirichlet starts -----
    for _ in range(30):
        conc = rng.uniform(0.2, 2.0)
        g = rng.gamma(conc, 1.0, size=n)
        cand = g / g.sum()
        _try(cand)

    # ------------------------------------------------------------
    # gradient‑based refinement using L‑BFGS on a softmax parametrisation
    # ------------------------------------------------------------
    objgrad = _ObjGrad(n)
    z = np.log(best_vec + 1e-12)          # logits for the current best

    improve_target = 1.50274349
    stage = 0
    no_improve = 0

    while time.monotonic() < deadline:
        # recompute max convolution for epsilon scaling
        cur_a = _softmax(z)
        cur_sc = _score(cur_a)
        max_conv = cur_sc / (2.0 * n)

        # schedule epsilon (start relatively big, shrink each stage)
        eps = max(max_conv * 0.2, 1e-9) * (0.7 ** stage)
        objgrad.set_eps(eps)

        # L‑BFGS optimisation (few iterations per stage)
        res = minimize(objgrad,
                       z,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 200,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})

        # retrieve and symmetrise
        z = res.x
        a = _softmax(z)
        a = (a + a[::-1]) * 0.5           # occasional symmetrisation
        a = a / a.sum()                   # re‑normalise (should stay 1)

        # update best if improved
        sc = _score(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # prepare next start point
        z = np.log(a + 1e-12)

        # early exit
        if best_sc < improve_target:
            break

        # occasional random restart when stuck
        if no_improve >= 3000:
            g = rng.gamma(0.5, 1.0, size=n)
            a = g / g.sum()
            z = np.log(a + 1e-12)
            no_improve = 0

        stage += 1

    # final safeguard: return the best vector we have
    return [float(x) for x in best_vec]
