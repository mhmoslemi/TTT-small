def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # initial vector
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
    else:
        n_fallback = 2500
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 120.0)
        a0 = a0 / a0.sum()

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())
    # score function
    def _score_vec(v):
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())

    cur = a0.copy()
    cur_sc = _score_vec(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # parameters
    it = 0
    no_improve = 0
    eps = None
    eps_decay = 0.998
    base_alpha = 1.5     # maybe larger
    beta = 0.95          # momentum
    momentum = np.zeros_like(cur)
    sym_interval = 150
    improve_target = 1.50274349
    max_trials = 12
    restart_threshold = 2500
    perturb_threshold = 1200

    while time.monotonic() < deadline:
        it += 1

        # FFT and conv
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()

        # update epsilon
        if eps is None:
            eps = max(max_val * 0.2, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)

        # compute softmax weights
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient approximate
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum
        momentum = beta * momentum + (1 - beta) * g

        # adaptive step size
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / math.sqrt(it) / norm_m

        # backtracking line search
        improved = False
        cand = None
        cand_sc = None
        alpha = alpha0
        for _ in range(max_trials):
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score_vec(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5

        # occasional uphill acceptance
        if not improved:
            # try one more with current alpha
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score_vec(cand_try)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # periodic symmetrisation
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # random mass transfer if stagnation
        if no_improve > perturb_threshold:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_vec(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # restart if hopeless
        if no_improve > restart_threshold:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            eps = None
            no_improve = 0

        # early exit if target reached
        if best_sc < improve_target:
            break

    return [float(x) for x in best_vec]

import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound for a probability vector
# ------------------------------------------------------------
def _score_vec(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ----- initialise a feasible point on the simplex -----
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full_like(a0, 1.0 / a0.size)
    else:
        n_fallback = 2500
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 120.0)
        a0 = a0 / a0.sum()

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())          # power‑of‑two length for FFT

    cur = a0.copy()
    cur_sc = _score_vec(cur, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ----- optimisation parameters -----
    it = 0
    no_improve = 0
    eps = None
    eps_decay = 0.998
    base_alpha = 1.5
    beta = 0.95
    momentum = np.zeros_like(cur)
    sym_interval = 150
    improve_target = 1.50274349
    max_trials = 12
    perturb_thresh = 1200
    restart_thresh = 2500

    while time.monotonic() < deadline:
        it += 1

        # convolution of current iterate
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # smoothing temperature for soft‑max approximation of the max
        if eps is None:
            eps = max(max_val * 0.2, 1e-12)
        else:

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:

        else:
            w /= w_sum

        # gradient of the smoothed max (FFT‑based)
        w_rev = w[::-1]

        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        i = rng.integers(0, n)
        if best_vec[i] <= 1e-14:
            continue
        delta = best_vec[i] * rng.random() * 0.3
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score_vec(cand, L)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand

    return [float(x) for x in best_vec]
