import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=Z}."""
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Maximum value of a*a via FFT (full linear convolution)
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# ------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
# ------------------------------------------------------------
def _grad_max_subgrad(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """Return a sub‑gradient of max(conv) w.r.t. a (averaged over ties)."""
    n = a.shape[0]
    # indices where conv is (almost) maximal
    eps = 1e-12
    idxs = np.where(conv >= max_val - eps)[0]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    if idxs.size > 0:
        g /= idxs.size
    return g

        # ----- current full convolution and max -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_sc = 2.0 * n * max_val

        # ----- keep the best -----
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- sub‑gradient step (projected SGD with momentum) -----
        g = _grad_max_subgrad(a, conv_full, max_val)
        momentum = beta * momentum + (1.0 - beta) * g
        step = base_step / math.sqrt(it)
        a_tmp = a - step * momentum
        a_tmp = _project_simplex(a_tmp, Z=1.0)

        # ----- evaluate candidate -----
        cand_sc = 2.0 * n * _conv_max(a_tmp, L)

        # ----- accept if improvement or via Metropolis rule -----
        if cand_sc < cur_sc:
            a = a_tmp
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = a_tmp.copy()
                no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                a = a_tmp
                cur_sc = cand_sc

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym, Z=1.0)
            sym_sc = 2.0 * n * _conv_max(a_sym, L)
            if sym_sc < cur_sc:
                a = a_sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = a_sym.copy()
                    no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 1500:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
            no_improve = 0

        # ----- occasional random restart (exponential shape) -----
        if it % 5000 == 0 and (deadline - time.monotonic()) > 30.0:
            alpha = rng.uniform(0.01, 5.0)
            idx = np.arange(n)
            a = np.exp(-alpha * idx)
            a = a / a.sum()
            momentum = np.zeros_like(a)
            no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
