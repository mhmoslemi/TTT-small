    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # init
    if "height_sequence_1" in globals():

        a = np.maximum(a, 0.0)
        if a.sum() <= 0:

        a = np.ones(n) / n

    a = a / a.sum()  # simplex
    n = a.size

    L_grad = 1 << ((3 * n - 2).bit_length())

    # initial gradient and score
    conv = _conv_self_fft(a, L_self)  # returns conv array length 2n-1
    max_c = conv.max()
    cur_sc = 2.0 * n * max_c / (a.sum() ** 2)  # sum=1 => 2 n max_c
    best_vec = a.copy()
    best_sc = cur_sc

    # optimizer parameters
    base_alpha = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    tau = 0.5    # smoothing parameter
    tau_max = 12.0
    tau_factor = 1.003  # incremental increase

    # stagnation handling
    no_improve = 0
    improve_thr = 2500

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # compute smoothed gradient
        conv = _conv_self_fft(a, L_self)
        max_c = conv.max()
        # softmax weights
        # improve numeric stability: subtract max_c
        exp_vals = np.exp(tau * (conv - max_c))
        w = exp_vals / exp_vals.sum()
        # gradient
        # compute convolution of a with reversed w
        w_rev = w[::-1]
        # FFT convolution
        Af = np.fft.rfft(a, L_grad)
        Wf = np.fft.rfft(w_rev, L_grad)
        conv_all = np.fft.irfft(Af * Wf, L_grad)[: 3 * n - 2]

        grad = 2.0 * conv_all[2 * n - 2 - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_alpha / (np.sqrt(it) + 1.0) * m_hat / (np.sqrt(v_hat) + eps)

        max_c_new = conv_new.max()

            best_vec = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

def _conv_self_fft(a, L):
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)
    # conv length L, but we only need first 2n-1 entries

import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_self_fft(a, L):
    """Convolution a * a via FFT, returns length 2*len(a)-1 array."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)
    return conv[: 2 * a.shape[0] - 1]

def _smooth_gradient(a, weights, L_grad):
    """Gradient of smooth max w.r.t. a given softmax weights (sum to 1)."""
    w_rev = weights[::-1]
    Af = np.fft.rfft(a, L_grad)
    Wf = np.fft.rfft(w_rev, L_grad)
    conv_all = np.fft.irfft(Af * Wf, L_grad)[: 3 * a.shape[0] - 2]
    grads = 2.0 * conv_all[2 * a.shape[0] - 2 - np.arange(a.shape[0])]
    return grads

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full convolution a*a using FFT (length 2*len(a)-1)."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)
    return conv[: 2 * a.shape[0] - 1]


def _smooth_gradient(a: np.ndarray, w: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of the smooth approximation of max(conv) w.r.t. a.
    w – softmax weights over the convolution values (sum to 1).
    """
    w_rev = w[::-1]
    Af = np.fft.rfft(a, L_grad)
    Wf = np.fft.rfft(w_rev, L_grad)
    conv_all = np.fft.irfft(Af * Wf, L_grad)[: 3 * a.shape[0] - 2]
    grads = 2.0 * conv_all[2 * a.shape[0] - 2 - np.arange(a.shape[0])]
    return grads

    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():

    else:
        n_fallback = 2000
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    a = a / a.sum()                         # now sum = 1

    # initial score

    best_vec = a.copy()
