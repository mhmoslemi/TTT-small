import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast AC1 bound for a sequence with sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Gradient of the maximal convolution entry (exact)
# ------------------------------------------------------------
def _gradient_max(a: np.ndarray, imax: int) -> np.ndarray:
    n = a.shape[0]
    shift = imax - (n - 1)          # zero‑lag corresponds to index n‑1
    g = np.zeros_like(a)
    if shift >= 0:
        s = shift
        if s < n:
            g[: n - s] += a[s:]          # a_{k+shift}
            g[s:] += a[: n - s]          # a_{k-shift}
    else:
        s = -shift
        if s < n:
            g[s:] += a[: n - s]          # a_{k+shift} (shift negative)
            g[: n - s] += a[s:]          # a_{k-shift}
    return g

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- initialise ------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        cur = a0 / s                     # sum = 1
    else:                                 # robust fallback
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        cur = np.exp(-t / 100.0)
        cur = cur / cur.sum()

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ---- optimisation parameters ------------------------------------
    momentum = np.zeros_like(cur)
    beta = 0.9
    base_alpha = 1.2
    sym_interval = 100
    improve_target = 1.50274349
    max_backtrack = 10
    it = 0
    no_improve = 0
    mass_thresh = 3000
    restart_thresh = 5000

    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- convolution & maximal entry -------------------------
        af = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        imax = int(conv_full.argmax())

        # ---- compute (sub)gradient of the max entry ---------------
        # if several entries are almost maximal, average their gradients
        eps_tol = 1e-12

        if not improved:

            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)

                improved = True

        # ---- accept or reject --------------------------------------
        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:

                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:

        # ---- periodic symmetrisation -------------------------------

                    no_improve = 0

            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.8
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                no_improve = 0

            momentum = np.zeros_like(cur)
