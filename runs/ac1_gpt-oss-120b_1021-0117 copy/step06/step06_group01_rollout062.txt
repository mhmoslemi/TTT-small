def _grad_max(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    max_val = conv.max()
    # tolerance: maybe 1e-12 relative
    idx = np.where(np.abs(conv - max_val) <= 1e-12)[0]
    grad = np.zeros_like(a)
    for i in idx:
        shift = i - (n - 1)
        if shift >= 0:
            # shift part
            if shift < n:
                grad[shift:] += a[:n-shift]
                grad[:n-shift] += a[shift:]
        else:
            s = -shift
            # same as shift = s
            if s < n:
                grad[s:] += a[:n-s]
                grad[:n-s] += a[s:]
    return grad
```

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum=1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b)


# ------------------------------------------------------------
# Sub‑gradient of max‑convolution (exact, uses only the
# positions attaining the maximum)
# ------------------------------------------------------------
def _grad_max(a: np.ndarray) -> np.ndarray:
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    # indices where the max is achieved (tolerance for floating error)
    tol = 1e-12 * max_val
    idx = np.where(conv >= max_val - tol)[0]
    # work with absolute shifts only (symmetry of convolution)
    shifts = np.unique(np.abs(idx - (n - 1)))
    grad = np.zeros_like(a)
    for s in shifts:
        s = int(s)
        if s >= n:
            continue
        # contribution a_{j-s}
        grad[s:] += a[: n - s]
        # contribution a_{j+s}
        grad[: n - s] += a[s:]
    return grad
