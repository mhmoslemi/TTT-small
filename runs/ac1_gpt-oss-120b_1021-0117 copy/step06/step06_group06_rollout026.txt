import time, math
import numpy as np

# ------------------------------------------------------------
# Project a vector onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 score (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of a smooth approximation of the max (matrix‑free)
# ------------------------------------------------------------
def _gradient(a: np.ndarray, L: int, eps: float) -> np.ndarray:
    """Return gradient of the smoothed max of a*a (without the 2n factor)."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.shape[0] - 1]
    max_val = conv_full.max()
    # weight vector (softmax or hard max)
    if eps < max_val * 1e-6:
        mask = conv_full == max_val
        w = mask.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv_full, 1.0 / conv_full.size)
        else:
            w /= w_sum
    else:
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv_full, 1.0 / conv_full.size)
        else:
            w /= w_sum
    # gradient via convolution with reversed weights
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * a.shape[0] - 2) - np.arange(a.shape[0])
    grad = 2.0 * grad_full[idx]
    return grad

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    start = time.monotonic()
    # keep a safety margin for finalisation

    # ---------- initial feasible vector ----------
    if "height_sequence_1" in globals():

        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 100.0)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
        s = 1.0
    a0 = a0 / s                     # project onto simplex (sum = 1)

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length

    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ---------- optimisation hyper‑parameters ----------
    m = np.zeros_like(cur)          # first moment (Adam)
    v = np.zeros_like(cur)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.5
    eps_smooth = None
    eps_decay = 0.95
    max_trials = 4
    sym_interval = 30
    no_improve = 0
    target = 1.50274349
    it = 0

    # ---------- main optimisation loop ----------
    while time.monotonic() < deadline:
        it += 1

        # current convolution & max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # smoothing epsilon schedule
        if eps_smooth is None:
            eps_smooth = max_val * 0.2 + 1e-12
        else:
            eps_smooth = max(eps_smooth * eps_decay, max_val * 1e-12)

        # gradient of smoothed max
        grad = _gradient(cur, L, eps_smooth)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr / math.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # optional norm clipping (helps stability)
        step_norm = np.linalg.norm(step)

        cand_sc = None

                improved = True
                break

                cand = cand_try
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            no_improve += 1

        # ---------- periodic symmetrisation ----------
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ---------- random mass transfer when stuck ----------
        if no_improve > 500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0

        # ---------- restart if hopeless ----------
        if no_improve > 2000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            eps_smooth = None
            no_improve = 0
            it = 0   # reset bias‑correction counters

        # early exit if target reached
        if best_sc < target:
            break

        # safety time check
        if time.monotonic() >= deadline:
            break

    # ---------- final cheap refinement ----------
    while time.monotonic() < deadline - 0.04:
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        delta = cur[i] * rng.random() * 0.05
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            cur = cand
            cur_sc = cand_sc

    # Return as plain Python list
    return [float(x) for x in best_vec]
