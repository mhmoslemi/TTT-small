import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) :  x >= 0 , sum(x) = Z
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Maximum entry of a self‑convolution using FFT
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max_k (a*a)[k] using an FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# ------------------------------------------------------------
# Gradient approximation using the top‑k convolution values
# ------------------------------------------------------------
def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    n = a.size
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            # slice of a that contributes to this k
            rev = k - np.arange(i_start, i_end + 1, dtype=int)
            g[i_start:i_end + 1] += 2.0 * a[rev]
    # normalise contribution size
    g /= max(1, top_k)
    return g

# ------------------------------------------------------------

    best_sc = 2.0 * n * _conv_max(a, L)

    beta = 0.92
    momentum = np.zeros_like(a)
    top_k = 32
    sym_interval = 200
    restart_interval = 5000
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
