import time, math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound for a sequence with sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length() + 1)          # power of two ≥ 2n
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Generate a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # --------------------------------------------------------
    # initialise a feasible point on the simplex
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s                     # now sum = 1
    else:                               # simple fallback
        n_fallback = 2500
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 200.0)
        a0 = a0 / a0.sum()

    n = a0.shape[0]
    # length for FFTs (zero‑padded to avoid circular effects)
    L = 1 << ((3 * n - 2).bit_length())
    # indices that extract the gradient of the linear convolution
    grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)

    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------
    it = 0
    no_improve = 0
    eps = None
    eps_decay = 0.9995            # slower decay → smoother gradients longer
    base_alpha = 0.55
    beta = 0.90                   # momentum
    momentum = np.zeros_like(cur)
    sym_interval = 120            # more frequent symmetrisation
    improve_target = 1.50274349
    max_trials = 16               # back‑tracking attempts

    # -----------------------------------------------------------------
    # Helper: cheap random mass transfer (used when progress stalls)
    # -----------------------------------------------------------------
    def _random_transfer(vec):
        i = rng.integers(0, n)
        if vec[i] <= 1e-12:
            return vec
        delta = vec[i] * rng.random() * 0.5
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        out = vec.copy()
        out[i] -= delta
        out[j] += delta
        return _project_simplex(out)

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & smoothed max weight -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if eps is None:
            eps = max(max_val * 0.4, 1e-10)
        else:
            eps = max(eps * eps_decay, 1e-10)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max (via FFT) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[grad_idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- step size (decaying) -----
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / math.sqrt(it) / norm_m

        # ----- back‑tracking line search -----
        improved = False
        cand = None
        cand_sc = None
        alpha = alpha0
        for _ in range(max_trials):
            trial = cur - alpha * momentum
            trial = _project_simplex(trial)
            sc = _score(trial)
            if sc < cur_sc:
                cand, cand_sc = trial, sc
                improved = True
                break
            alpha *= 0.5

        # ----- occasional uphill acceptance (simulated annealing) -----
        if not improved:
            trial = cur - alpha * momentum
            trial = _project_simplex(trial)
            sc = _score(trial)
            prob = math.exp(-(sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand, cand_sc = trial, sc
                improved = True

        # ----- accept / reject -----
        if improved:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation (helps reduce peaks) -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----- aggressive random transfer when stuck -----
        if no_improve > 800:
            tmp = _random_transfer(cur)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0
            else:
                # occasionally accept a non‑improving move to escape local minima
                if rng.random() < 0.07:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0

        # ----- full restart from best with small noise if hopeless -----
        if no_improve > 3000:
            cur = best_vec + 0.015 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum.fill(0.0)
            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < improve_target:
            break

        # ----- periodic time check -----
        if it % 400 == 0 and time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
