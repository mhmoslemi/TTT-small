    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # margin

        s = a0.sum()

            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
    else:
        n0 = 2000
        t = np.arange(n0, dtype=np.float64)
        a0 = np.exp(-t / 120.0)
        a0 /= a0.sum()

    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length() + 1)
    grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)

    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # adam variables
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    eps = None
    eps_decay = 0.9995
    min_eps = 1e-12

    base_lr = 0.2
    sym_interval = 200
    no_improve = 0

    it = 0
    while time.monotonic() < deadline:
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()

        if eps is None:
            eps = max_val * 0.5
        else:
            eps = max(eps * eps_decay, min_eps)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)

import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ------------------------------------------------------------
# Helper: maximum of linear convolution a*a (full length)
# ------------------------------------------------------------
def _max_convolution(a: np.ndarray, L: int) -> float:
    """Return max_{k} (a * a)[k] using FFT of length L (L >= 2*len(a))."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.shape[0] - 1]
    return float(conv.max())
