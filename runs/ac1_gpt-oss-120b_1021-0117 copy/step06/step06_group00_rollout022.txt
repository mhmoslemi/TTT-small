def _softmax(u):
    shift = u - u.max()
    e = np.exp(shift)
    return e / e.sum()

def _score_and_grad(u, n, L, alpha):
    a = _softmax(u)
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2*n - 1]
    max_val = conv.max()

    w = np.exp(alpha * (conv - max_val))

import time, math, numpy as np


def _softmax(u):
    """Return softmax of u (guarantees sum 1, all positive)."""
    shift = u - u.max()
    e = np.exp(shift)
    return e / e.sum()


def _score_and_grad(u, n, L, alpha):
    """
    For logits u, return
      a       – softmax(u)
      score   – 2*n*max(conv(a,a))   (sum(a)=1)
      grad_u  – gradient of the smooth‑max objective w.r.t. u
      max_val – max of the autocorrelation (before smoothing)
    """
    a = _softmax(u)

    # autocorrelation via FFT (linear convolution)
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()

    # smooth max weights (softmax on conv values)
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient of weighted sum of conv entries
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad_a = 2.0 * grad_full[idx]

    # chain rule through softmax
    dot = np.dot(a, grad_a)
    grad_u = a * (grad_a - dot)

    score = 2.0 * n * max_val
    return a, score, grad_u, max_val
