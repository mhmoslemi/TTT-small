import time
import math
import numpy as np
from scipy import special

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum=1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(a*a) for a with sum=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Candidate generators (return raw non‑negative vectors)
# ------------------------------------------------------------
def _exp_decay(alpha, n):
    return np.exp(-alpha * np.arange(n, dtype=np.float64))


def _sym_exp(alpha, n):
    d = np.abs(np.arange(n, dtype=np.float64) - (n - 1) / 2.0)
    return np.exp(-alpha * d)


def _triangular(n):
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(idx - mid) / mid)


def _power_cos(p, n):
    return np.cos(np.pi * np.arange(n, dtype=np.float64) / (n - 1)) ** p


def _hann(n):
    return 0.5 * (1 - np.cos(2 * np.pi * np.arange(n, dtype=np.float64) / (n - 1)))


def _hamming(n):
    return 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n, dtype=np.float64) / (n - 1))


def _blackman(n):
    k = np.arange(n, dtype=np.float64) / (n - 1)
    return (
        0.42
        - 0.5 * np.cos(2 * np.pi * k)
        + 0.08 * np.cos(4 * np.pi * k)
    )


def _kaiser(beta, n):
    k = np.arange(n, dtype=np.float64) / (n - 1)
    return special.i0(beta * np.sqrt(1 - (2 * k - 1) ** 2)) / special.i0(beta)


def _logistic(k, n):
    x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    return 1.0 / (1.0 + np.exp(k * x))


def _random_dirichlet(rng, n):
    # Gamma(1) -> exponential, then normalize
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    return g


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # Reserve a tiny safety margin
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # --------------------------------------------------------
    # initialise feasible point (sum = 1)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
    else:
        # fallback: modest exponential decay
        n_fallback = 2000
        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 120.0)
        a0 = a0 / a0.sum()

    n = a0.size
    # FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = a0.copy()
    best_sc = _score(best_vec, L)

    # --------------------------------------------------------
    # helper to test and keep the best candidate
    # --------------------------------------------------------
    def _try(vec):
        nonlocal best_vec, best_sc
        sc = _score(vec, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()

    # --------------------------------------------------------
    # Explore parametric families (cheap dense search)
    # --------------------------------------------------------
    families = [
        ("exp", lambda: _exp_decay(10 ** rng.uniform(-4, 0), n)),
        ("sym_exp", lambda: _sym_exp(10 ** rng.uniform(-4, 0), n)),
        ("tri", lambda: _triangular(n)),
        ("power_cos", lambda: _power_cos(rng.uniform(0.5, 8.0), n)),
        ("hann", lambda: _hann(n)),
        ("hamming", lambda: _hamming(n)),
        ("blackman", lambda: _blackman(n)),
        ("kaiser", lambda: _kaiser(rng.uniform(0.0, 12.0), n)),
        ("logistic", lambda: _logistic(rng.uniform(-0.2, 0.2), n)),
    ]

    # Number of samples per family (adjusted if time runs low)
    samples_per_family = 300

    for name, gen in families:
        for _ in range(samples_per_family):
            if time.monotonic() > deadline:
                break
            cand = gen()
            cand = np.maximum(cand, 0.0)
            if cand.sum() == 0.0:
                continue
            cand = cand / cand.sum()
            _try(cand)
        if time.monotonic() > deadline:
            break

    # Random Dirichlet points
    for _ in range(500):
        if time.monotonic() > deadline:
            break
        cand = _random_dirichlet(rng, n)
        cand = cand / cand.sum()
        _try(cand)

    # --------------------------------------------------------
    # Gradient‑based refinement (soft‑max surrogate)
    # --------------------------------------------------------
    cur = best_vec.copy()
    momentum = np.zeros_like(cur)
    eps = None
    eps_decay = 0.995
    beta = 0.9
    base_alpha = 0.8
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # FFT of current iterate
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # temperature (smoothness) adaptation
        if eps is None:
            eps = max(max_val * 0.2, 1e-9)
        else:
            eps = max(eps * eps_decay, 1e-9)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smoothed max (see derivation in previous program)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
        idx = (2 * n - 2) - np.arange(n)   # alignment
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # step size (inverse of momentum norm, decaying with sqrt(iter))
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / (math.sqrt(it) * norm_m)

        # line‑search with backtracking
        improved = False
        alpha = alpha0
        for _ in range(12):
            cand = cur - alpha * momentum
            cand = _project_simplex(cand)
            cand_sc = _score(cand, L)
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                cur = cand
                improved = True
                no_improve = 0
                break
            if cand_sc < _score(cur, L):
                cur = cand
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            # occasional random mass transfer to escape plateaus
            if no_improve % 250 == 0:
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    delta = cur[i] * rng.random() * 0.4
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp, L)
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                        cur = tmp
                        no_improve = 0

        # periodic symmetrisation (helps reduce peaks)
        if it % 100 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()
                cur = sym
                no_improve = 0

        # restart with small jitter if stuck too long
        if no_improve > 5000:
            cur = best_vec + 0.005 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum.fill(0.0)
            eps = None
            no_improve = 0

        # early exit if we have reached the target bound
        if best_sc < target:
            break

    # --------------------------------------------------------
    # Return the best solution found as a plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
