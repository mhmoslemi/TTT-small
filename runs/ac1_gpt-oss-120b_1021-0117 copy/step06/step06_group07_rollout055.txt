import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (non‑negative, sum = Z)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=Z}."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Maximum of the full linear convolution via FFT
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Maximum entry of a * a (full linear convolution)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# ------------------------------------------------------------
# Sub‑gradient based on the top‑k convolution entries (FFT‑based)
# ------------------------------------------------------------
def _grad_topk_fft(a: np.ndarray, conv_full: np.ndarray,
                   top_k: int, L_fft: int) -> np.ndarray:
    """Approximate gradient of max‑conv using the top‑k entries."""
    n = a.shape[0]
    if top_k >= conv_full.size:
        mask = np.ones(conv_full.shape, dtype=np.float64)
    else:
        idxs = np.argpartition(-conv_full, top_k - 1)[:top_k]
        mask = np.zeros(conv_full.shape, dtype=np.float64)
        mask[idxs] = 1.0
    rev_a = a[::-1]
    # convolution mask * rev_a gives the needed sums
    fft_mask = np.fft.rfft(mask, n=L_fft)
    fft_rev = np.fft.rfft(rev_a, n=L_fft)
    conv = np.fft.irfft(fft_mask * fft_rev, n=L_fft)
    start = n - 1                       # index where length‑n segment begins
    grad = 2.0 * conv[start:start + n] / max(1, top_k)
    return grad

# ------------------------------------------------------------
# Optional LP refinement (few constraints, matrix‑free)
# ------------------------------------------------------------
def _lp_refine(a: np.ndarray, max_conv: float, budget_left: float):
    """Solve a tiny LP to increase total mass while respecting the max‑conv."""
    try:
        import cvxpy as cp
    except Exception:                     # cvxpy not available
        return None

    n = a.size
    # pick the most critical convolution positions
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    top = min(200, conv.size)
    idxs = np.argpartition(-conv, top - 1)[:top]

    # build a small dense constraint matrix A (top × n)
    A = np.zeros((top, n), dtype=np.float64)
    for row, k in enumerate(idxs):
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i = np.arange(i_low, i_high + 1, dtype=int)
        j = k - i
        A[row, j] = a[i]

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= max_conv + 1e-12]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # give the solver a tiny time budget
    try:

                    verbose=False)
    except Exception:
        return None
    if g.value is None:
        return None
    g_opt = np.maximum(g.value, 0.0)
    # mix a little bit of the LP solution into the current vector
    mix = (1.0 - 0.2) * a + 0.2 * g_opt
    mix = _project_simplex(mix)
    return mix
