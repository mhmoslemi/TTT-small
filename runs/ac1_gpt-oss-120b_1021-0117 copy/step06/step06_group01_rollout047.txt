def _optimize(init_vec, time_limit):

    start = time.monotonic()

    cur = init_vec.copy()
    n = len(cur)
    best_vec = cur.copy()
    best_sc = _score(cur)

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps2 = 1e-8
    lr0 = 0.1
    t = 0

    # loops
    while time.monotonic() < time_limit:
        t += 1
        g, M, eps = _gradient(cur)
        # update moments
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g**2)

        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        lr = lr0 / np.sqrt(t)  # decaying

        step = lr * m_hat / (np.sqrt(v_hat) + eps2)
        cur_new = cur - step
        # project onto simplex
        cur_new = _project_simplex(cur_new)
        # symmetrize
        cur_new = (cur_new + cur_new[::-1]) * 0.5
        cur_new = _project_simplex(cur_new)

        cur_sc = _score(cur_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur_new.copy()
        cur = cur_new

        # occasional random perturbation every 200 iterations
        if t % 200 == 0:
            # random mass transfer mixture with new random exponential
            rng = np.random.default_rng()
            alpha = 10**rng.uniform(-4, 0)
            rand = np.exp(-alpha * np.arange(n, dtype=np.float64))
            rand = rand / rand.sum()

            cand = (1 - t_mix) * cur + t_mix * rand

                best_vec = cand.copy()
                cur = cand

def _gradient(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    M = conv.max()
    eps = max(M * 0.2, 1e-9)
    w = np.exp((conv - M) / eps)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)[:2*n-1]
    idx = (2*n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g, M, eps

def _score(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of soft‑max approximation of max(conv)
# ------------------------------------------------------------
def _gradient(a: np.ndarray):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    eps = max(max_val * 0.2, 1e-9)
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g, max_val, eps


# ------------------------------------------------------------
# Adam‑style projected gradient optimisation (symmetrised)
# ------------------------------------------------------------
def _optimize(init_vec: np.ndarray, deadline: float, rng: np.random.Generator,
              target: float = 1.50274349):
    cur = init_vec.copy()
    n = cur.size
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps2 = 1e-8
    lr0 = 0.2
    t = 0

    while time.monotonic() < deadline:
        t += 1
        g, _, _ = _gradient(cur)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / math.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps2)

        cand = cur - step
        cand = _project_simplex(cand)

        # enforce symmetry
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cur_sc = _score(cand)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cand.copy()
        cur = cand

        # occasional random mixing (helps escape plateaus)
        if t % 250 == 0:
            alpha = 10 ** rng.uniform(-4, 0)
            rand = np.exp(-alpha * np.arange(n, dtype=np.float64))
            rand /= rand.sum()
            mix = rng.random()
            mixed = (1 - mix) * cur + mix * rand
            mixed = _project_simplex(mixed)
            mixed = (mixed + mixed[::-1]) * 0.5
            mixed = _project_simplex(mixed)
            mixed_sc = _score(mixed)
            if mixed_sc < best_sc:
                best_sc = mixed_sc
                best_vec = mixed.copy()
                cur = mixed

        if best_sc < target:
            break

    return best_vec, best_sc
