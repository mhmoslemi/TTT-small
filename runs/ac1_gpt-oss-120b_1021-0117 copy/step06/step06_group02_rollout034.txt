import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 score (assumes sum == 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L_fft: int) -> float:
    """Return 2*n*max(convolution) for a (sum≈1)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(af * af, n=L_fft)[: 2 * n - 1]
    return float(2.0 * n * conv.max()), conv, af

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
