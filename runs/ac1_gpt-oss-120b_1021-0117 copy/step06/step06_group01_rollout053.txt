import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    idx = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if idx.size == 0:
        theta = 0.0
    else:
        rho = idx[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound for a sequence with sum = 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Gradient of the hard‑max objective together with its value
# ------------------------------------------------------------
def _grad_and_score(a: np.ndarray):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    k = int(np.argmax(conv))
    max_val = conv[k]

    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    # d(max conv)/da_i = 2 * a_{k-i}; objective multiplies by 2*n
    grad[mask] = 4.0 * n * a[j[mask]]
    return grad, float(2.0 * n * max_val)
