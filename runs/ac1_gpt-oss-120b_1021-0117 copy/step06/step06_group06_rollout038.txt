import time, math

# ------------------------------------------------------------
# Project a vector onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Score for a vector whose entries sum to 1:
#   score = 2 * n * max(conv(a, a))
# ------------------------------------------------------------
def _score_from_max(max_val: float, n: int) -> float:
    return 2.0 * n * max_val

    cur = cur / cur.sum()                        # ensure sum == 1

    a_fft = np.fft.rfft(cur, n=L)
