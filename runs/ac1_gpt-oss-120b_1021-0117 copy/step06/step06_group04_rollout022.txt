# ------------------------------------------------------------
# Helper: convolution via FFT and index of the maximal entry
# ------------------------------------------------------------
def _conv_max_and_index(a: np.ndarray, L: int):
    """Return (max_value, argmax_index, full convolution array)."""
    af = np.fft.rfft(a, L)
    conv_full = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv_full.argmax())
    return float(conv_full[idx]), idx, conv_full


# ------------------------------------------------------------
# Helper: sub‑gradient of the maximal convolution entry
# ------------------------------------------------------------
def _subgrad_max(a: np.ndarray, max_idx: int):
    """
    For a non‑negative vector a and the index (in the full convolution) where
    the maximum occurs, return a sub‑gradient of that maximal value w.r.t. a.
    """
    n = a.shape[0]
    # shift k = lag corresponding to max_idx (zero lag at n-1)
    k = max_idx - (n - 1)
    grad = np.zeros_like(a)

    if k >= 0:
        # a_{j+k} contribution
        if k < n:
            grad[: n - k] += a[k:]
        # a_{j-k} contribution
        if k > 0:
            grad[k:] += a[: n - k]
    else:
        s = -k
        if s < n:
            grad[s:] += a[: n - s]
        if s > 0:
            grad[: n - s] += a[s:]

    return grad
