import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    # make a copy to avoid modifying the input
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: ensure sum is exactly 1
    s = w.sum()
    if s == 0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _conv_max_and_index(a: np.ndarray, L: int):
    """
    Compute a*a via FFT, return (max value, index of the max, full conv array).
    The length of the full convolution is 2*len(a)-1.
    """
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) beats the target bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # -----------------------------------------------------------------
    # initialise a feasible point on the simplex
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0:
            a0 = np.ones_like(a0)
    else:
        # fallback: simple uniform vector
        a0 = np.ones(2000, dtype=np.float64)

    # project onto the simplex (sum = 1)
    a0 = _project_simplex(a0)
    n = a0.shape[0]

    # -----------------------------------------------------------------
    # generate a few cheap candidate families and keep the best as start
    # -----------------------------------------------------------------
    def evaluate(a):
        L = 1 << ((2 * n - 1).bit_length())
        mx, _, _ = _conv_max_and_index(a, L)
        return 2.0 * n * mx

    best_vec = a0
    best_sc = evaluate(best_vec)

    # power‑law families (both increasing and decreasing)
    for p in np.linspace(-2.0, 3.0, 11):
        a = (np.arange(1, n + 1, dtype=np.float64) ** p)
        a = _project_simplex(a)
        sc = evaluate(a)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()

    # sine window (zero at ends, peak centre)
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a = _project_simplex(a)
    sc = evaluate(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    # raised‑cosine (Hann) window
    a = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    a = _project_simplex(a)
    sc = evaluate(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    # symmetric triangular (tent) shape
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(i - (n - 1) / 2) * (2.0 / (n - 1))
    a = np.clip(a, 0, None)
    a = _project_simplex(a)
    sc = evaluate(a)
    if sc < best_sc:
        best_sc, best_vec = sc, a.copy()

    # -----------------------------------------------------------------
    # optimisation parameters
    # -----------------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc
    best_vec = cur.copy()
    best_sc = cur_sc

    # FFT lengths (reuse the same power‑of‑two for all transforms)
    L_conv = 1 << ((2 * n - 1).bit_length())          # for a*a
    L_grad = 1 << ((3 * n - 2).bit_length())          # for gradient of flatness

    avg_conv = 1.0 / (2 * n - 1)                     # target flat value

    momentum = np.zeros_like(cur)
    beta = 0.9
    eta0 = 0.5
    no_improve = 0
    target_score = 1.50274349

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # -------------------------------------------------------------
        # evaluate current score
        # -------------------------------------------------------------
        af = np.fft.rfft(cur, L_conv)
        conv = np.fft.irfft(af * af, L_conv)[: 2 * n - 1]
        cur_max = float(conv.max())
        cur_sc = 2.0 * n * cur_max
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # early exit
        if best_sc < target_score:
            break

        # -------------------------------------------------------------
        # gradient that flattens the auto‑convolution (squared error)
        # -------------------------------------------------------------
        b = conv - avg_conv                       # length 2n‑1
        a_rev = cur[::-1]

        bf = np.fft.rfft(b, L_grad)
        ar = np.fft.rfft(a_rev, L_grad)
        grad_full = np.fft.irfft(bf * ar, L_grad)   # length = 3n‑2

        # indices that map the full convolution back to the n‑dim gradient
        idx1 = np.arange(n) + n - 1
        idx2 = 2 * n - 2 - np.arange(n)
        grad = 2.0 * (grad_full[idx1] + grad_full[idx2])

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # adaptive step size
        eta = eta0 / math.sqrt(it)

        # take a projected step
        cand = cur - eta * momentum
        cand = _project_simplex(cand)

        # -------------------------------------------------------------
        # occasional symmetrisation (helps to keep the shape regular)
        # -------------------------------------------------------------
        if it % 250 == 0:
            sym = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(sym)

        # -------------------------------------------------------------
        # occasional random mass transfer when progress stalls
        # -------------------------------------------------------------
        if no_improve > 800:
            # choose a large entry and a tiny one
            i_big = int(np.argmax(cand))
            i_small = int(np.argmin(cand))
            if cand[i_big] > 1e-12:
                delta = cand[i_big] * rng.random() * 0.4
                cand[i_big] -= delta
                cand[i_small] += delta
                cand = _project_simplex(cand)

        cur = cand

    # -----------------------------------------------------------------
    # return the best vector found (as a plain Python list)
    # -----------------------------------------------------------------
    return [float(x) for x in best_vec]
