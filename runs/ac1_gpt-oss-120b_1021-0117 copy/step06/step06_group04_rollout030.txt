import time

import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return conv[: 2 * a.shape[0] - 1]

                idxs = np.arange(i_start, i_end + 1)
                prods = cur[idxs] * cur[kmax - idxs]
                total = prods.sum()
                if total > 0:
                    probs = prods / total
                    i = rng.choice(idxs, p=probs)
                else:
                    i = rng.integers(i_start, i_end + 1)

                # pick a destination index outside the overlap region
                mask = np.ones(n, dtype=bool)
                mask[i_start:i_end + 1] = False
                possible = np.where(mask)[0]
                if possible.size:
                    j = rng.choice(possible)
                    delta = cur[i] * rng.random() * 0.3
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)

                    conv_tmp = _conv_fft(tmp, L)
                    sc_tmp = 2.0 * n * conv_tmp.max() / (tmp.sum() ** 2)
                    if sc_tmp < cur_sc:
                        cur, cur_sc = tmp, sc_tmp
                        if sc_tmp < best_sc:
                            best, best_sc = tmp.copy(), sc_tmp

        # ---- early exit -------------------------------------------------
        if best_sc < target_score:
            break

    # return plain Python list
    return [float(x) for x in best]
