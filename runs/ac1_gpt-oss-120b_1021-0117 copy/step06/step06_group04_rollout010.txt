import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_grad(a: np.ndarray, L: int):
    """
    Compute max value of the linear convolution a*a, its index,
    and the exact sub‑gradient at that index.
    """
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    k = int(conv.argmax())
    max_val = float(conv[k])
    n = a.shape[0]
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad = a.copy()
    grad[mask] += a[idx[mask]]
    return max_val, k, grad


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()

        return v / s if s > 0 else np.full_like(v, 1.0 / v.size)

    # fallback random starts

    # exponential decays give smooth shapes

    # -----------------------------------------------------------------

    sym_interval = 200
    transfer_thresh = 1200
    reinit_thresh = 3000
    target_score = 1.50274349

    momenta = [np.zeros_like(c) for c in candidates]
    best_vec = None

    no_improve = 0
    it = 0

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        # pick a candidate cyclically

        a = candidates[ci]
        mom = momenta[ci]

        # convolution, max index, gradient
        cur_max, cur_idx, grad = _conv_max_and_grad(a, L)
        cur_sc = 2.0 * n * cur_max / (a.sum() ** 2)   # a.sum() == 1

        # keep global best
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # momentum update
        mom = beta * mom + (1.0 - beta) * grad

        # adaptive step size
        norm_m = np.linalg.norm(mom) + 1e-12

        # backtracking line search
        improved = False

            max_val, _, _ = _conv_max_and_grad(cand_try, L)

                tmp[j] += delta
                tmp = _project_simplex(tmp)
                max_val, _, _ = _conv_max_and_grad(tmp, L)
                tmp_sc = 2.0 * n * max_val / (tmp.sum() ** 2)
                if tmp_sc < cur_sc:
                    candidates[ci] = tmp
                    momenta[ci] = np.zeros_like(tmp)
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0
                else:
                    no_improve = max(0, no_improve - 300)

        # -----------------------------------------------------------------
        # restart from the best point with a small perturbation
        # -----------------------------------------------------------------
        if no_improve > reinit_thresh:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            candidates[ci] = cur
            momenta[ci] = np.zeros_like(cur)
            no_improve = 0

        if best_sc < target_score:
            break

    # -----------------------------------------------------------------
    # fallback safety
    # -----------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
