import time, math, numpy as np, scipy.sparse as sp, cvxpy as cp

def _project_simplex(v, Z=1.0):
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=Z}."""
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max_idx(a, L):
    """Return (max value, index) of the linear convolution a*a using FFT length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx

def _build_conv_matrix(f):
    """Return sparse Toeplitz matrix T s.t. T @ g = f * g (full linear convolution)."""
    n = f.shape[0]
    # each diagonal offset d contains constant f[d]
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)
    T = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return T

import time
import math
import numpy as np
import scipy.sparse as sp

# ------------------------------------------------------------
# Projection onto the simplex {x >= 0, sum(x)=Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum Z."""
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Full linear convolution via FFT – returns max value and its index
# ------------------------------------------------------------
def _conv_max_idx(a: np.ndarray, L: int):
    """Return (max value, index) of a*a using FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


# ------------------------------------------------------------
# Sub‑gradient from the top‑k largest convolution entries
# ------------------------------------------------------------
def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    n = a.size
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # reversed slice of the corresponding a‑segment
        g[lo : hi + 1] += 2.0 * a[lo : hi + 1][::-1]
    if top_k > 0:
        g /= float(top_k)
    return g


# ------------------------------------------------------------
# Sparse convolution matrix: (T @ g) = f * g (full linear convolution)
# ------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray):
    n = f.shape[0]
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)
    T = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return T

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2048
        a = rng.random(n, dtype=np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # simplex normalisation

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # --------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------
    step0 = 0.2
    beta = 0.9
    momentum = np.zeros_like(a)
    top_k = 5
    sym_interval = 100
    no_improve = 0
    stuck_limit = 4000
    best_vec = a.copy()
    max_val, _ = _conv_max_idx(a, L)
    best_sc = cur_sc = 2.0 * n * max_val
    target = 1.50274349
    it = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution and its maximum
        max_val, max_idx = _conv_max_idx(a, L)
        cur_sc = 2.0 * n * max_val

        # keep best so far
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_sc < target:
            break

        # gradient from the top‑k largest entries
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        g = _grad_topk(a, conv_full, top_k)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # step size (slowly decreasing)
        step = step0 / math.sqrt(it)

        # candidate step and projection onto simplex
        cand = a - step * momentum
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_val, _ = _conv_max_idx(cand, L)
        cand_sc = 2.0 * n * cand_val

        # accept improvement (or occasional uphill move)
        if cand_sc < cur_sc:
            a = cand
            cur_sc = cand_sc
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                a = cand
                cur_sc = cand_sc

        # periodic symmetrisation
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            sym_val, _ = _conv_max_idx(a_sym, L)
            sym_sc = 2.0 * n * sym_val
            if sym_sc < cur_sc:
                a = a_sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = a_sym.copy()
                    no_improve = 0

        # random perturbation when stuck
        if no_improve > stuck_limit:
            rnd = rng.random(n)
            rnd = rnd / rnd.sum()
            a = 0.75 * a + 0.25 * rnd
            a = _project_simplex(a)
            no_improve = 0

    # --------------------------------------------------------
    # Optional LP refinement (if time permits)
    # --------------------------------------------------------
    if time.monotonic() < deadline - 4.0:
        try:
            import cvxpy as cp
        except Exception:
            cp = None
        if cp is not None:
            M = _conv_max_idx(best_vec, L)[0]
            A = _build_conv_matrix(best_vec)
            g = cp.Variable(n, nonneg=True)
            prob_lp = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M + 1e-12])
            try:
                prob_lp.solve(solver="ECOS", warm_start=True, max_iters=3000)
            except Exception:
                prob_lp = None
            if prob_lp is not None and g.value is not None:
                g_opt = np.maximum(g.value, 0.0)
                mix = (best_vec + g_opt) * 0.5
                mix = _project_simplex(mix)
                mix_val, _ = _conv_max_idx(mix, L)
                mix_sc = 2.0 * n * mix_val
                if mix_sc < best_sc:
                    best_sc = mix_sc
                    best_vec = mix.copy()

    # --------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
