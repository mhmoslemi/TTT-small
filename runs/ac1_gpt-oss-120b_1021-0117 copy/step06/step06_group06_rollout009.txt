import time, math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast score for a vector already normalized to sum = 1
# ------------------------------------------------------------
def _score_norm(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)

            n_tmp = 2500
            a0 = dpss(n_tmp, NW=4.0, sym=False)

            n_tmp = 2500

    a0 = np.maximum(a0, 0.0)

    if s <= 0.0:

    a0 = a0 / s                           # normalize to sum = 1
