import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum = 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Generate a few parametric families (non‑negative, sum=1)
# ------------------------------------------------------------
def _exp_vec(n, alpha):
    v = np.exp(-alpha * np.arange(n, dtype=np.float64))
    return v / v.sum()


def _sym_exp_vec(n, alpha):
    d = np.minimum(np.arange(n, dtype=np.float64),
                   n - 1 - np.arange(n, dtype=np.float64))
    v = np.exp(-alpha * d)
    return v / v.sum()


def _gauss_vec(n, sigma):
    mid = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64)
    v = np.exp(-((x - mid) ** 2) / (2 * sigma ** 2))
    return v / v.sum()


def _cos2_vec(n):
    v = np.cos(np.pi * np.arange(n, dtype=np.float64) / (n - 1)) ** 2
    return v / v.sum()


def _triangular_vec(n, power=1.0):
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    v = np.minimum(idx, n - 1 - idx)
    if power != 1.0:
        v = v ** power
    v = v + 1e-12          # avoid all‑zero
    return v / v.sum()


def _powerlaw_vec(n, p):
    v = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return v / v.sum()


def _alternating_permutation(v):
    """Place biggest values alternately at the two ends."""
    n = v.size
    s = np.sort(v)[::-1]
    out = np.empty_like(v)
    l, r = 0, n - 1
    toggle = True
    for val in s:
        if toggle:
            out[l] = val
            l += 1
        else:
            out[r] = val
            r -= 1
        toggle = not toggle
    return out / out.sum()

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # initialise a feasible point (sum = 1)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
    else:                       # fallback: simple exponential decay
        n_fallback = 2000
        a0 = np.exp(-np.arange(n_fallback, dtype=np.float64) / 100.0)
        a0 = a0 / a0.sum()

    n = a0.size
    best_vec = a0.copy()
    best_sc = _score(best_vec)

    # --------------------------------------------------------
    # helper to test a candidate and keep the best so far
    # --------------------------------------------------------
    def _try_candidate(vec):
        nonlocal best_vec, best_sc
        sc = _score(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()

    # --------------------------------------------------------
    # Explore parametric families (cheap dense search)
    # --------------------------------------------------------
    families = [
        (_exp_vec, 200, {"alpha_min": -4, "alpha_max": 0}),
        (_sym_exp_vec, 200, {"alpha_min": -4, "alpha_max": 0}),
        (_gauss_vec, 200, {"sigma_min": 0.5, "sigma_max": 50.0}),
        (_cos2_vec, 200, {}),
        (_triangular_vec, 200, {"power_min": 0.5, "power_max": 3.0}),
        (_powerlaw_vec, 200, {"p_min": 0.2, "p_max": 2.0}),
    ]

    for func, repeats, params in families:
        for _ in range(repeats):
            if time.monotonic() > deadline:
                break
            if func is _exp_vec or func is _sym_exp_vec:
                alpha = 10 ** rng.uniform(params["alpha_min"], params["alpha_max"])
                cand = func(n, alpha)
            elif func is _gauss_vec:
                sigma = rng.uniform(params["sigma_min"], params["sigma_max"])
                cand = func(n, sigma)
            elif func is _triangular_vec:
                power = rng.uniform(params["power_min"], params["power_max"])
                cand = func(n, power)
            elif func is _powerlaw_vec:
                p = rng.uniform(params["p_min"], params["p_max"])
                cand = func(n, p)
            else:   # cosine²
                cand = func(n)
            _try_candidate(cand)

            # also test an alternating permutation of the same shape
            if rng.random() < 0.15:
                perm = _alternating_permutation(cand)
                _try_candidate(perm)

    # --------------------------------------------------------
    # Random Dirichlet draws (highly diverse)
    # --------------------------------------------------------
    for _ in range(300):
        if time.monotonic() > deadline:
            break
        alpha = rng.uniform(0.1, 4.0)
        cand = rng.dirichlet(np.full(n, alpha))
        _try_candidate(cand)

    # --------------------------------------------------------
    # Gradient‑based refinement (soft‑max surrogate of the max)
    # --------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc
    momentum = np.zeros_like(cur)

    beta = 0.92                # momentum coefficient
    base_alpha = 0.9           # base step‑size
    eps = None
    eps_decay = 0.993
    sym_interval = 80
    no_improve = 0
    improve_target = 1.50274349

    L = 1 << ((2 * n - 1).bit_length())

    while time.monotonic() < deadline:
        # ----- convolution & soft‑max weights -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # temperature for smoothed max
        if eps is None:
            eps = max(max_val * 0.2, 1e-9)
        else:
            eps = max(eps * eps_decay, 1e-9)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- adaptive step size -----
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / math.sqrt(no_improve + 1) / norm_m

        # ----- backtracking line search -----
        improved = False
        cand_sc = None
        for _ in range(12):
            cand_try = cur - alpha0 * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                improved = True
                break
            alpha0 *= 0.5

        if not improved:
            # occasional uphill acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (alpha0 + 1e-12))
            if rng.random() < prob:
                improved = True

        if improved:
            cur = cand_try
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand_try.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation (helps a lot) -----
        if (no_improve + 1) % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.45
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- restart with small perturbation if hopeless -----
        if no_improve > 6000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < improve_target:
            break

    return [float(x) for x in best_vec]
