# ------------------------------------------------------------
# Helper: projection onto the simplex {x >= 0, sum(x) = z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum z."""
    n = v.shape[0]
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

            score = 2.0 * n * _conv_max(trial, L)

            # selection
            if score <= pop_score[i]:
                pop[i] = trial
                pop_score[i] = score
                if score < best_score:
                    best_score = score
                    best_vec = trial.copy()
                    if best_score < target_score:
                        break
            # early exit check
            if time.monotonic() >= deadline:
                break
        if best_score < target_score or time.monotonic() >= deadline:
            break

    # --------------------------------------------------------
    # Local refinement: projected (sub‑)gradient descent
    # --------------------------------------------------------
    a = best_vec.copy()
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.2
    top_k = 32
    max_iter = 2000
    it = 0

    while it < max_iter and time.monotonic() < deadline:
        it += 1

        af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(af * af, n=L)[: 2 * n - 1]

        # sub‑gradient from the top‑k largest convolution entries
        if top_k >= conv_full.size:
            idxs = np.arange(conv_full.size)
        else:
            idxs = np.argpartition(-conv_full, top_k - 1)[:top_k]

        grad = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i = np.arange(i_start, i_end + 1, dtype=int)
            grad[i] += 2.0 * a[k - i]
        grad /= max(1, top_k)

        # momentum update and step
        momentum = beta * momentum + (1.0 - beta) * grad
        alpha = step0 / np.sqrt(it)
        a_tmp = a - alpha * momentum

        # projection onto simplex
        a = _project_simplex(a_tmp, 1.0)

        # enforce symmetry and renormalise
        a = (a + a[::-1]) * 0.5
        a = a / a.sum()

        cur_score = 2.0 * n * _conv_max(a, L)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            if best_score < target_score:
                break

    # --------------------------------------------------------
    # Return best sequence found (as plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
