                else:

        # ----- restart from the current best if hopeless -----
        if no_improve > 2000:
            a = best_a + 0.01 * rng.random(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            m[:] = 0.0
            v[:] = 0.0
            eps = None
            t_iter = 0
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # -------------------- final local refinement --------------------
    while time.monotonic() < deadline - 0.04:
        i = rng.integers(0, n)
        if best_a[i] <= 0.0:
            continue
        delta = best_a[i] * rng.random() * 0.05
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        af_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(af_c * af_c, n=L)[: 2 * n - 1]
        sc_c = float(2.0 * n * conv_c.max())
        if sc_c < best_sc:
            best_sc = sc_c
            best_a = cand

    # Return as a plain Python list
    return [float(x) for x in best_a]
