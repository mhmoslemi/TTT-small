def _project_simplex(v):
    v = np.asarray(v)
    if v.ndim != 1:
        v = v.ravel()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0)
    return w
