import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)  {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex with sum Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
# ------------------------------------------------------------
def _subgrad_max_convolution(a: np.ndarray, k: int) -> np.ndarray:
    """Return a sub‑gradient of the convolution entry with index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=int)
    j = k - i
    # each ordered pair (i,j) contributes a[j] to g[i] and a[i] to g[j];
    # the vectorised operations automatically double‑count when i!=j,
    # which is the correct factor of 2.
    g[i] += a[j]
    g[j] += a[i]
    return g

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a tiny safety margin for the external harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initial vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: a moderately large random vector
        n = 2048
        a = rng.random(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1 (simplex)

    n = a.size
    # FFT length – next power of two ≥ 2n‑1
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_vec = a.copy()
    # current bound = 2·n·max(conv) because sum(a)=1
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    cur_max = conv.max()
    best_score = cur_score = 2.0 * n * cur_max

    # optimisation parameters
    step0 = 1.0          # base step magnitude
    beta = 0.90          # momentum factor
    momentum = np.zeros_like(a)
    sym_interval = 250   # how often to symmetrise
    random_interval = 500
    no_improve = 0
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- compute current convolution and its maximum -----
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # ----- keep the best ever seen -----
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_score < target:
            break

        # ----- sub‑gradient of the active convolution entries -----
        eps = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - eps)[0]
        # limit the number of active indices to keep work linear
        max_active = 5
        if active.size > max_active:
            active = active[:max_active]

        g = np.zeros_like(a)
        for k in active:
            g += _subgrad_max_convolution(a, int(k))
        if active.size > 0:
            g /= float(active.size)

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- projected gradient step -----
        alpha = step0 / math.sqrt(it)
        a_tmp = a - alpha * momentum
        # small random jitter to avoid stagnation (scaled with step)
        jitter = rng.normal(scale=alpha * 0.02, size=n)
        a_tmp += jitter
        a_tmp = _project_simplex(a_tmp)   # keep on simplex (sum=1)

        # ----- evaluate candidate -----
        af_tmp = np.fft.rfft(a_tmp, n=L)
        conv_tmp = np.fft.irfft(af_tmp * af_tmp, n=L)[: 2 * n - 1]
        cand_score = 2.0 * n * conv_tmp.max()

        # ----- accept if improvement, otherwise Metropolis move -----
        if cand_score < cur_score:
            a = a_tmp
            cur_score = cand_score
            no_improve = 0
        else:
            prob = math.exp(-(cand_score - cur_score) / (alpha + 1e-12))
            if rng.random() < prob:
                a = a_tmp
                cur_score = cand_score

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            af_sym = np.fft.rfft(a_sym, n=L)
            conv_sym = np.fft.irfft(af_sym * af_sym, n=L)[: 2 * n - 1]
            sym_score = 2.0 * n * conv_sym.max()
            if sym_score < cur_score:
                a = a_sym
                cur_score = sym_score
                if sym_score < best_score:
                    best_score = sym_score
                    best_vec = a_sym.copy()
                    no_improve = 0

        # ----- occasional random mass transfer when stuck -----
        if it % random_interval == 0 and no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

        # ----- time guard -----
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
