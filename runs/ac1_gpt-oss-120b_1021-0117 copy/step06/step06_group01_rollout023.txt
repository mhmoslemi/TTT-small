def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        if rng.random() < 0.3:
            t = rng.random()
            cand = t * best_vec + (1.0 - t) * cand
            cand = _project_simplex(cand)
        else:
            s = cand.sum()
            if s <= 0.0:
                continue
            cand = cand / s

        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()
            if best_sc < 1.50274349:               # target reached
                return [float(x) for x in best_vec]

    # --------------------------------------------------------------
    # Phase 2 – stochastic local search / simulated annealing
    # --------------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc
    temperature = 0.01
    it = 0
    sym_interval = 400
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # ---- pick source index proportionally to its mass (high values) ----
        i = rng.choice(n, p=cur)                     # source (likely large)
        if cur[i] <= 0.0:
            continue
        # ---- pick a different destination uniformly ----
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # ---- transfer a random fraction of the source mass ----
        frac = rng.uniform(0.01, 0.25)                # up to 25 %
        delta = cur[i] * frac
        nxt = cur.copy()
        nxt[i] -= delta
        nxt[j] += delta

        nxt_sc = _score(nxt)

        # ---- Metropolis acceptance ----
        if nxt_sc < cur_sc or rng.random() < math.exp(-(nxt_sc - cur_sc) / temperature):
            cur = nxt
            cur_sc = nxt_sc
            if nxt_sc < best_sc:
                best_sc = nxt_sc
                best_vec = nxt.copy()
                no_improve = 0
                if best_sc < 1.50274349:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---- temperature cooling ----
        temperature *= 0.99995
        temperature = max(temperature, 1e-8)

        # ---- occasional symmetrisation (helps flatten the autocorrelation) ----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = sym / sym.sum()
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                    no_improve = 0

        # ---- restart from the best with a tiny perturbation if stuck ----
        if no_improve > 5000:
            cur = best_vec + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            cur_sc = _score(cur)
            temperature = 0.01
            no_improve = 0

    return [float(x) for x in best_vec]
