def _proj_pos_simplex(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:

    else:

    # ------------------------------------------------------------
    # final cheap random refinement
    # ------------------------------------------------------------
    while time.monotonic() < deadline - 0.05:
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        delta = cur[i] * rng.random() * 0.04
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _proj_pos_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            cur = cand
            cur_sc = cand_sc

    # return as plain Python list
    return [float(x) for x in best_vec]
