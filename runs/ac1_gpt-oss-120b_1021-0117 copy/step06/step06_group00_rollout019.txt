def propose_candidate(...):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.2)  # spare margin

    # initialization
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # random uniform
        cur = rng.random(2000, dtype=np.float64)
    # ensure nonnegative
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)   # sum=1

    n = cur.size
    L = 1 << ((2*n - 1).bit_length()) # FFT length power of 2 >= 2n-1

    # preallocate momentum
    momentum = np.zeros_like(cur)
    beta = 0.99
    step0 = 0.5
    decay = 1e-4  # step denominator factor
    alpha = 8.0  # softmax sharpness
    best = cur.copy()
    best_sc = _score(best)
    cur_sc = best_sc
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # compute fft of cur
        a_fft = np.fft.rfft(cur, n=L)
        # conv (full)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n - 1]  # length 2n-1
        max_val = conv.max()
        # compute softmax weights
        # shift by max to avoid overflow
        w_exp = np.exp(alpha * (conv - max_val))
        w = w_exp / w_exp.sum()
        # compute gradient
        w_rev = w[::-1]  # reversed
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # update momentum
        momentum = beta * momentum + (1 - beta) * grad

        # step size
        step = step0 / (1.0 + decay * it)  # slowly decreasing
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
                # early exit if good enough
                if best_sc < target:
                    break
        else:
            no_improve += 1
            # occasional acceptance: temperature decays
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
        # periodic symmetrization
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
        # random perturbation when stuck
        if no_improve > 5000:
            # random shuffle of mass among few positions
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                # move a fraction of mass from i to j
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:

        # optional early break if runs for many iterations
        if it % 20000 == 0:
            # check time left sp
            if time.monotonic() + 0.2 > deadline:
                break

    return [float(x) for x in best]

import time, math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Score for a vector already (approximately) normalized to sum=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave margin for final handling
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce non‑negativity & sum=1
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # optimizer state
    momentum = np.zeros_like(cur)
    beta = 0.9                # momentum coefficient
    step0 = 0.6               # initial step magnitude
    decay = 1e-4              # step decay factor
    alpha = 4.0               # softmax sharpness (gradual increase later)
    it = 0
    no_improve = 0
    target = 1.50274349

    best = cur.copy()
    best_sc = _score(best)
    cur_sc = best_sc

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ---------- convolution ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ---------- soft‑max weighting ----------
        # shift by max to keep numbers ≤1
        w_exp = np.exp(alpha * (conv - max_val))
        w = w_exp / w_exp.sum()

        # ---------- gradient ----------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---------- momentum ----------
        momentum = beta * momentum + (1.0 - beta) * grad

        # ---------- step ----------
        step = step0 / (1.0 + decay * it)

        # ---------- candidate ----------
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score(cand)

        if cand_sc < cur_sc:
