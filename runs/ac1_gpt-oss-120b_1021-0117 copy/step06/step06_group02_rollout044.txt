import time

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 score (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length() + 1)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    improve_target = 1.50274349
    max_trials_bt = 8          # back‑track trials when step is too large

    it = 0
    cur_sc = best_sc

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- convolution of cur with itself ----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ---- temperature for smoothed max ----
        if eps_temp is None:
            eps_temp = max(max_val * temp_factor, 1e-12)
        else:
            eps_temp = max(eps_temp * eps_decay, 1e-12)

        # ---- soft‑max weights ----
        w = np.exp((conv_full - max_val) / eps_temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smoothed max (via FFT) ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[grad_idx]

        # ---- Adam moment updates ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        # bias‑corrected moments
        bias_correction1 = 1.0 - beta1 ** it
        bias_correction2 = 1.0 - beta2 ** it
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2

        # ---- learning rate schedule ----
        lr = base_lr / math.sqrt(it)

        # ---- proposed update (with back‑tracking) ----
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        cand = None
        cand_sc = None
        for _ in range(max_trials_bt):
            cand_try = cur - step
            cand_try = _project_simplex(cand_try)
            cand_sc_try = _score(cand_try)
            if cand_sc_try < cur_sc:
                cand = cand_try
                cand_sc = cand_sc_try
                break
            step *= 0.5
        # if none improved, fall back to the last tried point (may be uphill)
        if cand is None:
            cand = cand_try
            cand_sc = cand_sc_try

        # ---- acceptance (allow occasional uphill moves) ----
        if cand_sc < cur_sc:
            accept = True
        else:
            # Metropolis‑like acceptance with temperature proportional to step norm
            prob = math.exp(-(cand_sc - cur_sc) / (np.linalg.norm(step) + 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            no_improve += 1

        # ---- periodic symmetrisation (helps reduce the peak) ----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()

        # ---- aggressive random mass transfer when stuck ----
        if no_improve > 1800:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.6
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                else:
                    # occasional acceptance of uphill move to escape plateau
                    if rng.random() < 0.04:
                        cur = tmp
                        cur_sc = tmp_sc
                        no_improve = 0

        # ---- restart from best with small noise if hopeless ----
        if no_improve > 3500:
            cur = best_vec + 0.02 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            m.fill(0.0)
            v.fill(0.0)
            eps_temp = None
            no_improve = 0
            cur_sc = _score(cur)

        # ---- early exit if target reached ----
        if best_sc < improve_target:
            break

        # ---- periodic time check ----
        if it % 400 == 0 and time.monotonic() >= deadline:
            break

    # Return as plain Python list
    return [float(x) for x in best_vec]
