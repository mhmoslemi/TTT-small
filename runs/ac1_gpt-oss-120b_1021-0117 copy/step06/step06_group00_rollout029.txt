import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # due to numerical noise, renormalise exactly
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    return w

def _conv_max(a, L):
    """Return the maximum value of the full convolution a*a using FFT."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), conv, a_fft

def _gradient(a, a_fft, conv, L, alpha=3.0):
    """
    Smooth sub‑gradient of the max‑convolution.
    Uses a soft‑max weighting with temperature 1/alpha.
    """
    max_val = conv.max()
    # softmax weights on the full convolution
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum
    # gradient = 2 * (a * w_rev) convolution (full)
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * a.size - 1]
    n = a.size
    # pick the n components that correspond to the gradient for each entry
    grad = 2.0 * grad_full[2 * n - 2 : n - 2 : -1]  # length n
    return grad

                    no_improve = 0

            # ----- update global best ---------------------------------
            if cur_score < best_score:
                best_score = cur_score
                best_vec = cur.copy()
                # early exit if target already reached
                if best_score < 1.50274349:
                    break

            # ----- stagnation handling --------------------------------
            if no_improve > max_no_improve:
                # perturb a random pair of entries
                i = rng.integers(0, n)
                if cur[i] > 0:
                    delta = cur[i] * rng.random() * 0.5
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_score = 2.0 * n * _conv_max(tmp, L)[0]
                    if tmp_score < cur_score:
                        cur, cur_score = tmp, tmp_score
                        no_improve = 0
                else:
                    # full random restart
                    cur = _project_simplex(rng.random(n))
                    cur_score = 2.0 * n * _conv_max(cur, L)[0]
                    m.fill(0.0)
                    v.fill(0.0)
                    no_improve = 0

            # ----- iteration limits ------------------------------------
            if it >= restart_interval:
                break

        # end of a restart
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # fallback: if nothing found, return a simple uniform vector
    if best_vec is None or not np.isfinite(best_score):
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
