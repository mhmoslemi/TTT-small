import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with sum Z."""
    n = v.size
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Convolution (linear) via FFT – returns the full array
# ------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length L (>=2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv


# ------------------------------------------------------------
# Max‑convolution and gradient of the active (max) entry
# ------------------------------------------------------------
def _max_and_grad(a: np.ndarray, L: int):
    """
    Return (max_value, gradient) where gradient is the sub‑gradient of the
    maximum convolution entry (the entry achieving the max).
    """
    n = a.size
    conv = _conv_full(a, L)
    k = int(np.argmax(conv))               # index of the active entry
    max_val = conv[k]

    # gradient for that entry: g[i] = 2 * a[k-i] where i is valid
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    g = np.zeros_like(a)
    if i_start <= i_end:
        idx = k - np.arange(i_start, i_end + 1, dtype=int)
        g[i_start : i_end + 1] = 2.0 * a[idx]
    return max_val, g


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence whose AC1 bound
    2*n*max(conv)/(sum**2) is as small as possible.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # Initialise
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a modest random start if no parent is provided
        a = rng.random(2048, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1  (simplex)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # --------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------
    max_val, _ = _max_and_grad(a, L)
    cur_sc = 2.0 * n * max_val
    best_sc = cur_sc
    best_vec = a.copy()

    # optimisation hyper‑parameters
    step0 = 0.7
    beta = 0.90
    momentum = np.zeros_like(a)
    sym_interval = 120          # how often we symmetrise
    no_improve = 0
    it = 0
    target = 1.50274349

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current max‑convolution and its gradient -----
        max_val, grad = _max_and_grad(a, L)
        cur_sc = 2.0 * n * max_val

        # ----- keep the best so far -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- projected gradient step (SGD with momentum) -----
        momentum = beta * momentum + (1.0 - beta) * grad
        alpha = step0 / math.sqrt(it)               # decaying step size
        a_tmp = a - alpha * momentum
        a_tmp = _project_simplex(a_tmp)               # back to simplex

        # ----- evaluate trial -----
        trial_max = _conv_full(a_tmp, L).max()
        trial_sc = 2.0 * n * trial_max

        # ----- accept if improvement, otherwise Metropolis style -----
        if trial_sc < cur_sc:

            cur_sc = trial_sc
            no_improve = 0
        else:

            if j >= i:

                a[j] += delta
                a = a / a.sum()
                no_improve = 0

    # --------------------------------------------------------

    # --------------------------------------------------------
    return [float(x) for x in best_vec]
