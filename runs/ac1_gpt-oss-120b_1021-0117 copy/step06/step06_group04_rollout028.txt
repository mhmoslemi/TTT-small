import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex. """
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_index(a: np.ndarray, L: int):
    """Return max(conv), its index and the full convolution a*a (length 2n‑1)."""
    af = np.fft.rfft(a, L)
    conv = np.fft.irfft(af * af, L)[: 2 * a.shape[0] - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    rng = np.random.default_rng(seed)
    start = time.monotonic()

    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / a.size)
            s = 1.0
        a = a / s
    else:
        # simple power‑law fallback
        n_fallback = 2000
        a = (np.arange(1, n_fallback + 1, dtype=np.float64) ** -0.7)
        a = a / a.sum()

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # FFT length for a*a

    # -----------------------------------------------------------------
    # helper for the objective (sum = 1 → score = 2·n·max(conv))
    # -----------------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        mx, _, _ = _conv_max_and_index(vec, L)
        return 2.0 * n * mx

    best_vec = a.copy()

    # Adam‑style optimisation on the simplex

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
