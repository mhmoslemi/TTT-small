def propose_candidate(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # get n
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64).copy()
        init = np.maximum(init, 0.0)
        # Normalize to sum = 1
        s = init.sum()
        if s <= 0:
            n = init.size
            init = np.full_like(init, 1.0 / n)
            s = 1.0
        init /= s
        n = init.shape[0]

        n = 2500

    L = 1 << ((2 * n - 1).bit_length())

        conv = np.fft.irfft(r * r, L)[: 2 * n - 1]
        return conv.max()

        s = v.sum()

        best_vec = np.full(n, 1.0 / n)

    return [float(x) for x in best_vec]

def _conv_max(a: np.ndarray, L: int) -> float:
    """Maximum value of the full autocorrelation a*a using FFT."""
    r = np.fft.rfft(a, L)
    conv = np.fft.irfft(r * r, L)[: 2 * a.size - 1]
    return float(conv.max())


def _score(a: np.ndarray, n: int, L: int) -> float:
    """Exact AC1 bound for a non‑negative vector a."""
    max_val = _conv_max(a, L)
    s = a.sum()
    return 2.0 * n * max_val / (s * s)
