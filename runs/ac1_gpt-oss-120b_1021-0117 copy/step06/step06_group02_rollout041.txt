def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # Determine n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2500  # fallback length

    i = np.arange(n, dtype=np.float64)
    # FFT length
    L = 1 << ((2 * n - 1).bit_length() + 1)

    def _score_vec(a):
        # assume a sum may not be 1; normalise
        s = a.sum()
        if s <= 0:
            return np.inf
        a_norm = a / s
        af = np.fft.rfft(a_norm, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    # Initial candidates list
    best_a = None
    best_sc = np.inf

    # Candidate families
    # Uniform
    a_uni = np.full(n, 1.0 / n)
    sc = _score_vec(a_uni)
    if sc < best_sc:
        best_sc = sc; best_a = a_uni.copy()

    # Exponential grid
    for alpha in np.linspace(0.0002, 0.0015, 120):
        a = np.exp(-alpha * i)
        a /= a.sum()
        sc = _score_vec(a)
        if sc < best_sc:
            best_sc = sc; best_a = a.copy()
        # also try reversed (increasing) shape
        a_rev = np.exp(-alpha * (n-1 - i))
        a_rev /= a_rev.sum()
        sc = _score_vec(a_rev)
        if sc < best_sc:
            best_sc = sc; best_a = a_rev.copy()

    # Power-law grid
    for p in np.linspace(0.1, 1.5, 100):
        a = (i + 1.0) ** (-p)
        a /= a.sum()
        sc = _score_vec(a)
        if sc < best_sc:
            best_sc = sc; best_a = a.copy()
        # also try reversed power law (increasing)
        a_rev = (n - i) ** (-p)
        a_rev /= a_rev.sum()
        sc = _score_vec(a_rev)
        if sc < best_sc:
            best_sc = sc; best_a = a_rev.copy()

    # If height_sequence_1 present use as candidate
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 0:
            a0 = a0 / s
            sc = _score_vec(a0)
            if sc < best_sc:
                best_sc = sc; best_a = a0.copy()

    # Now refine using projected gradient descent (like previous implementation)
    cur = best_a.copy()
    cur_sc = best_sc
    # momentum etc.
    momentum = np.zeros_like(cur)
    beta = 0.9
    base_alpha = 0.5
    eps = None
    eps_decay = 0.999
    max_trials = 10
    sym_interval = 300
    no_improve = 0
    improve_target = 1.50274349

    grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)

    while time.monotonic() < deadline:
        # convolution
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        if eps is None:
            eps = max(max_val * 0.5, 1e-9)
        else:
            eps = max(eps * eps_decay, 1e-9)

        # Softmax weights
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # Gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[grad_idx]

        # Momentum
        momentum = beta * momentum + (1 - beta) * g

        # Step size
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha0 = base_alpha / math.sqrt(time.monotonic() - start + 1.0) / norm_m

        # Line search
        improved = False
        cand = None
        cand_sc = None
        alpha = alpha0
        for _ in range(max_trials):
            cand_try = cur - alpha * momentum
            # project onto simplex (sum=1, nonnegative)
            cand_try = _project_simplex(cand_try)
            cand_sc = _score_vec(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5

            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score_vec(cand_try)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:

                improved = True

        if improved:

            cur_sc = cand_sc

                best_sc = cand_sc
                best_a = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:

        # Symmetrisation
        if it % sym_interval == 0:

                cur = sym

                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_a = sym.copy()
                no_improve = 0

        # Random mass transfer if stuck
        if no_improve > 2000:

            if cur[i_idx] > 1e-12:
                delta = cur[i_idx] * rng.random() * 0.5
                j_idx = rng.integers(0, n-1)
                if j_idx >= i_idx:
                    j_idx += 1
                tmp = cur.copy()
                tmp[i_idx] -= delta
                tmp[j_idx] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_vec(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp; cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc; best_a = tmp.copy()
                    no_improve = 0

        if best_sc < improve_target:
            break

        # Periodic time check
        if time.monotonic() >= deadline:
            break

    # Return best found as python list
    return [float(x) for x in best_a]

import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score_vec(a: np.ndarray, L: int, n: int) -> float:
    """Return 2*n*max(convolution) for a (not required to sum to 1)."""
    s = a.sum()
    if s <= 0.0:
        return np.inf
    a = a / s
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""

    # --------------------------------------------------------

    L = 1 << ((2 * n - 1).bit_length() + 1)

    # --------------------------------------------------------
    best_vec = None

        sc = _score_vec(a, L, n)
        if sc < best_sc:
            best_sc = sc
            best_vec = a.copy()

    # uniform
    try_candidate(np.full(n, 1.0 / n))

    for alpha in np.linspace(0.00018, 0.0016, 120):

        a /= a.sum()
