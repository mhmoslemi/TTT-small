import time

import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# Fast AC1 bound (assumes sum(a) = 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s                     # now sum=1
    else:                               # fallback: uniform
        n_fallback = 2000
        a0 = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    n = a0.shape[0]
    # FFT length (power of two) and index map for gradient extraction
    L = 1 << ((2 * n - 1).bit_length() + 1)
    grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)

    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------
    it = 0
    no_improve = 0
    eps = None
    eps_decay = 0.999          # very slow decay → smoother gradient longer
    base_alpha = 0.7           # larger base step size
    beta = 0.92                # momentum coefficient
    momentum = np.zeros_like(cur)
    improve_target = 1.50274349
    max_backtrack = 30

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution (FFT) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- temperature for smooth max -----
        if eps is None:
            eps = max(max_val * 0.5, 1e-12)
        else:
            eps = max(eps * eps_decay, 1e-12)

        # ----- soft‑max weights -----
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # ----- gradient of the smoothed max (via FFT) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[grad_idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- step size (decaying) -----
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / math.sqrt(it) / norm_m

        # ----- backtracking line‑search -----
        improved = False
        cand = None
        cand_sc = None
        for _ in range(max_backtrack):
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            alpha *= 0.5

        # ----- occasional uphill acceptance -----
        if not improved:
            cand_try = cur - alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cand = cand_try
                improved = True

        # ----- accept / reject -----
        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----- periodic symmetrisation (helps with the autocorrelation structure) -----
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- aggressive random mass‑transfer when stuck -----
        if no_improve > 1200:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.7
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                elif rng.random() < 0.08:   # occasional acceptance
                    cur = tmp
                    cur_sc = tmp_sc
                    no_improve = 0

        # ----- restart from best with a dash of noise if hopeless -----
        if no_improve > 3000:
            cur = best_vec + 0.015 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum.fill(0.0)
            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < improve_target:
            break

        # ----- periodic time check -----
        if it % 600 == 0 and time.monotonic() >= deadline:
            break

    # --------------------------------------------------------
    # final tiny local refinement (pure random hill‑climb)
    # --------------------------------------------------------
    for _ in range(2000):
        if time.monotonic() >= deadline:
            break
        i = rng.integers(0, n)
        if best_vec[i] <= 1e-12:
            continue
        delta = best_vec[i] * rng.random() * 0.3
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand

    # return as plain list of Python floats
    return [float(x) for x in best_vec]
