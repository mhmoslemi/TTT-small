import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # protect against numerical drift
    s = w.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / s

# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a) = 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length() + 1)  # power‑of‑two > 2n‑1
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
    else:
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a = np.exp(-t / 120.0)
        a = a / a.sum()

    n = a.shape[0]

    # FFT length (must exceed 2n‑1 for linear convolution)
    L = 1 << ((2 * n - 1).bit_length() + 1)

    # index map for gradient extraction (see derivation in description)
    grad_idx = (2 * n - 2) - np.arange(n, dtype=np.int64)

    # Adam optimizer state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = 0.5

    sigma = None                     # smoothing parameter for soft‑max
    sigma_decay = 0.998
    no_improve = 0
    best_vec = a.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc
    sym_interval = 250
    target = 1.50274349

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ----------------------------------------------------
        # convolution of current iterate
        # ----------------------------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----------------------------------------------------
        # smoothed max‑weight vector
        # ----------------------------------------------------
        if sigma is None:
            sigma = max_val * 0.5
        else:
            sigma = max(sigma * sigma_decay, 1e-12)

        w = np.exp((conv_full - max_val) / sigma)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # ----------------------------------------------------
        # gradient of the smoothed max (matrix‑free)
        # ----------------------------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[grad_idx]

        # ----------------------------------------------------
        # Adam update
        # ----------------------------------------------------
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = base_lr / math.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = a - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # ----------------------------------------------------
        # occasional uphill acceptance
        # ----------------------------------------------------
        improved = cand_sc < cur_sc
        if not improved:
            prob = math.exp(-(cand_sc - cur_sc) / (np.linalg.norm(step) + 1e-12))
            if rng.random() < prob:
                improved = True

        if improved:
            a = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----------------------------------------------------
        # periodic symmetrisation (helps flatten the convolution)
        # ----------------------------------------------------
        if it % sym_interval == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                a = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----------------------------------------------------
        # aggressive random mass‑transfer when stuck
        # ----------------------------------------------------
        if no_improve > 1500:
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                delta = a[i] * rng.random() * 0.6
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    a = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    if rng.random() < 0.05:
                        a = tmp
                        cur_sc = tmp_sc
                        no_improve = 0

        # ----------------------------------------------------
        # restart from best with small noise if hopeless
        # ----------------------------------------------------
        if no_improve > 3000:
            a = best_vec + 0.02 * rng.random(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            m.fill(0.0)
            v.fill(0.0)
            sigma = None
            no_improve = 0

        # ----------------------------------------------------
        # early exit if target reached
        # ----------------------------------------------------
        if best_sc < target:
            break

        # ----------------------------------------------------
        # periodic deadline check
        # ----------------------------------------------------
        if it % 500 == 0 and time.monotonic() >= deadline:
            break

    # return as plain list of Python floats
    return [float(x) for x in best_vec]
```
