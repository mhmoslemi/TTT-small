def propose_candidate(...):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # Determine length n from parent or fallback.
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2500  # fallback length
    # Prepare several starting vectors.
    candidates = []

    # 1: parent normalized to sum=1
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        a0 = a0 / a0.sum()
        candidates.append(a0)

    # 2: exponential decay
    t = np.arange(n, dtype=np.float64)
    exp_decay = np.exp(-t / (n/10.0))  # scale factor
    exp_decay = exp_decay / exp_decay.sum()
    candidates.append(exp_decay)

    # 3: linear decreasing
    lin = np.linspace(1, 0, n)
    lin = lin / lin.sum()
    candidates.append(lin)

    # 4: cosine shape (half-sine)
    theta = np.linspace(0, np.pi, n, endpoint=False) + np.pi/(2*n)  # shift avoid zeros
    cos_shape = np.sin(theta)  # sin from 0 to pi => half sine positive
    cos_shape = cos_shape / cos_shape.sum()
    candidates.append(cos_shape)

    # 5: random Dirichlet (maybe 3 samples)
    for _ in range(3):
        r = rng.random(n)
        r = r / r.sum()
        candidates.append(r)

    # function to compute score (without sum normalization)
    def _score(arr):
        # arr assumed sum=1
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    # evaluate and pick best as initial
    best_vec = None
    best_sc = float('inf')
    for cand in candidates:

        if sc < best_sc:
            best_sc = sc
            best_vec = cand.copy()

    # start SA
    cur = best_vec.copy()
    cur_sc = best_sc

    # hyperparameters
    move_factor = 0.2  # relative mass moved each modification
    num_moves = 2

    sym_interval = 200

    target = 1.50274349

    it = 0

        # propose candidate

            cand[j_idx] += delta

        # normalize and clip
        cand = np.maximum(cand, 0.0)
        total = cand.sum()
        if total == 0:

        cand = cand / total

        delta_sc = cand_sc - cur_sc

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2*n*max(a*a) for a sequence with sum≈1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
