import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of sum Z."""
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    The algorithm performs a smooth‑max gradient descent with
    simplex projection and periodic symmetrisation.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback size if no parent is provided
        n_fallback = 2000
        a = rng.random(n_fallback, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1
    n = a.size

    # FFT length large enough for 3n‑2 (needed for gradient)
    L = 1 << ((3 * n - 2).bit_length())

    # ----- hyper‑parameters -------------------------------------------
    step0 = 0.5                 # base step magnitude
    beta = 0.9                  # momentum decay
    momentum = np.zeros_like(a)
    tau0 = 0.8                  # initial temperature for soft‑max
    tau_decay = 0.9995
    tau_min = 1e-3
    target = 1.50274349
    no_improve = 0
    it = 0

    # ----- bookkeeping -------------------------------------------------
    # initial score
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max / (a.sum() ** 2)
    best_vec = a.copy()

    # ----- optimisation loop -----------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution and max
        A = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        # keep best
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # early exit
        if best_score < target:
            break

        # ---- smooth‑max weights ------------------------------------
        tau = max(tau_min, tau0 * (tau_decay ** it))
        shift = cur_max
        exp_w = np.exp((conv - shift) / tau)
        w = exp_w / exp_w.sum()               # size 2n‑1, sums to 1

        # ---- gradient via cross‑correlation ------------------------
        rev_a = a[::-1]
        R = np.fft.rfft(rev_a, n=L)
        W = np.fft.rfft(w, n=L)
        conv_rev_len = 3 * n - 2
        conv_rev = np.fft.irfft(R * W, n=L)[:conv_rev_len]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # length n

        # ---- normalised direction ----------------------------------
        dir_norm = np.linalg.norm(grad) + 1e-12
        direction = grad / dir_norm

        # ---- momentum & step ---------------------------------------
        momentum = beta * momentum + (1.0 - beta) * direction
        step = step0 / math.sqrt(it + 1.0)
        a_new = a - step * momentum

        # ---- feasibility projection --------------------------------
        a_new = np.maximum(a_new, 0.0)
        # enforce symmetry (helps the bound)
        a_new = (a_new + a_new[::-1]) * 0.5
        a_new = _project_simplex(a_new, Z=1.0)

        a = a_new

        # ---- occasional random perturbation if stuck -------------
        if no_improve > 3000:
            noise = rng.normal(scale=0.01, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a, Z=1.0)
            no_improve = 0

    # return the best candidate found
    return [float(x) for x in best_vec]
