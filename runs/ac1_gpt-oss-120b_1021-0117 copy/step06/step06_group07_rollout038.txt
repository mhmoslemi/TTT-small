import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x)=Z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, Z: float = 1.0) -> np.ndarray:
    n = v.shape[0]
    if Z < 0:
        raise ValueError("Z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - Z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Max of the full linear convolution using FFT (matrix‑free)
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Sub‑gradient approximation using the largest k convolution entries
# ------------------------------------------------------------
def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    n = a.size
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        if i0 <= i1:
            i = np.arange(i0, i1 + 1, dtype=int)
            g[i] += 2.0 * a[k - i]
    # average to keep scale reasonable
    return g / max(1.0, top_k)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise from the parent construction if present; otherwise random.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)

    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    # project onto simplex (sum = 1), scale‑invariant for the objective
    a = _project_simplex(a, Z=1.0)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_vec = a.copy()
    cur_sc = 2.0 * n * _conv_max(a, L)
    best_sc = cur_sc

    # optimisation hyper‑parameters
    step0 = 0.9                # base step magnitude
    beta = 0.95                # momentum factor
    momentum = np.zeros_like(a)
    top_k = 32                 # number of top convolution entries for gradient
    sym_interval = 200         # how often to symmetrise
    no_improve = 0
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Optimisation loop (projected SGD with momentum)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        cur_max = conv_full.max()
        cur_sc = 2.0 * n * cur_max

        # ----- keep the best -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- gradient step (projected momentum SGD) -----
        g = _grad_topk(a, conv_full, top_k)
        momentum = beta * momentum + (1.0 - beta) * g
        alpha = step0 / math.sqrt(it)          # simple decaying step size
        a_tmp = a - alpha * momentum
        a_tmp = _project_simplex(a_tmp, Z=1.0)

        # ----- evaluate candidate -----
        cand_sc = 2.0 * n * _conv_max(a_tmp, L)

        # ----- accept if improvement, otherwise Metropolis move -----
        if cand_sc < cur_sc:
            a, cur_sc = a_tmp, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, a_tmp.copy()
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                a, cur_sc = a_tmp, cand_sc

        # ----- periodic symmetrisation (helps for many optimal shapes) -----
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym, Z=1.0)
            sym_sc = 2.0 * n * _conv_max(a_sym, L)
            if sym_sc < cur_sc:
                a, cur_sc = a_sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                a[i] -= delta
                a[j] += delta
                # projection not needed (sum unchanged, non‑negative kept)
                a = np.clip(a, 0.0, 1000.0)
                no_improve = 0

        # ----- occasional gentle flattening towards uniform -----
        if it % 500 == 0:
            uniform = np.full_like(a, 1.0 / n)
            a = (1 - 0.02) * a + 0.02 * uniform
            a = _project_simplex(a, Z=1.0)

    # ------------------------------------------------------------------
    # Return the best candidate discovered (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
