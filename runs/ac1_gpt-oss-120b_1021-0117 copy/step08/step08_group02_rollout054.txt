import time, math, numpy as np

# ---------- helper utilities (top‑level, no closures) ----------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_vec(a, L, n):
    """Exact score (2·n·max(conv)) for a vector whose sum≈1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _grad_maxconv(a, L, n):
    """
    Return (grad, max_val) where grad is a sub‑gradient of
    max(conv(a,a)) and max_val = max(conv(a,a)).
    """
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    # binary mask of the maximal positions, then normalise
    w = (conv == max_val).astype(np.float64)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    # gradient via FFT (see derivation in the description)
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val

        if best_sc < target:
            break

    # return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
