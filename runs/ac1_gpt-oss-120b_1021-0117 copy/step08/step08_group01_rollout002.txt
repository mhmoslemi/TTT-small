        if restart_counter == 0:

        m = np.zeros_like(cur)
        v = np.zeros_like(cur)
        t_iter = 0
        temp = None

        while time.time() < deadline:
            t_iter += 1
            # compute convolution
            a_fft = np.fft.rfft(cur, n=L)

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast score for a vector already normalized to sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = alpha * m_hat / (np.sqrt(v_hat) + eps)

            cand = _project_simplex(cur - step)
            cand_sc = _score(cand)

            # Accept improvements; occasional uphill moves
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
            else:
                no_improve += 1
                prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
                if rng.random() < prob:
                    cur, cur_sc = cand, cand_sc

            # Random redistribution when stuck
            if no_improve > 4000:
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()

            # Periodic symmetrisation
            if it % 500 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best_vec = sym_sc, sym.copy()

            # Early exit if target reached
            if best_sc < target:
                break

        # ------------------------------------------------------------------
        if best_sc < target:
            break
        restart += 1
        if restart > 30:      # avoid endless restarts
            break

    # ---------------------------------------------------------------------- #
    # Final local refinement (hill climbing)
    cur = best_vec.copy()
    cur_sc = best_sc
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        eps = cur[i] * rng.random() * 0.02
        if eps <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                # occasional extra symmetrisation
                if rng.random() < 0.1:
                    sym = (cur + cur[::-1]) * 0.5
                    sym = _project_simplex(sym)
                    sym_sc = _score(sym)
                    if sym_sc < cur_sc:
                        cur, cur_sc = sym, sym_sc
                        if sym_sc < best_sc:
                            best_sc, best_vec = sym_sc, sym.copy()
        else:
            # tiny chance to escape flat regions
            if rng.random() < 0.001:
                cur = _project_simplex(rng.exponential(scale=1.0, size=n))
                cur_sc = _score(cur)

    # Return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
