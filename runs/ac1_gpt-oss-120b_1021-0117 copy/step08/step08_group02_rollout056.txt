import time, numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if 'height_sequence_1' in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64).copy()
        init = np.maximum(init, 0.0)
        sum_init = init.sum()
        if sum_init < 1e-12:
            init = rng.random(2000, dtype=np.float64)
            sum_init = init.sum()
        a = init / sum_init
    else:
        n = 2000
        a = rng.random(n, dtype=np.float64)
        a = a / a.sum()

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    # epsilon to avoid log(0)
    eps = 1e-12
    y = np.log(a + eps)

    # Adam state for y
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    lr0 = 0.1  # base learning rate

    best_a = a.copy()
    best_sc = _score(best_a)
    cur_sc = best_sc

    no_improve = 0
    it = 0
    target = 1.50274349
    # temperature schedule parameters
    temp = None
    decay = 0.999
    min_temp_factor = 1e-5

    # main loop
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # recompute a from y (ensure positivity)
        a = np.exp(y)
        a_sum = a.sum()
        a = a / a_sum   # normalize to sum 1

        # FFT of a
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]

        max_val = conv.max()

        # temperature
        if temp is None:
            temp = max_val * 0.2
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # compute weight w
        if temp > 0:
            w = np.exp((conv - max_val) / temp)
            w_sum = w.sum()
            if w_sum == 0.0 or not np.isfinite(w_sum):
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
            w = w / w_sum
        else:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(conv) / conv.size
                w_sum = 1.0
            w = w / w_sum

        # gradient wrt a
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * grad_full[idx]

        # convert to gradient wrt y (softmax parameterization)
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        lr = lr0 / np.sqrt(it)  # decreasing lr
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        y = y - step

        # optional symmetrization every few iterations
        if it % 500 == 0:
            a = np.exp(y)
            a = a / a.sum()
            a = (a + a[::-1]) * 0.5
            y = np.log(a + eps)

        # evaluate current score
        cur_sc = _score(a)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            j = rng.integers(0, n - 1)

            m.fill(0.0)

            beta1_pow = beta2_pow = 1.0
            no_improve = 0

        if best_sc < target:
            break

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    # initialization
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur_sum = cur.sum()
    if cur_sum < 1e-12:
        cur = rng.random(cur.shape[0], dtype=np.float64)
        cur_sum = cur.sum()
    cur = cur / cur_sum
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # param y
    eps = 1e-12
    y = np.log(cur + eps)

    # Adam
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    lr0 = 0.1

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    no_imp = 0
    it = 0

    temp = None
    decay = 0.9995
    min_temp_factor = 1e-5
    target = 1.50274349

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # reconstruct a from y, enforce simplex and symmetry
        a = np.exp(y)
        a = a / a.sum()
        # enforce symmetry
        a = (a + a[::-1]) * 0.5
        a = a / a.sum()

        # conv via FFT
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # temperature
        if temp is None:
            temp = max_val * 0.2
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # compute weight w
        if temp > 0:
            w = np.exp((conv - max_val) / temp)
            w_sum = w.sum()
            if w_sum == 0.0 or not np.isfinite(w_sum):
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
            w = w / w_sum
        else:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()

                w_sum = 1.0

            v.fill(0.0)
            beta1_pow = beta2_pow = 1.0
            no_imp = 0

        if best_sc < target:
            break

import time, numpy as np

def _score(a):
    """Score = 2 * n * max(convolution) for a vector a with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
