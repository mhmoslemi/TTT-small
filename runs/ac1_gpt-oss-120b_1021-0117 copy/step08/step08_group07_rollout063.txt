    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    max_val = conv.max()
    # temperature schedule:
    if t == 1:
        temp = max_val * 0.1
    else:
        temp = max(temp * 0.999, max_val * 1e-4)
    # softmax weighting
    w = np.exp((conv - max_val) / temp)
    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w = (conv == max_val).astype(float)
        w_sum = w.sum()
    w /= w_sum
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n-2) - np.arange(n)
    grad = 2.0 * grad_full[idx]  # derivative of weighted sum wrt cur
    # Adam update
    m = beta1 * m + (1 - beta1) * grad

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if ind.size == 0:
        theta = 0.0
    else:
        rho = ind[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast evaluation of 2·n·max(conv(a,a)) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
