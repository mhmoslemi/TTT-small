import time, math, numpy as np

# ---------- helper utilities ----------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    """Score = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _grad_max(a, conv, max_val, L):
    """Subgradient of max(conv) w.r.t. a (binary mask version)."""
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)                # length 2n-1
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

    while time.time() < deadline:
        it += 1

        # compute convolution and its maximum
        af = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(af * af, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # gradient of the (binary) maximum
        grad = _grad_max(cur, conv, max_val, L)

        # early exit if gradient vanished (likely optimal)
        if np.max(np.abs(grad)) < 1e-12:
            best_vec = cur.copy()
            best_sc = cur_sc
            break

        # AdaGrad step
        G += grad * grad
        step_adagrad = base_lr / (np.sqrt(G) + epsilon)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step_adam = base_lr * m_hat / (np.sqrt(v_hat) + epsilon)

        # blend both directions
        step = 0.5 * step_adagrad + 0.5 * step_adam

        # -------- attempt a descent step with backtracking -----------------
        cand = None
        cand_sc = None
        step_try = step.copy()
        for _ in range(5):
            # additive update then projection onto simplex
            tmp = cur - step_try * grad
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            sc = _score(tmp, L)
            if sc < cur_sc - 1e-12:
                cand, cand_sc = tmp, sc
                break
            step_try *= 0.5

        if cand is not None:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc - 1e-12:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1

        # -------- stochastic rescue if stuck --------------------------------
        if no_improve >= 800:
            # random Gaussian perturbation on the simplex
            pert = rng.normal(scale=0.05, size=n)
            tmp = cur + pert
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            sc = _score(tmp, L)
            if sc < cur_sc - 1e-12:
                cur, cur_sc = tmp, sc
                no_improve = 0
                if sc < best_sc - 1e-12:
                    best_vec, best_sc = tmp.copy(), sc
            else:
                # a cheap symmetric averaging move
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sc = _score(sym, L)
                if sc < cur_sc - 1e-12:
                    cur, cur_sc = sym, sc
                    no_improve = 0
                    if sc < best_sc - 1e-12:
                        best_vec, best_sc = sym.copy(), sc
                else:
                    no_improve = 0   # reset even if no improvement to avoid endless stalling

        # -------- periodic symmetrisation -----------------------------------
        if it % 1500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sc = _score(sym, L)
            if sc < cur_sc - 1e-12:
                cur, cur_sc = sym, sc
                if sc < best_sc - 1e-12:
                    best_vec, best_sc = sym.copy(), sc

        # early exit if target reached
        if best_sc < target:
            break

    # ----------------------------------------------------------------------
    # Return best discovered vector as plain Python floats
    return [float(x) for x in best_vec]
