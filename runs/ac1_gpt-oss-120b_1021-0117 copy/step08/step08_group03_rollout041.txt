import time, math, numpy as np, cvxpy as cp

# ---------- helper utilities ----------

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _conv_full(a, L):
    """Full linear convolution a*a via FFT (length L)."""
    af = np.fft.rfft(a, n=L)
    full = np.fft.irfft(af * af, n=L)
    return full


def _conv_max(a, L):
    """Return full convolution (first 2n‑1 entries) and its maximum."""
    full = _conv_full(a, L)
    n = a.shape[0]
    conv = full[: 2 * n - 1]
    return conv, conv.max()


def _grad_max(a, conv, max_val, L):
    """Sub‑gradient of max(conv) w.r.t. a (binary mask of active entries)."""
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]


def _solve_lp(cur, bound, mix_ratio=0.2):
    """
    Solve
        max   sum(g)
        s.t.  conv(cur, g) <= bound,  g >= 0
    Returns a convex combination (mix_ratio) of cur and the normalised solution.
    """
    n = cur.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.conv(cp.Constant(cur), g) <= bound]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                    max_iters=2000, abstol=1e-9,
                    reltol=1e-7, feastol=1e-9)
    except Exception:
        return None
    if g.value is None:
        return None
    g0 = np.maximum(g.value, 0.0)
    if g0.sum() == 0:
        return None
    g_norm = g0 / g0.sum()                 # keep sum = 1
    new_vec = (1.0 - mix_ratio) * cur + mix_ratio * g_norm
    return _project_simplex(new_vec)
