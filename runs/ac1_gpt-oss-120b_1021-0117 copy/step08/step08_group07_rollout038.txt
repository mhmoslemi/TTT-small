# --------------------------------------------------------------
#  Simple utilities

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _score(a):
    """Fast evaluation of the objective 2·n·max(conv(a,a)) for sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()

        # ----- occasional symmetrisation --------------------------------
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()

        # ----- early exit if target reached ------------------------------
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # final polishing with tiny Gaussian jitter
    while time.time() < deadline:
        pert = best_vec + rng.normal(scale=0.004, size=n)
        pert = np.clip(pert, 0.0, None)
        pert = _project_simplex(pert)
        sc = _score(pert)
        if sc < best_sc:
            best_sc = sc
            best_vec = pert.copy()
            if best_sc < target:
                break

    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
