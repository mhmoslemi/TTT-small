import time, math, numpy as np

def _project_simplex(v):
    """
    Project a vector onto the probability simplex (non‑negative, sum = 1).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(seq):
    """
    Compute the AC1 upper bound for a sequence (no clipping).  The function is
    scale‑invariant, but we evaluate it directly for compatibility with the
    provided evaluator.
    """
    seq = np.asarray(seq, dtype=np.float64)
    n = seq.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    sum_a = seq.sum()
    if sum_a <= 0:
        return float('inf')
    return float(2.0 * n * max_val / (sum_a * sum_a))
