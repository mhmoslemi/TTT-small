    idx = np.arange(n, dtype=np.float64)          # 0 … n‑1
    a = (n - idx) * (2.0 / (n * (n + 1)))        # a_i = 2·(n−i)/(n·(n+1))

    # Numerical safety: renormalise to exactly sum = 1.
    a /= a.sum()

    # Clip to the evaluator's hard bounds.
    a = np.clip(a, 0.0, 1000.0)

    # Return as a plain Python list of floats.
    return [float(v) for v in a]
