def _fast_score(a: np.ndarray, n: int, L: int) -> float:
    """Return the surrogate score 2*n*max(conv(a,a)) using an FFT."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        best_sc = evaluate_sequence(best_vec.tolist())
    except Exception:               # very unlikely, but guard against failures
        best_sc = float("inf")

    # Surrogate score for the current iterate (cheaper to compute)
    cur_sc = _fast_score(cur, n, L)

    # --------------------------------------------------------------
    # Simple stochastic hill‑climbing: move a random amount from one index to another.
    # The algorithm never leaves the simplex, respects non‑negativity,
    # and runs until the time budget is exhausted.
    while time.time() < deadline:
        # pick two distinct positions
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction (up to 40 %) of the mass at i to j
        delta = cur[i] * rng.random() * 0.4
        if delta == 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # fast surrogate evaluation
        cand_sc = _fast_score(cand, n, L)

        # keep the move if it improves the surrogate; otherwise reject.
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            # exact evaluation only when an improvement was observed
            try:
                true_sc = evaluate_sequence(cand.tolist())
            except Exception:
                true_sc = float("inf")
            if true_sc < best_sc:
                best_sc, best_vec = true_sc, cand.copy()
        # occasional random restart to avoid stagnation
        elif rng.random() < 0.001:
            cur = np.full(n, 1.0 / n, dtype=np.float64)  # back to uniform
            cur_sc = _fast_score(cur, n, L)

    # --------------------------------------------------------------
    # Final sanity check – guarantee a finite, non‑inf score.
    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not math.isfinite(final_sc):
            raise ValueError
    except Exception:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # Return as plain Python list of floats.
    return [float(x) for x in best_vec]
