def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)
    
    # init cur
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        cur = rng.random(n)
    cur = np.maximum(cur, 0.0)
    if cur.sum() == 0: cur[0] = 1.0
    cur = cur / cur.sum()   # normalize sum to 1

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_score = compute_score(best)

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.5   # learning rate

    # temperature for softmax
    temp = None

    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()

        # temperature schedule
        if temp is None:
            temp = max_val * 0.05
        else:
            temp = max(temp * 0.9999, max_val * 1e-4)

        # compute weights w
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum

        # compute gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)
        # update multiplicatively
        cur = cur * np.exp(-step * m_hat)
        # renormalize
        cur_sum = cur.sum()
        if cur_sum == 0:
            cur = np.full_like(cur, 1.0 / n)
        else:
            cur = cur / cur_sum

        cur_score = compute_score(cur)
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score

        # optional occasional symmetrisation
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym_sum = sym.sum()
            if sym_sum > 0:
                sym = sym / sym_sum
                sym_score = compute_score(sym)
                if sym_score < best_score:
                    best = sym.copy()
                    best_score = sym_score
                if sym_score < cur_score:
                    cur = sym
                    cur_score = sym_score

        # early stop
        if best_score < 1.50274349:
            break

    # final local refinement: hill climbing neighbor swap
    # run limited steps given remaining time
    remaining = deadline - time.time()
    if remaining > 1.0:
        hill_iters = int(min(5000, remaining * 500))  # approx 500 iters per sec
        for _ in range(hill_iters):
            # pick two indices
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # move small delta from i to j (or vice versa) randomly
            delta = cur[i] * rng.random() * 0.2
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            # ensure nonneg and renorm
            if cand[i] < 0:
                # compensate

                cand[i] = 0.0
                # adjust j to keep sum

            cand = np.clip(cand, 0, None)
            total = cand.sum()

            cand = cand / total
            cand_score = compute_score(cand)
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                cur = cand
            # accept if improvement maybe else not
    # final result
    return [float(x) for x in best]

import time, math, numpy as np

def _score_fft(a, L, n):
    """Fast computation of 2*n*max(conv(a,a)) assuming sum(a)=1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max()), conv, a_fft

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # --- initialise ----------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    if s == 0.0:
        cur[0] = 1.0
        s = 1.0
    cur = cur / s                     # normalise L1 to 1

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two for FFT

    # --- optimisation parameters ----------------------------------------
    best = cur.copy()
    best_sc, _, _ = _score_fft(best, L, n)

    # Adam‑style moments
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    eta0 = 0.8                       # base learning rate

    # soft‑max temperature
    temp = None
    target = 1.50274349
    it = 0

    # -------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution and max
        cur_sc, conv, a_fft = _score_fft(cur, L, n)
        max_val = conv.max()

        # temperature schedule (slow decay)
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * 0.9999, max_val * 1e-4)

        # soft‑max weights (focus on near‑max entries)
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:                         # fallback to hard max
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of weighted max (FFT‑based)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update (multiplicative version)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = eta0 / (np.sqrt(v_hat) + eps)          # element‑wise step

        # multiplicative (exponentiated) gradient step
        exp_arg = -step * m_hat
        exp_arg = np.clip(exp_arg, -20.0, 20.0)       # avoid overflow
        cur = cur * np.exp(exp_arg)

        # renormalise to L1 = 1, keep non‑negative
        cur_sum = cur.sum()
        if cur_sum == 0.0:
            cur = np.full_like(cur, 1.0 / n)
        else:
            cur = cur / cur_sum

        # keep best
        if cur_sc < best_sc:

        # occasional symmetrisation (helps flatten the convolution)
        if it % 500 == 0:

            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc

        if best_sc < target:

    remaining = deadline - time.time()
    if remaining > 1.0:

                best_sc, best, cur = cand_sc, cand.copy(), cand.copy()

    # return plain Python list
    return [float(x) for x in best]
