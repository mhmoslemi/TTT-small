def grad_max(a, conv, max_val, L):
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)
    w_rev = w[::-1]  # shape 2n-1
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2*n - 2) - np.arange(n)  # yields indices for gradient as in previous code
    grad = 2.0 * grad_full[idx]
    return grad

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L):
    """Score for a vector with sum≈1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _grad_max(a, conv, max_val, L, idx):
    """Subgradient of max(conv) w.r.t. a using FFT."""
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    grad = 2.0 * grad_full[idx]
    return grad
