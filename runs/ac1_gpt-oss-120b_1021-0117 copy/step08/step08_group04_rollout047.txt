import time, math, numpy as np

def _project_simplex(v):
   # as before, but ensure sum>0.
   v = np.asarray(v, dtype=np.float64)
   n = v.size
   # sort descending
   u = np.sort(v)[::-1]
   css = np.cumsum(u) - 1
   rho = np.where(u - css/np.arange(1, n+1) > 0)[0]
   if rho.size > 0:
       rho = rho[-1]
       theta = css[rho] / (rho + 1)
   else:
       theta = 0.0
   w = v - theta
   w[w < 0] = 0.0
   return w

def _score_norm1(a, L):
   a = np.asarray(a, dtype=np.float64)
   n = a.shape[0]
   af = np.fft.rfft(a, n=L)
   conv = np.fft.irfft(af * af, n=L)[:2*n-1]
   return float(2.0 * n * conv.max())

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (lr / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

        # step‑norm clipping
        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        # candidate and its score
        cand = _project_simplex(cur - step)
        cand_score = _score_norm1(cand, L)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / (temp + 1e-12)):
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            no_improve += 1

        # ----- occasional random mass redistribution ----------------------
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_score = _score_norm1(tmp, L)
            if tmp_score < cur_score:
                cur = tmp
                cur_score = tmp_score
                no_improve = 0
                if tmp_score < best_score:
                    best = tmp.copy()
                    best_score = tmp_score

        # ----- symmetrisation (helps convergence) ------------------------
        if it % 600 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_norm1(sym, L)
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                no_improve = 0
                if sym_score < best_score:
                    best = sym.copy()
                    best_score = sym_score

        # ----- restart from best with noise if really stuck ---------------
        if no_improve > 6000:
            cur = best + rng.normal(scale=0.005, size=n)
            cur = _project_simplex(np.maximum(cur, 0.0))
            cur_score = _score_norm1(cur, L)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        if best_score <= target:
            break

    # ----- final deterministic polishing (low temperature, tiny steps) --
    while time.time() < deadline:
        # gradient focused on the exact maximum locations
        a_fft = np.fft.rfft(best, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        w = (conv == max_val).astype(np.float64)
        if w.sum() == 0:
            break
        w = w / w.sum()
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # tiny gradient step
        step = -0.001 * grad / (np.linalg.norm(grad) + 1e-12)
        cand = _project_simplex(best + step)
        cand_score = _score_norm1(cand, L)
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
        else:
            break   # no further improvement

    # ----- return best found ------------------------------------------------
    return [float(x) for x in best]
