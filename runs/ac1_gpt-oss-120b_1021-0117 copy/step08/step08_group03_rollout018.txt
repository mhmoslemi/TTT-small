import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / (np.arange(1, n+1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L):
    # a sums to 1 (simplex)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[:2*a.shape[0]-1]
    return float(2.0 * a.shape[0] * conv.max())

                no_improve = 0
            elif no_improve > 6000:            # hard restart
                cur = _project_simplex(rng.random(n))
                momentum.fill(0.0)
                no_improve = 0

        # ---- occasional symmetrisation ---------------------------------
        if it % 1200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # early exit if target achieved
        if best_sc < target:
            break

    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
