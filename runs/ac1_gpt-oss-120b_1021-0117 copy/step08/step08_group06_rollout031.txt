def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # init vector
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        a = rng.random(n0, dtype=np.float64)
    # project onto simplex (sum=1)
    a = _project_simplex(np.maximum(a, 0.0))
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # keep best
    best = a.copy()
    best_sc = evaluate_sequence(best.tolist())
    cur_sc = _fast_score(a, n, L)

    # momentum
    momentum = np.zeros_like(a)
    beta = 0.92
    base_step = 0.8
    temp_factor = 0.1
    temp_min = 0.001
    temp_decay = 0.999
    it = 0
    no_improve = 0
    while time.time() < deadline:
        it += 1
        # FFT conv
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

        # temperature
        temp = max(temp_min * max_val, temp_factor * max_val * (temp_decay**it))
        # compute softmax weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient via convolution with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # adaptive step size
        step = base_step / math.sqrt(it)

        # try multiple scaling factors
        best_cand = None
        best_cand_sc = float('inf')
        for factor in (0.4, 0.7, 1.0, 1.3, 1.6):
            cand = _project_simplex(a - step * factor * momentum)
            cand_sc = _fast_score(cand, n, L)
            if cand_sc < best_cand_sc:
                best_cand_sc = cand_sc
                best_cand = cand

        # acceptance
        accept = False
        if best_cand_sc < cur_sc:
            accept = True
        else:
            # probability based on surrogate improvement
            prob = math.exp(-(best_cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a, cur_sc = best_cand, best_cand_sc
            no_improve = 0
            # occasional exact evaluation (e.g., every 500 iterations)
            if it % 500 == 0:
                true_sc = evaluate_sequence(a.tolist())
                if true_sc < best_sc:
                    best_sc, best = true_sc, a.copy()
        else:
            no_improve += 1

        # random perturbation when stuck
        if no_improve > 5000:
            # move a mass from random i to j
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            delta = a[i] * rng.random() * 0.3
            tmp = a.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _fast_score(tmp, n, L)
            if tmp_sc < cur_sc:
                a, cur_sc = tmp, tmp_sc
                no_improve = 0
                true_sc = evaluate_sequence(tmp.tolist())
                if true_sc < best_sc:
                    best_sc, best = true_sc, tmp.copy()
        # symmetrize periodically
        if it % 1000 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, n, L)
            if sym_sc < cur_sc:
                a, cur_sc = sym, sym_sc
                true_sc = evaluate_sequence(sym.tolist())
                if true_sc < best_sc:
                    best_sc, best = true_sc, sym.copy()
        # early exit if below target.
        if best_sc < 1.49:
            break

    try:

        if not math.isfinite(final_sc):
            raise ValueError
    except Exception:
        # fallback: uniform distribution
        best = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best]

import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
