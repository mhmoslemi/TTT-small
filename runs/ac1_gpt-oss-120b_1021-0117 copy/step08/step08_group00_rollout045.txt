import time, math, numpy as np

def _project_simplex(v):
    # similar as previous.
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # shift so that sum(v) =1 after projection onto non-negative simplex.
    # Sorting descending:
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_from_conv(max_val, n):
    return float(2.0 * n * max_val)

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

    rng = np.random.default_rng(seed)

    n = cur.shape[0]

    v = np.zeros_like(cur)       # 2nd moment

        return float(2.0 * n * max_val)   # sum(cur)=1 by construction

    # ----- main loop -------------------------------------------------------
