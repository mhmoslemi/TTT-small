def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3) # 0.3 sec margin

    # init
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        cur = rng.random(n, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    # project onto simplex
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2*n - 1).bit_length())
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # iteration parameters
    # Use exponential moving average for gradient (Adam)
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    beta1_pow = 1.0
    beta2_pow = 1.0
    lr = 0.2   # base learning rate
    eps = 1e-8

    # temperature schedule for softmax
    temp = None
    decay = 0.998

    # record iteration
    it = 0
    no_improve = 0

    # main loop
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # compute convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

        # compute weight for gradient
        # Use softmax temperature early and switch later after certain iteration
        if temp is None:
            temp = max_val * 0.1
        else:
            temp = max(temp * decay, max_val * 1e-6)
        # Use indicator if iteration > some threshold
        if it > 5000:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
            w = w / w_sum
        else:
            # softmax
            w = np.exp((conv - max_val) / temp)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
            w = w / w_sum

        # gradient computation
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = _project_simplex(cur - step_vec)

        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            # accept with prob
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
                if cand_sc < best_sc:
                    best_vec, best_sc = cand.copy(), cand_sc
            else:
                no_improve += 1

        # occasional random redistribution if stuck
        if no_improve > 2000:
            # perturb
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.2
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
            else:
                no_improve = 0

        # occasional symmetrisation (makes AC1 symmetrical)
        if it % 1500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

    return [float(x) for x in best_vec]

import time, math, numpy as np

# --------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def _project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# --------------------------------------------------------------
# Helper: fast evaluation of the AC1 bound (numerator only)
def _score(a):
    """Return 2 * n * max(convolution of a with itself). Assumes sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
