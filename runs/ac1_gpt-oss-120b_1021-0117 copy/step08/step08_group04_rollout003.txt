import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

def _score_norm1(a, L=None):
    """Score assuming sum(a)=1: returns 2*n*max(conv)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # --- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback random start if parent not present
        cur = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_score = _score_norm1(best, L)
    cur_score = best_score

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.03               # a bit more conservative than before
    max_step = 0.3

    # temperature for soft‑max weighting
    temp = None
    temp_decay = 0.9998
    min_temp_factor = 1e-5

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # --- convolution and soft‑max weights -------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.1
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:

        w = w / w_sum
