    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.size

    best_vec = cur.copy()
    best_sc = _score_fft(best_vec)
    cur_sc = best_sc

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    lr = 0.3
    eps = 1e-8

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:

        # convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # use relative tolerance to avoid missing due to fp errors
        tol = 1e-12

        if not np.any(mask):

        w = mask.astype(np.float64)

import time, math, numpy as np

def _project_simplex(v):
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_fft(a):
    """Return 2·n·max(conv) using an FFT based convolution."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a safety margin for the final return
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialise from the supplied parent if available
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score_fft(best_vec)
    cur_sc = best_sc

    # Adam optimiser state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    lr = 0.3
    eps = 1e-8

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # --------------------------------------------------------------
        # exact sub‑gradient (weights on all maximising positions)
        mask = np.abs(conv - max_val) <= 1e-12
        if not mask.any():
            mask = conv == max_val
        w = mask.astype(np.float64)
        w_sum = w.sum()
        w /= w_sum

        # --------------------------------------------------------------
        # gradient of the weighted max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # --------------------------------------------------------------
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        beta2_pow *= beta2
