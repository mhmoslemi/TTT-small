import time, math, numpy as np
import cvxpy as cp  # optional, not used but allowed

# --------------------------------------------------------------
# Helper utilities (simplex projection, fast surrogate score)
# --------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w


def _fast_score(a, n, L):
    """Surrogate: 2*n*max(conv) for a vector of length n (sum≈1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
