def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0., float(budget_s) - 0.3)  # margin

    # Initialization
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # generate random vector of length maybe 2000?
        n0 = 2000
        cur = rng.random(n0)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two

    # Evaluate initial
    cur_sc = _fast_score(cur, n, L)   # exact objective
    best_sc = cur_sc
    best_vec = cur.copy()

    # Optim variables
    momentum = np.zeros_like(cur)
    beta = 0.95
    step0 = 1.0  # base step
    decay_temp = 0.997
    min_temp_factor = 1e-6
    temp = None
    it = 0
    no_improve = 0

    # To vary between gradient and random steps
    prob_random = 0.1   # 10% random swaps

    # scale factors for line search
    scale_factors = (0.3, 0.6, 1.0, 1.5, 2.0)

    while time.time() < deadline:
        it += 1

        # Compute convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # Update temperature
        if temp is None:
            temp = max_val * 0.1
        else:
            temp = max(temp * decay_temp, max_val * min_temp_factor)

        # Choose step type
        if rng.random() < prob_random:
            # Random redistribution step
            # Choose source i with probability proportional to cur (larger mass more likely)
            src = rng.choice(n, p=cur/cur.sum())
            dst = rng.integers(0, n-1)
            if dst >= src:
                dst += 1
            # delta up to 30% of src's mass
            delta = cur[src] * rng.random() * 0.3
            cand = cur.copy()
            cand[src] -= delta
            cand[dst] += delta
            cand = _project_simplex(cand)
        else:
            # Softmax weights
            w = np.exp((conv - max_val) / temp)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
            w /= w_sum

            # Gradient
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # Momentum update
            momentum = beta * momentum + (1.0 - beta) * grad

            # Line search over scale_factors
            best_cand = None
            best_cand_sc = np.inf
            step = step0 / math.sqrt(it)
            for fac in scale_factors:
                cand_tmp = _project_simplex(cur - step * fac * momentum)
                cand_sc = _fast_score(cand_tmp, n, L)
                if cand_sc < best_cand_sc:
                    best_cand_sc = cand_sc
                    best_cand = cand_tmp
            cand = best_cand

        # Evaluate candidate
        cand_sc = _fast_score(cand, n, L)

        # Acceptance
        if cand_sc < cur_sc:
            accept = True
        else:
            # probabilistic acceptance
            prob_accept = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            accept = rng.random() < prob_accept

        if accept:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            no_improve += 1

        # Random redistribution when stuck
        if no_improve > 5000:
            # large perturbation
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _fast_score(tmp, n, L)
            if tmp_sc < cur_sc:
                cur = tmp
                cur_sc = tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
            else:
                no_improve = 0  # reset anyway

        # Symmetrisation occasionally
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, n, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()

        # Early exit if we beat target
        if best_sc < 1.5025:  # slight margin below target
            break

    # Ensure valid final sequence before return
    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not (math.isfinite(final_sc) and final_sc < 1e9):
            raise ValueError
    except Exception:
        # fallback to uniform simplex
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Exact AC1 bound (2*n*max(conv)) for a vector a on the simplex."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
