import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast score for a vector (sum not required to be 1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------ #
    # Determine problem size and initial vectors
    if "height_sequence_1" in globals():
        init_full = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 2000
        init_full = rng.random(n, dtype=np.float64)

    n = init_full.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    cur0 = _project_simplex(np.maximum(init_full, 0.0))
    best_vec = cur0.copy()
    best_sc = _score(best_vec)

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_sc = _score(uniform)
    if uni_sc < best_sc:
        best_sc, best_vec = uni_sc, uniform.copy()

    # ------------------------------------------------------------------ #
    # Hyper‑parameters
    beta = 0.95
    step0 = 0.8
    decay = 0.998
    min_temp_factor = 5e-5
    target = 1.50274349

    restart = 0
    while time.time() < deadline:
        # -------------------------------------------------------------- #
        # Initialise a new run
        if restart == 0:
            cur = best_vec.copy()
        else:
            cur = rng.exponential(scale=1.0, size=n)
            cur = _project_simplex(cur)

        cur_sc = _score(cur)
        momentum = np.zeros_like(cur)
        temp = None
        it = 0
        no_improve = 0

        # -------------------------------------------------------------- #
        # Inner optimisation loop
        while time.time() < deadline:
            it += 1

            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()

            # temperature schedule
            if temp is None:
                temp = max_val * 0.04
            else:
                temp = max(temp * decay, max_val * min_temp_factor)

            # soft‑max weighting
            w = np.exp((conv - max_val) / (temp + 1e-20))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
            w /= w_sum

            # gradient of weighted max
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # momentum & step size
            momentum = beta * momentum + (1.0 - beta) * grad
            step = step0 / math.sqrt(it)

            cand = _project_simplex(cur - step * momentum)
            cand_sc = _score(cand)

            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
            else:
                no_improve += 1
                prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
                if rng.random() < prob:
                    cur, cur_sc = cand, cand_sc

            # ---------------------------------------------------------- #
            # Targeted mass transfer every 200 iterations
            if it % 200 == 0:
                i_max = int(np.argmax(grad))
                i_min = int(np.argmin(grad))
                if cur[i_max] > 0:
                    eps = cur[i_max] * 0.02
                    tmp = cur.copy()
                    tmp[i_max] -= eps
                    tmp[i_min] += eps
                    if tmp[i_max] < 0:
                        tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        if tmp_sc < best_sc:
                            best_sc, best_vec = tmp_sc, tmp.copy()

            # ---------------------------------------------------------- #
            # Random redistribution when stuck
            if no_improve > 3000:
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()

            # ---------------------------------------------------------- #
            # Occasional symmetrisation
            if it % 800 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best_vec = sym_sc, sym.copy()

            # ---------------------------------------------------------- #
            if best_sc < target:
                break
            if time.time() >= deadline:
                break

        restart += 1
        if best_sc < target:
            break

    # ------------------------------------------------------------------ #
    # Final refinement: simple hill climbing
    cur = best_vec.copy()
    cur_sc = best_sc
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        eps = cur[i] * rng.random() * 0.02
        if eps <= 0.0:
            continue
        tmp = cur.copy()
        tmp[i] -= eps
        tmp[j] += eps
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
            if rng.random() < 0.05:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best_vec = sym_sc, sym.copy()
        else:
            if rng.random() < 0.001:
                cur, cur_sc = tmp, tmp_sc

    # Return the best sequence found
    return [float(x) for x in best_vec]
