import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort v descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Score assuming a sums to 1 (or close). Returns 2*n*max(conv)"""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

        init_full = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_full.shape[0]
    else:
        n = 2000
        init_full = rng.random(n, dtype=np.float64)

    # ensure non‑negative and simplex
    cur0 = _project_simplex(np.maximum(init_full, 0.0))
    best_vec = cur0.copy()
    best_sc = _score(best_vec)

    # a uniform baseline in case it is better
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_sc = _score(uniform)
    if uni_sc < best_sc:

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _score(a):
    """Score for a vector (assumed to sum to 1): 2·n·max(a∗a)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
