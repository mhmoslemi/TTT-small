import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast evaluation of 2·n·max(conv(a,a)) for a vector normalised to sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # initialise
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    eta0 = 0.5               # base learning rate for exponentiated gradient
    target = 1.50274349
    no_improve = 0
    it = 0

    # ------------------------------------------------------------------
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # ----- convolution -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- mask of maximal entries -----
        tol = 1e-12
        mask = np.abs(conv - max_val) <= tol
        w = mask.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ----- gradient of the (soft) max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- exponentiated‑gradient update -----
        eta = eta0 / np.sqrt(it)
        new_cur = cur * np.exp(-eta * grad)
        new_cur = _project_simplex(new_cur)
        new_sc = _score(new_cur)

        # ----- keep best -----
        if new_sc < best_sc:
            best_sc = new_sc
            best_vec = new_cur.copy()

        # ----- accept if improvement, else occasional jump -----
        if new_sc < cur_sc:
            cur, cur_sc = new_cur, new_sc
            no_improve = 0
