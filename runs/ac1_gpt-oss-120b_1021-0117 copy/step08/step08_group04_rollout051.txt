import time, math, numpy as np

# ---------- helper utilities ----------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("Input must be a 1-D array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _score_norm1(a, L=None):
    """Score assuming sum(a)=1: 2*n*max(convolution(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _gradient(cur, conv, max_val, temp, L):
    """Sub‑gradient of the smoothed max–convolution."""
    # soft‑max weights
    w = np.exp((conv - max_val) / temp)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = (conv == max_val).astype(np.float64)
        w_sum = w.sum()
    w = w / w_sum
    # reverse for correlation
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    cur_fft = np.fft.rfft(cur, n=L)
    grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
    idx = (2 * len(cur) - 2) - np.arange(len(cur))
    return 2.0 * grad_full[idx]


def _candidate_step(cur, step, max_step):
    """Take a step, clip its norm and project back onto the simplex."""
    norm = np.linalg.norm(step)
    if norm > max_step:
        step = step * (max_step / norm)
    cand = _project_simplex(cur - step)
    return cand


# ---------- main search ----------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # keep a safety margin

    # ---- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_score = _score_norm1(best, L)
    cur_score = best_score

    # ---- hyper‑parameters ------------------------------------------------
    max_step = 1.0
    step_factors = (0.05, 0.1, 0.2)          # multiple step sizes for exploration
    temp = None
    temp_decay = 0.9995
    min_temp_factor = 1e-5
    target = 1.50274349
    no_improve = 0
    it = 0

    # ---- main loop ------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution of current iterate
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature schedule (smooth max)
        if temp is None:
            temp = max_val * 0.08
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # gradient of smoothed max
        grad = _gradient(cur, conv, max_val, temp, L)

        # ---- generate candidates -----------------------------------------
        cand_best = cur
        cand_best_score = cur_score

        # gradient‑based steps with several step sizes
        for sf in step_factors:
            step = sf * grad
            cand = _candidate_step(cur, step, max_step)
            s = _score_norm1(cand, L)
            if s < cand_best_score:
                cand_best, cand_best_score = cand, s

        # power‑law perturbation (random exponent)
        if rng.random() < 0.3:
            p = rng.uniform(0.7, 1.4)
            cand = _project_simplex(cur ** p)
            s = _score_norm1(cand, L)
            if s < cand_best_score:
                cand_best, cand_best_score = cand, s

        # random mass transfer between two coordinates
        if rng.random() < 0.3:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)
            s = _score_norm1(cand, L)
            if s < cand_best_score:
                cand_best, cand_best_score = cand, s

        # occasional symmetrisation
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            s = _score_norm1(sym, L)
            if s < cand_best_score:
                cand_best, cand_best_score = sym, s

        # ---- acceptance ---------------------------------------------------
        accept = False
        if cand_best_score < cur_score:
            accept = True
        else:
            # simulated‑annealing style fallback
            prob = math.exp(-(cand_best_score - cur_score) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur, cur_score = cand_best, cand_best_score
            no_improve = 0
            if cand_best_score < best_score:
                best, best_score = cand_best.copy(), cand_best_score
        else:
            no_improve += 1

        # ---- stagnation handling -----------------------------------------
        if no_improve > 8000:
            # restart around the best solution with noise
            cur = best + rng.normal(scale=0.01, size=n)
            cur = _project_simplex(np.maximum(cur, 0.0))
            cur_score = _score_norm1(cur, L)
            no_improve = 0
            temp = None

        # ---- early exit --------------------------------------------------
        if best_score <= target:
            break

    # return the best sequence found (as plain Python floats)
    return [float(x) for x in best]
