import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Score assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

                idx_local = int(np.argmax(prod))
                i = shift + idx_local
                j = n - 1 - idx_local

            donor = i if cur[i] >= cur[j] else j
            rec = rng.integers(0, n)
            while rec == donor:
                rec = rng.integers(0, n)

            delta = cur[donor] * rng.random() * 0.18
            if delta > 0:
                tmp = cur.copy()
                tmp[donor] -= delta
                tmp[rec] += delta
                tmp_sc = _score(tmp)
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0

        # ---------- occasional symmetrisation ----------
        if it % 800 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_sc, best_vec = sym_sc, sym.copy()
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0

        # ---------- random restart on stagnation ----------
        if no_improve > 4000:
            cur = rng.exponential(scale=1.0, size=n)
            cur = _project_simplex(cur)
            cur_sc = _score(cur)
            no_improve = 0

        if best_sc < target:
            break

    # ------------------------------------------------------------------ #
    # Fine‑grained local hill‑climbing
    cur = best_vec.copy()
    cur_sc = best_sc
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        eps = cur[i] * rng.random() * 0.015
        if eps <= 0.0:
            continue
        tmp = cur.copy()
        tmp[i] -= eps
        tmp[j] += eps
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
            # occasional extra symmetrisation
            if rng.random() < 0.07:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best_vec = sym_sc, sym.copy()

    # Return the best sequence found
    return [float(x) for x in best_vec]
