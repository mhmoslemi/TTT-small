import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Fast surrogate: 2 * n * max(full linear convolution) for vectors with sum≈1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ------------------------------------------------------------------
    # initialise
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())          # FFT size (next power of two)

    # best-so-far (true score via evaluate_sequence)
    best_vec = cur.copy()
    best_sc = evaluate_sequence(best_vec.tolist())

    # fast surrogate for the current iterate
    cur_sc = _fast_score(cur, n, L)

    # optimisation state
    momentum = np.zeros_like(cur)
    beta = 0.90                 # momentum decay
    step0 = 0.8                 # base step size
    it = 0
    no_improve = 0
    temp = None
    decay = 0.997
    min_temp_factor = 1e-5
    exact_eval_interval = 800
    target_score = 1.50274349

    while time.time() < deadline:
        it += 1

        # ---------- convolution and temperature ----------
        a_fft = np.fft.rfft(cur, n=L)

        conv = conv_full[: 2 * n - 1]
