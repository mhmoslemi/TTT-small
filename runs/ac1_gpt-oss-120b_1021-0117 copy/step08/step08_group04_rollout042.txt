import time, math, numpy as np

def _project_simplex(v):
    """Project vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

def _score_norm1(a):
    """Score assuming sum(a)=1: 2*n*max(convolution(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------- initialisation -------------------------------------------------
    if "height_sequence_1" in globals():
        init_h = np.asarray(height_sequence_1, dtype=np.float64)
        init_h = _project_simplex(np.maximum(init_h, 0.0))
        score_h = _score_norm1(init_h)
    else:
        init_h = None
        score_h = np.inf

    # a second random start (length taken from the first if available)
    n_guess = init_h.shape[0] if init_h is not None else 2500
    init_r = rng.random(n_guess, dtype=np.float64)
    init_r = _project_simplex(np.maximum(init_r, 0.0))
    score_r = _score_norm1(init_r)

    if score_h < score_r:
        cur = init_h
        cur_score = score_h
    else:
        cur = init_r
        cur_score = score_r

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_score = cur_score

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.07               # base learning rate
    max_step = 0.8

    # temperature schedule
    temp = None
    temp_decay = 0.9999
    min_temp_factor = 1e-5

    target = 1.50274349

    while time.time() < deadline:

        # ----- convolution & temperature -----------------------------------
        a_fft = np.fft.rfft(cur, n=L)

        max_val = conv_full.max()

        if temp is None:
            temp = max_val * 0.15
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # soft‑max weighting of all lags (smooth sub‑gradient of max)
        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()

                if tmp_score < cur_score:

                    no_improve = 0

                        best, best_score = tmp.copy(), tmp_score

        # ----- full restart if really stuck ----------------------------------
        if no_improve > 8000:
            cur = best + rng.normal(scale=0.01, size=n)
            cur = _project_simplex(np.maximum(cur, 0.0))
            cur_score = _score_norm1(cur)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # ----- termination ---------------------------------------------------
        if best_score <= target:
            break

    return [float(x) for x in best]
