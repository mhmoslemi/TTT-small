import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_from_conv(conv, n):
    """Score given full convolution and length n (sum is already 1)."""
    return float(2.0 * n * conv.max())

            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()

        # -------------------------------------------------------------- #
        # Periodic symmetrisation (helps enforce known symmetry of optimum)
        if t % 600 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_from_conv(*_conv_max(sym), n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # -------------------------------------------------------------- #
        if best_sc < target:
            break

    # ------------------------------------------------------------------ #
    # Final fine‑grained local search (tiny mass swaps)
    cur = best_vec.copy()
    cur_sc = best_sc
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        eps_move = cur[i] * rng.random() * 0.015
        if eps_move <= 0.0:
            continue
        tmp = cur.copy()
        tmp[i] -= eps_move
        tmp[j] += eps_move
        tmp = _project_simplex(tmp)
        tmp_sc = _score_from_conv(*_conv_max(tmp), n)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
            # occasional extra symmetrisation
            if rng.random() < 0.12:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score_from_conv(*_conv_max(sym), n)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best_vec = sym_sc, sym.copy()
        else:
            # very small chance to accept worsening move
            if rng.random() < 0.0005:
                cur, cur_sc = tmp, tmp_sc

    # Return the best candidate as plain Python floats
    return [float(x) for x in best_vec]
