def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n):
    a = np.asarray(a, dtype=np.float64)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum = 1, v >= 0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n):
    """Score for a vector with sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # keep a safety margin

    # ---- initialise -------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)                     # sum = 1

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())            # FFT length (power of two)
    idx = (2 * n - 2) - np.arange(n)                # indices for gradient extraction

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_vec = cur.copy()
    best_sc = _score_simplex(best_vec, n)
    cur_sc = best_sc

    # deterministic candidates (different smooth windows)
    def _gen_windows():
        x = np.arange(n, dtype=np.float64)
        # Hann window
        hann = 0.5 - 0.5 * np.cos(2 * np.pi * x / (n - 1))
        hann /= hann.sum()
        # Hamming window
        hamming = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1))
        hamming /= hamming.sum()
        # Gaussian window
        sigma = n / 6.0
        gauss = np.exp(-0.5 * ((x - (n - 1) / 2) / sigma) ** 2)
        gauss /= gauss.sum()
        # Exponential decay
        dec = np.exp(-x / (n / 5))
        dec /= dec.sum()
        return [hann, hamming, gauss, dec]

    for w in _gen_windows():
        sc = _score_simplex(w, n)
        if sc < best_sc:
            best_sc, best_vec = sc, w.copy()

    # ---- optimisation loop -----------------------------------------------
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # convolution of current vector
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_conv = conv.max()

        # soft‑max temperature (starts relatively high, then cools)
        if it == 1:
            T = max_conv * 0.07
        else:
            T = max(T * 0.998, max_conv * 1e-4)

        # soft‑max weighting over convolution entries
        w = np.exp((conv - max_conv) / T)

        if w_sum == 0.0:

            w_sum = w.sum()
        w /= w_sum

        # gradient of the smoothed max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        j = rng.integers(0, n - 1)
        if j >= i:

        tmp = best_vec.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score_simplex(tmp, n)
        if tmp_sc < best_sc:
            best_sc, best_vec = tmp_sc, tmp.copy()
            if best_sc < target:
                break

    return [float(x) for x in best_vec]
