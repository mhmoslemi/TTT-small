import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Score assuming a already sums to 1 (i.e. 2*n*max(conv))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------ #
    # problem size & initial vectors
    if "height_sequence_1" in globals():
        init_full = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_full.shape[0]
    else:
        n = 2000
        init_full = rng.random(n, dtype=np.float64)

    L = 1 << ((2 * n - 1).bit_length())

    # start from projected parent and a uniform baseline
    cur0 = _project_simplex(np.maximum(init_full, 0.0))
    best_vec = cur0.copy()
    best_sc = _score(best_vec)

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_sc = _score(uniform)
    if uni_sc < best_sc:
        best_sc, best_vec = uni_sc, uniform.copy()

    # ------------------------------------------------------------------ #
    # hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha0 = 0.12           # base learning rate
    temp_factor = 0.07
    decay = 0.9995
    min_temp_factor = 1e-4
    target = 1.50274349
    restart = 0

    # ------------------------------------------------------------------ #
    while time.time() < deadline:
        # initial vector for this restart
        if restart == 0:
            cur = best_vec.copy()
        else:
            cur = rng.exponential(scale=1.0, size=n)
            cur = _project_simplex(cur)

        cur_sc = _score(cur)

        # Adam moments
        m = np.zeros_like(cur)
        v = np.zeros_like(cur)
        it = 0
        no_improve = 0
        temp = None

        while time.time() < deadline:
            it += 1
            # FFT of current vector
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()

            # temperature schedule
            if temp is None:
                temp = max_val * temp_factor
            else:
                temp = max(temp * decay, max_val * min_temp_factor)

            # soft‑max weights
            w = np.exp((conv - max_val) / (temp + 1e-20))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
            w /= w_sum

            # gradient of weighted max
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = alpha0 * m_hat / (np.sqrt(v_hat) + eps)

            cand = _project_simplex(cur - step)
            cand_sc = _score(cand)

            # accept if improvement, otherwise occasional uphill move
            if cand_sc < cur_sc:

                no_improve = 0
                if cand_sc < best_sc:

            else:

    cur_sc = best_sc

        if j >= i:

        if eps_move <= 0.0:

        tmp = cur.copy()
        tmp[i] -= eps_move
        tmp[j] += eps_move
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
            if rng.random() < 0.1:

        else:
            if rng.random() < 0.001:
                cur, cur_sc = tmp, tmp_sc

    # return as plain Python floats
    return [float(x) for x in best_vec]
