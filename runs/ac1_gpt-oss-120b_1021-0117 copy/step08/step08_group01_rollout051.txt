def _project_simplex(v):
    # as earlier: Euclidean projection onto simplex sum=1 (nonnegative)
    v = np.asarray(v, dtype=np.float64)
    # Ensure nonnegative: but algorithm itself yields nonnegative result; we can sort descending.
    # Code as before.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1,n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = v - theta
    w[w<0] = 0.0
    return w

import time, math, numpy as np

def _project_simplex(v, target=1.0):
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - target
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Score = 2*n*max(a*a) for a already on the simplex (sum=1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
