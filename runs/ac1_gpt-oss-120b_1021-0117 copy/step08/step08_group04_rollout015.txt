import time, math, numpy as np

# ---------- utilities -------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, L):
    """Score 2*n*max(conv) / (sum(a)**2) using FFT, assuming a is a 1‑D array."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s == 0:
        return np.inf
    return float(2.0 * n * max_val / (s * s))

        max_val = conv.max()

        # subgradient from arg‑max positions
        mask = (conv == max_val)
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
        else:
            w = mask.astype(np.float64) / cnt
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # normalized descent direction
        gn = np.linalg.norm(grad)
        if gn > 0:
            direction = grad / gn
            cand = _project_simplex(cur - step * direction)
        else:
            cand = cur.copy()
        cand_score = _fast_score(cand, L)

        # acceptance
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
            step = min(step * 1.07, max_step)
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            no_improve += 1
            if no_improve % 250 == 0:
                step = max(step * 0.70, min_step)

        # occasional random redistribution (escape local minima)
        if no_improve > improve_patience:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_score = _fast_score(tmp, L)
            if tmp_score < cur_score:
                cur, cur_score = tmp, tmp_score
                no_improve = 0
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score

        # random symmetrisation (helps due to inherent symmetry of objective)
        if rng.random() < 0.008:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _fast_score(sym, L)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                no_improve = 0
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score

        # hard restart if stuck too long
        if no_improve > restart_patience:
            cur = best + rng.normal(scale=0.001, size=n)
            cur = _project_simplex(np.maximum(cur, 0.0))
            cur_score = _fast_score(cur, L)
            no_improve = 0
            step = 0.12

        # early exit if we already beat the target bound
        if best_score <= target:
            break

    # ---------------------------------------------------------------
    return [float(x) for x in best]
