import time, math, numpy as np

def _project_simplex(v):
    # clip to >=0 and normalize to sum=1
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # uniform fallback
        return np.full_like(v, 1.0 / v.size)
    return v / s

def _score_vec(a, n, L):
    # compute 2 * n * max(conv)
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def _gradient(a, a_fft, w, n, L):
    # w is weight vector of length 2*n-1; compute gradient
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad

def run_opt(initial, deadline, rng, n, L):
    # returns best vector and score
    a = initial.copy()
    # initial projection
    a = _project_simplex(a)
    best = a.copy()
    best_sc = _score_vec(best, n, L)
    # Adam variables
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.1  # base step
    temp = None
    decay = 0.995
    min_temp_factor = 1e-5
    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        # temperature schedule
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        grad = _gradient(a, a_fft, w, n, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / math.sqrt(it)
        delta = step * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(a - delta)
        cand_sc = _score_vec(cand, n, L)

        if cand_sc < best_sc:
            best, best_sc = cand.copy(), cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # maybe accept candidate as new a to continue descent even if not better
        # For stability accept if cand_sc <= best_sc or random.
        if cand_sc <= best_sc or rng.random() < 0.05:
            a = cand
        else:
            # keep a unchanged maybe small chance to move
            a = cand if rng.random() < 0.01 else a

        # occasional random perturbation
        if no_improve > 2000:
            # random redistribution
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            delta_mass = a[i] * rng.random() * 0.4
            tmp = a.copy()
            tmp[i] -= delta_mass
            tmp[j] += delta_mass
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp, n, L)
            if tmp_sc < best_sc:
                best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0
            a = tmp if tmp_sc < _score_vec(a, n, L) else a

        # symmetrisation occasional
        if it % 1000 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym, n, L)
            if sym_sc < best_sc:
                best, best_sc = sym.copy(), sym_sc
                no_improve = 0
            a = sym if sym_sc < _score_vec(a, n, L) else a

        # early exit if reach target (with some tolerance)
        if best_sc < 1.50274349:

import time, math, numpy as np

# ---------- simple simplex projection (non‑negative, sum = 1) ----------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # uniform fallback
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ---------- score (the quantity the evaluator returns) ----------
def _score_vec(a: np.ndarray, n: int, L: int) -> float:
    """Return 2·n·max(conv) for a (already normalised to sum≈1)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ---------- gradient of the soft‑max‑smoothed max ----------
def _grad_softmax(a: np.ndarray, a_fft: np.ndarray,
                  w: np.ndarray, n: int, L: int) -> np.ndarray:
    """Gradient of Σ w_k·(a*a)_k with respect to a."""
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]


# ---------- optimiser (Adam + annealed soft‑max) ----------
def _run_opt(initial: np.ndarray, deadline: float,
             rng: np.random.Generator, n: int, L: int):
    a = _project_simplex(initial.copy())
    best = a.copy()
    best_sc = _score_vec(best, n, L)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 1.0
    temp = None
    decay = 0.995
    min_temp = 1e-5
    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        # temperature (soft‑max sharpness)
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp)

        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:               # pathological – fall back to hard max
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of the smoothed objective
        grad = _grad_softmax(a, a_fft, w, n, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / math.sqrt(it)
        delta = step * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(a - delta)
        cand_sc = _score_vec(cand, n, L)

        # accept if improvement, otherwise occasional stochastic move
        if cand_sc < cur_sc:
            a = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            no_improve += 1
            if rng.random() < 0.02:      # occasional exploration
                a = cand
                cur_sc = cand_sc

        # update global best
        if cand_sc < best_sc:
            best, best_sc = cand.copy(), cand_sc
            no_improve = 0

        # random redistribution when stuck
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta_mass = a[i] * rng.random() * 0.4
            tmp = a.copy()
            tmp[i] -= delta_mass
            tmp[j] += delta_mass
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp, n, L)
            if tmp_sc < best_sc:
                best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0
            a = tmp

        # periodic symmetrisation (helps flatten peaks)
        if it % 800 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym, n, L)
            if sym_sc < best_sc:
                best, best_sc = sym.copy(), sym_sc
                no_improve = 0
            a = sym

        # early exit if target reached
        if best_sc < 1.50274349:
            break

    return best, best_sc
