def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)
    # Determine length
    if 'height_sequence_1' in globals():
        n = int(height_sequence_1.shape[0])
        cur = np.asarray(height_sequence_1, dtype=np.float64)
        cur = _project_simplex(np.maximum(cur, 0.0))
    else:
        n = 2500
        cur = rng.random(n)
        cur = _project_simplex(np.maximum(cur, 0.0))
    L = 1 << ((2 * n - 1).bit_length())
    best = cur.copy()
    best_score = _score_norm1(best)
    cur_score = best_score
    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.05
    max_step = 0.5
    # temperature schedule
    temp = None
    temp_decay = 0.9995
    min_temp_factor = 1e-5
    it = 0
    no_improve = 0
    target = 1.50274349
    # main loop
    while time.time() < deadline:
        it += 1
        # FFT of current
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]  # length 2n-1
        max_val = conv.max()
        # temperature
        if temp is None:
            temp = max_val * 0.08
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)
        # weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum
        # gradient via FFT
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (lr0 / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)
        # clip step norm
        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)
        cand = _project_simplex(cur - step)
        cand_score = _score_norm1(cand)
        # acceptance
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / (temp + 1e-12)):
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            no_improve += 1
        # random redistribution if stuck
        if no_improve > 2000:
            # generate few random redistribution attempts
            for _ in range(3):
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score_norm1(tmp)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    no_improve = 0
                    if tmp_score < best_score:
                        best, best_score = tmp.copy(), tmp_score
                    break
        # symmetrisation more often
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_norm1(sym)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                no_improve = 0
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score
        # restart when stuck long
        if no_improve > 5000:
            cur = best + rng.normal(scale=0.02, size=n)
            cur = _project_simplex(np.maximum(cur, 0.0))
            cur_score = _score_norm1(cur)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
        if best_score <= target:
            break
    # final fine-tune short run with smaller lr if time left
    while time.time() < deadline - 0.1:
        # take small steps with decreasing lr
        a_fft = np.fft.rfft(best, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        # gradient approximated by max location only
        k_star = np.argmax(conv)
        # gradient for max shift
        grad = np.zeros_like(best)
        # compute shifted contributions
        # for idx i where 0 <= k_star - i < n
        low = max(0, k_star - (n - 1))
        high = min(n - 1, k_star)
        # for i in [low, high], grad[i] = 2 * best[k_star - i]
        # vectorized
        i_range = np.arange(low, high + 1)
        grad[i_range] = 2.0 * best[k_star - i_range]
        # simple step
        step = 0.01 * grad / (np.linalg.norm(grad) + eps)
        cand = _project_simplex(best - step)
        cand_score = _score_norm1(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
        else:
            break
    return [float(x) for x in best]

import time, math, numpy as np

def _project_simplex(v):
    """Project a vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

def _score_norm1(a):
    """Score for a vector with sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
        cur = _project_simplex(np.maximum(cur, 0.0))
    else:
        # fallback length if no parent is provided
        n = 2500
        cur = rng.random(n, dtype=np.float64)
        cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_score = _score_norm1(best)
    cur_score = best_score

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.1          # larger initial learning rate
    max_step = 0.8

    # temperature schedule
    temp = None
    temp_decay = 0.9995
    min_temp_factor = 1e-5

    it = 0
    no_improve = 0
    target = 1.50274349

    # ---- main optimisation loop -----------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution and temperature
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.08
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # softmax weighting of convolution entries
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum

        # gradient of weighted sum (matrix‑free)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (lr0 / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

        # clip step length
        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        cand = _project_simplex(cur - step)
        cand_score = _score_norm1(cand)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / (temp + 1e-12)):
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            no_improve += 1

        # ---- occasional diversification ----------------------------------
        if no_improve > 2000:
            for _ in range(3):
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score_norm1(tmp)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    no_improve = 0
                    if tmp_score < best_score:
                        best, best_score = tmp.copy(), tmp_score
                    break

        # ---- symmetrisation (more frequent) -----------------------------
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_norm1(sym)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                no_improve = 0
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score

        # ---- restart when truly stuck ------------------------------------
        if no_improve > 5000:
            cur = best + rng.normal(scale=0.02, size=n)
            cur = _project_simplex(np.maximum(cur, 0.0))
            cur_score = _score_norm1(cur)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # early exit if target reached
        if best_score <= target:
            break

    # ---- final fine‑tuning on the peak position --------------------------
    while time.time() < deadline - 0.1:
        a_fft = np.fft.rfft(best, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        k_star = int(np.argmax(conv))

        # gradient for the max‑position only (exact)
        low = max(0, k_star - (n - 1))
        high = min(n - 1, k_star)
        i_range = np.arange(low, high + 1)
        grad = np.zeros_like(best)
        grad[i_range] = 2.0 * best[k_star - i_range]

        step = 0.01 * grad / (np.linalg.norm(grad) + eps)
        cand = _project_simplex(best - step)
        cand_score = _score_norm1(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
        else:
            break

    # ---- try mixing with the reversed sequence -------------------------
    rev = best[::-1]
    for alpha in np.linspace(0.0, 1.0, 21):
        mixed = (1.0 - alpha) * best + alpha * rev
        mixed = _project_simplex(mixed)          # essentially a no‑op
        mixed_score = _score_norm1(mixed)
        if mixed_score < best_score:
            best, best_score = mixed, mixed_score

    # ---- fallback sanity check -----------------------------------------
    if best_score == float('inf') or best.size == 0:
        # uniform fallback
        best = np.full(n, 1.0 / n)
        best_score = _score_norm1(best)

    return [float(x) for x in best]
