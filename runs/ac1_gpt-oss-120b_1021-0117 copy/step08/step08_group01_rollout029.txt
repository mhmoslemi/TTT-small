def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # Determine n
    if 'height_sequence_1' in globals():
        init_full = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_full.shape[0]
    else:
        n = 2500
        init_full = rng.random(n).astype(np.float64)

    # projection and score
    def _project_simplex(v):
        v = np.asarray(v, dtype=np.float64)
        n_ = v.size
        # sort descending
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n_+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        return w

    def _score(a):
        a = np.asarray(a, dtype=np.float64)
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    # baseline
    cur = _project_simplex(np.maximum(init_full, 0.0))
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # Also try uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_sc = _score(uniform)
    if uni_sc < best_sc:
        best_sc, best_vec = uni_sc, uniform.copy()

    # Setup parameters
    L = 1 << ((2 * n - 1).bit_length())
    beta = 0.9
    step0 = 0.2
    it = 0
    momentum = np.zeros_like(cur)

    # main loop
    while time.time() < deadline:
        it += 1
        # compute conv via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        # find index of max (first)
        k0 = int(np.argmax(conv_full))

        # gradient for max index (subgradient)
        grad = np.zeros_like(cur)
        # range of valid idx
        start_idx = max(0, k0 - (n - 1))
        end_idx = min(k0, n - 1)
        idx = np.arange(start_idx, end_idx + 1)
        j = k0 - idx
        grad[j] = 2.0 * cur[idx]

        # momentum
        momentum = beta * momentum + (1 - beta) * grad

        # step size schedule
        step = step0 / math.sqrt(it)

        # candidate
        cand = _project_simplex(cur - step * momentum)

        # optionally enforce symmetry to reduce max
        if rng.random() < 0.05:  # occasional symmetrization
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        cand_sc = _score(cand)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights attempting to minimise the
    autocorrelation upper bound evaluated by `evaluate_sequence`.
    """

        n = 2500

    def _project_simplex(v):
        v = np.asarray(v, dtype=np.float64)
        n_ = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n_ + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        return w

    # ------------------------------------------------------------------ #
    # Helper: compute the score (2 * n * max convolution) for a simplex vector
    def _score(a):
        a = np.asarray(a, dtype=np.float64)
        L = 1 << ((2 * n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())
