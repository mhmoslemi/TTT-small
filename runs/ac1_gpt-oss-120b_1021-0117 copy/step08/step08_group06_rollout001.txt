def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

    """Maximum of the (circular) convolution a*a using FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * len(a) - 1]
    return float(conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------ #
    # initialise
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT size (next power of two)

    # best solution seen so far (true objective)
    try:
        best_sc = evaluate_sequence(cur.tolist())
        best_vec = cur.copy()
    except Exception:
        best_sc = float("inf")
        best_vec = None

    # surrogate best (2*n*max_conv)
    best_sur_sc = 2.0 * n * _max_conv(cur, L)

    # optimisation parameters
    beta = 0.90                # momentum decay
    step0 = 0.50               # initial step size factor
    momentum = np.zeros_like(cur)
    it = 0
    no_improve = 0
    temp = None
    decay = 0.995
    min_temp_factor = 1e-4
    exact_every = 1200        # evaluate true objective every this many iters

    # ------------------------------------------------------------------ #
    while time.time() < deadline:
        it += 1

        # convolution of current iterate
        af = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(af * af, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = float(conv.max())
        idx_max = int(conv.argmax())
        cur_sc = 2.0 * n * max_val

        # temperature for occasional uphill moves
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # gradient of the max‑convolution (exact at the active index)
        grad = np.zeros_like(cur)
        j_start = max(0, idx_max - (n - 1))
        j_end = min(n - 1, idx_max)
        if j_start <= j_end:
            idx_range = np.arange(j_start, j_end + 1)
            grad[idx_range] = 2.0 * cur[idx_max - idx_range]

        # momentum & step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        # candidate step and projection onto simplex
        cand = cur - step * momentum
        cand = _project_simplex(cand)

        # surrogate score of candidate
        cand_max = _max_conv(cand, L)
        cand_sc = 2.0 * n * cand_max

        # accept if better, otherwise with Metropolis probability
        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sur_sc:
                best_sur_sc = cand_sc
                true_sc = evaluate_sequence(cand.tolist())
                if true_sc < best_sc:
                    best_sc = true_sc
                    best_vec = cand.copy()
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # random redistribution when stuck
        if no_improve > 5000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = 2.0 * n * _max_conv(tmp, L)
            if tmp_sc < cur_sc:
                cur = tmp
                cur_sc = tmp_sc
                no_improve = 0
                true_sc = evaluate_sequence(tmp.tolist())
                if true_sc < best_sc:
                    best_sc = true_sc
                    best_vec = tmp.copy()

        # ------------------------------------------------------------------
        # occasional symmetrisation
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = 2.0 * n * _max_conv(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                true_sc = evaluate_sequence(sym.tolist())
                if true_sc < best_sc:
                    best_sc = true_sc
                    best_vec = sym.copy()

        # ------------------------------------------------------------------
        # periodic exact evaluation of the current iterate
        if it % exact_every == 0:
            true_sc = evaluate_sequence(cur.tolist())
            if true_sc < best_sc:
                best_sc = true_sc
                best_vec = cur.copy()

        # ------------------------------------------------------------------
        # early exit if we already beat a strong target
        if best_sc < 1.45:
            break

    # ---------------------------------------------------------------------- #
    # final sanity check / fallback
    if best_vec is None or not np.all(np.isfinite(best_vec)):
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not math.isfinite(final_sc):
            best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    except Exception:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
