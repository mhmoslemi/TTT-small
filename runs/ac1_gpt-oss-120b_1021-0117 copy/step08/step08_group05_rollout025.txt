import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a: np.ndarray, n: int) -> float:
    """Fast score for a vector a with sum(a)=1: 2*n*sum(a**2)."""
    # sum(a) is assumed to be 1 (or very close)
    return float(2.0 * n * np.dot(a, a))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function height vector that (heuristically)
    minimizes the AC1 upper bound evaluated by `evaluate_sequence`.
    The routine respects the time budget and keeps the best candidate seen.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # reserve a small margin for the final conversion
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # -------------------- initialization --------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
    else:
        n0 = 2500
        cur = rng.random(n0, dtype=np.float64)
        cur = np.maximum(cur, 0.0)

    # Normalise to the simplex (sum == 1)
    cur = _project_simplex(cur)

    n = cur.size
    best_vec = cur.copy()
    best_sc = _score_simplex(best_vec, n)

    # Adam‑style accumulator (optional but useful)
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-12
    # base learning rate – tuned for this problem
    base_lr = 0.05

    it = 0
    no_improve = 0
    target = 1.50274349  # theoretical target from the challenge description

    # -------------------- optimisation loop --------------------
    while time.time() < deadline:
        it += 1

        # fast objective and its gradient (exact for the simplified form)
        s = cur.sum()                     # should be 1 (but recompute for safety)
        q = np.dot(cur, cur)              # sum of squares
        # objective: f = 2 * n * q / s**2
        # gradient of f w.r.t. cur:
        grad = (4.0 * n / (s * s)) * (cur - q / s)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (base_lr / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + epsilon)

        cand = cur - step
        cand = _project_simplex(cand)

        # evaluate candidate (fast exact score)
        cand_sc = _score_simplex(cand, n)

        # accept if improvement
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()
            cur, no_improve = cand, 0
        else:
            no_improve += 1
            cur = cand  # still move (allows occasional uphill steps)

        # occasional random perturbation when stuck
        if no_improve > 2000:
            # pick two indices and shift a small fraction of mass
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_simplex(tmp, n)
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    cur = tmp
                else:
                    cur = tmp
            no_improve = 0

        # symmetrisation every 1500 iterations (helps for some structures)
        if it % 1500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_simplex(sym, n)
            if sym_sc < best_sc:
                best_sc, best_vec = sym_sc, sym.copy()
                cur = sym

        # early exit if we beat the target (unlikely but cheap to check)
        if best_sc < target:
            break

    # Return the best candidate discovered as a plain Python list of floats
    return [float(x) for x in best_vec]
