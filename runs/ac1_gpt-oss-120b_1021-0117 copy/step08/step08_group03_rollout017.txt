def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a = rng.random(n, dtype=np.float64)
    # project to simplex
    a = np.maximum(a, 0)
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0)
        s = a.sum()
    a = a / s

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    
    # helper scoring function
    def _score_vec(v):
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())
    
    best = a.copy()
    best_score = _score_vec(best)

    # parameters
    eta0 = 1.0
    alpha = 5.0
    # schedule for alpha increase every certain steps
    alpha_inc_iter = 200
    alpha_factor = 1.05
    max_alpha = 200.0
    # momentum optionally
    # no momentum for now

    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv of current a
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]

        max_val = conv.max()
        # compute softmax weights
        # shift by max to avoid overflow
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # compute gradient of logsumexp
        # gradient = 2 * alpha * conv(w, a_rev)[n-1:n-1+n]
        a_rev = a[::-1]
        # convolution via FFT
        wf = np.fft.rfft(w, n=L)
        arf = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(wf * arf, n=L)
        grad = 2.0 * alpha * grad_full[n-1:n-1+n]

        # step size schedule
        eta = eta0 / math.sqrt(it)

        # exponentiated gradient update
        exp_term = np.exp(-eta * grad)
        a = a * exp_term
        sum_a = a.sum()
        if sum_a <= 0:
            # fallback to uniform
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a

        # evaluate
        cur_score = _score_vec(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            # optional early exit
            if best_score < 1.50274349:
                break

        # update alpha occasionally
        if it % alpha_inc_iter == 0 and alpha < max_alpha:
            alpha = min(alpha * alpha_factor, max_alpha)

        # occasional random perturbation if stuck for many iterations
        # we can track no_improve count
        # implement simple check every 1000 iter

    return [float(x) for x in best]

import time, math, numpy as np

def _score_vec(v, L):
    """Exact AC1 score for a vector (already on the simplex)."""
    af = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * len(v) - 1]
    return float(2.0 * len(v) * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random vector of length ~2000
        a = rng.random(2000, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full_like(a, 1.0)
        s = a.sum()
    a = a / s                       # simplex normalisation

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two FFT length

    best_vec = a.copy()
    best_sc = _score_vec(best_vec, L)

    # ---- parameters ----------------------------------------------------
    eta0 = 1.0          # initial step size factor
    alpha = 5.0         # softmax sharpness
    alpha_inc = 200     # increase alpha every this many iterations
    alpha_factor = 1.05
    alpha_max = 200.0
    no_improve = 0
    it = 0

    # ---- main optimisation loop ----------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution of a with itself
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weights over convolution entries
        w = np.exp(alpha * (conv - max_val))      # shift avoids overflow
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of log‑sum‑exp (smooth approximation of max)
        a_rev = a[::-1]
        wf = np.fft.rfft(w, n=L)
        arf = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(wf * arf, n=L)
        grad = 2.0 * alpha * grad_full[n - 1 : n - 1 + n]

        # step size schedule (diminishing)
        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient update (mirror descent on simplex)
        exp_term = np.exp(-eta * grad)
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # evaluate current vector
        cur_sc = _score_vec(a, L)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # gradual sharpening of the soft‑max
        if it % alpha_inc == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_factor, alpha_max)

        # early exit if target reached
        if best_sc < 1.50274349:
            break

        # occasional random rescue when stuck
        if no_improve >= 2000:

            if i == j:

            a[i] -= delta
            a[j] += delta
            a = np.maximum(a, 0.0)

            best_sc, best_vec = cand_sc, cand.copy()
            if best_sc < 1.50274349:

    return [float(x) for x in best_vec]
