import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

                best = sym.copy()
                best_score = sym_score
                no_improve = 0
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score

        # ---- restart with small noise if very long stagnation ----------
        if no_improve > 7000:
            cur = best + rng.normal(scale=0.01, size=n)
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            cur_score = _score_norm1(cur)
            no_improve = 0

        # ---- early exit if target met ------------------------------------
        if best_score <= target:
            break

        # ---- time guard --------------------------------------------------
        if time.time() >= deadline:
            break

    # Return the best sequence found as plain Python floats
    return [float(x) for x in best]
