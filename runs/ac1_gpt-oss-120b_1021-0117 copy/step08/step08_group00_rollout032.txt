def grad_and_max(a, temp):
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]

def simplex_projection(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # ---- soft‑max weights (smooth max) ---------------------------
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:                     # fallback to hard max
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ---- gradient of smooth max ----------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]          # ∂(smooth max)/∂a
        grad_obj = 2.0 * n * grad            # ∂objective/∂a (sum fixed =1)

        # ---- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj * grad_obj)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # candidate and projection
        cand = cur - step
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        # evaluate candidate (objective = 2*n*max(conv) because sum=1)
        a_fft_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(a_fft_c * a_fft_c, n=L)[: 2 * n - 1]
        cand_obj = 2.0 * n * conv_c.max()

        # keep best
        if cand_obj < best_obj:
            best_obj = cand_obj
            best_vec = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur = cand
        # bias‑correction update for next iteration
        beta1_pow *= beta1
        beta2_pow *= beta2

        # ---- occasional symmetrisation (helps reduce peaks) ----------
        if it % 700 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            a_fft_s = np.fft.rfft(sym, n=L)
            conv_s = np.fft.irfft(a_fft_s * a_fft_s, n=L)[: 2 * n - 1]
            sym_obj = 2.0 * n * conv_s.max()
            if sym_obj < best_obj:
                best_obj, best_vec = sym_obj, sym.copy()
                cur = sym
                no_improve = 0

        # ---- occasional sparsification --------------------------------
        if it % 1500 == 0:
            thresh = np.quantile(cur, 0.2)          # prune bottom 20 %
            tmp = cur.copy()
            tmp[tmp < thresh] = 0.0
            tmp = _project_simplex(tmp)
            a_fft_t = np.fft.rfft(tmp, n=L)
            conv_t = np.fft.irfft(a_fft_t * a_fft_t, n=L)[: 2 * n - 1]
            tmp_obj = 2.0 * n * conv_t.max()
            if tmp_obj < best_obj:
                best_obj, best_vec = tmp_obj, tmp.copy()
                cur = tmp
                no_improve = 0

        # ---- random jitter when stuck ---------------------------------
        if no_improve > 8000:
            jitter = rng.normal(scale=0.01, size=n)
            cur = cur + jitter
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            no_improve = 0
            # reset Adam statistics to avoid stale moments
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = beta1, beta2

        # ---- early exit ------------------------------------------------
        if best_obj < target:
            break

    # return the best-found sequence as plain Python floats
    return [float(x) for x in best_vec]
