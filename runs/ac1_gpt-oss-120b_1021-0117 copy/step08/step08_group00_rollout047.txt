import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

        while time.time() < deadline:
            it += 1

            # ----- convolution & max -----
            cur_fft = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
            max_conv = conv.max()
            cur_score = 2.0 * n * max_conv

            # ----- keep global best -----
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                if best_score < target:
                    outer_break = True
                    break

            # ----- temperature for soft‑max -----
            if temp is None:
                temp = max_conv * 0.07
            else:
                temp = max(temp * 0.9995, max_conv * 1e-5)

            # ----- soft‑max weights -----
            w = np.exp((conv - max_conv) / temp)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = (conv == max_conv).astype(np.float64)
                w_sum = w.sum()
            w /= w_sum

            # ----- gradient (via FFT) -----
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr / (np.sqrt(v_hat) + eps)
            cand = cur - step * m_hat

            # ----- projection onto simplex -----
            cand = _project_simplex(cand)

            # ----- candidate score -----
            cand_fft = np.fft.rfft(cand, n=L)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
            cand_score = float(2.0 * n * cand_conv.max())

            # ----- acceptance test -----
            if cand_score < cur_score:
                cur, cur_score = cand, cand_score
                no_improve = 0
            else:
                no_improve += 1
                if rng.random() < 0.001:   # occasional uphill move
                    cur, cur_score = cand, cand_score

            # ----- periodic symmetrisation -----
            if it % 400 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_fft = np.fft.rfft(sym, n=L)
                sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
                sym_score = float(2.0 * n * sym_conv.max())
                if sym_score < cur_score:
                    cur, cur_score = sym, sym_score
                    no_improve = 0

            # ----- occasional smoothing -----
            if it % 800 == 0:
                kernel = np.ones(5) / 5.0
                smooth = np.convolve(cur, kernel, mode='same')
                smooth = _project_simplex(smooth)
                smooth_fft = np.fft.rfft(smooth, n=L)
                smooth_conv = np.fft.irfft(smooth_fft * smooth_fft, n=L)[: 2 * n - 1]
                smooth_score = float(2.0 * n * smooth_conv.max())
                if smooth_score < cur_score:
                    cur, cur_score = smooth, smooth_score
                    no_improve = 0

            # ----- random redistribution when stuck -----
            if no_improve > 2500:
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_score = float(2.0 * n * tmp_conv.max())
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    no_improve = 0

        # end of inner while
    # end of restarts

    # ------------------------------------------------------------------
    # 3.  Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _project_simplex(np.maximum(a0, 0.0))

    # Convert to plain Python floats
    return [float(x) for x in best_seq]
