def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w
```

def _score_simplex(a, n):
    L = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5) # reserve margin

    # init
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2*n - 1).bit_length())
    idx = (2*n - 2) - np.arange(n)
    # Adam
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    best_vec = cur.copy()
    best_sc = _score_simplex(best_vec, n)
    cur_sc = best_sc

    it = 0
    no_improve = 0
    T = None
    # main loop
    while time.time() < deadline:
        it += 1
        # compute FFT of current
        a_fft = np.fft.rfft(cur, n=L)

        # convolution for score and gradient
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_c = conv.max()
        # subgradient weights: uniform over max entries
        # find max indices
        # Use tolerance:
        tol = max_c * 1e-12 + 1e-15
        mask = np.abs(conv - max_c) <= tol
        w = np.zeros_like(conv)
        cnt = mask.sum()
        if cnt == 0:
            w[np.argmax(conv)] = 1.0
            cnt = 1
        else:
            w[mask] = 1.0 / cnt

        # compute gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # temperature for Metropolis acceptance (optional)
        if T is None:
            T = max_c * 0.05
        else:
            # decay
            T = max(T * 0.995, max_c * 1e-4)

        # Adam step size
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = (0.2 / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)  # base=0.2

        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score_simplex(cand, n)

        # Acceptance
        if cand_sc < cur_sc:

            no_improve = 0
            if cand_sc < best_sc:

import time, math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n):
    """Score for a vector on the simplex: 2*n*max(conv(a,a))."""
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    it = 0
