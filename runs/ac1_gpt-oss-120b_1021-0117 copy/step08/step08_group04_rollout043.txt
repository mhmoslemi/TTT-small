def _project_simplex(v):
    """Project vector v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _max_convolution(a, L):
    """Maximum of linear convolution a*a using FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv.max()

def _score(a, L):
    """Score = 2 * n * max(conv)  (sum(a)=1)."""
    n = a.size
    return 2.0 * n * _max_convolution(a, L)
