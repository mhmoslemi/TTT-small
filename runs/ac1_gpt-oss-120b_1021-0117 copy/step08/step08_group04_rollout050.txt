import time, math, numpy as np

def _project_simplex(v):
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
        cur = np.maximum(cur, 0.0)
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(cur)          # sum = 1, non‑negative
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())  # FFT length (next power of two)

    # ----- initial score --------------------------------------------------
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    cur_score = 2.0 * n * max_val          # sum = 1 => exact evaluator value
    best = cur.copy()
    best_score = cur_score

    # ----- parameters -----------------------------------------------------

    eta_min = 0.08
    decay = 0.9998
    no_improve = 0
    target = 1.50274349

    # ----- main optimisation loop -----------------------------------------
    while time.time() < deadline:
