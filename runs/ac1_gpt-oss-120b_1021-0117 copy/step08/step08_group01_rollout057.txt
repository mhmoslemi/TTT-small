import time, math, numpy as np

def _project_simplex(v):
    """
    Euclidean projection of v onto the probability simplex (sum = 1, v >= 0).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

    Fast evaluation of 2*n*max(conv(a,a)) for a vector a that already satisfies
    sum(a) == 1. Returns both the scalar score and the full convolution.

    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv_full.max()), conv_full

            # gradient of the smoothed max (matrix‑free via FFT)
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # candidate and projection onto simplex
            cand = _project_simplex(cur - step)
            cand_sc, _ = _score(cand, L)

            # acceptance rule (always accept improvement)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
            else:
                # occasional uphill move to escape flat regions
                no_improve += 1
                if rng.random() < 0.001:
                    cur, cur_sc = cand, cand_sc

            # ----------------------------------------------------------------
            # random redistribution when stuck for a while
            if no_improve > 3000:
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc, _ = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()

            # ----------------------------------------------------------------
            # periodic symmetrisation (helps when the optimum is symmetric)
            if it % sym_interval == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc, _ = _score(sym, L)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best_vec = sym_sc, sym.copy()

            # ----------------------------------------------------------------
            if time.time() >= deadline:
                break
        # --------------------------------------------------------------------
        restart += 1
        if best_sc < target:
            break

    # ---------------------------------------------------------------------- #
    # Final fine‑grained hill‑climbing (many cheap local moves)
    cur = best_vec.copy()
    cur_sc = best_sc
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        eps = cur[i] * rng.random() * 0.02
        if eps <= 0.0:
            continue
        tmp = cur.copy()
        tmp[i] -= eps
        tmp[j] += eps
        tmp = _project_simplex(tmp)
        tmp_sc, _ = _score(tmp, L)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
            # occasional extra symmetrisation
            if rng.random() < 0.07:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc, _ = _score(sym, L)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best_vec = sym_sc, sym.copy()
        else:
            # tiny chance to accept a non‑improving move
            if rng.random() < 0.001:
                cur, cur_sc = tmp, tmp_sc

    # Return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
