def _project_simplex(v):
   v = np.asarray(v, dtype=np.float64)
   n = v.size
   if n == 0:
      return v
   # Sort in descending order
   u = np.sort(v)[::-1]
   css = np.cumsum(u) - 1
   rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
   if rho.size == 0:
       theta = 0.0
   else:
       rho = rho[-1]
       theta = css[rho] / (rho + 1)
   w = v - theta
   w[w < 0] = 0.0
   return w

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w


def _conv_max(seq, L):
    """Maximum value of the full linear convolution seq*seq."""
    af = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * seq.shape[0] - 1].max()
