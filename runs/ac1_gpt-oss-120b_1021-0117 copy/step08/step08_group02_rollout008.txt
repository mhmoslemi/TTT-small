import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast computation of 2*n*max(conv) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            continue
        # move a small fraction of mass from i to j
        delta = best_vec[i] * rng.random() * 0.02
        if delta == 0.0:
            continue
        tmp = best_vec.copy()
        tmp[i] -= delta
        tmp[j] += delta
        # ensure non‑negativity (clipping can break sum, so re‑project)
        tmp = np.maximum(tmp, 0.0)
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < best_sc:
            best_vec, best_sc = tmp, tmp_sc

    # ------------------------------------------------------------------
    # return best found candidate as plain Python list
    return [float(x) for x in best_vec]
