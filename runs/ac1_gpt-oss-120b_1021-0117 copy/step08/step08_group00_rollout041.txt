    start = time.time()

    # project onto simplex sum=1
    cur = _project_simplex(cur)

    # compute initial score, best

    L = 1 << ((2 * n - 1).bit_length())
    def compute_conv(v):
        a_fft = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        return conv

    conv = compute_conv(cur)
    max_val = conv.max()
    cur_score = 2.0 * n * max_val
    best_vec = cur.copy()
    best_score = cur_score

    # Adam variables
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t_iter = 0
    lr0 = 0.8
    no_improve = 0

    while time.time() < deadline:

        # compute gradient at current cur
        # conv already computed and max_idx known
        max_idx = int(conv.argmax())
        # gradient of max (subgradient)
        grad = np.zeros_like(cur)
        low = max(0, max_idx - (n - 1))

        m_hat = m / (1.0 - beta1 ** t_iter)

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a simple exponentially‑decaying start if no parent is supplied
        n = 2500
        cur = np.exp(-0.001 * np.arange(n))
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # project onto simplex, sum = 1
    n = cur.shape[0]

    # ----- FFT helpers ----------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two length for FFT

    def _conv_fft(a):
        a_fft = np.fft.rfft(a, n=L)
        return np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]

    # ----- initial score --------------------------------------------------
    conv = _conv_fft(cur)
    max_val = conv.max()
    cur_score = 2.0 * n * max_val          # sum(cur)=1 after projection
    best_vec = cur.copy()
    best_score = cur_score

    # ----- Adam optimiser -------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    it = 0
    lr0 = 0.8
    no_improve = 0

    while time.time() < deadline:
        it += 1

        # --- gradient of the current (hard) max --------------------------------
        max_idx = int(conv.argmax())
        low = max(0, max_idx - (n - 1))
        high = min(n - 1, max_idx)
        idx = np.arange(low, high + 1)
        grad = np.zeros_like(cur)
        grad[low:high + 1] = 2.0 * cur[max_idx - idx]

        # --- Adam update -------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 / math.sqrt(it)
        update = step * (m_hat / (np.sqrt(v_hat) + eps))

        cand = cur - update
        cand = _project_simplex(cand)

        # --- evaluate candidate -------------------------------------------------
        conv_cand = _conv_fft(cand)
        cand_score = 2.0 * n * conv_cand.max()

        if cand_score < cur_score:
            cur, conv, max_val, cur_score = cand, conv_cand, conv_cand.max(), cand_score
            no_improve = 0
            if cand_score < best_score:
                best_vec, best_score = cand.copy(), cand_score
        else:
            no_improve += 1
            # occasional uphill move (simulated‑annealing flavour)
            prob = math.exp(-(cand_score - cur_score) / (step + 1e-12))
            if rng.random() < prob:
                cur, conv, max_val, cur_score = cand, conv_cand, conv_cand.max(), cand_score

        # --- random mass transfer when stuck -----------------------------------
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                conv_tmp = _conv_fft(tmp)
                tmp_score = 2.0 * n * conv_tmp.max()
                if tmp_score < cur_score:
                    cur, conv, max_val, cur_score = tmp, conv_tmp, conv_tmp.max(), tmp_score
                    no_improve = 0
                    if tmp_score < best_score:
                        best_vec, best_score = tmp.copy(), tmp_score

        # --- early exit if good enough -----------------------------------------
        if best_score < 1.5025:
            break

    # Return the best sequence found (as plain Python floats)
    return [float(x) for x in best_vec]
