import time, math, numpy as np

# ---------- helper utilities -------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L):
    """Fast evaluation of 2*n*max(conv) for a probability vector (sum≈1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ---------- main search ------------------------------------------------------

    rng = np.random.default_rng(seed)

    # ----- initial vector ---------------------------------------------------

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())        # power‑of‑two for fast FFT

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # ----- Adam optimizer state ---------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    lr, eps = 0.15, 1e-8                     # a slightly smaller step size

    # ----- temperature for simulated‑annealing -------------------------------
    temp = None
    decay = 0.996
    min_temp_factor = 1e-6

    it = 0

    while time.time() < deadline:
        it += 1

        # ----- full convolution via FFT (exact max) -------------------------
        a_fft = np.fft.rfft(cur, n=L)

        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        w = np.exp((conv - max_val) / temp)

        w /= w_sum
