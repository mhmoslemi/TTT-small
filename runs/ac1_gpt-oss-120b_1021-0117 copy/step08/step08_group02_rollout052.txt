def _lp_refine(f, target_sum, M, max_iter=2000):
    n = f.shape[0]
    T = _build_toeplitz_matrix(f)  # csr
    b = cp.Variable(n)
    constraints = [b >= 0, T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, abstol=1e-6, reltol=1e-6, max_iters=2000)
    except cp.error.SolverError:
        return None
    if b.value is None:
        return None
    b_opt = np.array(b.value, dtype=np.float64)
    # Scale to target sum:
    if b_opt.sum() == 0:
        return None
    scale = target_sum / b_opt.sum()
    return b_opt * scale

import time, math, numpy as np, cvxpy as cp, scipy.sparse as sp

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_fast(a):
    """Fast evaluation of the AC1 upper bound for a non‑negative vector."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    s = a.sum()
    if s == 0.0:
        return np.inf
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

def _lp_active_refine(f, target_sum, M, conv, max_active=20, max_iters=2000):
    """Solve a small LP using only the (approximately) active convolution constraints."""
    n = f.shape[0]
    # Identify practically active constraints
    eps = max(M * 1e-8, 1e-12)
    act = np.where(conv >= M - eps)[0]
    if act.size == 0:
        act = np.array([np.argmax(conv)], dtype=np.int64)
    if act.size > max_active:
        rng = np.random.default_rng()
        act = rng.choice(act, size=max_active, replace=False)
    # Build dense constraint matrix (|act| × n)
    C = np.zeros((act.size, n), dtype=np.float64)
    for r, k in enumerate(act):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        C[r, j_vals] = f[i_vals]
    # LP: maximise sum(b) s.t. b≥0, C b ≤ M
    b = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [b >= 0, C @ b <= M])
    try:

            w = mask / cnt
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        cand = cur - lr * grad
        cand = _project_simplex(cand) * target_sum
        cand_sc = _score_fast(cand)

        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
        else:
            # occasional probabilistic acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (M * 0.01 + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        lr *= decay
        it += 1

        # ------------------------------------------------------------------
        # occasional symmetrisation (helps many instances)
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym) * target_sum
            sym_sc = _score_fast(sym)
            if sym_sc < best_sc:
                best_sc, best_vec = sym_sc, sym.copy()
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc

        if it > 3000:      # safety stop
            break

    return [float(x) for x in best_vec]
