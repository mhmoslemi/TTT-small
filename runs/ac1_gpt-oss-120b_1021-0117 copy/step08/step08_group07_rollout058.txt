import time, math, numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of a vector onto the probability simplex. """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast evaluation of the AC1 bound for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _softmax_weights(conv, temp, max_val):
    """Compute soft‑max weights for the convolution vector."""
    w = np.exp((conv - max_val) / temp)
    s = w.sum()
    if s == 0.0 or np.isnan(s):
        w = (conv == max_val).astype(np.float64)
        s = w.sum()
    w /= s
    return w

def _gradient(a, w_rev_fft, a_fft, n, L):
    """Gradient of the (soft) max of a*a wrt a using FFTs."""
    grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]
