import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # ensure nonnegative
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = v - theta
    w[w < 0] = 0.0
    # due to numerical issues, may not sum exactly to 1; renorm
    s = w.sum()
    if s > 0:
        w = w / s
    else:
        # fallback uniform
        w = np.full_like(v, 1.0/n)
    return w

def _score_norm1(a, L=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed=42,budget_s=1100,**kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # generate simple initial distribution: exponential decay
        n = 2500
        cur = np.exp(-np.linspace(0,5,n))
        cur = _project_simplex(cur)

    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    # evaluate
    cur_score = _score_norm1(cur, L)
    best = cur.copy()
    best_score = cur_score

    # Adam variables
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.2
    max_step = 0.5

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1
        # compute FFT and convolution
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()
        # compute indicator for max entries
        # tolerance relative
        tol = 1e-12
        w = np.zeros_like(conv_full)
        w[conv_full >= max_val - tol] = 1.0
        # compute gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = (base_lr / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        # candidate generation with simple line search
        best_local = None
        best_local_score = np.inf
        for scale in (0.5, 1.0, 1.5):
            cand = _project_simplex(cur - step * scale)
            cand_score = _score_norm1(cand, L)
            if cand_score < best_local_score:
                best_local = cand
                best_local_score = cand_score
        cand = best_local
        cand_score = best_local_score

        # acceptance (simulated annealing)
        # define temperature
        temp = max_val * 0.02  # small relative
        accept = False
        if cand_score < cur_score:

            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:

                best_score = cand_score
        else:

            if j >= i:

    # ensure values within allowed range (0,1000)
    best = np.clip(best, 0.0, 1000.0)
    # convert to python floats
    return [float(x) for x in best]

import time, math, numpy as np

def _project_simplex(v):
    """Project onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # make non‑negative
    v = np.maximum(v, 0.0)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w.fill(1.0 / n)
    return w

def _score_norm1(a, L):
    """Return 2*n*max(convolution(a,a)) for a with sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # simple exponential decay if no parent is present
        n = 2500
        cur = np.exp(-np.linspace(0, 5, n, dtype=np.float64))
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    cur_score = _score_norm1(cur, L)
    best = cur.copy()
    best_score = cur_score

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.20
    max_step = 0.5

    it = 0
    no_improve = 0
    target = 1.50274349

    # -------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ---- convolution and gradient ---------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # hard max indicator (sub‑gradient of the max)
        tol = 1e-12
        w = np.zeros_like(conv_full)
        w[conv_full >= max_val - tol] = 1.0

        # gradient of weighted sum (here weighted by w)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---- Adam update ----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (base_lr / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        # ---- candidate generation (simple line‑search) -----------------
        cand = None

        for scale in (0.5, 1.0, 1.5):
            tmp = _project_simplex(cur - step * scale)
            sc = _score_norm1(tmp, L)
            if sc < cand_score:

                cand_score = sc

        temp = max_val * 0.02  # small temperature proportional to peak

            v.fill(0.0)
            no_improve = 0

        # ---- early exit if target reached --------------------------------
        if best_score <= target:
            break

    # -------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)   # respect evaluator limits
    return [float(x) for x in best]
