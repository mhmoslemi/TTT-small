    # compute weight vector w
    # conv: length 2n-1 array
    # alpha >0
    w = np.exp(alpha * (conv - max_val))

    af = np.fft.rfft(a, n=L)

import time, math, numpy as np

# --------------------------------------------------------------
# Helper utilities (all top‑level as required)
# --------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    """Fast evaluation of 2·n·max(conv(a,a)) for a vector already on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _grad_weighted(a, conv, max_val, L, alpha):
    """
    Gradient of a smoothed max of the convolution.
    w_i = exp(alpha*(conv_i - max_val)) / sum_j exp(alpha*(conv_j - max_val))
    grad = 2 * (a * reversed(w)) (convolution via FFT).
    """
    # build weight vector
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.zeros_like(w)
    else:
        w = w / w_sum
    # reversed weights for convolution
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

    # ---- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # best‑so‑far tracking
    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # Adam‑style moments
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.99
    eps = 1e-8
    step0 = 0.5               # base step magnitude
    alpha = 5.0               # start smoothing weight
    alpha_inc = 0.005
    max_alpha = 30.0

    no_improve = 0
    target = 1.50274349
    it = 0

    # --------------------------------------------------------------
    # optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution of current vector
        af = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(af * af, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val               # because cur is on the simplex

        # weighted gradient of smoothed max
        grad = _grad_weighted(cur, conv, max_val, L, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / math.sqrt(it)
        delta = step * m_hat / (np.sqrt(v_hat) + eps)

        # propose a new point and enforce feasibility
        cand = _project_simplex(cur - delta)
        cand_sc = _score(cand, L)

        # accept if it improves the current point
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # update global best
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()

        # gradually sharpen the soft‑max mask
        if alpha < max_alpha:
            alpha = min(max_alpha, alpha + alpha_inc)

        # occasional random rescue when progress stalls
        if no_improve > 800:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta_val = cur[i] * rng.random() * 0.2
                tmp = cur.copy()
                tmp[i] -= delta_val
                tmp[j] += delta_val
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                cur, cur_sc = tmp, tmp_sc
            no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

    # --------------------------------------------------------------
    # return the best candidate found (as plain Python floats)
    # --------------------------------------------------------------
    return [float(x) for x in best_vec]
