import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """2*n*max(conv) for a vector a with sum≈1 (no denominator)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # safety margin

    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())  # next power of two

    # best-so-far

    cur_sc = _fast_score(cur, n, L)

    # Adam parameters

    v = np.zeros_like(cur)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    alpha = 0.2  # base learning rate

    # temperature schedule

    decay = 0.998
    min_temp_factor = 1e-4

    while time.time() < deadline:

        max_val = conv.max()
