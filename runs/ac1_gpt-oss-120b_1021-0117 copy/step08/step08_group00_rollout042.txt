import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a list of non‑negative step heights with low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ----- initialise ----------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64) + 1e-6
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # sum ≈ 1, non‑negative
    n = cur.shape[0]

    # length for FFTs (must accommodate convolution of length 2n‑1 and
    # correlation of length 3n‑2)
    L = 1 << ((3 * n - 2).bit_length())

    # ----- helpers -------------------------------------------------------
    def _score_and_conv(a, a_fft=None):
        """Return (score, full convolution, a_fft)."""
        if a_fft is None:
            a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        return float(2.0 * n * max_val), conv_full, a_fft

    # ----- optimisation parameters ---------------------------------------
    beta = 0.9                     # momentum decay
    step0 = 0.5                    # initial step magnitude
    target = 1.50274349            # stop if achieved
    momentum = np.zeros_like(cur)
    best_vec = cur.copy()
    best_score, _, _ = _score_and_conv(best_vec)

    it = 0
    no_improve = 0

    # ----- main optimisation loop ----------------------------------------
    while time.time() < deadline:
        it += 1

        # current convolution, max and FFT
        cur_score, conv_full, cur_fft = _score_and_conv(cur)
        max_val = conv_full.max()

        tol = max_val * 1e-12 + 1e-15

        cand = _project_simplex(cur - step * momentum)
