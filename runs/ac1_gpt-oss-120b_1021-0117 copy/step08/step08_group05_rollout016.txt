import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_vec(vec, n, L):
    a_fft = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initialization -----
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # sum = 1

    n = cur.size
    # FFT length: power of two >= 2*n-1
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)    # indices to extract gradient slice

    # Adam state
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # hyper‑parameters
    base_lr = 0.2          # base learning rate (before sqrt decay)
    T = None               # temperature for smoothing
    it = 0
    no_improve = 0
    target = 1.50274349

    cur_sc = _score_vec(cur, n, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ----- main loop -----
    while time.time() < deadline:
        it += 1

        # convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_conv = conv_full.max()

        # temperature schedule (soft‑max smoothing)
        if T is None:
            T = max_conv * 0.2
        else:
            T = max(T * 0.995, max_conv * 1e-6)

        # soft‑max weights
        w = np.exp((conv_full - max_conv) / T)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv_full == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of smoothed max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate decay
        lr = base_lr / math.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)

        cand = cur - step
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_sc = _score_vec(cand, n, L)

        # acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            # simulated‑annealing uphill move
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # random redistribution when stuck
        if no_improve > 8000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp, n, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

        # symmetrisation periodically
        if it % 5000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # early exit
        if best_sc < target:
            break

    # final fine‑tuning phase (very low temperature)
    # use remaining time if any
    while time.time() < deadline - 0.1:
        # same as above but with hard max weight
        a_fft = np.fft.rfft(best_vec, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_conv = conv_full.max()
        # hard max weights
        w = (conv_full == max_conv).astype(np.float64)
        w /= w.sum()
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # simple gradient step (no Adam) with decreasing lr
        it += 1
        lr = 0.02 / math.sqrt(it)
        step = lr * grad
        cand = best_vec - step
        cand = _project_simplex(cand)
        cand_sc = _score_vec(cand, n, L)
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()
        else:
            # stop if no improvement for a while
            if it > 2000 and (best_sc - cand_sc) < 1e-8:
                break

    # return best found vector as plain Python list
    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Project vector v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_vec(vec, n, L):
    """Return 2·n·max(conv(vec,vec)) using an FFT of length L."""
    a_fft = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
