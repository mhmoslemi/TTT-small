def propose_candidate(...):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0., budget_s - 0.2)

    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = len(cur)
    L = 1 << ((2*n - 1).bit_length())
    best = cur.copy()
    best_sc = _score(best)
    cur_sc = best_sc

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.1
    t = 0
    no_improve = 0

    while time.time() < deadline:
        t += 1
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2*n-1]
        max_val = conv.max()
        # get mask of max positions
        mask = conv >= max_val - 1e-14
        w = mask.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
            w_sum = 1.0
        w /= w_sum
        # gradient via fft
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[(2*n-2) - np.arange(n)]

        # Adam update
        m = beta1 * m + (1-beta1) * grad

        m_hat = m / (1 - beta1**t)

        cand = cur - step

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast evaluation of 2·n·max(conv(a,a)) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
