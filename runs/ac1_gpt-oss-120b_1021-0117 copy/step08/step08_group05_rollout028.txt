import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n):
    a = np.asarray(a, dtype=np.float64)
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2*n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # keep half-second margin

    # ----- initial vector -----

    else:
        cur = rng.random(2500, dtype=np.float64)
    # Normalize to sum 1 and enforce nonnegativity
    cur = np.maximum(cur, 0.0)
    cur = cur / np.maximum(cur.sum(), 1e-12)
    cur = _project_simplex(cur)   # ensures exact sum=1

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    alpha = 0.12               # base learning rate
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    best_vec = cur.copy()
    best_sc = _score_simplex(best_vec, n)
    cur_sc = best_sc

    it = 0
    no_improve = 0
    target = 1.50274349
    warmup = 4000              # iterations where we use a softened max
    min_T_factor = 1e-4

    while time.time() < deadline:
        it += 1

        # ----- convolution -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # ----- weight for gradient (softmax early, hard max later) -----
        if it <= warmup:
            T = max(max_conv * 0.07, max_conv * min_T_factor)
            w = np.exp((conv - max_conv) / T)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = (conv == max_conv).astype(np.float64)
                w_sum = w.sum()
            w /= w_sum
        else:
            # hard max: uniform weight over all maximal entries
            mask = conv == max_conv
            w = mask.astype(np.float64)
            w_sum = w.sum()
            if w_sum == 0.0:   # numerical safety
                w = np.ones_like(conv) / conv.size
                w_sum = 1.0
            w /= w_sum

        # ----- gradient (sub‑gradient of the smoothed/ hard max) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)

        grad = 2.0 * grad_full[idx]

            break

import time, math, numpy as np

def _project_simplex(v):
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n):
    """Score for a vector with sum = 1: 2·n·max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
