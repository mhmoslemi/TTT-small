    rng = np.random.default_rng(seed)

    deadline = start + max(0.0, budget_s - 0.2)

    else:
        n = 2000
        cur0 = _project_simplex(rng.random(n))

    # Baseline best
    best_vec = cur0.copy()
    best_sc = _score(best_vec)

    # other baseline: maybe Dirichlet random
    # Then main optimization:

    # parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    step_size = 0.5   # base learning rate (scale)
    max_iters = 1000000   # not needed.

    L = 1 << ((2*n - 1).bit_length())

    # number of restarts
    restart = 0
    while time.time() < deadline:
        if restart == 0:
            cur = best_vec.copy()
        else:
            cur = _project_simplex(rng.exponential(scale=1.0, size=n))
        # ensure not all zeros
        cur_sc = _score(cur)

        m = np.zeros_like(cur)
        v = np.zeros_like(cur)
        t = 0
        no_improve = 0

        while time.time() < deadline:
            t += 1
            # compute conv and max
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[:2*n-1]
            max_val = conv.max()

            # compute active set
            eps = max_val * 1e-6
            active = np.where(conv >= max_val - eps)[0]
            if active.size == 0:
                active = np.array([np.argmax(conv)])
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.size

            # compute gradient
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2*n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            update = step_size * m_hat / (np.sqrt(v_hat) + epsilon)

            cand = _project_simplex(cur - update)
            cand_sc = _score(cand)

            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()

                tmp[j] += delta

                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:

                    no_improve = 0
                    if tmp_sc < best_sc:

            # early exit if best meets target
            if best_sc < 1.50274349:

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of vector v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast objective (sum of coefficients is assumed to be 1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
