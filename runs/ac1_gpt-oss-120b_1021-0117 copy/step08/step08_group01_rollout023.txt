import time
import math
import numpy as np
import cvxpy as cp
from scipy.sparse import diags, csr_matrix

def _project_simplex(v):
    """Project onto the probability simplex {x>=0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _score_vec(a, L):
    """Fast score for a vector already normalized to sum≈1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max()), conv  # also return full convolution


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------ #
    # Problem size and initial vector
    if "height_sequence_1" in globals():
        init_full = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_full.shape[0]
    else:
        n = 2000
        init_full = rng.random(n, dtype=np.float64)

    L = 1 << ((2 * n - 1).bit_length())

    cur = _project_simplex(np.maximum(init_full, 0.0))
    cur_score, _ = _score_vec(cur, L)
    best_vec = cur.copy()
    best_score = cur_score

    # uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_score, _ = _score_vec(uniform, L)
    if uni_score < best_score:
        best_score, best_vec = uni_score, uniform.copy()

    # ------------------------------------------------------------------ #
    # LP‑based monotone descent (few iterations)
    max_lp_iter = 4
    lp_iter = 0
    while time.time() < deadline and lp_iter < max_lp_iter:
        f = best_vec
        _, conv_f = _score_vec(f, L)
        M = conv_f.max()

        # Build sparse convolution matrix A where (A @ b)[k] = (f * b)[k]
        A = diags(f, offsets=range(n), shape=(2 * n - 1, n), format='csr')

        # Linear program: maximize sum(b) s.t. A @ b <= M, b >= 0
        b = cp.Variable(n, nonneg=True)
        constraints = [A @ b <= M * np.ones(2 * n - 1)]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)
        except Exception:
            break

        if b.value is None:
            break
        b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
        sum_b = b_val.sum()
        if sum_b <= 1.0 + 1e-12:
            break

        g = b_val / sum_b  # normalized version, sum(g)=1

        # Try to blend f towards g
        improved = False
        for t in (0.5, 0.25, 0.125):
