def _softmax_weights(conv, k):
    # conv: numpy array length 2n-1
    cmax = np.max(conv)
    exp = np.exp(k * (conv - cmax))   # stable
    w = exp / np.sum(exp)
    return w

def _grad_softmax(a, w, L2):
    # a: shape (n,)
    # w: size (2n-1)
    # compute convolution a * w (full) using FFT length L2
    af = np.fft.rfft(a, n=L2)
    wf = np.fft.rfft(w, n=L2)
    conv_full = np.fft.irfft(af * wf, n=L2)
    # take slice
    start = len(a) - 1  # n-1
    grad = 2.0 * conv_full[start:start+len(a)]
    return grad

import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of a vector onto the probability simplex. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_from_conv(conv, n):
    """ Score = 2 * n * max(convolution)  (sum of coefficients is 1). """
    return float(2.0 * n * np.max(conv))
