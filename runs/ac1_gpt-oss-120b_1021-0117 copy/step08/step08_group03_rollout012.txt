import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_vec(v, L_conv):
    """Exact evaluator (sum(v)=1) using FFT."""
    af = np.fft.rfft(v, n=L_conv)
    conv = np.fft.irfft(af * af, n=L_conv)[: 2 * v.size - 1]
    return float(2.0 * v.size * conv.max())
