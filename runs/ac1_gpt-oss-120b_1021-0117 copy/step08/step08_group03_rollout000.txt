import time, math, numpy as np, cvxpy as cp

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    """Fast evaluation for a vector already normalized to sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _solve_lp(cur, bound, t_mix=0.2):
    """
    Solve the LP:
        max   sum(g)
        s.t.  conv(cur, g) <= bound, g >= 0
    Returns a new vector after mixing with current one,
    already normalized to sum=1. Returns None on failure.
    """
    n = cur.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.conv(cp.Constant(cur), g) <= bound]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=3000,
                   abstol=1e-8, reltol=1e-6, feastol=1e-8)
    except Exception:
        return None
    if g.value is None:
        return None
    g0 = np.maximum(g.value, 0.0)
    s = g0.sum()
    if s <= 0:
        return None
    g_norm = g0 / s
    return (1.0 - t_mix) * cur + t_mix * g_norm

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.1)   # safety margin

    # ----- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # ----- Adam parameters -----------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha0 = 0.08          # base learning rate
    target = 1.50274349

    no_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # full convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # smoothed subgradient (log‑sum‑exp approximation)
        diff = conv - max_val          # ≤ 0, zero at maxima
        lam = min(100.0, 0.5 + 0.02 * it)   # increase smoothing gradually
        weights = np.exp(lam * diff)   # max weight = 1
        w_rev = weights[::-1]
        wf = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * wf, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        beta1_pow = beta1 ** it
        beta2_pow = beta2 ** it
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        direction = m_hat / (np.sqrt(v_hat) + eps)

        step = alpha0 / math.sqrt(it)
        cand = _project_simplex(cur - step * direction)
        cand_sc = _score(cand, L)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # ----- rescue when stuck -----------------------------------------
        if no_improve > 800:
            # small random transfer between two coordinates
            i = rng.integers(n)
            j = rng.integers(n)
            delta = cur[i] * 0.12 * rng.random()
            tmp = cur.copy()
            tmp[i] = max(tmp[i] - delta, 0.0)
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0
            else:
                # try a LP refinement as a last resort
                lp_vec = _solve_lp(cur, max_val, t_mix=0.18)
                if lp_vec is not None:
                    lp_sc = _score(lp_vec, L)
                    if lp_sc < cur_sc:
                        cur, cur_sc = lp_vec, lp_sc
                        if lp_sc < best_sc:
                            best_vec, best_sc = lp_vec.copy(), lp_sc
                        no_improve = 0

        # ----- occasional symmetrisation ---------------------------------
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # ----- rare LP refinement -----------------------------------------
        if it % 3000 == 0:
            lp_vec = _solve_lp(cur, max_val, t_mix=0.2)
            if lp_vec is not None:
                lp_sc = _score(lp_vec, L)
                if lp_sc < cur_sc:
                    cur, cur_sc = lp_vec, lp_sc
                    if lp_sc < best_sc:
                        best_vec, best_sc = lp_vec.copy(), lp_sc
                    no_improve = 0

        # ----- early exit -------------------------------------------------
        if best_sc < target:
            break

    return [float(x) for x in best_vec]
