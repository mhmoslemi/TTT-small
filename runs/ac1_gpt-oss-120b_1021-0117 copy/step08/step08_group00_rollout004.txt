import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

            tmp_score, tmp_M, tmp_conv = _fast_score(tmp)
            if tmp_score < cur_score:
                cur, cur_score, cur_M, cur_conv = tmp, tmp_score, tmp_M, tmp_conv
                no_improve = 0
                if tmp_score < best_sc:
                    best_sc, best_vec = tmp_score, tmp.copy()

        # ---- occasional symmetrisation -----------------------------------
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score, sym_M, sym_conv = _fast_score(sym)
            if sym_score < cur_score:
                cur, cur_score, cur_M, cur_conv = sym, sym_score, sym_M, sym_conv
                no_improve = 0
                if sym_score < best_sc:
                    best_sc, best_vec = sym_score, sym.copy()

        # early exit
        if best_sc < target:
            break

    # Return best-found sequence as plain Python floats
    return [float(x) for x in best_vec]
