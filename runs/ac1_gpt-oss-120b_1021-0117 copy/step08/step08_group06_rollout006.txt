def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 0.2)
    
    # initialization
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # random initialization: e.g., exponential
        cur = rng.random(2000, dtype=np.float64)
    # ensure nonnegative and project to simplex
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two

    # compute initial surrogate (true) score
    def compute_score(vec):
        # sum is expected 1
        a_fft = np.fft.rfft(vec, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        max_val = conv_full[:2*n-1].max()
        return float(2.0 * n * max_val)

    best_vec = cur.copy()
    best_sc = compute_score(best_vec)
    
    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5   # initial learning rate (to be tuned)
    # temperature parameters
    temp_factor = 0.3
    temp_decay = 0.9995

    temp = None
    it = 0
    no_improve = 0

    # main loop
    while time.time() < deadline:
        it += 1

        # compute convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # update temperature
        if temp is None:
            temp = max_val * temp_factor
        else:
            temp = max(temp * temp_decay, max_val * 1e-5)

        # compute softmax weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum

        # gradient: 2 * conv of cur with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # indices for gradient length n (mirrored)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update: m,v
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr / (np.sqrt(v_hat) + eps)

        cand = cur - step * m_hat
        cand = _project_simplex(cand)

        cand_sc = compute_score(cand)

        # acceptance
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
        # decide whether to accept as new current point
        if cand_sc < compute_score(cur) or rng.random() < 0.001:
            cur = cand
            no_improve = 0
        else:
            no_improve += 1

        # random perturbation if stagnating
        if no_improve > 500:
            # pick random i,j
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            # move some mass from i to j
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            cur = tmp
            no_improve = 0

        # occasional symmetrisation for better shape
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            cur = sym

        # early exit if we beat target
        if best_sc < 1.45:   # far below target
            break

    # final check via evaluate_sequence
    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not math.isfinite(final_sc) or final_sc >= 1e9:
            raise Exception()
    except Exception:
        # fallback uniform
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_fft(vec, n, L):
    """Exact score 2*n*max(conv) for a vector whose sum is 1."""
    a_fft = np.fft.rfft(vec, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    max_val = conv_full[: 2 * n - 1].max()
    return float(2.0 * n * max_val)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initialise ----------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    best_vec = cur.copy()
    best_sc = _score_fft(best_vec, n, L)

    # ---- optimisation state ---------------------------------------------------------
    m = np.zeros_like(cur)          # Adam first moment
    v = np.zeros_like(cur)          # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2                        # base learning rate
    temp_factor = 0.3
    temp_decay = 0.9995
    temp = None
    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1

        # convolution of current iterate
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        # temperature for soft‑max weighting
        if temp is None:
            temp = max_val * temp_factor
        else:
            temp = max(temp * temp_decay, max_val * 1e-5)

        # soft‑max weights over convolution entries
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        idx = (2 * n - 2) - np.arange(n)
