def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

import time, math, numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex (sum = 1, non‑negative)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# Compute full convolution a*a via FFT and return both full array and its maximum
def _conv_full_and_max(a: np.ndarray, L: int, n: int):
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    return conv_full, max_val

    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ------------------------------------------------------------------
    # Determine problem size
    if "height_sequence_1" in globals():
        n = int(height_sequence_1.shape[0])
    else:
        n = 2500                     # reasonable fallback length
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # ------------------------------------------------------------------
    # Initial vector generator (uses parent if available)
    def _initial_vector() -> np.ndarray:
        if "height_sequence_1" in globals():
            a = np.asarray(height_sequence_1, dtype=np.float64).copy()
            a = np.maximum(a, 0.0)
        else:
            a = rng.random(n, dtype=np.float64)
        # tiny random jitter to avoid exact symmetry
        a += rng.normal(scale=0.001, size=n)
        return _project_simplex(a)

    # ------------------------------------------------------------------
    best_vec = None
    best_sc = np.inf

    # Parameters for the outer restart loop
    max_restarts = 5
    restart = 0

    while time.time() < deadline and restart < max_restarts:
        a = _initial_vector()
        a_prev = a.copy()
        momentum = np.zeros_like(a)
        beta = 0.9                     # momentum factor
        it = 0
        no_improve = 0
        cur_sc = _score(a, L, n)

        # store local best (mostly for plateau handling)
        local_best = cur_sc

        while time.time() < deadline:
            it += 1

            # ----- convolution & subgradient ---------------------------------
            conv_full, max_val = _conv_full_and_max(a, L, n)
            grad = _subgrad_max(a, conv_full, max_val, L, n)

            # ----- Nesterov‑style momentum update ---------------------------
            momentum = beta * momentum + (1.0 - beta) * grad
            step = 2.0 / math.sqrt(it)          # initial step magnitude 2.0
            a_new = _project_simplex(a - step * momentum)

            # ----- bookkeeping ---------------------------------------------
            a_prev = a
            a = a_new
            cur_sc = _score(a, L, n)

            if cur_sc < best_sc:
                best_sc = cur_sc
                best_vec = a.copy()
            if cur_sc < local_best * 0.9999:
                local_best = cur_sc
                no_improve = 0
            else:
                no_improve += 1

            # ----- occasional stochastic rescue when stuck ------------------
            if no_improve > 800:
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.25
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)
                cur_sc = _score(a, L, n)
                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best_vec = a.copy()
                local_best = cur_sc
                no_improve = 0

            # ----- early exit if target reached -----------------------------
            if best_sc < 1.50274349:
                break

            # ----- periodic momentum reset to avoid stagnation -------------
            if it % 2000 == 0:
                momentum.fill(0.0)

        restart += 1
        if best_sc < 1.50274349:
            break

    # Fallback – should never happen but guarantees a valid return
    if best_vec is None:
        best_vec = _initial_vector()

    # Return plain Python list of floats
    return [float(x) for x in best_vec]
