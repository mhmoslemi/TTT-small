    start = time.time()

    # Determine n

        init = rng.random(n)

    # compute score via FFT (fast)
    # compute convex combination with reversed for sym.

    def simplex_proj(v):
        # same as before
        v = np.asarray(v, dtype=np.float64)
        n = v.size
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho+1)
        w = v - theta
        w[w < 0] = 0.0
        return w

    def score(v):
        # v assumed sum=1? Not necessary.
        a = np.asarray(v, dtype=np.float64)
        # compute conv via FFT
        L = 1 << ((2*n-1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af*af, n=L)[:2*n-1]
        return 2.0 * n * conv.max()

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L):
    """Return 2·n·max(conv(a,a)) for a (sum does not need to be 1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return 2.0 * a.size * conv.max()

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
