import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)            # sum = 1, non‑negative

    n = cur.shape[0]
    # FFT length: next power of two ≥ 2n‑1
    L = 1 << ((2 * n - 1).bit_length())

    # ----- helper for scoring --------------------------------------------
    def _score_vec(a):
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    # ----- optimisation state ---------------------------------------------
    momentum = np.zeros_like(cur)
    best_vec = cur.copy()
    best_sc = _score_vec(best_vec)
    cur_sc = best_sc

    # hyper‑parameters
    beta = 0.92               # momentum decay
    step0 = 1.5               # base step size
    temp_decay = 0.9995
    min_temp_factor = 1e-5
    no_improve = 0
    target = 1.50274349
    it = 0
    temp = None
    sym_interval = 300
    rand_interval = 2000

    # ----- main loop ------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution of current vector
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # temperature for occasional uphill moves
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # exact max mask (tolerance for floating noise)
        mask = np.isclose(conv_full, max_val, atol=1e-12).astype(np.float64)

        # gradient of the (exact) max via convolution with reversed mask
        w_rev = mask[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * n - 1]
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size inversely proportional to sqrt(iter) and gradient norm
        grad_norm = np.linalg.norm(momentum)
        if grad_norm == 0:
            break
        step = step0 / (math.sqrt(it) * (grad_norm + 1e-12))

        # candidate generation
        cand = _project_simplex(cur - step * momentum)

        # occasional forced symmetry
        if it % sym_interval == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        cand_sc = _score_vec(cand)

        # acceptance test (greedy + occasional uphill)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # random redistribution when stuck
        if no_improve > rand_interval:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # early exit if target reached
        if best_sc < target:
            break

        # time guard
        if time.time() >= deadline:
            break

    # return best vector as plain Python floats
    return [float(x) for x in best_vec]
