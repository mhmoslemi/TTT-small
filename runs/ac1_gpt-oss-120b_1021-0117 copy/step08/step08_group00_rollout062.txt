import time, math, numpy as np

# ------------------------------------------------------------
# utility: Euclidean projection onto the probability simplex
def _project_simplex(v):
    """Project a vector onto the simplex {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    # leave a short safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: uniform random start of moderate length
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)               # sum = 1, non‑negative
    n = a.shape[0]
    # FFT length – next power of two >= 2*n-1
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------
    # scoring (fast, uses the fact sum(a)=1)
    def _score_vec(v):
        fft = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    # Adam optimizer state
    m = np.zeros_like(a)
    v_acc = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.6               # initial learning‑rate scale

    # temperature for soft‑max (actually we’ll use a hard‑max mask)
    eps_mask = 1e-12

    # best solution bookkeeping
    best_vec = a.copy()
    best_sc = _score_vec(best_vec)

    it = 0
    # ------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ---- convolution and peak ------------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # ---- build sub‑gradient of the max (hard mask) ---------------
        mask = (conv >= max_val - eps_mask)
        w = mask.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0.0:          # safety fallback – should not happen
            w = np.ones_like(conv)
            w_sum = w.size
        w /= w_sum

        # ---- gradient of weighted sum w.r.t. a -----------------------
        # using convolution theorem; same pattern as previous code
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---- Adam update --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v_acc = beta2 * v_acc + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)

        cand = _project_simplex(cand)
