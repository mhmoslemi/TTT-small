def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        cur = rng.random(n, dtype=np.float64)
    # ensure nonnegative and simplex
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # optimizer state
    beta = 0.9
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.05   # base learning rate for additive ADAM
    eta = 0.2       # multiplicative step size
    it = 0
    no_improve = 0
    target = 1.50274349

    # temperature schedule
    init_temp_factor = 0.2
    min_temp_factor = 0.005
    temp_decay = 0.995

    while time.time() < deadline:
        it += 1
        # Fourier transforms
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature
        temp = max_val * max(min_temp_factor, init_temp_factor * (temp_decay ** it))

        # softmax weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ADAM additive update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step_vec = alpha * m_hat / (np.sqrt(v_hat) + eps)

        cand_add = cur - step_vec
        cand_add = _project_simplex(cand_add)
        cand_add_sc = _score(cand_add)

        # multiplicative update
        cand_mul = cur * np.exp(-eta * grad)
        # avoid all zeros
        if cand_mul.sum() == 0:
            cand_mul = cur.copy()
        cand_mul = cand_mul / cand_mul.sum()
        cand_mul_sc = _score(cand_mul)

        # pick better candidate
        if cand_add_sc < cand_mul_sc:
            cand = cand_add
            cand_sc = cand_add_sc
        else:
            cand = cand_mul
            cand_sc = cand_mul_sc

        # simple acceptance rule
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional probabilistic uphill move
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # update best if improved
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0
            if best_sc < target:
                break

        # occasional symmetrisation
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                    if best_sc < target:
                        break

        # mass transfer when stalled
        if no_improve > 2000:
            # find indices with highest and lowest gradient
            idx_max = np.argmax(grad)
            idx_min = np.argmin(grad)
            if idx_max != idx_min:
                delta = min(cur[idx_max] * 0.3, 0.002)
                tmp = cur.copy()
                tmp[idx_max] -= delta
                tmp[idx_min] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                        if best_sc < target:
                            break

        # random redistribution when very stuck
        if no_improve > 4000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                    if best_sc < target:
                        break

    # final perturbation stage
    while time.time() < deadline:
        perturb = best_vec + rng.normal(scale=0.003, size=n)
        perturb = np.clip(perturb, 0.0, None)
        perturb = _project_simplex(perturb)
        pert_sc = _score(perturb)
        if pert_sc < best_sc:
            best_sc = pert_sc
            best_vec = perturb.copy()
            if best_sc < target:
                break

    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast evaluation of 2·n·max(conv(a,a)) assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ---- initialise ----------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # ---- optimiser state -----------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    alpha = 0.05          # additive step base
    eta = 0.2             # multiplicative step factor
    it = 0
    no_improve = 0
    target = 1.50274349

    # temperature schedule for soft‑max
    init_temp_factor = 0.2
    min_temp_factor = 0.005
    temp_decay = 0.995

    # -------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- current convolution & temperature ------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        temp = max_val * max(min_temp_factor,
                             init_temp_factor * (temp_decay ** it))

        # ----- smooth‑max weights ---------------------------------------
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of smooth max -----------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- ADAM additive update ------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step_vec = alpha * m_hat / (np.sqrt(v_hat) + eps)

        cand_add = cur - step_vec
        cand_add = _project_simplex(cand_add)
        cand_add_sc = _score(cand_add)

        # ----- multiplicative (mirror‑descent) update -------------------
        cand_mul = cur * np.exp(-eta * grad)
        if cand_mul.sum() == 0:
            cand_mul = cur.copy()
        cand_mul = cand_mul / cand_mul.sum()
        cand_mul_sc = _score(cand_mul)

        # ----- pick better candidate ------------------------------------
        if cand_add_sc < cand_mul_sc:
            cand, cand_sc = cand_add, cand_add_sc
        else:
            cand, cand_sc = cand_mul, cand_mul_sc

        # ----- acceptance -----------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:

                cur, cur_sc = cand, cand_sc

        # ----- record best -----------------------------------------------
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0
            if best_sc < target:
                break

        # ----- occasional symmetrisation --------------------------------
        if it % 500 == 0:

            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                    if best_sc < target:

        if no_improve > 2000:
            i_max = np.argmax(grad)
            i_min = np.argmin(grad)
            if i_max != i_min:

                tmp = cur.copy()
                tmp[i_max] -= delta
                tmp[i_min] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)

                        best_vec = tmp.copy()

        # ----- random redistribution when very stuck --------------------
        if no_improve > 4000:

            j = rng.integers(0, n - 1)

            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                    if best_sc < target:
                        break

    # ---- final local perturbations ------------------------------------
    while time.time() < deadline:
        pert = best_vec + rng.normal(scale=0.003, size=n)
        pert = np.clip(pert, 0.0, None)
        pert = _project_simplex(pert)
        pert_sc = _score(pert)
        if pert_sc < best_sc:
            best_sc = pert_sc
            best_vec = pert.copy()
            if best_sc < target:
                break

    return [float(x) for x in best_vec]
