def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initial vector
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
        cur = np.maximum(cur, 0.0)
        cur = cur / cur.sum()
    else:
        n = 2000
        cur = rng.random(n, dtype=np.float64)
        cur = cur / cur.sum()
    n = cur.shape[0]
    # define L for FFT
    L = 1 << ((2 * n - 1).bit_length())

    def compute_score(a):
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    def project_simplex(v):
        # implementation of projection onto simplex (sum=1, v>=0)
        v = np.asarray(v, dtype=np.float64)
        if v.ndim != 1:
            v = v.ravel()
        # Following algorithm from Duchi et al.
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, len(v) + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        return w

    # initial score
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    # Adam variables
    m = np.zeros_like(cur)
    v_vec = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.04  # step size
    max_step = 0.5
    temp = None
    temp_decay = 0.9997
    min_temp_factor = 1e-5

    it = 0
    no_improve = 0
    target = 1.50274349

    # Main optimization loop
    while time.time() < deadline:
        it += 1

        # compute convolution and softmax weights
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.1
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of smoothed max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v_vec / (1.0 - beta2 ** it)
        step = (lr / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        cand = project_simplex(cur - step)
        cand_score = compute_score(cand)

        # acceptance
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / (temp + 1e-12)):
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            no_improve += 1

        # periodic symmetrization
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = project_simplex(sym)
            sym_score = compute_score(sym)
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                no_improve = 0
                if sym_score < best_score:
                    best = sym.copy()
                    best_score = sym_score

        # random redistribution when stuck
        if no_improve > 3000:
            # small random perturbation
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = project_simplex(tmp)
            tmp_score = compute_score(tmp)
            if tmp_score < cur_score:
                cur = tmp
                cur_score = tmp_score
                no_improve = 0
                if tmp_score < best_score:
                    best = tmp.copy()
                    best_score = tmp_score

        # restart with noise if very stuck
        if no_improve > 6000:
            cur = best + rng.normal(scale=0.01, size=n)
            cur = np.maximum(cur, 0.0)
            cur = project_simplex(cur)
            cur_score = compute_score(cur)
            m.fill(0.0)
            v_vec.fill(0.0)
            no_improve = 0

        # early exit if target reached
        if best_score <= target:
            break

    # If time remains, perform simple annealing fine-tuning
    # Use remaining time
    while time.time() < deadline:
        # small random move
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if cur[i] < 1e-12:
            continue
        delta = cur[i] * rng.random() * 0.1
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # no need to project: sum unchanged, nonneg ensured
        cand_score = compute_score(cand)
        # simple Metropolis acceptance
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / (temp + 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

    # Ensure best is valid list
    return [float(x) for x in best]

import time, math, numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, all entries ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v):
    """Return the Euclidean projection of v onto the simplex {x≥0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        v = v.ravel()
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# ----------------------------------------------------------------------
# Helper: evaluate the AC1 upper bound for a *normalized* vector.
# (The external evaluator works with any scale; we keep sum = 1.)
# ----------------------------------------------------------------------
def _score_norm1(a):
    """Score 2·n·max(conv(a,a)) assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----- initial vector ------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)

    else:

        cur = cur / cur.sum()
    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    # ----- bookkeeping ----------------------------------------------------
    cur_score = _score_norm1(cur)
    best = cur.copy()

    # ----- Adam state -----------------------------------------------------
    m = np.zeros_like(cur)
    v_vec = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.04
    max_step = 0.5

    # ----- temperature (soft‑max approximation of max) --------------------
    temp = None
    temp_decay = 0.9997
    min_temp_factor = 1e-5

    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Phase 1: smoothed‑max gradient descent (Adam)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution of cur with itself
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature schedule
        if temp is None:
            temp = max_val * 0.1
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # soft‑max weights over convolution entries
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:                 # degenerate case – fall back to hard max
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of the smooth max (via FFT, matrix‑free)
        w_rev = w[::-1]

    # Phase 2: simple simulated‑annealing fine‑tuning (uses remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if cur[i] < 1e-12:
            continue

        cand[j] += delta               # sum is preserved, non‑negativity kept

                -(cand_score - cur_score) / (temp + 1e-12)):

            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

    # ------------------------------------------------------------------
