import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Fast surrogate score: 2*n*max(conv) for a vector with sum≈1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return float(2.0 * n * np.max(conv[: 2 * n - 1]))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # initialise
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two FFT size

    best_vec = cur.copy()
    best_sc = evaluate_sequence(best_vec.tolist())
    cur_sc = _fast_score(cur, n, L)

    # optimisation state
    momentum = np.zeros_like(cur)
    beta = 0.93               # momentum decay
    step0 = 1.0               # base step size
    it = 0
    no_improve = 0
    exact_interval = 5000    # exact evaluations
    sym_interval = 700       # symmetrisation

    while time.time() < deadline:
        it += 1

        # ---- convolution and max index --------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        max_idx = int(conv.argmax())

        # ---- gradient of the active max (one‑hot) --------------------
        grad = np.zeros_like(cur)

        momentum = beta * momentum + (1.0 - beta) * grad
