def gradient_of_max_conv(a, conv, max_val, tol=1e-12):
    n = a.shape[0]
    max_inds = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in max_inds:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # iterable over i
        for i in range(i_start, i_end + 1):
            j = k - i
            if i == j:
                grad[i] += 2.0 * a[i]
            else:
                grad[i] += a[j]
                grad[j] += a[i]
    return grad
```

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _conv_max(a, L):
    """Return max of a*a using FFT of size L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv.max(), conv

def _score(a, L):
    """Fast score for a vector already normalized to sum≈1."""
    n = a.shape[0]
    max_val, _ = _conv_max(a, L)
    return float(2.0 * n * max_val)
