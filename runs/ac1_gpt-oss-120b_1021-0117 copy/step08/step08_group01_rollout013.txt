import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

def _score(a):
    """Score for a vector with sum≈1: 2·n·max(a∗a)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------ #
    # Initial sequence (use supplied parent if available)
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 2000
        a0 = rng.random(n, dtype=np.float64)

    n = a0.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    cur0 = _project_simplex(np.maximum(a0, 0.0))
    best_vec = cur0.copy()
    best_sc = _score(best_vec)

    # uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_sc = _score(uniform)
    if uni_sc < best_sc:
        best_sc, best_vec = uni_sc, uniform.copy()

    # ------------------------------------------------------------------ #
    # Hyper‑parameters
    beta = 0.95               # momentum
    step0 = 1.5               # initial step magnitude
    decay = 0.998             # temperature decay
    min_temp_factor = 1e-5    # lower bound for temperature
    target = 1.50274349       # goal to beat

    restart = 0

    # ------------------------------------------------------------------ #
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Choose starting point for this restart
        if restart == 0:
            cur = best_vec.copy()
        else:
            cur = rng.exponential(scale=1.0, size=n)
            cur = _project_simplex(cur)

        cur_sc = _score(cur)
        momentum = np.zeros_like(cur)
        temp = None
        it = 0
        no_improve = 0

        # --------------------------- inner optimisation -------------------
        while time.time() < deadline:
            it += 1

            # convolution via FFT
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()

            # temperature schedule (soft‑max smoothing)
            if temp is None:
                temp = max_val * 0.07
            else:
                temp = max(temp * decay, max_val * min_temp_factor)

            # soft‑max weights
            w = np.exp((conv - max_val) / (temp + 1e-20))
            w_sum = w.sum()
            if w_sum == 0.0:

                w_sum = w.sum()
