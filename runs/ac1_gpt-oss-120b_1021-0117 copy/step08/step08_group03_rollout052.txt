import time, math, numpy as np, cvxpy as cp

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_exact(a, L=None):
    """Exact AC1 score for a vector a that already satisfies sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    max_val = conv_full[: 2 * n - 1].max()
    return float(2.0 * n * max_val)

def _solve_lp(cur, bound, t_mix=0.2):
    """
    Solve the LP:
        max   sum(g)
        s.t.  conv(cur, g) <= bound  (elementwise)
              g >= 0
    Return a convex combination (1-t_mix)*cur + t_mix*(g/sum(g)),
    already normalized to sum=1. Return None on failure.
    """
    n = cur.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.conv(cp.Constant(cur), g) <= bound]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=3000,
                   abstol=1e-8, reltol=1e-6, feastol=1e-8)

    if g.value is None:
        return None

    if s <= 0:
        return None
    g_norm = g0 / s
    new_vec = (1.0 - t_mix) * cur + t_mix * g_norm
