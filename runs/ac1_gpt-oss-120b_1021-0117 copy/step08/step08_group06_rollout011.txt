def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math, random
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)
    
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # length for default: maybe 2000
        cur = rng.random(2000, dtype=np.float64)
    # Ensure nonnegative (should be) and project onto simplex to make sum=1
    cur = _project_simplex(np.maximum(cur, 0.0))
    
    n = cur.size
    # FFT size: next power of two >= 2*n-1
    L = 1 << ((2 * n - 1).bit_length())
    
    best_vec = cur.copy()
    best_sc = evaluate_sequence(best_vec.tolist())
    
    cur_sc = _fast_score(cur, n, L)
    
    momentum = np.zeros_like(cur)
    beta = 0.95
    step0 = 0.7   # base step maybe bigger?
    # Maybe tune: initial step should be moderate relative to scale.
    # as cur entries sum to 1, typical values ~ 1/n ~ .0004 for n=2500.
    # step size maybe 0.1? Actually step updates act on probability vector, projection will renormalize.
    # We'll use moderate step0.
    
    it = 0
    no_improve = 0
    temp = None
    decay = 0.997
    min_temp_factor = 0.001
    exact_eval_interval = 1000
    # Keep track of previous cur for Nesterov
    prev_cur = cur.copy()
    
    while time.time() < deadline:
        it += 1
        # compute convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # temperature schedule
        if temp is None:
            temp = max_val * 0.08
        else:
            temp = max(temp * decay, max_val * min_temp_factor)
        # compute softmax weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum
        
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]
        
        # Update momentum (exponential moving avg)
        momentum = beta * momentum + (1.0 - beta) * grad
        
        # Nesterov lookahead: compute tentative update using momentum
        step = step0 / math.sqrt(it + 1)   # avoid division by zero
        # generate candidates for varying step scaling
        best_cand = None

        for factor in (0.3, 0.6, 1.0, 1.5, 2.0):
            cand = cur - step * factor * momentum
            cand = _project_simplex(cand)
            cand_sc = _fast_score(cand, n, L)
            if cand_sc < best_cand_sc:
                best_cand_sc = cand_sc
                best_cand = cand
        
        cand = best_cand
        cand_sc = best_cand_sc
        
        # acceptance based on surrogate

            accept = True
        else:

            if rng.random() < prob:
                accept = True
        if accept:

            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if it % exact_eval_interval == 0:

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Surrogate score 2*n*max(conv) for a simplex‑normalised vector."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
