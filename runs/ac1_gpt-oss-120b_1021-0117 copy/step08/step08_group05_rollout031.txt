import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_vec(a, L=None):
    """Score = 2*n*max(conv(a,a)) for a summing to 1."""
    n = a.size
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initialization ----
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)               # sum = 1
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score_vec(best_vec, L)
    cur_sc = best_sc

    # ---- quick parametric sweep for a good start ----
    idx = np.arange(n, dtype=np.float64)
    # exponential family
    for lam in np.linspace(0.001, 2.0, 120):
        a = np.exp(-lam * idx)
        a /= a.sum()
        sc = _score_vec(a, L)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
        ar = a[::-1]
        sc = _score_vec(ar, L)
        if sc < best_sc:
            best_sc, best_vec = sc, ar.copy()
    # power‑law family
    for p in np.linspace(0.01, 5.0, 120):
        a = (idx + 1.0) ** (-p)
        a /= a.sum()
        sc = _score_vec(a, L)
        if sc < best_sc:
            best_sc, best_vec = sc, a.copy()
        ar = a[::-1]
        sc = _score_vec(ar, L)
        if sc < best_sc:
            best_sc, best_vec = sc, ar.copy()

    cur = best_vec.copy()
    cur_sc = best_sc

    # ---- Adam optimizer parameters ----
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    lr = 0.12
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1_pow = beta1
    beta2_pow = beta2

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # convolution and max location
        af = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_conv = conv.max()
        k = int(conv.argmax())

        # exact gradient of the max entry
        grad = np.zeros_like(cur)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        i_start = k - j_end
        i_end = k - j_start
        if i_end >= i_start:  # non‑empty overlap
            grad[i_start:i_end + 1] = 2.0 * cur[j_start:j_end + 1][::-1]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score_vec(cand, L)

        # accept if improved
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            # occasional uphill move
            if rng.random() < 0.003:
                cur, cur_sc = cand, cand_sc

        # random redistribution when stuck
        if no_improve > 800:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.35
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

        # occasional symmetrisation
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # decay learning rate gently
        lr *= 0.99995

        # early exit
        if best_sc < target:
            break

    return [float(x) for x in best_vec]
