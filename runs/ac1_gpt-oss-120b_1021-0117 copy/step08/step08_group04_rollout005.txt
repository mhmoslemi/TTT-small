import time, math, numpy as np

def _project_simplex(v):
    """Project a vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_norm1(a):
    """Score for a sequence a with sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        cand_add = _project_simplex(cand_add)

        # evaluate both
        score_mul = _score_norm1(cand_mul)
        score_add = _score_norm1(cand_add)

        if score_mul < score_add:
            cand, cand_score = cand_mul, score_mul
        else:
            cand, cand_score = cand_add, score_add

        # --- acceptance (simulated annealing style) -------------------
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / (temp + 1e-12)):
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional symmetrisation (helps for many instances)
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_norm1(sym)
            if sym_score < best_score:
                cur, cur_score = sym, sym_score
                best, best_score = sym.copy(), sym_score
                no_improve = 0

        # random redistribution if stuck
        if no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score_norm1(tmp)
                if tmp_score < best_score:
                    cur, cur_score = tmp, tmp_score
                    best, best_score = tmp.copy(), tmp_score
                    no_improve = 0
                else:
                    # gentle noise restart around best
                    cur = best + rng.normal(scale=0.005, size=n)
                    cur = _project_simplex(np.maximum(cur, 0.0))
                    cur_score = _score_norm1(cur)
                    no_improve = 0

        # decay learning rate
        eta = max(eta * eta_decay, min_eta)

        # early exit if target reached
        if best_score <= target:
            break

    # return the best sequence found
    return [float(x) for x in best]
