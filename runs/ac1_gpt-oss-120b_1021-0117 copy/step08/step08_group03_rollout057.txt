import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum == 1, v >= 0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    # ----- initialise ----------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: generate a random vector of length ~2500
        a = rng.random(2500, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)          # now sum(a) == 1
    n = a.shape[0]

    # FFT length – power of two >= 2n‑1
    L = 1 << ((2 * n - 1).bit_length())

    # ----- optimisation parameters ---------------------------------------
    best_a = a.copy()
    # initial score (sum==1) -> 2 * n * max(conv)
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    best_score = float(2.0 * n * conv.max())

    # smoothing temperature (start with range of conv)
    temp = max(conv.max() - conv.min(), 1e-12)
    min_temp = 1e-6
    decay_temp = 0.9999

    # step‑size schedule
    eta0 = 0.5

    # tracking stagnation
    last_improve_it = 0
    it = 0

    # pre‑compute index vector for gradient extraction
    idx = (2 * n - 2) - np.arange(n)

    while time.time() < deadline:
        it += 1

        # ---- convolution and smoothed weights -------------------------
        af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(af * af, n=L)
        conv = conv_full[: 2 * n - 1]

        max_conv = conv.max()
        cur_score = float(2.0 * n * max_conv)

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve_it = it
            if best_score < 1.50274349:      # target reached
                break

        # ---- temperature update (gradual cooling) --------------------
        temp = max(min_temp, temp * decay_temp)

        # ---- softmax weights for a smooth approximation of max -------
        # subtract max to avoid overflow, then exponentiate
        w = np.exp((conv - max_conv) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            # fall back to hard max (binary mask)
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # ---- gradient of the smoothed objective -----------------------
        w_rev = w[::-1]
        wf = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * wf, n=L)
        grad = 2.0 * grad_full[idx]

        # ---- exponentiated‑gradient update ----------------------------
        eta = eta0 / math.sqrt(it)
        # clip gradient to avoid overflow in the exponential
        grad_clip = np.clip(grad, -1e3, 1e3)
        factor = np.exp(-eta * grad_clip)
        a = a * factor

        # ---- enforce symmetry (helps flatten autocorrelation) -------
        a = (a + a[::-1]) * 0.5

        # ---- project back onto the simplex ----------------------------
        a = _project_simplex(a)

        # ---- occasional random perturbations to escape plateaus ------
        if it % 2000 == 0:
            # small Gaussian shake
            noise = rng.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

        if it - last_improve_it > 4000:
            # more aggressive restart: blend best with a fresh random vector
            fresh = rng.random(n, dtype=np.float64)
            fresh = _project_simplex(fresh)
            a = 0.7 * best_a + 0.3 * fresh
            a = _project_simplex(a)
            last_improve_it = it   # reset stagnation counter

    # ----- return best found sequence ------------------------------------
    # ensure final values are plain Python floats
    return [float(x) for x in best_a]
