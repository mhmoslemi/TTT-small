    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0., budget_s - 0.2)

    # init

        cur = np.array(height_sequence_1, dtype=float)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)  # ensures sum=1

    n = len(cur)
    L = 1 << ((2*n - 1).bit_length())

    best = cur.copy()
    best_score = _score_norm1(best) # computed via conv (M) and n

    # Adam variables
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    # learning rates schedule:
    base_lr = 0.06
    # step norm bound
    max_step = 0.4

    # temperature schedule but now we use hard mask after some iterations
    temp = None
    temp_start = 0.1  # fraction of max_val
    temp_decay = 0.9997
    min_temp_factor = 1e-8

    # main loop
    it=0
    no_improve=0
    target = 1.50274349

    while time.time() < deadline:
        it+=1

        # compute convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

        # temperature for softmax
        if temp is None:
            temp = max_val * temp_start
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # compute weight mask
        # Use hard mask if temperature is very small; else softmax 
        if temp < max_val * 1e-4:
            # hard mask: all max positions weight equally
            mask = (conv == max_val).astype(float)
        else:
            # softmax weighting
            w = np.exp((conv - max_val) / temp)
            w_sum = w.sum()
            if w_sum == 0:
                mask = (conv == max_val).astype(float)
            else:
                mask = w / w_sum

        # compute gradient via FFT: conv of cur with reversed mask (of length 2n-1)
        mask_rev = mask[::-1]
        mask_fft = np.fft.rfft(mask_rev, n=L)
        grad_full = np.fft.irfft(a_fft * mask_fft, n=L)
        # grad_full length L, take indices corresponding to convolution offset
        idx = (2*n - 2) - np.arange(n)  # mapping for gradient
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = (base_lr / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        cand = _project_simplex(cur - step)
        cand_score = _score_norm1(cand)

        # accept if improves or with prob arg for simulated annealing (use temp)
        if cand_score < best_score or rng.random() < math.exp(-(cand_score - best_score) / (temp + 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                no_improve = 0
        else:
            no_improve += 1

        # periodic symmetrization
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_norm1(sym)
            if sym_score < best_score:

            no_improve = 0

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, non‑negative)
def _project_simplex(v):
    """Project an array onto the simplex {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# Helper: compute the AC1 bound for a sequence with sum == 1
def _score_norm1(a):
    """Return 2 * n * max(a * a) for a non‑negative vector with sum 1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
