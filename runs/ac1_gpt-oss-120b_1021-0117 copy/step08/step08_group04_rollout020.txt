import time, math, numpy as np

def _project_simplex(v):
    """Projection of vector v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_norm1(a):
    """Score for sum(a)=1: 2*n*max(convolution(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        cur = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_score = _score_norm1(best)
    cur_score = best_score

    # Adam optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.1

    # temperature for smooth max
    temp = None
    temp_decay = 0.9997
    min_temp_factor = 1e-6

    it = 0
    no_improve = 0
    target = 1.50274349

    # switch to refinement after 80% of the budget
    refine_start = start + 0.8 * (total_budget - 0.2)

    while time.time() < deadline:
        now = time.time()
        it += 1

        if now < refine_start:
            # ---------- gradient descent (Adam) ----------
            af = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(af * af, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()

            if temp is None:
                temp = max_val * 0.08
            else:
                temp = max(temp * temp_decay, max_val * min_temp_factor)

            w = np.exp((conv - max_val) / temp)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
            w = w / w_sum

            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(af * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            lr = base_lr / math.sqrt(it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            step_norm = np.linalg.norm(step)
            max_step = 0.3
            if step_norm > max_step:
                step = step * (max_step / step_norm)

            cand = _project_simplex(cur - step)
        else:
            # ---------- refinement by random pairwise moves ----------
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.05  # move up to 5% of mass at i
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)

        cand_score = _score_norm1(cand)

        # acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        elif now < refine_start:
            # occasional uphill move (simulated annealing)
            if rng.random() < math.exp(-(cand_score - cur_score) / (temp + 1e-12)):
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            no_improve += 1

        # occasional symmetrisation
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_norm1(sym)
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                no_improve = 0
                if sym_score < best_score:
                    best = sym.copy()
                    best_score = sym_score

        # restart if stuck
        if no_improve > 12000:
            cur = best + rng.normal(scale=0.02, size=n)
            cur = _project_simplex(np.maximum(cur, 0.0))
            cur_score = _score_norm1(cur)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # early exit if target reached
        if best_score <= target:
            break

    # final optional symmetrisation
    final = (best + best[::-1]) * 0.5
    final = _project_simplex(final)
    final_score = _score_norm1(final)
    if final_score < best_score:
        best = final
        best_score = final_score

    return [float(x) for x in best]
