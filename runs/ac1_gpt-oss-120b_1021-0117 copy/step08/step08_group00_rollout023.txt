# ------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

def _fft_len(n):
    """Return the next power‑of‑two length suitable for a convolution of size n."""
    return 1 << ((2 * n - 1).bit_length())

def _score_and_conv(a, L):
    """
    Return (score, full convolution, rfft of a) for a vector a.
    The score is 2 * n * max(conv) assuming sum(a)=1.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    score = float(2.0 * n * max_val)
    return score, conv, a_fft

    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
