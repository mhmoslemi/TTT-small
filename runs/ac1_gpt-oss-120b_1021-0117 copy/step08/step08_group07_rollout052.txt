import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_fft(a, L):
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1 (fast via FFT)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # -------- smooth max (soft‑max) ----------------------------
        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = (conv_full == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # -------- gradient of the smooth max -----------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # -------- Adam update --------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

        # -------- candidate ----------------------------------------
        cand = cur - step_vec
        cand = _project_simplex(cand)
        cand_sc = _score_fft(cand, L)

        if cand_sc < best_score:
            best_score = cand_sc
            best_vec = cand.copy()

        # -------- accept / probabilistic move ----------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # -------- directed mass transfer when stuck ---------------
        if no_improve > 800:
            i_max = np.argmax(grad)
            i_min = np.argmin(grad)
            if i_max != i_min:
                delta = min(cur[i_max] * 0.2, 0.002)
                tmp = cur.copy()
                tmp[i_max] -= delta
                tmp[i_min] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_fft(tmp, L)
                if tmp_sc < best_score:
                    best_score = tmp_sc
                    best_vec = tmp.copy()
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0

        # -------- random redistribution when very stuck ----------
        if no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_fft(tmp, L)
            if tmp_sc < best_score:
                best_score = tmp_sc
                best_vec = tmp.copy()
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0

        # -------- occasional symmetrisation ------------------------
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, L)
            if sym_sc < best_score:
                best_score = sym_sc
                best_vec = sym.copy()
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0

        # -------- early exit if target reached --------------------
        if best_score < target:
            break

    # --------------------------------------------------------------
    # final local‑search polish (small Gaussian perturbations)
    while time.time() < deadline:
        pert = best_vec + rng.normal(scale=0.004, size=n)
        pert = np.clip(pert, 0.0, None)
        pert = _project_simplex(pert)
        sc = _score_fft(pert, L)
        if sc < best_score:
            best_score = sc
            best_vec = pert.copy()
            if best_score < target:
                break

    # Return the best vector as plain Python floats
    return [float(x) for x in best_vec]
