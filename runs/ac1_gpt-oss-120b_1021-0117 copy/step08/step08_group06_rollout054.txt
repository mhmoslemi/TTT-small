import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = css[rho[-1]] / (rho[-1] + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Fast surrogate: 2·n·max(full convolution) for a simplex‐normalized a."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    # best solution tracking
    best_vec = cur.copy()
    best_sc = evaluate_sequence(best_vec.tolist())
    cur_sc = _fast_score(cur, n, L)

    # temperature schedule
    temp = None
    decay = 0.996
    min_temp_factor = 1e-4

    # stagnation handling
    no_improve = 0
    exact_every = 2500      # exact evaluation interval
    target_score = 1.49     # early stop if we beat this

    while time.time() < deadline:
        t += 1

        # --------------------------------------------------------------
        # convolution and soft‑max weights
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # --------------------------------------------------------------
        # gradient (sub‑gradient of weighted max)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # --------------------------------------------------------------
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        step = alpha0 * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # try a few step scalings and keep the best surrogate
        best_cand = None
        best_cand_sc = float('inf')
        for factor in (0.5, 1.0, 1.5):
            cand = _project_simplex(cur - factor * step)
            cand_sc = _fast_score(cand, n, L)
            if cand_sc < best_cand_sc:
                best_cand_sc, best_cand = cand_sc, cand

        cand = best_cand
        cand_sc = best_cand_sc

        # --------------------------------------------------------------
        # acceptance (always accept improvement, sometimes accept worsening)
        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha0 + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if t % exact_every == 0:
                true_sc = evaluate_sequence(cand.tolist())
                if true_sc < best_sc:
                    best_sc, best_vec = true_sc, cand.copy()
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # random redistribution when stuck
        if no_improve > 5000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _fast_score(tmp, n, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                true_sc = evaluate_sequence(tmp.tolist())
                if true_sc < best_sc:
                    best_sc, best_vec = true_sc, tmp.copy()
        # --------------------------------------------------------------
        # occasional symmetrisation
        if t % 900 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                true_sc = evaluate_sequence(sym.tolist())
                if true_sc < best_sc:
                    best_sc, best_vec = true_sc, sym.copy()

        # early exit if we have already beaten the target
        if best_sc < target_score:
            break

    # ------------------------------------------------------------------
    # final sanity check – ensure the returned vector yields a finite score
    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not math.isfinite(final_sc) or final_sc >= 1e9:
            best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    except Exception:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
