import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex (sum=1, non‑negative)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Exact objective: 2·n·max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # ---- bookkeeping ----------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    alpha0 = 0.2          # base learning rate
    eps = 1e-8

    it = 0
    no_improve = 0
    target = 1.50274349

    # ---- main optimisation loop ----------------------------------------
    while time.time() < deadline:
        it += 1
        # FFT of current iterate
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        if cur_sc < best_sc:

        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
