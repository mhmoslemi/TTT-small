import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w
```

def _score_vec(a):
    a = np.asarray(a, dtype=np.float64)

    if S < 1e-12:
        return np.inf
    Q = np.dot(a, a)
    n = a.size

import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_vec(a):
    """Exact score used by the official evaluator, but faster for us."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 0.01:
        return np.inf
    q = np.dot(a, a)                # sum of squares = max of the full convolution
    n = a.size
    return float(2.0 * n * q / (s * s))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ------------------------------------------------------------------
    # Initialise: use the provided construction if it exists, otherwise
    # start from a uniform distribution.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        cur = _project_simplex(cur)
        n = cur.size
    else:
        n = 2000
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = cur.copy()
    best_score = _score_vec(best_vec)

    # ------------------------------------------------------------------
    # Helper: one gradient‑descent step on the exact score.
    # ------------------------------------------------------------------
    def _gd_step(x, step):
        s = x.sum()
        q = np.dot(x, x)
        # gradient of 2*n*q / s^2 w.r.t. x_i
        grad = 4.0 * n * (x * s - q) / (s ** 3)
        y = x - step * grad
        return _project_simplex(y)

    # ------------------------------------------------------------------
    # Main optimisation loop – a mixture of random restarts and a few
    # gradient steps from each restart.  All work stays O(n) in memory.
    # ------------------------------------------------------------------
    it = 0
    step0 = 0.5               # initial step size for gradient moves
    while time.time() < deadline:
        it += 1

        # -------- random restart on the simplex ----------
        r = rng.exponential(scale=1.0, size=n)
        vec = r / r.sum()                     # lies on the simplex

        # -------- a couple of cheap gradient refinements ----------
        for _ in range(3):
            step = step0 / math.sqrt(it)
            vec = _gd_step(vec, step)

        sc = _score_vec(vec)
        if sc < best_score:
            best_score, best_vec = sc, vec.copy()
            # early exit if we already beat the target (unlikely but cheap)
            if best_score < 1.50274349:
                return [float(x) for x in best_vec]

        # -------- occasional deterministic improvement ----------
        if it % 250 == 0:
            # simple symmetrisation (often helps for autocorrelation)
            sym = (vec + vec[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < best_score:
                best_score, best_vec = sym_sc, sym.copy()

    # ------------------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
