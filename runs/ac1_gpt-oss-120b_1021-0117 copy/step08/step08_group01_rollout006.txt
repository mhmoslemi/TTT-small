# Helper: Euclidean projection onto the probability simplex (sum = 1, non‑neg.)
def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# ----------------------------------------------------------------------
# Fast score assuming the vector is already normalized (sum ≈ 1)
def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # Initialise a feasible vector (symmetric, on the simplex)
    if "height_sequence_1" in globals():
        init_full = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_full.shape[0]
        cur = _project_simplex(np.maximum(init_full, 0.0))
    else:
        n = 2000
        cur = _project_simplex(rng.random(n))

    # enforce symmetry from the start
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)

    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters
    target = 1.50274349          # goal to beat
    beta1, beta2 = 0.9, 0.999    # Adam moments
    eps = 1e-8
    base_lr = 0.02               # base learning‑rate factor
    temp_factor = 0.07
    min_temp_factor = 1e-4
    decay = 0.997

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t_iter = 0
    temp = None
    no_improve = 0

    # FFT length (power of two)
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Main optimisation loop
    while time.time() < deadline:
        # ----- convolution and temperature --------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * temp_factor
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # ----- soft‑max weighting -----------------------------------------
        w = np.exp((conv - max_val) / (temp + 1e-20))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of the smoothed max -------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -------------------------------------------------
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = base_lr / (np.sqrt(v_hat) + eps)

        cand = cur - step * m_hat
        cand = _project_simplex(cand)
        # enforce symmetry after projection
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # ----- accept / reject ---------------------------------------------
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            if rng.random() < 0.01:   # occasional uphill move
                cur, cur_sc = cand, cand_sc

        # ----- occasional random local perturbation ------------------------
        if no_improve > 5000:
            # blend a small Dirichlet‑like noise
            noise = rng.exponential(scale=1.0, size=n)
            noise = _project_simplex(noise)
            cur = 0.9 * cur + 0.1 * noise
            cur = _project_simplex(cur)
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)
            cur_sc = _score(cur)
            no_improve = 0

        # ----- cheap hill‑climbing move ------------------------------------
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            eps_swap = cur[i] * rng.random() * 0.02
            if eps_swap > 0.0:
                tmp = cur.copy()
                tmp[i] -= eps_swap
                tmp[j] += eps_swap
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                elif rng.random() < 0.001:
                    cur, cur_sc = tmp, tmp_sc

        # ----- termination checks ------------------------------------------
        if best_sc < target:
            break
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final polishing: small pairwise swaps (hill climbing)
    cur = best_vec.copy()
    cur_sc = best_sc
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        eps_swap = cur[i] * rng.random() * 0.01
        if eps_swap <= 0.0:
            continue
        tmp = cur.copy()
        tmp[i] -= eps_swap
        tmp[j] += eps_swap
        tmp = _project_simplex(tmp)
        tmp = (tmp + tmp[::-1]) * 0.5
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
        elif rng.random() < 0.001:
            cur, cur_sc = tmp, tmp_sc

    # Ensure the final answer respects the simplex and symmetry
    best_vec = (best_vec + best_vec[::-1]) * 0.5
    best_vec = _project_simplex(best_vec)

    # Return a plain Python list of floats
    return [float(x) for x in best_vec]
