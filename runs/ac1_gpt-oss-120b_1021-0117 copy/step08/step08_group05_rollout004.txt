        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]

    cur_score = compute_score(cur)
    best_vec = cur.copy()
    best_score = cur_score

    no_improve = 0
    iteration = 0
    target = 1.50274349

    while time.time() < deadline:

        # FFT of current
        a_fft = np.fft.rfft(cur, n=L)

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum = 1, v >= 0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

def _score_simplex(a: np.ndarray, n: int) -> float:
    """Fast evaluation of 2*n*max(conv(a,a)) assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # keep headroom

    # ---- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)           # enforce sum = 1

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two FFT length
    idx = (2 * n - 2) - np.arange(n)      # index map for gradient extraction

    # ---- Adam hyper‑parameters -----------------------------------------
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    base_lr = 0.20                         # initial learning‑rate factor
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # ---- bookkeeping ----------------------------------------------------
    cur_score = _score_simplex(cur, n)
    best_vec = cur.copy()
    best_score = cur_score

    no_improve = 0
    it = 0
    target = 1.50274349

    # ---- main optimisation loop ----------------------------------------
    while time.time() < deadline:
        it += 1

        # FFT of current iterate
        a_fft = np.fft.rfft(cur, n=L)

        # full convolution a*a
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # --- sub‑gradient of max(conv) (uniform weight over maxima) -----
        tol = max(max_conv * 1e-12, 1e-12)
        mask = (conv >= max_conv - tol).astype(np.float64)
        w = mask / mask.sum()
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # --- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
