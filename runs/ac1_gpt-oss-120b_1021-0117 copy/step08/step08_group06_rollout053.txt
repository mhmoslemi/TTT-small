   v = np.asarray(v, dtype=np.float64)

   n = v.size

   u = np.sort(v)[::-1]

   css = np.cumsum(u) - 1

   rho = np.where(u - css / np.arange(1, n+1) > 0)[0]

   if rho.size == 0:

        theta = 0

        w = np.full_like(v, 1.0 / n)

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    # numerical safety: re‑normalize if sum deviates due to rounding
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w[:] = 1.0 / n
    return w

def _fast_score(a, n, L):
    """Surrogate score 2*n*max(conv) for a simplex‑normalised vector."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
