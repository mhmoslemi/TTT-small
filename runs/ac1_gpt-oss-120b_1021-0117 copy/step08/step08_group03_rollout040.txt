import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _grad_max(a, conv, max_val, L):
    """Subgradient of max(conv) w.r.t. a (conv = a*a)."""
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

            max_val_t = conv_full_t[: 2 * n - 1].max()
            tmp_sc = float(2 * n * max_val_t)

            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0
            else:
                # restart from a fresh random simplex
                cur = _rand_simplex()
                m.fill(0.0); v.fill(0.0)
                beta1_pow, beta2_pow = 1.0, 1.0
                no_improve = 0

        # ---- occasional symmetrisation ------------------------------------
        if it % 1200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            a_fft_s = np.fft.rfft(sym, n=L)
            conv_full_s = np.fft.irfft(a_fft_s * a_fft_s, n=L)
            max_val_s = conv_full_s[: 2 * n - 1].max()
            sym_sc = float(2 * n * max_val_s)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ---- early exit if target reached ----------------------------------
        if best_sc < target:
            break

    # return best found vector as plain Python floats
    return [float(x) for x in best_vec]
