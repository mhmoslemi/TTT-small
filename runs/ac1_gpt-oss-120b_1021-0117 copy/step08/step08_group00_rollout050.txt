import time, numpy as np, math

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # initialization
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # default length 2000
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)   # sum = 1

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # next power of two

    # scoring helper
    def _score_vec(a):
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())

    # initial best
    best_vec = cur.copy()
    best_sc = _score_vec(best_vec)
    cur_sc = best_sc

    # optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05   # base learning rate
    # temperature smoothing
    temp = None
    decay = 0.9995
    min_temp_factor = 1e-5

    no_improve = 0
    it = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # compute convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()

        # temperature schedule
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # softmax weights
        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv_full == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step)
        cand_sc = _score_vec(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            # occasional acceptance of worse solutions (simulated annealing)
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # random redistribution when stuck
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # symmetrisation every 500 iterations
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # sparsify every 1200 iterations
        if it % 1200 == 0:
            thresh = np.quantile(cur, 0.2)
            tmp = cur.copy()
            tmp[tmp < thresh] = 0.0
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        if best_sc < target:
            break

    # return best as list of floats
    return [float(x) for x in best_vec]

import time

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _refine_sequence(vec, L, n, time_budget):

    start = time.time()
    cur = vec.copy()

    def _score(a):

        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    cur_sc = _score(cur)
    it = 0
    # limit both by iteration count and remaining time

        it += 1

        s = mask.sum()
