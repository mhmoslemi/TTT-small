import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Fast surrogate: 2*n*max(conv) for a vector with sum≈1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---------- initial vector ----------
    if "height_sequence_1" in globals():

    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())          # FFT size (next power of two)

    # ---------- best-so-far ----------
    best_vec = cur.copy()

    cur_sc = _fast_score(cur, n, L)

    # ---------- optimisation state ----------
    momentum = np.zeros_like(cur)
    beta = 0.90                # momentum decay
    step0 = 0.80               # base step size
    it = 0
    no_improve = 0
    eval_interval = 1500
    sym_interval = 500
    max_no_improve = 4000

    # pre‑compute index mapping for gradient extraction
    idx = (2 * n - 2) - np.arange(n)

    while time.time() < deadline:
        it += 1

        # ----- convolution & max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- subgradient of the max (indicator weighting) -----
        active = np.where(conv == max_val)[0]

        momentum = beta * momentum + (1.0 - beta) * grad
