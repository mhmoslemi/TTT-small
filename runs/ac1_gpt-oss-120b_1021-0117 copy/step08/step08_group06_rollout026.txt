    v = np.asarray(v, dtype=np.float64)

        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        cand = cur - step
        cand = _project_simplex(np.maximum(cand, 0.0))

        # ----- accept if improves fast surrogate ----------------------
        cand_sc_fast = _fast_score(cand, n, L)
        if cand_sc_fast < cur_sc_fast:
            cur, cur_sc_fast = cand, cand_sc_fast
            no_improve = 0
        else:
            no_improve += 1
            # occasional stochastic acceptance
            if rng.random() < 0.01:
                cur, cur_sc_fast = cand, cand_sc_fast

        # ----- exact evaluation occasionally ---------------------------
        if it % exact_interval == 0:
            true_sc = evaluate_sequence(cur.tolist())
            if true_sc < best_sc:
                best_sc, best_vec = true_sc, cur.copy()
                best_sc_fast = cand_sc_fast
            if best_sc < 1.45:   # early exit if already very good
                break

        # ----- random redistribution if stuck -------------------------
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc_fast = _fast_score(tmp, n, L)
            if tmp_sc_fast < cur_sc_fast:
                cur, cur_sc_fast = tmp, tmp_sc_fast
                no_improve = 0

        # ----- occasional symmetrisation (often helpful) -------------
        if it % 800 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc_fast = _fast_score(sym, n, L)
            if sym_sc_fast < cur_sc_fast:
                cur, cur_sc_fast = sym, sym_sc_fast

    # ---- final local search (cheap random tweaks) --------------------
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = best_vec[i] * rng.random() * 0.07
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc_fast = _fast_score(cand, n, L)
        if cand_sc_fast < best_sc_fast:
            best_vec, best_sc_fast = cand, cand_sc_fast
            best_sc = evaluate_sequence(best_vec.tolist())

    # ---- final sanity check -----------------------------------------
    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not (math.isfinite(final_sc) and final_sc < 1e9):
            raise ValueError
    except Exception:
        # fallback: uniform simplex (guaranteed finite)
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
