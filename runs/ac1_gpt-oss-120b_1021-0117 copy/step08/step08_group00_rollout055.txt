def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # init
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

    cur = np.maximum(cur, 0.0)

    n = cur.shape[0]
    L = 1 << ((2*n-1).bit_length())  # next power of two

    # helper functions
    def compute_conv_and_max(a):
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        return conv, conv.max()

    def score_vec(a):
        conv, mx = compute_conv_and_max(a)
        s = np.sum(a)
        if s == 0:
            return float('inf')
        return 2.0 * n * mx / (s*s)

    # initialize best
    best_vec = cur.copy()
    best_sc = score_vec(best_vec)
    cur_sc = best_sc

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.02  # step size
    it = 0
    no_improve = 0
    temp = None

    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # compute conv, gradient weights
        conv, max_val = compute_conv_and_max(cur)

        # temperature schedule
        if temp is None:
            temp = max_val * 0.05
        else:
            temp = max(temp * 0.9999, max_val * 1e-5)

        # softmax weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]               # sort descending
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    # enforce exact sum=1 (tiny numerical drift)
    s = w.sum()
    if s > 0:
        w /= s
    return w

        """Return full linear convolution a*a and its maximum value."""
