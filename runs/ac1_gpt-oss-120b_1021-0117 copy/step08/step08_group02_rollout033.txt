def propose_candidate(seed=42, budget_s=1100):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0, float(budget_s)-0.5)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2000
    # initialize best vector
    if 'height_sequence_1' in globals():
        init = np.maximum(height_sequence_1.astype(np.float64), 0.0)
    else:
        init = rng.random(n)
    init = _project_simplex(init)
    best_vec = init.copy()
    best_sc = _score(best_vec)

    # Adam parameters
    m = np.zeros_like(best_vec)
    v = np.zeros_like(best_vec)
    beta1,beta2 = 0.9,0.999
    beta1_pow=beta2_pow=1.0
    lr = 0.15
    eps = 1e-8

    # Temperature for softmax
    temp = None
    decay = 0.998

    # Stagnation counters
    no_improve = 0
    iteration = 0
    target = 1.50274349

    while time.time() < deadline:
        iteration+=1

        # FFT conv
        a_fft = np.fft.rfft(best_vec, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

        # update temperature
        if temp is None:
            temp = max_val*0.2
        else:
            temp = max(temp*decay, max_val*1e-6)

        # weights
        # Use softmax but with minimal temperature; if temp < 1e-9, use indicator
        if temp < 1e-9:
            w = (conv == max_val).astype(np.float64)
        else:
            w = np.exp((conv - max_val)/temp)
        w_sum = w.sum()
        if w_sum == 0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        grad_full = np.fft.irfft(a_fft * np.fft.rfft(w[::-1], n=L), n=L)
        idx = (2*n-2) - np.arange(n)
        grad = 2.0*grad_full[idx]

        # Adam
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m/(1-beta1_pow)
        v_hat = v/(1-beta2_pow)
        step = lr * m_hat/(np.sqrt(v_hat)+eps)

        cand = _project_simplex(best_vec - step)
        cand_sc = _score(cand)

        if cand_sc < best_sc:
            best_vec = cand
            best_sc = cand_sc
            no_improve = 0
        else:
            no_improve +=1
            prob = math.exp(-(cand_sc - best_sc)/ (temp+1e-12))
            if rng.random() < prob:
                best_vec = cand
                best_sc = cand_sc

        # occasional perturbations

            i = rng.integers(0,n)
            j = rng.integers(0,n-1)

            tmp[i] -= delta

            tmp = _project_simplex(tmp)

import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L):
    """Return 2*n*max(conv) for a non‑negative vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small head‑room for final return
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialise
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback length
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    lr0 = 0.25          # initial learning rate
    eps = 1e-8

    temp = None
    decay = 0.998
    min_temp_factor = 1e-6

    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # --------------------------------------------------------------
        # full convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # --------------------------------------------------------------
        # temperature for soft‑max (smooth max)
        if temp is None:
            temp = max_val * 0.2
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # --------------------------------------------------------------
        # soft‑max weights (or hard indicator when temperature tiny)
        if temp < 1e-12:
            w = (conv == max_val).astype(np.float64)
        else:
            w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # --------------------------------------------------------------
        # gradient of the weighted max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)

        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

    # ------------------------------------------------------------------
    # return best found candidate as plain Python list
    return [float(x) for x in best_vec]
