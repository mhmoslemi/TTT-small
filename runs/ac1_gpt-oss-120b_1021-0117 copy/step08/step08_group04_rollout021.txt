def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np, random

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # init vector
    if 'height_sequence_1' in globals():
        cur = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2500
        cur = rng.random(n)
    cur = np.maximum(cur, 0)
    cur = _project_simplex(cur)

    n = cur.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    best = cur.copy()
    best_score = _score_norm1(best)
    cur_score = best_score

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    lr0 = 0.1
    max_step = 0.7
    temp = None
    temp_decay = 0.9995
    min_temp_factor = 1e-5

    iteration = 0
    no_improve = 0
    target_score = 1.50274349

    while time.time() < deadline:
        iteration += 1

        # FFT of cur
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # Temperature adapt
        if temp is None:
            temp = max_val * 0.08
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # compute soft weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum

        # compute gradient via convolution with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # The relevant gradient indices correspond to shift positions (2n-2 - np.arange(n))
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        step = (lr0 / math.sqrt(iteration)) * m_hat / (np.sqrt(v_hat) + eps)

        # clip step norm
        step_norm = np.linalg.norm(step)
        if step_norm > max_step:

        cand = _project_simplex(cur - step)
        cand_score = _score_norm1(cand)

import time, math, numpy as np

def _project_simplex(v):
    """Project vector onto the probability simplex (non‑negative, sum=1)."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        v = v.ravel()
    # ensure non‑negative
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # uniform fallback
        return np.full_like(v, 1.0 / v.size)
    v = v / s
    # optional sparsity‑preserving step (standard Euclidean projection)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

def _score_norm1(a):
    """Score for a vector already on the simplex (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 2500
        cur = rng.random(n, dtype=np.float64)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_score = _score_norm1(best)
    cur_score = best_score

    # Adam‑style moments
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    lr0 = 0.10
    max_step = 0.70
    temp = None
    temp_decay = 0.9995
    min_temp_factor = 1e-5

    it = 0
    no_improve = 0
    target = 1.50274349

    # ---- main optimisation loop -----------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature schedule
        if temp is None:
            temp = max_val * 0.08
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # softmax weights over convolution entries
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum

        # gradient of the weighted max (via convolution)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (lr0 / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

        # clip step length
        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        cand = _project_simplex(cur - step)
        cand_score = _score_norm1(cand)

        # acceptance test
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            if rng.random() < math.exp(-(cand_score - cur_score) / (temp + 1e-12)):
                cur, cur_score = cand, cand_score
            no_improve += 1

        # periodic symmetrisation
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_norm1(sym)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                no_improve = 0
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score

        # random redistribution if stagnating
        if no_improve > 3000:
            # pick a small random block and reshuffle its mass
            blk = rng.choice(n, size=5, replace=False)
            total = cur[blk].sum()
            rnd = rng.random(5)
            rnd = rnd / rnd.sum()
            cur[blk] = total * rnd
            cur = _project_simplex(cur)
            cur_score = _score_norm1(cur)
            no_improve = 0

        # restart with tiny noise after long stagnation
        if no_improve > 8000:
            cur = best + rng.normal(scale=0.005, size=n)
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            cur_score = _score_norm1(cur)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        if best_score <= target:
            break

    # ---- final short greedy refinement ----------------------------------
    ref_deadline = min(deadline - 0.05, time.time() + 5.0)
    while time.time() < ref_deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a small random fraction of mass from i to j
        delta = cur[i] * 0.1 * rng.random()
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_score = _score_norm1(cand)
        if cand_score < best_score:
            best, best_score = cand.copy(), cand_score
            cur, cur_score = cand, cand_score

    # return the best found vector as a plain Python list
    return [float(x) for x in best]
