import time, math, numpy as np

def _normalize(v):
    """Clip to non‑negative and scale to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

def _score_vec(a, L, n):
    """2·n·max(conv(a,a)) assuming sum(a)=1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    # --------------------------------------------------------------
    # problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000                     # fallback length
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two for FFT

    # --------------------------------------------------------------
    # generate a pool of diverse starting points
    pool = []

    # 1) parent (if any)
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        pool.append(_normalize(parent))

    # 2) pure exponential family (grid on λ)
    for lam in np.linspace(0.0002, 0.0025, 80):
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        pool.append(_normalize(a))

    # 3) truncated exponential (random m, λ)
    for _ in range(30):
        m = rng.integers(int(0.2 * n), n + 1)
        lam = rng.uniform(0.0002, 0.0025)
        a = np.zeros(n, dtype=np.float64)
        a[:m] = np.exp(-lam * np.arange(m, dtype=np.float64))
        pool.append(_normalize(a))

    # 4) random Dirichlet vectors
    for _ in range(20):
        a = rng.dirichlet(np.ones(n))
        pool.append(a)

    # --------------------------------------------------------------
    # evaluate pool, keep the best
    best_vec = None
    best_sc = float('inf')
    for vec in pool:
        sc = _score_vec(vec, L, n)
        if sc < best_sc:
            best_sc, best_vec = sc, vec.copy()
        if time.time() >= deadline:
            return [float(x) for x in best_vec]

    # --------------------------------------------------------------
    # Adam optimiser parameters
    cur = best_vec.copy()
    cur_sc = best_sc
    m_adam = np.zeros_like(cur)
    v_adam = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    lr, eps = 0.15, 1e-8

    temp = None
    decay = 0.997
    min_temp_factor = 1e-6

    it = 0
    no_improve = 0
    target = 1.50274349

    # --------------------------------------------------------------
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # ----- convolution and soft‑max temperature
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.1
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:               # fallback to exact max mask
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of the smoothed max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam step
        m_adam = beta1 * m_adam + (1.0 - beta1) * grad
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m_adam / (1.0 - beta1_pow)
        v_hat = v_adam / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = _normalize(cur - step)
        cand_sc = _score_vec(cand, L, n)

        # ----- accept / reject
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ----- occasional random reshuffle when stuck
        if no_improve > 2500:
            # take a fresh random Dirichlet vector
            cur = _normalize(rng.dirichlet(np.ones(n)))
            cur_sc = _score_vec(cur, L, n)
            no_improve = 0

        # ----- occasional symmetrisation (helps many instances)
        if it % 1100 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _normalize(sym)
            sym_sc = _score_vec(sym, L, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ----- early exit
        if best_sc < target:
            break

    return [float(x) for x in best_vec]
