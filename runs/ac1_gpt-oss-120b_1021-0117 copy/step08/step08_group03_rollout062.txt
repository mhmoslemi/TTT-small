import time, math, numpy as np

# ----------------------------------------------------------------------
# Helper functions (simplex projection, scoring, gradient of the max‑conv)
# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    """Score 2·n·max(conv(a,a)) for a already on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _grad_max(a, conv, max_val, L):
    """Sub‑gradient of the max‑convolution using a binary mask."""
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)          # reversed slice
    return 2.0 * grad_full[idx]
