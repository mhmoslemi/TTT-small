import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Score used during search (sum of a is assumed to be 1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        # ------------------------------ inner optimisation -------------
        while time.time() < deadline:
            it += 1

            # FFT of current vector
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()

            # temperature schedule (soft‑max temperature)
            if temp is None:
                temp = max_val * 0.07
            else:
                temp = max(temp * decay, max_val * min_temp_factor)

            # soft‑max weighting of convolution entries
            w = np.exp((conv - max_val) / (temp + 1e-20))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
            w /= w_sum

            # gradient of the smooth‑max (matrix‑free via FFT)
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # momentum update and step size
            momentum = beta * momentum + (1.0 - beta) * grad
            step = step0 / math.sqrt(it)

            cand = _project_simplex(cur - step * momentum)
            cand_sc = _score(cand)

            # accept improvement, otherwise Metropolis move
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
            else:
                prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
                if rng.random() < prob:
                    cur, cur_sc = cand, cand_sc

            # occasional symmetrisation (helps for many instances)
            if it % 800 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best_vec = sym_sc, sym.copy()

            # stop inner loop after a reasonable number of iterations
            if it >= 25000:
                break
        restart_counter += 1
        # give up early if only a tiny amount of time remains
        if time.time() + 2.0 >= deadline:
            break

    # ---------------------------------------------------------------------- #
    # Final fine‑grained hill climbing (tiny transfers)
    cur = best_vec.copy()
    cur_sc = best_sc
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        eps = cur[i] * rng.random() * 0.02
        if eps <= 0.0:
            continue
        tmp = cur.copy()
        tmp[i] -= eps
        tmp[j] += eps
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
            # occasional extra symmetrisation
            if rng.random() < 0.07:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best_vec = sym_sc, sym.copy()
        else:
            # tiny probability to escape flat regions
            if rng.random() < 0.001:
                cur, cur_sc = tmp, tmp_sc

    # Return the best vector found (plain Python floats)
    return [float(x) for x in best_vec]
