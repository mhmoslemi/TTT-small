import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_fast(a, n, L):
    """Return 2·n·max(a*a) assuming sum(a)=1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max()

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ---- initial vector ------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_score = _score_fast(best, n, L)
    cur_score = best_score

    # ---- optimisation state -------------------------------------------
    momentum = np.zeros_like(cur)
    beta = 0.9          # momentum decay
    step0 = 0.8         # initial step magnitude
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # ---- compute convolution max and argmax -----------------------
        af = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(af * af, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        argmax = int(conv.argmax())
        cur_score = 2.0 * n * max_val

        # ---- gradient of the max (sub‑gradient) ----------------------
        grad = np.zeros_like(cur)
        k = argmax
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1, dtype=np.int64)
            grad[js] = 2.0 * cur[k - js]

        # ---- momentum update -----------------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad

        # ---- step size ------------------------------------------------
        step = step0 / math.sqrt(it)

        # ---- candidate ------------------------------------------------
        cand = cur - step * momentum
        cand = _project_simplex(cand)
        cand_score = _score_fast(cand, n, L)

        # ---- keep global best -----------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()

        # ---- accept or reject -----------------------------------------
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_score - cur_score) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        # ---- directed mass transfer when progress stalls -------------
        if no_improve > 200:
            i_max = int(np.argmax(grad))
            i_min = int(np.argmin(grad))

        if it % 500 == 0:

                    best = sym.copy()

            tmp = best + rng.normal(scale=0.02, size=n)

            tmp = _project_simplex(tmp)
            tmp_score = _score_fast(tmp, n, L)

    # final local‑search polish (small Gaussian perturbations)
    while time.time() < deadline:

        pert_score = _score_fast(pert, n, L)

            best_score = pert_score
            best = pert.copy()
            if best_score < target:
