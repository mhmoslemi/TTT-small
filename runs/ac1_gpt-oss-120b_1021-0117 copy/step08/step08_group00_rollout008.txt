import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

        # acceptance
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best_vec, best_score = cand.copy(), cand_score
            no_improve = 0
        else:
            # occasional uphill move (simulated annealing flavour)
            prob = math.exp(-(cand_score - cur_score) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score
            no_improve += 1

        # ----- occasional extra moves ---------------------------------
        if no_improve > 2000:               # random redistribution when stuck
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            a_fft_t = np.fft.rfft(tmp, n=L)
            conv_t = np.fft.irfft(a_fft_t * a_fft_t, n=L)[: 2 * n - 1]
            tmp_score = _score_from_max(conv_t.max())
            if tmp_score < cur_score:
                cur, cur_score = tmp, tmp_score
                no_improve = 0
                if tmp_score < best_score:
                    best_vec, best_score = tmp.copy(), tmp_score

        if it % 800 == 0:                    # symmetrisation
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            a_fft_s = np.fft.rfft(sym, n=L)
            conv_s = np.fft.irfft(a_fft_s * a_fft_s, n=L)[: 2 * n - 1]
            sym_score = _score_from_max(conv_s.max())
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if sym_score < best_score:
                    best_vec, best_score = sym.copy(), sym_score

        if it % 1500 == 0:                   # sparsify (prune small entries)
            thresh = np.quantile(cur, 0.2)
            tmp = cur.copy()
            tmp[tmp < thresh] = 0.0
            tmp = _project_simplex(tmp)
            a_fft_sp = np.fft.rfft(tmp, n=L)
            conv_sp = np.fft.irfft(a_fft_sp * a_fft_sp, n=L)[: 2 * n - 1]
            sp_score = _score_from_max(conv_sp.max())
            if sp_score < cur_score:
                cur, cur_score = tmp, sp_score
                if sp_score < best_score:
                    best_vec, best_score = tmp.copy(), sp_score

        # early exit if target reached
        if best_score < target:
            break

    # return best solution as plain Python list
    return [float(x) for x in best_vec]
