    v = np.maximum(v, 0.0)

import time, math, numpy as np

def _normalize_nonneg(v: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and renormalise to sum‑one (simplex)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:                                 # fallback to uniform
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v

def _score_vec(a: np.ndarray) -> float:
    """Score for a vector already normalised to sum‑one: 2·n·max(conv)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())         # next power of two
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
