import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    # ensure sum exactly 1 due to numeric
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _max_conv(a_fft, L, n):
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    return max_val, conv

    L = 1 << ((2 * n - 1).bit_length())

    candidates.append(hann)

    tri = np.clip(tri, 0.0, None)

        w = np.exp((conv - max_val) / temp)

        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (lr / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

        # clip step size
        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        cand = _project_simplex(cur - step)
        cand_score = evaluate_sequence(cand.tolist())

        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / (temp + 1e-12)):

            no_improve = 0
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score

            sym_score = evaluate_sequence(sym.tolist())

import time, math, numpy as np

def _project_simplex(v):
    """Project onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w[:] = 1.0 / n
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
