def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2) # leave margin

    # initialization
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a = rng.random(2500, dtype=np.float64) # or length 2500? but we need same length n? If no parent, length unknown. Let's set n = 2500 maybe.
    # Project onto simplex sum=1 (nonnegative)
    a = project_simplex(np.maximum(a, 0.0))
    n = a.shape[0]

    # Setup FFT length L
    L = 1 << ((2*n - 1).bit_length())

    # State variables
    best = a.copy()
    best_score = score_norm1(best)
    cur_score = best_score

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.08   # initial learning rate maybe larger
    max_step = 0.5

    # temperature schedule
    temp = None
    temp_decay = 0.9995
    min_temp_factor = 1e-5

    it = 0
    no_improve = 0
    target = 1.50274349

    # Additional search loops
    while time.time() < deadline:
        it += 1

        # compute convolution and gradient
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.08
        else:
            # update temperature
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # weight distribution for gradient
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum

        # gradient via convolution with w reversed
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (lr0 / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        cand = project_simplex(a - step)
        cand_score = score_norm1(cand)

        # Acceptance
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / (temp + 1e-12)):
            a, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            no_improve += 1

        # Random redistribution when stuck
        if no_improve > 4000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = a[i] * rng.random() * 0.5
            tmp = a.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = project_simplex(tmp)
            tmp_score = score_norm1(tmp)
            if tmp_score < cur_score:
                a, cur_score = tmp, tmp_score
                no_improve = 0
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score

        # Symmetrisation every 800 steps maybe
        if it % 800 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = project_simplex(sym)
            sym_score = score_norm1(sym)
            if sym_score < cur_score:
                a, cur_score = sym, sym_score
                no_improve = 0
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score

        # Additional local pairwise move every 2000 steps
        if it % 2000 == 0:
            # attempt a few random pair moves
            for _ in range(5):
                i, j = rng.choice(n, size=2, replace=False)
                eps_move = a[i] * rng.random() * 0.1
                if eps_move <= 0: continue
                tmp = a.copy()
                tmp[i] -= eps_move
                tmp[j] += eps_move
                # ensure nonnegative
                if tmp[i] < 0:
                    tmp[i] = 0
                tmp = project_simplex(tmp)
                tmp_score = score_norm1(tmp)
                if tmp_score < cur_score:
                    a, cur_score = tmp, tmp_score
                    no_improve = 0
                    if tmp_score < best_score:
                        best, best_score = tmp.copy(), tmp_score

        # Restart with noise if really stuck
        if no_improve > 8000:
            a = best + rng.normal(scale=0.01, size=n)
            a = project_simplex(np.maximum(a, 0.0))
            cur_score = score_norm1(a)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # Check early exit
        if best_score <= target:
            break

    # fallback: also evaluate a deterministic shape (sin)
    # if not already better:
    if best_score > target:
        # try sin shape
        x = np.arange(n) + 1
        sin_vec = np.sin(np.pi * x / (n + 1))
        sin_vec = sin_vec.clip(min=0)
        sin_vec = project_simplex(sin_vec)
        sin_score = score_norm1(sin_vec)
        if sin_score < best_score:
            best, best_score = sin_vec, sin_score

    # convert to list
    return [float(v) for v in best]

import time, math, numpy as np

def _project_simplex(v):
    """Project vector v onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_norm1(a):
    """Score for sequences with sum = 1: 2*n*max(convolution(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
