import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative step heights with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ---------- initialise ----------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # sum = 1, non‑negative
    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two length for FFT

    # ---------- optimisation state ----------
    m = np.zeros_like(cur)                # first moment (Adam)
    v = np.zeros_like(cur)                # second moment (Adam)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.5                               # base learning rate
    temp = None
    decay = 0.9999
    min_temp_factor = 1e-5
    best_vec = cur.copy()
    # initial score (sum is 1)
    a_fft = np.fft.rfft(cur, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    best_sc = float(2.0 * n * conv.max())
    cur_sc = best_sc

    # ---------- control ----------
    target = 1.50274349
    it = 0
    no_improve = 0

    # ---------- main loop ----------
    while time.time() < deadline:
        it += 1

        # --- convolution and temperature ---
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_conv = conv.max()

        if temp is None:
            temp = max_conv * 0.07
        else:
            temp = max(temp * decay, max_conv * min_temp_factor)

        # --- soft‑max weights (smooth approximation of max) ---
        # avoid overflow/underflow
        w_exp = (conv - max_conv) / temp
        w_exp = np.clip(w_exp, -50, 50)
        w = np.exp(w_exp)
        w_sum = w.sum()
        if w_sum == 0.0:                     # fallback
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # --- gradient via FFT (cross‑correlation) ---
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)

        # --- evaluate candidate ---
        cand_fft = np.fft.rfft(cand, n=L)
        conv_cand = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_sc = float(2.0 * n * conv_cand.max())

        # --- acceptance ---
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            # occasional stochastic acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        if it % 700 == 0:

            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)

                    best_vec, best_sc = sym.copy(), sym_sc

                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # --- early exit if target achieved ---
        if best_sc < target:
            break

    # return best seen sequence as plain Python floats
    return [float(x) for x in best_vec]
