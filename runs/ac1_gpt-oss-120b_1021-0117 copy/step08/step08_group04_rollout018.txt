import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_norm1(a, L=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2 * n - 1]
    return float(2.0 * n * conv.max())

def _generate_symmetric_shape(n, kind, params):
    # distance from center
    d = np.abs(np.arange(n) - (n - 1) / 2.0)
    if kind == "gaussian":
        sigma = params[0]
        a = np.exp(-0.5 * (d / sigma) ** 2)
    elif kind == "exponential":

        a = np.exp(-lam * d)
    elif kind == "parabolic":
        R, p = params
        a = np.maximum(0.0, 1.0 - (d / R) ** 2) ** p
    elif kind == "raised_cosine":
        R = params[0]
        # cosine bell: positive for d <= R
        a = np.where(d <= R, (np.cos(np.pi * d / (2 * R))) ** 2, 0.0)
    else:  # default uniform

    # normalize to sum 1
    s = a.sum()
    if s <= 0:

    else:

    return a

import time, math, numpy as np

def _project_simplex(v):
    """Project onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_norm1(a, L=None):
    """Score assuming sum(a)=1: 2*n*max(convolution(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
