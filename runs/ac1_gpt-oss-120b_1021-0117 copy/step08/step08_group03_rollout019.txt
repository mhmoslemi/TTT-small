def _grad_weight(a, w, L):
    # a shape (n,)
    # w shape (2n-1,)
    n = a.shape[0]
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    idx = (2*n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import time, numpy as np, math, random
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # init
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)  # some length
    cur = np.maximum(cur, 0.0)
    cur = cur / (np.sum(cur) + 1e-12)  # normalize to sum=1

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5   # initial learning rate
    alpha = 0.5   # smoothing parameter start
    alpha_max = 30.0
    alpha_inc = (alpha_max - alpha) / 2000.0  # increase over ~2000 iters

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # compute convolution
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2*n - 1]
        max_val = conv.max()

        # compute weight vector for smoother max
        # w = exp(alpha * (conv - max_val))
        w = np.exp(alpha * (conv - max_val))

        # gradient
        grad = _grad_weight(cur, w, L)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)

        # step size
        lr = lr0 / math.sqrt(it)

        # descent direction
        step_dir = m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - lr * step_dir
        # project onto simplex
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0:
            cand = cur
        else:
            cand = cand / s

        # evaluate
        cand_sc = _score(cand, L)

        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        # increase smoothing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha + alpha_inc)

        # random rescue if stuck
        if no_improve >= 600:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            # transfer a fraction of mass from i to j
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = np.maximum(tmp, 0.0)
            s = tmp.sum()
            if s > 0:
                tmp = tmp / s
            tmp_sc = _score(tmp, L)
            if tmp_sc < cur_sc:
                cur = tmp
                cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                no_improve = 0
            else:
                # occasional symmetrization
                sym = (cur + cur[::-1]) * 0.5
                sym = np.maximum(sym, 0.0)
                sym = sym / sym.sum()
                sym_sc = _score(sym, L)
                if sym_sc < cur_sc:
                    cur = sym
                    cur_sc = sym_sc
                    if sym_sc < best_sc:
                        best_sc = sym_sc
                        best_vec = sym.copy()
                    no_improve = 0

        # early exit if reached target
        if best_sc < target:
            break

    # Return best as list
    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    """Fast evaluation for a vector with sum≈1 (returns 2·n·max(conv))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _grad_weight(a, w, L):
    """
    Gradient of sum_k w_k * (a*a)_k w.r.t. a.
    w has length 2n-1, L is FFT length.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # small safety margin

    # ---- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

    cur = np.maximum(cur, 0.0)
    cur = cur / (cur.sum() + 1e-12)        # simplex normalization

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # ---- Adam parameters ------------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.5               # base learning rate
    alpha = 0.5             # smoothing for max; will increase
    alpha_max = 30.0
    alpha_inc = (alpha_max - alpha) / 2000.0

    it = 0
    no_improve = 0
    target = 1.50274349

    # ---- main optimisation loop ----------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution and smoothed weights
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        w = np.exp(alpha * (conv - max_val))

        # gradient of smoothed objective
        grad = _grad_weight(cur, w, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # descent step
        lr = lr0 / math.sqrt(it)
        step_dir = m_hat / (np.sqrt(v_hat) + eps)
