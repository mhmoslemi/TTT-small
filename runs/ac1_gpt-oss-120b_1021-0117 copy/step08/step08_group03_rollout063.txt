import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function sequence.
    If the parent sequence ``height_sequence_1`` is available we use it
    directly (it already satisfies the required constraints).  Otherwise we
    fall back to a simple uniform distribution of length 2000.
    """
    # The harness may call this repeatedly; we honour the time budget
    _ = seed, budget_s, kwargs  # unused, keep signature

    # Use the provided parent construction when present.
    if "height_sequence_1" in globals():

        seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
