def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    if np.all(v <= 0):
        # Uniform distribution
        return np.full_like(v, 1.0 / v.size)
    # sort v descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _project_simplex(v):
    """Project onto the probability simplex (sum = 1, elements >= 0)."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    # If all entries are non‑positive, return uniform distribution.
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _score_norm1(a, L):
    """Score assuming sum(a) == 1 : 2 * n * max(convolution(a, a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            # --- soft‑max weights for gradient approximation ----------
            w = np.exp((conv - max_val) / temp)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
            w = w / w_sum

            # --- gradient of the smoothed max -------------------------
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # --- Adam update -----------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = (lr / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

            # --- step size clipping ----------------------------------
            step_norm = np.linalg.norm(step)
            if step_norm > max_step:
                step = step * (max_step / step_norm)

            cand = _project_simplex(cur - step)
            cand_score = _score_norm1(cand, L)

            # --- acceptance test --------------------------------------
            accept = False
            if cand_score < cur_score:
                accept = True
            else:
                prob = math.exp(-(cand_score - cur_score) / (temp + 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur, cur_score = cand, cand_score
                no_improve = 0
                if cand_score < best_score:
                    best, best_score = cand.copy(), cand_score
                    if best_score <= 1.50274349:
                        return [float(x) for x in best]
            else:
                no_improve += 1

            # --- tiny random perturbation (exploration) -------------
            if rng.random() < 0.001:
                cur = _project_simplex(cur + rng.normal(scale=0.001, size=n))
                cur_score = _score_norm1(cur, L)
                no_improve = 0

            # --- occasional local mass transfer ------------------------
            if rng.random() < 0.005:
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = min(cur[i], rng.random() * 0.02)
                if delta > 0:
                    cand = cur.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = _project_simplex(cand)
                    cand_score = _score_norm1(cand, L)
                    if cand_score < cur_score:
                        cur, cur_score = cand, cand_score
                        no_improve = 0
                        if cand_score < best_score:
                            best, best_score = cand.copy(), cand_score

            # --- periodic symmetrisation -------------------------------
            if it % 2000 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_score = _score_norm1(sym, L)
                if sym_score < cur_score:
                    cur, cur_score = sym, sym_score
                    no_improve = 0
                    if sym_score < best_score:
                        best, best_score = sym.copy(), sym_score

            # --- restart if we are stuck for a long time -------------
            if no_improve > 8000:
                cur = best + rng.normal(scale=0.01, size=n)
                cur = _project_simplex(np.maximum(cur, 0.0))
                cur_score = _score_norm1(cur, L)
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0

        run_idx += 1
        # early exit if target already met
        if best_score <= 1.50274349:
            break

    # -----------------------------------------------------------------
    # Return the best sequence found (as a plain python list of floats)
    return [float(x) for x in best]
