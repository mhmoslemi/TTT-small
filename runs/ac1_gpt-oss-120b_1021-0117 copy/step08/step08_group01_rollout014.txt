def propose_candidate(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.1) # small margin

    if 'height_sequence_1' in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
        n = init.shape[0]
    else:
        n = 2000
        init = rng.random(n, dtype=np.float64)

    # Normalize init to simplex
    cur0 = project_simplex(np.maximum(init, 0.0))
    best_vec = cur0.copy()
    best_sc = score(best_vec)

    # also evaluate uniform as baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_sc = score(uniform)
    if uni_sc < best_sc:
        best_sc, best_vec = uni_sc, uniform.copy()

    # Precompute constants
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)

    # optimization hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.1  # initial learning rate
    decay_lr = 0.9995  # maybe reduce gradually

    temp_decay = 0.998
    min_temp_factor = 1e-4
    target = 1.50274349
    restart_counter = 0
    max_restarts = 10  # maybe unlimited but limited by time

    # Outer loop
    while time.time() < deadline:
        # Choose starting point for this run
        if restart_counter == 0:
            cur = best_vec.copy()
        else:
            # random Dirichlet-like init via exponential
            cur = rng.exponential(scale=1.0, size=n)
            cur = project_simplex(cur)

        cur_sc = score(cur)

        # Adam state
        m = np.zeros_like(cur)
        v = np.zeros_like(cur)

        # Temperature
        temp = None

        it = 0
        no_improve = 0

        while time.time() < deadline:
            it += 1

            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()

            # set temperature
            if temp is None:
                temp = max_val * 0.07
            else:
                temp = max(temp * temp_decay, max_val * min_temp_factor)

            # compute weight for gradient
            # use softmax for early iterations (temp > some threshold)
            # else use indicator of max
            if temp > max_val * 0.001:  # using temperature-based criterion
                w = np.exp((conv - max_val) / (temp + 1e-20))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = (conv == max_val).astype(np.float64)
                    w_sum = w.sum()
                w /= w_sum
            else:
                # indicator of max
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
                w = w / w_sum

            # gradient
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            grad = 2.0 * grad_full[idx]

            # Adam updates
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)

            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)

            # learning rate schedule
            lr = lr0 * (decay_lr ** it)

            # step
            cand = cur - lr * m_hat / (np.sqrt(v_hat) + eps)
            cand = project_simplex(cand)

            cand_sc = score(cand)

            # acceptance
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
            else:
                no_improve += 1
                # occasional uphill move probabilistically
                prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
                if rng.random() < prob:
                    cur, cur_sc = cand, cand_sc

            # restart condition
            if no_improve > 5000:
                # random perturbation: move some mass between positions
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.3
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = project_simplex(tmp)
                tmp_sc = score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                else:
                    no_improve = 0  # reset anyway

            # occasional symmetrization
            if it % 1000 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = project_simplex(sym)
                sym_sc = score(sym)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best_vec = sym_sc, sym.copy()

            if best_sc < target:
                break

        restart_counter += 1
        if best_sc < target:
            break

    # final local refinement: hill climbing
    cur = best_vec.copy()
    cur_sc = best_sc
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        eps = cur[i] * rng.random() * 0.02
        if eps <= 0.0:
            continue
        tmp = cur.copy()
        tmp[i] -= eps
        tmp[j] += eps
        tmp = project_simplex(tmp)
        tmp_sc = score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
            # occasional symmetrization
            if rng.random() < 0.05:
                sym = (cur + cur[::-1]) * 0.5
                sym = project_simplex(sym)
                sym_sc = score(sym)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best_vec = sym_sc, sym.copy()
        else:
            # small probability accept uphill
            if rng.random() < 0.001:
                cur, cur_sc = tmp, tmp_sc

        # break early if target reached
        if best_sc < target:
            break

    # final scaling to ensure sum >= 0.01 maybe already sum=1
    best_seq = np.maximum(best_vec, 0.0)  # ensure nonnegative
    # convert to Python list
    return [float(x) for x in best_seq]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast objective (2*n*max(convolution)) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # --------------------------------------------------------------- #
    # Problem size & initial vector
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
        n = init.shape[0]
    else:
        n = 2000
        init = rng.random(n, dtype=np.float64)

    cur0 = _project_simplex(np.maximum(init, 0.0))
    best_vec = cur0.copy()
    best_sc = _score(best_vec)

    # baseline uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_sc = _score(uniform)
    if uni_sc < best_sc:
        best_sc, best_vec = uni_sc, uniform.copy()

    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)

    # --------------------------------------------------------------- #
    # Hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr0 = 0.1
    lr_decay = 0.9995
    temp_decay = 0.998
    min_temp_factor = 1e-4
    target = 1.50274349
    restart_counter = 0

    # --------------------------------------------------------------- #
    while time.time() < deadline:
        # ----- starting point for this outer run -----
        if restart_counter == 0:
            cur = best_vec.copy()
        else:
            cur = rng.exponential(scale=1.0, size=n)
            cur = _project_simplex(cur)

        cur_sc = _score(cur)

        # Adam state

        v = np.zeros_like(cur)
        beta1_pow, beta2_pow = 1.0, 1.0

        temp = None
        it = 0
        no_improve = 0

        # ---------------- inner optimisation -----------------
        while time.time() < deadline:
            it += 1

            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()

            # temperature schedule
            if temp is None:

            j += 1

        tmp[i] -= eps

    best_vec = np.maximum(best_vec, 0.0)
    return [float(x) for x in best_vec]
