def compute_score(a):
    n = len(a)
    L = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2*n-1]
    max_val = conv.max()
    # Since sum may not be exactly 1 (due to numerical). compute sum.
    sum_a = a.sum()
    return 2.0 * n * max_val / (sum_a**2)

import time, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    sum_a = a.sum()
    return 2.0 * n * max_val / (sum_a * sum_a)

import time, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum = 1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L):
    """Fast AC1 upper‑bound for a vector a (assumes sum(a)≈1)."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    return 2.0 * n * max_val / (s * s)

    start = time.time()

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # sum = 1, non‑negative
    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power‑of‑two FFT length

    best_sc = _score(best_vec, L)

        elif typ == 1:        # Dirichlet (random probability vector)
