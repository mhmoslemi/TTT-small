import time, math, numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w


def _score_norm1(a, fft_len):
    """Score for a vector whose entries sum to 1:
       2 * n * max(convolution(a, a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a short margin for the final return
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.maximum(np.asarray(height_sequence_1, dtype=np.float64), 0.0)
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(cur)            # sum = 1, non‑negative
    n = cur.size
    fft_len = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_score = _score_norm1(best, fft_len)

    # ---- Adam state ----------------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.05
    max_step = 0.6

    # ---- temperature schedule -----------------------------------------
    temp = None
    temp_decay = 0.9995
    min_temp_factor = 5e-6

    it = 0
    stagnant = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1
        # convolution of cur with itself
        a_fft = np.fft.rfft(cur, n=fft_len)
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
        max_val = conv_full.max()

        # temperature for sub‑gradient smoothing
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # smooth max‑indicator
        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:                # numerical safety
            w = (conv_full == max_val).astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum

        # sub‑gradient of the max‑convolution term
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=fft_len)
        grad_full = np.fft.irfft(a_fft * w_fft, n=fft_len)[: 2 * n - 1]
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (lr0 / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

        if step_norm > max_step:

            stagnant = 0
