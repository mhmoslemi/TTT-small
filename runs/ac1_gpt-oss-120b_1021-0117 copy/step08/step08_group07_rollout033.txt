import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum = 1, non‑negative)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ---- initialise -------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    a_fft = np.fft.rfft(cur, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    cur_sc = 2.0 * n * max_val
    best_vec = cur.copy()
    best_sc = cur_sc

    # ---- Adam optimiser ---------------------------------------------------
    lr = 0.2
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t = 0
    no_improve = 0
    target = 1.50274349

    # ----------------------------------------------------------------------
    while True:
        if time.time() >= deadline:
            break

        # gradient of the (soft) max of convolution
        tol = 1e-12
        w = (conv >= max_val - tol).astype(np.float64)
        if w.sum() == 0.0:                     # safety fallback
            w = np.ones_like(conv) / conv.size
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_fft = np.fft.rfft(cand, n=L)
        conv_cand = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        max_val_cand = conv_cand.max()
        cand_sc = 2.0 * n * max_val_cand

        # accept if better, otherwise occasional uphill move
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            a_fft, conv, max_val = cand_fft, conv_cand, max_val_cand
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
                a_fft, conv, max_val = cand_fft, conv_cand, max_val_cand
                no_improve = 0

        # keep global best
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
            if best_sc < target:
                break

        # ---- stagnation handling -----------------------------------------
        if no_improve > 2500:
            # random mass exchange
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_fft = np.fft.rfft(tmp, n=L)
            conv_tmp = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
            tmp_sc = 2.0 * n * conv_tmp.max()
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                a_fft, conv, max_val = tmp_fft, conv_tmp, conv_tmp.max()
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    if best_sc < target:
                        break

        if t % 1500 == 0:      # occasional symmetrisation
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            conv_sym = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * conv_sym.max()
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                a_fft, conv, max_val = sym_fft, conv_sym, conv_sym.max()
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                    if best_sc < target:
                        break

    # ---- final polishing --------------------------------------------------
    while time.time() < deadline:
        pert = best_vec + rng.normal(scale=0.003, size=n)
        pert = np.clip(pert, 0.0, None)
        pert = _project_simplex(pert)
        pert_fft = np.fft.rfft(pert, n=L)
        conv_pert = np.fft.irfft(pert_fft * pert_fft, n=L)[: 2 * n - 1]
        pert_sc = 2.0 * n * conv_pert.max()
        if pert_sc < best_sc:
            best_sc, best_vec = pert_sc, pert.copy()
            if best_sc < target:
                break

    return [float(x) for x in best_vec]
