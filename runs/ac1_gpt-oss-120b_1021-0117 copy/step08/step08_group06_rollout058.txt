    rng = np.random.default_rng(seed)
    start = time.time()

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    # best
    best_vec = cur.copy()
    best_sc = evaluate_sequence(best_vec.tolist())
    cur_sc = 2.0 * n * np.max(np.fft.irfft(np.fft.rfft(cur, n=L) * np.fft.rfft(cur, n=L), n=L)[:2*n-1])
    # momentum etc.
    momentum = np.zeros_like(cur)
    beta = 0.94
    step0 = 1.0
    temperature = None
    decay = 0.997
    min_temp_factor = 5e-5
    no_improve = 0
    it = 0
    exact_eval_every = 3000
    sym_interval = 500
    # maybe also random shake interval

    while time.time() < deadline:

        # compute conv and max

        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

        if temperature is None:
            temperature = max_val * 0.08
        else:
            temperature = max(temperature * decay, max_val * min_temp_factor)
        # softmax weights

        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):

            w_sum = w.sum()
        w = w / w_sum

        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)

        # But we compute using formula from previous code:
        idx = (2*n - 2) - np.arange(n)   # note rev mapping

        # try candidates with different scaling factors
        best_cand_sc = float('inf')

                best_cand = cand

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho_arr.size == 0 else css[rho_arr[-1]] / (rho_arr[-1] + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w
