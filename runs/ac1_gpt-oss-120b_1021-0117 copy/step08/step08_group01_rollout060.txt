def project_simplex(v):
    # onto simplex sum=1, v>=0
   n=len(v)
   u = np.sort(v)[::-1]
   css = np.cumsum(u) - 1.0
   rho = np.where(u - css/np.arange(1,n+1) > 0)[0]
   if rho.size==0:
       theta=0
   else:
       rho = rho[-1]
       theta = css[rho]/(rho+1)
   w = v - theta
   w[w<0]=0
   return w

def fast_score(a):
    # assume sum ~1
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0*n*conv.max())

def subgradient_refine(a, max_iter, deadline):
    n = a.shape[0]; L = compute length global maybe pass.
    momentum = np.zeros_like(a)
    beta = 0.9
    base_step = 1.0
    best_local = a.copy()
    best_sc_local = _fast_score(a)
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        # compute conv via fft
        af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(af * af, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # find indices where conv close to max (within small relative tolerance)
        # we can use np.where(conv >= max_val - 1e-12)[0] (absolute tolerance)
        max_idxs = np.where(conv >= max_val - 1e-12)[0]
        # compute gradient
        grad = np.zeros_like(a)
        for k in max_idxs:
            # shift: for j where 0 <= k-j < n => j in [k-(n-1), k]
            lo = max(0, k - (n-1))
            hi = min(k, n-1)
            # vectorized: for j in lo..hi, a_idx = k - j
            # We can slice: grad[lo:hi+1] += 2 * a[k - np.arange(lo, hi+1)]
            idxs = np.arange(lo, hi+1)
            grad[idxs] += 2 * a[k - idxs]
        # average over multiple max indices
        grad /= len(max_idxs)
        # momentum update
        momentum = beta * momentum + (1 - beta) * grad
        step = base_step / math.sqrt(it)
        a_new = _project_simplex(a - step * momentum)
        sc_new = _fast_score(a_new)
        if sc_new < best_sc_local:
            best_sc_local = sc_new; best_local = a_new.copy()
        # Accept if improved or maybe accept? We'll accept always if improved else keep old (no uphill)
        if sc_new < _fast_score(a):
            a = a_new
        else:
            # occasional random perturbation
            if rng.random() < 0.01:
                a = _project_simplex(a + 0.01 * (rng.random(n) - 0.5))
    return best_local, best_sc_local

import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def _grad_max(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[:2*n-1]
    max_val = conv.max()
    # tolerance
    eps = 1e-12 * max_val if max_val > 0 else 1e-12
    max_idxs = np.where(conv >= max_val - eps)[0]
    grad = np.zeros_like(a)
    n_minus_one = n - 1
    for k in max_idxs:
        lo = max(0, k - n_minus_one)
        hi = min(k, n_minus_one)
        idxs = np.arange(lo, hi+1)
        grad[idxs] += 2.0 * a[k - idxs]
    grad /= max_idxs.size
    return grad, max_val

import time, math, numpy as np

# --------------------------------------------------------------
# Helper functions (top‑level, no closures)
# --------------------------------------------------------------

def _project_simplex(v):
    """Project onto the probability simplex (sum=1, non‑negative)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _fast_score(a):
    """Score = 2·n·max(a∗a)  (a is already normalised to sum≈1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _grad_max(a):
    """
    Sub‑gradient of the max‑convolution.
    Returns (grad, max_val) where grad is the averaged shift
    of a corresponding to the maximal convolution entries.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = 1e-12 * max_val if max_val > 0 else 1e-12
    max_idxs = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    n1 = n - 1
    for k in max_idxs:
        lo = max(0, k - n1)
        hi = min(k, n1)
        if lo <= hi:
            idxs = np.arange(lo, hi + 1)
            grad[idxs] += 2.0 * a[k - idxs]
    grad /= max_idxs.size
    return grad, max_val

    Simple projected sub‑gradient descent with momentum.
    Returns the best vector found and its score.
    """

    best_local = a.copy()
    best_sc_local = _fast_score(a)
    no_improve = 0

        cand = _project_simplex(a - step * momentum)
