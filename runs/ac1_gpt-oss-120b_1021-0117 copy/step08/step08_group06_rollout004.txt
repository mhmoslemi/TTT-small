import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Surrogate score = 2 * n * max(convolution) for a sum‑1 vector."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _grad_max_conv(a, max_idxs):
    """Average sub‑gradient of max convolution over all maximal indices."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if max_idxs.size == 1:
        k = max_idxs[0]
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        idx = k - np.arange(i0, i1 + 1)
        g[i0:i1 + 1] = 2.0 * a[idx]
    else:
        # average over several argmax positions
        for k in max_idxs:
            i0 = max(0, k - (n - 1))
            i1 = min(n - 1, k)
            idx = k - np.arange(i0, i1 + 1)
            g[i0:i1 + 1] += 2.0 * a[idx]
        g /= max_idxs.size
    return g
