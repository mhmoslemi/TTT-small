import time, math, numpy as np

def _project_simplex(v):
    # ... implement O(n log n) algorithm.
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_norm1(a, L=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

            tmp_score = _score_norm1(tmp, L)
            if tmp_score < cur_score:
                cur, cur_score = tmp, tmp_score
                no_improve = 0
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score

        # occasional symmetrisation
        if it % 800 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_norm1(sym, L)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                no_improve = 0
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score

        # full random restart if really stuck
        if no_improve > 7000:
            cur = rng.random(n, dtype=np.float64)
            cur = _project_simplex(cur)
            cur_score = _score_norm1(cur, L)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        if best_score <= target:
            break

    # Return the best vector as a plain Python list of floats
    return [float(x) for x in best]
