import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = css[rho[-1]] / (rho[-1] + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def _grad_for_lag(a, lag):
    n = a.shape[0]
    grad = np.zeros_like(a)
    if lag < n:
        grad[:lag+1] = 2.0 * a[lag::-1]
    else:
        start = lag - (n - 1)
        length = 2 * n - 1 - lag
        grad[start:] = 2.0 * a[n-1:n-1-length:-1]
    return grad

def _contrib_for_lag(a, lag):
    n = a.shape[0]

    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()

    best_vec = cur.copy()

    # fast surrogate score for current vector (sum=1)
    cur_sc = _fast_score(cur, n, fft_len)

    # optimisation state (Adam)
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5   # base learning rate
    it = 0
    no_improve = 0
    exact_interval = 800   # evaluate true score every this many iterations

    while time.time() < deadline:
        it += 1

        # compute convolution and locate worst lag
        a_fft = np.fft.rfft(cur, n=fft_len)
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)
        conv = conv_full[:2 * n - 1]
        max_val = conv.max()
        max_idx = conv.argmax()

        # ----- gradient step targeting the worst lag -----
        grad = _grad_for_lag(cur, max_idx)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

                    no_improve = 0

import time, math, numpy as np

def _project_simplex(v):
    """Project vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = css[rho[-1]] / (rho[-1] + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Fast surrogate: 2*n*max(convolution) for a sequence with sum≈1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _grad_for_lag(a, lag):
    """Gradient of the convolution value at a given lag."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if lag < n:
        g[: lag + 1] = 2.0 * a[lag::-1]
    else:
        start = lag - (n - 1)
        length = 2 * n - 1 - lag
        g[start:] = 2.0 * a[n - 1 : n - 1 - length : -1]
    return g

def _contrib_for_lag(a, lag):
    """Per‑coordinate contribution to the convolution at a given lag."""
    n = a.shape[0]

    else:
        start = lag - (n - 1)
        length = 2 * n - 1 - lag
        c[start:] = a[start:] * a[n - 1 : n - 1 - length : -1]
    return c
