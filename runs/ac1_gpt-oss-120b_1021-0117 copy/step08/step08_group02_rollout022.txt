def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2) 
    # small headroom

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()  # normalize
    # symmetrize
    a = (a + a[::-1]) * 0.5
    a = _project_simplex(a)

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    best_a = a.copy()
    best_sc = _score(best_a)

    # parameters
    eta = 0.5
    eta_decay = 0.9995
    min_eta = 1e-5
    no_improve = 0
    target_score = 1.50274349
    iter_count = 0

    while time.time() < deadline:
        iter_count += 1
        # compute convolution
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # compute subgradient
        idxs = np.where(conv == max_val)[0]
        g = np.zeros_like(a)
        for k in idxs:
            # for j <= k: g[j] += 2 * a[k - j]
            # Equivalent: g[:k+1] += 2 * a[k::-1]
            g[:k+1] += 2 * a[k::-1]
        g /= len(idxs)

        # multiplicative update
        # clip gradient to some bound to avoid overflows
        grad_clip = np.clip(g, 0, 10)  # maybe clip
        a_new = a * np.exp(-eta * grad_clip)
        # renorm
        a_new_sum = a_new.sum()
        if a_new_sum == 0:
            a_new = _project_simplex(a)  # fallback
        else:
            a_new = a_new / a_new_sum
        # symmetrize
        a_new = (a_new + a_new[::-1]) * 0.5
        # ensure nonneg and sum=1 by projection
        a_new = _project_simplex(a_new)

        # evaluate score
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # acceptance: if improves accept, else sometimes accept using simulated annealing
        if sc < _score(a):
            a = a_new
        else:
            # probability decreasing with temperature (eta)
            # use temperature proportional to eta maybe
            prob = math.exp(-(sc - _score(a)) / (eta + 1e-12))
            if rng.random() < prob:
                a = a_new

        # decay eta
        eta = max(min_eta, eta * eta_decay)

        # occasional random perturbation if stuck
        if no_improve > 2000:
            # redistrib random mass
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j: continue
            delta = a[i] * rng.random() * 0.5
            a[i] -= delta
            a[j] += delta
            a = _project_simplex(a)
            a = (a + a[::-1]) * 0.5
            a = _project_simplex(a)
            no_improve = 0

        if best_sc < target_score:
            break

    # final evaluation using evaluate_sequence just to guarantee valid.
    # It expects list.
    return [float(x) for x in best_a]

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]

    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast score = 2*n*max(conv) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    vec = np.zeros(n, dtype=np.float64)
    vec[offset:offset + L] = w
