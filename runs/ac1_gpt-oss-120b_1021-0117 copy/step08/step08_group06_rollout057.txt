decay = 0.9995
min_temp_factor = 1e-5

iteration = 0

no_improve = 0

while time.time() < deadline:
    iteration+=1

    # compute FFT conv
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    max_val = conv.max()

    # temperature
    if temp is None:
        temp = max_val * 0.2
    else:
        temp = max(temp * decay, max_val * min_temp_factor)

    # compute softmax weights
    # to avoid overflow, subtract max_val
    w = np.exp((conv - max_val) / temp)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = (conv == max_val).astype(np.float64)
        w_sum = w.sum()
    w /= w_sum

    # gradient
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]

    # Adam update
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)

    # bias correction
    m_hat = m / (1.0 - beta1 ** iteration)
    v_hat = v / (1.0 - beta2 ** iteration)

    lr = lr0 / math.sqrt(iteration)  # decreasing lr
    step = lr * m_hat / (np.sqrt(v_hat) + eps)

    cand = _project_simplex(cur - step)

    cand_sc = _fast_score(cand, n, L)

    # acceptance
    if cand_sc < cur_sc:
        cur, cur_sc = cand, cand_sc
        no_improve = 0
    else:
        # maybe accept with probability
        prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
        if np.random.rand() < prob:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1

    # update best exact occasionally
    if iteration % 3000 == 0:
        true_sc = evaluate_sequence(cur.tolist())
        if true_sc < best_sc:

        perturb = rng.normal(scale=0.01, size=n)
        tmp = cur + perturb
        tmp = np.maximum(tmp, 0.0)
        tmp = _project_simplex(tmp)
        tmp_sc = _fast_score(tmp, n, L)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            no_improve = 0
            true_sc = evaluate_sequence(tmp.tolist())

import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Fast surrogate: 2*n*max(conv) assuming sum(a)=1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = _project_simplex(np.maximum(cur, 0.0))
        n = cur.size
    else:
        n = 2000
        cur = rng.random(n, dtype=np.float64)
        cur = _project_simplex(np.maximum(cur, 0.0))

    L = 1 << ((2 * n - 1).bit_length())   # next power of two for FFT

    best_vec = cur.copy()
    best_sc = evaluate_sequence(best_vec.tolist())

    cur_sc = _fast_score(cur, n, L)

    # Adam optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.8

    temp = None
    decay = 0.9995
    min_temp_factor = 1e-5

    it = 0
    no_improve = 0

    # main loop
    while time.time() < deadline:
        it += 1

        # ----- convolution & temperature -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.5
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # ----- soft‑max weights -----
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient (soft‑max) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step)
        cand_sc = _fast_score(cand, n, L)

        # ----- acceptance -----

            no_improve = 0
        else:

                cur, cur_sc = tmp, tmp_sc
                no_improve = 0

                if true_sc < best_sc:

            else:
                no_improve = 0

        # ----- symmetrisation -----

    except Exception:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
