def propose_candidate(seed=42,budget_s=1100,**kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)
    if 'height_sequence_1' in globals():
        init_full = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 2000
        init_full = rng.random(n)
    n = init_full.shape[0]
    cur0 = _project_simplex(np.maximum(init_full,0.0))
    cur0 = (cur0 + cur0[::-1]) * 0.5  # symmetrize
    cur0 = _project_simplex(cur0)    # reproject to ensure sum=1
    best_vec = cur0.copy()
    best_sc = _score(best_vec)
    # maybe uniform baseline
    uniform = np.full(n, 1.0/n)
    uni_sc = _score(uniform)
    if uni_sc < best_sc:
        best_sc = uni_sc
        best_vec = uniform.copy()
    # parameters
    L = 1 << ((2*n - 1).bit_length())
    lr0 = 0.2
    temp_factor = 0.07
    decay = 0.999
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # start main loop
    restart = 0
    while time.time() < deadline:
        # initialize current vector
        if restart == 0:
            cur = best_vec.copy()
        else:
            # random Dirichlet like
            cur = rng.exponential(scale=1.0,size=n)
            cur = _project_simplex(cur)
        # optionally symmetrize to reduce dimension
        cur = (cur + cur[::-1]) * 0.5
        cur = _project_simplex(cur)
        cur_sc = _score(cur)
        m = np.zeros_like(cur)
        v = np.zeros_like(cur)
        t = 0
        no_improve = 0
        temp = None
        while time.time() < deadline:
            t += 1
            # compute FFT and conv
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[:2*n-1]
            max_val = conv.max()
            # temperature
            if temp is None:
                temp = max_val * temp_factor
            else:
                temp = max(temp * decay, max_val * 1e-4)
            # weights
            w = np.exp((conv - max_val) / (temp + 1e-20))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
            w /= w_sum
            # gradient
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2*n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step = lr0 / (np.sqrt(v_hat) + eps)
            cand = cur - step * m_hat
            # enforce non-negativity via simplex projection
            cand = _project_simplex(cand)
            cand_sc = _score(cand)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
            else:
                no_improve += 1
                # occasional uphill acceptance
                prob = math.exp(-(cand_sc - cur_sc) / (lr0 + 1e-12))
                if rng.random() < prob:
                    cur, cur_sc = cand, cand_sc
            # random redistribution if stuck
            if no_improve > 2000:
                i = rng.integers(0,n)
                j = rng.integers(0,n-1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
            # occasional symmetrise
            if t % 500 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best_vec = sym_sc, sym.copy()
            # break if target reached
            if best_sc < 1.50274349:
                break
            # stop if time
            if time.time() >= deadline:
                break
        restart += 1
        if best_sc < 1.50274349:
            break
    # final perturbation local search
    cur = best_vec.copy()
    cur_sc = best_sc
    while time.time() < deadline:
        i = rng.integers(0,n)
        j = rng.integers(0,n-1)
        if j >= i:
            j += 1
        eps = cur[i] * rng.random() * 0.01
        if eps <= 0.0:
            continue
        tmp = cur.copy()
        tmp[i] -= eps
        tmp[j] += eps
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
    # return best vector as list of floats
    return [float(x) for x in best_vec]

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast objective for a vector whose sum is (approximately) 1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
