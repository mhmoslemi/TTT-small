        cur = np.maximum(cur, 0.0)
        cur = cur / cur.sum()  # project onto simplex (sum=1)

        cur = cur / cur.sum()

    L = 1 << ((2 * n - 1).bit_length())

    best_score = _score_norm1(best)
    cur_score = best_score

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    max_step = 1.0

    it = 0
    no_improve = 0
    while time.time() < deadline:
        it += 1
        # compute conv
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n - 1]
        max_val = conv.max()
        # indicator weights
        w = np.zeros_like(conv)
        max_idxs = np.where(conv == max_val)[0]
        w[max_idxs] = 1.0 / max_idxs.size
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # Clip step norm
        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        cand = _project_simplex(cur - step)
        cand_score = _score_norm1(cand)

        # Accept greedily or probabilistically if no improvement
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            # occasional random move
            no_improve += 1
            if no_improve % 1000 == 0:
                # perturb
                pert = cur + rng.normal(scale=0.01, size=n)
                pert = np.maximum(pert, 0.0)
                pert = _project_simplex(pert)
                pert_score = _score_norm1(pert)
                if pert_score < cur_score:
                    cur, cur_score = pert, pert_score
                    no_improve = 0
                    if pert_score < best_score:
                        best, best_score = pert.copy(), pert_score

        # symmetrization occasionally
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_norm1(sym)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score

    return [float(x) for x in best]

import time, math, numpy as np

def _project_simplex(v):
    """Project a non‑negative vector onto the probability simplex (sum=1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # shift to make all entries non‑negative
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

def _score_norm1(a):
    """Score for a vector already normalised to sum = 1: 2·n·max(conv)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
