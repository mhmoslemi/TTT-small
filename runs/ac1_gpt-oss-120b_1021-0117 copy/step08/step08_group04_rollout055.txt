import time
import numpy as np

def _project_simplex(v):
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as evaluated by the supplied `evaluate_sequence`.
    """
    start = time.time()
    # leave a small margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # Determine length n
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2500                     # a reasonable default if no parent is present

    # ----- candidate 1 : uniform distribution (sum = 1) -----------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = uniform
    best_score = evaluate_sequence(best_seq.tolist())

    # ----- candidate 2 : projected parent (if available) -----------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)          # enforce non‑negativity
        parent = _project_simplex(parent)         # enforce sum = 1
        parent_score = evaluate_sequence(parent.tolist())
        if parent_score < best_score:
            best_seq = parent
            best_score = parent_score

    # ----- lightweight local search ------------------------------------
    # a few very cheap random perturbations that keep us inside the simplex
    rng = np.random.default_rng(seed)
    max_iters = 2000
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # random redistributive move: pick two indices i!=j
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a small fraction of mass from i to j
        delta = best_seq[i] * rng.random() * 0.1
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        # re‑project onto simplex (fast using the same routine)
        cand = _project_simplex(cand)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score

    # Return the best sequence found as a plain Python list
    return [float(x) for x in best_seq]
