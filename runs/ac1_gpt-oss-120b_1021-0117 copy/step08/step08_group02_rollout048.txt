    L = 1 << ((2*n-1).bit_length())

    best_sc = _score(best_vec)
    cur_sc = best_sc

    # hyperparameters
    lr = 0.15
    min_lr = 1e-4
    decay_lr = 0.9995  # per iteration
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1_pow = 1.0
    beta2_pow = 1.0
    eps = 1e-8

    # temperature for softmax
    temp = None
    decay_temp = 0.998

    no_improve = 0
    it = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # compute conv via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

        # update temperature
        if temp is None:
            temp = max_val * 0.2
        else:
            temp = max(temp * decay_temp, max_val * 1e-6)

        # compute softmax weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast evaluation of 2*n*max(conv) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
