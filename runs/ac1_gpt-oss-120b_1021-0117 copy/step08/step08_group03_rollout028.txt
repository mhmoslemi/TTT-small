import time, math, numpy as np, cvxpy as cp

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _grad_max(a, conv, max_val, L):
    """Subgradient of max(conv) wrt a."""
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

def _solve_lp(cur, bound, t_mix=0.3):
    """LP: max sum(g) s.t. conv(cur,g) <= bound, g >= 0. Returns a mixed, simplex‑projected vector."""
    n = cur.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.conv(cp.Constant(cur), g) <= bound]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=3000,
                   abstol=1e-8, reltol=1e-6, feastol=1e-8)
    except Exception:
        return None
    if g.value is None:
        return None
    g0 = np.maximum(g.value, 0.0)
    s = g0.sum()
    if s <= 0:
        return None
    g_norm = g0 / s                     # stay on simplex
    return (1.0 - t_mix) * cur + t_mix * g_norm

            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = float(2.0 * n *
                           (np.fft.irfft(np.fft.rfft(tmp, n=L) ** 2, n=L)[:2 * n - 1].max()))
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0
            else:
                # fallback LP
                lp_vec = _solve_lp(cur, max_val, t_mix=0.4)
                if lp_vec is not None:
                    lp_sc = float(2.0 * n *
                                  (np.fft.irfft(np.fft.rfft(lp_vec, n=L) ** 2, n=L)[:2 * n - 1].max()))
                    if lp_sc < cur_sc:
                        cur, cur_sc = lp_vec, lp_sc
                        if lp_sc < best_sc:
                            best_vec, best_sc = lp_vec.copy(), lp_sc
                        no_improve = 0
                else:
                    # full restart
                    cur = _project_simplex(rng.exponential(scale=1.0, size=n))
                    momentum = np.zeros_like(cur)
                    no_improve = 0

        # ---- occasional symmetrisation ---------------------------------
        if it % 1200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = float(2.0 * n *
                           (np.fft.irfft(np.fft.rfft(sym, n=L) ** 2, n=L)[:2 * n - 1].max()))
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # early stop
        if best_sc < target:
            break

    # ---- return best found -----------------------------------------------
    return [float(x) for x in best_vec]
