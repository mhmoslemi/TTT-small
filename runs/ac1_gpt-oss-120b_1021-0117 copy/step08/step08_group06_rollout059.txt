def _subgrad(a):
    n = a.size
    L = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    max_val = conv.max()
    k = np.argmax(conv)  # first max
    # gradient
    grad = np.zeros_like(a)
    # we need grad[i] = 2 * a[k - i] if 0 <= k - i < n else 0
    # could compute via vectorized shift
    # for i = 0..n-1, j = k - i; if 0 <= j < n: grad[i] = 2*a[j]
    # Equivalent to grad = 2 * a[clip(k - np.arange(n),0,n-1?) but need mask.
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad[mask] = 2 * a[idx[mask]]
    return conv, max_val, grad
```

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

def _conv_fft(a, L):
    """Full linear convolution a*a using FFT (output length 2*len(a)-1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv

def _fast_score_from_conv(conv, n):
    """Fast surrogate of the true AC1 upper bound (sum is 1)."""
    return float(2.0 * n * conv.max())

def _descent_from(init, L, n, rng, deadline):
    """Projected (momentum) descent from a simplex start vector."""
    cur = _project_simplex(init)
    cur_sc = _fast_score_from_conv(_conv_fft(cur, L), n)
    # true score of the start (guaranteed finite because sum=1)
    best_vec = cur.copy()
    best_sc = evaluate_sequence(cur.tolist())
    momentum = np.zeros_like(cur)
    beta = 0.92
    step0 = 0.9
    it = 0
    no_improve = 0
    temp = None
    decay = 0.998
    min_temp_factor = 1e-4

    while time.time() < deadline:
        it += 1
        # ----- convolution and temperature -----
        af = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_val = conv.max()
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # ----- soft‑max weighting -----
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient from weighted max (smooth) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad_soft = 2.0 * grad_full[idx]

        # ----- sub‑gradient of the true max -----
        k = int(np.argmax(conv))
        grad_sub = np.zeros_like(cur)
        idx_sub = k - np.arange(n)
        mask = (idx_sub >= 0) & (idx_sub < n)
        grad_sub[mask] = 2.0 * cur[idx_sub[mask]]

        # ----- combined gradient and momentum -----
        grad = 0.6 * grad_soft + 0.4 * grad_sub
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- step size -----
        step = step0 / math.sqrt(it)

        # ----- try a few step scalings -----
        best_cand = None
        best_cand_sc = float('inf')
        for factor in (0.5, 1.0, 1.5):
            cand = _project_simplex(cur - step * factor * momentum)
            cand_sc = _fast_score_from_conv(_conv_fft(cand, L), n)
            if cand_sc < best_cand_sc:
                best_cand_sc = cand_sc
                best_cand = cand

        # ----- acceptance -----
        if best_cand_sc < cur_sc:
            cur, cur_sc = best_cand, best_cand_sc
            no_improve = 0
        else:
            prob = math.exp(-(best_cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = best_cand, best_cand_sc
                no_improve = 0
            else:
                no_improve += 1

        # ----- record best (exact) -----
        if cur_sc < best_sc:
            true_sc = evaluate_sequence(cur.tolist())
            if true_sc < best_sc:
                best_sc, best_vec = true_sc, cur.copy()

        # ----- random redistribution when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _fast_score_from_conv(_conv_fft(tmp, L), n)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
            true_sc = evaluate_sequence(tmp.tolist())
            if true_sc < best_sc:
                best_sc, best_vec = true_sc, tmp.copy()

        # ----- occasional symmetrisation -----
        if it % 800 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score_from_conv(_conv_fft(sym, L), n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                true_sc = evaluate_sequence(sym.tolist())
                if true_sc < best_sc:
                    best_sc, best_vec = true_sc, sym.copy()

        # ----- early exit if already excellent -----
        if best_sc < 1.45:   # well below historic target
            break

    return best_vec, best_sc


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ----- determine problem size -----
    if "height_sequence_1" in globals():
        base_seq = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        base_seq = rng.random(2000, dtype=np.float64)
    n = base_seq.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two FFT length

    # ----- generate several initial simplex vectors -----
    inits = []
    if "height_sequence_1" in globals():
        inits.append(_project_simplex(np.maximum(base_seq, 0.0)))
    inits.append(np.full(n, 1.0 / n, dtype=np.float64))   # uniform
    x = np.arange(n)
    c = (n - 1) / 2.0
    tri = 1.0 - np.abs(x - c) / c
    inits.append(_project_simplex(tri))
    sigma = n / 6.0
    gauss = np.exp(-0.5 * ((x - c) / sigma) ** 2)
    inits.append(_project_simplex(gauss))

    best_vec = None
    best_sc = float('inf')

    for init in inits:
        if time.time() >= deadline:
            break
        vec, sc = _descent_from(init, L, n, rng, deadline)
        if sc < best_sc:
            best_sc, best_vec = sc, vec
        if best_sc < 1.45:
            break

    # ----- fallback if nothing succeeded -----
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # ----- final sanity check -----
    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not math.isfinite(final_sc) or final_sc >= 1e9:
            best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    except Exception:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
