    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2*n-1]

    return float(2 * n * max_val / (sum_a * sum_a))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    # Adam variables
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    alpha = 0.1
    # baseline scoring
    best_vec = cur.copy()
    best_sc = _score_fft(best_vec, L)
    cur_sc = best_sc
    # counters
    it = 0
    no_improve = 0
    target = 1.50274349
    # main loop
    while time.time() < deadline:
        it += 1
        # compute FFT and conv
        af = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(af * af, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        # compute gradient via softmax smoothing
        beta = 2.0 + 0.001 * it  # increase slowly
        exps = np.exp(beta * (conv - max_val))
        w = exps / exps.sum()
        w_rev = w[::-1]
        wf = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * wf, n=L)
        idx = (2 * n - 2) - np.arange(n)
        gradient = 2.0 * grad_full[idx]
        # Adam update
        m = beta1 * m + (1 - beta1) * gradient
        v = beta2 * v + (1 - beta2) * (gradient * gradient)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score_fft(cand, L)
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            no_improve += 1
        # occasional random perturbation
        if no_improve > 600:
            # random small move
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.2
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_fft(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
            else:
                # LP rescue
                lp_vec = _solve_lp(cur, max_val, t_mix=0.12)
                if lp_vec is not None:
                    lp_sc = _score_fft(lp_vec, L)
                    if lp_sc < cur_sc:
                        cur, cur_sc = lp_vec, lp_sc
                        no_improve = 0
                        if lp_sc < best_sc:
                            best_sc, best_vec = lp_sc, lp_vec.copy()
        # occasional symmetry enforcement
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
        # early exit
        if best_sc < target:
            break
    # return best vector as list of floats
    return [float(x) for x in best_vec]

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _score_fft(a, L):
    """Fast AC1 score for a vector a (any positive scale)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s == 0:
        return float("inf")
    return float(2.0 * n * max_val / (s * s))
