import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    """Score = 2 * n * max(conv) for a (assumed sum(a)=1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # head‑room for cleanup

    # ------------------------------------------------------------------ #
    # problem size / initial vector
    if "height_sequence_1" in globals():
        init_full = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_full.shape[0]
    else:
        n = 2000
        init_full = rng.random(n, dtype=np.float64)

    cur = _project_simplex(np.maximum(init_full, 0.0))
    best = cur.copy()
    L = 1 << ((2 * n - 1).bit_length())
    best_sc = _score(best, L)

    # uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_sc = _score(uniform, L)
    if uni_sc < best_sc:
        best_sc, best = uni_sc, uniform.copy()

    # ------------------------------------------------------------------ #
    # hyper‑parameters
    beta = 0.95               # momentum decay
    step0 = 0.5               # initial step scale
    decay = 0.995             # temperature decay
    temp_factor = 0.07        # initial temperature as fraction of max conv
    min_temp_factor = 1e-4    # floor for temperature scaling
    max_no_improve = 4000     # trigger random shake
    sym_interval = 500        # symmetrisation frequency
    target = 1.50274349       # stop if reached

    momentum = np.zeros(n)
    cur_sc = best_sc
    it = 0
    no_improve = 0
    temp = None

    # ------------------------------------------------------------------ #
    # main optimisation loop
    while time.time() < deadline:
        it += 1

        # convolution & max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature schedule (soft‑max smoothing)
        if temp is None:
            temp = max_val * temp_factor
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # soft‑max weights for the max
        w = np.exp((conv - max_val) / (temp + 1e-20))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of weighted max (FFT‑based)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size
        step = step0 / math.sqrt(it)

        # ----------- additive candidate -----------------
        cand_add = _project_simplex(cur - step * momentum)
        cand_add_sc = _score(cand_add, L)

        # ----------- multiplicative candidate ----------
        grad_clipped = np.clip(grad, -20, 20)      # avoid overflow
        cand_mul = cur * np.exp(-step * grad_clipped)
        cand_mul = cand_mul / cand_mul.sum()      # simplex projection (all >0)
        cand_mul_sc = _score(cand_mul, L)

        # choose the better of the two
        if cand_add_sc < cand_mul_sc:
            cand, cand_sc = cand_add, cand_add_sc
        else:
            cand, cand_sc = cand_mul, cand_mul_sc

        # acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best = cand_sc, cand.copy()
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # random shake when stuck
        if no_improve > max_no_improve:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best = tmp_sc, tmp.copy()
            else:
                no_improve = 0

        # periodic symmetrisation (helps because the problem is symmetric)
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best = sym_sc, sym.copy()

        # early stop
        if best_sc < target:
            break

    # ------------------------------------------------------------------ #
    # final fine‑grained hill climbing
    cur = best.copy()
    cur_sc = best_sc
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        eps = cur[i] * rng.random() * 0.01
        if eps <= 0.0:
            continue
        tmp = cur.copy()
        tmp[i] -= eps
        tmp[j] += eps
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp, L)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best = tmp_sc, tmp.copy()
        else:
            if rng.random() < 0.001:          # rare uphill move
                cur, cur_sc = tmp, tmp_sc

    # return plain‑python list
    return [float(x) for x in best]
