import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]                     # descending sort
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_fast(a, n, L, len_w):
    """Efficient evaluation of 2*n*max(convolve(a,a))."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:len_w]
    return float(2.0 * n * conv.max())
