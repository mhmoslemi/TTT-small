def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
    else:
        cur = rng.random(2000, dtype=np.float64)

    cur = _project_simplex(cur)

    n = cur.shape[0]
    L = 1 << ((2*n-1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 1.0
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # compute convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # gradient of max
        grad = _grad_max(cur, conv, max_val, L)  # gradient of max conv

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 * (m_hat / (np.sqrt(v_hat) + eps))
        # gradient descent step (minimize)
        cand = _project_simplex(cur - step)
        cand_sc = _score(cand, L)

        improved = False
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            improved = True

        # also try random move if not improved
        if not improved:
            # random move: transfer mass from i to j
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            delta_max = cur[i] * 0.5  # transfer up to 50% of source
            delta = rng.random() * delta_max
            if delta > 0:
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                # projection not needed because sum unchanged, but ensure nonneg
                if tmp[i] >= 0:
                    tmp_sc = _score(tmp, L)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        improved = True

        # try split move occasionally
        if not improved and (it % 500 == 0):
            idx = np.argmax(cur)
            if cur[idx] > 0:
                delta = cur[idx] * 0.25  # split 25%
                left = idx - 1
                right = idx + 1
                tmp = cur.copy()
                tmp[idx] -= delta
                if left >= 0:

                if right < n:
                    tmp[right] += delta / 2
                # if neighbor missing, give whole to the other side
                if left < 0:

                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    improved = True

        if cur_sc < best_sc:

import time, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L):
    """Objective value for a vector a that sums to 1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _grad_max(a, conv, max_val, L):
    """Subgradient of max(conv) wrt a (a is real, non‑negative)."""
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ---- initial vector --------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(cur)          # sum = 1, non‑negative

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # power‑of‑2 FFT length

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # ---- Adam optimizer state -------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 1.0

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # convolution and max value
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # gradient of max(conv)
        grad = _grad_max(cur, conv, max_val, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step_scale = step0 / np.sqrt(it)
        step = step_scale * (m_hat / (np.sqrt(v_hat) + eps))

        # gradient descent candidate
        cand = _project_simplex(cur - step)
        cand_sc = _score(cand, L)

        improved = False
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            improved = True

        # random mass transfer if gradient step did not improve
        if not improved:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5   # up to 50% of source mass
            if delta > 0.0:
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                if tmp[i] >= 0.0:   # preserve non‑negativity
                    tmp_sc = _score(tmp, L)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        improved = True

        # occasional splitting of the biggest coefficient
        if not improved and (it % 500 == 0):
            idx = int(np.argmax(cur))
            if cur[idx] > 0.0:
                delta = cur[idx] * 0.25      # split 25%
                left, right = idx - 1, idx + 1
                tmp = cur.copy()
                tmp[idx] -= delta
                share = delta / 2.0
                if left >= 0:
                    tmp[left] += share
                else:
                    tmp[right] += share
                if right < n:
                    tmp[right] += share
                else:
                    tmp[left] += share
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    improved = True

        # occasional symmetrisation
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                improved = True

        # record best
        if cur_sc < best_sc:
            best_vec, best_sc = cur.copy(), cur_sc
            no_improve = 0
        else:
            no_improve += 1

        # restart near best if stuck too long
        if no_improve > 5000:
            cur = best_vec + 0.01 * rng.normal(size=n)
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            cur_sc = _score(cur, L)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # early exit if target achieved
        if best_sc < target:
            break

    # Return the best found sequence as plain Python floats
    return [float(x) for x in best_vec]
