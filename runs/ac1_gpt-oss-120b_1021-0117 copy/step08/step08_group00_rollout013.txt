import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_fft(a, L):
    """Fast AC1 score for a vector with sum(a)=1 using FFT."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    it = 0
    no_improve = 0

    # ---- main loop ------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution and its maximum (FFT‑based)
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # temperature schedule (used for occasional uphill moves)
        if it == 1:
            temp = max_val * temp_factor
        else:
            temp = max(temp * decay_temp, max_val * min_temp_factor)

        # mask of indices attaining the maximum (hard max)
        mask = (conv >= max_val - 1e-12).astype(np.float64)

        # gradient of the max‑convolution (FFT‑based)
        w_rev = mask[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # adaptive step size
        step = step0 / math.sqrt(it)

        # candidate after projected gradient step
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score_fft(cand, L)

        # acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ---- diversification when stuck ---------------------------------
        if no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_fft(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # periodic symmetrisation (helps to flatten peaks)
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # occasional sparsification (zero out small entries)
        if it % 1200 == 0:
            thresh = np.quantile(cur, 0.2)   # prune 20 % smallest
            tmp = cur.copy()
            tmp[tmp < thresh] = 0.0
            tmp = _project_simplex(tmp)
            tmp_sc = _score_fft(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # early exit if target reached
        if best_sc < target:
            break

    # ---- final light local search (uses remaining time) -----------------
    while time.time() < deadline:
        i = rng.integers(0, n)
        delta = best_vec[i] * rng.random() * 0.05
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        tmp = best_vec.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score_fft(tmp, L)
        if tmp_sc < best_sc:
            best_vec, best_sc = tmp.copy(), tmp_sc

    # return the best sequence found as plain Python floats
    return [float(x) for x in best_vec]
