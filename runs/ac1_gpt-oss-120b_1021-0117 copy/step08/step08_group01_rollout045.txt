import time, numpy as np, math

def _project_simplex(v):
   # typical algorithm for simplex sum=1
   v = np.asarray(v, dtype=np.float64)
   n = v.shape[0]
   # sort descending
   u = np.sort(v)[::-1]
   css = np.cumsum(u) - 1
   rho = np.where(u - css/np.arange(1, n+1) > 0)[0]
   if rho.size == 0:
       theta=0.0
   else:
       rho = rho[-1]
       theta = css[rho]/(rho+1)
   w = v - theta
   w[w<0]=0.0
   return w

def _score_fft(a, L, n):
   af = np.fft.rfft(a, n=L)
   conv = np.fft.irfft(af * af, n=L)[:2*n-1]
   max_val = conv.max()
   return 2.0 * n * max_val, max_val, conv, af

    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_and_data(a, L, n):
    """Return (score, max_convolution, full convolution, rfft of a)."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    score = 2.0 * n * max_val
    return score, max_val, conv, af

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------ #
    # Determine problem size and initial vector
    if "height_sequence_1" in globals():
        init_full = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 2000
        init_full = rng.random(n, dtype=np.float64)

    n = init_full.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑2 for FFT speed

    # Start from a projected, symmetrised version of the provided sequence
    cur0 = _project_simplex(np.maximum(init_full, 0.0))
    cur0 = (cur0 + cur0[::-1]) * 0.5
    cur0 = _project_simplex(cur0)

    cur = cur0.copy()
    cur_sc, cur_max, cur_conv, cur_fft = _score_and_data(cur, L, n)

    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------------ #
    # Adam optimiser parameters (matrix‑free)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.5                     # base learning‑rate
    m = np.zeros_like(cur)       # first moment
    v = np.zeros_like(cur)       # second moment
    it = 0

    # Soft‑max temperature schedule for the gradient of max‑conv
    temp = None
    temp_factor = 0.07
    min_temp_factor = 1e-4
    decay = 0.997

    target = 1.50274349

    # ------------------------------------------------------------------ #
    while time.time() < deadline:
        it += 1

        # ----- convolution & temperature ---------------------------------
        cur_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * temp_factor
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # ----- soft‑max weighting ----------------------------------------
        w = np.exp((conv - max_val) / (temp + 1e-20))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of weighted max (FFT‑based) ----------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr / (np.sqrt(v_hat) + eps)

        cand = cur - step * m_hat
        cand = _project_simplex(cand)
        cand = (cand + cand[::-1]) * 0.5          # enforce symmetry
        cand = _project_simplex(cand)

        cand_sc, _, _, _ = _score_and_data(cand, L, n)

        # ----- accept / occasional uphill move ---------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ----- occasional random restart if stuck -----------------------
        if it % 5000 == 0:
            rnd = rng.exponential(scale=1.0, size=n)
            rnd = _project_simplex(rnd)
            rnd = (rnd + rnd[::-1]) * 0.5
            rnd = _project_simplex(rnd)
            rnd_sc, _, _, _ = _score_and_data(rnd, L, n)
            if rnd_sc < cur_sc:
                cur, cur_sc = rnd, rnd_sc

        # ----- early exit ------------------------------------------------
        if best_sc < target:
            break

    # ------------------------------------------------------------------ #
    # Final fine‑grained hill climbing (small deterministic swaps)
    cur = best_vec.copy()
    cur_sc = best_sc
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        eps_move = cur[i] * rng.random() * 0.02
        if eps_move <= 0.0:
            continue
        tmp = cur.copy()
        tmp[i] -= eps_move
        tmp[j] += eps_move
        if tmp[i] < 0.0:
            continue
        tmp = _project_simplex(tmp)
        tmp = (tmp + tmp[::-1]) * 0.5
        tmp = _project_simplex(tmp)
        tmp_sc, _, _, _ = _score_and_data(tmp, L, n)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
            if best_sc < target:
                break

    # ------------------------------------------------------------------ #
    return [float(x) for x in best_vec]
