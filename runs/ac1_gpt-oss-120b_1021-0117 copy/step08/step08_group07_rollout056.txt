import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # initialise – use the provided construction if present
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())  # next power of two >=2n-1

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.5          # base learning rate
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t = 0

    # initial score
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    best_sc = float(2 * n * conv_full.max())
    best_vec = cur.copy()
    cur_sc = best_sc

    # bookkeeping
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.time() >= deadline:
            break

        # convolution and max location
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = float(conv_full.max())
        max_idx = int(conv_full.argmax())

        # sub‑gradient of the max‑convolution term
        grad = np.zeros_like(cur)
        low = max(0, max_idx - (n - 1))
        high = min(n - 1, max_idx)
        if low <= high:

            grad[idx] = 2.0 * cur[max_idx - idx]

        # Adam update
