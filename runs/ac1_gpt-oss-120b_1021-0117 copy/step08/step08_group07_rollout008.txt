import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _objective(a, L):
    """Return 2·n·max(conv(a,a)) / (sum(a)^2)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_val / (s * s))

    target = 1.50274349   # objective threshold to stop early

    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- current convolution & subgradient of max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # indices attaining (or nearly attaining) the maximum
        eps = max_val * 1e-14 + 1e-18
        idx_max = np.where(conv >= max_val - eps)[0]

        # weight vector: uniform on max positions (exact subgradient of max)
        w = np.zeros_like(conv)
        if idx_max.size:
            w[idx_max] = 1.0 / idx_max.size
        else:                     # safety fallback
            w[np.argmax(conv)] = 1.0

        # gradient of weighted sum = subgradient of max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- momentum and step size -----
        momentum = beta * momentum + (1.0 - beta) * grad
        step = base_step / math.sqrt(it + 1.0)

        # ----- line‑search over a few step factors -----
        best_cand = None
        best_cand_obj = float('inf')
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = cur - step * factor * momentum
            cand = _project_simplex(cand)
            cand_obj = _objective(cand, L)
            if cand_obj < best_cand_obj:
                best_cand_obj = cand_obj
                best_cand = cand

        # ----- acceptance -----
        cur_obj = 2.0 * n * max_val      # sum(cur)=1 after projection
        if best_cand_obj < cur_obj:
            cur = best_cand
            cur_obj = best_cand_obj
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(best_cand_obj - cur_obj) / (step + 1e-12))
            if rng.random() < prob:
                cur = best_cand
                cur_obj = best_cand_obj
                no_improve = 0

        # ----- update global best -----
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_vec = cur.copy()

        # ----- stagnation heuristics -----
        if no_improve > 2000:
            # small random mass transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_obj = _objective(tmp, L)
            if tmp_obj < best_obj:
                best_obj = tmp_obj
                best_vec = tmp.copy()
            if tmp_obj < cur_obj:
                cur = tmp
                cur_obj = tmp_obj
                no_improve = 0

        # occasional symmetrisation (helps for many optima)
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_obj = _objective(sym, L)
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_vec = sym.copy()
            if sym_obj < cur_obj:
                cur = sym
                cur_obj = sym_obj
                no_improve = 0

        if best_obj < target:
            break

    # ------------------------------------------------------------------
    # final polishing with tiny Gaussian tweaks
    while time.time() < deadline:
        pert = best_vec + rng.normal(scale=0.001, size=n)
        pert = np.clip(pert, 0.0, None)
        pert = _project_simplex(pert)
        pert_obj = _objective(pert, L)
        if pert_obj < best_obj:
            best_obj = pert_obj
            best_vec = pert.copy()
            if best_obj < target:
                break

    # Return Python list of floats
    return [float(x) for x in best_vec]
