import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_fft(a, n, L):
    """Return 2*n*max(a*a) using an FFT of length L (a is assumed non‑negative and sum≈1)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

                    cur, cur_sc = tmp, tmp_sc

        # ---- periodic sparsification (keep only larger entries) -------
        if it % 2500 == 0:
            thresh = np.quantile(cur, 0.15)     # prune the smallest 15 %
            tmp = cur.copy()
            tmp[tmp < thresh] = 0.0
            tmp = _project_simplex(tmp)
            tmp_sc = _score_fft(tmp, n, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc

    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
