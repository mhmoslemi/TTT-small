import time, math, numpy as np

def _largest_prime_leq(k: int) -> int:
    """Return the largest prime ≤ k (k ≥ 2). Simple trial division works for k≈50."""
    for p in range(k, 1, -1):
        if p % 2 == 0:
            if p == 2:
                return 2
            continue
        is_prime = True
        r = int(math.isqrt(p))
        for d in range(3, r + 1, 2):
            if p % d == 0:
                is_prime = False
                break
        if is_prime:
            return p
    return 2  # fallback, never reached for k≥2

def _sidon_set_prime(p: int) -> list[int]:
    """
    Classic Singer construction:
    S = { i * p + (i*i) % p : i = 0..p-1 } is a Sidon set of size p
    contained in [0, p*p-1].
    """
    return [i * p + (i * i) % p for i in range(p)]
