import time, math, numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex (sum = 1, x >= 0)
def _project_simplex(v):
    """Project vector v onto the simplex {x>=0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# ----------------------------------------------------------------------
# Fast objective (sum of coefficients is 1 → objective = 2 n max(conv))
def _objective(a, L):
    """Return 2·n·max(a*a) assuming sum(a)=1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    # ------------------------------------------------------------------
    # initialise – use provided parent if available
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fall back: random vector of length 2000
        cur = rng.random(2000, dtype=np.float64)
    # make non‑negative, clamp to allowed range, then project to simplex
    cur = np.clip(cur, 0.0, None)
    cur = _project_simplex(cur)

    n = cur.size
    # FFT length – next power of two >= 2n-1
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _objective(best_vec, L)
    cur_sc = best_sc

    # optimisation state
    momentum = np.zeros_like(cur)
    beta = 0.9                # momentum decay
    base_step = 0.8           # initial step magnitude
    it = 0
    no_improve = 0
    TARGET = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # ----- convolution & mask for max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # tolerance for “near‑max” indices (hard max)
        tol = max_val * 1e-8 + 1e-12
        mask = conv >= (max_val - tol)
        w = mask.astype(np.float64)

            w[conv.argmax()] = 1.0
        w /= w.sum()
