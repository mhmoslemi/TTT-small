import time, math, numpy as np

def _project_simplex(v):
    """Projection onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_norm1(a):
    """Assumes sum(a)=1; returns 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]

        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum

        # gradient of weighted max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size
        lr = lr0 / math.sqrt(it)
        step = -lr * momentum
        step_norm = np.linalg.norm(step)
        if step_norm > max_step_norm:
            step = step * (max_step_norm / step_norm)

        cand = cur + step
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_score = _score_norm1(cand)

        # accept if better or by probability
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / (temp + 1e-12)):
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            no_improve += 1

        # occasional random redistribution when stuck
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp = _project_simplex(tmp)
            tmp_score = _score_norm1(tmp)
            if tmp_score < cur_score:
                cur, cur_score = tmp, tmp_score
                no_improve = 0
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score

        # restart with small Gaussian noise if really stuck
        if no_improve > 8000:
            cur = best + rng.normal(scale=0.01, size=n)
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)
            cur_score = _score_norm1(cur)
            momentum.fill(0.0)
            no_improve = 0

        if best_score <= target:
            break

    # Return the best-found sequence as a plain list of floats
    return [float(x) for x in best]
