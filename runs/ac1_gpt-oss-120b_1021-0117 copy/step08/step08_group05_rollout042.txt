import time, math, numpy as np

def _project_simplex(v):
    """Project onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n):
    """Score = 2*n*max(conv(a,a)) for a with sum=1."""
    a = np.asarray(a, dtype=np.float64)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ---- initialization ----
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)            # enforce sum=1
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())    # FFT length (power of two)
    idx = (2 * n - 2) - np.arange(n)       # gradient slice indices

    # Adam state
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-12
    alpha_base = 0.12                      # base LR (larger than before)
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    best_vec = cur.copy()
    best_sc = _score_simplex(best_vec, n)
    cur_sc = best_sc

    T = None                               # temperature for soft‑max
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:

        conv_full = np.fft.irfft(a_fft * a_fft, n=L)

        max_conv = conv.max()

        # ---- temperature schedule (soft‑max) ----
        if T is None:
