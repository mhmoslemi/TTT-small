def _project_simplex(v):
   # same as earlier (fast)
   v = np.asarray(v, dtype=np.float64)  # ensure float
   n = v.size
   u = np.sort(v)[::-1]
   css = np.cumsum(u) - 1
   rho = np.where(u - css / np.arange(1, n+1) > 0)[0]

   else:

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)

    css = np.cumsum(u) - 1.0

        theta = css[rho] / (rho + 1)
    w = v - theta

    return w
