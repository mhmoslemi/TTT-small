import time, math, numpy as np

# ---------- helper functions ----------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n):
    """Score for a vector a with sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    while time.time() < deadline:
        it += 1

        # ----- convolution of current vector -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_c = conv.max()

        # ----- temperature schedule (soft‑max) -----
        if T is None:
            T = max_c * 0.15                     # start fairly smooth
        else:
            T = max(T * 0.997, max_c * 1e-5)     # slow exponential decay

        # ----- weights over convolution entries -----
        w = np.exp((conv - max_c) / T)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            # fallback to uniform weight on all active max entries
            mask = (conv == max_c)
            w = mask.astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of (smoothed) max convolution -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update (bias‑corrected) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + epsilon)

        # ----- candidate generation -----
        cand = cur - step
        # small isotropic noise to aid escape from plateaux
        noise_scale = 0.001 * (np.linalg.norm(step) / np.sqrt(n) + 1e-12)
        cand += rng.normal(scale=noise_scale, size=n)
        # enforce symmetry and simplex feasibility
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_sc = _score_simplex(cand, n)

        # ----- accept / reject (with mild uphill probability) -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            # probability to accept a worse move (simulated annealing)
            prob = math.exp(-(cand_sc - cur_sc) / (max_c * 1e-4 + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- random redistribution when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.25
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp = _project_simplex(tmp)
            tmp_sc = _score_simplex(tmp, n)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
            else:
                no_improve = 0  # give it another chance

        # ----- occasional symmetrisation (already enforced each step) -----
        if it % 1500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_simplex(sym, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ----- global perturbation to escape deep local minima -----
        if it % 5000 == 0:
            # re‑initialise around the best found so far
            cur = best_vec + rng.normal(scale=0.02, size=n)
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(cur)
            cur_sc = _score_simplex(cur, n)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # return the best-found sequence as plain Python floats
    return [float(x) for x in best_vec]
