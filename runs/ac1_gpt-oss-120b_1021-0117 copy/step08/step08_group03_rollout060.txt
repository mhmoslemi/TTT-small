import time, numpy as np, math

def _project_simplex(v):
    # Euclidean projection onto simplex sum=1
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.nonzero(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _smoothmax_and_grad(a, alpha, L_conv, L_grad):
    """Return smoothmax value (not used) and gradient of smoothmax w.r.t a."""
    # a: vector length n
    n = a.shape[0]
    # FFT for convolution
    af = np.fft.rfft(a, n=L_conv)
    conv_full = np.fft.irfft(af * af, n=L_conv)
    conv = conv_full[: 2 * n - 1]
    # compute softmax weights
    # subtract max to improve stability
    maxc = conv.max()
    expv = np.exp(alpha * (conv - maxc))
    w = expv / expv.sum()
    # gradients via convolution with w
    a_rev = a[::-1]
    af_rev = np.fft.rfft(a_rev, n=L_grad)

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex (sum=1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.nonzero(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    """Fast evaluation of 2·n·max(conv(a,a)) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _smoothmax_and_grad(a, alpha, Lc, Lg):
    """
    Smooth approximation of the max of the convolution and its gradient.
    Returns (max_conv, gradient) where gradient has the same shape as a.
    """
    n = a.shape[0]

    # --- convolution (linear) ---
    af = np.fft.rfft(a, n=Lc)
    conv_full = np.fft.irfft(af * af, n=Lc)
    conv = conv_full[: 2 * n - 1]

    # --- softmax weights ---
    maxc = conv.max()
    expv = np.exp(alpha * (conv - maxc))          # shift for stability
    w = expv / expv.sum()                         # shape (2n-1,)

    # --- gradient via two convolutions ---
    a_rev = a[::-1]
    af_rev = np.fft.rfft(a_rev, n=Lg)
    wf = np.fft.rfft(w, n=Lg)

    # a_rev * w
    g1_full = np.fft.irfft(af_rev * wf, n=Lg)
    # a * w
    af_full = np.fft.rfft(a, n=Lg)
    g2_full = np.fft.irfft(af_full * wf, n=Lg)

    # appropriate slices (see derivation in analysis)
    g1 = g1_full[n - 1 : n - 1 + n]   # length n
    g2 = g2_full[:n]                  # length n
    grad = g1 + g2
    return conv.max(), grad
