def propose_candidate(seed=42, budget_s=1100):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 0.2)
    # Set target threshold (maybe not needed)
    target = 1.50274349

    # Determine length n.
    if "height_sequence_1" in globals():
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_seq.size
        # ensure nonnegative and sum=1 by projection on simplex
        cur0 = _project_simplex(np.maximum(init_seq, 0.0))
    else:
        n = 2000 # fallback
        cur0 = rng.random(n)
        cur0 = _project_simplex(cur0)

    # Precompute FFT length L.
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = None
    best_score = np.inf

    # We'll perform several restarts
    restarts = 3
    restart_idx = 0

    while time.time() < deadline and restart_idx < restarts:
        # initialization per restart
        if restart_idx == 0:
            cur = cur0.copy()
        else:
            # Random init on simplex
            cur = rng.random(n)
            cur = _project_simplex(cur)
        # optimizer state
        m = np.zeros_like(cur)
        v = np.zeros_like(cur)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr = 0.2   # base learning rate
        temp = None
        decay_temp = 0.999
        min_temp_fac = 1e-4

        # iteration loop
        it = 0
        while time.time() < deadline:
            it += 1
            # compute convolution
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()
            # score
            cur_score = 2.0 * n * max_val

            # track best
            if cur_score < best_score:
                best_score = cur_score
                best_vec = cur.copy()
                # if reached target early exit
                if best_score < target:
                    break

            # temperature for softmax
            if temp is None:
                temp = max_val * 0.07
            else:
                temp = max(temp * decay_temp, max_val * min_temp_fac)

            # softmax weights
            w = np.exp((conv - max_val) / temp)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
            w = w / w_sum

            # gradient
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = cur - step
            # ensure nonnegative and project onto simplex
            cand = _project_simplex(cand)

            cur = cand

            # occasional random perturbation if stagnating
            if it % 1000 == 0:
                # random swap mass
                i = rng.integers(n)
                j = rng.integers(n-1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.2
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                cur = tmp

        restart_idx += 1

    # final polishing: random small Gaussian noise on best_vec
    if best_vec is None:
        # fallback uniform
        best_vec = np.full(n, 1.0/n)
    while time.time() < deadline:
        pert = best_vec + rng.normal(scale=0.003, size=n)
        pert = np.clip(pert, 0.0, None)
        pert = _project_simplex(pert)
        # compute score quickly via conv
        a_fft = np.fft.rfft(pert, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        max_val = conv_full[:2*n-1].max()
        sc = 2.0 * n * max_val
        if sc < best_score:
            best_score = sc
            best_vec = pert.copy()
            if best_score < target:
                break

    # return list of floats
    return [float(x) for x in best_vec]

import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # initialise – use the provided construction if present
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
        init = np.maximum(init, 0.0)
        cur0 = _project_simplex(init)
        n = cur0.size
    else:
        n = 2000
        cur0 = _project_simplex(rng.random(n))

    # FFT length (power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    best_vec = None
    best_score = np.inf
    target = 1.50274349

    # optimisation parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.05
    decay_temp = 0.999
    min_temp_fac = 1e-4

    # ------------------------------------------------------------------
    # Multiple short restarts – first uses the parent construction
    restarts = 3
    for r in range(restarts):
        if time.time() >= deadline:
            break
        if r == 0:
            cur = cur0.copy()
        else:
            cur = _project_simplex(rng.random(n))

        m = np.zeros_like(cur)
        v = np.zeros_like(cur)
        temp = None
        it = 0
        max_iter = 5000

        while time.time() < deadline and it < max_iter:
            it += 1

            # ----- convolution & current score -------------------------
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()
            cur_score = 2.0 * n * max_val

            # track global best
            if cur_score < best_score:
                best_score = cur_score
                best_vec = cur.copy()
                if best_score < target:
                    break

            # ----- temperature & soft‑max weights --------------------
            if temp is None:
                temp = max_val * 0.07
            else:
                temp = max(temp * decay_temp, max_val * min_temp_fac)

            w = np.exp((conv - max_val) / temp)
            w_sum = w.sum()
            if w_sum == 0.0 or not np.isfinite(w_sum):
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
            w /= w_sum

            # ----- gradient of the smoothed max ----------------------
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)

            best_vec = tmp.copy()
            if best_score < target:

    return [float(x) for x in best_vec]
