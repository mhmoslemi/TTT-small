def _subgradient_max(a, k):
    n = a.shape[0]
    g = np.zeros_like(a)
    if k < n:
        # indices 0..k
        g[:k+1] = 2 * a[k::-1]
    else:

        # indices offset..n-1
        # slice of a from n-1 down to offset
        g[offset:] = 2 * a[n-1:offset-1:-1]
    return g

import time, math, numpy as np

def _project_simplex(v):
    """Projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _subgradient(a, k):
    """Sub‑gradient of max‑convolution at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if k < n:
        # i = 0..k  ->  g[i] = 2 * a[k-i]
        g[:k + 1] = 2.0 * a[k::-1]
    else:
        offset = k - (n - 1)          # offset > 0
        # i = offset..n-1  ->  g[i] = 2 * a[n-1-(i-offset)]
        g[offset:] = 2.0 * a[n - 1:offset - 1:-1]
    return g

def _interleave_sort(v):
    """Place largest values at opposite ends, alternating."""
    vals = np.sort(v)[::-1]           # descending
    n = v.size
    out = np.empty_like(v)
    left, right = 0, n - 1
    toggle = True
    for val in vals:
        if toggle:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
        toggle = not toggle
    return out

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())       # power‑of‑two length for FFT

    # ----- scoring helper -------------------------------------------------
    def _score_vec(v):
        a_fft = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_idx = int(conv.argmax())
        max_val = float(conv[max_idx])
        return float(2.0 * n * max_val), max_idx, max_val

    cur_score, cur_max_idx, cur_max_val = _score_vec(cur)
    best_vec = cur.copy()
    best_score = cur_score

    # ----- Adam parameters ------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    alpha = 0.2                               # base step size
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    it = 0
    no_improve = 0
    stage = "grad"                            # two phases: gradient then swaps
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # ----- switch to swap stage after half the budget ---------------
        if stage == "grad" and (time.time() - start) > (budget_s - 0.3) * 0.5:
            stage = "swap"
            continue

        if stage == "grad":
            # ----- convolution and gradient -----------------------------
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
            k = int(conv_full.argmax())
            max_val = float(conv_full[k])
            cur_score = float(2.0 * n * max_val)

            g = _subgradient(cur, k)

            # ----- Adam update -----------------------------------------
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = (alpha / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + epsilon)

            cand = cur - step
            cand = _project_simplex(cand)

            cand_score, _, _ = _score_vec(cand)

            # ----- accept if improvement --------------------------------
            if cand_score < cur_score:
                cur, cur_score = cand, cand_score
                no_improve = 0
                if cand_score < best_score:
                    best_score, best_vec = cand_score, cand.copy()
                    if best_score < target:
                        break
            else:
                no_improve += 1
                T = max_val * 0.02
                if rng.random() < math.exp(-(cand_score - cur_score) / (T + 1e-12)):
                    cur, cur_score = cand, cand_score

            # ----- occasional random mass redistribution -----------------
            if no_improve > 2000:
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.3
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score, _, _ = _score_vec(tmp)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    no_improve = 0
                    if tmp_score < best_score:
                        best_score, best_vec = tmp_score, tmp.copy()
                        if best_score < target:
                            break

            # ----- symmetrisation ---------------------------------------
            if it % 500 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_score, _, _ = _score_vec(sym)
                if sym_score < cur_score:
                    cur, cur_score = sym, sym_score
                    if sym_score < best_score:
                        best_score, best_vec = sym_score, sym.copy()
                        if best_score < target:
                            break

            # ----- interleaved ordering ---------------------------------
            if it % 1000 == 0:
                reordered = _interleave_sort(cur)
                reordered = _project_simplex(reordered)
                reo_score, _, _ = _score_vec(reordered)
                if reo_score < cur_score:
                    cur, cur_score = reordered, reo_score
                    if reo_score < best_score:
                        best_score, best_vec = reo_score, reordered.copy()
                        if best_score < target:
                            break

        else:  # ----- swap‑based local search ----------------------------
            i, j = rng.integers(0, n, size=2)
            if i != j:
                cand = cur.copy()
                cand[i], cand[j] = cand[j], cand[i]          # swap values
                cand_score, _, _ = _score_vec(cand)
                if cand_score < cur_score:
                    cur, cur_score = cand, cand_score
                    if cand_score < best_score:
                        best_score, best_vec = cand_score, cand.copy()
                        if best_score < target:
                            break

            # occasional small mass move
            if rng.random() < 0.01:
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.2
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score, _, _ = _score_vec(tmp)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    if tmp_score < best_score:
                        best_score, best_vec = tmp_score, tmp.copy()
                        if best_score < target:
                            break

            # occasional symmetrisation (helps in later phase as well)
            if it % 500 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_score, _, _ = _score_vec(sym)
                if sym_score < cur_score:
                    cur, cur_score = sym, sym_score
                    if sym_score < best_score:
                        best_score, best_vec = sym_score, sym.copy()
                        if best_score < target:
                            break

    # Return the best found vector as a plain Python list
    return [float(x) for x in best_vec]
