def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # small headroom

    # Initialize sequence
    if "height_sequence_1" in globals():
        cur = np.maximum(np.asarray(height_sequence_1, dtype=np.float64), 0.0)
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur_sum = cur.sum()
    if cur_sum < 1e-12:
        cur = np.full_like(cur, 1.0)
        cur_sum = cur.sum()
    cur = cur / cur_sum   # sum = 1

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())  # FFT length power of two

    def score(vec):
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2 * n - 1]
        return float(2.0 * n * conv.max())
    cur_sc = score(cur)

    best_vec = cur.copy()
    best_sc = cur_sc

    # Temperature init
    # compute initial conv max via FFT
    af = np.fft.rfft(cur, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    max_conv = conv.max()
    temp = max_conv * 0.1
    min_temp = max_conv * 1e-8
    decay = 0.9995

    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1

        # occasional targeted move every 200 iterations
        if it % 200 == 0:
            # compute conv and max index
            af = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(af * af, n=L)[:2*n-1]
            max_idx = int(conv_full.argmax())
            # compute contribution of each coordinate to max index
            # contribution: a[i] * a[max_idx - i] if index in range
            contrib = np.zeros(n, dtype=np.float64)
            # valid i where j = max_idx - i in [0, n-1]
            lo = max(0, max_idx - (n-1))
            hi = min(n-1, max_idx)
            # indices i in [lo, hi]
            i_range = np.arange(lo, hi+1)
            j_range = max_idx - i_range
            contrib[i_range] = cur[i_range] * cur[j_range]
            # select source from top 5 contributions, target from bottom 5
            if contrib.max() > 0:
                src_idx = np.argpartition(-contrib, 5)[:5]  # top 5
                tgt_idx = np.argpartition(contrib, 5)[:5]   # bottom 5
                # pick random src and tgt distinct
                s = rng.choice(src_idx)
                t = rng.choice(tgt_idx)
                if s == t:
                    # pick different random index
                    t = (t+1) % n
                # move a fraction of source mass
                delta = cur[s] * rng.random() * 0.3
                cand = cur.copy()
                cand[s] -= delta
                cand[t] += delta
                cand_sc = score(cand)
                if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / max(temp, min_temp)):
                    cur, cur_sc = cand, cand_sc
                    if cand_sc < best_sc:
                        best_vec, best_sc = cand.copy(), cand_sc
                    no_improve = 0
                else:
                    no_improve += 1
                # continue loop to smaller steps
                continue

        # standard random pairwise transfer
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # choose delta up to a fraction of source
        max_delta = cur[i]
        if max_delta <= 0:
            continue
        delta = max_delta * rng.random() * 0.25  # up to 25%
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_sc = score(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            # SA acceptance
            prob = math.exp(-(cand_sc - cur_sc) / max(temp, min_temp))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
                no_improve += 1
            else:
                no_improve += 1

        # temperature decay
        temp = max(temp * decay, min_temp)

        # occasional symmetric projection every 1000 it
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym_sc = score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # optional restart if stuck
        if no_improve > 3000:
            # perturb current best slightly
            cur = best_vec.copy()
            # apply small random redistribution
            for _ in range(5):
                i = rng.integers(0, n)
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                max_delta = cur[i]
                delta = max_delta * rng.random() * 0.2
                cur[i] -= delta
                cur[j] += delta
            cur_sc = score(cur)
            no_improve = 0

        # early exit if reached target
        if best_sc < 1.50274349:
            break

    # Return best vector as list of floats
    return [float(x) for x in best_vec]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic search for a non‑negative step‑function minimizing the
    AC1 upper bound.  Uses simple simulated‑annealing style pairwise mass
    transfers with occasional targeted moves.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # leave a small head‑room for the return
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # initialise from the provided construction if available
    if "height_sequence_1" in globals():
        cur = np.maximum(np.asarray(height_sequence_1, dtype=np.float64), 0.0)
    else:
        cur = rng.random(2500, dtype=np.float64)

    total = cur.sum()
    if total < 1e-12:                     # avoid degenerate start
        cur = np.full_like(cur, 1.0)
        total = cur.sum()
    cur = cur / total                      # enforce sum = 1

    n = cur.size
    # FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # helper to evaluate 2·n·max(conv) (sum is 1, so denominator = 1)
    def _score(vec):
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------------
    # temperature schedule (simulated annealing)
    af = np.fft.rfft(cur, n=L)
    conv0 = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_conv = conv0.max()
    temp = max_conv * 0.5                 # initial temperature
    min_temp = max_conv * 1e-8
    decay = 0.9995

    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1
        # temperature decay (always applied)
        temp = max(temp * decay, min_temp)

        # --------------------------------------------------------------
        # occasional targeted transfer using the current maximal convolution term
        if it % 200 == 0:
            af = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
            k_max = int(conv_full.argmax())
            # contribution a[i] * a[k_max-i] for indices where it is defined
            contrib = np.zeros(n, dtype=np.float64)
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            jdx = k_max - idx
            contrib[idx] = cur[idx] * cur[jdx]

            if contrib.max() > 0:
                src_idx = np.argpartition(-contrib, 5)[:5]   # top‑5 contributors
                tgt_idx = np.argpartition(contrib, 5)[:5]   # bottom‑5
                s = rng.choice(src_idx)
                t = rng.choice(tgt_idx)
                if s == t:
                    t = (t + 1) % n
                delta = cur[s] * rng.random() * 0.30        # move up to 30 %
                cand = cur.copy()
                cand[s] -= delta
                cand[t] += delta
                cand_sc = _score(cand)

                if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / max(temp, min_temp)):
                    cur, cur_sc = cand, cand_sc
                    no_improve = 0
                    if cand_sc < best_sc:
                        best_vec, best_sc = cand.copy(), cand_sc
                else:
                    no_improve += 1
                continue   # go to next iteration

        # --------------------------------------------------------------
        # simple random pairwise mass transfer
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if cur[i] <= 0.0:
            continue
        delta = cur[i] * rng.random() * 0.25   # up to 25 % of source mass
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            # Metropolis acceptance
            prob = math.exp(-(cand_sc - cur_sc) / max(temp, min_temp))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
                no_improve += 1
            else:
                no_improve += 1

        # --------------------------------------------------------------
        # occasional symmetrisation (helps many instances)
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # --------------------------------------------------------------
        # restart with a slight perturbation if stuck too long
        if no_improve > 3000:
            cur = best_vec.copy()
            for _ in range(5):
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                if cur[i] <= 0.0:
                    continue
                delta = cur[i] * rng.random() * 0.20
                cur[i] -= delta
                cur[j] += delta
            cur_sc = _score(cur)
            no_improve = 0

        # --------------------------------------------------------------
        # early exit if we beat the target bound
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
