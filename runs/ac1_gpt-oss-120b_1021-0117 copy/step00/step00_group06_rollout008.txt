import numpy as np
import time

def _max_convolution_bound(indices, n, K):
    """Return (size, max_conv) for the set of indices under ordered‐pair limit K.
    The function also ensures the construction respects the limit.
    """
    sum_counts = {}
    m = 0
    for x in indices:
        # check self pair
        s_self = x + x
        if sum_counts.get(s_self, 0) + 1 > K:
            continue
        # check with already accepted elements
        ok = True
        for y in indices[:m]:
            s = x + y
            if sum_counts.get(s, 0) + 2 > K:
                ok = False
                break
        if not ok:
            continue
        # accept x
        # self
        sum_counts[s_self] = sum_counts.get(s_self, 0) + 1
        # pairs
        for y in indices[:m]:
            s = x + y
            sum_counts[s] = sum_counts.get(s, 0) + 2
        indices[m] = x  # keep accepted elements at the front
        m += 1
    # after construction, compute the true max convolution (ordered pairs)
    max_conv = max(sum_counts.values()) if sum_counts else 0
    return m, max_conv, sum_counts

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Construct a sparse 0/1 sequence with low max convolution."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # length of the problem (fallback to 5582 if parent not present)
    try:
        n = int(len(height_sequence_1))  # parent may be defined
    except Exception:
        n = 5582

    # start from parent if it is a valid sequence
    best_seq = None
    best_score = float('inf')
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size and np.isfinite(parent).all():
            parent = np.clip(parent, 0.0, 1000.0)
            best_seq = parent.copy()
            best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        pass
    if best_seq is None:
        # tiny fallback
        best_seq = np.ones(min(256, n), dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # parameters for the combinatorial construction
    K_target = 8                     # allowed ordered pairs per sum (<=K_target)
    attempts = 0

    while time.monotonic() < deadline:
        attempts += 1
        # random permutation of positions
        perm = rng.permutation(n).astype(np.int32)
        # workspace: we will fill accepted indices at the start of this array
        work = perm.copy()
        m, max_conv, _ = _max_convolution_bound(work, n, K_target)

        if m < 2:           # need at least two ones to have a meaningful bound

        # theoretical bound using binary values (exact for this construction)
        score_est = 2.0 * n * max_conv / (m * m)
