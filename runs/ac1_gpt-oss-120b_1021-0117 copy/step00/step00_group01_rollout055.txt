import numpy as np
import time

def _make_block_perturbation(base, rng):
    """Apply a small multiplicative perturbation to a random contiguous block."""
    a = base.copy()
    n = a.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    a[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # optional gentle symmetrisation
        a = 0.5 * (a + a[::-1])
    return np.clip(a, 0.0, 1000.0)


def _init_exponential(n):
    """Exponential shape a_i = r^i with r tuned to minimise the AC1 bound."""
    # solve e^x = 1 + 2x  (optimal x ≈ 1.25643)
    x = 1.2
    for _ in range(20):
        fx = np.exp(x) - 1.0 - 2.0 * x
        dfx = np.exp(x) - 2.0
        x -= fx / (dfx + 1e-12)
    eps = x / n                     # epsilon = 1 - r
    r = np.exp(np.log1p(-eps))      # r = 1 - eps  (more stable)
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(np.log(r) * idx)     # a_i = r^i
    return np.clip(a, 0.0, 1000.0)
