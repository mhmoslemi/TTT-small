import numpy as np
import time

def _convolution_max(seq: np.ndarray) -> float:
    """Return max of full convolution seq*seq (non‑negative)."""
    # np.convolve works fast enough for the intended sizes.
    return float(np.max(np.convolve(seq, seq)))

def _score(seq: np.ndarray) -> float:
    """Exact score used by the authoritative evaluator."""
    # The evaluator applies the same clipping and conversion rules.
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(np.sum(seq))
    if s < 0.01:
        return float('inf')
    return 2.0 * seq.size * _convolution_max(seq) / (s * s)


def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improve on the
    baseline bound 2.0.  The routine works entirely in O(n) memory,
    stays within the given time budget, and always falls back to a valid
    candidate (the uniform distribution) if no improvement is found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 1.  Determine length and initialise a uniform candidate.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            n = 256
        else:
            n = base.shape[0]
    else:
        n = 256

    # Uniform vector with sum 1.0 (the scaling does not affect the score).
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    best = uniform.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # 2.  Helper: a cheap block perturbation (as in the reference code).
    # ------------------------------------------------------------------
    def block_perturb(x: np.ndarray) -> np.ndarray:
        """Multiply a random contiguous block by a log‑normal factor."""
        cand = x.copy()
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        cand[start:stop] *= factor
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])
        np.clip(cand, 0.0, 1000.0, out=cand)
        return cand

    # ------------------------------------------------------------------
    # 3.  Helper: a simple mass‑transfer move that preserves the total sum.
    # ------------------------------------------------------------------
    def transfer_move(x: np.ndarray) -> np.ndarray:
        """Move a small amount of mass from one index to another."""
        cand = x.copy()
        i, j = rng.integers(0, n, size=2)
        if i == j:
            return cand
        # Take a fraction of the donor (no more than 10 % of its current value)
        donor = max(i, j)
        receiver = min(i, j)
        amount = cand[donor] * rng.uniform(0.0, 0.1)
        if amount <= 0.0:
            return cand
        cand[donor] -= amount
        cand[receiver] += amount
        # Numerical safety
        np.clip(cand, 0.0, 1000.0, out=cand)
        return cand

    # ------------------------------------------------------------------
    # 4.  Main optimisation loop.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # Randomly choose a perturbation type
        if rng.random() < 0.6:
            candidate = block_perturb(best)
        else:
            candidate = transfer_move(best)

        # Rescale to keep the total sum (optional – not needed for the score,
        # but keeps the vector in a comfortable numerical range).
        s = candidate.sum()
        if s > 0.0:
            candidate *= (1.0 / s)  # force sum = 1.0

        sc = _score(candidate)
        if np.isfinite(sc) and sc < best_score:
            best = candidate
            best_score = sc

    # ------------------------------------------------------------------
    # 5.  Return the best found candidate as a plain Python list.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
