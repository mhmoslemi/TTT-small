def evaluate_sequence(sequence: list) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

def solve_lp_step(f, M, active, t=0.3):
    n = len(f)
    g = cp.Variable(n)
    constraints = [g >= 0]
    for k in active:
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        i = np.arange(i_start, i_end+1)
        j = k - i
        constraints.append(cp.sum(cp.multiply(f[i], g[j])) <= M)
    # objective: maximize sum(g)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iter=2000)
    if g.value is None:
        return None
    g_val = np.asarray(g.value, dtype=np.float64)
    sum_f = np.sum(f)
    sum_g = np.sum(g_val)
    if sum_g <= 0:
        return None
    # Scale to same sum as f (both normalized to 1)
    scale = sum_f / sum_g
    g_scaled = g_val * scale
    # blend
    f_new = (1 - t) * f + t * g_scaled
    # renormalize sum to 1 (optional)
    s = np.sum(f_new)
    if s > 0:
        f_new = f_new / s
    return f_new

def reduce_max_step(seq, rng, step_factor=0.2):
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    k = np.argmax(conv)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_idx = np.arange(i_start, i_end+1)
    j_idx = k - i_idx
    prod = seq[i_idx] * seq[j_idx]
    # find i with max product
    idx_max = i_idx[np.argmax(prod)]
    # compute counterpart for all positions for this k
    counterpart = np.zeros(n)
    counterpart[i_idx] = seq[j_idx]
    # compute low product indices
    prod_all = seq * counterpart
    # define a mask for indices not equal idx_max and product less than something
    low_candidates = np.where((prod_all < max_val * 0.5) & (np.arange(n) != idx_max))[0]
    if low_candidates.size == 0:
        # fallback to block perturbation
        return None
    idx_low = rng.choice(low_candidates)
    # move epsilon from idx_max to idx_low
    # epsilon factor depends on mass at idx_max
    amount = seq[idx_max] * rng.random() * step_factor
    if amount > seq[idx_max]:
        amount = seq[idx_max]
    new_seq = seq.copy()
    new_seq[idx_max] -= amount
    new_seq[idx_low] += amount
    return new_seq
```

import time, numpy as np

# optional linear programming support
try:
    import cvxpy as cp
    HAVE_CVX = True
except Exception:               # pragma: no cover
    HAVE_CVX = False


def make_block_perturbation(base: np.ndarray, rng) -> np.ndarray:
    """Small random block perturbation (O(n) memory)."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def reduce_step(seq: np.ndarray, rng) -> np.ndarray | None:
    """Move a small amount of mass away from the dominant convolution term."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    k = int(np.argmax(conv))
    i0 = max(0, k - (n - 1))
    i1 = min(k, n - 1)
    i = np.arange(i0, i1 + 1)
    j = k - i
    prod = seq[i] * seq[j]
    idx_max = i[np.argmax(prod)]

    # counterpart for every index under this k
    counterpart = np.zeros(n, dtype=seq.dtype)
    counterpart[i] = seq[j]
    weight = seq * counterpart
    low = np.where((weight < max_val * 0.5) & (np.arange(n) != idx_max))[0]
    if low.size == 0:
        return None
    idx_low = rng.choice(low)
    amt = seq[idx_max] * rng.random() * 0.4
    if amt <= 0.0:
        return None
    out = seq.copy()
    out[idx_max] -= amt
    out[idx_low] += amt
    return out


def lp_update(f: np.ndarray, M: float, active: np.ndarray, t: float = 0.3):
    """Linear‑programming step: maximise sum(g) with f·g ≤ M on active indices."""
    if not HAVE_CVX:
        return None
    n = f.shape[0]
    g = cp.Variable(n)
    cons = [g >= 0]
    for k in active:
        i0 = max(0, k - (n - 1))
        i1 = min(k, n - 1)
        i = np.arange(i0, i1 + 1)
        j = k - i
        cons.append(cp.sum(cp.multiply(f[i], g[j])) <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), cons)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500,
                   abstol=1e-6, reltol=1e-6)
    except Exception:                      # pragma: no cover
        return None
    if g.value is None:
        return None
    g_val = np.asarray(g.value, dtype=np.float64)
    # simple convex combination
    new_f = (1.0 - t) * f + t * g_val
    return new_f


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Heuristic search for a step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # length of the problem (use the supplied parent if present)
    if 'height_sequence_1' in globals():

        n = parent_raw.size

    else:                                            # fallback length
        n = 256
        parent = None

    # start from a uniform sequence (scale is irrelevant for the score)
    best = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # possibly keep the given parent if it is better
    if parent is not None and parent.size == n:
        sc = float(evaluate_sequence(parent.tolist()))
        if sc < best_score:
            best = parent.copy()
            best_score = sc

    # a simple non‑uniform seed for LP‑based updates
    base_vals = np.exp(-0.0005 * np.arange(n))
    base_vals = base_vals / base_vals.sum()

    while time.monotonic() < deadline:
        r = rng.random()

        # ----- different proposal families ---------------------------------
        if r < 0.20:
            cand = make_block_perturbation(best, rng)

        elif r < 0.40:
            # Dirichlet‑like random draw (Gamma shape 0.5 gives variability)
            cand = rng.gamma(0.5, 1.0, size=n)
            cand = cand / cand.sum()

        elif r < 0.60:
            # permutation of a smooth decay vector
            cand = rng.permutation(base_vals)

        elif r < 0.80:
            # targeted reduction of the current maximal convolution term
            cand = reduce_step(best, rng)
            if cand is None:
                continue

        else:
            # linear‑programming refinement (active‑set)
            if not HAVE_CVX:
                continue
            conv = np.convolve(best, best)
            M = conv.max()
            K = 5
            if conv.size <= K:
                active = np.arange(conv.size)
            else:
                active = np.argpartition(conv, -K)[-K:]
            cand = lp_update(best, M, active, t=0.3)
            if cand is None:
                continue

        # ----- normalise, clip and evaluate ---------------------------------
        s = cand.sum()
        if s > 0.0:
            cand = cand / s
        cand = np.clip(cand, 0.0, 1000.0)

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # return a plain Python list (the evaluator expects a list)
    return [float(x) for x in best]
