        # random block scaling of best_seq
        cand = best_seq.copy()
        width = int(rng.integers(max(1, n // 100), max(2, n // 10)))

            best_score = score

def propose_candidate(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)
    # Determine sequence length
    n = None
    try:
        seq_parent = np.asarray(height_sequence_1, dtype=np.float64)
        if seq_parent.ndim == 1 and seq_parent.size > 0 and np.isfinite(seq_parent).all():
            n = seq_parent.size
    except Exception:
        seq_parent = None
    if n is None:
        n = 256
        seq_parent = np.ones(n, dtype=np.float64) * 500.0   # fallback
    # Initialize best with candidate based on monotonic power
    best_seq = None
    best_score = float('inf')
    # build index array from 0 to 1
    idx = np.arange(n, dtype=np.float64) / (n-1) if n > 1 else np.zeros(1)
    # try several exponent values:
    for p in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0):
        cand = np.power(idx, p) * 1000.0
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand.copy()
        # reversed version
        cand_rev = cand[::-1]
        score = float(evaluate_sequence(cand_rev.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand_rev.copy()
    # Also try parent if available:
    if seq_parent is not None:
        parent_seq = np.clip(seq_parent, 0.0, 1000.0)
        score = float(evaluate_sequence(parent_seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = parent_seq.copy()
    # Now local search
    while time.monotonic() < deadline:
        # random block scaling on current best
        cand = best_seq.copy()
        # Choose block size between 1% and 20% of n
        min_width = max(1, n // 200)
        max_width = max(2, n // 10)
        width = int(rng.integers(min_width, max_width + 1))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        factor = math.exp(rng.normal(0.0, 0.07))
        cand[start:stop] *= factor
        # occasional half-symmetrization
        if rng.random() < 0.15:
            cand = (cand + cand[::-1]) * 0.5
        np.clip(cand, 0.0, 1000.0, out=cand)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score
    return [float(x) for x in best_seq]

import time
import math
import numpy as np

# The evaluator `evaluate_sequence` and the optional parent array
# `height_sequence_1` are available in the execution environment.

def _seq_score(seq: np.ndarray) -> float:
    """Helper that returns the authoritative score for a NumPy array."""
    return float(evaluate_sequence(seq.tolist()))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that minimizes the AC‑1 bound.

    The routine builds a few simple analytic families (power ramps, their
    reverses) and keeps the best one.  Afterwards a very fast
    block‑perturbation loop is run for the remaining time to squeeze a little
    extra improvement.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------ #
    # 1. Determine the problem size and a safe fallback length.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = parent.size
    except Exception:
        # No parent supplied – use a modest default.
        n = 256
        parent = np.full(n, 500.0, dtype=np.float64)

    # ------------------------------------------------------------------ #
    # 2. Build a suite of analytic candidates (power ramps) and keep the best.
    idx = np.arange(n, dtype=np.float64) / (n - 1) if n > 1 else np.zeros(1)
    best_seq = None
    best_score = float("inf")

    # explore a range of exponents; the spacing is fine enough for the
    # modest time budget.
    for p in np.linspace(0.1, 6.0, 45):
        # monotone increasing ramp
        cand = np.power(idx, p) * 1000.0
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _seq_score(cand)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand.copy()

        # the reversed version (decreasing ramp) can behave differently
        cand_rev = cand[::-1]
        sc = _seq_score(cand_rev)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand_rev.copy()

    # also keep the provided parent if it happens to be better
    parent_clipped = np.clip(parent, 0.0, 1000.0)
    sc = _seq_score(parent_clipped)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = parent_clipped.copy()

    # ------------------------------------------------------------------ #
    # 3. Fast local search: random block scaling (the same idea as the
    #    reference solution, but starting from the much better base).
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # helper to generate a perturbed copy
    def _perturb(seq: np.ndarray) -> np.ndarray:
        cand = seq.copy()
        # block width between 0.5 % and 10 % of the length
        w_min = max(1, n // 200)
        w_max = max(2, n // 10)
        width = int(rng.integers(w_min, w_max + 1))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        factor = math.exp(rng.normal(0.0, 0.09))
        cand[start:stop] *= factor
        # occasional gentle symmetrisation (can improve the bound)
        if rng.random() < 0.12:
            cand = (cand + cand[::-1]) * 0.5
        np.clip(cand, 0.0, 1000.0, out=cand)
        return cand

    # run the stochastic improvement while we have time
    while time.monotonic() < deadline:
        trial = _perturb(best_seq)
        trial_score = _seq_score(trial)
        if np.isfinite(trial_score) and trial_score < best_score:
            best_score = trial_score
            best_seq = trial

    # ------------------------------------------------------------------ #
    # 4. Return the best sequence found as a plain Python list.
    return [float(v) for v in best_seq]
