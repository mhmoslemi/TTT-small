    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    # initial a
    try:
        a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
            raise Exception()
    except Exception:
        n = 4498
        a = np.full(n, 1.0, dtype=np.float64)
    n = a.size
    # ensure it's float and nonnegative
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur = a.copy()
    cur_score = best_score
    # SA parameters
    temp = 1.0
    cooling = 0.9995
    min_temp = 1e-6
    # step size scale for transfers
    base_step = 0.05  # fraction of mass moved
    while time.monotonic() < deadline:
        # compute convolution and peak
        b = np.convolve(cur, cur)
        k_max = int(np.argmax(b))
        # compute significance array: a partner values for this k
        idx = np.arange(n)
        partner = k_max - idx

        sig = np.zeros_like(cur)
        sig[mask] = cur[partner[mask]]
        # pick source index weighted by sig
        if sig.sum() > 0:
            # probability proportional to sig
            probs = sig / sig.sum()
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(n)
        # pick destination index far from i maybe, uniform random
        j = rng.integers(n)
        # ensure i != j
        if j == i:
            continue

        if delta <= 0:
            continue

        cand[i] -= delta
        cand[j] += delta
        # clip to bounds

import numpy as np
import time
import math

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Simulated‑annealing / local‑transfer search for a low AC1 bound."""
    rng = np.random.default_rng(seed)
    # leave a few seconds margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # initialise a feasible sequence (use the supplied parent when possible)
    try:
        a0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if a0.ndim != 1 or a0.size == 0 or not np.isfinite(a0).all():
            raise Exception()
        cur = a0.copy()
    except Exception:
        # fallback: uniform vector of length 4498 (the known length)
        n = 4498
        cur = np.full(n, 1.0, dtype=np.float64)

    n = cur.size
    # ------------------------------------------------------------------
    # helper to compute the exact AC1 score (same as evaluate_sequence)
    def score(vec):
        s = vec.sum()
        if s == 0.0:
            return float('inf')
        conv = np.convolve(vec, vec)
        return 2.0 * n * conv.max() / (s * s)

    cur_score = score(cur)
    best = cur.copy()
    best_score = cur_score

    # SA hyper‑parameters (all cheap, O(1) per iteration)
    temp = 1.0
    cooling = 0.9995
    min_temp = 1e-6
    base_step = 0.07          # typical fraction of mass moved

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- convolution, peak index, and a cheap “gradient” ----------
        conv = np.convolve(cur, cur)
        k_max = int(np.argmax(conv))

        # partner values for the current peak index
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        sig = np.zeros_like(cur)
        sig[mask] = cur[partner[mask]]

        # pick a source index biased by its contribution to the peak
        if sig.sum() > 0.0:
            probs = sig / sig.sum()
            i_src = rng.choice(n, p=probs)
        else:
            i_src = rng.integers(n)

        # pick a destination index uniformly (different from source)
        i_dst = rng.integers(n)
        if i_dst == i_src:
            continue

        # amount of mass to transfer
        delta = cur[i_src] * base_step * rng.random()
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[i_src] -= delta
        cand[i_dst] += delta
        # enforce bounds
        if cand[i_src] < 0.0:
            cand[i_src] = 0.0
        if cand[i_dst] > 1000.0:
            cand[i_dst] = 1000.0

        cand_score = score(cand)

        # ---- simulated‑annealing acceptance ---------------------------
        improve = cand_score < cur_score
        if improve or (temp > min_temp and rng.random() <
                       math.exp(-(cand_score - cur_score) / temp)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

        # temperature schedule
        temp = max(min_temp, temp * cooling)

        # ---- occasional larger perturbation to keep diversity ----------
        if rng.random() < 0.008:          # about 1 % of loops
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand2 = cur.copy()
            cand2[start:stop] *= factor
            # optional mild symmetrisation
            if rng.random() < 0.3:
                cand2 = 0.5 * (cand2 + cand2[::-1])
            np.clip(cand2, 0.0, 1000.0, out=cand2)
            cand2_score = score(cand2)
            if cand2_score < cur_score or (temp > min_temp and rng.random() <
                                          math.exp(-(cand2_score - cur_score) / temp)):
                cur = cand2
                cur_score = cand2_score
                if cand2_score < best_score:
                    best = cand2.copy()
                    best_score = cand2_score

    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
