import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Gradient‑descent flattening of the self‑convolution.

    The algorithm keeps the sequence non‑negative, normalises its sum to 1,
    and iteratively reduces the variance of the linear convolution a*a.
    The objective 2*n*max(conv)/(sum**2) is minimised.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # --- initialise ---------------------------------------------------------
    # use the provided parent if it looks valid, otherwise a uniform start
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise Exception()
        parent = parent_arr.copy()
    except Exception:
        parent = None

    if parent is not None:
        # normalise parent to sum = 1 for the optimisation (its score is a fallback)
        parent_sum = parent.sum()
        if parent_sum > 0:
            parent = np.clip(parent, 0.0, None)
            parent /= parent_sum
        else:
            parent = np.ones_like(parent) / parent.size
        best = parent.copy()
        best_score = float(evaluate_sequence(best.tolist()))
    else:
        n = 3809
        best = np.ones(n, dtype=np.float64) / n
        best_score = 2.0 * n * (np.max(np.convolve(best, best, mode="full")))  # =2
    n = best.size

    # --- parameters for the FFT based optimisation -------------------------
    Lconv = 2 * n - 1                     # length of linear self‑convolution
    Lcorr = n + Lconv - 1                 # length needed for a*e correlation
    Lfft = 1 << ((Lcorr - 1).bit_length())  # next power of two (≥ 3n)

    target = 1.0 / Lconv                  # target value of each convolution entry (sum=1)

    step = 0.01                           # initial step magnitude (euclidean norm)

    # --- main optimisation loop --------------------------------------------
    while time.monotonic() < deadline:
        # FFT of the current sequence (zero‑padded)
        fft_a = np.fft.rfft(best, n=Lfft)

        # linear self‑convolution via pointwise product in frequency domain
        conv_full = np.fft.irfft(fft_a * fft_a, n=Lfft)[:Lconv]

        # current objective (exact, since sum(best)=1)
        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best = best.copy()

        # error w.r.t. the flat target
        err = conv_full - target

        # correlation a ⋆ err = Σ_j a_j * err_{i+j}
        fft_err = np.fft.rfft(err, n=Lfft)
        corr = np.fft.irfft(np.conj(fft_a) * fft_err, n=Lfft)[:n]

        grad = 2.0 * corr                  # gradient of Σ(err²)

        # normalise gradient and take a gradient step
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            best -= step * grad / grad_norm
        # enforce non‑negativity and renormalise to sum = 1
        best = np.clip(best, 0.0, None)
        s = best.sum()
        if s == 0.0:
            best = np.ones(n, dtype=np.float64) / n
        else:
            best /= s

        # gentle annealing of the step size
        step *= 0.9995

        # occasional tiny random kick to escape plateaus
        if rng.random() < 0.001:
            best += 0.001 * rng.standard_normal(n)
            best = np.clip(best, 0.0, None)
            best /= best.sum()

    # final sanity – make sure the output respects the evaluator limits
    final = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in final]
