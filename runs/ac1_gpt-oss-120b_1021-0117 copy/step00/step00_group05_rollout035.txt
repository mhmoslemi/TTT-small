def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    n = len(height_sequence_1) if ... else 256
    # define candidate generation function
    def gen(alpha):
        i = np.arange(1, n+1, dtype=np.float64)
        a = i ** (-alpha)
        # maybe scale to have max 1000? Not needed but can multiply by 1000 (makes sum bigger proportionally)
        # scaling cancels, but may affect numeric stability, keep as is.
        return a

    alphas = [0.45, 0.5, 0.55, 0.6, 0.65, 0.7]
    best = None
    best_score = float('inf')
    for alpha in alphas:
        seq = gen(alpha)
        # Clip for safety
        seq = np.clip(seq, 0.0, 1000.0)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = seq.copy()
    # if best remains None, fallback to uniform
    if best is None:
        best = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))
    
    # Return best as list.
    return [float(x) for x in best]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    from scipy.optimize import linprog

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Determine n and initial f
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)
    else:
        n = int(parent.size)

    # Clip to allowed range
    f = np.clip(parent, 0.0, 1000.0)
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Precompute FFT length
    L = 1 << ((2*n - 1).bit_length())

    # Outer loop
    iteration = 0
    while time.monotonic() < deadline and iteration < 20:
        # Compute convolution f*f
        F = np.fft.rfft(f, n=L)
        conv_ff = np.fft.irfft(F * F, n=L)[:2*n-1]
        M = float(conv_ff.max())
        # Active constraints set
        active = set()
        max_idx = int(np.argmax(conv_ff))
        active.add(max_idx)
        # Cutting-plane
        while True:
            # Build A_ub
            rows = []
            for k in active:
                row = np.zeros(n, dtype=np.float64)
                start_j = max(0, k - (n-1))
                end_j = min(k, n-1)
                if start_j <= end_j:
                    j = np.arange(start_j, end_j+1, dtype=np.int64)
                    i = k - j
                    row[j] = f[i]
                rows.append(row)
            A_ub = np.vstack(rows) if rows else np.empty((0, n))
            b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)
            # Solve LP: maximize sum(g) => minimize -sum(g)
            c = -np.ones(n, dtype=np.float64)
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0.0, 1000.0)] * n,
                          method='highs')
            if not res.success:
                # fallback to current f
                break
            g = res.x
            # compute convolution f*g
            G = np.fft.rfft(g, n=L)
            conv_fg = np.fft.irfft(F * G, n=L)[:2*n-1]
            viol = conv_fg - M
            max_viol = float(viol.max())
            if max_viol <= 1e-8:
                # feasible found
                break
            # add most violating shift
            viol_idx = int(np.argmax(conv_fg))
            active.add(viol_idx)
            # limit active size to avoid explosion
            if len(active) > 100:
                break
        # Mix f and g
        t = 0.2
        f_new = (1 - t) * f + t * g
        # Clip to bounds
        f_new = np.clip(f_new, 0.0, 1000.0)
        score_new = float(evaluate_sequence(f_new.tolist()))
        if np.isfinite(score_new) and score_new < best_score:
            best = f_new.copy()
            best_score = score_new
        # update f for next iteration
        f = f_new
        iteration += 1

    # If best still not improved maybe fallback to uniform
    if best_score == float('inf') or best_score > 2.0:
        # fallback simple uniform
        best = np.full(n, 1.0, dtype=np.float64)
    return [float(x) for x in best]

import numpy as np
import time
from scipy.optimize import linprog

def _conv_fft(a, b, L):
    """Linear convolution of a and b using FFT of length L."""
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    return np.fft.irfft(A * B, n=L)[: 2 * a.size - 1]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a sequence with a low AC‑upper‑bound score.

    The algorithm starts from the supplied parent (if any) and repeatedly
    solves a tiny linear programme that moves mass to positions that are
    less constrained by the current maximal convolution value.
    """
    rng = np.random.default_rng(seed)
    end_time = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # initialise the sequence ------------------------------------------------
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is provided by the harness
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
    except Exception:                     # fall back to a simple uniform start
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    n = int(parent.size)
    f = np.clip(parent, 0.0, 1000.0)      # current working sequence

    best_seq = f.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # outer optimisation loop ------------------------------------------------
    # ------------------------------------------------------------------
    outer_iter = 0
    while time.monotonic() < end_time and outer_iter < 30:
        # current auto‑convolution and its maximal value
        conv_ff = _conv_fft(f, f, L)
        M = float(conv_ff.max())                # the current bound to respect

        # ------------------- cutting‑plane LP -----------------------------
        # start with the shift that attains the maximum
        active_shifts = {int(np.argmax(conv_ff))}

        # iterate adding violated shifts until the LP solution respects all
        while True:
            # build the (small) constraint matrix for the active shifts
            rows = []
            for k in active_shifts:
                row = np.zeros(n, dtype=np.float64)
                start_j = max(0, k - (n - 1))
                end_j = min(k, n - 1)
                if start_j <= end_j:
                    j = np.arange(start_j, end_j + 1, dtype=np.int64)
                    i = k - j
                    row[j] = f[i]                  # coeff = f_i
                rows.append(row)
            A_ub = np.vstack(rows) if rows else np.empty((0, n), dtype=np.float64)
            b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)

            # linear programme: maximise sum(g)  (i.e. minimise –sum(g))
            c = -np.ones(n, dtype=np.float64)      # minimise -∑g_i
            bounds = [(0.0, 1000.0)] * n
            res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
            if not res.success:
                # LP failed – give up and keep the current f
                break

            g = res.x
            # check for any new violations of the bound M
            conv_fg = _conv_fft(f, g, L)
            viol = conv_fg - M
            max_viol = float(viol.max())
            if max_viol <= 1e-8:
                # feasible solution found
                break

            # add the most violating shift and continue
            viol_idx = int(np.argmax(conv_fg))
            active_shifts.add(viol_idx)
            # keep the active set modest to stay within time limits
            if len(active_shifts) > 100:
                break

        # ------------------- mixing step --------------------------------
        # blend a fraction of the LP solution back into f
        t = 0.2                                   # step size, can be tuned
        f_new = (1.0 - t) * f + t * g
        f_new = np.clip(f_new, 0.0, 1000.0)

        # evaluate the new candidate
        score_new = float(evaluate_sequence(f_new.tolist()))
        if np.isfinite(score_new) and score_new < best_score:
            best_score = score_new
            best_seq = f_new.copy()

        # prepare for the next outer iteration
        f = f_new
        outer_iter += 1

    # ------------------------------------------------------------------
    # final fallback (should never be needed) -----------------------------
    # ------------------------------------------------------------------
    if best_seq.size == 0:
        best_seq = np.full(n, 1.0, dtype=np.float64)

    return [float(v) for v in best_seq]
