def max_convolution(p):
    N = len(p)
    L = 1 << ((2*N - 1).bit_length())
    fp = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(fp*fp, n=L)[:2*N-1]
    # due to rounding errors may be a bit negative; set to zero
    max_val = conv.max()
    return max_val

def propose_move(p, rng):
    q = p.copy()
    # decide number of moves avg 2
    num_moves = rng.integers(1, 4)  # 1 to 3 moves
    for _ in range(num_moves):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # choose delta up to proportion of source
        source_val = q[i]
        if source_val <= 0:
            continue
        # delta factor random between 0 and move_scale
        delta = source_val * rng.random() * 0.1  # up to 10%
        q[i] -= delta
        q[j] += delta
    return q

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = parent.size

        n = 256
    # if no valid parent, use fallback uniform of length n
    # we will search using probabilities (sum=1)
    n = int(n)
    # initialize probability vector uniform
    p = np.full(n, 1.0 / n, dtype=np.float64)
    # compute best
    L = 1 << ((2 * n - 1).bit_length())
    def max_convolution(p_vec):
        fp = np.fft.rfft(p_vec, n=L)
        conv = np.fft.irfft(fp * fp, n=L)[:2 * n - 1]
        # ensure nonnegative
        return float(conv.max())
    best_p = p.copy()
    best_max = max_convolution(p)
    best_score = 2.0 * n * best_max
    # temperature schedule
    temp = 0.01
    decay = 0.99995
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # propose new vector
        q = best_p.copy()
        # choose move type
        if rng.random() < 0.7:
            # local transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            if q[i] == 0:
                continue
            delta = q[i] * rng.random() * 0.2  # move up to 20% of source
            q[i] -= delta
            q[j] += delta

            alpha_param = 0.5 + rng.random() * 2.0  # between 0.5 and 2.5

import numpy as np
import time
import math

def _max_convolution(p, L):
    """Return max of the full convolution p*p using FFT of length L."""
    fp = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(fp * fp, n=L)[: 2 * p.size - 1]
    return float(conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a non‑negative step function with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # length of the problem (use the provided parent if present)
    try:
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = int(base.size)
        if n == 0:
            raise ValueError
    except Exception:
        n = 256                     # safe fallback length
        base = np.full(n, 0.0)

    # work with a probability vector (sum = 1); scaling is irrelevant for the score
    p = np.full(n, 1.0 / n, dtype=np.float64)

    # FFT length – the next power of two ≥ 2n‑1
    L = 1 << ((2 * n - 1).bit_length())

    # current best
    best_p = p.copy()
    best_max = _max_convolution(best_p, L)
    best_score = 2.0 * n * best_max

    # simple simulated‑annealing loop
    temp = 0.02
    decay = 0.99997
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # create a neighbour of the current best point
        # ------------------------------------------------------------------
        if rng.random() < 0.7:
            # small mass transfer between two random coordinates
            q = best_p.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:

            if q[i] == 0.0:

            delta = q[i] * rng.random() * 0.15               # up to 15 % of source
            q[i] -= delta
            q[j] += delta
        else:

            alpha = 0.5 + rng.random() * 2.0                # 0.5 … 2.5
            q = rng.dirichlet(np.full(n, alpha))

        # evaluate the candidate

        cur_max = _max_convolution(q, L)

        if cur_score < best_score or rng.random() < math.exp(-(cur_score - best_score) / (temp + 1e-12)):
