def propose_candidate(seed=42, budget_s=1100):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 5000
    # Initialize vector
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size != n or not np.isfinite(parent).all():
        # fallback: uniform
        a = np.ones(n, dtype=np.float64)
    else:
        a = parent.copy()
    # ensure sum >0
    if np.sum(a) < 0.01:
        a = np.ones(n, dtype=np.float64) * (0.01 / n)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur_score = best_score
    # Parameters
    max_iter = 0
    # Baseline uniform not optimal; we will random perturb initial a a bit.
    # We'll start by applying a few random perturbations to break symmetry
    for _ in range(10):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.random() * 0.1 * a[i]  # up to 10% of a[i]
        if a[i] < delta:
            continue
        a[i] -= delta
        a[j] += delta
    cur_score = float(evaluate_sequence(a.tolist()))
    if cur_score < best_score:
        best = a.copy()
        best_score = cur_score
    # Main loop
    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose move
        # Determine delta magnitude
        max_delta = a[i]  # can't withdraw more than current amount
        if max_delta <= 0:
            continue
        # delta is random fraction of a[i], up to maybe 5%
        delta = rng.random() * min(0.05 * max_delta, max_delta)
        a[i] -= delta
        a[j] += delta
        # compute new score
        new_score = float(evaluate_sequence(a.tolist()))
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta
        max_iter += 1
    return [float(x) for x in best]

import numpy as np
import time
import math

def _score_vector(v, n):
    """Fast exact score: 2 n * max(conv) / (sum(v))**2."""
    s = np.sum(v)
    if s < 0.01:
        return np.inf
    maxb = np.max(np.convolve(v, v))
    return 2.0 * n * maxb / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step function with a low AC1 upper bound.

    The search uses a small set of stochastic moves that keep memory O(n)
    and runs until the allotted time expires (or a good enough solution is
    found).  The result is a list of floats suitable for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    # leave a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, total_budget - 2.0)

    # ------------------------------------------------------------------
    # initialise a starting vector -------------------------------------------------
    try:
        # height_sequence_1 is supplied by the harness
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = base.size
    except Exception:               # pragma: no cover
        n = 6000
        base = np.ones(n, dtype=np.float64)

    if base.ndim != 1 or n == 0 or not np.isfinite(base).all():
        base = np.ones(n, dtype=np.float64)

    # a tiny random tilt helps to escape the perfectly symmetric uniform case
    cur = base.copy()
    cur += rng.normal(scale=0.001, size=n)
    cur = np.clip(cur, 0.0, 1000.0)

    # ensure a non‑trivial sum
    if np.sum(cur) < 0.01:
        cur += 0.01 / n

    cur_score = _score_vector(cur, n)
    best = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # parameters for the stochastic search ------------------------------------
    TARGET = 1.50274349   # the required upper bound (lower is better)
    # kernels for a cheap smoothing move
    smooth_kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)

    # ------------------------------------------------------------------
    # main optimisation loop -------------------------------------------------
    while time.monotonic() < deadline:
        # fraction of remaining time – used to scale move sizes
        remaining = max(0.0, deadline - time.monotonic())
        time_frac = remaining / total_budget

        # ----- decide which mutation to apply ---------------------------------
        move_type = rng.random()

        if move_type < 0.45:
            # ----------- high‑to‑low mass transfer -----------------------------
            total = np.sum(cur)
            if total == 0.0:
                continue
            # pick a source proportional to its magnitude
            probs_src = cur / total
            i = rng.choice(n, p=probs_src)
            # pick a destination biased toward small values
            inv = total - cur
            if np.allclose(inv, 0):
                continue
            probs_dst = inv / np.sum(inv)
            k = rng.choice(n, p=probs_dst)
            if i == k:
                continue
            max_move = cur[i] * (0.3 * time_frac + 0.01)   # larger early on
            delta = rng.random() * max_move
            if delta <= 0.0:
                continue
            new = cur.copy()
            new[i] -= delta
            new[k] += delta

        elif move_type < 0.80:
            # ----------- block scaling ----------------------------------------
            # block width chosen between 0.5 % and 15 % of the length
            lo = max(1, n // 200)
            hi = max(2, n // 12)
            width = int(rng.integers(lo, hi))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            factor = float(np.exp(rng.normal(0.0, 0.12 * time_frac + 0.02)))
            new = cur.copy()
            new[start:stop] *= factor

        elif move_type < 0.95:
            # ----------- smoothing (local averaging) --------------------------
            # convolution with a small fixed kernel; sum is preserved
            new = np.convolve(cur, smooth_kernel, mode='same')
        else:
            # ----------- random reversal (a cheap symmetry move) -------------
            new = cur[::-1].copy()

        # ----- enforce bounds -------------------------------------------------
        new = np.clip(new, 0.0, 1000.0)

        # ----- score the candidate -------------------------------------------
        new_score = _score_vector(new, n)

        # ----- acceptance rule ------------------------------------------------
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
                if best_score <= TARGET:
                    break
        else:
            # occasional uphill moves (simulated‑annealing style)
            temp = 0.001 * time_frac + 1e-9
            if rng.random() < math.exp(-(new_score - cur_score) / temp):
                cur = new
                cur_score = new_score
        # end of loop --------------------------------------------------------

    # ------------------------------------------------------------------
    # final safety check with the official evaluator (optional but cheap)
    try:
        from __main__ import evaluate_sequence  # provided by the harness
        final_score = float(evaluate_sequence(best.tolist()))
        if final_score < best_score:   # pragma: no cover
            best = np.asarray(best, dtype=np.float64)
    except Exception:                 # pragma: no cover
        pass

    return [float(x) for x in best]
