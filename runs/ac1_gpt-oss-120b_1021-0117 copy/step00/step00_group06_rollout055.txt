def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # Determine sequence length n
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 5000  # fallback length
    n = max(2, n)  # ensure at least 2

    # Build binomial series
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (1.0 - 1.0/(2.0*i))
    x = a / a.sum()
    # Ensure non-neg, clip
    x = np.clip(x, 0.0, 1000.0)

    best = x.copy()
    best_score = 2 * n * np.max(np.convolve(best, best)) / (best.sum()**2)

    # also have fallback uniform if needed
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    uniform_score = 2 * n * np.max(np.convolve(uniform, uniform)) / (uniform.sum()**2)
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # heuristics parameters
    step_factor = 0.2  # fraction of source value to move
    min_step = 1e-6

    # iterative improvement
    while time.monotonic() < deadline:
        # compute convolution and max index
        c = np.convolve(best, best)
        max_val = c.max()
        k = int(c.argmax())

        # compute partner array
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # partner[i_start:i_end+1] = best[k - np.arange(i_start, i_end+1)]
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = np.zeros_like(best)
        partner[idx] = best[k - idx]

        # pick source index (large partner) and dest index (small partner)
        src = int(partner.argmax())
        # choose dest among indices with minimal partner (prefer zeros)
        min_val = partner.min()
        candidates = np.where(partner == min_val)[0]
        dest = int(rng.choice(candidates))

        if src == dest:
            # no move possible, break
            break

        # step size
        delta = best[src] * step_factor
        if delta < min_step:
            delta = min_step
        # ensure we don't make src negative
        if delta > best[src]:
            delta = best[src]

        candidate = best.copy()
        candidate[src] -= delta
        candidate[dest] += delta

        # Clip and renormalize to avoid drift
        candidate = np.clip(candidate, 0.0, 1000.0)
        s = candidate.sum()
        if s < 1e-8:
            continue
        candidate = candidate / s * best.sum()  # keep total sum same as before (should be same). Actually sum unchanged if we move same amount.

        # compute score
        new_score = 2 * n * np.max(np.convolve(candidate, candidate)) / (candidate.sum()**2)

        # accept if improvement
        if new_score < best_score:
            best = candidate
            best_score = new_score
            # optional: reduce step factor gradually
            # step_factor = max(0.01, step_factor * 0.999)
        else:
            # occasional random restart or random perturbation
            if rng.random() < 0.01:
                # make a random perturbation via block
                width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                stop = min(n, start + width)
                pert = best.copy()
                pert[start:stop] *= np.exp(rng.normal(0.0, 0.1))
                pert = np.clip(pert, 0.0, 1000.0)
                pert_sum = pert.sum()
                if pert_sum > 0:
                    pert = pert / pert_sum * best.sum()
                    cand_score = 2*n*np.max(np.convolve(pert,pert))/(pert.sum()**2)
                    if cand_score < best_score:
                        best = pert
                        best_score = cand_score
    # return best as list of floats
    return [float(v) for v in best]

import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Gradient‑guided mass transfer to flatten the self‑convolution.
    Returns a non‑negative step‑function (list of floats)."""
    rng = np.random.default_rng(seed)
    # ---- timing ------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ---- sequence length ----------------------------------------------------
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 5000                     # a reasonable default if no parent

    # ---- initial guess: binomial series (a_i = C(2i,i)/4^i) -----------------
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))      # recurrence a_i = a_{i-1}*(1-1/(2i))
    x = a / a.sum()                                    # normalise to sum == 1
    x = np.clip(x, 0.0, 1000.0)

    # ---- fallback uniform sequence (score = 2) -------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best = x.copy()
    # score = 2 * n * max(conv) / (sum^2)   (sum == 1 for both candidates)
    def _score(v):
        c = np.convolve(v, v)
        return 2.0 * n * c.max() / (v.sum() ** 2)

    best_score = _score(best)
    uniform_score = _score(uniform)
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # ---- parameters ---------------------------------------------------------
    step_factor = 0.4          # fraction of source value to move
    min_step = 1e-7

    # ---- main optimisation loop ---------------------------------------------
    while time.monotonic() < deadline:
        # current convolution and its maximal index
        conv = np.convolve(best, best)
        cur_max = conv.max()
        k = int(conv.argmax())

        # partner[i] = best[k-i] if defined, else 0
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = np.zeros_like(best)
        partner[idx] = best[k - idx]

        src = int(partner.argmax())                     # index contributing most to current max
        # pick a destination among indices with minimal partner (often zero)
        min_val = partner.min()
        dest_candidates = np.where(partner == min_val)[0]
        if dest_candidates.size == 0:
            dest = rng.integers(0, n)
        else:
            dest = int(rng.choice(dest_candidates))

        if src == dest:
            # nothing to move – try a tiny random perturbation instead
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)

            cand = np.clip(cand, 0.0, 1000.0)

                cand_score = _score(cand)
