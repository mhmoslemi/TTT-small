            best = seq.copy()

    for target in targets:
        if time.monotonic() >= deadline:

        seq = np.clip(seq, 0.0, 1000.0)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = seq.copy()
    # fallback to parent (height_sequence_1) if best not found
    if best is None:
        best = np.clip(parent, 0.0, 1000.0)
        best_score = float(evaluate_sequence(best.tolist()))
    # Perform small random local perturbations to possibly improve further within remaining time.
    rng = np.random.default_rng(seed)
    while time.monotonic() < deadline:

        cand = best.copy()

import time
import numpy as np


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a non‑negative step‑function sequence with a low AC1 bound.

    The construction uses a geometric‑decay profile a_i = r**i, where r is
    chosen so that the theoretical bound 2·n·(1‑r)/(1+r) is well below the
    target 1.50274349.  Several target AC1 values are tried, the best candidate
    is kept, and a tiny random multiplicative jitter is applied until the
    time budget expires.
    """
    rng = np.random.default_rng(seed)
    # deadline leaves a short margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)

    # length of the problem (parent may be unavailable)
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.shape[0]
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)

    # initialise with the parent (or uniform) – always a valid fallback
    best_seq = np.clip(base, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # try a few geometric sequences aiming at decreasing AC1
    # target values are chosen below the required bound; r follows from
    #   AC1 = 2·n·(1‑r)/(1+r)  =>  r = (2n‑AC1)/(2n+AC1)
    targets = [1.5, 1.3, 1.0, 0.8, 0.5]
    for target in targets:
        if time.monotonic() >= deadline:
            break
        # r ∈ (0,1) for any realistic target << 2·n
        r = (2.0 * n - target) / (2.0 * n + target)
        if not (0.0 < r < 1.0):
            continue

        # forward geometric
        seq = np.power(r, np.arange(n, dtype=np.float64))
        seq = np.clip(seq, 0.0, 1000.0)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq, best_score = seq, score

        # reversed (large values at the end)
        rev = seq[::-1].copy()
        score_rev = float(evaluate_sequence(rev.tolist()))
        if np.isfinite(score_rev) and score_rev < best_score:
            best_seq, best_score = rev, score_rev

    # brief random local search to possibly improve the best candidate
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        factor = np.exp(rng.normal(0.0, 0.05))
        cand = best_seq.copy()
        cand[i] *= factor
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq, best_score = cand, score

    # convert to plain Python list of floats as required
    return [float(x) for x in best_seq]
