import numpy as np
import time

def _score(seq: np.ndarray) -> float:
    """Exact score (identical to evaluate_sequence) for a non‑negative sequence."""
    n = seq.size
    total = seq.sum()
    if total < 0.01:
        return float('inf')
    # convolution (full)
    conv = np.convolve(seq, seq)
    maxb = conv.max()
    return float(2.0 * n * maxb / (total * total))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Construct a non‑negative step‑function heights vector that minimizes the AC1
    upper bound.  The construction uses a power‑law sequence a_i = (i+1)^{-α},
    where α is tuned by a cheap grid search (and a tiny local refinement).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Determine the length n from the provided parent if possible
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            n = 256
        else:
            n = int(parent.size)
    except Exception:
        n = 256

    # ---- 1. Grid search over the exponent α ---------------------------------
    # Search in a sensible window; α=0.5 gives the classic √‑law, larger α
    # flattens the self‑convolution further.
    best_seq = None
    best_score = float('inf')
    best_alpha = None

    # coarse grid
    for alpha in np.linspace(0.50, 0.90, 41):      # step 0.01
        seq = (np.arange(1, n + 1, dtype=np.float64) ** (-alpha))
        sc = _score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            best_alpha = alpha

    # ---- 2. Small local refinement around the best α -------------------------
    if best_alpha is not None:
        # zoom in a bit
        lo = max(0.40, best_alpha - 0.05)
        hi = min(0.99, best_alpha + 0.05)
        for alpha in np.linspace(lo, hi, 31):      # finer step ~0.003
            seq = (np.arange(1, n + 1, dtype=np.float64) ** (-alpha))
            sc = _score(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                best_alpha = alpha

    # ---- 3. Optional stochastic polishing (time‑bounded) --------------------
    # Simple block‑wise multiplicative perturbations; keep changes only if they
    # improve the score.  The perturbations preserve non‑negativity.
    current_seq = best_seq.copy()
    current_score = best_score

    while time.monotonic() < deadline:
        # pick a random block
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        factor = float(np.exp(rng.normal(0.0, 0.02)))   # modest log‑normal tweak
        cand = current_seq.copy()
        cand[start:stop] *= factor
        # clip to the allowed range (no effect for our power‑law but guard anyway)
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = _score(cand)
        if np.isfinite(cand_score) and cand_score < current_score:
            current_seq = cand
            current_score = cand_score
            # keep the global best as well
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score

    # Return the best sequence found (as a plain Python list of floats)
    return [float(x) for x in best_seq]
