import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Generate a non‑negative step‑function that minimizes the AC1 upper bound.
    The construction uses a power‑law decay a_i = (i+1)^(-p) which yields a
    flat self‑convolution and therefore a small score.  Several exponents are
    tried and the best result (according to the authoritative evaluator) is
    returned.  The routine respects the time budget and falls back to a simple
    uniform vector if something goes wrong.
    """
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Length of the problem – use the provided parent if present.
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 2000                     # safe default when no parent is given

    best_seq = None
    best_score = float('inf')

    # Exponents to try; values >0.5 give a flat convolution.
    exponents = [0.55, 0.60, 0.65, 0.70, 0.75, 0.80,
                 0.85, 0.90, 0.95, 0.99]

    # Helper: evaluate a candidate sequence.
    def eval_seq(arr):
        return float(evaluate_sequence(arr.tolist()))

    # Main loop – try each exponent while we have time.
    for p in exponents:
        if time.monotonic() >= deadline:
            break
        # a_i = (i+1)^{-p}
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = np.power(idx, -p)
        # a is already non‑negative and well below the 1000 ceiling.
        try:
            score = eval_seq(a)
        except Exception:
            continue
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a

    # If for some reason nothing was successful, fall back to a uniform vector.
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)

    # Return a plain Python list of floats as required.
    return [float(x) for x in best_seq]
