import time
import numpy as np

def _linear_ramp(n: int) -> np.ndarray:
    """Return a non‑negative linear ramp of length n with maximum 1000."""
    # a_i = (i+1) * (1000 / n), i = 0..n-1  →  a_{n-1}=1000
    scale = 1000.0 / n
    return np.arange(1, n + 1, dtype=np.float64) * scale


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) perturbation: scale a random contiguous block."""
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12 (as before)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative log‑normal perturbation
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional mirroring for extra variety
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # keep within allowed bounds
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step‑function that (empirically) lowers the AC1 upper bound.

    The construction starts from the optimal linear ramp a_i = (i+1)*(1000/n).
    If a parent sequence is provided (height_sequence_1) it is kept as a fallback.
    A quick stochastic refinement is performed while time permits.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ----------------------------------------------------------------------
    # Determine problem length (use the parent if it exists and looks sane)
    # ----------------------------------------------------------------------
    parent = None
    parent_arr = globals().get("height_sequence_1")
    if isinstance(parent_arr, np.ndarray) and parent_arr.ndim == 1 and parent_arr.size > 0:
        tmp = np.clip(parent_arr.astype(np.float64), 0.0, 1000.0)
        if np.isfinite(tmp).all():
            parent = tmp

    if parent is None:
        # safe fallback length
        n = 256
        parent = np.ones(n, dtype=np.float64)
    else:
        n = parent.size

    # ----------------------------------------------------------------------
    # Baseline candidate: linear ramp (theoretical near‑optimal shape)
    # ----------------------------------------------------------------------
    best = _linear_ramp(n)
    best_score = float(evaluate_sequence(best.tolist()))

    # Compare with the provided parent – keep the better one as fallback
    parent_score = float(evaluate_sequence(parent.tolist()))
    if parent_score < best_score:
        best = parent.copy()
        best_score = parent_score

    # ----------------------------------------------------------------------
    # Light‑weight stochastic improvement (optional, respects time limit)
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = _make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return as plain Python list of floats
    return [float(v) for v in best]
