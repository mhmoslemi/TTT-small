def propose_candidate(seed=42,budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-0.1)

    # get base length
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
        n = base.size
    except Exception:
        n = 256
    # initialize uniform sequence
    a = np.ones(n, dtype=np.float64)
    # Optionally adjust to have sum=1 (not needed)
    # Keep best
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Helper functions for convolution and score
    def compute_score(x):
        # compute convolution via FFT
        nloc = x.size
        nfft = 1 << ((2*nloc-1).bit_length())
        fx = np.fft.rfft(x, n=nfft)
        conv = np.fft.irfft(fx * fx, n=nfft)[:2*nloc-1]
        maxb = float(conv.max())
        s = float(x.sum())
        if s < 0.01:
            return float('inf'), maxb
        return 2.0 * nloc * maxb / (s*s), maxb

    # Precompute scale to avoid recomputing nfft each change? n constant.

    while time.monotonic() < deadline:
        # propose random transfer
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # choose delta as fraction of source
        delta = a[i] * 0.01 * rng.random()
        if delta <= 0:
            continue
        # ensure not moving to below zero
        a[i] -= delta
        a[j] += delta
        # ensure within bounds 0-1000, though values remain small
        # compute score

import numpy as np
import time

def _convolution_max(arr: np.ndarray, nfft: int) -> float:
    """Return max of the linear convolution of arr with itself using FFT."""
    freqs = np.fft.rfft(arr, n=nfft)
    conv = np.fft.irfft(freqs * freqs, n=nfft)[: 2 * arr.size - 1]
    return float(conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Simple hill‑climbing search for a sequence that lowers the AC1 bound.
    Starts from a uniform (all‑ones) sequence of the same length as
    ``height_sequence_1`` (or 256 if unavailable) and repeatedly moves a
    tiny amount of mass from one position to another while keeping the total
    sum unchanged.  The best sequence seen before the time budget expires is
    returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # -----------------------------------------------------------------
    # Determine problem size (number of coefficients)
    # -----------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        n = 256            # fallback length
    # -----------------------------------------------------------------
    # Start from a uniform sequence; scaling does not affect the score.
    # -----------------------------------------------------------------
    cur = np.ones(n, dtype=np.float64)
    best = cur.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Pre‑compute FFT length (next power of two ≥ 2n‑1)
    nfft = 1 << ((2 * n - 1).bit_length())

    # -----------------------------------------------------------------
    # Helper: compute objective quickly (without the evaluator's safety
    # checks, which are already satisfied for our candidates).
    # -----------------------------------------------------------------
    def _score(arr: np.ndarray) -> float:
        maxb = _convolution_max(arr, nfft)
        s = float(arr.sum())
        # sum is guaranteed >0.01 for our candidates
        return 2.0 * n * maxb / (s * s)

    # -----------------------------------------------------------------
    # Main stochastic improvement loop
    # -----------------------------------------------------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move a tiny fraction of mass from i to j
        delta = cur[i] * 0.008 * rng.random()   # ≤0.8 % of source
        if delta <= 0.0:
            continue
        cur[i] -= delta
        cur[j] += delta
        # keep within the required range (values stay tiny, but we clip anyway)
        np.clip(cur, 0.0, 1000.0, out=cur)

        new_score = _score(cur)
        # accept only genuine improvements; otherwise revert
        if new_score < best_score and np.isfinite(new_score):
            best = cur.copy()
            best_score = new_score
        else:
            # revert the move
            cur[i] += delta
            cur[j] -= delta

    # Return the best candidate found (as plain Python floats)
    return [float(v) for v in best]
