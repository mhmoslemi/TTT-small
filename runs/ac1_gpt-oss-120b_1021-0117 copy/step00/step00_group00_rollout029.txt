import numpy as np, time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # init
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    a = parent.copy()
    n = a.size

    # evaluate
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # precompute FFT size
    L = 1 << ((2*n-1).bit_length())
    # define function to compute conv and max
    def conv_and_max(arr):
        fft = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fft*fft, n=L)[:2*n-1]
        max_idx = int(np.argmax(conv))
        return conv, conv[max_idx], max_idx

    # main loop
    iter_count = 0
    while time.monotonic() < deadline:
        # compute conv
        conv, max_b, max_idx = conv_and_max(a)
        sum_a = a.sum()
        if sum_a < 1e-12:
            # avoid degenerate
            a = np.maximum(a, 1e-6)
            sum_a = a.sum()
        # gradient descent step
        factor = 4.0 * n / (sum_a * sum_a)
        const = 4.0 * n * max_b / (sum_a * sum_a * sum_a)

        grad = np.full_like(a, -const)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        # compute partner values vectorized
        idx = max_idx - np.arange(i_start, i_end + 1)
        grad[i_start:i_end+1] = factor * a[idx] - const

        # learning rate schedule: start larger then decay
        # simple: lr = 0.5 * sum_a / n  (scale of average)
        avg = sum_a / n
        lr = 0.04 * avg  # 4% of average per step
        # limit step magnitude
        step = - lr * grad
        # if step too large clip
        max_step = 0.5 * avg
        max_abs = np.max(np.abs(step))
        if max_abs > max_step:
            step = step * (max_step / max_abs)
        candidate = a + step
        # clip
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        # occasionally add random perturbation to escape
        if rng.random() < 0.1:
            # block perturbation similar to previous
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.15)))
            if rng.random() < 0.3:
                candidate = 0.5 * (candidate + candidate[::-1])
            np.clip(candidate, 0.0, 1000.0, out=candidate)

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate.copy()
            best_score = score
            a = candidate  # accept as new state
        else:
            # accept gradient step anyway with probability maybe if not too bad
            if score < best_score * 1.001 and np.isfinite(score):
                a = candidate  # move even if not best
            else:
                # else keep old a
                pass

        iter_count += 1

    # return best found
    return [float(x) for x in best]

Define function compute_score(a):
    conv,maxb,_ = conv_and_max(a)
    score = 2 * n * maxb / (s**2)

Define function gradient_step(a, lr):
    compute conv, maxb, max_idx.
    sum_a = a.sum()
    factor = 4.0 * n / (sum_a * sum_a)
    const = 4.0 * n * maxb / (sum_a * sum_a * sum_a)
    grad = np.full_like(a, -const)
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)
    idx_arr = max_idx - np.arange(i_start, i_end+1)
    grad[i_start:i_end+1] = factor * a[idx_arr] - const
    # update a: a_new = a - lr * grad
    a_new = a - lr * grad
    np.clip(a_new, 0.0, 1000.0, out=a_new)
    return a_new

import numpy as np
import time

def _conv_max(arr, L):
    """Convolution of arr with itself via FFT and return (full, max, argmax)."""
    fft = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * len(arr) - 1]
    idx = int(np.argmax(conv))
    return conv, conv[idx], idx


def _gradient_step(arr, n, L, lr):
    """One sub‑gradient step to reduce 2*n*max/(sum**2)."""
    conv, maxb, maxk = _conv_max(arr, L)
    s = arr.sum()
    if s == 0.0:
        return arr  # should never happen
    factor = 4.0 * n / (s * s)
    const = 4.0 * n * maxb / (s * s * s)

    grad = np.full_like(arr, -const)

    i0 = max(0, maxk - (n - 1))
    i1 = min(n - 1, maxk)
    idx = maxk - np.arange(i0, i1 + 1)
    grad[i0 : i1 + 1] = factor * arr[idx] - const

    new = arr - lr * grad
    np.clip(new, 0.0, 1000.0, out=new)
    return new


def _block_perturb(arr, rng):
    """A cheap local random perturbation (like the reference implementation)."""
    cand = arr.copy()
    n = cand.size
    w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    st = int(rng.integers(0, max(1, n - w + 1)))
    en = min(n, st + w)
    cand[st:en] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
