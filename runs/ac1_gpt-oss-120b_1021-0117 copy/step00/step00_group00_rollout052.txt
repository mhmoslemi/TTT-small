   except Exception:
       default_n = 4000

   best_seq = np.clip(parent_arr, 0.0, 1000.0) if parent_arr.size>0 else np.ones(default_n, dtype=np.float64)
   best_score = float(evaluate_sequence(best_seq.tolist()))

   # Helper: compute score quickly via FFT convolution.
   def compute_score(arr):
       n = arr.size
       total = arr.sum()
       if total <= 0.01:
           return float('inf')
       # length for FFT under power of two >= 2*n-1
       m = 1 << ((2*n - 2).bit_length())
       # compute convolution via FFT
       fa = np.fft.rfft(arr, m)
       conv = np.fft.irfft(fa * fa, m)[:2*n-1]
       maxb = conv.max()
       # ensure nonnegative
       if maxb < 0:
           maxb = 0.0
       return 2.0 * n * maxb / (total*total)

   # Random search loop
   while time.monotonic() < deadline:
       # random length: choose within [int(default_n*0.5), default_n] perhaps
       # small chance to go bigger if allowed? can't exceed parent length? No restriction.

    Generate a non‑negative step‑function (list of heights) that
    minimizes the AC1 upper bound evaluated by `evaluate_sequence`.
    A simple random‑search with FFT‑based convolution is used;
    the best sequence found before the deadline is returned.
    """
    rng = np.random.default_rng(seed)
    # Leave a small safety margin before the sandbox timeout.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Use the supplied parent if it exists, otherwise a simple fallback.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    best_arr = parent.copy()
    # authoritative score for the fallback
    best_score = float(evaluate_sequence(best_arr.tolist()))

    # ------------------------------------------------------------------
    # Helper: compute the AC1 bound using an FFT‑based (linear) convolution.
    # ------------------------------------------------------------------
    def _score(arr: np.ndarray) -> float:
        n = arr.shape[0]
        tot = arr.sum()
        if tot <= 0.01:
            return float('inf')
        # Length for a linear convolution via zero‑padding.
        m = 1 << ((2 * n - 2).bit_length())   # next power of two ≥ 2n‑1
        f = np.fft.rfft(arr, m)
        conv = np.fft.irfft(f * f, m)[: 2 * n - 1]
        maxb = conv.max()
        if maxb < 0:
            maxb = 0.0
        return 2.0 * n * maxb / (tot * tot)

    # ------------------------------------------------------------------
    # Random search loop – generate many candidates of varying length
    # and distribution, keep the best.
    # ------------------------------------------------------------------
    # Minimal length to keep the problem non‑trivial.
    min_len = max(200, parent.size // 4)

    while time.monotonic() < deadline:
        # random length in a reasonable window
        n = int(rng.integers(min_len, parent.size + 1))
        # pick a distribution type
        dist = int(rng.integers(0, 4))
        if dist == 0:               # uniform on [0,1)
            cand = rng.random(n)
        elif dist == 1:             # exponential, normalized
            cand = rng.exponential(scale=1.0, size=n)
            cand = cand / cand.max()
        elif dist == 2:             # Beta(0.5,0.5) – U‑shaped
            cand = rng.beta(0.5, 0.5, size=n)
        else:                       # clipped normal
            cand = rng.normal(0.5, 0.2, size=n)
            cand = np.clip(cand, 0.0, 1.0)

        sc = _score(cand)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_arr = cand.copy()
            # early exit if we have already beaten the target bound
            if best_score < 1.50274349:
                break

    # Return the best known candidate as a plain Python list.
    return [float(x) for x in best_arr]
