    Theoretical analysis shows the AC1 upper bound 2 · n·∑a_i²/(∑a_i)²
    is minimized (and equals 2) by any uniform vector, independent of length.
    This implementation builds such a vector, using the provided
    `height_sequence_1` when available, otherwise a single entry.
    """
    # Try to use the supplied parent as a source of total mass.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        total = float(parent.sum())
        n = parent.size
        if total <= 0.0:
            # fallback to a single‑element uniform vector
            candidate = np.array([1.0], dtype=np.float64)
        else:
            # Uniform distribution preserving the total L1‑mass of the parent.
            uniform_val = total / n
            # Clip to the evaluator's admissible range.
            uniform_val = max(0.0, min(1000.0, uniform_val))
            candidate = np.full(n, uniform_val, dtype=np.float64)
    except Exception:
        # No parent available – use the minimal non‑empty uniform vector.
        candidate = np.array([1.0], dtype=np.float64)

    # Return a plain Python list of floats as required.
    return [float(v) for v in candidate]
