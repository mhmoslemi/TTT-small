import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as evaluated by the provided ``evaluate_sequence``.
    The theoretical minimum of the bound is 2, attained by any constant vector.
    This implementation returns a constant vector (fallback) and performs a
    brief random block perturbation search within the allocated time budget,
    keeping the best result found.
    """
    rng = np.random.default_rng(seed)
    # Reserve a small margin for the final return
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    # Determine a reasonable length.
    # Prefer the provided parent (height_sequence_1) if it is valid.
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim == 1 and parent_arr.size > 0 and np.isfinite(parent_arr).all():
            n = parent_arr.size
        else:
            raise ValueError
    except Exception:
        n = 256                     # fallback length

    # Start from a constant vector – this yields the theoretical lower bound 2.
    const_val = 1.0                # any positive value works
    best = np.full(n, const_val, dtype=np.float64)

    # Evaluate once (authoritative score)
    best_score = float(evaluate_sequence(best.tolist()))

    # Simple stochastic block perturbations while time permits.
    while time.monotonic() < deadline:
        # Pick a random contiguous block and scale it slightly.
        width = max(1, int(rng.integers(1, max(2, n // 20))))
        start = int(rng.integers(0, max(1, n - width + 1)))
        factor = float(np.exp(rng.normal(0.0, 0.01)))   # close to 1
        candidate = best.copy()
        candidate[start:start + width] *= factor
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        # Authoritative evaluation
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score

    # Return a plain Python list of floats
    return [float(x) for x in best]
