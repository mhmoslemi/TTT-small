def compute_conv_and_stats(a):

    size = 1 << ((2*n-1).bit_length())

def compute_gradient(a, p=4):
    n = a.shape[0]
    size = 1 << ((3*n-1).bit_length())
    # convolution a*a
    A = np.fft.rfft(a, size)
    B = A * A
    b = np.fft.irfft(B, size)[:2*n - 1]
    w = b ** (p-1)  # shape (2n-1)
    # correlation of a and w
    # using complex fft for correlation
    A_c = np.fft.fft(a, size)
    W_c = np.fft.fft(w, size)
    corr = np.fft.ifft(np.conj(A_c) * W_c).real
    grad = 2.0 * corr[:n]  # valid for i=0..n-1
    return grad, b

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time

    rng = np.random.default_rng(seed)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Determine n
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim != 1 or arr.size == 0:
            raise Exception
        n = arr.size
        # Use fallback uniform if we want.
    except Exception:
        n = 256
    # Initialize sequence
    a = np.full(n, 1.0 / n, dtype=np.float64)  # uniform
    # Keep best
    best_a = a.copy()
    best_score = float('inf')
    # Compute initial score
    def compute_score_vec(v):
        sv = v.sum()
        if sv < 1e-8:
            return np.inf
        # convolution via FFT for speed
        size = 1 << ((2*n - 1).bit_length())
        A = np.fft.rfft(v, size)
        B = A * A
        b = np.fft.irfft(B, size)[:2*n-1]
        max_b = b.max()
        return 2.0 * n * max_b / (sv * sv)
    # Evaluate initial
    best_score = compute_score_vec(best_a)
    # Optimization parameters
    p = 4  # smoothing power
    alpha = 0.3
    # Precompute FFT size for gradient (3n)
    size_grad = 1 << ((3*n - 1).bit_length())
    iter_count = 0
    while time.monotonic() < deadline:
        # Compute convolution for gradient
        A = np.fft.fft(a, size_grad)
        B = A * A  # convolution via freq domain
        b_vals = np.fft.ifft(B).real[:2*n-1]  # linear convolution length
        # Ensure nonnegative small errors
        b_vals = np.maximum(b_vals, 0.0)
        w = b_vals ** (p - 1)
        # correlation grad
        W = np.fft.fft(w, size_grad)
        corr_full = np.fft.ifft(np.conj(A) * W).real
        grad = 2.0 * corr_full[:n]  # gradient (positive)
        # Update candidate via inverse gradient mixing
        inv = 1.0 / (grad + 1e-12)
        a_candidate = (1 - alpha) * a + alpha * inv
        # Keep nonnegative and renorm
        a_candidate = np.maximum(a_candidate, 0.0)
        total = a_candidate.sum()
        if total <= 0:
            a_candidate = np.full_like(a, 1.0 / n)
            total = 1.0
        a_candidate /= total
        # Evaluate
        score = compute_score_vec(a_candidate)
        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()
        # Accept candidate as new state for next iteration (always move)
        a = a_candidate
        iter_count += 1
        # Occasionally randomize alpha / perturb to escape local minima
        if iter_count % 500 == 0:
            # small random perturbation
            perturb = rng.normal(0, 0.01, size=n)
            a = np.clip(a + perturb, 0, None)
            a = a / a.sum()
    # Return best candidate as list
    return [float(x) for x in best_a]

def pair_reduction_step(a):

   n = len(a)

   # compute convolution using FFT for speed.

   size = conv_size (precomputed: next pow of 2 >= 2n-1)

   A = np.fft.rfft(a, size)

   B = A * A

   b = np.fft.irfft(B, size)[:2*n-1]

   # find k_max

   k = int(np.argmax(b))

   # compute i range:

   i_start = max(0, k - (n-1))

   i_end = min(n-1, k)

   i_vals = np.arange(i_start, i_end+1)

   j_vals = k - i_vals

   # compute contributions

   contrib = a[i_vals] * a[j_vals]

   idx = int(np.argmax(contrib))

   i = i_vals[idx]; j = j_vals[idx]

   # Determine delta: we could remove fraction f of min(a_i, a_j)

   return a

import time
import numpy as np

def _convolution_fft(x, size):
    """Linear convolution of x with itself using FFT of given size."""
    X = np.fft.fft(x, size)
    Y = X * X
    conv = np.fft.ifft(Y).real[: 2 * x.size - 1]
    # numerical noise may produce tiny negatives
    return np.maximum(conv, 0.0)
