    try:

        n = 256
        seq = np.linspace(0.0, 1000.0, n, dtype=np.float64)

    # Ensure in bounds
    seq = np.clip(seq, 0.0, 1000.0)
    best = seq
    best_score = float(evaluate_sequence(best.tolist()))

    # simple local search.
    while time.monotonic() < deadline:
        # perturb a random contiguous block of width between n//200 and n//12
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        cand = best.copy()
        scale = np.exp(rng.normal(0.0, 0.12))
        cand[start:start+width] *= scale
        # optional mirror
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])
        # clip
        np.clip(cand, 0.0, 1000.0, out=cand)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
    return [float(x) for x in best]

import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound evaluated by `evaluate_sequence`.

    The construction starts from a linear ramp (0 → 1000) which attains the
    theoretical score 4/3·(n‑2)/(n‑1) ≈ 1.33 for n≈6000 and then performs a
    cheap local‑search to possibly improve it further within the time budget.
    """
    rng = np.random.default_rng(seed)
    # Give the evaluator a small safety margin before the hard timeout.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Determine the length of the required sequence.
    try:
        n = int(len(height_sequence_1))
        if n <= 0:
            raise Exception
        # Linear ramp from 0 to 1000 (inclusive).
        best = np.linspace(0.0, 1000.0, n, dtype=np.float64)
    except Exception:
        # Fallback length if the parent is unavailable.
        n = 256
        best = np.linspace(0.0, 1000.0, n, dtype=np.float64)

    # Clip to the evaluator's limits (redundant for the ramp but safe).
    np.clip(best, 0.0, 1000.0, out=best)

    # Initial score (authoritative).
    best_score = float(evaluate_sequence(best.tolist()))

    # Simple block‑perturbation local search.
    while time.monotonic() < deadline:
        # Random block width between n/200 and n/12.
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        cand = best.copy()

        # Random multiplicative scaling of the chosen block.
        scale = float(np.exp(rng.normal(0.0, 0.12)))
        cand[start:start + width] *= scale

        # Occasionally enforce (approximate) symmetry.
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])

        # Enforce the hard constraints.
        np.clip(cand, 0.0, 1000.0, out=cand)

        # Evaluate the candidate.
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return a plain Python list as required.
    return [float(x) for x in best]
