                break

            dest = (i_max + 1) % n
            if dest == j_max:

        eps = eps_factor * rng.uniform(min_eps, max_eps)
        delta_i = eps * a[i_max]
        delta_j = eps * a[j_max] if j_max != i_max else 0.0
        # Ensure we don't move more than available
        # (delta_i <= a[i_max]) guaranteed by eps<1
        candidate = a.copy()
        candidate[i_max] -= delta_i
        candidate[dest] += delta_i
        if j_max != i_max:
            candidate[j_max] -= delta_j
            candidate[dest] += delta_j

        # Clip to bounds
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        # Compute score
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            a = candidate  # continue from improved solution
        else:
            # optionally accept with small probability early on to escape
            # simple annealing schedule: if random < 0.001, accept
            if rng.random() < 0.001:
                a = candidate
        iter_count += 1
        # Periodically reassign a = best to ensure we keep best
        if iter_count % 100 == 0:
            a = best.copy()
    return [float(x) for x in best]
