def generate_sidon_set(n, target_m, rng):
    sums_used = np.zeros(2*n-1, dtype=bool) maybe bitset.
    chosen = []
    attempts = 0
    while len(chosen) < target_m and attempts < 10000:
        candidate = rng.integers(0, n)
        # Check collisions:
        conflict=False
        for x in chosen:
            s = candidate + x
            if sums_used[s]:
                conflict=True
                break
        s_self = candidate+candidate
        if sums_used[s_self]:
            conflict=True
        if not conflict:
            # Add
            for x in chosen:
                sums_used[candidate + x] = True
            sums_used[s_self]=True
            chosen.append(candidate)
        attempts+=1
    return chosen

import numpy as np, time

def arcsine_initial(n):
    i = np.arange(n, dtype=np.float64)
    a = 1.0/np.sqrt(i+1.0)
    return a

def perturb(block, rng, sigma):
    candidate = block.copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, sigma)))
    candidate[start:start+width] *= factor

    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

import numpy as np
import time

def _score(seq: np.ndarray) -> float:
    """Fast computation of the evaluator's score (ignoring its safety checks)."""
    total = seq.sum()
    if total == 0.0:
        return np.inf
    conv = np.convolve(seq, seq)
    maxb = conv.max()
    n = seq.size
    return float(2.0 * n * maxb / (total * total))


def _initial_arc(n: int) -> np.ndarray:
    """Arc‑sine inspired start: a_i = 1/sqrt(i+1)."""
    idx = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(idx + 1.0)
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a height sequence with a low AC1 bound.

    The routine starts from the analytic 1/√i seed and performs cheap,
    memory‑bounded local moves (targeted mass transfers and block scalings)
    while keeping the best result seen so far.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # 1.  Determine length and fallback parent (if provided)
    # ------------------------------------------------------------
    try:
        # height_sequence_1 is supplied by the harness (may be missing)
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
    except Exception:
        # No valid parent – choose a modest length
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------
    # 2.  Initial candidates
    # ------------------------------------------------------------
    parent = np.clip(parent, 0.0, 1000.0)
    best = parent.copy()
    best_score = _score(best)

    # analytic start (arc‑sine shape)
    arc = _initial_arc(n)
    arc_score = _score(arc)
    if arc_score < best_score:
        best = arc.copy()
        best_score = arc_score

    # Work on the current best; store the current vector separately
    cur = best.copy()
    cur_score = best_score

    # ------------------------------------------------------------
    # 3.  Search parameters
    # ------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)
    target = 1.50274349   # required upper bound
    # probabilities for the two move types
    prob_targeted = 0.65
    # maximum fraction of a value that may be moved in a single step
    transfer_frac = 0.12
    # block‑scale log‑std
    block_sigma = 0.10

    # ------------------------------------------------------------
    # 4.  Main optimisation loop
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        if rng.random() < prob_targeted:
            # ------- targeted mass transfer ---------------------------------
            conv = np.convolve(cur, cur)
            k_max = int(conv.argmax())

            # indices i that actually contribute to this k
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            i_range = np.arange(i_low, i_high + 1, dtype=np.int64)

            # contribution of each i to the peak
            contrib = cur[i_range] * cur[k_max - i_range]
            tot_contrib = contrib.sum()
            if tot_contrib == 0.0:
                continue

            # pick a heavy‑contributing index (probability ∝ contribution)
            probs = contrib / tot_contrib
            i_choice = int(i_range[rng.choice(i_range.size, p=probs)])

            # pick a random destination index (different from source)
            j_choice = int(rng.integers(0, n))
            if j_choice == i_choice:
                continue

            # amount to move
            delta = cur[i_choice] * rng.uniform(0.0, transfer_frac)
            if delta <= 0.0:
                continue

            cand = cur.copy()
            cand[i_choice] -= delta
            cand[j_choice] += delta
            np.clip(cand, 0.0, 1000.0, out=cand)

        else:
            # ------- block scaling -----------------------------------------
            cand = cur.copy()
            # block length between n/200 and n/12
            wmin = max(1, n // 200)
            wmax = max(2, n // 12)
            width = int(rng.integers(wmin, wmax + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, block_sigma)))
            cand[start:start + width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate candidate
        cand_score = _score(cand)
        if not np.isfinite(cand_score):
            continue

        # keep the best ever seen
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            # early exit if we have already passed the target
            if best_score <= target:
                break

        # move to the candidate if it improves the current state
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score

        # occasional random restart to avoid stagnation
        if rng.random() < 0.005:
            cur = _initial_arc(n)
            cur_score = _score(cur)

    # ------------------------------------------------------------
    # 5.  Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------
    return [float(x) for x in best]
