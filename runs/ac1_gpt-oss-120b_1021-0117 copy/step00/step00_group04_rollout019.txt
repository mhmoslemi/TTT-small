import time
import numpy as np

def _conv_score(arr):
    """Fast computation of the AC1 bound 2·n·max(conv)/sum² using FFT."""
    a = np.asarray(arr, dtype=np.float64)
    n = a.shape[0]
    L = 2 * n - 1                       # length of full linear convolution
    # real FFT of zero‑padded vector
    F = np.fft.rfft(a, n=L)
    # convolution = inverse FFT of squared spectrum
    c = np.fft.irfft(F * F, n=L)
    max_b = np.max(c)
    sum_a = np.sum(a)
    if sum_a <= 0.0:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Generate a non‑negative step sequence that minimizes the AC1 upper bound.
    The algorithm searches for a random binary pattern (0/1) with a density
    that yields a flat self‑convolution, which drives the bound below 2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Determine problem size (length of the required sequence)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
            n = base.size
        else:
            n = 256
    except Exception:
        n = 256

    # ------------------------------------------------------------------
    # Initialise best candidate – start with a uniform vector (bound = 2)
    # ------------------------------------------------------------------
    best = np.ones(n, dtype=np.float64)
    best_score = _conv_score(best)

    # If a parent sequence exists, try it as a fallback.
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size == n and np.isfinite(parent).all():
            parent_score = _conv_score(parent)
            if parent_score < best_score:
                best = parent.copy()
                best_score = parent_score
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Main random search: binary patterns with varying density.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # random density between 0.5 and 0.95 (inclusive)
        p = rng.uniform(0.5, 0.95)
        cand = (rng.random(n) < p).astype(np.float64)

        # guarantee a non‑zero sum (the bound is undefined otherwise)
        if cand.sum() < 0.01:
            continue

        score = _conv_score(cand)
        if score < best_score:
            best = cand
            best_score = score

            # ---------------------------------------------------------
            # Local refinement: toggle a small random block a few times.
            # ---------------------------------------------------------
            for _ in range(5):
                if time.monotonic() >= deadline:
                    break
                # size of block to flip – a small fraction of the length
                block = max(1, n // 200)

                if new.sum() < 0.01:

                new_score = _conv_score(new)
                if new_score < best_score:
                    best = new
                    best_score = new_score

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list of floats.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
