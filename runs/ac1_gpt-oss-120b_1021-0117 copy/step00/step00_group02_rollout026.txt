import time
import numpy as np

# --------------------------------------------------------------
# Helper: project a vector onto the probability simplex {x>=0, sum=1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find rho = max{i | u_i - cssv_i / i > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero -> uniform simplex point
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Helper: full convolution of a vector with itself (linear, not circular)
def _conv_full(v: np.ndarray) -> np.ndarray:
    """Convolution v*v using FFT (size is next power of two)."""
    n = v.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fv = np.fft.rfft(v, size)
    prod = fv * fv
    conv = np.fft.irfft(prod, size)[: 2 * n - 1]
    return conv

        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
        best = parent.copy()
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:
        # a tiny default if no parent is visible
        n = 256
        best = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))

    n = best.shape[0]          # length of the problem

    # ------------------------------------------------------------------
    # 2) Build a pool of analytic candidates
    cand_pool = []

    idx = np.arange(n, dtype=np.float64)

    # basic inverse‑sqrt shape (continuous 1/√x optimum)
    base = 1.0 / np.sqrt(idx + 1.0)
    cand_pool.append(base)

    # symmetric version (U‑shape) – average with reversed copy
    cand_pool.append((base + base[::-1]) * 0.5)

    # varying offsets for the inverse‑sqrt law
    for off in (0.5, 1.0, 2.0, 5.0, 10.0):
        w = 1.0 / np.sqrt(idx + off + 1.0)
        cand_pool.append(w)
        cand_pool.append((w + w[::-1]) * 0.5)

    # mixtures with the uniform distribution (helps break symmetry)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for w in list(cand_pool):
        for lam in (0.05, 0.10, 0.20, 0.30, 0.40):
            mix = (1.0 - lam) * w + lam * uniform
            cand_pool.append(mix)

    # ------------------------------------------------------------------
    # 3) Evaluate all static candidates (no optimizer yet)
    for vec in cand_pool:
        p = vec / vec.sum()                     # put on simplex
        conv = _conv_full(p)
        max_c = conv.max()
        score = 2.0 * n * max_c                  # same as evaluate_sequence
        if score < best_score:
            best = p.copy()
            best_score = score

    # ------------------------------------------------------------------
    # 4) Simple projected sub‑gradient refinement starting from the best so far
    p = best.copy()
    step0 = 0.7                                 # initial step magnitude
    it = 0
    while time.monotonic() < deadline:
        conv = _conv_full(p)
        max_val = conv.max()
        k_star = int(conv.argmax())

        # gradient of the active max entry  (∂/∂p_i  max_c = 2 p_{k*-i})
        L = max(0, k_star - (n - 1))
        R = min(k_star, n - 1)
        idx_range = np.arange(L, R + 1, dtype=np.int64)
        g = np.zeros_like(p)
        g[L : R + 1] = 2.0 * p[k_star - idx_range]

        # diminishing steps; if no progress, shrink step0
        eta = step0 / np.sqrt(it + 1.0)
        trial = _project_simplex(p - eta * g)
        conv_trial = _conv_full(trial)
        max_trial = conv_trial.max()

        if max_trial < max_val - 1e-15:               # improvement
            p = trial
            max_val = max_trial
            score = 2.0 * n * max_val
            if score < best_score:
                best = p.copy()
                best_score = score
        else:
            step0 *= 0.6                               # damp step size

        it += 1
        if best_score <= 1.5028:                      # target reached
            break

    # ------------------------------------------------------------------
    # 5) Return the best vector found (scaled back to a plain list)
    #    The evaluator will clip to [0,1000] and enforce sum≥0.01 automatically.
    return [float(x) for x in best]
