    # normalize to sum 1

    best = a.copy()

            best_score = score2

        # proportion of a[i_high] to move, adapt decreasing as iterations increase
        max_move = a[i_high] * 0.5
        step = min(max_move, 0.02)  # absolute limit
        if step <= 0:
            # random reset - break
            break
        cand = a.copy()
        cand[i_high] -= step
        cand[i_low] += step
        # check nonnegative
        if cand[i_high] < 0:
            continue
        # evaluate
        b2 = np.convolve(cand, cand)
        M2 = np.max(b2)
        score2 = 2 * n * M2  # sum=1
        if score2 < best_score:
            best = cand
            best_score = score2
            a = cand
        else:
            # optional annealing: occasionally accept worse move
            if rng.random() < 1e-4:
                a = cand
        # occasional random large perturbation to escape
        if it % 2000 == 0 and rng.random() < 0.05:
            # random block scaling
            width = int(rng.integers(max(2, n // 100), max(10, n // 10)))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.2)))
            cand2 = a.copy()
            cand2[start:start+width] *= factor
            if np.any(cand2 < 0):
                cand2 = np.clip(cand2, 0, None)
            # renormalize to sum 1
            cand2 = cand2 / np.sum(cand2)
            b2 = np.convolve(cand2, cand2)
            score2 = 2 * n * np.max(b2)
            if score2 < best_score:
                best = cand2
                best_score = score2
                a = cand2
        it += 1
    return [float(x) for x in best]

import time
import numpy as np

def _conv_max(seq: np.ndarray) -> float:
    """Return max entry of the full convolution of seq with itself."""
    return float(np.max(np.convolve(seq, seq)))

def _score(seq: np.ndarray) -> float:
    """Exact objective used by the evaluator (scale‑invariant)."""
    n = seq.shape[0]
    max_b = _conv_max(seq)
    s = np.sum(seq)
    if s == 0.0:
        return float("inf")
    return 2.0 * n * max_b / (s * s)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a non‑negative step‑function that lowers the AC1 bound.

    The routine works with O(n) memory and respects the time budget.
    It uses a simple greedy/annealing scheme driven by the active

    if a_sum == 0.0:

    # keep the best solution seen so far
