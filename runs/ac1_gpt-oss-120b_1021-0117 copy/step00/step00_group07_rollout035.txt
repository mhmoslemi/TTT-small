    g = np.zeros_like(a)
    g[i] = 2.0 * a[j]                     # each ordered pair counted twice
    diag = (i == j)                       # diagonal terms counted once
    if np.any(diag):
        idx = i[diag]
        g[idx] = a[idx]                   # replace 2*a with a for i=j
    return g


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a non‑negative step function with a low AC1 bound.

    The algorithm starts from the supplied ``height_sequence_1`` (or a
    uniform fallback), normalises it to unit sum (the objective is
    scale‑invariant) and then performs a cheap projected sub‑gradient
    descent that repeatedly moves mass from indices that contribute most
    to the current maximal convolution value toward indices that contribute
    least.  Random block perturbations are interleaved to avoid stagnation.
    The best candidate found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------ #
    #   initialise sequence                                            #
    # ------------------------------------------------------------------ #
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:                      # height_sequence_1 not supplied
        base = np.ones(256, dtype=np.float64)

    n = base.size
    a = np.clip(base, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s                           # unit‑sum normalisation

    # safety – evaluator rejects sums < 0.01
    if a.sum() < 0.01:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------ #
    #   main optimisation loop                                          #
    # ------------------------------------------------------------------ #
    step_frac = 0.10            # fraction of a[src] moved each iteration
    min_step = 1e-4

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # current maximal convolution entry
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))

        # gradient of that entry (how each a[i] participates)
        g = _grad_contrib(a, k_max)

        # source = index with largest contribution, target = smallest
        src = int(np.argmax(g))
        if g[src] <= 0:                     # degenerate – fall back to jitter
            jitter = rng.normal(scale=0.02, size=n)
            cand = a + jitter
            cand = np.clip(cand, 0.0, 1000.0)
            cand /= cand.sum()
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_seq, best_score, a = cand.copy(), sc, cand
            continue

        tgt = int(np.argmin(g))
        if src == tgt:                      # choose a different target at random
            cand_idx = np.arange(n)
            cand_idx = cand_idx[cand_idx != src]
            tgt = int(rng.choice(cand_idx))

        delta = step_frac * a[src]          # amount of mass to move
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta
        a_new[src] = max(a_new[src], 0.0)    # non‑negativity
        a_new /= a_new.sum()                # re‑normalise (scale invariant)

        sc = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_seq, best_score, a = a_new.copy(), sc, a_new
            step_frac = max(step_frac, 0.10)    # reset step size after improvement
        else:
            step_frac = max(step_frac * 0.99, min_step)

        # occasional random block perturbation to escape local minima
        if it % 250 == 0:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            blk = a.copy()
            blk[start:stop] *= factor
            blk = np.clip(blk, 0.0, 1000.0)
            blk /= blk.sum()
            sc_blk = float(evaluate_sequence(blk.tolist()))
            if np.isfinite(sc_blk) and sc_blk < best_score:
                best_seq, best_score, a = blk.copy(), sc_blk, blk
                step_frac = max(step_frac, 0.10)

    # ------------------------------------------------------------------ #
    #   return the best candidate found                                 #
    # ------------------------------------------------------------------ #
    return [float(x) for x in best_seq]
