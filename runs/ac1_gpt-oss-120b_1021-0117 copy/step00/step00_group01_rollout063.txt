import numpy as np

def _apply_delta(a: np.ndarray, b: np.ndarray, idx: int, delta: float) -> None:
    """
    Apply an additive change `delta` to a[idx] and update the full convolution
    a*a (stored in b) incrementally.  The operation is exact for ordered
    convolution and respects the quadratic term a_i^2.
    """
    n = a.shape[0]
    # update all positions that involve a[idx] as the first factor
    b[idx : idx + n] += 2.0 * delta * a
    # quadratic correction for the self‑product a_idx * a_idx
    b[2 * idx] += delta * delta
    a[idx] += delta


def _score(b: np.ndarray, sum_a: float, n: int) -> float:
    """Return the AC1 upper bound for the current vector."""
    max_b = b.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))

            a = np.clip(a, 0.0, 1000.0)
            # re‑normalise to keep total mass constant
            a /= a.sum()
            sum_a = a.sum()
            b = np.convolve(a, a)
            cur_score = _score(b, sum_a, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # Return the best vector found (as plain Python floats)
    return [float(v) for v in best_a]
