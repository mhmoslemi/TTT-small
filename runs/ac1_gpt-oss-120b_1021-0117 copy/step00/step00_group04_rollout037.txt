def improve(a, rng, max_iter, deadline):
    a = a.copy()
    # Normalize sum to 1
    a = a / a.sum()
    best_metric = evaluate_sequence(a.tolist())
    best_a = a.copy()
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        # compute convolution
        b = np.convolve(a, a)
        s = np.argmax(b)
        # compute gradient for that s
        low = max(0, s - (len(a)-1))
        high = min(len(a)-1, s)
        idx = np.arange(low, high+1)
        g = np.zeros_like(a)
        g[idx] = 2 * a[s - idx]
        # step size
        eta = 0.1 / np.sqrt(it+1)  # can adjust
        a = a - eta * g
        # add small random noise to avoid stagnation?
        # projection to nonnegative
        a = np.clip(a, 0, None)
        # renormalize to sum=1
        sum_a = a.sum()
        if sum_a <= 0:
            # reinitialize randomly
            a = rng.random(len(a))
            a = a / a.sum()
            continue
        a = a / sum_a
        # add small random perturbation to avoid local minima (optional)
        # compute metric
        metric = evaluate_sequence(a.tolist())
        if metric < best_metric:
            best_metric = metric
            best_a = a.copy()
    return best_a, best_metric
