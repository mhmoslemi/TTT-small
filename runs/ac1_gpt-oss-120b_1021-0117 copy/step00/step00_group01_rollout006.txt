import time
import math
import numpy as np

def _make_candidate(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a simple O(n) perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # Choose a block width roughly between n/200 and n/12
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    # Scale the block by a log‑normal factor
    factor = math.exp(rng.normal(0.0, 0.12))
    cand[start:start + width] *= factor
    # Occasionally symmetrize to keep the shape balanced
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # Enforce the problem bounds
    cand = np.clip(cand, 0.0, 1000.0)
    return cand
