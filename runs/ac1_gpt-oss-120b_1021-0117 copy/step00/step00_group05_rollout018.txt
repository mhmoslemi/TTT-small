def propose_move(a, rng, b, k_max):
    # pick index i randomly but weighted by contribution a_i*a_{k_max-i}
    contributions = np.zeros_like(a)
    low = max(0, k_max - (len(a)-1))
    high = min(len(a)-1, k_max)
    # compute for i in low..high contributions
    for i in range(low, high+1):
        j = k_max - i
        if 0 <= j < len(a):
            contributions[i] = a[i] * a[j]
    # pick i with probability proportional to contributions (or top few)
    total = contributions.sum()
    if total == 0:
        i = rng.integers(len(a))
    else:
        # select i by random weighted sample
        # Use cumulative distribution
        cum = np.cumsum(contributions)
        r = rng.random()*total
        i = np.searchsorted(cum, r)
    # choose j target far from i
    j = rng.integers(len(a))
    # if j==i skip or pick again
    if j == i:
        j = (i+len(a)//2) % len(a)
    # compute delta = min(a[i], fraction)
    max_delta = a[i] * 0.2
    if max_delta <= 0:
        return None
    delta = max_delta * rng.random()
    new = a.copy()
    new[i] -= delta
    new[j] += delta
    new = np.clip(new, 0.0, 1000.0)
    return new
```

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    try:
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
        n = base.size
    except Exception:
        n = 256
        base = np.full(n, 1000.0, dtype=np.float64)
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # grid search over exponential shape
    # Determine lambda range
    # Typically lambda in (0, 5) works; we also add lambda=0 -> uniform
    # exponents: a_i = 1000 * exp(-lam * i / n)
    # Use lambda values spaced more densely near optimum maybe
    for lam in np.linspace(0.1, 6.0, 60):
        if time.monotonic() >= deadline:
            break
        log_factor = -lam / n  # exponent per step
        exponents = np.arange(n, dtype=np.float64) * log_factor
        a = 1000.0 * np.exp(exponents)   # decreasing shape
        # clip is not needed as range within 0..1000
        score = float(evaluate_sequence(a.tolist()))
        if score < best_score:
            best = a
            best_score = score
    # optionally add power shape
    for p in np.linspace(0.1, 5.0, 30):
        if time.monotonic() >= deadline:
            break
        # decreasing power: (1 - i/(n-1))**p
        idx = np.arange(n, dtype=np.float64) / (n - 1)
        a = 1000.0 * (1.0 - idx) ** p
        score = float(evaluate_sequence(a.tolist()))
        if score < best_score:
            best = a
            best_score = score
    # local random pairwise transfer search
    while time.monotonic() < deadline:
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if best[i] <= 0.0:
            continue
        delta = best[i] * rng.uniform(0.01, 0.3)
        # ensure j not exceed bound
        max_add = 1000.0 - best[j]
        if delta > max_add:
            delta = max_add
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure still nonnegative
        if cand[i] < 0.0 or cand[j] > 1000.0:
            continue
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best = cand
            best_score = score
    # final fallback safety: ensure not empty and finite.
    if not np.isfinite(best_score) or best.size == 0:
        # fallback uniform
        best = np.full(n, 1.0, dtype=np.float64)
    return [float(x) for x in best]

        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
        n = base.size
    except Exception:
        n = 256
        base = np.full(n, 1000.0, dtype=np.float64)

    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # FFT length (next power of two ≥ 2n‑1)
    fft_len = 1 << int(math.ceil(math.log2(2 * n - 1)))

    # ----- random parametric shapes -----------------------------------
    while time.monotonic() < deadline:
        # random shape parameters
        p = rng.uniform(0.0, 5.0)          # power on the linear decay
        lam = rng.uniform(0.0, 3.0)        # exponential decay strength

        idx = np.arange(n, dtype=np.float64)

        # combined polynomial‑exponential form
        shape = np.exp(-lam * idx / n) * (1.0 - idx / (n - 1)) ** p
        # force non‑negativity (tiny negatives can appear from rounding)
        shape = np.maximum(shape, 0.0)
        # scale so the maximum equals 1000 (clipping not needed afterwards)
        shape *= 1000.0 / shape.max()

        # cheap evaluation first
        sc = _fast_ac1_score(shape, n, fft_len)
        if sc < best_score:
            # authoritative check
            exact = float(evaluate_sequence(shape.tolist()))
            if exact < best_score:
                best = shape
                best_score = exact

    # ----- local pairwise transfers -----------------------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or best[i] <= 0.0:
            continue
        # move a modest fraction of mass from i to j
        delta = best[i] * rng.uniform(0.01, 0.25)
        max_add = 1000.0 - best[j]
        if delta > max_add:
            delta = max_add
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _fast_ac1_score(cand, n, fft_len)
        if sc < best_score:
            exact = float(evaluate_sequence(cand.tolist()))
            if exact < best_score:
                best = cand
                best_score = exact

    # safety fallback – should never be needed
    if not np.isfinite(best_score) or best_score == np.inf:
        best = np.full(n, 1.0, dtype=np.float64)

    return [float(x) for x in best]
