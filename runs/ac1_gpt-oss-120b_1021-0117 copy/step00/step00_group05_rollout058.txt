import numpy as np
import time
import math

def _score(a, n):
    """Return the AC1 bound for a non‑negative sequence a (sum not assumed 1)."""
    b = np.convolve(a, a)
    max_b = b.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Stochastic local search for a low AC1 bound.
    Returns a list of non‑negative heights (already clipped to [0,1000]).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Length of the problem
    n = int(len(height_sequence_1)) if 'height_sequence_1' in globals() else 256

    # ----- initial candidate -------------------------------------------------
    # start from a random Dirichlet distribution (more diverse than uniform)
    a = rng.gamma(0.5, size=n)          # shape 0.5 → many small, few large values
    if a.sum() == 0:
        a = np.ones(n, dtype=np.float64)
    a /= a.sum()                        # normalise to sum = 1
    cur_score = _score(a, n)

    # keep the best ever seen
    best_a = a.copy()
    best_score = cur_score

    # ----- optional fallback: use the provided parent sequence ----------------
    if 'height_sequence_1' in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent_score = _score(parent, n)
            if parent_score < best_score:
                best_score = parent_score
                best_a = parent.copy()
                cur_score = parent_score
                a = parent.copy()

    # ----- simulated‑annealing style search ---------------------------------
    # temperature schedule
    temp = 0.2
    decay = 0.9995
    # maximal fraction of mass moved in one trial (will shrink with temperature)
    base_move_frac = 0.3

    while time.monotonic() < deadline:
        # pick two distinct indices i (source) and j (target)
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose moving a fraction of a[i] to a[j]
        if a[i] <= 0.0:
            continue
        # step size adapts with temperature
        move_frac = base_move_frac * (temp / 0.2)   # shrink as temp drops
        delta = a[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = _score(a, n)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert the move
            a[i] += delta
            a[j] -= delta

        # cool down
        temp *= decay
        # occasional larger random perturbation to escape plateaus
        if rng.random() < 0.001:
            # add small Gaussian noise and renormalise
            noise = rng.normal(0.0, 0.02, size=n)
            a = np.clip(a + noise, 0.0, None)
            if a.sum() == 0.0:
                a = np.ones(n, dtype=np.float64) / n
            else:
                a /= a.sum()
            cur_score = _score(a, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        # safety: if we already beat the target, we can stop early
        if best_score < 1.5028:
            break

    # final clipping to satisfy evaluator's bounds
    result = np.clip(best_a, 0.0, 1000.0).tolist()
    # ensure a tiny positive sum (evaluator needs sum >= 0.01)
    if sum(result) < 0.01:
        # fall back to a uniform tiny vector
        result = [0.01 / n] * n
    return result
