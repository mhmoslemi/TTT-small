            break

import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) that substantially
    improves the AC1 upper bound.
    The construction is a simple geometric decay:
        a[i] = q**i ,   q = 1 - α/n
    with α chosen near log(n).  This makes the maximum convolution value
    close to 1 while keeping the sum large, yielding a very small score.
    """
    # Time budget handling
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Length of the problem (use the supplied parent if present)
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    # Candidate alphas to test (centered around log(n))
    logn = np.log(n)
    alphas = [logn * r for r in (0.8, 0.9, 1.0, 1.1, 1.2)]

    best_seq = None
    best_score = float('inf')

    # Evaluate each geometric candidate
    for alpha in alphas:
        if time.monotonic() >= deadline:
            break
        eps = alpha / n               # ε = α / n
        q = 1.0 - eps
        if q <= 0.0:
            continue
        # a[i] = q**i  (non‑negative, automatically ≤1)
        seq = np.power(q, np.arange(n, dtype=np.float64))
        seq = np.clip(seq, 0.0, 1000.0)    # ensure within evaluator limits
        score = evaluate_sequence(seq.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq

    # If none succeeded (should not happen), fall back to a simple uniform vector
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())

    # Optional tiny local refinement (few cheap perturbations)
    rng = np.random.default_rng(seed)
    while time.monotonic() < deadline:
        # Randomly scale a short block
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 100))
        stop = min(n, start + width)
        factor = np.exp(rng.normal(0.0, 0.04))
        cand = best_seq.copy()
        cand[start:stop] *= factor
        # Occasionally symmetrize to keep shape balanced

        score = evaluate_sequence(cand.tolist())
        if np.isfinite(score) and score < best_score:
