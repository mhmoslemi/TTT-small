        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        return conv, np.max(conv), int(np.argmax(conv))

    # iterative improvement
    while time.monotonic() < deadline:

        # compute gradient of M w.r.t a
        idx = np.arange(n)

        a_new = np.clip(a_new, 0.0, 1000.0)
        # re-normalize sum (scale to keep sum constant)
        if np.sum(a_new) > 0:
            a_new = a_new / np.sum(a_new) * np.sum(a)  # keep same sum
        else:
            a_new = a  # fallback

        # maybe occasional random jitter
        if rng.random() < 0.1:
            jitter = rng.normal(0.0, 0.1, size=n)

        # compute fast bound

        # Keep if better (based on fast estimate)

            best_a = a_new.copy()
            best_score = score_est

        # assign a = a_new for next iteration

        F = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(F * F, n=L)[:2*n-1]
        idx = int(np.argmax(conv))
        return conv, float(conv[idx]), idx

    # compute initial best

        best = parent.copy()
        best_score = parent_score

    # iteration loop
    while time.monotonic() < deadline:

        # gradient
        i = np.arange(n)
        j = kmax - i
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        # use factor 2 except maybe double-count; ok
        g[mask] = 2.0 * a[j[mask]]

        s = a_new.sum()
        if s > 0:
            a_new = a_new / s * np.sum(a)  # keep same sum as before, e.g., 1
        else:
            a_new = a  # fallback

        # occasional random jitter to escape stagnation
        if rng.random() < 0.05:

            a_new = np.clip(a_new + jitter, 0.0, 1000.0)

    best_score = fast_bound(best_arr)
    # also consider parent as fallback
    if parent_score < best_score:
        best_arr = parent_arr.copy()
        best_score = parent_score
    # search parameters
    sigma = 0.2  # multiplicative log-std
    it = 0
    # main loop
    while time.monotonic() < deadline:
        cand = best_arr.copy()
        # random block perturb
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))

        # compute fast bound
        sc = fast_bound(cand)
        if np.isfinite(sc) and sc < best_score:

    # parent construction, if provided
    try:

        if s <= 0.0:
            return np.inf

        conv = np.fft.irfft(F * F, n=L_fft)[:2 * n - 1]

    # start from a simple two‑block construction (ratio 2 gives bound ≈1.78)

    best = cur.copy()

    if parent_score < best_score:
