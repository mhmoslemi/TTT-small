    rng = np.random.default_rng(seed)

    # also try power law base
    def make_power(p, shift):

        a = 1.0 / np.power(indices + shift, p)

        p = 0.5 + rng.uniform(-0.1, 0.1)
        shift = 10**rng.uniform(-6, 2)  # random offset between 1e-6 and 100
        candidate = make_power(p, shift)
        # optionally apply mirror symmetry
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
        # occasional local perturbation of best
        if rng.random() < 0.2:
            cand = best.copy()
            # random block perturbation (multiplicative)
            width = int(rng.integers(1, max(2, n // 100)))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.05))
            cand[start:start+width] *= factor
            cand = np.clip(cand, 0.0, 1000.0)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
    return [float(x) for x in best]

import numpy as np
import time
import math

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step‑function height sequence that reduces the AC1 bound.
    Uses a simple parametric family a_i = (i+1)^{-p} for a prefix and a constant
    tail, which analytically approaches the known optimum.
    """
    rng = np.random.default_rng(seed)
    # Time budget (leave a couple seconds margin for the return)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Length of the problem sequence (provided by the harness)
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    # Fallback: uniform sequence (score = 2.0)
    best_seq = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # Helper to build the piece‑wise power‑law / constant candidate
    def build_candidate(alpha: float, p: float) -> np.ndarray:
        """alpha ∈ (0,1] fraction of the array that follows the power law."""
        K = max(1, int(round(alpha * n)))
        # power‑law part: a_i = (i+1)^{-p}
        idx = np.arange(K, dtype=np.float64) + 1.0
        early = np.power(idx, -p)
        # constant tail, equal to the last power‑law value for continuity
        tail_val = np.power(K + 1.0, -p)
        tail = np.full(n - K, tail_val, dtype=np.float64)
        return np.clip(np.concatenate([early, tail]), 0.0, 1000.0)

    # Main stochastic search (parameter sampling + occasional local tweaks)
    while time.monotonic() < deadline:
        # Sample parameters around the analytically motivated region
        p = 0.5 + rng.uniform(-0.1, 0.1)          # exponent near 0.5
        alpha = rng.uniform(0.15, 0.85)          # size of the decreasing prefix
        cand = build_candidate(alpha, p)

        # Occasionally smooth by mirroring (helps a bit on some runs)
        if rng.random() < 0.2:
            cand = 0.5 * (cand + cand[::-1])

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score

        # Small local perturbation of the current best (multiplicative noise)
        if rng.random() < 0.15:
            tmp = best_seq.copy()
            width = max(1, int(rng.integers(1, max(2, n // 120))))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.07))
            tmp[start:start + width] *= factor
            tmp = np.clip(tmp, 0.0, 1000.0)
            s2 = float(evaluate_sequence(tmp.tolist()))
            if np.isfinite(s2) and s2 < best_score:
                best_seq = tmp
                best_score = s2

    # Ensure a valid (non‑empty) list is returned
    return [float(x) for x in best_seq]
