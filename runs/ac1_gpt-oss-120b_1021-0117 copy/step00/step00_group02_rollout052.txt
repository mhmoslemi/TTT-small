def simplex_projection(v):
    """Project vector v onto probability simplex (sum=1, v>=0)"""
    # from literature
    v = v - np.max(v) # shift for numeric stability? Not needed. We'll implement.
    # Actually we can implement straightforward:
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last True index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

# ------------------------------------------------------------
# Helper: project a vector onto the probability simplex
# ------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x >= 0, sum x = 1}."""
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – fallback to the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]          # last index where condition holds (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Helper: max of the self‑convolution (via FFT, O(n log n))
# ------------------------------------------------------------
def _conv_max(p: np.ndarray, fft_len: int) -> float:
    """Return max_{s} (p * p)[s] for a non‑negative vector p, sum(p)=1."""
    f = np.fft.rfft(p, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * p.size - 1]
    # due to round‑off some values may be tiny negative numbers
    return float(np.max(conv))

    return [float(x) for x in best.tolist()]
