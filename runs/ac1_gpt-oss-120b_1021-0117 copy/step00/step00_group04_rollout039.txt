    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 1.0
    n = len(height_sequence_1) if known else 1000
    S_target = np.sqrt(2 * n)
    # initialize a with random Dirichlet distribution
    a = rng.random(n)
    a = a / a.sum() * S_target
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        M = conv.max()
        # early exit: if M < best_score maybe update best
        # but score equals M for sum fixed; but given a may have sum close S_target
        score = float(M)  # because sum fixed.
        if score < best_score:
            best = a.copy()
            best_score = score
        # gradient
        # compute mask of near-max indices
        mask = conv >= M * (1 - 1e-12)   # includes values equal to max
        ks = np.where(mask)[0]
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # i range inclusive
            i_range = np.arange(i_start, i_end + 1)
            j = k - i_range
            g[i_range] += a[j]   # contributions
        # factor 2 for off-diagonal, but not needed for direction; we can just use g
        # compute step size proportionally e.g., eta = alpha * S_target / (np.max(g)+1e-12)
        if np.all(g == 0):
            # random restart
            a = rng.random(n)
            a = a / a.sum() * S_target
            continue
        # set step size decreasing over time
        remaining = deadline - time.monotonic()
        # use step size scaling with sqrt(remaining) maybe
        eta = 0.1 * (remaining / budget_s)   # something
        # more rational: base step factor = 0.5 * S_target / (np.max(g) + eps)
        base = S_target / (np.max(g) + 1e-12)
        eta = 0.5 * base * (remaining / budget_s)  # decreasing
        # update a
        a = a - eta * g
        a = np.maximum(a, 0.0)
        # renormalize sum
        sum_a = a.sum()
        if sum_a > 0:
            a = a * (S_target / sum_a)
        else:
            a = rng.random(n)
            a = a / a.sum() * S_target
    return [float(x) for x in best]
