import time
import numpy as np

# -------------------------------------------------------------
# Fast linear convolution using FFT (zero‑padded to next power of two)
# -------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Length of the FFT that can hold the full linear convolution of
    two length‑n arrays."""
    conv_len = 2 * n - 1
    return 1 << (conv_len - 1).bit_length()


def _conv_fft(x: np.ndarray, fft_len: int) -> np.ndarray:
    """Return np.convolve(x, x) using FFT (real input)."""
    X = np.fft.rfft(x, fft_len)
    Y = X * X
    conv = np.fft.irfft(Y, fft_len)[: 2 * x.size - 1]
    # tiny negative round‑off can appear; clip to zero for safety
    return np.maximum(conv, 0.0)


def _ac1_score_fft(x: np.ndarray, n: int, fft_len: int) -> float:
    """Fast version of the AC1 bound used by the authoritative evaluator."""
    conv = _conv_fft(x, fft_len)
    max_b = conv.max()
    s = x.sum()
    return 2.0 * n * max_b / (s * s)

    # -----------------------------------------------------------------
