import time
import math
import numpy as np

def _score(arr, n):
    """Fast evaluation of the objective (without the evaluator’s extra checks)."""
    s = np.sum(arr)
    if s <= 0.0:
        return np.inf
    b = np.convolve(arr, arr)
    return float(2.0 * n * np.max(b) / (s * s))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step‑function with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    # length of the problem (height_sequence_1 is supplied by the harness)
    n = int(np.asarray(height_sequence_1).size) if 'height_sequence_1' in globals() else 256

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1. generate a collection of cheap parametric candidates
    # ------------------------------------------------------------------
    best = None
    best_score = np.inf

    # exponential families
    for r in np.linspace(0.90, 0.9999, 200):
        cand = r ** np.arange(n, dtype=np.float64)
        sc = _score(cand, n)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # power‑law families a_i = (i+1)^(-p)
    for p in np.linspace(0.30, 0.70, 81):
        cand = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        sc = _score(cand, n)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # polynomial‑decay families a_i = (1 - i/(n‑1))**q
    for q in np.linspace(0.3, 5.0, 71):

        sc = _score(cand, n)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    if best is None:
        best = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    current = best.copy()
    current_score = best_score

    # parameters controlling the moves
    early_frac = 0.25          # region considered “early”
    late_frac = 0.75           # region considered “late”
    max_step = 0.12            # max fraction of a[i] that can be moved per trial
    block_prob = 0.2           # probability of a random block perturbation
    block_width_max = max(2, n // 20)

    while time.monotonic() < deadline:
        # occasionally try a small random block perturbation (exploration)
        if rng.random() < block_prob:
            width = int(rng.integers(1, block_width_max + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.08))
            cand = current.copy()
            cand[start:start + width] *= factor
        else:
            # move a little mass from an early index to a late index
            i = int(rng.integers(0, max(1, int(early_frac * n))))
            j = int(rng.integers(int(late_frac * n), n))
            if i == j:
                continue
            delta = current[i] * rng.random() * max_step
            if delta <= 0.0:
                continue
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta

        # respect the hard bounds required by the evaluator
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _score(cand, n)

        # accept only improvements (greedy hill‑climb)
        if sc < current_score:
            current = cand
            current_score = sc
            if sc < best_score:
                best = cand.copy()
                best_score = sc

        # occasional reset to the best found so far (helps escape tiny plateaus)
        if rng.random() < 0.001:
            current = best.copy()
            current_score = best_score

    # ------------------------------------------------------------------
    # 3. return the best candidate as a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best]
