    rng = np.random.default_rng(seed)

    # Length of the problem (use the provided parent if present)
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    # ------------------------------------------------------------------
    # Base distribution: a_i = (i+1)^{-0.5}  (p = 0.5 gives the optimal
    # constant convolution asymptotically).  This already yields a score
    # around π/2 ≈ 1.57.  We can improve it by reducing the centre values
    # a little (parameter k) and then renormalising the sum.
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)
    base = idx ** -0.5                     # a_i = 1/sqrt(i+1)
    sum_base = float(base.sum())

    # Normalised distance from the centre (0 at centre, 1 at the edges)
    if n > 1:
        mid = (n - 1) / 2.0
        dist = np.abs(np.arange(n) - mid) / mid
    else:
        dist = np.zeros(1, dtype=np.float64)

    best_seq = base.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # --------------------------------------------------------------
    # Search over the centre‑reduction parameter k ∈ [0, 0.9].
    #   w_i = (1‑k) + k * dist_i   (edges keep weight 1, centre → 1‑k)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    for k in np.linspace(0.0, 0.9, 13):          # 0,0.075,…,0.9
        if time.monotonic() > deadline:
            break
        # weight vector
        w = (1.0 - k) + k * dist
        cand = base * w
        # renormalise to the original sum (scale‑invariant objective)
        cand *= (sum_base / cand.sum())
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand.copy()

    # --------------------------------------------------------------
    # Small random local perturbations around the current best.
    # This cheap hill‑climbing often squeezes a bit more improvement.
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = best_seq.copy()
        # pick a random block and perturb it multiplicatively
        start = rng.integers(0, n)
        width = int(rng.integers(1, max(2, n // 200)))
        stop = min(n, start + width)
        factor = math.exp(rng.normal(0.0, 0.06))   # modest log‑normal tweak
        cand[start:stop] *= factor
        np.maximum(cand, 0, out=cand)             # ensure non‑negative

        # renormalise to keep the sum unchanged (objective is scale‑free)
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand *= (sum_base / cand_sum)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand.copy()

    # Return as a plain Python list of floats
    return [float(x) for x in best_seq]
