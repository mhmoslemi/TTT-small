def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # get parent
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        base = None
    n = len(base) if base is not None else 256
    # start with either parent or uniform
    if base is None or base.ndim !=1 or base.size==0 or not np.isfinite(base).all():
        # generate uniform shape
        base = np.ones(n, dtype=np.float64)
    # Ensure clipping as per evaluator
    base = np.clip(base, 0.0, 1000.0)

    # Evaluate parent best
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Predefined alphas
    alphas = np.linspace(0.0, 1.0, 21) # 0..1
    for alpha in alphas:
        if time.monotonic() >= deadline: break
        den = (np.arange(n)+1)*(n - np.arange(n))
        # avoid zero denominator (should be >0)
        a = np.power(den, -alpha)
        # scaling to max 1000
        a = a * (1000.0 / np.max(a))
        a = np.clip(a, 0.0, 1000.0)
        score = float(evaluate_sequence(a.tolist()))
        if score < best_score:
            best = a
            best_score = score

    # Hill climb from best
    current = best.copy()
    current_score = best_score

    while time.monotonic() < deadline:
        # compute convolution and max index for current
        b = np.convolve(current, current)
        k_max = int(np.argmax(b))
        # Compute contributions for k_max
        low = max(0, k_max - (n-1))
        high = min(k_max, n-1)
        idx = np.arange(low, high+1)
        partners = k_max - idx
        contributions = current[idx] * current[partners]
        # pick index with largest contribution
        i = idx[np.argmax(contributions)]
        # find partner j correspond (k_max - i)
        # We'll reduce mass from i (or optionally its partner)
        # Choose delta as fraction of current[i] (if >0)
        val = current[i]
        if val <= 0:
            continue
        # random fraction
        frac = rng.uniform(0.01, 0.5)  # avoid zero delta
        delta = val * frac
        # Choose a random target index not equal i
        j = int(rng.integers(0, n))
        while j == i:
            j = int(rng.integers(0, n))
        candidate = current.copy()
        candidate[i] -= delta
        candidate[j] += delta
        # Clip just in case
        # Not needed as delta <= current[i]
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < current_score:
            current = candidate
            current_score = score
            if score < best_score:
                best = candidate
                best_score = score
        # else maybe try small random move to diversify Occasionally
        # Also occasionally random perturbation.
        if rng.random() < 0.01:
            # produce random small perturbation across many indices
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            candidate2 = current.copy()
            factor = np.exp(rng.normal(0.0, 0.12))
            candidate2[start:stop] *= factor
            if rng.random() < 0.35:
                candidate2 = 0.5 * (candidate2 + candidate2[::-1])
            candidate2 = np.clip(candidate2, 0.0, 1000.0)
            score2 = float(evaluate_sequence(candidate2.tolist()))
            if np.isfinite(score2) and score2 < current_score:
                current = candidate2
                current_score = score2
                if score2 < best_score:
                    best = candidate2
                    best_score = score2

    return [float(x) for x in best]

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that (hopefully) beats the 2.0 bound.

    The algorithm:
      1. Initialise from the supplied ``height_sequence_1`` (if any) or a
         uniform vector.
      2. Scan a one‑parameter family a_i = ((i+1)*(n-i))^{-α} (α∈[0,1])
         – after scaling to the allowed height range – and keep the best.
      3. Perform a simple hill‑climb: repeatedly move a small amount of mass
         from an index that contributes most to the current maximal convolution
         entry to a random other index, accepting only improvements.
      4. Return the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------ #
    # 1️⃣  Initial candidate (parent if available, else uniform)
    # ------------------------------------------------------------------ #
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
    except Exception:
        base = None

    if base is None:
        n = 256
        base = np.ones(n, dtype=np.float64)
    else:
        n = base.size

    base = np.clip(base, 0.0, 1000.0)
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------ #
    # 2️⃣  Scan simple U‑shaped families a_i = ((i+1)*(n-i))^{-α}
    # ------------------------------------------------------------------ #
    idx = np.arange(n, dtype=np.float64)
    den = (idx + 1.0) * (n - idx)          # >0 for all i
    alphas = np.linspace(0.0, 1.0, 21)    # 0 → uniform, 1 → very U‑shaped

    for alpha in alphas:
        if time.monotonic() >= deadline:
            break
        # a_i ∝ den^{-α}
        a = np.power(den, -alpha, dtype=np.float64)
        # scale so the largest entry is exactly 1000 (scaling does not affect score)
        a *= 1000.0 / np.max(a)
        a = np.clip(a, 0.0, 1000.0)
        score = float(evaluate_sequence(a.tolist()))
        if score < best_score:
            best = a
            best_score = score

    # ------------------------------------------------------------------ #
    # 3️⃣  Hill‑climb from the best candidate found so far
    # ------------------------------------------------------------------ #
    current = best.copy()
    current_score = best_score

    while time.monotonic() < deadline:
        # full convolution to locate the current maximal entry
        conv = np.convolve(current, current)
        k_max = int(np.argmax(conv))

        # indices i that contribute to this entry (j = k_max - i)
        low = max(0, k_max - (n - 1))
        high = min(k_max, n - 1)
        i_vals = np.arange(low, high + 1, dtype=np.int64)
        j_vals = k_max - i_vals
        contrib = current[i_vals] * current[j_vals]

        # pick the i with the largest product (greedy)
        i = int(i_vals[np.argmax(contrib)])
        if current[i] <= 0.0:
            continue

        # move a random fraction of its mass to a random position
        frac = rng.uniform(0.02, 0.30)           # not too tiny, not too large
        delta = current[i] * frac
        j = int(rng.integers(0, n))
        while j == i:
            j = int(rng.integers(0, n))

        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta
        # (delta ≤ current[i] guarantees non‑negativity)
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < current_score:
            current, current_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand, cand_score
            continue

        # occasional random perturbation to jump out of plateaus
        if rng.random() < 0.015:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            cand2 = current.copy()
            factor = np.exp(rng.normal(0.0, 0.12))
            cand2[start:stop] *= factor
            if rng.random() < 0.35:
                cand2 = 0.5 * (cand2 + cand2[::-1])
            cand2 = np.clip(cand2, 0.0, 1000.0)
            cand2_score = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(cand2_score) and cand2_score < current_score:
                current, current_score = cand2, cand2_score
                if cand2_score < best_score:
                    best, best_score = cand2, cand2_score

    # ------------------------------------------------------------------ #
    # 4️⃣  Return the best sequence found
    # ------------------------------------------------------------------ #
    return [float(x) for x in best]
