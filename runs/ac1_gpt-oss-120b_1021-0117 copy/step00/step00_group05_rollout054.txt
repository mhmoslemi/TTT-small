def make_block_perturbation(base, rng):

    cand = np.asarray(base, dtype=np.float64).copy()

    start = int(rng.integers(0, max(1, n - width + 1)))

    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _random_sparse_candidate(rng, n):
    """Select a modest fraction of indices and give them non‑zero weights."""
    # 8‑20 % of the coordinates – enough to make the average pair count large,
    # but sparse enough that the maximum does not dominate.
    frac = rng.uniform(0.08, 0.20)
    m = max(2, int(frac * n))
    idx = rng.choice(n, size=m, replace=False)

    # Either unit weight or a small random weight (exponential tail).
    if rng.random() < 0.5:

    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return np.clip(a, 0.0, 1000.0)


def _random_shape_candidate(rng, n):
    """Generate a dense vector from a few simple parametric families."""
    typ = rng.integers(0, 5)
