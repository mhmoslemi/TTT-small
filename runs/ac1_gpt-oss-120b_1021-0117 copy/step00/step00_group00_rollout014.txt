def _build_sidon_set(n, target_k, max_allowed, rng):
    """
    Greedy construction of a set of indices S (size target_k) such that
    for the multiset of ordered sums i+j (i,j in S) no value exceeds max_allowed.
    Returns the positions array (size may be < target_k) and the full sum-count array.
    """
    # sum counts for indices 0 .. 2*n-2
    counts = np.zeros(2 * n - 1, dtype=np.int8)
    positions = np.empty(target_k, dtype=np.int32)
    used = np.zeros(n, dtype=np.bool_)
    cur = 0
    attempts = 0
    # generous bound on attempts; reset after each successful insertion
    max_attempts = target_k * 500
    while cur < target_k and attempts < max_attempts:
        i = rng.integers(0, n)
        if used[i]:
            attempts += 1
            continue
        if cur > 0:
            existing = positions[:cur]
            sums = i + existing                       # vector of candidate sums
            # each new unordered pair adds 2 ordered contributions
            if np.any(counts[sums] + 2 > max_allowed):
                attempts += 1
                continue
        # self‑pair contributes a single ordered term
        if counts[2 * i] + 1 > max_allowed:
            attempts += 1
            continue
        # accept this index
        if cur > 0:
            counts[sums] += 2
        counts[2 * i] += 1
        positions[cur] = i
        used[i] = True
        cur += 1
        attempts = 0          # reset after a success
    return positions[:cur], counts
