import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Construct a non‑negative step function (list of heights) that aims to
    minimize the AC1 upper bound evaluated by `evaluate_sequence`.
    The algorithm builds a large set of indices whose pairwise sums occur
    at most a few times, which keeps the peak of the self‑convolution low.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ---- fallback parent ----------------------------------------------------
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                         0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    n = parent.size

    # ---- parameters ---------------------------------------------------------
    # maximal allowed unordered collisions for any sum (1 = Sidon, 2 = low‑energy)
    occ_limits = [2, 3]               # try a couple of limits
    attempts_per_limit = 4            # how many random orderings per limit

    # ---- main search loop ---------------------------------------------------
    while time.monotonic() < deadline:
        for occ_limit in occ_limits:
            for _ in range(attempts_per_limit):
                if time.monotonic() >= deadline:
                    break

                occ = np.zeros(2 * n - 1, dtype=np.int8)   # unordered pair counts
                indices = []

                order = rng.permutation(n)
                for i in order:
                    # diagonal sum (i+i)
                    if occ[2 * i] >= occ_limit:
                        continue

                    if indices:
                        prev = np.array(indices, dtype=np.int32)
                        sums = i + prev
                        # would any unordered sum exceed the limit?
                        if np.any(occ[sums] >= occ_limit):
                            continue

                    # accept i
                    if indices:
                        occ[i + np.array(indices, dtype=np.int32)] += 1
                    occ[2 * i] += 1
                    indices.append(int(i))

                # build candidate vector (binary heights)
                cand = np.zeros(n, dtype=np.float64)
                cand[indices] = 1.0

                # evaluate
                score = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_seq = cand
                    best_score = score

                # early exit if we already beat the target bound
                if best_score < 1.50274349:
                    return [float(x) for x in best_seq]

        # if we already have a good solution, stop early
        if best_score < 1.50274349:
            break

    return [float(x) for x in best_seq]
