    """One O(n) non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

# Greedy construction of a low‑load binary pattern
def greedy_set(L, n, rng):
    """
    Build a set S⊂{0,…,n‑1} such that the convolution of the
    0/1 indicator of S never exceeds L.
    Returns the indices in S and the final load array.
    """
    loads = np.zeros(2 * n - 1, dtype=np.int32)   # convolution loads
    selected = []                                 # indices in S
    order = np.arange(n, dtype=np.int32)
    rng.shuffle(order)

    for i in order:
        # self‑contribution (i,i) gives +1
        if loads[2 * i] + 1 > L:
            continue
        # check pair contributions with previously selected indices
        bad = False
        for j in selected:
            if loads[i + j] + 2 > L:   # (i,j) and (j,i) each add 1
                bad = True
                break
        if bad:
            continue
        # accept i
        selected.append(i)
        loads[2 * i] += 1
        for j in selected[:-1]:
            loads[i + j] += 2
    return np.array(selected, dtype=np.int32), loads

# ------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a step‑function with low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------- fallback parent -------------------------------------------------
    try:
        parent_arr = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64),
            0.0, 1000.0)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise Exception
        parent = parent_arr
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = best.size

    # ------- parameters -------------------------------------------------------
    eps = 0.01          # small background value added to every entry
    L_vals = range(3, 9)   # target maximal convolution loads to try
    attempts_per_L = 6

    # ------- greedy binary construction --------------------------------------
    for L in L_vals:
        for _ in range(attempts_per_L):
            if time.monotonic() >= deadline:
                return [float(x) for x in best]

            sel, loads = greedy_set(L, n, rng)
            # actual maximal load (may be < L)
            max_load = int(loads.max())

            # build candidate: 1 on selected positions, eps elsewhere
            cand = np.full(n, eps, dtype=np.float64)
            cand[sel] = 1.0

            # exact evaluation
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
                # early exit if outstanding
                if best_score < 1.5:
                    # keep searching a little for possible improvements
                    pass

    # ------- optional local refinement (fast, cheap) -------------------------
    while time.monotonic() < deadline:
        cand = make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    return [float(x) for x in best]
