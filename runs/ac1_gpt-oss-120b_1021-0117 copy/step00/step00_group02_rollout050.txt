def conv_fft(x, y):
    n = len(x) + len(y) - 1
    N = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    Z = X * Y
    z = np.fft.irfft(Z, N)
    return z[:n]

import numpy as np, time

def conv_fft(a, b):
    n = len(a)
    m = len(b)
    size = 1 << ((n + m - 1).bit_length())
    # compute FFT using real transforms
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    C = A * B
    c = np.fft.irfft(C, size)[:n+m-1]
    return c

# Helper: linear convolution using FFT (memory O(n))
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution of two 1‑D arrays using FFT."""
    n = len(x) + len(y) - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    Z = X * Y
    z = np.fft.irfft(Z, N)[:n]
    return z

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic optimisation for the AC1 upper bound.
    Works with the pre‑imported `evaluate_sequence` and optional
    `height_sequence_1`.  Returns a list of non‑negative coefficients.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)

    # ------------------------------------------------------------------
    # initialise from the provided parent if possible
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
            a = np.full(256, 1.0, dtype=np.float64)
    else:
        a = np.full(256, 1.0, dtype=np.float64)

    # enforce bounds and a tiny positive sum
    np.clip(a, 0.0, 1000.0, out=a)
    if a.sum() < 0.01:
        a += (0.01 - a.sum()) / a.size

    n = a.size

    # ------------------------------------------------------------------
    # fast evaluation (identical to the official evaluator but without
    # the extra Python overhead)
    def _fast_score(vec: np.ndarray) -> float:
        s = vec.sum()
        if s < 0.01:
            return np.inf
        c = _conv_fft(vec, vec)
        return 2.0 * n * np.max(c) / (s * s)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))   # authoritative
    best_fast = _fast_score(best)

    # ------------------------------------------------------------------
    # main optimisation loop
    eps = 0.4               # initial reduction fraction for the max pair
    eps_decay = 0.9999      # slowly shrink epsilon
    lr = 0.5                # base learning‑rate for the pair‑shift step
    min_lr = 1e-5

    while time.monotonic() < deadline:
        # current convolution and its maximal index
        conv_cur = _conv_fft(best, best)
        max_idx = int(np.argmax(conv_cur))

        # pairs (i,j) with i+j == max_idx
        i0 = max(0, max_idx - (n - 1))
        i1 = min(n - 1, max_idx)
        i_range = np.arange(i0, i1 + 1, dtype=np.int64)
        j_range = max_idx - i_range
        products = best[i_range] * best[j_range]

        # pick the contributing pair with largest product
        pair_pos = int(np.argmax(products))
        i_max = i_range[pair_pos]
        j_max = j_range[pair_pos]

        # reduction of the selected pair (keep total mass constant)
        red_frac = eps * (0.5 + 0.5 * rng.random())   # randomise a bit
        red_i = red_frac * best[i_max]
        if i_max == j_max:
            red_j = 0.0
        else:
            red_j = red_frac * best[j_max]

        total_removed = red_i + red_j

        cand = best.copy()
        cand[i_max] -= red_i
        if i_max != j_max:
            cand[j_max] -= red_j

        # distribute the removed mass uniformly (preserves the sum)
        add = total_removed / n
        cand += add

        # enforce bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        # fast check – accept only if we really improve the objective
        cand_fast = _fast_score(cand)
        if cand_fast < best_fast:
            # confirm with the authoritative evaluator (rare call)
            exact = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best = cand
                best_score = exact
                best_fast = cand_fast

        # ------------------------------------------------------------------
        # occasional random perturbation to escape plateaus
        if rng.random() < 0.01:
            w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            s = int(rng.integers(0, max(1, n - w + 1)))
            e = min(n, s + w)
            pert = best.copy()
            pert[s:e] *= float(np.exp(rng.normal(0.0, 0.2)))
            np.clip(pert, 0.0, 1000.0, out=pert)
            pert_fast = _fast_score(pert)
            if pert_fast < best_fast:
                exact = float(evaluate_sequence(pert.tolist()))
                if np.isfinite(exact) and exact < best_score:
                    best = pert
                    best_score = exact
                    best_fast = pert_fast

        # ------------------------------------------------------------------
        # decay parameters so the search becomes finer over time
        eps = max(1e-4, eps * eps_decay)
        lr = max(min_lr, lr * 0.995)

        # safety: if the algorithm has stalled, re‑initialise a small amount
        if rng.random() < 0.002:
            # tiny random bump on a random block
            w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            s = int(rng.integers(0, max(1, n - w + 1)))
            e = min(n, s + w)
            tmp = best.copy()
            tmp[s:e] *= float(np.exp(rng.normal(0.0, 0.5)))
            np.clip(tmp, 0.0, 1000.0, out=tmp)
            tmp_fast = _fast_score(tmp)
            if tmp_fast < best_fast:
                exact = float(evaluate_sequence(tmp.tolist()))
                if np.isfinite(exact) and exact < best_score:
                    best = tmp
                    best_score = exact
                    best_fast = tmp_fast

    # ------------------------------------------------------------------
    # final authoritative evaluation to guarantee a valid return value
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # fallback to a safe uniform vector
        best = np.full(n, 1.0, dtype=np.float64)
    return [float(x) for x in best]
