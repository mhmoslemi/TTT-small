    else:
        avg = 1.0

    best_score = float(evaluate_sequence(best.tolist()))
    
    # also evaluate parent baseline if exists
    if parent is not None:
        ps = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(ps) and ps < best_score:
            best = parent
            best_score = ps
    
    # simple hill climbing: random block scaling
    while time.monotonic() < deadline:
        # pick random indices block
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n//10))
        stop = min(n, start + width)
        cand = best.copy()
        scale = np.exp(rng.normal(0.0, 0.1))
        cand[start:stop] *= scale
        # ensure bounds
        np.clip(cand, 0.0, 1000.0, out=cand)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score - 1e-12:
            best = cand
            best_score = score
        # occasional random reset to uniform to avoid stuck
        if rng.random() < 0.01:
            cand = uniform.copy()
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
    return [float(x) for x in best]

import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound.

    The algorithm starts from the provided parent (if any), compares it with a
    uniform candidate (the known minimizer of the bound), and then performs a
    very cheap hill‑climbing search while time permits.  The best sequence found
    before the deadline is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    # give a small safety margin before the hard sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.5)

    # ----------------------------------------------------------------------
    # 1.  Acquire a parent sequence if it exists and looks valid.
    # ----------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:   # height_sequence_1 missing or invalid
        parent = None

    # ----------------------------------------------------------------------
    # 2.  Determine problem size.
    # ----------------------------------------------------------------------
    n = parent.size if parent is not None else 256

    # ----------------------------------------------------------------------
    # 3.  Build the uniform candidate – the theoretical minimizer of the bound.
    # ----------------------------------------------------------------------
    if parent is not None:
        avg = float(parent.mean())
    else:
        avg = 1.0
    avg = max(0.0, min(1000.0, avg))
    uniform = np.full(n, avg, dtype=np.float64)

    # ----------------------------------------------------------------------
    # 4.  Initialise the best holder.
    # ----------------------------------------------------------------------
    best = uniform
    best_score = float(evaluate_sequence(best.tolist()))

    if parent is not None:
        pscore = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(pscore) and pscore < best_score:
            best = parent.copy()
            best_score = pscore

    # ----------------------------------------------------------------------
    # 5.  Tiny stochastic hill‑climb – keeps memory O(n) and respects time.
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        # random block perturbation
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 12))
        stop = min(n, start + width)
        cand = best.copy()
        # multiplicative log‑normal perturbation (keeps values non‑negative)
        factor = float(np.exp(rng.normal(0.0, 0.07)))
        cand[start:stop] *= factor
        # enforce admissible range
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score + 1e-12 < best_score:
            best = cand
            best_score = score

        # occasional random reset to the uniform baseline (helps escape plateaus)
        if rng.random() < 0.005:
            cand = uniform.copy()
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score

    # ----------------------------------------------------------------------
    # 6.  Return the best sequence as a plain list of floats.
    # ----------------------------------------------------------------------
    return [float(x) for x in best]
