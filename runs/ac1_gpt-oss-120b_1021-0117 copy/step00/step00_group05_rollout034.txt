import time
import numpy as np

def _conv_score(seq: np.ndarray) -> float:
    """Helper: evaluate sequence using the authoritative evaluator."""
    # the evaluator works on Python lists; conversion is cheap
    return float(evaluate_sequence(seq.tolist()))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients whose AC1 bound
    (as measured by ``evaluate_sequence``) is as low as possible within the
    time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ---------------------------------------------------------------------
    # Base length (use the provided parent if it exists, otherwise a default)
    # ---------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise ValueError
        n = parent_arr.size
        parent_arr = np.clip(parent_arr, 0.0, 1000.0)
    except Exception:
        n = 256
        parent_arr = np.ones(n, dtype=np.float64)

    best_seq = parent_arr.copy()
    best_score = _conv_score(best_seq)

    # ---------------------------------------------------------------------
    # Parameterised families (power‑law shapes).  The score is scale‑invariant,
    # so we keep the raw values (they stay well below the 1000 cap).
    # ---------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # search grid (adjustable if time permits)
    alphas = np.linspace(0.45, 0.70, 13)          # 0.45 … 0.70
    betas  = np.linspace(0.5, 30.0, 25)          # 0.5 … 30

    for alpha in alphas:
        if time.monotonic() >= deadline:
            break
        for beta in betas:
            if time.monotonic() >= deadline:
                break
            # a_i = (i + beta)^{-alpha}
            seq = 1.0 / np.power(idx + beta, alpha)
            score = _conv_score(seq)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = seq.copy()
                # early stop if we already beat the target bound
                if best_score < 1.50274349:
                    # keep a tiny margin before the deadline
                    deadline = min(deadline, time.monotonic() + 0.5)

    # ---------------------------------------------------------------------
    # Local stochastic refinement around the current best candidate.
    # Simple multiplicative block perturbations + occasional smoothing.
    # ---------------------------------------------------------------------
    while time.monotonic() < deadline:
        if best_seq is None:
            cand = np.abs(rng.normal(size=n))
        else:
            cand = best_seq.copy()
            # random contiguous block
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            factor = float(np.exp(rng.normal(0.0, 0.06)))
            cand[start:stop] *= factor
            # occasional averaging with neighbour (smooths spikes)
            if rng.random() < 0.25:
                cand = 0.5 * (cand + np.roll(cand, 1))
        # enforce hard bounds before evaluation
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = _conv_score(cand)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand.copy()
            if best_score < 1.50274349:
                # we already surpassed the target – give ourselves a tiny
                # safety buffer before the sandbox kills us.
                deadline = min(deadline, time.monotonic() + 0.3)

    # ---------------------------------------------------------------------
    # Fallback safety: if nothing valid was found, use a uniform vector.
    # ---------------------------------------------------------------------
    if best_seq is None or not np.isfinite(best_score):
        best_seq = np.ones(n, dtype=np.float64)

    return [float(x) for x in best_seq]
