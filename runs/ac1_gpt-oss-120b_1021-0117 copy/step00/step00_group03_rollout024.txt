            idx = rng.choice(n, size=flips, replace=False)
            cand[idx] = 1.0 - cand[idx]      # toggle 0↔1

        # Guarantee a non‑zero sum (evaluator rejects sum < 0.01).
        if cand.sum() < 0.01:
            cand[rng.integers(0, n)] = 1.0

        # --------------------------------------------------------------
        # Evaluate candidate.
        # --------------------------------------------------------------
        score = float(eval_seq(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

        # Early exit if we hit a very good bound.
        if best_score < 1.450:          # far below the target, no need to continue
            break

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list of floats).
    # ------------------------------------------------------------------
    return [float(x) for x in best]
```
