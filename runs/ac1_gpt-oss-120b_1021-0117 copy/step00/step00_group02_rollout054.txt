        # pick a modest block and perturb it multiplicatively
        block_w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - block_w + 1)))
        factor = float(np.exp(rng.normal(0.0, 0.02)))  # very mild change
        cand = best.copy()
        cand[start:start + block_w] *= factor
        np.clip(cand, 0.0, max_val, out=cand)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return plain Python list as required by the harness
    return best.tolist()
