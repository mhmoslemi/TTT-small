import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Generate a step‑function using a power‑law decay a_i = (i+1)^{-p}.
    Several exponents are tried; the best (lowest AC1 bound) is kept.
    A cheap random local search runs until the time budget expires,
    guaranteeing a valid fallback.
    """
    rng = np.random.default_rng(seed)
    # length of the problem (height_sequence_1 is guaranteed when available)
    n = len(height_sequence_1) if "height_sequence_1" in globals() else 1024

    # ------------------------------------------------------------------
    # 1.  Evaluate a few analytic candidates a_i = (i+1)^{-p}
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # include the provided parent as a candidate
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(score) and score < best_score:
                best_seq, best_score = parent, score
    except Exception:
        pass

    # a few exponents p ∈ (0,1) – the smaller p, the flatter the sequence
    # the asymptotic analysis predicts p≈0.6–0.8 to beat the target bound.
    for p in (0.55, 0.60, 0.65, 0.70, 0.75, 0.80):
        seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        seq = np.clip(seq, 0.0, 1000.0)
        try:
            score = float(evaluate_sequence(seq.tolist()))
        except Exception:
            continue
        if np.isfinite(score) and score < best_score:
            best_seq, best_score = seq, score

    # ------------------------------------------------------------------
    # 2.  Simple bounded random perturbation loop (optional refinement)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    while time.monotonic() < deadline:
        # pick a random contiguous block and perturb multiplicatively
        width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width

        cand = best_seq.copy()
        # log‑normal multiplicative noise (small sigma)
        noise = np.exp(rng.normal(0.0, 0.07, size=width))
        cand[start:stop] *= noise
        cand = np.clip(cand, 0.0, 1000.0)

        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue
        if np.isfinite(score) and score < best_score:
            best_seq, best_score = cand, score

    # ------------------------------------------------------------------
    # 3.  Return the best found candidate as a plain Python list
    # ------------------------------------------------------------------
    if best_seq is None:
        # ultimate fallback – a uniform vector (guaranteed to be valid)
        best_seq = np.full(n, np.sqrt(2 * n) / n, dtype=np.float64)

    return [float(x) for x in best_seq]
