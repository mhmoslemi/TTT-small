import numpy as np
import time
import math

def _generate_sidon_set(n):
    """
    Greedy construction of a Sidon (B2) set in {0,…,n-1}.
    The set has size ≈√n and satisfies that all pairwise sums are distinct.
    """
    used = np.zeros(2 * n, dtype=bool)          # used[s] == True  ⇒  sum s already taken
    pos = []
    for i in range(n):
        # if i+i already used we cannot add i (its self‑sum would clash)
        if used[i + i]:
            continue
        conflict = False
        for p in pos:
            if used[i + p]:
                conflict = True
                break
        if conflict:
            continue
        # i is admissible
        pos.append(i)
        # mark all new sums i+p (including i+i)
        for p in pos:
            used[i + p] = True
    return pos

    if m:
        seq[np.array(sidon, dtype=np.int64)] = H
    seq = np.clip(seq, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Return the best (and only) candidate as a plain Python list.
    # ------------------------------------------------------------------
    return [float(v) for v in seq]
